1/*
2 * WHAT: Async kXR_Qcksum — compute a single-file checksum (adler32/crc32c/md5/sha1/sha256) off the event loop via nginx thread pool.
3 * The thread worker opens a confined fd, computes checksum via brix_checksum_hex_fd, writes "algo hex" result to t->resp.
4 * On failure sets error_code + error_msg. Never touches nginx event-loop state (no ctx, no c, no ngx_pool).
5 *
6 * WHY: Qcksum file checksums can be expensive for large files. Running on the event loop blocks all other connections while reading.
7 * The async path uses ngx_thread_pool_run to execute the open+checksum in a background worker thread, keeping nginx responsive.
8 * Results delivered via aio_done callback which restores request streamid, sends response to client, and resumes connection.
9 *
10 * HOW: brix_cksum_aio_thread() parses algo from t->algo string, computes checksum via hex_fd on pre-opened fd,
11 * formats result as "algo hex" into resp buffer. On parse failure sets kXR_ArgInvalid; on compute failure sets kXR_IOError.
12 * brix_cksum_aio_done() closes fd if path-based request (t->close_fd), restores streamid via aio_restore_request, sends ok+resp or error, frees resources, resumes client connection event loop via aio_resume().
13 */

21/* One slot per connection, shared by the serialized checksum query forms.
22 * Keeping it behind a pool cleanup avoids a public connection-layout change. */
23typedef struct {
24 ngx_thread_task_t *task;
25 size_t capacity;
26} brix_checksum_task_slot_t;

28/* WHAT: Release a connection's checksum task after its deferred I/O is done.
29 * WHY: The embedded event must not be freed from its own completion callback.
30 * HOW: 1. Free its heap block at connection-pool teardown. 2. Retire the slot. */
31static void
32brix_checksum_task_cleanup(void *data)
33{
34 brix_checksum_task_slot_t *slot = data;
35
36 if (slot->task != NULL) {
37 ngx_free(slot->task);
38 slot->task = NULL;
39 }
40 slot->capacity = 0;
41}

43/* WHAT: Find or register the one checksum-task owner on a connection pool.
44 * WHY: Queue-overflow retries must not allocate a fresh task on every request;
45 * ngx_pfree cannot reclaim small pool allocations on large-page hosts.
46 * HOW: 1. Match this concern's cleanup handler. 2. Allocate one zeroed slot. */
47static brix_checksum_task_slot_t *
48brix_checksum_task_slot(ngx_connection_t *c)
49{
50 ngx_pool_cleanup_t *cleanup;
51
52 for (cleanup = c->pool->cleanup; cleanup != NULL; cleanup = cleanup->next) {
53 if (cleanup->handler == brix_checksum_task_cleanup) {
54 return cleanup->data;
55 }
56 }
57 cleanup = ngx_pool_cleanup_add(c->pool, sizeof(brix_checksum_task_slot_t));
58 if (cleanup == NULL) {
59 return NULL;
60 }
61 ngx_memzero(cleanup->data, sizeof(brix_checksum_task_slot_t));
62 cleanup->handler = brix_checksum_task_cleanup;
63 return cleanup->data;
64}

66/* WHAT: Obtain an inactive reusable checksum/scan task, or NULL on failure.
67 * WHY: Both accepted queries and busy retries need bounded connection memory;
68 * a queued task's event/context must remain untouched until completion.
69 * HOW: 1. Find its owner. 2. Reject overlap. 3. Grow before replacing the old
70 * block. 4. Reset inactive storage and expose the requested context. */
71ngx_thread_task_t *
72brix_query_checksum_task(ngx_connection_t *c, size_t size)
73{
74 brix_checksum_task_slot_t *slot;
75 ngx_thread_task_t *task;
76
77 if (size > SIZE_MAX - sizeof(ngx_thread_task_t)) {
78 return NULL;
79 }
80 slot = brix_checksum_task_slot(c);
81 if (slot == NULL) {
82 return NULL;
83 }
84 if (slot->task != NULL && slot->task->event.active) {
85 return NULL;
86 }
87 if (slot->task == NULL || slot->capacity < size) {
88 task = ngx_alloc(sizeof(ngx_thread_task_t) + size, c->log);
89 if (task == NULL) {
90 return NULL;
91 }
92 brix_checksum_task_cleanup(slot);
93 slot->task = task;
94 slot->capacity = size;
95 }
96 ngx_memzero(slot->task, sizeof(ngx_thread_task_t) + size);
97 slot->task->ctx = slot->task + 1;
98 return slot->task;
99}

102/*
103 * brix_cksum_aio_thread — runs on a thread-pool worker.
104 *
105 * Computes the checksum of t->fd using t->algo and writes the result
106 * into t->resp. On failure sets t->error_code to the appropriate kXR_*
107 * code and t->error_msg to a human-readable description. Never touches
108 * nginx event-loop state (no ctx, no c, no ngx_pool).
109 */
110/* public API: brix_cksum_aio_thread() — async qcksum thread worker
111 * WHAT: Thread pool worker that computes single-file checksum.
112 * - Parses algo string via brix_checksum_parse
113 * - Computes checksum via brix_checksum_hex_fd on pre-opened fd
114 * - Writes "algo hex" result to resp buffer
115 * - On parse failure: sets kXR_ArgInvalid
116 * - On compute failure: sets kXR_IOError + error message
117 * - Never touches nginx event-loop state (no ctx, no c, no ngx_pool)
118 */
119
120void
121brix_cksum_aio_thread(void *data, ngx_log_t *log)
122{
123 brix_cksum_aio_t *t = data;
124 brix_integrity_info_t info;
125 brix_integrity_opts_t iopts;
126 char token[256];
127
128 ngx_memzero(&iopts, sizeof(iopts));
129 iopts.allow_xattr_cache = 1;
130 iopts.update_xattr_cache = 1;
131 iopts.mutation_policy = t->mutation_policy;
132 iopts.proto = BRIX_PROTO_ROOT;
133
134 if (brix_integrity_get_fd(log, t->fd, &t->obj, t->resolved, t->algo,
135 &iopts, &info) != NGX_OK)
136 {
137 t->error_code = kXR_IOError;
138 snprintf(t->error_msg, sizeof(t->error_msg),
139 "checksum computation failed");
140 return;
141 }
142
143 /* kXR_Qcksum wire format: "algo hexvalue" (space-separated) */
144 snprintf(token, sizeof(token), "%s %s", info.alg_name, info.hex);
145 ngx_cpystrn((u_char *) t->resp, (u_char *) token, sizeof(t->resp));
146}

148/*
149 * brix_cksum_aio_done — main-thread completion callback.
150 *
151 * Closes t->fd if t->close_fd is set (path-based requests), then sends
152 * the response built by the thread worker and re-arms the connection.
153 * The fd is closed before the destroy check so it is never leaked even
154 * when the client disconnected while the AIO was in flight.
155 */
156/* public API: brix_cksum_aio_done() — async qcksum completion callback
157 * WHAT: Main-thread event handler when thread worker completes.
158 * - Closes t->fd if path-based request (t->close_fd)
159 * - Restores request streamid via aio_restore_request
160 * - Sends error response or ok+checksum data to client
161 * - Resumes client connection event loop via brix_aio_resume()
162 */
163
164void
165brix_cksum_aio_done(ngx_event_t *ev)
166{
167 brix_cksum_aio_t *t = ((ngx_thread_task_t *) ev->data)->ctx;
168 brix_ctx_t *ctx = t->ctx;
169 ngx_connection_t *c = t->c;
170
171 if (t->close_fd) {
172 /* Path-based request: release the VFS handle we opened (it owns the fd).
173 * The handle is freed before the destroy check so it is never leaked even
174 * when the client disconnected while the AIO was in flight. */
175 if (t->fh != NULL) {
176 brix_vfs_close(t->fh, c->log);
177 t->fh = NULL;
178 } else if (t->fd >= 0) {
179 close(t->fd);
180 }
181 t->fd = -1;
182 }
183
184 if (!brix_aio_restore_request(ctx, t->streamid)) {
185 return;
186 }
187
188 if (t->error_code != 0) {
189 BRIX_OP_ERR(ctx, BRIX_OP_QUERY_CKSUM);
190 brix_send_error(ctx, c, (uint16_t) t->error_code, t->error_msg);
191 } else {
192 BRIX_OP_OK(ctx, BRIX_OP_QUERY_CKSUM);
193 brix_log_access(ctx, c, "QUERY", t->resolved, "cksum", 1, 0, NULL, 0);
194 brix_send_ok(ctx, c, t->resp, (uint32_t) (strlen(t->resp) + 1));
195 }
196
197 brix_aio_resume(c);
198}
