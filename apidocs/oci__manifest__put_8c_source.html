1/*
2 * oci_manifest_put.c — manifest PUT and the DELETE surface (§D4.3, §D4.4).
3 *
4 * WHAT: PUT /v2/<name>/manifests/<ref> — validate the document, prove every
5 * blob it names is already in the store, then commit it and (for a tag)
6 * swap the tag pointer atomically. Plus DELETE of a manifest by digest.
7 * WHY: the manifest is the only object in a registry that REFERS to others,
8 * which makes it the only place a push can leave the store incoherent:
9 * accept a manifest whose layers were never uploaded and every later
10 * pull of that tag 404s halfway through, after the client has already
11 * committed to the image. The spec's answer — MANIFEST_BLOB_UNKNOWN at
12 * PUT time — is therefore not a nicety, it is the invariant that makes
13 * "the tag resolves" mean "the image is complete".
14 * HOW: jansson parses the document (never json_min: this is attacker-supplied
15 * nesting on the write path, and the bounded scanner cannot answer
16 * structural questions about it), the body is size-capped BEFORE it is
17 * read, and the commit order is blob-check → CAS write → sidecar → tag
18 * swap, so a failure at any step leaves the previous tag intact.
19 */

62/* Is `d` present in the store, either as a blob or as a manifest? An index
63 * names manifests, a manifest names blobs, and both live in the same CAS —
64 * so one existence question covers both, and neither has to know which kind
65 * of child it is holding. */
66static int
67oci_ref_present(const brix_oci_store_t *st, const char *name, size_t name_len,
68 const brix_oci_digest_t *d)
69{
70 char path[PATH_MAX];
71
72 if (brix_oci_store_blob_path(st, d, path, sizeof(path)) == 0
73 && brix_oci_store_exists(path, NULL))
74 {
75 return 1;
76 }
77 return brix_oci_store_manifest_path(st, name, name_len, d, NULL,
78 path, sizeof(path)) == 0
79 && brix_oci_store_exists(path, NULL);
80}

83/* Walk one descriptor array ("layers", "manifests", or the lone "config"
84 * object) and stop at the first digest the store cannot produce. Returns 0,
85 * or -1 with the offending digest string copied into `missing`. */
86static int
87oci_walk_refs(const brix_oci_store_t *st, const brix_oci_req_t *req,
88 json_t *node, char *missing, size_t missing_len)
89{
90 brix_oci_digest_t d;
91 const char *dg;
92 json_t *elem;
93 size_t i;
94
95 if (json_is_object(node)) {
96 dg = json_string_value(json_object_get(node, "digest"));
97 if (dg == NULL
98 || brix_oci_digest_parse(dg, ngx_strlen(dg), &d) != 0)
99 {
100 (void) snprintf(missing, missing_len, "%s",
101 (dg != NULL) ? dg : "(no digest)");
102 return -1;
103 }
104 if (!oci_ref_present(st, req->name, req->name_len, &d)) {
105 (void) snprintf(missing, missing_len, "%s", dg);
106 return -1;
107 }
108 return 0;
109 }
110
111 if (!json_is_array(node)) {
112 return 0; /* absent member: nothing to prove */
113 }
114
115 json_array_foreach(node, i, elem) {
116 if (oci_walk_refs(st, req, elem, missing, missing_len) != 0) {
117 return -1;
118 }
119 if (i + 1 >= BRIX_OCI_MAX_REFS) {
120 (void) snprintf(missing, missing_len, "(too many references)");
121 return -1;
122 }
123 }
124 return 0;
125}

128/* Every object the document names is already here. NGX_OK = commit it;
129 * NGX_DONE = refused, envelope written (never NGX_OK — a refusal that reads
130 * as success is a manifest committed over its own 400).
131 *
132 * The `subject` descriptor is SHAPE-checked but deliberately not walked: a
133 * referrer may be pushed before, or entirely without, the thing it describes
134 * — that is what lets a signature exist for an image this registry never
135 * held — but a subject we cannot read at all is a refusal, because storing it
136 * would publish an artifact whose edge nothing can follow. */
137static ngx_int_t
138oci_manifest_validate(ngx_http_request_t *r, const oci_manifest_ctx_t *m,
139 const brix_oci_req_t *req, json_t *doc)
140{
141 char missing[BRIX_OCI_DIGEST_STRLEN + 32];
142 const char *why = NULL;
143 int bad;
144
145 if (brix_oci_referrers_subject_ok(doc, &why) != 0) {
146 return brix_oci_refuse(r, NGX_HTTP_BAD_REQUEST,
147 BRIX_OCI_ERR_MANIFEST_INVALID, why);
148 }
149
150 missing[0] = '\0';
151 bad = oci_walk_refs(&m->st, req, json_object_get(doc, "config"),
152 missing, sizeof(missing))
153 || oci_walk_refs(&m->st, req, json_object_get(doc, "layers"),
154 missing, sizeof(missing))
155 || oci_walk_refs(&m->st, req, json_object_get(doc, "manifests"),
156 missing, sizeof(missing));
157
158 if (bad) {
159 return brix_oci_refuse(r, NGX_HTTP_BAD_REQUEST,
160 BRIX_OCI_ERR_MANIFEST_BLOB_UNKNOWN, missing);
161 }
162 return NGX_OK;
163}

166/* Write the manifest into the CAS, describe it in the sidecar, and — when the
167 * push named a tag rather than a digest — point that tag at it. The tag swap
168 * is last and atomic: until it lands, a reader sees the previous image, and
169 * after it lands, a reader sees a complete one. There is no moment in between
170 * where the tag names something half-written. */
171static ngx_int_t
172oci_manifest_commit(ngx_http_request_t *r, const oci_manifest_ctx_t *m,
173 const brix_oci_req_t *req, json_t *doc, const u_char *body, size_t len,
174 const brix_oci_digest_t *d)
175{
176 brix_oci_meta_t meta;
177 char path[PATH_MAX];
178 char dg[BRIX_OCI_DIGEST_STRLEN];
179 char subj[BRIX_OCI_DIGEST_STRLEN];
180 char loc[BRIX_OCI_KEY_MAX];
181 ngx_log_t *log = r->connection->log;
182
183 if (brix_oci_digest_format(d, dg, sizeof(dg)) < 0
184 || brix_oci_store_manifest_path(&m->st, req->name, req->name_len,
185 d, NULL, path, sizeof(path)) != 0)
186 {
187 return NGX_HTTP_INTERNAL_SERVER_ERROR;
188 }
189 if (brix_oci_store_publish_bytes(&m->st, path, body, len,
190 BRIX_VFS_DOMAIN_REGISTRY, log) != NGX_OK)
191 {
192 return NGX_HTTP_INTERNAL_SERVER_ERROR;
193 }
194
195 ngx_memzero(&meta, sizeof(meta));
196 (void) snprintf(meta.content_type, sizeof(meta.content_type), "%s",
197 m->ctype);
198 (void) snprintf(meta.digest, sizeof(meta.digest), "%s", dg);
199 meta.fetched_at = ngx_time();
200 meta.verified = 1;
201
202 /* mtime 0: the manifest is CAS-addressed, so its path already pins its
203 * bytes and the size is all the coherence a memo beside it can need. */
204 if (brix_oci_meta_store(path, (off_t) len, 0, &meta, log) != NGX_OK) {
205 return NGX_HTTP_INTERNAL_SERVER_ERROR;
206 }
207
208 /* The referrers edge is recorded BEFORE the tag swap, for the same reason
209 * the blob walk runs before either: once a name resolves to this
210 * manifest, everything the spec says about it must already be true. */
211 if (brix_oci_referrers_index(&m->st, req, doc, d, m->ctype, (off_t) len,
212 subj, sizeof(subj), log) != NGX_OK)
213 {
214 return NGX_HTTP_INTERNAL_SERVER_ERROR;
215 }
216
217 if (!req->ref_is_digest
218 && brix_oci_store_tag_set(&m->st, req, dg, log) != NGX_OK)
219 {
220 return NGX_HTTP_INTERNAL_SERVER_ERROR;
221 }
222
223 /* OCI-Subject is how the client learns the registry UNDERSTOOD the
224 * subject field. Its absence tells a signing tool to fall back to the
225 * tag-schema workaround, so emitting it is not cosmetic — it is the
226 * handshake that turns the referrers API on for that client. */
227 if (subj[0] != '\0'
228 && brix_http_set_header(r, "OCI-Subject", subj, NULL) != NGX_OK)
229 {
230 return NGX_HTTP_INTERNAL_SERVER_ERROR;
231 }
232
233 if ((size_t) snprintf(loc, sizeof(loc), "/v2/%.*s/manifests/%s",
234 (int) req->name_len, req->name, dg) >= sizeof(loc)
235 || brix_http_set_header(r, "Location", loc, NULL) != NGX_OK
236 || brix_http_set_header(r, "Docker-Content-Digest", dg, NULL) != NGX_OK)
237 {
238 return NGX_HTTP_INTERNAL_SERVER_ERROR;
239 }
240 return brix_oci_send_body(r, NGX_HTTP_CREATED, "application/json",
241 (const u_char *) "", 0);
242}

245/* Parse once, then validate and commit against the same document. The parse
246 * lives here rather than inside each stage because the referrers index needs
247 * the `subject` descriptor at commit time, and re-parsing a 4 MiB body to
248 * read one field would be a second chance for the two readings to disagree
249 * about what was pushed. */
250static ngx_int_t
251oci_manifest_stages(ngx_http_request_t *r, oci_manifest_ctx_t *m,
252 ngx_http_brix_oci_ctx_t *ctx, u_char *body, size_t len,
253 const brix_oci_digest_t *d)
254{
255 json_error_t jerr;
256 json_t *doc;
257 ngx_int_t rc;
258
259 doc = json_loadb((const char *) body, len, 0, &jerr);
260 if (doc == NULL) {
261 ctx->disp = BRIX_OCI_OUT_REFUSED;
262 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
263 BRIX_OCI_ERR_MANIFEST_INVALID, jerr.text);
264 }
265 if (!json_is_object(doc)) {
266 json_decref(doc);
267 ctx->disp = BRIX_OCI_OUT_REFUSED;
268 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
269 BRIX_OCI_ERR_MANIFEST_INVALID,
270 "a manifest is a JSON object");
271 }
272
273 rc = oci_manifest_validate(r, m, &ctx->req, doc);
274 if (rc != NGX_OK) {
275 json_decref(doc);
276 ctx->disp = BRIX_OCI_OUT_REFUSED;
277 return (rc == NGX_DONE) ? NGX_OK : rc;
278 }
279
280 rc = oci_manifest_commit(r, m, &ctx->req, doc, body, len, d);
281 json_decref(doc);
282
283 return rc;
284}

287static void
288oci_manifest_body_handler(ngx_http_request_t *r)
289{
290 ngx_http_brix_oci_ctx_t *ctx =
291 ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
292 oci_manifest_ctx_t *m = (ctx != NULL) ? ctx->reg : NULL;
293 brix_oci_digest_t d;
294 u_char *body;
295 size_t len;
296
297 if (m == NULL
298 || brix_http_body_read_all(r, BRIX_OCI_MANIFEST_MAX, &body, &len)
299 != NGX_OK)
300 {
301 ngx_http_finalize_request(r, brix_oci_error(r,
302 NGX_HTTP_REQUEST_ENTITY_TOO_LARGE, BRIX_OCI_ERR_SIZE_INVALID,
303 "the manifest exceeds brix's 4 MiB document cap"));
304 return;
305 }
306
307 /* A push BY DIGEST is checked under the algorithm the client named —
308 * anything else compares two different functions' output and rejects an
309 * honest sha512 push. A push by tag has nothing to be checked against,
310 * so it is filed under the algorithm we produce, and the tag therefore
311 * always resolves to a sha256-addressed manifest. */
312 if (brix_oci_digest_hash(ctx->req.ref_is_digest
313 ? ctx->req.digest.alg
314 : BRIX_OCI_ALG_SHA256,
315 body, len, &d) != 0)
316 {
317 ngx_http_finalize_request(r, NGX_HTTP_INTERNAL_SERVER_ERROR);
318 return;
319 }
320
321 /* Pushed by digest? Then the bytes must be the bytes that digest names.
322 * Skipping this is how a registry ends up serving one document under
323 * another document's immutable, infinitely-cacheable name. */
324 if (ctx->req.ref_is_digest
325 && !brix_oci_digest_eq(&d, &ctx->req.digest))
326 {
327 ctx->disp = BRIX_OCI_OUT_REFUSED;
328 ngx_http_finalize_request(r, brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
329 BRIX_OCI_ERR_DIGEST_INVALID,
330 "the manifest does not hash to the digest it was pushed under"));
331 return;
332 }
333
334 ngx_http_finalize_request(r,
335 oci_manifest_stages(r, m, ctx, body, len, &d));
336}

339ngx_int_t
340brix_oci_manifest_put(ngx_http_request_t *r,
341 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx,
342 brix_oci_store_t *st)
343{
344 oci_manifest_ctx_t *m;
345
346 (void) lcf;
347
348 /* The cap is checked against the declared length here and enforced again
349 * against the delivered bytes in the callback: a client may lie in the
350 * header, but it cannot lie about how much it sent. */
351 if (r->headers_in.content_length_n > (off_t) BRIX_OCI_MANIFEST_MAX) {
352 return brix_oci_error(r, NGX_HTTP_REQUEST_ENTITY_TOO_LARGE,
353 BRIX_OCI_ERR_SIZE_INVALID,
354 "the manifest exceeds brix's 4 MiB document cap");
355 }
356
357 m = ngx_pcalloc(r->pool, sizeof(*m));
358 if (m == NULL) {
359 return NGX_HTTP_INTERNAL_SERVER_ERROR;
360 }
361 m->st = *st;
362
363 if (r->headers_in.content_type != NULL) {
364 (void) snprintf(m->ctype, sizeof(m->ctype), "%.*s",
365 (int) r->headers_in.content_type->value.len,
366 r->headers_in.content_type->value.data);
367 }
368 if (!oci_type_known(m->ctype)) {
369 ctx->disp = BRIX_OCI_OUT_REFUSED;
370 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
371 BRIX_OCI_ERR_MANIFEST_INVALID,
372 "unsupported manifest media type");
373 }
374
375 ctx->reg = m;
376 return brix_http_read_body(r, oci_manifest_body_handler);
377}

381
382ngx_int_t
383brix_oci_registry_delete(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
384 brix_oci_store_t *st)
385{
386 char path[PATH_MAX];
387 char meta[PATH_MAX];
388 ngx_log_t *log = r->connection->log;
389
390 /* Deletion is by digest only. A tag is a NAME, and deleting the object a
391 * name happens to point at is how two clients racing a re-tag destroy an
392 * image neither of them meant to touch — so a tag DELETE removes the
393 * pointer and nothing else. */
394 if (ctx->req.cls == BRIX_OCI_REQ_MANIFEST && !ctx->req.ref_is_digest) {
395 if (brix_oci_store_tag_path(st, &ctx->req, path, sizeof(path)) != 0) {
396 return NGX_HTTP_INTERNAL_SERVER_ERROR;
397 }
398 if (!brix_oci_store_exists(path, NULL)) {
399 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
400 BRIX_OCI_ERR_MANIFEST_UNKNOWN, NULL);
401 }
402 if (brix_oci_store_remove(path, log) != NGX_OK) {
403 return NGX_HTTP_INTERNAL_SERVER_ERROR;
404 }
405 return brix_oci_reply_empty(r, NGX_HTTP_ACCEPTED);
406 }
407
408 if (ctx->req.cls == BRIX_OCI_REQ_MANIFEST) {
409 if (brix_oci_store_manifest_path(st, ctx->req.name, ctx->req.name_len,
410 &ctx->req.digest, NULL,
411 path, sizeof(path)) != 0
412 || brix_oci_store_manifest_path(st, ctx->req.name,
413 ctx->req.name_len,
414 &ctx->req.digest, ".meta",
415 meta, sizeof(meta)) != 0)
416 {
417 return NGX_HTTP_INTERNAL_SERVER_ERROR;
418 }
419 if (!brix_oci_store_exists(path, NULL)) {
420 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
421 BRIX_OCI_ERR_MANIFEST_UNKNOWN, NULL);
422 }
423 brix_oci_referrers_forget(st, ctx->req.name, ctx->req.name_len,
424 &ctx->req.digest, log);
425 (void) brix_oci_store_remove(meta, log);
426 if (brix_oci_store_remove(path, log) != NGX_OK) {
427 return NGX_HTTP_INTERNAL_SERVER_ERROR;
428 }
429 return brix_oci_reply_empty(r, NGX_HTTP_ACCEPTED);
430 }
431
432 /* A blob DELETE drops this repository's reference mark. The CAS object
433 * itself stays: another repository may hold the same layer, and content
434 * addressing means it is byte-identical, not merely similar. Reclaiming
435 * unreferenced blobs is a GC pass (`brixoci rm`), not a request handler. */
436 if (brix_oci_store_layer_path(st, ctx->req.name, ctx->req.name_len,
437 &ctx->req.digest, path,
438 sizeof(path)) != 0)
439 {
440 return NGX_HTTP_INTERNAL_SERVER_ERROR;
441 }
442 if (!brix_oci_store_exists(path, NULL)) {
443 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
444 BRIX_OCI_ERR_BLOB_UNKNOWN, NULL);
445 }
446 if (brix_oci_store_remove(path, log) != NGX_OK) {
447 return NGX_HTTP_INTERNAL_SERVER_ERROR;
448 }
449 return brix_oci_reply_empty(r, NGX_HTTP_ACCEPTED);
450}
