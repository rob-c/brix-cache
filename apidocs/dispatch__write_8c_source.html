12/* manager_mutation_t — decoded policy inputs for one manager-side mutation.
13 *
14 * WHAT: Carries the operation's logical path(s), lexical export path(s), and
15 * authorization metadata from decode through authorization and routing.
16 * WHY: A manager does not probe its usually-empty local export, but it must
17 * make the same client-identity/path decision as a storage node before a
18 * proxy or CMS fan-out can execute with server authority.
19 * HOW: manager_mutation_prepare() fills it from the authoritative wire
20 * decoders; manager_mutation_authorize() consumes it without mutation.
21 */
22typedef struct {
23 const char *verb;
24 ngx_uint_t op_id;
25 int auth_level;
26 brix_acc_op_t aop;
27 char path[BRIX_MAX_PATH + 1];
28 char resolved[PATH_MAX];
29 char dst[BRIX_MAX_PATH + 1];
30 char dst_resolved[PATH_MAX];
31} manager_mutation_t;

33/* manager_mutation_error — uniform manager-boundary decode/policy failure.
34 *
35 * WHAT: Emits the access record, operation metric, and one ROOT error reply.
36 * WHY: A rejected manager request must be observable exactly like a local
37 * mutation and terminate before registry selection or CMS fan-out.
38 * HOW: Records the stable client-facing path/message and returns the send
39 * result directly to the dispatcher.
40 */
41static ngx_int_t
42manager_mutation_error(brix_ctx_t *ctx, ngx_connection_t *c,

44{

48 BRIX_OP_ERR(ctx, m->op_id);

52/* manager_mutation_lexical_path — validate one manager path without I/O.
53 *
54 * WHAT: Applies the shared depth, forbidden-component, reserved-name, and
55 * root-join checks with BRIX_PATH_NOEXIST.
56 * WHY: Pure redirectors commonly have no local object to probe. Authorization
57 * still needs a canonical lexical backing path; existence remains the
58 * selected data node's responsibility.
59 * HOW: Delegates to brix_path_resolve_beneath() in no-probe mode and maps
60 * malformed/reserved names to the local handlers' public error classes.
61 */
62static ngx_int_t
63manager_mutation_lexical_path(brix_ctx_t *ctx, ngx_connection_t *c,
64 ngx_stream_brix_srv_conf_t *conf, manager_mutation_t *m, int destination)
65{
66 const char *path = destination ? m->dst : m->path;
67 char *resolved = destination ? m->dst_resolved : m->resolved;
68 ngx_int_t rc;
69
70 if (path[0] != '/') {
71 return manager_mutation_error(ctx, c, m, kXR_ArgInvalid,
72 "path must be absolute");
73 }
74
75 rc = brix_path_resolve_beneath(conf, c->log, path, BRIX_PATH_NOEXIST,
76 resolved, PATH_MAX);
77 if (rc == NGX_OK) {
78 return BRIX_DISPATCH_CONTINUE;
79 }
80 if (rc == NGX_DECLINED) {
81 return manager_mutation_error(ctx, c, m, kXR_NotFound,
82 "no such file or directory");
83 }
84 return manager_mutation_error(ctx, c, m, kXR_ArgInvalid,
85 "invalid path");
86}

88/* manager_mutation_decode_single — extract a single-path mutation payload.
89 *
90 * WHAT: Copies the request path into the manager policy object.
91 * WHY: Keeping the absent/malformed distinction here keeps prepare() flat and
92 * prevents unauthenticated payload parsing (the caller authenticates first).
93 * HOW: Uses the shared path extractor and returns a fully formatted error.

112/* manager_mutation_prepare — decode one routable namespace mutation.
113 *
114 * WHAT: Selects the authorization class, decodes one path (or both MV paths),
115 * and produces lexical policy paths without touching storage.
116 * WHY: The manager must authorize exactly the operands the data node executes;
117 * the old generic extraction treated an MV pair as one selection string.
118 * HOW: Uses brix_extract_path for single-path opcodes and the shared MV decoder
119 * for kXR_mv, then validates each operand through the no-probe path seam.
120 */
121static ngx_int_t
122manager_mutation_prepare(brix_ctx_t *ctx, ngx_connection_t *c,
123 ngx_stream_brix_srv_conf_t *conf, manager_mutation_t *m)
124{
125 brix_mv_paths_t mv;
126 ngx_int_t rc;
127
128 ngx_memzero(m, sizeof(*m));
129 m->aop = BRIX_AOP_ANY;
130
131 switch (ctx->recv.cur_reqid) {
132 case kXR_mkdir:
133 m->verb = "MKDIR"; m->op_id = BRIX_OP_MKDIR;
134 m->auth_level = BRIX_AUTH_MKDIR;
135 break;
136 case kXR_rm:
137 m->verb = "RM"; m->op_id = BRIX_OP_RM;
138 m->auth_level = BRIX_AUTH_DELETE;
139 break;
140 case kXR_rmdir:
141 m->verb = "RMDIR"; m->op_id = BRIX_OP_RMDIR;
142 m->auth_level = BRIX_AUTH_DELETE;
143 break;
144 case kXR_chmod:
145 m->verb = "CHMOD"; m->op_id = BRIX_OP_CHMOD;
146 m->auth_level = BRIX_AUTH_UPDATE;
147 break;
148 case kXR_truncate:
149 m->verb = "TRUNCATE"; m->op_id = BRIX_OP_TRUNCATE;
150 m->auth_level = BRIX_AUTH_UPDATE;
151 break;
152 case kXR_mv:
153 m->verb = "MV"; m->op_id = BRIX_OP_MV;
154 m->auth_level = BRIX_AUTH_DELETE;
155 m->aop = BRIX_AOP_RENAME;
156 ngx_memzero(&mv, sizeof(mv));
157 if (brix_mv_parse_paths(ctx, c, &mv) != NGX_OK) {
158 return ctx->write_rc;
159 }
160 ngx_cpystrn((u_char *) m->path, (u_char *) mv.src_buf,
161 sizeof(m->path));
162 ngx_cpystrn((u_char *) m->dst, (u_char *) mv.dst_buf,
163 sizeof(m->dst));
164 break;
165 default:
166 return BRIX_DISPATCH_CONTINUE;
167 }
168
169 if (ctx->recv.cur_reqid != kXR_mv) {
170 rc = manager_mutation_decode_single(ctx, c, m);
171 if (rc != BRIX_DISPATCH_CONTINUE) {
172 return rc;
173 }
174 }
175
176 rc = manager_mutation_lexical_path(ctx, c, conf, m, 0);
177 if (rc != BRIX_DISPATCH_CONTINUE || ctx->recv.cur_reqid != kXR_mv) {
178 return rc;
179 }
180 return manager_mutation_lexical_path(ctx, c, conf, m, 1);
181}

183/* manager_mutation_authorize — front-side path authorization for orchestration.
184 *
185 * WHAT: Runs native/XrdAcc, VO, and write-token checks for the source path and,
186 * for MV, independent destination Insert authorization.
187 * WHY: Proxy and fan-out execute with server authority, so relying on the data
188 * node discards the initiating client's path policy.
189 * HOW: Reuses brix_auth_gate_op; MV supplies its two operation-specific XrdAcc
190 * verbs, while other opcodes use the local handler's derived mapping.
191 */
192static ngx_int_t
193manager_mutation_authorize(brix_ctx_t *ctx, ngx_connection_t *c,
194 ngx_stream_brix_srv_conf_t *conf, const manager_mutation_t *m)
195{
196 if (brix_auth_gate_op(ctx, c, m->op_id, m->verb, m->path, m->resolved,
197 conf, m->auth_level, 1, m->aop) != NGX_OK)
198 {
199 return ctx->write_rc;
200 }
201
202 if (ctx->recv.cur_reqid != kXR_mv) {
203 return BRIX_DISPATCH_CONTINUE;

206 m->dst_resolved, conf, BRIX_AUTH_UPDATE, 1,

208 {
209 return ctx->write_rc;
210 }
211 return BRIX_DISPATCH_CONTINUE;

214/* manager_mutation_applies — classify manager-routable namespace writes.
215 *
216 * WHAT: Identifies path-form mkdir/rm/rmdir/mv/chmod/truncate in manager mode.
217 * WHY: Preflight and later orchestration must agree exactly on which requests
218 * may execute through manager authority.
219 * HOW: Excludes handle-form truncate and every handle/data-write opcode.
220 */
221static int
222manager_mutation_applies(const brix_ctx_t *ctx,
223 const ngx_stream_brix_srv_conf_t *conf)
224{
225 if (!conf->manager_mode) {
226 return 0;
227 }
228
229 /* Handle-based truncate carries no path and belongs to the ordinary local
230 * write gate. Every other unlisted opcode also falls through unchanged. */
231 if (ctx->recv.cur_reqid == kXR_truncate && ctx->recv.cur_dlen == 0) {
232 return 0;
233 }
234
235 switch (ctx->recv.cur_reqid) {
236 case kXR_mkdir:
237 case kXR_rm:
238 case kXR_rmdir:
239 case kXR_mv:
240 case kXR_chmod:
241 case kXR_truncate:
242 return 1;
243 default:
244 return 0;
245 }
246}

248/* brix_dispatch_manager_mutation_preflight — authorize before proxy routing.
249 *
250 * WHAT: Authenticates the session, rejects bound secondaries, decodes both MV
251 * operands, and applies client path policy for every manager mutation.
252 * WHY: The top-level proxy shortcut precedes dispatch_write. A busy CMS pin or
253 * configured proxy could otherwise relay a mutation under broader service
254 * credentials without ever reaching the manager boundary below.
255 * HOW: The central dispatcher calls this once after session posture and before
256 * any proxy/rate-limit/orchestration side effect. Successful requests are
257 * decoded again by manager_redirect_mutation only to obtain the route key.
258 */
259ngx_int_t
260brix_dispatch_manager_mutation_preflight(brix_ctx_t *ctx, ngx_connection_t *c,
261 ngx_stream_brix_srv_conf_t *conf)
262{
263 manager_mutation_t m;
264 ngx_int_t rc;
265
266 if (!manager_mutation_applies(ctx, conf)) {
267 return BRIX_DISPATCH_CONTINUE;
268 }
269
270 /* A hard read-only manager is rejected at config merge. Keep a runtime
271 * tripwire so a future configuration path cannot route around that policy;
272 * endpoint posture precedes identity and token evaluation. */
273 if (conf->common.read_only == 1) {
274 brix_io_monitor_record_err(&ctx->io_monitor, BRIX_ERR_FORBIDDEN);
275 return brix_send_error(ctx, c, kXR_fsReadOnly,
276 "this is a read-only server");
277 }
278
279 rc = brix_dispatch_require_auth(ctx, c);
280 if (rc != BRIX_DISPATCH_CONTINUE) {
281 return rc;
282 }
283 if (ctx->is_bound) {
284 return brix_send_error(ctx, c, kXR_NotAuthorized,
285 "bound streams may only read or write primary handles");
286 }
287
288 rc = manager_mutation_prepare(ctx, c, conf, &m);
289 if (rc != BRIX_DISPATCH_CONTINUE) {
290 return rc;
291 }
292 return manager_mutation_authorize(ctx, c, conf, &m);
293}

295/* manager_redirect_mutation — Plane B manager orchestration.
296 *
297 * WHAT: Routes a preflighted manager mutation to CMS fan-out or one selected
298 * data node, bypassing only the manager's local-export allow_write flag.
299 * WHY: Pure redirectors usually have no writable local export; executing there
300 * is wrong, while the selected node retains the typed mutation-policy gate.
301 * HOW: Re-decodes the immutable route key after the dispatcher-level preflight,
302 * then starts fan-out or select/redirect/proxy with no intervening policy.
303 */
304static ngx_int_t
305manager_redirect_mutation(brix_ctx_t *ctx, ngx_connection_t *c,
306 ngx_stream_brix_srv_conf_t *conf)
307{
308 manager_mutation_t m;
309 char host[256];
310 uint16_t port;
311 ngx_int_t rc;
312
313 if (!manager_mutation_applies(ctx, conf)) {
314 return BRIX_DISPATCH_CONTINUE;
315 }
316
317 /* The central dispatcher already completed the security preflight before
318 * its proxy shortcut. Decode again only to obtain the immutable route key;
319 * this function must perform no authorization after orchestration starts. */
320 rc = manager_mutation_prepare(ctx, c, conf, &m);
321 if (rc != BRIX_DISPATCH_CONTINUE) {
322 return rc;
323 }
324
325 /* Phase-89 W8: with brix_cms_fanout on, a delete of a replicated path
326 * fans out to EVERY holder node instead of redirecting to one (NGX_AGAIN
327 * parks the client until the reply window settles). NGX_DECLINED means
328 * fan-out did not engage — fall through to the single-node redirect. */
329 rc = brix_cms_fanout_mutation(ctx, c, conf, m.path);
330 if (rc != NGX_DECLINED) {
331 return rc;
332 }
333
334 if (brix_srv_select_or_blacklisted(m.path, 1, host, sizeof(host), &port)) {
335 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, c->log, 0,
336 "brix: manager redirect mutation \"%s\" -> %s",
337 m.path, host);
338 return brix_cms_answer_selected(ctx, c, conf, host, port);
339 }
340
341 return brix_send_error(ctx, c, kXR_FSError,
342 "no data server available for path");
343}

346 *
347 * WHAT: binds a write/mutation wire opcode to its handler. Exactly one of

351 * altering any of them.
352 * WHY: a single table row per opcode is the readable source of truth for
353 * routing and collapses the former 14-case gate ladder into one scan,
354 * keeping the dispatcher under the complexity cap. Row order is
355 * irrelevant to behaviour (opcodes are distinct); it follows the historic

388/* brix_dispatch_write_opcode — phase 2 routing (from handshake/dispatch.c) for
389 * the mutating opcodes: table lookup on ctx->recv.cur_reqid over data writes
390 * (kXR_write, kXR_pgwrite CRC32c-checked, kXR_writev), kXR_sync, structure
391 * changes (truncate/mkdir/rm/rmdir), mv/chmod, kXR_chkpoint, and the vendor
392 * ext ops. Each matched opcode passes the write gate — brix_dispatch_require_write,
393 * stricter than require_auth (needs auth AND conf->common.allow_write) — before
394 * its handler. In manager_mode, path-based mutations pass the manager's
395 * authentication/bound/path-policy preflight before redirect or fan-out;
396 * local-export allow_write is intentionally not applied to a pure redirector.
397 * Returns the handler result, or
398 * BRIX_DISPATCH_CONTINUE when not a write opcode. */
399ngx_int_t
400brix_dispatch_write_opcode(brix_ctx_t *ctx, ngx_connection_t *c,
401 ngx_stream_brix_srv_conf_t *conf)
402{
403 const brix_wr_route_t *route;
404 ngx_uint_t i;
405 ngx_int_t rc;
406
407 /* Plane B: in manager mode, redirect path-based mutations to a data node
408 * before the local write gate (which the manager may not satisfy). */
409 rc = manager_redirect_mutation(ctx, c, conf);
410 if (rc != BRIX_DISPATCH_CONTINUE) {
411 return rc;
412 }
413
414 for (i = 0; i < sizeof(brix_wr_routes) / sizeof(brix_wr_routes[0]); i++) {
415 route = &brix_wr_routes[i];
416 if (route->op != (int) ctx->recv.cur_reqid) {
417 continue;
418 }
419
420 /* Matched a write opcode: enforce the write gate before dispatching. */
421 rc = brix_dispatch_require_write(ctx, c, conf);
422 if (rc != BRIX_DISPATCH_CONTINUE) {
423 return rc;
424 }
425
426 return route->fn2 ? route->fn2(ctx, c)
427 : route->fn3(ctx, c, conf);
428 }
429
430 return BRIX_DISPATCH_CONTINUE;
431}
