1/*
2 * cred_stage.h — one private staging area for short-lived credential material.
3 *
4 * Several subsystems must hand a live secret (an OAuth2 subject/bearer token, a
5 * delegated X.509 proxy PEM) to a fork/exec'd helper that reads it from a FILE
6 * path rather than a fd — curl -d @file, the GSI blocking client, etc. Writing
7 * that file into world-traversable /tmp opens a co-tenant race (CWE-377): during
8 * the brief window the file exists a shell on the same host / mount namespace can
9 * open() the inode by name and read the secret.
10 *
11 * This module is the SINGLE place that decides where such files live and enforces
12 * the directory's safety. Every credential stager routes through it instead of
13 * open-coding an mkstemp("/tmp/..."). The staging dir is a per-uid 0700 tmpfs
14 * directory (/dev/shm/brix-creds.<euid>); a 0700 parent owned by the worker uid
15 * removes the co-tenant open() entirely, and tmpfs keeps nothing on disk or in
16 * backups. There is deliberately NO /tmp fallback: if a private directory cannot
17 * be guaranteed, callers fail closed rather than stage a secret insecurely.
18 *
19 * Pure libc (no nginx headers) so the path/permission logic is unit-testable
20 * standalone — see tests/c/test_cred_stage.c.
21 */
22

28/*
29 * brix_cred_stage_dir — resolve (creating if absent) the process's private
30 * credential staging directory and copy its NUL-terminated path into out.
31 *
32 * The directory is created 0700 and REQUIRED to be a real directory owned by the
33 * effective uid with no group/other access bits; any deviation (foreign owner, a
34 * loosened mode, a symlink swap, tmpfs absent) is treated as unsafe.
35 *
36 * Returns 0 on success; -1 with errno set when a private directory cannot be
37 * guaranteed (EPERM if a pre-existing path fails the owner/mode check). Callers
38 * MUST fail closed on -1 — never fall back to a world-traversable location.
39 */
40int brix_cred_stage_dir(char *out, size_t outsz);

42/*
43 * brix_cred_stage_write — atomically stage len bytes into a fresh 0600 file under
44 * the private staging directory and return its path in path_out.
45 *
46 * The file name begins with prefix (e.g. "tpc_token_body_") followed by mkstemp's
47 * unique suffix. On any failure no file is left behind and -1 is returned with
48 * errno set. The caller owns `bytes`; this function does not retain or scrub them
49 * (the caller cleanses its own buffer) and remains responsible for unlink()ing the
50 * returned path once the helper subprocess has consumed it.
51 *
52 * Returns 0 on success, -1 with errno set on failure.
53 */
54int brix_cred_stage_write(const char *prefix, const void *bytes, size_t len,
55 char *path_out, size_t path_outsz);

57/*
58 * ---- The shared credential-write verb (phase-108 C11) ----
59 *
60 * One engine for EVERY file this server creates whose content is a live
61 * secret. Before it, four sites hand-rolled the same dance with divergent
62 * invariants (deleg_capture wrote a forwarded TGT into getenv("TMPDIR")||
63 * "/tmp" — the exact CWE-377 this module exists to prevent). The engine owns
64 * the invariant table once: O_CREAT|O_EXCL|O_WRONLY|O_NOFOLLOW|O_CLOEXEC
65 * create at 0600 (+ defensive fchmod), a directory that is 0700/euid-owned/
66 * (mode & 0077)==0 re-checked on every call for BOTH arms, EINTR-safe
67 * full-length write, close() checked as a write error, unlink on every
68 * failure branch.
69 *
70 * Forward-declared so this header stays nginx-free (ngx_log_t is a typedef of
71 * struct ngx_log_s); the pure engine and the standalone unit never touch it.
72 */
73struct ngx_log_s;

75/* The lifetime arm is stated by the CALLER, never inferred (no statfs probe:
76 * a probe would turn a mount change into a silent durability change). */
77typedef enum {
78 BRIX_CRED_ARM_VOLATILE = 0, /* per-uid tmpfs staging dir; consumed and
79 * unlinked by the caller; NEVER fsynced —
80 * the §3.3 tmpfs carve-out, weaker on
81 * purpose: durability of a secret that must
82 * not survive reboot is an anti-goal */
83 BRIX_CRED_ARM_PERSISTENT, /* caller-named dir + final name; fsync data,
84 * publish by rename, fsync parent dir */
85 BRIX_CRED_ARM_COUNT
86} brix_cred_arm_t;

107/*
108 * brix_cred_write_engine — the pure-libc mechanics of both arms.
109 *
110 * VOLATILE: stages under brix_cred_stage_dir() as "<prefix><random>"; the
111 * temp file IS the product (path_out) and no rename or fsync happens.
112 * PERSISTENT: creates "<dir>/.<name>.<random>", fsyncs the data, publishes it
113 * as "<dir>/<name>" by rename, then fsyncs the parent directory; path_out is
114 * the final path. len == 0 is valid on both arms (an empty 0600 file — the
115 * krb5 ccache pre-creation shape).
116 *
117 * Returns 0, or -1 with errno: EINVAL (request shape — missing/'/'-bearing
118 * name or prefix, out-of-range arm/kind, NULL bytes with len > 0), EPERM (the
119 * directory safety check failed — fail closed, never a fallback location),
120 * ENAMETOOLONG, or the failing syscall's errno (write/fsync/close/rename all
121 * checked). On failure before publish the temp file is always unlinked; a
122 * PERSISTENT parent-fsync failure after rename reports -1 but does NOT unlink
123 * the published file (the rename may have replaced a live credential —
124 * destroying it would be worse than the lost durability barrier).
125 *
126 * Pure libc; unit-tested standalone in tests/c/test_cred_stage.c.
127 */
128int brix_cred_write_engine(const brix_cred_write_req_t *req,
129 const void *bytes, size_t len,
130 char *path_out, size_t path_outsz);
