1/*
2 * server_conf_merge_cluster.c — server-block merge for the third-party-copy and
3 * cluster/session configuration areas.
4 *
5 * WHAT: Owns brix_merge_srv_tpc() (TPC allowances, key TTL, transfer caps +
6 * reaper age, SSI/CNS, outbound credentials) and brix_merge_srv_cluster()
7 * (manager/redirector mode, staged uploads, pipeline/registry/session
8 * sizing, health checks, traffic mirror, CMS client, listen port,
9 * checksum-scan limits, VO/group/manager-map rule arrays + redirector
10 * inheritance), together with the file-local per-concern helpers.
11 * WHY: Split (phase-79 file-size cap) out of the former 1249-line
12 * server_conf.c. Both entry points are non-static (declared in
13 * server_conf_internal.h) because the top-level orchestrator in
14 * server_conf.c calls them in linear order; every sub-helper stays
15 * file-local.
16 * HOW: Standard nginx parent->child inheritance via ngx_conf_merge_* and the
17 * BRIX_MERGE_* macros, one helper per concern group, invoked in the
18 * original order so cross-group derivations (pipeline-depth clamp, CMS
19 * read-timeout from the heartbeat interval) still observe their
20 * already-merged inputs. Config-time side effects (CNS collect flag, TPC
21 * registry reaper age) run once inputs settle, exactly as before. No
22 * behaviour change from the split.
23 */

40/* Third-party copy (TPC): local/private allowances, key TTL, transfer caps and
41 * the abandoned-slot reaper age, and the outbound OAuth2/bearer credentials. */
42char *
43brix_merge_srv_tpc(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
44 ngx_stream_brix_srv_conf_t *prev)
45{
46 ngx_conf_merge_value(conf->ssi_enable, prev->ssi_enable, 0);
47 ngx_conf_merge_value(conf->ssi_cta_enable, prev->ssi_cta_enable, 0);
48 /* defaults mirror BRIX_SSI_MAX_INFLIGHT (8) and the 1 MiB req/resp caps. */
49 ngx_conf_merge_uint_value(conf->ssi_max_inflight, prev->ssi_max_inflight, 8);
50 ngx_conf_merge_size_value(conf->ssi_request_max, prev->ssi_request_max, 1u << 20);
51 ngx_conf_merge_size_value(conf->ssi_response_max, prev->ssi_response_max, 1u << 20);
52 ngx_conf_merge_str_value(conf->ssi_cta_journal, prev->ssi_cta_journal, "");
53 ngx_conf_merge_uint_value(conf->ssi_cta_executor, prev->ssi_cta_executor, 0);
54 ngx_conf_merge_uint_value(conf->cns_mode, prev->cns_mode, BRIX_CNS_OFF);
55 if (conf->cns_mode == BRIX_CNS_COLLECT) {
56 brix_cns_set_collect(1); /* §6: this node maintains the CNS inventory */
57 }
58 ngx_conf_merge_value(conf->tpc_delegate, prev->tpc_delegate, 0);
59 ngx_conf_merge_value(conf->tpc_push, prev->tpc_push, 0);
60 /* Phase-70/opportunistic: passthrough of the client's own inbound bearer JWT
61 * to the TPC source is ON by default so token-authenticated pulls forward the
62 * end-user identity without any per-server opt-in. The default path is
63 * OPPORTUNISTIC (token_mode "passthrough-opt", set in tpc_init_dst_file): an
64 * inbound token is used when present, but its ABSENCE falls back to GSI proxy
65 * delegation / static bearer file / anonymous exactly as before this default
66 * flip — never a new denial. Only an explicit client tpc.token_mode=passthrough
67 * stays STRICT/fail-closed. Operators can still set the directive to `off`. */
68 ngx_conf_merge_msec_value(conf->tpc_key_ttl_ms, prev->tpc_key_ttl_ms,
69 BRIX_TPC_KEY_TTL_MS);
70 /* Phase 51 (B2): default native-TPC absolute wall-clock cap to a generous
71 * 24h so a wedged transfer cannot pin a thread-pool worker forever; 0 still
72 * means unlimited (back-compat). Progress-based stall detection (the curl
73 * low-speed bounds, webdav/tpc_config.c) is the primary guard — this is only
74 * the absolute backstop, large enough never to clip a real transfer. */
75 ngx_conf_merge_uint_value(conf->tpc_max_transfer_secs,
76 prev->tpc_max_transfer_secs, BRIX_CONFIG_TPC_MAX_TRANSFER_SECS);
77 /* F7: multihop hop budget and the multi-stream cap. Both are range-checked
78 * here (not in a setter) so a block that only inherits still fails loud. */
79 ngx_conf_merge_uint_value(conf->tpc_max_hops, prev->tpc_max_hops,
80 TPC_HOPS_DEFAULT);
81 ngx_conf_merge_uint_value(conf->tpc_streams, prev->tpc_streams, 1);
82 if (conf->tpc_max_hops > BRIX_TPC_HOPS_MAX) {
83 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
84 "brix_tpc_max_hops must be between 0 and %d",
85 BRIX_TPC_HOPS_MAX);
86 return NGX_CONF_ERROR;
87 }
88 if (conf->tpc_streams < 1 || conf->tpc_streams > TPC_STREAMS_MAX) {
89 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
90 "brix_tpc_streams must be between 1 and %d",
91 TPC_STREAMS_MAX);
92 return NGX_CONF_ERROR;
93 }
94 /* Hostile-network completion/integrity gates for the native TPC pull, both
95 * default off (a size mismatch always fails regardless; these only govern the
96 * "no size" and "verify content checksum" postures). */
97 /* tpc_verify_checksum merge -> ngx_http_brix_shared_merge
98 * (common.tpc_verify_checksum, str "" = off) — phase-101 W4. */
99 ngx_conf_merge_value(conf->tpc_transfer_max_age,
100 prev->tpc_transfer_max_age, 0);
101 /* Phase 39 (WS5): publish the abandoned-slot reaper age to the shared TPC
102 * registry (config-time, before fork). Guarded so a 0-default block does not
103 * disable a sibling block that enabled it. */
104 if (conf->tpc_transfer_max_age > 0) {
105 brix_tpc_registry_set_max_age((time_t) conf->tpc_transfer_max_age);
106 }
107 return NGX_CONF_OK;
108}

110/*
111 * WHAT: merge the node sizing group — manager mode, staged/resumable uploads,
112 * and the pipeline/registry/session/redirector-cache slot counts.
113 * WHY: the pipeline-window clamp is its own decision; isolating it keeps the
114 * cluster orchestrator flat.
115 * HOW: standard child<-parent inheritance, then clamp pipeline_depth into
116 * [MIN, MAX] (the recv-loop ring arithmetic needs a positive bounded
117 * modulus).
118 */
119static void
120brix_merge_srv_cluster_sizing(ngx_stream_brix_srv_conf_t *conf,
121 ngx_stream_brix_srv_conf_t *prev)
122{
123 ngx_conf_merge_value(conf->manager_mode, prev->manager_mode, 0);
124 brix_node_caps_conf_merge(&conf->caps, &prev->caps);
125 /* Uploads are staged + resumable by DEFAULT (atomic commit-on-close, resume
126 * across a restart). Set brix_upload_resume off to opt out. */
127 ngx_conf_merge_value(conf->upload_resume, prev->upload_resume, 1);
128 ngx_conf_merge_str_value(conf->upload_stage_dir, prev->upload_stage_dir, "");
129 ngx_conf_merge_uint_value(conf->pipeline_depth, prev->pipeline_depth,
130 BRIX_PIPELINE_DEPTH_DEFAULT);
131 /* Clamp the in-flight pipeline window to a sane range: >=1 (the recv loop and
132 * ring arithmetic require a positive modulus) and <=MAX (bounds per-connection
133 * out_ring/rd_pool memory). */
134 if (conf->pipeline_depth < BRIX_PIPELINE_DEPTH_MIN) {
135 conf->pipeline_depth = BRIX_PIPELINE_DEPTH_MIN;
136 } else if (conf->pipeline_depth > BRIX_PIPELINE_DEPTH_MAX) {
137 conf->pipeline_depth = BRIX_PIPELINE_DEPTH_MAX;
138 }
139 ngx_conf_merge_uint_value(conf->registry_slots, prev->registry_slots, 128);
140 ngx_conf_merge_uint_value(conf->session_slots, prev->session_slots,
141 BRIX_SESSION_REGISTRY_SLOTS);
142 /* Left UNSET when unconfigured; postconfiguration treats UNSET as
143 * "use the compile-time default" (BRIX_REDIR_CACHE_SLOTS). */
144 ngx_conf_merge_uint_value(conf->redir_cache_slots, prev->redir_cache_slots,
145 NGX_CONF_UNSET_UINT);
146}

191/*
192 * WHAT: merge the CMS selection knobs — load weight, path affinity, rm/rmdir
193 * fan-out, cluster role, state relay, the delay-server floor and the
194 * scheduler weight vector.
195 * WHY: several of these are process-wide set-once installs (one registry, one
196 * weight vector), so they must all be resolved in the same pass, before
197 * fork — grouping them makes that shared property visible.
198 * HOW: merge each directive, clamp the ones that are hints rather than
199 * contracts, then install the process-wide values.
200 */
201static void
202brix_merge_srv_cms_selection(ngx_stream_brix_srv_conf_t *conf,
203 ngx_stream_brix_srv_conf_t *prev)
204{
205 /* Phase-89 W4: load-weighted selection knob (0 = legacy scoring). Values
206 * are clamped, not rejected — the weight is a balancing hint. Applied
207 * process-wide like brix_manager_stale_after (set once before fork). */
208 ngx_conf_merge_value(conf->cms.load_weight, prev->cms.load_weight, 0);
209 if (conf->cms.load_weight < 0) {
210 conf->cms.load_weight = 0;
211 }
212 if (conf->cms.load_weight > 100) {
213 conf->cms.load_weight = 100;
214 }
215 if (conf->cms.load_weight > 0) {
216 brix_srv_set_load_weight((ngx_uint_t) conf->cms.load_weight);
217 }
218
219 /* Phase-89 W5: path-affinity sticky selection (process-wide, like the
220 * load weight — set once before fork) + multi-source locate replies. */
221 ngx_conf_merge_value(conf->cms.affinity, prev->cms.affinity, 0);
222 ngx_conf_merge_value(conf->cms.locate_multi, prev->cms.locate_multi, 0);
223 if (conf->cms.affinity) {
224 brix_srv_set_affinity(1);
225 }
226
227 /* Phase-89 W8: rm/rmdir fan-out to all holder nodes + its reply window.
228 * The window is the success deadline (node executor is silent on success),
229 * so it must be short enough to sit inside any client request timeout. */
230 ngx_conf_merge_value(conf->cms.fanout, prev->cms.fanout, 0);
231 ngx_conf_merge_msec_value(conf->cms.fanout_window,
232 prev->cms.fanout_window, 500);
233
234 /* Phase-61 W7: explicit upward-leg cluster role (auto = legacy mode
235 * derivation + permissive inbound dispatch). */
236 ngx_conf_merge_uint_value(conf->cms.role, prev->cms.role,
237 BRIX_CMS_ROLE_AUTO);
238
239 /* Phase-115 W2.1: selection answer — redirect keeps the stock wire
240 * behaviour; proxy is opt-in per server block. */
241 ngx_conf_merge_uint_value(conf->cms.response, prev->cms.response,
242 BRIX_CMS_RESPONSE_REDIRECT);
243
244 /* Phase-61 W7: multi-tier kYR_state recursion (off = registry-only). */
245 ngx_conf_merge_value(conf->cms.state_relay, prev->cms.state_relay, 0);
246
247 /* §2.2 (cms.delay servers): SUPCount floor + hold seconds. Process-wide
248 * set-once like the load weight — the registry is one table. */
249 ngx_conf_merge_value(conf->cms.delay_servers, prev->cms.delay_servers, 0);
250 ngx_conf_merge_value(conf->cms.delay_hold, prev->cms.delay_hold, 5);
251 if (conf->cms.delay_servers > 0) {
252 brix_srv_set_delay_servers((ngx_uint_t) conf->cms.delay_servers);
253 }
254 if (conf->cms.delay_hold < 1) {
255 conf->cms.delay_hold = 1; /* kXR_wait 0 would busy-loop clients */
256 }
257
258 /* §2.3 (cms.sched): component weights + fuzz + maxload; any weight set
259 * installs the process-wide vector (all-zero = legacy scoring). */
260 ngx_conf_merge_value(conf->cms.sched_cpu, prev->cms.sched_cpu, 0);
261 ngx_conf_merge_value(conf->cms.sched_io, prev->cms.sched_io, 0);
262 ngx_conf_merge_value(conf->cms.sched_runq, prev->cms.sched_runq, 0);
263 ngx_conf_merge_value(conf->cms.sched_mem, prev->cms.sched_mem, 0);
264 ngx_conf_merge_value(conf->cms.sched_pag, prev->cms.sched_pag, 0);
265 ngx_conf_merge_value(conf->cms.sched_space, prev->cms.sched_space, 0);
266 ngx_conf_merge_value(conf->cms.sched_fuzz, prev->cms.sched_fuzz, 0);
267 ngx_conf_merge_value(conf->cms.sched_maxload, prev->cms.sched_maxload, 0);
268 if (conf->cms.sched_cpu | conf->cms.sched_io | conf->cms.sched_runq
269 | conf->cms.sched_mem | conf->cms.sched_pag | conf->cms.sched_space
270 | conf->cms.sched_fuzz | conf->cms.sched_maxload)
271 {
272 brix_srv_sched_t sched;
273 sched.cpu = (ngx_uint_t) conf->cms.sched_cpu;
274 sched.io = (ngx_uint_t) conf->cms.sched_io;
275 sched.runq = (ngx_uint_t) conf->cms.sched_runq;
276 sched.mem = (ngx_uint_t) conf->cms.sched_mem;
277 sched.pag = (ngx_uint_t) conf->cms.sched_pag;
278 sched.space = (ngx_uint_t) conf->cms.sched_space;
279 sched.fuzz = (ngx_uint_t) conf->cms.sched_fuzz;
280 sched.maxload = (ngx_uint_t) conf->cms.sched_maxload;
281 brix_srv_set_sched(&sched);
282 }

284 /* §2.4 (cms.space, manager half): process-wide set-once like the sched

287 ngx_conf_merge_value(conf->cms.space_enforce, prev->cms.space_enforce, 0);

295
296 /* §2.5: stage-aware selection (off = legacy least-utilised pick). */

303 * mode, the external perf program and the advertised alternate data server.
304 * WHY: these describe where the cluster's load and location facts come from,

370/*
371 * WHAT: merge the heartbeat interval and the resilience deadlines that derive
372 * from it (read/send/keepalive/tcp_user + the connect-backoff pair left
373 * UNSET for worker-time choice).
374 * WHY: the read-timeout auto-derivation depends on the just-merged interval;
375 * isolating it keeps that ordering explicit and unbreakable — no directive
376 * merged after this point can change what the deadlines derived from.
377 * HOW: merge the interval (floored at 1s to avoid a busy-loop heartbeat), then
378 * derive read_timeout = max(3×interval, 90s) when unset, and default
379 * tcp_user_timeout to the read timeout.
380 */
381static void
382brix_merge_srv_cms_deadlines(ngx_stream_brix_srv_conf_t *conf,
383 ngx_stream_brix_srv_conf_t *prev)
384{
385 ngx_conf_merge_value(conf->cms.interval, prev->cms.interval, 30);
386 if (conf->cms.interval < 1) {
387 /* 0 would arm a 0ms heartbeat timer AND zero the reconnect backoff
388 * (connect.c) — both busy-loops. Floor the heartbeat at 1s. */
389 conf->cms.interval = 1;
390 }
391
392 /*
393 * Phase 50: CMS client resilience deadlines. Resolve here (after
394 * cms_interval is merged) so an unset directive auto-derives a generous
395 * ON-by-default value from the heartbeat interval; an explicit 0 disables.
396 * - read timeout: max(3 x interval, 90s) — a healthy real cmsd pings well
397 * within its interval, so this never trips a conformant manager.
398 * - send timeout: 10s — bounds a manager that stops draining our writes.
399 * - tcp_user_timeout: defaults to the read-timeout as a kernel backstop.
400 */
401 if (conf->cms.read_timeout == NGX_CONF_UNSET_MSEC) {
402 if (prev->cms.read_timeout != NGX_CONF_UNSET_MSEC) {
403 conf->cms.read_timeout = prev->cms.read_timeout;
404 } else {
405 ngx_msec_t d = (ngx_msec_t) conf->cms.interval * 3 * 1000; /* 3×interval in seconds */
406 conf->cms.read_timeout = (d > BRIX_CMS_READ_TIMEOUT_MAX_MS) ? d : BRIX_CMS_READ_TIMEOUT_MAX_MS;
407 }
408 }
409 ngx_conf_merge_msec_value(conf->cms.send_timeout, prev->cms.send_timeout,
410 BRIX_CMS_SEND_TIMEOUT_DEFAULT_MS);
411 ngx_conf_merge_value(conf->cms.tcp_keepalive, prev->cms.tcp_keepalive, 1);
412 if (conf->cms.tcp_user_timeout == NGX_CONF_UNSET_MSEC) {
413 conf->cms.tcp_user_timeout =
414 (prev->cms.tcp_user_timeout != NGX_CONF_UNSET_MSEC)
415 ? prev->cms.tcp_user_timeout
416 : conf->cms.read_timeout;
417 }
418
419 /* Leave these UNSET through the merge so connect.c can pick the manager-locality
420 * (loopback vs remote) profile default at worker start; an explicit directive
421 * still wins and is inherited child<-parent. */
422 ngx_conf_merge_msec_value(conf->cms.initial_delay, prev->cms.initial_delay,
423 NGX_CONF_UNSET_MSEC);
424 ngx_conf_merge_msec_value(conf->cms.connect_retry, prev->cms.connect_retry,
425 NGX_CONF_UNSET_MSEC);
426}

434 * constraint (deadlines derive from the merged heartbeat interval) is
435 * easiest to enforce as call order.
436 * HOW: merge the three plain directives, then run the three groups in the
437 * order their dependencies require — deadlines last.

439static char *

493 * WHAT: merge the VO/group/manager-map rule arrays.
494 * WHY: the three arrays share an identical merge-and-fail-check shape; a helper
495 * keeps the NGX_CONF_ERROR handling out of the orchestrator.
496 * HOW: brix_merge_arrays() each child array over the parent; a NULL result with
497 * a non-empty parent or child signals allocation failure → NGX_CONF_ERROR.
498 */
499static char *
500brix_merge_srv_rules(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
501 ngx_stream_brix_srv_conf_t *prev)
502{

508 child_group_rules,

520 if (conf->manager_map == NULL
521 && (prev->manager_map != NULL || child_manager_map != NULL)) {
522 return NGX_CONF_ERROR;
523 }
524
525 return NGX_CONF_OK;
526}

528/*
529 * WHAT: refuse `brix_manager_mode on` together with `brix_read_only on` at
530 * config time — nginx -t fails and the server does not start.
531 * WHY: the two directives make CONTRADICTORY promises about the same request.
532 * manager_redirect_mutation() (dispatch_write.c) answers a path mutation
533 * with a REDIRECT to a data node, and it runs BEFORE the local write gate
534 * that brix_read_only arms — by design, because a manager holds no data
535 * and therefore cannot answer for it. So on a manager, `brix_read_only on`
536 * does NOT make the namespace read-only: it hands the client the address
537 * of a node that will happily accept the write. An operator who wrote both
538 * directives believes they have a read-only endpoint and does not. That is
539 * a security misconfiguration, not a preference, so it is fatal rather
540 * than a warning: the two are different ROLES and belong in different
541 * server {} blocks.
542 * HOW: both flags have settled by now (common.read_only in the security merge,
543 * manager_mode in the sizing merge above), so a plain conjunction suffices.
544 * Tested by nginx -t; see docs/03-configuration/read-only-root-gateway.md.
545 */
546static char *
547brix_merge_srv_readonly_role_check(ngx_conf_t *cf,
548 ngx_stream_brix_srv_conf_t *conf)
549{
550 if (conf->common.read_only != 1 || conf->manager_mode != 1) {
551 return NGX_CONF_OK;
552 }
553
554 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
555 "brix_manager_mode and brix_read_only are mutually exclusive: a manager "
556 "redirects mkdir/rm/rmdir/mv/chmod/truncate to a data node BEFORE the "
557 "local read-only gate runs, so the endpoint would not be read-only. "
558 "Put the manager and the read-only gateway in separate server {} blocks%s",
559 conf->common.read_only_public == 1
560 ? " (brix_read_only_public implies brix_read_only)" : "");
561
562 return NGX_CONF_ERROR;
563}

565/* Cluster & sessions: manager/redirector mode, write recovery + staged uploads,
566 * pipeline/registry/session sizing, active health checks, the traffic mirror,
567 * the CMS client (+ resilience-timeout derivation), listen port, checksum-scan
568 * limits, and the VO/group/manager-map rule arrays + redirector inheritance.
569 * Also the point where the read-only role check runs: brix_read_only has settled
570 * in the security merge and manager_mode settles in the sizing merge above. */
571char *

573 ngx_stream_brix_srv_conf_t *prev)
574{

577 return NGX_CONF_ERROR;
578 }

580 brix_merge_srv_mirror(conf, prev);
581 if (brix_merge_srv_cms(cf, conf, prev) != NGX_CONF_OK) {
582 return NGX_CONF_ERROR;
583 }

593 prev->http_handoff_idle_timeout, 75000);
594
595 brix_merge_srv_cluster_addrs(conf, prev);
596
597 return brix_merge_srv_rules(cf, conf, prev);
598}
