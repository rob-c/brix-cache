10/*
11 * Phase 94 — TRUE concurrent striped download (opt-in, --parallel).
12 *
13 * WHAT: one thread per bound connection (primary + secondaries), each streaming a
14 * DISJOINT contiguous byte range of the source and pwrite()-ing it into the
15 * destination at its absolute offset. Reassembly is by offset, so the
16 * stripes are order-independent and need no coordination — the real
17 * multi-stream throughput the serial round-robin fan-out cannot give (the
18 * serial pump has one in-flight request at a time).
19 *
20 * WHY: on a high-latency link N connections reading in parallel hide N× the RTT.
21 * kXR_read carries no pathid, so a read on a bound secondary is served on
22 * that connection against the primary-published handle — the same handle the
23 * server publishes to its SHM table for cross-worker bound reads.
24 *
25 * SAFETY: `brix_file` is a POD (fhandle + codec bytes) and the read path only
26 * READS it (streamid comes from each connection's own brix_send), so the
27 * single handle is shared read-only across the workers with no lock; each
28 * worker owns its connection and its own brix_status, and the pwrite ranges
29 * are disjoint. The path is FAIL-CLOSED: any worker error drops the temp and
30 * fails the copy (no single-link resilient ride-out — that is why --parallel
31 * is opt-in and the resilient serial fan-out stays the default).
32 */
33typedef struct {
34 brix_conn *c; /* this worker's connection (never shared) */
35 brix_file *f; /* SHARED read-only handle (POD; read never mutates) */
36 brix_vfs_file *vf; /* SHARED dest VFS handle — disjoint pwrite ranges,
37 * io_uring OFF so posix_pwrite is a plain thread-safe
38 * pwrite(2) at an explicit offset (no shared cursor) */
39 int64_t start;/* inclusive */
40 int64_t end; /* exclusive */
41 int rc; /* 0 ok / -1 fail */
42 brix_status st; /* per-worker status (never shared) */
43} dl_stripe_t;

46static void *
47dl_stripe_worker(void *arg)
48{
49 dl_stripe_t *w = (dl_stripe_t *) arg;
50 uint8_t *buf = (uint8_t *) malloc(XRDC_COPY_CHUNK);
51 int64_t off = w->start;
52
53 if (buf == NULL) {
54 brix_status_set(&w->st, XRDC_EPROTO, 0, "out of memory");
55 w->rc = -1;
56 return NULL;
57 }
58 while (off < w->end) {
59 int64_t remain = w->end - off;
60 size_t want = (remain < (int64_t) XRDC_COPY_CHUNK)
61 ? (size_t) remain : XRDC_COPY_CHUNK;
62 ssize_t n;
63
64 if (brix_copy_quit_requested()) {
65 brix_status_set(&w->st, XRDC_ESOCK, EINTR, "cancelled (signal)");
66 w->rc = -1;
67 free(buf);
68 return NULL;
69 }
70 n = brix_file_read(w->c, w->f, off, buf, want, &w->st);
71 if (n <= 0) {
72 if (n == 0) {
73 brix_status_set(&w->st, XRDC_EPROTO, 0,
74 "short read at %lld", (long long) off);
75 }
76 w->rc = -1;
77 free(buf);
78 return NULL;
79 }
80 if (brix_vfs_pwrite(w->vf, off, buf, (size_t) n, &w->st) != 0) {
81 w->rc = -1;
82 free(buf);
83 return NULL;
84 }
85 off += n;
86 }
87 w->rc = 0;
88 free(buf);
89 return NULL;
90}

93/* Eligible for the concurrent striped path? Explicit opt-in, a real local
94 * destination, and a plain known-size object big enough to be worth ≥2 stripes;
95 * everything else (stdout, --compress, --pgrw, tiny/unknown size) uses the serial
96 * pump. All cheap, no I/O. */
97static int
98dl_parallel_eligible(const download_job_t *job)
99{
100 const brix_copy_opts *o = job->o;
101
102 return o->parallel
103 && job->du->scheme != XRDC_SCHEME_STDIO
104 && !o->pgrw
105 && !(o->compress != NULL && o->compress[0] != '\0')
106 && job->si->size >= 2 * (int64_t) XRDC_COPY_CHUNK;
107}

110/* Partition [0,size) into `nconns` contiguous stripes; the last absorbs the
111 * remainder. Worker 0 uses the primary connection, worker k the (k-1)-th
112 * secondary. All workers share the read-only handle `f` and dest handle `vf`. */
113static void
114dl_plan_stripes(dl_stripe_t *w, int nconns, brix_streamset *ss, brix_conn *primary,
115 brix_file *f, brix_vfs_file *vf, int64_t size)
116{
117 int64_t stride = (size + nconns - 1) / nconns;
118 int k;
119
120 for (k = 0; k < nconns; k++) {
121 w[k].c = (k == 0) ? primary : &ss->sec[k - 1];
122 w[k].f = f;
123 w[k].vf = vf;
124 w[k].start = (int64_t) k * stride;
125 w[k].end = w[k].start + stride;
126 if (w[k].end > size) {
127 w[k].end = size;
128 }
129 if (w[k].start > size) {
130 w[k].start = size; /* empty tail stripe (size < nconns*stride) */
131 }
132 brix_status_clear(&w[k].st);
133 w[k].rc = 0;
134 }
135}

138/* Spawn workers 1..n-1 on their own threads (a spawn failure runs that stripe
139 * inline so none is dropped), run worker 0 inline, join, then fold the verdicts:
140 * fail-closed — any stripe error returns -1 with that stripe's status in *st. */
141static int
142dl_run_stripes(dl_stripe_t *w, pthread_t *th, int nconns, brix_status *st)
143{
144 int k, rc = 0;
145
146 for (k = 1; k < nconns; k++) {
147 if (pthread_create(&th[k], NULL, dl_stripe_worker, &w[k]) != 0) {
148 dl_stripe_worker(&w[k]);
149 th[k] = 0;
150 }
151 }
152 dl_stripe_worker(&w[0]);
153 for (k = 1; k < nconns; k++) {
154 if (th[k] != 0) {
155 pthread_join(th[k], NULL);
156 }
157 }
158 for (k = 0; k < nconns; k++) {
159 if (w[k].rc != 0) {
160 *st = w[k].st; /* surface the first failing stripe's status */
161 rc = -1;
162 break;
163 }
164 }
165 return rc;
166}

169/* Commit (fsync+rename) a fully-successful stripe set and verify its checksum, or
170 * abort the temp so the destination is never partial — mirroring
171 * download_to_local_file. Always closes `vf`. Returns the final verdict. */
172static int
173dl_commit_or_abort(const download_job_t *job, brix_vfs_file *vf, int rc,
174 brix_status *st)
175{
176 int committed = 0;
177
178 if (rc == 0) {
179 rc = brix_vfs_commit(vf, st);
180 if (rc == 0) {
181 committed = 1;
182 if (job->o->cksum != NULL) {
183 rc = download_reconcile_cksum(job, job->du->path, st);
184 if (rc != 0) {
185 unlink(job->du->path); /* drop committed-but-bad file */
186 }
187 }
188 }
189 }
190 if (rc != 0 && !committed) {
191 brix_vfs_abort(vf);
192 }
193 brix_vfs_close(vf);
194 return rc;
195}

198/* Run the concurrent striped download end-to-end. Returns 1 when it handled the
199 * transfer (verdict in *out_rc), or 0 when the request is not eligible for the
200 * parallel path and the caller should run the serial pump. On the handled path
201 * it opens its own read handle + binds the secondaries into job->ss (the caller
202 * still owns the streamset teardown). */
203int
204copy_download_parallel(const download_job_t *job, int *out_rc, brix_status *st)
205{
206 const brix_copy_opts *o = job->o;
207 int64_t size = job->si->size;
208 brix_file f;
209 brix_vfs_file *vf = NULL;
210 brix_vfs_open_opts vopts = {0};
211 int nconns, rc;
212 dl_stripe_t *w;
213 pthread_t *th;
214
215 if (!dl_parallel_eligible(job)) {
216 return 0;
217 }
218 if (brix_file_open_read(job->c, job->su->path, &f, st) != 0) {
219 *out_rc = -1;
220 return 1;
221 }
222 brix_streams_open(job->ss, job->c, o->streams, st); /* best-effort; n may be 0 */
223 nconns = 1 + job->ss->n;
224 if (getenv("BRIX_STREAMS_DEBUG") != NULL) {
225 fprintf(stderr, "brix: parallel-download stripes=%d\n", nconns);
226 }
227
228 /* Atomic temp+rename via the VFS (INV-12), io_uring OFF so posix_pwrite is a
229 * plain pwrite(2) at an explicit offset — thread-safe for the disjoint stripe
230 * ranges (no shared ring / cursor). */
231 vopts.io_uring = XRDC_IO_URING_OFF;
232 vopts.expected_size = size;
233 if (brix_vfs_open(job->du->path,
234 XRDC_VFS_WRITE | (o->force ? XRDC_VFS_FORCE : 0),
235 &vopts, &vf, st) != 0) {
236 brix_file_close(job->c, &f, st);
237 *out_rc = -1;
238 return 1;
239 }
240
241 w = (dl_stripe_t *) calloc((size_t) nconns, sizeof(*w));
242 th = (pthread_t *) calloc((size_t) nconns, sizeof(*th));
243 if (w == NULL || th == NULL) {
244 brix_status_set(st, XRDC_EPROTO, 0, "out of memory");
245 rc = -1;
246 } else {
247 dl_plan_stripes(w, nconns, job->ss, job->c, &f, vf, size);
248 rc = dl_run_stripes(w, th, nconns, st);
249 }
250
251 rc = dl_commit_or_abort(job, vf, rc, st);
252 brix_file_close(job->c, &f, st);
253 free(w);
254 free(th);
255 *out_rc = rc;
256 return 1;
257}
