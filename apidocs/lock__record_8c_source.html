17ngx_int_t
18brix_lock_record_encode(const brix_lock_record_t *e, char *out, size_t outsz)
19{
20 int n;
21
22 /*
23 * Schema v2: `expires` is absolute Unix WALL-CLOCK seconds (not the legacy
24 * v1 monotonic-msec value, which was meaningless after a reboot). The leading
25 * `v=2` lets the decoder reject/expire any pre-upgrade v1 record. `null=1`
26 * marks a lock-null placeholder (RFC 4918 §9.10.1).
27 */
28 n = snprintf(out, outsz,
29 "v=2|token=%s|owner=%s|expires=%lld|scope=%s|depth=%s|null=%d",
30 e->token, e->owner,
31 (long long) e->expires,
32 e->exclusive ? "exclusive" : "shared",
33 e->depth_infinity ? "infinity" : "0",
34 e->is_null ? 1 : 0);
35 return (n > 0 && (size_t) n < outsz) ? NGX_OK : NGX_ERROR;
36}

38/* ---- Apply one decoded `key`/`val` field pair to the lock record ----
39 *
40 * WHAT: Dispatches a single `key=val` pair (already split, NUL-terminated) from
41 * a lock xattr into the matching field of `*e`, and captures the schema version
42 * into `*version` for the `v` key. Unknown keys are ignored (forward-compat).
43 *
44 * WHY: The per-key branch ladder is the sole source of decode complexity; hoisting
45 * it out of brix_lock_record_decode keeps that parser a small loop and makes the
46 * field-mapping table reviewable on its own. Field semantics (schema v2 encoding,
47 * scope/depth string forms, null placeholder) are preserved byte-for-byte.
48 *
49 * HOW:
50 * 1. Compare `key` against each known field name in turn.
51 * 2. On match, parse `val` into the field with the same primitive the inline
52 * ladder used (strtol/strtoll for numerics, ngx_cpystrn for bounded strings,
53 * exact-string equality for the enum-like scope/depth/null fields).
54 * 3. Return with no effect when `key` matches nothing.
55 */
56static void
57brix_lock_record_apply_field(const char *key, char *val,
58 brix_lock_record_t *e, int *version)
59{
60 if (strcmp(key, "v") == 0) {
61 *version = (int) strtol(val, NULL, 10);
62 } else if (strcmp(key, "token") == 0) {
63 ngx_cpystrn((u_char *) e->token, (u_char *) val, sizeof(e->token));
64 } else if (strcmp(key, "owner") == 0) {
65 ngx_cpystrn((u_char *) e->owner, (u_char *) val, sizeof(e->owner));
66 } else if (strcmp(key, "expires") == 0) {
67 e->expires = (int64_t) strtoll(val, NULL, 10);
68 } else if (strcmp(key, "scope") == 0) {
69 e->exclusive = (strcmp(val, "exclusive") == 0) ? 1 : 0;
70 } else if (strcmp(key, "depth") == 0) {
71 e->depth_infinity = (strcmp(val, "infinity") == 0) ? 1 : 0;
72 } else if (strcmp(key, "null") == 0) {
73 e->is_null = (strcmp(val, "1") == 0) ? 1 : 0;
74 }
75}

77ngx_int_t
78brix_lock_record_decode(const char *raw, size_t rawlen, brix_lock_record_t *e)
79{
80 char buf[BRIX_LOCK_XATTR_MAXLEN];
81 char *p, *end, *val, *next;
82 int version = 0;
83
84 if (rawlen == 0 || rawlen >= sizeof(buf)) {
85 return NGX_DECLINED;
86 }
87
88 ngx_memcpy(buf, raw, rawlen);
89 buf[rawlen] = '\0';
90 ngx_memzero(e, sizeof(*e));
91
92 p = buf;
93 end = buf + rawlen;
94
95 while (p < end) {
96 next = strchr(p, '|');
97 if (next != NULL) {
98 *next = '\0';
99 }
100
101 val = strchr(p, '=');
102 if (val != NULL) {
103 *val++ = '\0';
104 brix_lock_record_apply_field(p, val, e, &version);
105 }
106
107 p = next ? next + 1 : end;
108 }
109
110 if (e->token[0] == '\0') {
111 return NGX_DECLINED;
112 }
113
114 /*
115 * Migration guard: a legacy v1 record (no `v=2`) carries a MONOTONIC `expires`
116 * that is meaningless in this process (especially after a reboot). Force it to
117 * 0 (already-expired) so every caller's existing expired-lock cleanup path
118 * deletes it and proceeds — a downgrade therefore releases stale locks rather
119 * than honouring a bogus deadline. Returning NGX_OK (not NGX_DECLINED) is
120 * deliberate: NGX_DECLINED would leave the physical xattr in place and a fresh
121 * XATTR_CREATE LOCK would then wrongly hit EEXIST -> 423.
122 */
123 if (version != 2) {
124 e->expires = 0;
125 }
126
127 return NGX_OK;
128}
