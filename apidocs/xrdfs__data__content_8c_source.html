2 * xrdfs_data_content.c — xrdfs content-inspection data ops (Phase-38 split).
3 *
4 * WHAT: the content-examining busybox ops — cmp (byte compare two files),
5 * grep (regex scan), hexdump (canonical hex+ASCII), and dd (windowed,
6 * rate-limited block copy to stdout).

33/* cmp <path1> <path2> — compare two files on this endpoint. Fast path: same-algo
34 * server checksums (adler32); if they match the files are identical (exit 0), if they
35 * differ exit 1. Falls back to a byte-exact compare when checksums are unavailable.
36 * Quiet on a match (cmp(1) convention); reports the first differing offset otherwise. */
37int
38do_cmp(brix_conn *c, const char *cwd, int argc, char **argv)
39{
40 brix_status st;
41 char p1[XRDC_PATH_MAX], p2[XRDC_PATH_MAX];
42 char h1[160], h2[160];
43
44 if (argc < 3) { fprintf(stderr, "usage: cmp <path1> <path2>\n"); return 50; }
45 build_path(cwd, argv[1], p1, sizeof(p1));
46 build_path(cwd, argv[2], p2, sizeof(p2));
47
48 /* Fast path: compare server checksums (cheap, no bulk transfer). */
49 brix_status_clear(&st);
50 if (brix_query_cksum(c, p1, "adler32", h1, sizeof(h1), &st) == 0) {
51 brix_status s2;
52 brix_status_clear(&s2);
53 if (brix_query_cksum(c, p2, "adler32", h2, sizeof(h2), &s2) == 0) {
54 if (strcmp(h1, h2) == 0) { return 0; }
55 printf("%s %s differ: checksum adler32 (%s vs %s)\n", p1, p2, h1, h2);
56 return 1;
57 }
58 }
59
60 /* Fallback: byte-exact compare. */
61 {
62 uint8_t *b1 = NULL, *b2 = NULL;
63 int64_t l1 = 0, l2 = 0, i, rc;
64 brix_status_clear(&st);
65 if (slurp_file(c, p1, &b1, &l1, &st) != 0) {
66 return xrdfs_report_err("cmp", p1, &st, 0, c);
67 }
68 if (slurp_file(c, p2, &b2, &l2, &st) != 0) {
69 free(b1);
70 return xrdfs_report_err("cmp", p2, &st, 0, c);
71 }
72 rc = 0;
73 for (i = 0; i < l1 && i < l2; i++) {
74 if (b1[i] != b2[i]) {
75 printf("%s %s differ: byte %lld\n", p1, p2, (long long) (i + 1));
76 rc = 1;
77 break;
78 }
79 }
80 if (rc == 0 && l1 != l2) {
81 printf("%s %s differ: EOF (sizes %lld vs %lld)\n", p1, p2,
82 (long long) l1, (long long) l2);
83 rc = 1;
84 }
85 free(b1);
86 free(b2);
87 return (int) rc;
88 }
89}

103/* Scan one read chunk of `got` bytes, reassembling complete lines and matching each
104 * against `re` (numbered = prefix line numbers). Returns 0 normally, 2 on OOM growing
105 * the line buffer. */
106static int
107grep_scan_chunk(const uint8_t *buf, ssize_t got, const regex_t *re, int numbered,
108 grep_scan_t *g)
109{
110 ssize_t k;
111
112 for (k = 0; k < got; k++) {
113 if (buf[k] == '\n') {
114 if (g->llen + 1 > g->lcap) {
115 char *nl = (char *) realloc(g->line, g->llen + 1);
116 if (nl == NULL) { return 2; }
117 g->line = nl; g->lcap = g->llen + 1;
118 }
119 g->line[g->llen] = '\0';
120 g->lineno++;
121 if (regexec(re, g->line, 0, NULL, 0) == 0) {
122 g->matched = 1;
123 if (numbered) { printf("%ld:", g->lineno); }
124 printf("%s\n", g->line);
125 }
126 g->llen = 0;
127 } else {
128 if (g->llen + 1 > g->lcap) {
129 size_t ncap = g->lcap ? g->lcap * 2 : 256;
130 char *nl = (char *) realloc(g->line, ncap);
131 if (nl == NULL) { return 2; }
132 g->line = nl; g->lcap = ncap;
133 }
134 g->line[g->llen++] = (char) buf[k];
135 }
136 }
137 return 0;
138}

192/* grep [-i] [-n] PATTERN <path> — POSIX-regex line match over a streamed file. Lines
193 * are reassembled across read chunks. -i case-insensitive, -n prefix line numbers.
194 * Exit 0 if any line matched, 1 if none, >1 on error (grep(1) convention). */
195int
196do_grep(brix_conn *c, const char *cwd, int argc, char **argv)
197{
198 brix_status st;
199 char path[XRDC_PATH_MAX];
200 grep_args_t a = {0};
201 int cflags = REG_NEWLINE;
202 regex_t re;
203 brix_rfile f;
204 grep_scan_t g = { NULL, 0, 0, 0, 0 };
205 grep_sink_t sink;
206 int rc = 0;
207
208 rc = grep_parse_args(argc, argv, &a);
209 if (rc != 0) { return rc; }
210 rc = 0;
211 if (a.icase) { cflags |= REG_ICASE; }
212 if (regcomp(&re, a.pattern, cflags) != 0) {
213 fprintf(stderr, "xrdfs: grep: bad pattern '%s'\n", a.pattern);
214 return 2;
215 }
216 build_path(cwd, a.arg, path, sizeof(path));
217
218 brix_status_clear(&st);
219 if (brix_rfile_open_read(c, path, NULL, 0, -1, &f, &st) != 0) {
220 rc = xrdfs_report_err("grep", path, &st, 0, c);
221 regfree(&re);
222 return rc > 1 ? rc : 2;
223 }
224 sink.regex = &re;
225 sink.numbered = a.numbered;
226 sink.scan = &g;
227 rc = brix_rfile_pump(&f, 0, -1, 0, grep_sink, &sink, NULL, &st);
228 free(g.line);
229 brix_rfile_close(&f, &st);
230 regfree(&re);
231 if (rc != 0) {
232 (void) xrdfs_report_err("grep", path, &st, 0, c);
233 return 2;
234 }
235 return g.matched ? 0 : 1;
236}

239/* Print one xxd-style row for the `row` bytes at buf[base..], labelled with the
240 * absolute file offset `abs_off`: 8-hex-digit offset, 16 hex columns (padded), then
241 * the printable-ASCII gutter. */
242static void
243hexdump_row(const uint8_t *buf, ssize_t base, ssize_t row, int64_t abs_off)
244{
245 ssize_t j;
246 printf("%08llx ", (unsigned long long) abs_off);
247 for (j = 0; j < 16; j++) {
248 if (j < row) { printf("%02x ", buf[base + j]); }
249 else { printf(" "); }
250 }
251 printf(" |");
252 for (j = 0; j < row; j++) {
253 int ch = buf[base + j];
254 putchar((ch >= 32 && ch < 127) ? ch : '.');
255 }
256 printf("|\n");
257}

311/* dd [if=]<path> [bs=BYTES] [skip=BLOCKS] [count=BLOCKS] [rate=BYTES/s] — read a
312 * windowed, optionally rate-limited slice of a remote file to stdout. bs defaults to
313 * 1 MiB; the window starts at skip*bs and is count*bs bytes (count omitted = to EOF).
314 * rate accepts a K/M/G suffix; 0 = unlimited. A one-line byte summary goes to stderr. */
315/* Parse the dd operand list (bs=/skip=/count=/rate=/if=/bare path) into `a` (pre-seeded
316 * with the command defaults). On a bad operand prints the diagnostic and returns 50;
317 * otherwise returns 0 (a->arg = source path, NULL if none was given). */
318static int
319dd_parse_args(int argc, char **argv, dd_args_t *a)
320{
321 int i;
322
323 for (i = 1; i < argc; i++) {
324 if (strncmp(argv[i], "bs=", 3) == 0) {
325 a->bs = parse_bytes(argv[i] + 3);
326 if (a->bs <= 0 || a->bs > XRDFS_DD_MAXBS) {
327 fprintf(stderr, "xrdfs: dd: bad bs (max 256M)\n"); return 50;
328 }
329 } else if (strncmp(argv[i], "skip=", 5) == 0) {
330 a->skip = strtoll(argv[i] + 5, NULL, 10);
331 if (a->skip < 0) { fprintf(stderr, "xrdfs: dd: bad skip\n"); return 50; }
332 } else if (strncmp(argv[i], "count=", 6) == 0) {
333 a->count = strtoll(argv[i] + 6, NULL, 10);
334 if (a->count < 0) { fprintf(stderr, "xrdfs: dd: bad count\n"); return 50; }
335 } else if (strncmp(argv[i], "rate=", 5) == 0) {
336 int64_t r = parse_bytes(argv[i] + 5);
337 if (r < 0) { fprintf(stderr, "xrdfs: dd: bad rate\n"); return 50; }
338 a->rate = (double) r;
339 } else if (strncmp(argv[i], "if=", 3) == 0) {
340 a->arg = argv[i] + 3;
341 } else if (argv[i][0] != '-') {
342 a->arg = argv[i];
343 }
344 }
345 return 0;
346}

371int
372do_dd(brix_conn *c, const char *cwd, int argc, char **argv)
373{
374 brix_status st;
375 char path[XRDC_PATH_MAX];
376 dd_args_t a = {0};
377 int64_t want_total, off, produced = 0;
378 int rc = 0;
379 brix_rfile f;
380 struct timespec start;
381 dd_sink_t sink;
382
383 a.bs = 1 << 20; a.count = -1; /* skip/rate default to 0 via zero-init */
384
385 rc = dd_parse_args(argc, argv, &a);
386 if (rc != 0) { return rc; }
387 if (a.arg == NULL) {
388 fprintf(stderr, "usage: dd [if=]<path> [bs=N] [skip=N] [count=N] [rate=R]\n");
389 return 50;
390 }
391 build_path(cwd, a.arg, path, sizeof(path));
392 off = a.skip * a.bs;
393 want_total = (a.count >= 0) ? a.count * a.bs : -1;
394
395 brix_status_clear(&st);
396 if (brix_rfile_open_read(c, path, NULL, 0, -1, &f, &st) != 0) {
397 return xrdfs_report_err("dd", path, &st, 0, c);
398 }
399 clock_gettime(CLOCK_MONOTONIC, &start);
400 sink.start = &start;
401 sink.rate = a.rate;
402 sink.produced = 0;
403 rc = brix_rfile_pump(&f, off, want_total, (size_t) a.bs,
404 dd_sink, &sink, &produced, &st);
405 produced = sink.produced;
406 brix_rfile_close(&f, &st);
407 if (rc != 0) {
408 return xrdfs_report_err("dd", path, &st, 0, c);
409 }
410 fprintf(stderr, "%lld bytes copied\n", (long long) produced);
411 return 0;
412}
