12/* WHAT: Send one checkpoint sub-operation for an already-open write handle.
13 * WHY: The client library has no convenience wrapper for the transactional op.
14 * HOW: Pack the authoritative fixed body and retain the response for Query. */
15static int
16checkpoint_op(xrd_conform_execution *execution, brix_conn *connection,
17 const brix_file *file, uint8_t opcode, xrd_conform_wire_reply *reply,
18 brix_status *status)
19{
20 ClientChkPointRequest request;
21 xrdw_chkpoint_req_t fields = { .opcode = opcode };
22 uint16_t wire_status = 0;
23 brix_resp_out response = { &wire_status, &reply->body, &reply->length };
24
25 memset(reply, 0, sizeof(*reply));
26 if (!xrd_conform_reserve_status(execution, sizeof(request), status)) return -1;
27 memcpy(fields.fhandle, file->fhandle, sizeof(fields.fhandle));
28 xrd_creq_begin(&request, sizeof(request), 0, kXR_chkpoint);
29 xrdw_chkpoint_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
30 brix_status_clear(status);
31 return brix_roundtrip(connection, &request, NULL, &response, status);
32}

56 const xrd_conform_scratch_object *object, const void *data, size_t length,

65 return -1;

79{
80 uint8_t actual[64];
81 brix_file file;

83

85 || brix_file_open_read(connection, path, &file, status) != 0) return 0;
86 if (xrd_conform_reserve_status(execution, length, status))
87 got = brix_file_read(connection, &file, 0, actual, length, status);

102static int
103query_checkpoint_capacity(xrd_conform_execution *execution,
104 brix_conn *connection, brix_file *file, uint32_t *maximum,
105 brix_status *status)
106{
107 xrd_conform_wire_reply reply = { 0 };
108 uint32_t used = 0;
109 int rc = checkpoint_op(execution, connection, file, kXR_ckpQuery,
110 &reply, status);
111
112 if (rc != 0 && status->kxr == kXR_Unsupported) {
113 grade(execution, "CKP-001", 1, "checkpoint-explicitly-unsupported",
114 "checkpoint-wrong-answer", status);
115 xrd_conform_result_set(execution, "CKP-002",
116 BRIX_CONFORM_VERDICT_SKIPPED, "checkpoint-unsupported", NULL, NULL);
117 xrd_conform_result_set(execution, "CKP-003",
118 BRIX_CONFORM_VERDICT_SKIPPED, "checkpoint-unsupported", NULL, NULL);
119 xrd_conform_result_set(execution, "CKP-004",
120 BRIX_CONFORM_VERDICT_SKIPPED, "checkpoint-unsupported", NULL, NULL);
121 free(reply.body);
122 return 0;
123 }
124 grade(execution, "CKP-001", rc == 0 && reply.length == 8,
125 "checkpoint-query-implemented", "checkpoint-query-malformed", status);
126 if (rc == 0 && reply.length == 8) {
127 *maximum = xrdw_get_u32(reply.body);
128 used = xrdw_get_u32(reply.body + 4);
129 }
130 grade(execution, "CKP-002", *maximum >= used && *maximum > 0,
131 "checkpoint-capacity-consistent", "checkpoint-capacity-invalid",
132 status);
133 free(reply.body);
134 return 1;
135}

137static int
138rollback_restores_bytes(xrd_conform_execution *execution,
139 brix_conn *connection, brix_file *file,
140 const xrd_conform_scratch_object *object, brix_status *status)
141{
142 static const char before[] = "before-checkpoint";
143 static const char after[] = "after-checkpoint!";
144 xrd_conform_wire_reply reply = { 0 };
145 int began;
146 int wrote = 0;
147 int rolled_back = 0;
148
149 began = checkpoint_op(execution, connection, file, kXR_ckpBegin, &reply,
150 status) == 0;
151 free(reply.body);
152 if (began && xrd_conform_reserve_status(execution, sizeof(after), status))
153 wrote = brix_file_write(connection, file, 0, after, sizeof(after),
154 status) == 0;
155 if (began) {
156 rolled_back = checkpoint_op(execution, connection, file,
157 kXR_ckpRollback, &reply, status) == 0;
158 free(reply.body);
159 }
160 if (xrd_conform_reserve_status(execution, 0, status))
161 brix_file_close(connection, file, status);
162 return began && wrote && rolled_back
163 && read_matches(execution, connection, object->path, before,
164 sizeof(before), status);
165}

167static int
168overlimit_refused_without_loss(xrd_conform_execution *execution,
169 brix_conn *connection, const xrd_conform_scratch_object *object,
170 uint32_t maximum, brix_status *status)

172 xrd_conform_wire_reply reply = { 0 };

176 int refusal_kxr = 0;

178
179 rc = xrd_conform_reserve_status(execution, 0, status)
180 ? brix_truncate(connection, object->path, (int64_t) oversized, status)

182 if (rc == 0 && xrd_conform_reserve_status(execution, 0, status))
183 rc = brix_file_open_update(connection, object->path, 0, &file, status);
184 if (rc == 0) {

189 if (xrd_conform_reserve_status(execution, 0, status))

191 }
192 memset(&info, 0, sizeof(info));

194 && xrd_conform_reserve_status(execution, 0, status)
195 && brix_stat(connection, object->path, &info, status) == 0
196 && (uint64_t) info.size == oversized;

199/* WHAT: Execute the complete safe checkpoint family against one scratch file.
200 * WHY: Query, rollback, and over-limit refusal share one advertised capacity.
201 * HOW: Query first, snapshot/write/rollback, then create a sparse max+1 object
202 * and require refusal without a size change before final cleanup. */
203void
204xrd_conform_probe_checkpoints(xrd_conform_execution *execution,
205 brix_conn *connection)
206{
207 static const char before[] = "before-checkpoint";
208 xrd_conform_scratch_object object;
209 brix_file file;
210 brix_status status;
211 uint32_t maximum = 0;
212 int rollback_ok;
213 int overlimit_ok;
214
215 if (!xrd_conform_family_active(execution, "CKP")
216 || brix_conform_scratch_path(execution->run->config.scratch,
217 "brix-conform-checkpoint", object.path, sizeof(object.path)) != 0)
218 return;
219 brix_status_clear(&status);
220 if (seed_file(execution, connection, &object, before, sizeof(before),
221 &status) != 0 || !xrd_conform_reserve_status(execution, 0, &status)
222 || brix_file_open_update(connection, object.path, 0, &file, &status) != 0)
223 return;
224 if (!query_checkpoint_capacity(execution, connection, &file, &maximum,
225 &status)) {
226 if (xrd_conform_reserve_status(execution, 0, &status))
227 brix_file_close(connection, &file, &status);
228 cleanup_checkpoint_object(execution, connection, &object, &status);
229 return;
230 }
231 rollback_ok = rollback_restores_bytes(execution, connection, &file,
232 &object, &status);
233 grade(execution, "CKP-003", rollback_ok,
234 "checkpoint-rollback-restored-bytes", "checkpoint-rollback-mismatch",
235 &status);
236 overlimit_ok = overlimit_refused_without_loss(execution, connection,
237 &object, maximum, &status);
238 grade(execution, "CKP-004", overlimit_ok,
239 "overlimit-checkpoint-refused-before-data-loss",
240 "overlimit-checkpoint-wrong-answer", &status);
241 cleanup_checkpoint_object(execution, connection, &object, &status);
242}
