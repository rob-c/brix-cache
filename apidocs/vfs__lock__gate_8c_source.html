1/*
2 * vfs_lock_gate.c — the cross-protocol lock gate (phase-107 C7).
3 *
4 * WHAT: brix_vfs_require_unlocked(): may this mutation proceed against the
5 * ctx's target, or does a live foreign lock cover it (EBUSY)?
6 *
7 * WHY: A WebDAV LOCK used to bind only WebDAV verbs; an XRootD or GridFTP
8 * write landed on a locked file unopposed. The gate makes lock coverage
9 * a VFS question, asked once, after the mutation kernel — so EROFS still
10 * precedes every other refusal and a read-only endpoint discloses
11 * nothing about lock state.
12 *
13 * HOW: Walk the resolved target up to the export root (the same ancestor walk
14 * and coverage rule as the WebDAV edge's webdav_check_locks), reading
15 * the shared lock record (core/compat/lock_record.h) through the QUIET
16 * xattr path so the per-request OP_XATTR counter deltas the metrics
17 * conformance contract pins stay depth-independent. A record past expiry
18 * is treated as absent and NOT reaped — reaping is itself a mutation,
19 * and a read-only export must never reap. Ownership is the ctx's
20 * lock-token presentation (ctx->lock_token, the raw If/Lock-Token header
21 * value) matched by substring, exactly as the edge matches, so the two
22 * planes agree on who owns a lock. The enforcement mode comes from the
23 * backend registry keyed on root_canon (absence = STRICT, failing toward
24 * enforcement); a call site cannot opt itself out. Descendant scans for
25 * collection DELETE/MOVE/COPY stay at the WebDAV edge — this gate is the
26 * ancestor half every plane shares. Impure by design (reads xattrs),
27 * which is why it is not part of vfs_policy.c's pure kernel.
28 */

41/* One level of the ancestor walk: the (mutable) ancestor path being examined
42 * plus the (immutable) target path the whole walk is checking. Mirrors the
43 * WebDAV edge's walk struct so the coverage rule reads identically. */
44typedef struct {
45 const char *check;
46 size_t check_len;
47 const char *path;
48 size_t path_len;
49} brix_vfs_lock_walk_t;

51/* The absent-class xattr errnos — the WebDAV lock DB's own list
52 * (prop_xattr.c): no record is stored here, or the backend cannot store one.
53 * EACCES/EPERM stay absent-class so a backend that hides the attribute cannot
54 * wedge every mutation. */
55static int
56brix_vfs_lock_errno_absent(int err)
57{
58 return err == ENODATA || err == ENOATTR || err == ENOENT
59 || err == ENOTSUP || err == EOPNOTSUPP || err == ENOSYS
60 || err == EACCES || err == EPERM;
61}

63/*
64 * brix_vfs_lock_probe_level — evaluate the lock (if any) recorded on one
65 * ancestor path against the walk's target.
66 *
67 * WHAT: NGX_OK when nothing here blocks the target (no record, not a valid
68 * record, expired, not covering, or owned by the presented token);
69 * NGX_BUSY when a live foreign lock covers the target; NGX_ERROR with
70 * errno preserved on a hard xattr read fault.
71 * WHY: Isolating the per-level decision keeps the ascent loop in the entry
72 * point a simple path-walk, exactly as the WebDAV edge splits it.
73 * HOW: Quiet getxattr (no OP_XATTR observation) → decode → expiry → the
74 * edge's coverage rule (exact match, or a depth-infinity ancestor on a
75 * path boundary) → substring token match. Absent-class errnos admit
76 * (brix_vfs_lock_errno_absent above).
77 */
78static ngx_int_t
79brix_vfs_lock_probe_level(brix_vfs_ctx_t *ctx, const brix_vfs_lock_walk_t *w)
80{
81 char raw[BRIX_LOCK_XATTR_MAXLEN + 1];
82 brix_lock_record_t rec;
83 ssize_t n;
84 int covers;
85
86 n = brix_vfs_getxattr_quiet_at(ctx, w->check, BRIX_LOCK_XATTR_KEY,
87 raw, BRIX_LOCK_XATTR_MAXLEN);
88 if (n < 0) {
89 if (brix_vfs_lock_errno_absent(errno)) {
90 return NGX_OK; /* absent-class: no lock recorded here */
91 }
92 return NGX_ERROR; /* hard read fault; errno preserved */
93 }
94
95 if (brix_lock_record_decode(raw, (size_t) n, &rec) != NGX_OK) {
96 return NGX_OK; /* not a lock record */
97 }
98
99 if (rec.expires <= (int64_t) ngx_time()) {
100 /* Expired = absent. Deliberately NOT reaped: reaping is a mutation,
101 * this gate also runs for read-only-adjacent surfaces, and the WebDAV
102 * edge's own expiry cleanup (which already declines on a read-only
103 * export) remains the reaper. */
104 return NGX_OK;
105 }
106
107 /* Lock covers the target if: exact match, or a depth-infinity ancestor on
108 * a path boundary — byte-identical to webdav_check_lock_at. */
109 covers = (w->check_len == w->path_len)
110 || (rec.depth_infinity
111 && w->check_len < w->path_len
112 && (w->check[w->check_len - 1] == '/'
113 || w->path[w->check_len] == '/'));
114
115 if (covers
116 && !(ctx->lock_token != NULL
117 && strstr(ctx->lock_token, rec.token) != NULL))
118 {
119 return NGX_BUSY;
120 }
121
122 return NGX_OK;
123}

125/* Shared refusal tail: book the per-proto refusal metric (in BOTH strict and
126 * advisory — the advisory count is what tells an operator the relaxed mode is
127 * masking real contention), then refuse (strict: EBUSY) or warn-and-admit
128 * (advisory). The warning names the path and operation, NEVER the held token —
129 * the token is a bearer secret. */
130static ngx_int_t
131brix_vfs_lock_refuse(const brix_vfs_ctx_t *ctx, brix_vfs_mutation_op_t op,
132 ngx_uint_t mode)
133{
134 brix_metric_vfs_lock_refused(brix_vfs_metrics_proto(ctx));
135
136 if (mode == BRIX_VFS_LOCK_ADVISORY) {
137 if (ctx->log != NULL) {
138 ngx_log_error(NGX_LOG_WARN, ctx->log, 0,
139 "brix: %s on \"%s\" proceeds under a live foreign "
140 "lock (brix_lock_enforcement advisory)",
141 brix_vfs_mutation_op_name(op),
142 brix_vfs_ctx_path(ctx));
143 }
144 return NGX_OK;
145 }
146
147 errno = EBUSY;
148 return NGX_ERROR;
149}

151/* ---- May this mutation proceed, or is the target under a foreign lock? ----
152 *
153 * WHAT: NGX_OK when no live foreign lock covers ctx's target (or the export
154 * relaxed enforcement); NGX_ERROR with EINVAL for a missing/unconfined
155 * ctx or an out-of-range op, EBUSY under a live foreign lock in strict
156 * mode, or the read fault's errno when a lock record cannot be read in
157 * strict mode (unlocked cannot be proven — fail closed).
158 *
159 * WHY: Position 2 in the mutation-gate order (§3.4): after the policy kernel,
160 * before confinement-derived leaf work — EROFS always precedes EBUSY.
161 *
162 * HOW: Validate op → require confinement → registry mode (OFF reads nothing)
163 * → ancestor walk with the per-level probe above. The walk's stop
164 * conditions are the WebDAV edge's own: the export root is the last
165 * level examined.
166 */
167/*
168 * brix_vfs_lock_walk_gate — the ancestor walk itself, shared by the single and
169 * batch entry points (op validation, confinement and mode lookup are the
170 * caller's).
171 *
172 * cleared_parent (batch only, else NULL): an ancestor path this SWEEP already
173 * walked to the export root and proved lock-free — reaching it ends the walk
174 * without re-reading the chain above. The memo is never consulted for the
175 * target itself (check_len < path_len), so the exact-node probe always runs.
176 *
177 * *clean is set to 0 when any level went unproven — an advisory admit past a
178 * live foreign lock, or an advisory-swallowed read fault — so the batch caller
179 * only memoizes chains that were actually walked clean to the root.
180 */
181static ngx_int_t
182brix_vfs_lock_walk_gate(brix_vfs_ctx_t *ctx, brix_vfs_mutation_op_t op,
183 ngx_uint_t mode, const char *cleared_parent, int *clean)
184{
185 char check[PATH_MAX];
186 size_t root_len;
187 ngx_int_t rc;
188 brix_vfs_lock_walk_t w;
189
190 *clean = 1;
191
192 w.path = brix_vfs_ctx_path(ctx);
193 if (w.path == NULL) {
194 errno = EINVAL;
195 return NGX_ERROR;
196 }
197 w.path_len = strlen(w.path);
198 root_len = strlen(ctx->root_canon);
199
200 ngx_cpystrn((u_char *) check, (u_char *) w.path, sizeof(check));
201 w.check = check;
202
203 for ( ;; ) {
204 w.check_len = strlen(check);
205
206 if (cleared_parent != NULL && w.check_len < w.path_len
207 && strcmp(check, cleared_parent) == 0)
208 {
209 break; /* chain above proven lock-free earlier this sweep */
210 }
211
212 rc = brix_vfs_lock_probe_level(ctx, &w);
213 if (rc == NGX_BUSY) {
214 *clean = 0;
215 return brix_vfs_lock_refuse(ctx, op, mode);
216 }
217 if (rc != NGX_OK) {
218 if (mode != BRIX_VFS_LOCK_ADVISORY) {
219 return NGX_ERROR; /* strict: unreadable ⇒ fail closed */
220 }
221 *clean = 0; /* advisory swallows the fault; level unproven */
222 }
223
224 /* Stop at or above export root (the root itself was just checked). */
225 if (w.check_len <= root_len) {
226 break;
227 }
228
229 if (!brix_lock_path_ascend(check, w.check_len)) {
230 break;
231 }
232
233 if (strlen(check) < root_len) {
234 break;
235 }
236 }
237
238 return NGX_OK;
239}

242brix_vfs_require_unlocked(brix_vfs_ctx_t *ctx, brix_vfs_mutation_op_t op)
243{
244 ngx_uint_t mode;
245 int clean;
246
247 if ((ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT) {
248 errno = EINVAL;
249 return NGX_ERROR;
250 }
251

254 }

257 if (mode == BRIX_VFS_LOCK_OFF) {
258 return NGX_OK;

260

264/* Alternate-target form: gate `path` (a validated confined destination in the
265 * same export) with ctx's identity, token presentation, and enforcement mode.
266 * A stack copy of the ctx with the resolved path swapped keeps the walk, the
267 * quiet reads, and the refusal tail in exactly one body above. */
268ngx_int_t
269brix_vfs_require_unlocked_at(brix_vfs_ctx_t *ctx, const char *path,
270 brix_vfs_mutation_op_t op)
271{

273
274 if (ctx == NULL || path == NULL || path[0] == '\0') {
275 errno = EINVAL;
276 return NGX_ERROR;
277 }
278
279 alt = *ctx;
280 alt.resolved.resolved.data = (u_char *) path;
281 alt.resolved.resolved.len = strlen(path);
282
283 return brix_vfs_require_unlocked(&alt, op);

286/*
287 * Batch form: gate every resolved path of a delete window in ONE sweep.
288 *
289 * WHY: gating a 1,000-key batch through the per-path entry costs ~3 quiet
290 * xattr reads per key over a remote leaf (exact node, parent, export
291 * root) — ~3,000 upstream round trips for one request, and ONE transient
292 * fault among them fail-closes the whole batch. Batch keys overwhelmingly
293 * share a parent directory, and a parent chain walked clean to the export
294 * root once needs no second read.
295 *
296 * HOW: per key the exact-node probe ALWAYS runs; the ancestor walk stops on
297 * reaching the previous key's memoized parent (a flat batch costs n + 2
298 * probes instead of 3n). The memo holds only chains proven fully clean —
299 * an advisory admit or swallowed fault leaves it unset, so advisory-mode
300 * contention books its refusal metric per key exactly as the per-path
301 * gate does. Refusal semantics are unchanged and ATOMIC: the first strict
302 * EBUSY or read fault returns before the caller runs any arm.
303 */
304ngx_int_t
305brix_vfs_require_unlocked_many(brix_vfs_ctx_t *ctx, const char *const *paths,
306 size_t n, brix_vfs_mutation_op_t op)
307{
308 char cleared[PATH_MAX];
309 size_t i, root_len;
310 ngx_uint_t mode;
311 int clean;
312 brix_vfs_ctx_t alt;
313
314 if (paths == NULL || (ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT) {
315 errno = EINVAL;
316 return NGX_ERROR;
317 }
318
319 if (brix_vfs_require_confined(ctx) != NGX_OK) {
320 return NGX_ERROR;
321 }
322
323 mode = brix_vfs_backend_lock_enforcement(ctx->root_canon);
324 if (mode == BRIX_VFS_LOCK_OFF) {
325 return NGX_OK;
326 }
327
328 root_len = strlen(ctx->root_canon);
329 cleared[0] = '\0';
330 alt = *ctx;
331
332 for (i = 0; i < n; i++) {
333 if (paths[i] == NULL || paths[i][0] == '\0') {
334 errno = EINVAL;
335 return NGX_ERROR;
336 }
337
338 alt.resolved.resolved.data = (u_char *) paths[i];
339 alt.resolved.resolved.len = strlen(paths[i]);
340
341 if (brix_vfs_lock_walk_gate(&alt, op, mode,
342 (cleared[0] != '\0') ? cleared : NULL,
343 &clean) != NGX_OK)
344 {
345 return NGX_ERROR; /* EBUSY or the read fault's errno */
346 }
347
348 if (clean) {
349 ngx_cpystrn((u_char *) cleared, (u_char *) paths[i],
350 sizeof(cleared));
351 if (!brix_lock_path_ascend(cleared, strlen(cleared))
352 || strlen(cleared) < root_len)
353 {
354 cleared[0] = '\0'; /* key sits at the root: nothing above */
355 }
356 }
357 }
358
359 return NGX_OK;
360}
