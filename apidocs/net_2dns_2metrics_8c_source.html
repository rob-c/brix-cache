1/*
2 * metrics.c — Prometheus view of the runtime-DNS driver (phase-116).
3 *
4 * WHAT: brix_dns_metrics_emit(): a `brix_dns_targets{state}` gauge (one row
5 * per registry state), the summed `brix_dns_resolutions_total` and
6 * `brix_dns_failures_total` target counters, the per-query
7 * `brix_dns_lookups_total{result}` counter (ok / nxdomain / timeout /
8 * error — completed queries only, never cache hits) and the forward
9 * (`brix_dns_cache_*`) and reverse (`brix_dns_reverse_cache_*`) cache
10 * entries gauge + hits / misses / negative-hits counters, and the
11 * thread-pool bridge's `brix_dns_bridge_requests_total` /
12 * `brix_dns_bridge_timeouts_total`.
13 * WHY: A target stuck in "failed" or "resolving" is the one signal an
14 * operator needs when a brix_cms_manager / brix_upstream / mirror name
15 * never came up; the lookup counter separates "the nameserver says no"
16 * from "the nameserver is unreachable"; the cache counters prove one
17 * query per TTL window. The error text itself lives on the dashboard
18 * panel (api_snapshot_dns.c) and in the error log, never in a label
19 * (I-8: both label sets are closed enumerations).
20 * HOW: Reads the per-worker registry (targets.c) and caches — the worker
21 * answering /metrics reports its own view; every worker runs the same
22 * schedule so the views agree to within one refresh interval. The
23 * lookup counters are ngx_atomic_t because brix_dns_resolve_sync()
24 * completes libc lookups on thread-pool threads (per-process runtime
25 * state, the origin_probe precedent — not a new global policy).
26 */

66static void
67dns_metrics_emit_targets(metrics_writer_t *mw)
68{
69 ngx_uint_t counts[BRIX_DNS_STATE_COUNT];
70 ngx_uint_t i, n, resolutions = 0, failures = 0;
71 const brix_dns_target_t *t;
72
73 brix_dns_targets_count(counts);
74 n = brix_dns_targets_n();
75 for (i = 0; i < n; i++) {
76 t = brix_dns_target_at(i);
77 resolutions += t->resolutions;
78 failures += t->failures;
79 }
80
81 mw_printf(mw, "# HELP brix_dns_targets Runtime-DNS targets registered "
82 "from the configuration, by state (this worker's view).\n"
83 "# TYPE brix_dns_targets gauge\n");
84 for (i = 0; i < BRIX_DNS_STATE_COUNT; i++) {
85 mw_printf(mw, "brix_dns_targets{state=\"%s\"} %lu\n",
86 brix_dns_state_name(i), (unsigned long) counts[i]);
87 }
88 mw_printf(mw, "# HELP brix_dns_resolutions_total Successful runtime "
89 "resolutions of registered targets (this worker).\n"
90 "# TYPE brix_dns_resolutions_total counter\n"
91 "brix_dns_resolutions_total %lu\n"
92 "# HELP brix_dns_failures_total Failed runtime resolution "
93 "attempts of registered targets (this worker).\n"
94 "# TYPE brix_dns_failures_total counter\n"
95 "brix_dns_failures_total %lu\n",
96 (unsigned long) resolutions, (unsigned long) failures);
97}

117/* brix_dns_bridge_* — the thread-pool -> event-loop bridge (resolve_bridge.c).
118 * A blocking caller (TPC pin, cache-origin/gsiftp connect, cvmfs probe, OCSP)
119 * either crosses the bridge and resolves under the worker's own `brix_resolver`
120 * policy, or falls back to libc with a different resolv.conf, a different search
121 * list and no shared cache. requests counts the crossings, timeouts the ones
122 * that gave up waiting for the loop; requests staying at 0 while thread-pool
123 * sites are busy is the signal that every one of them silently took libc. Both
124 * are label-free per-worker counters (I-8, I-DNS-4). */
125static void
126dns_metrics_emit_bridge(metrics_writer_t *mw)
127{
128 ngx_uint_t requests = 0, timeouts = 0;
129
130 brix_dns_bridge_stats(&requests, &timeouts);
131 mw_printf(mw, "# HELP brix_dns_bridge_requests_total Blocking resolutions "
132 "handed to the event loop by a thread-pool caller (this "
133 "worker).\n"
134 "# TYPE brix_dns_bridge_requests_total counter\n"
135 "brix_dns_bridge_requests_total %lu\n",
136 (unsigned long) requests);
137 mw_printf(mw, "# HELP brix_dns_bridge_timeouts_total Bridge crossings that "
138 "timed out waiting for the event loop and fell back to "
139 "libc.\n"
140 "# TYPE brix_dns_bridge_timeouts_total counter\n"
141 "brix_dns_bridge_timeouts_total %lu\n",
142 (unsigned long) timeouts);
143}

170void
171brix_dns_metrics_emit(metrics_writer_t *mw)
172{
173 ngx_uint_t entries, hits, misses, negative_hits;
174
175 dns_metrics_emit_targets(mw);
176 dns_metrics_emit_lookups(mw);
177 dns_metrics_emit_bridge(mw);
178 brix_dns_cache_stats(&entries, &hits, &misses, &negative_hits);
179 dns_metrics_emit_cache(mw, "brix_dns_cache", "forward-DNS", entries,
180 hits, misses, negative_hits);
181 brix_dns_rcache_stats(&entries, &hits, &misses, &negative_hits);
182 dns_metrics_emit_cache(mw, "brix_dns_reverse_cache", "reverse-DNS",
183 entries, hits, misses, negative_hits);
184}
