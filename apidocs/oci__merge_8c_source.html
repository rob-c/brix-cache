1/*
2 * oci_merge.c — the OCI location merge: refusals first, then the export build.
3 *
4 * WHAT: ngx_http_brix_oci_merge_loc_conf() and its four helpers — scalar
5 * inheritance, the §0.6.3 config-load refusal matrix, the upstream
6 * descriptor built from `brix_oci_mirror`, and the export/backend/tier
7 * build the enabled surfaces share.
8 * WHY: a registry mirror is a cache whose upstream is named by ONE directive,
9 * so almost everything this file does is turning that one URL into the
10 * shapes the existing machinery already understands: a storage backend
11 * string for sd_http, an export root for the confinement fd, a verify
12 * mode for the fill. The part that is genuinely new is the refusal
13 * matrix: every combination below is one an operator can write and would
14 * otherwise get a silently WRONG deployment from (a "mirror" that
15 * accepts pushes, a push registry open to the world, a credential file
16 * every local user can read). Those are nginx -t failures on purpose.
17 * HOW: the order is load-bearing and matches the cvmfs merge it is modelled
18 * on: adopt the unified directives → merge our scalars → pre-seed the
19 * verify default while the sentinel is still readable → shared merge →
20 * refuse → build. A location with neither surface enabled returns after
21 * the shared merge having allocated nothing (J.2).
22 */

49/* Inherit every OCI scalar and pre-seed the verify default.
50 *
51 * The verify pre-seed must land BEFORE the shared merge: the sentinel is what
52 * distinguishes "the operator said nothing" from "the operator said off", and
53 * the shared merge collapses the former to the latter. An OCI cache key names
54 * a sha256 digest for every immutable object, so the self-verifying mode is
55 * the only default that makes a mirror trustworthy by construction. */
56static char *
57oci_merge_scalars(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *prev,
58 ngx_http_brix_oci_loc_conf_t *conf)
59{
60 brix_http_common_adopt(cf, &conf->common);
61
62 ngx_conf_merge_value(conf->mirror, prev->mirror, 0);
63 ngx_conf_merge_str_value(conf->mirror_url, prev->mirror_url, "");
64 ngx_conf_merge_str_value(conf->mirror_user, prev->mirror_user, "");
65 ngx_conf_merge_str_value(conf->mirror_pwfile, prev->mirror_pwfile, "");
66 ngx_conf_merge_str_value(conf->upstream_ns, prev->upstream_ns, "");
67 ngx_conf_merge_str_value(conf->token_zone_name, prev->token_zone_name,
68 "oci_tokens");
69 ngx_conf_merge_value(conf->token_zone_set, prev->token_zone_set, 0);
70 ngx_conf_merge_sec_value(conf->manifest_ttl, prev->manifest_ttl, 60);
71 ngx_conf_merge_value(conf->insecure, prev->insecure, 0);
72
73 ngx_conf_merge_value(conf->delegate, prev->delegate, 0);
74 ngx_conf_merge_str_value(conf->delegate_realm, prev->delegate_realm,
75 "brix-oci");
76 ngx_conf_merge_sec_value(conf->deleg_proof_ttl, prev->deleg_proof_ttl,
77 BRIX_OCI_DELEG_TTL_DEFAULT);
78 ngx_conf_merge_value(conf->deleg_insecure, prev->deleg_insecure, 0);
79
80 ngx_conf_merge_value(conf->registry, prev->registry, 0);
81 ngx_conf_merge_value(conf->registry_anon, prev->registry_anon, 0);
82 ngx_conf_merge_str_value(conf->registry_root, prev->registry_root, "");
83 ngx_conf_merge_str_value(conf->token_issuers, prev->token_issuers, "");
84 ngx_conf_merge_size_value(conf->max_blob, prev->max_blob, 0);
85 ngx_conf_merge_sec_value(conf->upload_grace, prev->upload_grace,
86 24 * 60 * 60);
87 ngx_conf_merge_msec_value(conf->gc_interval, prev->gc_interval, 0);
88 ngx_conf_merge_sec_value(conf->gc_grace, prev->gc_grace,
89 BRIX_OCI_GC_GRACE_DEFAULT);
90 if (conf->issuers == NULL) {
91 conf->issuers = prev->issuers;
92 }
93 if (conf->auth_realms == NULL) {
94 conf->auth_realms = prev->auth_realms;
95 }
96 if (conf->up == NULL) {
97 conf->up = prev->up;
98 }
99
100 if ((conf->mirror || conf->registry)
101 && conf->common.cache_verify_mode == NGX_CONF_UNSET_UINT)
102 {
103 conf->common.cache_verify_mode = BRIX_CACHE_VERIFY_OCI_DIGEST;
104 }
105
106 return ngx_http_brix_shared_merge(cf, &prev->common, &conf->common, "");
107}

112/* The §0.6.3 refusal matrix for the mirror surface. */
113static char *
114oci_reject_mirror(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *conf,
115 const brix_oci_url_t *url)
116{
117 if (conf->common.allow_write == 1) {
118 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
119 "brix_allow_write on: an OCI pull-through mirror is read-only "
120 "by construction; use brix_oci_registry for a push surface");
121 return NGX_CONF_ERROR;
122 }
123
124 if (conf->common.stage_store.len > 0) {
125 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
126 "brix_stage_store: an OCI pull-through mirror never stages "
127 "writes; remove it from this block");
128 return NGX_CONF_ERROR;
129 }
130
131 if (!url->tls && !conf->insecure) {
132 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
133 "brix_oci_mirror \"%V\": a cleartext upstream would hand every "
134 "pulled token to the network; use https, or brix_oci_mirror_"
135 "insecure on for a test fixture", &conf->mirror_url);
136 return NGX_CONF_ERROR;
137 }
138
139 if ((conf->mirror_user.len == 0) != (conf->mirror_pwfile.len == 0)) {
140 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
141 "brix_oci_mirror_auth needs both a user and a password file");
142 return NGX_CONF_ERROR;
143 }
144
145 /* Delegate mode asks clients for THEIR registry password; a server that
146 * does not terminate TLS would carry that secret in cleartext, and by
147 * then it is burned. Refuse at load, where the operator is looking —
148 * the request path re-checks per connection for mixed listeners. */
149 if (conf->delegate && !conf->deleg_insecure && !oci_ssl_terminates(cf)) {
150 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
151 "brix_oci_mirror_delegate on a server with no TLS certificate: "
152 "a delegated credential on cleartext is already compromised; add "
153 "ssl, or state a test fixture with brix_oci_delegate_insecure on");
154 return NGX_CONF_ERROR;
155 }
156
157 return NGX_CONF_OK;
158}

160/* Does this server terminate TLS at all? `certificates` unset means every
161 * downstream byte — including a delegated Basic credential — rides cleartext.
162 * Same read-it-off-the-ssl-conf discipline as oci_ssl_verifies_client(). */
163static int
164oci_ssl_terminates(ngx_conf_t *cf)
165{
166#if (NGX_HTTP_SSL)
167 ngx_http_ssl_srv_conf_t *sslcf =
168 ngx_http_conf_get_module_srv_conf(cf, ngx_http_ssl_module);
169
170 return (sslcf != NULL
171 && sslcf->certificates != NGX_CONF_UNSET_PTR
172 && sslcf->certificates != NULL
173 && sslcf->certificates->nelts > 0);
174#else
175 (void) cf;
176 return 0;
177#endif
178}

201/* The §0.6.3 refusal matrix for the surfaces as a pair, plus the registry's
202 * one structural demand: a push endpoint must have said, in the config, who
203 * is allowed to push. */
204static char *
205oci_reject_surfaces(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *conf)
206{
207 if (conf->mirror && conf->registry) {
208 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
209 "brix_oci_mirror and brix_oci_registry are different locations: "
210 "one caches somebody else's registry, the other IS a registry");
211 return NGX_CONF_ERROR;
212 }
213
214 /* A mirror's objects are cache entries: the cache tier is what decides
215 * when they go, and a sweep that unlinked them behind its back would be
216 * deleting another subsystem's bookkeeping. Only the mirror is refused
217 * here and not every non-registry block: the directive inherits, so an
218 * operator who sets it once at server level and enables the registry in
219 * one location underneath is writing a correct config, and the outer
220 * block's copy is simply inert. */
221 if (conf->gc_interval > 0 && conf->mirror) {
222 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
223 "brix_oci_gc_interval: there is nothing for a registry sweep to "
224 "collect on a pull-through mirror; the cache tier owns eviction "
225 "there");
226 return NGX_CONF_ERROR;
227 }
228
229 if (!conf->registry) {
230 return NGX_CONF_OK;
231 }
232
233 /* A pass costs a full walk of the store. Below a second apart they
234 * overlap into a busy loop over the disk rather than maintenance, and the
235 * timer would spend the worker on nothing else. */
236 if (conf->gc_interval > 0 && conf->gc_interval < BRIX_OCI_GC_MIN_INTERVAL) {
237 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
238 "brix_oci_gc_interval: %M ms is a busy loop over the store, not "
239 "maintenance; use %M ms or more", conf->gc_interval,
240 (ngx_msec_t) BRIX_OCI_GC_MIN_INTERVAL);
241 return NGX_CONF_ERROR;
242 }
243
244 /* An unauthenticated push registry is a supply-chain hole, so it can only
245 * be reached through a directive that says so in as many words. TLS client
246 * verification counts as the authenticated context: the peer is already
247 * proven at the transport before a byte of the API is parsed. */
248 if (conf->token_issuers.len == 0 && !conf->registry_anon
249 && !oci_ssl_verifies_client(cf))
250 {
251 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
252 "brix_oci_registry on without an authenticated context: add "
253 "brix_oci_token_issuers <scitokens.cfg>, enable ssl_verify_client, "
254 "or state the intent with brix_oci_registry_allow_anonymous on");
255 return NGX_CONF_ERROR;
256 }
257
258 /* One issuer table per configured file, built at load so a malformed
259 * scitokens.cfg fails nginx -t instead of every push. */
260 if (conf->token_issuers.len > 0 && conf->issuers == NULL) {
261 if (brix_token_registry_build(cf,
262 (const char *) conf->token_issuers.data,
263 BRIX_AUTHZ_CAPABILITY, &conf->issuers) != NGX_OK)
264 {
265 return NGX_CONF_ERROR;
266 }
267 }
268
269 return NGX_CONF_OK;
270}

273/* Read the token-endpoint password and compose the "user:pass" pair the D1
274 * dance base64s. The file is read ONCE, here, at config load: a per-request
275 * read would put a secret on the hot path and a reload is the operator's
276 * signal that the credential changed. */
277static char *
278oci_read_pwfile(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *conf,
279 brix_oci_upstream_t *up)
280{
281 char line[OCI_PWFILE_LINE_MAX];
282 char *path = (char *) conf->mirror_pwfile.data;
283 struct stat st;
284 FILE *f;
285 size_t n;
286
287 f = fopen(path, "re");
288 if (f == NULL) {
289 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
290 "brix_oci_mirror_auth: cannot open password file \"%V\"",
291 &conf->mirror_pwfile);
292 return NGX_CONF_ERROR;
293 }
294
295 if (fstat(fileno(f), &st) != 0) {
296 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
297 "brix_oci_mirror_auth: cannot stat \"%V\"", &conf->mirror_pwfile);
298 (void) fclose(f);
299 return NGX_CONF_ERROR;
300 }
301
302 if (st.st_mode & OCI_PWFILE_MODE_MASK) {
303 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
304 "brix_oci_mirror_auth: password file \"%V\" is mode %04o - "
305 "group/world access must be removed (chmod 0600)",
306 &conf->mirror_pwfile, (unsigned) (st.st_mode & 07777));
307 (void) fclose(f);
308 return NGX_CONF_ERROR;
309 }
310
311 if (fgets(line, sizeof(line), f) == NULL) {
312 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
313 "brix_oci_mirror_auth: password file \"%V\" is empty",
314 &conf->mirror_pwfile);
315 (void) fclose(f);
316 return NGX_CONF_ERROR;
317 }
318 (void) fclose(f);
319
320 n = strcspn(line, "\r\n");
321 line[n] = '\0';
322 if (n == 0) {
323 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
324 "brix_oci_mirror_auth: password file \"%V\" first line is blank",
325 &conf->mirror_pwfile);
326 return NGX_CONF_ERROR;
327 }
328
329 if ((size_t) snprintf(up->basic, sizeof(up->basic), "%.*s:%s",
330 (int) conf->mirror_user.len, conf->mirror_user.data,
331 line)
332 >= sizeof(up->basic))
333 {
334 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
335 "brix_oci_mirror_auth: user and password exceed %uz bytes",
336 sizeof(up->basic));
337 return NGX_CONF_ERROR;
338 }
339
340 return NGX_CONF_OK;
341}

344/* Turn `brix_oci_mirror <base-url>` into the upstream descriptor the fill
345 * thread reads and into the storage-backend string sd_http is configured
346 * from. The two must agree byte-for-byte: the canonical cache key is an
347 * absolute "/v2/..." path, so the backend base carries the scheme, authority
348 * and any path prefix, and nothing else. */
349static char *
350oci_build_upstream(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *conf)
351{
352 brix_oci_upstream_t *up;
353 brix_oci_url_t url;
354 u_char *backend;
355 size_t len;
356
357 if (conf->mirror_url.len == 0) {
358 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
359 "brix_oci_mirror requires an upstream base URL");
360 return NGX_CONF_ERROR;
361 }
362
363 if (brix_oci_url_parse((const char *) conf->mirror_url.data,
364 conf->mirror_url.len, &url) != 0)
365 {
366 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
367 "brix_oci_mirror \"%V\" is not an absolute http(s) URL with a "
368 "plain host and path", &conf->mirror_url);
369 return NGX_CONF_ERROR;
370 }
371
372 if (oci_reject_mirror(cf, conf, &url) != NGX_CONF_OK) {
373 return NGX_CONF_ERROR;
374 }
375
376 BRIX_PCALLOC_OR_RETURN(up, cf->pool, sizeof(*up), NGX_CONF_ERROR);
377
378 (void) ngx_cpystrn((u_char *) up->host, (u_char *) url.host,
379 sizeof(up->host));
380 (void) ngx_cpystrn((u_char *) up->base_path, (u_char *) url.path,
381 sizeof(up->base_path));
382 up->port = url.port;
383 up->tls = url.tls;
384 up->insecure = conf->insecure;
385 if (conf->auth_realms != NULL) {
386 /* Copied by value: the fill thread reads the allowlist on every
387 * dance, and it reads it off the upstream it already holds. */
388 up->realms = *conf->auth_realms;
389 }
390
391 len = (size_t) snprintf(up->base_url, sizeof(up->base_url),
392 "%s://%s%s%s:%d%s",
393 url.tls ? "https" : "http",
394 strchr(url.host, ':') ? "[" : "",
395 url.host,
396 strchr(url.host, ':') ? "]" : "",
397 url.port, url.path);
398 if (len >= sizeof(up->base_url)) {
399 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
400 "brix_oci_mirror \"%V\" is too long", &conf->mirror_url);
401 return NGX_CONF_ERROR;
402 }
403
404 if (conf->mirror_pwfile.len > 0
405 && oci_read_pwfile(cf, conf, up) != NGX_CONF_OK)
406 {
407 return NGX_CONF_ERROR;
408 }
409
410 /* The composable backend is the same http(s) driver a CVMFS Stratum-1
411 * rides; the mirror's contribution is the base it points at. An explicit
412 * brix_storage_backend in the same block is a second answer to a question
413 * that already has one. */
414 if (conf->common.storage_backend.len > 0) {
415 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
416 "brix_storage_backend and brix_oci_mirror both name the upstream "
417 "for this location; brix_oci_mirror is the OCI spelling - "
418 "configure exactly one");
419 return NGX_CONF_ERROR;
420 }
421
422 backend = ngx_pnalloc(cf->pool, len + 1);
423 if (backend == NULL) {
424 return NGX_CONF_ERROR;
425 }
426 ngx_memcpy(backend, up->base_url, len + 1);
427 conf->common.storage_backend.data = backend;
428 conf->common.storage_backend.len = len;
429
430 conf->up = up;
431 return NGX_CONF_OK;
432}

435/* Anchor the namespace, open the confinement fd, register the backend and
436 * compose the cache tiers — the build order every brix HTTP protocol shares.
437 * The mirror is a pure cache node: no local export tree, so the namespace
438 * anchors at "/" and every object lives in the cache store. */
439static char *
440oci_merge_export(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *conf)
441{
442 if (conf->mirror) {
443 conf->common.allow_write = 0;
444 }
445
446 brix_storage_backend_posix_root(&conf->common);
447
448 if (conf->registry && conf->registry_root.len > 0) {
449 conf->common.root = conf->registry_root;
450 }
451 if (brix_http_merge_export_anchor(cf, &conf->common,
452 conf->mirror ? "brix_oci_mirror"
453 : "brix_oci_registry",
454 conf->registry ? 1 : 0) != NGX_CONF_OK)
455 {
456 return NGX_CONF_ERROR;
457 }
458
459 /* MANIFEST-class entries get the tag freshness window stamped; every
460 * digest-addressed object is immutable and ignores it (§D2.2). */
461 conf->common.cache_manifest_ttl = conf->manifest_ttl;
462
463 if (brix_tier_register_stores(cf, &conf->common) != NGX_OK) {
464 return NGX_CONF_ERROR;
465 }
466
467 return NGX_CONF_OK;
468}

471/* Resolve — or, for the documented `oci_tokens 1m` default, create — the SHM
472 * zone the D1 dance caches bearers in. A mirror is expected to work from the
473 * six directives §0.6.4 shows, so the absence of an explicit
474 * brix_oci_token_zone is a default rather than an error; an explicit NAME that
475 * was never declared still fails nginx -t, because that is a typo. */
476static char *
477oci_bind_token_zone(ngx_conf_t *cf, ngx_http_brix_oci_loc_conf_t *conf)
478{
479 brix_kv_t *kv = brix_kv_find(&conf->token_zone_name);
480
481 if (kv != NULL) {
482 conf->up->tokens = kv;
483 return NGX_CONF_OK;
484 }
485
486 if (conf->token_zone_set) {
487 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
488 "brix_oci_mirror: token zone \"%V\" is not declared - add "
489 "brix_oci_token_zone %V <size> to the http block",
490 &conf->token_zone_name, &conf->token_zone_name);
491 return NGX_CONF_ERROR;
492 }
493
494 BRIX_PCALLOC_OR_RETURN(kv, cf->pool, sizeof(*kv), NGX_CONF_ERROR);
495
496 if (brix_kv_configure(cf, kv, &conf->token_zone_name,
497 BRIX_OCI_TOKEN_ZONE_DEFAULT_SIZE,
498 BRIX_OCI_TOKEN_KEYLEN, BRIX_OCI_TOKEN_MAX,
499 &ngx_http_brix_oci_module) != NGX_OK)
500 {
501 return NGX_CONF_ERROR;
502 }
503
504 conf->up->tokens = kv;
505 return NGX_CONF_OK;
506}

509char *
510ngx_http_brix_oci_merge_loc_conf(ngx_conf_t *cf, void *parent, void *child)
511{
512 ngx_http_brix_oci_loc_conf_t *prev = parent;
513 ngx_http_brix_oci_loc_conf_t *conf = child;
514
515 if (oci_merge_scalars(cf, prev, conf) != NGX_CONF_OK) {
516 return NGX_CONF_ERROR;
517 }
518
519 if (!conf->mirror && !conf->registry) {
520 return NGX_CONF_OK; /* surface off: nothing is built */
521 }
522
523 if (oci_reject_surfaces(cf, conf) != NGX_CONF_OK) {
524 return NGX_CONF_ERROR;
525 }
526
527 if (conf->mirror && oci_build_upstream(cf, conf) != NGX_CONF_OK) {
528 return NGX_CONF_ERROR;
529 }
530
531 if (oci_merge_export(cf, conf) != NGX_CONF_OK) {
532 return NGX_CONF_ERROR;
533 }
534
535 /* Registered from the CANONICAL root, which is the only spelling the
536 * store itself is built under — the raw directive string would hand the
537 * sweep back the symlinks brix_prepare_export_root just removed. */
538 if (conf->registry) {
539 brix_oci_gc_register((const char *) conf->common.root_canon,
540 conf->gc_interval, conf->gc_grace);
541 /* C10: the service-publish verb stages every object through a
542 * "<final>.xrd-tmp.<pid>.<rand>" temp (brix_staged_open →
543 * brix_make_tmp_path). A crash between stage and rename orphans one
544 * under the store tree; register the canonical root so worker-0's
545 * brix_tmp_reap_all() nftw-walks it and removes dead-owner orphans.
546 * (The flat-.commit reaper brix_stage_dir_register sweeps does NOT
547 * see these scattered temps — this recursive one is the right sweep;
548 * the tag lister additionally skips them (oci_store.c tag_list) so an
549 * orphan never surfaces as a tag in the window before the next
550 * restart's reap; the referrers lister is already immune via its exact
551 * sha256-hexlen entry gate.) */
552 brix_tmp_reap_register((const char *) conf->common.root_canon);
553 }
554
555 if (conf->mirror && oci_bind_token_zone(cf, conf) != NGX_CONF_OK) {
556 return NGX_CONF_ERROR;
557 }
558
559 return NGX_CONF_OK;
560}
