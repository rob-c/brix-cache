18/* Per-user backend credential policy gates (vfs_cred.c).
19 *
20 * WHAT: Examine ctx->storage_cred_dir and the ctx identity, select a per-user
21 * x509 proxy via brix_sd_ucred_select, and report whether the op should
22 * proceed with a user credential (use_cred=1, *cred filled from *store),
23 * the service credential (use_cred=0), or be refused (NGX_ERROR, errno/
24 * *err_out = EACCES).
25 *
26 * WHY: brix_vfs_backend_cred gates data-plane opens (open/staged_open), keyed
27 * on driver->open_cred. brix_vfs_ns_cred gates namespace ops (stat/unlink/
28 * mkdir/rename/copy/setattr/xattr/opendir), keyed on driver->stat_cred.
29 * Both share the same select+deny+fallback decision body in vfs_cred.c.
30 *
31 * HOW: The gates are stateless — each probes the credential file at call time.
32 * *store and *cred are stack-allocated by the callers and live for the
33 * duration of the driver call that follows. */
34ngx_int_t brix_vfs_backend_cred(brix_vfs_ctx_t *ctx, brix_sd_ucred_t *store,
35 brix_sd_cred_t *cred, int *use_cred, int *err_out);

53/* ---- brix_vfs_cred_gate_active ---------------------------------------------
54 *
55 * WHAT: True when the per-user backend credential gate (brix_vfs_ns_cred /
56 * brix_vfs_backend_cred) must run for this ctx — i.e. a per-user
57 * credential SOURCE is bound: either the directory-based SELECT policy
58 * (storage_cred_dir) OR a live delegation bag (PASSTHROUGH/EXCHANGE).
59 *
60 * WHY: The namespace dispatch sites (vfs_xattr/stat/unlink/mkdir/rename/dir/
61 * copy) originally guarded the gate on `storage_cred_dir != NULL` alone.
62 * That drops the credential in pure PASSTHROUGH mode (a deleg bag bound
63 * with NO storage_credential_dir): the ns op then runs on the static
64 * service credential, which a per-user (e.g. token-only) backend rejects —
65 * asymmetric with the data-plane open, whose gate (vfs_backend_cred_decide)
66 * already consults the deleg bag before the dir. This predicate makes the
67 * ns guard consider BOTH sources so a passthrough bearer/proxy reaches the
68 * backend on namespace ops (e.g. the WebDAV lock-state getxattr) exactly
69 * as it does on data-plane opens.
70 *
71 * HOW: storage_cred_dir set, OR brix_vfs_backend_mode(ctx) != BRIX_CRED_SELECT
72 * (a bag is bound). A no-op change for the dir-only and no-cred configs. */
73int brix_vfs_cred_gate_active(brix_vfs_ctx_t *ctx);

75/* Delegation live-cred materialiser (phase-70 §5.1/§5.4, vfs_deleg.c).
76 *
77 * WHAT: For a ctx carrying a bound live bag in PASSTHROUGH mode, validate the
78 * captured bytes and materialise them into *cred: a bearer token is copied
79 * straight through (cred->bearer); a full x509 proxy PEM is written to a
80 * 0600 temp path (cred->x509_proxy) with a pool cleanup that unlink()s +
81 * zeroes the path. Sets cred->mode. On success *use_cred=1, NGX_OK. On a
82 * missing/invalid live cred: *err_out=EACCES and NGX_ERROR when the ctx is
83 * in fallback-deny, else *use_cred=0 + NGX_OK (fall to service cred).
84 *
85 * WHY: The one place the front door's raw forwardable credential becomes the
86 * exact cred form the backend GSI/ZTN presenter already consumes, so no
87 * new origin-leg code is needed.
88 *
89 * HOW: Reuses brix_proxy_gsi_write_pem_temp() (net/proxy) for the bytes→path
90 * adaptor and PEM_read_bio_X509 to reject non-PEM. When the capture site
91 * stamped a CA store via brix_vfs_deleg_set_ca_store(), the full RFC-3820
92 * chain-trust verify re-runs here at the gate (P90-70.4); DN-match
93 * happens at capture. Unit: tests/c/deleg_gate_test.c. */
94ngx_int_t brix_vfs_deleg_live_cred(brix_vfs_ctx_t *ctx, brix_sd_cred_t *cred,
95 int *use_cred, int *err_out);
