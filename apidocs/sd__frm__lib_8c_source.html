1/*
2 * sd_frm_lib.c — the library-native MSS adapter: drives a real HSM through an
3 * operator-supplied shared object (dlopen) instead of forking a stage command.
4 *
5 * Same residency/recall/migrate/purge model as the exec adapter (sd_frm_exec.c),
6 * and it reuses the same online-buffer filesystem helpers (frm_mkparents /
7 * open / pread live here or in sd_frm_stub.c); the ONLY difference is that each
8 * verb is an in-process dlsym'd call rather than a posix_spawn + waitpid. On a
9 * busy silo that removes the per-verb fork+exec that dominates small-object
10 * staging latency — the phase-64 "library-native adapter" residual.
11 *
12 * The vendor .so is a runtime plug-in (exactly like a driver): if it is absent or
13 * missing a required symbol, brix_mss_lib_create returns NULL and sd_frm.c falls
14 * back to the exec/stub transport. The symbol ABI is sd_frm_lib_abi.h.
15 */

40/* The lib adapter's frm_mss_invoke_fn: each MSS verb is a direct dlsym'd call.
41 * `purge` is optional in the ABI — absent means the MSS-side drop is a no-op
42 * (the shared op already unlinked the online buffer). */
43static int
44lib_invoke(void *mss, const char *verb, const char *key, const char *online)
45{
46 lib_ctx_t *c = mss;
47
48 if (strcmp(verb, "exists") == 0) {
49 return c->fn_exists(key);
50 }
51 if (strcmp(verb, "recall") == 0) {
52 return c->fn_recall(key, online);
53 }
54 if (strcmp(verb, "migrate") == 0) {
55 return c->fn_migrate(key, online);
56 }
57 if (strcmp(verb, "purge") == 0) {
58 return (c->fn_purge != NULL) ? c->fn_purge(key) : 0;
59 }
60 return -1;
61}

63static void
64lib_destroy(void *mss)
65{
66 lib_ctx_t *c = mss;
67
68 if (c == NULL) {
69 return;
70 }
71 if (c->dl != NULL) {

91/* brix_mss_lib_create — dlopen `libpath` and bind the HSM ABI.
92 *
93 * Returns the adapter context on success, or NULL when the library cannot be
94 * opened or a REQUIRED symbol (exists/recall/migrate) is missing — the caller
95 * (sd_frm.c) then WARNs and falls back to the exec/stub transport. `purge` is
96 * optional. RTLD_LOCAL keeps the HSM library's symbols out of the global scope
97 * so it cannot collide with nginx or another dlopen'd module. */
98void *
99brix_mss_lib_create(const char *location, const char *libpath, ngx_log_t *log)
100{
101 lib_ctx_t *c;
102
103 if (libpath == NULL || libpath[0] == '\0') {
104 return NULL;
105 }
106 c = calloc(1, sizeof(*c));
107 if (c == NULL) {
108 return NULL;
109 }
110 ngx_cpystrn((u_char *) c->head.base, (u_char *) location,
111 sizeof(c->head.base));
112 c->head.invoke = lib_invoke;
113 c->log = log;
114
115 c->dl = dlopen(libpath, RTLD_NOW | RTLD_LOCAL);
116 if (c->dl == NULL) {
117 ngx_log_error(NGX_LOG_WARN, log, 0,
118 "xrootd frm: cannot dlopen HSM library \"%s\": %s",
119 libpath, dlerror());
120 free(c);
121 return NULL;
122 }
123 (void) dlerror();
124 c->fn_exists = (brix_frm_hsm_exists_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_EXISTS);
125 c->fn_recall = (brix_frm_hsm_recall_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_RECALL);
126 c->fn_migrate = (brix_frm_hsm_migrate_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_MIGRATE);
127 c->fn_purge = (brix_frm_hsm_purge_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_PURGE);
128
129 if (c->fn_exists == NULL || c->fn_recall == NULL || c->fn_migrate == NULL) {
130 ngx_log_error(NGX_LOG_WARN, log, 0,
131 "xrootd frm: HSM library \"%s\" is missing a required symbol "
132 "(need %s, %s, %s)", libpath, BRIX_FRM_HSM_SYM_EXISTS,
133 BRIX_FRM_HSM_SYM_RECALL, BRIX_FRM_HSM_SYM_MIGRATE);
134 (void) dlclose(c->dl);
135 free(c);
136 return NULL;
137 }
138 return c;
139}
