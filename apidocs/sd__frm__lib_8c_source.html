1/*
2 * sd_frm_lib.c — the library-native MSS adapter: drives a real HSM through an
3 * operator-supplied shared object (dlopen) instead of forking a stage command.
4 *
5 * Same residency/recall/migrate/purge model as the exec adapter (sd_frm_exec.c),
6 * and it reuses the same online-buffer filesystem helpers (frm_mkparents /
7 * open / pread live here or in sd_frm_stub.c); the ONLY difference is that each
8 * verb is an in-process dlsym'd call rather than a posix_spawn + waitpid. On a
9 * busy silo that removes the per-verb fork+exec that dominates small-object
10 * staging latency — the phase-64 "library-native adapter" residual.
11 *
12 * The vendor .so is a runtime plug-in (exactly like a driver): if it is absent or
13 * missing a required symbol, brix_mss_lib_create returns NULL and sd_frm.c falls
14 * back to the exec/stub transport. The symbol ABI is sd_frm_lib_abi.h.
15 */

49static int
50lib_residency(void *mss, const char *key, off_t *size_out, time_t *mtime_out)
51{
52 lib_ctx_t *c = mss;
53 char online[PATH_MAX];
54 struct stat sb;
55
56 if (lib_online_path(c, key, online, sizeof(online)) == 0
57 && stat(online, &sb) == 0)
58 {
59 if (size_out) { *size_out = sb.st_size; }
60 if (mtime_out) { *mtime_out = sb.st_mtime; }
61 return BRIX_RESIDENCY_ONLINE;
62 }
63 /* Ask the library: 0 = on tape (offline), non-zero = absent. Size is unknown
64 * until recalled; the cache fill restats the online buffer. */
65 if (c->fn_exists(key) == 0) {
66 if (size_out) { *size_out = 0; }
67 if (mtime_out) { *mtime_out = time(NULL); }
68 return BRIX_RESIDENCY_OFFLINE;
69 }
70 return BRIX_RESIDENCY_ABSENT;
71}

73static int
74lib_recall_begin(void *mss, const char *key)
75{
76 lib_ctx_t *c = mss;
77 char online[PATH_MAX];
78
79 if (lib_online_path(c, key, online, sizeof(online)) != 0) {
80 return -1;
81 }
82 if (access(online, F_OK) == 0) {
83 return 0; /* already online */
84 }
85 frm_mkparents(online); /* the library writes the online buffer */
86 return (c->fn_recall(key, online) == 0) ? 0 : -1;
87}

115{
116 lib_ctx_t *c = mss;
117 char online[PATH_MAX];
118

182/* brix_mss_lib_create — dlopen `libpath` and bind the HSM ABI.
183 *
184 * Returns the adapter context on success, or NULL when the library cannot be
185 * opened or a REQUIRED symbol (exists/recall/migrate) is missing — the caller
186 * (sd_frm.c) then WARNs and falls back to the exec/stub transport. `purge` is
187 * optional. RTLD_LOCAL keeps the HSM library's symbols out of the global scope
188 * so it cannot collide with nginx or another dlopen'd module. */
189void *
190brix_mss_lib_create(const char *location, const char *libpath, ngx_log_t *log)
191{
192 lib_ctx_t *c;
193
194 if (libpath == NULL || libpath[0] == '\0') {
195 return NULL;
196 }
197 c = calloc(1, sizeof(*c));
198 if (c == NULL) {
199 return NULL;
200 }
201 ngx_cpystrn((u_char *) c->base, (u_char *) location, sizeof(c->base));
202 c->log = log;
203
204 c->dl = dlopen(libpath, RTLD_NOW | RTLD_LOCAL);
205 if (c->dl == NULL) {
206 ngx_log_error(NGX_LOG_WARN, log, 0,
207 "xrootd frm: cannot dlopen HSM library \"%s\": %s",
208 libpath, dlerror());
209 free(c);
210 return NULL;
211 }
212 (void) dlerror();
213 c->fn_exists = (brix_frm_hsm_exists_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_EXISTS);
214 c->fn_recall = (brix_frm_hsm_recall_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_RECALL);
215 c->fn_migrate = (brix_frm_hsm_migrate_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_MIGRATE);
216 c->fn_purge = (brix_frm_hsm_purge_fn) dlsym(c->dl, BRIX_FRM_HSM_SYM_PURGE);
217
218 if (c->fn_exists == NULL || c->fn_recall == NULL || c->fn_migrate == NULL) {
219 ngx_log_error(NGX_LOG_WARN, log, 0,
220 "xrootd frm: HSM library \"%s\" is missing a required symbol "
221 "(need %s, %s, %s)", libpath, BRIX_FRM_HSM_SYM_EXISTS,
222 BRIX_FRM_HSM_SYM_RECALL, BRIX_FRM_HSM_SYM_MIGRATE);
223 (void) dlclose(c->dl);
224 free(c);
225 return NULL;
226 }
227 return c;
228}
