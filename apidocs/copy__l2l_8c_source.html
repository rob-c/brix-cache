1/*
2 * copy_l2l.c — local→local copy (§7.17: file:// / bare-path both sides).
3 *
4 * WHAT: brix_copy_local_to_local() moves bytes between two LOCAL endpoints —
5 * a file (or file:// URL) or the '-' stdio stream on either side.
6 * WHY: Stock xrdcp copies local→local; BriX rejected the direction outright
7 * ("local→local not supported"), so `xrdcp file:///a file:///b`, a
8 * plain `xrdcp /a /b`, and the `cat | xrdcp - out` / `xrdcp in -`
9 * stdio idioms all failed. file:// already parses to the local scheme
10 * (net/url.c); the only missing piece was the copy direction itself.
11 * HOW: Reuse the transfer pump. A file source opens through the VFS
12 * (size known → bounded pump); stdin is an EOF-driven fd read. A file
13 * destination opens through the VFS (atomic temp+rename on commit,
14 * --force honored); stdout is a raw fd write with no commit. --xrate
15 * pacing rides the shared pump; --cksum runs post-commit against the
16 * committed file (literal/print modes — :source is meaningless with no
17 * server and is reported skipped). A mismatch drops the destination,
18 * matching every other download path.
19 */
20#include "copy_internal.h"
21
22#include <fcntl.h>
23#include <unistd.h>

25/* ---- Sequential fd source/sink for the stdio ('-') endpoints ---- */
26typedef struct { int fd; } l2l_fd_io;
27
28static ssize_t
29l2l_src_fd(void *ctx, uint8_t *buf, int64_t off, size_t cap, brix_status *st)
30{
31 l2l_fd_io *s = ctx;
32 ssize_t n;
33
34 (void) off; /* stdin is a stream: sequential, offset ignored */
35 do {
36 n = read(s->fd, buf, cap);
37 } while (n < 0 && errno == EINTR);
38 if (n < 0) {
39 brix_status_set(st, XRDC_ESOCK, errno, "read stdin: %s",
40 strerror(errno));
41 }
42 return n;
43}
44
45static int
46l2l_sink_fd(void *ctx, const uint8_t *buf, int64_t off, size_t n,
47 brix_status *st)
48{
49 l2l_fd_io *s = ctx;
50 size_t done = 0;
51
52 (void) off; /* stdout is a stream: sequential, offset ignored */
53 while (done < n) {
54 ssize_t w = write(s->fd, buf + done, n - done);
55
56 if (w < 0) {
57 if (errno == EINTR) {
58 continue;
59 }
60 brix_status_set(st, XRDC_ESOCK, errno, "write stdout: %s",
61 strerror(errno));
62 return -1;
63 }
64 done += (size_t) w;
65 }
66 return 0;
67}

69/* ---- Open the local source: stdin fd or a VFS file handle ----
70 *
71 * WHAT: Populates the pump src function + ctx and *expected (byte count, or
72 * -1 for the EOF-driven stdin stream). Returns 0 / -1 (st set).
73 *
74 * WHY: The two source shapes differ only in "how many bytes and by what
75 * reader" — isolating the open keeps the orchestrator flat.
76 *
77 * HOW: STDIO → fd 0, expected -1. A file → brix_vfs_open READ + fstat for the
78 * size so the pump terminates on the byte count (not an EOF probe).
79 */
80static int
81l2l_open_src(const brix_url *su, brix_vfs_file **vf_out, l2l_fd_io *fdio,
82 pump_src_fn *src, void **sctx, int64_t *expected, brix_status *st)
83{
84 if (su->scheme == XRDC_SCHEME_STDIO) {
85 fdio->fd = STDIN_FILENO;
86 *src = l2l_src_fd;
87 *sctx = fdio;
88 *expected = -1;
89 return 0;
90 }
91 {
92 brix_vfs_open_opts opts = {0};
93 brix_vfs_stat vst;
94
95 if (brix_vfs_open(su->path, XRDC_VFS_READ, &opts, vf_out, st) != 0) {
96 return -1;
97 }
98 if (brix_vfs_fstat(*vf_out, &vst, st) != 0) {
99 brix_vfs_close(*vf_out);
100 *vf_out = NULL;
101 return -1;
102 }
103 *expected = vst.size;
104 return 0;
105 }
106}

108/* ---- Run one local→local transfer end to end ----
109 *
110 * WHAT: Returns 0 on a byte-complete copy (committed, and --cksum-verified
111 * when asked), -1 (st set) otherwise. Handles every local/stdio pair.
112 *
113 * WHY: §7.17 — the missing copy direction. Kept a peer of the download/upload
114 * routers rather than folded into brix_copy_route so each direction is
115 * one function.
116 *
117 * HOW: 1. Open src (stdin | VFS). 2. Open dst (stdout | VFS temp, --force).
118 * 3. transfer_pump with the chosen src/sink (pacing + cancel inside).
119 * 4. File dst: commit on success, then --cksum against the committed
120 * file (drop on a genuine mismatch); abort the temp on any failure.
121 */
122/* Open the destination side: stdout, or a VFS temp (--force honoured).
123 * On success the sink fn/ctx are armed; -1 with st set on open failure. */
124static int
125l2l_open_dst(const brix_url *du, const brix_copy_opts *o, int64_t expected,
126 brix_vfs_file **dst_vf, l2l_fd_io *dst_fd, pump_sink_fn *sink_fn,
127 void **kctx, pump_local_t *dst_lc, brix_status *st)
128{
129 brix_vfs_open_opts vopts = {0};
130
131 if (du->scheme == XRDC_SCHEME_STDIO) {
132 dst_fd->fd = STDOUT_FILENO;
133 *sink_fn = l2l_sink_fd;
134 *kctx = dst_fd;
135 return 0;
136 }
137 vopts.expected_size = expected;
138 if (brix_vfs_open(du->path,
139 XRDC_VFS_WRITE | (o->force ? XRDC_VFS_FORCE : 0),
140 &vopts, dst_vf, st) != 0) {
141 return -1;
142 }
143 dst_lc->vf = *dst_vf;
144 *sink_fn = pump_sink_local_vfs;
145 *kctx = dst_lc;
146 return 0;
147}

149/* File-destination tail: commit on success, then --cksum against the
150 * committed file (drop on a genuine mismatch); abort the temp on any failure.
151 * Owns closing dst_vf; a stdout destination (dst_vf == NULL) passes through. */
152static int
153l2l_finish_dst(int rc, brix_vfs_file *dst_vf, const brix_url *du,
154 const brix_copy_opts *o, brix_status *st)
155{
156 int committed = 0;
157
158 if (dst_vf == NULL) {
159 return rc;
160 }
161 if (rc == 0) {
162 rc = brix_vfs_commit(dst_vf, st);
163 if (rc == 0) {
164 committed = 1;
165 if (o->cksum != NULL
166 && cksum_verify(NULL, NULL, du->path, o->cksum, o->silent, st)
167 == XRDC_CK_MISMATCH) {
168 unlink(du->path); /* committed-but-bad: drop it */
169 rc = -1;
170 }
171 }
172 }
173 if (rc != 0 && !committed) {
174 brix_vfs_abort(dst_vf);
175 }
176 brix_vfs_close(dst_vf);
177 return rc;
178}

180int
181brix_copy_local_to_local(const brix_url *su, const brix_url *du,
182 const brix_copy_opts *o, brix_status *st)
183{
184 brix_vfs_file *src_vf = NULL, *dst_vf = NULL;
185 l2l_fd_io src_fd = {0}, dst_fd = {0};
186 pump_local_t src_lc, dst_lc;
187 pump_src_fn src_fn = NULL;
188 pump_sink_fn sink_fn = NULL;
189 void *sctx = NULL, *kctx = NULL;
190 int64_t expected = -1;
191 int dst_stdout = (du->scheme == XRDC_SCHEME_STDIO);
192 int rc;
193
194 /* Friendly destination-exists refusal (matches the download path's
195 * message) BEFORE opening the source, so the user sees "-f", not the
196 * VFS layer's internal flag name. stdout and --force skip it. */
197 if (!dst_stdout && !o->force && access(du->path, F_OK) == 0) {
198 brix_status_set(st, XRDC_EUSAGE, 0,
199 "destination exists (use -f to overwrite): %s",
200 du->path);
201 return -1;
202 }
203
204 if (l2l_open_src(su, &src_vf, &src_fd, &src_fn, &sctx, &expected,
205 st) != 0) {
206 return -1;
207 }
208 if (src_vf != NULL) {
209 src_lc.vf = src_vf;
210 src_fn = pump_src_local_vfs;
211 sctx = &src_lc;
212 }
213
214 if (l2l_open_dst(du, o, expected, &dst_vf, &dst_fd, &sink_fn, &kctx,
215 &dst_lc, st) != 0) {
216 if (src_vf != NULL) {
217 brix_vfs_close(src_vf);
218 }
219 return -1;
220 }
221
222 rc = transfer_pump(src_fn, sctx, sink_fn, kctx, expected, o,
223 expected, st);
224
225 rc = l2l_finish_dst(rc, dst_vf, du, o, st);
226 if (src_vf != NULL) {
227 brix_vfs_close(src_vf);
228 }
229 return rc;
230}
