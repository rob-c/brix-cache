9int
10brix_srv_hc_claim(char *host_out, size_t host_size, uint16_t *port_out,
11 ngx_msec_t interval_ms, ngx_msec_t *next_due_ms)
12{
13 brix_srv_table_t *tbl;
14 brix_srv_entry_t *e;
15 ngx_uint_t i;
16 ngx_msec_t now;
17 ngx_msec_t soonest = interval_ms; /* idle cap when none are due */
18
19 if (next_due_ms != NULL) {
20 *next_due_ms = interval_ms;
21 }
22
23 tbl = srv_table();
24 if (tbl == NULL) {
25 return 0;
26 }
27 now = ngx_current_msec;
28
29 ngx_shmtx_lock(&brix_srv_mutex);
30
31 for (i = 0; i < tbl->capacity; i++) {
32 e = &tbl->slots[i];
33
34 if (!e->in_use || e->hc_in_progress) {
35 continue;
36 }
37 if (e->hc_next_check > now) {
38 /* Not due yet — remember how soon it becomes due so the caller can
39 * sleep to that deadline instead of polling at a fixed floor. */
40 ngx_msec_int_t d = (ngx_msec_int_t) (e->hc_next_check - now);
41 if (d > 0 && (ngx_msec_t) d < soonest) {
42 soonest = (ngx_msec_t) d;
43 }
44 continue;
45 }
46
47 e->hc_in_progress = 1;
48 e->hc_next_check = now + interval_ms;
49 ngx_cpystrn((u_char *) host_out, (u_char *) e->host, host_size);
50 *port_out = e->port;
51
52 ngx_shmtx_unlock(&brix_srv_mutex);
53 return 1; /* claimed; caller spreads the rest */
54 }
55
56 ngx_shmtx_unlock(&brix_srv_mutex);
57
58 if (next_due_ms != NULL) {
59 *next_due_ms = soonest; /* nothing due: sleep until soonest */
60 }
61 return 0;
62}

120/* WHAT
121 * Builds a kXR_locate response body listing all non-blacklisted servers whose
122 * exported path set covers the requested path. Entries are space-separated
123 * "S<r|w>host:port" strings, NUL-terminated.
124 *
125 * WHY
126 * Returning the full set of matching servers lets the client pick based on
127 * network locality, eliminating the need for chained redirects through the
128 * hierarchy ("lateral redirect"). One kXR_ok response replaces what would
129 * otherwise be a kXR_redirect chain through multiple manager tiers.
130 *
131 * HOW
132 * Locks mutex → scans all in_use, non-blacklisted, path-matching slots →
133 * appends "Sr<host>:<port>" (or "Sw" for writes) to buf with space separator.
134 * Stops early if the next entry would overflow bufsz. Returns bytes written
135 * (not counting NUL); 0 if no servers match.
136 */
137int
138brix_srv_locate_all(const char *path, int for_write,
139 char *buf, size_t bufsz)
140{
141 brix_srv_table_t *tbl;
142 brix_srv_entry_t *e;
143 ngx_uint_t i;
144 int written, entry_len, first;
145 ngx_msec_t now;
146 char entry[300];
147 char hostport[288]; /* host[256] + "[]" + ":65535" + NUL */
148
149 tbl = srv_table();
150 if (tbl == NULL || bufsz < 2) {
151 return 0;
152 }
153
154 now = ngx_current_msec;
155 written = 0;
156 first = 1;
157
158 ngx_shmtx_lock(&brix_srv_mutex);
159
160 for (i = 0; i < tbl->capacity; i++) {
161 e = &tbl->slots[i];
162
163 if (!e->in_use) {
164 continue;
165 }
166 if (e->blacklisted_until != 0 && e->blacklisted_until > now) {
167 continue;
168 }
169 if (!srv_path_matches(e->paths, path)) {
170 continue;
171 }
172
173 /* Bracket IPv6 literals so "Sr[::1]:1094" (not the unparseable
174 * "Sr::1:1094") — the host is stored canonically bare; bracket on emit. */
175 brix_format_host_port(e->host, e->port, hostport, sizeof(hostport));
176 entry_len = snprintf(entry, sizeof(entry), "%sS%c%s",
177 first ? "" : " ",
178 for_write ? 'w' : 'r',
179 hostport);
180 if (entry_len <= 0 || written + entry_len + 1 >= (int) bufsz) {
181 break;
182 }
183
184 ngx_memcpy(buf + written, entry, (size_t) entry_len);
185 written += entry_len;
186 first = 0;
187 }
188
189 ngx_shmtx_unlock(&brix_srv_mutex);
190
191 if (written > 0) {
192 buf[written] = '\0';
193 }
194
195 return written;
196}

200/* WHAT
201 * Removes a single path token from an existing server entry's paths field.
202 * Used when a data server revokes access to a specific directory.
203
204 * WHY
205 * A server may serve multiple colon-delimited path tokens. Removing one token
206 * keeps the entry alive for other paths without full unregister.
207
208 * HOW
209 * Locks mutex → scans slots for host+port match → in-place token walk: copies
210 * non-matching tokens to dst buffer, drops matching ones. Safe overlap because
211 * dst <= p always (copy forward direction). Null-terminates result.
212 */
213void
214brix_srv_unregister_path(const char *host, uint16_t port, const char *path)
215{
216 brix_srv_table_t *tbl;
217 brix_srv_entry_t *e;
218 ngx_uint_t i;
219 const char *p, *end;
220 char *dst;
221 size_t tok_len, path_len;
222 int first;
223
224 tbl = srv_table();
225 if (tbl == NULL) {
226 return;
227 }
228
229 path_len = strlen(path);
230
231 ngx_shmtx_lock(&brix_srv_mutex);
232
233 for (i = 0; i < tbl->capacity; i++) {
234 e = &tbl->slots[i];
235 if (!e->in_use || e->port != port
236 || ngx_strcmp(e->host, host) != 0)
237 {
238 continue;
239 }
240
241 /* In-place removal: walk tokens, copying those that don't match. */
242 p = e->paths;
243 dst = e->paths;
244 first = 1;
245
246 while (*p) {
247 end = strchr(p, ':');
248 if (end == NULL) {
249 end = p + strlen(p);
250 }
251 tok_len = (size_t)(end - p);
252
253 if (tok_len == path_len && ngx_strncmp(p, path, tok_len) == 0) {
254 /* Drop this token. */
255 } else {
256 if (!first) {
257 *dst++ = ':';
258 }
259 /* dst <= p always — safe overlap direction for memcpy. */
260 ngx_memcpy(dst, p, tok_len);
261 dst += tok_len;
262 first = 0;
263 }
264
265 p = *end ? end + 1 : end;
266 }
267 *dst = '\0';
268 break;
269 }
270
271 ngx_shmtx_unlock(&brix_srv_mutex);
272}

276/* WHAT
277 * Aggregates free-space and utilisation metrics across all registered servers.
278 * Returns total free MB and average utilisation percentage via output pointers.
279
280 * WHY
281 * Used by the redirector to report cluster-wide capacity to CMS management or
282 * for S3 gateway decisions about which region has sufficient space.
283
284 * HOW
285 * Locks mutex → sums free_mb and util_pct across all occupied slots, counts
286 * entries. Unlocks → computes average = sum_util / count (0 if no servers).
287 */
288void
289brix_srv_aggregate_space(uint32_t *total_free_mb, uint32_t *avg_util_pct)
290{
291 brix_srv_table_t *tbl;
292 brix_srv_entry_t *e;
293 ngx_uint_t i;
294 uint32_t sum_free;
295 uint64_t sum_util;
296 ngx_uint_t count;
297
298 *total_free_mb = 0;
299 *avg_util_pct = 0;
300
301 tbl = srv_table();
302 if (tbl == NULL) {
303 return;
304 }
305
306 sum_free = 0;
307 sum_util = 0;
308 count = 0;
309
310 ngx_shmtx_lock(&brix_srv_mutex);
311
312 for (i = 0; i < tbl->capacity; i++) {
313 e = &tbl->slots[i];
314 if (!e->in_use) {
315 continue;
316 }
317 sum_free += e->free_mb;
318 sum_util += e->util_pct;
319 count++;
320 }
321
322 ngx_shmtx_unlock(&brix_srv_mutex);
323
324 *total_free_mb = sum_free;
325 *avg_util_pct = count > 0 ? (uint32_t) (sum_util / count) : 0;
326}
