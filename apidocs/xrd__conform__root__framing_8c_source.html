63static int
64run_read_probes(xrd_conform_execution *execution, brix_conn *connection,
65 brix_file *file, const char *path, const uint8_t *baseline, uint64_t size)
66{
67 const brix_conform_result *result;
68
69 if (xrd_conform_family_active(execution, "RD"))
70 xrd_conform_run_rd(execution, connection, file, baseline, size);
71 if (xrd_conform_family_active(execution, "PGR"))
72 xrd_conform_run_pgr(execution, connection, file, baseline, size);
73 result = xrd_conform_result_find(execution, "PGR-020");
74 if ((result != NULL
75 && result->verdict == BRIX_CONFORM_VERDICT_DIVERGENT)
76 || pgread_structure_poisoned(execution)) return 1;
77 if (xrd_conform_family_active(execution, "STA"))
78 xrd_conform_probe_stat(execution, connection, file, path, size);
79 if (xrd_conform_family_active(execution, "RDV"))
80 xrd_conform_probe_readv(execution, connection, file, path, baseline,
81 size);
82 if (xrd_conform_family_active(execution, "RDV")
83 && size == XRDC_CONFORM_CHUNKED_SIZE)
84 xrd_conform_probe_readv_chunked(execution, connection, file, baseline,
85 (size_t) size);
86 result = xrd_conform_result_find(execution, "RDV-018");

88 && result->verdict == BRIX_CONFORM_VERDICT_DIVERGENT;

178static int
179send_protocol(xrd_conform_execution *execution, brix_conn *connection,
180 uint16_t sid, int padded, framing_reply *reply, brix_status *status)
181{
182 ClientProtocolRequest request;
183 char reason[64];
184
185 if (!xrd_conform_reserve(execution, sizeof(request), reason,
186 sizeof(reason))) return -1;
187 make_protocol(&request, sid, padded);
188 return brix_write_full(&connection->io, &request, sizeof(request), status)
189 == 0
190 && read_reply(connection, reply, status) == 0 ? 0 : -1;
191}

193static void
194probe_sequential(xrd_conform_execution *execution, brix_conn *connection,
195 brix_status *status)
196{
197 framing_reply first, second;
198 int first_ok = send_protocol(execution, connection, 0, 0, &first,
199 status) == 0;
200 int second_ok = first_ok
201 && send_protocol(execution, connection, 0, 0, &second, status) == 0;
202 int body_ok = first_ok && first.status == kXR_ok
203 && first.length >= sizeof(ServerProtocolBody);
204
205 xrd_conform_grade(execution, "FRM-001", body_ok, "zero-payload-request-accepted",
206 "zero-payload-request-refused");
207 xrd_conform_grade(execution, "FRM-002", first_ok, "eight-byte-response-header-read",
208 "response-header-not-eight-bytes");
209 xrd_conform_grade(execution, "FRM-003", first_ok && first.sid == 0,
210 "streamid-echoed", "streamid-not-echoed");
211 xrd_conform_grade(execution, "FRM-005", first_ok && first.sid == 0,
212 "zero-streamid-echoed", "zero-streamid-refused");
213 xrd_conform_grade(execution, "FRM-006", second_ok && second.sid == 0
214 && second.status == kXR_ok, "completed-streamid-reused",

218static void
219probe_pipeline(xrd_conform_execution *execution, brix_conn *connection,
220 brix_status *status)
221{
222 ClientProtocolRequest requests[2];
223 framing_reply first, second;
224 int wrote;
225 int read_ok;
226 int identities;
227 char reason[64];
228
229 make_protocol(&requests[0], 0x1234, 0);
230 make_protocol(&requests[1], 0x5678, 0);
231 wrote = xrd_conform_reserve(execution, sizeof(requests[0]), reason,
232 sizeof(reason))
233 && xrd_conform_reserve(execution, sizeof(requests[1]), reason,
234 sizeof(reason))
235 && brix_write_full(&connection->io, requests, sizeof(requests), status)
236 == 0;
237 read_ok = wrote && read_reply(connection, &first, status) == 0
238 && read_reply(connection, &second, status) == 0;
239 identities = read_ok
240 && ((first.sid == 0x1234 && second.sid == 0x5678)
241 || (first.sid == 0x5678 && second.sid == 0x1234));
242 xrd_conform_grade(execution, "FRM-004", identities,
243 "inflight-streamids-preserved", "inflight-streamids-confused");
244 xrd_conform_grade(execution, "FRM-022", read_ok,
245 "pipelined-requests-served", "pipelined-request-lost");
246 xrd_conform_grade(execution, "FRM-023", read_ok,
247 "same-connection-pipeline-served", "response-wait-required");
248}

250static void
251probe_padding(xrd_conform_execution *execution, brix_conn *connection,
252 brix_status *status)
253{
254 framing_reply plain, padded;
255 int ok = send_protocol(execution, connection, 0x2200, 0, &plain,
256 status) == 0
257 && send_protocol(execution, connection, 0x2201, 1, &padded,
258 status) == 0;
259
260 ok = ok && plain.status == padded.status && plain.length == padded.length
261 && memcmp(plain.body, padded.body, plain.length) == 0;

267probe_invalid_opcode(xrd_conform_execution *execution, brix_conn *connection,
268 uint16_t opcode, const char *id)
269{
270 ClientRequestHdr request;
271 uint16_t sid;
272 uint16_t status = 0;
273 uint8_t *body = NULL;
274 uint32_t length = 0;
275 brix_resp_out output = { &status, &body, &length };
276 brix_status error;
277 int refused;
278 char reason[64];
279
280 memset(&request, 0, sizeof(request));
281 request.requestid = htons(opcode);
282 brix_status_clear(&error);
283 refused = xrd_conform_reserve(execution, sizeof(request), reason,

286 && brix_recv(connection, sid, &output, &error) != 0
287 && error.kxr == kXR_InvalidRequest;
288 xrd_conform_grade(execution, id, refused, "invalid-opcode-defined-error",

293void
294xrd_conform_probe_framing(xrd_conform_execution *execution,
295 const brix_url *url)
296{
297 brix_opts opts = execution->run->opts;
298 brix_conn connection;
299 brix_status status;
300
301 if (!xrd_conform_family_active(execution, "FRM")) return;
302 opts.capture = NULL;
303 brix_status_clear(&status);
304 if (brix_connect(&connection, url, &opts, &status) != 0) return;
305 if (xrd_conform_row_active(execution, "FRM-001")
306 || xrd_conform_row_active(execution, "FRM-002")
307 || xrd_conform_row_active(execution, "FRM-003")
308 || xrd_conform_row_active(execution, "FRM-005")
309 || xrd_conform_row_active(execution, "FRM-006"))
310 probe_sequential(execution, &connection, &status);
311 if (xrd_conform_row_active(execution, "FRM-004")
312 || xrd_conform_row_active(execution, "FRM-022")
313 || xrd_conform_row_active(execution, "FRM-023"))
314 probe_pipeline(execution, &connection, &status);
315 if (xrd_conform_row_active(execution, "FRM-016"))
316 probe_padding(execution, &connection, &status);
317 if (xrd_conform_row_active(execution, "FRM-017"))
318 probe_invalid_opcode(execution, &connection, kXR_REQFENCE, "FRM-017");
319 if (xrd_conform_row_active(execution, "FRM-018"))
320 probe_invalid_opcode(execution, &connection, kXR_1stRequest - 1,
321 "FRM-018");
322 brix_close(&connection);
323}
