1/*
2 * sd_mirage.c — sizes-only SYNTHETIC storage backend (parity audit §3 row 14,
3 * the Mirage zero-storage pattern-read analog).
4 *
5 * WHAT: a read-only backend that stores NOTHING: every path opens as a regular
6 * file of the configured size, and reads return a deterministic
7 * offset-derived byte pattern (byte at absolute offset o is
8 * (o * 131 + 7) & 0xFF), so any range read is independently verifiable
9 * by the client.
10 * WHY: protocol/throughput testing without disks — drive the full root://
11 * (or HTTP) stack at wire speed with zero storage behind it, exactly
12 * what stock's Mirage OSS is used for on WLCG links.
13 * HOW: no syscalls at all (tier-1 rule satisfied trivially): open heap-allocs
14 * the object shell (heap_shell — mirroring sd_block), pread/preadv fill
15 * the pattern clamped to the synthetic size, fstat/stat synthesize a
16 * fixed regular-file record. No CAP_FD (nothing to sendfile — reads are
17 * served through the memory path), no write caps: a write open is
18 * refused EROFS at the door.
19 */

35static ngx_int_t
36sd_mirage_init(brix_sd_instance_t *inst, void *driver_conf)
37{
38 const brix_sd_mirage_conf_t *conf = driver_conf;
39 sd_mirage_state_t *st;
40
41 if (conf == NULL || conf->size < 0) {
42 errno = EINVAL;
43 return NGX_ERROR;
44 }
45 st = ngx_pcalloc(inst->pool, sizeof(*st));
46 if (st == NULL) {
47 errno = ENOMEM;
48 return NGX_ERROR;
49 }
50 st->size = conf->size;
51 inst->state = st;
52 return NGX_OK;
53}

55static brix_sd_obj_t *
56sd_mirage_open(brix_sd_instance_t *inst, const char *path, int sd_flags,
57 mode_t mode, int *err_out)
58{
59 brix_sd_obj_t *obj;
60
61 (void) path; /* every path names the same synthetic object */
62 (void) mode;
63
64 if (sd_flags & (BRIX_SD_O_WRITE | BRIX_SD_O_CREATE | BRIX_SD_O_TRUNC
65 | BRIX_SD_O_APPEND))
66 {
67 if (err_out != NULL) { *err_out = EROFS; }
68 return NULL;
69 }
70
71 /* Heap shell, exactly like sd_block: open may run off the event loop and
72 * inst->pool is not thread-safe. */
73 obj = ngx_calloc(sizeof(*obj), inst->log);
74 if (obj == NULL) {
75 if (err_out != NULL) { *err_out = ENOMEM; }
76 return NULL;
77 }
78 obj->driver = inst->driver;
79 obj->inst = inst;
80 obj->fd = NGX_INVALID_FILE; /* nothing kernel-backed */
81 obj->state = NULL;
82 obj->heap_shell = 1;
83 return obj;
84}

93static ssize_t
94sd_mirage_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
95{
96 const sd_mirage_state_t *st = obj->inst->state;
97 u_char *p = buf;
98 size_t i;
99
100 if (off < 0) {
101 errno = EINVAL;
102 return -1;
103 }
104 if (off >= st->size) {
105 return 0; /* EOF */
106 }
107 if ((int64_t) len > st->size - off) {
108 len = (size_t) (st->size - off);
109 }
110 for (i = 0; i < len; i++) {
111 p[i] = MIRAGE_BYTE((uint64_t) off + i);
112 }
113 return (ssize_t) len;
114}

116static ssize_t
117sd_mirage_preadv(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
118 off_t off)
119{
120 ssize_t total = 0;
121 int i;
122
123 for (i = 0; i < iovcnt; i++) {
124 ssize_t n = sd_mirage_pread(obj, iov[i].iov_base, iov[i].iov_len,
125 off + total);
126 if (n < 0) {
127 return total > 0 ? total : n;
128 }
129 total += n;
130 if ((size_t) n < iov[i].iov_len) {
131 break; /* EOF inside this segment */
132 }
133 }
134 return total;
135}
