1/*
2 * recv_frame_bounds.h — pure XRootD request-framing bounds.
3 *
4 * WHAT: brix_max_payload_for_request() is the per-opcode payload-size cap table;
5 * brix_root_frame_dlen_ok() decodes a 24-byte ClientRequestHdr and applies
6 * that cap to its dlen field — the "reject an oversized dlen BEFORE any
7 * allocation" invariant, made callable on raw bytes.
8 * WHY: the request header is the first attacker-controlled bytes off the root://
9 * wire; the dlen cap is the bound that stops a hostile length from driving
10 * an allocation (hyper-hardening C-2). Carving it into a pure, nginx-free
11 * TU makes the invariant executable under a libFuzzer harness rather than
12 * review-only — recv_process.c (the framing state machine) is far too
13 * nginx-coupled to harness whole.
14 * HOW: pure — integer arithmetic over wire constants only; no allocation, no
15 * ctx, no I/O. recv_process.c reuses brix_max_payload_for_request from here
16 * so the cap table is single-source.
17 */

19#define BRIX_RECV_FRAME_BOUNDS_H

30/*
31 * Decode the fixed 24-byte ClientRequestHdr (streamid[2]@0, requestid@2 BE16,
