178/* WHAT: Require an asserted offline object to answer a nowait open promptly.
179 * WHY: The ordinary roundtrip helper retries kXR_wait; this clause must inspect
180 * the first answer before any client-side sleep can hide server blocking.
181 * HOW: Send/receive exactly once and accept success or any defined non-server
182 * error within two seconds, retaining the first wire status as evidence. */
183static void
184probe_staged_nowait(xrd_conform_execution *execution, const char *target)
185{
186 brix_url url;
187 brix_conn connection;
188 brix_status status;
189 uint8_t *body = NULL;
190 uint32_t body_length = 0;
191 uint16_t wire_status = 0;
192 uint64_t started;
193 uint64_t elapsed_ms = 0;
194 int connected;
195 int rc = -1;
196 int defined;
197 char received[160];
198
199 if (!xrd_conform_row_active(execution, "OPN-020") || target == NULL) return;
200 connected = connect_fixture(execution, target, 0, &url, &connection,
201 &status) == 0;
202 started = brix_mono_ns();
203 if (connected)
204 rc = send_open_once(execution, &connection, url.path,
205 kXR_open_read | kXR_nowait, &wire_status, &body, &body_length,
206 &status);
207 if (started != 0) elapsed_ms = (brix_mono_ns() - started) / 1000000U;
208 defined = nowait_answer_defined(connected, elapsed_ms, rc, wire_status,
209 &status);
210 snprintf(received, sizeof(received), "elapsed_ms=%llu wire=%u kxr=%d",
211 (unsigned long long) elapsed_ms, (unsigned) wire_status, status.kxr);
212 xrd_conform_result_set(execution, "OPN-020",
213 defined ? BRIX_CONFORM_VERDICT_CONFORMANT
214 : BRIX_CONFORM_VERDICT_DIVERGENT,
215 defined ? "offline-nowait-returned-prompt-defined-answer"
216 : "offline-nowait-blocked-or-returned-undefined-answer",
217 "first answer in <=2000ms", received);
218 close_open_response(execution, &connection, rc, wire_status, body,
219 body_length, &status);
220 free(body);
221 if (connected) brix_close(&connection);
222}

233/* WHAT: Exercise a handle after an asserted sub-two-second idle threshold.
234 * WHY: A transport drop masquerading as handle expiry is not a defined answer.
235 * HOW: Open, idle for two seconds, then require either one byte or a native kXR
236 * refusal; a socket/protocol failure is divergent. */
237static void
238probe_idle(xrd_conform_execution *execution, const char *target)
239{
240 brix_url url;
241 brix_conn connection;
242 brix_file file;
243 brix_status status;
244 struct timespec idle = { 2, 0 };
245 uint8_t byte;
246 ssize_t got = -1;
247 int connected;
248 int opened = 0;
249 int pass;
250 char received[128];
251
252 if (!xrd_conform_row_active(execution, "OPN-033") || target == NULL) return;
253 connected = connect_fixture(execution, target, 0, &url, &connection,
254 &status) == 0;
255 if (connected && xrd_conform_reserve_status(execution, 0, &status)
256 && brix_file_open_read(&connection, url.path, &file, &status) == 0) {
257 opened = 1;
258 while (nanosleep(&idle, &idle) != 0 && errno == EINTR) { }
259 if (xrd_conform_reserve_status(execution, 1, &status))
260 got = brix_file_read(&connection, &file, 0, &byte, 1, &status);
261 }
262 pass = idle_answer_defined(opened, got, &status);
263 snprintf(received, sizeof(received), "opened=%d read=%lld kxr=%d",
264 opened, (long long) got, status.kxr);
265 xrd_conform_result_set(execution, "OPN-033",
266 pass ? BRIX_CONFORM_VERDICT_CONFORMANT
267 : BRIX_CONFORM_VERDICT_DIVERGENT,
268 pass ? (got == 1 ? "idle-handle-survived"
269 : "idle-handle-returned-defined-error")
270 : "idle-handle-silently-invalidated",
271 "two-second idle then byte or native kXR error", received);
272 if (opened && xrd_conform_reserve_status(execution, 0, &status))
273 brix_file_close(&connection, &file, &status);
274 if (connected) brix_close(&connection);
275}
