1/*
2 * tpc_verify.c - HTTP-TPC pull completion gate (size + checksum).
3 *
4 * WHAT: webdav_tpc_verify_pulled() runs after the last byte of a COPY pull has
5 * landed in the staged temp and before it is committed. It re-probes the
6 * source with one HEAD (carrying Want-Digest when a checksum algorithm is
7 * configured), then compares the source's declared Content-Length against
8 * the bytes on disk and, when asked, recomputes the returned RFC-3230
9 * Digest over the staged temp. Any disagreement fails the pull.
10 *
11 * WHY: the pull's only in-band completion signal is "curl stopped without an
12 * error", which a chunked source that dies mid-body, a truncating
13 * middlebox, or a corrupting one can all produce. The native root:// TPC
14 * has carried both gates for this reason (brix_tpc_require_source_size /
15 * brix_tpc_verify_checksum, src/tpc/outbound/source_stream.c); the HTTP
16 * plane had neither, so the truncation and bit-flip classes proven on the
17 * native plane were unimplemented here. Semantics are deliberately the
18 * same as the native pair so an operator reasons about one contract.
19 *
20 * HOW: both halves are off by default, so an existing deployment sees no new
21 * refusal. Whenever either is on the size comparison runs (it is free
22 * once the HEAD is paid for); tpc_require_source_size decides only what a
23 * source that declares NO length means. The checksum half is fail-closed
24 * exactly like the native one: no Digest, an unparseable one, an algorithm
25 * brix cannot compute, or a mismatch all refuse. Every refusal returns
26 * NGX_HTTP_BAD_GATEWAY, which the three pull tiers already treat as a
27 * failed transfer — the staged temp is aborted, never committed. The gate
28 * is hooked into webdav_tpc_run_curl_pull() and the multi-stream driver,
29 * so all three tiers (sync, thread, marker) pass through it.
30 */

39/* Longest "alg=value" a source can hand back and still be usable: an algorithm
40 * name plus a SHA-512 hex digest fits in well under half of this. */
41#define WEBDAV_TPC_DIGEST_MAX 256

43/*
44 * tpc_verify_src_t — what the completion HEAD learned about the source.
45 * `size` is -1 when the source declared no Content-Length (a chunked origin);
46 * `digest` is the first RFC-3230 "alg=value" pair of its Digest: header, or "".
47 */
48typedef struct {
49 off_t size;
50 char digest[WEBDAV_TPC_DIGEST_MAX];
51} tpc_verify_src_t;

54/*
55 * tpc_verify_header_cb — libcurl header sink that keeps only Digest:.
56 * Stores the trimmed value (first one wins, over-long ones are dropped so the
57 * checksum half fails closed rather than comparing a truncated digest) and
58 * always reports the full byte count so curl never treats this as a write error.
59 */
60static size_t
61tpc_verify_header_cb(char *buf, size_t size, size_t nitems, void *userdata)
62{
63 tpc_verify_src_t *src = userdata;
64 size_t total = size * nitems;
65 size_t len = total;
66 const char *v = buf;
67 static const char key[] = "digest:";
68 const size_t klen = sizeof(key) - 1;
69
70 if (len <= klen || src->digest[0] != '\0'
71 || ngx_strncasecmp((u_char *) buf, (u_char *) key, klen) != 0)
72 {
73 return total;
74 }
75
76 v += klen;
77 len -= klen;
78 while (len > 0 && (*v == ' ' || *v == '\t')) {
79 v++;
80 len--;
81 }
82 while (len > 0 && (v[len - 1] == '\r' || v[len - 1] == '\n'
83 || v[len - 1] == ' ' || v[len - 1] == '\t'))
84 {
85 len--;
86 }
87 if (len > 0 && len < sizeof(src->digest)) {
88 ngx_memcpy(src->digest, v, len);
89 src->digest[len] = '\0';
90 }
91 return total;
92}

95/*
96 * tpc_verify_want_digest — append "Want-Digest: <alg>" to the probe's header
97 * list and re-install it. No-op when the checksum half is off. Returns NGX_OK,
98 * or NGX_ERROR on slist OOM (the caller frees the partial list).
99 */
100static ngx_int_t
101tpc_verify_want_digest(CURL *curl, ngx_http_brix_webdav_loc_conf_t *conf,
102 struct curl_slist **hdrs)
103{
104 char want[64];
105 struct curl_slist *next;
106
107 if (conf->common.tpc_verify_checksum.len == 0) {
108 return NGX_OK;
109 }
110 ngx_snprintf((u_char *) want, sizeof(want), "Want-Digest: %V%Z",
111 &conf->common.tpc_verify_checksum);
112 next = curl_slist_append(*hdrs, want);
113 if (next == NULL) {
114 return NGX_ERROR;
115 }
116 *hdrs = next;
117 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, *hdrs);
118 return NGX_OK;
119}

122/*
123 * tpc_verify_probe_source — one HEAD of the source over the same secured handle
124 * configuration the pull itself used (TLS pin, rebind guard, client credential,
125 * transfer headers), capturing Content-Length and Digest:.
126 * NGX_OK with *out filled, or NGX_ERROR when the probe could not be made.
127 */
128static ngx_int_t
129tpc_verify_probe_source(ngx_log_t *log,
130 ngx_http_brix_webdav_loc_conf_t *conf, const char *url,
131 ngx_array_t *transfer_headers, const char *user_cert, const char *user_key,
132 tpc_verify_src_t *out)
133{
134 CURL *curl;
135 struct curl_slist *hdrs = NULL;
136 struct curl_slist *resolve = NULL;
137 CURLcode res;
138 curl_off_t cl = -1;
139
140 ngx_memzero(out, sizeof(*out));
141 out->size = -1;
142
143 curl = curl_easy_init();
144 if (curl == NULL) {
145 return NGX_ERROR;
146 }
147 if (tpc_curl_apply_conf(curl, conf, url, transfer_headers, log,
148 user_cert, user_key, &hdrs, &resolve) < 0
149 || tpc_verify_want_digest(curl, conf, &hdrs) != NGX_OK)
150 {
151 return webdav_tpc_curl_finish(NGX_ERROR, curl, hdrs, resolve, NULL);
152 }
153
154 curl_easy_setopt(curl, CURLOPT_NOBODY, 1L);
155 curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, tpc_verify_header_cb);
156 curl_easy_setopt(curl, CURLOPT_HEADERDATA, out);
157
158 /* SECURITY (phase-106 W5 follow-up): this HEAD probe runs on a thread-pool
159 * thread (webdav_tpc_verify_pulled <- the TPC pull task). Without a
160 * timeout a black-holed source pins that thread forever, and enough of
161 * them exhaust the pool — a slow-loris against the completion gate. Bound
162 * both the connect and the whole probe; reuse the operator's TPC budgets so
163 * one knob governs the transfer and its verification alike. */
164 curl_easy_setopt(curl, CURLOPT_NOSIGNAL, 1L);
165 curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, BRIX_TPC_CONNECT_TIMEOUT_SECS);
166 if (conf->tpc_timeout > 0) {
167 curl_easy_setopt(curl, CURLOPT_TIMEOUT, (long) conf->tpc_timeout);
168 }
169
170 res = curl_easy_perform(curl);
171 if (res != CURLE_OK) {
172 ngx_log_error(NGX_LOG_WARN, log, 0,
173 "brix_webdav: HTTP-TPC completion probe failed: %s",
174 curl_easy_strerror(res));
175 return webdav_tpc_curl_finish(NGX_ERROR, curl, hdrs, resolve, NULL);
176 }
177
178 curl_easy_getinfo(curl, CURLINFO_CONTENT_LENGTH_DOWNLOAD_T, &cl);
179 out->size = (cl >= 0) ? (off_t) cl : -1;
180 return webdav_tpc_curl_finish(NGX_OK, curl, hdrs, resolve, NULL);
181}

184/*
185 * tpc_verify_size — compare what arrived against what the source declared.
186 * A declared length that disagrees with the staged temp is unambiguous
187 * truncation (or over-read) and always refuses; an undeclared length refuses
188 * only under brix_webdav_tpc_require_source_size.
189 */
190static ngx_int_t
191tpc_verify_size(ngx_log_t *log, ngx_http_brix_webdav_loc_conf_t *conf,
192 const tpc_verify_src_t *src, int fd)
193{
194 struct stat st;
195
196 if (src->size < 0) {
197 if (!conf->common.tpc_require_source_size) {
198 return NGX_OK;
199 }
200 ngx_log_error(NGX_LOG_ERR, log, 0,
201 "brix_webdav: HTTP-TPC pull refused: source declared no "
202 "size and brix_webdav_tpc_require_source_size is on");
203 return NGX_HTTP_BAD_GATEWAY;
204 }
205 if (fstat(fd, &st) != 0) {
206 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
207 "brix_webdav: HTTP-TPC pull refused: cannot stat the "
208 "staged temp for the size gate");
209 return NGX_HTTP_BAD_GATEWAY;
210 }
211 if (st.st_size != src->size) {
212 ngx_log_error(NGX_LOG_ERR, log, 0,
213 "brix_webdav: HTTP-TPC pull truncated: wrote %O of %O "
214 "source bytes", st.st_size, src->size);
215 return NGX_HTTP_BAD_GATEWAY;
216 }
217 return NGX_OK;
218}

221/*
222 * tpc_verify_split — split an RFC-3230 Digest value into algorithm and value.
223 * Accepts the first pair of a comma-separated list ("adler32=1f2e,md5=..." →
224 * "adler32" + "1f2e"). NGX_ERROR when there is no '=' or either half does not
225 * fit, which the caller turns into a refusal.
226 */
227static ngx_int_t
228tpc_verify_split(const char *digest, char *alg, size_t algsz,
229 char *val, size_t valsz)
230{
231 const char *eq = strchr(digest, '=');
232 size_t alen;
233 size_t vlen;
234
235 if (eq == NULL) {
236 return NGX_ERROR;
237 }
238 alen = (size_t) (eq - digest);
239 vlen = strcspn(eq + 1, ",");
240 if (alen == 0 || alen >= algsz || vlen == 0 || vlen >= valsz) {
241 return NGX_ERROR;
242 }
243 ngx_memcpy(alg, digest, alen);
244 alg[alen] = '\0';
245 ngx_memcpy(val, eq + 1, vlen);
246 val[vlen] = '\0';
247 return NGX_OK;
248}

251/*
252 * tpc_verify_checksum — recompute the source's digest over the staged temp.
253 * Fail-closed on every uncertainty, matching tpc_verify_source_checksum() on the
254 * native plane: no Digest returned, an unparseable one, an algorithm brix cannot
255 * compute, or a value mismatch all refuse the pull.
256 */
257static ngx_int_t
258tpc_verify_checksum(ngx_log_t *log, ngx_http_brix_webdav_loc_conf_t *conf,
259 const tpc_verify_src_t *src, int fd, const char *tmp_path)
260{
261 char alg[32];
262 char src_hex[2 * EVP_MAX_MD_SIZE + 1];
263 char local_hex[2 * EVP_MAX_MD_SIZE + 1];
264 char normalized[32];
265
266 if (conf->common.tpc_verify_checksum.len == 0) {
267 return NGX_OK;
268 }
269 if (src->digest[0] == '\0') {
270 ngx_log_error(NGX_LOG_ERR, log, 0,
271 "brix_webdav: HTTP-TPC pull refused: source supplied no "
272 "Digest for Want-Digest: %V", &conf->common.tpc_verify_checksum);
273 return NGX_HTTP_BAD_GATEWAY;
274 }
275 if (tpc_verify_split(src->digest, alg, sizeof(alg),
276 src_hex, sizeof(src_hex)) != NGX_OK)
277 {
278 ngx_log_error(NGX_LOG_ERR, log, 0,
279 "brix_webdav: HTTP-TPC pull refused: malformed source "
280 "Digest header");
281 return NGX_HTTP_BAD_GATEWAY;
282 }
283 if (brix_checksum_hex_name_fd(alg, fd, tmp_path, log, local_hex,
284 sizeof(local_hex), normalized,
285 sizeof(normalized)) != NGX_OK)
286 {
287 ngx_log_error(NGX_LOG_ERR, log, 0,
288 "brix_webdav: HTTP-TPC pull refused: cannot compute "
289 "\"%s\" on the staged temp", alg);
290 return NGX_HTTP_BAD_GATEWAY;
291 }
292 if (strcasecmp(local_hex, src_hex) != 0) {
293 ngx_log_error(NGX_LOG_ERR, log, 0,
294 "brix_webdav: HTTP-TPC checksum mismatch: source %s=%s "
295 "destination=%s", alg, src_hex, local_hex);
296 return NGX_HTTP_BAD_GATEWAY;
297 }
298 return NGX_OK;
299}

302ngx_int_t
303webdav_tpc_verify_pulled(ngx_log_t *log,
304 ngx_http_brix_webdav_loc_conf_t *conf, const char *source_url,
305 const char *tmp_path, ngx_array_t *transfer_headers,
306 const char *user_cert, const char *user_key)
307{
308 tpc_verify_src_t src;
309 ngx_int_t rc;
310 int fd;
311
312 if (!conf->common.tpc_require_source_size && conf->common.tpc_verify_checksum.len == 0) {
313 return NGX_OK;
314 }
315
316 if (tpc_verify_probe_source(log, conf, source_url, transfer_headers,
317 user_cert, user_key, &src) != NGX_OK)
318 {
319 ngx_log_error(NGX_LOG_ERR, log, 0,
320 "brix_webdav: HTTP-TPC pull refused: the completion gate "
321 "could not re-probe the source");
322 return NGX_HTTP_BAD_GATEWAY;
323 }
324
325 /* Confined read-open of the staged temp — the same seam the pull wrote it
326 * through, so the gate cannot be pointed at a planted symlink target. */
327 fd = brix_vfs_open_fd(log, conf->common.root_canon, tmp_path,
328 O_RDONLY | O_CLOEXEC | O_NOFOLLOW, 0);
329 if (fd < 0) {
330 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
331 "brix_webdav: HTTP-TPC pull refused: cannot re-open the "
332 "staged temp for the completion gate");
333 return NGX_HTTP_BAD_GATEWAY;
334 }
335
336 rc = tpc_verify_size(log, conf, &src, fd);
337 if (rc == NGX_OK) {
338 rc = tpc_verify_checksum(log, conf, &src, fd, tmp_path);
339 }
340 (void) close(fd);
341 return rc;
342}
