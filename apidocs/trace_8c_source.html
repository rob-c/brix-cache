2 * trace.c — §15 client diagnostics: wire-trace + per-opcode timing.
3 *
4 * WHAT: Decode requestid/status codes to names, emit one human-readable line per
5 * wire frame (optionally with a bounded hexdump), and accumulate/print
6 * per-opcode round-trip times.
7 * WHY: The native client owns both ends of the protocol and `libxrdproto` can
8 * name every kXR_* struct it frames — so `--wire-trace`/`--timing` give a

10 * hook in frame.c/conn.c is a single `if (c->diag.*)` when disabled.
11 * HOW: Switch tables over our own opcodes.h constants (clean-room: names from the
12 * wire spec, never from XrdCl). All output goes to stderr so stdout stays
13 * clean for the conformance harness.

15 * names: src/protocols/root/protocol/opcodes.h request opcodes (3000–3032) + response statuses.

49/*
50 * Phase 40 (a): bounded backoff jitter for the SYNCHRONOUS retry / kXR_wait
51 * paths. Returns a pseudo-random value in [0, span_ms). Lazily seeded from the
52 * monotonic clock so concurrent clients/processes spread their wake-ups instead
53 * of retrying in lockstep (thundering-herd). Same xorshift the async core uses
54 * (aio.c rc_worker_main); kept here as the single leaf helper both frame.c and
55 * xrdcp.c reuse, with NO dependency on the aio/thread machinery.
56 */
57unsigned
58brix_jitter_ms(unsigned span_ms)
59{
60 static uint64_t state; /* lazily seeded; 0 until first use */
61 if (span_ms == 0) {
62 return 0;
63 }
64 if (state == 0) {
65 state = brix_mono_ns() | 1ull;
66 }
67 state ^= state << 13;
68 state ^= state >> 7;
69 state ^= state << 17;
70 return (unsigned) (state % span_ms);
71}

73/*
74 * WHAT: Emit a bounded hexdump (256 bytes, 16 per event).
75 * WHY: Fragmented stderr writes interleave with other threads' diagnostics.
76 * HOW: Format each row locally, then emit one complete DEBUG record.
77 */
78static void
79hexdump(const uint8_t *buf, uint32_t n)
80{
81 uint32_t cap = (n < 256) ? n : 256;
82 uint32_t i;
83 for (i = 0; i < cap; i += 16) {
84 uint32_t j;
85 char row[49];
86 size_t used = 0;
87 for (j = i; j < i + 16 && j < cap; j++) {
88 used += (size_t) snprintf(row + used, sizeof(row) - used,
89 "%02x ", buf[j]);
90 }
91 BRIX_CLIENT_LOG("DEBUG", "offset=%04x bytes=%s", i, row);
92 }
93 if (n > cap) {
94 BRIX_CLIENT_LOG("DEBUG", " … (%u more bytes)\n", n - cap);
95 }
96}

98/* ---- Trace one kXR_error response frame ----
99 *
100 * WHAT: Emits a single stderr line decoding an error frame: the 4-byte
101 * big-endian errnum, its symbolic name, and the trailing message text
102 * (bytes past the errnum, printed with an explicit length so the buffer
103 * need not be null-terminated).
104 *
105 * WHY: The errnum-plus-message layout is specific to kXR_error; isolating it
106 * keeps brix_trace_response a flat dispatch and each per-status decoder
107 * independently reviewable. Caller guarantees b != NULL and blen >= 4.
108 *
109 * HOW: 1. Read the leading 32-bit BE errnum via the unaligned-safe accessor.
110 * 2. Print errnum, its brix_kxr_name(), and the message slice
111 * (blen - 4 bytes when present, else an empty string).
112 */
113static void
114brix_trace_error(int dir, uint16_t sid, const uint8_t *b, uint32_t blen)
115{
116 int errnum = (int) xrd_get_u32_be(b);
117 BRIX_CLIENT_LOG("DEBUG", "%c sid=%u error errnum=%d (%s) \"%.*s\"\n",
118 dir, sid, errnum, brix_kxr_name(errnum),
119 (int) (blen > 4 ? blen - 4 : 0),
120 blen > 4 ? (const char *) (b + 4) : "");
121}

123/* ---- Trace one kXR_redirect response frame ----
124 *
125 * WHAT: Emits a single stderr line decoding a redirect frame: the 4-byte
126 * big-endian port followed by the target host slice (bytes past the
127 * port), rendered as host:port.
128 *
129 * WHY: The port-then-host layout is unique to kXR_redirect; a dedicated
130 * decoder mirrors brix_trace_error and keeps the dispatcher branch-free
131 * of field math. Caller guarantees b != NULL and blen >= 4.
132 *
133 * HOW: 1. Read the leading 32-bit BE port via the unaligned-safe accessor.
134 * 2. Print the host slice (blen - 4 bytes when present, else empty)
135 * joined to the port with a colon.
136 */
137static void
138brix_trace_redirect(int dir, uint16_t sid, const uint8_t *b, uint32_t blen)
139{
140 int port = (int) xrd_get_u32_be(b);
141 BRIX_CLIENT_LOG("DEBUG", "%c sid=%u redirect → %.*s:%d\n",
142 dir, sid, (int) (blen > 4 ? blen - 4 : 0),
143 blen > 4 ? (const char *) (b + 4) : "", port);
144}

146/* ---- Trace one kXR_wait response frame ----
147 *
148 * WHAT: Emits a single stderr line decoding a wait frame's 4-byte big-endian
149 * wait interval, in seconds.
150 *
151 * WHY: kXR_wait carries only the retry delay; a leaf decoder keeps it uniform
152 * with the other per-status helpers. Caller guarantees b != NULL and
153 * blen >= 4.
154 *
155 * HOW: 1. Read the 32-bit BE interval via the unaligned-safe accessor.
156 * 2. Print it with a trailing 's' for seconds.
157 */
158static void
159brix_trace_wait(int dir, uint16_t sid, const uint8_t *b)
160{
161 BRIX_CLIENT_LOG("DEBUG", "%c sid=%u wait %us\n",
162 dir, sid, xrd_get_u32_be(b));
163}

165/* ---- Trace a response (non-request) frame ----
166 *
167 * WHAT: Emits one stderr line for a response frame, decoding the few
168 * high-value bodies (error, redirect, wait) when the body is present and
169 * long enough, or falling back to a generic status-name + dlen line.
170 *
171 * WHY: Splitting the response side out of brix_trace_frame collapses the
172 * status branch ladder into a flat early-dispatch and lets each body
173 * decoder be a small single-purpose helper — the same wire fields are
174 * decoded from our own structs, never from XrdCl.
175 *
176 * HOW: 1. If code is kXR_error with a >= 4-byte body, delegate to
177 * brix_trace_error.
178 * 2. Else if kXR_redirect with a >= 4-byte body, delegate to
179 * brix_trace_redirect.
180 * 3. Else if kXR_wait with a >= 4-byte body, delegate to
181 * brix_trace_wait.
182 * 4. Otherwise print the generic status-name + dlen line.
183 */
184static void
185brix_trace_response(int dir, uint16_t sid, int code, uint32_t dlen,
186 const uint8_t *b, uint32_t blen)
187{
188 if (code == kXR_error && b != NULL && blen >= 4) {
189 brix_trace_error(dir, sid, b, blen);
190 } else if (code == kXR_redirect && b != NULL && blen >= 4) {
191 brix_trace_redirect(dir, sid, b, blen);
192 } else if (code == kXR_wait && b != NULL && blen >= 4) {
193 brix_trace_wait(dir, sid, b);
194 } else {
195 BRIX_CLIENT_LOG("DEBUG", "%c sid=%u %s dlen=%u\n",
196 dir, sid, brix_status_name(code), dlen);
197 }
198}

200void
201brix_trace_frame(brix_conn *c, int dir, uint16_t sid, int code,
202 int is_request, uint32_t dlen, const void *body, uint32_t blen)
203{
204 const uint8_t *b = (const uint8_t *) body;
205
206 if (is_request) {
207 BRIX_CLIENT_LOG("DEBUG", "%c sid=%u %s dlen=%u\n",
208 dir, sid, brix_reqid_name(code), dlen);
209 } else {
210 brix_trace_response(dir, sid, code, dlen, b, blen);
211 }
212
213 /* Handshake bodies contain credentials/session material, never dump them.
214 * Later signing and authentication requests remain sensitive as well. */
215 if (c->diag.wire_trace >= 2 && b != NULL && blen > 0
216 && c->diag.phase_ns[3] != 0
217 && (!is_request || (code != kXR_auth && code != kXR_login
218 && code != kXR_sigver))
219 && (is_request || code != kXR_authmore)) {
220 hexdump(b, blen);
221 }
222}

224void
225brix_timing_report(const brix_conn *c)
226{
227 int i, any = 0;
228 for (i = 0; i < XRDC_NOP; i++) {
229 uint64_t n = c->diag.rtt[i].n;
230 double tot, avg;
231 if (n == 0) {
232 continue;
233 }
234 if (!any) {
235 fprintf(stderr, "--- per-opcode RTT (n / total / min / avg / max ms) ---\n");
236 any = 1;
237 }
238 tot = (double) c->diag.rtt[i].tot_ns / 1e6;
239 avg = tot / (double) n;
240 BRIX_CLIENT_LOG("DEBUG", " %-12s n=%-4llu tot=%.3f min=%.3f avg=%.3f max=%.3f\n",
241 brix_reqid_name(i + kXR_1stRequest),
242 (unsigned long long) n, tot,
243 (double) c->diag.rtt[i].min_ns / 1e6, avg,
244 (double) c->diag.rtt[i].max_ns / 1e6);
245 }
