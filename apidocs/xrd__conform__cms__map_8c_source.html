16static int
17endpoint_is_numeric(const char *endpoint)
18{
19 char url_text[CONFORM_LOCATE_URL_MAX];
20 struct in_addr ipv4;
21 struct in6_addr ipv6;
22 brix_url url;
23 brix_status status;
24
25 if ((size_t) snprintf(url_text, sizeof(url_text), "root://%s/", endpoint)
26 >= sizeof(url_text)) return 0;
27 brix_status_clear(&status);
28 if (brix_url_parse(url_text, &url, &status) != 0) return 0;
29 return inet_pton(AF_INET, url.host, &ipv4) == 1
30 || inet_pton(AF_INET6, url.host, &ipv6) == 1;
31}

131static int
132map_tool_read_all(int fd, char *output, size_t output_size, int *overflow)
133{
134 size_t used = 0;
135
136 while (used + 1 < output_size) {
137 ssize_t got = read(fd, output + used, output_size - used - 1);
138
139 if (got < 0 && errno == EINTR) continue;
140 if (got <= 0) break;
141 used += (size_t) got;
142 }
143 *overflow = used + 1 == output_size;
144 output[used] = '\0';
145 return 0;
146}

159/* WHAT: Execute the exact co-installed xrdmapc and capture bounded stdout.
160 * WHY: Calling a parser in-process would make CMS-007 circular and could not
161 * detect packaging or CLI transformation drift.
162 * HOW: Resolve the sibling from this executable, fork/execv without a shell,
163 * cap output, and accept only a clean zero exit. */
164static int
165run_map_tool(const xrd_conform_execution *execution, char *output,
166 size_t output_size)
167{
168 char tool[PATH_MAX];
169 char stall[32];
170 char *args[CMS_MAP_ARG_MAX];
171 int fds[2];
172 int status = 0;
173 int overflow = 0;
174 pid_t child;
175 pid_t waited;
176
177 if (map_tool_path(tool, sizeof(tool)) != 0) return -2;
178 map_tool_args(execution, tool, stall, sizeof(stall), args);
179 if (pipe(fds) != 0) return -1;
180 child = fork();
181 if (child < 0) {
182 close(fds[0]);
183 close(fds[1]);
184 return -1;
185 }
186 if (child == 0) {
187 close(fds[0]);
188 if (dup2(fds[1], STDOUT_FILENO) < 0) _exit(126);
189 close(fds[1]);
190 execv(tool, args);
191 _exit(127);
192 }
193 close(fds[1]);
194 (void) map_tool_read_all(fds[0], output, output_size, &overflow);
195 close(fds[0]);
196 waited = map_tool_wait(child, &status);
197 if (overflow || waited != child) return -1;
198 return WIFEXITED(status) && WEXITSTATUS(status) == 0 ? 0 : -1;
199}

201static int
202machine_holder_set(const char *output, char *canonical,
203 size_t canonical_size)
204{
205 char locate[CMS_MAP_OUTPUT_MAX];
206 char copy[CMS_MAP_OUTPUT_MAX];
207 char *line;
208 char *save;
209 size_t used = 0;
210 size_t count = 0;
211
212 if ((size_t) snprintf(copy, sizeof(copy), "%s", output) >= sizeof(copy))
213 return -1;
214 for (line = strtok_r(copy, "\n", &save); line != NULL;
215 line = strtok_r(NULL, "\n", &save)) {
216 char host[256];
217 int port;
218 size_t length = strlen(line);
219 int written;
220
221 if (length > 0 && line[length - 1] == '\r') line[--length] = '\0';
222 if (length == 0 || strpbrk(line, " \t") != NULL
223 || brix_split_host_port(line, host, sizeof(host), &port, 0) != 0
224 || ++count > 32) return -1;
225 written = snprintf(locate + used, sizeof(locate) - used,
226 "%sS=%s", used == 0 ? "" : " ", line);
227 if (written < 0 || (size_t) written >= sizeof(locate) - used)
228 return -1;
229 used += (size_t) written;
230 }
231 if (count == 0) return -1;
232 return xrd_conform_canonical_holder_set(locate, canonical,
233 canonical_size);
234}
