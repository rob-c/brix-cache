11/* `authdb <path>` — load identity-based ACL rules. Requires brix_auth gsi,
12 * token, or both; stores the path and parses its entries into authdb_rules.
13 * Returns NGX_CONF_OK, or NGX_CONF_ERROR (emerg-logged) on a bad prerequisite
14 * or parse error. */
15char *
16brix_conf_set_authdb(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
17{
18 ngx_stream_brix_srv_conf_t *xcf = conf;
19 ngx_str_t *value;
20
21 value = cf->args->elts;
22
23 /*
24 * The auth-mode requirement (authdb needs gsi/token for the native engine)
25 * is validated at merge time, where `brix_authdb_engine` is final — the
26 * xrdacc engine also authorizes anonymous `u *` rules, so it is exempt.
27 * (Directive order means xcf->auth / acc_format are not yet settled here.)
28 */
29 xcf->common.acc.authdb = value[1];
30
31 if (xcf->authdb_rules == NULL) {
32 xcf->authdb_rules = ngx_array_create(cf->pool, 4,
33 sizeof(brix_authdb_rule_t));
34 if (xcf->authdb_rules == NULL) {
35 return NGX_CONF_ERROR;
36 }
37 }
38
39 if (brix_parse_authdb(cf, &xcf->common.acc.authdb,
40 xcf->authdb_rules) != NGX_OK)
41 {
42 return NGX_CONF_ERROR;
43 }
44
45 return NGX_CONF_OK;
46}

73/* Shared `brix_require_vo <path> <vo>` grammar (phase-101 W4): append one VO ACL
74 * rule to *slot, lazily creating the array. The stream setter below, the HTTP
75 * setter (registered on the common module) AND the stream_common owner (W3) all
76 * call this, so require_vo parses byte-identically on every plane. Non-static so
77 * stream_common.c can reuse it (declared in ngx_brix_module.h). Returns
78 * NGX_CONF_OK / NGX_CONF_ERROR. */
79char *
80brix_vo_rules_append(ngx_conf_t *cf, ngx_str_t *value, ngx_array_t **slot)
81{
82 brix_vo_rule_t *rule;
83
84 if (*slot == NULL) {
85 *slot = ngx_array_create(cf->pool, 2, sizeof(brix_vo_rule_t));
86 if (*slot == NULL) {
87 return NGX_CONF_ERROR;
88 }
89 }
90
91 rule = ngx_array_push(*slot);
92 if (rule == NULL) {
93 return NGX_CONF_ERROR;
94 }
95
96 ngx_memzero(rule, sizeof(*rule));
97
98 if (brix_normalize_policy_path(cf->pool, &value[1], &rule->path) != NGX_OK) {
99 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
100 "brix_require_vo: invalid path \"%V\"", &value[1]);
101 return NGX_CONF_ERROR;
102 }
103
104 return brix_copy_conf_string(cf, &value[2], &rule->vo);
105}

118/* `brix_tpc_verify_checksum on|off|<alg>` on EVERY plane (phase-101 W4): unify the
119 * native-TPC boolean grammar and the webdav <alg> grammar into one. Normalizes to
120 * common.tpc_verify_checksum — "off"/absent => "" (off); "on" => "adler32" (the
121 * XRootD/WLCG default checksum); a checksum algorithm name => its canonical
122 * spelling. `common` is member 0 of every plane's conf, so the cast is valid.
123 * The native (root) TPC reads the field as a boolean gate (kXR_Qcksum negotiates
124 * its own algorithm); the webdav curl-COPY uses the algorithm for Want-Digest and
125 * the post-copy recompute. Returns NGX_CONF_OK / NGX_CONF_ERROR. */
126char *
127brix_conf_set_tpc_verify_checksum(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
128{
129 ngx_http_brix_shared_conf_t *sc = conf;
130 ngx_str_t *value = cf->args->elts;
131 brix_checksum_alg_t alg;
132 char norm[32];
133
134 (void) cmd;
135
136 if (sc->tpc_verify_checksum.data != NULL) {
137 return "is duplicate"; /* parity with the stock str-slot grammar */
138 }
139 if (value[1].len == 3
140 && ngx_strncasecmp(value[1].data, (u_char *) "off", 3) == 0)
141 {
142 /* Explicit off: a non-NULL empty marker, so a later duplicate is
143 * still caught and the merge never re-inherits over the opt-out. */
144 ngx_str_set(&sc->tpc_verify_checksum, "");
145 return NGX_CONF_OK;
146 }
147 if (value[1].len == 2
148 && ngx_strncasecmp(value[1].data, (u_char *) "on", 2) == 0)
149 {
150 ngx_str_set(&sc->tpc_verify_checksum, "adler32"); /* XRootD/WLCG default */
151 return NGX_CONF_OK;
152 }
153 if (brix_checksum_parse((const char *) value[1].data, value[1].len,
154 &alg, norm, sizeof(norm)) != NGX_OK)
155 {
156 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
157 "brix_tpc_verify_checksum: expected on, off, or a checksum algorithm "
158 "(adler32, crc32, crc32c, md5, sha1, sha256, crc64), got \"%V\"",
159 &value[1]);
160 return NGX_CONF_ERROR;
161 }
162 if (brix_pstrdupz(cf->pool, &sc->tpc_verify_checksum,
163 (u_char *) norm, ngx_strlen(norm)) != NGX_OK)
164 {
165 return NGX_CONF_ERROR;
166 }
167 return NGX_CONF_OK;
168}

246/* Postconfiguration finalization for the policy rules: validate cross-directive
247 * prerequisites once every directive has settled. Returns NGX_OK / NGX_ERROR. */
248ngx_int_t
249brix_config_finalize_policy(ngx_conf_t *cf,
250 ngx_stream_brix_srv_conf_t *xcf)
251{
252 if (xcf->common.vo_rules != NULL
253 && xcf->auth != BRIX_AUTH_GSI
254 && xcf->auth != BRIX_AUTH_TOKEN
255 && xcf->auth != BRIX_AUTH_BOTH)
256 {
257 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
258 "brix_require_vo requires brix_auth gsi, token or both");
259 return NGX_ERROR;
260 }
261
262 if (xcf->common.vo_rules != NULL) {
263 if (!brix_voms_available()) {
264 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
265 "brix_require_vo requires libvomsapi.so.1 at runtime "
266 "(install voms-libs on EL9)");
267 return NGX_ERROR;
268 }
269 if (xcf->common.vomsdir.len == 0 || xcf->common.voms_cert_dir.len == 0) {
270 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
271 "brix_require_vo requires brix_vomsdir and brix_voms_cert_dir");
272 return NGX_ERROR;
273 }
274
275 if (brix_validate_path(cf, "brix_vomsdir", &xcf->common.vomsdir,
276 BRIX_PATH_DIRECTORY, R_OK | X_OK)
277 != NGX_OK
278 || brix_validate_path(cf, "brix_voms_cert_dir",
279 &xcf->common.voms_cert_dir,
280 BRIX_PATH_DIRECTORY, R_OK | X_OK)
281 != NGX_OK)
282 {
283 return NGX_ERROR;
284 }
285 }
286 if (brix_finalize_vo_rules(cf->log, &xcf->common.root,
287 xcf->common.vo_rules)
288 != NGX_OK)
289 {
290 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
291 "brix: failed to finalize brix_require_vo rules for root \"%V\"",
292 &xcf->common.root);
293 return NGX_ERROR;
294 }
295
296 if (brix_finalize_authdb_rules(cf->log, &xcf->common.root, xcf->authdb_rules)
297 != NGX_OK)
298 {
299 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
300 "brix: failed to finalize brix_authdb rules for root \"%V\"",
301 &xcf->common.root);
302 return NGX_ERROR;
303 }
304
305 if (brix_finalize_group_rules(cf->log, &xcf->common.root,
306 xcf->group_rules) != NGX_OK)
307 {
308 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
309 "brix: failed to finalize brix_inherit_parent_group rules for root \"%V\"",
310 &xcf->common.root);
311 return NGX_ERROR;
312 }
313
314 return NGX_OK;
315}
