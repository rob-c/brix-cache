1/*
2 * xfer_mover_agent.c — the single crash-safe out-of-process transfer agent.
3 *
4 * WHAT: A payload-agnostic harness that runs an external transfer in a
5 * double-forked, reparented-to-init agent process and shuttles fixed-size
6 * request/reply frames over a socketpair, driving completions on the nginx
7 * event loop. Each transfer kind supplies an brix_xfer_agent_ops_t (frame
8 * sizes + agent-side process() + worker-side on_reply()).
9 *
10 * WHY: See xfer.h: reaping a worker child crashes the nginx master via the
11 * SHM-zone-as-slab-pool SIGCHLD handler. The reparented agent is the only
12 * safe way to fork/exec/waitpid a transfer command from a worker. This was
13 * proven in the FRM tape path (src/frm/stage.c); it is lifted here verbatim
14 * so write-through and TPC inherit the same one fork/reap path instead of
15 * each rolling their own (the posix_spawn-at-close hazard).
16 *
17 * HOW: attach: socketpair → block SIGCHLD → fork → fork(agent) → intermediate
18 * _exit (agent reparents to init) → reap the intermediate while SIGCHLD is
19 * blocked (nginx never sees it). The worker side is non-blocking and
20 * registered as an nginx read event. The agent loop reads a request, calls
21 * ops->process(), writes a reply; the worker read handler drains replies
22 * (ops->on_reply each), respawns on agent death, and calls ops->after_drain
23 * once. No goto; early-return throughout.
24 */

41/* Read exactly len bytes. Returns len, 0 on EOF, -1 on error. EINTR retried. */
42static ssize_t
43xfer_read_full(int fd, void *buf, size_t len)
44{
45 u_char *p = buf;
46 size_t got = 0;
47
48 while (got < len) {
49 ssize_t n = read(fd, p + got, len - got);
50 if (n == 0) {
51 return 0;
52 }
53 if (n < 0) {
54 if (errno == EINTR) {
55 continue;
56 }
57 return -1;
58 }
59 got += (size_t) n;
60 }
61 return (ssize_t) len;
62}

64/* Write exactly len bytes. Returns len, or -1 on error. EINTR retried. */
65static ssize_t

67{
68 const u_char *p = buf;
69 size_t put = 0;
70

72 ssize_t n = write(fd, p + put, len - put);
73 if (n < 0) {
74 if (errno == EINTR) {
75 continue;
76 }
77 return -1;
78 }
79 put += (size_t) n;
80 }
81 return (ssize_t) len;

86/* The agent loop: read a request, process it into a reply, send the reply.
87 * Exits on EOF (worker closed) or a fatal frame error. Never returns. */
88static void
89xfer_agent_loop(int fd, const brix_xfer_agent_ops_t *ops)
90{
91 void *req;
92 void *rep;
93
94 /* Previous handler is irrelevant in the fresh agent child — discard it. */
95 (void) signal(SIGCHLD, SIG_DFL); /* our own waitpid() must work */
96
97 req = malloc(ops->req_size);
98 rep = malloc(ops->rep_size);
99 if (req == NULL || rep == NULL) {
100 _exit(0);
101 }
102
103 for ( ;; ) {
104 if (xfer_read_full(fd, req, ops->req_size) <= 0) {
105 _exit(0); /* worker closed → shut down */
106 }
107 ops->process(req, rep, ops->data);
108 if (xfer_write_full(fd, rep, ops->rep_size) < 0) {
109 _exit(0);
110 }
111 }
112}

114/* Double-fork the agent (reparented to init so nginx never reaps it), reaping the
115 * intermediate while SIGCHLD is blocked. Returns the non-blocking worker-side fd,
116 * or -1. */
117static int
118xfer_agent_spawn(const brix_xfer_agent_ops_t *ops, ngx_log_t *log)
119{
120 int sv[2];
121 sigset_t block, prev;
122 pid_t inter;
123
124 if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) != 0) {
125 BRIX_LOG(NGX_LOG_ERR, log, ngx_errno, "xfer: %s socketpair failed",
126 ops->name);
127 return -1;
128 }
129
130 sigemptyset(&block);
131 sigaddset(&block, SIGCHLD);
132 sigprocmask(SIG_BLOCK, &block, &prev);
133
134 inter = fork();
135 if (inter < 0) {
136 sigprocmask(SIG_SETMASK, &prev, NULL);
137 close(sv[0]);
138 close(sv[1]);
139 BRIX_LOG(NGX_LOG_ERR, log, ngx_errno, "xfer: %s agent fork failed",
140 ops->name);
141 return -1;
142 }
143 if (inter == 0) {
144 pid_t agent = fork();
145 if (agent == 0) {
146 close(sv[0]);
147 xfer_agent_loop(sv[1], ops); /* never returns */
148 _exit(0);
149 }
150 _exit(0); /* intermediate → agent reparents */
151 }
152
153 /* parent: reap the intermediate ourselves (nginx never sees it). */
154 close(sv[1]);
155 while (waitpid(inter, NULL, 0) < 0 && errno == EINTR) { }
156 sigprocmask(SIG_SETMASK, &prev, NULL);
157
158 (void) fcntl(sv[0], F_SETFL, O_NONBLOCK);
159 return sv[0];
160}

166/* Recover from a dead agent: tear down the old socketpair/connection and spawn a
167 * fresh agent so transfers keep flowing. On respawn failure a->fd stays -1 and
168 * dispatch fails closed. */
169static void
170xfer_agent_respawn(brix_xfer_agent_t *a)
171{
172 brix_xfer_agent_teardown(a);
173 if (brix_xfer_agent_attach(a, a->ops, a->log) != NGX_OK) {
174 BRIX_LOG(NGX_LOG_ERR, a->log, 0,
175 "xfer: %s agent respawn failed — transfers paused until "
176 "the next reload", a->ops->name);
177 } else {
178 BRIX_LOG(NGX_LOG_NOTICE, a->log, 0,
179 "xfer: %s agent respawned after exit", a->ops->name);
180 }
181}

183/* nginx read-event handler: drain all pending agent replies. */
184static void
185xfer_agent_on_reply(ngx_event_t *rev)
186{
187 ngx_connection_t *c = rev->data;
188 brix_xfer_agent_t *a = c->data;
189 const size_t rep_size = a->ops->rep_size;
190
191 for ( ;; ) {
192 ssize_t n = recv(c->fd, a->rep_buf, rep_size, 0);
193
194 if (n == 0) {
195 /* Agent exited (EOF): respawn frees this connection, so return
196 * immediately — c/rev must not be touched afterward. */
197 BRIX_LOG(NGX_LOG_ERR, c->log, 0, "xfer: %s agent exited",
198 a->ops->name);
199 xfer_agent_respawn(a);
200 return;
201 }
202 if (n < 0) {
203 if (ngx_errno == NGX_EAGAIN) {
204 break;
205 }
206 if (ngx_errno == NGX_EINTR) {
207 continue;
208 }
209 BRIX_LOG(NGX_LOG_ERR, c->log, ngx_errno, "xfer: %s agent recv",
210 a->ops->name);
211 xfer_agent_respawn(a);
212 return;
213 }
214 if (n != (ssize_t) rep_size) {
215 /* partial frame: finish it (small, rare) */
216 if (xfer_read_full(c->fd, (u_char *) a->rep_buf + n,
217 rep_size - (size_t) n) <= 0)
218 {
219 xfer_agent_respawn(a);
220 return;
221 }
222 }
223 a->ops->on_reply(a->rep_buf, a->ops->data);
224 }
225
226 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
227 BRIX_LOG(NGX_LOG_ERR, c->log, 0, "xfer: %s agent read-event re-arm",
228 a->ops->name);
229 }
230 if (a->ops->after_drain != NULL) {
231 a->ops->after_drain(a->ops->data);
232 }

245ngx_int_t
246brix_xfer_agent_attach(brix_xfer_agent_t *a,
247 const brix_xfer_agent_ops_t *ops, ngx_log_t *log)
248{
249 a->ops = ops;
250 a->log = log;
251
252 if (a->rep_buf == NULL) {
253 a->rep_buf = malloc(ops->rep_size);
254 if (a->rep_buf == NULL) {
255 BRIX_LOG(NGX_LOG_ERR, log, 0, "xfer: %s reply-buf alloc failed",
256 ops->name);
257 return NGX_ERROR;
258 }
259 }
260
261 a->fd = xfer_agent_spawn(ops, log);
262 if (a->fd < 0) {
263 return NGX_ERROR;
264 }
265
266 a->conn = ngx_get_connection(a->fd, log);
267 if (a->conn == NULL) {
268 close(a->fd);
269 a->fd = -1;
270 return NGX_ERROR;
271 }
272 a->conn->data = a; /* recover the handle in handler */
273 a->conn->read->handler = xfer_agent_on_reply;
274 a->conn->read->log = log;
275 a->conn->recv = ngx_recv;
276 if (ngx_handle_read_event(a->conn->read, 0) != NGX_OK) {
277 BRIX_LOG(NGX_LOG_ERR, log, 0, "xfer: %s cannot arm agent read event",
278 ops->name);
279 brix_xfer_agent_teardown(a);
280 return NGX_ERROR;
281 }
282 return NGX_OK;
283}

285ngx_int_t
286brix_xfer_agent_dispatch(brix_xfer_agent_t *a, const void *req)
287{
288 const size_t req_size = a->ops->req_size;
289 ssize_t n;
290
291 if (a->fd < 0) {
292 return NGX_ERROR;
293 }
294
295 n = write(a->fd, req, req_size);
296 if (n == (ssize_t) req_size) {
297 return NGX_OK;
298 }
299 if (n < 0 && (ngx_errno == NGX_EAGAIN || ngx_errno == NGX_EINTR)) {
300 return NGX_AGAIN; /* agent backed up — retry later */
301 }
302 /* a partial/failed write: finish it blocking (small, rare) */
303 if (n >= 0 && xfer_write_full(a->fd, (const u_char *) req + n,
304 req_size - (size_t) n)
305 == (ssize_t) (req_size - (size_t) n))
306 {
307 return NGX_OK;
308 }
309 return NGX_ERROR;
310}
