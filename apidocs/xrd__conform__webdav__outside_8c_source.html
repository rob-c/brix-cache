136/* WHAT: Refuse COPY and MOVE to an explicitly authorized outside-export URL.
137 * WHY: A scratch-only destination cannot establish the export boundary.
138 * HOW: Snapshot the sacrificial URL, plant two scratch sources, require both
139 * mutations to fail, verify both sources and the unchanged destination. */
140void xrd_conform_probe_webdav_outside(xrd_conform_execution *execution,
141 const brix_weburl *base, const char *token)
142{
143 brix_weburl outside;
144 brix_weburl copy_source;
145 brix_weburl move_source;
146 dav_observation before;
147 char fields[768];
148 int copy_status = 0;
149 int move_status = 0;
150 int pass;
151
152 if (!row_active(execution)) return;
153 if (!outside_fixture_valid(execution, base, &outside)) {
154 xrd_conform_result_set(execution, "DAV-008",
155 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
156 "fixture-required:outside-export-destination", NULL, NULL);
157 return;
158 }
159 if (!prepare_outside_probe(execution, base, token, &outside, &copy_source,
160 &move_source, &before, fields, sizeof(fields))) {
161 xrd_conform_result_set(execution, "DAV-008",
162 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
163 "fixture-setup:outside-export-destination", NULL, NULL);
164 return;
165 }
166 pass = outside_mutations_refused(execution, &outside, &copy_source,
167 &move_source, token, fields, &before, &copy_status, &move_status);
168 xrd_conform_result_set(execution, "DAV-008",
169 pass ? BRIX_CONFORM_VERDICT_CONFORMANT
170 : BRIX_CONFORM_VERDICT_DIVERGENT,
171 pass ? "outside-export-copy-move-refused"
172 : "outside-export-destination-accepted-or-mutated",
173 NULL, pass ? "COPY/MOVE refused; sources and destination unchanged"
174 : "outside-export contract changed");
175 cleanup_outside_probe(execution, &outside, &copy_source, &move_source,
176 token, copy_status, move_status);
177}
