1/* brix_fault_udp.c — Phase-99 Wave C: a minimal UDP relay so the "UDP vs TCP"
2 * middlebox pathologies (the class where UDP fired ahead of / alongside TCP is
3 * silently dropped, held, or reaped) become reproducible from userland.
4 *
5 * WHAT: `--udp <listen> <host:port>` opens a UDP listener; each distinct client
6 * source address is a flow with its own connected upstream socket. The
7 * levers act on real datagrams: udp-drop (loss), udp-hold-until-tcp (delay
8 * a flow's FIRST datagram — the "allowed only as related to TCP" model),
9 * udp-reap (short flow-map timeout), udp-reorder (per-datagram hold-back).
10 *
11 * WHY: the TCP relay cannot express any of this; UDP is a separate data path.
12 *
13 * HOW: one poll loop over the listen socket + every live flow's upstream socket.
14 * Root-free. Counters feed the same status/JSON oracle. clear_all() resets
15 * the levers (the listener itself lives for the process, like a route). */
16

41/* cmd_set_udp — the udp-* control verbs. Returns 1 once one is claimed. */
42int
43cmd_set_udp(const char *verb, char *args, char *reply, size_t rsz)
44{
45 (void) reply; (void) rsz;
46 if (strcmp(verb, "udp-drop") == 0) {
47 int p = atoi(args);
48 g_udp_drop_ppm = p < 0 ? 0 : (p > 1000000 ? 1000000 : p);
49 } else if (strcmp(verb, "udp-hold-until-tcp") == 0) {
50 g_udp_hold_ms = atoi(args);
51 } else if (strcmp(verb, "udp-reap") == 0) {
52 g_udp_reap_ms = atoi(args);
53 } else if (strcmp(verb, "udp-reorder") == 0) {
54 int p = 0, ms = 0;
55 sscanf(args, "%d %d", &p, &ms);
56 g_udp_reorder_ppm = p; g_udp_reorder_ms = ms;
57 } else {
58 return 0;
59 }
60 return 1;
61}

63static long
64udp_elapsed_ms(const struct timespec *t0)
65{
66 struct timespec now;
67 clock_gettime(CLOCK_MONOTONIC, &now);
68 return (now.tv_sec - t0->tv_sec) * 1000 + (now.tv_nsec - t0->tv_nsec) / 1000000;
69}

71/* Dial a connected UDP socket to host:port; -1 on failure. */
72static int
73udp_dial(const char *host, int port)
74{
75 char portstr[16];
76 snprintf(portstr, sizeof(portstr), "%d", port);
77 struct addrinfo hints = { 0 }, *res = NULL;
78 hints.ai_family = AF_UNSPEC;
79 hints.ai_socktype = SOCK_DGRAM;
80 if (getaddrinfo(host, portstr, &hints, &res) != 0 || !res) {
81 return -1;
82 }
83 int fd = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
84 if (fd >= 0 && connect(fd, res->ai_addr, res->ai_addrlen) != 0) {
85 close(fd);
86 fd = -1;
87 }
88 freeaddrinfo(res);
89 return fd;
90}

92/* Find (or create) the flow for client `sa`; NULL when the table is full. */
93static fp_udp_flow *
94udp_flow_for(fp_udp_flow *flows, const struct sockaddr_storage *sa, socklen_t sl,
95 const fp_udp_cfg *cfg, int *created)
96{
97 int free_slot = -1;
98 for (int i = 0; i < FP_UDP_MAX_FLOWS; i++) {
99 if (flows[i].ufd < 0) {
100 if (free_slot < 0) { free_slot = i; }
101 continue;
102 }
103 if (flows[i].clilen == sl && memcmp(&flows[i].cli, sa, sl) == 0) {
104 *created = 0;
105 return &flows[i];
106 }
107 }
108 if (free_slot < 0) {
109 return NULL;
110 }
111 int ufd = udp_dial(cfg->host, cfg->port);
112 if (ufd < 0) {
113 return NULL;
114 }
115 fp_udp_flow *f = &flows[free_slot];
116 memcpy(&f->cli, sa, sl);
117 f->clilen = sl;
118 f->ufd = ufd;
119 clock_gettime(CLOCK_MONOTONIC, &f->last);
120 *created = 1;
121 return f;
122}

124/* Reap flows idle beyond g_udp_reap_ms (a UDP flow-map timeout far shorter than
125 * TCP — long-lived UDP monitoring/streams get evicted early). */
126static void
127udp_reap_idle(fp_udp_flow *flows)
128{
129 if (g_udp_reap_ms <= 0) {
130 return;
131 }
132 for (int i = 0; i < FP_UDP_MAX_FLOWS; i++) {
133 if (flows[i].ufd >= 0 && udp_elapsed_ms(&flows[i].last) >= g_udp_reap_ms) {
134 close(flows[i].ufd);
135 flows[i].ufd = -1;
136 CBUMP(udp_reaped, 1);
137 }
138 }
139}

141/* True (and counts) when this datagram should be dropped per udp-drop. */
142static int
143udp_should_drop(unsigned *seed)
144{
145 if (g_udp_drop_ppm > 0 &&
146 (unsigned) (rand_r(seed) % 1000000) < (unsigned) g_udp_drop_ppm) {
147 CBUMP(udp_dropped, 1);
148 return 1;
149 }
150 return 0;
151}

153/* Apply the timing levers before forwarding a datagram: first-datagram hold and
154 * probabilistic reorder hold-back. */
155static void
156udp_apply_timing(int is_first, unsigned *seed)
157{
158 if (is_first && g_udp_hold_ms > 0) {
159 usleep((useconds_t) g_udp_hold_ms * 1000);
160 CBUMP(udp_held, 1);
161 }
162 if (g_udp_reorder_ppm > 0 && g_udp_reorder_ms > 0 &&
163 (unsigned) (rand_r(seed) % 1000000) < (unsigned) g_udp_reorder_ppm) {
164 usleep((useconds_t) g_udp_reorder_ms * 1000);
165 }
166}

168/* Build and bind the UDP listener on the vetted bind template (port only). */
169static int
170udp_listen(int port)
171{
172 struct sockaddr_storage ss = g_bind_ss;
173 if (ss.ss_family == AF_INET) {
174 ((struct sockaddr_in *) &ss)->sin_port = htons((uint16_t) port);
175 } else {
176 ((struct sockaddr_in6 *) &ss)->sin6_port = htons((uint16_t) port);
177 }
178 int fd = socket(ss.ss_family, SOCK_DGRAM, 0);
179 if (fd < 0) {
180 return -1;
181 }
182 int one = 1;
183 setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &one, sizeof(one));
184 if (bind(fd, (struct sockaddr *) &ss, g_bind_len) != 0) {
185 close(fd);
186 return -1;
187 }
188 return fd;
189}

191/* Upstream -> client (response path): forward each ready flow's datagram back
192 * to its client, applying the drop/reorder timing levers. */
193static void
194udp_pump_upstream(struct pollfd *pf, int nf, const int *idx, fp_udp_flow *flows,
195 int lfd, unsigned char *dg, unsigned *seed)
196{
197 for (int j = 1; j < nf; j++) {
198 if (!(pf[j].revents & POLLIN)) {
199 continue;
200 }
201 fp_udp_flow *f = &flows[idx[j]];
202 ssize_t n = recv(f->ufd, dg, FP_UDP_DGRAM, 0);
203 if (n <= 0) {
204 continue;
205 }
206 clock_gettime(CLOCK_MONOTONIC, &f->last);
207 if (udp_should_drop(seed)) {
208 continue;
209 }
210 udp_apply_timing(0, seed);
211 sendto(lfd, dg, (size_t) n, 0, (struct sockaddr *) &f->cli, f->clilen);
212 CBUMP(udp_out, 1);
213 }
214}

216/* Client -> upstream (request path): relay one inbound datagram, creating its
217 * flow on demand and applying the first-datagram hold / drop levers. */
218static void
219udp_pump_client(struct pollfd *pf, fp_udp_flow *flows, int lfd, unsigned char *dg,
220 const fp_udp_cfg *cfg, unsigned *seed)
221{
222 if (!(pf[0].revents & POLLIN)) {
223 return;
224 }
225 struct sockaddr_storage sa;
226 socklen_t sl = sizeof(sa);
227 ssize_t n = recvfrom(lfd, dg, FP_UDP_DGRAM, 0, (struct sockaddr *) &sa, &sl);
228 if (n <= 0) {
229 return;
230 }
231 CBUMP(udp_in, 1);
232 if (udp_should_drop(seed)) {
233 return;
234 }
235 int created = 0;
236 fp_udp_flow *f = udp_flow_for(flows, &sa, sl, cfg, &created);
237 if (!f) {
238 return; /* table full or dial failed — drop */
239 }
240 clock_gettime(CLOCK_MONOTONIC, &f->last);
241 udp_apply_timing(created, seed);
242 send(f->ufd, dg, (size_t) n, 0);
243}

245/* Build the pollfd set: the listener at [0], every live flow's upstream after
246 * it (with idx[k] mapping pollfd slot k back to its flow). Returns the count. */
247static int
248udp_build_pollset(struct pollfd *pf, int *idx, fp_udp_flow *flows, int lfd)
249{
250 pf[0].fd = lfd; pf[0].events = POLLIN; pf[0].revents = 0;
251 int nf = 1;
252 for (int i = 0; i < FP_UDP_MAX_FLOWS; i++) {
253 if (flows[i].ufd >= 0) {
254 pf[nf].fd = flows[i].ufd; pf[nf].events = POLLIN; pf[nf].revents = 0;
255 idx[nf] = i;
256 nf++;
257 }
258 }
259 return nf;
260}

262void *
263fp_udp_thread(void *arg)
264{
265 fp_udp_cfg *cfg = arg;
266 fp_udp_flow flows[FP_UDP_MAX_FLOWS];
267 unsigned seed = g_seed ? g_seed : 0x1234567u;
268 unsigned char dg[FP_UDP_DGRAM];
269
270 for (int i = 0; i < FP_UDP_MAX_FLOWS; i++) {
271 flows[i].ufd = -1;
272 }
273 int lfd = udp_listen(cfg->listen_port);
274 if (lfd < 0) {
275 fprintf(stderr, "brix-fault-proxy: udp bind failed on port %d\n",
276 cfg->listen_port);
277 free(cfg);
278 return NULL;
279 }
280
281 struct pollfd pf[1 + FP_UDP_MAX_FLOWS];
282 int idx[1 + FP_UDP_MAX_FLOWS];
283 for (;;) {
284 udp_reap_idle(flows);
285 int nf = udp_build_pollset(pf, idx, flows, lfd);
286 int pr = poll(pf, (nfds_t) nf, 200);
287 if (pr < 0) {
288 if (errno == EINTR) { continue; }
289 break;
290 }
291 if (pr == 0) {
292 continue;
293 }
294 udp_pump_upstream(pf, nf, idx, flows, lfd, dg, &seed);
295 udp_pump_client(pf, flows, lfd, dg, cfg, &seed);
296 }
297 close(lfd);
298 free(cfg);
299 return NULL;
300}
