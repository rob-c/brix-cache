17/* ---- frame.c ---- */
18/* One outbound request payload: the bytes written after the 24-byte header.
19 * A NULL brix_payload argument means "no payload" ({NULL, 0}). */
20typedef struct {
21 const void *data; /* payload bytes (may be NULL when len == 0) */
22 uint32_t len; /* payload byte count (brix_send frames dlen == len) */
23} brix_payload;
24
25/* brix_send_ext payload: `dlen` — the wire dlen written into hdr[20..23] (also
26 * the sigver-signed span) — may be smaller than `len`, the payload bytes
27 * actually written. Needed by kXR_writev, whose dlen frames only the 16-byte
28 * descriptor block while the segment data streams after the frame (stock
29 * XrdXrootdProtocol::do_WriteV). brix_send == brix_send_ext with dlen == len. */
30typedef struct {
31 const void *data; /* payload bytes (may be NULL when len == 0) */
32 uint32_t len; /* payload bytes actually written after the header */
33 uint32_t dlen; /* wire dlen framed in the header */
34} brix_payload_ext;
35
36/* Caller-provided response out-params. The struct pointer itself must be
37 * non-NULL; any field may be NULL to decline that value. *body receives a
38 * malloc'd buffer the caller frees (NULL when the reply has no body). */
39typedef struct {
40 uint16_t *status;
41 uint8_t **body;
42 uint32_t *blen;
43} brix_resp_out;
44
45/* Assign a fresh streamid into hdr[0..1] and write dlen into hdr[20..23] (the
46 * caller has already filled requestid + the 16-byte body). Sends header+payload. */
47int brix_send(brix_conn *c, void *hdr24, const brix_payload *pl,
48 uint16_t *out_sid, brix_status *st);
49/* As brix_send, but with independent wire dlen (see brix_payload_ext). */
50int brix_send_ext(brix_conn *c, void *hdr24, const brix_payload_ext *pl,
51 uint16_t *out_sid, brix_status *st);
52/* Read one response frame for streamid want_sid. Returns 0 with *out->status set
53 * and a malloc'd *out->body / *out->blen (caller frees) for kXR_ok/oksofar/authmore
54 * AND kXR_redirect/kXR_wait (so the roundtrip wrapper can act on them). Returns
55 * -1 on kXR_error (st filled from errnum+errmsg) or any other status / transport
56 * fault. */
57int brix_recv(brix_conn *c, uint16_t want_sid, brix_resp_out *out,
58 brix_status *st);
59
60/* Send a request and read its reply, transparently following kXR_redirect
61 * (reconnect+replay, bounded by XRDC_REDIR_MAX + a visited-set loop guard) and
62 * honoring kXR_wait (sleep+resend). Use this for path-based ops so cluster
63 * redirectors work. hdr24 is re-stamped (streamid/dlen) on each attempt.
64 * out->body and out->blen must be non-NULL here. Returns 0 with *out->status =
65 * kXR_ok/oksofar + body/blen; -1 (st set) on error. */
66int brix_roundtrip(brix_conn *c, void *hdr24, const brix_payload *pl,
67 brix_resp_out *out, brix_status *st);

69/* ---- conn.c ---- */
70/* connect → handshake → [TLS upgrade] → kXR_protocol → kXR_login → [auth].
71 * opts may be NULL (anonymous, no TLS). */
72int brix_connect(brix_conn *c, const brix_url *u, const brix_opts *o,
73 brix_status *st);
74/* Tear down the current transport and re-establish the full session (handshake →
75 * [TLS] → login → auth) against host:port, preserving the stored opts/creds. Used
76 * by the redirect follower. 0 / -1. */
77int brix_reconnect(brix_conn *c, const char *host, int port, brix_status *st);
78void brix_close(brix_conn *c);

80/* ---- pool.c — thread-safe pool of connections for concurrent callers ---- */
81/* An brix_conn is one-request-in-flight and NOT thread-safe; a multi-threaded
82 * consumer (e.g. the FUSE driver) checks out an independent connected conn per
83 * operation. The struct is opaque; callers hold only the handle. */
84typedef struct brix_pool brix_pool;
85/* Create a pool of `n` connections to `u` (opts `o`, may be NULL). Connects one
86 * eagerly so a bad endpoint/auth fails up front; the rest connect on demand.
87 * Returns NULL + sets st on failure. */
88brix_pool *brix_pool_create(const brix_url *u, const brix_opts *o, int n,
89 brix_status *st);
90/* Borrow a connected conn, blocking until one is free; reconnects a dropped slot
91 * transparently. Returns NULL + sets st only if (re)connect fails. */
92brix_conn *brix_pool_checkout(brix_pool *p, brix_status *st);
93/* Return a checked-out conn. healthy==0 (the op hit a connection-level error,
94 * i.e. st->kxr == XRDC_ESOCK/XRDC_EPROTO) drops the conn so the next checkout
95 * reconnects on a clean session. */
96void brix_pool_checkin(brix_pool *p, brix_conn *c, int healthy);
97void brix_pool_destroy(brix_pool *p);
98/* Establish a secondary data stream bound to `primary`'s session: handshake +
99 * kXR_protocol [+ TLS] then kXR_bind{primary->sessid}, skipping kXR_login (the
100 * server inherits identity from the primary). `sec` is fully initialised here.
101 * Tear it down with brix_streams_close (no endsess). 0 / -1. */
102int brix_bind(brix_conn *sec, const brix_conn *primary, brix_status *st);
103
104/* ---- streams.c (M8 parallel streams) ---- */
105#define XRDC_MAX_STREAMS 16
106typedef struct {
107 int n; /* secondaries actually bound */
108 brix_conn sec[XRDC_MAX_STREAMS - 1];
109} brix_streamset;
110/* Best-effort: bind up to (streams-1) secondaries to `primary` (capped at
111 * XRDC_MAX_STREAMS-1). Never fails the caller — returns the number bound; a
112 * secondary that won't bind is simply skipped. */
113int brix_streams_open(brix_streamset *ss, brix_conn *primary, int streams,
114 brix_status *st);
115void brix_streams_close(brix_streamset *ss);
