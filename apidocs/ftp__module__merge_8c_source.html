22/* brix_ftp_install_backend_credential — bind the named brix_credential block's
23 * upstream identity (SigV4 keys for an s3:// backend) to this export's registry
24 * entry, so per-request ctx resolution builds a signed backend instance. Mirrors
25 * the root/webdav wiring via the ONE shared credential→backend_cred mapper
26 * (P80.1). A no-op when no credential is named; a hard error on an unknown name
27 * or an unreadable token_file (a misconfigured upstream must fail the config, not
28 * serve unauthenticated). */
29static char *
30brix_ftp_install_backend_credential(ngx_conf_t *cf,
31 ngx_stream_brix_ftp_srv_conf_t *conf)
32{
33 char name[256];
34 char bearer[4096];
35 const brix_credential_t *cred;
36 brix_vfs_backend_cred_t bcred;
37
38 if (conf->common.storage_credential.len == 0) {
39 return NGX_CONF_OK;
40 }
41 ngx_cpystrn((u_char *) name, conf->common.storage_credential.data,
42 ngx_min(conf->common.storage_credential.len + 1, sizeof(name)));
43 cred = brix_credential_lookup(name);
44 if (cred == NULL) {
45 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
46 "brix_storage_credential \"%s\" names no brix_credential "
47 "block", name);
48 return NGX_CONF_ERROR;
49 }
50 if (brix_credential_to_backend_cred(cred, bearer, sizeof(bearer),
51 &bcred, cf->log) != NGX_OK)
52 {
53 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
54 "brix_gridftp: cannot derive credential \"%s\" for export \"%s\"",
55 name, conf->common.root_canon);
56 return NGX_CONF_ERROR;
57 }
58 brix_vfs_backend_set_credential(conf->common.root_canon, &bcred);
59
60 /* An explicit `mode` on the block overrides the gateway's default per-request
61 * delegation mode (e.g. `mode select` pins the service credential and never
62 * forwards the client's proxy). */
63 if (cred->mode != NGX_CONF_UNSET) {
64 conf->deleg_mode = (enum brix_cred_mode) cred->mode;
65 }
66 return NGX_CONF_OK;
67}

98/* Register this export's storage backend so every per-request ftp_vfs_ctx()
99 * (which calls brix_vfs_ctx_init → brix_vfs_backend_resolve on root_canon)
100 * routes through the selected driver — "pblock" today, POSIX when unset.
101 * The gateway itself only ever touches storage through brix_vfs_*, so no
102 * data-path change is needed once the choice is on the registry. */
103static char *
104brix_ftp_merge_storage(ngx_conf_t *cf, ngx_stream_brix_ftp_srv_conf_t *conf)
105{
106 if (!conf->enable || conf->common.root_canon[0] == '\0') {
107 return NGX_CONF_OK;
108 }
109 if (brix_vfs_backend_config_str(cf, conf->common.root_canon,
110 &conf->common.storage_backend, 0, BRIX_AF_AUTO) != NGX_OK)
111 {
112 return NGX_CONF_ERROR;
113 }
114 /* Default: forward the client's delegated proxy (PASSTHROUGH). A named
115 * brix_credential block's `mode` may override this inside install below;
116 * the request-time bind (ftp_ev_path.c) additionally no-ops on backends
117 * that do not consume a proxy, so posix/pblock exports are unaffected. */
118 conf->deleg_mode = BRIX_CRED_PASSTHROUGH;
119 return brix_ftp_install_backend_credential(cf, conf);
120}

144static char *
145ftp_resolve_root(ngx_conf_t *cf, ngx_stream_brix_ftp_srv_conf_t *conf)
146{
147 ngx_str_t dir;
148 char raw[PATH_MAX];
149
150 if (conf->common.root.len == 0) {
151 return NGX_CONF_OK;
152 }
153 dir = conf->common.root;
154 if (ngx_conf_full_name(cf->cycle, &dir, 1) != NGX_OK) {
155 return NGX_CONF_ERROR;
156 }
157 if (dir.len >= sizeof(raw)) {
158 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
159 "brix_export path too long: %V", &dir);
160 return NGX_CONF_ERROR;
161 }
162 ngx_memcpy(raw, dir.data, dir.len);
163 raw[dir.len] = '\0';
164 if (realpath(raw, conf->common.root_canon) == NULL) {
165 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
166 "brix_export \"%s\" cannot be resolved (does it exist?)", raw);
167 return NGX_CONF_ERROR;
168 }
169 conf->common.root = dir;
170 return NGX_CONF_OK;
171}
