2/*
3 * common.c — shared helpers for write-side opcode handlers.
4 *
5 * Provides brix_try_post_write_aio() — AIO task setup and thread-pool
6 * dispatch for kXR_write / kXR_pgwrite syscalls — and the two §6 CNS emit
7 * wrappers whose only difference from the bare brix_cns_emit seam is that they
8 * must first observe the mutated object (size / mtime / dir-ness) through the
9 * VFS. They live here rather than in net/cms because the observation needs a
10 * request-identity VFS context, which is root-plane knowledge.
11 *
12 * Note: path-based write opcodes (mkdir, mv, rmdir, truncate) perform auth via
13 * brix_auth_gate() directly; chmod/rm dispatch through the op-descriptor
14 * table in op_table.c. The former shared resolver
15 * (brix_write_resolve_existing_path) was retired once those callers migrated
16 * and has been removed.
17 */

24/*
25 * root_cns_probe — stat one mutated object with the requesting user's identity.
26 *
27 * WHAT: Fills *st for `path` through the VFS seam, bound exactly as every other
28 * namespace op in this plane binds (identity + backend credential +
29 * delegation), so a remote-backed export is probed as the CLIENT.
30 * WHY: A CNS event carries size/mtime, and for a rename also the destination's
31 * dir-ness. Reading them with a raw stat() would both bypass the VFS seam
32 * (INVARIANT 12) and read the wrong thing on a non-POSIX backend.
33 * HOW: ctx_init + bind_backend_cred + bind_deleg + brix_vfs_probe, no-follow so
34 * a symlink is reported as itself. Returns NGX_OK on a hit.
35 */
36static ngx_int_t
37root_cns_probe(brix_ctx_t *ctx, ngx_connection_t *c,
38 ngx_stream_brix_srv_conf_t *conf, const char *path, brix_vfs_stat_t *st)
39{
40 brix_vfs_ctx_t vctx;
41
42 brix_vfs_ctx_init(&vctx, c->pool, c->log, BRIX_PROTO_ROOT,
43 conf->common.root_canon, NULL,
44 brix_vfs_policy_from_write_enable(conf->common.allow_write),
45 0 /* is_tls */, ctx->identity, path);
46 brix_vfs_ctx_bind_backend_cred(&vctx,
47 &conf->common.storage_credential_dir,
48 conf->common.storage_credential_fallback);
49 brix_root_vfs_bind_session(ctx, conf, &vctx);
50
51 return brix_vfs_probe(&vctx, 1 /* no-follow */, st);
52}

54void
55brix_root_cns_emit_moved(brix_ctx_t *ctx, ngx_connection_t *c,
56 ngx_stream_brix_srv_conf_t *conf, const char *src_resolved,
57 const char *dst_resolved)
58{
59 brix_vfs_stat_t st;
60
61 /* Gate BEFORE the probe: the probe is a real syscall, and CNS is off on
62 * every node that is not a federation data server. */
63 if (conf->cns_mode != BRIX_CNS_EMIT) {
64 return;
65 }
66 if (root_cns_probe(ctx, c, conf, dst_resolved, &st) != NGX_OK) {
67 /* The rename succeeded but the destination cannot be observed (a racing
68 * unlink, a backend that lost it). Emitting a rename whose metadata we
69 * had to invent would seed the manager with a wrong size, so drop the
70 * source entry instead and let the next ADD re-establish the truth. */
71 brix_cns_emit(conf, BRIX_CNS_DEL, src_resolved, 0, 0);
72 return;
73 }
74 brix_cns_emit_rename(conf, src_resolved, dst_resolved,
75 (uint64_t) st.size, (uint64_t) st.mtime,
76 st.is_directory);
77}

96brix_try_post_write_aio(brix_ctx_t *ctx, ngx_connection_t *c, int idx,
97 off_t offset, const u_char *data, size_t len, int64_t req_offset,
98 ngx_uint_t is_pgwrite, u_char *payload_to_free,
99 const xrdp_pg_bad_t *bad, size_t bad_count, const char *fallback_log,
100 ngx_flag_t *posted)
101{
102 ngx_stream_brix_srv_conf_t *conf;
103 ngx_thread_task_t *task;
104 brix_write_aio_t *t;
105
106 *posted = 0;
107
108 conf = ngx_stream_get_module_srv_conf((ngx_stream_session_t *) (c->data),
109 ngx_stream_brix_module);
110 if (conf->common.thread_pool == NULL) {
111 return NGX_OK;
112 }
113
114 task = ngx_thread_task_alloc(c->pool, sizeof(brix_write_aio_t));
115 if (task == NULL) {
116 return NGX_ERROR;
117 }
118
119 t = task->ctx;
120 t->c = c;
121 t->ctx = ctx;
122 t->conf = conf;
123 t->fd = ctx->files[idx].fd;
124 t->handle_idx = idx;
125 t->offset = offset;
126 t->data = data;
127 t->len = len;
128 t->req_offset = req_offset;
129 t->is_pgwrite = is_pgwrite;
130 t->nwritten = -1;
131 t->io_errno = 0;
132 t->csi = ctx->files[idx].csi; /* phase-59 W2: tag update */
133 t->obj = ctx->files[idx].sd_obj; /* Layer 3: driver obj or zeroed */
134 t->payload_to_free = payload_to_free;
135 t->bad_page_count = (bad != NULL && bad_count > kXR_pgMaxEpr)
136 ? kXR_pgMaxEpr : bad_count;
137 if (bad != NULL && t->bad_page_count > 0) {
138 ngx_memcpy(t->bad_pages, bad,
139 t->bad_page_count * sizeof(xrdp_pg_bad_t));
140 }
141 t->streamid[0] = ctx->recv.cur_streamid[0];
142 t->streamid[1] = ctx->recv.cur_streamid[1];
143 ngx_cpystrn((u_char *) t->path,
144 (u_char *) (ctx->files[idx].path != NULL
145 ? ctx->files[idx].path : "-"),
146 sizeof(t->path));
147 t->start_ns = brix_phase_now_ns(); /* phase-56 D-2 */
148
149 brix_task_bind(task, brix_write_aio_thread, brix_write_aio_done);
150
151 return brix_aio_post_task(ctx, c, conf->common.thread_pool, task, fallback_log,
152 posted);
153}
154/* WHY: Provides uniform thread-pool dispatch for write syscalls, enabling parallel disk I/O without blocking the main event loop during large file transfers. Detaches payload from ctx->recv.payload_buf so the main thread can safely read next request headers while write happens in worker threads. The posted flag enables callers to distinguish between dispatched and fallback cases — dispatched=1 means completion callback handles response; dispatched=0 means caller must perform synchronous pwrite. */
155/* HOW: Sets *posted=0 initially; retrieves conf via ngx_stream_get_module_srv_conf(); returns NGX_OK if thread_pool==NULL (no AIO configured). Allocates task struct with ngx_thread_task_alloc() — if OOM returns NGX_ERROR. Populates t=brix_write_aio_t context: c, ctx, conf, fd from files[idx], handle_idx, offset, data, len, req_offset, is_pgwrite, nwritten=-1, io_errno=0, payload_to_free, streamid copy, path copy via ngx_cpystrn(). Binds the worker + done callbacks via brix_task_bind(task, brix_write_aio_thread, brix_write_aio_done). Calls brix_aio_post_task() which sets posted=1 on success or 0 if queue full. Returns result from post_task call. */
