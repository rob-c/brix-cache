8/* WHAT: Send an exact readv payload outside the typed client's safety checks.
9 * WHY: Over-limit and malformed vectors must reach the remote implementation.
10 * HOW: Use the repository wire codec and the shared bounded exchange seam. */
11static int
12raw_readv(xrd_conform_execution *execution, brix_conn *connection,
13 const void *payload_data, uint32_t payload_length, size_t expected_bytes,
14 uint16_t *wire_status, uint8_t **body, uint32_t *body_length,
15 brix_status *status)
16{
17 ClientRequestHdr request;
18 brix_payload payload = { payload_data, payload_length };
19 brix_resp_out response = { wire_status, body, body_length };
20 uint16_t stream_id;
21 char reason[64];
22
23 if (!xrd_conform_reserve(execution, expected_bytes, reason,
24 sizeof(reason))) {
25 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
26 return -1;
27 }

30 xrdw_empty_req_pack(request.body);

33 return -1;

37/* WHAT: Inspect one successful response segment without typed normalization.
38 * WHY: RDV-002 requires evidence that rlen, offset and handle are echoed.
39 * HOW: Send one byte and parse the exact 16-byte response descriptor. */
40static void
41probe_echoed_descriptor(xrd_conform_execution *execution,
42 brix_conn *connection, brix_file *file, const uint8_t *baseline,
43 uint64_t size)
44{
45 uint8_t request_segment[BRIX_READV_SEGSIZE];
46 uint8_t *body = NULL;
47 uint32_t body_length = 0;
48 uint16_t wire_status = 0;
49 brix_status status;
50 uint64_t offset = size > 1 ? 1 : 0;
51 int rc, ok;
52
53 if (size == 0) return;
54 brix_readv_seg_pack(request_segment, file->fhandle, 1, offset);
55 rc = raw_readv(execution, connection, request_segment,
56 sizeof(request_segment), 1, &wire_status, &body, &body_length,
57 &status);

60 && memcmp(body, file->fhandle, XRDC_FHANDLE_LEN) == 0
61 && brix_readv_seg_rlen(body) == 1
62 && xrd_get_u64_be(body + 8) == offset
63 && body[BRIX_READV_SEGSIZE] == baseline[offset];
64 xrd_conform_result_set(execution, "RDV-002",
65 ok ? BRIX_CONFORM_VERDICT_CONFORMANT
66 : BRIX_CONFORM_VERDICT_DIVERGENT,
67 ok ? "response-descriptor-echoed-exactly"
68 : "response-descriptor-mismatch",

70 xrd_conform_result_set(execution, "RDV-004",
71 ok ? BRIX_CONFORM_VERDICT_CONFORMANT
72 : BRIX_CONFORM_VERDICT_DIVERGENT,

75 free(body);
76}

291/* WHAT: Read one positive decimal Qconfig limit under the run ceiling.
292 * WHY: RDV-019 compares advertised caps with the exact exercised boundaries.
293 * HOW: Query one key and reject suffixes, zero, overflow and empty replies. */
294static int
295query_limit(xrd_conform_execution *execution, brix_conn *connection,
296 const char *key, size_t *limit)
297{
298 char value[64], reason[64], *end;
299 unsigned long long parsed;
300 brix_status status;
301
302 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return 0;
303 brix_status_clear(&status);
304 if (brix_query(connection, kXR_Qconfig, key, value, sizeof(value),
305 &status) != 0) return 0;
306 errno = 0;
307 parsed = strtoull(value, &end, 10);
308 while (*end == '\r' || *end == '\n' || *end == ' ' || *end == '\t') end++;
309 if (errno != 0 || end == value || *end != '\0' || parsed == 0
310 || parsed > SIZE_MAX) return 0;
311 *limit = (size_t) parsed;
312 return 1;
313}

315/* WHAT: Issue a bounded typed readv and require every returned byte.
316 * WHY: Cap tests must distinguish exact transfer from silent truncation.
317 * HOW: Reserve aggregate bytes, call the API, then compare each segment. */
318static int
319readv_exact(xrd_conform_execution *execution, brix_conn *connection,
320 brix_file *file, brix_readv_seg *segments, size_t count,
321 const uint8_t *baseline, brix_status *status)
322{
323 size_t index, total = 0;
324 char reason[64];
325 ssize_t got;
326
327 for (index = 0; index < count; index++) total += segments[index].len;
328 if (!xrd_conform_reserve(execution, total, reason, sizeof(reason))) return 0;
329 brix_status_clear(status);
330 got = brix_file_readv(connection, file, segments, count, status);
331 if (got != (ssize_t) total) return 0;
332 for (index = 0; index < count; index++) {
333 if (segments[index].got != segments[index].len
334 || memcmp(segments[index].buf,
335 baseline + (size_t) segments[index].offset,
336 segments[index].len) != 0) return 0;
337 }
338 return 1;
339}
