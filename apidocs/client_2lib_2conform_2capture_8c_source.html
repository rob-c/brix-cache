31/* WHAT: Read an exact big-endian u16. WHY: Reject truncated record fields.
32 * HOW: Combine a fixed two-byte buffer. */
33static int
34read_u16(FILE *input, uint16_t *value)
35{
36 uint8_t bytes[2];
37
38 if (fread(bytes, 1, sizeof(bytes), input) != sizeof(bytes)) return -1;
39 *value = (uint16_t) ((uint16_t) bytes[0] << 8) | bytes[1];
40 return 0;
41}

43/* WHAT: Read an exact big-endian u32. WHY: Bound wire record allocation.
44 * HOW: Combine a fixed four-byte buffer. */
45static int
46read_u32(FILE *input, uint32_t *value)
47{
48 uint8_t bytes[4];
49
50 if (fread(bytes, 1, sizeof(bytes), input) != sizeof(bytes)) return -1;
51 *value = ((uint32_t) bytes[0] << 24) | ((uint32_t) bytes[1] << 16)
52 | ((uint32_t) bytes[2] << 8) | bytes[3];
53 return 0;
54}

56/* WHAT: Write the numeric half of a frame record. WHY: Preserve replay format.
57 * HOW: Split sid, code and length into one network-order byte array. */
58static int
59write_numbers(FILE *output, uint16_t sid, uint16_t code, uint32_t length)
60{
61 uint8_t bytes[8] = {
62 (uint8_t) (sid >> 8), (uint8_t) sid,
63 (uint8_t) (code >> 8), (uint8_t) code,
64 (uint8_t) (length >> 24), (uint8_t) (length >> 16),
65 (uint8_t) (length >> 8), (uint8_t) length
66 };
67
68 return fwrite(bytes, 1, sizeof(bytes), output) == sizeof(bytes) ? 0 : -1;
69}

96/* WHAT: Consume exact bytes. WHY: Seeking beyond EOF would accept truncation.
97 * HOW: Read bounded scratch chunks until complete. */
98static int
99discard_bytes(FILE *input, uint32_t length)
100{
101 uint8_t scratch[512];
102
103 while (length > 0) {
104 size_t chunk = length < sizeof(scratch) ? length : sizeof(scratch);
105
106 if (fread(scratch, 1, chunk, input) != chunk) return -1;
107 length -= (uint32_t) chunk;
108 }
109 return 0;
110}

114static int

118 uint16_t value_length;

126/* WHAT: Decode one bounded frame record. WHY: Scrub and validator must share
127 * identical truncation checks. HOW: Read fields, bound length, allocate bytes. */
128static int
129read_frame(FILE *input, capture_frame *frame)
130{
131 memset(frame, 0, sizeof(*frame));
132 frame->direction = fgetc(input);
133 frame->request = fgetc(input);
134 if (frame->direction == EOF || frame->request == EOF) return -1;
135 if (read_u16(input, &frame->sid) != 0) return -1;
136 if (read_u16(input, &frame->code) != 0) return -1;
137 if (read_u32(input, &frame->length) != 0) return -1;
138 if (frame->length > CAP_WIRE_MAX) return -1;
139 frame->wire = malloc(frame->length == 0 ? 1 : frame->length);
140 if (frame->wire == NULL) return -1;
141 if (fread(frame->wire, 1, frame->length, input) != frame->length) {
142 free(frame->wire);
143 frame->wire = NULL;
144 return -1;
145 }
146 return 0;
147}

149/* WHAT: Emit one already-validated frame. WHY: Retained evidence must preserve
150 * exact replay bytes. HOW: Write marker, scalar fields and wire body in order. */
151static int
152write_frame(FILE *output, const capture_frame *frame)
153{
154 if (fputc('F', output) == EOF) return -1;
155 if (fputc(frame->direction, output) == EOF) return -1;
156 if (fputc(frame->request, output) == EOF) return -1;
157 if (write_numbers(output, frame->sid, frame->code, frame->length) != 0)
158 return -1;
159 return fwrite(frame->wire, 1, frame->length, output) == frame->length
160 ? 0 : -1;
161}

163/* WHAT: Validate and conditionally retain one frame. WHY: Operation frames are
164 * replayable evidence but session frames are secrets. HOW: Track sid and copy. */
165static int
166scrub_frame(FILE *input, FILE *output, sensitive_streams *streams)
167{
168 capture_frame frame;
169 int rc = 0;
170
171 if (read_frame(input, &frame) != 0) return -1;
172 if (frame.request == 1 && frame.direction == '>'
173 && sensitive_opcode(frame.code)) stream_mark(streams, frame.sid);
174 if (!stream_sensitive(streams, frame.sid)) rc = write_frame(output, &frame);
175 free(frame.wire);
176 return rc;
177}

194/* WHAT: Scrub a complete XRDCAP1 stream. WHY: Never publish partial or
195 * credential-bearing evidence. HOW: Verify magic and dispatch every record. */
196static int
197scrub_stream(FILE *input, FILE *output)
198{
199 char magic[CAP_MAGIC_SIZE];
200 sensitive_streams streams;
201 int type;
202
203 memset(&streams, 0, sizeof(streams));
204 if (fread(magic, 1, sizeof(magic), input) != sizeof(magic)
205 || memcmp(magic, CAP_MAGIC, sizeof(magic)) != 0
206 || fwrite(magic, 1, sizeof(magic), output) != sizeof(magic)) return -1;
207 while ((type = fgetc(input)) != EOF) {
208 if (type == 'M') {
209 if (discard_meta(input) != 0) return -1;
210 } else if (type == 'F') {
211 if (scrub_frame(input, output, &streams) != 0) return -1;
212 } else {
213 return -1;
214 }
215 }
216 return ferror(input) || fflush(output) != 0 ? -1 : 0;
217}

219/* WHAT: Scrub one clause artifact beneath a dirfd. WHY: Refuse symlink and

221static int
222artifact_one(int dirfd, const char *raw_capture, const char *clause_id)
223{
224 char name[64];
225 FILE *input, *output;
226 int output_fd;
227 int failed = 0;
228 int failure_errno = 0;
229
230 if ((size_t) snprintf(name, sizeof(name), "%s.xrdcap", clause_id)
231 >= sizeof(name)) return -1;
232 input = fopen(raw_capture, "rb");
233 if (input == NULL) return -1;
234 output_fd = openat(dirfd, name,
235 O_WRONLY | O_CREAT | O_EXCL | O_NOFOLLOW, 0600);
236 if (output_fd < 0) {
237 fclose(input);
238 return -1;
239 }
240 output = fdopen(output_fd, "wb");
241 if (output == NULL) {
242 close(output_fd);
243 fclose(input);
244 return -1;
245 }
246 if (scrub_stream(input, output) != 0) {
247 failure_errno = EBADMSG;
248 failed = 1;
249 } else if (fsync(fileno(output)) != 0) {
250 failure_errno = errno;
251 failed = 1;
252 }
253 if (fclose(input) != 0) {
254 if (!failed) failure_errno = errno;
255 failed = 1;
256 }
257 if (fclose(output) != 0) {
258 if (!failed) failure_errno = errno;
259 failed = 1;
260 }
261 if (failed) unlinkat(dirfd, name, 0);
262 if (failed) errno = failure_errno == 0 ? EIO : failure_errno;
263 return failed ? -1 : 0;
264}

302static int
303write_reproducer_stream(FILE *output,
304 const brix_conform_reproducer_frame *frames, size_t frame_count)
305{
306 size_t index;
307
308 if (fwrite(CAP_MAGIC, 1, CAP_MAGIC_SIZE, output) != CAP_MAGIC_SIZE)
309 return -1;
310 for (index = 0; index < frame_count; index++) {
311 capture_frame frame;
312
313 if (frames[index].wire == NULL || frames[index].wire_length == 0
314 || frames[index].wire_length > CAP_WIRE_MAX
315 || sensitive_opcode(frames[index].code)) return -1;
316 frame.direction = '>';
317 frame.request = 1;
318 frame.sid = frames[index].sid;
319 frame.code = frames[index].code;
320 frame.length = frames[index].wire_length;
321 frame.wire = (uint8_t *) frames[index].wire;
322 if (write_frame(output, &frame) != 0) return -1;
323 }
324 return fflush(output) == 0 && fsync(fileno(output)) == 0 ? 0 : -1;
325}
