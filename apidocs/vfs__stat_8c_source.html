3 *
4 * WHAT: Implements brix_vfs_stat(), which lstat()s the resolved ctx path and

24/* ---- phase-56 C-2: per-worker bounded negative-stat cache ----------------
25 *
26 * Collapses repeat ENOENT probes of the same absent path (xcache fill races,
27 * "does this exist?" namespace browsing) to one syscall per TTL window.
28 * Default OFF; BRIX_NEG_STAT_CACHE=1 in the worker environment enables it.
29 *
30 * Safety envelope (§13 of the phase-56 doc — a negative cache can turn a real
31 * file into a false ENOENT, so every mitigation is mandatory):
32 * - per-worker only, never shared cross-worker (plain statics, no SHM);
33 * - short TTL (1 s) bounds staleness from cross-worker creates;
34 * - only a genuine ENOENT is inserted — never EACCES/EIO, never positives
35 * (they carry size/mtime that must be live);
36 * - every same-worker create/mkdir/rename/staged-commit clears the target's
37 * slot via brix_vfs_neg_stat_forget();
38 * - fully disabled while impersonation MAP mode is wired (an ENOENT for one
39 * mapped identity may be an EACCES mask for another);
40 * - the lstat (nofollow) and stat (follow) arms are keyed SEPARATELY: a
41 * follow-stat ENOENT can be a dangling symlink whose lstat legitimately
42 * succeeds, so the two must never share a cached negative — but each arm's
43 * own repeat probes are safely collapsible (the wire's default kXR_stat is
44 * the follow arm, so caching only nofollow would miss the probe storms
45 * this exists for). Forget clears BOTH arms.
46 */
47#define BRIX_NEG_STAT_SLOTS 256 /* power of two */
48#define BRIX_NEG_STAT_TTL ((ngx_msec_t) 1000)
49
50typedef struct {
51 uint64_t hash;
52 ngx_msec_t expire;
53} brix_neg_stat_ent_t;

55static brix_neg_stat_ent_t brix_neg_stat[BRIX_NEG_STAT_SLOTS];

57/* Env knob, resolved once per worker. Identity scoping is checked separately
58 * (per call) because MAP mode wires up lazily on first broker contact. */
59static int
60brix_neg_stat_env_on(void)
61{
62 static int on = -1;
63
64 if (on < 0) {
65 const char *env = getenv("BRIX_NEG_STAT_CACHE");
66
67 on = (env != NULL && env[0] == '1' && env[1] == '\0');
68 }
69 return on;
70}

72/* key = FNV1a(root_canon) ⊕ FNV1a(resolved_path), salted per stat arm so a
73 * follow negative and a nofollow negative never alias (dangling-symlink
74 * hazard). Canonical basis/prime from core/fnv.h (NOT the frozen
75 * pblock/rados variants). */
76#define BRIX_NEG_STAT_FOLLOW_SALT UINT64_C(0x9e3779b97f4a7c15)
77
78static uint64_t
79brix_neg_stat_fnv(const char *s)
80{
81 uint64_t h = BRIX_FNV1A64_OFFSET_BASIS;
82 const u_char *p;
83
84 if (s == NULL) {
85 return 0;
86 }
87 for (p = (const u_char *) s; *p != '\0'; p++) {
88 h = (h ^ *p) * BRIX_FNV1A64_PRIME;
89 }
90 return h;
91}

93static uint64_t
94brix_neg_stat_key(const char *root_canon, const char *path, int nofollow)
95{
96 uint64_t key = brix_neg_stat_fnv(root_canon) ^ brix_neg_stat_fnv(path);
97
98 return nofollow ? key : key ^ BRIX_NEG_STAT_FOLLOW_SALT;
99}

101/* 1 iff a live cached ENOENT covers (root_canon, path) on this stat arm —
102 * lookup and insert are both no-ops under MAP-mode impersonation or with the
103 * knob off. */
104static int
105brix_neg_stat_lookup(const char *root_canon, const char *path, int nofollow)
106{
107 uint64_t key;
108 brix_neg_stat_ent_t *ent;
109
110 if (!brix_neg_stat_env_on() || brix_imp_enabled()
111 || root_canon == NULL || path == NULL)
112 {
113 return 0;
114 }
115
116 key = brix_neg_stat_key(root_canon, path, nofollow);
117 ent = &brix_neg_stat[key & (BRIX_NEG_STAT_SLOTS - 1)];
118 return ent->hash == key && ngx_current_msec < ent->expire;
119}

121static void
122brix_neg_stat_insert(const char *root_canon, const char *path, int nofollow)
123{
124 uint64_t key;
125 brix_neg_stat_ent_t *ent;
126
127 if (!brix_neg_stat_env_on() || brix_imp_enabled()
128 || root_canon == NULL || path == NULL)
129 {
130 return;
131 }
132
133 key = brix_neg_stat_key(root_canon, path, nofollow);
134 ent = &brix_neg_stat[key & (BRIX_NEG_STAT_SLOTS - 1)];
135 ent->hash = key;

139/* Drop any cached negative for (root_canon, path) on BOTH stat arms. Called by
140 * every same-worker mutator that can materialise the path (create-open, mkdir,
141 * rename target, staged commit). Deliberately NOT gated on impersonation: an
142 * impersonated create must still clear an entry inserted by an earlier
143 * non-impersonated probe of the same worker. */
144void
145brix_vfs_neg_stat_forget(const char *root_canon, const char *path)
146{
147 int arm;
148 uint64_t key;
149 brix_neg_stat_ent_t *ent;
150
151 if (!brix_neg_stat_env_on() || root_canon == NULL || path == NULL) {
152 return;
153 }
154
155 for (arm = 0; arm <= 1; arm++) {
156 key = brix_neg_stat_key(root_canon, path, arm);
157 ent = &brix_neg_stat[key & (BRIX_NEG_STAT_SLOTS - 1)];
158 if (ent->hash == key) {
159 ent->hash = 0;
160 ent->expire = 0;
161 }
162 }
163}

165/*
166 * Shared confined-stat body for both the lstat (no-follow) and stat (follow)
167 * entrypoints. nofollow!=0 reports a trailing symlink as itself; nofollow=0
168 * follows it chroot-style within the export. Both run AS THE MAPPED USER under
169 * impersonation (broker-routed) — otherwise a metadata op (WebDAV
170 * HEAD/DELETE/PROPFIND-on-file, kXR_stat/statx, S3 HEAD) whose target sits
171 * inside a directory the unprivileged worker cannot traverse (a 0700 user-owned
172 * subdir, or a group-restricted 0770 dir the mapped user reaches only via a
173 * supplementary group) would EACCES and fail (500) for the legitimate
174 * owner/group-member. Off impersonation this is the same bare lstat/stat.
175 */
176static ngx_int_t
177vfs_stat_precheck(brix_vfs_ctx_t *ctx, brix_vfs_stat_t *stat_out,
178 const char *path, uint64_t start)
179{
180 int saved_errno;
181
182 if (stat_out == NULL) {
183 errno = EINVAL;
184 } else if (brix_vfs_require_confined(ctx) != NGX_OK) {
185 errno = EINVAL;
186 } else if (brix_vfs_require_authorized_lookup(ctx) == NGX_OK) {
187 return NGX_OK;
188 }
189
190 saved_errno = errno;
191 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
192 NGX_ERROR, saved_errno, start);
193 return NGX_ERROR;
194}

196static ngx_int_t
197brix_vfs_stat_impl(brix_vfs_ctx_t *ctx, brix_vfs_stat_t *stat_out,
198 int nofollow)
199{
200 struct stat st;
201 const char *path;
202 uint64_t start;
203 int saved_errno;
204 const brix_sd_driver_t *drv;
205
206 start = brix_vfs_now_ns();
207 path = brix_vfs_ctx_path(ctx);
208
209 if (vfs_stat_precheck(ctx, stat_out, path, start) != NGX_OK) {
210 return NGX_ERROR;
211 }
212
213 /* Non-POSIX backend: the namespace lives in the driver, not the export tree.
214 * nofollow is moot (an object/block backend has no symlinks). */
215 drv = brix_vfs_ctx_driver(ctx);
216 if (drv != NULL) {
217 brix_sd_ucred_t store;
218 brix_sd_cred_t cred;
219 brix_sd_stat_t sd_st;
220 int use_cred = 0, cred_err = 0;
221 ngx_int_t grc;
222 char physical[PATH_MAX];
223
224 /* Zero the cred before the gate: the gate fills only the ACTIVE kind
225 * (an x509 proxy path OR a bearer OR an s3/ceph tuple) and leaves the
226 * inactive pointers untouched, so an unzeroed struct hands a garbage
227 * pointer to the driver's stat_cred slot (a bearer-only PASSTHROUGH
228 * cred left a live-stack x509_proxy that sd_xroot dereferenced → SIGSEGV
229 * on the two-hop token read). Mirrors the data-plane callers, which all
230 * memzero their ucred before brix_vfs_backend_cred. */
231 ngx_memzero(&cred, sizeof(cred));
232
233 if (brix_path_resolved_to_pfn(ctx, path, physical,
234 sizeof(physical)) != NGX_OK)
235 {
236 saved_errno = errno;
237 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
238 NGX_ERROR, saved_errno, start);
239 return NGX_ERROR;
240 }
241
242 if (brix_vfs_cred_gate_active(ctx)) {
243 grc = brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err);
244 if (grc != NGX_OK) {
245 saved_errno = cred_err ? cred_err : EACCES;
246 errno = saved_errno;
247 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL,
248 0, NGX_ERROR, saved_errno, start);
249 return NGX_ERROR;
250 }
251 }
252
253 /* Dispatch on the leaf instance so brix_sd_stat_maybe_cred finds the
254 * leaf driver's stat_cred slot (decorators have only plain relays). */
255 if (drv->stat == NULL
256 || brix_sd_stat_maybe_cred(brix_vfs_ns_leaf(ctx->sd),
257 physical, &sd_st,
258 use_cred ? &cred : NULL) != NGX_OK)
259 {
260 saved_errno = errno;
261 brix_sd_ucred_wipe(&store); /* secret consumed; erase (A-4/T4) */
262 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
263 NGX_ERROR, saved_errno, start);
264 return NGX_ERROR;
265 }
266 brix_sd_ucred_wipe(&store); /* secret consumed; erase (A-4/T4) */
267 brix_vfs_sd_stat_fill(&sd_st, stat_out);
268 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
269 NGX_OK, 0, start);
270 return NGX_OK;
271 }
272
273 /* C-2: a live cached negative answers the probe without the syscall (or
274 * broker IPC). The hit is metered exactly like the real ENOENT it stands
275 * in for. Keyed per stat arm — see the safety envelope above. */
276 if (brix_neg_stat_lookup(ctx->root_canon, path, nofollow)) {
277 errno = ENOENT;
278 saved_errno = errno;
279 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
280 NGX_ERROR, saved_errno, start);
281 return NGX_ERROR;
282 }
283
284 if ((ctx->rootfd >= 0
285 ? brix_lstat_confined_canon_at(ctx->log, ctx->rootfd,
286 ctx->root_canon, path, &st,
287 nofollow)
288 : brix_lstat_confined_canon(ctx->log, ctx->root_canon, path,
289 &st, nofollow)) != 0)
290 {
291 saved_errno = errno;
292 if (saved_errno == ENOENT) {
293 brix_neg_stat_insert(ctx->root_canon, path, nofollow);
294 }
295 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
296 NGX_ERROR, saved_errno, start);
297 return NGX_ERROR;
298 }
299
300 brix_vfs_fill_stat(&st, stat_out);
301 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_STAT, NULL, 0,
302 NGX_OK, 0, start);
303 return NGX_OK;
304}

336/* Classify the resolved ctx path's nearline residency (online/nearline/offline/
337 * lost) so protocol handlers can advertise tape state — the HTTP Tape REST API, S3
338 * InvalidObjectState / x-amz-storage-class, root:// stat's nearline flag — WITHOUT
339 * forcing a recall. Walks any cache/stage decorators down to the CAP_NEARLINE
340 * driver and reads its residency model; an export with no nearline tier is always
341 * ONLINE (a plain disk/object store is resident). Emits no metric (it is an
342 * internal classification the caller's own op accounts for). NGX_OK with *out set,
343 * or NGX_ERROR (errno) on a guard failure or a driver error. */
344ngx_int_t
345brix_vfs_residency(brix_vfs_ctx_t *ctx, brix_sd_residency_t *out,
346 int *nearline_export)
347{
348 brix_sd_instance_t *inst;
349 char physical[PATH_MAX];
350
351 if (nearline_export != NULL) {
352 *nearline_export = 0;
353 }
354 if (out == NULL || brix_vfs_require_confined(ctx) != NGX_OK) {
355 errno = EINVAL;
356 return NGX_ERROR;
357 }
358 if (brix_vfs_require_authorized_lookup(ctx) != NGX_OK) {
359 return NGX_ERROR;
360 }
361
362 *out = BRIX_SD_RES_ONLINE;
363 for (inst = ctx->sd; inst != NULL;
364 inst = brix_vfs_decorator_source(inst))
365 {
366 if ((brix_sd_caps(inst) & BRIX_SD_CAP_NEARLINE) != 0
367 && inst->driver->residency != NULL)
368 {
369 if (brix_path_resolved_to_pfn(ctx, brix_vfs_ctx_path(ctx), physical,
370 sizeof(physical)) != NGX_OK)
371 {
372 return NGX_ERROR;
373 }
374 if (nearline_export != NULL) {
375 *nearline_export = 1;
376 }
377 return inst->driver->residency(inst, physical, out);
378 }
379 }
380 return NGX_OK;
381}

392

433/*
434 * brix_vfs_probe — confined existence/type probe for pre-op resolution and
435 * ACL gates. Unlike brix_vfs_stat/statf this emits NO OP_STAT metric or
436 * access-log line: it is an internal namespace pre-check that the caller's own
437 * operation accounts for (routing it through the metered stat would record a
438 * phantom STAT for every rm/chmod/mkdir/mv). nofollow selects lstat vs stat
439 * semantics. Returns NGX_OK with *stat_out filled when the path is present,
440 * NGX_DECLINED when it is absent (errno preserved from the underlying stat), or
441 * NGX_ERROR on a confinement-guard failure.
442 */
443ngx_int_t
444brix_vfs_probe(brix_vfs_ctx_t *ctx, int nofollow,
445 brix_vfs_stat_t *stat_out)
446{
447 struct stat st;
448 const brix_sd_driver_t *drv;
449
450 if (stat_out == NULL || brix_vfs_require_confined(ctx) != NGX_OK) {
451 errno = EINVAL;
452 return NGX_ERROR;
453 }
454 if (brix_vfs_require_authorized_lookup(ctx) != NGX_OK) {
455 return NGX_ERROR;
456 }
457
458 drv = brix_vfs_ctx_driver(ctx);
459 if (drv != NULL) {
460 brix_sd_ucred_t store;
461 brix_sd_cred_t cred;
462 brix_sd_stat_t sd_st;
463 int use_cred = 0, cred_err = 0;
464 char physical[PATH_MAX];
465
466 /* Zero before the gate: it fills only the active credential kind and
467 * leaves the inactive pointers as-is, so an unzeroed cred would pass a
468 * garbage x509_proxy/bearer pointer to the driver (see the companion
469 * note in brix_vfs_stat_impl). */
470 ngx_memzero(&cred, sizeof(cred));
471
472 if (brix_path_resolved_to_pfn(ctx, brix_vfs_ctx_path(ctx), physical,
473 sizeof(physical)) != NGX_OK)
474 {
475 return NGX_ERROR;
476 }
477
478 /* Credential gate for the probe: a denied pre-flight MUST return
479 * NGX_ERROR (EACCES), not NGX_DECLINED — a denied probe must not be
480 * silently treated as "absent", which would let the caller proceed.
481 * The caller maps EACCES → 403. */
482 if (brix_vfs_cred_gate_active(ctx)) {
483 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
484 != NGX_OK)
485 {
486 errno = cred_err ? cred_err : EACCES;
487 return NGX_ERROR;
488 }
489 }
490
491 /* Dispatch on the leaf instance so brix_sd_stat_maybe_cred finds the
492 * leaf driver's stat_cred slot (decorators have only plain relays). */
493 if (drv->stat == NULL
494 || brix_sd_stat_maybe_cred(brix_vfs_ns_leaf(ctx->sd),
495 physical, &sd_st, use_cred ? &cred : NULL) != NGX_OK)
496 {
497 brix_sd_ucred_wipe(&store); /* secret consumed; erase (A-4/T4) */
498 return NGX_DECLINED; /* absent (or unsupported) — caller's errno */
499 }
500 brix_sd_ucred_wipe(&store); /* secret consumed; erase (A-4/T4) */
501 brix_vfs_sd_stat_fill(&sd_st, stat_out);
502 return NGX_OK;
503 }
504
505 if ((ctx->rootfd >= 0
506 ? brix_lstat_confined_canon_at(ctx->log, ctx->rootfd,
507 ctx->root_canon,
508 brix_vfs_ctx_path(ctx), &st,
509 nofollow)
510 : brix_lstat_confined_canon(ctx->log, ctx->root_canon,
511 brix_vfs_ctx_path(ctx), &st,
512 nofollow)) != 0)
513 {
514 return NGX_DECLINED;
515 }
516
517 brix_vfs_fill_stat(&st, stat_out);
518 return NGX_OK;
519}
