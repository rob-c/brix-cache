1/*
2 * net_target.h — shared external transfer target parser and SSRF policy.
3 *
4 * Centralises URL parsing and DNS-based address classification for all
5 * outbound connections made by the module: native root:// TPC, WebDAV
6 * HTTP-TPC (curl), and any future S3 remote-copy feature.
7 *
8 * Two public helpers:
9 * brix_net_target_parse() — split URL into scheme/host/port/path
10 * brix_net_target_check_dns() — resolve host and reject prohibited addrs
11 *
12 * check_dns() resolves synchronously (brix_dns_resolve_sync), which blocks. It MUST be called only
13 * from a background thread (ngx_thread_pool_run or equivalent), never from
14 * the nginx event-loop worker.
15 */
16

24/*
25 * Parsed view of an outbound URL. All ngx_str_t fields point into the
26 * original URL buffer — no pool allocation is performed by the parser.
27 * The caller must keep the original buffer alive while this struct is used.
28 */
29typedef struct {
30 ngx_str_t raw_url; /* the full original URL */
31 ngx_str_t scheme; /* e.g. "https", "root" (no trailing colon) */
32 ngx_str_t host; /* hostname or IP literal (no brackets for IPv6) */
33 ngx_str_t path; /* everything from the first "/" onward, or empty */
34 uint16_t port; /* 0 when not explicitly present in the URL */
35 ngx_flag_t has_port; /* 1 when port was explicitly present */
36} brix_net_target_t;
37
38/*
39 * Policy controlling which destinations are acceptable.
40 *
41 * require_https: 1 → reject any non-https scheme URL
42 * allow_root_scheme: 1 → also accept root:// URLs (used by native TPC)
43 * allow_local: 0 → reject loopback (127/8, ::1) and link-local
44 * allow_private: 0 → reject RFC-1918 (10/8, 172.16/12, 192.168/16)
45 * and IPv6 ULA (fc00::/7)
46 * default_https_port: port substituted when URL has no port and scheme=https
47 * default_root_port: port substituted when URL has no port and scheme=root
48 * dns: brix DNS policy the checkers resolve through (the
49 * export/server `brix_resolver`); NULL = worker default
50 */
51typedef struct {
52 ngx_flag_t require_https;
53 ngx_flag_t allow_root_scheme;
54 ngx_flag_t allow_local;
55 ngx_flag_t allow_private;
56 uint16_t default_https_port;
57 uint16_t default_root_port;
58 const struct brix_dns_policy_s *dns; /* resolver policy for the DNS
59 * checkers; NULL = worker default
60 * (phase-116) */
61} brix_net_target_policy_t;

91/*
92 * brix_net_target_check_dns — resolve host and verify against policy.
93 *
94 * Resolves target->host through brix_dns_resolve_sync() (BLOCKING — call
95 * from a thread-pool task only: literal -> per-worker cache -> the event
96 * loop's resolver via the bridge -> libc). Rejects the target if any
97 * resolved address is in a prohibited range under the given policy.

118/*
119 * brix_net_target_check_cached — the event-loop flavour of check_dns.
120 *
121 * Never blocks and starts nothing: answers from an IP literal or the
122 * per-worker DNS cache. NGX_OK = every cached address passes policy;
123 * NGX_ERROR = a cached address is prohibited or the name is definitively

132/*
133 * brix_net_target_check_dns_pin — like check_dns, but also returns the first
134 * permitted resolved address as a numeric string in out_ip[0..out_ipsz).
135 *
136 * Closes the DNS-rebind TOCTOU window: the caller validates and resolves once,
137 * then pins out_ip into the transfer agent (e.g. curl CURLOPT_RESOLVE) so the
138 * connection target cannot differ from the address that passed policy. Same
139 * BLOCKING/thread-only contract as check_dns(). Returns NGX_OK with out_ip
140 * filled, NGX_ERROR otherwise (err filled). Every address comes from the
141 * brix DNS driver (policy->dns), so `brix_resolver` governs this path too.
142 */

144 const brix_net_target_t *target,
145 const brix_net_target_policy_t *policy,
146 char *out_ip, size_t out_ipsz,
147 char *err, size_t errsz);

149/*
150 * brix_net_host_chars_valid — reject host strings that could inject control
151 * bytes or alternate schemes into a redirect/registration string.
152 *
153 * Allows only the characters legal in a hostname or IPv4/IPv6 literal:
154 * [A-Za-z0-9.:-]. Rejects empty, anything with '/', whitespace, NUL, or any
155 * byte outside that set (so "evil\r\nLocation:" or "http://x" never reach a
156 * client that parses a redirect host). Returns 1 if valid, 0 otherwise.
157 */
158int brix_net_host_chars_valid(const char *host, size_t len);
