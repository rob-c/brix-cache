1/*
2 * authz_endpoint.c — the auth_request endpoint (phase-106 W4) and the
3 * X-Accel-Redirect handoff (phase-106 W3).
4 *
5 * WHAT: Two content-phase behaviours for a location that brix gates but does
6 * not serve:
7 * brix_webdav_authz on -> answer 204 + X-Brix-* identity
8 * headers, for nginx's auth_request
9 * brix_webdav_accel_redirect PFX -> answer X-Accel-Redirect: PFX<uri>
10 * with no body, handing the request
11 * to an nginx `internal` location
12 *
13 * WHY: brix owns a large authz corpus — WLCG tokens, VOMS FQANs, macaroons,
14 * GSI/X.509 chains, ZTN — and before this file NONE of it was reachable
15 * unless brix also served the bytes. Adoption was all-or-nothing. These
16 * two seams let an operator put brix's authorization in front of an
17 * existing nginx deployment and keep their own data path.
18 *
19 * HOW: Neither behaviour performs a single authorization check of its own.
20 * nginx runs the ACCESS phase before the CONTENT phase, and webdav's
21 * access handler (access.c) has by then already run
22 * access_authenticate() plus the write-method, token-scope and XrdAcc
23 * gates — returning 401/403 itself on refusal. So reaching the content
24 * phase IS the verdict, and these handlers only REPORT it. That is the
25 * property that makes them safe: there is no second, weaker copy of the
26 * policy to drift out of step with the first.
27 *
28 * SECURITY, the load-bearing points:
29 * - A refused request never reaches here at all, so neither seam can
30 * turn a denial into a serve.
31 * - X-Accel-Redirect is only honoured by nginx on a response from an
32 * upstream/handler, and the target MUST be an `internal` location.
33 * A client cannot inject it: request headers do not become response
34 * headers, and this handler builds the value itself from a
35 * CONFIGURED prefix — never from anything the client sent.
36 * - The X-Brix-* headers carry the SUBJECT of an identity (DN, VO,
37 * issuer, sub) and never the credential that proved it, so an
38 * operator copying them onward with auth_request_set cannot leak a
39 * bearer token or a proxy key.
40 */

46/*
47 * webdav_authz_set_header — add one X-Brix-* response header.
48 *
49 * Skips an empty value entirely rather than emitting an empty header: an
50 * absent header and an empty one are different things to auth_request_set,
51 * and "no VO" should read as absent.
52 */
53static ngx_int_t
54webdav_authz_set_header(ngx_http_request_t *r, const char *name,
55 const ngx_str_t *value)
56{
57 ngx_table_elt_t *h;
58
59 if (value == NULL || value->len == 0) {
60 return NGX_OK;
61 }
62
63 h = ngx_list_push(&r->headers_out.headers);
64 if (h == NULL) {
65 return NGX_ERROR;
66 }
67 h->hash = 1;
68 h->key.len = ngx_strlen(name);
69 h->key.data = (u_char *) name;
70 h->value = *value;
71 return NGX_OK;
72}

75/*
76 * webdav_authz_identity_headers — publish the authenticated SUBJECT.
77 *
78 * Only subject attributes: DN, VO, token issuer, token subject. Never the
79 * bearer token, the macaroon, or the proxy key — an auth_request consumer
80 * routinely copies these into an upstream request with auth_request_set, so a
81 * credential here would be forwarded to a backend that has no business seeing
82 * it.
83 */
84static ngx_int_t
85webdav_authz_identity_headers(ngx_http_request_t *r)
86{
87 ngx_http_brix_webdav_req_ctx_t *rx =
88 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
89 brix_identity_t *id;
90
91 if (rx == NULL || rx->identity == NULL) {
92 return NGX_OK; /* anonymous: no subject to report */
93 }
94 id = rx->identity;
95
96 if (webdav_authz_set_header(r, "X-Brix-DN", &id->dn) != NGX_OK
97 || webdav_authz_set_header(r, "X-Brix-Sub", &id->subject) != NGX_OK
98 || webdav_authz_set_header(r, "X-Brix-Issuer", &id->issuer) != NGX_OK
99 || webdav_authz_set_header(r, "X-Brix-VO", &id->vo_csv) != NGX_OK)
100 {
101 return NGX_ERROR;
102 }
103 return NGX_OK;
104}

107/*
108 * webdav_authz_enforce — make sure the policy has actually run.
109 *
110 * SECURITY, load-bearing. nginx SKIPS the entire ACCESS phase for subrequests:
111 *
112 * ngx_http_core_access_phase()
113 * if (r != r->main) { r->phase_handler = ph->next; return NGX_AGAIN; }
114 *
115 * and auth_request issues exactly such a subrequest. So "reaching the CONTENT
116 * phase means the ACCESS phase admitted me" — true for a main request — is
117 * FALSE for the one case this endpoint exists to serve. Relying on it would
118 * make the endpoint answer 204 unconditionally: brix would become a rubber
119 * stamp that admits every request an auth_request consumer sends it, which is
120 * an authorization bypass in the security-critical direction. (This is not
121 * hypothetical: the first version of this file had that bug, and the
122 * "refusing target denies the outer location" test is what caught it.)
123 *
124 * So when the access phase was skipped, run the SAME handler explicitly. It is
125 * the identical code path a main request takes — auth gate, token scope,
126 * XrdAcc, rate limit — so there is no second copy of the policy to drift.
127 *
128 * NGX_DECLINED means the location is not brix-enabled and therefore has no
129 * policy to apply. That fails CLOSED (403): an authorization endpoint with no
130 * configured authorization must never admit.
131 */
132static ngx_int_t
133webdav_authz_enforce(ngx_http_request_t *r)
134{
135 ngx_int_t rc;
136
137 if (r == r->main) {
138 return NGX_OK; /* the access phase already ran */
139 }
140
141 rc = ngx_http_brix_webdav_access_handler(r);
142 if (rc == NGX_DECLINED) {
143 ngx_log_error(NGX_LOG_ERR, r->connection->log, 0,
144 "brix_webdav_authz: no brix policy on this location — "
145 "refusing (an authz endpoint must fail closed)");
146 return NGX_HTTP_FORBIDDEN;
147 }
148 return rc;
149}

152/*
153 * webdav_authz_endpoint — the auth_request target.
154 *
155 * Reaching the content phase means the ACCESS phase admitted the request, so
156 * the answer is 204 plus the subject headers. 204 (not 200) because
157 * auth_request treats any 2xx as success and a body would be pointless — the
158 * subrequest's body is discarded.
159 */
160ngx_int_t
161webdav_authz_endpoint(ngx_http_request_t *r)
162{
163 ngx_int_t rc;
164
165 rc = webdav_authz_enforce(r);
166 if (rc != NGX_OK) {
167 return rc;
168 }
169
170 if (webdav_authz_identity_headers(r) != NGX_OK) {
171 return NGX_HTTP_INTERNAL_SERVER_ERROR;
172 }
173
174 r->headers_out.status = NGX_HTTP_NO_CONTENT;
175 r->headers_out.content_length_n = 0;
176 r->header_only = 1;
177
178 return ngx_http_send_header(r);
179}

182/*
183 * webdav_accel_redirect — hand the admitted request to an internal location.
184 *
185 * MECHANISM NOTE. X-Accel-Redirect is an UPSTREAM-RESPONSE feature: nginx acts
186 * on it when it appears in a response the upstream module received, not when a
187 * content handler sets it on its own response. A handler that merely emits the
188 * header produces a 200 with an empty body — which is precisely what the first
189 * version of this function did. The handler-side primitive is
190 * ngx_http_internal_redirect(), so that is what this uses; the directive keeps
191 * the familiar X-Accel-Redirect NAME because that is the concept operators
192 * know, and the observable behaviour is the same.
193 *
194 * The target is <configured prefix><request uri>. The prefix comes from the
195 * config and the URI is nginx's already-normalised r->uri (traversal resolved
196 * by nginx's own parser before any handler runs), so nothing the client sends
197 * can steer the request outside the operator's chosen internal namespace.
198 */
199ngx_int_t
200webdav_accel_redirect(ngx_http_request_t *r, const ngx_str_t *prefix)
201{
202 ngx_str_t target;
203 u_char *p;
204
205 target.len = prefix->len + r->uri.len;
206 p = ngx_pnalloc(r->pool, target.len);
207 if (p == NULL) {
208 return NGX_HTTP_INTERNAL_SERVER_ERROR;
209 }
210 ngx_memcpy(p, prefix->data, prefix->len);
211 ngx_memcpy(p + prefix->len, r->uri.data, r->uri.len);
212 target.data = p;
213
214 /* Same subrequest protection as the authz endpoint: never hand off on a
215 * request whose access phase nginx skipped. */
216 if (webdav_authz_enforce(r) != NGX_OK) {
217 return NGX_HTTP_FORBIDDEN;
218 }
219
220 /* Publish the subject before handing off, so the internal location (or an
221 * upstream behind it) can be given the identity with proxy_set_header. */
222 if (webdav_authz_identity_headers(r) != NGX_OK) {
223 return NGX_HTTP_INTERNAL_SERVER_ERROR;
224 }
225
226 return ngx_http_internal_redirect(r, &target, &r->args);
227}

230/*
231 * webdav_gate_only_dispatch — run whichever gate-only seam this location
232 * configures, or NGX_DECLINED when it configures neither.
233 *
234 * Lives here rather than inline in dispatch.c so the branch sits next to the
235 * handlers it selects (and so dispatch.c stays under the 600-line ratchet).
236 */
237ngx_int_t
238webdav_gate_only_dispatch(ngx_http_request_t *r,
239 ngx_http_brix_webdav_loc_conf_t *conf)
240{
241 if (conf->authz_endpoint) {
242 return webdav_authz_endpoint(r);
243 }
244 if (conf->accel_redirect.len > 0) {
245 return webdav_accel_redirect(r, &conf->accel_redirect);
246 }
247 return NGX_DECLINED;
248}
