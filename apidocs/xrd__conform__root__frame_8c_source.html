59static void
60probe_short_handshake(xrd_conform_execution *execution, const brix_url *url)
61{
62 ClientInitHandShake hello;
63 brix_io io;
64 brix_conn check;
65 brix_status status;
66 char reason[64];
67 int recovered = 0;
68 brix_opts opts = execution->run->opts;
69
70 brix_status_clear(&status);
71 opts.capture = NULL;
72 xrd_pack_handshake(&hello);
73 if (xrd_conform_reserve(execution, 19, reason, sizeof(reason))

75 (void) brix_write_full(&io, &hello, 19, &status);
76 raw_close(&io);
77 brix_status_clear(&status);

79 recovered = 1;
80 brix_close(&check);

84 recovered ? BRIX_CONFORM_VERDICT_CONFORMANT

87 : "short-handshake-poisoned-endpoint",
88 "subsequent-full-connection-succeeds", status.msg);

91static void

93{
