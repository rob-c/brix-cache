37/* cms_state_extract_path — pure validation of a kYR_state payload: bound the
38 * NUL-terminated namespace path, require an absolute path that fits the buffer,
39 * and reject any ".." traversal before the registry/filesystem is touched
40 * (cheap defence-in-depth ahead of the kernel-confined probe). Copies the
41 * NUL-terminated path into pathz and returns its length via *pl_out; returns
42 * NGX_OK or NGX_ERROR (caller stays silent, matching real cmsd). */
43static ngx_int_t
44cms_state_extract_path(const u_char *payload, size_t plen, char *pathz,
45 size_t pathz_size, size_t *pl_out)
46{
47 size_t pl;
48 size_t k;
49
50 /* bounded length of the NUL-terminated path */
51 for (pl = 0; pl < plen && payload[pl] != '\0'; pl++) { /* void */ }
52 if (pl == 0 || payload[0] != '/' || pl >= pathz_size) {
53 return NGX_ERROR;
54 }
55
56 /* reject path traversal before touching the registry/filesystem */
57 for (k = 0; k + 1 < pl; k++) {
58 if (payload[k] == '.' && payload[k + 1] == '.') {
59 return NGX_ERROR;
60 }
61 }
62
63 ngx_memcpy(pathz, payload, pl);
64 pathz[pl] = '\0';
65 *pl_out = pl;
66 return NGX_OK;
67}

69/* cms_frame_state — kYR_state (raw): the manager asks "do you hold <path>?" as
70 * part of on-demand selection. The payload is the raw NUL-terminated namespace
71 * path (no Pup framing). We answer kYR_have (echoing streamid = path hash) if
72 * we can serve the path, else stay silent so the manager won't select us —
73 * matching real cmsd.
74 *
75 * Two ways to "have" a path:
76 * - manager_mode (a sub-manager registered UP to a meta-manager): forward the
77 * query to our own server registry — if any registered leaf data node
78 * exports a prefix covering the path, we have it (the client will be
79 * redirected to us and we then redirect down to the leaf). This is what
80 * makes a multi-tier meta->nginx->leaf mesh resolve.
81 * - data node: the file exists on our local export filesystem. */
82ngx_int_t
83cms_frame_state(ngx_brix_cms_ctx_t *ctx, uint32_t streamid, u_char code)
84{
85 const u_char *payload = ctx->inbuf + NGX_BRIX_CMS_HDR_LEN;
86 size_t plen = ctx->in_need - NGX_BRIX_CMS_HDR_LEN;
87 char pathz[1024];
88 char safe[256];
89 size_t pl;
90 struct stat st;
91
92 if (cms_state_extract_path(payload, plen, pathz, sizeof(pathz), &pl)
93 != NGX_OK)
94 {
95 return NGX_OK;
96 }
97
98 /* WS6: cms_state_extract_path bounds the path and rejects "..", but it
99 * accepts every other byte — including CR/LF. The probe comes from the
100 * manager leg, so an unsanitised %s here lets a hostile (or compromised)
101 * manager forge whole lines into error.log. Every log site below uses the
102 * escaped rendering; the wire/VFS calls keep the real path. */
103 brix_sanitize_log_string(pathz, safe, sizeof(safe));
104
105 if (ctx->conf->manager_mode) {
106 char host[256];
107 uint16_t dport;
108 if (brix_srv_select(pathz, 0, host, sizeof(host), &dport)) {
109 ngx_log_debug1(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
110 "brix: CMS state(mgr): registry serves "
111 "\"%s\", replying kYR_have", safe);
112 brix_cms_log_action(ctx->cycle->log, "state-probe",
113 (const char *) ctx->mgr_name.data,
114 "in", safe, 1,
115 "sub-manager: a downstream node holds it "
116 "-> kYR_have");
117 return ngx_brix_cms_send_have(ctx, streamid, pathz, pl);
118 }
119
120 /*
121 * Phase-61 W7: registry miss. With brix_cms_state_relay on, recurse
122 * — park the parent's leg and re-ask our own nodes; the first child
123 * kYR_have (server-leg ingest) is echoed back up under the parent's
124 * streamid. Off, or table full, or no child took the probe: stay
125 * silent, which the parent reads as "not here" (stock semantics).
126 */
127 if (ctx->conf->cms.state_relay) {
128 uint32_t down_sid;
129 ngx_uint_t i, sent = 0;
130 brix_cms_srv_ctx_t *node;
131
132 down_sid = brix_cms_state_relay_add(ctx, streamid, pathz);
133 if (down_sid == 0) {
134 return NGX_OK;
135 }
136 for (i = 0; i < brix_cms_srv_node_count(); i++) {
137 node = brix_cms_srv_node_at(i);
138 if (node == NULL || !node->logged_in || node->c == NULL) {
139 continue;
140 }
141 if (brix_cms_srv_send_state(node, down_sid, pathz) == NGX_OK) {
142 sent++;
143 }
144 }
145 if (sent == 0) {
146 /* Parked a leg no child will ever answer: the entry now just
147 * holds relay-table capacity until its TTL reaps it, and the
148 * parent degrades to "not here". Logged unconditionally —
149 * a debug-only counter is invisible in the builds that hit
150 * this, which is every production one. */
151 ngx_log_error(NGX_LOG_INFO, ctx->cycle->log, 0,
152 "brix: CMS state(mgr): relayed \"%s\" down to no "
153 "eligible node; parent will read silence",
154 safe);
155 }
156 ngx_log_debug2(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
157 "brix: CMS state(mgr): relayed \"%s\" down to "
158 "%ui node(s)", safe, sent);
159 }
160 return NGX_OK;
161 }

163 /*
164 * Kernel-confined existence probe. A malicious manager can ask
165 * "do you hold <path>?" for ANY path; the raw stat() this replaced
166 * followed symlinks, so a symlink planted under the export root
167 * (e.g. /link -> /etc) would make us answer kYR_have for a file
168 * OUTSIDE the root — a cross-root information leak and a
169 * cluster-poisoning vector. brix_stat_beneath() resolves the
170 * path under the persistent export rootfd with openat2
171 * RESOLVE_BENEATH, so any symlink or ".." that escapes the root is
172 * rejected by the kernel and we correctly stay silent. (The ".."
173 * pre-check in cms_state_extract_path remains as cheap
174 * defence-in-depth.) A node with no local export root (rootfd < 0)
175 * never holds files locally.
176 */
177 if (ctx->conf->rootfd >= 0
178 && brix_stat_beneath(ctx->conf->rootfd, pathz, &st) == 0)
179 {
180 ngx_log_debug1(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
181 "brix: CMS state: have \"%s\", "
182 "replying kYR_have", safe);
183 brix_cms_log_action(ctx->cycle->log, "state-probe",
184 (const char *) ctx->mgr_name.data,
185 "in", safe, 1, "file present on export -> kYR_have");
186 return ngx_brix_cms_send_have(ctx, streamid, pathz, pl);
187 }
188
189 ngx_log_debug1(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
190 "brix: CMS state: do not have \"%s\"", safe);
191 return NGX_OK;
192}

194/* cms_super_fan_down — Phase-61 W7: an explicit supervisor pushes a manager-
195 * forwarded namespace op DOWN to its own logged-in data nodes instead of
196 * executing it locally (supVOps marks these Forward in stock cmsd's
197 * initSUProuting). The Pup payload is decoded once and re-encoded per child
198 * hop; rrdata spans are NUL-terminated in place on the wire, so they pass
199 * straight through as C strings. Silent toward the manager either way —
200 * per-child failures are the child's to report on its own leg. */
201ngx_int_t
202cms_super_fan_down(ngx_brix_cms_ctx_t *ctx, u_char code,
203 const u_char *payload, size_t plen)
204{
205 brix_cms_rrdata_t rr;
206 brix_cms_srv_ctx_t *node;
207 ngx_uint_t i, sent;
208 uint32_t sid;
209
210 if (brix_cms_rrdata_parse(code, payload, plen, &rr) != 0) {
211 ngx_log_error(NGX_LOG_WARN, ctx->cycle->log, 0,
212 "brix: CMS super: malformed forwarded op=%ui — dropped",
213 (ngx_uint_t) code);
214 return NGX_OK;
215 }
216
217 sid = brix_cms_srv_next_streamid();
218 sent = 0;
219
220 for (i = 0; i < brix_cms_srv_node_count(); i++) {
221 node = brix_cms_srv_node_at(i);
222 if (node == NULL || !node->logged_in || node->c == NULL) {
223 continue;
224 }
225 if (brix_cms_forward_to_node(node->c, code, sid,
226 (const char *) rr.ident,
227 (const char *) rr.path,
228 (const char *) rr.path2,
229 (const char *) rr.mode,
230 (const char *) rr.opaque) == NGX_OK)
231 {
232 sent++;
233 }
234 }
235
236 if (sent == 0) {
237 /* A forwarded op that reached nobody is a silently dropped op — the
238 * one outcome of this function worth seeing without a debug build. */
239 ngx_log_error(NGX_LOG_WARN, ctx->cycle->log, 0,
240 "brix: CMS super: forwarded op=%ui reached no node — "
241 "dropped", (ngx_uint_t) code);
242 }
243 ngx_log_debug2(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
244 "brix: CMS super: forwarded op=%ui down to %ui node(s)",
245 (ngx_uint_t) code, sent);
246 return NGX_OK;
247}
