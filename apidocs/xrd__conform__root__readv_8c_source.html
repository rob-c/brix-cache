4/* WHAT: Reserve and issue one vector read. WHY: Every scenario must honor run
5 * request/byte ceilings. HOW: Sum lengths overflow-safely, reserve, call API. */
6static ssize_t
7readv_bounded(xrd_conform_execution *execution, brix_conn *connection,
8 brix_file *file, brix_readv_seg *segments, size_t count, brix_status *status)
9{
10 size_t index;
11 size_t total = 0;
12 char reason[64];
13
14 for (index = 0; index < count; index++) {
15 if (segments[index].len > SIZE_MAX - total) return -1;
16 total += segments[index].len;
17 }
18 if (!xrd_conform_reserve(execution, total, reason, sizeof(reason))) {
19 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
20 return -1;
21 }
22 brix_status_clear(status);
23 return brix_file_readv(connection, file, segments, count, status);
24}

39/* WHAT: Recognize calibrated whole-request EOF errors. WHY: Stock v5 reports
40 * FSError while BriX reports IOError; both safely reject without segment data.
41 * HOW: Accept only the two observed, defined wire errors and a failed call. */
42static int
43readv_eof_error(ssize_t got, const brix_readv_seg *segment,
44 const brix_status *status)
45{
46 return got < 0 && segment->got == 0
47 && (status->kxr == kXR_IOError || status->kxr == kXR_FSError);
48}

50/* WHAT: Exercise one-element, duplicate, descending and overlap geometries.
51 * WHY: These rows catch element loss, reordering and cross-element confusion.
52 * HOW: Issue small deterministic vectors and compare each returned extent. */
53static void
54probe_descending(xrd_conform_execution *execution, brix_conn *connection,
55 brix_file *file, const uint8_t *baseline)
56{
57 uint8_t buffers[2][16];
58 brix_readv_seg segments[2];
59 brix_status status;
60 ssize_t got;
61
62 memset(segments, 0, sizeof(segments));

64 segments[1].offset = 0; segments[1].len = 2; segments[1].buf = buffers[1];

78/* WHAT: Exercise single and duplicate vector geometries. WHY: Keep independent
79 * ordering work out of the basic equality probe. HOW: Compare both buffers. */
80static void
81probe_geometries(xrd_conform_execution *execution, brix_conn *connection,
82 brix_file *file, const uint8_t *baseline, uint64_t size)
83{
84 uint8_t buffers[2][16];
85 brix_readv_seg segments[2];
86 brix_status status;
87 size_t length = size < 8 ? (size_t) size : 8;
88 ssize_t got;
89
90 if (length == 0) return;
91 memset(segments, 0, sizeof(segments));
92 segments[0].offset = 0; segments[0].len = length; segments[0].buf = buffers[0];
93 got = readv_bounded(execution, connection, file, segments, 1, &status);
94 grade(execution, "RDV-001", got == (ssize_t) length
95 && segments[0].got == length && memcmp(buffers[0], baseline, length) == 0,
96 "single-readv-equals-read", &status);
97 segments[1] = segments[0]; segments[1].buf = buffers[1];
98 got = readv_bounded(execution, connection, file, segments, 2, &status);
99 grade(execution, "RDV-005", got == (ssize_t) (length * 2)
100 && segments[0].got == length && segments[1].got == length
101 && memcmp(buffers[0], baseline, length) == 0
102 && memcmp(buffers[1], baseline, length) == 0,
103 "duplicate-elements-preserved", &status);
104 if (size >= 2) probe_descending(execution, connection, file, baseline);
105}

107/* WHAT: Exercise zero, past-EOF and straddling vector extents. WHY: Fractional
108 * vector answers must preserve elements and clamp exactly. HOW: Use one element. */
109static void
110probe_edges(xrd_conform_execution *execution, brix_conn *connection,
111 brix_file *file, uint64_t size)
112{
113 uint8_t buffer[2];
114 brix_readv_seg segment;
115 brix_status status;
116 ssize_t got;
117
118 memset(&segment, 0, sizeof(segment)); segment.buf = buffer;
119 got = readv_bounded(execution, connection, file, &segment, 1, &status);
120 grade(execution, "RDV-008", got == 0 && segment.got == 0,
121 "zero-element-preserved", &status);
122 if (size <= INT64_MAX - 1) {
123 segment.offset = (int64_t) size + 1; segment.len = 1; segment.got = 0;
124 got = readv_bounded(execution, connection, file, &segment, 1, &status);
125 grade(execution, "RDV-009",
126 readv_eof_error(got, &segment, &status),
127 "past-eof-request-failed-wholesale", &status);
128 }
129 if (size == 0 || size > INT64_MAX) return;
130 segment.offset = (int64_t) size - 1; segment.len = 2; segment.got = 0;
131 got = readv_bounded(execution, connection, file, &segment, 1, &status);
132 grade(execution, "RDV-010",
133 readv_eof_error(got, &segment, &status),
134 "straddling-request-failed-wholesale", &status);
135}

150/* WHAT: Prove a >16 MiB vector reply stays one logical response and is exact.
151 * WHY: Official XrdCl does not reassemble kXR_oksofar for readv.
152 * HOW: Split into legal elements, count responses and compare every byte. */
153void
154xrd_conform_probe_readv_chunked(xrd_conform_execution *execution,
155 brix_conn *connection, brix_file *file, const uint8_t *baseline,
156 size_t size)
157{
158 const size_t segment_size = 2U * 1024U * 1024U - BRIX_READV_SEGSIZE;
159 brix_readv_seg *segments;
160 uint8_t *buffer;
161 brix_status status;
162 char reason[64];
163 unsigned partials = 0;
164 size_t count, index, cursor = 0;
165 ssize_t got;
166 int exact = 1;
167
168 count = (size + segment_size - 1U) / segment_size;
169 segments = calloc(count, sizeof(*segments));
170 buffer = malloc(size);
171 if (segments == NULL || buffer == NULL) {
172 free(segments); free(buffer);
173 return;
174 }
175 for (index = 0; index < count; index++) {
176 size_t length = size - cursor;
177
178 if (length > segment_size) length = segment_size;
179 segments[index].offset = (int64_t) cursor;
180 segments[index].len = length;
181 segments[index].buf = buffer + cursor;
182 cursor += length;
183 }
184 if (!xrd_conform_reserve(execution, size, reason, sizeof(reason))) {
185 free(segments); free(buffer);
186 return;
187 }
188 brix_status_clear(&status);
189 got = brix_file_readv_profile(connection, file, segments, count,
190 &partials, &status);
191 for (index = 0; index < count && exact; index++)
192 exact = segments[index].got == segments[index].len;
193 exact = exact && got == (ssize_t) size
194 && memcmp(buffer, baseline, size) == 0 && partials == 0;
195 xrd_conform_result_set(execution, "RDV-018",
196 exact ? BRIX_CONFORM_VERDICT_CONFORMANT
197 : BRIX_CONFORM_VERDICT_DIVERGENT,
198 exact ? "single-logical-response-streamed-byte-exact"
199 : "large-readv-framing-mismatch",
200 "one kXR_ok response; byte-exact", status.msg);
201 free(segments); free(buffer);
