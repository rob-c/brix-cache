1/*
2 * connect_upstream_select.c — upstream endpoint selection for the lazy connect.
3 *
4 * WHAT: Chooses where brix_proxy_connect() goes: a redirect target, a
5 * CMS-pinned data server (phase-115 W2.1), an already-authenticated
6 * pooled connection, a healthy member of the round-robin upstream array,
7 * or the single configured host.
8 *
9 * WHY: Selection is pure policy over the conf and per-session state — the one
10 * self-contained phase of connect_upstream.c. Splitting it keeps that
11 * file to the socket/resolve/TLS mechanics (and under the size gate).
12 *
13 * HOW: brix_proxy_select_endpoint() fills a brix_proxy_target_t (host + port)
14 * and returns NGX_DECLINED for "go resolve/connect", NGX_OK when a pooled
15 * connection was adopted (connect complete), NGX_ERROR when every
16 * upstream is down. pc_pick_healthy_upstream() walks the worker-local
17 * upstream health table from the shared round-robin cursor.
18 */
19
20#include "proxy_internal.h"

22/* Round-robin counter for multiple upstream endpoints. */
23static ngx_atomic_t proxy_upstream_rr;

26/*
27 * WHAT: Picks a healthy upstream index from the round-robin array, honouring
28 * the lazily-allocated upstream health table.
29 *
30 * WHY: When every upstream is marked down (and none has aged past the retry
31 * window) we must fail the connect rather than fall through to a known-dead
32 * endpoint and hammer it in a tight loop.
33 *
34 * HOW: Advances the shared RR counter, then walks up to nelts entries from that
35 * start looking for one that is up (or stale enough to retry). A NULL
36 * health table means "all healthy" — the RR pick stands. Returns NGX_OK
37 * with *idx_out set, or NGX_ERROR when all are down.
38 */
39static ngx_int_t
40pc_pick_healthy_upstream(ngx_stream_brix_srv_conf_t *conf,
41 ngx_uint_t *idx_out)
42{
43 ngx_uint_t nelts = conf->proxy.upstreams->nelts;
44 ngx_uint_t idx;
45 ngx_uint_t i;
46 int found = 0;
47 const brix_proxy_up_status_t *status = brix_proxy_up_status_get();
48
49 idx = ngx_atomic_fetch_add(&proxy_upstream_rr, 1) % nelts;
50
51 /* The health table is lazily allocated by the health-tracking path and is
52 * NULL until a failure marks an upstream down (the mark_fail/is_down
53 * accessors are all NULL-tolerant no-ops). Treat a NULL table as "every
54 * upstream healthy" so the round-robin pick stands — same semantics,
55 * without dereferencing a NULL array. */
56 for (i = 0; status != NULL && i < nelts; i++) {
57 ngx_uint_t cur = (idx + i) % nelts;
58 if (!status[cur].down ||
59 ngx_time() - status[cur].checked >= BRIX_PROXY_FAIL_TIMEOUT)
60 {
61 idx = cur;
62 found = 1;
63 break;
64 }
65 }
66 if (status == NULL) {
67 found = 1; /* no health table → RR pick is authoritative */
68 }
69
70 if (!found) {
71 return NGX_ERROR;
72 }
73
74 *idx_out = idx;
75 return NGX_OK;
76}

78/*
79 * WHAT: Chooses the upstream endpoint into *tgt (host+port) with priority
80 * redirect > pooled connection > round-robin healthy array > single host.
81 *
82 * WHY: A pooled connection short-circuits the whole connect: it is already
83 * authenticated and bootstrapped, so we adopt it and either dispatch the
84 * saved request or resume the client read loop. GSI-as-user connections
85 * are per-user authenticated and must never reuse a pooled (foreign
86 * identity) connection.
87 *
88 * HOW: Returns NGX_OK when a pooled connection was adopted (caller returns OK to
89 * its own caller — connect is complete), NGX_DECLINED when *tgt was filled
90 * and the caller must proceed to resolve/connect, or NGX_ERROR when all
91 * upstreams are down.
92 */
93ngx_int_t
94brix_proxy_select_endpoint(brix_proxy_ctx_t *proxy, ngx_connection_t *client_conn,
95 ngx_stream_brix_srv_conf_t *conf, brix_proxy_target_t *tgt)
96{
97 ngx_connection_t *uconn;
98 int pooled_idx = -1;
99 ngx_uint_t idx;
100
101 if (proxy->redirect_host.len > 0) {
102 tgt->host = &proxy->redirect_host;
103 tgt->port = (ngx_int_t) proxy->redirect_port;
104 /* upstream_idx stays what it was, or -1 if we started redirected */
105 return NGX_DECLINED;
106 }
107
108 /* Phase-115 W2.1: a CMS-selected session goes to its pinned data server —
109 * the configured upstream list (usually empty on a manager) and the shared
110 * pool (keyed by that list) do not apply. */
111 if (proxy->pinned_host.len > 0) {
112 tgt->host = &proxy->pinned_host;
113 tgt->port = (ngx_int_t) proxy->pinned_port;
114 proxy->upstream_idx = -1;
115 return NGX_DECLINED;
116 }
117
118 /* GSI-as-user connections are per-user authenticated — never reuse a
119 * pooled connection (it carries a different identity). */
120 uconn = (conf->proxy.auth == BRIX_PROXY_AUTH_GSI)
121 ? NULL : brix_proxy_pool_get(proxy, conf, &pooled_idx);
122 if (uconn != NULL) {
123 proxy->conn = uconn;
124 proxy->upstream_idx = pooled_idx;
125 proxy->state = XRD_PX_IDLE;
126 proxy->from_pool = 1;
127 uconn->data = proxy;
128 uconn->log = client_conn->log;
129 uconn->read->log = client_conn->log;
130 uconn->write->log = client_conn->log;
131
132 if (proxy->saved_req != NULL) {
133 brix_proxy_dispatch_pending(proxy);
134 } else {
135 proxy->client_ctx->state = XRD_ST_REQ_HEADER;
136 brix_schedule_read_resume(client_conn);
137 }
138 return NGX_OK;
139 }
140
141 if (conf->proxy.upstreams != NULL && conf->proxy.upstreams->nelts > 0) {
142 brix_proxy_upstream_t *ups = conf->proxy.upstreams->elts;
143
144 if (pc_pick_healthy_upstream(conf, &idx) != NGX_OK) {
145 ngx_log_error(NGX_LOG_ERR, client_conn->log, 0,
146 "xrootd proxy: all %ui upstream(s) down — "
147 "failing request",
148 (ngx_uint_t) conf->proxy.upstreams->nelts);
149 return NGX_ERROR;
150 }
151
152 tgt->host = &ups[idx].host;
153 tgt->port = (ngx_int_t) ups[idx].port;
154 proxy->upstream_idx = (int) idx;
155 } else {
156 tgt->host = &conf->proxy.host;
157 tgt->port = conf->proxy.port;
158 proxy->upstream_idx = -1;
159 }
160
161 return NGX_DECLINED;
162}
