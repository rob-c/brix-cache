182/*
183 * brix_shared_apply_read_only() — enforce the hard read-only switch. When
184 * common->read_only is on, force allow_write off so EVERY existing write gate
185 * (root:// brix_dispatch_require_write, the WebDAV/S3 write-method gate, the
186 * write-open gate) rejects writes at the protocol edge - before the VFS, and
187 * before token scope (allow_write is checked first), so a write-scoped token
188 * cannot bypass it. ngx_http_brix_shared_merge() applies it after the
189 * allow_write/read_only merges so no protocol can forget the enforcement;
190 * callers with later allow_write-dependent validations (e.g. WebDAV's
191 * "writes need auth" check) simply run them after the shared merge.
192 */
193static inline void
194brix_shared_apply_read_only(ngx_http_brix_shared_conf_t *common,
195 ngx_log_t *log)
196{
197 /*
198 * brix_read_only_public is the STRICTER posture, so it implies the weaker
199 * one: turning read_only on here (rather than only gating queries) means an
200 * operator cannot get the introspection restrictions while leaving the
201 * export writable, and means every existing write gate keyed on allow_write
202 * covers a public gateway without knowing the directive exists.
203 */
204 if (common->read_only_public == 1 && common->read_only != 1) {
205 common->read_only = 1;
206 if (log != NULL) {
207 ngx_log_error(NGX_LOG_NOTICE, log, 0,
208 "brix: read_only_public on - implies read_only; the export is "
209 "read-only and server-introspection queries are refused");
210 }
211 }
212
213 if (common->read_only != 1) {
214 return;
215 }
216 if (common->allow_write == 1 && log != NULL) {
217 ngx_log_error(NGX_LOG_NOTICE, log, 0,
218 "brix: read_only on - the export is read-only; all write "
219 "operations are rejected at the protocol edge (overrides allow_write)");
220 }
221 common->allow_write = 0;
222}

227/*
228 * brix_shared_security_gate() — E-1: a valid-but-dangerous config setting is
229 * loud at load and refused under strict mode.
230 *
231 * WHY: several configurations parse cleanly yet leave the export wide open —
232 * anonymous S3 (no SigV4/token verification), WebDAV writes with auth optional,
233 * an anonymous dashboard. Each is a legitimate choice for a closed lab and a
234 * foot-gun in production, so the default must be loud (an operator who never
235 * reads the config still sees the warning in the error log at every reload),
236 * and a site that wants the guarantee flips `brix_strict_security on` to turn
237 * every such setting into a hard `nginx -t` failure — fail-closed, opt-in.
238 *
239 * HOW: emit NGX_LOG_WARN (default) or NGX_LOG_EMERG (strict) naming the insecure
240 * setting `what` and the directive `remedy` that closes it. Return NGX_OK when
241 * the merge may proceed (warn-only) and NGX_ERROR when strict mode requires the
242 * caller to return NGX_CONF_ERROR. The caller owns the return so the diagnostic
243 * points at the offending location's own merge.
244 */
245static inline ngx_int_t
246brix_shared_security_gate(ngx_conf_t *cf, ngx_flag_t strict,
247 const char *what, const char *remedy)
248{
249 ngx_conf_log_error(strict ? NGX_LOG_EMERG : NGX_LOG_WARN, cf, 0,
250 "brix: insecure configuration — %s; set %s to close it%s",
251 what, remedy,
252 strict ? " (refused: brix_strict_security on)" : "");
253 return strict ? NGX_ERROR : NGX_OK;
254}

277/*
278 * brix_shared_thread_pool() — resolve the async-I/O thread pool for a merged
279 * common loc-conf, lazily and idempotently.
280 *
281 * WHY: postconfig only wires common.thread_pool for a *server-level* enabled
282 * loc-conf (webdav_postconf_setup_thread_pool walks cscf->ctx->loc_conf). A
283 * protocol enabled per-`location` (the common production shape for WebDAV TPC
284 * and PUT) leaves common.thread_pool NULL, so any path that only NULL-checks it
285 * falls back to a synchronous, event-loop-blocking transfer regardless of how
286 * many workers are configured. Resolving by name at first use — the pool is
287 * fully initialized by the time any request runs — closes that gap for every
288 * offload site with one shared helper (previously copy/paste'd into move.c,
289 * copy_collection.c, open_or_fill.c, http_serve_offload.c, http_cache_fill.c).
290 *
291 * HOW: returns the cached handle if already resolved; else looks it up by the
292 * configured name (default "default"), caches a hit back onto common, and
293 * returns it. NULL means no such pool exists — the caller must run synchronously.
294 */
295static inline ngx_thread_pool_t *
296brix_shared_thread_pool(ngx_http_brix_shared_conf_t *common)
297{
298 ngx_thread_pool_t *pool;
299
300 if (common == NULL) {
301 return NULL;
302 }
303
304 pool = common->thread_pool;
305 if (pool == NULL) {
306 static ngx_str_t default_name = ngx_string("default");
307 ngx_str_t *pname = common->thread_pool_name.len > 0
308 ? &common->thread_pool_name : &default_name;
309 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, pname);
310 if (pool != NULL) {
311 common->thread_pool = pool;
312 }
313 }
314
315 return pool;
316}
