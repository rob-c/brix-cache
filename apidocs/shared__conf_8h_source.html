197/*
198 * brix_shared_apply_read_only() — enforce the hard read-only switch. When
199 * common->read_only is on, force allow_write off so EVERY existing write gate
200 * (root:// brix_dispatch_require_write, the WebDAV/S3 write-method gate, the
201 * write-open gate) rejects writes at the protocol edge - before the VFS, and
202 * before token scope (allow_write is checked first), so a write-scoped token
203 * cannot bypass it. ngx_http_brix_shared_merge() applies it after the
204 * allow_write/read_only merges so no protocol can forget the enforcement;
205 * callers with later allow_write-dependent validations (e.g. WebDAV's
206 * "writes need auth" check) simply run them after the shared merge.
207 */
208static inline void
209brix_shared_apply_read_only(ngx_http_brix_shared_conf_t *common,
210 ngx_log_t *log)
211{
212 /*
213 * brix_read_only_public is the STRICTER posture, so it implies the weaker
214 * one: turning read_only on here (rather than only gating queries) means an
215 * operator cannot get the introspection restrictions while leaving the
216 * export writable, and means every existing write gate keyed on allow_write
217 * covers a public gateway without knowing the directive exists.
218 */
219 if (common->read_only_public == 1 && common->read_only != 1) {
220 common->read_only = 1;
221 if (log != NULL) {
222 BRIX_LOG(NGX_LOG_NOTICE, log, 0,
223 "brix: read_only_public on - implies read_only; the export is "
224 "read-only and server-introspection queries are refused");
225 }
226 }
227
228 if (common->read_only != 1) {
229 return;
230 }
231 if (common->allow_write == 1 && log != NULL) {
232 BRIX_LOG(NGX_LOG_NOTICE, log, 0,
233 "brix: read_only on - the export is read-only; all write "
234 "operations are rejected at the protocol edge (overrides allow_write)");
235 }
236 common->allow_write = 0;
237}

242/*
243 * brix_shared_security_gate() — E-1: a valid-but-dangerous config setting is
244 * loud at load and refused under strict mode.
245 *
246 * WHY: several configurations parse cleanly yet leave the export wide open —
247 * anonymous S3 (no SigV4/token verification), WebDAV writes with auth optional,
248 * an anonymous dashboard. Each is a legitimate choice for a closed lab and a
249 * foot-gun in production, so the default must be loud (an operator who never
250 * reads the config still sees the warning in the error log at every reload),
251 * and a site that wants the guarantee flips `brix_strict_security on` to turn
252 * every such setting into a hard `nginx -t` failure — fail-closed, opt-in.
253 *
254 * HOW: emit NGX_LOG_WARN (default) or NGX_LOG_EMERG (strict) naming the insecure
255 * setting `what` and the directive `remedy` that closes it. Return NGX_OK when
256 * the merge may proceed (warn-only) and NGX_ERROR when strict mode requires the
257 * caller to return NGX_CONF_ERROR. The caller owns the return so the diagnostic
258 * points at the offending location's own merge.
259 */
260static inline ngx_int_t
261brix_shared_security_gate(ngx_conf_t *cf, ngx_flag_t strict,
262 const char *what, const char *remedy)
263{
264 BRIX_CONF_LOG(strict ? NGX_LOG_EMERG : NGX_LOG_WARN, cf, 0,
265 "brix: insecure configuration — %s; set %s to close it%s",
266 what, remedy,
267 strict ? " (refused: brix_strict_security on)" : "");
268 return strict ? NGX_ERROR : NGX_OK;
269}

292/*
293 * brix_shared_thread_pool() — resolve the async-I/O thread pool for a merged
294 * common loc-conf, lazily and idempotently.
295 *
296 * WHY: postconfig only wires common.thread_pool for a *server-level* enabled
297 * loc-conf (webdav_postconf_setup_thread_pool walks cscf->ctx->loc_conf). A
298 * protocol enabled per-`location` (the common production shape for WebDAV TPC
299 * and PUT) leaves common.thread_pool NULL, so any path that only NULL-checks it
300 * falls back to a synchronous, event-loop-blocking transfer regardless of how
301 * many workers are configured. Resolving by name at first use — the pool is
302 * fully initialized by the time any request runs — closes that gap for every
303 * offload site with one shared helper (previously copy/paste'd into move.c,
304 * copy_collection.c, open_or_fill.c, http_serve_offload.c, http_cache_fill.c).
305 *
306 * HOW: returns the cached handle if already resolved; else looks it up by the
307 * configured name (default "default"), caches a hit back onto common, and
308 * returns it. NULL means no such pool exists — the caller must run synchronously.
309 */
310static inline ngx_thread_pool_t *
311brix_shared_thread_pool(ngx_http_brix_shared_conf_t *common)
312{
313 ngx_thread_pool_t *pool;
314
315 if (common == NULL) {
316 return NULL;
317 }
318
319 pool = common->thread_pool;
320 if (pool == NULL) {
321 static ngx_str_t default_name = ngx_string("default");
322 ngx_str_t *pname = common->thread_pool_name.len > 0
323 ? &common->thread_pool_name : &default_name;
324 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, pname);
325 if (pool != NULL) {
326 common->thread_pool = pool;
327 }
328 }
329
330 return pool;
331}
