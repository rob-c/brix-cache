1/*
2 * oci_key.c — the canonical cache key (§D2.3).
3 *
4 * WHAT: turn a classified route into the one string that is simultaneously
5 * the cache key, the export-relative store path, and the upstream URL
6 * suffix: "/v2/" [namespace "/"] name "/" terminal "/" reference.
7 * WHY: three consumers must agree on it byte for byte, and two of them
8 * RE-CLASSIFY it. brix_cache_verify_oci_digest() reads the reference
9 * back off the key to know which digest a fill must hash to, and
10 * sd_cache_is_manifest_key() reads it to know whether the object is a
11 * mutable tag manifest (TTL) or immutable content (forever). A key in
12 * any other shape — a hash, a flattened name, an Accept-qualified
13 * variant — silently disables both: the verify becomes a no-op and the
14 * tag manifest becomes permanent. So the key IS the route, and the one
15 * normalization applied (the DockerHub `library/` expansion) is applied
16 * HERE, before anyone reads it, rather than at each consumer.
17 * HOW: a single bounded append into ctx->key. No allocation: the classifier
18 * already validated every component, so this is concatenation, not
19 * parsing — the only failure is the cap.
20 */
21

24/* Whether `name` needs the operator's upstream namespace prefixed. A
25 * single-component name is the DockerHub shorthand ("alpine"), which every
26 * client resolves against an implicit namespace ("library/alpine") before it
27 * hits the wire; a multi-component name is already fully qualified and must
28 * be passed through untouched or it would name a different repository. */
29static int
30oci_needs_namespace(const ngx_http_brix_oci_loc_conf_t *lcf,
31 const brix_oci_req_t *req)
32{
33 return lcf->upstream_ns.len > 0 && req->name_components == 1;
34}

36ngx_int_t
37brix_oci_build_key(ngx_http_brix_oci_loc_conf_t *lcf,
38 ngx_http_brix_oci_ctx_t *ctx)
39{
40 const brix_oci_req_t *req = &ctx->req;
41 const char *terminal;
42 char digest[BRIX_OCI_DIGEST_STRLEN];
43 const char *ref;
44 size_t ref_len;
45 size_t need, off;
46
47 terminal = (req->cls == BRIX_OCI_REQ_MANIFEST) ? "manifests" : "blobs";
48
49 /* A blob route is named ONLY by its digest, and the classifier already
50 * parsed it — rebuild the canonical spelling from the parsed form rather
51 * than echoing the wire span, so a legal-but-unusual encoding cannot
52 * produce two keys for one object. A manifest reference is a tag or a
53 * digest and rides verbatim: it is the client's chosen name for the
54 * object, and the TTL classifier keys off exactly that distinction. */
55 if (req->cls == BRIX_OCI_REQ_BLOB) {
56 if (brix_oci_digest_format(&req->digest, digest, sizeof(digest)) < 0) {
57 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
58 }
59 ref = digest;
60 ref_len = ngx_strlen(digest);
61
62 } else {
63 ref = req->ref;
64 ref_len = req->ref_len;
65 }
66
67 need = sizeof("/v2/") - 1 + req->name_len + 1 + ngx_strlen(terminal) + 1
68 + ref_len;
69 if (oci_needs_namespace(lcf, req)) {
70 need += lcf->upstream_ns.len + 1;
71 }
72 if (need >= sizeof(ctx->key)) {
73 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
74 }
75
76 off = 0;
77 ngx_memcpy(ctx->key + off, "/v2/", sizeof("/v2/") - 1);
78 off += sizeof("/v2/") - 1;
79
80 if (oci_needs_namespace(lcf, req)) {
81 ngx_memcpy(ctx->key + off, lcf->upstream_ns.data, lcf->upstream_ns.len);
82 off += lcf->upstream_ns.len;
83 ctx->key[off++] = '/';
84 }
85
86 ngx_memcpy(ctx->key + off, req->name, req->name_len);
87 off += req->name_len;
88 ctx->key[off++] = '/';
89
90 ngx_memcpy(ctx->key + off, terminal, ngx_strlen(terminal));
91 off += ngx_strlen(terminal);
92 ctx->key[off++] = '/';
93
94 ngx_memcpy(ctx->key + off, ref, ref_len);
95 off += ref_len;
96 ctx->key[off] = '\0';
97
98 ctx->key_len = off;
99 return NGX_OK;
100}
