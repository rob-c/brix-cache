9/*
10 * ---- The data-plane I/O monitor: $brix_bytes_served / $brix_backend_time /
11 * $brix_checksum ------------------------------------------------------
12 *
13 * Unlike the identity/config variables above, these three report values the
14 * data plane produces while serving — bytes moved, backend time, page-CRC —
15 * which no HTTP plane retained per-request. The retention layer is a single
16 * brix_io_monitor_t on the request pool, allocated ON THE EVENT LOOP and stored
17 * in ngx_http_brix_common_module's request-ctx slot (the common module owns the
18 * variable surface, and no plane uses its ctx slot). The VFS post-op observer
19 * folds each op into it; these handlers read it at log time. See io_monitor.h
20 * for the single-writer/thread contract.
21 */
22
23/* Get-or-create the request's monitor. MUST be first called on the event loop
24 * (it allocates on r->pool). NULL only on allocation failure. */
25static brix_io_monitor_t *
26brix_http_monitor_get(ngx_http_request_t *r)
27{
28 brix_io_monitor_t *m = ngx_http_get_module_ctx(r, ngx_http_brix_common_module);
29
30 if (m != NULL) {
31 return m;
32 }
33 m = ngx_pcalloc(r->pool, sizeof(*m));
34 if (m == NULL) {
35 return NULL;
36 }
37 ngx_http_set_ctx(r, m, ngx_http_brix_common_module);
38 return m;
39}

42/* phase-110 W7: the client address as a NUL-terminated string on r->pool.
43 * r->connection->addr_text is NOT guaranteed NUL-terminated (ngx_sock_ntop
44 * writes the address but not a terminator), so copy it once — the established
45 * pattern (guard_audit_http.c). Event-loop only (allocates on r->pool). */
46static const char *
47brix_http_peer_cstr(ngx_http_request_t *r)
48{
49 ngx_connection_t *c;
50 u_char *p;
51 size_t n;
52
53 if (r == NULL || r->connection == NULL
54 || r->connection->addr_text.len == 0)
55 {
56 return NULL;
57 }
58 c = r->connection;
59 p = ngx_pnalloc(r->pool, c->addr_text.len + 1);
60 if (p == NULL) {
61 return NULL;
62 }
63 n = c->addr_text.len;
64 ngx_memcpy(p, c->addr_text.data, n);
65 p[n] = '\0';
66 return (const char *) p;
67}

70void
71brix_http_monitor_bind(ngx_http_request_t *r, brix_vfs_ctx_t *vctx)
72{
73 if (r == NULL || vctx == NULL) {
74 return;

98brix_http_monitor_record_served(ngx_http_request_t *r, off_t bytes)
99{
100 if (r == NULL || bytes <= 0) {
101 return;
102 }

112/*
113 * $brix_bytes_served — total bytes brix moved through its VFS for this request.
114 *
115 * The brix-measured figure, distinct from nginx's $body_bytes_sent (which is
116 * what left the socket, after range trimming and on-the-wire framing); logging
117 * both lets an operator see cache/backend read amplification. "-" when brix
118 * moved no bytes (a metadata request, a 304, a request brix never served).
119 * NOCACHEABLE: a data-plane outcome, unknown until the request is served.
120 */
121ngx_int_t
122brix_var_bytes_served(ngx_http_request_t *r, ngx_http_variable_value_t *v,
123 uintptr_t data)
124{
125 brix_io_monitor_t *m = brix_http_monitor_peek(r);
126 u_char *p;
127
128 (void) data;
129 /* "-" not "0": a request that ran a brix op but served nothing (a 404, a
130 * metadata op, a 304) served no bytes — that is "no serve happened", not a
131 * measured zero. Gate on the served count itself, not on `any`. */
132 if (m == NULL || m->bytes == 0) {
133 return brix_var_set_static(v, "-", 1);
134 }
135 p = ngx_pnalloc(r->pool, NGX_INT64_LEN);
136 if (p == NULL) {
137 return brix_var_set_static(v, "-", 1);
138 }
139 v->data = p;
140 v->len = (unsigned) (ngx_sprintf(p, "%uL", m->bytes) - p);
141 v->valid = 1;
142 v->no_cacheable = 1;
143 v->not_found = 0;
144 return NGX_OK;
145}

148/*
149 * $brix_backend_time — time brix spent in its own VFS I/O for this request,
150 * in SECONDS with millisecond precision, deliberately the SAME format as
151 * nginx's $request_time so the two sit side by side in a log line and subtract
152 * cleanly. It measures backend/storage service time (summed across the
153 * request's ops), which $request_time does not isolate — on a cache hit it is
154 * near zero, on a cold miss it is the origin fetch. "-" when brix did no I/O.
155 * NOCACHEABLE.
156 */
157ngx_int_t
158brix_var_backend_time(ngx_http_request_t *r, ngx_http_variable_value_t *v,
159 uintptr_t data)
160{
161 brix_io_monitor_t *m = brix_http_monitor_peek(r);
162 u_char *p;
163 uint64_t usec, sec, ms;
164
165 (void) data;
166 if (m == NULL || !m->any) {
167 return brix_var_set_static(v, "-", 1);
168 }
169 usec = (uint64_t) m->backend_usec;
170 sec = usec / 1000000ULL;
171 ms = (usec % 1000000ULL) / 1000ULL;
172 p = ngx_pnalloc(r->pool, NGX_INT64_LEN + sizeof(".000"));
173 if (p == NULL) {
174 return brix_var_set_static(v, "-", 1);
175 }
176 v->data = p;
177 v->len = (unsigned) (ngx_sprintf(p, "%uL.%03uL", sec, ms) - p);
178 v->valid = 1;
179 v->no_cacheable = 1;
180 v->not_found = 0;
181 return NGX_OK;
182}

200 * request (the WebDAV `Digest` response header, the root kXR_Qcksum reply),

280/*
281 * $brix_status — the brix OUTCOME CLASS of the request, plane-neutral, in the
282 * SAME vocabulary as the JSON access log's "status" and the io_ops{status}
283 * label (brix_metric_err_name): ok / not_found / forbidden / io / other.
284 * $status is an HTTP code on one plane and a stream code on the other, and a
285 * brix refusal that never became a response has neither; this is the one word
286 * that means the same thing everywhere (a read-only-export refusal is
287 * "forbidden" on WebDAV, S3 and root:// alike). Source: the outcome recorded
288 * with the primary op, else the mutation gate's / plane's explicit record,
289 * else — for a request brix refused before any VFS op ran (401/403/404 from
290 * the auth or resolve step) — the class of the HTTP status it answered with
291 * (brix_metric_err_from_http_status). "-" only when brix never ran at all.
292 * NOCACHEABLE.
293 */
294ngx_int_t
295brix_var_status(ngx_http_request_t *r, ngx_http_variable_value_t *v,
296 uintptr_t data)
297{
298 brix_io_monitor_t *m = brix_http_monitor_peek(r);
299 brix_err_class_t err;
300
301 (void) data;
302 if (m != NULL && (m->have_op || m->err != BRIX_ERR_NONE)) {
303 /* The primary op's own outcome — the most specific answer. */
304 err = m->err;
305 } else if (r->headers_out.status != 0
306 && brix_request_shared_conf(r) != NULL) {
307 /* brix owns this location but refused/served at the HTTP layer before
308 * any VFS op ran (a 403 allow_write refusal at the access phase, a 401
309 * auth refusal): classify the status brix answered with, so the outcome
310 * is never a bare "-" for a request brix demonstrably handled. */
311 err = brix_metric_err_from_http_status(r->headers_out.status);
312 } else {
313 return brix_var_set_static(v, "-", 1);
314 }
315 return brix_var_set_static(v, brix_metric_err_name(err), 1);
316}

389/*
390 * $brix_duration — wall time of the whole request, seconds.mmm, byte-identical
391 * to nginx's $request_time (same clock, same formula:
392 * ngx_http_variable_request_time). It exists for ONE reason: nginx spells this
393 * fact $request_time on HTTP and $session_time on stream, and a log_format
394 * that must serve both planes needs one name (phase-110 rule 6 — the only
395 * transport fact brix twins). Per-request cacheable within the log phase.
396 */
397ngx_int_t
398brix_var_duration(ngx_http_request_t *r, ngx_http_variable_value_t *v,
399 uintptr_t data)
400{
401 u_char *p;
402 ngx_time_t *tp;
403 ngx_msec_int_t ms;
404
405 (void) data;
406 p = ngx_pnalloc(r->pool, NGX_TIME_T_LEN + 4);
407 if (p == NULL) {
408 return brix_var_set_static(v, "-", 1);
409 }
410 tp = ngx_timeofday();
411 ms = (ngx_msec_int_t) ((tp->sec - r->start_sec) * 1000
412 + (tp->msec - r->start_msec));
413 ms = ngx_max(ms, 0);
414 v->data = p;
415 v->len = (unsigned) (ngx_sprintf(p, "%T.%03M", (time_t) ms / 1000,
416 ms % 1000) - p);
417 v->valid = 1;
418 v->no_cacheable = 0;
419 v->not_found = 0;
420 return NGX_OK;
421}
