14/*
15 * launch_push.c — F16, the SOURCE side of a native push (tpc.stage=push).
16 *
17 * A push read-open is an ORDINARY read-open that happens to name a remote
18 * destination: the client opens the local file for read with tpc.key / tpc.dst /
19 * tpc.dlfn, and the two kXR_syncs that follow arm and then fire the transfer.
20 * The role is decided in brix_open_handle_tpc(), which runs BEFORE path
21 * resolution, authz and open — so nothing here reimplements any of that.
22 * Instead brix_tpc_prepare_push runs the security ladder and PARKS the decision
23 * (brix_tpc_push_pending_t on ctx->tpc_push_pending, allocated from c->pool),
24 * and brix_open_finalize_handle() stamps it onto the handle the normal open
25 * produced. A parked intent that is never stamped is simply dropped with the
26 * connection pool.
27 */
28
29/*
30 * brix_tpc_push_pending_t — the push intent parked between the TPC role

33 * rendezvous key to present, and the (already clamped) stream count.

46/*
47 * brix_tpc_prepare_push — the kXR_open leg. Runs the operator opt-in
48 * (brix_tpc_push), the destination host/path validation, the source-egress
49 * allowlist (with its fail2ban signal and refusal counter) and the cached SSRF
50 * verdict on the host this server would DIAL, then parks the intent. An
51 * unknown name is NOT parked on DNS here: nothing has been dialled yet and the
52 * pull thread re-checks every address it actually connects to (I-DNS-3), so the
53 * open proceeds and an offending address is refused at connect time.
54 * Returns NGX_DECLINED when the intent is parked and the ordinary open must
55 * proceed, or NGX_OK once a refusal has been answered.
