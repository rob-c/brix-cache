35/* Parse the optional "rounds=N$" clause; advances *p past it when present. */
36static int
37sha_crypt_parse_rounds(const char **p, sha_crypt_setting_t *s)
38{
39 const char *q = *p;
40 char *end;
41 unsigned long n;
42
43 if (strncmp(q, "rounds=", 7) != 0) {
44 s->rounds = SHA_CRYPT_ROUNDS_DEF;
45 return 0;
46 }
47 n = strtoul(q + 7, &end, 10);
48 if (end == q + 7 || *end != '$') {
49 return -1;
50 }
51 if (n < SHA_CRYPT_ROUNDS_MIN) {
52 n = SHA_CRYPT_ROUNDS_MIN;
53 } else if (n > SHA_CRYPT_ROUNDS_MAX) {
54 n = SHA_CRYPT_ROUNDS_MAX;
55 }
56 s->rounds = n;
57 s->rounds_custom = 1;
58 *p = end + 1;
59 return 0;
60}

62static int
63sha_crypt_parse(const char *setting, sha_crypt_setting_t *s)
64{
65 const char *p;
66 size_t n;
67
68 memset(s, 0, sizeof(*s));
69 if (!brix_sha_crypt_handles(setting)) {
70 return -1;
71 }
72 s->id = setting[1];
73 s->md = (s->id == '6') ? EVP_sha512() : EVP_sha256();
74 p = setting + 3;
75 if (sha_crypt_parse_rounds(&p, s) != 0) {
76 return -1;
77 }
78 n = strcspn(p, "$");
79 s->salt = p;
80 s->salt_len = (n > SHA_CRYPT_SALT_MAX) ? SHA_CRYPT_SALT_MAX : n;
81 return 0;
82}

84/* Feed `len` bytes taken cyclically from a `srclen`-byte block. */
85static void
86sha_crypt_update_repeat(EVP_MD_CTX *ctx, const unsigned char *src,
87 size_t srclen, size_t len)
88{
89 while (len > 0) {
90 size_t take = (len > srclen) ? srclen : len;
91 EVP_DigestUpdate(ctx, src, take);
92 len -= take;
93 }

96/* Steps 1-8: the "alternate" digest A = H(key salt key). */
97static void
98sha_crypt_alt(const sha_crypt_setting_t *s, const unsigned char *key,
99 size_t keylen, unsigned char *alt)
100{
101 EVP_MD_CTX *ctx = EVP_MD_CTX_new();
102
103 EVP_DigestInit_ex(ctx, s->md, NULL);
104 EVP_DigestUpdate(ctx, key, keylen);
105 EVP_DigestUpdate(ctx, s->salt, s->salt_len);
106 EVP_DigestUpdate(ctx, key, keylen);
107 EVP_DigestFinal_ex(ctx, alt, NULL);
108 EVP_MD_CTX_free(ctx);
109}

111/* Steps 9-12: digest B from key, salt, A and the key-length bit pattern. */
112static void
113sha_crypt_intermediate(const sha_crypt_setting_t *s, const unsigned char *key,
114 size_t keylen, unsigned char *alt)
115{
116 EVP_MD_CTX *ctx = EVP_MD_CTX_new();
117 size_t dlen = (size_t) EVP_MD_size(s->md);
118 size_t cnt;
119
120 EVP_DigestInit_ex(ctx, s->md, NULL);
121 EVP_DigestUpdate(ctx, key, keylen);
122 EVP_DigestUpdate(ctx, s->salt, s->salt_len);
123 sha_crypt_update_repeat(ctx, alt, dlen, keylen);
124 for (cnt = keylen; cnt > 0; cnt >>= 1) {
125 if (cnt & 1) {
126 EVP_DigestUpdate(ctx, alt, dlen);
127 } else {
128 EVP_DigestUpdate(ctx, key, keylen);
129 }
130 }
131 EVP_DigestFinal_ex(ctx, alt, NULL);
132 EVP_MD_CTX_free(ctx);
133}

135/* Steps 13-15 / 17-19: DP = H(key x keylen), DS = H(salt x (16 + alt[0])). */
136static void
137sha_crypt_derive_block(const EVP_MD *md, const unsigned char *src,

151/* Step 21: the rounds loop, in place on `alt`. */
152static void
153sha_crypt_rounds(const sha_crypt_setting_t *s, unsigned char *alt,
154 const unsigned char *dp, size_t keylen, const unsigned char *ds)
155{
156 EVP_MD_CTX *ctx = EVP_MD_CTX_new();
157 size_t dlen = (size_t) EVP_MD_size(s->md);
158 unsigned long r;
159
160 for (r = 0; r < s->rounds; r++) {
161 EVP_DigestInit_ex(ctx, s->md, NULL);
162 if (r & 1) {
163 sha_crypt_update_repeat(ctx, dp, dlen, keylen);
164 } else {
165 EVP_DigestUpdate(ctx, alt, dlen);
166 }
167 if (r % 3) {
168 EVP_DigestUpdate(ctx, ds, s->salt_len);
169 }
170 if (r % 7) {
171 sha_crypt_update_repeat(ctx, dp, dlen, keylen);
172 }
173 if (r & 1) {
174 EVP_DigestUpdate(ctx, alt, dlen);
175 } else {
176 sha_crypt_update_repeat(ctx, dp, dlen, keylen);
177 }
178 EVP_DigestFinal_ex(ctx, alt, NULL);
179 }
180 EVP_MD_CTX_free(ctx);
181}

183/* Emit `n` base-64 characters of the 24-bit group (b2 b1 b0). */
184static char *
185sha_crypt_b64_24(char *out, unsigned b2, unsigned b1, unsigned b0, int n)
186{
187 unsigned w = (b2 << 16) | (b1 << 8) | b0;
188
189 while (n-- > 0) {
190 *out++ = sha_crypt_b64[w & 0x3f];
191 w >>= 6;
192 }
193 return out;
194}

231/* "$<id>$[rounds=N$]<salt>$" prefix; returns the write position or NULL. */
232static char *
233sha_crypt_prefix(const sha_crypt_setting_t *s, char *out, size_t outsz)
234{
235 int n;
236
237 if (s->rounds_custom) {
238 n = snprintf(out, outsz, "$%c$rounds=%lu$%.*s$", s->id, s->rounds,
239 (int) s->salt_len, s->salt);
240 } else {
241 n = snprintf(out, outsz, "$%c$%.*s$", s->id, (int) s->salt_len, s->salt);
242 }
243 if (n < 0 || (size_t) n + 87 > outsz) { /* longest hash (86) + NUL */
244 return NULL;
245 }
246 return out + n;

249int
250brix_sha_crypt(const char *key, const char *setting, char *out, size_t outsz)
251{
252 sha_crypt_setting_t s;
253 unsigned char alt[SHA_CRYPT_DIGEST_MAX];
254 unsigned char dp[SHA_CRYPT_DIGEST_MAX];
255 unsigned char ds[SHA_CRYPT_DIGEST_MAX];
256 const unsigned char *k = (const unsigned char *) key;
257 size_t keylen;
258 char *p;
259
260 if (key == NULL || out == NULL || sha_crypt_parse(setting, &s) != 0) {
261 return -1;
262 }
263 p = sha_crypt_prefix(&s, out, outsz);
264 if (p == NULL) {
265 return -1;
266 }
267 keylen = strlen(key);
268 sha_crypt_alt(&s, k, keylen, alt);
269 sha_crypt_intermediate(&s, k, keylen, alt);
270 sha_crypt_derive_block(s.md, k, keylen, keylen, dp);
271 sha_crypt_derive_block(s.md, (const unsigned char *) s.salt, s.salt_len,
272 16 + (size_t) alt[0], ds);
273 sha_crypt_rounds(&s, alt, dp, keylen, ds);
274 p = sha_crypt_encode(s.id, alt, p);
275 *p = '\0';
276 OPENSSL_cleanse(alt, sizeof(alt));
277 OPENSSL_cleanse(dp, sizeof(dp));
278 OPENSSL_cleanse(ds, sizeof(ds));
279 return 0;
