1/*
2 * oci_gate.c — method policing, route classification, locally-answered routes.
3 *
4 * WHAT: the first step of every request on an brix_oci_mirror location.
5 * Refuses write-class methods (a pull-through mirror is GET/HEAD only),
6 * classifies the URI through the shared kernel, answers the two routes
7 * that never touch an upstream object store (`GET /v2/` and an upload
8 * session that cannot exist here), routes tags/list to the uncached
9 * passthrough, and lets manifests and blobs fall through to the cache.
10 * WHY: a mirror that quietly accepted a push would be a supply-chain hole
11 * with a 200 on it: `podman push` would report success, the operator
12 * would believe the image was published, and nothing would hold it.
13 * So the refusal is loud on three channels at once — the spec's 405 +
14 * Allow for the client, the UNSUPPORTED envelope for the tooling, and
15 * one GUARD_R_OCIPUSH audit line for the operator's fail2ban jail.
16 * Classifying here (once, at the edge) is the other half: everything
17 * downstream builds paths out of `name` and `reference`, and this is
18 * the only place those wire bytes are ever validated.
19 * HOW: early-return ladder over the classifier verdict. NGX_DECLINED means
20 * "a cacheable object route" — the handler's tier path takes it from
21 * there; every other return is terminal and has already written its
22 * response.
23 */

36/* Classifier verdict → HTTP status. NAME_UNKNOWN is the one 404 in the set:
37 * the grammar was legal but named nothing this registry could route. */
38static ngx_uint_t
39oci_bad_status(brix_oci_err_t err)
40{
41 return (err == BRIX_OCI_ERR_NAME_UNKNOWN) ? NGX_HTTP_NOT_FOUND
42 : NGX_HTTP_BAD_REQUEST;
43}

45/* Route → metric traffic class (INVARIANT #8: a fixed enum, never the name). */
46static brix_oci_mclass_metric_e
47oci_mclass_of(brix_oci_class_t cls)
48{
49 switch (cls) {
50 case BRIX_OCI_REQ_API_ROOT: return BRIX_OCI_MCLASS_API;
51 case BRIX_OCI_REQ_MANIFEST: return BRIX_OCI_MCLASS_MANIFEST;
52 case BRIX_OCI_REQ_BLOB: return BRIX_OCI_MCLASS_BLOB;
53 case BRIX_OCI_REQ_UPLOAD_START: return BRIX_OCI_MCLASS_UPLOAD;
54 case BRIX_OCI_REQ_UPLOAD_SESSION: return BRIX_OCI_MCLASS_UPLOAD;
55 case BRIX_OCI_REQ_TAGS_LIST: return BRIX_OCI_MCLASS_TAGS;
56 case BRIX_OCI_REQ_REFERRERS: return BRIX_OCI_MCLASS_REFERRERS;
57 default: return BRIX_OCI_MCLASS_BAD;
58 }
59}

61/* A write-class method aimed at a read-only mirror. Three channels, one
62 * event: the spec's 405 + Allow, the UNSUPPORTED envelope, and the audit
63 * line the [brix-oci-push] jail bans on. */
64static ngx_int_t
65oci_refuse_push(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx)
66{
67 ctx->disp = BRIX_OCI_OUT_REFUSED;
68 ctx->mclass = BRIX_OCI_MCLASS_BAD;
69
70 brix_oci_guard_emit(r, GUARD_R_OCIPUSH, GUARD_OP_WRITE,
71 NGX_HTTP_NOT_ALLOWED);
72
73 if (brix_http_set_header(r, "Allow", "GET, HEAD", NULL) != NGX_OK) {
74 return NGX_HTTP_INTERNAL_SERVER_ERROR;
75 }
76 return brix_oci_error(r, NGX_HTTP_NOT_ALLOWED, BRIX_OCI_ERR_UNSUPPORTED,
77 "this endpoint is a read-only pull-through mirror");
78}

80/* `GET /v2/` — answered here, with zero upstream traffic. The probe is what
81 * every client issues before its first pull; forwarding it would multiply
82 * one image pull into two upstream round-trips and, worse, would make the
83 * mirror's own availability depend on the upstream being reachable. */
84static ngx_int_t
85oci_answer_api_root(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx)
86{
87 ctx->disp = BRIX_OCI_OUT_LOCAL;
88
89 return brix_oci_send_body(r, NGX_HTTP_OK, "application/json",
90 oci_api_root_body,
91 sizeof(oci_api_root_body) - 1);
92}

94/* Validate and classify r->uri into ctx->req. NGX_OK, or a terminal rc whose
95 * response has already been written. Shared by both surfaces: the grammar is
96 * the same on the read and the write side, and two copies of it would be two
97 * chances to disagree about what a legal repository name is.
98 *
99 * The spans in ctx->req point into `uri`, so it is the CALLER's buffer — a
100 * classifier verdict does not outlive the storage it describes. */
101static ngx_int_t
102oci_classify_uri(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
103 char *uri, size_t uri_size)
104{
105 size_t n;
106
107 /* The classifier is a pure kernel over a NUL-terminated span; r->uri is
108 * neither NUL-terminated nor bounded by our key cap. A URI that cannot
109 * fit a legal route cannot BE a legal route. */
110 if (r->uri.len >= uri_size) {
111 ctx->disp = BRIX_OCI_OUT_REFUSED;
112 ctx->mclass = BRIX_OCI_MCLASS_BAD;
113 return brix_oci_error(r, NGX_HTTP_REQUEST_URI_TOO_LARGE,
114 BRIX_OCI_ERR_NAME_INVALID, NULL);
115 }
116 n = r->uri.len;
117 ngx_memcpy(uri, r->uri.data, n);
118 uri[n] = '\0';
119
120 if (brix_oci_classify(uri, n, &ctx->req) != 0) {
121 ctx->classified = 1;
122 ctx->mclass = BRIX_OCI_MCLASS_BAD;
123 ctx->disp = BRIX_OCI_OUT_REFUSED;
124 return brix_oci_error(r, oci_bad_status(ctx->req.err), ctx->req.err,
125 NULL);
126 }
127 ctx->classified = 1;
128 ctx->mclass = oci_mclass_of(ctx->req.cls);
129
130 return NGX_OK;
131}

133ngx_int_t
134brix_oci_gate(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
135 ngx_http_brix_oci_ctx_t *ctx)
136{
137 char uri[BRIX_OCI_KEY_MAX];
138 ngx_int_t rc;
139
140 /* GET/HEAD only. Everything else — PUT, POST, PATCH, DELETE and the
141 * long tail — is a push attempt or a probe, and both get the same
142 * answer. Ordering matters: police the METHOD before parsing the URI,
143 * so a push at a malformed path is still audited as a push. */
144 if (!(r->method & (NGX_HTTP_GET | NGX_HTTP_HEAD))) {
145 return oci_refuse_push(r, ctx);
146 }
147
148 rc = oci_classify_uri(r, ctx, uri, sizeof(uri));
149 if (rc != NGX_OK) {
150 return rc;
151 }
152
153 switch (ctx->req.cls) {
154
155 case BRIX_OCI_REQ_API_ROOT:
156 return oci_answer_api_root(r, ctx);
157
158 case BRIX_OCI_REQ_TAGS_LIST:
159 case BRIX_OCI_REQ_REFERRERS:
160 /* Never cached: both are mutable listings whose staleness the client
161 * cannot detect. Forwarded verbatim, pagination and filters and all.
162 * Gated FIRST in delegate mode — a listing is exactly where a
163 * private repository's metadata would otherwise leak (D16), and
164 * ctx->req's name span is only alive inside this frame. */
165 rc = brix_oci_delegate_gate(r, lcf, ctx);
166 if (rc != NGX_DECLINED) {
167 return rc;
168 }
169 return brix_oci_listing_passthrough(r, lcf, ctx);
170
171 case BRIX_OCI_REQ_UPLOAD_START:
172 case BRIX_OCI_REQ_UPLOAD_SESSION:
173 /* A GET here is a client resuming an upload it believes it started.
174 * On a mirror it never did — and saying "unknown session" is both
175 * true and exactly what makes the client abandon cleanly instead of
176 * retrying its PATCH loop against a surface that will 405 it. */
177 ctx->disp = BRIX_OCI_OUT_REFUSED;
178 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
179 BRIX_OCI_ERR_BLOB_UPLOAD_UNKNOWN, NULL);
180
181 case BRIX_OCI_REQ_MANIFEST:
182 case BRIX_OCI_REQ_BLOB:
183 break; /* the cacheable object routes */
184
185 default:
186 ctx->mclass = BRIX_OCI_MCLASS_BAD;
187 ctx->disp = BRIX_OCI_OUT_REFUSED;
188 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
189 BRIX_OCI_ERR_NAME_UNKNOWN, NULL);
190 }
191
192 rc = brix_oci_build_key(lcf, ctx);
193 if (rc != NGX_OK) {
194 ctx->disp = BRIX_OCI_OUT_REFUSED;
195 return brix_oci_error(r, (ngx_uint_t) rc, BRIX_OCI_ERR_NAME_INVALID,
196 "canonical route exceeds the key cap");
197 }
198 ctx->keyed = 1;
199
200 return NGX_DECLINED;
201}

204/* The registry surface's gate. Same grammar, no method refusal: here a POST
205 * IS the point. The API root is still answered locally — a push client probes
206 * `/v2/` exactly like a pull client does, and it must get the same answer
207 * before it has any credential to present. */
208ngx_int_t
209brix_oci_registry_gate(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
210 char *uri, size_t uri_size)
211{
212 ngx_int_t rc;
213
214 rc = oci_classify_uri(r, ctx, uri, uri_size);
215 if (rc != NGX_OK) {
216 return rc;
217 }
218 if (ctx->req.cls == BRIX_OCI_REQ_API_ROOT) {
219 return oci_answer_api_root(r, ctx);
220 }
221 return NGX_DECLINED;
222}
