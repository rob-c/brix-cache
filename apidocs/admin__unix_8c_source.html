28/* Length of `line` with any trailing CR/LF run stripped. */
29static size_t
30admin_unix_strip_eol(const u_char *line, size_t len)
31{
32 while (len > 0 && (line[len - 1] == '\n' || line[len - 1] == '\r')) {
33 len--;
34 }
35 return len;
36}

38/*
39 * Match one line against one verb. A bare verb must match the whole line; a
40 * verb taking operands must be followed by a space AND at least one operand
41 * byte, so "disc" alone is unknown-command rather than a disc of the empty
42 * session id.
43 */
44static int
45admin_unix_verb_match(const brix_admin_verb_t *v, const u_char *line,
46 size_t len, size_t *args_off)
47{
48 if (len < v->nlen || ngx_strncmp(line, v->name, v->nlen) != 0) {
49 return 0;
50 }
51 if (!v->wants_args) {
52 return len == v->nlen;
53 }
54 if (len <= v->nlen + 1 || line[v->nlen] != ' ') {
55 return 0;
56 }
57 *args_off = v->nlen + 1;
58 return 1;
59}

61/* Dispatch one complete command line into a pool-allocated reply. */
62static void
63admin_unix_dispatch(ngx_connection_t *c, admin_unix_conn_t *st)
64{
65 const brix_admin_unix_t *srv = st->srv;
66 size_t len = admin_unix_strip_eol(st->cmd, st->cmd_len);
67 size_t off = 0;
68 ngx_uint_t i;
69
70 st->rep.buf = ngx_pnalloc(c->pool, BRIX_ADMIN_REPLY_MAX);
71 if (st->rep.buf == NULL) {
72 return;
73 }
74 st->rep.cap = BRIX_ADMIN_REPLY_MAX;
75 st->rep.out = st->rep.buf;
76 st->rep.len = 0;
77 st->sent = 0;
78
79 for (i = 0; i < srv->nverbs; i++) {
80 if (admin_unix_verb_match(&srv->verbs[i], st->cmd, len, &off)) {
81 srv->verbs[i].handler(srv->ud, off ? st->cmd + off : NULL,
82 off ? len - off : 0, &st->rep);
83 return;
84 }
85 }
86 brix_admin_reply_set(&st->rep, "err unknown-command\n");
87}

102static void
103admin_unix_write_handler(ngx_event_t *wev)
104{
105 ngx_connection_t *c = wev->data;
106 admin_unix_conn_t *st = c->data;
107 ssize_t n;
108
109 while (st->sent < st->rep.len) {
110 n = c->send(c, st->rep.out + st->sent, st->rep.len - st->sent);
111 if (n > 0) {
112 st->sent += (size_t) n;
113 continue;
114 }
115 if (n == NGX_AGAIN) {
116 if (ngx_handle_write_event(wev, 0) != NGX_OK) {
117 admin_unix_close(c);
118 }
119 return;
120 }
121 admin_unix_close(c);
122 return;
123 }
124
125 /* Reply fully flushed — reset for the next command on the same conn. */
126 st->rep.len = 0;
127 st->sent = 0;
128 st->cmd_len = 0;
129}

131static void
132admin_unix_read_handler(ngx_event_t *rev)
133{
134 ngx_connection_t *c = rev->data;
135 admin_unix_conn_t *st = c->data;
136 ssize_t n;
137
138 for ( ;; ) {
139 if (st->cmd_len >= sizeof(st->cmd)) {
140 admin_unix_close(c); /* oversized command line — refuse */
141 return;
142 }
143 n = c->recv(c, st->cmd + st->cmd_len, sizeof(st->cmd) - st->cmd_len);
144 if (n > 0) {
145 st->cmd_len += (size_t) n;
146 if (ngx_strlchr(st->cmd, st->cmd + st->cmd_len, '\n') != NULL) {
147 admin_unix_dispatch(c, st);
148 admin_unix_write_handler(c->write);
149 return; /* one command per read pass; next arrives as data */
150 }
151 continue;
152 }
153 if (n == NGX_AGAIN) {
154 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
155 admin_unix_close(c);
156 }
157 return;
158 }
159 admin_unix_close(c); /* EOF or error */
160 return;
161 }
162}

164/* Wire one accepted fd into a connection with its own pool and state. */
165static void
166admin_unix_setup_conn(ngx_connection_t *c, ngx_event_t *rev)
167{
168 admin_unix_conn_t *st;
169
170 /* ngx_get_connection() hands out a BARE connection: the I/O vtable is
171 * ngx_event_accept()'s job, so wire it here or c->recv/c->send are
172 * NULL function pointers. */
173 c->recv = ngx_recv;
174 c->send = ngx_send;
175 c->recv_chain = ngx_recv_chain;
176 c->send_chain = ngx_send_chain;
177 c->log = rev->log;
178 c->read->log = c->log;
179 c->write->log = c->log;
180 c->type = SOCK_STREAM;
181 c->pool = ngx_create_pool(BRIX_ADMIN_POOL_SIZE, rev->log);
182 if (c->pool == NULL) {
183 ngx_close_connection(c);
184 return;
185 }
186 st = ngx_pcalloc(c->pool, sizeof(*st));
187 if (st == NULL) {
188 admin_unix_close(c);
189 return;
190 }
191 st->srv = ((ngx_connection_t *) rev->data)->data;
192 c->data = st;
193 c->read->handler = admin_unix_read_handler;
194 c->write->handler = admin_unix_write_handler;
195 if (ngx_handle_read_event(c->read, 0) != NGX_OK) {
196 admin_unix_close(c);
197 }
198}

200static void
201admin_unix_accept_handler(ngx_event_t *rev)
202{
203 ngx_connection_t *lc = rev->data;
204 ngx_socket_t fd;
205 ngx_connection_t *c;
206
207 for ( ;; ) {
208 fd = accept(lc->fd, NULL, NULL);
209 if (fd == (ngx_socket_t) -1) {
210 return; /* EAGAIN or transient — nothing more to accept */
211 }
212 if (ngx_nonblocking(fd) == -1) {
213 (void) close(fd);
214 continue;
215 }
216 c = ngx_get_connection(fd, rev->log);
217 if (c == NULL) {
218 (void) close(fd);
219 continue;
220 }
221 admin_unix_setup_conn(c, rev);
222 }
223}

227/*
228 * This worker's socket path. Worker 0 serves the configured <path>; worker n
229 * serves "<path>.<n>". Returns the length, or -1 when the result would exceed
230 * sun_path (a silently truncated path would be a DIFFERENT socket).
231 */
232static int
233admin_unix_worker_path(char *out, size_t cap, const char *path)
234{
235 int n;
236
237 if (ngx_worker == 0) {
238 n = snprintf(out, cap, "%s", path);
239 } else {
240 n = snprintf(out, cap, "%s.%lu", path, (unsigned long) ngx_worker);
241 }
242 return (n < 0 || (size_t) n >= cap) ? -1 : n;
243}

245/* Bind + chmod + listen a fresh AF_UNIX socket; -1 on any failure. */
246static ngx_socket_t
247admin_unix_bind(const char *path, size_t plen, ngx_log_t *log)
248{
249 struct sockaddr_un sa;
250 ngx_socket_t fd;
251
252 fd = socket(AF_UNIX, SOCK_STREAM, 0);
253 if (fd == (ngx_socket_t) -1) {
254 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
255 "brix admin: socket(%s) failed", path);
256 return (ngx_socket_t) -1;
257 }
258
259 ngx_memzero(&sa, sizeof(sa));
260 sa.sun_family = AF_UNIX;
261 ngx_memcpy(sa.sun_path, path, plen + 1);
262
263 if (bind(fd, (struct sockaddr *) &sa, sizeof(sa)) == -1
264 || chmod(path, 0600) == -1 /* vfs-seam-allow: NOT_STORAGE — admin socket 0600 privilege boundary */
265 || listen(fd, 8) == -1
266 || ngx_nonblocking(fd) == -1)
267 {
268 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
269 "brix admin: bind/listen(%s) failed", path);
270 (void) close(fd);
271 return (ngx_socket_t) -1;
272 }
273 return fd;
274}

276void
277brix_admin_unix_listen(ngx_cycle_t *cycle, const char *path,
278 const brix_admin_unix_t *srv)
279{
280 char wpath[sizeof(((struct sockaddr_un *) 0)->sun_path)];
281 ngx_socket_t fd;
282 ngx_connection_t *lc;
283 int n;
284
285 if (path == NULL || path[0] == '\0') {
286 return;
287 }
288
289 n = admin_unix_worker_path(wpath, sizeof(wpath), path);
290 if (n < 0) {
291 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
292 "brix %s: per-worker path for \"%s\" exceeds unix "
293 "socket path limit — socket disabled on worker %ui",
294 srv->label, path, ngx_worker);
295 return;
296 }
297
298 (void) unlink(wpath); /* drop a stale socket from a prior run */ /* vfs-seam-allow: NOT_STORAGE — admin unix socket, not export namespace */
299
300 fd = admin_unix_bind(wpath, (size_t) n, cycle->log);
301 if (fd == (ngx_socket_t) -1) {
302 return;
303 }
304
305 lc = ngx_get_connection(fd, cycle->log);
306 if (lc == NULL) {
307 (void) close(fd);
308 return;
309 }
310 lc->data = (void *) srv; /* the accept handler's route to the verbs */
311 /* Bare connections have zeroed events; debug event registration logs. */
312 lc->read->log = lc->log;
313 lc->write->log = lc->log;
314 lc->read->handler = admin_unix_accept_handler;
315 if (ngx_handle_read_event(lc->read, 0) != NGX_OK) {
316 ngx_close_connection(lc);
317 return;
318 }
319
320 ngx_log_error(NGX_LOG_INFO, cycle->log, 0,
321 "brix %s: listening on \"%s\" (worker %ui)", srv->label,
322 wpath, ngx_worker);
323}
