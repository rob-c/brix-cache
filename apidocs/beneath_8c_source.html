1/*
2 * beneath.c — kernel-enforced path-confinement primitives (openat2/RESOLVE_BENEATH).
3 *
4 * WHAT: Implements the "beneath" API declared in beneath.h: a small set of
5 * filesystem primitives (open/stat/unlink/mkdir/rename/link) that operate
6 * on a client-supplied reqpath but are GUARANTEED never to escape a
7 * per-worker O_PATH rootfd anchored on the export root. The core wrapper
8 * do_openat2() drives the openat2(2) syscall with RESOLVE_BENEATH |
9 * RESOLVE_NO_MAGICLINKS; brix_open_beneath()/brix_stat_beneath() use it
10 * directly, while the path-MUTATING ops route through beneath_open_parent().
11 *
12 * WHY: This layer IS the confinement boundary (see op_path.c: the lexical
13 * 'resolved' string is for ACL/logging only — it is NOT the security
14 * boundary). XRootD/WebDAV/S3 clients send arbitrary paths; the kernel,
15 * not nginx code, must refuse any traversal out of the export root.
16 * RESOLVE_BENEATH rejects absolute paths, "..", and symlinks that would
17 * leave the rootfd subtree, atomically and without TOCTOU races.
18 * RESOLVE_NO_MAGICLINKS additionally blocks /proc/<pid>/fd-style escapes.
19 * An escape attempt surfaces to callers as EXDEV/ELOOP, which the error
20 * mapper turns into kXR_NotAuthorized / HTTP 403.
21 *
22 * HOW: openat2(2) is a hard build requirement (the #error guards below fail the
23 * compile on kernels/headers < 5.6). do_openat2() masks O_CREAT mode bits
24 * to 07777 because openat2() — unlike open(2) — rejects S_IFMT type bits.
25 * The crucial subtlety: RESOLVE_BENEATH protects ONLY the openat2() call
26 * itself; the legacy *at() syscalls (mkdirat/unlinkat/renameat/linkat) do
27 * NOT honour it, so a symlink in an intermediate component could be
28 * followed out of the root. beneath_open_parent() closes that hole by
29 * resolving the PARENT directory under RESOLVE_BENEATH and then performing
30 * the *at() op on the final component name only (which those syscalls do
31 * not dereference as a symlink for delete/rename/create).
32 */

52/*
53 * IMPERSONATION SEAM (phase 40).
54 *
55 * When `brix_idmap map` is active and a per-request principal is set,
56 * brix_imp_client_active() returns true and the confined helpers below delegate
57 * the open/metadata syscall to the privileged broker, which performs it as the
58 * mapped UNIX user under its OWN rootfd (the kernel enforces RESOLVE_BENEATH
59 * there too). In that mode the worker's local `rootfd` argument is unused — the
60 * broker is the confinement authority — but the helper signatures are unchanged,
61 * so no caller is touched. When impersonation is off (the default) or no
62 * principal is set, brix_imp_client_active() is false and the original local
63 * openat2 path below runs exactly as before.

74/*
75 * Open an O_PATH directory fd on root_canon to anchor every beneath() call.
76 * O_PATH gives a traversal-only handle (no read/write to the dir itself),
77 * O_DIRECTORY enforces it is a directory, O_CLOEXEC stops fd leaks across exec.
78 * Returns fd >= 0 (caller closes) or -1 with errno set. Hot paths should reuse
79 * a persistent per-worker rootfd rather than calling this per request.

81int

83{

87/*
88 * do_openat2 — the single chokepoint through which every confined open passes.
89 * `rel` is a root-RELATIVE path (no leading '/'); RESOLVE_BENEATH treats rootfd
90 * as the top of the tree and refuses to follow anything — "..", absolute paths,
91 * or symlinks — that would resolve outside it, returning EXDEV/ELOOP instead.
92 * RESOLVE_NO_MAGICLINKS additionally blocks /proc/<pid>/fd magic symlinks.
93 * NOTE: this guarantee applies ONLY to the open performed here; mutating *at()
94 * ops must pre-resolve their parent via beneath_open_parent() (see below).
95 */
96static int
97do_openat2_resolve(int rootfd, const char *rel, int flags, mode_t mode,
98 uint64_t resolve)
99{
100 struct open_how how;
101 /* empty rel means root dir itself; "." opens the rootfd directory */
102 if (rel[0] == '\0') { rel = "."; }
103 ngx_memzero(&how, sizeof(how));
104 /*
105 * Force O_NONBLOCK on real (data-bearing) opens so the open(2) itself can
106 * NEVER block the worker. Opening a FIFO O_RDONLY (or a device with a
107 * blocking open routine) without O_NONBLOCK parks the caller in the kernel
108 * "wait_for_partner" state until a peer appears — which, on a single-threaded
109 * nginx worker, freezes the event loop and stalls EVERY connection pinned to
110 * that worker (a remotely triggerable DoS: a directory entry that is a named
111 * pipe wedges the worker on the kXR_open / GET path). The flag is a no-op
112 * for the regular files we serve (local-fs reads never return EAGAIN), and
113 * callers reject any non-regular object after fstat()ing the fd.
114 *
115 * It must NOT be added to an O_PATH open: openat2() is stricter than open(2)
116 * and rejects O_PATH|O_NONBLOCK with EINVAL. O_PATH opens only reference the
117 * inode (stat/lstat, the persistent root anchor, parent resolution for
118 * mutating ops) and never start file I/O, so they cannot block on a FIFO and
119 * need no guard.
120 */
121 how.flags = (uint64_t)(flags | O_CLOEXEC
122 | ((flags & O_PATH) ? 0 : O_NONBLOCK));
123 how.resolve = resolve;
124 /*
125 * openat2() is stricter than open()/openat(): it rejects (EINVAL) any
126 * how.mode bit outside 07777. Callers legitimately pass a full struct
127 * stat st_mode (e.g. staged_file copying a source's permissions), which
128 * carries the S_IFMT type bits. Mask to the permission bits, exactly as
129 * open(2) does by ignoring the type bits, so a struct-stat mode is accepted.
130 */
131 if (flags & O_CREAT) { how.mode = (uint64_t)(mode & 07777); }
132 return (int)syscall(SYS_openat2, rootfd, rel, &how, sizeof(how));
133}

135/* The default confinement for open/mutate paths: RESOLVE_BENEATH (no symlinks,
136 * no "..", no absolute paths) + RESOLVE_NO_MAGICLINKS (refuse /proc/PID/fd). */
137static int
138do_openat2(int rootfd, const char *rel, int flags, mode_t mode)
139{
140 return do_openat2_resolve(rootfd, rel, flags, mode,
141 RESOLVE_BENEATH | RESOLVE_NO_MAGICLINKS);
142}

153int
154brix_stat_beneath(int rootfd, const char *reqpath, struct stat *st)
155{
156 int fd, rc;
157
158 if (brix_imp_client_active()) {
159 return brix_imp_stat(reqpath, st, 0 /* follow */);
160 }
161 /* O_PATH without O_NOFOLLOW: follow symlinks. RESOLVE_IN_ROOT (not BENEATH)
162 * so an in-export symlink — including one with an ABSOLUTE target that points
163 * back into the export — is followed chroot-style (the kernel resolves "/" and
164 * ".." relative to rootfd) and still CANNOT escape the root. BENEATH rejected
165 * every absolute/symlink target outright (EXDEV), diverging from stock, which
166 * follows in-export links (test_conf_stattypes). */
167 fd = do_openat2_resolve(rootfd, brix_beneath_rel(reqpath), O_PATH, 0,
168 RESOLVE_IN_ROOT | RESOLVE_NO_MAGICLINKS);
169 if (fd < 0) { return -1; }
170 rc = fstat(fd, st);
171 close(fd);
172 return rc;
173}

175int
176brix_lstat_beneath(int rootfd, const char *reqpath, struct stat *st)
177{
178 int fd, rc;
179
180 if (brix_imp_client_active()) {
181 return brix_imp_stat(reqpath, st, 1 /* nofollow */);
182 }
183 /* O_PATH | O_NOFOLLOW: do NOT follow a trailing symlink, so fstat() reports
184 * the link itself (lstat semantics). RESOLVE_IN_ROOT (chroot-style) confines
185 * the path while still permitting in-export symlinks in INTERMEDIATE
186 * components; the trailing link is opened as itself via O_NOFOLLOW. (BENEATH
187 * rejected even a trailing symlink here with EXDEV, so lstat of a link failed.) */
188 fd = do_openat2_resolve(rootfd, brix_beneath_rel(reqpath),
189 O_PATH | O_NOFOLLOW, 0,
190 RESOLVE_IN_ROOT | RESOLVE_NO_MAGICLINKS);
191 if (fd < 0) { return -1; }
192 rc = fstat(fd, st);
193 close(fd);
194 return rc;
195}

197DIR *
198brix_opendir_beneath(int rootfd, const char *reqpath)
199{
200 int fd;
201 DIR *d;
202
203 /* O_RDONLY | O_DIRECTORY under RESOLVE_IN_ROOT (chroot-style): a trailing
204 * in-export directory symlink is followed within the export — matching stock,
205 * which lists through in-export links — while an outward/absolute target is
206 * resolved relative to rootfd and so lands on a non-existent in-root path
207 * (ENOENT) or is refused (EXDEV) rather than escaping the export root. The
208 * legacy bare opendir() followed an outward link straight out of the root. */
209 fd = do_openat2_resolve(rootfd, brix_beneath_rel(reqpath),
210 O_RDONLY | O_DIRECTORY, 0,
211 RESOLVE_IN_ROOT | RESOLVE_NO_MAGICLINKS);
212 if (fd < 0) { return NULL; }
213 d = fdopendir(fd);
214 if (d == NULL) { close(fd); }
215 return d; /* closedir() closes the fd */
216}

218/*
219 * SECURITY: the *at() family (mkdirat / unlinkat / renameat / linkat) does NOT
220 * honour RESOLVE_BENEATH — only openat2() does. Handing them a multi-component
221 * relative path lets a symlink in an INTERMEDIATE component (e.g. /link -> /etc,
222 * then "/link/x") be followed straight out of the export root. openat2()'s
223 * RESOLVE_BENEATH only guards the open itself, so for every path-MUTATING op we
224 * must resolve the PARENT directory ourselves under RESOLVE_BENEATH and then
225 * operate on the final component name only. The final component is safe for
226 * these ops: unlinkat/renameat never follow a trailing symlink, and mkdir/link
227 * create a fresh name.
228 *
229 * beneath_open_parent() returns an O_PATH fd on the confined parent of reqpath
230 * — rootfd ITSELF (borrowed) when the parent is the root — and points *base
231 * at the final component (into reqpath). Callers must release the fd with
232 * beneath_close_parent(), never a bare close(). pbuf must be >= PATH_MAX.
233 * Returns -1 (errno: EXDEV/ELOOP on escape).
234 */
235static int
236beneath_open_parent(int rootfd, const char *reqpath,
237 char *pbuf, size_t pbufsz, const char **base)
238{
239 const char *rel = brix_beneath_rel(reqpath); /* strip leading '/'s */
240 const char *slash = strrchr(rel, '/');
241 size_t plen;
242
243 if (slash == NULL || slash == rel) {
244 /* Single component, or only a leading-slash remnant → the parent IS
245 * the root: hand back rootfd itself, BORROWED — callers release via
246 * beneath_close_parent(), which never closes it. Re-opening the root
247 * as "." only re-derived an fd we already hold, at an openat2+close
248 * per top-level mutation. */
249 *base = (slash == NULL) ? rel : slash + 1;
250 return rootfd;
251 }
252
253 *base = slash + 1;
254 plen = (size_t) (slash - rel);
255 if (plen >= pbufsz) {
256 errno = ENAMETOOLONG;
257 return -1;
258 }
259 memcpy(pbuf, rel, plen);
260 pbuf[plen] = '\0';
261 return do_openat2(rootfd, pbuf, O_PATH | O_DIRECTORY, 0);
262}

264/* Release a parent fd from beneath_open_parent(), preserving errno; a no-op
265 * when the parent was the borrowed rootfd itself. */
266static void
267beneath_close_parent(int pfd, int rootfd)
268{
269 int e;
270
271 if (pfd != rootfd) {
272 e = errno;
273 close(pfd);
274 errno = e;
275 }
276}

278/* Strip trailing '/' (keeping a lone root) into norm[] and return the
279 * normalized path: a trailing slash denotes the same object, and without this
280 * beneath_open_parent() would split "name/" into parent="name" + an empty
281 * leaf. Two callers hit it from opposite directions: globus-url-copy -cd
282 * issues `MKD interop/` (parent does not exist yet -> ENOENT), and an S3
283 * folder marker is addressed as "key/", so DeleteObject of one — and any
284 * rmdir arriving through it — died with EINVAL. Matches the recursive
285 * mkdir path's brix_mkdir_normalise(). NULL/ENAMETOOLONG when it cannot fit. */
286static const char *
287beneath_strip_trailing_slash(const char *reqpath, char *norm, size_t cap)
288{
289 size_t n;
290
291 n = strlen(reqpath);
292 if (n >= cap) { errno = ENAMETOOLONG; return NULL; }
293 memcpy(norm, reqpath, n + 1);
294 while (n > 1 && norm[n - 1] == '/') { norm[--n] = '\0'; }
295 return norm;
296}

300{
301 char pbuf[PATH_MAX];
302 char norm[PATH_MAX];
303 const char *base;

305

319 beneath_close_parent(pfd, rootfd);

363/*
364 * renameat2(RENAME_NOREPLACE) on already-resolved parent fds + final
365 * components. On a kernel/filesystem without RENAME_NOREPLACE
366 * (ENOSYS/EINVAL) it falls back to a plain renameat — logged once — so
367 * behaviour degrades to the legacy last-writer-wins rather than spuriously
368 * failing (callers still ran their stat-based precondition, so this is no
369 * worse than before W6b on such hosts). Shared with the confined-parent
370 * rename in resolve_confined_ops.c.
371 */
372int
373brix_renameat_noreplace_fallback(ngx_log_t *log, int sfd, const char *sbase,
374 int dfd, const char *dbase)
375{
376 int rc = (int) syscall(SYS_renameat2, sfd, sbase, dfd, dbase,
377 (unsigned int) RENAME_NOREPLACE);
378
379 if (rc != 0 && (errno == ENOSYS || errno == EINVAL)) {
380 noreplace_degraded = 1;
381 if (!noreplace_warned) {
382 noreplace_warned = 1;
383 ngx_log_error(NGX_LOG_WARN, log, errno,
384 "brix: renameat2(RENAME_NOREPLACE) unsupported; "
385 "create-if-absent falls back to non-atomic rename");
386 }
387 /* phase74-fp: prototype order (olddirfd, oldpath, newdirfd, newpath). */
388 rc = renameat(sfd, sbase, dfd, dbase); /* NOLINT(readability-suspicious-call-argument) */
389 }
390 return rc;
391}

393/* The two-path mutating ops share one confined body: impersonation dispatch,
394 * RESOLVE_BENEATH resolution of BOTH parents, empty-component rejection, the
395 * op on the final components, and fd cleanup with errno preserved. */
396typedef enum {
397 BENEATH_2P_RENAME,
398 BENEATH_2P_RENAME_EXCL,
399 BENEATH_2P_EXCHANGE,
400 BENEATH_2P_LINK,
401} beneath_two_path_op_t;
402
403static int
404beneath_two_path(beneath_two_path_op_t op, int rootfd, const char *src,
405 const char *dst)
406{
407 char sbuf[PATH_MAX], dbuf[PATH_MAX];
408 const char *sbase, *dbase;
409 int sfd, dfd, rc;
410
411 if (brix_imp_client_active()) {
412 switch (op) {
413 case BENEATH_2P_RENAME: return brix_imp_rename(src, dst);
414 case BENEATH_2P_RENAME_EXCL: return brix_imp_rename_noreplace(src, dst);
415 case BENEATH_2P_EXCHANGE:
416 /* No impersonation-broker exchange verb, and §3.5 forbids a
417 * two-rename emulation: refuse rather than fake atomicity. */
418 errno = ENOTSUP;
419 return -1;
420 default: return brix_imp_link(src, dst);
421 }
422 }
423
424 sfd = beneath_open_parent(rootfd, src, sbuf, sizeof(sbuf), &sbase);
425 if (sfd < 0) { return -1; }
426 dfd = beneath_open_parent(rootfd, dst, dbuf, sizeof(dbuf), &dbase);
427 if (dfd < 0) { beneath_close_parent(sfd, rootfd); return -1; }
428 if (sbase[0] == '\0' || dbase[0] == '\0') {
429 beneath_close_parent(sfd, rootfd);
430 beneath_close_parent(dfd, rootfd);
431 errno = EINVAL; return -1;
432 }
433
434 switch (op) {
435 case BENEATH_2P_RENAME:
436 /* phase74-fp: renameat(olddirfd, oldpath, newdirfd, newpath) — (sfd,
437 * sbase, dfd, dbase) IS the prototype order; same class as
438 * broker_ops.c. */
439 rc = renameat(sfd, sbase, dfd, dbase); /* NOLINT(readability-suspicious-call-argument) */
440 break;
441 case BENEATH_2P_RENAME_EXCL:
442 rc = brix_renameat_noreplace_fallback(ngx_cycle->log, sfd, sbase,
443 dfd, dbase);
444 break;
445 case BENEATH_2P_EXCHANGE:
446 /* Atomic two-name swap; no glibc wrapper predates 2.28, so raw
447 * SYS_renameat2 like the NOREPLACE arm above. A kernel or filesystem
448 * without the flag answers ENOSYS/EINVAL — reported as ENOTSUP, and
449 * NEVER degraded to two renames: unlike NOREPLACE there is no
450 * pre-checked consolation whose only failure mode is under-claiming
451 * (sd.h exchange contract, phase-107 §3.5). */
452 rc = (int) syscall(SYS_renameat2, sfd, sbase, dfd, dbase,
453 (unsigned int) RENAME_EXCHANGE);
454 if (rc != 0 && (errno == ENOSYS || errno == EINVAL)) {
455 errno = ENOTSUP;
456 }
457 break;
458 default:
459 rc = linkat(sfd, sbase, dfd, dbase, 0);
460 break;
461 }
462 beneath_close_parent(sfd, rootfd);
463 beneath_close_parent(dfd, rootfd);
464 return rc;
465}
