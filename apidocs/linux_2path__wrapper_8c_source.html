2 * src/platform/linux/path_wrapper.c - Linux path primitives beneath the VFS seam
3 *
4 * WHAT: openat2(2), the confined stat, renameat2(2) and the statx birth time
5 * declared in platform_api_posix.h.
6 * WHY: These are the calls whose Darwin form differs in semantics, not
7 * spelling; the portable tree calls brix_plat_* and never names the
8 * syscall (guard: tools/ci/check_platform_leak.py).
9 * HOW: Thin syscall wrappers. A kernel or filesystem without the feature

11 * their existing fallbacks.

40int
41brix_plat_openat2(int rootfd, const char *rel, int flags, mode_t mode,
42 uint64_t resolve)
43{
44 struct open_how how;
45
46 memset(&how, 0, sizeof(how));
47 how.flags = (uint64_t) (flags | O_CLOEXEC);
48 how.resolve = resolve;
49 if (flags & O_CREAT) {
50 how.mode = (uint64_t) (mode & 07777);
51 }
52#if defined(SYS_openat2)
53 return (int) syscall(SYS_openat2, rootfd, rel, &how, sizeof(how)); /* vfs-seam-allow: SEAM_CORRECT - PAL confinement primitive beneath brix_open_beneath */
54#else
55 (void) rootfd;
56 errno = ENOSYS;
57 return -1;
58#endif
59}

73int
74brix_plat_stat_resolve(int rootfd, const char *rel, int nofollow,
75 uint64_t resolve, struct stat *st)
76{
77 int fd, rc;
78
79 fd = brix_plat_openat2(rootfd, rel[0] == '\0' ? "." : rel,
80 O_PATH | (nofollow ? O_NOFOLLOW : 0), 0, resolve);
81 if (fd < 0) {
82 return -1;
83 }
84 rc = fstat(fd, st); /* vfs-seam-allow: SEAM_CORRECT - fstat on the PAL-confined handle */
85 close(fd);
86 return rc;

99#if defined(SYS_renameat2)

106 kflags |= 2u; /* RENAME_EXCHANGE */

123brix_plat_fstatat_btime(int dirfd, const char *name, int flags,
124 struct stat *st, time_t *btime)
125{
126 struct statx stx;
127
128 if (statx(dirfd, name, flags,
129 STATX_BASIC_STATS | STATX_BTIME, &stx) != 0) /* vfs-seam-allow: SEAM_CORRECT - PAL metadata primitive beneath the VFS */
130 {
131 return -1;
132 }
133 memset(st, 0, sizeof(*st));
134 st->st_mode = (mode_t) stx.stx_mode;
135 st->st_size = (off_t) stx.stx_size;
136 st->st_uid = (uid_t) stx.stx_uid;
137 st->st_gid = (gid_t) stx.stx_gid;
138 st->st_nlink = (nlink_t) stx.stx_nlink;
139 st->st_ino = (ino_t) stx.stx_ino;
140 st->st_mtime = (time_t) stx.stx_mtime.tv_sec;
141 st->st_atime = (time_t) stx.stx_atime.tv_sec;
142 st->st_ctime = (time_t) stx.stx_ctime.tv_sec;
143 *btime = (stx.stx_mask & STATX_BTIME) ? (time_t) stx.stx_btime.tv_sec : 0;
144 return 0;
145}
