1/*
2 * opdsl.c — declarative operation pipeline executor (§7.16). See opdsl.h.
3 *
4 * The pipeline is a small growable array of steps plus one optional failure
5 * handler. Execution is a linear scan that stops at the first non-zero step and
6 * runs the handler — the plain-C shape of the upstream Then/Else operation DSL.
7 */

35brix_opd_t *
36brix_opd_step(brix_opd_t *p, const char *label, brix_opd_step_fn fn, void *arg)
37{
38 if (p == NULL || fn == NULL || p->n >= BRIX_OPD_MAX) {
39 return NULL;
40 }
41 if (p->n == p->cap) {
42 size_t ncap = p->cap ? p->cap * 2 : 8;
43 brix_opd_entry_t *ns = realloc(p->steps, ncap * sizeof(*ns));
44
45 if (ns == NULL) {
46 return NULL;
47 }
48 p->steps = ns;
49 p->cap = ncap;
50 }
51 p->steps[p->n].label = label;
52 p->steps[p->n].fn = fn;
53 p->steps[p->n].arg = arg;
54 p->n++;
55 return p;
56}

67int
68brix_opd_run(brix_opd_t *p, brix_conn *c, brix_status *st, size_t *ran)
69{
70 size_t i;
71
72 if (p == NULL) {
73 return -1;
74 }
75 for (i = 0; i < p->n; i++) {
76 if (p->steps[i].fn(c, p->steps[i].arg, st) != 0) {
77 if (ran != NULL) {
78 *ran = i; /* the index that failed */
79 }
80 if (p->otherwise_fn != NULL) {
81 (void) p->otherwise_fn(c, p->otherwise_arg, st);
82 }
83 return -1;
84 }
85 }
86 if (ran != NULL) {
87 *ran = p->n; /* all ran */
88 }
89 return 0;
90}
