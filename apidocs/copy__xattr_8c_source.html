1/*
2 * copy_xattr.c — xrdcp --xattr: preserve extended attributes (§7.13).
3 *
4 * WHAT: brix_copy_preserve_xattrs() mirrors user-namespace extended
5 * attributes across a completed root://↔local copy: a download lists
6 * the source's kXR_fattr attributes and setxattr()s them onto the
7 * local destination; an upload lists the local user.* attributes and
8 * kXR_fattr-sets them on the remote destination.
9 * WHY: Stock xrdcp --xattr ("preserve extended attributes"); scientific
10 * workflows carry checksums/provenance in xattrs. Preservation is
11 * BEST-EFFORT exactly like stock: a failed attribute warns and the
12 * copy stays successful — attributes are auxiliary to the bytes.
13 * HOW: Direction from the URL schemes; names travel VERBATIM in both
14 * directions with two hard rules: (a) only user-namespace attributes
15 * ever transfer — system./security./trusted. names are skipped
16 * outright in BOTH directions, so a hostile remote name like
17 * "security.capability" can never be planted on a local file; (b) a
18 * remote name without a user. prefix lands locally under
19 * "user.<name>". Values are bounded by the fattr wire caps (64 KiB).
20 */

28/* ---- May this attribute name cross the copy boundary? ----
29 *
30 * WHAT: 1 only for user-namespace names (with or without the "user."
31 * prefix spelled out); anything claiming another namespace is refused.
32 *
33 * WHY: system./security./trusted./btrfs. attributes change LOCAL KERNEL
34 * behavior (capabilities, ACLs) — a remote server must never be able to
35 * plant one via --xattr, and uploading them would leak host metadata.
36 *
37 * HOW: Explicit deny on a '.'-delimited first token that is a known kernel
38 * namespace other than user; bare names (no namespace) are allowed and
39 * later normalised under user. locally.
40 */
41static int
42xattr_name_allowed(const char *name)
43{
44 if (name == NULL || name[0] == '\0'
45 || strlen(name) >= XATTR_NAME_MAX_) {
46 return 0;
47 }
48 if (strncmp(name, "user.", 5) == 0) {
49 return 1;
50 }
51 if (strncmp(name, "system.", 7) == 0 || strncmp(name, "security.", 9) == 0
52 || strncmp(name, "trusted.", 8) == 0) {
53 return 0;
54 }
55 return strchr(name, '.') == NULL; /* bare names ok; odd namespaces not */
56}

58/* ---- Download direction: remote fattr list → local setxattr ---- */
59static void
60xattr_pull(brix_conn *c, const char *rpath, const char *lpath, int silent)
61{
62 brix_status st;
63 char names[8192];
64 size_t nlen = 0, cursor = 0;
65
66 brix_status_clear(&st);
67 if (brix_fattr_list(c, rpath, names, sizeof(names), &nlen, &st) != 0) {
68 return; /* no attributes / unsupported: nothing to preserve */
69 }
70 if (nlen > sizeof(names)) {
71 nlen = sizeof(names);
72 }
73 while (cursor < nlen) {
74 const char *name = names + cursor;
75 size_t span = strnlen(name, nlen - cursor);
76 char localname[XATTR_NAME_MAX_ + 8];
77 char *val;
78 size_t vlen = 0;
79
80 cursor += span + 1;
81 if (span == 0 || !xattr_name_allowed(name)) {
82 continue;
83 }
84 val = (char *) malloc(XATTR_VAL_MAX);
85 if (val == NULL) {
86 return;
87 }
88 brix_status_clear(&st);
89 if (brix_fattr_get(c, rpath, name, val, XATTR_VAL_MAX, &vlen,
90 &st) == 0
91 && vlen <= XATTR_VAL_MAX) {
92 snprintf(localname, sizeof(localname),
93 strncmp(name, "user.", 5) == 0 ? "%s" : "user.%s", name);
94 if (setxattr(lpath, localname, val, vlen, 0) != 0 && !silent) {
95 fprintf(stderr, "xrdcp: --xattr: could not set %s on %s: %s\n",
96 localname, lpath, strerror(errno));
97 }
98 } else if (!silent) {
99 fprintf(stderr, "xrdcp: --xattr: could not read %s from %s: %s\n",
100 name, rpath, st.msg);
101 }
102 free(val);
103 }
104}

106/* ---- Upload direction: local user.* list → remote fattr set ---- */
107static void
108xattr_push(brix_conn *c, const char *lpath, const char *rpath, int silent)
109{
110 char names[8192];
111 ssize_t nlen = listxattr(lpath, names, sizeof(names));
112 size_t cursor = 0;
113
114 if (nlen <= 0) {
115 return;
116 }
117 while (cursor < (size_t) nlen) {
118 const char *name = names + cursor;
119 size_t span = strnlen(name, (size_t) nlen - cursor);
120 char *val;
121 ssize_t vlen;
122
123 cursor += span + 1;
124 if (span == 0 || strncmp(name, "user.", 5) != 0
125 || !xattr_name_allowed(name)) {
126 continue; /* only the user namespace ever leaves the host */
127 }
128 val = (char *) malloc(XATTR_VAL_MAX);
129 if (val == NULL) {
130 return;
131 }
132 vlen = getxattr(lpath, name, val, XATTR_VAL_MAX);
133 if (vlen >= 0) {
134 brix_status st;
135
136 brix_status_clear(&st);
137 if (brix_fattr_set(c, rpath, name, val, (size_t) vlen,
138 0 /* overwrite ok */, &st) != 0
139 && !silent) {
140 fprintf(stderr, "xrdcp: --xattr: could not set %s on %s: %s\n",
141 name, rpath, st.msg);
142 }
143 }
144 free(val);
145 }
146}

148/* ---- Preserve xattrs across one completed copy ----
149 *
150 * WHAT: Applies the direction-appropriate mirror for a root://↔local pair;
151 * other scheme pairs are silently out of scope (stock --xattr is a
152 * root-protocol feature). Never fails the copy: warnings only.
153 *
154 * WHY: Called by the transfer wrapper AFTER the bytes (and any --cksum
155 * verdict) succeeded — attributes describe a file that now exists.
156 *
157 * HOW: Parse both URLs; open one throwaway connection to the remote side
158 * (the data connection is already torn down by the time the wrapper
159 * runs); dispatch pull/push.
160 */
161void
162brix_copy_preserve_xattrs(const char *src, const char *dst,
163 const brix_opts *co, int silent)
164{
165 brix_url su, du;
166 brix_status st;
167 brix_conn c;
168
169 brix_status_clear(&st);
170 if (brix_url_parse(src, &su, &st) != 0
171 || brix_url_parse(dst, &du, &st) != 0) {
172 return;
173 }
174 if ((su.scheme == XRDC_SCHEME_ROOT || su.scheme == XRDC_SCHEME_ROOTS)
175 && du.scheme == XRDC_SCHEME_LOCAL) {
176 if (brix_connect(&c, &su, co, &st) == 0) {
177 xattr_pull(&c, su.path, du.path, silent);
178 brix_close(&c);
179 }
180 return;
181 }
182 if (su.scheme == XRDC_SCHEME_LOCAL
183 && (du.scheme == XRDC_SCHEME_ROOT || du.scheme == XRDC_SCHEME_ROOTS)) {
184 if (brix_connect(&c, &du, co, &st) == 0) {
185 xattr_push(&c, su.path, du.path, silent);
186 brix_close(&c);
187 }
188 }
189}
