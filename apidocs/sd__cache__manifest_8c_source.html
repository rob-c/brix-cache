1/*
2 * sd_cache_manifest.c — CVMFS manifest/whitelist signature verify on fill
3 * (phase-85 F1: the verifying proxy).
4 *
5 * WHAT: Before a MANIFEST-class staged fill commits, verify the full CVMFS
6 * trust chain against the operator-configured repo master public key:
7 * .cvmfswhitelist — signature vs master key + expiry;
8 * .cvmfspublished — whitelist sig vs master → whitelist not expired →
9 * manifest cert fingerprint ∈ whitelist → manifest
10 * sig vs cert → repo-name cross-check.
11 * A failed chain never publishes: the caller quarantines the part,
12 * aborts the fill, emits signal=cvmfs_tamper and reports EBADMSG.
13 *
14 * WHY: cvmfs-cas already makes every CAS object self-certifying at the edge,
15 * but the CAS *names* flow from the manifest — an origin (or MITM) that
16 * forges .cvmfspublished redirects every client to attacker-named
17 * content. Verifying the signature chain ONCE here protects every
18 * downstream client of this proxy, exactly as the FUSE client's mount
19 * path (shared/cvmfs/client/client.c load_trust_and_catalog) protects
20 * one mount. The verify primitives are the same shared pure-C code.
21 *
22 * HOW: Runs on the blocking fill THREAD (like the cvmfs-cas verify), so the
23 * sibling fetches (.cvmfswhitelist, certificate CAS object) go straight
24 * through the source tier with plain blocking preads. All buffers are
25 * malloc'd (fill threads carry small stacks). The staged part is read
26 * from its posix staged_path — enforced at config time.
27 */

44/* Signed-metadata artifacts are small (manifest ~400B, whitelist ~1KB, cert
45 * ~2KB); 64KB matches the FUSE client's raw_fetch bound and caps a hostile
46 * origin's memory cost. */
47#define SD_CACHE_META_MAX 65536
48
49enum {
50 META_UNSIGNED = 0,
51 META_PUBLISHED,
52 META_WHITELIST
53};

55/*
56 * WHAT: Classify the signed metadata names handled by the cache trust gate.
57 * WHY: Keeping exact-name recognition separate makes the verifier linear.
58 * HOW: Compare the already-classified relative path against both wire names.
59 */
60static int
61meta_signed_kind(const cvmfs_url_info_t *info)
62{
63 if (info->rel == NULL)
64 return META_UNSIGNED;
65 if (info->rel_len == sizeof(".cvmfspublished") - 1
66 && memcmp(info->rel, ".cvmfspublished", info->rel_len) == 0)
67 return META_PUBLISHED;
68 if (info->rel_len == sizeof(".cvmfswhitelist") - 1
69 && memcmp(info->rel, ".cvmfswhitelist", info->rel_len) == 0)
70 return META_WHITELIST;
71 return META_UNSIGNED;
72}

74/* Read the whole staged part at `pp` into a malloc'd buffer (cap
75 * SD_CACHE_META_MAX). Returns the byte count, or -1 (oversize/unreadable). */
76static ssize_t
77meta_read_part(const char *pp, unsigned char **out)
78{
79 unsigned char *buf;
80 ssize_t n, off = 0;
81 int fd;
82
83 fd = open(pp, O_RDONLY | O_CLOEXEC);
84 if (fd == -1) {
85 return -1;
86 }
87 buf = malloc(SD_CACHE_META_MAX);
88 if (buf == NULL) {
89 (void) close(fd);
90 return -1;
91 }
92 for ( ;; ) {
93 n = read(fd, buf + off, (size_t) (SD_CACHE_META_MAX - off));
94 if (n < 0) {
95 if (errno == EINTR) {
96 continue;
97 }
98 break;
99 }
100 if (n == 0) { /* clean EOF within the cap */
101 (void) close(fd);
102 *out = buf;
103 return off;
104 }
105 off += n;
106 if (off >= SD_CACHE_META_MAX) { /* oversize for signed meta */
107 break;
108 }
109 }
110 (void) close(fd);
111 free(buf);
112 return -1;
113}

115/* Fetch `key` COMPLETE from the source tier into a malloc'd buffer (cap
116 * SD_CACHE_META_MAX). Blocking — fill-thread only. Returns the byte count,
117 * or -1 on any failure (open, read, oversize). */
118static ssize_t
119meta_fetch_source(sd_cache_inst_state *st, const char *key,
120 unsigned char **out)
121{
122 brix_sd_obj_t *so;
123 unsigned char *buf;
124 ssize_t n, off = 0;
125 int err = 0;
126
127 if (st->source == NULL || st->source->driver->open == NULL) {
128 return -1;
129 }
130 so = brix_sd_open_maybe_cred(st->source, key, BRIX_SD_O_READ, 0, NULL,
131 &err);
132 if (so == NULL || so->driver->pread == NULL) {
133 if (so != NULL) {
134 brix_sd_obj_release(so);
135 }
136 return -1;
137 }
138 buf = malloc(SD_CACHE_META_MAX);
139 if (buf == NULL) {
140 brix_sd_obj_release(so);
141 return -1;
142 }
143 for ( ;; ) {
144 n = so->driver->pread(so, buf + off,
145 (size_t) (SD_CACHE_META_MAX - off), off);
146 if (n < 0) {
147 if (errno == EINTR) {
148 continue;
149 }
150 break;
151 }
152 if (n == 0) {
153 brix_sd_obj_release(so);
154 *out = buf;
155 return off;
156 }
157 off += n;
158 if (off >= SD_CACHE_META_MAX) {
159 break;
160 }
161 }
162 brix_sd_obj_release(so);
163 free(buf);
164 return -1;
165}

167/* Parse `buf` as a whitelist and verify it against the policy master key +
168 * wall-clock expiry. 0 on success; fills *out for the caller's fingerprint
169 * check. */
170static int
171meta_check_whitelist(sd_cache_inst_state *st, const unsigned char *buf,
172 size_t len, const char *key, cvmfs_whitelist_t *out)
173{
174 if (cvmfs_whitelist_parse(buf, len, out) != 0) {
175 ngx_log_error(NGX_LOG_WARN, st->log, 0,
176 "sd_cache: cvmfs whitelist for \"%s\" is malformed", key);
177 return -1;
178 }
179 if (cvmfs_verify_whitelist(out, st->policy.cvmfs_master_pub,
180 st->policy.cvmfs_master_pub_len) != 0)
181 {
182 ngx_log_error(NGX_LOG_WARN, st->log, 0,
183 "sd_cache: cvmfs whitelist for \"%s\" FAILED master-key "
184 "signature verification", key);
185 return -1;
186 }
187 if (cvmfs_whitelist_expired(out, (long) time(NULL))) {
188 ngx_log_error(NGX_LOG_WARN, st->log, 0,
189 "sd_cache: cvmfs whitelist for \"%s\" is EXPIRED", key);
190 return -1;
191 }
192 return 0;
193}

195/*
196 * WHAT: Verify that a signed manifest names the repository in its request key.
197 * WHY: A valid signature from repository A must not authorize a splice into B.
198 * HOW: Permit an omitted name, otherwise require exact length and bytes.
199 */
200static int
201meta_manifest_repo_matches(const cvmfs_manifest_t *manifest,
202 const cvmfs_url_info_t *info)
203{
204 if (manifest->repo_name[0] == '\0')
205 return 1;
206 return strlen(manifest->repo_name) == info->repo_len
207 && strncmp(manifest->repo_name, info->repo, info->repo_len) == 0;
208}

210/*
211 * WHAT: Check that the manifest certificate is admitted by the whitelist.
212 * WHY: Fingerprint calculation and membership form one trust-chain decision.
213 * HOW: Calculate the fingerprint and require its exact whitelist presence.
214 */
215static int
216meta_certificate_allowed(const cvmfs_whitelist_t *whitelist,
217 const unsigned char *pem, size_t pem_len, char *fingerprint,
218 size_t fingerprint_cap)
219{
220 return cvmfs_cert_fingerprint(pem, pem_len, fingerprint,
221 fingerprint_cap) == 0
222 && cvmfs_whitelist_lists_fp(whitelist, fingerprint);
223}

225/* Verify a staged .cvmfspublished: parse, cross-check the repo name against
226 * the fill key, fetch the live whitelist + signing cert through the source,
227 * and run the full chain. 0 = verified, -1 = definitive reject (tamper),
228 * -2 = the chain could not be evaluated (sibling fetch failed — an origin
229 * fault, not evidence of tampering). */
230static int
231meta_check_manifest(sd_cache_inst_state *st, const char *key,
232 const cvmfs_url_info_t *info, const unsigned char *buf, size_t len)
233{
234 cvmfs_manifest_t m;
235 cvmfs_whitelist_t wl;
236 unsigned char *wlbuf = NULL, *cert = NULL, *pem = NULL;
237 ssize_t wln, certn;
238 size_t pemn = 0;
239 char fp[64];
240 char sib[1024];
241 char objp[160];
242 int n, rc = -1;
243
244 if (cvmfs_manifest_parse(buf, len, &m) != 0) {
245 ngx_log_error(NGX_LOG_WARN, st->log, 0,
246 "sd_cache: cvmfs manifest \"%s\" is malformed", key);
247 return -1;
248 }
249 /* The signed 'N' must name the repo the key claims — a valid manifest for
250 * repo A must not publish under repo B's namespace (cross-repo splice). */
251 if (!meta_manifest_repo_matches(&m, info)) {
252 ngx_log_error(NGX_LOG_WARN, st->log, 0,
253 "sd_cache: cvmfs manifest \"%s\" names repo \"%s\" — "
254 "cross-repo splice rejected", key, m.repo_name);
255 return -1;
256 }
257
258 /* Trust anchor: the repo's live whitelist, fetched through the SAME
259 * source tier (ranked failover applies). */
260 n = snprintf(sib, sizeof(sib), "/cvmfs/%.*s/.cvmfswhitelist",
261 (int) info->repo_len, info->repo);
262 if (n <= 0 || (size_t) n >= sizeof(sib)) {
263 return -1;
264 }
265 wln = meta_fetch_source(st, sib, &wlbuf);
266 if (wln < 0) {
267 ngx_log_error(NGX_LOG_WARN, st->log, 0,
268 "sd_cache: cannot fetch \"%s\" to verify \"%s\"", sib, key);
269 return -2; /* origin fault, not tamper */
270 }
271 if (meta_check_whitelist(st, wlbuf, (size_t) wln, key, &wl) != 0) {
272 free(wlbuf);
273 return -1;
274 }
275
276 /* Signing cert: the manifest's 'X' CAS object, inflated to PEM/DER. */
277 if (cvmfs_hash_to_object_path(&m.certificate, 'X', objp,
278 sizeof(objp)) < 0)
279 {
280 free(wlbuf);
281 return -1;
282 }
283 n = snprintf(sib, sizeof(sib), "/cvmfs/%.*s/data/%s",
284 (int) info->repo_len, info->repo, objp);
285 if (n <= 0 || (size_t) n >= sizeof(sib)) {
286 free(wlbuf);
287 return -1;
288 }
289 certn = meta_fetch_source(st, sib, &cert);
290 if (certn < 0) {
291 ngx_log_error(NGX_LOG_WARN, st->log, 0,
292 "sd_cache: cannot fetch cert \"%s\" to verify \"%s\"", sib, key);
293 free(wlbuf);
294 return -2; /* origin fault, not tamper */
295 }
296 /* Stored cert objects are zlib-deflated; fall back to the raw bytes for
297 * an uncompressed store (cvmfs_cert_fingerprint takes PEM or DER). */
298 pem = malloc(SD_CACHE_META_MAX);
299 if (pem == NULL
300 || cvmfs_object_inflate(cert, (size_t) certn, pem,
301 SD_CACHE_META_MAX, &pemn) != 0)
302 {
303 free(pem);
304 pem = cert;
305 pemn = (size_t) certn;
306 cert = NULL;
307 }
308
309 if (!meta_certificate_allowed(&wl, pem, pemn, fp, sizeof(fp)))
310 {
311 ngx_log_error(NGX_LOG_WARN, st->log, 0,
312 "sd_cache: cvmfs manifest \"%s\" signing cert is NOT in the "
313 "whitelist — rejected", key);
314 } else if (cvmfs_verify_manifest(&m, pem, pemn) != 0) {
315 ngx_log_error(NGX_LOG_WARN, st->log, 0,
316 "sd_cache: cvmfs manifest \"%s\" FAILED certificate signature "
317 "verification", key);
318 } else {
319 rc = 0;
320 }
321
322 free(cert);
323 free(pem);
324 free(wlbuf);
325 return rc;
326}

328/* Verify a MANIFEST-class staged fill before commit (phase-85 F1). See the
329 * declaration in sd_cache_internal.h for the contract; NGX_OK also covers
330 * "not a signed-metadata shape" (.cvmfsreflog is unsigned) and "verify not
331 * configured". */
332ngx_int_t
333sd_cache_verify_manifest(sd_cache_inst_state *st, const char *key,
334 const char *pp)
335{
336 cvmfs_url_info_t info;
337 cvmfs_whitelist_t wl;
338 unsigned char *buf = NULL;
339 ssize_t len;
340 int kind;
341 ngx_int_t rc;
342
343 if (st->policy.cvmfs_master_pub == NULL
344 || st->policy.cvmfs_master_pub_len == 0)
345 {
346 return NGX_OK;
347 }
348 (void) cvmfs_classify_url(key, strlen(key), &info);
349 if (info.cls != CVMFS_URL_MANIFEST) {
350 return NGX_OK; /* CAS/GEO — not this gate */
351 }
352 kind = meta_signed_kind(&info);
353 if (kind == META_UNSIGNED) {
354 return NGX_OK; /* .cvmfsreflog etc. — unsigned */
355 }
356
357 if (pp == NULL) {
358 ngx_log_error(NGX_LOG_ERR, st->log, 0,
359 "sd_cache: cvmfs manifest verify has no staged path for \"%s\" "
360 "- failing the fill closed", key);
361 return NGX_DECLINED;
362 }
363 len = meta_read_part(pp, &buf);
364 if (len < 0) {
365 ngx_log_error(NGX_LOG_ERR, st->log, errno,
366 "sd_cache: cvmfs manifest verify cannot read the staged part "
367 "for \"%s\" - failing the fill closed", key);
368 return NGX_DECLINED;
369 }
370
371 if (kind == META_WHITELIST) {
372 rc = (meta_check_whitelist(st, buf, (size_t) len, key, &wl) == 0)
373 ? NGX_OK : NGX_ERROR; /* every whitelist fail is
374 * evaluated on the artifact
375 * itself — definitive */
376 } else {
377 switch (meta_check_manifest(st, key, &info, buf, (size_t) len)) {
378 case 0: rc = NGX_OK; break;
379 case -2: rc = NGX_DECLINED; break;
380 default: rc = NGX_ERROR; break;
381 }
382 }
383 free(buf);
384 return rc;
385}
