20/* Copy one optional template string into instance storage; the template
21 * pointer is then re-aimed at the copy (NULL when unset). */
22static const char *
23sd_xroot_fwd_copy_opt(char *dst, size_t cap, const char *src)
24{
25 if (src == NULL || src[0] == '\0') {
26 dst[0] = '\0';
27 return NULL;
28 }
29 ngx_cpystrn((u_char *) dst, (u_char *) src, cap);
30 return dst;
31}

33static void
34sd_xroot_fwd_state_init(sd_xroot_fwd_inst_state *st,
35 const brix_sd_xroot_fwd_cfg_t *cfg)
36{
37 const brix_sd_xroot_origin_cfg_t *o = &cfg->origin;
38
39 st->tmpl.af_policy = o->af_policy;
40 st->tmpl.verify_pages = o->verify_pages;
41 st->tmpl.nearline = o->nearline;
42 st->tmpl.dns = o->dns;
43 st->tmpl.bearer = sd_xroot_fwd_copy_opt(st->bearer,
44 sizeof(st->bearer), o->bearer);
45 st->tmpl.x509_proxy = sd_xroot_fwd_copy_opt(st->x509_proxy,
46 sizeof(st->x509_proxy),
47 o->x509_proxy);
48 st->tmpl.x509_key = sd_xroot_fwd_copy_opt(st->x509_key,

53 st->tmpl.sss_keytab = sd_xroot_fwd_copy_opt(st->sss_keytab,

55 o->sss_keytab);
56 ngx_cpystrn((u_char *) st->permit, (u_char *) cfg->permit,

58 st->allow_root = cfg->allow_root ? 1 : 0;
59 st->allow_roots = cfg->allow_roots ? 1 : 0;

62brix_sd_instance_t *
63brix_sd_xroot_fwd_create(const brix_sd_xroot_fwd_cfg_t *cfg, ngx_log_t *log)
64{
65 brix_sd_instance_t *inst;
66 sd_xroot_fwd_inst_state *st;
67
68 if (cfg == NULL || cfg->permit == NULL || cfg->permit[0] == '\0'
69 || (!cfg->allow_root && !cfg->allow_roots))
70 {
71 errno = EINVAL;
72 return NULL;
73 }
74 inst = calloc(1, sizeof(*inst));
75 st = calloc(1, sizeof(*st));
76 if (inst == NULL || st == NULL) {
77 free(inst);
78 free(st);
79 errno = ENOMEM;
80 return NULL;
81 }
82 sd_xroot_fwd_state_init(st, cfg);
83 if (pthread_mutex_init(&st->mtx, NULL) != 0) {
84 free(inst);
85 free(st);
86 errno = ENOMEM;
87 return NULL;
88 }
89 inst->driver = &brix_sd_xroot_fwd_driver;
90 inst->log = log;
91 inst->pool = NULL;
92 inst->state = st;
93 /* Same caps as one origin — the operator's `nearline` arms recall on
94 * every child exactly as it does on a fixed root+tape:// origin. */
95 inst->caps = brix_sd_xroot_fwd_driver.caps;
96 if (cfg->origin.nearline) {
97 inst->caps |= BRIX_SD_CAP_NEARLINE;
98 }
99 inst->domain = BRIX_VFS_DOMAIN_EXPORT;
100 return inst;
101}

103void
104brix_sd_xroot_fwd_destroy(brix_sd_instance_t *inst)
105{
106 sd_xroot_fwd_inst_state *st;
107 sd_xroot_fwd_child_t *c, *next;
108
109 if (inst == NULL) {
110 return;
111 }
112 st = inst->state;
113 if (st != NULL) {
114 for (c = st->children; c != NULL; c = next) {
115 next = c->next;
116 brix_sd_xroot_destroy(c->inst);
117 free(c);
118 }
119 pthread_mutex_destroy(&st->mtx);
120 free(st);
121 }
122 free(inst);
123}

143sd_xroot_fwd_child_find(const sd_xroot_fwd_inst_state *st,
144 const brix_sd_xroot_fwd_target_t *t)
145{
146 sd_xroot_fwd_child_t *c;
147
148 for (c = st->children; c != NULL; c = c->next) {
149 if (c->port == t->port && c->tls == t->tls
150 && strcasecmp(c->host, t->host) == 0)

152 return c;
153 }
154 }
155 return NULL;

158/* Create the sd_xroot child for `t` from the template and link it. Caller
159 * holds st->mtx. NULL with errno on the cap or the factory's failure. */
160static sd_xroot_fwd_child_t *
161sd_xroot_fwd_child_new(brix_sd_instance_t *inst, sd_xroot_fwd_inst_state *st,
162 const brix_sd_xroot_fwd_target_t *t)
163{
164 brix_sd_xroot_origin_cfg_t cfg = st->tmpl;
165 sd_xroot_fwd_child_t *c;
166
167 if (st->nchildren >= SD_XROOT_FWD_MAX_CHILDREN) {
168 ngx_log_error(NGX_LOG_ERR, inst->log, 0,
169 "brix: forward: origin table full (%ud distinct origins), "
170 "refusing %s:%d", st->nchildren, t->host, t->port);
171 errno = EMFILE;
172 return NULL;
173 }
174 c = calloc(1, sizeof(*c));
175 if (c == NULL) {
176 errno = ENOMEM;
177 return NULL;
178 }
179 cfg.host = t->host;
180 cfg.port = t->port;
181 cfg.tls = t->tls;
182 c->inst = brix_sd_xroot_create_origin(&cfg, inst->log);
183 if (c->inst == NULL) {
184 int saved = errno;
185
186 free(c);
187 errno = saved;
188 return NULL;
189 }
190 c->inst->domain = inst->domain;
191 ngx_cpystrn((u_char *) c->host, (u_char *) t->host, sizeof(c->host));
192 c->port = t->port;
193 c->tls = t->tls;
194 c->next = st->children;
195 st->children = c;
196 st->nchildren++;
197 ngx_log_error(NGX_LOG_INFO, inst->log, 0,
198 "brix: forward: origin %s://%s:%d admitted (%ud of %d)",
199 t->tls ? "roots" : "root", t->host, t->port, st->nchildren,
200 SD_XROOT_FWD_MAX_CHILDREN);
201 return c;
202}

204/* The two admission verdicts, in the order a refusal should be reported: a
205 * scheme outside the protocol list is a client error (kXR_Unsupported); a
206 * host outside the permit list is the security refusal (kXR_NotAuthorized)
207 * and is logged as such. */
208static int
209sd_xroot_fwd_admit(brix_sd_instance_t *inst, const sd_xroot_fwd_inst_state *st,
210 const brix_sd_xroot_fwd_target_t *t)
211{
212 if ((t->tls && !st->allow_roots) || (!t->tls && !st->allow_root)) {
213 return ENOTSUP;
214 }
215 if (!brix_sd_xroot_fwd_host_permitted(st->permit, t->host)) {
216 ngx_log_error(NGX_LOG_NOTICE, inst->log, 0,
217 "brix: forward: origin \"%s\" is not in the permit list — refused",
218 t->host);
219 return EACCES;
220 }
221 return 0;
222}

240brix_sd_instance_t *
241sd_xroot_fwd_resolve(brix_sd_instance_t *inst, const char *key,
242 brix_sd_xroot_fwd_target_t *t)
243{
244 sd_xroot_fwd_inst_state *st = inst->state;
245 sd_xroot_fwd_child_t *c;
246 int rc;
247
248 rc = brix_sd_xroot_fwd_parse_key(key, t);
249 if (rc == 0) {
250 rc = sd_xroot_fwd_admit(inst, st, t);
251 }
252 if (rc != 0) {
253 errno = rc;
254 return NULL;
255 }
256 pthread_mutex_lock(&st->mtx);
257 c = sd_xroot_fwd_child_find(st, t);
258 if (c == NULL) {
259 c = sd_xroot_fwd_child_new(inst, st, t);
260 }
261 pthread_mutex_unlock(&st->mtx);
262 return (c != NULL) ? c->inst : NULL;
263}
