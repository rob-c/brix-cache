14/*
15 * When session reuse is OFF, disable TLS session resumption on `ctx`: clear the
16 * server session cache AND session tickets so every connection performs a full
17 * handshake (per-connection forward secrecy; no resumption state to capture or
18 * replay). Reuse ON (or a NULL ctx) leaves the OpenSSL/nginx defaults untouched,
19 * so the knob is inert unless an operator opts out.
20 */
21static inline void
22brix_tls_apply_session_reuse(SSL_CTX *ctx, int reuse_on)
23{
24 if (ctx == NULL || reuse_on) {
25 return;
26 }
27 SSL_CTX_set_session_cache_mode(ctx, SSL_SESS_CACHE_OFF);
28 SSL_CTX_set_options(ctx, SSL_OP_NO_TICKET);
29}
