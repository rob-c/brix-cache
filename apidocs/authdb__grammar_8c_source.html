21/*
22 * WHAT: map ONE native-engine privilege letter onto its BRIX_AUTH_* bits, or 0
23 * when the letter is not in the alphabet.
24 * WHY: 2.0 F20 — the old switch folded "unknown letter" into `default: break;`
25 * so a typo (or the xrdacc-only letters `i`/`n`) silently evaporated and
26 * the operator got a rule weaker than the one they wrote. Returning the
27 * bits lets the caller tell "no bits" from "not a letter".
28 * HOW: pure lookup. 'r' implies 'l' (you cannot read what you cannot look
29 * up); 'a' (append) folds into UPDATE because the FS-level write
30 * permission is identical; 'x' is the 2.0 stage/recall privilege.
31 */
32static uint32_t
33adb_priv_bit(u_char c)
34{
35 switch (c) {
36 case 'r': return BRIX_AUTH_READ | BRIX_AUTH_LOOKUP;
37 case 'l': return BRIX_AUTH_LOOKUP;
38 case 'w': return BRIX_AUTH_UPDATE;
39 case 'a': return BRIX_AUTH_UPDATE; /* append is update */
40 case 'd': return BRIX_AUTH_DELETE;
41 case 'm': return BRIX_AUTH_MKDIR;
42 case 'k': return BRIX_AUTH_ADMIN;
43 case 'x': return BRIX_AUTH_STAGE; /* 2.0 F20: tape recall / evict */
44 default: return 0;
45 }
46}

48/*
49 * WHAT: OR-accumulate a whole privilege string into *out. NGX_OK, or NGX_ERROR
50 * with *bad set to the first letter that is not in the alphabet.
51 * WHY: a privilege field is all-or-nothing: half a rule is a rule the operator
52 * never wrote, so the caller refuses the LINE rather than the character.
53 * HOW: adb_priv_bit per byte; an empty field legitimately yields 0 bits (a
54 * rule that grants nothing, which is how an operator spells "deny here").
55 */
56static ngx_int_t
57brix_parse_privs(const char *p, size_t len, uint32_t *out, u_char *bad)
58{
59 uint32_t privs = 0;
60 uint32_t bit;
61 size_t i;
62
63 for (i = 0; i < len; i++) {
64 bit = adb_priv_bit((u_char) p[i]);
65 if (bit == 0) {
66 *bad = (u_char) p[i];
67 return NGX_ERROR;
68 }
69 privs |= bit;
70 }
71
72 *out = privs;
73 return NGX_OK;
74}
75/*
76 * WHAT: record (or immediately raise) a grammar defect on the current line.
77 * Always returns NGX_DECLINED so callers can `return adb_reject(...)`.
78 * WHY: one place owns the message shape, so every refusal an operator can hit
79 * names the file, the line number, and what was wrong with it.
80 * HOW: render into a stack buffer; with a defect slot keep the FIRST message
81 * (later lines are noise once the config is already refused), else log
82 * NGX_LOG_EMERG through cf directly.
83 */
84ngx_int_t
85brix_adb_reject(adb_parse_ctx_t *pc, const char *reason)
86{
87 u_char buf[512];
88 u_char *end;
89 size_t len;
90
91 end = ngx_snprintf(buf, sizeof(buf) - 1,
92 "brix_authdb \"%V\" line %ui: %s",
93 pc->filename, pc->lineno, reason);
94 *end = '\0';
95 len = (size_t) (end - buf);
96
97 if (pc->defect == NULL) {
98 ngx_conf_log_error(NGX_LOG_EMERG, pc->cf, 0, "%s", buf);
99 return NGX_DECLINED;
100 }
101 if (pc->defect->len == 0) {
102 pc->defect->data = ngx_pnalloc(pc->cf->pool, len + 1);
103 if (pc->defect->data != NULL) {
104 ngx_memcpy(pc->defect->data, buf, len + 1);
105 pc->defect->len = len;
106 }
107 }
108 return NGX_DECLINED;
109}

111/* As adb_reject, naming the offending byte (printable as a char, else hex). */
112static ngx_int_t
113adb_reject_char(adb_parse_ctx_t *pc, const char *what, u_char c)
114{
115 u_char reason[128];
116
117 if (c >= 0x20 && c < 0x7f) {
118 ngx_snprintf(reason, sizeof(reason), "%s '%c'%Z", what, c);
119 } else {
120 ngx_snprintf(reason, sizeof(reason), "%s byte 0x%02xd%Z", what,
121 (int) c);
122 }
123 return brix_adb_reject(pc, (const char *) reason);
124}

126/* WHAT: 1 when `c` is a legal field-1 identity selector.
127 * WHY: the alphabet is the security surface — anything outside it must refuse
128 * the line, never be skipped (a skipped selector widens the rule).
129 * HOW: `u` DN, `g` VO/group, `p` peer host or CIDR, `a` any identity,
130 * `v` VOMS virtual organisation, `l` VOMS role (2.0 F20). */
131static ngx_flag_t
132adb_selector_ok(u_char c)
133{
134 return c == 'u' || c == 'g' || c == 'p' || c == 'a'
135 || c == 'v' || c == 'l';
136}

152/*
153 * WHAT: read field 1 into rule->sel[] / rule->nsel.
154 * WHY: 2.0 F20 (i) — a compound token used to be truncated to its lead byte,
155 * so `usergroup` was accepted as `u` and matched every DN. Every byte is
156 * now either a distinct selector or a refused line.
157 * HOW: reject an out-of-alphabet byte, a repeat, `a` mixed with anything (it
158 * already means "any identity", so AND-ing it is either a no-op or a
159 * contradiction the operator did not intend), and more bytes than the
160 * alphabet holds.
161 */
162static ngx_int_t
163adb_bind_selectors(adb_parse_ctx_t *pc, brix_authdb_rule_t *rule,
164 const adb_line_t *line)
165{
166 size_t n = (size_t) (line->type_end - line->type_p);
167 size_t i, j;
168
169 if (n > BRIX_AUTHDB_MAX_SELECTORS) {
170 return brix_adb_reject(pc, "field 1 lists more identity selectors than the "
171 "alphabet `u g p a v l` holds");
172 }
173
174 for (i = 0; i < n; i++) {
175 u_char c = line->type_p[i];
176
177 if (!adb_selector_ok(c)) {
178 return adb_reject_char(pc, "unknown identity selector", c);
179 }
180 if (c == 'a' && n > 1) {
181 return brix_adb_reject(pc, "the `a` (any identity) selector cannot be "
182 "combined with another selector");
183 }
184 for (j = 0; j < i; j++) {
185 if ((u_char) rule->sel[j] == c) {
186 return adb_reject_char(pc, "identity selector repeated", c);
187 }
188 }
189 rule->sel[i] = (brix_auth_type_t) c;
190 }
191
192 rule->nsel = (ngx_uint_t) n;
193 return NGX_OK;
194}

196/* Count the '|'-separated components of an id token (always >= 1). */
197static ngx_uint_t
198adb_id_components(const u_char *p, const u_char *end)
199{
200 ngx_uint_t n = 1;
201
202 for (; p < end; p++) {
203 if (*p == '|') {
204 n++;
205 }
206 }
207 return n;
208}

210/*
211 * WHAT: bind field 2 to the selectors bound by adb_bind_selectors.
212 * WHY: a single-selector rule must keep taking its id VERBATIM — DNs are full
213 * of punctuation and splitting them would break every deployed authdb —
214 * so only a compound rule reads '|' as a separator.
215 * HOW: arity must match exactly (one component per selector) and no component
216 * may be empty; both are refused lines, never a silently dropped
217 * selector.
218 */
219static ngx_int_t
220adb_bind_ids(adb_parse_ctx_t *pc, brix_authdb_rule_t *rule,
221 const adb_line_t *line)
222{
223 const u_char *p = line->id_p;
224 const u_char *end = line->id_end;
225 ngx_uint_t i;
226
227 if (rule->nsel == 1) {
228 return adb_dup(pc->cf, &rule->sel_id[0], p, (size_t) (end - p));
229 }
230
231 if (adb_id_components(p, end) != rule->nsel) {
232 return brix_adb_reject(pc, "a compound rule needs exactly one "
233 "'|'-separated id component per selector");
234 }
235
236 for (i = 0; i < rule->nsel; i++) {
237 const u_char *bar = p;
238
239 while (bar < end && *bar != '|') {
240 bar++;
241 }
242 if (bar == p) {
243 return brix_adb_reject(pc, "a compound rule has an empty id component");
244 }
245 if (adb_dup(pc->cf, &rule->sel_id[i], p,
246 (size_t) (bar - p)) != NGX_OK)
247 {
248 return NGX_ERROR;
249 }
250 p = bar + 1;
251 }
252
253 return NGX_OK;
254}

256/*
257 * WHAT: build ONE rule from a tokenized line and push it into `rules`.
258 * NGX_OK pushed; NGX_DECLINED the line was refused (nothing pushed);
259 * NGX_ERROR allocation failure.
260 * WHY: isolates the array-push + pool ownership from the tokenizer. 2.0 F20:
261 * the rule is assembled in a SCRATCH struct and only pushed once every
262 * field parsed, so a refused line can never leave a half-built rule (and
263 * therefore a half-built matcher) in the array.
264 * HOW: selectors -> ids -> path -> privs, each fallible; `type`/`id` are then
265 * set as the selector-0 alias every pre-F20 reader still uses, and
266 * resolved[] stays zeroed (filled later by brix_finalize_authdb_rules —
267 * deferred realpath).
268 */
269ngx_int_t
270brix_adb_append(adb_parse_ctx_t *pc, ngx_array_t *rules,
271 const adb_line_t *line)
272{
273 brix_authdb_rule_t scratch;
274 brix_authdb_rule_t *rule;
275 u_char bad = 0;
276 ngx_int_t rc;
277
278 ngx_memzero(&scratch, sizeof(scratch));
279
280 rc = adb_bind_selectors(pc, &scratch, line);
281 if (rc != NGX_OK) {
282 return rc;
283 }
284
285 rc = adb_bind_ids(pc, &scratch, line);
286 if (rc != NGX_OK) {
287 return rc;
288 }
289
290 if (adb_dup(pc->cf, &scratch.path, line->path_p,
291 (size_t) (line->path_end - line->path_p)) != NGX_OK)
292 {
293 return NGX_ERROR;
294 }
295
296 if (brix_parse_privs((const char *) line->privs_p,
297 (size_t) (line->privs_end - line->privs_p),
298 &scratch.privs, &bad) != NGX_OK)
299 {
300 return adb_reject_char(pc, "unknown privilege letter", bad);
301 }
302
303 scratch.type = scratch.sel[0];
304 scratch.id = scratch.sel_id[0];
305
306 rule = ngx_array_push(rules);
307 if (rule == NULL) {
308 return NGX_ERROR;
309 }
310 *rule = scratch;
311 return NGX_OK;
312}

314/*
315 * WHAT: turn a deferred authdb grammar defect into an `nginx -t` failure, but
316 * only for the engine that actually reads the native grammar.
317 * WHY: `brix_authdb` is parsed by BOTH engines' parsers because the directive
318 * runs before `brix_authdb_engine` has settled. An xrdacc authfile is
319 * full of bytes the native grammar has never accepted (record types
320 * `= x s h n o r t`, multi-pair lines, the `i`/`n` privilege letters), so
321 * raising at parse time would refuse every xrdacc deployment.
322 * HOW: called from the merge, where `format` is final. Returns NGX_CONF_OK
323 * unless the native engine is selected AND a defect was recorded.
324 */
325char *
326brix_authdb_defect_refuse(ngx_conf_t *cf, ngx_uint_t format,
327 const ngx_str_t *defect)
328{
329 if (format == BRIX_AUTHDB_FORMAT_XRDACC || defect == NULL
330 || defect->len == 0)
331 {
332 return NGX_CONF_OK;
333 }
334
335 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
336 "%V (the native authdb engine refuses a line it cannot "
337 "parse rather than silently dropping part of it; use "
338 "`brix_authdb_engine xrdacc` for XrdAcc-format files)",
339 defect);
340 return NGX_CONF_ERROR;
341}
