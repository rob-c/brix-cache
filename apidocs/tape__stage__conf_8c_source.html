41/* ---- 2.0 F1: the process-wide engine snapshot ------------------------------
42 * One snapshot per configuration cycle (the phase-116 dns_registry precedent:
43 * a cycle-keyed static, reset by the first merge of a new cycle so a reload
44 * that drops every `brix_frm on` block also drops the published values). The
45 * exec MSS adapter is told about the stage command / deadline right here, in
46 * the master, so `nginx -t`, master-built tier instances and every forked
47 * worker see the same program without a per-worker hand-off. */
48static brix_frm_engine_conf_t frm_engine;
49
50static brix_frm_engine_conf_t *
51frm_engine_for(ngx_conf_t *cf)
52{
53 if (frm_engine.cycle != cf->cycle) {
54 ngx_memzero(&frm_engine, sizeof(frm_engine));
55 frm_engine.cycle = cf->cycle;
56 frm_engine.copymax = BRIX_FRM_COPYMAX_DEFAULT;
57 frm_engine.fail_retries = BRIX_FRM_FAIL_RETRIES_DEFAULT;
58 frm_engine.fail_backoff_ms = BRIX_FRM_FAIL_BACKOFF_DEFAULT_MS;
59 brix_sd_frm_set_exec_defaults(NULL, 0);
60 }
61 return &frm_engine;
62}
63
64const brix_frm_engine_conf_t *
65brix_frm_engine_conf(const ngx_cycle_t *cycle)
66{
67 if (cycle == NULL || frm_engine.cycle != cycle || !frm_engine.published) {
68 return NULL;
69 }
70 return &frm_engine;

73/* Publish one string value: the first `brix_frm on` block sets it, a later one
74 * must repeat it verbatim or leave it unset. */
75static char *
76frm_publish_str(ngx_conf_t *cf, const char *name, char *slot, size_t size,
77 const ngx_str_t *value)
78{
79 if (value->len == 0) {
80 return NGX_CONF_OK;
81 }
82 if (value->len >= size) {
83 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "%s \"%V\" is too long",
84 name, value);
85 return NGX_CONF_ERROR;
86 }
87 if (slot[0] != '\0'
88 && (ngx_strlen(slot) != value->len
89 || ngx_strncmp(slot, value->data, value->len) != 0))
90 {
91 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
92 "%s \"%V\" differs from the value another brix_frm server "
93 "published (\"%s\"): the stage engine is process-wide",
94 name, value, slot);
95 return NGX_CONF_ERROR;
96 }
97 ngx_memcpy(slot, value->data, value->len);
98 slot[value->len] = '\0';
99 return NGX_CONF_OK;
100}

102/* Publish one numeric value (ngx_uint_t and ngx_msec_t share the width). */
103static char *
104frm_publish_num(ngx_conf_t *cf, const char *name, ngx_uint_t *slot,
105 unsigned bit, ngx_uint_t value, ngx_uint_t unset)
106{
107 brix_frm_engine_conf_t *e = frm_engine_for(cf);
108
109 if (value == unset) {
110 return NGX_CONF_OK;
111 }
112 if ((e->explicit_mask & bit) && *slot != value) {
113 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
114 "%s %ui differs from the value another brix_frm server "
115 "published (%ui): the stage engine is process-wide",
116 name, value, *slot);
117 return NGX_CONF_ERROR;
118 }
119 *slot = value;
120 e->explicit_mask |= bit;
121 return NGX_CONF_OK;
122}

124/* Grammar checks on a program the worker runs (the exec MSS command, the
125 * 2.0 F4 purge policy program): an absolute path that, when it already
126 * exists, nobody but its owner can rewrite (the worker runs it with the MSS
127 * credentials; a group/world-writable program is a privilege hand-off to
128 * whoever can write it — the same rule brix_checksum_plugin applies). */
129char *
130brix_frm_check_program(ngx_conf_t *cf, const char *directive,
131 const ngx_str_t *cmd, const char *usage)
132{
133 char path[BRIX_TAPE_STAGE_PATH_BUF];
134 struct stat st;
135
136 if (cmd->len == 0) {
137 return NGX_CONF_OK;
138 }
139 if (cmd->data[0] != '/') {
140 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
141 "%s \"%V\" must be an absolute program path (%s)",
142 directive, cmd, usage);
143 return NGX_CONF_ERROR;
144 }
145 if (cmd->len >= sizeof(path)) {
146 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "%s \"%V\" is too long",
147 directive, cmd);
148 return NGX_CONF_ERROR;
149 }
150 ngx_memcpy(path, cmd->data, cmd->len);
151 path[cmd->len] = '\0';
152 if (stat(path, &st) == 0 && (st.st_mode & (S_IWGRP | S_IWOTH))) {
153 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
154 "%s \"%V\" is group- or world-writable; refusing to run a "
155 "program anyone else can rewrite", directive, cmd);
156 return NGX_CONF_ERROR;
157 }
158 return NGX_CONF_OK;
159}

168/* brix_frm_stagemsg (2.0 F2): the StageEvents feed is appended to by every
169 * worker, so an existing file must be a private regular file -- a feed anyone
170 * can append to is a forged tape-event stream for whatever tails it. A
171 * missing file is fine: the worker creates it 0600 (stage_events.c). */
172static char *
173frm_check_stagemsg(ngx_conf_t *cf, const ngx_str_t *file)
174{
175 struct stat sb;
176 char path[BRIX_TAPE_STAGE_LINE_BUF];
177
178 if (file->len == 0) {
179 return NGX_CONF_OK;
180 }
181 if (file->data[0] != '/') {
182 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
183 "brix_frm_stagemsg \"%V\" must be an absolute path", file);
184 return NGX_CONF_ERROR;
185 }
186 if (file->len >= sizeof(path)) {
187 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
188 "brix_frm_stagemsg \"%V\" is too long", file);
189 return NGX_CONF_ERROR;
190 }
191 ngx_memcpy(path, file->data, file->len);
192 path[file->len] = '\0';
193 if (stat(path, &sb) != 0) {
194 return NGX_CONF_OK; /* created 0600 by the worker */
195 }
196 if (!S_ISREG(sb.st_mode)) {
197 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
198 "brix_frm_stagemsg \"%V\" is not a regular file", file);
199 return NGX_CONF_ERROR;
200 }
201 if (sb.st_mode & (S_IWGRP | S_IWOTH)) {
202 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
203 "brix_frm_stagemsg \"%V\" is group- or world-writable; "
204 "refusing to feed a file anyone else can append to", file);
205 return NGX_CONF_ERROR;
206 }
207 return NGX_CONF_OK;
208}

210/* Publish this `brix_frm on` block's engine values (raw = pre-merge, so only
211 * an explicit line counts). */
212static char *
213frm_engine_publish(ngx_conf_t *cf, const brix_frm_conf_t *conf,
214 const brix_frm_conf_t *raw)
215{
216 brix_frm_engine_conf_t *e = frm_engine_for(cf);
217
218 if (frm_publish_str(cf, "brix_frm_queue_path", e->queue_path,
219 sizeof(e->queue_path), &conf->queue_path)
220 != NGX_CONF_OK
221 || frm_publish_str(cf, "brix_frm_stagecmd", e->stagecmd,
222 sizeof(e->stagecmd), &conf->stagecmd) != NGX_CONF_OK
223 || frm_publish_str(cf, "brix_frm_stagemsg", e->stagemsg,
224 sizeof(e->stagemsg), &conf->stagemsg) != NGX_CONF_OK
225 || frm_publish_num(cf, "brix_frm_copymax", &e->copymax,
226 BRIX_FRM_ENGINE_COPYMAX, raw->copymax,
227 NGX_CONF_UNSET_UINT) != NGX_CONF_OK
228 || frm_publish_num(cf, "brix_frm_fail_retries", &e->fail_retries,
229 BRIX_FRM_ENGINE_FAIL_RETRIES, raw->fail_retries,
230 NGX_CONF_UNSET_UINT) != NGX_CONF_OK
231 || frm_publish_num(cf, "brix_frm_fail_backoff",
232 (ngx_uint_t *) &e->fail_backoff_ms,
233 BRIX_FRM_ENGINE_FAIL_BACKOFF,
234 (ngx_uint_t) raw->fail_backoff_ms,
235 (ngx_uint_t) NGX_CONF_UNSET_MSEC) != NGX_CONF_OK
236 || frm_publish_num(cf, "brix_frm_copy_timeout",
237 (ngx_uint_t *) &e->copy_timeout_ms,
238 BRIX_FRM_ENGINE_COPY_TIMEOUT,
239 (ngx_uint_t) raw->copy_timeout,
240 (ngx_uint_t) NGX_CONF_UNSET_MSEC) != NGX_CONF_OK)
241 {
242 return NGX_CONF_ERROR;
243 }
244 e->published = 1;
245 brix_sd_frm_set_exec_defaults(e->stagecmd[0] ? e->stagecmd : NULL,
246 e->copy_timeout_ms);
247 return NGX_CONF_OK;
248}

250/* A `brix_frm off` block that still carries an engine knob configures
251 * nothing: say so at load time rather than let the line rot silently. */
252static void
253frm_warn_inert_knobs(ngx_conf_t *cf, const brix_frm_conf_t *raw)
254{
255 static const char *names[] = {
256 "brix_frm_queue_path", "brix_frm_stagecmd", "brix_frm_copymax",
257 "brix_frm_fail_retries", "brix_frm_fail_backoff",
258 "brix_frm_copy_timeout",
259 "brix_frm_stagemsg",
260 };
261 unsigned set[7];
262 size_t i;
263
264 set[0] = raw->queue_path.len != 0;
265 set[1] = raw->stagecmd.len != 0;
266 set[2] = raw->copymax != NGX_CONF_UNSET_UINT;
267 set[3] = raw->fail_retries != NGX_CONF_UNSET_UINT;
268 set[4] = raw->fail_backoff_ms != NGX_CONF_UNSET_MSEC;
269 set[5] = raw->copy_timeout != NGX_CONF_UNSET_MSEC;
270 set[6] = raw->stagemsg.len != 0;
271 for (i = 0; i < sizeof(names) / sizeof(names[0]); i++) {
272 if (set[i]) {
273 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
274 "%s is ignored: brix_frm is off in this server", names[i]);
275 }
276 }
277}

279static char *
280frm_check_enabled(ngx_conf_t *cf, const brix_frm_conf_t *conf)
281{
282 if (conf->queue_path.len == 0) {
283 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
284 "brix_frm on requires brix_frm_queue_path");
285 return NGX_CONF_ERROR;
286 }
287 if (conf->queue_path.data[0] != '/') {
288 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
289 "brix_frm_queue_path \"%V\" must be an absolute path",
290 &conf->queue_path);
291 return NGX_CONF_ERROR;
292 }
293 if (conf->copymax == 0) {
294 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
295 "brix_frm_copymax must be at least 1");
296 return NGX_CONF_ERROR;
297 }
298 if (conf->fail_retries == 0) {
299 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
300 "brix_frm_fail_retries must be at least 1");
301 return NGX_CONF_ERROR;
302 }
303 if (frm_check_stagecmd(cf, &conf->stagecmd) != NGX_CONF_OK) {
304 return NGX_CONF_ERROR;
305 }
306 return frm_check_stagemsg(cf, &conf->stagemsg);
307}

309char *
310brix_frm_conf_merge(ngx_conf_t *cf, brix_frm_conf_t *conf,
311 brix_frm_conf_t *prev)
312{
313 brix_frm_conf_t raw = *conf; /* pre-merge: which lines were written */
314
315 (void) frm_engine_for(cf); /* reset the snapshot on a new cycle */
316
317 ngx_conf_merge_value(conf->enable, prev->enable, 0);
318 ngx_conf_merge_str_value(conf->queue_path, prev->queue_path, "");
319 ngx_conf_merge_uint_value(conf->max_inflight, prev->max_inflight, 64);
320 ngx_conf_merge_str_value(conf->stagecmd, prev->stagecmd, "");
321 ngx_conf_merge_uint_value(conf->copymax, prev->copymax,
322 BRIX_FRM_COPYMAX_DEFAULT);
323 ngx_conf_merge_msec_value(conf->stage_ttl, prev->stage_ttl, 600000);
324 ngx_conf_merge_uint_value(conf->stage_wait, prev->stage_wait, 30);
325 ngx_conf_merge_value(conf->async_recall, prev->async_recall, 0);
326 ngx_conf_merge_msec_value(conf->fail_backoff_ms, prev->fail_backoff_ms,
327 BRIX_FRM_FAIL_BACKOFF_DEFAULT_MS);
328 ngx_conf_merge_uint_value(conf->fail_retries, prev->fail_retries,
329 BRIX_FRM_FAIL_RETRIES_DEFAULT);
330 ngx_conf_merge_msec_value(conf->copy_timeout, prev->copy_timeout, 0);
331 ngx_conf_merge_str_value(conf->stagemsg, prev->stagemsg, "");
332 ngx_conf_merge_str_value(conf->control_dir, prev->control_dir, "");
333 ngx_conf_merge_uint_value(conf->purge_hi_ppm, prev->purge_hi_ppm, 0);
334 ngx_conf_merge_uint_value(conf->purge_lo_ppm, prev->purge_lo_ppm, 0);
335 ngx_conf_merge_off_value(conf->purge_max_bytes, prev->purge_max_bytes, 0);
336 ngx_conf_merge_msec_value(conf->purge_interval_ms, prev->purge_interval_ms,
337 300000);
338 if (brix_frm_purge_policy_merge(cf, conf, prev) != NGX_CONF_OK) {
339 return NGX_CONF_ERROR; /* 2.0 F4: independent of enable */
340 }
341
342 if (!conf->enable) {
343 frm_warn_inert_knobs(cf, &raw);
344 return NGX_CONF_OK;
345 }
346 if (frm_check_enabled(cf, conf) != NGX_CONF_OK) {
347 return NGX_CONF_ERROR;
348 }
349 return frm_engine_publish(cf, conf, &raw);
350}

353static ngx_uint_t
354frm_ratio_to_ppm(const ngx_str_t *s)
355{
356 char buf[32];
357 double v;
358 size_t n = s->len;
359
360 if (n == 0 || n >= sizeof(buf)) {
361 return 0;
362 }
363 ngx_memcpy(buf, s->data, n);
364 buf[n] = '\0';
365 v = strtod(buf, NULL);
366 if (buf[n - 1] == '%' || v > 1.0) { /* a percentage like "95" or "95%" */
367 v /= 100.0;
368 }
369 if (v < 0.0) { v = 0.0; }
370 if (v > 1.0) { v = 1.0; }
371 return (ngx_uint_t) (v * 1000000.0 + 0.5);
372}

389 }
390 return NGX_CONF_OK;
