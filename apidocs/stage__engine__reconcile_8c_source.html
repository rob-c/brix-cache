1/*
2 * stage_engine_reconcile.c — restart replay of the durable staging journal
3 * (split from stage_engine.c, phase-79).
4 *
5 * WHAT: On worker start, re-drives every persisted staged-flush record so no
6 * write-back transfer is lost across a crash: snapshot the *.req names,
7 * decode each (size-tolerantly), and re-flush the recoverable ones through
8 * the export's rebuilt stage decorator with the persisted owner credential.
9 *
10 * WHY: stage_engine.c owned three concerns and exceeded the file-size cap.
11 * Restart reconcile is a self-contained concern reached only through the
12 * journal seam (stage_journal_dir) and the public dead-letter helper. The
13 * replayable-vs-droppable classification (only a FLUSH with a durable stage
14 * copy replays; an UPLOAD/MULTIPART body is gone so its record is dropped)
15 * and the deny dead-letter cap are unchanged by the split — this is a
16 * re-home, not a behavior change.
17 *
18 * HOW: brix_stage_reconcile snapshots the journal names first (records are
19 * unlinked while driving), then calls stage_reconcile_one per name, which
20 * reads + decodes the record, rebuilds the export stack via
21 * brix_vfs_backend_resolve, unwraps to the stage decorator, and calls
22 * brix_sd_stage_reflush; a permanent deny goes through stage_deny_terminal.
23 */

41/* Re-flush ONE persisted record.
42 *
43 * WHAT: Reads the journal record (up to sizeof brix_sreq_t bytes), decodes it
44 * with size-tolerance via brix_sreq_decode, then re-drives a FLUSH through
45 * the export's stage decorator with the persisted owner credential.
46 *
47 * WHY: Only a staged-write FLUSH is replayable: its bytes are durable on the
48 * stage store and the export anchor lets us rebuild both tiers. An
49 * UPLOAD/MULTIPART (client body -> stage) cannot be replayed — the body is
50 * gone after a crash, so the client retries; such records are dropped.
51 * The size-tolerant decode allows journals written before the cred field
52 * was appended to be replayed with a zeroed cred (= service-credential flush,
53 * preserving pre-feature semantics).
54 *
55 * HOW: Read into a max-size buffer, call brix_sreq_decode; on decode failure
56 * drop the record. Pass the cred (NULL when key[0]=='\0') to reflush.
57 * Returns 1 replayed / 0 kept (retry) / -1 dropped. */
58/* 2.0 F3: the tape tier at the bottom of the export a journaled archive seal
59 * is anchored to, or NULL with *why = the drop reason. */
60static brix_sd_instance_t *
61stage_archive_tier(const brix_sreq_t *rec, ngx_log_t *log, const char **why)
62{
63 brix_sd_instance_t *inst;
64
65 if (rec->export_root[0] == '\0') {
66 *why = "not-anchored";
67 return NULL;
68 }
69 inst = brix_vfs_backend_resolve(rec->export_root, log);
70 if (brix_sd_cache_instance_is(inst)) {
71 inst = brix_sd_cache_source_instance(inst);
72 }
73 if (brix_sd_stage_instance_is(inst)) {
74 inst = brix_sd_stage_source_instance(inst);
75 }
76 if (!brix_sd_frm_instance_is(inst)) {
77 *why = "no-tape-tier";
78 return NULL;
79 }
80 return inst;
81}

83/* A seal errno that no retry can cure: the record is dropped with this reason. */
84static const char *
85stage_seal_drop_reason(int err)
86{
87 switch (err) {
88 case EINVAL: return "not-a-marker"; /* the key is not a completion marker */
89 case ENOENT: return "not-online"; /* the marker was withdrawn */
90 case ENOTSUP: return "no-archiver"; /* the tier lost its ?arc= decorator */
91 default: return NULL;
92 }

95/* 2.0 F3: re-drive a journaled archive seal (kind archive) -- the restart
96 * replay and the brix_frm_fail_backoff sweep both land here. Same contract
97 * as the flush branch: 1 sealed (record removed), -1 dropped/dead-lettered,
98 * 0 kept FAILED for a later retry. */
99static int
100stage_reconcile_archive(const char *path, brix_sreq_t *rec, ngx_log_t *log)
101{
102 brix_sd_instance_t *inst;
103 const char *why = NULL;
104 int saved_errno = 0;
105 char att[24];
106
107 inst = stage_archive_tier(rec, log, &why);
108 if (inst != NULL) {
109 errno = 0;
110 if (brix_sd_frm_seal(inst, rec->dst_key) == NGX_OK) {
111 brix_stage_events_emit("engine", "replayed", rec->reqid, rec->dst_key,
112 "attempts",
113 brix_stage_events_num(att, sizeof(att),
114 rec->attempts),
115 NULL);
116 (void) unlink(path); /* sealed: the record is done */
117 return 1;
118 }
119 saved_errno = errno ? errno : EIO;
120 why = stage_seal_drop_reason(saved_errno);
121 }
122 if (why != NULL) {
123 brix_stage_events_emit("engine", "dropped", rec->reqid, rec->dst_key,
124 "reason", why, NULL);
125 ngx_log_error(NGX_LOG_WARN, log, 0,
126 "xrootd stage: archive seal of \"%s\" (export \"%s\") dropped: %s",
127 rec->dst_key, rec->export_root, why);
128 (void) unlink(path);
129 return -1;
130 }
131 if (stage_retry_terminal(brix_stage_engine_journal_dir(), rec, saved_errno, log)) {
132 return -1; /* dead-lettered = dropped from active journal */
133 }
134 ngx_log_error(NGX_LOG_WARN, log, 0,
135 "xrootd stage: archive seal of \"%s\" (export \"%s\") failed "
136 "(errno %d attempts=%uD) - record kept FAILED for retry",
137 rec->dst_key, rec->export_root, saved_errno, rec->attempts);
138 return 0;
139}

141static int
142stage_reconcile_one(const char *path, ngx_log_t *log)
143{
144 char rbuf[sizeof(brix_sreq_t)];
145 brix_sreq_t rec;
146 brix_sd_instance_t *inst;
147 const brix_stage_cred_t *credp;
148 int fd;
149 ssize_t n;
150 char att[24];
151
152 fd = open(path, O_RDONLY | O_CLOEXEC);
153 if (fd < 0) {
154 return 0;
155 }
156 n = read(fd, rbuf, sizeof(rbuf));
157 (void) close(fd);
158
159 if (brix_sreq_decode(rbuf, (size_t) n, &rec) != NGX_OK) {
160 brix_stage_events_emit("engine", "dropped", NULL, path,
161 "reason", "corrupt", NULL);
162 (void) unlink(path); /* corrupt/short/oversized record - drop */
163 return -1;
164 }
165 if (rec.kind == BRIX_STAGE_ARCHIVE) {
166 return stage_reconcile_archive(path, &rec, log);
167 }
168 if (rec.kind != BRIX_STAGE_FLUSH || rec.export_root[0] == '\0') {
169 brix_stage_events_emit("engine", "dropped", rec.reqid, rec.dst_key,
170 "reason", "not-a-flush", NULL);
171 (void) unlink(path); /* not a recoverable staged write */
172 return -1;
173 }
174
175 credp = (rec.cred.key[0] != '\0') ? &rec.cred : NULL;
176
177 /* Rebuild the export's composed stack and unwrap to its stage decorator.
178 * On success the stage copy is already dropped by reflush. On a permanent
179 * deny (EACCES from reflush + deny cred), apply the dead-letter cap so the
180 * reconcile does not re-drive the same record forever. On any other failure,
181 * keep the record for a later transient retry. */
182 inst = brix_vfs_backend_resolve(rec.export_root, log);
183 if (brix_sd_cache_instance_is(inst)) {
184 inst = brix_sd_cache_source_instance(inst);
185 }
186 if (brix_sd_stage_instance_is(inst)) {
187 ngx_int_t rc;
188 int saved_errno;
189
190 errno = 0;
191 rc = brix_sd_stage_reflush(inst, rec.dst_key, credp);
192 saved_errno = errno;
193
194 if (rc == NGX_OK) {
195 brix_stage_events_emit("engine", "replayed", rec.reqid, rec.dst_key,
196 "attempts",
197 brix_stage_events_num(att, sizeof(att),
198 rec.attempts),
199 NULL);
200 (void) unlink(path); /* re-flushed + stage copy dropped */
201 return 1;
202 }
203
204 /* Distinguish a permanent deny (EACCES, deny cred) from a transient
205 * error so dead-letter logic applies only to the former. */
206 if (saved_errno == EACCES && rec.cred.deny) {
207 /* stage_deny_terminal bumps attempts, persists, moves to deadletter
208 * when the cap is reached. The `path` is the active record; we use
209 * stage_journal_dir + reqid (not path) so the helper can rebuild the
210 * active and deadletter paths itself from the canonical dir. */
211 if (stage_deny_terminal(brix_stage_engine_journal_dir(), rec.reqid, &rec, log)) {
212 return -1; /* dead-lettered = dropped from active journal */
213 }
214 /* Below the cap: keep for retry (same as transient). */
215 ngx_log_error(NGX_LOG_WARN, log, 0,
216 "xrootd stage: reconcile denied flush \"%s\" (export \"%s\" "
217 "attempts=%uD) - record kept (below dead-letter cap)",
218 rec.dst_key, rec.export_root, rec.attempts);
219 return 0;
220 }
221
222 /* Transient re-drive failure (the origin is still unreachable): bump
223 * attempts and re-persist the record FAILED. The higher attempt count is
224 * the durable evidence that the replay re-drove this transfer — against
225 * a recovered origin the reflush above would instead have completed and
226 * unlinked the record. Kept in the active journal for the retry sweep /
227 * next restart until brix_frm_fail_retries dead-letters it (2.0 F1). */
228 if (stage_retry_terminal(brix_stage_engine_journal_dir(), &rec, saved_errno, log)) {
229 return -1; /* dead-lettered = dropped from active journal */
230 }
231 ngx_log_error(NGX_LOG_WARN, log, 0,
232 "xrootd stage: reconcile re-flush of \"%s\" (export \"%s\") failed "
233 "(errno %d attempts=%uD) - record kept FAILED for retry",
234 rec.dst_key, rec.export_root, saved_errno, rec.attempts);
235 return 0;
236 }
237
238 /* Backend unreachable / no stage tier now: keep the record for a later retry. */
239 ngx_log_error(NGX_LOG_WARN, log, 0,
240 "xrootd stage: reconcile could not re-flush \"%s\" (export \"%s\") - "
241 "record kept for retry", rec.dst_key, rec.export_root);
242 return 0;
243}

245/* Snapshot the *.req journal record names from `dir` into `names`.
246 *
247 * WHAT: Opens `dir`, collects up to `cap` names ending in ".req" (each shorter
248 * than the per-slot width) into `names`, and returns the count. Returns 0
249 * when the directory cannot be opened.
250 *
251 * WHY: The driving loop unlinks records as it replays/drops them, so the name
252 * set must be snapshotted before driving rather than iterated live.
253 *
254 * HOW: readdir loop with a suffix + length filter, copying each accepted name
255 * (with its NUL) into the caller's fixed-width array. */
256static ngx_uint_t
257stage_reconcile_snapshot(const char *dir, char names[][256], ngx_uint_t cap)
258{
259 DIR *d;
260 struct dirent *de;
261 ngx_uint_t ncount = 0;
262
263 d = opendir(dir);
264 if (d == NULL) {
265 return 0;
266 }
267 while ((de = readdir(d)) != NULL && ncount < cap) {
268 size_t nlen = strlen(de->d_name);
269
270 if (nlen > 4 && nlen < 256
271 && strcmp(de->d_name + nlen - 4, ".req") == 0)
272 {
273 memcpy(names[ncount], de->d_name, nlen + 1);
274 ncount++;
275 }
276 }
277 closedir(d);
278 return ncount;
279}

281void
282brix_stage_reconcile(brix_stage_queue_t *queue)
283{
284 char names[1024][256];
285 ngx_uint_t ncount, i, replayed = 0, kept = 0, dropped = 0;
286 ngx_log_t *log;
287
288 (void) queue;
289 if (brix_stage_engine_journal_dir()[0] == '\0') {
290 return; /* no durable journal configured */
291 }
292 /* Reconcile REPORTS every replay/keep/drop decision through ngx_log_error,
293 * whose macro dereferences the log unconditionally — a NULL log here is a
294 * crash, not a quiet run. Without a cycle (pre-init or a bare harness)
295 * DEFER: return with the journal untouched so a properly-initialised boot
296 * replays it. The in-tree caller (worker-0 init_process) always has one. */
297 if (ngx_cycle == NULL) {
298 return;
299 }
300 log = ngx_cycle->log;
301
302 /* Snapshot the *.req names first (we unlink while driving). */
303 ncount = stage_reconcile_snapshot(brix_stage_engine_journal_dir(), names, 1024);
304
305 for (i = 0; i < ncount; i++) {
306 char path[1300];
307 int r;
308
309 if ((size_t) snprintf(path, sizeof(path), "%s/%s",
310 brix_stage_engine_journal_dir(), names[i]) >= sizeof(path))
311 {
312 continue;
313 }
314 r = stage_reconcile_one(path, log);
315 if (r > 0) { replayed++; }
316 else if (r < 0) { dropped++; }
317 else { kept++; }
318 }
319
320 if (replayed > 0 || kept > 0 || dropped > 0) {
321 ngx_log_error(NGX_LOG_NOTICE, log, 0,
322 "xrootd stage: restart reconcile - %ui staged flush(es) replayed, "
323 "%ui kept for retry, %ui dropped", replayed, kept, dropped);
324 }
325}

329static int
330stage_retry_due(const char *path, ngx_uint_t min_age_sec)
331{
332 brix_sreq_t rec;
333
334 if (stage_journal_load(path, &rec) != 0 || rec.state != BRIX_SREQ_FAILED) {
335 return 0;
336 }
337 return (int64_t) time(NULL) - rec.finished_at >= (int64_t) min_age_sec;
338}

340ngx_uint_t
341brix_stage_retry_sweep(ngx_uint_t min_age_sec, ngx_log_t *log)
342{
343 char names[1024][256];
344 ngx_uint_t ncount, i, driven = 0, replayed = 0, dropped = 0;
345
346 if (brix_stage_engine_journal_dir()[0] == '\0' || log == NULL) {
347 return 0;
348 }
349 ncount = stage_reconcile_snapshot(brix_stage_engine_journal_dir(), names, 1024);
350 for (i = 0; i < ncount; i++) {
351 char path[1300];
352 int r;
353
354 if ((size_t) snprintf(path, sizeof(path), "%s/%s",
355 brix_stage_engine_journal_dir(), names[i]) >= sizeof(path)
356 || !stage_retry_due(path, min_age_sec))
357 {
358 continue;
359 }
360 driven++;
361 r = stage_reconcile_one(path, log);
362 if (r > 0) { replayed++; }
363 else if (r < 0) { dropped++; }
364 }
365 if (driven > 0) {
366 ngx_log_error(NGX_LOG_NOTICE, log, 0,
367 "xrootd stage: retry sweep - %ui FAILED record(s) re-driven, "
368 "%ui completed, %ui dead-lettered/dropped",
369 driven, replayed, dropped);
370 }
371 return driven;
372}
