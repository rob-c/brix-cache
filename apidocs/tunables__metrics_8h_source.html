178/* Increment a per-operation metric counter. No-op when metrics are disabled. */
179#define BRIX_OP_OK(ctx, op) \
180 do { if ((ctx)->metrics) { \
181 ngx_atomic_fetch_add(&(ctx)->metrics->op_ok[(op)], 1); \
182 } } while (0)
183
184#define BRIX_OP_ERR(ctx, op) \
185 do { if ((ctx)->metrics) { \
186 ngx_atomic_fetch_add(&(ctx)->metrics->op_err[(op)], 1); \
187 } } while (0)

189/*
190 * Collapse the common three-line pattern into a single macro call.
191 * Use only when brix_send_ok sends no body (NULL, 0).
192 * Handlers that return a body (read data, pgwrite status, query results)
193 * must keep the three lines explicit.
194 */
195#define BRIX_RETURN_OK(ctx, c, op, verb, path, detail, bytes) \
196 do { \
197 brix_log_access((ctx), (c), (verb), (path), (detail), \
198 1, kXR_ok, NULL, (bytes)); \
199 BRIX_OP_OK((ctx), (op)); \
200 return brix_send_ok((ctx), (c), NULL, 0); \
201 } while (0)
202
203#define BRIX_RETURN_ERR(ctx, c, op, verb, path, detail, code, msg) \
204 do { \
205 brix_log_access((ctx), (c), (verb), (path), (detail), \
206 0, (code), (msg), 0); \
207 BRIX_OP_ERR((ctx), (op)); \
208 return brix_send_error((ctx), (c), (code), (msg)); \
209 } while (0)

211/*
212 * Collapse: log_access + BRIX_OP_OK + return send_redirect.
213 * Used wherever the outcome is a successful redirect (locate, manager, etc.)
214 */
215#define BRIX_RETURN_REDIR(ctx, c, op, verb, path, detail, host, port) \
216 do { \
217 brix_log_access((ctx), (c), (verb), (path), (detail), \
218 1, kXR_ok, NULL, 0); \
219 BRIX_OP_OK((ctx), (op)); \
220 return brix_send_redirect((ctx), (c), (host), (port)); \
221 } while (0)

223/*
224 * Selection answer (phase-115 W2.1): like BRIX_RETURN_REDIR but the answer is
225 * the manager's `brix_cms_response` policy — kXR_redirect (default) or pin the
226 * session to the selected server and proxy. Use at dynamic selection sites
227 * (registry / caches / stage); static manager_map redirects keep RETURN_REDIR.
228 */
229#define BRIX_RETURN_SELECTED(ctx, c, conf, op, verb, path, detail, host, port) \
230 do { \
231 brix_log_access((ctx), (c), (verb), (path), (detail), \
232 1, kXR_ok, NULL, 0); \
233 BRIX_OP_OK((ctx), (op)); \
234 return brix_cms_answer_selected((ctx), (c), (conf), (host), (port)); \
235 } while (0)

237/*
238 * Collapse: log_access + BRIX_OP_ERR + *rc=send_error + return 0.
239 * Used in helper functions (validate_handle, parse_op_path, etc.) that
240 * signal failure to callers via an out-parameter and return int 0.
241 */
242#define BRIX_BAIL_ERR(ctx, c, op, verb, path, detail, code, msg, rc) \
243 do { \
244 brix_log_access((ctx), (c), (verb), (path), (detail), \
245 0, (code), (msg), 0); \
246 BRIX_OP_ERR((ctx), (op)); \
247 *(rc) = brix_send_error((ctx), (c), (code), (msg)); \
248 return 0; \
249 } while (0)
