26/* WHAT: Transfer bytes and advance the offset by actual progress.
27 * WHY: FreeBSD sendfile can return partial progress together with EAGAIN.
28 * HOW: Bound the request, publish completed bytes, otherwise preserve errno. */
29ssize_t brix_plat_sendfile(int destination, int source, off_t *offset, size_t count)
30{
31 off_t sent = 0;
32 int result;
33 if (offset == NULL || *offset < 0) { errno = EINVAL; return -1; }
34 if (count == 0) { return 0; }
35 if (count > SSIZE_MAX) { count = SSIZE_MAX; }
36 result = sendfile(source, destination, *offset, count, NULL, &sent, 0); /* vfs-seam-allow: SEAM_CORRECT - native PAL sendfile */
37 if (sent > 0) { *offset += sent; return (ssize_t) sent; }
38 return result < 0 ? -1 : 0;
