1/*
2 * cms_select.c — the one place a manager turns "data server X was selected for
3 * this request" into an answer for the client (phase-115 W2.1).
4 *
5 * WHAT: brix_cms_answer_selected() emits kXR_redirect — the stock cmsd/xrootd
6 * behaviour and the default — or, under `brix_cms_response proxy`, pins
7 * the session to the selected server and relays the in-flight request
8 * through the transparent proxy (src/net/proxy) so the client never
9 * leaves the gateway.
10 * WHY: NAT'd / firewalled sites and stable-endpoint deployments cannot expose
11 * every data server to clients; a single chokepoint keeps the
12 * redirect-vs-proxy policy out of the six selection sites (open, locate,
13 * stat, dirlist, query-checksum, mutations) and the CMS wake path.
14 * HOW: policy check → brix_send_redirect; else brix_proxy_dispatch_to() with
15 * the selected host:port as the pinned upstream. The host has already
16 * passed the registry store choke point (brix_net_host_chars_valid) or
17 * the wake path's identical check, so nothing is re-parsed here. The
18 * proxy's own fail budget (BRIX_PROXY_MAX_CONN_FAILS) bounds a dead
19 * selection: the client gets kXR_error, never a hang.
20 */
21#include "protocols/root/response/response.h"
22#include "protocols/root/read/locate.h"
23#include "proxy.h"

25ngx_int_t
26brix_cms_answer_selected(brix_ctx_t *ctx, ngx_connection_t *c,
27 ngx_stream_brix_srv_conf_t *conf, const char *host, uint16_t port)
28{
29 if (conf->cms.response != BRIX_CMS_RESPONSE_PROXY) {
30 return brix_send_redirect(ctx, c, host, port);
31 }
32
33 /* A proxying gateway is, to its clients, a data server: kXR_locate names
34 * the gateway itself. Forwarding it would leak the selected node's
35 * address and the client's next open would bypass the gateway. */
36 if (ctx->recv.cur_reqid == kXR_locate) {
37 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
38 "brix: cms select: locate answered as self "
39 "(selected %s:%ud)", host, (unsigned) port);
40 return brix_locate_answer_self(ctx, c, conf);
41 }
42
43 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
44 "brix: cms select: proxying session to %s:%ud",
45 host, (unsigned) port);
46 return brix_proxy_dispatch_to(ctx, c, conf, host, port);
47}
