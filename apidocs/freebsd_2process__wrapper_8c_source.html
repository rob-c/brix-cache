15/* WHAT: Open a native counter channel. WHY: Preserve paired descriptor ABI.
16 * HOW: Translate PAL flags and publish only a successfully opened eventfd. */
17int brix_plat_wakefd_open(int *reader, int *writer, int flags)
18{
19 int descriptor = eventfd(0, ((flags & BRIX_EVENTFD_CLOEXEC) ? EFD_CLOEXEC : 0)
20 | ((flags & BRIX_EVENTFD_NONBLOCK) ? EFD_NONBLOCK : 0));
21 if (descriptor < 0) { return -1; }
22 *reader = descriptor;
23 *writer = descriptor;
24 return 0;

55 * WHY: The impersonation broker authenticates its worker through this boundary.

98/* WHAT: Return the canonical current executable path, never a truncated prefix.
99 * WHY: Helpers must execute the exact installed image. HOW: Query and realpath. */
100int brix_plat_self_exe(char *output, size_t capacity)
101{
102 int selector[] = {CTL_KERN, KERN_PROC, KERN_PROC_PATHNAME, -1};
103 char path[PATH_MAX], canonical[PATH_MAX];
104 size_t length = sizeof(path);
105
106 if (sysctl(selector, 4, path, &length, NULL, 0) != 0) { return -1; }
107 if (length == 0 || length > sizeof(path) || path[length - 1] != '\0') {
108 errno = EIO; return -1;
109 }
110 if (realpath(path, canonical) == NULL) { return -1; }
111 length = strlen(canonical) + 1;
112 if (output == NULL || length > capacity) { errno = ENAMETOOLONG; return -1; }
113 memcpy(output, canonical, length);
114 return 0;
115}

138/* WHAT: Wait for exit without reaping, returning 1 only on the deadline.
139 * WHY: Let the owning lifecycle code collect its own child status.
140 * HOW: Use kqueue NOTE_EXIT and a monotonic absolute deadline across EINTR. */
141int brix_plat_wait_pid_timeout(pid_t pid, unsigned timeout_ms)
142{
143 struct kevent event;
144 struct timespec started, now, remaining;
145 uint64_t deadline, current;
146 int queue = kqueue(), ready;
147
148 if (queue < 0) { return 0; }
149 EV_SET(&event, (uintptr_t) pid, EVFILT_PROC, EV_ADD | EV_ONESHOT, NOTE_EXIT, 0, NULL);
150 if (kevent(queue, &event, 1, NULL, 0, NULL) < 0) { close(queue); return 0; }
151 clock_gettime(CLOCK_MONOTONIC, &started);
152 deadline = (uint64_t) started.tv_sec * 1000000000u + started.tv_nsec
153 + (uint64_t) timeout_ms * 1000000u;
154 do {
155 clock_gettime(CLOCK_MONOTONIC, &now);
156 current = (uint64_t) now.tv_sec * 1000000000u + now.tv_nsec;
157 current = deadline > current ? deadline - current : 0;
158 remaining.tv_sec = (time_t) (current / 1000000000u);
159 remaining.tv_nsec = (long) (current % 1000000000u);
160 ready = kevent(queue, NULL, 0, &event, 1, &remaining);
161 } while (ready < 0 && errno == EINTR);
162 close(queue);
163 return ready == 0;
164}
