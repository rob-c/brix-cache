1/*
2 * backend_async_queue.h — durable write-behind queue for backend namespace
3 * mutations (brix_backend_async).
4 *
5 * WHAT: A per-worker coalescing queue for backend MUTATIONS (unlink / rmdir /
6 * rename / mkdir). When `brix_backend_async on` is set for a server, a mutation
7 * is (1) fsync'd to a durable journal record, (2) appended to the worker's
8 * pending batch with a protocol-supplied completion callback, and (3) the
9 * client is parked. The batch is drained in bulk — driven against the backend
10 * with the pool-less VFS path primitives — when it reaches `batch` ops OR the
11 * oldest op has waited `wait_ms` (whichever comes first). Each op's real result
12 * is then delivered to its parked client via the callback, and the durable
13 * record is removed. On worker start, brix_baq_reconcile() re-applies any op
14 * left un-drained by a crash (idempotently), so a mutation that the client was
15 * told to wait for is never silently lost across a reboot.
16 *
17 * WHY: Operators front high-latency / rate-limited origins (tape, object stores,
18 * remote XRootD) where a burst of individual namespace mutations is far more
19 * expensive than one coalesced flush. The queue trades per-op latency (the
20 * client blocks, and may time out — an accepted contract) for bulk backend
21 * efficiency and, crucially, durability: the reply is only sent once the op is
22 * truly on the backend, and a crash mid-batch replays rather than loses it.
23 * Reads / stats / lists are never queued — only mutations, so the namespace a
24 * subsequent request sees is always the post-flush truth for THIS client.
25 *
26 * HOW: The durable substrate mirrors the stage engine's journal (a fsync'd fixed-
27 * size record per op, scanned + replayed on boot) but keeps its OWN record type
28 * and OWN journal subdirectory (<stage_journal_dir>/backend/) so the two
29 * concerns stay decoupled — a namespace mutation is not a byte transfer and must
30 * not ride the stage mover / deny-cred flush machinery. The pending list and the
31 * park state are per-worker (the worker that parked a client is the only one
32 * that may wake it), so no SHM is needed — unlike the cross-worker stage_waiter.
33 * The core is protocol-agnostic: the root:// and S3/WebDAV adapters each supply
34 * a brix_baq_done_pt that wakes their own client with the delivered errno.
35 */

45/* The four coalescable namespace mutations. Values are stable on-disk identities
46 * (a journal record survives a restart), so they are append-only — never renumber. */
47typedef enum {
48 BRIX_BAQ_UNLINK = 0, /* remove a regular file */
49 BRIX_BAQ_RMDIR = 1, /* remove an empty directory */
50 BRIX_BAQ_RENAME = 2, /* rename src_key -> dst_key */
51 BRIX_BAQ_MKDIR = 3 /* create a directory (mode) */
52} brix_baq_op_t;
53
54/* The durable on-disk record — one per queued mutation, fixed size (fsync'd whole,
55 * decoded size-exactly on replay). root_canon anchors the backend the op is
56 * re-driven against on reconcile; src_key/dst_key are export-relative logical
57 * paths (dst_key used only by RENAME). APPEND-ONLY layout: never reorder or shrink
58 * a field, or old journals become unreadable. */
59typedef struct {
60 char reqid[40]; /* minted request id (NUL-terminated) */
61 uint32_t op; /* brix_baq_op_t */
62 char root_canon[1024]; /* export root (canonical) to resolve backend */
63 char src_key[1024]; /* target path; RENAME source */
64 char dst_key[1024]; /* RENAME destination; "" for single-target ops */
65 uint32_t mode; /* MKDIR mode; 0 otherwise */
66 int64_t enqueued_at; /* wall-clock enqueue time (seconds) */
67 uint32_t attempts; /* re-drive count (durable across restart) */
68} brix_baq_rec_t;

70/*
71 * Completion callback: the core calls this exactly once per queued op when the op
72 * has been driven against the backend, on the SAME worker that enqueued it.
73 * `client` is the opaque token the adapter passed to brix_baq_enqueue (an
74 * ngx_connection_t* for root://, an ngx_http_request_t* for S3/WebDAV);
75 * `op_errno` is 0 on success or the POSIX errno the backend mutation failed with.
76 * The adapter delivers the corresponding wire/HTTP status and resumes the client.
77 */
78typedef void (*brix_baq_done_pt)(void *client, int op_errno);

80/*
81 * Configure the queue's durable journal subdirectory for this worker.
82 *
83 * WHAT: Derives <stage_journal_dir>/backend/ from the stage engine's journal dir
84 * (the single source of truth, set by brix_stage_engine_init) and creates it
85 * (0700). A NULL/empty stage journal dir leaves the queue in-memory only (no
86 * crash recovery) — the coalescing still works, only durability is off.
87 *
88 * WHY: Keeps the async-queue records physically separate from the stage engine's
89 * *.req records so each subsystem's boot reconcile scans only its own.
90 *
91 * HOW: Called from the worker stage-engine init, AFTER brix_stage_engine_init.
92 */
93void brix_baq_init(void);

95/*
96 * Enqueue one mutation and park the caller's client.
97 *
98 * WHAT: Persists a durable record (when a journal dir is configured), appends the
99 * op to the worker's pending batch with its completion callback, and — if
100 * the batch now meets the size trigger — drains immediately (which invokes
101 * `done` for every op including this one, inline). Otherwise the op stays
102 * parked until a later size/time-triggered drain.
103 *
104 * WHY: The single entry point every protocol adapter calls after it has resolved
105 * + authorized the mutation and decided (from brix_backend_async) to defer.
106 *
107 * HOW: req->root_canon / src_key / dst_key / mode populate the record; batch and
108 * wait_ms are this server's coalescing thresholds (stored per-op so a worker
109 * shared by servers with different tunables still flushes each correctly).
110 * Returns NGX_OK when the op is queued (or drained inline), NGX_ERROR on a
111 * bad argument, a READ-ONLY endpoint (errno EROFS — phase-105: deferring a
112 * mutation is still a mutation, and the refusal must land before the client
113 * is parked), or a full queue that could not be drained (the caller then
114 * runs the op inline / errors, never silently drops it).
115 */
116/*
117 * WHAT: one deferred mutation as the adapter describes it — the verb, the
118 * endpoint's phase-105 write posture, the confining export root, the
119 * target key(s), the MKDIR mode, and this server's coalescing thresholds.
120 * WHY: the queue outlives the request that fills it, so the authority to
121 * mutate has to travel WITH the op rather than be re-derived at drain
122 * time from configuration that may since have been reloaded. Bundling
123 * also keeps brix_baq_enqueue at three parameters.
124 * HOW: a caller-owned value; brix_baq_enqueue copies everything it keeps.
125 * `policy` zero-initialises to READ_ONLY, so an adapter that forgets it
126 * has its op refused rather than silently queued.
127 */
128typedef struct {
129 brix_baq_op_t op;
130 brix_vfs_mutation_policy_t policy; /* endpoint write posture */
131 brix_proto_t proto; /* attribution for a refusal */
132 const char *root_canon; /* confining export root */
133 const char *src_key; /* target; RENAME source */
134 const char *dst_key; /* RENAME destination, else NULL */
135 uint32_t mode; /* MKDIR mode, else 0 */
136 ngx_uint_t batch; /* size trigger */
137 ngx_msec_t wait_ms; /* time trigger */
138} brix_baq_req_t;

143/*
144 * Per-worker timer tick: drain the batch if the time trigger has fired.
145 *
146 * WHAT: If any parked op has waited >= its wait_ms, drains the whole batch now
147 * (delivering each op's result to its client). A no-op when the batch is
148 * empty or nothing has aged out yet.
149 *
150 * WHY: The size trigger fires inline in enqueue; the time trigger needs a clock
151 * source. Called from the same per-worker timer that drives the stage
152 * scheduler tick.
153 *
154 * HOW: Compares ngx_current_msec against each pending op's enqueue stamp.
155 */
156void brix_baq_tick(void);
157
158/*
159 * Drop any parked op belonging to `client` without delivering (best-effort).
160 *
161 * WHAT: Removes pending ops whose opaque client token equals `client` (their
162 * durable records are LEFT on disk, so a still-un-drained mutation is
163 * replayed at next boot rather than lost). Used when a parked client
164 * disconnects before its batch drains.
165 *
166 * WHY: A woken callback must never touch a freed connection/request. The adapter
167 * calls this from its disconnect/cleanup handler so the drain skips the dead
168 * client — the mutation itself still lands (inline drain of the rest, or
169 * reconcile).
170 */
171void brix_baq_drop_client(void *client);
172
173/*
174 * Restart replay: re-apply every persisted mutation left un-drained by a crash.
175 *
176 * WHAT: Scans <stage_journal_dir>/backend/, decodes each record, re-runs the op
177 * against its backend IDEMPOTENTLY (unlink/rmdir of a missing target = done,
178 * mkdir of an existing dir = done, rename of an already-moved source = done),
179 * and removes the record on success. Called on worker 0 at startup.
180 *
181 * WHY: The client was told to wait for the op; durability means a crash between
182 * "journalled" and "drained" must not lose it.
183 *
184 * HOW: Mirrors brix_stage_reconcile's snapshot-then-drive loop; no client to wake
185 * (the parked connections died with the crash), so it only re-drives the
186 * backend effect.
187 */
188void brix_baq_reconcile(void);
