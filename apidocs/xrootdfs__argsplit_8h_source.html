1/*
2 * xrootdfs_argsplit.h — the command-line split rule shared by both FUSE
3 * transports (xrootdfs.c's aio parser and xrootdfs_legacy_ext.c's legacy
4 * parser).
5 *
6 * WHAT: xfs_arg_passthrough() places one token the transport's own option
7 * matcher did NOT consume — the first bare word is the endpoint, later
8 * bare words and any unrecognised dash-arg fall through to libfuse.
9 * WHY: the two parsers honour different option sets but must split the
10 * endpoint / mountpoint / fuse-passthrough the same way; one helper
11 * keeps that rule from drifting between transports.
12 * HOW: header-only, static inline — no build registration, no shared TU.
13 */
14#ifndef XROOTDFS_ARGSPLIT_H
15#define XROOTDFS_ARGSPLIT_H

24/* Dispatch one command-line token the caller's option matcher did not consume:
25 * a bare word (is_opt == 0) becomes *endpoint when none is set yet, otherwise it
26 * and every unrecognised dash-arg pass through to fuse_argv (bounded at
27 * XFS_FUSE_ARGV_MAX). */
28static inline void
29xfs_arg_passthrough(char *arg, int is_opt, char **fuse_argv, int *fuse_argc,
30 const char **endpoint)
31{
32 if (!is_opt && *endpoint == NULL) {
33 *endpoint = arg;
34 } else if (*fuse_argc < XFS_FUSE_ARGV_MAX) {
35 fuse_argv[(*fuse_argc)++] = arg;
36 }
37}
38
39/* Append the host's own mount options (brix_plat_fuse_host_opts: none on
40 * Linux, "noappledouble" for macFUSE) as one "-o" pair, ahead of the user's
41 * arguments so an explicit later "-o" can still override them. */
42static inline void
43xfs_add_host_opts(char **fuse_argv, int *fuse_argc)
44{
45 const char *opts = brix_plat_fuse_host_opts();
46
47 if (opts != NULL && *fuse_argc + 2 <= XFS_FUSE_ARGV_MAX) {
48 fuse_argv[(*fuse_argc)++] = (char *) "-o";
49 fuse_argv[(*fuse_argc)++] = (char *) opts;
50 }
51}

53/* Rewrite a libfuse "-o" comma list in place, dropping every element this
54 * host's libfuse would reject (brix_plat_fuse_opt_supported): one option
55 * string then mounts on Linux libfuse3 and macFUSE alike. Returns the new
56 * length; 0 means nothing survived (the caller drops the "-o" pair). */
57static inline size_t
58xfs_filter_fuse_opts(char *list)
59{
60 char *out = list;
61 char *save = NULL;
62 char *copy = strdup(list);
63 char *tok;
64
65 if (copy == NULL) {
66 return strlen(list); /* ENOMEM: pass the list unchanged */
67 }
68 *out = '\0';
69 for (tok = strtok_r(copy, ",", &save); tok; tok = strtok_r(NULL, ",", &save)) {
70 if (!brix_plat_fuse_opt_supported(tok)) {
71 continue;
72 }
73 if (out != list) {
74 *out++ = ',';
75 }
76 memcpy(out, tok, strlen(tok) + 1);
77 out += strlen(tok);
78 }
79 free(copy);
80 return (size_t) (out - list);
81}
