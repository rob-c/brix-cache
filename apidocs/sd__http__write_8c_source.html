1/*
2 * sd_http_write.c — write path for the HTTP-origin storage driver (SP3): the
3 * HTTP/WebDAV origin as a writable cache_store / stage_store.
4 *
5 * WHAT: The staged-upload vtable slots — staged_open/staged_open_cred (buffer a
6 * whole object, capturing the per-user credential for the commit),
7 * staged_write (sequential append into the growable buffer), staged_commit
8 * (one PUT of the whole object — atomic from the reader's view), and
9 * staged_abort.
10 *
11 * WHY: Split out of sd_http.c (phase-79 file-size split): the write path is one
12 * concept, distinct from the read/credential path (sd_http_read.c),
13 * selection/failover (sd_http_select.c), the driver vtable/lifecycle
14 * (sd_http.c), and the namespace mutations DELETE/MKCOL/MOVE
15 * (sd_http_mutate.c). Writes never fail over — a write to a non-primary
16 * origin would split-brain the store — so they always target endpoint 0
17 * via sd_http_write_path (sd_http_select.c) rather than sd_http_request_fo.
18 *
19 * HOW: The per-user identity (bearer header / x509 client-cert path) is captured
20 * at staged_open and applied at commit — mirroring the read leg exactly,
21 * because an HTTP origin has no kernel fd / session to re-scope per user.
22 * cred_gate (sd_http_read.c) refuses a proxy-only cred the transport cannot
23 * present in deny mode, just like the read leg.
24 */

35/* #12 outbound integrity: base64(MD5(buf,len)) → out[25] ("" on failure). The
36 * classic RFC 1864 Content-MD5 the origin re-computes over the received body and
37 * rejects on mismatch (400/412) — the outbound analogue of the ingest
38 * s3_content_md5_verify gate. ngx-free (this backend layer builds into
39 * libxrdproto): OpenSSL EVP only, no ngx_* / ngx pool. */
40static void
41sd_http_content_md5(const void *buf, size_t len, char out[25])
42{
43 unsigned char md[EVP_MAX_MD_SIZE];
44 unsigned int mdlen = 0;
45
46 out[0] = '\0';
47 if (EVP_Digest(buf, len, md, &mdlen, EVP_md5(), NULL) != 1 || mdlen != 16) {
48 return;
49 }
50 EVP_EncodeBlock((unsigned char *) out, md, 16); /* 16 → 24 chars + NUL */
51}

53/* Memory-safety cap on the whole-object staged buffer (phase-107 C1, W2): this
54 * transport PUTs from memory, so the buffer holds the ENTIRE object — before this
55 * cap a large upload grew worker heap without bound. Past it the append refuses
56 * ENOSPC (capacity error -> kXR_NoSpace / 507), never a truncated object. Reorder
57 * absorption does NOT live here — that is the VFS spill (vfs_writer_spill.c),
58 * which drains sequentially and is disk-backed; this bounds what one commit PUT
59 * may hold in heap. A streaming/multipart PUT lifting the limit is a transport
60 * follow-up. */
61#define SD_HTTP_STAGED_MAX (1u << 30) /* 1 GiB */
62
63/* Per-staged-write state: HTTP has no streaming PUT through this transport, so the
64 * object is buffered and PUT whole at commit (a remote stage/cache store of typical
65 * file sizes; very large objects are a multipart follow-up).
66 *
67 * auth_hdr / cert_pem carry the per-open (per-user) credential captured at
68 * staged_open_cred so the commit PUT authenticates to the origin AS the requesting
69 * user rather than the static service credential (phase-70 §5.1, write leg). Both
70 * are COPIES — the cred fields are borrowed only for the staged_open() call. "" in
71 * either falls back to the instance static (is->auth_hdr) / no client cert. Exactly
72 * one kind is ever set (the VFS gate populates one of bearer / x509_proxy). */
73typedef struct {
74 char path[SD_HTTP_PATH_MAX];
75 u_char *buf;
76 size_t len;
77 size_t cap;
78 char auth_hdr[SD_HTTP_AUTH_MAX];
79 char cert_pem[SD_HTTP_PATH_MAX];
80} sd_http_staged_state;

82/* sd_http_staged_open_common — shared staged-open path for the plain and
83 * credential-scoped staged-open slots.
84 *
85 * WHAT: Allocates the staged buffer state, composes the write-target URL path
86 * and, when a `cred` is present, captures the per-user credential (bearer →
87 * Authorization header; x509 proxy → mutual-TLS client-cert PATH) into the
88 * staged state so the commit PUT presents THAT identity, not the static
89 * service credential.
90 * WHY: Phase-70 §5.1 write leg — an HTTP/WebDAV origin authenticates a PUT purely
91 * on the request credential, and this driver has no kernel fd / session to
92 * re-scope per user, so the per-user identity travels as staged state copied
93 * at open time (mirroring the read-leg sd_http_open_common exactly).
94 * HOW: cred==NULL (plain .staged_open) leaves auth_hdr AND cert_pem empty, so the
95 * commit falls back to the instance static header and no client cert. The
96 * same cred_gate the read path uses refuses a proxy-only cred in deny mode
97 * when the transport cannot mutual-TLS (request_cred==NULL) → EACCES. A
98 * bearer is snprintf'd into the staged auth_hdr; an x509 proxy path is copied
99 * into cert_pem only when the transport can present it (request_cred!=NULL).
100 * Both are COPIES — cred fields are borrowed only for this call. */
101static brix_sd_staged_t *
102sd_http_staged_open_common(brix_sd_instance_t *inst, const char *final_path,
103 mode_t mode, off_t declared_size, const brix_sd_cred_t *cred, int *err_out)
104{
105 sd_http_inst_state *is = inst->state;
106 sd_http_staged_state *ss;
107 brix_sd_staged_t *h;
108
109 (void) mode;
110
111 /* Phase-107 C5: this driver stages the WHOLE object in one heap buffer, so
112 * a declared final size past the buffer cap can only end in ENOSPC after
113 * the client streams a gigabyte — refuse it at open instead. */
114 if (declared_size > (off_t) SD_HTTP_STAGED_MAX) {
115 if (err_out) { *err_out = ENOSPC; }
116 errno = ENOSPC;
117 return NULL;
118 }
119
120 /* Same credential gate the read leg applies: a proxy-only cred that cannot be
121 * presented as a client cert must be refused in deny mode rather than served
122 * on the anonymous/service credential. */
123 if (sd_http_cred_gate(is, cred) != 0) {
124 if (err_out) { *err_out = errno; }
125 return NULL;
126 }
127
128 ss = calloc(1, sizeof(*ss));
129 h = calloc(1, sizeof(*h));
130 if (ss == NULL || h == NULL) {
131 free(ss);
132 free(h);
133 if (err_out) { *err_out = ENOMEM; }
134 return NULL;
135 }
136 sd_http_write_path(is, final_path, ss->path, sizeof(ss->path));
137
138 /* Capture the per-open credential for the commit PUT: bearer → header;
139 * x509 proxy → cert path (only when the transport can present it). Empty
140 * leaves the commit on the instance static / anonymous credential. */
141 if (cred != NULL && cred->bearer != NULL && cred->bearer[0] != '\0') {
142 snprintf(ss->auth_hdr, sizeof(ss->auth_hdr),
143 "Authorization: Bearer %s\r\n", cred->bearer);
144 }
145 if (cred != NULL && cred->x509_proxy != NULL && cred->x509_proxy[0] != '\0'
146 && is->transport->request_cred != NULL)
147 {
148 snprintf(ss->cert_pem, sizeof(ss->cert_pem), "%s", cred->x509_proxy);
149 }
150
151 h->inst = inst;
152 h->state = ss;
153 return h;
154}

183ssize_t
184sd_http_staged_write(brix_sd_staged_t *h, const void *buf, size_t len,
185 off_t off)
186{
187 sd_http_staged_state *ss = h->state;
188
189 /* Sequential append only (whole-object PUT has no random write). */
190 if ((size_t) off != ss->len) {
191 errno = ESPIPE;
192 return -1;
193 }
194 if (ss->len + len > SD_HTTP_STAGED_MAX) {
195 errno = ENOSPC; /* whole-object buffer cap — see SD_HTTP_STAGED_MAX */
196 return -1;
197 }
198 if (ss->len + len > ss->cap) {
199 size_t ncap = ss->cap ? ss->cap * 2 : (1u << 20);
200 u_char *nbuf;
201
202 while (ncap < ss->len + len) {
203 ncap *= 2;
204 }
205 if (ncap > SD_HTTP_STAGED_MAX) {
206 ncap = SD_HTTP_STAGED_MAX; /* never allocate past the cap */
207 }
208 nbuf = realloc(ss->buf, ncap);
209 if (nbuf == NULL) {
210 errno = ENOMEM;
211 return -1;
212 }
213 ss->buf = nbuf;
214 ss->cap = ncap;
215 }
216 ngx_memcpy(ss->buf + ss->len, buf, len);
217 ss->len += len;
218 return (ssize_t) len;
219}

242/* Does the write origin HONOUR RFC 7232 conditional PUT (phase-107 C6)?
243 * Lazily probed once per instance and latched in is->cond_probe: PUT an empty
244 * body to a driver-private probe key with an If-Match etag that can never
245 * hold. 412 ⇒ the origin evaluates preconditions (1). 2xx ⇒ it IGNORED the
246 * header — it published the probe object unconditionally (deleted again
247 * best-effort) — so a real precondition would be a silent lie there (-1).
248 * Any other outcome (auth refusal, transport error) is NOT latched: the
249 * verdict stays unknown and this commit refuses without condemning the
250 * origin. Returns the verdict for this call: 1 / -1 / 0 (unknown). */
251static int
252sd_http_cond_probe(sd_http_inst_state *is, sd_http_staged_state *ss,
253 const char *auth_hdr)
254{
255 char probe_path[SD_HTTP_PATH_MAX];
256 char hdrs[SD_HTTP_AUTH_MAX + 64];
257 brix_s3_resp_t resp;
258 char errbuf[256];
259 int verdict = atomic_load(&is->cond_probe);
260
261 if (verdict != 0) {
262 return verdict;
263 }
264 sd_http_write_path(is, "/.brix-cond-probe", probe_path,
265 sizeof(probe_path));
266 snprintf(hdrs, sizeof(hdrs), "If-Match: \"brix-cond-probe-never\"\r\n%s",
267 auth_hdr ? auth_hdr : "");
268 if (sd_http_staged_request(is, ss, "PUT", probe_path, hdrs, "", 0,
269 &resp, errbuf, sizeof(errbuf)) != 0)
270 {
271 return 0; /* transport: unknown, unlatched */
272 }
273 if (resp.status == 412) {
274 verdict = 1;
275 } else if (resp.status >= 200 && resp.status < 300) {
276 verdict = -1; /* ignored — and it published */
277 }
278 is->transport->resp_free(&resp);
279 if (verdict == -1) {
280 /* best-effort cleanup of the probe object the origin let through */
281 if (sd_http_staged_request(is, ss, "DELETE", probe_path, auth_hdr,
282 NULL, 0, &resp, errbuf,
283 sizeof(errbuf)) == 0)
284 {
285 is->transport->resp_free(&resp);
286 }
287 }
288 if (verdict != 0) {
289 atomic_store(&is->cond_probe, verdict);
290 }
291 return verdict;
292}

294/* Build the RFC 7232 conditional line for the commit PUT (phase-107 C6). The
295 * typed precondition is carried on the one commit PUT so the ORIGIN decides
296 * atomically — but only after the lazy probe has seen this origin answer 412.
297 * An origin that ignores the header would publish unconditionally, so anything
298 * short of a positive probe refuses ENOTSUP (§3.5 — a refusal over an
299 * emulation that lies). MATCH_META is np: HTTP has no size/mtime conditional.
300 * Leaves cond_line empty for an unconditional commit. */
301static ngx_int_t
302sd_http_cond_line(sd_http_inst_state *is, sd_http_staged_state *ss,
303 const brix_sd_precond_t *pre, const char *auth_hdr,
304 char *cond_line, size_t cap)
305{
306 cond_line[0] = '\0';
307 if (pre == NULL || pre->kind == BRIX_SD_PRECOND_NONE) {
308 return NGX_OK;
309 }
310 if (pre->kind == BRIX_SD_PRECOND_ABSENT) {
311 snprintf(cond_line, cap, "If-None-Match: *\r\n");
312 } else if (pre->kind == BRIX_SD_PRECOND_MATCH_ETAG
313 && pre->etag != NULL && pre->etag_len > 0
314 && pre->etag_len < cap - 16)
315 {
316 snprintf(cond_line, cap, "If-Match: %.*s\r\n",
317 (int) pre->etag_len, pre->etag);
318 } else {
319 errno = (pre->kind == BRIX_SD_PRECOND_MATCH_ETAG) ? EINVAL : ENOTSUP;
320 return NGX_ERROR;
321 }
322 if (sd_http_cond_probe(is, ss, auth_hdr) != 1) {
323 errno = ENOTSUP;
324 return NGX_ERROR;
325 }
326 return NGX_OK;
327}

329/* Map the commit PUT's terminal status to the staged-commit contract. A 412 on
330 * a conditional commit is the origin refusing the precondition — ECANCELED is
331 * the typed verdict, an ABSENT refusal the contract's EEXIST — and the refusal
332 * was decided AT the origin, so it is atomic (C6 advisory metric). */
333static ngx_int_t
334sd_http_commit_status(long status, int conditional, brix_sd_precond_t *pre)
335{
336 if (status == 412 && conditional) {
337 errno = brix_sd_precond_absent(pre) ? EEXIST : ECANCELED;
338 pre->atomic = 1;
339 return NGX_ERROR;
340 }
341 if (status != 200 && status != 201 && status != 204) {
342 errno = (status == 403 || status == 401) ? EACCES : EIO;
343 return NGX_ERROR;
344 }
345 return NGX_OK;
346}

348ngx_int_t
349sd_http_staged_commit(brix_sd_staged_t *h, brix_sd_precond_t *pre)
350{
351 sd_http_staged_state *ss = h->state;
352 sd_http_inst_state *is = h->inst->state;
353 brix_s3_resp_t resp;
354 char errbuf[256];
355 const char *auth_hdr;
356 const char *hdrs;
357 char hdr_block[SD_HTTP_AUTH_MAX + 256];
358 char cond_line[160];
359 int rq;
360 ngx_int_t rc = NGX_OK;
361
362 /* Per-user commit: a per-open bearer (staged_open_cred) wins over the
363 * instance's static bearer_token; "" (plain staged_open, or no usable cred)
364 * falls back. A per-open x509 proxy path (ss->cert_pem) is presented as the
365 * mutual-TLS client cert on the PUT via the transport's request_cred slot —
366 * exactly mirroring the read leg (sd_http_request_fo / sd_http_pread). The
367 * cred gate at staged_open already refused a proxy-only cred that cannot be
368 * presented in deny mode, so reaching here with a cert path guarantees a
369 * request_cred-capable transport. */
370 auth_hdr = ss->auth_hdr[0] ? ss->auth_hdr
371 : (is->auth_hdr[0] ? is->auth_hdr : NULL);
372
373 /* Typed publish precondition (phase-107 C6): the RFC 7232 line for the
374 * commit PUT, probe-gated — see sd_http_cond_line. */
375 if (sd_http_cond_line(is, ss, pre, auth_hdr, cond_line,
376 sizeof(cond_line)) != NGX_OK)
377 {
378 return NGX_ERROR;
379 }
380
381 /* #12 outbound integrity: when put_checksum is on, prepend a Content-MD5 line
382 * (base64 MD5 of the whole staged object) to the header block so the origin
383 * re-computes it over the received body and rejects a wire-corrupted PUT with
384 * 400/412 rather than silently committing poison. The transport splits this
385 * CRLF block into request headers (s3o_build_slist). Off ⇒ the auth header
386 * passes through unchanged (byte-frozen prior behaviour). The C6 conditional
387 * line joins the same block. */
388 hdrs = auth_hdr;
389 {
390 char md5b64[25];
391
392 md5b64[0] = '\0';
393 if (is->put_checksum) {
394 sd_http_content_md5(ss->buf, ss->len, md5b64);
395 }
396 if (md5b64[0] != '\0' || cond_line[0] != '\0') {
397 snprintf(hdr_block, sizeof(hdr_block), "%s%s%s%s%s",
398 cond_line,
399 md5b64[0] ? "Content-MD5: " : "", md5b64,
400 md5b64[0] ? "\r\n" : "",
401 auth_hdr ? auth_hdr : "");
402 hdrs = hdr_block;
403 }
404 }
405
406 rq = sd_http_staged_request(is, ss, "PUT", ss->path, hdrs, ss->buf,
407 ss->len, &resp, errbuf, sizeof(errbuf));
408 /* Ownership contract (brix_vfs_staged_commit / sd_remote_staged_commit):
409 * free the handle ONLY on success. On failure it stays valid for the
410 * caller's staged_abort — freeing here would double-free. */
411 if (rq != 0) {
412 errno = EIO;
413 return NGX_ERROR;
414 }
415 rc = sd_http_commit_status(resp.status, cond_line[0] != '\0', pre);
416 is->transport->resp_free(&resp);
417 if (rc != NGX_OK) {
418 return rc;
419 }
420 if (cond_line[0] != '\0') {
421 pre->atomic = 1; /* the probed origin decided */
422 }
423 free(ss->buf);
424 free(ss);
425 free(h);
426 return NGX_OK;
427}
