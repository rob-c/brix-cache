1/* request.c — proxy-mode target extraction + allowlist.
2 *
3 * WHAT: for absolute-form request lines ("GET http://s1:8000/cvmfs/..."),
4 * extract the authority nginx already parsed (r->host_start/end,
5 * r->port_start/end) and check it against brix_cvmfs_upstream_allow.
6 * WHY: a CVMFS site proxy is a proxy for the site's Stratum-1s ONLY —
7 * without the allowlist it is an open HTTP proxy.
8 * HOW: no parsing of our own: nginx validates absolute-form targets during
9 * request-line parsing and r->uri already holds just the path, so the
10 * classifier and every layer below see exactly the reverse-mode shape.
11 */
12#include "cvmfs.h"

14/* ---- Validate the absolute-form request scheme ----
15 *
16 * WHAT: returns NGX_OK when the absolute-form target scheme is acceptable on
17 * this listener, else NGX_HTTP_FORBIDDEN. http:// is always allowed;
18 * https:// is allowed only on a secure (scvmfs://) listener.
19 *
20 * WHY: WLCG proxy traffic is plain HTTP, so an https target on a cleartext
21 * cvmfs:// listener means a misconfigured client, not a feature request.
22 * The secure (T22) listener lifts this to permit https authorities (and
23 * TLS upstream connects), gated on ctx->secure.
24 *
25 * HOW: 1. Fetch the per-request cvmfs ctx (carries the secure flag).
26 * 2. Classify the parsed schema token as exactly "http" or "https".
27 * 3. Refuse anything that is neither.
28 * 4. Refuse https unless the ctx exists and is marked secure.
29 */
30static ngx_int_t
31brix_cvmfs_proxy_scheme_check(ngx_http_request_t *r)
32{
33 ngx_http_brix_cvmfs_ctx_t *ctx =
34 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
35 size_t sl = (size_t) (r->schema_end - r->schema_start);
36 int is_http = (sl == 4
37 && ngx_strncasecmp(r->schema_start, (u_char *) "http", 4) == 0);
38 int is_https = (sl == 5
39 && ngx_strncasecmp(r->schema_start, (u_char *) "https", 5) == 0);
40
41 if (!is_http && !is_https) {
42 return NGX_HTTP_FORBIDDEN;
43 }
44 if (is_https && (ctx == NULL || !ctx->secure)) {
45 return NGX_HTTP_FORBIDDEN;
46 }
47 return NGX_OK;
48}

50static u_char *
51brix_cvmfs_authority_host_end(u_char *start, u_char *authority_end,
52 u_char **port_start)
53{
54 u_char *cursor;
55
56 *port_start = NULL;
57 if (*start == '[') {
58 cursor = start + 1;
59 while (cursor < authority_end && *cursor != ']') {
60 cursor++;
61 }
62 if (cursor == authority_end || (cursor + 1 < authority_end
63 && cursor[1] != ':'))
64 {
65 return NULL;
66 }
67 if (cursor + 1 < authority_end) {
68 *port_start = cursor + 2;
69 }
70 return cursor + 1;
71 }
72
73 for (cursor = start; cursor < authority_end; cursor++) {
74 if (*cursor == ':') {
75 *port_start = cursor + 1;
76 return cursor;
77 }
78 }
79 return authority_end;
80}

82/* ---- Extract the host and port from the validated authority span ----
83 *
84 * WHAT: fills host and port from the absolute-form target, defaulting the port
85 * to 80. Returns NGX_OK, or NGX_HTTP_BAD_REQUEST for malformed bounds,
86 * bracket syntax, separators, or a port outside 1..65535.
87 *
88 * WHY: nginx versions disagree on whether r->host_end points at the colon or
89 * after the complete host:port authority. The security allowlist is
90 * host-only, so using that unstable endpoint can reject valid packaged
91 * builds or accidentally compare the wrong authority bytes.
92 *
93 * HOW: 1. Bound the authority by nginx's validated host_start and uri_start.
94 * 2. For bracketed IPv6, retain the brackets and find the closing one;
95 * otherwise the first colon separates the host from its port.
96 * 3. Require any trailing authority bytes to be exactly :<valid-port>.
97 */
98static ngx_int_t
99brix_cvmfs_proxy_extract_authority(ngx_http_request_t *r, ngx_str_t *host,
100 in_port_t *port)
101{
102 u_char *authority_end, *host_end, *port_start;
103 ngx_int_t parsed_port;
104
105 if (r->host_start == NULL || r->uri_start == NULL
106 || r->host_start >= r->uri_start)
107 {
108 return NGX_HTTP_BAD_REQUEST;
109 }
110
111 authority_end = r->uri_start;
112 host_end = brix_cvmfs_authority_host_end(r->host_start, authority_end,
113 &port_start);
114 if (host_end == NULL) {
115 return NGX_HTTP_BAD_REQUEST;
116 }
117
118 if (host_end == r->host_start) {
119 return NGX_HTTP_BAD_REQUEST;
120 }
121 host->data = r->host_start;
122 host->len = (size_t) (host_end - r->host_start);
123 *port = 80;
124 if (port_start != NULL) {
125 if (port_start == authority_end) {
126 return NGX_HTTP_BAD_REQUEST;
127 }
128 parsed_port = ngx_atoi(port_start,
129 (size_t) (authority_end - port_start));
130 if (parsed_port < 1 || parsed_port > 65535) {
131 return NGX_HTTP_BAD_REQUEST;
132 }
133 *port = (in_port_t) parsed_port;
134 }
135 return NGX_OK;
136}

138/* ---- Match the target host against the Stratum-1 allowlist ----
139 *
140 * WHAT: returns NGX_OK when host is listed in cc->upstream_allow (case-
141 * insensitive), else NGX_HTTP_FORBIDDEN.
142 *
143 * WHY: the allowlist is the sole thing separating a CVMFS site proxy from an
144 * open HTTP proxy; an unset or empty allowlist means proxy mode is off,
145 * so every absolute-form target is refused.
146 *
147 * HOW: 1. Treat an unset/empty allowlist as proxy-off and refuse.
148 * 2. Scan the entries for a length-exact, case-insensitive host match.
149 * 3. Return NGX_OK on a hit, NGX_HTTP_FORBIDDEN otherwise.
150 */
151static ngx_int_t
152brix_cvmfs_proxy_host_allowed(const brix_cvmfs_conf_t *cc, const ngx_str_t *host)
153{
154 ngx_str_t *allow;
155 ngx_uint_t i;
156
157 if (cc->upstream_allow == NULL || cc->upstream_allow->nelts == 0) {
158 /* allowlist unset = proxy mode off: absolute-form always refused */
159 return NGX_HTTP_FORBIDDEN;
160 }
161 allow = cc->upstream_allow->elts;
162 for (i = 0; i < cc->upstream_allow->nelts; i++) {
163 if (allow[i].len == host->len
164 && ngx_strncasecmp(allow[i].data, host->data, host->len) == 0)
165 {
166 return NGX_OK;
167 }
168 }
169 return NGX_HTTP_FORBIDDEN;
170}

172/* ---- Extract and authorize an absolute-form proxy target ----
173 *
174 * WHAT: for an absolute-form request line, fills host/port with the target
175 * authority and returns NGX_OK when it is an allowed Stratum-1. Returns
176 * NGX_DECLINED for origin-form (reverse mode), or an HTTP error status
177 * (FORBIDDEN/BAD_REQUEST) when the scheme, port, or host is rejected.
178 *
179 * WHY: a CVMFS site proxy proxies the site's Stratum-1s ONLY; this gate keeps
180 * it from acting as an open HTTP proxy while leaving reverse-mode
181 * requests to fall through to the classifier.
182 *
183 * HOW: 1. Absent host_start means origin-form → NGX_DECLINED.
184 * 2. Enforce the scheme policy (http always, https only when secure).
185 * 3. Extract the bounded host and port independent of nginx-version
186 * host_end semantics.
187 * 4. Authorize the request-target host against the Stratum-1 allowlist.
188 */
189ngx_int_t
190brix_cvmfs_proxy_target(ngx_http_request_t *r, const brix_cvmfs_conf_t *cc,
191 ngx_str_t *host, in_port_t *port)
192{
193 ngx_int_t rc;
194
195 if (r->host_start == NULL) {
196 return NGX_DECLINED; /* origin-form: reverse mode */
197 }
198
199 rc = brix_cvmfs_proxy_scheme_check(r);
200 if (rc != NGX_OK) {
201 return rc;
202 }
203
204 rc = brix_cvmfs_proxy_extract_authority(r, host, port);
205 if (rc != NGX_OK) {
206 return rc;
207 }
208
209 return brix_cvmfs_proxy_host_allowed(cc, host);
210}
