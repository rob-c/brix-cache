1/* File: tpc io_xfer — the blocking full-transfer byte loop, free of nginx
2 * WHAT: brix_tpc_xfer_all() — loop until exactly `len` bytes have moved in one
3 * direction, over a raw blocking fd or an OpenSSL session.
4 *
5 * WHY: see io_xfer.h. The single copy of the EINTR / <=0 / INT_MAX rules is
6 * the point: TPC's wire framing reads a fixed-size ServerResponseHdr and
7 * then a length-prefixed body, so a loop that returns early on a partial
8 * transfer desynchronises the stream rather than failing it.
9 *
10 * HOW: cursor + remaining, decremented by whatever the syscall reports moved.
11 * EINTR retries; every other short-circuit (<=0) is a hard failure, which
12 * covers peer close, RST and SSL errors alike — the caller has no useful
13 * recovery from a half-moved frame, so the distinction is not surfaced.
14 * The SSL branch clamps to INT_MAX because SSL_read/SSL_write take int. */

22brix_tpc_xfer_all(SSL *ssl, int fd, void *buf, size_t len, int sending)
23{
24 unsigned char *cursor = buf;
25
26 while (len > 0) {
27 ssize_t moved;
28
29 if (ssl != NULL) {
30 int chunk = (int) (len > INT_MAX ? INT_MAX : len);
31 int n = sending ? SSL_write(ssl, cursor, chunk)
32 : SSL_read(ssl, cursor, chunk);
33 if (n <= 0) {
34 return -1;
35 }
36 moved = n;
37 } else {
38 moved = sending ? send(fd, cursor, len, 0)
39 : recv(fd, cursor, len, 0);
40 if (moved < 0 && errno == EINTR) {
41 continue;
42 }
43 if (moved <= 0) {
44 return -1;
45 }
46 }
47
48 cursor += (size_t) moved;
49 len -= (size_t) moved;
50 }
51
52 return 0;
53}
