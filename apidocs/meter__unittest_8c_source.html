46static void
47test_loadavg(void)
48{
49 uint8_t pct = 77;
50
51 CHECK(brix_cms_meter_parse_loadavg("2.00 1.50 1.00 2/345 6789\n", 4,
52 &pct) == 0);
53 CHECK(pct == 50); /* 2.0 load on 4 cpus = 50% */
54
55 CHECK(brix_cms_meter_parse_loadavg("8.00 0 0", 4, &pct) == 0);
56 CHECK(pct == 100); /* clamped at 100 */
57
58 CHECK(brix_cms_meter_parse_loadavg("0.00 0 0", 4, &pct) == 0);
59 CHECK(pct == 0);
60
61 /* error legs: garbage / no cpus / NULL */
62 pct = 77;
63 CHECK(brix_cms_meter_parse_loadavg("garbage", 4, &pct) == -1);
64 CHECK(brix_cms_meter_parse_loadavg("1.0 0 0", 0, &pct) == -1);
65 CHECK(brix_cms_meter_parse_loadavg(NULL, 4, &pct) == -1);
66 CHECK(pct == 77); /* out untouched on failure */
67}

153static void
154test_rate_pct(void)
155{
156 /* 125 MB over 1s against a 125 MB/s reference = 100% */
157 CHECK(brix_cms_meter_rate_pct(125000000, 1000, 125000000) == 100);
158 /* half rate = 50% */
159 CHECK(brix_cms_meter_rate_pct(62500000, 1000, 125000000) == 50);
160 /* over-reference clamps at 100 */
161 CHECK(brix_cms_meter_rate_pct(999999999, 1000, 125000000) == 100);
162 /* zero elapsed / zero reference never divide */
163 CHECK(brix_cms_meter_rate_pct(1000, 0, 125000000) == 0);
164 CHECK(brix_cms_meter_rate_pct(1000, 1000, 0) == 0);
165}

168static void
169test_sample_driver(void)
170{
171 /* The first sample reports zero deltas; the second consumes monotonic PAL
172 * counters without ever accepting a configurable host path. */
173 brix_cms_meter_t m;
174 uint8_t out5[5] = {1, 2, 3, 4, 5};
175
176 memset(&m, 0, sizeof(m));
177 brix_cms_meter_sample(&m, 1000, out5);
178 CHECK(out5[0] == 50);
179 CHECK(out5[1] == 0); /* unprimed delta */
180 CHECK(out5[2] == 0); /* xeq always 0 */
181 CHECK(out5[3] == 75);
182 CHECK(out5[4] == 0); /* unprimed delta */
183
184 brix_cms_meter_sample(&m, 2000, out5); /* primed second sample */
185 CHECK(out5[1] <= 100);
186 CHECK(out5[4] <= 100);
187}
