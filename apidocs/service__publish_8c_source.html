25/*
26 * Durability is a property of the domain (§3.3), not of the medium or a
27 * per-operator directive: REGISTRY answers 201 Created, a JOURNAL replay is the
28 * recovery mechanism, and CONFIG is operator truth — all three must be on stable
29 * storage before the caller proceeds. CACHE and STAGE are reconstructible (a
30 * re-fetch, a retried upload), so they rename only. CREDENTIAL never routes here
31 * (its arm-explicit verb lives in cred_write.c); EXPORT is refused by the domain
32 * claim before this is consulted.
33 */
34static int
35service_domain_durable(brix_vfs_domain_t domain)
36{
37 return domain == BRIX_VFS_DOMAIN_REGISTRY
38 || domain == BRIX_VFS_DOMAIN_JOURNAL
39 || domain == BRIX_VFS_DOMAIN_CONFIG;
40}

42/*
43 * Fsync a caller-staged file that arrived without a held fd, so its data is
44 * stable before its name is (the durable-domain _fd path). Confined beneath
45 * root_canon and O_NOFOLLOW so a symlink swapped in at stage_path can redirect
46 * neither the flush nor, by extension, the file the rename will publish.
47 * Returns 0, or -1 with errno preserved.
48 */
49static int
50service_fsync_stage(ngx_log_t *log, const char *root_canon,
51 const char *stage_path)
52{
53 const char *rel;
54 int rootfd, fd, e;
55
56 rootfd = brix_beneath_open_root(root_canon);
57 if (rootfd < 0) {
58 return -1;
59 }
60 rel = brix_beneath_strip_root(root_canon, stage_path);
61 if (rel == NULL) {
62 close(rootfd);
63 errno = EXDEV;
64 return -1;
65 }
66 fd = brix_open_beneath(rootfd, rel,
67 O_RDONLY | O_NOFOLLOW | O_CLOEXEC, 0);
68 e = errno;
69 close(rootfd);
70 if (fd < 0) {
71 errno = e;
72 return -1;
73 }
74 if (fsync(fd) != 0) {
75 e = errno;
76 ngx_log_error(NGX_LOG_ERR, log, e,
77 "brix: service publish fsync of staged \"%s\" failed — "
78 "not publishing", stage_path);
79 close(fd);
80 errno = e;
81 return -1;
82 }
83 close(fd);
84 return 0;
85}

87/*
88 * Seal the write handle before the commit: fsync the data on a durable domain
89 * (D1), restore the published mode, then CHECK the close (D3) — a deferred
90 * write-back error surfaces here, not silently at a bare close(). Leaves
91 * staged->fd = NGX_INVALID_FILE so the commit's own seal is a no-op. Returns 0,
92 * or -1 with errno set and the fd already closed.
93 */
94static int
95service_seal_fd(const brix_service_publish_req_t *req,
96 brix_staged_file_t *staged)
97{
98 int fd = staged->fd;
99 int e;
100
101 if (service_domain_durable(req->domain) && fsync(fd) != 0) {
102 e = errno;
103 ngx_log_error(NGX_LOG_ERR, req->log, e,
104 "brix: service publish fsync failed — not publishing "
105 "\"%s\"", req->final_path);
106 close(fd);
107 staged->fd = NGX_INVALID_FILE;
108 errno = e;
109 return -1;
110 }
111 /* The temp was created 0600 (private in-flight); publish the intended mode.
112 * Service storage is never impersonated, so the fd fchmod reaches it — no
113 * broker path is needed. Best-effort, as in staged_seal_temp. */
114 if (req->mode != 0) {
115 (void) fchmod(fd, req->mode);
116 }
117 /* close() on a modern Linux frees the fd even on EINTR, so a retry would
118 * close a fd we no longer own; treat only a genuine error as a failure. */
119 if (close(fd) != 0 && errno != EINTR) {
120 e = errno;
121 ngx_log_error(NGX_LOG_ERR, req->log, e,
122 "brix: service publish close failed — not publishing "
123 "\"%s\"", req->final_path);
124 staged->fd = NGX_INVALID_FILE;
125 errno = e;
126 return -1;
127 }
128 staged->fd = NGX_INVALID_FILE;
129 return 0;
130}

132/*
133 * The shared tail: seal the staged handle (staged->fd, the write fd for a bytes
134 * publish or the caller's held fd adopted onto the struct), then commit through
135 * the confined durable publish. When no fd is held, a durable domain fsyncs the
136 * staged file by path first. On any failure no file survives — the pre-commit
137 * paths abort the staged temp; the commit owns its own cleanup and needs no
138 * second abort. errno (EEXIST on the excl arm included) is preserved.
139 */
140static ngx_int_t
141service_publish_finish(const brix_service_publish_req_t *req,
142 brix_staged_file_t *staged)
143{
144 int e;
145
146 if (staged->fd != NGX_INVALID_FILE) {
147 if (service_seal_fd(req, staged) != 0) {
148 e = errno;
149 brix_staged_abort(req->log, req->root_canon, staged, 1);
150 errno = e;
151 return NGX_ERROR;
152 }
153 } else if (service_domain_durable(req->domain)
154 && service_fsync_stage(req->log, req->root_canon,
155 staged->tmp_path) != 0)
156 {
157 e = errno;
158 brix_staged_abort(req->log, req->root_canon, staged, 1);
159 errno = e;
160 return NGX_ERROR;
161 }
162
163 if ((req->excl
164 ? brix_staged_commit_excl(req->log, req->root_canon, staged,
165 req->final_path)
166 : brix_staged_commit(req->log, req->root_canon, staged,
167 req->final_path)) != NGX_OK)
168 {
169 return NGX_ERROR; /* commit preserved errno and cleaned up */
170 }
171 return NGX_OK;
172}

174ngx_int_t
175brix_service_publish_bytes(const brix_service_publish_req_t *req,
176 const void *bytes, size_t len)
177{
178 brix_staged_open_req_t open_req;
179 brix_staged_file_t staged;
180 const u_char *p = bytes;
181 ssize_t n;
182 int e;
183
184 if (req == NULL || req->root_canon == NULL || req->final_path == NULL
185 || (len > 0 && bytes == NULL))
186 {
187 errno = EINVAL;
188 return NGX_ERROR;
189 }
190
191 if (brix_vfs_domain_claim(req->log, req->domain, BRIX_VFS_MUTATE_PUBLISH)
192 != NGX_OK)
193 {
194 return NGX_ERROR; /* EROFS on EXPORT, EINVAL out of range */
195 }
196
197 ngx_memzero(&open_req, sizeof(open_req));
198 open_req.root_canon = req->root_canon;
199 open_req.final_path = req->final_path;
200 open_req.mode = req->mode;
201 open_req.open_flags = O_WRONLY; /* staged_open adds O_CREAT | O_EXCL */
202
203 if (brix_staged_open(req->log, &open_req, &staged) != NGX_OK) {
204 return NGX_ERROR;
205 }
206
207 while (len > 0) {
208 n = write(staged.fd, p, len);
209 if (n < 0) {
210 if (errno == EINTR) {
211 continue;
212 }
213 e = errno;
214 ngx_log_error(NGX_LOG_ERR, req->log, e,
215 "brix: service publish write to \"%s\" failed",
216 req->final_path);
217 brix_staged_abort(req->log, req->root_canon, &staged, 1);
218 errno = e;
219 return NGX_ERROR;
220 }
221 p += n;
222 len -= (size_t) n;
223 }
224
225 return service_publish_finish(req, &staged);
226}

228ngx_int_t
229brix_service_publish_fd(const brix_service_publish_req_t *req,
230 ngx_fd_t fd, const char *stage_path)
231{
232 brix_staged_file_t staged;
233
234 if (req == NULL || req->root_canon == NULL || req->final_path == NULL
235 || stage_path == NULL)
236 {
237 errno = EINVAL;
238 return NGX_ERROR;
239 }
240
241 if (brix_vfs_domain_claim(req->log, req->domain, BRIX_VFS_MUTATE_PUBLISH)
242 != NGX_OK)
243 {
244 return NGX_ERROR;
245 }
246
247 if (ngx_strlen(stage_path) >= sizeof(staged.tmp_path)) {
248 errno = ENAMETOOLONG;
249 return NGX_ERROR;
250 }
251
252 /* Adopt the caller's already-written file as the staged temp: the commit
253 * renames it confined beneath root_canon and runs the C3 barrier, exactly
254 * as for a temp this unit opened. A held fd (not NGX_INVALID_FILE) is sealed
255 * by the shared tail; NGX_INVALID_FILE means the caller kept no fd, and a
256 * durable domain reopens and fsyncs stage_path by path. */
257 ngx_memzero(&staged, sizeof(staged));
258 staged.fd = fd;
259 staged.active = 1;
260 staged.final_mode = req->mode;
261 ngx_cpystrn((u_char *) staged.tmp_path, (u_char *) stage_path,
262 sizeof(staged.tmp_path));
263
264 return service_publish_finish(req, &staged);
265}
