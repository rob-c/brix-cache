4/*
5 * Path prefix rewriting and file handle translation helpers for the transparent XRootD proxy.
6 *
7 * WHAT:
8 * Path prefix rewriting and file handle translation helpers.
9 *
10 * WHY:
11 * The proxy translates paths between client-facing namespace and upstream
12 * filesystem namespace (e.g., /data → /mnt/storage).
13 * kXR_prepare payloads contain newline-separated path lists that need
14 * per-line rewriting.
15 * File handles are local to the client session but must be translated to
16 * upstream handle IDs before forwarding.
17 *
18 * INVARIANT:
19 * All rewrote buffers allocated via ngx_alloc;
20 * original request freed on success when reallocation occurred
21 * (caller responsibility tracked via total_out).
22 *
23 * HOW:
24 * proxy_rewrite_path:
25 * - Three strategies: no-match pass-through, same-length in-place rewrite,
26 * longer-path reallocation with dlen update
27 * proxy_rewrite_prepare_payload:
28 * - Two-pass approach: calculate length + check needs_rewrite first,
29 * then copy/rewrite
30 * proxy_translate_fh:
31 * - Looks up local handle in fh_map and replaces byte at offset
32 * - Returns -1 for invalid/unmapped handles
33 */
34

37/* proxy_rewrite_path — apply the strip/add prefix transform to a request payload
38 * carrying one filesystem path (transparent proxy: client /data → upstream
39 * /mnt/storage); same-length rewrites are in-place, longer results reallocate and
40 * update the dlen header. */
41u_char *
42proxy_rewrite_path(ngx_connection_t *c,
43 ngx_stream_brix_srv_conf_t *conf,
44 u_char *req, size_t total,
45 size_t path_off, size_t path_len,
46 size_t *total_out)
47{
48 const u_char *strip = conf->proxy.path_strip.data;
49 size_t slen = conf->proxy.path_strip.len;
50 const u_char *add = conf->proxy.path_add.data;
51 size_t alen = conf->proxy.path_add.len;
52 size_t new_path_len, new_total;
53 u_char *new_req;
54
55 if (slen == 0 || path_len < slen
56 || ngx_strncmp(req + path_off, strip, slen) != 0)
57 {
58 *total_out = total;
59 return req; /* no match — pass through unchanged */
60 }
61
62 new_path_len = alen + (path_len - slen);
63 new_total = total - path_len + new_path_len;
64
65 if (new_path_len == path_len) {
66 /* Same length: rewrite in-place */
67 ngx_memcpy(req + path_off, add, alen);
68 *total_out = total;
69 return req;
70 }
71
72 new_req = ngx_alloc(new_total, c->log);
73 if (new_req == NULL) {
74 *total_out = total;
75 return req; /* OOM: pass original unchanged */
76 }
77
78 /* Copy header portion before path, then new path, then anything after */
79 ngx_memcpy(new_req, req, path_off);
80 ngx_memcpy(new_req + path_off, add, alen);
81 ngx_memcpy(new_req + path_off + alen,
82 req + path_off + slen,
83 total - path_off - slen);
84
85 /* Update the 4-byte dlen field in the request header (bytes 20-23) */
86 {
87 uint32_t new_dlen = htonl((uint32_t) (new_total - XRD_REQUEST_HDR_LEN));
88 ngx_memcpy(new_req + 20, &new_dlen, 4);
89 }
90
91 *total_out = new_total;
92 return new_req;
93}

95/*
96 * WHAT: File-local view of the strip/add prefix transform plus the payload
97 * slice a kXR_prepare rewrite operates over.
98 * WHY: proxy_rewrite_prepare_payload's two passes (measure, then copy) share the
99 * same conf-derived prefixes and payload bounds. Bundling them into one
100 * frozen, file-local struct lets the extracted pass helpers stay under the
101 * 5-parameter cap without threading six loose scalars through each call.
102 * HOW: Populated once by proxy_rewrite_prepare_payload from conf + req/total,
103 * then passed by const pointer to the pass helpers. Never mutated after
104 * construction; carries no ownership.
105 */
106typedef struct {
107 const u_char *strip; /* prefix to strip from each matching line */
108 size_t slen; /* length of strip */
109 const u_char *add; /* prefix to prepend in place of strip */
110 size_t alen; /* length of add */
111 const u_char *payload; /* first byte after the request header */
112 size_t plen; /* payload length in bytes */
113} proxy_prepare_rewrite_t;
114
115/*
116 * WHAT: Compute the length of the line that ends at cursor p, given its start.
117 * WHY: Both prepare passes must treat a final line without a trailing newline as
118 * inclusive of p, but a '\n'-terminated line as exclusive of p. Centralizing
119 * the off-by-one keeps the two passes byte-identical.
120 * HOW: p is a boundary (either '\n' or the payload's last byte). If p is the last
121 * byte and is not itself '\n', the line includes p (+1); otherwise it ends
122 * just before p.
123 */
124static size_t
125proxy_prepare_line_len(const u_char *line_start, const u_char *p,
126 const u_char *last)
127{
128 if (p == last && *p != '\n') {
129 return (size_t)(p - line_start + 1);
130 }
131 return (size_t)(p - line_start);
132}
133
134/*
135 * WHAT: Report whether a payload line begins with the strip prefix.
136 * WHY: The match test (long-enough line AND prefix equality) gates every
137 * rewrite decision in both passes; a single helper keeps them consistent.
138 * HOW: True when line_len covers slen and the leading slen bytes equal strip.
139 */
140static ngx_uint_t
141proxy_prepare_line_matches(const proxy_prepare_rewrite_t *rw,
142 const u_char *line_start, size_t line_len)
143{
144 return (line_len >= rw->slen
145 && ngx_strncmp(line_start, rw->strip, rw->slen) == 0);
146}

148/*
149 * WHAT: First pass — compute the rewritten payload's total request size and
150 * whether any line actually needs rewriting.
151 * WHY: The copy pass must allocate the exact output size up front, and the whole
152 * operation is skipped (original buffer returned) when no line matches.
153 * HOW: Walk the payload line by line; matching lines contribute alen +
154 * (line_len - slen), others contribute line_len; a terminating '\n' adds
155 * one byte. Accumulates onto the header length and sets *needs_rewrite.
156 */
157static size_t
158proxy_prepare_measure(const proxy_prepare_rewrite_t *rw,
159 ngx_uint_t *needs_rewrite)
160{
161 const u_char *p = rw->payload;
162 const u_char *line_start = rw->payload;
163 const u_char *last = rw->payload + rw->plen - 1;
164 size_t new_total = XRD_REQUEST_HDR_LEN;
165
166 *needs_rewrite = 0;
167
168 while (p < rw->payload + rw->plen) {
169 if (*p == '\n' || p == last) {
170 size_t line_len = proxy_prepare_line_len(line_start, p, last);
171
172 if (proxy_prepare_line_matches(rw, line_start, line_len)) {
173 new_total += rw->alen + (line_len - rw->slen);
174 *needs_rewrite = 1;
175 } else {
176 new_total += line_len;
177 }
178 if (*p == '\n') {
179 new_total++;
180 }
181 line_start = p + 1;
182 }
183 p++;
184 }
185 return new_total;
186}
187
188/*
189 * WHAT: Second pass — emit the rewritten payload into dst.
190 * WHY: Once the size is known and a rewrite is required, each line is copied
191 * verbatim or with its strip prefix replaced by add, preserving line order
192 * and newline placement byte-for-byte.
193 * HOW: Walk the payload identically to the measure pass; for matching lines write
194 * add then the post-strip remainder, otherwise copy the line as-is; re-emit
195 * a trailing '\n' when the boundary byte was one.
196 */
197static void
198proxy_prepare_emit(const proxy_prepare_rewrite_t *rw, u_char *dst)
199{
200 const u_char *p = rw->payload;
201 const u_char *line_start = rw->payload;
202 const u_char *last = rw->payload + rw->plen - 1;
203
204 while (p < rw->payload + rw->plen) {
205 if (*p == '\n' || p == last) {
206 size_t line_len = proxy_prepare_line_len(line_start, p, last);
207
208 if (proxy_prepare_line_matches(rw, line_start, line_len)) {
209 ngx_memcpy(dst, rw->add, rw->alen);
210 dst += rw->alen;
211 ngx_memcpy(dst, line_start + rw->slen, line_len - rw->slen);
212 dst += (line_len - rw->slen);
213 } else {
214 ngx_memcpy(dst, line_start, line_len);
215 dst += line_len;
216 }
217 if (*p == '\n') {
218 *dst++ = '\n';
219 }
220 line_start = p + 1;
221 }
222 p++;
223 }
224}

226/*
227 * proxy_rewrite_prepare_payload — apply path rewriting to a kXR_prepare payload.
228 * The payload is a newline-separated list of paths.
229 */
230u_char *
231proxy_rewrite_prepare_payload(ngx_connection_t *c,
232 ngx_stream_brix_srv_conf_t *conf,
233 u_char *req, size_t total,
234 size_t *total_out)
235{
236 proxy_prepare_rewrite_t rw;
237 u_char *new_req;
238 size_t new_total;
239 ngx_uint_t needs_rewrite = 0;
240
241 if (conf->proxy.path_strip.len == 0 || total <= XRD_REQUEST_HDR_LEN) {
242 *total_out = total;
243 return req;
244 }
245
246 rw.strip = conf->proxy.path_strip.data;
247 rw.slen = conf->proxy.path_strip.len;
248 rw.add = conf->proxy.path_add.data;
249 rw.alen = conf->proxy.path_add.len;
250 rw.payload = req + XRD_REQUEST_HDR_LEN;
251 rw.plen = total - XRD_REQUEST_HDR_LEN;
252
253 new_total = proxy_prepare_measure(&rw, &needs_rewrite);
254 if (!needs_rewrite) {
255 *total_out = total;
256 return req;
257 }
258
259 new_req = ngx_alloc(new_total, c->log);
260 if (new_req == NULL) {
261 *total_out = total;
262 return req;
263 }
264
265 ngx_memcpy(new_req, req, XRD_REQUEST_HDR_LEN);
266 proxy_prepare_emit(&rw, new_req + XRD_REQUEST_HDR_LEN);
267
268 /* Update dlen in header */
269 {
270 uint32_t dlen_be = htonl((uint32_t)(new_total - XRD_REQUEST_HDR_LEN));
271 ngx_memcpy(new_req + 20, &dlen_be, 4);
272 }
273
274 ngx_free(req);
275 *total_out = new_total;
276 return new_req;
277}

281/*
282 * proxy_reject_request — free the forwarded-request buffer and queue a
283 * kXR_error to the client.
284 *
285 * WHY: brix_send_error() returns NGX_OK once the error frame is queued, so a
286 * translate handler that did `ngx_free(req); return brix_send_error(...)`
287 * handed NGX_OK back to brix_proxy_forward_request(), which read that as
288 * "continue forwarding" and went on to copy (save_wait_retry), tap and queue
289 * the freed buffer as an owned wbuf — a use-after-free and a later double
290 * free. Mapping the queued-error case to NGX_ABORT gives the forwarding
291 * pipeline an unambiguous "stop, req is gone, session continues" signal.
292 */
293ngx_int_t
294proxy_reject_request(brix_ctx_t *ctx, ngx_connection_t *c,
295 u_char *req, uint16_t errcode, const char *msg)
296{
297 ngx_int_t rc;
298
299 ngx_free(req);
300 rc = brix_send_error(ctx, c, errcode, msg);
301 return (rc == NGX_OK) ? NGX_ABORT : rc;
302}

304/*
305 * Translate the local file handle named by buf[offset] into the upstream's own
306 * 4-byte fhandle, written over buf[offset..offset+3].
307 *
308 * brix's own open response hands the client a 1-byte local index followed by
309 * three zero bytes, so buf[offset] alone identifies the slot on the way in;
310 * what goes back out is the upstream's handle in full. A slot that is merely
311 * PENDING has no upstream handle yet and is refused rather than translated to
312 * a sentinel. Returns 0 on success, -1 if the local handle is invalid.
313 */
314int
315proxy_translate_fh(brix_proxy_ctx_t *proxy, u_char *buf, size_t offset)
316{
317 int local_fh = (int)(unsigned char) buf[offset];
318
319 if (local_fh < 0 || local_fh >= BRIX_MAX_FILES
320 || proxy->fh_map[local_fh].fh_state != BRIX_PROXY_FH_BOUND)
321 {
322 return -1;
323 }
324 brix_proxy_fh_put(&proxy->fh_map[local_fh], buf + offset);
325 return 0;
326}
