87char *
88brix_conf_set_seccomp(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
89{
90 ngx_uint_t *field = (ngx_uint_t *) ((char *) conf + cmd->offset);
91 ngx_conf_enum_t *e = cmd->post;
92 ngx_str_t *value = cf->args->elts;
93 ngx_uint_t i;
94
95 if (*field != NGX_CONF_UNSET_UINT && *field != BRIX_SECCOMP_OFF) {
96 return "is duplicate";
97 }
98 for (i = 0; e[i].name.len != 0; i++) {
99 if (e[i].name.len == value[1].len
100 && ngx_strcasecmp(e[i].name.data, value[1].data) == 0)
101 {
102 *field = e[i].value;
103 /* Bump the process-global to the strictest requested anywhere. */
104 if (e[i].value > brix_seccomp_get_worker_mode()) {
105 brix_seccomp_set_worker_mode(e[i].value);
106 }
107 return NGX_CONF_OK;
108 }
109 }
110 return "invalid value";
111}

126{
127 ngx_str_t *value = cf->args->elts;
128
129 (void) cmd;
130 (void) conf;
131
132 if (value[1].len == 3 && ngx_strncasecmp(value[1].data,
133 (u_char *) "off", 3) == 0)
134 {
135 brix_seccomp_set_allow_exec(0); /* ratchet toward strict: off wins */
136 return NGX_CONF_OK;
137 }
138 if (value[1].len == 2 && ngx_strncasecmp(value[1].data,
139 (u_char *) "on", 2) == 0)
140 {
141 /* on is the default; a no-op that must NOT override a prior `off`
142 * (fail-secure — once any server disabled exec, restart to re-enable). */
143 return NGX_CONF_OK;
144 }
145 return "must be \"on\" or \"off\"";
146}

148/* Sink passed to the core: turns a per-syscall or whole-filter failure into an
149 * EMERG line. `name == NULL` denotes a whole-filter failure (init/load). */
150static void
151brix_seccomp_log_err(void *ud, const char *name, int rc)
152{
153 ngx_cycle_t *cycle = ud;
154
155 if (name != NULL) {
156 ngx_log_error(NGX_LOG_EMERG, cycle->log, 0,
157 "brix_seccomp: rule add for \"%s\" failed: %s",
158 name, strerror(-rc));
159 } else if (rc != 0) {
160 ngx_log_error(NGX_LOG_EMERG, cycle->log, 0,
161 "brix_seccomp: filter load failed: %s", strerror(-rc));
162 } else {
163 ngx_log_error(NGX_LOG_EMERG, cycle->log, 0,
164 "brix_seccomp: filter init failed");
165 }

168ngx_int_t
169brix_seccomp_install(ngx_cycle_t *cycle, ngx_uint_t mode, ngx_uint_t allow_exec)
170{
171 unsigned n_allow = 0;
172 unsigned n_deny = 0;
173 int rc;
174
175 if (mode == BRIX_SECCOMP_OFF) {
176 return NGX_OK;
177 }
178
179 rc = brix_seccomp_core_apply((unsigned) mode, (unsigned) allow_exec,
180 brix_seccomp_log_err, cycle, &n_allow, &n_deny);
181
182 if (rc == BRIX_SECCOMP_CORE_UNAVAIL) {
183 /* Fail closed: the operator asked for a syscall filter (audit or
184 * enforce) but this binary was built without libseccomp. Refusing the
185 * worker is the honest outcome — serving while the operator believes a
186 * filter is active would be a silent security downgrade. Rebuild with
187 * libseccomp-devel or set "brix_seccomp off". */
188 ngx_log_error(NGX_LOG_EMERG, cycle->log, 0,
189 "brix_seccomp: mode=%ui requested but this binary was "
190 "built without libseccomp (install libseccomp-devel and "
191 "rebuild, or set \"brix_seccomp off\")", mode);
192 return NGX_ERROR;
193 }
194
195 if (rc != BRIX_SECCOMP_CORE_OK) {
196 /* The sink already logged the specific failure. */
197 return NGX_ERROR;
198 }
199
200 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
201 "brix_seccomp: worker syscall filter active (mode=%s, exec=%s, "
202 "%ui allowed, %ui denied)",
203 (mode == BRIX_SECCOMP_ENFORCE) ? "enforce" : "audit",
204 allow_exec ? "allowed" : "killed",
205 (ngx_uint_t) n_allow, (ngx_uint_t) n_deny);
206 return NGX_OK;
207}
