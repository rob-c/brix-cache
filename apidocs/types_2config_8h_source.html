4/* ---- File: config.h — Per-server configuration struct (ngx_stream_brix_srv_conf_t) ----
5 *
6 * PURPOSE:
7 * One ngx_stream_brix_srv_conf_t per `server {}` block.
8 * nginx allocates via create_srv_conf, merges in merge_srv_conf.
9 * Encapsulates all tunables for a server instance.
10 *
11 * HELPER TYPES:
12 * - brix_sss_key_t: SSS credential key (id/exp/opts/key/name/user/group)
13 * - brix_auth_type_t: ACL rule type (user DN, VO, hostname, all-match)
14 * - brix_authdb_rule_t: ACL rule with auth type + identity + privileges
15 * - brix_vo_rule_t: VO access rule (path prefix + VOMS VO name)
16 * - brix_group_rule_t: Group inheritance rule (path prefix)
17 * - brix_manager_map_t: CMS manager map (prefix + host/port)
18 *
19 * CONFIGURATION AREAS:
20 * 1. Authentication: GSI/token/SSS/anonymous mode
21 * 2. TLS: cert/key/trusted CA/CRL/VOMS dirs, OCSP stapling
22 * 3. Token Auth: JWKS/issuer/audience, macaroon secrets (grace rotation)
23 * 4. VO ACLs: path prefix + VO name arrays
24 * 5. Access Log: file descriptor
25 * 6. Metrics: Prometheus slot
26 * 7. Upstream Redirector: host/port/addr/tls_ctx/token_file
27 * 8. TPC: SSRF policy + bearer file + OAuth2 endpoints
28 * 9. Read-through Cache: origin/eviction/size limits/include regex
29 * 10. Write-through: mode (sync/async) + origin + deny/allow prefixes
30 * 11. CMS Manager: heartbeat fields
31 * 12. Transparent Proxy: upstream TLS/auth/login/audit/reconnect/path rewrite
32 * 13. Security Level: enforcement mode
33 * 14. In-Protocol TLS: flag + tls_ctx
34 * 15. Manager Mode: registry slots
35 * 16. ckscan: depth/files limits
36 *
37 * DESIGN DECISIONS:
38 * - Inline bracket annotations map fields to nginx directives
39 * - OpenSSL objects (X509/EVP_PKEY/X509_STORE) for GSI
40 * - Timer events (crl_timer/jwks_timer) for cert refresh
41 * - Compiled regex (cache_include_regex) for cache filtering
42 * - Arrays for VO ACLs, authdb rules, group rules, proxy upstreams
43 *
44 * LIFECYCLE:
45 * Create: ngx_stream_brix_create_srv_conf()
46 * Merge: ngx_stream_brix_merge_srv_conf() (parent → child)
47 * Free: ngx_stream_brix_free_srv_conf() (on server reload)
48 */

117/*
118 * One native-engine authdb rule.
119 *
120 * `nsel` selectors are AND-ed: the rule applies only to a subject that
121 * satisfies EVERY sel[i]/sel_id[i] pair. A single-selector rule (the only
122 * shape that existed before 2.0 F20) takes its id verbatim, so a DN containing
123 * any character at all keeps working; a compound rule splits its id token on
124 * '|' into exactly `nsel` values, positionally aligned with the selectors.
125 *
126 * `type` / `id` are kept as the selector-0 alias so every reader that predates
127 * the compound form still sees what it saw before.
128 */
