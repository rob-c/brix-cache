31/* WHAT: Adopt one GridFTP object into the common size/digest oracle.
32 * WHY: XPR-001/002 must compare the actual FTP data channel, not an HTTP twin.
33 * HOW: SIZE first, charge the exact byte budget, then RETR into a bounded body. */
34int
35xrd_conform_run_ftp(xrd_conform_execution *execution)
36{
37 brix_ftpurl url;
38 brix_ftp_sess *session;
39 brix_status status;
40 conform_ftp_body body = { 0 };
41 int64_t size = -1;
42 int64_t mtime = -1;
43 char reason[64];
44 int rc = -1;
45
46 if (brix_ftpurl_parse(execution->target, &url) != 0) return -1;
47 session = calloc(1, sizeof(*session));
48 if (session == NULL) return -1;
49 brix_status_clear(&status);
50 if (brix_ftp_session_open(session, &url, &execution->run->opts,
51 &status) != 0
52 || brix_ftp_stat(session, url.path, &size, &mtime, &status) != 0
53 || size < 0 || (uint64_t) size > SIZE_MAX
54 || !xrd_conform_reserve(execution, (size_t) size, reason,
55 sizeof(reason))) {
56 brix_ftp_close(session);
57 free(session);
58 return -1;
59 }
60 body.capacity = (size_t) size;
61 body.bytes = malloc(body.capacity == 0 ? 1 : body.capacity);
62 if (body.bytes != NULL
63 && brix_ftp_retr(session, url.path, ftp_body_sink, &body, NULL, size,
64 &status) == 0
65 && body.length == body.capacity) {
66 execution->ftp_size = (uint64_t) size;
67 brix_s3_sha256_hex(body.bytes, body.length, execution->ftp_digest);
68 execution->have_ftp = 1;
69 rc = 0;
70 }
71 free(body.bytes);
72 brix_ftp_close(session);
73 free(session);
74 (void) mtime;
75 return rc;
76}
