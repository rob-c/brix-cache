1/*
2 * sd_stage_wb.c - the write-BACK object of the write-stage decorator (Option A /
3 * §12.2). Split from sd_stage_write.c (600-line ratchet); see sd_stage.h for the
4 * decorator contract and sd_stage_internal.h for the shared seam.
5 *
6 * WHAT: A random-access write open lands on the STAGE STORE as a normal writable
7 * object; pwrite buffers there and fsync/close flush the whole object to
8 * the backend through the one staging engine (brix_stage_run_inline_cred /
9 * brix_stage_submit FLUSH). Includes the read-modify-write hydration that
10 * seeds the stage copy from the backend before an update open, so a flush
11 * can never truncate regions the client did not write.
12 * WHY: The write-back object is one concept — the OTHER interposed write path
13 * (the staged-upload / HTTP-PUT slot) stays in sd_stage_write.c — so each
14 * file owns one concept and stays under the cap. The write-back path lets
15 * root:// kXR_write reuse the SAME stage mechanism as the staged path.
16 * HOW: The open dispatch in sd_stage.c calls sd_stage_open_writeback here for
17 * write opens; the driver table in sd_stage.c routes byte-I/O to the
18 * methods below. The captured cred records the owner identity at open time
19 * (sd_stage_record_cred, shared seam) so a deferred or async flush
20 * authenticates as the original user.
21 */

31/* Write-BACK object state (Option A / §12.2): a random-access write open lands on
32 * the STAGE STORE as a normal writable object; pwrite buffers there, and fsync/close
33 * flush the whole object to the backend through the one staging engine. This lets
34 * root:// kXR_write (direct pwrite at arbitrary offsets) use the SAME stage mechanism
35 * as the staged (HTTP PUT) path, so writethrough_flush's bespoke loop is retired.
36 * `cred` records the owner identity at open time so the flush can authenticate as
37 * the original user even if the flush runs on a background thread or after a restart. */
38typedef struct {
39 sd_stage_inst_state *is; /* back-ref: source / store / policy */
40 char key[PATH_MAX]; /* export-relative object key */
41 brix_sd_obj_t store_obj; /* the writable stage-store object */
42 off_t high_water; /* max offset+len written (flush size) */
43 unsigned dirty:1; /* written since the last flush */
44 brix_stage_cred_t cred; /* per-user identity (zeroed = service) */
45} sd_stage_wb_state;

47/* Seed the stage-store copy of `path` from the SOURCE object for a
48 * read-modify-write open.
49 *
50 * WHAT: When an update open (no TRUNC/EXCL) targets an object that already
51 * exists on the backend, copy it source -> store through the one staging
52 * engine (BRIX_STAGE_RECALL, the same generic pread -> staged_write
53 * mover the nearline recall uses) BEFORE the write-back object opens.
54 *
55 * WHY: The eventual flush replaces the WHOLE backend object with the staged
56 * bytes, so an unhydrated partial overwrite would silently truncate every
57 * region the client did not write. Hydrating also makes wb_pread return
58 * the real object bytes during a read-write session.
59 *
60 * HOW: 1. No source stat slot -> nothing to probe, keep the old behaviour.
61 * 2. A non-empty stage copy already present (a durable staged object
62 * kept for a flush retry) is NEWER than the backend - never clobber.
63 * 3. Source stat ENOENT / unreachable -> create semantics, proceed
64 * unhydrated (the dead-origin write-back tier keeps absorbing
65 * writes; unreachable is WARNed). Empty source -> nothing to copy.
66 * 4. Otherwise run the inline RECALL with the caller's cred threaded;
67 * a copy failure for an object that provably EXISTS fails the open
68 * (errno set) - flushing a truncated object is never acceptable. */
69static ngx_int_t
70sd_stage_wb_hydrate(sd_stage_inst_state *is, const char *path,
71 const brix_sd_cred_t *cred)
72{
73 brix_sd_stat_t sst, dst_st;
74 brix_stage_cred_t scred;
75
76 if (is->source->driver->stat == NULL) {
77 return NGX_OK;
78 }
79 if (is->store->driver->stat != NULL
80 && is->store->driver->stat(is->store, path, &dst_st) == NGX_OK
81 && dst_st.size > 0)
82 {
83 return NGX_OK;
84 }
85 if (is->source->driver->stat(is->source, path, &sst) != NGX_OK) {
86 if (errno != ENOENT) {
87 ngx_log_error(NGX_LOG_WARN, is->log, errno,
88 "stage write-back: cannot probe source \"%s\" — proceeding "
89 "as a whole-object create (unreachable backend)", path);
90 }
91 return NGX_OK;
92 }
93 if (sst.size == 0 || !sst.is_reg) {
94 return NGX_OK;
95 }
96
97 ngx_memzero(&scred, sizeof(scred));
98 sd_stage_record_cred(&scred, cred);
99 if (brix_stage_run_inline_cred(BRIX_STAGE_RECALL, is->source, path,
100 is->store, path,
101 sd_stage_cred_present(&scred) ? &scred : NULL) != NGX_OK)
102 {
103 int e = errno ? errno : EIO;
104
105 ngx_log_error(NGX_LOG_ERR, is->log, e,
106 "stage write-back: hydrating existing object \"%s\" (%O bytes) "
107 "from the backend failed — refusing the update open (an "
108 "unhydrated flush would truncate it)", path, sst.size);
109 errno = e;
110 return NGX_ERROR;
111 }
112
113 /* The mover preserves the SOURCE-reported permission bits, and a remote
114 * driver may synthesize a read-only mode (sd_xroot reports r--r--r-- for
115 * an anonymous origin) — but this copy is the WRITE spool the wb open is
116 * about to open O_RDWR as the (cap-dropped) worker. Force owner-rw on the
117 * spool copy; the extra bits keep the source's visibility for the flush's
118 * mode-preserving PUT. Best-effort: a store without setattr keeps the
119 * mover's mode. */
120 if (is->store->driver->setattr != NULL) {
121 brix_sd_setattr_t at;
122
123 ngx_memzero(&at, sizeof(at));
124 at.set_mode = 1;
125 at.mode = (sst.mode & 0777) | 0600;
126 (void) is->store->driver->setattr(is->store, path, &at);
127 }
128 return NGX_OK;
129}

131/* Wrap an opened stage-store object in a write-back handle: malloc the obj
132 * shell + wb state, adopt the store object BY VALUE (freeing a heap shell),
133 * record the owner identity, and stamp the stage driver so byte-I/O dispatches
134 * to the write-back methods below. Owns store_obj on both paths — adopted on
135 * success, closed and freed on ENOMEM (the only failure). */
136static brix_sd_obj_t *
137sd_stage_wb_wrap(brix_sd_instance_t *inst, sd_stage_inst_state *is,
138 const char *path, brix_sd_obj_t *store_obj, const brix_sd_cred_t *cred)
139{
140 brix_sd_obj_t *obj;
141 sd_stage_wb_state *wb;
142
143 obj = calloc(1, sizeof(*obj));
144 wb = calloc(1, sizeof(*wb));
145 if (obj == NULL || wb == NULL) {
146 if (store_obj->driver->close != NULL) { store_obj->driver->close(store_obj); }
147 if (store_obj->heap_shell) { free(store_obj); }
148 free(obj);
149 free(wb);
150 return NULL;
151 }
152
153 wb->is = is;
154 wb->high_water = 0;
155 wb->dirty = 0;
156 ngx_cpystrn((u_char *) wb->key, (u_char *) path, sizeof(wb->key));
157 wb->store_obj = *store_obj; /* adopt by value */
158 if (store_obj->heap_shell) { free(store_obj); }
159 wb->store_obj.heap_shell = 0;
160
161 /* Record the owner identity for the flush; zeroed cred = service account. */
162 sd_stage_record_cred(&wb->cred, cred);
163
164 obj->driver = &brix_sd_stage_driver; /* → the write-back methods below */
165 obj->inst = inst;
166 obj->fd = wb->store_obj.fd; /* expose the stage fd (posix sendfile) */
167 obj->snap = wb->store_obj.snap;
168 obj->state = wb;
169 obj->heap_shell = 1; /* malloc'd shell; caller frees post-copy */
170 return obj;
171}

173/* Open the write-BACK object on the stage store (a normal writable object).
174 *
175 * WHAT: Opens a writable object on the stage store and wires a sd_stage_wb_state
176 * that carries the owner identity for the eventual flush.
177 *
178 * WHY: The returned obj carries the stage driver so its pwrite/pread/fsync/close
179 * dispatch to the write-back methods below; the stage-store object is held
180 * BY VALUE so the handle is self-contained. Recording the caller's cred
181 * in wb->cred ensures sd_stage_wb_flush can authenticate to the backend
182 * as the original user rather than the service account.
183 *
184 * HOW: Opens the stage STORE (always local) with a plain open — the store is
185 * service-owned — then hands the opened object to sd_stage_wb_wrap above,
186 * which adopts it and records the owner cred (zeroed = service path). */
187brix_sd_obj_t *
188sd_stage_open_writeback(brix_sd_instance_t *inst, sd_stage_inst_state *is,
189 const char *path, int sd_flags, mode_t mode,
190 const brix_sd_cred_t *cred, int *err_out)
191{
192 brix_sd_obj_t *store_obj;
193 brix_sd_obj_t *obj;
194 int e = 0;
195
196 if (is->store->driver->open == NULL || is->store->driver->pwrite == NULL) {
197 if (err_out != NULL) { *err_out = ENOSYS; }
198 return NULL;
199 }
200 /* Read-modify-write hydration: an update open (not truncating, not an
201 * exclusive create) must seed the stage copy from the existing backend
202 * object first — the flush later replaces the WHOLE object. */
203 if ((sd_flags & (BRIX_SD_O_TRUNC | BRIX_SD_O_EXCL)) == 0
204 && sd_stage_wb_hydrate(is, path, cred) != NGX_OK)
205 {
206 if (err_out != NULL) { *err_out = errno ? errno : EIO; }
207 return NULL;
208 }
209 /* The stage store is service-owned (local POSIX); always open it plain. */
210 store_obj = is->store->driver->open(is->store, path, sd_flags, mode, &e);
211 if (store_obj == NULL) {
212 if (err_out != NULL) { *err_out = e; }
213 return NULL;
214 }
215
216 obj = sd_stage_wb_wrap(inst, is, path, store_obj, cred);
217 if (obj == NULL) {
218 if (err_out != NULL) { *err_out = ENOMEM; }
219 return NULL;
220 }
221 return obj;
222}

226/* Flush the buffered stage object to the backend through the one staging engine.
227 * Persists the stage buffer first (fsync), then FLUSHes store→source by key.
228 *
229 * The flush MODE is a property of the tier (BRIX_WT_MODE_ASYNC comes from the
230 * `brix_stage_flush async` directive on this stage decorator), so this routing is
231 * defined by the backend, not global: a plain/sync stage export flushes inline as
232 * before; only an async (tape write-back) tier defers.
233 *
234 * ASYNC (tape write-back): the whole object is durable on the stage store after
235 * the fsync above, so the flush is submitted to the async staging queue and the
236 * close/fsync returns immediately. The scheduler drains it store→source and drops
237 * the stage copy on success; a transient origin failure leaves a FAILED record in
238 * the durable journal (last_errno stamped) and the restart reconcile re-drives it
239 * from the export anchor (o.export_root). The owner cred rides on the opts so the
240 * deferred flush authenticates as the original user.
241 *
242 * SYNC: flush inline and reflect the result; on success clear dirty, on failure
243 * keep the stage copy for retry. */
244static ngx_int_t
245sd_stage_wb_flush(sd_stage_wb_state *wb)
246{
247 sd_stage_inst_state *is = wb->is;
248 ngx_int_t rc;
249
250 if (wb->store_obj.driver->fsync != NULL) {
251 (void) wb->store_obj.driver->fsync(&wb->store_obj);
252 }
253 if (!wb->dirty) {
254 return NGX_OK;
255 }
256
257 if (is->policy.flush_mode == BRIX_WT_MODE_ASYNC) {
258 brix_stage_opts_t o;
259
260 ngx_memzero(&o, sizeof(o));
261 o.async = 1;
262 o.export_root = (is->root_canon[0] != '\0') ? is->root_canon : NULL;
263 o.cred = sd_stage_cred_present(&wb->cred) ? &wb->cred : NULL;
264 /* phase74-fp: FLUSH moves bytes FROM the stage `store` TO the backend
265 * `source`, so store is the src and source the dst — same verified
266 * src/dst order as sd_stage_staged_commit's async submit. */
267 (void) brix_stage_submit(BRIX_STAGE_FLUSH, is->store, wb->key, /* NOLINT(readability-suspicious-call-argument) */
268 is->source, wb->key, &o);
269 wb->dirty = 0; /* handed off to the durable queue */
270 return NGX_OK;
271 }
272
273 rc = brix_stage_run_inline_cred(BRIX_STAGE_FLUSH, is->store, wb->key,
274 is->source, wb->key,
275 sd_stage_cred_present(&wb->cred) ? &wb->cred : NULL);
276 if (rc == NGX_OK) {
277 wb->dirty = 0;
278 }
279 return rc;
280}

283sd_stage_wb_pwrite(brix_sd_obj_t *obj, const void *buf, size_t len, off_t off)
284{
285 sd_stage_wb_state *wb = obj->state;
286 ssize_t n = wb->store_obj.driver->pwrite(&wb->store_obj, buf, len, off);
287
288 if (n > 0) {
