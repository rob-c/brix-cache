1/*
2 * cvmfs/secure_x509.c — X.509 / VOMS client-certificate authz for CVMFS.
3 *
4 * WHAT: The `brix_scvmfs_authz x509` and `authz voms` back-ends — locate the
5 * end-entity cert behind any RFC 3820 proxy chain, glob-gate its subject
6 * DN (brix_scvmfs_x509_dn), and, in VOMS mode, glob-gate the extracted
7 * VO/FQAN set (brix_scvmfs_voms). Both fail CLOSED when TLS is absent or
8 * the peer presented no VERIFIED cert.
9 *
10 * WHY: Split out of secure.c (coding-standards §1, 600-line cap). secure.c
11 * keeps the bearer-token path, the repo-authz directive plumbing, the
12 * preamble that dispatches between modes, and the QoS class glue; the
13 * OpenSSL chain-walking lives here so the crypto surface is one
14 * reviewable unit and the #if (NGX_HTTP_SSL) fencing stays in one place.
15 */

30/* ---- x509 client-cert authz (phase-92) -----------------------------------
31 * Authenticate the TLS-verified peer by its end-entity subject DN. nginx's
32 * ssl_verify_client chain does the crypto; we require a peer cert that VERIFIED
33 * (X509_V_OK) — a location that forgot ssl_verify_client presents no cert and
34 * fails CLOSED here. The EEC is the leaf when the client used a bare cert, else
35 * the first non-proxy cert in the presented chain (RFC 3820 proxies chain
36 * leaf → EEC), so a GSI proxy authenticates as its issuing identity. An
37 * optional brix_scvmfs_x509_dn glob list gates the DN; an empty list accepts
38 * any verified client. The validated DN becomes the F9 QoS / attest subject. */
39#if (NGX_HTTP_SSL)
40static X509 *
41scvmfs_find_eec(SSL *sc, X509 *leaf)
42{
43 STACK_OF(X509) *chain;
44 int i, n;
45
46 if (brix_px_classify(leaf) == BRIX_PX_NONE) {
47 return leaf;
48 }
49 chain = SSL_get_peer_cert_chain(sc); /* server side: excludes leaf */
50 n = (chain != NULL) ? sk_X509_num(chain) : 0;
51 for (i = 0; i < n; i++) {
52 X509 *cert = sk_X509_value(chain, i);
53 if (brix_px_classify(cert) == BRIX_PX_NONE) {
54 return cert;
55 }
56 }
57 return NULL;
58}

60static ngx_int_t
61scvmfs_dn_allowed(ngx_array_t *globs, const char *dn)
62{
63 ngx_str_t *g;
64 ngx_uint_t i;
65
66 if (globs == NULL || globs->nelts == 0) {
67 return 1; /* no list = any verified peer */
68 }
69 g = globs->elts;
70 for (i = 0; i < globs->nelts; i++) {
71 char pat[SCVMFS_DN_MAX];
72
73 if (g[i].len >= sizeof(pat)) {
74 continue; /* pathological glob; skip */
75 }
76 ngx_memcpy(pat, g[i].data, g[i].len);
77 pat[g[i].len] = '\0';
78 if (brix_sp_glob_match(pat, dn)) {
79 return 1;
80 }
81 }
82 return 0;
83}

85/* voms mode: gate the proxy's carried VO name(s) against an allow-glob list.
86 * vo_csv is the comma-separated VO list lifted by brix_extract_voms_info; an
87 * empty glob list admits any client carrying at least one VO, otherwise one
88 * glob must match one carried VO. Fails CLOSED (0) on the empty-VO case — the
89 * caller has already rejected a proxy with no VOMS AC. */
90static ngx_int_t
91scvmfs_vo_allowed(ngx_array_t *globs, const char *vo_csv)
92{
93 ngx_str_t *g;
94 ngx_uint_t i;
95 const char *tok;
96
97 if (globs == NULL || globs->nelts == 0) {
98 return 1; /* no list = any carried VO */
99 }
100 g = globs->elts;
101 for (tok = vo_csv; tok != NULL && *tok != '\0'; ) {
102 const char *comma = strchr(tok, ',');
103 size_t tlen = (comma != NULL) ? (size_t) (comma - tok)
104 : ngx_strlen(tok);
105 char vo[256];
106
107 if (tlen < sizeof(vo)) {
108 ngx_memcpy(vo, tok, tlen);
109 vo[tlen] = '\0';
110 for (i = 0; i < globs->nelts; i++) {
111 char pat[256];
112
113 if (g[i].len >= sizeof(pat)) {
114 continue; /* pathological glob; skip */
115 }
116 ngx_memcpy(pat, g[i].data, g[i].len);
117 pat[g[i].len] = '\0';
118 if (brix_sp_glob_match(pat, vo)) {
119 return 1;
120 }
121 }
122 }
123 tok = (comma != NULL) ? comma + 1 : NULL;
124 }
125 return 0;
126}

129ngx_int_t
130scvmfs_check_x509(ngx_http_request_t *r,
131 ngx_http_brix_cvmfs_loc_conf_t *lcf)
132{
133#if (NGX_HTTP_SSL)
134 ngx_connection_t *c = r->connection;
135 X509 *leaf, *eec;
136 char dn[SCVMFS_DN_MAX];
137
138 if (c->ssl == NULL
139 || SSL_get_verify_result(c->ssl->connection) != X509_V_OK)
140 {
141 return NGX_HTTP_UNAUTHORIZED; /* no verified client chain */
142 }
143
144 leaf = SSL_get_peer_certificate(c->ssl->connection);
145 if (leaf == NULL) {
146 return NGX_HTTP_UNAUTHORIZED; /* verify off / no cert presented */
147 }
148
149 dn[0] = '\0';
150 eec = scvmfs_find_eec(c->ssl->connection, leaf);
151 if (eec != NULL) {
152 brix_x509_oneline(X509_get_subject_name(eec), dn, sizeof(dn));
153 }
154 X509_free(leaf);
155
156 if (dn[0] == '\0') {
157 return NGX_HTTP_UNAUTHORIZED; /* every presented cert a proxy */
158 }
159 if (!scvmfs_dn_allowed(lcf->scvmfs_x509_dn, dn)) {
160 ngx_log_error(NGX_LOG_INFO, c->log, 0,
161 "scvmfs: client DN \"%s\" not in brix_scvmfs_x509_dn allow-list "
162 "- 403", dn);
163 return NGX_HTTP_FORBIDDEN; /* verified, but out of policy */
164 }
165
166 {
167 ngx_http_brix_cvmfs_ctx_t *ctx =
168 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
169
170 if (ctx != NULL) {
171 ngx_cpystrn((u_char *) ctx->token_sub, (u_char *) dn,
172 sizeof(ctx->token_sub));
173 }
174 }
175 return NGX_DECLINED; /* authenticated: proceed */
176#else
177 (void) r; (void) lcf;
178 return NGX_HTTP_UNAUTHORIZED; /* no TLS built in: fail closed */
179#endif
180}

182/* voms mode = x509 authentication PLUS a VOMS-FQAN authorisation gate. The
183 * TLS-verified peer is authenticated by its EEC DN exactly as x509 mode, then
184 * brix_extract_voms_info lifts+verifies the proxy's VOMS VO(s) against the
185 * per-VO LSC dir (vomsdir) and VOMS signing-CA trust (voms_cert_dir); the VO(s)
186 * are gated by the brix_scvmfs_voms allow-glob list. Fails CLOSED: a plain GSI
187 * proxy carrying no VOMS AC is 403, never admitted. The crypto is nginx's
188 * ssl_verify_client chain plus the shared brix_extract_voms_info engine — this
189 * TU stays policy glue. */
190ngx_int_t
191scvmfs_check_voms(ngx_http_request_t *r,
192 ngx_http_brix_cvmfs_loc_conf_t *lcf)
193{
194#if (NGX_HTTP_SSL)
195 ngx_connection_t *c = r->connection;
196 X509 *leaf, *eec;
197 STACK_OF(X509) *chain;
198 char dn[SCVMFS_DN_MAX];
199 char primary_vo[256] = "";
200 char vo_list[1024] = "";
201
202 if (c->ssl == NULL
203 || SSL_get_verify_result(c->ssl->connection) != X509_V_OK)
204 {
205 return NGX_HTTP_UNAUTHORIZED; /* no verified client chain */
206 }
207 leaf = SSL_get_peer_certificate(c->ssl->connection);
208 if (leaf == NULL) {
209 return NGX_HTTP_UNAUTHORIZED; /* verify off / no cert presented */
210 }
211 chain = SSL_get_peer_cert_chain(c->ssl->connection); /* borrowed */
212
213 dn[0] = '\0';
214 eec = scvmfs_find_eec(c->ssl->connection, leaf);
215 if (eec != NULL) {
216 brix_x509_oneline(X509_get_subject_name(eec), dn, sizeof(dn));
217 }
218 (void) brix_extract_voms_info(c->log, leaf, chain,
219 &lcf->scvmfs_vomsdir,
220 &lcf->scvmfs_voms_cert_dir,
221 primary_vo, sizeof(primary_vo),
222 vo_list, sizeof(vo_list));
223 X509_free(leaf);
224
225 if (dn[0] == '\0') {
226 return NGX_HTTP_UNAUTHORIZED; /* every presented cert a proxy */
227 }
228 if (vo_list[0] == '\0') {
229 ngx_log_error(NGX_LOG_INFO, c->log, 0,
230 "scvmfs: client DN \"%s\" carries no VOMS attribute - 403", dn);
231 return NGX_HTTP_FORBIDDEN; /* voms mode requires a VO */
232 }
233 if (!scvmfs_vo_allowed(lcf->scvmfs_voms, vo_list)) {
234 ngx_log_error(NGX_LOG_INFO, c->log, 0,
235 "scvmfs: client VO(s) \"%s\" not in brix_scvmfs_voms allow-list "
236 "- 403", vo_list);
237 return NGX_HTTP_FORBIDDEN; /* verified, but out of policy */
238 }
239
240 {
241 ngx_http_brix_cvmfs_ctx_t *ctx =
242 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
243
244 if (ctx != NULL) {
245 ngx_cpystrn((u_char *) ctx->token_sub, (u_char *) dn,
246 sizeof(ctx->token_sub));
247 }
248 }
249 ngx_log_error(NGX_LOG_INFO, c->log, 0,
250 "scvmfs: x509+voms admit DN=\"%s\" primary_vo=\"%s\"",
251 dn, primary_vo);
252 return NGX_DECLINED; /* authenticated + authorised */
253#else
254 (void) r; (void) lcf;
255 return NGX_HTTP_UNAUTHORIZED; /* no TLS built in: fail closed */
256#endif
257}
