8/* File: launch_dest.c — opening the destination of a native root:// TPC pull.
9 *
10 * WHAT: Everything between "the guard ladder said yes" and "the file slot has a
11 * write handle": the VFS flag derivation, the identity/credential/delegation-
12 * bound VFS context, the two ways a destination can be opened (a random-write
13 * handle when the selected storage leaf supports pwrite, a staged whole-object
14 * writer otherwise), and the stat each produces.
15 *
16 * WHY: split out of launch_prepare.c on 2026-08-23, which had reached the
17 * 600-line cap. The seam is a real one — launch_prepare.c now holds the guard
18 * ladder and the request orchestration, this file holds the storage-facing open
19 * — and it keeps both sides reviewable on their own terms.
20 *
21 * HOW: brix_tpc_open_destination builds the bound context and flags, tries the
22 * random-write path, and falls back to the staged writer when the leaf declines.
23 * Every refusal is answered through brix_tpc_refuse and reported as
24 * TPC_ANSWERED, never as NGX_OK; see the note above that helper.
25 */

27static ngx_uint_t
28tpc_destination_vfs_flags(uint16_t options)
29{
30 ngx_uint_t flags = BRIX_VFS_O_WRITE | BRIX_VFS_O_CREATE;
31
32 if (options & kXR_new) {
33 flags |= BRIX_VFS_O_EXCL;
34 }
35 if ((options & kXR_delete)
36 || !(options & (kXR_new | kXR_delete)))
37 {
38 flags |= BRIX_VFS_O_TRUNC;
39 }
40 if (options & kXR_mkpath) {
41 flags |= BRIX_VFS_O_MKDIRPATH;
42 }
43 return flags;
44}

86tpc_try_random_destination(brix_ctx_t *ctx, ngx_connection_t *c,
87 brix_vfs_ctx_t *vctx, ngx_uint_t vflags, int idx, brix_file_t *file,
88 struct stat *st, const char *dst_path)
89{
90 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(vctx->sd);
91 brix_vfs_file_t *vfh;
92 /* Zeroed: a driver may return NGX_OK without setting every field. */
93 brix_vfs_stat_t vst = { 0 };
94 int fd;
95
96 if (!(vctx->sd == NULL || leaf == NULL
97 || (brix_sd_caps(leaf) & BRIX_SD_CAP_RANDOM_WRITE)
98 || (leaf->driver != NULL && leaf->driver->pwrite != NULL)))
99 {
100 return NGX_DECLINED;
101 }
102
103 vfh = brix_vfs_open(vctx, vflags, &fd);
104 if (vfh == NULL) {
105 return NGX_DECLINED;
106 }
107 if (brix_vfs_file_stat(vfh, &vst) != NGX_OK) {
108 int err = errno;
109
110 (void) brix_vfs_close(vfh, c->log);
111 brix_free_fhandle(ctx, idx);
112 return brix_tpc_refuse(ctx, c, dst_path, kXR_IOError, strerror(err));
113 }
114 brix_vfs_file_sd_obj(vfh, &file->sd_obj);
115 file->fd = brix_vfs_file_fd(vfh);
116 tpc_stat_from_vfs(&vst, st);
117 return NGX_OK;
118}

120static ngx_int_t
121tpc_open_staged_destination(brix_ctx_t *ctx, ngx_connection_t *c,
122 ngx_stream_brix_srv_conf_t *conf, brix_vfs_ctx_t *vctx, ngx_uint_t vflags,
123 int idx, brix_file_t *file, struct stat *st, mode_t create_mode,
124 const char *dst_path)
125{
126 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(vctx->sd);
127 brix_vfs_writer_t *writer;
128 int fd = 0;
129
130 if (leaf == NULL || ((brix_sd_caps(leaf) & BRIX_SD_CAP_RANDOM_WRITE)
131 && leaf->driver != NULL
132 && leaf->driver->pwrite != NULL))
133 {
134 int err = errno != 0 ? errno : EIO;
135
136 brix_free_fhandle(ctx, idx);
137 return brix_tpc_refuse(ctx, c, dst_path, kXR_IOError, strerror(err));
138 }
139
140 writer = brix_vfs_writer_open(vctx, vflags & BRIX_VFS_O_TRUNC,
141 conf->common.verify_write ? 1 : 0, &fd);
142 if (writer == NULL) {
143 int err = fd != 0 ? fd : (errno != 0 ? errno : EIO);
144
145 brix_free_fhandle(ctx, idx);
146 return brix_tpc_refuse(ctx, c, dst_path, kXR_IOError, strerror(err));
147 }
148 file->writer = writer;
149 file->fd = brix_vfs_writer_fd(writer);
150 ngx_memzero(st, sizeof(*st));
151 st->st_mode = S_IFREG | create_mode;
152 return NGX_OK;
153}

155/* WHAT: Open and classify a TPC destination through the VFS — a random-write
156 * handle when the selected leaf supports pwrite, else a staged whole-object
157 * writer — populating the caller's file slot and stat. NGX_OK on success;
158 * TPC_ANSWERED when the failure was already answered on the wire (the
159 * fhandle is freed by then); NGX_ERROR when it could not be.
160 * WHY: POSIX/cache backends and HTTP/S3-like backends carry different write
161 * handles, but both must pass through the same identity-bound VFS context;
162 * keeping the branch out of the orchestrator makes the security and
163 * resource transitions explicit. */
164ngx_int_t
165brix_tpc_open_destination(brix_ctx_t *ctx, ngx_connection_t *c,
166 ngx_stream_brix_srv_conf_t *conf, const char *dst_path,
167 uint16_t options, uint16_t mode_bits, int idx,
168 brix_file_t *file, struct stat *st)
169{
170 brix_vfs_ctx_t vctx;
171 ngx_uint_t vflags;
172 mode_t create_mode;
173 ngx_int_t rc;
174
175 tpc_destination_vfs_ctx(ctx, c, conf, dst_path, &vctx);
176 vflags = tpc_destination_vfs_flags(options);
177 create_mode = (mode_bits & 0777) ? (mode_t) (mode_bits & 0777) : 0644;
178
179 /* NGX_DECLINED alone means "this leaf cannot take a random-write handle";
180 * every other answer, refusals included, is the final one. */
181 rc = tpc_try_random_destination(ctx, c, &vctx, vflags, idx, file, st,
182 dst_path);
183 if (rc != NGX_DECLINED) {
184 return rc;
185 }
186 return tpc_open_staged_destination(ctx, c, conf, &vctx, vflags, idx,
187 file, st, create_mode, dst_path);
188}
