1/*
2 * sd_http_digest.c — checksum offload for the HTTP-origin storage driver.
3 *
4 * WHAT: The `query_checksum` vtable slot. Asks the origin for its own digest of
5 * the open object with an RFC-3230 `Want-Digest:` request header on a
6 * HEAD, and answers from the `Digest:` header it replies with.
7 *
8 * WHY: Without this slot a checksum request against an http:// primary (a WLCG
9 * storage endpoint, a dCache/StoRM/XrdHttp door, a cvmfs mirror) drags the
10 * entire object across the network through pread just to hash bytes the
11 * origin already hashed and stores as metadata. One HEAD replaces the
12 * whole-object transfer — the same trade the root:// driver's kXR_Qcksum
13 * offload makes, over the protocol HTTP origins actually speak.
14 *
15 * HOW: The requested algorithm is canonical ("sha256"); the wire wants the
16 * registered token ("sha-256"), so brix_digest_wire_token maps it and an
17 * algorithm with no registered token declines before any I/O. The probe
18 * reuses the object's own key and resolved identity (bearer header and/or
19 * mutual-TLS client cert), so an origin that authorizes per-object sees
20 * the same requester it saw on the open. The reply is parsed by the shared
21 * RFC-3230 grammar (core/compat/digest_header.c) asking for EXACTLY the
22 * requested algorithm: an origin that answers with a digest in some OTHER
23 * algorithm declines to the compute fallback rather than mislabelling it.
24 *
25 * A decline is never a failure: per the slot contract the caller falls back to
26 * reading the bytes, so an origin that ignores Want-Digest, speaks RFC 9530's
27 * `Repr-Digest:` instead, or holds no digest at all simply costs one HEAD.
28 */

37/* An origin may list several digests in one header ("adler32=…,md5=…,sha-256=…");
38 * a base64 sha-512 alone is 88 bytes. 512 holds every realistic reply, and a
39 * longer one is truncated by the transport, which the grammar then rejects
40 * rather than half-parses. */
41#define SD_HTTP_DIGEST_HDR_MAX 512
42
43/*
44 * WHAT: Turn one HEAD response into a lowercase-hex digest in `algo`, or
45 * NGX_DECLINED.
46 * WHY: Split from the request leg so the "what did the origin actually say"
47 * policy — no header, wrong algorithm, unusable value — reads as one
48 * sequence of refusals.
49 * HOW: Fetch `Digest:` by name → scan it for exactly `algo` → re-pad the value
50 * to the algorithm's fixed width (origins trim leading zeros off an
51 * adler32, and this digest is handed to clients as authoritative) → copy
52 * out only when it fits the caller's buffer whole.
53 */
54static ngx_int_t
55sd_http_digest_from_resp(sd_http_inst_state *is, const brix_s3_resp_t *resp,
56 const char *algo, char *hex_out, size_t hex_sz)
57{
58 char hdr[SD_HTTP_DIGEST_HDR_MAX];
59 char hex[BRIX_DIGEST_HEX_MAX];
60
61 if (is->transport->resp_header(resp, "Digest", hdr, sizeof(hdr)) != 0) {
62 return NGX_DECLINED; /* origin advertised no digest */
63 }
64 if (brix_digest_header_scan((u_char *) hdr, ngx_strlen(hdr), algo, NULL,
65 hex, sizeof(hex)) != BRIX_DIGEST_FOUND)
66 {
67 return NGX_DECLINED; /* other algorithm, or garbage */
68 }
69 brix_digest_hex_pad(algo, hex, sizeof(hex));
70 if (ngx_strlen(hex) + 1 > hex_sz) {
71 return NGX_DECLINED;
72 }
73 ngx_cpystrn((u_char *) hex_out, (u_char *) hex, hex_sz);
74 return NGX_OK;
75}

77ngx_int_t
78sd_http_query_checksum(brix_sd_obj_t *obj, const char *algo, char *hex_out,
79 size_t hex_sz)
80{
81 sd_http_inst_state *is;
82 sd_http_obj_state *st;
83 const char *token;
84 const char *auth;
85 char hdrs[SD_HTTP_AUTH_MAX + 64];
86 brix_s3_resp_t resp;
87 int auth_failed = 0;
88 ngx_int_t rc;
89
90 if (obj == NULL || obj->state == NULL || algo == NULL || hex_sz == 0) {
91 return NGX_DECLINED;
92 }
93 token = brix_digest_wire_token(algo);
94 if (token == NULL) {
95 return NGX_DECLINED; /* no RFC-3230 token — never ask, never guess */
96 }
97
98 is = obj->inst->state;
99 st = obj->state;
100
101 /* Same identity precedence as pread: a per-open bearer wins over the
102 * instance's static one, and a per-open x509 proxy rides as the client
103 * cert — a digest we present as this object's must be the digest the
104 * origin shows to the identity that opened it. */
105 auth = st->auth_hdr[0] ? st->auth_hdr : is->auth_hdr;
106 snprintf(hdrs, sizeof(hdrs), "Want-Digest: %s\r\n%s", token, auth);
107
108 sd_http_req_t rq = { is, "HEAD", st->key, hdrs,
109 st->cert_pem[0] ? st->cert_pem : NULL, &resp,
110 g_sd_http_force_primary, &auth_failed,
111 NULL, 0 /* no request entity */ };
112 if (sd_http_request_fo(&rq, NULL) != 0) {
113 return NGX_ERROR; /* transport fault → compute */
114 }
115 if (resp.status != 200) {
116 is->transport->resp_free(&resp);
117 return NGX_DECLINED;
118 }
119 rc = sd_http_digest_from_resp(is, &resp, algo, hex_out, hex_sz);
120 is->transport->resp_free(&resp);
121 return rc;
122}
