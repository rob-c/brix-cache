13/* Darwin's unlinkat(2) rejects a directory (no AT_REMOVEDIR) with EPERM
14 * where Linux answers EISDIR; callers key their "unlink a file, rmdir a
15 * directory" fallback on EISDIR, so restore it when the name is a dir. */
16int
17brix_plat_unlinkat(int dirfd, const char *name, int flags)
18{
19 struct stat st;
20 int saved;
21
22 if (unlinkat(dirfd, name, flags) == 0) { /* vfs-seam-allow: SEAM_CORRECT - PAL unlink primitive beneath the VFS */
23 return 0;
24 }
25 if (errno != EPERM || (flags & AT_REMOVEDIR)) {
26 return -1;
27 }
28 saved = errno;
29 if (fstatat(dirfd, name, &st, AT_SYMLINK_NOFOLLOW) == 0 /* vfs-seam-allow: SEAM_CORRECT - PAL unlink primitive beneath the VFS */
30 && S_ISDIR(st.st_mode))
31 {
32 errno = EISDIR;
33 return -1;
34 }
35 errno = saved;
36 return -1;
37}
