36/* ---- Drain an unexpected bounded response body ----
37 *
38 * WHAT: Consumes length bytes and returns zero, or -1 on transport failure.
39 * WHY: A hostile pgwrite status is evidence for the selected clause, but its
40 * body must not remain queued and corrupt the following cleanup frame.
41 * HOW: Read fixed-size chunks through the shared exact-read helper until the
42 * complete advertised body has been discarded.
43 */
44static int
45discard_reply_body(brix_conn *connection, uint32_t length,
46 brix_status *status)
47{
48 uint8_t chunk[256];
49
50 while (length > 0) {
51 size_t take = length < sizeof(chunk) ? length : sizeof(chunk);
52
53 if (brix_read_full(&connection->io, chunk, take, status) != 0)
54 return -1;
55 length -= (uint32_t) take;
56 }
57 return 0;
58}

60static int
61read_pgwrite_reply(brix_conn *connection, uint16_t want_stream,
62 pgw_reply *reply, brix_status *status)
63{
64 uint8_t header[XRD_RESPONSE_HDR_LEN];
65 uint16_t stream_id;
66 uint32_t length;
67
68 memset(reply, 0, sizeof(*reply));
69 if (brix_read_full(&connection->io, header, sizeof(header), status) != 0)
70 return -1;
71 xrd_resp_hdr_unpack(header, &stream_id, &reply->status, &length);
72 if (stream_id != want_stream || reply->status != kXR_status
73 || length != sizeof(reply->body)) {
74 discard_reply_body(connection, length, status);
75 return -1;
76 }
77 if (brix_read_full(&connection->io, reply->body, length, status) != 0)
78 return -1;
79 reply->body_length = length;
80 reply->trailer_length = xrd_get_u32_be(reply->body + 12);
81 if (reply->trailer_length > 4096) return -1;
82 if (reply->trailer_length > 0) {
83 reply->trailer = malloc(reply->trailer_length);
84 if (reply->trailer == NULL
85 || brix_read_full(&connection->io, reply->trailer,
86 reply->trailer_length, status) != 0) {
87 free(reply->trailer);
88 reply->trailer = NULL;
89 return -1;
90 }
91 }
92 return 0;
93}
