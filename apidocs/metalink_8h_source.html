1/*
2 * metalink.h — parsed metalink (RFC 5854 v4 + legacy v3) mirror-list contract.
3 *
4 * WHAT: One concept: the brix_metalink result struct (ranked mirror URLs +
5 * optional size/digest) and the pure parser/detector API over it.
6 * WHY: Metalink files are the client's "virtual redirector" input (phase-100):
7 * the copy engine fails over across the ranked mirrors and can feed them
8 * to the extreme-copy engine. The parser is pure (bytes in, struct out,
9 * no I/O) so it is unit-testable against hostile documents.
10 * HOW: copy_metalink.c loads the document (local read or a bounded remote
11 * fetch) and calls brix_metalink_parse; consumers walk urls[0..n_urls)
12 * which is already sorted best-first.
13 *
14 * Requires: nothing beyond brix.h (included here for brix_status). Not a
15 * public API: include only from client/lib/ and client/tests/c/.
16 */

24/* Hard caps against hostile documents (see phase-100 doc §2.1). */
25#define XRDC_METALINK_MAX_BYTES (4u * 1024u * 1024u) /* refuse larger docs */
26#define XRDC_METALINK_MAX_URLS 16 /* mirrors kept (ranked) */

28#define XRDC_METALINK_HEX_MAX 129 /* digest hex + NUL */
29#define XRDC_METALINK_ALGO_MAX 16 /* digest name + NUL */

31/* One ranked mirror: the absolute URL and the rank it sorted under (ascending =
32 * better; v4 priority used as-is, v3 preference mapped to 101-preference). */
33typedef struct {
34 char rank_url[XRDC_METALINK_URL_MAX];
35 int rank;
36} brix_metalink_url;
37
38typedef struct {
39 brix_metalink_url urls[XRDC_METALINK_MAX_URLS];
40 size_t n_urls;
41 size_t n_skipped; /* mirrors dropped: bad scheme/length/overflow */
42 int64_t size; /* <size> of the file, -1 when absent */
43 /* Strongest client-supported digest found (md5 > crc32c > adler32), or
44 * empty strings when the document carries none we can verify. */
45 char hash_algo[XRDC_METALINK_ALGO_MAX];
46 char hash_hex[XRDC_METALINK_HEX_MAX];
47} brix_metalink;
