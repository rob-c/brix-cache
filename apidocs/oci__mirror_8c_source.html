1/*
2 * oci_mirror.c — the /v2/ content handler (Appendix J).
3 *
4 * WHAT: the entry point for every request on a brix_oci_mirror location:
5 * ctx + observer → gate → canonical key → offloaded serve → coalesced
6 * fill → open + ranged response with the registry's two headers.
7 * WHY: a pull-through registry mirror is a cache with a URL grammar bolted
8 * on. Everything below the grammar — request coalescing so a hundred
9 * simultaneous `podman pull`s of the same layer cause ONE upstream GET,
10 * digest verify at the edge, ranged/conditional serving, bounded
11 * stale-if-error — is machinery this tree already runs for CVMFS. The
12 * job of this file is therefore to compose those pieces in the exact
13 * order Appendix J pins, and to add nothing of its own that could drift
14 * from them.
15 * HOW: the J-wiring, verbatim: brix_http_serve_offload_remote →
16 * brix_http_cache_fill_if_needed → brix_vfs_open →
17 * brix_http_serve_file_ranged. Never brix_cache_open_or_fill (it fills
18 * inline on the event loop), never a direct open/read/sendfile in this
19 * directory, and never `Accept` in the cache key — a manifest is one
20 * object, and content negotiation on it would fragment the cache along
21 * a dimension the upstream does not.
22 */

49/* ---- the fill-failure map (Appendix J.4) --------------------------------- */
50
51/* Seconds a client is told to wait after the upstream rate-limited us. Long
52 * enough that a pull farm backing off in lockstep does not immediately re-form
53 * the burst, short enough that a transient limit does not stall a deployment. */
54#define OCI_RATE_LIMIT_HOLDOFF "5"
55
56/* Translate the tier's fill verdict into the registry's own vocabulary. The
57 * tier reports HTTP statuses because that is what its origin spoke; a registry
58 * client reads the ERROR CODE, and "MANIFEST_UNKNOWN" vs "BLOB_UNKNOWN" is the
59 * difference between `podman pull` reporting a missing tag and reporting a
60 * corrupt image. */
61static brix_oci_err_t
62oci_fill_err(const ngx_http_brix_oci_ctx_t *ctx, ngx_int_t status)
63{
64 switch (status) {
65 case NGX_HTTP_NOT_FOUND:
66 return (ctx->req.cls == BRIX_OCI_REQ_MANIFEST)
67 ? BRIX_OCI_ERR_MANIFEST_UNKNOWN : BRIX_OCI_ERR_BLOB_UNKNOWN;
68 case NGX_HTTP_FORBIDDEN:
69 case NGX_HTTP_UNAUTHORIZED:
70 return BRIX_OCI_ERR_DENIED;
71 case NGX_HTTP_TOO_MANY_REQUESTS:
72 return BRIX_OCI_ERR_TOOMANYREQUESTS;
73 default:
74 return BRIX_OCI_ERR_UNAVAILABLE;
75 }
76}
77
78/* Fill-failure interceptor. The tier has already exhausted its own stale-serve
79 * window by the time it calls this, so there is nothing left to salvage — the
80 * job here is to say WHY in the shape the client parses, and to charge the
81 * upstream-error bucket an operator alerts on. */
82static ngx_int_t
83oci_fill_fail(ngx_http_request_t *r, void *data, ngx_int_t status)
84{
85 ngx_http_brix_oci_ctx_t *ctx =
86 ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
87 ngx_uint_t out;
88
89 (void) data;
90
91 if (ctx == NULL) {
92 return status;
93 }
94 ctx->disp = BRIX_OCI_OUT_ERROR;
95
96 /* An upstream 401 that survived the token dance is a genuine denial, not a
97 * challenge to relay: relaying it would make the client believe OUR mirror
98 * wants credentials it has no way to mint. */
99 out = (status == NGX_HTTP_UNAUTHORIZED) ? NGX_HTTP_FORBIDDEN
100 : (ngx_uint_t) status;
101 if (out < 400) {
102 out = NGX_HTTP_BAD_GATEWAY;
103 }
104
105 BRIX_OCI_METRIC_INC(upstream_errors_total[brix_oci_uperr_bucket(out)]);
106
107 /* A 429 without a Retry-After is a refusal the client can only answer by
108 * guessing, and a guessing client retries immediately — the exact traffic
109 * the upstream limit exists to stop. This is OUR hold-off, not the
110 * origin's: the upstream header does not survive the fill's thread-pool
111 * boundary (DRIFT vs D1.5's "echoed"), and a mirror-local constant is the
112 * honest thing to publish anyway, since our next attempt is what the
113 * client's retry actually costs. */
114 if (out == NGX_HTTP_TOO_MANY_REQUESTS
115 && brix_http_set_header(r, "Retry-After", OCI_RATE_LIMIT_HOLDOFF, NULL)
116 != NGX_OK)
117 {
118 return NGX_HTTP_INTERNAL_SERVER_ERROR;
119 }
120
121 return brix_oci_error(r, out, oci_fill_err(ctx, status), NULL);
122}
123
124/* Re-entry trampoline: the completed fill re-runs the whole handler, which now
125 * takes the hit path. The disposition stays FILL — the bytes came from a fresh
126 * upstream pull, and an operator reading $brix_cache_status wants to know
127 * that. */
128static ngx_int_t
129oci_reenter(ngx_http_request_t *r, void *data)
130{
131 (void) data;
132 return ngx_http_brix_oci_handler(r);
133}

138/* The two off-loop steps: a socket-wire serve, then a coalesced miss fill.
139 * NGX_DECLINED means the bytes are local now and the caller should open. */
140static ngx_int_t
141oci_serve_or_fill(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
142 ngx_http_brix_oci_ctx_t *ctx, brix_sd_instance_t *sd, const char *path)
143{
144 brix_http_serve_opts_t sopts;
145 ngx_int_t rc;
146
147 /* No presentation record yet: the offload path re-serves through the same
148 * ranged pipeline, but the object is not open here and the digest cannot
149 * be derived without it. The offload only ever fires for socket-wire
150 * backends, which an OCI mirror (http upstream) is not — so this stays a
151 * correctness guard rather than a live path. */
152 brix_oci_present_serve_opts(&sopts, NULL);
153
154 rc = brix_http_serve_offload_remote(r, sd, ctx->key, path, &sopts,
155 &lcf->common, NULL, NULL);
156 if (rc == NGX_DONE) {
157 return NGX_DONE;
158 }
159 if (rc == NGX_ERROR) {
160 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
161 BRIX_OCI_ERR_UNAVAILABLE, NULL);
162 }
163
164 rc = brix_http_cache_fill_if_needed(r, sd, ctx->key, &lcf->common, NULL,
165 oci_reenter, NULL, oci_fill_fail);
166 if (rc == NGX_DONE) {
167 ctx->disp = BRIX_OCI_OUT_FILL;
168 return NGX_DONE;
169 }
170 if (rc == NGX_ERROR) {
171 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
172 BRIX_OCI_ERR_UNAVAILABLE, NULL);
173 }
174
175 return NGX_DECLINED;
176}

178/* Where this object's sidecar goes: beside the cached body, in the cache
179 * STORE's own directory. The mirror's export root is the pure-cache anchor "/",
180 * so the request path is the key and nothing more — the bytes physically live
181 * under the store, and asking the store where that is (rather than assuming the
182 * advertised root) is the same discipline the reaper follows. A non-local store
183 * (s3/rados) answers NULL: it has no directory to put a memo in, and the caller
184 * then derives the pair on every hit, which is correct but not free.
185 * 0 = written, -1 = no sidecar is possible for this object. */
186static int
187oci_meta_base(brix_sd_instance_t *sd, const ngx_http_brix_oci_ctx_t *ctx,
188 char *out, size_t outsz)
189{
190 const char *store_root = brix_cstore_local_root(brix_sd_cache_cstore(sd));
191 size_t rn;
192
193 if (store_root == NULL) {
194 return -1;
195 }
196 rn = ngx_strlen(store_root);
197 while (rn > 0 && store_root[rn - 1] == '/') {
198 rn--; /* the key carries the slash */
199 }
200 if (rn + ctx->key_len >= outsz) {
201 return -1;
202 }
203 ngx_memcpy(out, store_root, rn);
204 ngx_memcpy(out + rn, ctx->key, ctx->key_len);
205 out[rn + ctx->key_len] = '\0';
206
207 return 0;
208}

211/* Conditional evaluation against the validator this surface actually
212 * publishes. When the object's digest is known that IS the ETag (App. B.1),
213 * and it must be the one compared: the generic evaluator derives its own
214 * from mtime+size, which a refill of byte-identical bytes changes — so a
215 * client revalidating an unchanged tag past its TTL would be handed the whole
216 * manifest again. Everything else (If-Match, the date forms) stays with the
217 * shared RFC 9110 §13.2.2 evaluator. */
218static ngx_int_t
219oci_eval_conditional(ngx_http_request_t *r, const brix_oci_present_t *pres,
220 const brix_vfs_stat_t *vst)
221{
222 char etag[BRIX_OCI_DIGEST_STRLEN + 3];
223
224 if (r->headers_in.if_none_match != NULL
225 && r->headers_in.if_match == NULL
226 && brix_oci_present_etag(pres, etag, sizeof(etag)))
227 {
228 return (brix_http_etag_list_contains(
229 &r->headers_in.if_none_match->value, etag,
230 BRIX_HTTP_COND_WEAK_EQUIV) == NGX_OK)
231 ? NGX_HTTP_NOT_MODIFIED : NGX_OK;
232 }
233
234 return brix_http_eval_preconditions(r, 1, vst->mtime, vst->size,
235 BRIX_ETAG_WEAK,
236 BRIX_HTTP_COND_READ
237 | BRIX_HTTP_COND_TIME
238 | BRIX_HTTP_COND_WEAK_EQUIV);
239}

242/* Open the now-local object and answer. Errors map through J.5, not through
243 * the generic table: an ENOSPC on a read surface is an upstream/store problem
244 * (502), and an EACCES is the containment check refusing a path escape. */
245static ngx_int_t
246oci_open_respond(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
247 ngx_http_brix_oci_ctx_t *ctx, brix_vfs_ctx_t *vctx, const char *path)
248{
249 brix_oci_present_t *pres;
250 brix_http_serve_opts_t opts;
251 brix_http_serve_result_t result;
252 brix_vfs_file_t *fh;
253 brix_vfs_stat_t vst;
254 char meta_base[PATH_MAX];
255 int vfs_err = 0;
256 ngx_int_t rc;
257
258 fh = brix_vfs_open(vctx, BRIX_VFS_O_READ, &vfs_err);
259 if (fh == NULL) {
260 ngx_uint_t status = brix_oci_errno_status(r, vfs_err);
261
262 ctx->disp = BRIX_OCI_OUT_ERROR;
263 return brix_oci_error(r, status, oci_fill_err(ctx, (ngx_int_t) status),
264 NULL);
265 }
266 if (ctx->disp == BRIX_OCI_OUT_HIT || ctx->disp == BRIX_OCI_OUT_LOCAL) {
267 /* pcalloc'd zero == BRIX_OCI_OUT_HIT; only a fill overrides it. */
268 ctx->disp = BRIX_OCI_OUT_HIT;
269 }
270
271 if (brix_vfs_file_stat(fh, &vst) != NGX_OK) {
272 brix_vfs_close(fh, r->connection->log);
273 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
274 BRIX_OCI_ERR_UNAVAILABLE, NULL);
275 }
276 if (vst.is_directory) {
277 brix_vfs_close(fh, r->connection->log);
278 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
279 oci_fill_err(ctx, NGX_HTTP_NOT_FOUND), NULL);
280 }
281
282 pres = ngx_pcalloc(r->pool, sizeof(*pres));
283 if (pres == NULL) {
284 brix_vfs_close(fh, r->connection->log);
285 return NGX_HTTP_INTERNAL_SERVER_ERROR;
286 }
287 if (oci_meta_base(vctx->sd, ctx, meta_base, sizeof(meta_base)) != 0) {
288 meta_base[0] = '\0';
289 }
290 if (brix_oci_present_prepare(r, lcf, ctx, fh, &vst,
291 (meta_base[0] != '\0') ? meta_base : NULL,
292 pres) != NGX_OK)
293 {
294 brix_vfs_close(fh, r->connection->log);
295 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
296 BRIX_OCI_ERR_UNAVAILABLE, NULL);
297 }
298
299 rc = oci_eval_conditional(r, pres, &vst);
300 if (rc == NGX_HTTP_NOT_MODIFIED) {
301 brix_vfs_close(fh, r->connection->log);
302 ctx->disp = BRIX_OCI_OUT_LOCAL; /* answered with no bytes */
303 r->headers_out.status = NGX_HTTP_NOT_MODIFIED;
304 r->headers_out.content_length_n = 0;
305 r->headers_out.last_modified_time = vst.mtime;
306 (void) brix_http_add_etag_header(r, vst.mtime, vst.size,
307 BRIX_ETAG_WEAK, 1);
308 /* RFC 9110 §15.4.5: a 304 carries the validators — and, for a
309 * registry, the digest the 200 would have carried, because a client
310 * revalidating a manifest still needs to pin what it holds. */
311 brix_oci_present_headers(r, (ngx_fd_t) -1, 0, pres);
312 ngx_http_send_header(r);
313 return ngx_http_send_special(r, NGX_HTTP_LAST);
314 }
315 if (rc != NGX_OK) {
316 brix_vfs_close(fh, r->connection->log);
317 return rc;
318 }
319
320 /* Blobs are the only route a client ever ranges (a resumed layer pull);
321 * a manifest is fetched whole, but allowing ranges on it costs nothing and
322 * keeps one code path. */
323 r->allow_ranges = 1;
324
325 brix_oci_present_serve_opts(&opts, pres);
326 return brix_http_serve_file_ranged(r, fh, &vst, path, &opts, &result);
327}

329static ngx_int_t
330oci_tier_get(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
331 ngx_http_brix_oci_ctx_t *ctx)
332{
333 char path[PATH_MAX];
334 const char *root = lcf->common.root_canon;
335 brix_vfs_ctx_t vctx;
336 int is_tls = 0;
337 ngx_int_t rc;
338
339 rc = brix_http_mirror_key_path(root, ctx->key, ctx->key_len,
340 path, sizeof(path));
341 if (rc != NGX_OK) {
342 return brix_oci_error(r, (ngx_uint_t) rc, BRIX_OCI_ERR_NAME_INVALID,
343 NULL);
344 }
345
346 is_tls = brix_http_request_is_tls(r);
347 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log, BRIX_PROTO_OCI,
348 root, "", BRIX_VFS_MUTATION_READ_ONLY, is_tls, NULL, path);
349 brix_http_vfs_bind_no_rules(&lcf->common, &vctx);
350
351 vctx.sd = brix_vfs_backend_resolve(root, r->connection->log);
352 if (vctx.sd == NULL) {
353 ngx_log_error(NGX_LOG_ERR, r->connection->log, 0,
354 "oci: no storage backend registered for \"%s\" — check "
355 "brix_oci_mirror / brix_storage_backend", root);
356 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
357 BRIX_OCI_ERR_UNAVAILABLE, NULL);
358 }
359
360 /* Per-worker one-shot: the upstream token supplier can only be attached
361 * once the instance exists, and instances are built lazily post-fork. */
362 brix_oci_bind_bearer(lcf, vctx.sd, r->connection->log);
363
364 (void) brix_dashboard_http_start_identity(r, ctx->key,
365 (ctx->deleg_user != NULL) ? ctx->deleg_user : "anonymous", "",
366 BRIX_XFER_PROTO_OCI, BRIX_XFER_DIR_READ, "GET", -1);
367
368 rc = oci_serve_or_fill(r, lcf, ctx, vctx.sd, path);
369 if (rc != NGX_DECLINED) {
370 return rc;
371 }
372
373 return oci_open_respond(r, lcf, ctx, &vctx, path);
374}

378void
379brix_oci_finalize_observe(void *data)
380{
381 ngx_http_request_t *r = data;
382 ngx_http_brix_oci_ctx_t *ctx =
383 ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
384
385 if (ctx == NULL || ctx->counted) {
386 return;
387 }
388 ctx->counted = 1;
389
390 /* One request, one row. The surface is fixed per location and the class /
391 * disposition are enum indices — invariant #8 holds by construction: no
392 * repository name, tag or digest is ever a label. */
393 BRIX_OCI_METRIC_INC(
394 requests_total[BRIX_OCI_SURFACE_MIRROR][ctx->mclass][ctx->disp]);
395 /* phase-110 W10: also feed the unified cache family, so an OCI cache
396 * hit-rate is one query across every plane (brix_cache_requests_total).
397 * HIT/LOCAL served from the local store; FILL is an origin fill (a miss);
398 * REFUSED/ERROR are not cache outcomes. */
399 if (ctx->disp == BRIX_OCI_OUT_HIT || ctx->disp == BRIX_OCI_OUT_LOCAL) {
400 brix_metric_cache_result(BRIX_PROTO_OCI, 1, 0);
401 } else if (ctx->disp == BRIX_OCI_OUT_FILL) {
402 brix_metric_cache_result(BRIX_PROTO_OCI, 0, 0);
403 }
404}

408static ngx_http_brix_oci_ctx_t *
409oci_handler_ctx(ngx_http_request_t *r)
410{
411 ngx_http_brix_oci_ctx_t *ctx =
412 ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
413 ngx_pool_cleanup_t *cln;
414
415 if (ctx != NULL) {
416 return ctx; /* re-entry after a parked fill */
417 }
418 ctx = ngx_pcalloc(r->pool, sizeof(*ctx));
419 if (ctx == NULL) {
420 return NULL;
421 }
422 ngx_http_set_ctx(r, ctx, ngx_http_brix_oci_module);
423
424 cln = ngx_pool_cleanup_add(r->pool, 0);
425 if (cln == NULL) {
426 return NULL;
427 }
428 cln->handler = brix_oci_finalize_observe;
429 cln->data = r;
430
431 return ctx;
432}

434ngx_int_t
435ngx_http_brix_oci_handler(ngx_http_request_t *r)
436{
437 ngx_http_brix_oci_loc_conf_t *lcf =
438 ngx_http_get_module_loc_conf(r, ngx_http_brix_oci_module);
439 ngx_http_brix_oci_ctx_t *ctx;
440 ngx_int_t rc;
441
442 ctx = oci_handler_ctx(r);
443 if (ctx == NULL) {
444 return NGX_HTTP_INTERNAL_SERVER_ERROR;
445 }
446
447 /* The registry surface reads bodies (a PATCH IS the layer), so it must
448 * take the request BEFORE the body is discarded. The two surfaces are
449 * mutually exclusive per location — the merge refuses both directives on
450 * one location — so this is a branch, not a fallthrough. */
451 if (lcf->registry) {
452 return brix_oci_registry_handle(r, lcf, ctx);
453 }
454
455 /* The mirror is GET/HEAD-only, so no request body is ever meaningful; the
456 * gate still polices the method itself, because a discarded body must not
457 * make a push look like it was accepted. */
458 rc = ngx_http_discard_request_body(r);
459 if (rc != NGX_OK) {
460 return rc;
461 }
462
463 /* Delegated pull (D16): establish WHO is asking before any route is
464 * dispatched — the listing routes answered inside the gate need the
465 * identity too — then, for the object routes, require the per-
466 * (credential, repository) proof before any byte is served, hit or
467 * miss alike. Re-entries (parked fill, granted proof) pass straight
468 * through on ctx->deleg_proved. */
469 rc = brix_oci_delegate_ident(r, lcf, ctx);
470 if (rc != NGX_DECLINED) {
471 return rc;
472 }
473
474 rc = brix_oci_gate(r, lcf, ctx);
475 if (rc != NGX_DECLINED) {
476 return rc;
477 }
478
479 rc = brix_oci_delegate_gate(r, lcf, ctx);
480 if (rc != NGX_DECLINED) {
481 return rc;
482 }
483
484 return oci_tier_get(r, lcf, ctx);
485}
