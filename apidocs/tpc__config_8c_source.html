1/* File: tpc_config.c — HTTP-TPC location defaults and inheritance
2 * WHAT: Configures HTTP third-party copy (TPC) settings for WebDAV locations. This file contains two lifecycle functions: create_loc_conf initializes TPC fields to NGX_CONF_UNSET sentinel values marking them as unset; merge_loc_conf inherits parent config values using ngx_conf_merge_* macros and applies smart defaults for curl path (/usr/bin/curl), timeout (0 = no limit), OAuth2/OIDC token delegation credentials (empty endpoint, storage.read default scope). Also implements cross-field inheritance: if TPC-specific cert/key paths are unset but general WebDAV cafile/cadir exist, copies those values to ensure TPC has certificate access even when not explicitly configured.
3 *
4 * WHY: HTTP-TPC requires curl as the transfer tool and TLS credentials for source authentication — these defaults must be inherited from parent/server configuration so individual locations can override only what they need. OAuth2/OIDC token delegation enables TPC to acquire temporary credentials for accessing remote endpoints without requiring client-side certificate exchange. The smart inheritance logic (cafile/cadir fallback, cert→key copy) reduces configuration burden by providing reasonable defaults when operators don't explicitly set every field. Thread safety: config setup runs once during nginx startup; no concurrent access after initialization. */

67void
68ngx_http_brix_webdav_tpc_merge_loc_conf(
69 ngx_http_brix_webdav_loc_conf_t *conf,
70 ngx_http_brix_webdav_loc_conf_t *prev)
71{
72 ngx_conf_merge_value(conf->tpc, prev->tpc, 0);
73 /* SSRF policy: deny local, allow private by default (HEP federation nodes
74 * commonly reside on private networks, but loopback must stay blocked). */
75 ngx_conf_merge_value(conf->tpc_allow_local, prev->tpc_allow_local, 0);
76 ngx_conf_merge_value(conf->tpc_allow_private, prev->tpc_allow_private, 1);
77 /* Source-host NAMING allowlist: opt-in (default off) and fail-closed when
78 * on — an unset/empty list denies every source. Complements the range
79 * policy above; enforced in tpc_marker_start.c before curl dials out. */
80 ngx_conf_merge_value(conf->tpc_source_guard, prev->tpc_source_guard, 0);
81 ngx_conf_merge_ptr_value(conf->tpc_source_allow, prev->tpc_source_allow,
82 NULL);
83 ngx_conf_merge_str_value(conf->tpc_curl, prev->tpc_curl,
84 "/usr/bin/curl");
85 ngx_conf_merge_str_value(conf->tpc_cert, prev->tpc_cert, "");
86 ngx_conf_merge_str_value(conf->tpc_key, prev->tpc_key, "");
87 ngx_conf_merge_str_value(conf->tpc_cadir, prev->tpc_cadir, "");
88 ngx_conf_merge_str_value(conf->tpc_cafile, prev->tpc_cafile, "");
89 /* Phase 51 (B2): default the curl low-speed stall detector ON (abort a pull/
90 * push that averages < 1KB/s for 60s) so a stalled remote cannot hold a
91 * thread-pool worker indefinitely. This measures lack of PROGRESS, not
92 * duration, so a slow-but-advancing transfer is never clipped. The absolute
93 * total timeout stays opt-in (0 = unlimited) — operators who need a hard cap
94 * set brix_webdav_tpc_timeout. 0 on either low-speed knob disables it. */
95 ngx_conf_merge_uint_value(conf->tpc_timeout, prev->tpc_timeout, 0);
96 ngx_conf_merge_uint_value(conf->tpc_low_speed_bytes,
97 prev->tpc_low_speed_bytes, 1024);
98 ngx_conf_merge_uint_value(conf->tpc_low_speed_secs,
99 prev->tpc_low_speed_secs, 60);
100 ngx_conf_merge_uint_value(conf->tpc_marker_interval,
101 prev->tpc_marker_interval, 0);
102 ngx_conf_merge_uint_value(conf->tpc_max_streams, prev->tpc_max_streams, 1);
103
104 /* Completion gate: both halves default OFF so an existing deployment sees no
105 * new refusal. A size comparison still runs whenever either half is on and
106 * the source declared a length — that is free and catches truncation. */
107 ngx_conf_merge_value(conf->tpc_require_source_size,
108 prev->tpc_require_source_size, 0);
109 ngx_conf_merge_str_value(conf->tpc_verify_digest,
110 prev->tpc_verify_digest, "");
111
112 /* Per-user TPC credential forwarding defaults ON — a PULL presents the
113 * requesting user's delegated proxy / forwards their bearer to the source by
114 * default (the HTTP equivalent of the native root:// bearer passthrough).
115 * Opportunistic: absence of a per-user credential silently falls back to the
116 * service cert, so enabling it by default never introduces a new denial. */
117 ngx_conf_merge_value(conf->tpc_credential_forward,
118 prev->tpc_credential_forward, 1);
119
120 /* Merge OAuth2/OIDC token-delegation config. */
121 ngx_conf_merge_str_value(conf->tpc_cred.token_endpoint,
122 prev->tpc_cred.token_endpoint, "");
123 ngx_conf_merge_str_value(conf->tpc_cred.token_client_id,
124 prev->tpc_cred.token_client_id, "");
125 ngx_conf_merge_str_value(conf->tpc_cred.token_client_secret,
126 prev->tpc_cred.token_client_secret, "");
127 ngx_conf_merge_str_value(conf->tpc_cred.token_scope,
128 prev->tpc_cred.token_scope, "storage.read");
129
130 if (conf->tpc_cadir.len == 0 && conf->cadir.len > 0) {
131 conf->tpc_cadir = conf->cadir;
132 }
133 if (conf->tpc_cafile.len == 0 && conf->cafile.len > 0) {
134 conf->tpc_cafile = conf->cafile;
135 }
136 if (conf->tpc_key.len == 0 && conf->tpc_cert.len > 0) {
137 conf->tpc_key = conf->tpc_cert;
138 }
139}
