20static int
21split_parent(const char *path, char parent[XRDC_PATH_MAX], char self[XRDC_NAME_MAX])
22{
23 const char *slash = strrchr(path, '/');
24 size_t parent_length;
25 size_t self_length;
26
27 if (slash == NULL || slash[1] == '\0') return -1;
28 parent_length = slash == path ? 1U : (size_t) (slash - path);
29 self_length = strlen(slash + 1);
30 if (parent_length >= XRDC_PATH_MAX || self_length >= XRDC_NAME_MAX)
31 return -1;
32 memcpy(parent, path, parent_length);
33 parent[parent_length] = '\0';
34 memcpy(self, slash + 1, self_length + 1);
35 return 0;
36}

39add_name(namespace_names *set, const char *name, size_t length)
40{
41 size_t index;
42
43 if (length == 0 || length >= XRDC_NAME_MAX
44 || set->count >= XPR_NAMESPACE_MAX) return -1;
45 for (index = 0; index < length; index++)
46 if (name[index] == '\0' || name[index] == '/' || name[index] == '&')
47 return -1;
48 memcpy(set->names[set->count], name, length);
49 set->names[set->count][length] = '\0';
50 set->count++;

64namespace_digest(namespace_names *set, char digest[65])
65{
66 unsigned char *serialized;
67 size_t length = 0;
68 size_t offset = 0;
69 size_t index;
70
71 if (set->count > 64) return -1;
72 qsort(set->names, set->count, XRDC_NAME_MAX, compare_name);
73 for (index = 0; index < set->count; index++) {
74 if (index > 0 && strcmp(set->names[index - 1], set->names[index]) == 0)
75 return -1;
76 length += strlen(set->names[index]) + 1;
77 }
78 serialized = malloc(length == 0 ? 1 : length);
79 if (serialized == NULL) return -1;
80 for (index = 0; index < set->count; index++) {
81 size_t name_length = strlen(set->names[index]) + 1;
82
83 memcpy(serialized + offset, set->names[index], name_length);
84 offset += name_length;
85 }
86 brix_s3_sha256_hex(serialized, length, digest);

170

172 key = bucket_end + 1;
173 slash = strrchr(key, '/');
174 prefix_length = slash == NULL ? 0 : (size_t) (slash - key + 1);
175 if (prefix_length >= XRDC_PATH_MAX) return -1;

178 if (brix_http_urlencode((const unsigned char *) prefix, prefix_length,

180 bucket_length = (size_t) (bucket_end - url->path);

184 (int) bucket_length, url->path, prefix_length == 0 ? "" : "&prefix=",
185 encoded) < sizeof(listing->path) ? 0 : -1;
