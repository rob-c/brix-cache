1/* rpm_repomd.h — the freshness root, read for what a dnf client asks next
2 * (phase-104 D15.10).
3 *
4 * WHAT: extract from a repomd.xml the `<location href="...">` of the metadata
5 * files a package client fetches unconditionally right after it, and
6 * compose each one into the mirror's own cache key.
7 * WHY: Appendix X, finding X-3: a stock EL9 dnf fetches repomd.xml and then
8 * primary AND filelists, every time its metadata expires, before it can
9 * answer any question at all. On a cold mirror those two fetches are
10 * the client's whole wait, and they are perfectly predictable — the
11 * repomd the mirror has just pulled names them. Warming them is the one
12 * piece of speculation a repository mirror can do that is never wrong:
13 * the objects are named by the index the client already has, they are
14 * digest-named (so they are verified on arrival like any other fill),
15 * and if the client never asks, the mirror has cached two files the
16 * next client will ask for.
17 * HOW: pure C over the caller's buffer — no nginx types, no allocation,
18 * spans point into the XML. Not an XML parser: a repomd is a fixed
19 * shape written by createrepo, and the only thing read out of it is an
20 * attribute value that must then survive the SAME path grammar the gate
21 * applied to the request (brix_rpm_classify) before it can become a
22 * fetch. An href this file cannot vouch for is dropped, never repaired.
23 */
24#ifndef BRIX_PROTOCOLS_RPM_REPOMD_H

29/* The warm set is {primary, filelists} — `other` is never fetched by dnf4 and
30 * is optional in dnf5 (finding X-3), so speculating on it would be paying for
31 * the one file the client does not want. */
32#define BRIX_RPM_REPOMD_WARM_MAX 2

34/* The longest href this reader will carry. Repository layouts nest, but a
35 * location inside one repository is a short relative path; anything longer is
36 * not a file createrepo named. */
37#define BRIX_RPM_REPOMD_HREF_MAX 512

39/* A repomd.xml larger than this is not a repository index this mirror is
40 * willing to speculate on: the biggest real one is a few kilobytes, and the
41 * cap is what bounds the read the serving path does inline. */
42#define BRIX_RPM_REPOMD_MAX (256 * 1024)

48/* Fill `out` (capacity `max`) with the warm-set hrefs `xml` names, in the
49 * order they appear. Returns how many were written (0 when the document names
50 * none it can vouch for). An href that is absolute, escapes its repository,
51 * carries a scheme or an XML entity, or is empty is DROPPED — the caller
52 * cannot tell a dropped one from an absent one, which is the point. */
53size_t brix_rpm_repomd_warm_set(const char *xml, size_t len,

56/* Compose the cache key of a file `href` names relative to the repository
57 * root, given the key of the repomd.xml that named it (".../repodata/
58 * repomd.xml"). Writes a NUL-terminated key into `out` and returns 0, or -1
59 * when the repomd key is not repodata-shaped or the composition would not
60 * fit. The result is NOT trusted on the strength of this composition: the
61 * caller classifies it, and only a metadata-class key is ever fetched. */
62int brix_rpm_repomd_sibling_key(const char *repomd_key, size_t key_len,
