13/*
14 * cns_emit_ready — is this worker in a position to report a mutation at all?
15 *
16 * EMIT mode plus at least one manager link that is connected and logged in.
17 * Shared by both emit entry points so the gate is stated once.
18 */
19static ngx_flag_t
20cns_emit_ready(ngx_stream_brix_srv_conf_t *conf)
21{
22 ngx_uint_t i;
23
24 if (conf == NULL || conf->cns_mode != BRIX_CNS_EMIT) {
25 return 0;
26 }
27
28 for (i = 0; i < conf->cms.nctxs; i++) {
29 if (conf->cms.ctxs[i]->connection != NULL
30 && conf->cms.ctxs[i]->logged_in)
31 {
32 return 1;
33 }
34 }
35
36 return 0;
37}

39/*
40 * cns_emit_send — fan one encoded CNS event out to EVERY ready manager link.
41 *
42 * Redundant managers each keep their own inventory, so an event that reached
43 * only one of them would leave the others' catalogues stale — exactly the
44 * split-brain redundancy is meant to prevent. Best-effort per link.
45 */
46static void
47cns_emit_send(ngx_stream_brix_srv_conf_t *conf, const uint8_t *buf, size_t n)
48{
49 ngx_uint_t i;
50
51 for (i = 0; i < conf->cms.nctxs; i++) {
52 if (conf->cms.ctxs[i]->connection != NULL
53 && conf->cms.ctxs[i]->logged_in)
54 {
55 (void) brix_cms_send_frame(conf->cms.ctxs[i]->connection, 0,
56 CMS_RR_CNS, CMS_MOD_RAW, buf, n);
57 }
58 }
59}

61/*
62 * cns_logical — the client-facing path for an on-disk resolved path.
63 *
64 * Strips the export root_canon prefix, keeping the leading '/'. A path that is
65 * not under the export root is returned unchanged (the encode step then rejects
66 * anything oversized).
67 *
68 * Takes the root as a plain string rather than the conf it came from: an HTTP
69 * plane resolves against its OWN export root, which is the prefix that has to
70 * come off, even though the frame leaves over the stream plane's manager link.
71 */
72static const char *
73cns_logical(const char *root, const char *resolved)
74{
75 size_t rlen = (root != NULL) ? ngx_strlen(root) : 0;
76
77 if (rlen > 0 && ngx_strncmp(resolved, root, rlen) == 0
78 && resolved[rlen] == '/')
79 {
80 return resolved + rlen;
81 }
82 return resolved;
83}

85/*
86 * cns_emit_conf — the server block whose manager link this worker reports over.
87 *
88 * WHY: the CMS link hangs off a stream{} server conf, but WebDAV/S3/gridftp are
89 * http{} planes with no path to it. Rather than a new directive and a
90 * per-worker cached pointer (a global, which CLAUDE.md forbids), this
91 * derives the emitter from the cycle on demand — the same walk
92 * config/process.c already does at worker init, so it introduces no state
93 * at all and cannot go stale across a reload.
94 * HOW: first server block that is in EMIT mode with a live, logged-in link.
95 * A node's manager links all hang off one server block (worker-0-gated,
96 * see cms_start.c), so "first ready" is "the" emitter, not an arbitrary
97 * pick among equals — and the send fans out to every ready link.
98 *
99 * The walk is over server blocks (a handful), and every caller is already
100 * committing a namespace mutation — a syscall plus a frame — so it is noise
101 * against the work it accompanies.
102 */
103static ngx_stream_brix_srv_conf_t *
104cns_emit_conf(void)
105{
106 ngx_stream_core_main_conf_t *cmcf;
107 ngx_stream_core_srv_conf_t **cscfp;
108 ngx_stream_brix_srv_conf_t *xcf;
109 ngx_uint_t i;
110
111 if (ngx_cycle == NULL) {
112 return NULL;
113 }
114
115 cmcf = ngx_stream_cycle_get_module_main_conf(ngx_cycle,
116 ngx_stream_core_module);
117 if (cmcf == NULL) {
118 return NULL; /* no stream{} block → no CMS link → no CNS */
119 }
120
121 cscfp = cmcf->servers.elts;
122
123 for (i = 0; i < cmcf->servers.nelts; i++) {
124 xcf = ngx_stream_conf_get_module_srv_conf(cscfp[i],
125 ngx_stream_brix_module);
126 if (cns_emit_ready(xcf)) {
127 return xcf;
128 }
129 }
130
131 return NULL;
132}

134/*
135 * brix_cns_emit — report one namespace mutation to the manager.
136 *
137 * Best-effort, fire-and-forget: only in EMIT mode, only when the worker's manager
138 * link is connected + logged in.
139 */
140void
141brix_cns_emit(ngx_stream_brix_srv_conf_t *conf, uint8_t op,
142 const char *resolved, uint64_t size, uint64_t mtime)
143{
144 uint8_t buf[BRIX_CNS_HDR_LEN + BRIX_CNS_PATH_MAX];
145 size_t n;
146
147 if (resolved == NULL || !cns_emit_ready(conf)) {
148 return;
149 }
150
151 n = brix_cns_event_encode(op, cns_logical(conf->common.root_canon, resolved),
152 size, mtime, buf, sizeof(buf));
153 if (n == 0) {
154 return; /* empty or over-long logical path → skip (torn/oversized) */
155 }
156
157 cns_emit_send(conf, buf, n);
158}

160/*
161 * brix_cns_emit_rename — report a rename as ONE event, not a DEL plus an ADD.
162 *
163 * A DEL/ADD pair would be wrong for a directory: the manager would drop the
164 * directory entry and add the new one while every recorded child stayed at a
165 * path the cluster no longer serves. The single BRIX_CNS_MV event lets the
166 * manager carry the subtree atomically under its inventory lock, and it also
167 * survives the pair arriving out of order, which two independent frames cannot.
168 */
169void
170brix_cns_emit_rename(ngx_stream_brix_srv_conf_t *conf, const char *src_resolved,
171 const char *dst_resolved, uint64_t size, uint64_t mtime, int is_dir)
172{
173 uint8_t buf[BRIX_CNS_HDR_LEN + 2 * BRIX_CNS_PATH_MAX + 2];
174 size_t n;
175
176 if (src_resolved == NULL || dst_resolved == NULL || !cns_emit_ready(conf)) {
177 return;
178 }
179
180 n = brix_cns_event_encode_mv(cns_logical(conf->common.root_canon,
181 src_resolved),
182 cns_logical(conf->common.root_canon,
183 dst_resolved),
184 size, mtime, is_dir, buf, sizeof(buf));
185 if (n == 0) {
186 return;
187 }
188
189 cns_emit_send(conf, buf, n);
190}
