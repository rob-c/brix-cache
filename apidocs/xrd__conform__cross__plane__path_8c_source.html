5/* WHAT: Submit one literal parent segment through the ROOT stat operation.
6 * WHY: XPR-008 needs the same hostile relative shape on each selected plane.
7 * HOW: Append /../outside and retain only a defined protocol refusal or success. */
8void
9xrd_conform_probe_root_traversal(xrd_conform_execution *execution,
10 brix_conn *connection, const char *path)
11{
12 const char *const suffixes[] = {
13 "/../brix-conform-outside",
14 "/%2e%2e%2fbrix-conform-outside"
15 };
16 brix_statinfo info;
17 brix_status status;
18 char hostile[XRDC_PATH_MAX];
19 char reason[64];
20 size_t index;
21 int state = 1;
22
23 if (!xrd_conform_row_active(execution, "XPR-008")) return;
24 for (index = 0; index < sizeof(suffixes) / sizeof(suffixes[0]); index++) {
25 if ((size_t) snprintf(hostile, sizeof(hostile), "%s%s", path,

28 state = 0;
29 break;
30 }

34 }

104/* WHAT: Derive one bounded path differing only in basename letter case.
105 * WHY: XPR-010 must not accidentally compare a different directory or object.
106 * HOW: Copy the path and toggle the first ASCII letter after its final slash. */
107static int
108case_variant(const char *path, char out[XRDC_PATH_MAX])
109{
110 char *name;
111 char *cursor;
112
113 if ((size_t) snprintf(out, XRDC_PATH_MAX, "%s", path) >= XRDC_PATH_MAX)
114 return -1;
115 name = strrchr(out, '/');
116 name = name == NULL ? out : name + 1;
117 for (cursor = name; *cursor != '\0'; cursor++) {
118 if (*cursor >= 'a' && *cursor <= 'z') {
119 *cursor = (char) (*cursor - 'a' + 'A');
120 return 0;
121 }
122 if (*cursor >= 'A' && *cursor <= 'Z') {
123 *cursor = (char) (*cursor - 'A' + 'a');
124 return 0;
125 }
126 }
127 return -1;
128}

201/* WHAT: Construct the UTF-8, Latin-1 and raw-byte sibling names for one plane.
202 * WHY: HTTP needs percent-encoded octets while S3 canonicalization starts from
203 * raw key bytes; using one representation would double-encode one plane.
204 * HOW: Append a fixed class tag plus either canonical escapes or literal bytes. */
205static int
206filename_variant(const char *path, int web_encoded, size_t index,
207 char out[XRDC_PATH_MAX])
208{
209 static const char *const raw_suffixes[] = {
210 ".brix-conform-utf8-\xc3\xa9",
211 ".brix-conform-latin1-\xe9",
212 ".brix-conform-raw-\xff"
213 };
214 static const char *const encoded_suffixes[] = {
215 ".brix-conform-utf8-%C3%A9",
216 ".brix-conform-latin1-%E9",
217 ".brix-conform-raw-%FF"
218 };
219 const char *suffix = web_encoded ? encoded_suffixes[index]
220 : raw_suffixes[index];
221
222 return (size_t) snprintf(out, XRDC_PATH_MAX, "%s%s", path, suffix)
223 < XRDC_PATH_MAX ? 0 : -1;
224}

251web_filename_state(xrd_conform_execution *execution, const brix_weburl *url,
252 const char *token, size_t index)
253{
254 brix_weburl variant = *url;
255 brix_http_resp response;
256 brix_status status;

258 const char *expected = url->is_s3 ? execution->s3_digest
259 : execution->http_digest;
260 int rc;
261
262 if (filename_variant(url->path, !url->is_s3, index, variant.path) != 0)
263 return 0;
264 memset(&response, 0, sizeof(response));
265 brix_status_clear(&status);
266 rc = xrd_conform_http_request_fields(execution, &variant, "GET", token,
267 NULL, 65536, &response, &status);
268 if (rc != 0) return 0;
269 if (response.status == 400 || response.status == 403) rc = 1;
270 else if (response.status >= 200 && response.status < 300) {
271 brix_s3_sha256_hex(response.body, response.body_len, digest);
272 rc = strcmp(digest, expected) == 0 ? 2 : -1;
273 } else rc = 0;
274 brix_http_resp_free(&response);
275 return rc;
