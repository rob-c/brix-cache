41/* HTTP -> kXR status mapping */
42
43/*
44 * WHAT: One row of the HTTP-status to kXR-error mapping table.
45 * WHY: A data table replaces a long switch ladder, keeping the lookup
46 * function trivially simple and the mapping auditable at a glance.
47 * HOW: http_status is the numeric HTTP code (nginx macro value); xrd_code
48 * is the corresponding kXR_* error constant.
49 */
50typedef struct {
51 ngx_int_t http_status;
52 int xrd_code;
53} xrdhttp_status_map_row_t;
54

75/*
76 * WHAT: Map any non-2xx HTTP status to a kXR_* error code by class.
77 * WHY: HTTP codes without an exact table entry still need a sensible kXR
78 * error so clients never see kXR_ok on a failure.
79 * HOW: 4xx -> kXR_ArgInvalid, 5xx (and above) -> kXR_ServerError, and any
80 * remaining code (1xx/3xx) -> kXR_ok, matching the original ladder.
81 */
82static int
83xrdhttp_status_class_default(ngx_int_t http_status)
84{
85 if (http_status >= 400 && http_status < 500) {
86 return kXR_ArgInvalid;
87 }
88 if (http_status >= 500) {
89 return kXR_ServerError;
90 }
91 return kXR_ok;
92}

94int
95xrdhttp_http_to_xrd_status(ngx_int_t http_status)
96{
97 size_t i;
98
99 if (http_status >= 200 && http_status < 300) {
100 return kXR_ok;
101 }
102
103 for (i = 0; i < sizeof(xrdhttp_status_map) / sizeof(xrdhttp_status_map[0]);
104 i++)
105 {
106 if (xrdhttp_status_map[i].http_status == http_status) {
107 return xrdhttp_status_map[i].xrd_code;
108 }
109 }
110
111 return xrdhttp_status_class_default(http_status);
112}

114ngx_int_t
115xrdhttp_add_response_headers(ngx_http_request_t *r, ngx_int_t http_status)
116{
117 xrdhttp_req_ctx_t *ctx;
118 int xrd_code;
119
120 ctx = (xrdhttp_req_ctx_t *)
121 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
122 if (ctx == NULL || !ctx->is_xrdhttp) {
123 return NGX_OK;
124 }
125
126 /* Guard against double-injection: filter + direct call in get.c. */
127 if (ctx->headers_injected) {
128 return NGX_OK;
129 }
130 ctx->headers_injected = 1;
131
132 /* X-Xrootd-Requuid — echo the request UUID back to the client. */
133 if (ctx->requuid[0]) {
134 if (brix_http_set_header(r, "X-Xrootd-Requuid", ctx->requuid,
135 NULL) != NGX_OK)
136 {
137 return NGX_ERROR;
138 }
139 }
140
141 /* X-Xrootd-Status — always emit per XrdHttp spec: 0 = success, non-zero = error. */
142 xrd_code = xrdhttp_http_to_xrd_status(http_status);
143 if (brix_http_set_header_num(r, "X-Xrootd-Status",
144 (long) xrd_code) != NGX_OK)
145 {
146 return NGX_ERROR;
147 }
148
149 /* X-Xrootd-Wait — back-pressure for async/tape-recall scenarios. */
150 if (ctx->wait_seconds > 0) {
151 if (brix_http_set_header_num(r, "X-Xrootd-Wait",
152 (long) ctx->wait_seconds) != NGX_OK)
153 {
154 return NGX_ERROR;
155 }
156 }
157
158 /* X-Xrootd-Retry — hint for client retry interval. */
159 if (ctx->retry_seconds > 0) {
160 if (brix_http_set_header_num(r, "X-Xrootd-Retry",
161 (long) ctx->retry_seconds) != NGX_OK)
162 {
163 return NGX_ERROR;
164 }
165 }
166
167 return NGX_OK;
168}
