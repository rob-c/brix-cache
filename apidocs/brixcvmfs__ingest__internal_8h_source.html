1/*
2 * brixcvmfs_ingest_internal.h — seam of the `brixcvmfs ingest` conductor
3 * (phase-104 D8/D9): the dir/image/prune verbs that turn a folder or an OCI
4 * image into a Stratum-0 publish.
5 *
6 * WHAT: the ingest exit-code table, the small filesystem/lock helpers the
7 * ingest TUs share, and the per-verb entry points.
8 * WHY: ingest is five TUs under the file-size cap (brixcvmfs_ingest.c:
9 * dispatch + `dir` + helpers; _image.c: `image`; _prune.c: `prune`;
10 * _diffid.c: the config/diff_id comparator; _layout.c: the shape
11 * of the published namespace);
12 * the transaction-lock primitives stay owned by brixcvmfs_publish.c so
13 * `repo transaction` and `ingest` can never drift on lock discipline.
14 * HOW: pure tool surface (G14): no FUSE, no ngx, no root. Everything here
15 * is single-threaded CLI plumbing.
16 */

98/* One layer of an image published under `--layout layered` (D15.6): its own
99 * digest, the diff_id naming its uncompressed bytes, and whether this run
100 * had to materialize it at all. A layer root is content-addressed, so an
101 * image whose base is already published pays for nothing but its own top
102 * layers — which is the entire point of the layout. */
103typedef struct {
104 brix_oci_digest_t dig;
105 ing_diffid_t diffid; /* "" = not known */
106 int reused; /* 1 = already published; not fetched */
107} ing_layer_t;

117/* Decide whether one layer must be materialized, and where. `updir` gets
