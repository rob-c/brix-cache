17static int
18qprep_status_valid(const char *reply, const char *path)
19{
20 const char *cursor = reply;
21
22 while (cursor != NULL && *cursor != '\0') {
23 const char *end = strchr(cursor, '\n');
24 size_t length = end == NULL ? strlen(cursor) : (size_t) (end - cursor);
25
26 if (length >= 3 && strchr("AMqsf", cursor[0]) != NULL
27 && cursor[1] == ' ' && strlen(path) == length - 2
28 && memcmp(cursor + 2, path, length - 2) == 0) return 1;
29 cursor = end == NULL ? NULL : end + 1;
30 }
31 return 0;
32}

61/* WHAT: Compare cancellation of a foreign live id with an absent id.
62 * WHY: Different wire results disclose request existence or ownership.
63 * HOW: Issue both cancels in one fresh session and compare result classes. */
64static int
65cancel_is_non_enumerating(xrd_conform_execution *execution,
66 const char *request_id, brix_status *status)
67{
68 static const char absent_id[] = "brix-conform-absent-request-id";
69 const char *real[] = { request_id };
70 const char *absent[] = { absent_id };
71 char reply[256];
72 brix_status real_status;
73 brix_status absent_status;
74 brix_url url;
75 brix_conn auxiliary;
76 brix_opts options = execution->run->opts;
77 int real_rc;
78 int absent_rc;
79
80 options.capture = NULL;
81 if (!xrd_conform_reserve_zero_status(execution, status)
82 || brix_endpoint_parse(execution->target, &url, status) != 0
83 || brix_connect(&auxiliary, &url, &options, status) != 0)
84 return 0;
85 brix_status_clear(&real_status);
86 real_rc = xrd_conform_reserve_zero_status(execution, &real_status)
87 ? brix_prepare(&auxiliary, real, 1, kXR_cancel, 0, 0,
88 reply, sizeof(reply), &real_status) : -1;
89 brix_status_clear(&absent_status);
90 absent_rc = xrd_conform_reserve_zero_status(execution, &absent_status)
91 ? brix_prepare(&auxiliary, absent, 1, kXR_cancel, 0, 0,
92 reply, sizeof(reply), &absent_status) : -1;
93 brix_close(&auxiliary);
94 if (real_rc == absent_rc
95 && (real_rc == 0 || real_status.kxr == absent_status.kxr)) return 1;
96 brix_status_set(status, XRDC_EPROTO, real_status.kxr,
97 "foreign cancel rc=%d/kxr=%d differs from absent rc=%d/kxr=%d",
98 real_rc, real_status.kxr, absent_rc, absent_status.kxr);
99 return 0;
100}
