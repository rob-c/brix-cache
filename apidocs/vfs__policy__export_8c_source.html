1/*
2 * vfs_policy_export.c — policy-bearing wrappers over the raw export helpers
3 * (phase-105 W3).
4 *
5 * WHAT: One thin, gated form for every confinement-only mutator in vfs_ops.h
6 * (open_fd, open_fd_at, unlink_path, unlink_at, rmdir_path, mkdir_path,
7 * backend_mkpath, rename_path, copyfile, copytree). Each takes a
8 * brix_vfs_export_op_ctx_t carrying the request's endpoint mutation
9 * policy, refuses with EROFS on a read-only endpoint, and otherwise
10 * delegates verbatim to the raw helper.
11 *
12 * WHY: The raw helpers exist because bulk and off-thread consumers (multipart
13 * assembly, the async namespace queue, TPC, CMS forwarding, checkpoint
14 * recovery) cannot use the pool-allocating, metered handle API. They
15 * carry confinement but knew nothing about whether the endpoint may be
16 * written at all, which made them a complete alternate route around every
17 * gate the handle API applies. Rather than thread a boolean through ten
18 * signatures, the authority travels as one typed bundle, so a caller
19 * cannot accidentally pass the wrong argument in the right position.
20 *
21 * HOW: Every wrapper is: gate, then delegate. The gate is the same
22 * brix_vfs_export_require_mutation() kernel the context forms use, so the
23 * decision, the errno, and the single denial observation are identical on
24 * and off the event loop. The int-returning helpers translate the
25 * kernel's NGX_ERROR into their own -1/errno contract; brix_vfs_open_fd's
26 * wrapper additionally classifies its raw O_* flags, so a read-only open
27 * through a raw helper stays legal on a read-only export.
28 *
29 * The un-gated originals stay in vfs_walk.c/vfs_read.c for VFS-internal and
30 * explicitly service-owned domains (the cache store, the staging area, a
31 * VFS-owned unpublished temporary). Protocol, TPC, CMS, and queue code calls
32 * the forms in THIS file.
33 */
34#include "vfs_internal.h"
35#include "vfs_policy.h"

37/* ---- Gated confined open beneath an export root ----
38 *
39 * WHAT: Returns a raw confined fd, or -1 with errno set; -1/EROFS when the
40 * endpoint is read-only and `flags` describe a mutating open.
41 *
42 * WHY: A raw open is the widest of the raw helpers: O_CREAT|O_TRUNC through it
43 * creates and empties an export object with no other gate in the way.
44 * Reads must stay free, so the flag word — not the caller — decides
45 * whether the kernel runs.
46 *
47 * HOW: 1. Classify the flags; 2. gate a mutating open as MUTATE_OPEN;
48 * 3. delegate to brix_vfs_open_fd with the bundle's log and root.
49 */
50int
51brix_vfs_export_open_fd(const brix_vfs_export_op_ctx_t *opctx,
52 const char *logical, int flags, mode_t mode)
53{
54 if (opctx == NULL) {
55 errno = EINVAL;
56 return -1;
57 }
58
59 if (brix_vfs_open_flags_mutate(flags)
60 && brix_vfs_export_require_mutation(opctx, BRIX_VFS_MUTATE_OPEN)
61 != NGX_OK)
62 {
63 return -1;
64 }
65
66 return brix_vfs_open_fd(opctx->log, opctx->root_canon, logical, flags,
67 mode);
68}

192/* ---- Gated recursive backend mkdir of a path and its parents ----
193 *
194 * WHAT: 0 on success, NGX_DECLINED for a default POSIX export (the caller uses
195 * its own confined mkpath), -1 with errno set otherwise; -1/EROFS on a
196 * read-only endpoint.
197 *
198 * WHY: The NGX_DECLINED answer is the one a caller must still be able to get
199 * cheaply, but only after the policy question: probing which backend an
200 * export selected is exactly the capability disclosure §6 forbids ahead
201 * of the decision.
202 *
203 * HOW: 1. Gate as MUTATE_MKDIR; 2. delegate to brix_vfs_backend_mkpath.
204 */
205int
206brix_vfs_export_mkpath(const brix_vfs_export_op_ctx_t *opctx,
207 const char *logical, mode_t mode)
208{
209 char physical[PATH_MAX];
210
211 if (brix_vfs_export_require_mutation(opctx, BRIX_VFS_MUTATE_MKDIR)
212 != NGX_OK)
213 {
214 return -1;
215 }
216
217 if (brix_path_export_to_pfn(opctx->root_canon, opctx->n2n, logical,
218 physical, sizeof(physical)) != NGX_OK)
219 {
220 return -1;
221 }
222 return brix_vfs_backend_mkpath(opctx->root_canon, physical, mode,
223 opctx->log);
224}
