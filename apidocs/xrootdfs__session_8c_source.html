35/* WHAT: Copy a live inode name, returning negative errno if absent.
36 * WHY: Never substitute root or another file for a forgotten/unlinked inode.
37 * HOW: 1. Resolve its exact id. 2. Require a surviving alias. 3. Copy bounded key. */
38int
39xfs_req_path(xfs_session *session, fuse_ino_t ino, char *out)
40{
41 const char *path = xfs_node_path(xfs_nodes_find(&session->nodes, ino));
42 if (!path) {
43 return -ESTALE;
44 }
45 memcpy(out, path, strlen(path) + 1);
46 return 0;
47}

49/* WHAT: Construct one child key, returning negative errno on malformed input.
50 * WHY: FUSE node names cannot escape their parent or alias another namespace.
51 * HOW: 1. Reject traversal/separators. 2. Resolve parent. 3. Append with bounds. */
52int
53xfs_req_child(xfs_session *session, fuse_ino_t parent, const char *name, char *out)
54{
55 if (!name[0] || strchr(name, '/') || strcmp(name, ".") == 0
56 || strcmp(name, "..") == 0) {
57 return -EINVAL;
58 }
59 int rc = xfs_req_path(session, parent, out);
60 if (rc != 0) {
61 return rc;
62 }
63 size_t length = strlen(out), extra = strlen(name);
64 size_t slash = length > 1 ? 1 : 0;
65 if (length + slash + extra >= XRDC_PATH_MAX
66 || strlen(g_base) + length + slash + extra >= XRDC_PATH_MAX) {
67 return -ENAMETOOLONG;
68 }
69 if (slash) {
70 out[length++] = '/';
71 }
72 memcpy(out + length, name, extra + 1);
73 return 0;
74}

76/* WHAT: Reply with one stable identity and balanced lookup/open owners.
77 * WHY: LINK must return its source node id, not a fresh path-based id.
78 * HOW: 1. Authorize/stat as the caller. 2. Bind reserved alias. 3. Reply and
79 * compensate references if the kernel rejects an interrupted reply. */
80int
81xfs_ll_entry(fuse_req_t req, xfs_session *session, const char *path,
82 struct fuse_file_info *fi, xfs_node_ticket *ticket, xfs_node *preferred)
83{
84 struct fuse_entry_param entry;
85 memset(&entry, 0, sizeof(entry));
86 int rc;
87 if (fi && fi->fh && !g_web) {
88 afh *handle = (afh *) (uintptr_t) fi->fh;
89 brix_statinfo info;
90 brix_status status;
91 brix_status_clear(&status);
92 rc = brix_mfile_stat(handle->mf, &info, &status);
93 if (rc == 0) {
94 xfs_fill_stat(&info, &entry.attr);
95 } else {
96 rc = xfs_err(&status);
97 }
98 } else {
99 rc = xfs_getattr(xfs_req_uid(req), path, &entry.attr, fi);
100 }
101 if (rc != 0) {
102 return rc;
103 }
104 xfs_node *node = xfs_nodes_bind(&session->nodes, ticket,
105 (uint64_t) entry.attr.st_ino, preferred);
106 entry.ino = node->ino;
107 entry.generation = node->generation;
108 entry.attr.st_ino = (ino_t) node->ino;
109 entry.attr_timeout = g_sss_ident ? 0 : g_attr_timeout;
110 entry.entry_timeout = g_sss_ident ? 0 : g_entry_timeout;
111 node->lookups++;
112 if (fi) {
113 ((afh *) (uintptr_t) fi->fh)->attr = entry.attr;
114 xfs_ll_handle_attach(node, fi, xfs_req_uid(req));
115 rc = fuse_reply_create(req, &entry, fi);
116 } else {
117 rc = fuse_reply_entry(req, &entry);
118 }
119 if (rc != 0) {
120 if (fi) {
121 node->open_handles = ((afh *) (uintptr_t) fi->fh)->node_next;
122 xfs_release(path, fi);
123 node->opens--;
124 }
125 xfs_nodes_forget(&session->nodes, node, 1);
126 }
127 return 0; /* a reply, including an interrupted one, was already sent */
128}

156/* WHAT: Mount/run one prepared session. WHY: Balance signals and unmount on exit.
157 * HOW: 1. Mount. 2. Install signal hooks. 3. Run requested threading. 4. Unwind. */
158static int
159xfs_session_run(struct fuse_session *session, struct fuse_cmdline_opts *opts)
160{
161 if (fuse_session_mount(session, opts->mountpoint) != 0) {
162 return 1;
163 }
164 if (fuse_set_signal_handlers(session) != 0) {
165 fuse_session_unmount(session);
166 return 1;
167 }
168 int rc = opts->singlethread ? fuse_session_loop(session)
169 : fuse_session_loop_mt(session, opts->clone_fd);
170 fuse_remove_signal_handlers(session);
171 fuse_session_unmount(session);
172 return rc < 0 ? 1 : 0;
173}

175/* WHAT: Run the low-level driver, returning shell status.
176 * WHY: A single inode-aware path serves every supported host and transport.
177 * HOW: 1. Initialize mount-owned state. 2. Parse public libfuse arguments.
178 * 3. Create/run/destroy the session. 4. Free all mount resources. */
179int
180xfs_session_main(int argc, char **argv)
181{
182 xfs_session *state = calloc(1, sizeof(*state));
183 if (!state) {
184 return 1;
185 }
186 int rc = xfs_nodes_init(&state->nodes);
187 if (rc != 0) {
188 free(state);
189 return 1;
190 }
191 pthread_mutex_init(&state->mutex, NULL);
192 struct fuse_args args = FUSE_ARGS_INIT(argc, argv);
193 struct fuse_cmdline_opts opts;
194 memset(&opts, 0, sizeof(opts));
195 rc = 1;
196 if (fuse_parse_cmdline(&args, &opts) == 0 && opts.mountpoint) {
197 struct fuse_session *session = fuse_session_new(&args, &xfs_lowlevel_ops,
198 sizeof(xfs_lowlevel_ops), state);
199 if (session) {
200 rc = xfs_session_run(session, &opts);
201 fuse_session_destroy(session);
202 }
203 }
204 free(opts.mountpoint);
205 fuse_opt_free_args(&args);
206 xfs_nodes_destroy(&state->nodes);
207 pthread_mutex_destroy(&state->mutex);
208 free(state);
209 return rc;
210}
