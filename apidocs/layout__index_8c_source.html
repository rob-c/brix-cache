18/* JSON-safe check for a media type we are about to embed verbatim. */
19static int
20lay_mt_safe(const char *mt)
21{
22 for (; *mt != '\0'; mt++) {
23 if (*mt < 0x20 || *mt == '"' || *mt == '\\' ||
24 (unsigned char) *mt > 0x7e) {
25 return 0;
26 }
27 }
28 return 1;
29}

59/* Keep one old manifest unless the binding being written replaces it. */
60static int
61lay_index_keep_entry(char **acc, size_t *alen, size_t *acap, int *first,
62 const char *el, size_t en, const char *refname,
63 const char *digest)
64{
65 char eref[160], edig[80] = "";
66
67 lay_elem_refname(el, en, eref, sizeof(eref));
68 brix_json_get_str(el, en, "digest", edig, sizeof(edig));
69 if ((refname != NULL && strcmp(eref, refname) == 0) ||
70 (refname == NULL && eref[0] == '\0' &&
71 strcmp(edig, digest) == 0)) {
72 return 0;
73 }
74 if (!*first && regc_buf_append(acc, alen, acap, ",", 1) != 0) {
75 return -1;
76 }
77 if (regc_buf_append(acc, alen, acap, el, en) != 0) {
78 return -1;
79 }
80 *first = 0;
81 return 0;
82}

84/* Copy retained raw entries so unfamiliar fields survive an index rewrite. */
85static int
86lay_index_keep_old(char **acc, size_t *alen, size_t *acap, int *first,
87 const char *old, size_t olen, const char *refname,
88 const char *digest)
89{
90 const char *arr, *el;
91 size_t an, en, cur = 0;

94 return 0;
95 }
96 while (brix_json_arr_next(arr, an, &cur, &el, &en) == 1) {

99 return -1;
100 }
101 }
102 return 0;

106static int
107lay_index_new_entry(char *entry, size_t elen, int first, const char *refname,
108 const char *dg, const char *mt, size_t size)
109{
110 int n;
111
112 n = snprintf(entry, elen,
113 "%s{\"mediaType\":\"%s\",\"digest\":\"%s\","
114 "\"size\":%zu", first ? "" : ",", mt, dg, size);
115 if (refname != NULL && n > 0 && (size_t) n < elen) {
116 n += snprintf(entry + n, elen - (size_t) n,
117 ",\"annotations\":{\"org.opencontainers.image."
118 "ref.name\":\"%s\"}", refname);
119 }
120 if (n < 0 || (size_t) n + 3 >= elen) {
121 return -1;
122 }
123 memcpy(entry + n, "}]}\n", 4);
124 return n + 4;
125}

127/* Rebuild index.json text: the surviving old entries (raw spans, unknown
128 * fields preserved) + the new binding. 0 with a malloc'd *acc_out / -1
129 * (everything freed). */
130static int
131lay_index_rebuild(const char *old, size_t olen, const char *refname,
132 const char *digest, const char *dg, const char *mt,
133 size_t size, char **acc_out, size_t *alen_out)
134{
135 static const char head[] = "{\"schemaVersion\":2,\"manifests\":[";
136 char *acc = NULL;
137 size_t alen = 0, acap = 0;
138 char entry[1024];
139 int n, first = 1;
140
141 if (regc_buf_append(&acc, &alen, &acap, head, sizeof(head) - 1) != 0 ||
142 lay_index_keep_old(&acc, &alen, &acap, &first, old, olen, refname,
143 digest) != 0) {
144 free(acc);
145 return -1;
146 }
147 n = lay_index_new_entry(entry, sizeof(entry), first, refname, dg, mt,
148 size);
149 if (n < 0 || regc_buf_append(&acc, &alen, &acap, entry, (size_t) n) != 0) {
150 free(acc);
151 return -1;
152 }
153 *acc_out = acc;
154 *alen_out = alen;
155 return 0;
156}

158int
159brix_oci_layout_index_set(brix_oci_layout_t *l, const char *refname,
160 const char *digest, const char *mt, size_t size,
161 char *err, size_t errlen)
162{
163 brix_oci_digest_t d;
164 char dg[BRIX_OCI_DIGEST_STRLEN];
165 char *old = NULL, *acc = NULL;
166 size_t olen, alen = 0;
167 int rc;
168
169 /* The entry carries the canonical "<alg>:<hex>" the parser produced,
170 * not the caller's spelling and not a hardcoded prefix — the index is
171 * what another runtime reads the algorithm back out of. */
172 if (brix_oci_digest_parse(digest, strlen(digest), &d) != 0
173 || brix_oci_digest_format(&d, dg, sizeof(dg)) < 0)
174 {
175 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
176 "invalid digest \"%s\"", digest);
177 }
178 if (refname != NULL &&
179 brix_oci_tag_valid(refname, strlen(refname)) != 0) {
180 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
181 "invalid ref name \"%s\"", refname);
182 }
183 if (!lay_mt_safe(mt)) {
184 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
185 "unsafe media type");
186 }
187 rc = lay_index_load(l, &old, &olen, err, errlen);
188 if (rc != BRIX_OCI_REG_OK) {
189 return rc;
190 }
191 if (lay_index_rebuild(old, olen, refname, digest, dg, mt, size,
192 &acc, &alen) != 0) {
193 free(old);
194 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
195 "index rebuild failed");
196 }
197 free(old);
198 rc = layx_write_atomic(l->dir, "index.json", acc, alen, err, errlen);
199 free(acc);
200 return rc;
201}
