20/* Per-staged-write state: the delegated S3 write handle, plus the composed
21 * object path so a noreplace commit can HEAD the destination (P80.2). When the
22 * upload was opened under a per-user credential (P80.3) the triple is copied
23 * here — the caller's cred store does not outlive the open call, and the
24 * noreplace HEAD must present the same identity as the upload itself. */
25typedef struct {
26 sd_s3_file *s3;
27 char objpath[768];
28 int has_cred;
29 char ak[128];
30 char sk[256];
31 char region[64];
32 char session[2048]; /* STS X-Amz-Security-Token (phase-70 §5.5); "" static */
33} sd_remote_staged_state;

35/* Multipart part size for a staged upload of unknown final size (S3's 5 MiB
36 * minimum for non-final parts; 16 MiB balances request count vs. buffering). */
37#define SD_REMOTE_PART_SIZE (16 * 1024 * 1024)
38
39/* S3 caps a multipart upload at 10,000 parts, so a fixed part size caps the
40 * object: 10,000 x 16 MiB = 160 GB was this driver's silent ceiling. */
41#define SD_REMOTE_MAX_PARTS 10000
42
43/* sd_remote_part_size — phase-107 C5: derive a LEGAL multipart part size from
44 * the size the client declared up front (oss.asize / Content-Length / ALLO).
45 *
46 * max(SD_REMOTE_PART_SIZE, ceil(declared / 10,000)), rounded up to a MiB so
47 * parts stay aligned (rounding UP only lowers the part count, never breaks the
48 * 10,000-part cap). A declared 5 TB object gets ~525 MiB parts (~9,987 of
49 * them); undeclared (0) keeps the 16 MiB default and with it the historic
50 * 160 GB ceiling — declaring the size is precisely how a client lifts it. S3's
51 * own 5 GiB-per-part / 5 TiB-per-object limits govern beyond that; a
52 * declaration those cannot satisfy fails at the origin, not here. */
53int64_t
54sd_remote_part_size(off_t declared_size)
55{
56 int64_t need;
57
58 if (declared_size <= 0) {
59 return SD_REMOTE_PART_SIZE;
60 }
61 need = ((int64_t) declared_size + SD_REMOTE_MAX_PARTS - 1)
62 / SD_REMOTE_MAX_PARTS;
63 if (need <= SD_REMOTE_PART_SIZE) {
64 return SD_REMOTE_PART_SIZE;
65 }
66 return (need + (1 << 20) - 1) & ~(int64_t) ((1 << 20) - 1);
67}

69/* Shared staged-open body: start the upload, optionally signing with a
70 * per-user ak/sk/region override (NULL = the static service credential). The
71 * override triple is copied into the staged state so the noreplace commit's
72 * HEAD (P80.2) presents the same identity as the upload (P80.3). */
73static brix_sd_staged_t *
74sd_remote_staged_open_impl(brix_sd_instance_t *inst, const char *final_path,
75 off_t declared_size, const char *ak, const char *sk, const char *region,
76 const char *session, int *err_out)
77{
78 const brix_sd_remote_cfg_t *cfg = inst->state;
79 sd_s3_open_params p;
80 char objpath[768];
81 char errbuf[256];
82 sd_s3_file *s3;
83 sd_remote_staged_state *ss;
84 brix_sd_staged_t *h;
85
86 sd_remote_s3_key(cfg, final_path, objpath, sizeof(objpath));
87 sd_remote_s3_params(cfg, objpath, &p);
88 if (ak != NULL) { p.ak = ak; }
89 if (sk != NULL) { p.sk = sk; }
90 if (region != NULL) { p.region = region; }
91 if (session != NULL) { p.session_token = session; }
92
93 /* Declared final size (phase-107 C5): hand it to sd_s3 so a small object
94 * gets an exactly-sized single-PUT buffer and a large one starts multipart
95 * immediately with a part size that keeps the upload inside S3's
96 * 10,000-part cap. Unknown (-1): sd_s3 buffers a single PUT and lazily
97 * upgrades to multipart past the 16 MiB default (P80.2), so small objects
98 * cost one request while anything under 160 GB still works. */
99 s3 = sd_s3_open_write(&p, declared_size > 0 ? (int64_t) declared_size : -1,
100 sd_remote_part_size(declared_size),
101 errbuf, sizeof(errbuf));
102 if (s3 == NULL) {
103 if (err_out) { *err_out = EIO; }
104 return NULL;
105 }
106 ss = calloc(1, sizeof(*ss));
107 h = calloc(1, sizeof(*h));
108 if (ss == NULL || h == NULL) {
109 free(ss);
110 free(h);
111 sd_s3_abort(s3);
112 sd_s3_close(s3);
113 if (err_out) { *err_out = ENOMEM; }
114 return NULL;
115 }
116 ss->s3 = s3;
117 snprintf(ss->objpath, sizeof(ss->objpath), "%s", objpath);
118 if (ak != NULL && sk != NULL) {
119 ss->has_cred = 1;
120 snprintf(ss->ak, sizeof(ss->ak), "%s", ak);
121 snprintf(ss->sk, sizeof(ss->sk), "%s", sk);
122 snprintf(ss->region, sizeof(ss->region), "%s",
123 (region != NULL) ? region : "");
124 snprintf(ss->session, sizeof(ss->session), "%s",
125 (session != NULL) ? session : "");
126 }
127 h->inst = inst;
128 h->state = ss;
129 return h;
130}

183ngx_int_t
184sd_remote_staged_commit(brix_sd_staged_t *h, brix_sd_precond_t *pre)
185{
186 sd_remote_staged_state *ss = h->state;
187 char errbuf[256];
188 int rc;
189
190 /* Typed publish precondition (phase-107 C6): the ORIGIN decides, in the
191 * publish request itself. ABSENT arms If-None-Match: * and MATCH_ETAG arms
192 * If-Match on the final PUT / CompleteMPU — atomic at S3, so the pre-W7
193 * HEAD-then-PUT existence probe (racy against a concurrent external
194 * writer landing the object between the two requests) is gone entirely.
195 * MATCH_META is np on this driver: S3 has no size/mtime conditional and
196 * §3.5 forbids emulating one with a HEAD. */
197 if (pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE) {
198 char etag[128]; /* >= the client's cond_val cap; a longer
199 * tag is refused by the setter below */
200
201 switch (pre->kind) {
202 case BRIX_SD_PRECOND_ABSENT:
203 rc = sd_s3_set_publish_cond(ss->s3, "if-none-match", "*");
204 break;
205 case BRIX_SD_PRECOND_MATCH_ETAG:
206 if (pre->etag == NULL || pre->etag_len == 0
207 || pre->etag_len >= sizeof(etag))
208 {
209 errno = EINVAL;
210 return NGX_ERROR;
211 }
212 memcpy(etag, pre->etag, pre->etag_len);
213 etag[pre->etag_len] = '\0';
214 rc = sd_s3_set_publish_cond(ss->s3, "if-match", etag);
215 break;
216 default:
217 errno = ENOTSUP;
218 return NGX_ERROR;
219 }
220 if (rc != 0) {
221 errno = EINVAL;
222 return NGX_ERROR;
223 }
224 }
225
226 rc = sd_s3_commit(ss->s3, errbuf, sizeof(errbuf));
227 if (rc != 0) {
228 /* Failure contract: leave the staged handle intact — the caller's
229 * staged_abort discards the upload (sd_s3_abort) and frees
230 * ss->s3/ss/h. Freeing here too double-frees / uses-after-free that
231 * abort (stage_engine_move always aborts a failed commit): the reused
232 * sd_s3_file surfaced as free(put_buf==0x1). The origin's 412 arrives
233 * as ECANCELED (sd_s3_status_err); an ABSENT refusal retypes to the
234 * contract's EEXIST -> kXR_ItExists. Everything else flattens to EIO. */
235 if (errno == ECANCELED) {
236 if (pre != NULL) {
237 pre->atomic = 1; /* the origin decided — atomically — even
238 * when the answer is no (C6 advisory) */
239 }
240 if (brix_sd_precond_absent(pre)) {
241 errno = EEXIST;
242 }
243 } else {
244 errno = EIO;
245 }
246 return NGX_ERROR;
247 }
248 if (pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE) {
249 pre->atomic = 1; /* the origin evaluated it in the PUT */
250 }
251 sd_s3_close(ss->s3);
252 free(ss);
253 free(h);
254 return NGX_OK;
255}

268/* Shared unlink body: DELETE the object, optionally signing with a per-user
269 * ak/sk/region override (NULL = the instance's static service credential). A
270 * directory (is_dir, from rmdir) targets the zero-byte "path/" marker object
271 * (#4) — the VFS rmtree has already emptied its children — rather than a
272 * same-named file key. */
273static ngx_int_t
274sd_remote_unlink_impl(brix_sd_instance_t *inst, const char *path, int is_dir,
275 const char *ak, const char *sk, const char *region, const char *session)
276{
277 const brix_sd_remote_cfg_t *cfg = inst->state;
278 sd_s3_open_params p;
279 char objpath[768];
280 char errbuf[256];
281
282 if (is_dir) {
283 sd_remote_s3_dirkey(cfg, path, objpath, sizeof(objpath));
284 } else {
285 sd_remote_s3_key(cfg, path, objpath, sizeof(objpath));
286 }
287 sd_remote_s3_params(cfg, objpath, &p);
288 if (ak != NULL) { p.ak = ak; }
289 if (sk != NULL) { p.sk = sk; }
290 if (region != NULL) { p.region = region; }
291 if (session != NULL) { p.session_token = session; }
292
293 if (sd_s3_delete(&p, errbuf, sizeof(errbuf)) != 0) {
294 if (errno == 0) { errno = EIO; }
295 return NGX_ERROR;
296 }
297 return NGX_OK;

325/* ---- bulk delete (phase-107 C4: S3 DeleteObjects) -------------------------
326 *
327 * WHAT: unlink_many/_cred - one signed "POST /bucket?delete" removes up to
328 * 1,000 keys (sd_s3_delete_many), against the per-key loop's 1,000
329 * signed round trips. This driver is the item's whole motivation and the
330 * only one that advertises BRIX_SD_CAP_BULK_DELETE.
331 * WHY: Contract (sd_batch_types.h): paths arrive confined and non-directory,
332 * errs is per-key, and S3's own idempotency means an absent key reports
333 * 0 while AccessDenied stays on ITS key alone.
334 * HOW: The request target is the bare "/bucket"; each XML <Key> is the
335 * export-relative path minus its leading '/' - exactly the in-bucket
336 * remainder of what sd_remote_s3_key composes for the single DELETE.
337 * DeleteObjects is transactional at the transport: the origin attempts
338 * every key of an accepted POST, so done is n or (batch refused) 0.
339 */
340static ngx_int_t
341sd_remote_unlink_many_impl(brix_sd_instance_t *inst, brix_sd_unlink_batch_t *b,
342 const char *ak, const char *sk, const char *region, const char *session)
343{
344 const brix_sd_remote_cfg_t *cfg = inst->state;
345 sd_s3_open_params p;
346 const char **keys;
347 char bucket[300];
348 char errbuf[256];
349 size_t i;
350 int rc;
351
352 if (b->n > BRIX_SD_BULK_DELETE_WINDOW) {
353 errno = EINVAL;
354 return NGX_ERROR;
355 }
356 keys = malloc(b->n * sizeof(*keys));
357 if (keys == NULL) {
358 errno = ENOMEM;
359 return NGX_ERROR;
360 }
361 for (i = 0; i < b->n; i++) {
362 keys[i] = (b->paths[i][0] == '/') ? b->paths[i] + 1 : b->paths[i];
363 }
364 snprintf(bucket, sizeof(bucket), "/%s", cfg->bucket);
365 sd_remote_s3_params(cfg, bucket, &p);
366 sd_remote_params_cred(&p, ak, sk, region, session);
367
368 errno = 0;
369 rc = sd_s3_delete_many(&p, keys, b->n, b->errs, &b->done,
370 errbuf, sizeof(errbuf));
371 free(keys);
372 if (rc != 0) {
373 if (errno == 0) { errno = EIO; }
374 return NGX_ERROR;
375 }
376 return NGX_OK;
377}

405/* ---- server-side copy (S3 CopyObject, x-amz-copy-source) ----------------
406 *
407 * WHAT: Copy the export-relative `src` object to `dst` entirely within the S3
408 * origin — the bytes never traverse this host — signing with a per-user
409 * ak/sk/region/session override when given (NULL = the instance's static
410 * service credential).
411 * WHY: finding #4 — WebDAV COPY / xrdcp server-side copy over an s3:// backend
412 * previously fell through to ENOSYS. S3 exposes CopyObject natively, so
413 * this is a single signed PUT rather than a read-back/re-upload. It lives
414 * here, next to staged_open/unlink, because it is a MUTATION: it is the
415 * one slot on this driver that both reads and writes an object, and the
416 * identity it presents governs both halves at once.
417 * HOW: compose the "/bucket/src" copy-source and the "/bucket/dst" request
418 * target, call the shared sd_s3_copy primitive, then best-effort HEAD the
419 * destination for the copied byte count (mirrors the POSIX server_copy;
420 * 0 when the follow-up stat cannot confirm the size). The follow-up HEAD
421 * presents the SAME identity as the copy — the noreplace-commit rule from
422 * the staged path: a probe signed by anyone else answers about visibility
423 * the copying identity may not have.
424 */
425static ngx_int_t
426sd_remote_server_copy_impl(brix_sd_instance_t *inst, const char *src,
427 const char *dst, off_t *bytes_out, const brix_sd_cred_t *cred,
428 const char *ak, const char *sk, const char *region, const char *session)
429{
430 const brix_sd_remote_cfg_t *cfg = inst->state;
431 sd_s3_open_params p;
432 char srcpath[768];
433 char dstpath[768];
434 char errbuf[256];
435
436 sd_remote_s3_key(cfg, src, srcpath, sizeof(srcpath)); /* /bucket/src */
437 sd_remote_s3_key(cfg, dst, dstpath, sizeof(dstpath)); /* /bucket/dst */
438 sd_remote_s3_params(cfg, dstpath, &p); /* target = dst */
439 sd_remote_params_cred(&p, ak, sk, region, session);
440
441 errno = 0;
442 if (sd_s3_copy(&p, srcpath, errbuf, sizeof(errbuf)) != 0) {
443 if (errno == 0) { errno = EIO; }
444 return NGX_ERROR;
445 }
446 if (bytes_out != NULL) {
447 brix_sd_stat_t st;
448 ngx_int_t rc = (cred != NULL)
449 ? sd_remote_stat_cred(inst, dst, &st, cred)
450 : sd_remote_stat(inst, dst, &st);
451 *bytes_out = (rc == NGX_OK) ? st.size : 0;
452 }
453 return NGX_OK;
454}
