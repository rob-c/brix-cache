1/* sd_s3_sign_ext.c — extended SigV4 signer (arbitrary verb + extra headers).
2 *
3 * WHAT: SigV4 signing for the metadata/copy verbs: unlike sd_s3_sign() in
4 * sd_s3_sign.c (which signs a fixed GET/PUT shape), this signer takes a
5 * caller-supplied method, canonical query string and a sorted set of
6 * extra x-amz-* headers, so PUT-with-metadata and CopyObject can sign the
7 * headers they actually send.
8 *
9 * WHY: Split out of sd_s3_meta.c, which crossed the 600-line cap
10 * (coding-standards §1). Signing is a self-contained crypto stage with no
11 * knowledge of the metadata verbs that call it.
12 *
13 * HOW: prepare → canonical request → string-to-sign → derive key + sign →
14 * emit Authorization header, each a separate early-return step. The
15 * request-description types and the entry point are declared in
16 * sd_s3_internal.h; the working context stays private to this TU. */
17

42/*
43 * sd_s3_sign_ctx_t — file-local working state for the extended SigV4 signer.
44 * Carries the request identity (method / query string / extra headers), the
45 * fixed signed-header values (host, x-amz-date, payload hash) and every
46 * intermediate the pipeline stages hand to each other (sorted header set,
47 * canonical-request hash, credential scope, signed-header list, signature).
48 * One struct instead of 7+ loose parameters per helper.
49 */
50typedef struct {
51 const char *method; /* HTTP verb being signed */
52 const char *canon_qs; /* canonical query string ("" ok) */
53 const char *payload_hex; /* payload hash (UNSIGNED-PAYLOAD) */
54 const sd_s3_sign_hdr_t *extra; /* caller's additional headers */
55 size_t n_extra;
56 sd_s3_sign_hdr_t all[3 + 32]; /* fixed + extra, sorted for canon */
57 size_t n_all;
58 char host[300]; /* Host header value */
59 char amzdate[20]; /* x-amz-date (ISO8601 basic) */
60 char datestamp[12]; /* YYYYMMDD for the scope */
61 char enc_uri[2048]; /* URI-encoded object key */
62 char signed_list[2048]; /* "a;b;c" SignedHeaders */
63 char canon_hex[65]; /* SHA-256(canonical request) hex */
64 char scope[160]; /* date/region/s3/aws4_request */
65 char sighex[65]; /* final signature, hex */
66} sd_s3_sign_ctx_t;

68/*
69 * sd_s3_sign_prepare — stage 1: bind the request identity into the signing
70 * context. Formats host:port, stamps the current UTC time, URI-encodes the
71 * object key, then assembles the full signed-header set (fixed host /
72 * x-amz-content-sha256 / x-amz-date plus the caller's extras) sorted by name
73 * as SigV4 canonicalization requires. 0 / -1 (key too long to encode).
74 */
75static int
76sd_s3_sign_prepare(const sd_s3_file *f, sd_s3_sign_ctx_t *sx)
77{
78 size_t i;
79
80 brix_format_host_port(f->host, (uint16_t) f->port, sx->host,
81 sizeof(sx->host));
82 sd_s3_utc_now(sx->amzdate, sx->datestamp);
83 if (brix_http_urlencode((const unsigned char *) f->key, strlen(f->key),
84 sx->enc_uri, sizeof(sx->enc_uri), "/") < 0)
85 {
86 return -1;
87 }
88
89 sx->n_all = 0;
90 sx->all[sx->n_all].name = "host";
91 sx->all[sx->n_all++].value = sx->host;
92 sx->all[sx->n_all].name = "x-amz-content-sha256";
93 sx->all[sx->n_all++].value = sx->payload_hex;
94 sx->all[sx->n_all].name = "x-amz-date";
95 sx->all[sx->n_all++].value = sx->amzdate;
96 for (i = 0; i < sx->n_extra; i++) {
97 sx->all[sx->n_all++] = sx->extra[i];
98 }
99 qsort(sx->all, sx->n_all, sizeof(sx->all[0]), sd_s3_sign_hdr_cmp);
100 return 0;
101}

103/*
104 * sd_s3_canonical_request — stage 2: build the SigV4 canonical request
105 * (method \n uri \n qs \n sorted "name:value\n" headers \n signed-list \n
106 * payload-hash) and hash it into sx->canon_hex; also accumulates the
107 * ";"-joined SignedHeaders list the Authorization line reuses. The canonical
108 * buffer itself never leaves this frame — only its SHA-256 matters. 0 / -1
109 * on truncation.
110 */
111static int
112sd_s3_canonical_request(sd_s3_sign_ctx_t *sx)
113{
114 char canon[16384];
115 size_t i, sl = 0;
116 int off, cn;
117
118 off = snprintf(canon, sizeof(canon), "%s\n%s\n%s\n", sx->method,
119 sx->enc_uri, sx->canon_qs);
120 if (off < 0 || (size_t) off >= sizeof(canon)) {
121 return -1;
122 }
123 for (i = 0; i < sx->n_all; i++) {
124 cn = snprintf(canon + off, sizeof(canon) - (size_t) off, "%s:%s\n",
125 sx->all[i].name, sx->all[i].value);
126 if (cn < 0 || (size_t) off + (size_t) cn >= sizeof(canon)) {
127 return -1;
128 }
129 off += cn;
130 cn = snprintf(sx->signed_list + sl, sizeof(sx->signed_list) - sl,
131 "%s%s", (i ? ";" : ""), sx->all[i].name);
132 if (cn < 0 || sl + (size_t) cn >= sizeof(sx->signed_list)) {
133 return -1;
134 }
135 sl += (size_t) cn;
136 }
137 cn = snprintf(canon + off, sizeof(canon) - (size_t) off, "\n%s\n%s",
138 sx->signed_list, sx->payload_hex);
139 if (cn < 0 || (size_t) off + (size_t) cn >= sizeof(canon)) {
140 return -1;
141 }
142 sd_s3_sha256_hex(canon, strlen(canon), sx->canon_hex);
143 return 0;
144}

146/*
147 * sd_s3_string_to_sign — stage 3a: derive the credential scope
148 * (date/region/s3/aws4_request → sx->scope) and format the SigV4
149 * string-to-sign ("AWS4-HMAC-SHA256\n date \n scope \n canon-hash") into the
150 * caller's buffer. 0 / -1 on truncation.
151 */
152static int
153sd_s3_string_to_sign(const sd_s3_file *f, sd_s3_sign_ctx_t *sx,
154 char *sts, size_t stscap)
155{
156 int cn;
157
158 cn = snprintf(sx->scope, sizeof(sx->scope), "%s/%s/s3/aws4_request",
159 sx->datestamp, f->region);
160 if (cn < 0 || (size_t) cn >= sizeof(sx->scope)) {
161 return -1;
162 }
163 cn = snprintf(sts, stscap, "AWS4-HMAC-SHA256\n%s\n%s\n%s", sx->amzdate,
164 sx->scope, sx->canon_hex);
165 if (cn < 0 || (size_t) cn >= stscap) {
166 return -1;
167 }
168 return 0;
169}

171/*
172 * sd_s3_derive_key_and_sign — stage 3b: run the SigV4 key-derivation chain
173 * (secret → date → region → service → aws4_request) and HMAC the
174 * string-to-sign with the derived key, hex-encoding the signature into
175 * sx->sighex. 0 / -1 if any crypto primitive fails.
176 */
177static int
178sd_s3_derive_key_and_sign(const sd_s3_file *f, sd_s3_sign_ctx_t *sx,
179 const char *sts)
180{
181 uint8_t k4[32], sig[32];
182
183 if (!brix_sigv4_signing_key((const uint8_t *) f->sk, strlen(f->sk),
184 sx->datestamp, f->region, "s3", k4)
185 || !brix_hmac_sha256(k4, 32, (const uint8_t *) sts, strlen(sts), sig))
186 {
187 return -1;
188 }
189 brix_hex_encode(sig, 32, sx->sighex);
190 return 0;
191}

193/*
194 * sd_s3_emit_auth_header — stage 4: write the complete request header block
195 * the transport sends: x-amz-date, x-amz-content-sha256, every extra header
196 * verbatim (in caller order — the sorted set was for canonicalization only),
197 * then the Authorization line carrying credential scope, SignedHeaders and
198 * the signature. 0 / -1 on truncation.
199 */
200static int
201sd_s3_emit_auth_header(const sd_s3_file *f, const sd_s3_sign_ctx_t *sx,
202 char *hdrs, size_t hdrsz)
203{
204 size_t i;
205 int off, cn;
206
207 off = snprintf(hdrs, hdrsz,
208 "x-amz-date: %s\r\nx-amz-content-sha256: %s\r\n",
209 sx->amzdate, sx->payload_hex);
210 if (off < 0 || (size_t) off >= hdrsz) {
211 return -1;
212 }
213 for (i = 0; i < sx->n_extra; i++) {
214 cn = snprintf(hdrs + off, hdrsz - (size_t) off, "%s: %s\r\n",
215 sx->extra[i].name, sx->extra[i].value);
216 if (cn < 0 || (size_t) off + (size_t) cn >= hdrsz) {
217 return -1;
218 }
219 off += cn;
220 }
221 cn = snprintf(hdrs + off, hdrsz - (size_t) off,
222 "Authorization: AWS4-HMAC-SHA256 Credential=%s/%s, "
223 "SignedHeaders=%s, Signature=%s\r\n",
224 f->ak, sx->scope, sx->signed_list, sx->sighex);
225 if (cn < 0 || (size_t) off + (size_t) cn >= hdrsz) {
226 return -1;
227 }
228 return 0;
229}

231/*
232 * sd_s3_sign_ext — SigV4 over an arbitrary set of additional x-amz-* headers (on
233 * top of the fixed host / x-amz-content-sha256 / x-amz-date). Used by set-meta,
234 * whose copy-onto-self carries x-amz-copy-source, x-amz-metadata-directive and
235 * the x-amz-meta-* pairs — all of which AWS requires in the signed header set.
236 * Emits the complete request header block to send (the extra headers + the
237 * Authorization line). 0 / -1. Reuses every kernel sd_s3_sign uses. Pure
238 * pipeline: prepare → canonical request → string-to-sign → sign → emit.
239 */
240int
241sd_s3_sign_ext(const sd_s3_file *f, const sd_s3_sign_req_t *req,
242 char *hdrs, size_t hdrsz)
243{
244 sd_s3_sign_ctx_t sx;
245 char sts[640];
246
247 if (req->n_extra > 32) {
248 return -1;
249 }
250 sx.method = req->method;
251 sx.canon_qs = (req->canon_qs != NULL) ? req->canon_qs : "";
252 sx.payload_hex = "UNSIGNED-PAYLOAD";
253 sx.extra = req->extra;
254 sx.n_extra = req->n_extra;
255
256 if (sd_s3_sign_prepare(f, &sx) != 0
257 || sd_s3_canonical_request(&sx) != 0
258 || sd_s3_string_to_sign(f, &sx, sts, sizeof(sts)) != 0
259 || sd_s3_derive_key_and_sign(f, &sx, sts) != 0)
260 {
261 return -1;
262 }
263 return sd_s3_emit_auth_header(f, &sx, hdrs, hdrsz);
264}
