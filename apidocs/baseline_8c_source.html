18/* WHAT: Load a regular no-follow baseline under a hard size bound. WHY: A user
19 * supplied baseline cannot cause unbounded allocation or device reads.
20 * HOW: open/fstat/read exact bytes and append one private NUL terminator. */
21static char *
22baseline_load(const char *path, char *error, size_t error_size)
23{
24 struct stat info;
25 char *body;
26 size_t done = 0;
27 int fd = open(path, O_RDONLY | O_NOFOLLOW);
28
29 if (fd < 0 || fstat(fd, &info) != 0 || !S_ISREG(info.st_mode)
30 || info.st_size < 0 || (uint64_t) info.st_size > BASELINE_MAX) {
31 if (fd >= 0) close(fd);
32 snprintf(error, error_size, "not a bounded regular report file");
33 return NULL;
34 }
35 body = malloc((size_t) info.st_size + 1U);
36 if (body == NULL) {
37 close(fd);
38 snprintf(error, error_size, "baseline allocation failed");
39 return NULL;
40 }
41 while (done < (size_t) info.st_size) {
42 ssize_t count = read(fd, body + done, (size_t) info.st_size - done);
43
44 if (count < 0 && errno == EINTR) continue;
45 if (count <= 0) {
46 free(body); close(fd);
47 snprintf(error, error_size, "baseline read failed");
48 return NULL;
49 }
50 done += (size_t) count;
51 }
52 close(fd);
53 body[done] = '\0';
54 return body;
55}

57/* WHAT: Find one canonical result's verdict. WHY: Baselines compare stable ids,
58 * never array positions. HOW: Locate generated id key then bounded verdict key. */
59static const char *
60baseline_verdict(const char *body, const char *id, char out[24])
61{
62 char needle[64];
63 const char *row;
64 const char *end;
65 const char *value;
66 size_t length;
67
68 if ((size_t) snprintf(needle, sizeof(needle), "\"id\":\"%s\"", id)
69 >= sizeof(needle)) return NULL;
70 row = strstr(body, needle);
71 if (row == NULL) return NULL;
72 end = strchr(row, '}');
73 value = strstr(row, "\"verdict\":\"");
74 if (end == NULL || value == NULL || value >= end) return NULL;
75 value += strlen("\"verdict\":\"");
76 length = strcspn(value, "\"");
77 if (length == 0 || length >= 24 || value + length > end) return NULL;
78 memcpy(out, value, length); out[length] = '\0';
