1/*
2 * xrdhttp.h — XrdHttp protocol extension support for the nginx-xrootd WebDAV module.
3 *
4 * WHAT: Public API for the XrdHttp compatibility layer. Covers:
5 * - Detection of XrdHttp clients via X-Xrootd-Proto header
6 * - Extraction of ?xrd.* and ?tpc.* query parameters per the XrdHttp URL dialect
7 * - Response header injection: X-Xrootd-Requuid, X-Xrootd-Status, X-Xrootd-Wait/Retry
8 * - Multipart/byteranges GET for efficient vector reads (kXR_readv over HTTP)
9 * - URI-based TPC via ?tpc.src= / ?tpc.dst= query parameters
10 * - ?xrd.stats XML statistics endpoint
11 *
12 * WHY: The XRootD reference implementation (XrdHttp) uses these extensions so that
13 * XRootD-aware HTTP clients (xrdcp --prefer-xrdhttp, ROOT TFile, davix with xrd hints)
14 * get richer error diagnostics, server-side checksum hints, redirect loop prevention,
15 * and high-performance vector reads. Without this layer those clients fall back to
16 * slower code paths or emit spurious errors.
17 *
18 * USAGE: Call xrdhttp_parse_request() once per request (in dispatch, after auth).
19 * Call xrdhttp_add_response_headers() before returning from any method handler.
20 * For multi-range GETs, check xrdhttp_request_is_multirange() and route to
21 * xrdhttp_handle_multipart_get() instead of the standard single-range path.
22 */
23

48/*
49 * Per-request XrdHttp context. Allocated lazily in xrdhttp_parse_request()
50 * from r->pool on first use. Retrieved with xrdhttp_get_ctx().
51 */
52typedef struct {
53 /* Client identification (from X-Xrootd-Proto header). */
54 unsigned is_xrdhttp:1; /* 1 = client sent X-Xrootd-Proto */
55 unsigned headers_injected:1; /* 1 = xrdhttp response headers already added */
56 char proto_version[XRDHTTP_PROTO_MAX]; /* e.g. "5.2", NUL-term */
57
58 /* Parsed once at the WebDAV access boundary and shared by auth, redirect,
59 * XrdHttp, and TPC consumers. */
60 brix_http_query_t query;
61
62 /* ?xrd.* query parameters — logged and used for routing/metrics. */
63 char clnt_uuid[XRDHTTP_UUID_MAX]; /* xrd.clnt.uuid */
64 char clnt_app[XRDHTTP_APP_MAX]; /* xrd.clnt.app */
65 char want_cksum[XRDHTTP_CKSUM_MAX]; /* xrd.want.cksum alg */
66 char opaque[XRDHTTP_OPAQUE_MAX]; /* xrd.opaque blob */
67
68 /* ?tpc.* parameters — alternate TPC signaling (besides headers). */
69 char tpc_src[XRDHTTP_TPC_URL_MAX]; /* tpc.src pull URL */
70 char tpc_dst[XRDHTTP_TPC_URL_MAX]; /* tpc.dst push URL */
71 char tpc_key[XRDHTTP_TPC_KEY_MAX]; /* tpc.key security cookie */
72 char tpc_token[XRDHTTP_TPC_KEY_MAX]; /* X-Xrootd-Tpc-Token */
73
74 /* Request UUID echo — sent back in every response as X-Xrootd-Requuid. */
75 char requuid[XRDHTTP_UUID_MAX]; /* from X-Xrootd-Requuid or generated */
76
77 /* Wait/retry — populated by handlers that need back-pressure. */
78 int wait_seconds; /* emit X-Xrootd-Wait: <N> if > 0 */
79 int retry_seconds; /* emit X-Xrootd-Retry: <N> if > 0 */
80
81 /* Phase 21 Step B — streaming Digest over the response body. Set by the
82 * GET path when the client sent "Want-Digest: adler32" and the digest was
83 * not already produced from the open fd. The body filter folds each output
84 * buffer into `adler` and emits "Digest: adler32=<hex>" as a trailer. */
85 unsigned compute_digest:1; /* 1 = accumulate adler32 over the body */
86 unsigned digest_emitted:1; /* 1 = trailer already queued */
87 uint32_t adler; /* running adler32 (seed 1) */
88} xrdhttp_req_ctx_t;

100/*
101 * Parse all XrdHttp signals from the incoming request:
102 * - X-Xrootd-Proto → is_xrdhttp + proto_version
103 * - X-Xrootd-Requuid → requuid (echoed in response)
104 * - X-Xrootd-Tpc-Token → tpc_token
105 * - Query string → clnt_uuid, clnt_app, want_cksum, opaque, tpc_src, tpc_dst, tpc_key
106 * Logs clnt_app and clnt_uuid at debug level for auditing.
107 * Always succeeds (any unparseable value is silently ignored).
108 */
109ngx_int_t xrdhttp_parse_request(ngx_http_request_t *r);

111/* ---- response header injection ---- */
112
113/*
114 * Inject XrdHttp response headers into r->headers_out before the caller
115 * sends headers:
116 * - X-Xrootd-Requuid: <requuid> (always if is_xrdhttp)
117 * - X-Xrootd-Status: <xrd_code> (for non-2xx status; 0 = success)
118 * - X-Xrootd-Wait: <N> (if ctx->wait_seconds > 0)
119 * - X-Xrootd-Retry: <N> (if ctx->retry_seconds > 0)
120 * Safe to call even when is_xrdhttp==0 — emits nothing in that case.
121 * http_status is the HTTP status code about to be sent (for kXR mapping).
122 */
123ngx_int_t xrdhttp_add_response_headers(ngx_http_request_t *r,
124 ngx_int_t http_status);

126/* NOTE: the XrdHttp redirect dialect (xrdhttp_send_redirect + the
127 * X-Xrootd-Redir-Host/Port emitters) was removed in phase-95 — it had zero call

129 * needs a mesh-selection call site and a signed-CGI handoff; the root:// plane
130 * does its own redirects via brix_send_redirect (response/control.c). */
