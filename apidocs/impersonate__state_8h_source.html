16/*
17 * brix_imp_state_t — runtime state for impersonation broker.
18 *
19 * Fields:
20 * broker_allow_uid: UID allowed to connect to broker (0 = disabled)
21 * base_uid: Base UID for impersonation operations
22 * base_gid: Base GID for impersonation operations
23 * base_groups: Supplementary group array
24 * base_ngroups: Count of supplementary groups
25 * self_uid: Service user's own UID (for restore operations)
26 * initialized: Flag indicating state has been initialized
27 */
28typedef struct {
29 uid_t broker_allow_uid;
30 uid_t base_uid;
31 gid_t base_gid;
32 gid_t base_groups[BRIX_IDMAP_MAXGROUPS];
33 int base_ngroups;
34 uid_t self_uid;

36} brix_imp_state_t;
