1/*
2 * cta_service.c — the "cta" SSI service. See cta_service.h.
3 *
4 * Two phases (the defer-or-respond pattern):
5 * submit — decode the cta.xrd.Request, queue it, and defer (so a long
6 * archive/retrieve answers asynchronously with progress alerts).
7 * query/error answer inline.
8 * completion — run the executor on the stashed queue entry (alerts pushed live),
9 * then answer with a cta.xrd.Response.
10 */

71int
72brix_ssi_cta_process(const unsigned char *req, size_t req_len,
73 brix_ssi_responder_t *r)
74{
75 void **slot = r->svc_slot != NULL ? r->svc_slot(r) : NULL;
76 cta_req_t *e = slot != NULL ? (cta_req_t *) *slot : NULL;
77
78 if (e == NULL) {
79 /* ---- submit phase ---- */
80 cta_request_t creq;
81
82 if (cta_pb_decode_request(req, req_len, &creq) != 0) {
83 cta_respond(r, CTA_RSP_ERR_PROTOBUF, "malformed CTA request", 0);
84 return 0;
85 }
86 if (creq.op == CTA_OP_QUERY) {
87 char msg[64];
88 snprintf(msg, sizeof(msg), "%d active request(s)",
89 brix_cta_shm_active_count());
90 cta_respond(r, CTA_RSP_SUCCESS, msg, 0);
91 return 0;
92 }
93 if (creq.op == CTA_OP_UNKNOWN) {
94 cta_respond(r, CTA_RSP_ERR_USER, "unsupported workflow event", 0);
95 return 0;
96 }
97 if (brix_cta_shm_queue() == NULL) {
98 /* No zone: refuse. Falling back to a private per-worker queue is
99 * exactly the defect W8.6 removes — it would hand out ids that
100 * collide with another worker's. */
101 cta_respond(r, CTA_RSP_ERR_CTA, "CTA queue unavailable", 0);
102 return 0;
103 }
104 e = brix_cta_shm_submit(&creq, creq.owner_user);
105 if (e == NULL) {
106 cta_respond(r, CTA_RSP_ERR_CTA, "request queue full", 0);
107 return 0;
108 }
109 if (slot != NULL) {
110 *slot = e;
111 }
112 if (r->defer != NULL && r->defer(r) == 0) {
113 return 0; /* deferred — the executor runs at completion */
114 }
115 /* defer unavailable: fall through and run inline */
116 }
117
118 /* ---- completion phase ---- */
119 {
120 /* The executor transitions through brix_cta_shm_transition, which takes
121 * the zone lock once per transition — never for the length of an
122 * archive or retrieve. */
123 cta_progress_t prog = { brix_cta_shm_queue(), brix_cta_shm_transition,
124 cta_prog_alert, r };
125 int rc = cta_exec_run(cta_exec_vtbl(), e, &prog);
126 cta_respond(r, rc == 0 ? CTA_RSP_SUCCESS : CTA_RSP_ERR_CTA,
127 rc == 0 ? "request completed" : "request failed",
128 e->req.archive_id);
129 }
130 return 0;
131}
