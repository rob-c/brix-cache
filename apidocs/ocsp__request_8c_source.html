1/*
2 * OCSP request/response — build the query, POST it, and verify the reply.
3 *
4 * WHAT: The OCSP protocol crypto proper. do_ocsp_request() builds an
5 * OCSP_REQUEST (cert id + anti-replay nonce), opens/connects the responder BIO
6 * via the transport layer, bounds every network phase under
7 * BRIX_OCSP_TIMEOUT_SECS, POSTs the request, and returns the parsed
8 * OCSP_RESPONSE. check_ocsp_response() verifies that response — overall status,
9 * responder signature against a trust store, nonce match — and maps the
10 * certificate status to GOOD(0) / REVOKED(-1) / UNKNOWN(1).
11 *
12 * WHY: Split out of ocsp.c under the phase-79 file-size guard. This file is the
13 * security-critical heart of the revocation check: the nonce guards against
14 * replay, the signature verification proves the response came from a trusted
15 * responder, and the status mapping is the actual revocation decision. Keeping
16 * it separate from the URL/BIO transport (ocsp_transport.c) and the public entry
17 * points (ocsp.c) keeps each concern reviewable in isolation.
18 *
19 * HOW: OCSP_request_add0_id() takes ownership of `id` on success (the caller
20 * must not free it after a successful add). do_ocsp_request() owns the BIO
21 * teardown (ocsp_conn_free, cbio then ssl_ctx) and the request free on every
22 * path. check_ocsp_response() frees the BASICRESP on every path; a REVOKED
23 * status is never overridden. Shared structs/decls live in ocsp_internal.h.
24 */

34/*
35 * ocsp_build_request — allocate an OCSP request for cert id `id` with a nonce.
36 *
37 * WHAT: Returns a new OCSP_REQUEST carrying a COPY of `id` and a fresh nonce,
38 * or NULL on allocation failure. Ownership of `id` stays with the caller.
39 * WHY: Splits request construction out of the network path so do_ocsp_request()
40 * reads as build → connect → send.
41 * HOW: OCSP_request_add0_id() takes ownership of what it is given, but `id`
42 * belongs to brix_ocsp_check_cert() / brix_ocsp_staple_fetch() for the whole
43 * AIA loop: they build it once, query every responder URL with it, and free it
44 * afterwards. Handing the request the caller's `id` therefore made
45 * OCSP_REQUEST_free() free it out from under them — a use-after-free on a
46 * second AIA URL and a double free on the caller's OCSP_CERTID_free(), which
47 * segfaulted the worker on EVERY completed OCSP round trip. So add0 gets a
48 * dup and the caller's pointer is never transferred. The nonce
49 * (OCSP_request_add1_nonce) guards against replay. add0 does not take
50 * ownership when it fails, so the copy is freed on that path along with the
51 * half-built request.
52 */
53static OCSP_REQUEST *
54ocsp_build_request(OCSP_CERTID *id)
55{
56 OCSP_REQUEST *req = OCSP_REQUEST_new();
57 OCSP_CERTID *copy;
58
59 if (req == NULL) {
60 return NULL;
61 }
62 copy = OCSP_CERTID_dup(id);
63 if (copy == NULL) {
64 OCSP_REQUEST_free(req);
65 return NULL;
66 }
67 if (OCSP_request_add0_id(req, copy) == NULL) {
68 OCSP_CERTID_free(copy);
69 OCSP_REQUEST_free(req);
70 return NULL;
71 }
72 /* Add a nonce to prevent replay attacks */
73 OCSP_request_add1_nonce(req, NULL, -1);
74 return req;
75}

77/*
78 * do_ocsp_request — build request, POST to responder, return parsed response.
79 *
80 * The caller must call OCSP_RESPONSE_free() on the returned pointer.
81 * Returns NULL on any network or protocol error.
82 */
83/* WHAT: resolve the responder host through the brix DNS driver and open +
84 * connect a BIO to the first candidate that answers.
85 * WHY: BIO_set_conn_hostname() with a name would run getaddrinfo() on the
86 * event loop (I-DNS-1) — a second resolver with its own idea of
87 * resolv.conf. The BIO only ever sees a numeric "ip:port" now, so
88 * OpenSSL's resolver is never entered; the TLS leg still verifies the
89 * NAME (ocsp_open_bio sets SNI + SSL_set1_host from u->host).
90 * HOW: brix_dns_resolve_sync(): literal → per-worker cache → libc (the
91 * nginx-resolver bridge declines on the loop thread, which is where this
92 * blocking exchange runs by design). Each candidate is formatted with
93 * ngx_sock_ntop() ("[v6]:port") and tried in answer order under the
94 * E1 connect deadline. On success *conn is open + connected and
95 * hostport holds the numeric peer. */
96static int
97ocsp_connect_responder(ngx_log_t *log, const ocsp_url_t *u,
98 const brix_dns_policy_t *dns, ocsp_conn_t *conn, char *hostport,
99 size_t hostport_sz)
100{
101 brix_dns_addr_t addrs[BRIX_DNS_MAX_ADDRS];
102 char reason[BRIX_DNS_ERROR_LEN];
103 ngx_uint_t i, n;
104 size_t len;
105
106 n = brix_dns_resolve_sync(dns, u->host, (in_port_t) u->port, BRIX_AF_AUTO,
107 SOCK_STREAM, addrs, BRIX_DNS_MAX_ADDRS, reason,
108 sizeof(reason));
109 if (n == 0) {
110 ngx_log_error(NGX_LOG_WARN, log, 0,
111 "brix_ocsp: cannot resolve responder host \"%s\": %s",
112 u->host, reason);
113 return -1;
114 }
115 for (i = 0; i < n; i++) {
116 len = ngx_sock_ntop((struct sockaddr *) &addrs[i].ss, addrs[i].len,
117 (u_char *) hostport, hostport_sz - 1, 1);
118 hostport[len] = '\0';
119 if (len == 0 || ocsp_open_bio(log, u, hostport, conn) != 0) {
120 continue;
121 }
122 /* E1: bound the connect with a deadline (a black-holed responder must
123 * not freeze the worker for the kernel TCP timeout). */
124 if (ocsp_connect_bio(log, conn, hostport) == 0) {
125 return 0;
126 }
127 ocsp_conn_free(conn);
128 }
129 return -1;
130}

132/*
133 * HOW: Multi-stage OCSP request and response handling:
134 *
135 * 1. Parse OCSP URL
136 * - parse_ocsp_url() → ocsp_url_t struct
137 * - Extracts scheme, host, port, path components
138 *
139 * 2. Build OCSP_REQUEST
140 * - ocsp_build_request(id): adds certificate ID + nonce
141 *
142 * 3. Resolve responder via phase-116 DNS driver
143 * - brix_dns_resolve() with policy enforcement
144 * - Returns list of candidate addresses
145 *
146 * 4. Open + connect BIO per candidate via ocsp_connect_responder()
147 * - ocsp_open_bio(): plain TCP or verifying TLS with SNI
148 * - ocsp_connect_bio(): deadline-bounded connection
149 *
150 * 5. For HTTPS: bound handshake+verify under BRIX_OCSP_TIMEOUT_SECS
151 * - ocsp_tls_handshake(): enforces timeout
152 *
153 * 6. Send request and read reply via OCSP_REQ_CTX
154 * - Response length capped at OCSP_MAX_RESPONSE_BYTES
155 * - A-6/T2 security: untrusted responder must not
156 * stream unbounded body (DoS prevention)
157 *
158 * 7. Tear down connection
159 * - Free cbio (BIO object)
160 * - Free ssl_ctx (SSL context if TLS used)
161 * - Free OCSP_REQUEST
162 *
163 * Returns: OCSP_RESPONSE* on success, NULL on:
164 * - Network failure (DNS, connection, timeout)
165 * - Protocol failure (malformed response)
166 * - Response exceeds OCSP_MAX_RESPONSE_BYTES cap
167 */
168OCSP_RESPONSE *
169do_ocsp_request(ngx_log_t *log, const brix_dns_policy_t *dns, const char *url,
170 X509 *leaf, X509 *issuer, OCSP_CERTID *id, OCSP_REQUEST **req_out)
171{
172 ocsp_url_t u;
173 char hostport[320];
174 OCSP_REQUEST *req = NULL;
175 OCSP_RESPONSE *resp = NULL;
176 OCSP_REQ_CTX *rctx;
177 int rc;
178 ocsp_conn_t conn;
179
180 *req_out = NULL;
181
182 if (parse_ocsp_url(url, &u) != 0) {
183 ngx_log_error(NGX_LOG_WARN, log, 0,
184 "brix_ocsp: cannot parse URL \"%s\"", url);
185 return NULL;
186 }
187
188 req = ocsp_build_request(id);
189 if (req == NULL) {
190 return NULL;
191 }
192
193 /* phase-116: resolve through the brix driver and connect to the first
194 * answering candidate; hostport becomes the numeric peer ("[v6]:port"). */
195 if (ocsp_connect_responder(log, &u, dns, &conn, hostport,
196 sizeof(hostport)) != 0)
197 {
198 OCSP_REQUEST_free(req);
199 return NULL;
200 }
201
202 /* E1: bound the TLS handshake + request/response read/write phases. */
203 ocsp_set_io_timeouts(conn.cbio, BRIX_OCSP_TIMEOUT_SECS);
204
205 if (u.use_ssl
206 && ocsp_tls_handshake(log, &conn, u.host, hostport) != 0)
207 {
208 ocsp_conn_free(&conn);
209 OCSP_REQUEST_free(req);
210 return NULL;
211 }
212
213 /* A2/T2: bound the response body. The one-shot sendreq_bio helper reads
214 * under only OpenSSL's internal default cap; drive the request context
215 * ourselves so we pin the ceiling at OCSP_MAX_RESPONSE_BYTES — an untrusted
216 * (and usually
217 * plaintext/MITM-able) responder must not stream an unbounded body into the
218 * worker. Blocking BIO + SO_RCVTIMEO bound the wait; the -1 retry arm
219 * covers a spurious BIO_should_retry. */
220 rctx = OCSP_sendreq_new(conn.cbio, u.path, req, -1);
221 if (rctx == NULL) {
222 ocsp_conn_free(&conn);
223 OCSP_REQUEST_free(req);
224 ngx_log_error(NGX_LOG_WARN, log, 0,
225 "brix_ocsp: cannot build request context for \"%s\"", url);
226 return NULL;
227 }
228 OCSP_set_max_response_length(rctx, OCSP_MAX_RESPONSE_BYTES);
229
230 do {
231 rc = OCSP_sendreq_nbio(&resp, rctx);
232 } while (rc == -1 && BIO_should_retry(conn.cbio));
233
234 OCSP_REQ_CTX_free(rctx);
235 ocsp_conn_free(&conn);
236
237 if (rc <= 0 || resp == NULL) {
238 /* rc == 0 also fires when the reply exceeded OCSP_MAX_RESPONSE_BYTES:
239 * OpenSSL aborts the read at the cap and returns failure. */
240 ngx_log_error(NGX_LOG_WARN, log, 0,
241 "brix_ocsp: no valid response from \"%s\" (capped at %d bytes)",
242 url, OCSP_MAX_RESPONSE_BYTES);
243 if (resp != NULL) {
244 OCSP_RESPONSE_free(resp);
245 resp = NULL;
246 }
247 OCSP_REQUEST_free(req);
248 return NULL;
249 }
250
251 /* Success: hand the request (with its nonce) to the caller for the nonce
252 * match; the caller owns it now and frees it after check_ocsp_response(). */
253 *req_out = req;
254 return resp;
255}

257/*
258 * check_ocsp_response — verify the OCSP response and return the cert status.
259 *
260 * Returns:
261 * 0 — certificate is GOOD
262 * -1 — certificate is REVOKED or response is invalid
263 * 1 — status is UNKNOWN
264 */
265/*
266 * HOW: Five-step OCSP response validation:
267 *
268 * 1. Check response status
269 * - OCSP_response_status(resp)
270 * - Returns -1 if not OCSP_RESPONSE_STATUS_SUCCESSFUL
271 *
272 * 2. Extract BASICRESP
273 * - OCSP_response_get1_basic() → OCSP_BASICRESP*
274 *
275 * 3. Optionally verify response signature
276 * - OCSP_basic_verify(basicresp, NULL, trust_store, flags)
277 * - NULL trust_store: uses OpenSSL defaults
278 * - Returns: 1=valid, 0=invalid
279 *
280 * 4. Check nonce match (if original request available)
281 * - OCSP_check_nonce(req, basicresp)
282 * - Mismatch: failure (return -1)
283 * - Missing nonce: warning only, continue
284 *
285 * 5. Find certificate status and map to return code
286 * - OCSP_resp_find_status(basicresp, id, &status, &reason)
287 * - Switch on V_OCSP_CERTSTATUS_* enum:
288 * * V_OCSP_CERTSTATUS_GOOD → return 0
289 * * V_OCSP_CERTSTATUS_REVOKED → return -1
290 * * V_OCSP_CERTSTATUS_UNKNOWN → return 1
291 */
292int
293check_ocsp_response(ngx_log_t *log, OCSP_RESPONSE *resp,
294 X509_STORE *store, OCSP_CERTID *id, OCSP_REQUEST *req_for_nonce,
295 int require_nonce)
296{
297 OCSP_BASICRESP *bresp;
298 int status, reason;
299 ASN1_GENERALIZEDTIME *rev = NULL, *thisupd = NULL, *nextupd = NULL;
300 int rc = -1;
301
302 if (OCSP_response_status(resp) != OCSP_RESPONSE_STATUS_SUCCESSFUL) {
303 ngx_log_error(NGX_LOG_WARN, log, 0,
304 "brix_ocsp: OCSP response status = %d (not successful)",
305 OCSP_response_status(resp));
306 return -1;
307 }
308
309 bresp = OCSP_response_get1_basic(resp);
310 if (bresp == NULL) {
311 return -1;
312 }
313
314 /* Verify the response signature against the trust store */
315 if (store != NULL && OCSP_basic_verify(bresp, NULL, store, 0) <= 0) {
316 ngx_log_error(NGX_LOG_WARN, log, 0,
317 "brix_ocsp: OCSP response signature verification failed");
318 OCSP_BASICRESP_free(bresp);
319 return -1;
320 }
321
322 /* Check the nonce if we have the original request */
323 if (req_for_nonce != NULL) {
324 int nonce_rc = OCSP_check_nonce(req_for_nonce, bresp);
325 if (nonce_rc < 0) {
326 /* Nonce present in request but missing in response. A missing nonce
327 * lets an on-path attacker replay a captured (still-signed) GOOD
328 * response. Under brix_ocsp_require_nonce this is a hard failure
329 * (A-6 item 2); otherwise warn only — most CA responders serve
330 * pre-signed, nonce-less responses, so hard-fail is opt-in. */
331 if (require_nonce) {
332 ngx_log_error(NGX_LOG_WARN, log, 0,
333 "brix_ocsp: nonce missing in OCSP response and "
334 "brix_ocsp_require_nonce is on — denying (replay guard)");
335 OCSP_BASICRESP_free(bresp);
336 return -1;
337 }
338 ngx_log_error(NGX_LOG_WARN, log, 0,
339 "brix_ocsp: nonce missing in OCSP response");
340 } else if (nonce_rc == 0) {
341 ngx_log_error(NGX_LOG_WARN, log, 0,
342 "brix_ocsp: OCSP response nonce mismatch");
343 OCSP_BASICRESP_free(bresp);
344 return -1;
345 }
346 }
347
348 if (!OCSP_resp_find_status(bresp, id, &status, &reason,
349 &rev, &thisupd, &nextupd))
350 {
351 ngx_log_error(NGX_LOG_WARN, log, 0,
352 "brix_ocsp: certificate not found in OCSP response");
353 OCSP_BASICRESP_free(bresp);
354 return -1;
355 }
356
357 /* Max-staleness (phase-28 C3): a signed response is only evidence within
358 * its thisUpdate..nextUpdate window (+clock skew), and a nonce-less
359 * pre-signed response could otherwise be replayed long after the CA
360 * revoked the cert. A stale/not-yet-valid response proves nothing
361 * CURRENT, so it degrades to UNKNOWN (rc=1 — the soft_fail policy
362 * decides, same as a responder that answered UNKNOWN outright) — except
363 * REVOKED, which is never overridden: even stale evidence of revocation
364 * still denies. */
365 if (!OCSP_check_validity(thisupd, nextupd,
366 BRIX_OCSP_VALIDITY_SKEW_SEC,
367 BRIX_OCSP_VALIDITY_MAX_AGE_SEC)
368 && status != V_OCSP_CERTSTATUS_REVOKED)
369 {
370 ngx_log_error(NGX_LOG_WARN, log, 0,
371 "brix_ocsp: response outside its validity window "
372 "(stale or not yet valid) — treating as UNKNOWN");
373 OCSP_BASICRESP_free(bresp);
374 return 1; /* caller decides based on soft_fail */
375 }
376
377 switch (status) {
378 case V_OCSP_CERTSTATUS_GOOD:
379 rc = 0;
380 break;
381 case V_OCSP_CERTSTATUS_REVOKED:
382 ngx_log_error(NGX_LOG_WARN, log, 0,
383 "brix_ocsp: certificate is REVOKED (reason=%d)", reason);
384 rc = -1;
385 break;
386 case V_OCSP_CERTSTATUS_UNKNOWN:
387 default:
388 ngx_log_error(NGX_LOG_WARN, log, 0,
389 "brix_ocsp: certificate status is UNKNOWN");
390 rc = 1; /* caller decides based on soft_fail */
391 break;
392 }
393
394 OCSP_BASICRESP_free(bresp);
395 return rc;
396}
