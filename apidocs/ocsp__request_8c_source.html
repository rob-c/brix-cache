1/*
2 * OCSP request/response — build the query, POST it, and verify the reply.
3 *
4 * WHAT: The OCSP protocol crypto proper. do_ocsp_request() builds an
5 * OCSP_REQUEST (cert id + anti-replay nonce), opens/connects the responder BIO
6 * via the transport layer, bounds every network phase under
7 * BRIX_OCSP_TIMEOUT_SECS, POSTs the request, and returns the parsed
8 * OCSP_RESPONSE. check_ocsp_response() verifies that response — overall status,
9 * responder signature against a trust store, nonce match — and maps the
10 * certificate status to GOOD(0) / REVOKED(-1) / UNKNOWN(1).
11 *
12 * WHY: Split out of ocsp.c under the phase-79 file-size guard. This file is the
13 * security-critical heart of the revocation check: the nonce guards against
14 * replay, the signature verification proves the response came from a trusted
15 * responder, and the status mapping is the actual revocation decision. Keeping
16 * it separate from the URL/BIO transport (ocsp_transport.c) and the public entry
17 * points (ocsp.c) keeps each concern reviewable in isolation.
18 *
19 * HOW: OCSP_request_add0_id() takes ownership of `id` on success (the caller
20 * must not free it after a successful add). do_ocsp_request() owns the BIO
21 * teardown (ocsp_conn_free, cbio then ssl_ctx) and the request free on every
22 * path. check_ocsp_response() frees the BASICRESP on every path; a REVOKED
23 * status is never overridden. Shared structs/decls live in ocsp_internal.h.
24 */

33/*
34 * ocsp_build_request — allocate an OCSP request for cert id `id` with a nonce.
35 *
36 * WHAT: Returns a new OCSP_REQUEST carrying a COPY of `id` and a fresh nonce,
37 * or NULL on allocation failure. Ownership of `id` stays with the caller.
38 * WHY: Splits request construction out of the network path so do_ocsp_request()
39 * reads as build → connect → send.
40 * HOW: OCSP_request_add0_id() takes ownership of what it is given, but `id`
41 * belongs to brix_ocsp_check_cert() / brix_ocsp_staple_fetch() for the whole
42 * AIA loop: they build it once, query every responder URL with it, and free it
43 * afterwards. Handing the request the caller's `id` therefore made
44 * OCSP_REQUEST_free() free it out from under them — a use-after-free on a
45 * second AIA URL and a double free on the caller's OCSP_CERTID_free(), which
46 * segfaulted the worker on EVERY completed OCSP round trip. So add0 gets a
47 * dup and the caller's pointer is never transferred. The nonce
48 * (OCSP_request_add1_nonce) guards against replay. add0 does not take
49 * ownership when it fails, so the copy is freed on that path along with the
50 * half-built request.
51 */
52static OCSP_REQUEST *
53ocsp_build_request(OCSP_CERTID *id)
54{
55 OCSP_REQUEST *req = OCSP_REQUEST_new();
56 OCSP_CERTID *copy;
57
58 if (req == NULL) {
59 return NULL;
60 }
61 copy = OCSP_CERTID_dup(id);
62 if (copy == NULL) {
63 OCSP_REQUEST_free(req);
64 return NULL;
65 }
66 if (OCSP_request_add0_id(req, copy) == NULL) {
67 OCSP_CERTID_free(copy);
68 OCSP_REQUEST_free(req);
69 return NULL;
70 }
71 /* Add a nonce to prevent replay attacks */
72 OCSP_request_add1_nonce(req, NULL, -1);
73 return req;
74}

76/*
77 * do_ocsp_request — build request, POST to responder, return parsed response.
78 *
79 * The caller must call OCSP_RESPONSE_free() on the returned pointer.
80 * Returns NULL on any network or protocol error.
81 */
82/* HOW: Parses the OCSP URL via parse_ocsp_url() into an ocsp_url_t. Builds the OCSP_REQUEST (id + nonce) via ocsp_build_request(). Opens the responder BIO via ocsp_open_bio() (plain TCP or verifying TLS with SNI). Bounds the connect (ocsp_connect_bio) and, for HTTPS, the handshake+verify (ocsp_tls_handshake) under BRIX_OCSP_TIMEOUT_SECS. Sends the request and reads the reply through an OCSP_REQ_CTX whose response length is capped at OCSP_MAX_RESPONSE_BYTES (A-6/T2 — an untrusted responder must not stream an unbounded body), then tears down the connection (cbio then ssl_ctx) and the request. Returns NULL on any network/protocol failure or if the reply exceeds the cap. */
83OCSP_RESPONSE *
84do_ocsp_request(ngx_log_t *log, const char *url,
85 X509 *leaf, X509 *issuer, OCSP_CERTID *id, OCSP_REQUEST **req_out)
86{
87 ocsp_url_t u;
88 char hostport[320];
89 OCSP_REQUEST *req = NULL;
90 OCSP_RESPONSE *resp = NULL;
91 OCSP_REQ_CTX *rctx;
92 int rc;
93 ocsp_conn_t conn;
94
95 *req_out = NULL;
96
97 if (parse_ocsp_url(url, &u) != 0) {
98 ngx_log_error(NGX_LOG_WARN, log, 0,
99 "brix_ocsp: cannot parse URL \"%s\"", url);
100 return NULL;
101 }
102
103 req = ocsp_build_request(id);
104 if (req == NULL) {
105 return NULL;
106 }
107
108 /* Construct "host:port" for BIO */
109 snprintf(hostport, sizeof(hostport), "%s:%d", u.host, u.port);
110
111 if (ocsp_open_bio(log, &u, hostport, &conn) != 0) {
112 OCSP_REQUEST_free(req);
113 return NULL;
114 }
115
116 /* E1: bound the connect with a deadline (a black-holed responder must not
117 * freeze the worker for the kernel TCP timeout). */
118 if (ocsp_connect_bio(log, &conn, hostport) != 0) {
119 ocsp_conn_free(&conn);
120 OCSP_REQUEST_free(req);
121 return NULL;
122 }
123
124 /* E1: bound the TLS handshake + request/response read/write phases. */
125 ocsp_set_io_timeouts(conn.cbio, BRIX_OCSP_TIMEOUT_SECS);
126
127 if (u.use_ssl
128 && ocsp_tls_handshake(log, &conn, u.host, hostport) != 0)
129 {
130 ocsp_conn_free(&conn);
131 OCSP_REQUEST_free(req);
132 return NULL;
133 }
134
135 /* A2/T2: bound the response body. The one-shot sendreq_bio helper reads
136 * under only OpenSSL's internal default cap; drive the request context
137 * ourselves so we pin the ceiling at OCSP_MAX_RESPONSE_BYTES — an untrusted
138 * (and usually
139 * plaintext/MITM-able) responder must not stream an unbounded body into the
140 * worker. Blocking BIO + SO_RCVTIMEO bound the wait; the -1 retry arm
141 * covers a spurious BIO_should_retry. */
142 rctx = OCSP_sendreq_new(conn.cbio, u.path, req, -1);
143 if (rctx == NULL) {
144 ocsp_conn_free(&conn);
145 OCSP_REQUEST_free(req);
146 ngx_log_error(NGX_LOG_WARN, log, 0,
147 "brix_ocsp: cannot build request context for \"%s\"", url);
148 return NULL;
149 }
150 OCSP_set_max_response_length(rctx, OCSP_MAX_RESPONSE_BYTES);
151
152 do {
153 rc = OCSP_sendreq_nbio(&resp, rctx);
154 } while (rc == -1 && BIO_should_retry(conn.cbio));
155
156 OCSP_REQ_CTX_free(rctx);
157 ocsp_conn_free(&conn);
158
159 if (rc <= 0 || resp == NULL) {
160 /* rc == 0 also fires when the reply exceeded OCSP_MAX_RESPONSE_BYTES:
161 * OpenSSL aborts the read at the cap and returns failure. */
162 ngx_log_error(NGX_LOG_WARN, log, 0,
163 "brix_ocsp: no valid response from \"%s\" (capped at %d bytes)",
164 url, OCSP_MAX_RESPONSE_BYTES);
165 if (resp != NULL) {
166 OCSP_RESPONSE_free(resp);
167 resp = NULL;
168 }
169 OCSP_REQUEST_free(req);
170 return NULL;
171 }
172
173 /* Success: hand the request (with its nonce) to the caller for the nonce
174 * match; the caller owns it now and frees it after check_ocsp_response(). */
175 *req_out = req;
176 return resp;
177}

179/*
180 * check_ocsp_response — verify the OCSP response and return the cert status.
181 *
182 * Returns:
183 * 0 — certificate is GOOD
184 * -1 — certificate is REVOKED or response is invalid
185 * 1 — status is UNKNOWN
186 */
187/* HOW: Checks response status — returns -1 if not successful. Extracts the BASICRESP via OCSP_response_get1_basic(). Optionally verifies the response signature against a trust store (NULL means use OpenSSL defaults). If original request is available, checks nonce match — mismatch causes failure; missing nonce is a warning only. Finds certificate status via OCSP_resp_find_status() and maps to GOOD(0)/REVOKED(-1)/UNKNOWN(1) using a switch on the V_OCSP_CERTSTATUS_* enum. */
188int
189check_ocsp_response(ngx_log_t *log, OCSP_RESPONSE *resp,
190 X509_STORE *store, OCSP_CERTID *id, OCSP_REQUEST *req_for_nonce,
191 int require_nonce)
192{
193 OCSP_BASICRESP *bresp;
194 int status, reason;
195 ASN1_GENERALIZEDTIME *rev = NULL, *thisupd = NULL, *nextupd = NULL;
196 int rc = -1;
197
198 if (OCSP_response_status(resp) != OCSP_RESPONSE_STATUS_SUCCESSFUL) {
199 ngx_log_error(NGX_LOG_WARN, log, 0,
200 "brix_ocsp: OCSP response status = %d (not successful)",
201 OCSP_response_status(resp));
202 return -1;
203 }
204
205 bresp = OCSP_response_get1_basic(resp);
206 if (bresp == NULL) {
207 return -1;
208 }
209
210 /* Verify the response signature against the trust store */
211 if (store != NULL && OCSP_basic_verify(bresp, NULL, store, 0) <= 0) {
212 ngx_log_error(NGX_LOG_WARN, log, 0,
213 "brix_ocsp: OCSP response signature verification failed");
214 OCSP_BASICRESP_free(bresp);
215 return -1;
216 }
217
218 /* Check the nonce if we have the original request */
219 if (req_for_nonce != NULL) {
220 int nonce_rc = OCSP_check_nonce(req_for_nonce, bresp);
221 if (nonce_rc < 0) {
222 /* Nonce present in request but missing in response. A missing nonce
223 * lets an on-path attacker replay a captured (still-signed) GOOD
224 * response. Under brix_ocsp_require_nonce this is a hard failure
225 * (A-6 item 2); otherwise warn only — most CA responders serve
226 * pre-signed, nonce-less responses, so hard-fail is opt-in. */
227 if (require_nonce) {
228 ngx_log_error(NGX_LOG_WARN, log, 0,
229 "brix_ocsp: nonce missing in OCSP response and "
230 "brix_ocsp_require_nonce is on — denying (replay guard)");
231 OCSP_BASICRESP_free(bresp);
232 return -1;
233 }
234 ngx_log_error(NGX_LOG_WARN, log, 0,
235 "brix_ocsp: nonce missing in OCSP response");
236 } else if (nonce_rc == 0) {
237 ngx_log_error(NGX_LOG_WARN, log, 0,
238 "brix_ocsp: OCSP response nonce mismatch");
239 OCSP_BASICRESP_free(bresp);
240 return -1;
241 }
242 }
243
244 if (!OCSP_resp_find_status(bresp, id, &status, &reason,
245 &rev, &thisupd, &nextupd))
246 {
247 ngx_log_error(NGX_LOG_WARN, log, 0,
248 "brix_ocsp: certificate not found in OCSP response");
249 OCSP_BASICRESP_free(bresp);
250 return -1;
251 }
252
253 /* Max-staleness (phase-28 C3): a signed response is only evidence within
254 * its thisUpdate..nextUpdate window (+clock skew), and a nonce-less
255 * pre-signed response could otherwise be replayed long after the CA
256 * revoked the cert. A stale/not-yet-valid response proves nothing
257 * CURRENT, so it degrades to UNKNOWN (rc=1 — the soft_fail policy
258 * decides, same as a responder that answered UNKNOWN outright) — except
259 * REVOKED, which is never overridden: even stale evidence of revocation
260 * still denies. */
261 if (!OCSP_check_validity(thisupd, nextupd,
262 BRIX_OCSP_VALIDITY_SKEW_SEC,
263 BRIX_OCSP_VALIDITY_MAX_AGE_SEC)
264 && status != V_OCSP_CERTSTATUS_REVOKED)
265 {
266 ngx_log_error(NGX_LOG_WARN, log, 0,
267 "brix_ocsp: response outside its validity window "
268 "(stale or not yet valid) — treating as UNKNOWN");
269 OCSP_BASICRESP_free(bresp);
270 return 1; /* caller decides based on soft_fail */
271 }
272
273 switch (status) {
274 case V_OCSP_CERTSTATUS_GOOD:
275 rc = 0;
276 break;
277 case V_OCSP_CERTSTATUS_REVOKED:
278 ngx_log_error(NGX_LOG_WARN, log, 0,
279 "brix_ocsp: certificate is REVOKED (reason=%d)", reason);
280 rc = -1;
281 break;
282 case V_OCSP_CERTSTATUS_UNKNOWN:
283 default:
284 ngx_log_error(NGX_LOG_WARN, log, 0,
285 "brix_ocsp: certificate status is UNKNOWN");
286 rc = 1; /* caller decides based on soft_fail */
287 break;
288 }
289
290 OCSP_BASICRESP_free(bresp);
291 return rc;
292}
