1/*
2 * cred_write.c — the domain-gated, audited credential-write verb (phase-108
3 * C11).
4 *
5 * WHAT: Implements brix_cred_write(): the form every credential
6 * materialization site calls. It claims the CREDENTIAL storage domain
7 * through the typed policy seam, runs the pure engine in cred_stage.c,
8 * and emits exactly one structured audit line per materialization.
9 *
10 * WHY: The engine must stay pure libc (it links into the standalone unit and
11 * both nginx modules), but the domain claim and the audit line are
12 * nginx-territory — so the gated form lives here, a deliberate,
13 * recorded deviation from the phase-108 §9.4 ledger's letter (which
14 * named only cred_stage.c) in service of the ledger's intent: one
15 * engine, one gate, one audit shape. An EXPORT-domain claim can never
16 * be laundered through this path — brix_vfs_domain_claim routes it to
17 * the phase-105 kernel fail-closed and the answer is EROFS.
18 *
19 * HOW: Shape-check, claim the domain (books the
20 * vfs_domain_mutation_total{credential,credential} sample), run the
21 * engine, audit. The audit line carries arm/kind/dir/outcome ONLY —
22 * never the bytes, never a secret-bearing path component (a
23 * credential's basename can encode a subject identity; the directory
24 * cannot).
25 */

64int
65brix_cred_write(const brix_cred_write_req_t *req, const void *bytes,
66 size_t len, char *path_out, size_t path_outsz)
67{
68 int saved;
69 char stage_dir[64];
70 const char *dir;
71
72 /* Shape first, so the name tables and audit line below can index the
73 * enums safely; the engine re-checks the rest of the shape itself. */
74 if (req == NULL
75 || (unsigned) req->arm >= BRIX_CRED_ARM_COUNT
76 || (unsigned) req->kind >= BRIX_CRED_KIND_COUNT)
77 {
78 errno = EINVAL;
79 return -1;
80 }
81
82 if (req->arm == BRIX_CRED_ARM_PERSISTENT) {
83 dir = req->dir;
84 } else if (brix_cred_stage_dir(stage_dir, sizeof(stage_dir)) == 0) {
85 dir = stage_dir;
86 } else {
87 dir = NULL; /* staging dir unsafe/unavailable —
88 * the engine will refuse below */
89 }
90
91 if (brix_vfs_domain_claim((ngx_log_t *) req->log,
92 BRIX_VFS_DOMAIN_CREDENTIAL,
93 BRIX_VFS_MUTATE_CREDENTIAL)
94 != NGX_OK)
95 {
96 saved = errno;
97 cred_write_audit((ngx_log_t *) req->log, req, dir, "denied", saved);
98 errno = saved;
99 return -1;
100 }
101
102 if (brix_cred_write_engine(req, bytes, len, path_out, path_outsz) != 0) {
103 saved = errno;
104 cred_write_audit((ngx_log_t *) req->log, req, dir, "err", saved);
105 errno = saved;
106 return -1;
107 }
108
109 cred_write_audit((ngx_log_t *) req->log, req, dir, "ok", 0);
110 return 0;
111}
