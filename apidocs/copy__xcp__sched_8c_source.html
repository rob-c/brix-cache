6/* ---- Release an unstarted worker's reserved block ----
7 *
8 * WHAT: Return a startup reservation to the normal claim pool.
9 *
10 * WHY: A replica can fail before it reaches the block loop; its reserved
11 * block must not remain invisible to the surviving workers.
12 *
13 * HOW: Atomically change only XCP_RESERVED to XCP_TODO. A worker that has
14 * already claimed the reservation owns the ordinary release path.
15 */
16static void
17xcp_release_reserved_block(xcp_shared_t *sh, size_t idx)
18{
19 unsigned char expected = XCP_RESERVED;
20 atomic_compare_exchange_strong(&sh->state[idx], &expected, XCP_TODO);
21}

24/* ---- Drive one worker's claim/fetch loop ----
25 *
26 * WHAT: Fetch the startup reservation and then claim or steal blocks until the
27 * destination is complete, cancellation is requested, or a fetch fails.
28 * Returns 0 for a worker that can leave normally and -1 for its failure.
29 *
30 * WHY: The thread entry point owns connection/buffer lifetime; keeping the
31 * scheduler in this helper makes those resource edges independent of the
32 * block-stealing state machine and keeps both paths below the complexity
33 * gate.
34 *
35 * HOW: 1. Convert the startup reservation into a normal busy claim and fetch it.
36 * 2. Repeatedly claim/steal, fetch, and release the stealer latch.
37 * 3. Wait briefly when a straggler owns the remaining work; stop on DONE or
38 * cancellation and return the worker verdict.
39 */
40static int
41xcp_worker_blocks(xcp_worker_t *w, brix_conn *c, brix_file *f, uint8_t *buf)
42{
43 xcp_shared_t *sh = w->sh;
44 int rc = 0;
45
46 if (w->initial_reserved) {
47 unsigned char expected = XCP_RESERVED;
48
49 if (atomic_compare_exchange_strong(&sh->state[w->initial_idx],
50 &expected, XCP_BUSY)) {
51 w->initial_reserved = 0;
52 if (xcp_fetch_block(w, c, f, w->initial_idx, buf) != 0) {
53 xcp_release_block(sh, w->initial_idx, 0);
54 rc = -1;
55 }
56 } else {
57 w->initial_reserved = 0;
58 }
59 }
60
61 while (rc == 0) {
62 int stole = 0;
63 ssize_t idx = xcp_claim(sh, w->claim_hint, &stole);
64
65 if (idx < 0) {
66 if (xcp_all_done(sh)) {
67 break;
68 }
69 if (brix_copy_quit_requested()) {
70 brix_status_set(&w->st, XRDC_ESOCK, EINTR,
71 "cancelled (signal)");
72 rc = -1;
73 break;
74 }
75 /* Blocks remain but none claimable: idle-wait for a straggler
76 * to finish, die, or free its steal slot. Bounded tail spin. */
77 {
78 struct timespec ts = { 0, 20 * 1000 * 1000 };
79 nanosleep(&ts, NULL);
80 }
81 continue;
82 }
83
84 w->claim_hint = (unsigned) idx + 1;
85 if (xcp_fetch_block(w, c, f, (size_t) idx, buf) != 0) {
86 xcp_release_block(sh, (size_t) idx, stole);
87 rc = -1;
88 break;
89 }
90 if (stole) {
91 atomic_store(&sh->stealer[(size_t) idx], 0);
92 }
93 }
94
95 return rc;
96}

99/* ---- Wait for every source's open attempt to resolve (bounded) ----
100 *
101 * WHAT: Block until sh->resolved reaches the spawn width, the grace cap
102 * expires, or the operator cancels.
103 *
104 * WHY: Without a start rendezvous the first source to finish its XRootD open
105 * can drain the whole block table before a sibling's handshake completes
106 * — on a fast link --sources N silently degrades to one source. Holding
107 * the claim start until every open RESOLVES (success or failure — a dead
108 * mirror lifts the gate as fast as a live one) guarantees each opened
109 * source at least its first claim. The grace cap keeps a black-holed
110 * mirror from stalling the transfer start beyond a bounded delay.
111 *
112 * HOW: 1. Poll resolved vs nworkers in 2 ms sleeps. 2. Give up waiting at
113 * XRDC_XCP_JOIN_GRACE_MS or on quit — the claim loop handles cancel.
114 */
115static void
116xcp_join_gate(const xcp_shared_t *sh)
117{
118 unsigned waited_ms = 0;
119
120 while (atomic_load(&sh->resolved) < sh->nworkers
121 && waited_ms < XRDC_XCP_JOIN_GRACE_MS
122 && !brix_copy_quit_requested()) {
123 struct timespec ts = { 0, 2 * 1000 * 1000 };
124 nanosleep(&ts, NULL);
125 waited_ms += 2;
126 }
127}
