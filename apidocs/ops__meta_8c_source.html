1/*
2 * ops_meta.c — metadata operations: kXR_stat and kXR_dirlist.
3 *
4 * WHAT: stat() returns id/size/flags/mtime; dirlist() returns entries, optionally
5 * with per-entry stat (kXR_dstat).
6 * WHY: These back `xrdfs stat` and `xrdfs ls`, and stat also serves xrdcp size
7 * checks later.
8 * HOW: The path is sent as the request payload with dlen=strlen (no trailing NUL
9 * — the server reads exactly dlen bytes). dirlist accumulates kXR_oksofar
10 * chunks until the final kXR_ok; with dstat the body begins with the
11 * ".\n0 0 0 0\n" lead-in followed by name/stat line pairs.
12 *
13 * wire: kXR_stat response — ASCII "<id> <size> <flags> <mtime>".
14 * wire: kXR_dirlist dstat — ".\n0 0 0 0" prefix, then name\nstat\n pairs.
15 */

25/* WHAT: Decode one stat line, returning 0 or -1. WHY: Path and handle replies
26 * must share the authoritative grammar. HOW: 1. Parse fields. 2. Adopt extensions. */
27int
28brix_statinfo_parse(const char *s, brix_statinfo *out)
29{
30 brix_statline_ext ext;
31
32 /* The stat-line grammar (mandatory "<id> <size> <flags> <mtime>" + optional
33 * EOS "<ctime> <atime> <mode> <owner> <group>" tail) is owned by the shared
34 * protocol/stat_line.h so this decoder stays in lockstep with the server's
35 * encoder. */
36 if (brix_statline_parse(s, &out->id, &out->size, &out->flags, &out->mtime,
37 &ext) != 0) {
38 return -1;
39 }
40
41 out->have_ext = ext.have_ext;
42 out->ctime = ext.ctime;
43 out->atime = ext.atime;
44 out->mode = ext.mode;
45 snprintf(out->owner, sizeof(out->owner), "%s", ext.owner);
46 snprintf(out->group, sizeof(out->group), "%s", ext.group);
47 return 0;
48}

50/* Shared core: stat a path with the given kXR_stat options byte (0 = follow
51 * symlinks, kXR_statNoFollow = lstat). */
52static int
53stat_opt(brix_conn *c, const char *path, uint8_t options, brix_statinfo *out,
54 brix_status *st)
55{
56 ClientStatRequest req;
57 uint16_t status;
58 uint8_t *body = NULL;
59 uint32_t blen = 0;
60 char tmp[256];
61 uint32_t n;
62
63 memset(&req, 0, sizeof(req));
64 req.requestid = htons(kXR_stat);
65 {
66 xrdw_stat_req_t b = { .options = options, .wants = 0 };
67 xrdw_stat_req_pack(&b, ((ClientRequestHdr *) &req)->body);
68 }
69
70 if (brix_roundtrip_resilient(c, &req, path, (uint32_t) strlen(path),
71 XRDC_OP_READONLY, 0,
72 &status, &body, &blen, st) != 0) {
73 return -1;
74 }
75
76 n = (blen < sizeof(tmp) - 1) ? blen : (uint32_t) (sizeof(tmp) - 1);
77 memcpy(tmp, body, n);
78 tmp[n] = '\0';
79 free(body);
80
81 if (brix_statinfo_parse(tmp, out) != 0) {
82 brix_status_set(st, XRDC_EPROTO, 0, "unparseable stat body: \"%s\"", tmp);
83 return -1;
84 }
85 return 0;
86}

103/* Append n bytes to a growing heap buffer; returns 0 / -1. */
104static int
105buf_append(uint8_t **buf, size_t *len, size_t *cap, const uint8_t *src, size_t n)
106{
107 if (*len + n > *cap) {
108 size_t newcap = (*cap == 0) ? 65536 : *cap;
109 uint8_t *nb;
110 while (*len + n > newcap) {
111 newcap *= 2;
112 }
113 nb = (uint8_t *) realloc(*buf, newcap);
114 if (nb == NULL) {
115 return -1;
116 }
117 *buf = nb;
118 *cap = newcap;
119 }
120 memcpy(*buf + *len, src, n);
121 *len += n;
122 return 0;
123}

125/* Find the next line within [i, len): a run ending at '\n' or '\0'. Sets *start
126 * and *linelen, returns the index just past the terminator, or len if none. The
127 * caller stops when it encounters a '\0' terminator (returned via *was_nul). */
128static size_t
129next_line(const uint8_t *b, size_t len, size_t i, size_t *start, size_t *linelen,
130 int *was_nul)
131{
132 size_t j = i;
133 *was_nul = 0;
134 while (j < len && b[j] != '\n' && b[j] != '\0') {
135 j++;
136 }
137 *start = i;
138 *linelen = j - i;
139 if (j < len && b[j] == '\0') {
140 *was_nul = 1;
141 }
142 return (j < len) ? j + 1 : len;
143}

145static int
146dirent_push(brix_dirent **ents, size_t *count, size_t *cap, const brix_dirent *e)
147{
148 if (*count == *cap) {
149 size_t newcap = (*cap == 0) ? 64 : *cap * 2;
150 brix_dirent *na = (brix_dirent *) realloc(*ents, newcap * sizeof(**ents));
151 if (na == NULL) {
152 return -1;
153 }
154 *ents = na;
155 *cap = newcap;
156 }
157 (*ents)[*count] = *e;
158 (*count)++;
159 return 0;
160}

162/* ---- Accumulate every dirlist response chunk into one heap buffer ----
163 *
164 * WHAT: Sends the prepared kXR_dirlist request on c and drains the response,
165 * concatenating the first frame and every subsequent kXR_oksofar chunk
166 * into a single freshly-allocated buffer returned via the acc_out and acclen_out out-params.
167 * Returns 0 on success; -1 on any transport or OOM error, having freed all
168 * working buffers so nothing leaks and the caller starts clean.
169 *
170 * WHY: A directory listing arrives split across an arbitrary number of
171 * kXR_oksofar frames terminated by a final kXR_ok. Isolating the
172 * receive-and-concatenate loop keeps dirlist_once a flat orchestrator and
173 * lets the parse step work over one contiguous buffer.
174 *
175 * HOW: 1. Send the first frame via brix_roundtrip (which follows a cluster
176 * redirect to the data server); read every later chunk from that same
177 * post-redirect connection with brix_recv.
178 * 2. Append each non-empty body to acc via buf_append; free each per-frame
179 * body immediately after copying.
180 * 3. Stop once a frame's status is no longer kXR_oksofar.
181 */
182static int
183dirlist_accumulate(brix_conn *c, ClientDirlistRequest *req, const char *path,
184 uint8_t **acc_out, size_t *acclen_out,
185 size_t *continuations, brix_status *st)
186{
187 uint16_t status;
188 uint8_t *acc = NULL;
189 size_t acclen = 0, acccap = 0;
190 uint8_t *body = NULL;
191 uint32_t blen = 0;
192 int first = 1;
193 brix_payload pl = { path, (uint32_t) strlen(path) };
194 brix_resp_out out = { &status, &body, &blen };
195
196 if (continuations != NULL) *continuations = 0;
197
198 for (;;) {
199 if (first) {
200 if (brix_roundtrip(c, req, &pl, &out, st) != 0) {
201 free(acc);
202 return -1;
203 }
204 first = 0;
205 } else if (brix_recv(c, 0xffff, &out, st) != 0) {
206 free(acc);
207 return -1;
208 }
209 if (blen > 0
210 && buf_append(&acc, &acclen, &acccap, body, blen) != 0) {
211 free(body);
212 free(acc);
213 brix_status_set(st, XRDC_EPROTO, 0, "out of memory in dirlist");
214 return -1;
215 }
216 free(body);
217 body = NULL;
218 blen = 0;
219 if (status != kXR_oksofar) {
220 break;
221 }
222 if (continuations != NULL) (*continuations)++;
223 }
224
225 *acc_out = acc;
226 *acclen_out = acclen;
227 return 0;
228}

230/* ---- Skip the dstat lead-in prefix, returning the first entry offset ----
231 *
232 * WHAT: Returns the byte offset within acc at which directory entries begin. For
233 * a kXR_dstat listing that starts with the ".\n0 0 0 0" sentinel, that is
234 * the offset just past the sentinel's terminating '\n'; otherwise 0.
235 *
236 * WHY: With kXR_dstat the body opens with a fixed ".\n0 0 0 0\n" lead-in (the
237 * "." pseudo-entry plus its zeroed stat line) that must be consumed before
238 * real name/stat pairs. The sentinel and its match length are shared with
239 * the server via dirlist_fmt.h so both sides agree byte-for-byte.
240 *
241 * HOW: 1. Return 0 unless want_stat is set and acc begins with the shared
242 * BRIX_DSTAT_LEADIN sentinel of BRIX_DSTAT_PREFIX_LEN bytes.
243 * 2. Advance past the first '\n' (end of "."), then past the second '\n'
244 * (end of the "0 0 0 0" stat line), guarding each step against acclen.
245 */
246static size_t
247dirlist_skip_dstat_leadin(const uint8_t *acc, size_t acclen, int want_stat)
248{
249 size_t i = 0;
250
251 if (want_stat && acclen >= BRIX_DSTAT_PREFIX_LEN
252 && memcmp(acc, BRIX_DSTAT_LEADIN, BRIX_DSTAT_PREFIX_LEN) == 0) {
253 while (i < acclen && acc[i] != '\n') { i++; } /* ".\n..." first \n */
254 if (i < acclen) { i++; } /* now at "0 0 0 0\n" */
255 while (i < acclen && acc[i] != '\n') { i++; } /* end of "0 0 0 0" */
256 if (i < acclen) { i++; }
257 }
258 return i;
259}

261static void
262dirent_parse_stat_line(const uint8_t *acc, size_t acclen, size_t *offset,
263 int *term, int want_cksum, brix_dirent *entry)
264{
265 char statbuf[256];
266 size_t start, len, copy_len;
267 char *open, *close;
268
269 *offset = next_line(acc, acclen, *offset, &start, &len, term);
270 copy_len = len < sizeof(statbuf) - 1 ? len : sizeof(statbuf) - 1;
271 memcpy(statbuf, acc + start, copy_len);
272 statbuf[copy_len] = '\0';
273 if (brix_statinfo_parse(statbuf, &entry->st) == 0) {
274 entry->have_stat = 1;
275 }
276 if (!want_cksum) {
277 return;
278 }
279
280 open = strchr(statbuf, '[');
281 close = open == NULL ? NULL : strchr(open + 1, ']');
282 if (open == NULL || close == NULL) {
283 return;
284 }
285 start = (size_t) (open + 1 - statbuf);
286 len = (size_t) (close - statbuf);
287 while (start < len && statbuf[start] == ' ') {
288 start++;
289 }
290 while (len > start && statbuf[len - 1] == ' ') {
291 len--;
292 }
293 copy_len = len - start;
294 if (copy_len >= sizeof(entry->checksum)) {
295 copy_len = sizeof(entry->checksum) - 1;
296 }
297 memcpy(entry->checksum, statbuf + start, copy_len);
298 entry->checksum[copy_len] = '\0';
299}

301/* ---- Parse name (and optional per-entry stat) lines into a dirent array ----
302 *
303 * WHAT: Walks acc from offset i, building a heap array of brix_dirent handed off
304 * via the ents_out and count_out out-params. Returns 0 on success (ownership transferred);
305 * -1 on OOM, having freed its partial array and set a status on st.
306 *
307 * WHY: The entry region is a sequence of name lines, each optionally followed
308 * (under kXR_dstat) by a stat line, terminated by an empty line or a '\0'.
309 * Keeping the parse pure over the already-accumulated buffer makes the
310 * grammar auditable and independent of the transport loop.
311 *
312 * HOW: 1. For each line via next_line: stop on an empty name (trailing line).
313 * 2. Copy the name into the fixed dirent field, truncating to capacity.
314 * 3. When want_stat and the name line was not itself terminated, read the
315 * following stat line and decode it with parse_statinfo (set have_stat
316 * only on a clean parse).
317 * 4. Push the entry with dirent_push; stop when a '\0' terminator was seen.
318 */
319static int
320dirlist_parse_entries(const uint8_t *acc, size_t acclen, size_t i,
321 int want_stat, int want_cksum, brix_dirent **ents_out,
322 size_t *count_out, brix_status *st)
323{
324 brix_dirent *ents = NULL;
325 size_t count = 0, entcap = 0;
326
327 while (i < acclen) {
328 size_t ns, nl;
329 int term = 0;
330 brix_dirent e;
331
332 i = next_line(acc, acclen, i, &ns, &nl, &term);
333 if (nl == 0) {
334 break; /* empty trailing line */
335 }
336
337 memset(&e, 0, sizeof(e));
338 {
339 size_t cp = (nl < sizeof(e.name) - 1) ? nl : sizeof(e.name) - 1;
340 memcpy(e.name, acc + ns, cp);
341 e.name[cp] = '\0';
342 }
343
344 if (want_stat && !term) {
345 dirent_parse_stat_line(acc, acclen, &i, &term, want_cksum, &e);
346 }
347
348 if (dirent_push(&ents, &count, &entcap, &e) != 0) {
349 brix_status_set(st, XRDC_EPROTO, 0, "out of memory in dirlist");
350 free(ents);
351 return -1;
352 }
353 if (term) {
354 break;
355 }
356 }
357
358 *ents_out = ents;
359 *count_out = count;
360 return 0;
361}

363/* One full listing pass (a kXR_dirlist roundtrip plus the kXR_oksofar chunk
364 * accumulation) on the current connection. On any failure it leaves *ents_out
365 * NULL and frees its working buffers, so a resilient retry starts clean. */
366static int
367dirlist_once(brix_conn *c, const char *path, uint8_t options,
368 brix_dirent **ents_out, size_t *count_out,
369 size_t *continuations, brix_status *st)
370{
371 ClientDirlistRequest req;
372 uint8_t *acc = NULL;
373 size_t acclen = 0;
374 size_t i;
375 int rc;
376 int want_stat = (options & (kXR_dstat | kXR_dcksm)) != 0;
377
378 *ents_out = NULL;
379 *count_out = 0;
380
381 memset(&req, 0, sizeof(req));
382 req.requestid = htons(kXR_dirlist);
383 {
384 xrdw_dirlist_req_t b = { .options = options };
385 xrdw_dirlist_req_pack(&b, ((ClientRequestHdr *) &req)->body);
386 }
387
388 /* Drain the whole listing into one buffer (helper frees acc on failure). */
389 if (dirlist_accumulate(c, &req, path, &acc, &acclen, continuations,
390 st) != 0) {
391 return -1;
392 }
393
394 /* Skip the dstat lead-in, then parse the name/stat pairs. On a parse OOM the
395 * parse helper frees its partial array; the accumulation buffer we still own
396 * is freed once below, on both the success and error paths. */
397 i = dirlist_skip_dstat_leadin(acc, acclen, want_stat);
398 rc = dirlist_parse_entries(acc, acclen, i, want_stat,
399 (options & kXR_dcksm) != 0, ents_out, count_out, st);
400 free(acc);
401 return rc;
402}
