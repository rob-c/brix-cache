1/*
2 * vfs_unlink.c — VFS delete family (unlink / rmdir).
3 *
4 * WHAT: Implements brix_vfs_unlink() (remove a regular file) and
5 * brix_vfs_rmdir() (remove a directory, recursively or only when empty),
6 * both thin wrappers over the shared brix_vfs_delete() helper.
7 *
8 * WHY: Deletes are write-gated and must be applied through the namespace layer
9 * (../compat/namespace_ops) so confinement and the recursive / require-
10 * empty semantics match the rest of the namespace mutators, with one
11 * metric/access-log emission per call.
12 *
13 * HOW: brix_vfs_delete() enforces brix_vfs_require_confined_mutation()
14 * (MUTATE_REMOVE — EROFS on a read-only endpoint) and a non-NULL
15 * root_canon, builds an brix_ns_delete_opts_t (recursive,
16 * require_empty_dir), and calls brix_ns_delete(); the namespace status
17 * is mapped back to errno (sys_errno or EIO) and observed as
18 * BRIX_METRIC_OP_DELETE. rmdir requests require_empty_dir only when not
19 * recursive.
20 */

25/* brix_vfs_driver_rmtree — depth-first delete of `logical` through the storage
26 * driver: a file is unlinked directly; a directory has its children removed
27 * (opendir/readdir recursion) before the now-empty directory itself.
28 *
29 * WHAT: Recursive WebDAV DELETE of a collection on a non-POSIX backend:
30 * dispatches stat/opendir/readdir/unlink through the LEAF driver so that
31 * per-user credentials (stat_cred / unlink_cred / opendir_cred) are
32 * presented at every level of the tree, not just the top-level call.
33 *
34 * WHY: Calling drv->stat(ctx->sd, …) / drv->unlink(ctx->sd, …) bypasses the
35 * cred entirely because ctx->sd may be a decorator without *_cred slots.
36 * Dispatching through brix_sd_*_maybe_cred on the leaf ensures that a
37 * user credential, when present (use_cred=1), is threaded through the
38 * entire recursive walk — not just the outermost call.
39 *
40 * HOW: Accepts the pre-resolved leaf `leaf` and pre-computed `cred` pointer
41 * (NULL when use_cred=0) from the caller, uses brix_sd_*_maybe_cred on
42 * `leaf` for every driver call, and recurses with the same arguments.
43 * Non-static: brix_vfs_rmtree_dispatch (vfs_unlink_many.c) falls back to
44 * this per-key walk when the leaf has no real batch slot (C4). */
45ngx_int_t
46brix_vfs_driver_rmtree(brix_sd_instance_t *leaf, const brix_sd_driver_t *drv,
47 const char *logical, const brix_sd_cred_t *cred, ngx_uint_t depth)
48{
49 brix_sd_stat_t st;
50
51 /* D-6/T2: bound the independent readdir recursion so a hostile deep remote
52 * collection aborts cleanly instead of overflowing the worker stack. */
53 if (depth > BRIX_FS_TREE_MAX_DEPTH) {
54 errno = ELOOP;
55 return NGX_ERROR;
56 }
57
58 if (drv->stat == NULL || drv->unlink == NULL) {
59 errno = ENOTSUP;
60 return NGX_ERROR;
61 }
62 if (brix_sd_stat_maybe_cred(leaf, logical, &st, cred) != NGX_OK) {
63 return NGX_ERROR; /* ENOENT etc. — errno set by the driver */
64 }
65
66 if (st.is_dir && drv->opendir != NULL) {
67 brix_sd_dir_t *dir;
68 int err = 0;
69
70 dir = brix_sd_opendir_maybe_cred(leaf, logical, &err, cred);
71 if (dir != NULL) {
72 brix_sd_dirent_t de;
73 ngx_int_t drc;
74
75 while ((drc = drv->readdir(dir, &de)) == NGX_OK) {
76 char child[PATH_MAX];
77
78 ngx_snprintf((u_char *) child, sizeof(child), "%s/%s%Z",
79 (logical[0] == '/' && logical[1] == '\0')
80 ? "" : logical,
81 de.name);
82 if (brix_vfs_driver_rmtree(leaf, drv, child, cred, depth + 1)
83 != NGX_OK)
84 {
85 drv->closedir(dir);
86 return NGX_ERROR;
87 }
88 }
89 drv->closedir(dir);
90 if (drc == NGX_ERROR) {
91 return NGX_ERROR;
92 }
93 }
94 return brix_sd_unlink_maybe_cred(leaf, logical, 1, cred);
95 }
96
97 return brix_sd_unlink_maybe_cred(leaf, logical, 0, cred);
98}

100/* brix_vfs_delete_via_driver — perform a delete through a non-POSIX storage
101 * driver's namespace, presenting per-user credentials at every level.
102 *
103 * WHAT: Deletes `path` through `drv` (already confirmed non-NULL by the
104 * caller): a recursive request walks the collection tree, a
105 * non-recursive request unlinks a file or empty directory. Returns
106 * NGX_OK on success and NGX_ERROR (with errno set) on any failure,
107 * emitting exactly one BRIX_METRIC_OP_DELETE observation for `start`.
108 *
109 * WHY: Object/S3 backends have no POSIX namespace, so the delete must be
110 * dispatched on the LEAF instance where the driver's *_cred slots live
111 * (decorators carry only plain relays). Threading the resolved user
112 * credential through the entire walk keeps confinement identical to the
113 * top-level call — a partial-credential walk would delete children as
114 * the wrong identity.
115 *
116 * HOW: 1. Resolve the export-relative logical path and the leaf instance.
117 * 2. Zero the credential struct BEFORE the gate so an inactive kind is
118 * never handed to the driver as a dangling pointer.
119 * 3. When the cred gate is active, resolve the user credential; on
120 * failure observe EACCES (or the gate's errno) and return NGX_ERROR.
121 * 4. Recursive → brix_vfs_driver_rmtree; else unlink via the driver
122 * (require_empty_dir selects directory semantics), or ENOTSUP when the
123 * driver has no unlink.
124 * 5. Observe the result and return it. */
125static ngx_int_t
126brix_vfs_delete_via_driver(brix_vfs_ctx_t *ctx, const brix_sd_driver_t *drv,
127 unsigned recursive, unsigned require_empty_dir, const char *path,
128 uint64_t start)
129{
130 char logical[PATH_MAX];
131 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(ctx->sd);
132 brix_sd_ucred_t store;
133 brix_sd_cred_t cred;
134 ngx_int_t rc;
135 int saved_errno;
136 int use_cred = 0, cred_err = 0;
137
138 /* Zero before the gate: it fills only the active credential kind; an
139 * unzeroed cred hands a garbage inactive pointer to the driver's
140 * cred slot (bearer PASSTHROUGH would leave x509_proxy dangling). */
141 ngx_memzero(&cred, sizeof(cred));
142
143 if (brix_path_resolved_to_pfn(ctx, path, logical, sizeof(logical))
144 != NGX_OK)
145 {
146 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE,
147 NULL, 0, NGX_ERROR, errno, start);
148 return NGX_ERROR;
149 }
150
151 if (brix_vfs_cred_gate_active(ctx)) {
152 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
153 != NGX_OK)
154 {
155 saved_errno = cred_err ? cred_err : EACCES;
156 errno = saved_errno;
157 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE,
158 NULL, 0, NGX_ERROR, saved_errno,
159 start);
160 return NGX_ERROR;
161 }
162 }
163
164 if (recursive) {
165 /* phase-107 C4: routes to the windowed bulk walk when the leaf
166 * advertises CAP_BULK_DELETE, else to brix_vfs_driver_rmtree. */
167 rc = brix_vfs_rmtree_dispatch(leaf, drv, logical,
168 use_cred ? &cred : NULL);
169 } else if (drv->unlink != NULL) {
170 rc = brix_sd_unlink_maybe_cred(leaf, logical,
171 require_empty_dir ? 1 : 0, use_cred ? &cred : NULL);
172 } else {
173 errno = ENOTSUP;
174 rc = NGX_ERROR;
175 }
176 saved_errno = errno;
177 brix_sd_ucred_wipe(&store); /* secret consumed by unlink; erase (A-4/T4) */
178 if (rc == NGX_OK) {
179 /* The leaf dispatch above skipped the cache decorator's own unlink (and
180 * so its cstore evict); evict here so a DELETE does not leave a stale
181 * cached copy on the store. No-op when ctx->sd is not a cache. */
182 brix_metric_cache_evicted(brix_vfs_metrics_proto(ctx),
183 brix_sd_cache_evict(ctx->sd, logical));
184 }
185 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
186 rc, saved_errno, start);
187 return rc;
188}

190/* brix_vfs_delete_via_namespace — perform a delete through the POSIX namespace
191 * layer (brix_ns_delete) under the confined root_canon.
192 *
193 * WHAT: Builds a brix_ns_delete_opts_t from `recursive` / `require_empty_dir`
194 * and calls brix_ns_delete on the confined root. Returns NGX_OK on
195 * success and NGX_ERROR (with errno mapped from the namespace status) on
196 * failure, emitting exactly one BRIX_METRIC_OP_DELETE observation.
197 *
198 * WHY: This is the default (POSIX) backend path; it must apply the same
199 * recursive / require-empty semantics as the driver path while keeping
200 * confinement in brix_ns_delete's hands (root_canon + logical path).
201 *
202 * HOW: 1. Zero opts and set recursive / require_empty_dir.
203 * 2. require_directory tracks require_empty_dir: a non-recursive
204 * empty-dir delete is rmdir and must reject a regular file (ENOTDIR),
205 * while a recursive delete removes a file directly.
206 * 3. Call brix_ns_delete; on BRIX_NS_OK observe success and return
207 * NGX_OK.
208 * 4. Otherwise map sys_errno (or the status) to errno, observe the
209 * error, and return NGX_ERROR. */
210static ngx_int_t
211brix_vfs_delete_via_namespace(brix_vfs_ctx_t *ctx, unsigned recursive,
212 unsigned require_empty_dir, const char *path, uint64_t start)
213{
214 brix_ns_delete_opts_t opts;
215 brix_ns_result_t res;
216 int saved_errno;
217
218 ngx_memzero(&opts, sizeof(opts));
219 opts.recursive = recursive ? 1 : 0;
220 opts.require_empty_dir = require_empty_dir ? 1 : 0;
221 /* A non-recursive "empty dir" delete is rmdir, which must reject a regular
222 * file (ENOTDIR) — kXR_rmdir parity. Recursive deletes (require_empty_dir=0)
223 * remove a file directly, so require_directory stays off there. */
224 opts.require_directory = require_empty_dir ? 1 : 0;
225
226 if (ctx->rootfd >= 0) {
227 /* Borrow the ctx's persistent confinement rootfd: same beneath
228 * semantics, minus a root open/close per delete. */
229 res = brix_ns_delete_at(ctx->log, ctx->rootfd, ctx->root_canon,
230 brix_vfs_ctx_path(ctx), &opts);
231 } else {
232 res = brix_ns_delete(ctx->log, ctx->root_canon,
233 brix_vfs_ctx_path(ctx), &opts);
234 }
235 if (res.status == BRIX_NS_OK) {
236 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
237 NGX_OK, 0, start);
238 return NGX_OK;
239 }
240
241 errno = res.sys_errno != 0 ? res.sys_errno
242 : brix_vfs_ns_status_errno(res.status);
243 saved_errno = errno;
244 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
245 NGX_ERROR, saved_errno, start);
246 return NGX_ERROR;
247}

249/* Shared delete body for unlink/rmdir: write-gate, then dispatch to the driver
250 * or POSIX namespace delete with the given recursive / require_empty_dir
251 * options; metered as OP_DELETE.
252 *
253 * WHAT: Enforces the write gate and a non-NULL root_canon, then routes the
254 * delete to brix_vfs_delete_via_driver (non-POSIX backend present) or
255 * brix_vfs_delete_via_namespace (POSIX default). Returns NGX_OK on
256 * success and NGX_ERROR (with errno set) on any failure; each failure
257 * gate emits its own BRIX_METRIC_OP_DELETE observation.
258 *
259 * WHY: unlink and rmdir share identical gating and observation; centralising
260 * it here keeps confinement (write gate + root_canon) uniform and leaves
261 * only the backend-specific mechanics in the two dispatch helpers.
262 *
263 * HOW: 1. Snapshot start time and the logical path.
264 * 2. Fail closed on write-gate denial or a NULL root_canon, observing
265 * each error.
266 * 3. If a storage driver is bound, delegate to the driver helper.
267 * 4. Otherwise delegate to the POSIX namespace helper. */
268static ngx_int_t
269brix_vfs_delete(brix_vfs_ctx_t *ctx, unsigned recursive,
270 unsigned require_empty_dir)
271{
272 const char *path;
273 uint64_t start;
274 int saved_errno;
275 const brix_sd_driver_t *drv;
276
277 start = brix_vfs_now_ns();
278 path = brix_vfs_ctx_path(ctx);
279
280 if (brix_vfs_confined_mutation_checked(ctx,
281 BRIX_VFS_MUTATE_REMOVE) != NGX_OK)
282 {
283 saved_errno = errno;
284 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
285 NGX_ERROR, saved_errno, start);
286 return NGX_ERROR;
287 }
288
289 /* phase-107 C7: lock gate after the mutation gate (EROFS precedes EBUSY).
290 * Ancestor coverage only — the recursive descendant scan for collection
291 * deletes stays at the WebDAV edge. */
292 if (brix_vfs_require_unlocked(ctx, BRIX_VFS_MUTATE_REMOVE) != NGX_OK) {
293 saved_errno = errno;
294 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
295 NGX_ERROR, saved_errno, start);
296 return NGX_ERROR;
297 }
298
299 if (ctx->root_canon == NULL) {
300 errno = EINVAL;
301 saved_errno = errno;
302 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
303 NGX_ERROR, saved_errno, start);
304 return NGX_ERROR;
305 }
306
307 /* Confinement anchor (phase-107 C4): a recursive delete of the export root
308 * itself is refused outright - the walk would empty the entire export off
309 * one one-character URI, and the root is the object every confined path is
310 * resolved AGAINST, not one of the objects it resolves. EPERM, not EACCES:
311 * no credential makes this legal. */
312 if (recursive) {
313 const char *logical = brix_vfs_export_relative(ctx, path);
314
315 if (logical[0] == '\0'
316 || (logical[0] == '/' && logical[1] == '\0'))
317 {
318 errno = EPERM;
319 saved_errno = errno;
320 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DELETE, NULL, 0,
321 NGX_ERROR, saved_errno, start);
322 return NGX_ERROR;
323 }
324 }

326 /* Non-POSIX backend: delete through the driver namespace. A recursive delete
327 * (WebDAV DELETE of a collection) walks the tree; a non-recursive delete is a
328 * file unlink or empty-rmdir (require_directory selects is_dir).
329 * Dispatch on the leaf instance so *_maybe_cred forwarders find the leaf
330 * driver's *_cred slots (decorators have only plain relays). */
331 drv = brix_vfs_ctx_driver(ctx);
332 if (drv != NULL) {
333 return brix_vfs_delete_via_driver(ctx, drv, recursive,
334 require_empty_dir, path, start);
335 }
336
337 return brix_vfs_delete_via_namespace(ctx, recursive, require_empty_dir,
338 path, start);
339}
