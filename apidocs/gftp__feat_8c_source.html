8/* One advertised feature name and the bit it lights. A FEAT line is a bare
9 * verb optionally followed by parameters (RFC 2389 §3.2), so the match is on
10 * the leading token of a trimmed line and never on a substring of the reply —
11 * "ERET" must not be found inside a door's banner or inside "SITE ERETSTAT". */
12static const struct {
13 const char *name;
14 size_t len;
15 unsigned bit;

17 { "ERET", 4, GFTP_FEAT_ERET },
18 { "SPAS", 4, GFTP_FEAT_SPAS },
19};

22/* True when `line` (already trimmed of its leading space) begins with the
23 * feature `name` as a whole token.
24 *
25 * The line terminator is a token boundary, and saying so is the whole point of
26 * this helper: FEAT lines arrive CRLF-terminated (RFC 2389 §3.2 over RFC 959's
27 * control channel), so a boundary set of only ' ' / '\t' / NUL matched NOTHING
28 * on a conforming origin — every advertised name is followed by the CR. The
29 * mask came back empty every time, ERET and SPAS were never once issued, and
30 * nothing said so: the driver simply fell back to REST+RETR and single-stream
31 * transfers and worked, slower and unbounded. */
32static int
33gftp_feat_line_is(const char *line, const char *name, size_t len)
34{
35 char after;
36
37 if (strncasecmp(line, name, len) != 0) {
38 return 0;
39 }
40 after = line[len];
41
42 return after == '\0' || after == ' ' || after == '\t'
43 || after == '\r' || after == '\n';
44}

47/* Parse one FEAT reply body into a mask. */
48static unsigned
49gftp_feat_parse(const char *text)
50{
51 unsigned mask = 0;
52 const char *line = text;
53
54 while (line != NULL && *line != '\0') {
55 const char *end = strchr(line, '\n');
56 size_t i;
57
58 while (*line == ' ' || *line == '\t' || *line == '\r') {
59 line++;
60 }
61 for (i = 0; i < sizeof(gftp_feat_names) / sizeof(*gftp_feat_names);
62 i++)
63 {
64 if (gftp_feat_line_is(line, gftp_feat_names[i].name,
65 gftp_feat_names[i].len))
66 {
67 mask |= gftp_feat_names[i].bit;
68 }
69 }
70 line = (end != NULL) ? end + 1 : NULL;
71 }
72 return mask;
73}

82 /* Probed exactly once whatever the outcome: an origin that answers FEAT
83 * with 500 is telling us something permanent about itself, and re-asking on

85 session->feat_probed = 1;
86 session->feat = 0;
