28static int
29read_reply(brix_conn *connection, hostile_reply *reply, brix_status *status)
30{
31 uint8_t header[XRD_RESPONSE_HDR_LEN];
32
33 memset(reply, 0, sizeof(*reply));
34 if (brix_read_full(&connection->io, header, sizeof(header), status) != 0)
35 return -1;
36 xrd_resp_hdr_unpack(header, &reply->sid, &reply->status, &reply->length);
37 if (reply->length > sizeof(reply->body)) return -1;
38 if (reply->length > 0 && brix_read_full(&connection->io, reply->body,
39 reply->length, status) != 0) return -1;
40 return 0;
41}
