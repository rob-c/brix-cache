1/* fd_table_bound.c — the kXR_bind secondary's view of the primary's handles.
2 *
3 * A bound data stream never owns file handles: it follows the primary

19/* brix_local_file_matches_shared_handle — 1 iff the local fd still corresponds to
20 * the primary's published handle (fd valid + readable, same device/inode/path), used
21 * by brix_ensure_read_handle to decide whether a bound secondary must reopen. */
22
23static ngx_flag_t
24brix_local_file_matches_shared_handle(const brix_file_t *file,
25 const brix_shared_handle_entry_t *shared)
26{
27 if (file->fd < 0 || file->path == NULL || !file->readable
28 || !shared->readable)
29 {
30 return 0;
31 }
32
33 return file->device == shared->device
34 && file->inode == shared->inode
35 && ngx_strcmp(file->path, shared->path) == 0;
36}

38/* brix_bound_confined_open — cache-aware confined (re)open of a primary-published
39 * shared entry's path with the given flags, validating the reopened object's
40 * device/inode against the published tuple. Returns a fresh caller-owned fd, or
41 * -1 on open/fstat failure or a device/inode mismatch (path replaced since publish;
42 * the fd is closed before returning). Shared by the bound READ (O_RDONLY) and
43 * bound WRITE (O_WRONLY) reopen paths so the confinement + stale-reference check
44 * lives in exactly one place. */
45static int
46brix_bound_confined_open(brix_ctx_t *ctx, ngx_connection_t *c,
47 const brix_shared_handle_entry_t *shared, int open_flags, struct stat *st)
48{
49 ngx_stream_brix_srv_conf_t *conf;
50 int fd;
51
52 conf = ngx_stream_get_module_srv_conf(ctx->session,
53 ngx_stream_brix_module);
54 if (conf == NULL) {
55 return -1;
56 }
57
58 /*
59 * The primary published a canonical, ACL-checked path. The secondary
60 * still opens it with the same confinement helper so a stale or corrupted
61 * shared-memory entry cannot escape the configured export root.
62 */
63 if (shared->from_cache) {
64 fd = open(shared->path, open_flags | O_NOFOLLOW); /* vfs-seam-allow: DOMAIN_CACHE — separate server-managed cache-root domain (from_cache), opened as worker; O_NOFOLLOW so a stray symlink in the svc-owned cache tree is not followed */
65 } else {
66 /* shared->path is the absolute path; strip root_canon to get the
67 * path relative to rootfd for openat2 RESOLVE_BENEATH. */
68 const char *rel = shared->path;
69 size_t root_len = strlen(conf->common.root_canon);
70 if (root_len > 0
71 && ngx_strncmp((u_char *) shared->path,
72 (u_char *) conf->common.root_canon,
73 root_len) == 0
74 && shared->path[root_len] == '/')
75 {
76 rel = shared->path + root_len;
77 }
78 /* phase-105 (Appendix K.8): a bound data stream reopens a handle the
79 * PRIMARY published, on a connection whose own endpoint decides what it
80 * may do. A writable-looking bound handle must therefore meet this
81 * server's write posture here — the gated form refuses a mutating
82 * open_flags with EROFS and leaves a read-only reopen untouched. */
83 brix_vfs_export_op_ctx_t opctx;
84
85 brix_vfs_export_op_ctx_init(&opctx, c->log, conf->common.root_canon,
86 brix_vfs_policy_from_write_enable(conf->common.allow_write),
87 BRIX_PROTO_ROOT);
88 fd = brix_vfs_export_open_fd_at(&opctx, conf->rootfd, rel, open_flags,
89 0);
90 }
91
92 if (fd < 0) {
93 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, errno,
94 "brix: bound handle confined open failed path=%s flags=%d",
95 shared->path, open_flags);
96 return -1;
97 }
98
99 if (fstat(fd, st) != 0) {
100 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, errno,
101 "brix: fstat failed on bound reopen path=%s", shared->path);
102 close(fd);
103 return -1;
104 }
105
106 if (st->st_dev != shared->device || st->st_ino != shared->inode) {
107 close(fd);
108 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
109 "brix: bound handle path changed before reopen path=%s",
110 shared->path);
111 return -1;
112 }
113
114 return fd;
115}

117/* brix_reopen_bound_read_handle — reopen a bound secondary's stale fd O_RDONLY to
118 * match the primary's shared entry (via brix_bound_confined_open), refreshing all
119 * file metadata from the fresh fstat; NGX_DECLINED (revoke) if the object's
120 * device/inode changed. */
121
122static ngx_int_t
123brix_reopen_bound_read_handle(brix_ctx_t *ctx, ngx_connection_t *c,
124 int handle_index, const brix_shared_handle_entry_t *shared)
125{
126 brix_file_t *file;
127 struct stat st;
128 int fd;
129
130 if (!shared->readable || shared->path[0] == '\0') {
131 return NGX_DECLINED;
132 }
133
134 fd = brix_bound_confined_open(ctx, c, shared,
135 O_RDONLY | O_NOCTTY | O_CLOEXEC, &st);
136 if (fd < 0) {
137 return NGX_DECLINED;
138 }
139
140 file = &ctx->files[handle_index];
141 file->fd = fd;
142 file->readable = 1;
143 file->writable = 0;
144 file->from_cache = shared->from_cache ? 1 : 0;
145 file->is_regular = S_ISREG(st.st_mode) ? 1 : 0;
146 file->device = st.st_dev;
147 file->inode = st.st_ino;
148 file->cached_size = (off_t) st.st_size;
149 file->read_last_end = -1;
150 file->read_ahead_end = 0;
151 file->bytes_read = 0;
152 file->bytes_written = 0;
153 file->open_time = ngx_current_msec;
154
155 if (brix_set_fhandle_path(ctx, c, handle_index, shared->path)
156 != NGX_OK)
157 {
158 brix_free_fhandle(ctx, handle_index);
159 return NGX_ERROR;
160 }
161
162 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
163 "brix: bound handle=%d reopened shared path=%s",
164 handle_index, shared->path);
165 return NGX_OK;
166}

187/* brix_reopen_bound_write_handle — Phase 94: reopen a bound secondary's fd O_WRONLY
188 * to match the primary's published writable entry. Disjoint-offset pwrites from
189 * this fd are POSIX-safe against the primary's and other secondaries' fds on the
190 * same inode. NO O_CREAT/O_TRUNC — the primary already created the file at open, so
191 * a substream write only fills its own byte range. NGX_DECLINED on a device/inode
192 * change (path replaced since publish). */
193
194static ngx_int_t
195brix_reopen_bound_write_handle(brix_ctx_t *ctx, ngx_connection_t *c,
196 int handle_index, const brix_shared_handle_entry_t *shared)
197{
198 brix_file_t *file;
199 struct stat st;
200 int fd;
201
202 if (!shared->writable || shared->path[0] == '\0') {
203 return NGX_DECLINED;
204 }
205
206 fd = brix_bound_confined_open(ctx, c, shared,
207 O_WRONLY | O_NOCTTY | O_CLOEXEC, &st);
208 if (fd < 0) {
209 return NGX_DECLINED;
210 }
211
212 file = &ctx->files[handle_index];
213 file->fd = fd;
214 file->writable = 1;
215 /* Phase-105: the reopen above already refused unless this endpoint allows
216 * writes (see brix_ensure_write_handle), so the secondary's handle carries
217 * the same ALLOWED posture the primary's does — recorded, not re-derived,
218 * because the per-op gate reads only the handle. */
219 file->mutation_policy = BRIX_VFS_MUTATION_ALLOWED;
220 file->readable = shared->readable ? 1 : 0;
221 file->from_cache = shared->from_cache ? 1 : 0;
222 file->is_regular = S_ISREG(st.st_mode) ? 1 : 0;
223 file->device = st.st_dev;
224 file->inode = st.st_ino;
225 file->cached_size = (off_t) st.st_size;
226 file->read_last_end = -1;
227 file->read_ahead_end = 0;
228 file->bytes_read = 0;
229 file->bytes_written = 0;
230 file->open_time = ngx_current_msec;
231
232 if (brix_set_fhandle_path(ctx, c, handle_index, shared->path)
233 != NGX_OK)
234 {
235 brix_free_fhandle(ctx, handle_index);
236 return NGX_ERROR;
237 }
238
239 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
240 "brix: bound handle=%d reopened WRITABLE shared path=%s",
241 handle_index, shared->path);
242 return NGX_OK;
243}

245/* brix_ensure_read_handle — for an unbound session, NGX_OK iff the fd exists. For
246 * a bound secondary, re-validate against the primary's shared entry every time: if
247 * the primary closed/reused the slot, revoke (free + NGX_DECLINED); if stale but
248 * still published, reopen a fresh matching fd. Invariant: a bound secondary never
249 * reads from a handle no longer in the primary's published table. */
250
251ngx_int_t
252brix_ensure_read_handle(brix_ctx_t *ctx, ngx_connection_t *c,
253 int handle_index)
254{
255 brix_shared_handle_entry_t shared;
256
257 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES) {
258 return NGX_DECLINED;
259 }
260
261 if (!ctx->is_bound) {
262 /* A NULL table means no handle was ever opened on this session. */
263 if (ctx->files == NULL) {
264 return NGX_DECLINED;
265 }
266
267 /* fd >= 0 (POSIX) OR a driver-backed object/remote handle (no kernel fd;
268 * reads route through sd_obj.driver via the buffered serve path). */
269 return (ctx->files[handle_index].fd >= 0
270 || ctx->files[handle_index].sd_obj.driver != NULL)
271 ? NGX_OK : NGX_DECLINED;
272 }
273
274 /* A bound secondary materialises the primary's handle into its own table
275 * (slot hint + lazy reopen below), so the table must exist first. */
276 if (brix_files_ensure(ctx, c) != NGX_OK) {
277 return NGX_ERROR;
278 }
279
280 /*
281 * A bound stream only follows the primary session's current handle table.
282 * Re-check the shared slot on every read request so a primary close, reuse,
283 * or session teardown immediately revokes the secondary's local fd.
284 */
285 if (!brix_session_handle_lookup_hint(ctx->bound_sessid, handle_index,
286 &ctx->files[handle_index]
287 .shared_handle_slot_hint,
288 &shared))
289 {
290 if (ctx->files[handle_index].fd >= 0) {
291 brix_free_fhandle(ctx, handle_index);
292 }
293 return NGX_DECLINED;
294 }
295
296 if (brix_local_file_matches_shared_handle(&ctx->files[handle_index],
297 &shared))
298 {
299 return NGX_OK;
300 }
301
302 if (ctx->files[handle_index].fd >= 0
303 || ctx->files[handle_index].path != NULL)
304 {
305 brix_free_fhandle(ctx, handle_index);
306 }
307
308 return brix_reopen_bound_read_handle(ctx, c, handle_index, &shared);
309}

311/* brix_revalidate_bound_write_handle — the bound-secondary half of
312 * brix_ensure_write_handle. Re-checks the primary's published entry on every
313 * write: an absent entry or a published read-only one revokes the local handle
314 * (free + NGX_DECLINED); a stale-but-still-published one is reopened as a fresh
315 * matching O_WRONLY fd. Callers have already established that ctx is bound and
316 * that the server permits writes. */
317
318static ngx_int_t
319brix_revalidate_bound_write_handle(brix_ctx_t *ctx, ngx_connection_t *c,
320 int handle_index)
321{
322 brix_shared_handle_entry_t shared;
323
324 if (!brix_session_handle_lookup_hint(ctx->bound_sessid, handle_index,
325 &ctx->files[handle_index]
326 .shared_handle_slot_hint,
327 &shared))
328 {
329 if (ctx->files[handle_index].fd >= 0) {
330 brix_free_fhandle(ctx, handle_index);
331 }
332 return NGX_DECLINED;
333 }
334
335 /* A published read-only handle is a data channel a bound conn may only read. */
336 if (!shared.writable) {
337 if (ctx->files[handle_index].fd >= 0
338 && ctx->files[handle_index].writable)
339 {
340 brix_free_fhandle(ctx, handle_index);
341 }
342 return NGX_DECLINED;
343 }
344
345 if (brix_local_file_matches_shared_write_handle(&ctx->files[handle_index],
346 &shared))
347 {
348 return NGX_OK;
349 }
350
351 if (ctx->files[handle_index].fd >= 0
352 || ctx->files[handle_index].path != NULL)
353 {
354 brix_free_fhandle(ctx, handle_index);
355 }
356
357 return brix_reopen_bound_write_handle(ctx, c, handle_index, &shared);
358}

360/* brix_ensure_write_handle — Phase 94 write-side mirror of brix_ensure_read_handle.
361 * For an unbound (primary) session the write handle is already open locally, so
362 * NGX_OK iff a kernel fd / driver object / staged writer exists. For a bound
363 * secondary, re-validate against the primary's shared WRITABLE entry every write:
364 * a published-read-only or absent entry → revoke (NGX_DECLINED); a stale-but-still-
365 * published entry → reopen a fresh matching O_WRONLY fd. Invariant: a bound
366 * secondary never writes a handle the primary did not publish as writable. */
367ngx_int_t
368brix_ensure_write_handle(brix_ctx_t *ctx, ngx_connection_t *c,
369 int handle_index)
370{
371 ngx_stream_brix_srv_conf_t *conf;
372
373 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES) {
374 return NGX_DECLINED;
375 }
376
377 /*
378 * brix_read_only takes priority over EVERY write-side path, including this
379 * one, and it does so directly rather than by implication.
380 *
381 * On a read-only server no writable handle can be published in the first
382 * place (brix_open_mode_guard refuses a write open before a handle exists),
383 * so the shared-entry lookup below would already decline — but that is a
384 * DERIVED guarantee, and this function is reached from the recv-header hook
385 * (brix_recv_write_hdr_hook) which runs BEFORE brix_dispatch_require_write
386 * arms the allow_write gate. Checking allow_write here makes the ordering
387 * irrelevant: no read-only server ever reaches an O_WRONLY reopen, whatever
388 * a bound secondary sends and in whatever order.
389 */
390 conf = ngx_stream_get_module_srv_conf(ctx->session, ngx_stream_brix_module);
391 if (conf == NULL || !conf->common.allow_write) {
392 return NGX_DECLINED;
393 }
394
395 if (!ctx->is_bound) {
396 /* A NULL table means no handle was ever opened on this session. */
397 if (ctx->files == NULL) {
398 return NGX_DECLINED;
399 }
400
401 return (ctx->files[handle_index].fd >= 0
402 || ctx->files[handle_index].sd_obj.driver != NULL
403 || ctx->files[handle_index].writer != NULL)
404 ? NGX_OK : NGX_DECLINED;
405 }
406
407 /* A bound secondary materialises the primary's handle into its own table
408 * (slot hint + lazy reopen), so the table must exist first. */
409 if (brix_files_ensure(ctx, c) != NGX_OK) {
410 return NGX_ERROR;
411 }
412
413 return brix_revalidate_bound_write_handle(ctx, c, handle_index);
414}
