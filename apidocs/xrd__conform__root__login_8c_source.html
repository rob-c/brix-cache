32static int
33request(xrd_conform_execution *execution, brix_conn *connection,
34 ClientRequestHdr *header, const void *payload, uint16_t *wire_status,
35 brix_status *status)
36{
37 uint16_t sid;
38 uint8_t *body = NULL;
39 uint32_t length = 0;
40 brix_resp_out output = { wire_status, &body, &length };
41 char reason[64];
42 int rc;
43
44 if (!xrd_conform_reserve(execution, sizeof(*header), reason,
45 sizeof(reason))) return -2;
46 brix_status_clear(status);
47 if (brix_send(connection, header, payload, &sid, status) != 0) return -1;
48 rc = brix_recv(connection, sid, &output, status);
49 free(body);
50 return rc;
51}

87static void
88probe_session_ids(xrd_conform_execution *execution, const brix_url *url,
89 const brix_conn *primary)
90{
91 uint8_t ids[64][BRIX_SESSION_ID_LEN];
92 brix_conn connection;
93 brix_status status;
94 size_t count = 0;
95 size_t index;
96 int unique;
97 int increasing = 1;
98 int decreasing = 1;
99 int shared_prefix = 1;
100 int malformed_length = primary->login_body_len < BRIX_SESSION_ID_LEN;
101
102 memcpy(ids[count++], primary->sessid, BRIX_SESSION_ID_LEN);
103 while (count < 64 && connect_aux(execution, url, &connection, &status) == 0) {
104 memcpy(ids[count++], connection.sessid, BRIX_SESSION_ID_LEN);
105 brix_close(&connection);
106 }
107 if (strstr(status.msg, "login reply too short") != NULL)
108 malformed_length = 1;
109 for (index = 1; index < count; index++) {
110 int order = memcmp(ids[index - 1], ids[index], BRIX_SESSION_ID_LEN);
111
112 if (order >= 0) increasing = 0;
113 if (order <= 0) decreasing = 0;
114 if (memcmp(ids[0], ids[index], 8) != 0) shared_prefix = 0;
115 }
116 unique = session_ids_unique(ids, count);
117 xrd_conform_grade(execution, "LOG-001", count > 0 && !malformed_length,
118 "session-id-is-16-bytes",
119 "session-id-missing");
120 xrd_conform_grade(execution, "LOG-002", count >= 2 && unique,
121 "independent-session-ids-differ", "duplicate-session-id");
122 xrd_conform_grade(execution, "LOG-003", count == 64 && unique && !increasing
123 && !decreasing && !shared_prefix, "session-ids-pass-quality-checks",
124 count == 64 ? "predictable-or-low-entropy-session-ids"
125 : "session-id-sample-incomplete");
126}

205static void
206probe_endsess(xrd_conform_execution *execution, const brix_url *url)
207{
208 brix_conn first, second;
209 brix_status status;
210 uint8_t unknown[BRIX_SESSION_ID_LEN];
211 int current = 0;
212 int foreign = 0;
213 int random_noop = 0;
214
215 memset(unknown, 0xa5, sizeof(unknown));
216 if (connect_aux(execution, url, &first, &status) == 0) {
217 current = send_endsess(execution, &first, first.sessid)
218 && !session_can_stat(&first, url->path);
219 brix_close(&first);
220 }
221 if (connect_aux(execution, url, &first, &status) == 0
222 && connect_aux(execution, url, &second, &status) == 0) {
223 foreign = send_endsess(execution, &first, second.sessid)
224 && session_can_stat(&first, url->path);

226 && session_can_stat(&first, url->path);
227 brix_close(&second);
228 brix_close(&first);
229 }

231 "current-session-survived-endsess");
232 xrd_conform_grade_active(execution, "ENV-003", current, "current-session-ended",
233 "current-session-survived-endsess");
234 xrd_conform_grade_active(execution, "LOG-019", foreign,
235 "foreign-cleanup-preserved-carrier",
236 "foreign-cleanup-deauthenticated-carrier");
237 xrd_conform_grade_active(execution, "ENV-004", foreign,
238 "foreign-cleanup-preserved-carrier",
239 "foreign-cleanup-deauthenticated-carrier");

295/* WHAT: Reconcile the login security list with explicit working endpoints.
296 * WHY: Advertising a mechanism that no client can complete is a false claim;
297 * silently probing only the preferred mechanism would miss that defect.
298 * HOW: Grade the primary list shape, require an explicit fixture for every
299 * advertised name, force that protocol on its fixture, and verify the
300 * authentication driver records the same completed mechanism. */
301static void
302probe_security_claims(xrd_conform_execution *execution,
303 const brix_conn *primary)
304{
305 size_t index;
306 size_t complete = 0;
307 char received[96];
308 int covered;
309
310 xrd_conform_grade_active(execution, "LOG-004", strstr(primary->sec_list, "&P=") != NULL,
311 "security-continuation-names-mechanism",
312 "security-continuation-names-no-mechanism");
313 if (!xrd_conform_row_active(execution, "LOG-005")) return;
314 if (execution->run->auth_fixture_count == 0) {
315 xrd_conform_result_set(execution, "LOG-005",
316 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
317 "fixture-required:advertised-auth-protocols", NULL, NULL);
318 return;
319 }
320 covered = xrd_conform_auth_fixtures_cover(execution->run,
321 primary->sec_list);
322 for (index = 0; index < execution->run->auth_fixture_count; index++)
323 if (auth_fixture_completes(execution,
324 &execution->run->auth_fixtures[index])) complete++;
325 snprintf(received, sizeof(received), "protocols=%zu complete=%zu covered=%d",
326 execution->run->auth_fixture_count, complete, covered);
327 xrd_conform_result_set(execution, "LOG-005",
328 covered && complete == execution->run->auth_fixture_count
329 ? BRIX_CONFORM_VERDICT_CONFORMANT
330 : BRIX_CONFORM_VERDICT_DIVERGENT,
331 covered && complete == execution->run->auth_fixture_count
332 ? "all-advertised-auth-protocols-complete"
333 : "advertised-auth-protocol-false-or-uncovered",
334 "every advertised protocol has a forced successful fixture", received);
335}

337/* WHAT: Decide whether the protocol trailer claims request signing is needed.
338 * WHY: A security mechanism can be advertised while every signing requirement
339 * remains disabled; that endpoint must claim-skip SIG, not leave ten
340 * misleading probe-not-executed rows.
341 * HOW: Accept either a non-zero global level or a non-ignore secvec entry,
342 * after bounding the trailer against the captured protocol body. */
343static int
344signing_claimed(const brix_conn *primary)
345{
346 const uint8_t *body = primary->protocol_body;
347 size_t count;
348 size_t index;
349
350 if (primary->protocol_body_len < 14 || body[8] != 'S') return 0;
351 if (body[12] != 0) return 1;
352 count = body[13];
353 if (count > (primary->protocol_body_len - 14U) / 2U) return 0;
354 for (index = 0; index < count; index++)
355 if (body[15U + index * 2U] != 0) return 1;
356 return 0;
357}

384static void
385finalize_mechanism_specific_rows(xrd_conform_execution *execution,
386 const brix_conn *primary)
387{
388 static const char *const gsi_ids[] = {
389 "LOG-009", "LOG-010", "LOG-011", "LOG-012"
390 };
391 static const char *const token_ids[] = {
392 "LOG-022", "LOG-023", "LOG-024"
393 };
394 int have_gsi = brix_sec_proto_advertised(primary->sec_list, "gsi",
395 NULL, 0);
396 int have_ztn = brix_sec_proto_advertised(primary->sec_list, "ztn",
397 NULL, 0);
398 int have_ztn_fixture = 0;
399 int working_ztn_fixture = 0;
400 size_t index;
401
402 for (index = 0; index < sizeof(gsi_ids) / sizeof(gsi_ids[0]); index++)
403 set_if_pending(execution, gsi_ids[index],
404 have_gsi ? BRIX_CONFORM_VERDICT_INCONCLUSIVE
405 : BRIX_CONFORM_VERDICT_SKIPPED,
406 have_gsi ? "fixture-required:gsi-auth-transcript"
407 : "claim-absent:gsi");
408 for (index = 0; index < execution->run->auth_fixture_count; index++) {
409 const xrd_conform_auth_fixture *fixture =
410 &execution->run->auth_fixtures[index];
411
412 if (strcmp(fixture->protocol, "ztn") != 0) continue;
413 have_ztn_fixture = 1;
414 working_ztn_fixture = auth_fixture_completes(execution, fixture);
415 }
416 if (have_ztn)
417 xrd_conform_grade_active(execution, "LOG-021", 1,
418 "bearer-auth-advertised-as-ztn", NULL);
419 else if (have_ztn_fixture && working_ztn_fixture)
420 xrd_conform_grade_active(execution, "LOG-021", 0,
421 "bearer-auth-advertised-as-ztn",
422 "working-bearer-fixture-not-advertised-as-ztn");
423 else if (have_ztn_fixture)
424 set_if_pending(execution, "LOG-021", BRIX_CONFORM_VERDICT_INCONCLUSIVE,
425 "ztn-fixture-did-not-complete");
426 else
427 set_if_pending(execution, "LOG-021", BRIX_CONFORM_VERDICT_SKIPPED,
428 "claim-absent:ztn");
429 for (index = 0; index < sizeof(token_ids) / sizeof(token_ids[0]); index++)
430 set_if_pending(execution, token_ids[index],
431 have_ztn ? BRIX_CONFORM_VERDICT_INCONCLUSIVE
432 : BRIX_CONFORM_VERDICT_SKIPPED,
433 have_ztn ? "fixture-required:token-claim-negative"
434 : "claim-absent:ztn");
435}

479/* WHAT: Exercise the signed success and unsigned-refusal sides of the policy.
480 * WHY: SIG-001/002 are observable with the production client once GSI retained
481 * a session cipher; accepting an unsigned request would defeat the claim.
482 * HOW: Use isolated authenticated sessions. The positive keeps automatic
483 * signing enabled; the negative disables only the client-side prefix and
484 * requires the protocol's signature-verification error spelling. */
485static void
486probe_signature_baseline(xrd_conform_execution *execution,
487 const brix_url *url)
488{
489 brix_conn connection;
490 brix_statinfo info;
491 brix_status status;
492 int positive = 0;
493 int negative = 0;
494
495 if (connect_aux(execution, url, &connection, &status) == 0) {
496 if (connection.signing_active)
497 positive = brix_stat(&connection, url->path, &info, &status) == 0;
498 brix_close(&connection);
499 }
500 if (connect_aux(execution, url, &connection, &status) == 0) {
501 if (connection.signing_active) {
502 connection.signing_active = 0;
503 brix_status_clear(&status);
504 negative = brix_stat(&connection, url->path, &info, &status) != 0
505 && status.kxr == kXR_SigVerErr;
506 }
507 brix_close(&connection);
508 }
509 xrd_conform_grade_active(execution, "SIG-001", negative,
510 "unsigned-required-request-returned-sigver-error",
511 "unsigned-required-request-not-classified-as-sigver-error");
512 xrd_conform_grade_active(execution, "SIG-002", positive,
513 "correctly-signed-request-served",
514 "correctly-signed-request-refused");
515}

517static void
518probe_signature_family(xrd_conform_execution *execution, const brix_url *url,
519 const brix_conn *primary)
520{
521 if (!xrd_conform_family_active(execution, "SIG")) return;
522 if (primary->sec_list[0] != '\0' && !signing_claimed(primary)) {
523 set_signature_family(execution, BRIX_CONFORM_VERDICT_SKIPPED,
524 "claim-absent:request-signing");
525 return;
526 }
527 if (!signing_claimed(primary)) return;
528 if (!primary->signing_active) {
529 set_signature_family(execution, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
530 "signing-claim-present:no-session-signing-key");
531 return;
532 }
533 probe_signature_baseline(execution, url);
534 xrd_conform_probe_sigver_active(execution, url);
535 xrd_conform_probe_sigver_bound(execution, url);
536 if (primary->protocol_body_len < 14
537 || primary->protocol_body[8] != 'S'
538 || !(primary->protocol_body[11] & kXR_secOFrce)) {
539 xrd_conform_result_set(execution, "SIG-010",
540 BRIX_CONFORM_VERDICT_SKIPPED,
541 "claim-absent:secOFrce", NULL, NULL);
542 return;
543 }
544 xrd_conform_probe_sigver_unencrypted(execution, url);
545}
