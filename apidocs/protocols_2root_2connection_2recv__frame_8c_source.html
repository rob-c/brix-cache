13 * recv_frame.c — the READ side of the recv framing loop (see recv_frame.h):
14 * deferred-request drain, fresh-request housekeeping, non-XRootD handoff, and
15 * reading/accumulating the next PDU unit. The process side (buffer management,
16 * body extension, header/payload dispatch) lives in recv_process.c. Bodies are
17 * the original ngx_stream_brix_recv loop-body blocks moved verbatim; only
18 * loop-exit statements (break/continue/return) became step codes.
19 */

21brix_recv_step_t
22brix_recv_run_deferred(ngx_stream_session_t *s, ngx_connection_t *c,
23 ngx_stream_brix_srv_conf_t *conf, brix_ctx_t *ctx, ngx_event_t *rev,
24 size_t *rx_pending)
25{
26 ngx_int_t rc;
27
28 /*
29 * A non-pipelinable request was fully read (header in cur_*, any payload in
30 * payload_buf) while pipelined reads OR writes were still in flight. Keep
31 * it parked until BOTH the ack/response queue (out.count) and the in-flight
32 * pwrites (wr_inflight) have drained — a write completion or the send-side
33 * drain re-enters recv to re-check. This is what lets a kXR_close safely
34 * follow a burst of pipelined writes: every pwrite lands before the retire.
35 * phase-32 WS3: also hold until in-flight cold reads (rd.aio_inflight) drain,
36 * so the deferred op never runs while a read worker still holds a handle.
37 */
38 if (ctx->out.count > 0 || ctx->out.wr_inflight > 0
39 || ctx->rd.aio_inflight > 0)
40 {
41 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
42 return BRIX_RECV_STEP_BREAK;
43 }
44 return BRIX_RECV_STEP_RETURN;
45 }
46
47 ctx->out.recv_deferred = 0;
48
49 /*
50 * Reinstall the parked request's stream-id: a pipelined pwrite/read ack
51 * completing during the park went through brix_aio_restore_stream(), which
52 * overwrote recv.cur_streamid with the completed op's sid. Dispatching the
53 * deferred request with that stale sid misdirects its reply onto a sid the
54 * client has already retired (and may have recycled for a newer request),
55 * while the deferred request itself never gets answered — the client's
56 * stream timeout then kills the whole multiplexed connection.
57 */
58 ctx->recv.cur_streamid[0] = ctx->out.deferred_streamid[0];
59 ctx->recv.cur_streamid[1] = ctx->out.deferred_streamid[1];
60
61 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
62 *rx_pending = 0;
63 rc = brix_dispatch(ctx, c, conf);
64 if (rc == NGX_ERROR) {
65 return BRIX_RECV_STEP_BREAK;
66 }
67 if (ctx->state == XRD_ST_AIO) {
68 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
69 return BRIX_RECV_STEP_BREAK;
70 }
71 return BRIX_RECV_STEP_RETURN;
72 }
73 if (ctx->state != XRD_ST_SENDING) {
74 if (ctx->tls_pending) {
75 brix_start_tls(ctx, c, conf);
76 return BRIX_RECV_STEP_RETURN;
77 }
78 ctx->state = XRD_ST_REQ_HEADER;
79 ctx->recv.hdr_pos = 0;
80 }
81 return BRIX_RECV_STEP_CONTINUE;
82}

84/*
85 * Fresh-request-boundary housekeeping at the top of a REQ_HEADER read (hdr_pos
86 * == 0): apply pipelining backpressure, trim grown scratch, reconcile the heap
87 * budget, and handle graceful-quit teardown of an idle keepalive. RETURN when
88 * the connection suspended or was torn down; NEXT to proceed reading the header.
89 */
90static brix_recv_step_t
91brix_recv_header_prep(ngx_stream_session_t *s, ngx_connection_t *c,
92 brix_ctx_t *ctx, size_t rx_pending)
93{
94 /*
95 * Phase 29 pipelining backpressure: if pipeline_depth responses are already
96 * queued/draining (out.count) OR pwrites are still in flight (wr_inflight),
97 * stop reading new requests and suspend with state=SENDING. Bounding
98 * out.count + wr_inflight at pipeline_depth caps BOTH in-flight pwrites and
99 * queued acks, so the out_ring can never overflow. (Both counters are 0 for
100 * the common serial case, so this is a no-op there.)
101 *
102 * phase-32 WS3: cold reads in flight (rd.aio_inflight) each hold a rd_pool
103 * buffer and will each become a queued response, so count them here too —
104 * this is what caps the concurrent-read depth (and guarantees a free pool
105 * slot / out_ring slot for the next read the recv loop admits).
106 */
107 if (ctx->out.count + ctx->out.wr_inflight + ctx->rd.aio_inflight
108 >= ctx->out.pipeline_depth)
109 {
110 ctx->state = XRD_ST_SENDING;
111 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, rx_pending);
112 return BRIX_RECV_STEP_RETURN;
113 }
114
115 /*
116 * Top of a fresh request with the queue empty — the previous response is
117 * fully sent, so no chain references the scratch buffers. Phase 31: trim
118 * scratch a prior large read/pgwrite grew back to the streaming window, then
119 * reconcile the transfer-heap footprint with the SHM-global budget. Gated
120 * on out.count==0 so a pipelined read still in flight is never disturbed.
121 * phase-32 WS3: also gate on rd.aio_inflight==0 — a cold read on a worker
122 * thread still owns a pool buffer, and the ngx_exiting idle-close below would
123 * otherwise free it (via brix_on_disconnect) mid-pread.
124 */
125 if (ctx->out.count == 0 && ctx->out.wr_inflight == 0
126 && ctx->rd.aio_inflight == 0)
127 {
128 brix_trim_scratch(ctx, c);
129 brix_budget_sync(ctx);
130
131 /*
132 * Fast teardown: the worker is draining and this connection is quiescent
133 * at a fresh request boundary (no queued response, no in-flight pwrite).
134 * Close now rather than parking; the client reconnects to the new worker
135 * and resumes. BUT only when no file is open: a connection holding an
136 * open handle is mid-transfer (a streaming read parked between chunks, or
137 * a cache slice-fill in progress) — forcing a reconnect there loses the
138 * in-flight fill and surfaces as a spurious kXR_NotFound. Let the active
139 * transfer finish on the old worker (bounded by worker_shutdown_timeout).
140 */
141 ngx_flag_t has_open_file = brix_ctx_has_open_file(ctx);
142
143 brix_shutdown_hold_sync(c, ctx, has_open_file);
144
145 if (ngx_exiting && !has_open_file) {
146 brix_on_disconnect(ctx, c);
147 brix_close_all_files(ctx);
148 ngx_stream_finalize_session(s, NGX_STREAM_OK);
149 return BRIX_RECV_STEP_RETURN;
150 }
151
152 /*
153 * Mark idle so a graceful quit's ngx_close_idle_connections() drops this
154 * parked keepalive immediately. An open handle means an active transfer,
155 * not a keepalive idle — leave it un-idle so the reload lets it complete.
156 */
157 c->idle = has_open_file ? 0 : 1;
158 }
159
160 return BRIX_RECV_STEP_NEXT;
161}

163/*
164 * The handshake's first byte is non-zero — not an XRootD client hello (the
165 * client hello begins with a zero streamid word); it is an HTTP method letter or
166 * a TLS 0x16. Splice to the local HTTP/WebDAV listener when brix_http_handoff
167 * is configured (one port serves both protocols), else close. RETURN when the
168 * relay took the connection; BREAK to close it.
169 */
170static brix_recv_step_t
171brix_recv_handle_nonxrootd(ngx_stream_session_t *s, ngx_connection_t *c,
172 ngx_stream_brix_srv_conf_t *conf, brix_ctx_t *ctx)
173{
174 if (conf->http_handoff_addr != NULL
175 && brix_http_handoff_start(s, c, conf,
176 ctx->recv.hdr_buf, ctx->recv.hdr_pos) == NGX_OK)
177 {
178 return BRIX_RECV_STEP_RETURN; /* the relay owns the connection now */
179 }
180 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
181 "brix: non-XRootD client (first byte 0x%02xd)"
182 " closing immediately",
183 (unsigned) ctx->recv.hdr_buf[0]);
184 return BRIX_RECV_STEP_BREAK;
185}

187/*
188 * Fold `avail` newly-received bytes into the current PDU state and decide the
189 * loop action: detect a non-XRootD handshake byte (→ handle_nonxrootd), CONTINUE
190 * on a short read (more of this unit still to come), or NEXT when the unit is now
191 * complete.
192 */
193static brix_recv_step_t
194brix_recv_advance(ngx_stream_session_t *s, ngx_connection_t *c,
195 ngx_stream_brix_srv_conf_t *conf, brix_ctx_t *ctx, size_t avail,
196 size_t need, size_t *rx_pending)
197{
198 *rx_pending += avail;
199
200 if (ctx->state == XRD_ST_HANDSHAKE) {
201 ctx->recv.hdr_pos += avail;
202
203 if (ctx->recv.hdr_pos >= 1 && ctx->recv.hdr_buf[0] != 0) {
204 return brix_recv_handle_nonxrootd(s, c, conf, ctx);
205 }
206
207 } else if (ctx->state == XRD_ST_REQ_HEADER) {
208 ctx->recv.hdr_pos += avail;
209
210 } else {
211 ctx->recv.payload_pos += avail;
212 }
213
214 if (avail < need) {
215 return BRIX_RECV_STEP_CONTINUE;
216 }
217
218 return BRIX_RECV_STEP_NEXT;
219}

221/*
222 * Buffered frame read. Exact-size frame reads cost two extra syscalls each
223 * inside ngx_unix_recv (an ioctl(FIONREAD) follows every recv that exactly
224 * fills its buffer) plus a separate recv per framing stage. Once a session
225 * can no longer be handed to a raw-socket consumer — auth complete (no
226 * handshake sniffing, no post-protocol STARTTLS, and a pre-bind secondary
227 * never qualifies, which keeps §1.4 fd migration exact-read), and no
228 * upstream/relay configured (those subsystems read the client socket
229 * directly and would never see stashed bytes) — pull one big recv into a
230 * per-connection stash and feed the framing stages from it: a whole
231 * metadata request (often several pipelined ones) costs ONE partial recv
232 * that nginx never follows with an ioctl. Reads of at least the stash size
233 * (bulk write payloads) bypass the stash straight into their destination —
234 * no bounce copy — but only once the stash is drained, so byte order is
235 * preserved. Returns exactly what c->recv returns (bytes / NGX_AGAIN /
236 * NGX_ERROR / 0), so the caller's result handling is unchanged.
237 */
238static ssize_t
239brix_recv_buffered(brix_ctx_t *ctx, ngx_connection_t *c,
240 ngx_stream_brix_srv_conf_t *conf, u_char *dest, size_t need)
241{
242 brix_ctx_recv_t *rv = &ctx->recv;
243 size_t have = (size_t) (rv->stash_len - rv->stash_head);
244 ssize_t n;
245
246 if (have == 0) {
247 if (need >= BRIX_RECV_STASH_SIZE
248 || !ctx->login.auth_done
249 || conf->upstream_host.len > 0
250 || conf->relay_addr != NULL)
251 {
252 return c->recv(c, dest, need);
253 }
254
255 if (rv->stash == NULL) {
256 rv->stash = ngx_palloc(c->pool, BRIX_RECV_STASH_SIZE);
257 if (rv->stash == NULL) {
258 return c->recv(c, dest, need);
259 }
260 }
261
262 n = c->recv(c, rv->stash, BRIX_RECV_STASH_SIZE);
263 if (n <= 0) {
264 return n; /* NGX_AGAIN / NGX_ERROR / closed */
265 }
266 rv->stash_head = 0;
267 rv->stash_len = (uint32_t) n;
268 have = (size_t) n;
269 }
270
271 if (have > need) {
272 have = need;
273 }
274 ngx_memcpy(dest, rv->stash + rv->stash_head, have);
275 rv->stash_head += (uint32_t) have;
276 if (rv->stash_head == rv->stash_len) {
277 rv->stash_head = 0;
278 rv->stash_len = 0;
279 }
280 return (ssize_t) have;
281}

283brix_recv_step_t
284brix_recv_read_frame(ngx_stream_session_t *s, ngx_connection_t *c,
285 ngx_stream_brix_srv_conf_t *conf, brix_ctx_t *ctx, ngx_event_t *rev,
286 size_t *rx_pending)
287{
288 u_char *dest;
289 size_t need;
290 ssize_t n;
291
292 if (ctx->state == XRD_ST_HANDSHAKE) {
293 dest = ctx->recv.hdr_buf + ctx->recv.hdr_pos;
294 need = XRD_HANDSHAKE_LEN - ctx->recv.hdr_pos;
295
296 } else if (ctx->state == XRD_ST_REQ_HEADER) {
297 if (ctx->recv.hdr_pos == 0) {
298 brix_recv_step_t st = brix_recv_header_prep(s, c, ctx, *rx_pending);
299 if (st != BRIX_RECV_STEP_NEXT) {
300 return st;
301 }
302 }
303 dest = ctx->recv.hdr_buf + ctx->recv.hdr_pos;
304 need = XRD_REQUEST_HDR_LEN - ctx->recv.hdr_pos;
305
306 } else {
307 /* cur_body_extra is non-zero only for kXR_writev (segment data streams
308 * after the dlen-framed descriptor block) and for kXR_chkpoint/ckpXeq
309 * (the sub-request body streams after the dlen-framed sub-header). */
310 dest = ctx->recv.payload + ctx->recv.payload_pos;
311 need = (size_t) ctx->recv.cur_dlen + ctx->recv.cur_body_extra
312 - ctx->recv.payload_pos;
313 }
314
315 if (need > 0) {
316 /* No forced rev->available reset here: -1 made ngx_unix_recv follow
317 * every exact-fill read with an ioctl(FIONREAD) this loop never
318 * consulted (it reads to EAGAIN regardless of rev->ready). */
319 n = brix_recv_buffered(ctx, c, conf, dest, need);
320
321 if (n == NGX_AGAIN) {
322 ngx_log_debug5(NGX_LOG_DEBUG_STREAM, c->log, 0,
323 "brix: recv AGAIN st=%d hdr_pos=%uz avail=%d"
324 " ready=%d active=%d",
325 (int) ctx->state, ctx->recv.hdr_pos,
326 rev->available, (int) rev->ready,
327 (int) rev->active);
328 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
329 return BRIX_RECV_STEP_BREAK;
330 }
331 /* Phase 39: genuine incompletion — arm the read deadline once
332 * (idempotent, so repeated partial reads of the same PDU do NOT
333 * reset it), bounding time-to-complete against a slowloris. */
334 brix_arm_read_deadline(c, ctx);
335 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
336 return BRIX_RECV_STEP_RETURN;
337 }
338
339 if (n == NGX_ERROR || n == 0) {
340 ngx_log_debug0(NGX_LOG_DEBUG_STREAM, c->log, 0,
341 "brix: client disconnected");
342 if (brix_defer_teardown_if_writing(ctx, c, NGX_STREAM_OK)) {
343 return BRIX_RECV_STEP_RETURN;
344 }
345 brix_on_disconnect(ctx, c);
346 brix_close_all_files(ctx);
347 ngx_stream_finalize_session(s, NGX_STREAM_OK);
348 return BRIX_RECV_STEP_RETURN;
349 }
350
351 return brix_recv_advance(s, c, conf, ctx, (size_t) n, need, rx_pending);
352 }
353
354 return BRIX_RECV_STEP_NEXT;
355}
