1/*
2 * sd_s3_batch.c - S3 DeleteObjects: one signed POST for up to 1,000 keys.
3 *
4 * WHAT: sd_s3_delete_many(), the batch twin of sd_s3_delete() (sd_s3.c). Builds
5 * the <Delete> XML body, signs "POST /bucket?delete" with SigV4, sends it
6 * once, and maps the per-key <Error> entries back onto the caller's
7 * result vector.
8 * WHY: Phase-107 C4: a DeleteObjects of 1,000 keys over an s3:// backend was
9 * 1,000 signed DELETE round trips whose whole purpose was to avoid
10 * exactly that. The S3 protocol has carried the batch verb since day one;
11 * this driver just never spoke it.
12 * HOW: Quiet mode (<Quiet>true</Quiet>): the origin answers ONLY the failures,
13 * so success is the absence of an <Error> block, and a 1,000-key
14 * all-green response is a few hundred bytes instead of ~100 KB. AWS
15 * requires Content-MD5 on this verb (MinIO enforces it as
16 * MissingContentMD5), so the body digest rides as a SIGNED header via
17 * sd_s3_sign_ext - which also lets an STS session token join the signed
18 * set, unlike the fixed-shape signer. ngx-free (transport-injected) like
19 * every sibling in this directory.
20 */

30/* Bytes the XML wrapper needs per key, beyond the escaped key text itself. */
31#define S3_BATCH_PER_KEY_OVERHEAD (sizeof("<Object><Key></Key></Object>") - 1)
32
33/* Escaped length of one key: '&'->&amp; '<'->&lt; '>'->&gt; '"'->&quot;
34 * '\''->&apos;. The five XML predefined entities - a key is client-supplied
35 * text and MUST NOT be able to open or close markup in the body we sign. */
36static size_t
37s3_batch_escape_len(const char *s)
38{
39 size_t n = 0;
40
41 for (; *s != '\0'; s++) {
42 switch (*s) {
43 case '&': n += sizeof("&amp;") - 1; break;
44 case '<': n += sizeof("&lt;") - 1; break;
45 case '>': n += sizeof("&gt;") - 1; break;
46 case '"': n += sizeof("&quot;") - 1; break;
47 case '\'': n += sizeof("&apos;") - 1; break;
48 default: n += 1; break;
49 }
50 }
51 return n;
52}

54/* Append the XML-escaped key at dst; the caller sized dst via
55 * s3_batch_escape_len, so this cannot overflow. Returns bytes written. */
56static size_t
57s3_batch_escape(char *dst, const char *s)
58{
59 size_t off = 0;
60
61 for (; *s != '\0'; s++) {
62 const char *rep = NULL;
63
64 switch (*s) {
65 case '&': rep = "&amp;"; break;
66 case '<': rep = "&lt;"; break;
67 case '>': rep = "&gt;"; break;
68 case '"': rep = "&quot;"; break;
69 case '\'': rep = "&apos;"; break;
70 default: dst[off++] = *s; continue;
71 }
72 memcpy(dst + off, rep, strlen(rep));
73 off += strlen(rep);
74 }
75 return off;
76}

78/* malloc the <Delete> body for `keys[n]`; *blen_out = body length. NULL on
79 * OOM. Quiet mode - the response then carries only the failures. */
80static char *
81s3_batch_build_body(const char *const *keys, size_t n, size_t *blen_out)
82{
83 static const char head[] =
84 "<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
85 "<Delete xmlns=\"http://s3.amazonaws.com/doc/2006-03-01/\">"
86 "<Quiet>true</Quiet>";
87 static const char tail[] = "</Delete>";
88 char *body;
89 size_t cap, off, i;
90
91 cap = sizeof(head) - 1 + sizeof(tail) - 1 + 1;
92 for (i = 0; i < n; i++) {
93 cap += S3_BATCH_PER_KEY_OVERHEAD + s3_batch_escape_len(keys[i]);
94 }
95 body = malloc(cap);
96 if (body == NULL) {
97 return NULL;
98 }
99 memcpy(body, head, sizeof(head) - 1);
100 off = sizeof(head) - 1;
101 for (i = 0; i < n; i++) {
102 memcpy(body + off, "<Object><Key>", sizeof("<Object><Key>") - 1);
103 off += sizeof("<Object><Key>") - 1;
104 off += s3_batch_escape(body + off, keys[i]);
105 memcpy(body + off, "</Key></Object>", sizeof("</Key></Object>") - 1);
106 off += sizeof("</Key></Object>") - 1;
107 }
108 memcpy(body + off, tail, sizeof(tail) - 1);
109 off += sizeof(tail) - 1;
110 body[off] = '\0';
111 *blen_out = off;
112 return body;
113}

115/* base64(MD5(body)) -> out[25] ("" on failure): the RFC 1864 Content-MD5 value
116 * AWS requires on DeleteObjects. Same EVP shape as sd_http_content_md5. */
117static void
118s3_batch_content_md5(const void *buf, size_t len, char out[25])
119{
120 unsigned char md[EVP_MAX_MD_SIZE];
121 unsigned int mdlen = 0;
122
123 out[0] = '\0';
124 if (EVP_Digest(buf, len, md, &mdlen, EVP_md5(), NULL) != 1 || mdlen != 16) {
125 return;
126 }
127 EVP_EncodeBlock((unsigned char *) out, md, 16); /* 16 -> 24 chars + NUL */
128}

130/* One S3 <Error><Code> -> errno. AccessDenied is the security-relevant one
131 * (the caller answers it per key, never for the whole batch); NoSuchKey stays
132 * ENOENT verbatim - the CALLER decides idempotency (sd_batch_types.h). */
133static int
134s3_batch_errno_from_code(const char *code, size_t len)
135{
136 if (len == sizeof("AccessDenied") - 1
137 && memcmp(code, "AccessDenied", len) == 0)
138 {
139 return EACCES;
140 }
141 if ((len == sizeof("NoSuchKey") - 1
142 && memcmp(code, "NoSuchKey", len) == 0)
143 || (len == sizeof("NoSuchBucket") - 1
144 && memcmp(code, "NoSuchBucket", len) == 0))
145 {
146 return ENOENT;
147 }
148 return EIO;
149}

151/* Decode the XML-escaped `src[len]` into dst[dstcap] (NUL-terminated); the
152 * origin echoes each failed key exactly as escaped in our request body. */
153static void
154s3_batch_unescape(const char *src, size_t len, char *dst, size_t dstcap)
155{
156 static const struct { const char *ent; size_t n; char ch; } tab[] = {
157 { "&amp;", 5, '&' }, { "&lt;", 4, '<' }, { "&gt;", 4, '>' },
158 { "&quot;", 6, '"' }, { "&apos;", 6, '\'' },
159 };
160 size_t i = 0, o = 0, t;
161
162 while (i < len && o + 1 < dstcap) {
163 if (src[i] == '&') {
164 for (t = 0; t < sizeof(tab) / sizeof(tab[0]); t++) {
165 if (len - i >= tab[t].n
166 && memcmp(src + i, tab[t].ent, tab[t].n) == 0)
167 {
168 break;
169 }
170 }
171 if (t < sizeof(tab) / sizeof(tab[0])) {
172 dst[o++] = tab[t].ch;
173 i += tab[t].n;
174 continue;
175 }
176 }
177 dst[o++] = src[i++];
178 }
179 dst[o] = '\0';
180}

182/* Pull the text of `<tag>` out of the error block [blk, blk_end). Returns the
183 * start and sets *tlen, or NULL when the tag is absent. */
184static const char *
185s3_batch_block_tag(const char *blk, const char *blk_end, const char *open,
186 const char *close, size_t *tlen)
187{
188 const char *s, *e;
189
190 s = strstr(blk, open);
191 if (s == NULL || s >= blk_end) {
192 return NULL;
193 }
194 s += strlen(open);
195 e = strstr(s, close);
196 if (e == NULL || e > blk_end) {
197 return NULL;
198 }
199 *tlen = (size_t) (e - s);
200 return s;
201}

203/* Walk the quiet-mode <DeleteResult> and set errs[i] for every <Error> block
204 * whose <Key> matches one of ours. An unmatched or malformed block is a
205 * response-integrity failure worth surfacing on the WHOLE batch (-1/EIO):
206 * a per-key verdict we cannot attribute must not be silently dropped. */
207static int
208s3_batch_apply_errors(const char *xml, const char *const *keys, size_t n,
209 int *errs)
210{
211 const char *p = xml, *blk, *blk_end, *k, *c;
212 char raw[SD_S3_KEY_MAX];
213 size_t klen = 0, clen = 0, i;
214
215 while ((blk = strstr(p, "<Error>")) != NULL) {
216 blk_end = strstr(blk, "</Error>");
217 if (blk_end == NULL) {
218 return -1;
219 }
220 k = s3_batch_block_tag(blk, blk_end, "<Key>", "</Key>", &klen);
221 c = s3_batch_block_tag(blk, blk_end, "<Code>", "</Code>", &clen);
222 if (k == NULL || c == NULL) {
223 return -1;
224 }
225 s3_batch_unescape(k, klen, raw, sizeof(raw));
226 for (i = 0; i < n; i++) {
227 if (strcmp(raw, keys[i]) == 0) {
228 errs[i] = s3_batch_errno_from_code(c, clen);
229 break;
230 }
231 }
232 if (i == n) {
233 return -1; /* a verdict for a key we never sent */
234 }
235 p = blk_end + sizeof("</Error>") - 1;
236 }
237 return 0;
238}

240/*
241 * sd_s3_delete_many - delete up to 1,000 object keys in ONE signed request.
242 *
243 * p->key is the "/bucket" request target (no object part); keys[i] are the
244 * object keys WITHIN the bucket (no leading '/'). errs[n] receives 0 or a
245 * positive errno per key - S3 DeleteObjects is idempotent, so an absent key
246 * comes back deleted (errs 0), while AccessDenied stays per-key. On success
247 * *done = n (the origin attempted every key). On a batch-level failure (sign /
248 * transport / non-200 / unparseable response) *done = 0, errno is set, errbuf
249 * describes it, and errs is untouched: the caller pre-filled it, and no key
250 * was verifiably attempted. Returns 0 / -1.
251 */
252/* Build, MD5-seal, SigV4-sign, and POST the <Delete> body for keys[0..n)
253 * over f's transport. On success the response (whatever its status) is handed
254 * back in *resp for the caller to judge and free; -1 with errno set on an
255 * alloc, sign, or transport failure (errbuf describes it, *resp untouched). */
256static int
257s3_batch_post(sd_s3_file *f, const char *const *keys, size_t n,
258 brix_s3_resp_t *resp, char *errbuf, size_t errcap)
259{
260 sd_s3_sign_hdr_t extra[2];
261 sd_s3_sign_req_t req = { "POST", "delete=", extra, 1 };
262 char auth[SD_S3_AUTH_HDRS_CAP];
263 char wire[SD_S3_KEY_MAX + 16];
264 char md5b64[25];
265 char *body;
266 size_t blen = 0;
267 int pn;
268
269 body = s3_batch_build_body(keys, n, &blen);
270 if (body == NULL) {
271 errno = ENOMEM;
272 return -1;
273 }
274 s3_batch_content_md5(body, blen, md5b64);
275 extra[0].name = "content-md5";
276 extra[0].value = md5b64;
277 if (f->session_token[0] != '\0') {
278 extra[1].name = "x-amz-security-token";
279 extra[1].value = f->session_token;
280 req.n_extra = 2;
281 }
282 pn = snprintf(wire, sizeof(wire), "%s?delete", f->key);
283 if (pn < 0 || (size_t) pn >= sizeof(wire) || md5b64[0] == '\0'
284 || sd_s3_sign_ext(f, &req, auth, sizeof(auth)) != 0)
285 {
286 sd_s3_set_err(errbuf, errcap, "s3 DeleteObjects: SigV4 sign failed");
287 free(body);
288 errno = EIO;
289 return -1;
290 }
291 if (f->transport->request(f->tctx, f->host, f->port, f->tls, "POST", wire,
292 auth, body, blen, f->timeout_ms, resp,
293 errbuf, errcap) != 0)
294 {
295 if (errno == 0) { errno = EIO; }
296 free(body);
297 return -1;
298 }
299 free(body);
300 return 0;
301}

303int
304sd_s3_delete_many(const sd_s3_open_params *p, const char *const *keys,
305 size_t n, int *errs, size_t *done, char *errbuf, size_t errcap)
306{
307 sd_s3_file *f;
308 const void *rbody;
309 size_t rlen = 0, i;
310 brix_s3_resp_t resp;
311 int rc;
312
313 *done = 0;
314 if (n == 0) {
315 return 0;
316 }
317 f = sd_s3_open_read(p, errbuf, errcap); /* handle only; no I/O */
318 if (f == NULL) {
319 errno = ENOMEM;
320 return -1;
321 }
322 if (s3_batch_post(f, keys, n, &resp, errbuf, errcap) != 0) {
323 sd_s3_close(f);
324 return -1;
325 }
326 if (resp.status != 200) {
327 rc = sd_s3_status_err(resp.status, "DeleteObjects", f->key,
328 errbuf, errcap);
329 f->transport->resp_free(&resp);
330 sd_s3_close(f);
331 return rc;
332 }
333 for (i = 0; i < n; i++) {
334 errs[i] = 0;
335 }
336 rbody = f->transport->resp_body(&resp, &rlen);
337 if (rbody != NULL && rlen > 0
338 && s3_batch_apply_errors((const char *) rbody, keys, n, errs) != 0)
339 {
340 sd_s3_set_err(errbuf, errcap,
341 "s3 DeleteObjects: unattributable <Error> in response");
342 f->transport->resp_free(&resp);
343 sd_s3_close(f);
344 errno = EIO;
345 return -1;
346 }
347 f->transport->resp_free(&resp);
348 sd_s3_close(f);
349 *done = n;
350 return 0;
351}
