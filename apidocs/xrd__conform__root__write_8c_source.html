74truncate_handle(xrd_conform_execution *execution, brix_conn *connection,
75 const brix_file *file, int64_t size, brix_status *status)
76{
77 ClientTruncateRequest request;
78 xrdw_truncate_req_t fields = { .offset = size };
79 uint16_t wire_status = 0;
80 uint8_t *body = NULL;
81 uint32_t body_length = 0;
82 brix_resp_out response = { &wire_status, &body, &body_length };
83 int rc;
84
85 if (!xrd_conform_reserve_status(execution, 0, status)) return -1;

89 xrdw_truncate_req_pack(&fields, ((ClientRequestHdr *) &request)->body);

309static void
310probe_writev_sync(xrd_conform_execution *execution, brix_conn *connection,
311 const xrd_conform_scratch_object *object)
312{
313 static const uint8_t left[] = "left";
314 static const uint8_t right[] = "right";
315 brix_writev_seg segments[3] = {
316 { 0, sizeof(left), left },
317 { 6, 0, left },
318 { 8, sizeof(right), right },

320 brix_file file;

322 int rc;
323
324 brix_status_clear(&status);
325 if (open_write(execution, connection, object, 1, &file, &status) != 0) return;

327 close_file(execution, connection, &file, &status);

331 if (rc != 0 && status.kxr == kXR_Unsupported) {

337 rc == 0 ? BRIX_CONFORM_VERDICT_SKIPPED
338 : BRIX_CONFORM_VERDICT_DIVERGENT,
339 rc == 0 ? "capability-present:writev"
340 : "unsupported-writev-returned-wrong-error",
341 "kXR_Unsupported or implemented writev", status.msg);
342 xrd_conform_grade_status(execution, "WVC-001", rc == 0, "writev-elements-landed",
343 "writev-refused", &status);
344 xrd_conform_grade_status(execution, "WVC-003", rc == 0, "writev-doSync-completed",
345 "writev-doSync-failed", &status);
346 xrd_conform_grade_status(execution, "WVC-004", rc == 0, "zero-element-was-noop",
347 "zero-element-broke-writev", &status);
348 if (xrd_conform_reserve_status(execution, 0, &status)) {
349 rc = brix_file_sync(connection, &file, &status);

351 "sync-failed", &status);
352 }
353 close_file(execution, connection, &file, &status);
354 grade_writev_readback(execution, connection, object, left, sizeof(left),
355 right, sizeof(right), &status);
356 probe_read_sync(execution, connection, object, &status);
357}

421 *

449static void
450probe_pgwrite(xrd_conform_execution *execution, brix_conn *connection,
451 const xrd_conform_scratch_object *object)
452{
453 uint8_t expected[8192];
454 uint8_t actual[8192];
455 brix_file file;
456 brix_status status;
457 ssize_t got;
458 int rc;
459
460 if ((connection->server_flags & kXR_suppgrw) == 0) {
461 skip_pgwrite_absent(execution);
462 probe_pgwrite_absent(execution, connection, object);
463 return;
464 }
465 xrd_conform_result_set(execution, "PGW-014",
466 BRIX_CONFORM_VERDICT_SKIPPED, "claim-present:kXR_suppgrw", NULL, NULL);
467 brix_status_clear(&status);
468 brix_conform_fill(&execution->run->key, 0, expected, sizeof(expected));
469 if (open_write(execution, connection, object, 1, &file, &status) != 0) return;
470 if (!xrd_conform_reserve_status(execution, sizeof(expected), &status)) {
471 close_file(execution, connection, &file, &status);
472 return;
473 }
474 rc = brix_file_pgwrite(connection, &file, 0, expected, sizeof(expected),
475 &status);
476 xrd_conform_grade_status(execution, "PGW-001", rc == 0, "typed-status-response-validated",
477 "pgwrite-status-invalid", &status);
478 xrd_conform_grade_status(execution, "PGW-002", rc == 0, "crc-first-pages-accepted",
479 "crc-first-pages-refused", &status);
480 xrd_conform_grade_status(execution, "PGW-003", rc == 0, "status-offset-validated",
481 "status-offset-mismatch", &status);
482 if (rc == 0 && xrd_conform_reserve_status(execution, sizeof(actual), &status)) {
483 got = brix_file_pgread(connection, &file, 0, actual, sizeof(actual),
484 &status);
485 xrd_conform_grade_status(execution, "PGW-004", got == (ssize_t) sizeof(actual)
486 && memcmp(actual, expected, sizeof(actual)) == 0,
487 "pgwrite-pgread-byte-exact", "pgwrite-readback-mismatch", &status);
488 }
489 if (xrd_conform_reserve_status(execution, 17, &status)) {
490 rc = brix_file_pgwrite(connection, &file, 1, expected + 1, 17, &status);
491 xrd_conform_grade_status(execution, "PGW-011", rc == 0 || status.kxr == kXR_ArgInvalid,
492 rc == 0 ? "fractional-pgwrite-honoured"
493 : "fractional-pgwrite-defined-refusal",
494 "fractional-pgwrite-silently-realigned", &status);
495 }
496 if (xrd_conform_reserve_status(execution, 0, &status)) {
497 rc = brix_file_pgwrite(connection, &file, 0, expected, 0, &status);
498 xrd_conform_grade_status(execution, "PGW-012", rc == 0, "zero-pgwrite-noop",
499 "zero-pgwrite-failed", &status);
500 }
501 close_file(execution, connection, &file, &status);
502}

517static void
518run_write_probes(xrd_conform_execution *execution, brix_conn *connection,
519 const xrd_conform_scratch_object *object)
520{
521 static const char *const ids[] = {
522 "WRT-001", "WRT-002", "WRT-003", "WRT-004", "WRT-005", "WRT-009"
523 };
524
525 if (xrd_conform_any_row_active(execution, ids, sizeof(ids) / sizeof(ids[0])))
526 probe_write(execution, connection, object);
527 if (xrd_conform_row_active(execution, "WRT-011"))
528 xrd_conform_probe_concurrent_write(execution, connection, object);
