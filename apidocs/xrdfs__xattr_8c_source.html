11/* xattr ls|get|set|rm — extended attributes via kXR_fattr (client/lib/fattr.c).
12 * xattr ls <path> list attribute names
13 * xattr get <path> <name> print one value
14 * xattr set <path> <name> <value> set/replace a value
15 * xattr rm <path> <name> delete an attribute
16 * `xattr <path>` with no subcommand is treated as `ls`. */
17int
18xattr_ls(brix_conn *c, const char *path)
19{
20 brix_status st;
21 char names[8192];
22 size_t total = 0, off;
23
24 brix_status_clear(&st);
25 if (brix_fattr_list(c, path, names, sizeof(names), &total, &st) != 0) {
26 return xrdfs_report_err("xattr ls", path, &st, 0, c);
27 }
28 /* The server returns a NUL-separated list of managed names tagged with a
29 * one-letter namespace prefix ("U.<name>" for the user namespace). Strip the
30 * "<X>." tag so the printed names round-trip directly through xattr get/set. */
31 for (off = 0; off < total && names[off] != '\0'; ) {
32 const char *name = names + off;
33 if (name[0] >= 'A' && name[0] <= 'Z' && name[1] == '.') { name += 2; }
34 printf("%s\n", name);
35 off += strlen(names + off) + 1;
36 }
37 return 0;
38}

41/* ---- Is argv[1] a recognised xattr subcommand keyword? ----
42 *
43 * WHAT: Returns 1 when the token is one of ls/get/set/rm, 0 otherwise.
44 *
45 * WHY: `xattr <path>` with no subcommand is shorthand for `xattr ls <path>`;
46 * factoring the four-way keyword test out keeps do_xattr's dispatch under
47 * the complexity cap.
48 *
49 * HOW: Compare the token against each of the four recognised subcommand names.
50 */
51static int
52xattr_is_subcommand(const char *s)
53{
54 return strcmp(s, "ls") == 0 || strcmp(s, "get") == 0
55 || strcmp(s, "set") == 0 || strcmp(s, "rm") == 0;
56}

59/* ---- xattr get <path> <name> ----
60 *
61 * WHAT: Fetches one attribute value and writes it to stdout followed by a
62 * newline. Returns 0 on success, 50 on a usage error, or the mapped shell
63 * code on a protocol failure.
64 *
65 * WHY: Splits the get branch out of do_xattr's dispatch so each subcommand is
66 * independently readable and the dispatcher stays flat.
67 *
68 * HOW: 1. Require the <name> argument.
69 * 2. Call brix_fattr_get; on error report the message and map the status.
70 * 3. Write the raw value bytes (clamped to the buffer) and a newline.
71 */
72static int
73xattr_get(brix_conn *c, const char *path, int argc, char **argv)
74{
75 brix_status st;
76 char val[8192];
77 size_t vlen = 0;
78
79 if (argc < 4) { fprintf(stderr, "usage: xattr get <path> <name>\n"); return 50; }
80 brix_status_clear(&st);
81 if (brix_fattr_get(c, path, argv[3], val, sizeof(val), &vlen, &st) != 0) {
82 fprintf(stderr, "xrdfs: xattr get %s [%s]: %s\n", path, argv[3], st.msg);
83 return brix_shellcode(&st);
84 }
85 fwrite(val, 1, vlen < sizeof(val) ? vlen : sizeof(val), stdout);
86 printf("\n");
87 return 0;
88}

152/* ---- Stock xrdfs grammar: `xattr <path> <code> <params>` ----
153 *
154 * WHAT: Handles the STOCK spelling (path first, then set/get/del/list; set
155 * takes a single `name=value`) by mapping onto BriX's own xattr helpers.
156 * Returns 0 / shell code.
157 *
158 * WHY: §7.12 — a script written for stock xrdfs (`xattr /f set a=b`) used to
159 * be silently mis-read as a bare list of `/f`. Accepting the stock form
160 * as a drop-in alias closes the grammar incompatibility; BriX's own
161 * `xattr <code> <path>` form is unchanged and still takes priority
162 * (disambiguated by the caller: the stock form is reached only when the
163 * FIRST token is a path, not a BriX subcommand).
164 *
165 * HOW: Build a BriX-shaped argv ({prog, code, path, name[, value]}) and call
166 * the same xattr_ls/get/set/rm helpers, splitting set's name=value in
167 * place. `path` is already cwd-resolved by the caller.
168 */
169static int
170xattr_stock_form(brix_conn *c, const char *path, int argc, char **argv)
171{
172 const char *code = argv[2];
173 char *nargv[5] = { argv[0], NULL, (char *) path, NULL, NULL };
174
175 if (strcmp(code, "list") == 0) {
176 return xattr_ls(c, path);
177 }
178 if (argc < 4) {
179 fprintf(stderr, "usage: xattr <path> %s <args>\n", code);
180 return 50;
181 }
182 if (strcmp(code, "get") == 0) {
183 nargv[1] = "get"; nargv[3] = argv[3];
184 return xattr_get(c, path, 4, nargv);
185 }
186 if (strcmp(code, "del") == 0) {
187 nargv[1] = "rm"; nargv[3] = argv[3];
188 return xattr_rm(c, path, 4, nargv);
189 }
190 /* set <name=value> — split at the first '=' (in place; argv is mutable). */
191 {
192 char *eq = strchr(argv[3], '=');
193
194 if (eq == NULL) {
195 fprintf(stderr, "xrdfs: xattr set: expected name=value, got '%s'\n",
196 argv[3]);
197 return 50;
198 }
199 *eq = '\0';
200 nargv[1] = "set"; nargv[3] = argv[3]; nargv[4] = eq + 1;
201 return xattr_set(c, path, 5, nargv);
202 }
203}

205int
206do_xattr(brix_conn *c, const char *cwd, int argc, char **argv)
207{
208 char path[XRDC_PATH_MAX];
209
210 if (argc < 2) {
211 fprintf(stderr, "usage: xattr ls|get|set|rm <path> [name] [value]\n"
212 " or: xattr <path> set|get|del|list [args] (stock)\n");
213 return 50;
214 }
215 /* First token is NOT a BriX subcommand → it is a path: either the bare
216 * `xattr <path>` list shorthand, or the STOCK `xattr <path> <code>` form
217 * (§7.12) when a stock code follows. BriX's own `xattr <code> <path>`
218 * form (below) keeps priority so existing usage is unaffected. */
219 if (!xattr_is_subcommand(argv[1])) {
220 build_path(cwd, argv[1], path, sizeof(path));
221 if (argc >= 3 && xattr_is_stock_code(argv[2])) {
222 return xattr_stock_form(c, path, argc, argv);
223 }
224 return xattr_ls(c, path);
225 }
226 if (argc < 3) {
227 fprintf(stderr, "usage: xattr %s <path> ...\n", argv[1]);
228 return 50;
229 }
230 build_path(cwd, argv[2], path, sizeof(path));
231
232 if (strcmp(argv[1], "ls") == 0) { return xattr_ls(c, path); }
233 if (strcmp(argv[1], "get") == 0) { return xattr_get(c, path, argc, argv); }
234 if (strcmp(argv[1], "set") == 0) { return xattr_set(c, path, argc, argv); }
235 return xattr_rm(c, path, argc, argv); /* the only remaining subcommand */
236}
