8/* paged I/O with per-page CRC32c (kXR_pgread / kXR_pgwrite, M6) *
9 * Both replies use kXR_status (4007) framing, NOT the kXR_ok path, so they are
10 * read here rather than via brix_recv. One status frame is:
11 * ServerResponseHdr{status=kXR_status, dlen=24}
12 * ServerResponseBody_Status{crc32c[4], streamID[2], requestid[1], resptype[1],
13 * reserved[4], dlen[4]} (16 bytes)
14 * trailing offset[8] (pgRead/pgWrite body)
15 * The header crc32c covers the 20 bytes streamID..offset; bdy.dlen is the size of
16 * the page-data that follows the 24-byte body (0 for pgwrite). Page units are
17 * [crc32c_be 4][data ≤4096], aligned to the FILE offset (short first page).
18 */

21/* Decode one frame's page-data buffer (units of [crc 4][data], file-offset
22 * aligned at file_off) into dst, verifying each page's CRC32c. Returns decoded
23 * data bytes, or -1 (st set) on a CRC mismatch or malformed framing. */
24ssize_t
25decode_pages(const uint8_t *pg, uint32_t pglen, int64_t file_off,
26 uint8_t *dst, size_t dstcap, brix_status *st)
27{
28 int64_t bad = file_off;
29 ssize_t n = xrdp_pg_decode(pg, (size_t) pglen, file_off, dst, dstcap, &bad);
30
31 if (n == -1) {
32 brix_status_set(st, XRDC_EINTEGRITY, 0,
33 "pgread CRC mismatch at offset %lld", (long long) bad);
34 return -1;
35 }
36 if (n < 0) {
37 brix_status_set(st, XRDC_EPROTO, 0, "pgread malformed page framing");
38 return -1;
39 }
40 return n;
41}

51static int
52pgread_frame_type(uint8_t resptype, unsigned *partial_frames, brix_status *st)
53{
54 if (resptype == kXR_FinalResult) return 0;
55 if (resptype == kXR_PartialResult) {
56 (*partial_frames)++;
57 return 0;

60 "unexpected pgread response type %u", resptype);

64static ssize_t
65pgread_args_frames(brix_conn *c, uint16_t sid, int64_t pgoff, void *dst,
66 size_t pglen, unsigned *partial_frames, brix_status *st)
67{
68 size_t total = 0;
69
70 *partial_frames = 0;
71
72 for (;;) {
73 uint8_t resptype = 0;
74 uint32_t pgdlen = 0;
75 int64_t foff = 0;
76 uint8_t *pg;
77 ssize_t decoded;
78
79 if (read_status_frame(c, sid, kXR_pgread - kXR_1stRequest,
80 &resptype, &pgdlen, &foff, st) != 0) return -1;
81 if (foff != pgoff + (int64_t) total) {
82 brix_status_set(st, XRDC_EPROTO, 0,
83 "pgread argument offset mismatch");
84 return -1;
85 }
86 if (pgread_frame_type(resptype, partial_frames, st) != 0) return -1;
87 if (pgdlen == 0) {
88 if (resptype == kXR_FinalResult) break;
89 continue;
90 }
91 pg = (uint8_t *) malloc(pgdlen);
92 if (pg == NULL) {
93 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%u)", pgdlen);
94 return -1;
95 }
96 if (brix_read_full(&c->io, pg, pgdlen, st) != 0) {
97 free(pg);
98 return -1;
99 }
100 decoded = decode_pages(pg, pgdlen, foff, (uint8_t *) dst + total,
101 pglen - total, st);
102 free(pg);
103 if (decoded < 0) return -1;
104 total += (size_t) decoded;
105 if (resptype == kXR_FinalResult) break;
106 }
107 if (total != pglen) {
108 brix_status_set(st, XRDC_EPROTO, 0,
109 "pgread argument request short answer (%zu of %zu bytes)",
110 total, pglen);
111 return -1;
112 }
113 return (ssize_t) total;
114}

116/* ---- Issue one pgread with an explicit optional argument payload ----
117 *
118 * WHAT: Issues a fresh kXR_pgread of exactly [pgoff, pgoff+pglen) carrying the
119 * caller's payload, decodes it STRICTLY (any CRC mismatch fails), and
120 * writes the page into dst. Returns the exact byte count / -1 (st set).
121 *
122 * WHY: §7.14 — a single flipped bit between server and client used to abort
123 * the whole transfer even though the server still holds good bytes; the
124 * write side already re-sends corrupt pages (pgwrite_retry_one), so the
125 * read side re-REQUESTS them. The args payload is the §1.2 wire surface:
126 * stock and BriX servers both serve it as a fresh re-read (verified live
127 * against 5.6.9 — unknown flags tolerated, pathid 0 always valid).
128 *
129 * HOW: 1. pgread request with offset=pgoff, rlen=pglen and caller payload.
130 * 2. Frame loop mirroring file_pgread_frames, but decode_pages (strict)
131 * — a retry that comes back corrupt again is a failed retry.
132 * 3. Total must equal pglen: a short answer is a protocol error.
133 */
134ssize_t
135brix_file_pgread_args(brix_conn *c, brix_file *f, int64_t pgoff, void *dst,
136 size_t pglen, const void *args, uint32_t args_len,
137 brix_status *st)
138{
139 ClientPgReadRequest req;
140 brix_payload pl;
141 uint16_t sid;
142
143 if (pglen > INT32_MAX || (args_len > 0 && args == NULL)) {
144 brix_status_set(st, XRDC_EUSAGE, 0, "invalid pgread argument request");
145 return -1;
146 }
147
148 memset(&req, 0, sizeof(req));
149 req.requestid = htons(kXR_pgread);
150 {
151 xrdw_pgread_req_t b = { .offset = pgoff, .rlen = (int32_t) pglen };
152 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
153 xrdw_pgread_req_pack(&b, ((ClientRequestHdr *) &req)->body);
154 }
155 pl.data = args;
156 pl.len = args_len;
157
158 if (brix_send(c, &req, args_len == 0 ? NULL : &pl, &sid, st) != 0) {
159 return -1;
160 }
161 {
162 unsigned partial_frames;
163 return pgread_args_frames(c, sid, pgoff, dst, pglen,
164 &partial_frames, st);
165 }
166}

168/* §7.14: re-request each corrupt page (bounded), overwriting its bytes in
169 * buf. A page that stays corrupt after the attempts is the old hard error,
170 * same message shape as before. 0 = all recovered, -1 = error (st set). */
171static int
172pgread_refetch_bad(brix_conn *c, brix_file *f, int64_t offset, void *buf,
173 const xrdp_pg_bad_t *bad, size_t nbad, brix_status *st)
174{
175 size_t k;
176
177 for (k = 0; k < nbad; k++) {
178 int attempt, fixed = 0;
179 uint8_t args[2] = { 0, kXR_pgRetry };
180
181 for (attempt = 0; attempt < XRDC_PGREAD_RETRIES && !fixed; attempt++) {
182 brix_status rst;
183 brix_status_clear(&rst);
184 if (brix_file_pgread_args(c, f, bad[k].off,
185 (uint8_t *) buf + (size_t) (bad[k].off - offset),
186 bad[k].dlen, args, sizeof(args), &rst)
187 == (ssize_t) bad[k].dlen)
188 {
189 fixed = 1;
190 } else if (rst.kxr != XRDC_EINTEGRITY) {
191 /* transport/protocol failure — surface it, do not loop */
192 *st = rst;
193 return -1;
194 }
195 }
196 if (!fixed) {
197 brix_status_set(st, XRDC_EINTEGRITY, 0,
198 "pgread CRC mismatch at offset %lld "
199 "(unrecovered after %d retries)",
200 (long long) bad[k].off, XRDC_PGREAD_RETRIES);
201 return -1;
202 }
203 }
204 return 0;
205}

207static ssize_t
208file_pgread_frames(brix_conn *c, brix_file *f, int64_t offset, void *buf,
209 size_t len, unsigned *partial_frames, brix_status *st)
210{
211 ClientPgReadRequest req;
212 uint16_t sid;
213 size_t total = 0;
214 xrdp_pg_bad_t bad[XRDC_PGREAD_BADMAX];
215 size_t nbad = 0;
216
217 *partial_frames = 0;
218
219 memset(&req, 0, sizeof(req));
220 req.requestid = htons(kXR_pgread);
221 {
222 xrdw_pgread_req_t b = { .offset = offset, .rlen = (int32_t) len };
223 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
224 xrdw_pgread_req_pack(&b, ((ClientRequestHdr *) &req)->body);
225 }
226 /* dlen (offset 20) is set to 0 by brix_send (no args payload). */
227
228 if (brix_send(c, &req, NULL, &sid, st) != 0) {
229 return -1;
230 }
231
232 /* Accumulate kXR_status frames until resptype=Final (the module sends one
233 * Final frame per request; Partial is handled defensively). §7.14: pages
234 * are decoded with the COLLECT decoder — corrupt pages land in buf like
235 * everything else and are recorded rather than fatal, then re-requested
236 * below once the response is fully drained (the wire must be clean before
237 * a new request goes out). */
238 for (;;) {
239 uint8_t resptype = 0;
240 uint32_t pgdlen = 0;
241 int64_t foff = 0;
242 uint8_t *pg;
243 ssize_t decoded;
244 size_t frame_bad = 0;
245
246 if (read_status_frame(c, sid, kXR_pgread - kXR_1stRequest,
247 &resptype, &pgdlen, &foff, st) != 0) {
248 return -1;
249 }
250 if (foff != offset + (int64_t) total) {
251 brix_status_set(st, XRDC_EPROTO, 0,
252 "pgread frame offset mismatch");
253 return -1;
254 }
255 if (pgread_frame_type(resptype, partial_frames, st) != 0) return -1;
256 if (pgdlen == 0) {
257 if (resptype == kXR_FinalResult) {
258 break;
259 }
260 continue;
261 }
262 pg = (uint8_t *) malloc(pgdlen);
263 if (pg == NULL) {
264 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%u)", pgdlen);
265 return -1;
266 }
267 if (brix_read_full(&c->io, pg, pgdlen, st) != 0) {
268 free(pg);
269 return -1;
270 }
271 decoded = xrdp_pg_decode_collect(pg, pgdlen, foff,
272 (uint8_t *) buf + total,
273 len - total,
274 bad + nbad, XRDC_PGREAD_BADMAX - nbad,
275 &frame_bad);
276 free(pg);
277 if (decoded == -3) { /* collect cap hit: bad-page flood */
278 brix_status_set(st, XRDC_EINTEGRITY, 0,
279 "pgread: more than %d corrupt pages in one "
280 "request — refusing to retry a poisoned stream",
281 XRDC_PGREAD_BADMAX);
282 return -1;
283 }
284 if (decoded < 0) {
285 brix_status_set(st, XRDC_EPROTO, 0,
286 "pgread malformed page framing");
287 return -1;
288 }
289 nbad += frame_bad;
290 total += (size_t) decoded;
291 if (resptype == kXR_FinalResult) {
292 break;
293 }
294 }
295
296 if (pgread_refetch_bad(c, f, offset, buf, bad, nbad, st) != 0) {
297 return -1;
298 }
299 return (ssize_t) total;
300}

332/* Resend one page (kXR_pgRetry) from the original buffer and report the server's
333 * verdict. pgoff is the page's file offset; the page data is sliced out of buf
334 * at (pgoff - base). Returns 0 = corrected, 1 = still bad, -1 = error (st set). */
335int
336pgwrite_retry_one(brix_conn *c, brix_file *f, const uint8_t *buf, int64_t base,
337 size_t len, int64_t pgoff, brix_status *st)
338{
339 ClientPgWriteRequest req;
340 uint16_t sid;
341 uint8_t rp[kXR_pgPageSZ + 4];
342 size_t doff = (size_t) (pgoff - base);
343 size_t to_boundary = (size_t) kXR_pgPageSZ
344 - (size_t) (pgoff & (int64_t) (kXR_pgPageSZ - 1));
345 size_t remaining, page_len, rplen;
346 uint8_t resptype = 0;
347 uint32_t pgdlen = 0;
348 int64_t foff = 0;
349
350 if (pgoff < base || doff >= len) {
351 brix_status_set(st, XRDC_EPROTO, 0,
352 "pgwrite CSE offset %lld out of range", (long long) pgoff);
353 return -1;
354 }
355 remaining = len - doff;
356 page_len = remaining < to_boundary ? remaining : to_boundary;
357 rplen = xrdp_pg_encode(buf + doff, page_len, pgoff, rp);
358
359 memset(&req, 0, sizeof(req));
360 req.requestid = htons(kXR_pgwrite);
361 {
362 xrdw_pgwrite_req_t b = { .offset = pgoff, .pathid = 0,
363 .reqflags = kXR_pgRetry };
364 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
365 xrdw_pgwrite_req_pack(&b, ((ClientRequestHdr *) &req)->body);
366 }
367
368 {
369 brix_payload pl = { rp, (uint32_t) rplen };
370 if (brix_send(c, &req, &pl, &sid, st) != 0) {
371 return -1;
372 }
373 }
374 if (read_status_frame(c, sid, kXR_pgwrite - kXR_1stRequest,
375 &resptype, &pgdlen, &foff, st) != 0) {
376 return -1;
377 }
378 if (foff != pgoff) {
379 brix_status_set(st, XRDC_EPROTO, 0,
380 "pgwrite retry offset mismatch");
381 return -1;
382 }
383 if (pgdlen != 0) {
384 /* Still bad — drain and discard the CSE trailer, report not-yet-clean. */
385 uint8_t *cse = (uint8_t *) malloc(pgdlen);
386 if (cse != NULL) {
387 (void) brix_read_full(&c->io, cse, pgdlen, st);
388 free(cse);
389 }
390 return 1;
391 }
392 return 0;
393}

396/* ---- Drive one corrupt page back to a clean server verdict ----
397 *
398 * WHAT: Resend the page at file offset pgoff (sliced from buf at pgoff-base)
399 * via kXR_pgRetry up to XRDC_PGW_MAX_RETRY times. Returns 0 once the server
400 * accepts the page, or -1 (st set) if it stays corrupt past the retry budget
401 * or a wire error occurs.
402 *
403 * WHY: The pgwrite CSE handler must apply an identical bounded-retry policy to
404 * every failed page. Isolating that per-page loop keeps the CSE trailer walk
405 * flat and keeps the retry/verdict bookkeeping in one testable place.
406 *
407 * HOW:
408 * 1. Loop up to XRDC_PGW_MAX_RETRY: resend the page via pgwrite_retry_one;
409 * a verdict of 0 (corrected) or -1 (error) breaks out immediately, 1
410 * (still bad) retries.
411 * 2. verdict > 0 after the budget is exhausted -> set XRDC_EINTEGRITY, -1.
412 * 3. verdict < 0 -> propagate the already-set error, -1.
413 * 4. verdict 0 -> page corrected, 0.
414 */
415static int
416pgwrite_correct_one_page(brix_conn *c, brix_file *f, const uint8_t *buf,
417 int64_t base, size_t len, int64_t pgoff,
418 brix_status *st)
419{
420 int attempt, verdict = 1;
421
422 for (attempt = 0; attempt < XRDC_PGW_MAX_RETRY; attempt++) {
423 verdict = pgwrite_retry_one(c, f, buf, base, len, pgoff, st);
424 if (verdict <= 0) {
425 break; /* 0 = corrected, -1 = error */
426 }
427 }
428 if (verdict > 0) {
429 brix_status_set(st, XRDC_EINTEGRITY, 0,
430 "pgwrite page %lld uncorrectable after %d retries",
431 (long long) pgoff, XRDC_PGW_MAX_RETRY);
432 return -1;
433 }
434 if (verdict < 0) {
435 return -1;
436 }
437 return 0;
438}

441/* ---- Read and act on a pgwrite CSE (checksum-error) trailer ----
442 *
443 * WHAT: Read the pgdlen-byte CSE trailer the server sent after a pgwrite,
444 * validate its shape, and resend every listed bad page via
445 * pgwrite_correct_one_page. Returns 0 when all pages end clean, -1 (st set)
446 * on OOM, a malformed trailer, or an uncorrectable/errored page. Does NOT own
447 * the caller's request payload.
448 *
449 * WHY: The server writes the data but reports pages whose CRC32c failed on the
450 * wire (our encode is always correct); the client must retransmit those pages.
451 * Confining the trailer read and bad-page walk here keeps brix_file_pgwrite a
452 * flat send/read orchestrator.
453 *
454 * HOW:
455 * 1. Allocate and read the trailer; OOM or a short read -> -1 (st set).
456 * 2. Trailer layout is cseCRC(4) dlFirst(2) dlLast(2) then int64 bof[n];
457 * reject any length < 8 or not 8-aligned in the tail as malformed.
458 * 3. For each of the n bad-page offsets, decode the big-endian int64 and
459 * drive it clean via pgwrite_correct_one_page, stopping on the first
460 * failure.
461 * 4. Free the trailer and return the accumulated verdict.
462 */
463static int
464pgwrite_handle_cse(brix_conn *c, brix_file *f, const uint8_t *buf,
465 int64_t base, size_t len, uint32_t pgdlen, brix_status *st)
466{
467 uint8_t *cse = (uint8_t *) malloc(pgdlen);
468 size_t nbad, i;
469 int rc_ret = 0;
470
471 if (cse == NULL || brix_read_full(&c->io, cse, pgdlen, st) != 0) {
472 if (cse == NULL) {
473 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%u)", pgdlen);
474 }
475 free(cse);
476 return -1;
477 }
478
479 /* Trailer: cseCRC(4) dlFirst(2) dlLast(2) then int64 bof[n]. */
480 if (pgdlen < 8 || ((pgdlen - 8) % 8) != 0) {
481 free(cse);
482 brix_status_set(st, XRDC_EPROTO, 0,
483 "malformed pgwrite CSE trailer (%u bytes)", pgdlen);
484 return -1;
485 }
486 nbad = (size_t) (pgdlen - 8) / 8;
487
488 for (i = 0; i < nbad && rc_ret == 0; i++) {
489 uint64_t bo_be;
490 int64_t bo;
491
492 memcpy(&bo_be, cse + 8 + i * 8, 8);
493 bo = (int64_t) be64toh(bo_be);
494 rc_ret = pgwrite_correct_one_page(c, f, buf, base, len, bo, st);
495 }
496
497 free(cse);
498 return rc_ret;
499}

502int
503brix_file_pgwrite(brix_conn *c, brix_file *f, int64_t offset, const void *buf,
504 size_t len, brix_status *st)
505{
506 ClientPgWriteRequest req;
507 uint16_t sid;
508 uint8_t *payload = NULL;
509 size_t cap, plen = 0;
510 uint8_t resptype = 0;
511 uint32_t pgdlen = 0;
512 int64_t foff = 0;
513
514 /* Worst case: one 4-byte CRC per page plus a short first/last page → at most
515 * (len/pagesz + 2) checksums. Build the [crc][data] payload via the shared
516 * page-mode encoder (libxrdproto) so client-encode == server-decode. */
517 cap = len + ((len / kXR_pgPageSZ) + 2) * 4;
518 payload = (uint8_t *) malloc(cap);
519 if (payload == NULL) {
520 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%zu)", cap);
521 return -1;
522 }
523 plen = xrdp_pg_encode((const uint8_t *) buf, len, offset, payload);
524
525 memset(&req, 0, sizeof(req));
526 req.requestid = htons(kXR_pgwrite);
527 {
528 xrdw_pgwrite_req_t b = { .offset = offset, .pathid = 0, .reqflags = 0 };
529 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
530 xrdw_pgwrite_req_pack(&b, ((ClientRequestHdr *) &req)->body);
531 }
532 /* dlen (offset 20) is set to plen by brix_send. */
533
534 {
535 brix_payload pl = { payload, (uint32_t) plen };
536 if (brix_send(c, &req, &pl, &sid, st) != 0) {
537 free(payload);
538 return -1;
539 }
540 }
541 if (read_status_frame(c, sid, kXR_pgwrite - kXR_1stRequest,
542 &resptype, &pgdlen, &foff, st) != 0) {
543 free(payload); /* kXR_ChkSumErr (CRC rejected) surfaces here */
544 return -1;
545 }
546 if (foff != offset) {
547 brix_status_set(st, XRDC_EPROTO, 0,
548 "pgwrite response offset mismatch");
549 free(payload);
550 return -1;
551 }
552 if (pgdlen != 0) {
553 /* CSE: the server wrote the data but reported pages that failed CRC32c
554 * (typically wire corruption — our encode is always correct). Read the
555 * retransmit list and resend each page with kXR_pgRetry until the server
556 * accepts it or we exhaust the bounded attempts. */
557 int rc_ret = pgwrite_handle_cse(c, f, (const uint8_t *) buf, offset,
558 len, pgdlen, st);
559 free(payload);
560 return rc_ret;
561 }
562 free(payload);
563 return 0;
564}
