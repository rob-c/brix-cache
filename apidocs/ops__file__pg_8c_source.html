8/* paged I/O with per-page CRC32c (kXR_pgread / kXR_pgwrite, M6) *
9 * Both replies use kXR_status (4007) framing, NOT the kXR_ok path, so they are
10 * read here rather than via brix_recv. One status frame is:
11 * ServerResponseHdr{status=kXR_status, dlen=24}
12 * ServerResponseBody_Status{crc32c[4], streamID[2], requestid[1], resptype[1],
13 * reserved[4], dlen[4]} (16 bytes)
14 * trailing offset[8] (pgRead/pgWrite body)
15 * The header crc32c covers the 20 bytes streamID..offset; bdy.dlen is the size of
16 * the page-data that follows the 24-byte body (0 for pgwrite). Page units are
17 * [crc32c_be 4][data ≤4096], aligned to the FILE offset (short first page).
18 */

21/* Read and validate one kXR_status frame's 24-byte body. On success sets
22 * *resptype (0=Final,1=Partial), *pgdlen (page-data bytes that follow), and
23 * *foff (the frame's file offset). kXR_error is surfaced via st. 0 / -1. */
24int
25read_status_frame(brix_conn *c, uint16_t want_sid, uint8_t *resptype,
26 uint32_t *pgdlen, int64_t *foff, brix_status *st)
27{
28 uint8_t hdr[XRD_RESPONSE_HDR_LEN];
29 uint8_t sb[XRDC_PG_STATUSBODY];
30 uint16_t sid, stat;
31 uint32_t dlen, want_crc, got_crc;
32 uint64_t off_be;
33
34 if (brix_read_full(&c->io, hdr, sizeof(hdr), st) != 0) {
35 return -1;
36 }
37 xrd_resp_hdr_unpack(hdr, &sid, &stat, &dlen); /* unaligned-safe */
38
39 if (stat == kXR_error) {
40 uint8_t *eb = NULL;
41 int errnum = 0;
42 if (dlen > 0 && dlen <= XRDC_DLEN_MAX) {
43 eb = (uint8_t *) malloc(dlen);
44 if (eb != NULL && brix_read_full(&c->io, eb, dlen, st) == 0) {
45 const char *emsg = "";
46 size_t emlen = 0;
47 /* bounded msg slice — eb is not NUL-terminated. */
48 xrd_error_body_decode(eb, dlen, &errnum, &emsg, &emlen);
49 brix_status_set(st, errnum, 0, "%.*s (%s)", (int) emlen,
50 emsg ? emsg : "", brix_kxr_name(errnum));
51 }
52 }
53 free(eb);
54 if (st->kxr == 0) {
55 brix_status_set(st, XRDC_EPROTO, 0, "pg op error (status %u)", stat);
56 }
57 return -1;
58 }
59 if (stat != kXR_status) {
60 brix_status_set(st, XRDC_EPROTO, 0,
61 "expected kXR_status, got %u", stat);
62 return -1;
63 }
64 if (want_sid != 0xffff && sid != want_sid) {
65 brix_status_set(st, XRDC_EPROTO, 0,
66 "stream id mismatch (got %u, want %u)", sid, want_sid);
67 return -1;
68 }
69 if (dlen != XRDC_PG_STATUSBODY) {
70 brix_status_set(st, XRDC_EPROTO, 0,
71 "bad kXR_status body length %u (want %d)",
72 dlen, XRDC_PG_STATUSBODY);
73 return -1;
74 }
75 if (brix_read_full(&c->io, sb, sizeof(sb), st) != 0) {
76 return -1;
77 }
78
79 /* crc32c covers sb[4..24): streamID(2) requestid(1) resptype(1) reserved(4)
80 * dlen(4) offset(8) = 20 bytes. */
81 want_crc = xrd_get_u32_be(sb); /* unaligned-safe */
82 got_crc = brix_crc32c_value(sb + 4, XRDC_PG_STATUSBODY - 4);
83 if (want_crc != got_crc) {
84 brix_status_set(st, XRDC_EINTEGRITY, 0,
85 "kXR_status header CRC mismatch (got %08x want %08x)",
86 got_crc, want_crc);
87 return -1;
88 }
89
90 *resptype = sb[7];
91 *pgdlen = xrd_get_u32_be(sb + 12); /* unaligned-safe */
92 memcpy(&off_be, sb + 16, 8);
93 *foff = (int64_t) be64toh(off_be);
94 return 0;
95}

98/* Decode one frame's page-data buffer (units of [crc 4][data], file-offset
99 * aligned at file_off) into dst, verifying each page's CRC32c. Returns decoded
100 * data bytes, or -1 (st set) on a CRC mismatch or malformed framing. */
101ssize_t
102decode_pages(const uint8_t *pg, uint32_t pglen, int64_t file_off,
103 uint8_t *dst, size_t dstcap, brix_status *st)
104{
105 int64_t bad = file_off;
106 ssize_t n = xrdp_pg_decode(pg, (size_t) pglen, file_off, dst, dstcap, &bad);
107
108 if (n == -1) {
109 brix_status_set(st, XRDC_EINTEGRITY, 0,
110 "pgread CRC mismatch at offset %lld", (long long) bad);
111 return -1;
112 }
113 if (n < 0) {
114 brix_status_set(st, XRDC_EPROTO, 0, "pgread malformed page framing");
115 return -1;
116 }
117 return n;
118}

121/* Corrupt pages tolerated per pgread before failing outright: a flipped bit
122 * on the wire hits one page; a server producing MORE corrupt pages than this
123 * is not "a bit flip" and retrying each would let it stall the client. */
124#define XRDC_PGREAD_BADMAX 16
125/* kXR_pgRetry attempts per corrupt page before the integrity verdict. */
126#define XRDC_PGREAD_RETRIES 2
127
128/* ---- Re-request ONE corrupt page with the kXR_pgRetry request args ----
129 *
130 * WHAT: Issues a fresh kXR_pgread of exactly [pgoff, pgoff+pglen) carrying the
131 * ClientPgReadReqArgs payload {pathid=0, reqflags=kXR_pgRetry}, decodes
132 * it STRICTLY (any CRC mismatch fails), and writes the page into dst.
133 * Returns 0 corrected / -1 (st set).
134 *
135 * WHY: §7.14 — a single flipped bit between server and client used to abort
136 * the whole transfer even though the server still holds good bytes; the
137 * write side already re-sends corrupt pages (pgwrite_retry_one), so the
138 * read side re-REQUESTS them. The args payload is the §1.2 wire surface:
139 * stock and BriX servers both serve it as a fresh re-read (verified live
140 * against 5.6.9 — unknown flags tolerated, pathid 0 always valid).
141 *
142 * HOW: 1. pgread request with offset=pgoff, rlen=pglen, 2-byte args payload.
143 * 2. Frame loop mirroring file_pgread_frames, but decode_pages (strict)
144 * — a retry that comes back corrupt again is a failed retry.
145 * 3. Total must equal pglen: a short answer is a protocol error.
146 */
147static int
148pgread_retry_page(brix_conn *c, brix_file *f, int64_t pgoff, uint32_t pglen,
149 uint8_t *dst, brix_status *st)
150{
151 ClientPgReadRequest req;
152 brix_payload pl;
153 uint8_t args[2] = { 0, kXR_pgRetry };
154 uint16_t sid;
155 size_t total = 0;
156
157 memset(&req, 0, sizeof(req));
158 req.requestid = htons(kXR_pgread);
159 {
160 xrdw_pgread_req_t b = { .offset = pgoff, .rlen = (int32_t) pglen };
161 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
162 xrdw_pgread_req_pack(&b, ((ClientRequestHdr *) &req)->body);
163 }
164 pl.data = args;
165 pl.len = sizeof(args);
166
167 if (brix_send(c, &req, &pl, &sid, st) != 0) {
168 return -1;
169 }
170 for (;;) {
171 uint8_t resptype = 0;
172 uint32_t pgdlen = 0;
173 int64_t foff = 0;
174 uint8_t *pg;
175 ssize_t decoded;
176
177 if (read_status_frame(c, sid, &resptype, &pgdlen, &foff, st) != 0) {
178 return -1;
179 }
180 if (pgdlen == 0) {
181 if (resptype == kXR_FinalResult) {
182 break;
183 }
184 continue;
185 }
186 pg = (uint8_t *) malloc(pgdlen);
187 if (pg == NULL) {
188 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%u)", pgdlen);
189 return -1;
190 }
191 if (brix_read_full(&c->io, pg, pgdlen, st) != 0) {
192 free(pg);
193 return -1;
194 }
195 decoded = decode_pages(pg, pgdlen, foff, dst + total,
196 (size_t) pglen - total, st);
197 free(pg);
198 if (decoded < 0) {
199 return -1;
200 }
201 total += (size_t) decoded;
202 if (resptype == kXR_FinalResult) {
203 break;
204 }
205 }
206 if (total != (size_t) pglen) {
207 brix_status_set(st, XRDC_EPROTO, 0,
208 "pgread retry short answer (%zu of %u bytes)",
209 total, pglen);
210 return -1;
211 }
212 return 0;
213}

215/* §7.14: re-request each corrupt page (bounded), overwriting its bytes in
216 * buf. A page that stays corrupt after the attempts is the old hard error,
217 * same message shape as before. 0 = all recovered, -1 = error (st set). */
218static int
219pgread_refetch_bad(brix_conn *c, brix_file *f, int64_t offset, void *buf,
220 const xrdp_pg_bad_t *bad, size_t nbad, brix_status *st)
221{
222 size_t k;
223
224 for (k = 0; k < nbad; k++) {
225 int attempt, fixed = 0;
226
227 for (attempt = 0; attempt < XRDC_PGREAD_RETRIES && !fixed; attempt++) {
228 brix_status rst;
229 brix_status_clear(&rst);
230 if (pgread_retry_page(c, f, bad[k].off, bad[k].dlen,
231 (uint8_t *) buf
232 + (size_t) (bad[k].off - offset),
233 &rst) == 0)
234 {
235 fixed = 1;
236 } else if (rst.kxr != XRDC_EINTEGRITY) {
237 /* transport/protocol failure — surface it, do not loop */
238 *st = rst;
239 return -1;
240 }
241 }
242 if (!fixed) {
243 brix_status_set(st, XRDC_EINTEGRITY, 0,
244 "pgread CRC mismatch at offset %lld "
245 "(unrecovered after %d retries)",
246 (long long) bad[k].off, XRDC_PGREAD_RETRIES);
247 return -1;
248 }
249 }
250 return 0;
251}

253static ssize_t
254file_pgread_frames(brix_conn *c, brix_file *f, int64_t offset, void *buf,
255 size_t len, brix_status *st)
256{
257 ClientPgReadRequest req;
258 uint16_t sid;
259 size_t total = 0;
260 xrdp_pg_bad_t bad[XRDC_PGREAD_BADMAX];
261 size_t nbad = 0;
262
263 memset(&req, 0, sizeof(req));
264 req.requestid = htons(kXR_pgread);
265 {
266 xrdw_pgread_req_t b = { .offset = offset, .rlen = (int32_t) len };
267 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
268 xrdw_pgread_req_pack(&b, ((ClientRequestHdr *) &req)->body);
269 }
270 /* dlen (offset 20) is set to 0 by brix_send (no args payload). */
271
272 if (brix_send(c, &req, NULL, &sid, st) != 0) {
273 return -1;
274 }
275
276 /* Accumulate kXR_status frames until resptype=Final (the module sends one
277 * Final frame per request; Partial is handled defensively). §7.14: pages
278 * are decoded with the COLLECT decoder — corrupt pages land in buf like
279 * everything else and are recorded rather than fatal, then re-requested
280 * below once the response is fully drained (the wire must be clean before
281 * a new request goes out). */
282 for (;;) {
283 uint8_t resptype = 0;
284 uint32_t pgdlen = 0;
285 int64_t foff = 0;
286 uint8_t *pg;
287 ssize_t decoded;
288 size_t frame_bad = 0;
289
290 if (read_status_frame(c, sid, &resptype, &pgdlen, &foff, st) != 0) {
291 return -1;
292 }
293 if (pgdlen == 0) {
294 if (resptype == kXR_FinalResult) {
295 break;
296 }
297 continue;
298 }
299 pg = (uint8_t *) malloc(pgdlen);
300 if (pg == NULL) {
301 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%u)", pgdlen);
302 return -1;
303 }
304 if (brix_read_full(&c->io, pg, pgdlen, st) != 0) {
305 free(pg);
306 return -1;
307 }
308 decoded = xrdp_pg_decode_collect(pg, pgdlen, foff,
309 (uint8_t *) buf + total,
310 len - total,
311 bad + nbad, XRDC_PGREAD_BADMAX - nbad,
312 &frame_bad);
313 free(pg);
314 if (decoded == -3) { /* collect cap hit: bad-page flood */
315 brix_status_set(st, XRDC_EINTEGRITY, 0,
316 "pgread: more than %d corrupt pages in one "
317 "request — refusing to retry a poisoned stream",
318 XRDC_PGREAD_BADMAX);
319 return -1;
320 }
321 if (decoded < 0) {
322 brix_status_set(st, XRDC_EPROTO, 0,
323 "pgread malformed page framing");
324 return -1;
325 }
326 nbad += frame_bad;
327 total += (size_t) decoded;
328 if (resptype == kXR_FinalResult) {
329 break;
330 }
331 }
332
333 if (pgread_refetch_bad(c, f, offset, buf, bad, nbad, st) != 0) {
334 return -1;
335 }
336 return (ssize_t) total;
337}

355/* Resend one page (kXR_pgRetry) from the original buffer and report the server's
356 * verdict. pgoff is the page's file offset; the page data is sliced out of buf
357 * at (pgoff - base). Returns 0 = corrected, 1 = still bad, -1 = error (st set). */
358int
359pgwrite_retry_one(brix_conn *c, brix_file *f, const uint8_t *buf, int64_t base,
360 size_t len, int64_t pgoff, brix_status *st)
361{
362 ClientPgWriteRequest req;
363 uint16_t sid;
364 uint8_t rp[kXR_pgPageSZ + 4];
365 size_t doff = (size_t) (pgoff - base);
366 size_t to_boundary = (size_t) kXR_pgPageSZ
367 - (size_t) (pgoff & (int64_t) (kXR_pgPageSZ - 1));
368 size_t remaining, page_len, rplen;
369 uint8_t resptype = 0;
370 uint32_t pgdlen = 0;
371 int64_t foff = 0;
372
373 if (pgoff < base || doff >= len) {
374 brix_status_set(st, XRDC_EPROTO, 0,
375 "pgwrite CSE offset %lld out of range", (long long) pgoff);
376 return -1;
377 }
378 remaining = len - doff;
379 page_len = remaining < to_boundary ? remaining : to_boundary;
380 rplen = xrdp_pg_encode(buf + doff, page_len, pgoff, rp);
381
382 memset(&req, 0, sizeof(req));
383 req.requestid = htons(kXR_pgwrite);
384 {
385 xrdw_pgwrite_req_t b = { .offset = pgoff, .pathid = 0,
386 .reqflags = kXR_pgRetry };
387 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
388 xrdw_pgwrite_req_pack(&b, ((ClientRequestHdr *) &req)->body);
389 }
390
391 {
392 brix_payload pl = { rp, (uint32_t) rplen };
393 if (brix_send(c, &req, &pl, &sid, st) != 0) {
394 return -1;
395 }
396 }
397 if (read_status_frame(c, sid, &resptype, &pgdlen, &foff, st) != 0) {
398 return -1;
399 }
400 if (pgdlen != 0) {
401 /* Still bad — drain and discard the CSE trailer, report not-yet-clean. */
402 uint8_t *cse = (uint8_t *) malloc(pgdlen);
403 if (cse != NULL) {
404 (void) brix_read_full(&c->io, cse, pgdlen, st);
405 free(cse);
406 }
407 return 1;
408 }
409 return 0;
410}

413/* ---- Drive one corrupt page back to a clean server verdict ----
414 *
415 * WHAT: Resend the page at file offset pgoff (sliced from buf at pgoff-base)
416 * via kXR_pgRetry up to XRDC_PGW_MAX_RETRY times. Returns 0 once the server
417 * accepts the page, or -1 (st set) if it stays corrupt past the retry budget
418 * or a wire error occurs.
419 *
420 * WHY: The pgwrite CSE handler must apply an identical bounded-retry policy to
421 * every failed page. Isolating that per-page loop keeps the CSE trailer walk
422 * flat and keeps the retry/verdict bookkeeping in one testable place.
423 *
424 * HOW:
425 * 1. Loop up to XRDC_PGW_MAX_RETRY: resend the page via pgwrite_retry_one;
426 * a verdict of 0 (corrected) or -1 (error) breaks out immediately, 1
427 * (still bad) retries.
428 * 2. verdict > 0 after the budget is exhausted -> set XRDC_EINTEGRITY, -1.
429 * 3. verdict < 0 -> propagate the already-set error, -1.
430 * 4. verdict 0 -> page corrected, 0.
431 */
432static int
433pgwrite_correct_one_page(brix_conn *c, brix_file *f, const uint8_t *buf,
434 int64_t base, size_t len, int64_t pgoff,
435 brix_status *st)
436{
437 int attempt, verdict = 1;
438
439 for (attempt = 0; attempt < XRDC_PGW_MAX_RETRY; attempt++) {
440 verdict = pgwrite_retry_one(c, f, buf, base, len, pgoff, st);
441 if (verdict <= 0) {
442 break; /* 0 = corrected, -1 = error */
443 }
444 }
445 if (verdict > 0) {
446 brix_status_set(st, XRDC_EINTEGRITY, 0,
447 "pgwrite page %lld uncorrectable after %d retries",
448 (long long) pgoff, XRDC_PGW_MAX_RETRY);
449 return -1;
450 }
451 if (verdict < 0) {
452 return -1;
453 }
454 return 0;
455}

458/* ---- Read and act on a pgwrite CSE (checksum-error) trailer ----
459 *
460 * WHAT: Read the pgdlen-byte CSE trailer the server sent after a pgwrite,
461 * validate its shape, and resend every listed bad page via
462 * pgwrite_correct_one_page. Returns 0 when all pages end clean, -1 (st set)
463 * on OOM, a malformed trailer, or an uncorrectable/errored page. Does NOT own
464 * the caller's request payload.
465 *
466 * WHY: The server writes the data but reports pages whose CRC32c failed on the
467 * wire (our encode is always correct); the client must retransmit those pages.
468 * Confining the trailer read and bad-page walk here keeps brix_file_pgwrite a
469 * flat send/read orchestrator.
470 *
471 * HOW:
472 * 1. Allocate and read the trailer; OOM or a short read -> -1 (st set).
473 * 2. Trailer layout is cseCRC(4) dlFirst(2) dlLast(2) then int64 bof[n];
474 * reject any length < 8 or not 8-aligned in the tail as malformed.
475 * 3. For each of the n bad-page offsets, decode the big-endian int64 and
476 * drive it clean via pgwrite_correct_one_page, stopping on the first
477 * failure.
478 * 4. Free the trailer and return the accumulated verdict.
479 */
480static int
481pgwrite_handle_cse(brix_conn *c, brix_file *f, const uint8_t *buf,
482 int64_t base, size_t len, uint32_t pgdlen, brix_status *st)
483{
484 uint8_t *cse = (uint8_t *) malloc(pgdlen);
485 size_t nbad, i;
486 int rc_ret = 0;
487
488 if (cse == NULL || brix_read_full(&c->io, cse, pgdlen, st) != 0) {
489 if (cse == NULL) {
490 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%u)", pgdlen);
491 }
492 free(cse);
493 return -1;
494 }
495
496 /* Trailer: cseCRC(4) dlFirst(2) dlLast(2) then int64 bof[n]. */
497 if (pgdlen < 8 || ((pgdlen - 8) % 8) != 0) {
498 free(cse);
499 brix_status_set(st, XRDC_EPROTO, 0,
500 "malformed pgwrite CSE trailer (%u bytes)", pgdlen);
501 return -1;
502 }
503 nbad = (size_t) (pgdlen - 8) / 8;
504
505 for (i = 0; i < nbad && rc_ret == 0; i++) {
506 uint64_t bo_be;
507 int64_t bo;
508
509 memcpy(&bo_be, cse + 8 + i * 8, 8);
510 bo = (int64_t) be64toh(bo_be);
511 rc_ret = pgwrite_correct_one_page(c, f, buf, base, len, bo, st);
512 }
513
514 free(cse);
515 return rc_ret;
516}

519int
520brix_file_pgwrite(brix_conn *c, brix_file *f, int64_t offset, const void *buf,
521 size_t len, brix_status *st)
522{
523 ClientPgWriteRequest req;
524 uint16_t sid;
525 uint8_t *payload = NULL;
526 size_t cap, plen = 0;
527 uint8_t resptype = 0;
528 uint32_t pgdlen = 0;
529 int64_t foff = 0;
530
531 /* Worst case: one 4-byte CRC per page plus a short first/last page → at most
532 * (len/pagesz + 2) checksums. Build the [crc][data] payload via the shared
533 * page-mode encoder (libxrdproto) so client-encode == server-decode. */
534 cap = len + ((len / kXR_pgPageSZ) + 2) * 4;
535 payload = (uint8_t *) malloc(cap);
536 if (payload == NULL) {
537 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%zu)", cap);
538 return -1;
539 }
540 plen = xrdp_pg_encode((const uint8_t *) buf, len, offset, payload);
541
542 memset(&req, 0, sizeof(req));
543 req.requestid = htons(kXR_pgwrite);
544 {
545 xrdw_pgwrite_req_t b = { .offset = offset, .pathid = 0, .reqflags = 0 };
546 memcpy(b.fhandle, f->fhandle, XRD_FHANDLE_LEN);
547 xrdw_pgwrite_req_pack(&b, ((ClientRequestHdr *) &req)->body);
548 }
549 /* dlen (offset 20) is set to plen by brix_send. */
550
551 {
552 brix_payload pl = { payload, (uint32_t) plen };
553 if (brix_send(c, &req, &pl, &sid, st) != 0) {
554 free(payload);
555 return -1;
556 }
557 }
558 if (read_status_frame(c, sid, &resptype, &pgdlen, &foff, st) != 0) {
559 free(payload); /* kXR_ChkSumErr (CRC rejected) surfaces here */
560 return -1;
561 }
562 if (pgdlen != 0) {
563 /* CSE: the server wrote the data but reported pages that failed CRC32c
564 * (typically wire corruption — our encode is always correct). Read the
565 * retransmit list and resend each page with kXR_pgRetry until the server
566 * accepts it or we exhaust the bounded attempts. */
567 int rc_ret = pgwrite_handle_cse(c, f, (const uint8_t *) buf, offset,
568 len, pgdlen, st);
569 free(payload);
570 return rc_ret;
571 }
572 free(payload);
573 return 0;
574}
