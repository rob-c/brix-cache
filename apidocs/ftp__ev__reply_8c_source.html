4/*
5 * ftp_ev_reply.c — reply framing for the event engine.
6 *
7 * WHAT: append-only reply helpers that write into the connection's outbound
8 * buffer (->ob): a printf-style status reply, an AUTH/ADAT token continuation,
9 * plus the base64 codecs the RFC 2228 handshake needs.
10 *
11 * WHY: the engine is non-blocking, so replies are never written inline — they are
12 * queued into ->ob and drained by brix_ftp_ev_flush() under the event loop. Once
13 * the GSI security layer is active every status reply is transparently GSS-wrapped
14 * into a 631/632/633 safe/private frame; the pre-auth handshake (220/334/335/235)
15 * is emitted while ->sec_active is still 0 and so goes out cleartext.
16 *
17 * HOW: ev_out_put() is the single choke point that copies bytes into ->ob with a
18 * hard capacity check. Because commands are processed lock-step (the engine
19 * drains ->ob before framing the next command), at most one command's replies are
20 * resident at a time, so the fixed 64 KiB buffer cannot be overrun by a
21 * well-behaved reply path.
22 */

25/* Copy `len` reply bytes into the outbound buffer. Fails closed (NGX_ERROR) if
26 * the reply would exceed the buffer — a reply that large is a bug, not traffic. */
27static ngx_int_t
28ev_out_put(ftp_ev_t *fc, const u_char *data, size_t len)
29{
30 if (len > BRIX_FTP_EV_OB_CAP - fc->ob_len) {
31 BRIX_LOG(NGX_LOG_ERR, fc->c->log, 0,
32 "brix: GridFTP(ev) reply exceeds outbound buffer (%uz)",
33 len);
34 return NGX_ERROR;
35 }
36 ngx_memcpy(fc->ob + fc->ob_len, data, len);
37 fc->ob_len += len;
38 return NGX_OK;
39}

75/* GSS-wrap `n` reply bytes and queue them as an RFC 2228 safe/private reply
76 * ("631/632/633 <b64>") over the protected control channel. */
77static ngx_int_t
78ev_reply_wrapped(ftp_ev_t *fc, const u_char *plain, size_t n)
79{
80 ngx_str_t tok, b64;
81 u_char *line, *p;
82 size_t need;
83
84 if (brix_gssapi_wrap(fc->gss, plain, n, &tok) != NGX_OK
85 || ev_b64_encode(fc->c->pool, tok.data, tok.len, &b64) != NGX_OK)
86 {
87 return NGX_ERROR;
88 }
89 need = b64.len + ngx_strlen(fc->wrap_code) + sizeof(" \r\n");
90 line = ngx_pnalloc(fc->c->pool, need);
91 if (line == NULL) {
92 return NGX_ERROR;
93 }
94 p = ngx_slprintf(line, line + need, "%s %V\r\n", fc->wrap_code, &b64);
95 return ev_out_put(fc, line, (size_t) (p - line));
96}
