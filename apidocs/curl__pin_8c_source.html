13size_t
14brix_dns_curl_pin_entry(const char *host, in_port_t port,
15 const brix_dns_addr_t *addrs, ngx_uint_t naddrs, char *out, size_t sz)
16{
17 char ip[BRIX_DNS_NTOP_LEN];
18 size_t n;
19 ngx_uint_t i;
20
21 n = brix_format_host_port(host, port, out, sz);
22 if (n == 0 || naddrs == 0) {
23 return 0;
24 }
25 for (i = 0; i < naddrs; i++) {
26 if (brix_dns_addr_ntop(&addrs[i], ip, sizeof(ip)) == 0) {
27 continue;
28 }
29 if (n + 1 + ngx_strlen(ip) + 1 > sz) {
30 return 0;
31 }
32 out[n++] = (i == 0) ? ':' : ',';
33 n += ngx_cpystrn((u_char *) out + n, (u_char *) ip, sz - n)
34 - ((u_char *) out + n);
35 }
36 return n;
37}

40static in_port_t
41curl_pin_port(const brix_net_target_t *t)
42{

44 return t->port;

46 if (t->scheme.len == 4 && ngx_strncasecmp(t->scheme.data,

48 {
49 return CURL_PIN_DEFAULT_HTTP;
50 }
51 return CURL_PIN_DEFAULT_HTTPS;
52}

55ngx_int_t
56brix_dns_curl_pin(CURL *curl, const brix_dns_policy_t *policy, const char *url,
57 struct curl_slist **resolve_out, char *err, size_t errsz)
58{
59 brix_net_target_t tgt;
60 brix_dns_addr_t addrs[BRIX_DNS_MAX_ADDRS], lit;
61 ngx_str_t url_str, host;
62 char hostz[BRIX_RESOLV_DOMAIN_LEN];
63 char entry[BRIX_RESOLV_DOMAIN_LEN + 8
64 + BRIX_DNS_MAX_ADDRS * BRIX_DNS_NTOP_LEN];
65 struct curl_slist *list;
66 ngx_uint_t n;
67 in_port_t port;
68
69 *resolve_out = NULL;
70 /* a reused handle must never keep a list its previous request freed */
71 (void) curl_easy_setopt(curl, CURLOPT_RESOLVE, NULL);
72 url_str.data = (u_char *) url;
73 url_str.len = ngx_strlen(url);
74 if (brix_net_target_parse(NULL, &url_str, &tgt, err, errsz) != NGX_OK) {
75 return NGX_ERROR;
76 }
77 host = tgt.host;
78 if (host.len == 0 || host.len >= sizeof(hostz)) {
79 (void) ngx_snprintf((u_char *) err, errsz, "invalid URL host%Z");
80 return NGX_ERROR;
81 }
82 if (brix_dns_parse_literal(&host, &lit) == NGX_OK) {
83 return NGX_OK; /* literal: libcurl needs no DNS */
84 }
85 ngx_cpystrn((u_char *) hostz, host.data, host.len + 1);
86 port = curl_pin_port(&tgt);
87
88 n = brix_dns_resolve_sync(policy, hostz, port, BRIX_AF_AUTO, SOCK_STREAM,
89 addrs, BRIX_DNS_MAX_ADDRS, err, errsz);
90 if (n == 0) {
91 return NGX_ERROR;
92 }
93 if (brix_dns_curl_pin_entry(hostz, port, addrs, n, entry, sizeof(entry))
94 == 0)
95 {
96 (void) ngx_snprintf((u_char *) err, errsz, "resolve entry overflow%Z");
97 return NGX_ERROR;
98 }
99 list = curl_slist_append(NULL, entry);
100 if (list == NULL) {
101 (void) ngx_snprintf((u_char *) err, errsz, "curl_slist_append failed%Z");
102 return NGX_ERROR;
103 }
104 if (curl_easy_setopt(curl, CURLOPT_RESOLVE, list) != CURLE_OK) {
105 curl_slist_free_all(list);
106 (void) ngx_snprintf((u_char *) err, errsz, "CURLOPT_RESOLVE failed%Z");
107 return NGX_ERROR;
108 }
109 *resolve_out = list;
110 return NGX_OK;
111}

127/* WHAT: the transfer's total ms budget — the caller's, or the default.
128 * WHY: a caller that names none must still be bounded; 0 would mean
129 * "for ever" to libcurl, which is the denial-of-service this guards. */
130static long
131curl_pin_budget(const brix_dns_curl_transfer_t *t)
132{
133 return (t->timeout_ms > 0) ? t->timeout_ms
134 : (long) BRIX_DNS_CURL_TIMEOUT_MS_DEFAULT;
135}
136
137
138/* WHAT: what is left of the budget, 0 once it is spent.
139 * HOW: `spent_us` is the sum of CURLINFO_TOTAL_TIME_T over the hops already
140 * performed — libcurl's own clock, so the loop needs none of its own
141 * (ngx_current_msec does not advance on a thread-pool thread). */
142static long
143curl_pin_remaining(long budget_ms, curl_off_t spent_us)
144{
145 long spent_ms = (long) (spent_us / BRIX_DNS_US_TO_MS_DIVISOR);
146
147 return (spent_ms >= budget_ms) ? 0 : budget_ms - spent_ms;
148}

151/* WHAT: microseconds the attempt just performed took, 0 if libcurl cannot say.
152 */
153static curl_off_t
154curl_pin_spent(CURL *curl)
155{
156 curl_off_t us = 0;
157
158 if (curl_easy_getinfo(curl, CURLINFO_TOTAL_TIME_T, &us) != CURLE_OK) {
159 return 0;
160 }
161 return (us > 0) ? us : 0;
162}

165/* WHAT: bound this hop.
166 * WHY: the wrapper performs on a handle it did not create, so the bound has
167 * to be set here rather than trusted to the caller; NOSIGNAL keeps
168 * libcurl from raising SIGALRM on a thread-pool thread. */
169static void
170curl_pin_bound(CURL *curl, long timeout_ms)
171{
172 (void) curl_easy_setopt(curl, CURLOPT_NOSIGNAL, 1L);
173 (void) curl_easy_setopt(curl, CURLOPT_TIMEOUT_MS, timeout_ms);
174 (void) curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT_MS, timeout_ms);
175}

178/* WHAT: one attempt of the pinned loop — pin, aim, rewind the sink, perform,
179 * and read the Location libcurl would have followed.
180 * HOW: *next is NULL unless the response was a redirect; it points into the
181 * handle and is valid until the next perform, so the caller copies it
182 * before looping. */
183static CURLcode
184curl_pin_attempt(CURL *curl, const brix_dns_curl_transfer_t *t,
185 const char *target, long timeout_ms, char **next)
186{
187 struct curl_slist *resolve;
188 CURLcode res;
189
190 *next = NULL;
191 if (brix_dns_curl_pin(curl, t->policy, target, &resolve, t->err, t->errsz)
192 != NGX_OK)
193 {
194 return CURLE_COULDNT_RESOLVE_HOST;
195 }
196 curl_easy_setopt(curl, CURLOPT_URL, target);
197 curl_pin_bound(curl, timeout_ms);
198 if (t->on_hop != NULL) {
199 t->on_hop(t->hop_data);
200 }
201 res = curl_easy_perform(curl);
202 if (res == CURLE_OK) {
203 curl_easy_getinfo(curl, CURLINFO_REDIRECT_URL, next);
204 }
205 curl_slist_free_all(resolve);
206 return res;
207}

210CURLcode
211brix_dns_curl_perform_pinned(CURL *curl, const brix_dns_curl_transfer_t *t)
212{
213 char target[BRIX_DNS_CURL_URL_MAX];
214 char *next;
215 CURLcode res;
216 ngx_uint_t hop;
217 long budget, left;
218 curl_off_t spent;
219
220 if (t->err != NULL && t->errsz > 0) {
221 t->err[0] = '\0';
222 }
223 budget = curl_pin_budget(t);
224 spent = 0;
225 res = curl_pin_copy_target(t->url, target, sizeof(target), t->err,
226 t->errsz);
227 for (hop = 0; res == CURLE_OK; hop++) {
228 left = curl_pin_remaining(budget, spent);
229 if (left == 0) {
230 (void) ngx_snprintf((u_char *) t->err, t->errsz,
231 "timeout budget spent after %ui hop(s)%Z", hop);
232 return CURLE_OPERATION_TIMEDOUT;
233 }
234 res = curl_pin_attempt(curl, t, target, left, &next);
235 spent += curl_pin_spent(curl);
236 if (res != CURLE_OK || next == NULL) {
237 return res;
238 }
239 if (hop >= t->max_redirects) {
240 (void) ngx_snprintf((u_char *) t->err, t->errsz,
241 "too many redirects (%ui)%Z", hop);
242 return CURLE_TOO_MANY_REDIRECTS;
243 }
244 res = curl_pin_copy_target(next, target, sizeof(target), t->err,
245 t->errsz);
246 }
247 return res;
248}
