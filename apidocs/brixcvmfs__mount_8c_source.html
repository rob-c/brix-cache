1/*
2 * brixcvmfs_mount.c — CVMFS-brix trust-chain mount pipeline (Phase-38 split).
3 *
4 * WHAT: bring-up of one mounted repo — master-key loading (single file or a
5 * rotated *.pub directory), the failover/transport/quota config cascade,
6 * root-catalog pin + G4/G5 cache format, and the verify-mount itself —
7 * plus the two no-mount commands `--check` and `--prewarm`.
8 * WHY: split from brixcvmfs.c to keep each TU within the file-size budget;
9 * this is the "prove the repo is trustworthy and open it" concern,
10 * separate from FUSE dispatch, the transport, and option parsing.
11 * HOW: brixcvmfs_open assembles the client, seeds the process-global transport
12 * config, brings the libcurl pool up, and hands the injected transport to
13 * the shared cvmfs core, which owns the whitelist/manifest/catalog trust

29static unsigned char *read_file(const char *path, size_t *len) {
30 FILE *f = fopen(path, "rb");
31 if (f == NULL) return NULL;
32 fseek(f, 0, SEEK_END);
33 long n = ftell(f);
34 fseek(f, 0, SEEK_SET);
35 unsigned char *b = malloc(n > 0 ? (size_t) n : 1);
36 if (b && fread(b, 1, (size_t) n, f) != (size_t) n) { free(b); b = NULL; }
37 fclose(f);
38 if (b) *len = (size_t) n;
39 return b;
40}

42/* Load the repo master key(s). If `path` is a directory (the stock
43 * /etc/cvmfs/keys/<domain>/ layout), concatenate every *.pub in it — CVMFS
44 * rotates master keys and the whitelist is signed by one of them. */
45static unsigned char *load_master_key(const char *path, size_t *len) {
46 struct stat st;
47 if (stat(path, &st) != 0) return NULL;
48 if (!S_ISDIR(st.st_mode)) return read_file(path, len);
49
50 DIR *d = opendir(path);
51 if (d == NULL) return NULL;
52 unsigned char *buf = NULL; size_t cap = 0, used = 0;
53 struct dirent *e;
54 while ((e = readdir(d)) != NULL) {
55 const char *dot = strrchr(e->d_name, '.');
56 if (dot == NULL || strcmp(dot, ".pub") != 0) continue;
57 char full[1024];
58 snprintf(full, sizeof(full), "%s/%s", path, e->d_name);
59 size_t kl = 0;
60 unsigned char *k = read_file(full, &kl);
61 if (k == NULL) continue;
62 if (used + kl + 1 > cap) { cap = (used + kl + 1) * 2;
63 unsigned char *nb = realloc(buf, cap); if (nb == NULL) { free(k); continue; } buf = nb; }
64 memcpy(buf + used, k, kl); used += kl;
65 buf[used++] = '\n';
66 free(k);
67 }
68 closedir(d);
69 if (buf) *len = used;
70 return buf;
71}

73/* Populate the client failover set for `repo`. $BRIXCVMFS_SERVER pins a single
74 * DIRECT stratum-1; otherwise apply the loaded config cascade, and if that yields
75 * no hosts, synthesise the conventional cvmfs-stratum-one.<domain> fallback. */
76static void brixcvmfs_build_failover(cvmfs_client_t *cl, const cvmfs_conf_t *cf,
77 const char *repo) {
78 cvmfs_failover_init(&cl->fo, 60);
79 const char *env_server = getenv("BRIXCVMFS_SERVER");
80 if (env_server != NULL) {
81 cvmfs_failover_add_proxy(&cl->fo, "DIRECT", 0);
82 cvmfs_failover_add_host(&cl->fo, env_server);
83 return;
84 }
85 int hosts = cvmfs_conf_apply(cf, repo, &cl->config, &cl->fo);
86 if (hosts == 0) {
87 char s1[400];
88 snprintf(s1, sizeof(s1), "http://cvmfs-stratum-one.%s/cvmfs/%s",
89 strchr(repo, '.') + 1, repo);
90 cvmfs_failover_add_host(&cl->fo, s1);
91 }
92}

107/* Resolve the effective cache quota in bytes: a positive `quota_override` wins;
108 * otherwise fall back to CVMFS_QUOTA_LIMIT (MB) from the conf cascade; else 0. */
109static long brixcvmfs_resolve_quota(const cvmfs_conf_t *cf, long quota_override) {
110 if (quota_override > 0) return quota_override;
111 const char *q = cvmfs_conf_get(cf, "CVMFS_QUOTA_LIMIT"); /* MB */
112 if (q && atol(q) > 0) return atol(q) * 1024L * 1024L;
113 return 0;
114}

116/* Load the repo master key: $BRIXCVMFS_PUBKEY overrides the config default path.
117 * A config-derived CVMFS_KEYS_DIR/<domain>.pub that does not exist falls back to
118 * the key-chain DIRECTORY itself (stock layouts rotate keys as e.g.
119 * keys/cern.ch/cern-it4.cern.ch.pub with no cern.ch.pub; load_master_key
120 * concatenates every *.pub and the verifier tries each). Returns the allocated
121 * key blob (caller frees) + `*klen`, or NULL on read failure (message printed). */
122static unsigned char *brixcvmfs_load_repo_key(const cvmfs_client_t *cl, size_t *klen) {
123 const char *env_key = getenv("BRIXCVMFS_PUBKEY");
124 const char *keypath = env_key ? env_key : cl->config.master_pub_path;
125 unsigned char *master = load_master_key(keypath, klen);
126 if (master == NULL && env_key == NULL) {
127 char chain[512];
128 snprintf(chain, sizeof(chain), "%s", keypath);
129 char *slash = strrchr(chain, '/');
130 if (slash != NULL && slash != chain) {
131 *slash = '\0';
132 master = load_master_key(chain, klen);
133 }
134 }
135 if (master == NULL)
136 fprintf(stderr, "brixcvmfs: cannot read master key %s\n", keypath);
137 return master;
138}

153 * in overlay-dirfd mode (`cache_dirfd < 0`). Precedence: explicit override →

162 char mk[600];

167/* Root-catalog pin + G4/G5 cache format (phase-87): -o pin= / cache_format=
168 * packed / cache_tiering, env fallbacks for each. $BRIXCVMFS_CACHE_SEG_BYTES
169 * is a test hook to shrink segments so rollover/eviction is exercisable with
170 * tiny corpora. -1 = bad pin (message printed). */
171static int brixcvmfs_open_pin_cache(cvmfs_client_t *cl, const char *pin_opt,
172 int packed_opt, int tiering_opt) {
173 const char *pin = (pin_opt != NULL && pin_opt[0]) ? pin_opt : getenv("BRIXCVMFS_PIN");
174 if (pin != NULL && pin[0] && cvmfs_client_pin_root(cl, pin) != 0) {
175 fprintf(stderr, "brixcvmfs: bad pin '%s' (want a root-catalog hash)\n", pin);
176 return -1;
177 }
178
179 const char *env_cf = getenv("BRIXCVMFS_CACHE_FORMAT");
180 const char *env_ct = getenv("BRIXCVMFS_CACHE_TIERING");
181 const char *env_sb = getenv("BRIXCVMFS_CACHE_SEG_BYTES");
182 int packed = packed_opt || (env_cf != NULL && strcmp(env_cf, "packed") == 0);
183 int tiering = tiering_opt || (env_ct != NULL && atoi(env_ct) != 0);
184 if (packed)
185 cvmfs_client_cache_config(cl, 1, tiering, env_sb ? atol(env_sb) : 0);
186
187 return 0;
188}

190/* Build failover + config and verify-mount the repo trust chain. Cache backing:
191 * `cache_dirfd >= 0` = overlay dirfd mode; else `cache_dir_override` (or the
192 * default/env path). quota/retries fall back to the config cascade when the
193 * override is unset (<=0 quota, <0 retries). Returns a mounted client or NULL. */
194cvmfs_client_t *brixcvmfs_open(const char *repo, const char *cache_dir_override,
195 int cache_dirfd, long quota_override, int retries_override,
196 const char *pin_opt, int packed_opt, int tiering_opt) {
197 cvmfs_client_t *cl = calloc(1, sizeof(*cl));
198 if (cl == NULL) { fprintf(stderr, "brixcvmfs: out of memory\n"); return NULL; }
199
200 if (cvmfs_repo_config_defaults(repo, &cl->config) != 0) {
201 fprintf(stderr, "brixcvmfs: '%s' is not a fully-qualified repo name\n", repo);
202 free(cl); return NULL;
203 }
204
205 if (brixcvmfs_open_pin_cache(cl, pin_opt, packed_opt, tiering_opt) != 0) {
206 free(cl); return NULL;
207 }
208
209 cvmfs_conf_t cf;
210 cvmfs_conf_init(&cf);
211 cvmfs_conf_load_cascade(&cf, getenv("BRIXCVMFS_ETC"), repo); /* for quota/retries */
212
213 brixcvmfs_build_failover(cl, &cf, repo);
214 brixcvmfs_build_transport_cfg(cl, &cf, repo, retries_override);
215 long quota = brixcvmfs_resolve_quota(&cf, quota_override);
216
217 size_t klen = 0;
218 unsigned char *master = brixcvmfs_load_repo_key(cl, &klen);
219 if (master == NULL) { free(cl); return NULL; }
220
221 char tmp_dir[512];
222 brixcvmfs_prepare_tmp_dir(repo, tmp_dir, sizeof(tmp_dir));
223
224 char cache_dir[512] = "";
225 brixcvmfs_prepare_cache_dir(repo, cache_dir_override, cache_dirfd,
226 cache_dir, sizeof(cache_dir));
227
228 curl_global_init(CURL_GLOBAL_DEFAULT);
229 /* Bring the shared curl-handle pool up before the mount fetches the root
230 * catalog (eager slot-0 connect surfaces a curl_easy_init failure here). */
231 brix_status pst; brix_status_clear(&pst);
232 if (brixcvmfs_transport_pool_init(&pst) != 0) {
233 fprintf(stderr, "brixcvmfs: curl handle pool init failed: %s\n", pst.msg);
234 curl_global_cleanup();
235 free(cl); return NULL;
236 }
237 int mrc = cvmfs_client_mount(cl, repo, master, klen,
238 cache_dirfd < 0 ? cache_dir : NULL, tmp_dir,
239 quota, cache_dirfd, brixcvmfs_transport, NULL, mono_now());
240 free(master);
241 if (mrc != 0) {
242 fprintf(stderr, "brixcvmfs: mount of %s failed (trust/catalog error %d)\n", repo, mrc);
243 transport_cleanup();
244 curl_global_cleanup();
245 free(cl); return NULL;
246 }
247 return cl;
248}

250/* --check: verify the trust chain + root catalog and print a summary WITHOUT
251 * mounting (the stock `cvmfs_config chksetup` analog). Exit 0 = healthy. */
252int brixcvmfs_check(const char *repo) {
253 cvmfs_client_t *cl = brixcvmfs_open(repo, NULL, -1, 0, -1, NULL, 0, 0);
254 if (cl == NULL) return 1;
255
256 long now = mono_now();
257 char rev[64] = "?", root[128] = "?", host[256] = "?", proxy[64] = "?";
258 int n;
259 n = cvmfs_client_getxattr(cl, "/", "user.revision", rev, sizeof(rev) - 1, now);
260 if (n > 0) rev[n] = 0;
261 n = cvmfs_client_getxattr(cl, "/", "user.root_hash", root, sizeof(root) - 1, now);
262 if (n > 0) root[n] = 0;
263 n = cvmfs_client_getxattr(cl, "/", "user.host", host, sizeof(host) - 1, now);
264 if (n > 0) host[n] = 0;
265 n = cvmfs_client_getxattr(cl, "/", "user.proxy", proxy, sizeof(proxy) - 1, now);
266 if (n > 0) proxy[n] = 0;
267
268 cvmfs_dirent_t root_e;
269 /* the catalog root entry is the empty path "" (FUSE maps "/" → "" via cat_path). */
270 int root_ok = cvmfs_client_resolve(cl, "", &root_e, now) == 1
271 && (root_e.flags & CVMFS_FLAG_DIR);
272 int entries = cvmfs_catalog_readdir(cl->root_catalog, "", NULL, NULL);
273
274 printf("CVMFS-brix repository check: %s\n", repo);
275 printf(" trust chain .... OK (whitelist + manifest signature verified)\n");
276 printf(" revision ....... %s\n", rev);
277 printf(" root catalog ... %s\n", root);
278 printf(" root dir ....... %s (%d entries)\n", root_ok ? "OK" : "MISSING", entries);
279 printf(" active server .. %s\n", host);
280 printf(" active proxy ... %s\n", proxy);
281 printf(" ttl ............ %lds\n", cl->ttl);
282 printf("HEALTHY\n");
283
284 cvmfs_client_umount(cl);
285 transport_cleanup();
286 curl_global_cleanup();
287 free(cl);
288 return root_ok ? 0 : 1;
289}

291/* --prewarm (phase-85 F5): walk the WHOLE snapshot (the pin when
292 * $BRIXCVMFS_PIN is set, else the current root) and pull every referenced CAS
293 * object into the local cache, so a shared cache dir is fully warm before a
294 * job wave. No mount, single-threaded: reuses the client's own fetch ctx.
295 * Exit 0 = every object landed; a fetch error or a tampered catalog ⇒ 1. */
296typedef struct {
297 cvmfs_client_t *cl;
298 cvmfs_fetch_ctx_t *fx;
299 unsigned char *out;
300 cvmfs_failover_t fo0; /* pristine snapshot (blacklist reset, cf. F4) */
301 long objs, bytes, errs;
302} prewarm_ud_t;
303
304static int prewarm_visit(const cvmfs_walk_item_t *it, void *ud) {
305 prewarm_ud_t *p = ud;
306 if (it->kind == CVMFS_WALK_CATALOG) return 0; /* the walk itself caches it */
307 size_t n = 0;
308 /* This walk drives cvmfs_fetch_object directly, so it owns the landing-buffer
309 * sizing the read path does for itself. `it->size` is the catalog's plaintext
310 * size (0 when unrecorded — fall back to the whole object cap). */
311 if (cvmfs_client_scratch_reserve(p->cl, it->size ? (size_t) it->size
312 : (size_t) BRIX_PF_OBJCAP) != 0) {
313 p->errs++;
314 return 0;
315 }
316 if (cvmfs_fetch_object(p->fx, &it->hash, it->suffix,
317 p->out, BRIX_PF_OBJCAP, &n, mono_now()) == 0) {
318 p->objs++;
319 p->bytes += (long) n;
320 } else {
321 p->errs++;
322 *p->fx->fo = p->fo0; /* one bad object blacklists its route — restore
323 * so it can't shadow the rest of the sweep */
324 }
325 return 0;
326}

328int brixcvmfs_prewarm(const char *repo) {
329 cvmfs_client_t *cl = brixcvmfs_open(repo, NULL, -1, 0, -1, NULL, 0, 0);
330 if (cl == NULL) return 1;
331
332 prewarm_ud_t ud = { cl, &cl->fetch, malloc(BRIX_PF_OBJCAP), cl->fo, 0, 0, 0 };
333 int rc = -1;
334 if (ud.out != NULL) {
335 const cvmfs_hash_t *root = cl->pin_set ? &cl->pin_root
336 : &cl->manifest.root_catalog;
337 rc = cvmfs_walk_catalog(&cl->fetch, root, cl->catalog_tmp,
338 INT_MAX, prewarm_visit, &ud, mono_now());
339 char hex[48];
340 cvmfs_hash_to_hex(root, 0, hex, sizeof(hex));
341 printf("CVMFS-brix prewarm: %s\n", repo);
342 printf(" root catalog ... %s%s\n", hex, cl->pin_set ? " (pinned)" : "");
343 printf(" objects ........ %ld fetched (%ld bytes)\n", ud.objs, ud.bytes);
344 printf(" errors ......... %ld%s\n", ud.errs,
345 rc != 0 ? " (walk aborted: catalog fetch/verify failure)" : "");
346 printf("%s\n", rc == 0 && ud.errs == 0 ? "WARM" : "INCOMPLETE");
347 } else {
348 fprintf(stderr, "brixcvmfs: out of memory\n");
349 }
350 free(ud.out);
351
352 cvmfs_client_umount(cl);
353 transport_cleanup();
354 curl_global_cleanup();
355 free(cl);
356 return rc == 0 && ud.errs == 0 ? 0 : 1;
357}
