22/* usage_check — print usage and return rc. */
23static int
24usage_check(const char *prog, int rc)
25{
26 fprintf(stderr,
27 "usage: %s <manifest> <root> [--algo NAME]\n"
28 " Verify every file listed in a tree manifest against its stored digest.\n"
29 " manifest path to manifest file produced by `xrdcksum tree`\n"
30 " root local directory or root:// URL that was checksummed\n"
31 " --algo N digest algorithm (default: infer from hex length)\n"
32 " Without --algo, digests are assumed: adler32 (8 hex), crc64 (16 hex),\n"
33 " md5 (32 hex); use --algo for crc32c, zcrc32, crc64nvme manifests.\n"
34 " For each line prints 'OK <rel>' or 'FAILED <rel>'.\n"
35 " exit: 0 all OK, 1 any mismatch, 2 errors (parse/I/O); 2 (errors)\n"
36 " takes precedence over 1 (mismatch) when both occur\n",
37 prog);
38 return rc;
39}

54/* ckcheck_parse_args — decode the `xrdcksum check` command line.
55 *
56 * WHAT: Validate argc and parse the optional --algo flag; on any usage error
57 * (or -h) set *rc to the exit code and return -1; return 0 to proceed
58 * (manifest = argv[1], root = argv[2]).
59 * WHY: Keeps the option ladder out of the main flow.
60 * HOW: argv[1]/argv[2] are positional; scan from argv[3]; --algo consumes
61 * and validates the next argument via brix_cksum_algo_parse. */
62static int
63ckcheck_parse_args(int argc, char **argv, const char **algo_str,
64 brix_cksum_algo *algo, int *rc)
65{
66 int i;
67
68 *algo_str = NULL;
69 if (argc < 3) {
70 *rc = usage_check(argv[0], XRDC_EXIT_USAGE);
71 return -1;
72 }
73 for (i = 3; i < argc; i++) {
74 if (strcmp(argv[i], "--algo") == 0) {
75 if (++i >= argc) {
76 *rc = usage_check(argv[0], XRDC_EXIT_USAGE);
77 return -1;
78 }
79 *algo_str = argv[i];
80 if (brix_cksum_algo_parse(*algo_str, algo) != 0) {
81 fprintf(stderr, "%s: unknown algorithm '%s'\n",
82 argv[0], *algo_str);
83 *rc = XRDC_EXIT_USAGE;
84 return -1;
85 }
86 } else if (strcmp(argv[i], "-h") == 0 || strcmp(argv[i], "--help") == 0) {
87 *rc = usage_check(argv[0], 0);
88 return -1;
89 } else {
90 fprintf(stderr, "%s: unexpected argument '%s'\n", argv[0], argv[i]);
91 *rc = usage_check(argv[0], XRDC_EXIT_USAGE);
92 return -1;
93 }
94 }
95 return 0;
96}

98/* ckcheck_connect — parse the remote root URL and open the one connection.
99 *
100 * WHAT: Fill *u from `root`, enforce root/roots scheme, and connect *c.
101 * Return 0 on success, -1 after printing the error.
102 * WHY: One connection is reused across every manifest line; setup failures
103 * must abort the whole run before any line is processed.
104 * HOW: brix_url_parse + scheme check → brix_connect; messages match the
105 * historical output exactly. */
106static int
107ckcheck_connect(const char *prog, const char *root, brix_url *u, brix_conn *c)
108{
109 brix_status st;
110
111 brix_status_clear(&st);
112 if (brix_url_parse(root, u, &st) != 0
113 || (u->scheme != XRDC_SCHEME_ROOT
114 && u->scheme != XRDC_SCHEME_ROOTS)) {
115 fprintf(stderr, "%s: bad root URL: %s\n",
116 prog, st.msg[0] ? st.msg : "parse error");
117 return -1;
118 }
119 if (brix_connect(c, u, NULL, &st) != 0) {
120 fprintf(stderr, "%s: connect: %s\n", prog, st.msg);
121 return -1;
122 }
123 return 0;
124}

143/* ckcheck_line_algo — resolve the digest algorithm for one manifest line.
144 *
145 * WHAT: Set *out_algo / *out_name for the line: --algo when given, otherwise
146 * inferred from the hex digest length. Return 0 on success, -1 after
147 * printing the unrecognised-length error (caller counts it).
148 * WHY: Manifests do not record the algorithm; the historical inference is
149 * part of the CLI contract and must stay EXACT: adler32 = 8 hex chars,
150 * crc64 (CRC-64/XZ) = 16, md5 = 32. crc32c and zcrc32 also produce
151 * 8 hex chars and are indistinguishable by length — they require
152 * --algo. CRITICAL (INVARIANT 9): crc64 and crc64nvme are DIFFERENT
153 * polynomials; 16 hex maps to crc64 ONLY, and crc64nvme manifests must
154 * use --algo. Never merge or "normalize" the two.
155 * HOW: --algo short-circuit, else switch on strlen(hex). */
156static int
157ckcheck_line_algo(const ckcheck_ctx *cx, const char *hex, const char *rel,
158 brix_cksum_algo *out_algo, const char **out_name)
159{
160 size_t hexlen;
161
162 if (cx->algo_str != NULL) {
163 *out_algo = cx->algo;
164 *out_name = cx->algo_str;
165 return 0;
166 }
167 hexlen = strlen(hex);
168 switch (hexlen) {
169 case 8: *out_algo = XRDC_CK_ADLER32; *out_name = "adler32"; return 0;
170 case 16: *out_algo = XRDC_CK_CRC64; *out_name = "crc64"; return 0;
171 case 32: *out_algo = XRDC_CK_MD5; *out_name = "md5"; return 0;
172 default:
173 fprintf(stderr, "%s: unrecognised hex length %zu for '%s'\n",
174 cx->prog, hexlen, rel);
175 return -1;
176 }
177}

179/* ckcheck_digest_local — compute the actual digest of a local file.
180 *
181 * WHAT: Open cx->root/rel and compute its checksum into got (TREE_HEX_MAX
182 * bytes). Return 0 on success, -1 after printing the error.
183 * WHY: The local half of per-line verification; every failure is one
184 * stderr line and one counted error at the call site.
185 * HOW: Overflow-checked ckt_path_join → open O_RDONLY → brix_cksum_fd. */
186static int
187ckcheck_digest_local(const ckcheck_ctx *cx, const char *rel,
188 brix_cksum_algo algo, char *got)
189{
190 char lpath[XRDC_PATH_MAX];
191 brix_status st;
192 int fd;
193
194 if (ckt_path_join(cx->root, rel, lpath, sizeof(lpath)) != 0) {
195 fprintf(stderr, "%s: path too long: %s/%s\n",
196 cx->prog, cx->root, rel);
197 return -1;
198 }
199 fd = open(lpath, O_RDONLY);
200 if (fd < 0) {
201 fprintf(stderr, "%s: open %s: %s\n",
202 cx->prog, lpath, strerror(errno));
203 return -1;
204 }
205 brix_status_clear(&st);
206 if (brix_cksum_fd(fd, algo, got, TREE_HEX_MAX, &st) != 0) {
207 fprintf(stderr, "%s: %s: %s\n", cx->prog, lpath,
208 st.msg[0] ? st.msg : "checksum error");
209 close(fd);
210 return -1;
211 }
212 close(fd);
213 return 0;
214}

216/* ckcheck_digest_remote — query the server for a remote file's digest.
217 *
218 * WHAT: Build root_path + "/" + rel and ask the server (over the reused
219 * connection) for its digest into got (TREE_HEX_MAX bytes). Return 0
220 * on success, -1 after printing the error.
221 * WHY: The remote half of per-line verification; the server computes, we
222 * compare.
223 * HOW: Overflow-checked snprintf → brix_query_cksum with the per-line
224 * algorithm name. */
225static int
226ckcheck_digest_remote(const ckcheck_ctx *cx, const char *rel,
227 const char *algo_name, char *got)
228{
229 char rpath[XRDC_PATH_MAX];
230 brix_status st;
231 int n;
232
233 n = snprintf(rpath, sizeof(rpath), "%s/%s", cx->root_path, rel);
234 if (n <= 0 || (size_t) n >= sizeof(rpath)) {
235 fprintf(stderr, "%s: path too long: %s/%s\n",
236 cx->prog, cx->root_path, rel);
237 return -1;
238 }
239 brix_status_clear(&st);
240 if (brix_query_cksum(cx->c, rpath, algo_name,
241 got, TREE_HEX_MAX, &st) != 0) {
242 fprintf(stderr, "%s: query_cksum %s: %s\n", cx->prog, rpath,
243 st.msg[0] ? st.msg : "query error");
244 return -1;
245 }
246 return 0;
247}

249/* ckcheck_process_line — verify one manifest line and print its verdict.
250 *
251 * WHAT: Parse "<hex> <rel>" via brix_ckmf_parse_line, resolve the line's
252 * algorithm, obtain the actual digest (local compute or remote query),
253 * and print "OK <rel>" or "FAILED <rel>".
254 * WHY: One line = one independent verification; every failure mode counts
255 * an error and moves on so a single bad record never aborts the run.
256 * HOW: brix_ckmf_parse_line is the sole gate on the check path (hostile
257 * manifests cannot escape the root) → ckcheck_line_algo →
258 * ckcheck_digest_local/_remote → case-insensitive compare. */
259static void
260ckcheck_process_line(const ckcheck_ctx *cx, char *line)
261{
262 char hex[TREE_HEX_MAX];
263 char rel[XRDC_PATH_MAX];
264 char got[TREE_HEX_MAX];
265 brix_cksum_algo line_algo;
266 const char *line_algo_name;
267 int drc;
268
269 if (brix_ckmf_parse_line(line, hex, sizeof(hex),
270 rel, sizeof(rel)) != 0) {
271 ckcheck_report_malformed(cx->prog, line);
272 (*cx->errors)++;
273 return;
274 }
275 if (ckcheck_line_algo(cx, hex, rel, &line_algo, &line_algo_name) != 0) {
276 (*cx->errors)++;
277 return;
278 }
279
280 got[0] = '\0';
281 if (!cx->is_remote) {
282 drc = ckcheck_digest_local(cx, rel, line_algo, got);
283 } else {
284 drc = ckcheck_digest_remote(cx, rel, line_algo_name, got);
285 }
286 if (drc != 0) {
287 (*cx->errors)++;
288 return;
289 }
290
291 if (strcasecmp(hex, got) == 0) {
292 printf("OK %s\n", rel);
293 } else {
294 printf("FAILED %s\n", rel);
295 (*cx->mismatches)++;
296 }
297}

299/*
300 * brix_xrdckcheck_main — xrdcksum check: manifest verification.
301 *
302 * WHAT: Parse a manifest produced by `xrdcksum tree` and verify each recorded
303 * digest against the actual file content, local or remote.
304 * WHY: Periodic at-rest integrity checks without regenerating the full manifest;
305 * a single run covers the whole tree with one output line per file.
306 * HOW: Parse args (ckcheck_parse_args); open the manifest; for a remote root
307 * open one connection reused across all files (ckcheck_connect); verify
308 * every line with ckcheck_process_line (which owns algorithm inference —
309 * adler32=8 hex, crc64=16, md5=32 — and the OK/FAILED verdicts).
310 * Exit 0 all-OK, 1 any mismatch, 2 on any parse/I/O error.
311 */
312int
313brix_xrdckcheck_main(int argc, char **argv)
314{
315 const char *algo_str = NULL;
316 brix_cksum_algo algo = XRDC_CK_ADLER32; /* valid iff algo_str set */
317 FILE *mf;
318 char line[XRDC_PATH_MAX + TREE_HEX_MAX + 8];
319 int mismatches = 0;
320 int errors = 0;
321 int rc = 0;
322 brix_url u;
323 brix_conn c;
324 ckcheck_ctx cx;
325
326 if (ckcheck_parse_args(argc, argv, &algo_str, &algo, &rc) != 0) {
327 return rc;
328 }
329
330 mf = fopen(argv[1], "r");
331 if (mf == NULL) {
332 fprintf(stderr, "%s: cannot open manifest '%s': %s\n",
333 argv[0], argv[1], strerror(errno));
334 return 2;
335 }
336
337 cx.prog = argv[0];
338 cx.root = argv[2];
339 cx.root_path = NULL;
340 cx.c = NULL;
341 cx.is_remote = ckt_is_root_url(argv[2]);
342 cx.algo_str = algo_str;
343 cx.algo = algo;
344 cx.errors = &errors;
345 cx.mismatches = &mismatches;
346
347 if (cx.is_remote) {
348 if (ckcheck_connect(argv[0], argv[2], &u, &c) != 0) {
349 fclose(mf);
350 return XRDC_EXIT_USAGE;
351 }
352 cx.root_path = u.path;
353 cx.c = &c;
354 }
355
356 while (fgets(line, sizeof(line), mf) != NULL) {
357 ckcheck_process_line(&cx, line);
358 }
359
360 fclose(mf);
361 if (cx.c != NULL) {
362 brix_close(&c);
363 }
364
365 if (errors > 0) {
366 return 2;
367 }
368 return mismatches > 0 ? 1 : 0;
369}
