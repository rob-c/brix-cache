1/*
2 * http_peer_name.c — PREACCESS-phase wait for the peer's reverse-DNS name
3 * (phase-116).
4 *
5 * WHAT: For locations whose policy consults the peer's hostname (XrdAcc `h`
6 * rules via brix_acc_resolve_hosts, or a protbind host template), hold
7 * the request in NGX_HTTP_PREACCESS_PHASE until the PTR answer is in the
8 * reverse cache; every later brix_acc_resolve_peer() probe is then a
9 * hit. Locations that never consult the name pay two flag tests.
10 * WHY: The WebDAV/S3 handlers and the protbind access check run on the event
11 * loop, where the blocking getnameinfo() they used to call stalled the
12 * worker. limit_req shows the nginx-native way to pause a request in a
13 * phase: return NGX_AGAIN without advancing r->phase_handler and call
14 * ngx_http_core_run_phases() when the wait is over.
15 * HOW: brix_http_peer_name_postconfiguration() pushes the handler onto the
16 * PREACCESS phase (the common module owns the whole HTTP plane's hook).
17 * The handler probes brix_dns_reverse_cached(); on NGX_AGAIN it starts a
18 * brix_dns_rev_req_t on r->pool (a pool cleanup cancels an in-flight
19 * lookup if the client goes away first), parks the request behind
20 * ngx_http_test_reading so a client abort is noticed, and resumes the
21 * phase engine from the completion handler. A request waits at most
22 * once (a flag on the common module's request ctx): a second pass — a
23 * resolver failure the cache could not record — falls through and the
24 * consumers decide numerically.
25 */

42static ngx_flag_t
43http_peer_name_needed(const ngx_http_brix_common_conf_t *ccf)
44{
45 return ccf->common.acc.resolve_hosts > 0

65static void
66http_peer_name_done(brix_dns_rev_req_t *req)
67{
68 http_peer_name_wait_t *w = req->data;
69 ngx_http_request_t *r = w->r;
70
71 w->done = 1;
72 if (w->inline_call) {
73 return; /* http_peer_name_start() falls through */
74 }
75 r->read_event_handler = ngx_http_block_reading;
76 r->write_event_handler = ngx_http_core_run_phases;
77 ngx_http_core_run_phases(r);
78}

81/* NGX_AGAIN = parked until http_peer_name_done(); NGX_DECLINED = decide now */
82static ngx_int_t
83http_peer_name_start(ngx_http_request_t *r, const brix_dns_policy_t *policy)
84{
85 http_peer_name_wait_t *w;
86 ngx_pool_cleanup_t *cln;
87 ngx_connection_t *c = r->connection;
88
89 w = ngx_pcalloc(r->pool, sizeof(*w));
90 cln = (w != NULL) ? ngx_pool_cleanup_add(r->pool, 0) : NULL;
91 if (cln == NULL) {
92 return NGX_DECLINED;
93 }
94 cln->handler = http_peer_name_cleanup;
95 cln->data = w;
96
97 ngx_memcpy(&w->req.ss, c->sockaddr, c->socklen);
98 w->req.len = c->socklen;
99 w->req.policy = policy;
100 w->req.log = c->log;
101 w->req.handler = http_peer_name_done;
102 w->req.data = w;
103 w->r = r;
104
105 w->inline_call = 1;
106 if (brix_dns_reverse(&w->req) != NGX_OK) {
107 w->done = 1;
108 return NGX_DECLINED;
109 }
110 w->inline_call = 0;
111 if (w->done) {
112 return NGX_DECLINED;
113 }
114
115 r->read_event_handler = ngx_http_test_reading;
116 r->write_event_handler = ngx_http_request_empty_handler;
117 return NGX_AGAIN;
118}

121static ngx_int_t
122http_peer_name_handler(ngx_http_request_t *r)
123{
124 ngx_http_brix_common_conf_t *ccf;
125 ngx_connection_t *c = r->connection;
126 brix_io_monitor_t *m;
127 char name[BRIX_DNS_REVERSE_NAME_LEN];
128
129 ccf = ngx_http_get_module_loc_conf(r, ngx_http_brix_common_module);
130 if (ccf == NULL || !http_peer_name_needed(ccf) || c->sockaddr == NULL
131 || c->socklen > sizeof(struct sockaddr_storage)
132 || brix_dns_reverse_cached(c->sockaddr, c->socklen, name,
133 sizeof(name)) != NGX_AGAIN)
134 {
135 return NGX_DECLINED;
136 }
137
138 m = brix_http_monitor_get(r);
139 if (m == NULL || m->peer_name_waited) {
140 return NGX_DECLINED;
141 }
142 m->peer_name_waited = 1;
143
144 return http_peer_name_start(r, ccf->common.dns.policy);
145}
