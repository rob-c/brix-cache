1/*
2 * cms/recv_frame.c — opcode frame handlers + dispatch for the CMS client-side
3 * (manager-connection) receive path.
4 *
5 * WHAT: The per-opcode handlers a connected CMS node runs on frames from its
6 * manager — PING→PONG, SPACE→AVAIL, STATUS→suspend/resume, SELECT/TRY→redirect
7 * a waiting client, STATE→kYR_have existence probe, forwarded namespace ops,
8 * UPDATE, DISC — plus the pending-client redirect/wake plumbing
9 * (brix_cms_wake_pending_session / brix_cms_client_conn_by_fd), the frame dispatch
10 * table, and the router ngx_brix_cms_process_frame().
11 *
12 * WHY: Split (Phase-79 file-size split) from recv.c so the opcode-handler set
13 * and its dispatch table live in one focused, independently reviewable file
14 * under the size guideline, separate from the read/framing event loop (recv.c)
15 * and the Plane-B storage-mutation executor (recv_forward.c). The one entry
16 * point the read loop calls (ngx_brix_cms_process_frame) is declared in
17 * recv_internal.h; the forwarded-op executor it invokes (cms_node_exec_forward)
18 * is declared there too.
19 *
20 * HOW: cms_frame_* handlers share the cms_frame_handler_pt signature and derive
21 * their payload view from ctx; cms_frame_table maps opcode → handler;
22 * ngx_brix_cms_process_frame decodes the header's streamid/rrCode and dispatches
23 * through the table, silently ignoring unknown opcodes.
24 */

44/* Resolve a saved client fd to its live connection (recycle-guarded by the
45 * caller via c->number). Exported (recv_internal.h) since Phase-89 W8: the
46 * rm/rmdir fan-out finalizer (fanout.c) resolves its parked client the same
47 * way the locate wake below does. */
48ngx_connection_t *
49brix_cms_client_conn_by_fd(int fd)
50{
51 ngx_uint_t i;
52 ngx_connection_t *c;
53
54 if (fd < 0) {
55 return NULL;
56 }
57
58 if (ngx_cycle->files != NULL && (ngx_uint_t) fd < ngx_cycle->files_n) {
59 c = ngx_cycle->files[fd];
60 if (c != NULL && c->fd == fd) {
61 return c;
62 }
63 return NULL;
64 }
65
66 for (i = 0; i < ngx_cycle->connection_n; i++) {
67 c = &ngx_cycle->connections[i];
68 if (c->fd == fd) {
69 return c;
70 }
71 }
72
73 return NULL;
74}

76/* brix_cms_wake_pending_session — wake the suspended XRootD client waiting on
77 * a pending locate: look up the pending entry by streamid+pid, resolve its
78 * saved fd to the live connection (same worker, per-worker design), set
79 * XRD_ST_REQ_HEADER, brix_send_redirect to the resolved server, and resume
80 * reading. Exported (recv_internal.h) because two ingest paths converge on it:
81 * kYR_select/kYR_try from a parent manager (this file) and — Phase-89 W3 —
82 * kYR_have from a child node (server_recv_frame.c, state fan-out wake). */
83
84ngx_int_t
85brix_cms_wake_pending_session(ngx_log_t *log, uint32_t streamid,
86 const char *host, uint16_t port)
87{
88 brix_pending_locate_t *pending;
89 ngx_connection_t *client_conn;
90 ngx_stream_session_t *session;
91 brix_ctx_t *xrd_ctx;
92 int conn_fd;
93 ngx_atomic_uint_t conn_number;
94 u_char client_streamid[2];
95
96 pending = brix_pending_lookup(streamid, ngx_pid);
97 if (pending == NULL) {
98 ngx_log_debug1(NGX_LOG_DEBUG_EVENT, log, 0,
99 "brix: CMS wake: streamid=%uD not found in pending table",
100 streamid);
101 return NGX_OK; /* session timed out and was already removed */
102 }
103
104 conn_fd = pending->conn_fd;
105 conn_number = pending->conn_number;
106 client_streamid[0] = pending->client_streamid[0];
107 client_streamid[1] = pending->client_streamid[1];
108 brix_pending_unlock();
109
110 brix_pending_remove(streamid, ngx_pid);
111
112 client_conn = brix_cms_client_conn_by_fd(conn_fd);
113 if (client_conn == NULL || client_conn->number != conn_number) {
114 return NGX_OK; /* fd was recycled after the client disconnected */
115 }
116
117 session = client_conn->data;
118 if (session == NULL) {
119 return NGX_OK;
120 }
121
122 xrd_ctx = ngx_stream_get_module_ctx(session, ngx_stream_brix_module);
123 if (xrd_ctx == NULL || xrd_ctx->state != XRD_ST_WAITING_CMS) {
124 return NGX_OK;
125 }
126
127 /*
128 * WS6: the redirect host comes straight from the manager's kYR_select /
129 * kYR_try payload and is copied verbatim into the "Shost:port" redirect the
130 * client parses. A compromised/hostile manager could inject control bytes or
131 * an alternate scheme here, so validate it with the same character allowlist
132 * the registry uses as its store choke point (brix_net_host_chars_valid).
133 * On reject, drop the redirect and leave the client in XRD_ST_WAITING_CMS to
134 * hit its own cms_locate_timeout — we never emit a poisoned host.
135 */
136 if (host == NULL
137 || !brix_net_host_chars_valid(host, ngx_strlen(host)))
138 {
139 char safe[256];
140 brix_sanitize_log_string(host, safe, sizeof(safe));
141 ngx_log_error(NGX_LOG_WARN, log, 0,
142 "brix: CMS select: rejected redirect to invalid host "
143 "\"%s\" for fd=%d", safe, conn_fd);
144 return NGX_OK;
145 }
146
147 ngx_log_error(NGX_LOG_INFO, log, 0,
148 "brix: CMS select: redirecting client fd=%d to %s:%ud",
149 conn_fd, host, (unsigned) port);
150
151 ngx_del_timer(client_conn->read);
152 xrd_ctx->state = XRD_ST_REQ_HEADER;
153 xrd_ctx->recv.cur_streamid[0] = client_streamid[0];
154 xrd_ctx->recv.cur_streamid[1] = client_streamid[1];
155 if (brix_send_redirect(xrd_ctx, client_conn, host, port) == NGX_ERROR) {
156 ngx_log_error(NGX_LOG_ERR, log, 0,
157 "brix: CMS select: failed to queue redirect for fd=%d",
158 conn_fd);
159 return NGX_ERROR;
160 }
161 brix_schedule_read_resume(client_conn);
162 return NGX_OK;
163}

190/* cms_frame_status — kYR_status: the manager flips our login gate. The modifier
191 * byte (offset 5) selects suspend (pause new logins) or resume; any other
192 * modifier is a no-op, matching stock cmsd. */
193static ngx_int_t
194cms_frame_status(ngx_brix_cms_ctx_t *ctx, uint32_t streamid, u_char code)
195{
196 u_char mod = ctx->inbuf[5];
197 if (mod & CMS_ST_SUSPEND) {
198 ctx->conf->cms.suspended = 1;
199 ngx_log_error(NGX_LOG_NOTICE, ctx->cycle->log, 0,
200 "brix: CMS suspend received — new logins paused");
201 brix_cms_log_action(ctx->cycle->log, "status",
202 (const char *) ctx->mgr_name.data,
203 "in", NULL, 1, "manager suspended this node");
204 } else if (mod & CMS_ST_RESUME) {
205 ctx->conf->cms.suspended = 0;
206 ngx_log_error(NGX_LOG_NOTICE, ctx->cycle->log, 0,
207 "brix: CMS resume received — accepting logins");
208 brix_cms_log_action(ctx->cycle->log, "status",
209 (const char *) ctx->mgr_name.data,
210 "in", NULL, 1, "manager resumed this node");
211 } else {
212 ngx_log_debug1(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
213 "brix: CMS status modifier=0x%02xi (no action)",
214 (ngx_uint_t) mod);
215 }
216 return NGX_OK;
217}

219/*
220 * cms_frame_login_retarget — §2.9: an UNSOLICITED kYR_try (streamid 0) is
221 * the manager redirecting this node's LOGIN elsewhere (ManTree supervisor
222 * offload, or a blacklist `redirect` entry).
223 *
224 * WHAT: Re-points this heartbeat link at the named manager and reconnects.
225 * Returns NGX_OK always (a refused/malformed retarget just keeps the
226 * current link).
227 * WHY: Without honoring the login redirect, a manager at its direct-server
228 * cap can only drop us — the tree never forms. The original address
229 * is preserved so repeated failures (or a redirect chain > 4 deep)
230 * fall back to configuration.
231 * HOW: Validate host chars (same allowlist as the locate redirect), accept
232 * IP literals only (registry hosts are the peer's numeric address —
233 * a hostname here would mean blocking DNS in the event loop), build
234 * the sockaddr on the cycle pool, swap ctx->mgr_addr/mgr_name, then
235 * disconnect + immediate reconnect.
236 */
237static ngx_int_t
238cms_frame_login_retarget(ngx_brix_cms_ctx_t *ctx, const char *host,
239 uint16_t port)
240{
241 ngx_addr_t *addr;
242 u_char *name;
243 size_t name_len;
244
245 if (ctx->retarget_depth >= 4) {
246 ngx_log_error(NGX_LOG_WARN, ctx->cycle->log, 0,
247 "brix: CMS login retarget chain too deep — staying on %V",
248 &ctx->mgr_name);
249 return NGX_OK;
250 }
251
252 addr = ngx_pcalloc(ctx->cycle->pool, sizeof(ngx_addr_t));
253 name = ngx_pnalloc(ctx->cycle->pool,
254 ngx_strlen(host) + sizeof(":65535"));
255 if (addr == NULL || name == NULL) {
256 return NGX_OK;
257 }
258
259 if (ngx_parse_addr_port(ctx->cycle->pool, addr, (u_char *) host,
260 ngx_strlen(host)) != NGX_OK)
261 {
262 ngx_log_error(NGX_LOG_WARN, ctx->cycle->log, 0,
263 "brix: CMS login retarget to non-literal host \"%s\" refused "
264 "(IP literals only)", host);
265 return NGX_OK;
266 }
267 ngx_inet_set_port(addr->sockaddr, port);
268
269 name_len = (size_t)
270 (ngx_snprintf(name, ngx_strlen(host) + sizeof(":65535"),
271 "%s:%d%Z", host, (int) port) - name) - 1;
272 addr->name.data = name;
273 addr->name.len = name_len;
274
275 if (!ctx->retargeted) {
276 ctx->orig_addr = ctx->mgr_addr;
277 ctx->orig_name = ctx->mgr_name;
278 ctx->retargeted = 1;
279 }
280 ctx->retarget_depth++;
281 ctx->retarget_fails = 0;
282 ctx->mgr_addr = addr;
283 ctx->mgr_name.data = name;
284 ctx->mgr_name.len = name_len;
285
286 ngx_log_error(NGX_LOG_NOTICE, ctx->cycle->log, 0,
287 "brix: CMS login redirected — re-dialing %V (was %V)",
288 &ctx->mgr_name, &ctx->orig_name);
289 brix_cms_log_action(ctx->cycle->log, "login-retarget",
290 (const char *) ctx->mgr_name.data, "out", NULL, 1,
291 "manager answered login with kYR_try");
292
293 ngx_brix_cms_set_end_hint(ctx, BRIX_SESS_END_SERVER);
294 ngx_brix_cms_disconnect(ctx);
295 ngx_brix_cms_schedule(ctx, 10);
296 return NGX_OK;
297}

299/* cms_frame_redirect — kYR_select / kYR_try: the manager resolved a pending
300 * kYR_locate and names a server. Both payloads carry a NUL-terminated hostname
301 * + 2-byte big-endian port (kYR_try is an ordered list of such entries — use
302 * only the first; the client retries remaining entries itself), so one handler
303 * serves both opcodes. Truncated payloads are silently ignored. §2.9: an
304 * unsolicited kYR_try on streamid 0 is a LOGIN retarget, not a locate answer. */
305static ngx_int_t
306cms_frame_redirect(ngx_brix_cms_ctx_t *ctx, uint32_t streamid, u_char code)
307{
308 const u_char *payload = ctx->inbuf + NGX_BRIX_CMS_HDR_LEN;
309 size_t payload_len = ctx->in_need - NGX_BRIX_CMS_HDR_LEN;
310 char host[256];
311 size_t host_len;
312 uint16_t port;
313
314 if (payload_len < 3) {
315 /* need at least one host byte, a NUL, and two port bytes */
316 return NGX_OK;
317 }
318
319 ngx_cpystrn((u_char *) host, (u_char *) payload, sizeof(host));
320 host_len = ngx_strlen(host);
321
322 if (host_len + 3 > payload_len) {
323 /* port bytes would fall outside the received payload */
324 return NGX_OK;
325 }
326
327 port = ngx_brix_cms_get16(payload + host_len + 1);
328
329 /* §2.9: an unsolicited kYR_try (streamid 0 — never a locate correlation
330 * id, see ngx_brix_cms_next_streamid) is a LOGIN retarget. Host chars
331 * are validated with the same allowlist the locate wake uses, so a
332 * hostile manager cannot inject a poisoned dial target. */
333 if (streamid == 0 && code == CMS_RR_TRY) {
334 if (!brix_net_host_chars_valid(host, host_len)) {
335 char safe[256];
336 brix_sanitize_log_string(host, safe, sizeof(safe));
337 ngx_log_error(NGX_LOG_WARN, ctx->cycle->log, 0,
338 "brix: CMS login retarget to invalid host \"%s\" refused",
339 safe);
340 return NGX_OK;
341 }
342 return cms_frame_login_retarget(ctx, host, port);
343 }
344
345 {
346 u_char detail[320];
347 u_char *dp = ngx_snprintf(detail, sizeof(detail) - 1,
348 "%s selected -> redirect waiting client to "
349 "%s:%d",
350 code == CMS_RR_TRY ? "kYR_try" : "kYR_select",
351 host, (int) port);
352 *dp = '\0';
353 brix_cms_log_action(ctx->cycle->log, "redirect",
354 (const char *) ctx->mgr_name.data,
355 "in", NULL, 1, (const char *) detail);
356 }
357
358 return brix_cms_wake_pending_session(ctx->cycle->log, streamid, host, port);
359}

361/* cms_frame_forward — kYR_chmod/mkdir/mkpath/mv/rm/rmdir/trunc (Plane B): a
362 * manager-forwarded namespace mutation. An explicit supervisor relays it down
363 * to its own data nodes (cms_super_fan_down); every other role executes it
364 * locally under kernel confinement and replies silent-on-success /
365 * kYR_error-on-failure. */
366static ngx_int_t
367cms_frame_forward(ngx_brix_cms_ctx_t *ctx, uint32_t streamid, u_char code)
368{
369 const u_char *payload = ctx->inbuf + NGX_BRIX_CMS_HDR_LEN;
370 size_t plen = ctx->in_need - NGX_BRIX_CMS_HDR_LEN;
371 const brix_cms_route_t *r;
372 ngx_int_t rc;
373
374 /* A supervisor relays the op down to its own data nodes — it neither
375 * executes locally nor logs a node action (the receiving nodes do). */
376 if (ctx->conf->cms.role == BRIX_CMS_ROLE_SUPERVISOR) {
377 return cms_super_fan_down(ctx, code, payload, plen);
378 }
379
380 rc = cms_node_exec_forward(ctx, code, streamid, payload, plen);
381
382 r = brix_cms_route_lookup(XRDCMS_ROLE_NODE, code);
383 brix_cms_log_action(ctx->cycle->log, r != NULL ? r->name : "forward-op",
384 (const char *) ctx->mgr_name.data,
385 "in", NULL, rc == NGX_OK,
386 "manager-forwarded namespace op executed on this node");
387 return rc;
388}

461/* cms_frame_role_ok — Phase-61 W7 valid-ops gate: under an explicit
462 * brix_cms_role, only the ops in that role's upward-leg table (manVOps for a
463 * sub-manager, supVOps for a supervisor) may arrive from the parent manager —
464 * stock cmsd "prohibit[s] a meta-manager from requesting potentially
465 * destructive actions". kYR_select / kYR_try are exempt: they are the
466 * streamid-correlated replies to our own kYR_locate, not parent requests.
467 * Role auto keeps the legacy accept-everything behaviour. */
468static ngx_uint_t
469cms_frame_role_ok(ngx_brix_cms_ctx_t *ctx, u_char code)
470{
471 brix_cms_role_t role;
472
473 switch (ctx->conf->cms.role) {
474 case BRIX_CMS_ROLE_MANAGER:
475 role = XRDCMS_ROLE_SUBMAN;
476 break;
477 case BRIX_CMS_ROLE_SUPERVISOR:
478 role = XRDCMS_ROLE_SUPER;
479 break;
480 case BRIX_CMS_ROLE_PEER:
481 case BRIX_CMS_ROLE_PROXY:
482 /* §2.17: a peer/proxy leg accepts the same restricted (non-
483 * destructive) upward set as a sub-manager — a remote cluster's
484 * manager must not drive namespace mutations into this one. */
485 role = XRDCMS_ROLE_SUBMAN;
486 break;
487 default:
488 return 1;
489 }
490
491 if (code == CMS_RR_SELECT || code == CMS_RR_TRY) {
492 return 1;
493 }
494
495 if (brix_cms_route_lookup(role, code) != NULL) {
496 return 1;
497 }
498
499 ngx_log_error(NGX_LOG_NOTICE, ctx->cycle->log, 0,
500 "brix: CMS %s: manager sent an invalid request "
501 "(op=%ui) — dropped",
502 role == XRDCMS_ROLE_SUBMAN ? "subman" : "super",
503 (ngx_uint_t) code);
504 return 0;
505}

507/* ngx_brix_cms_process_frame — decode a complete CMS frame's streamid + rrCode
508 * (first 4 bytes + offset 4), apply the per-role valid-ops gate
509 * (cms_frame_role_ok), and dispatch by opcode through cms_frame_table:
510 * PING→PONG, SPACE→AVAIL, STATS→Cluster.Stats, STATUS→suspend/resume conf
511 * flags, SELECT/TRY→client redirect, STATE→kYR_have probe, forwarded
512 * namespace ops, UPDATE, DISC. Unknown opcodes are silently ignored (debug
513 * log). */
514
515ngx_int_t
516ngx_brix_cms_process_frame(ngx_brix_cms_ctx_t *ctx)
517{
518 uint32_t streamid;
519 u_char code;
520 ngx_uint_t i;
521 const brix_cms_route_t *r;
522
523 streamid = ngx_brix_cms_get32(ctx->inbuf);
524 code = ctx->inbuf[4];
525
526 ngx_log_debug2(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
527 "brix: CMS process frame code=%ui streamid=%uD",
528 (ngx_uint_t) code, streamid);
529
530 if (!cms_frame_role_ok(ctx, code)) {
531 return NGX_OK; /* logged + dropped, conn kept */
532 }
533
534 for (i = 0; i < sizeof(cms_frame_table) / sizeof(cms_frame_table[0]); i++) {
535 if (cms_frame_table[i].code == code) {
536 return cms_frame_table[i].handler(ctx, streamid, code);
537 }
538 }
539
540 r = brix_cms_route_lookup(XRDCMS_ROLE_NODE, code);
