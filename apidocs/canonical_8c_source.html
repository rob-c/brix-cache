37/*
38 * brix_realpath_existing — realpath(3) semantics for an existing path,
39 * priced for the miss.
40 *
41 * glibc realpath() readlink-walks every component, so a caller probing a
42 * path that usually does NOT exist (the stat handler's symlink-follow
43 * fallback) pays ~one readlink per directory level just to learn ENOENT.
44 * open(O_PATH) makes the kernel do the whole resolution in one syscall —
45 * a miss costs exactly that failing open — and the canonical result is
46 * read back through /proc/self/fd. Falls back to realpath(3) when /proc
47 * is not mounted. Returns `resolved` (a PATH_MAX buffer) or NULL with
48 * errno describing the failing resolution.
49 */
50char *
51brix_realpath_existing(const char *path, char *resolved)
52{

54 ssize_t n;
55 int fd, e;
56
57 fd = open(path, O_PATH | O_CLOEXEC); /* vfs-seam-allow: SEAM_CORRECT — canonicaliser core: kernel-side path resolution, no data plane */
58 if (fd < 0) {
59 return NULL;
60 }
61
62 (void) snprintf(proc, sizeof(proc), "/proc/self/fd/%d", fd);
63 n = readlink(proc, resolved, PATH_MAX - 1);
64 e = errno;
65 close(fd);

68 if (n >= 0 || e == ENOENT) {

70 return realpath(path, resolved);
71 }
72 errno = e;
73 return NULL;
74 }
