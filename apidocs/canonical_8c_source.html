44/*
45 * brix_realpath_existing — realpath(3) semantics for an existing path,
46 * priced for the miss.
47 *
48 * glibc realpath() readlink-walks every component, so a caller probing a
49 * path that usually does NOT exist (the stat handler's symlink-follow
50 * fallback) pays ~one readlink per directory level just to learn ENOENT.
51 * open(O_PATH) makes the kernel do the whole resolution in one syscall —
52 * a miss costs exactly that failing open — and the canonical result is
53 * read back through /proc/self/fd. Falls back to realpath(3) when /proc
54 * is not mounted. Returns `resolved` (a PATH_MAX buffer) or NULL with
55 * errno describing the failing resolution.
56 */
57char *
58brix_realpath_existing(const char *path, char *resolved)
59{
60 char proc[48];
61 ssize_t n;
62 int fd, e;
63
64 fd = open(path, O_PATH | O_CLOEXEC); /* vfs-seam-allow: SEAM_CORRECT — canonicaliser core: kernel-side path resolution, no data plane */
65 if (fd < 0) {
66 return NULL;
67 }
68
69 (void) snprintf(proc, sizeof(proc), "/proc/self/fd/%d", fd);
70 n = readlink(proc, resolved, PATH_MAX - 1);
71 e = errno;
72 close(fd);
73
74 if (n <= 0 || resolved[0] != '/') {
75 if (n >= 0 || e == ENOENT) {
76 /* /proc absent (or gave a non-path) — resolve the slow way. */
77 return realpath(path, resolved);
78 }
79 errno = e;
80 return NULL;
81 }
82
83 resolved[n] = '\0';
84 return resolved;
85}
