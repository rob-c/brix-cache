1/*
2 * sd_posix_dedup.c — the POSIX realisation of the commit-time dedup slots
3 * (phase-88 W1: the G13 hardlink farm moved below the SD seam).
4 *
5 * WHAT: Implements sd_posix_dedup_publish / sd_posix_dedup_gc: bind a
6 * content-verified store object and its canonical content-derived alias
7 * (`canon`, e.g. "/.gcas/<2hex>/<hex><sfx>") onto ONE inode via link(2),
8 * and reap the canonical name once it is the last remaining link.
9 *
10 * WHY: These mechanics lived above the seam in src/fs/cache/gcas.c as raw
11 * syscalls on the store's local root. Expressing them as driver slots
12 * (a) moves the raw filesystem work where invariant 12 wants it and
13 * (b) makes the caller (gcas.c) storage-neutral, so a refcounting
14 * backend (pblock) can serve brix_cache_global_cas with its own slot.
15 *
16 * HOW: publish: first appearance registers the canonical via link(2); later
17 * byte-identical publishes ADOPT the canonical via link-to-temp +
18 * rename(2) (atomic — open readers keep their inode). The filesystem's
19 * st_nlink IS the combined refcount, so there is no bookkeeping to
20 * corrupt; every path is best-effort and any failure leaves plain,
21 * correct per-name copies. gc: unlink the canonical at st_nlink <= 1.
22 * Both run on cache-fill worker threads: no pool access, inst->log only.
23 */

39/* ---- posix_dedup_abs — absolute store path for a store-relative name ----
40 *
41 * WHAT: Joins the instance's root_canon with `rel` (always leading '/') into
42 * out[cap]. Returns 0, or -1 on overflow / missing root.
43 *
44 * WHY: The dedup names travel store-relative (the caller is storage-neutral);
45 * only the POSIX driver knows the physical directory they land in.
46 *
47 * HOW: 1. Trim trailing '/' runs off root_canon. 2. snprintf-join and
48 * bounds-check.
49 */
50static int
51posix_dedup_abs(const sd_posix_state_t *st, const char *rel, char *out,
52 size_t cap)
53{
54 size_t rlen;
55 int n;
56
57 if (st == NULL || st->root_canon == NULL) {
58 return -1;
59 }
60 rlen = strlen(st->root_canon);
61 while (rlen > 0 && st->root_canon[rlen - 1] == '/') {
62 rlen--;
63 }
64 n = snprintf(out, cap, "%.*s%s", (int) rlen, st->root_canon, rel);
65 return (n > 0 && (size_t) n < cap) ? 0 : -1;
66}

68/* ---- posix_dedup_ensure_parent — mkdir -p the parent of an absolute path ----
69 *
70 * WHAT: Creates the directory chain above `path` (0700). Returns 0 when the
71 * parent exists or was created, -1 on error (errno set).
72 *
73 * WHY: The first canonical under a fresh "/.gcas/<2hex>/" prefix needs its
74 * directory; 0700 matches the svc-owned, never-client-listable cache
75 * tree convention.
76 *
77 * HOW: 1. Copy + bounds-check. 2. Truncate at the last '/'. 3. Delegate to
78 * brix_mkdir_recursive.
79 */
80static int
81posix_dedup_ensure_parent(const char *path)
82{
83 char parent[PATH_MAX];
84 char *slash;
85 int n;
86
87 n = snprintf(parent, sizeof(parent), "%s", path);
88 if (n < 0 || (size_t) n >= sizeof(parent)) {
89 errno = ENAMETOOLONG;
90 return -1;
91 }
92 slash = strrchr(parent, '/');
93 if (slash == NULL || slash == parent) {
94 return 0;
95 }
96 *slash = '\0';
97 return brix_mkdir_recursive(parent, 0700);
98}

100/* ---- posix_dedup_adopt — swap a published name onto an EXISTING canonical ----
101 *
102 * WHAT: Links the canonical's inode to a ".gclnk" temp next to `obj` and
103 * renames it over `obj` (atomic; open readers keep their old inode).
104 * Returns 0 when publish is finished (bound, skipped, or terminally
105 * failed), -1 when the canonical vanished underneath us (caller retries).
106 *
107 * WHY: Adopting — rather than re-linking the fresh copy — is what collapses a
108 * later byte-identical fill onto the shared inode without ever exposing
109 * a missing name.
110 *
111 * HOW: 1. Same-inode short-circuit. 2. Size-mismatch guard (hash collision /
112 * damaged canonical: never collapse mismatched bytes). 3. link canonical
113 * -> temp (failure = canonical evicted; retry). 4. rename temp -> obj.
114 */
115static int
116posix_dedup_adopt(brix_sd_instance_t *inst, const char *path,
117 const char *canon, const char *obj, const struct stat *cst,
118 const struct stat *ost)
119{
120 char tmp[PATH_MAX];
121 int n;
122
123 if (cst->st_ino == ost->st_ino) {
124 return 0; /* already bound */
125 }
126
127 if (cst->st_size != ost->st_size) {
128 ngx_log_error(NGX_LOG_WARN, inst->log, 0,
129 "gcas: canonical size mismatch for \"%s\" (%O vs %O) — "
130 "dedup skipped", path,
131 (off_t) cst->st_size, (off_t) ost->st_size);
132 return 0;
133 }
134
135 n = snprintf(tmp, sizeof(tmp), "%s.gclnk", obj);
136 if (n < 0 || (size_t) n >= sizeof(tmp)) {
137 return 0;
138 }
139 unlink(tmp);
140 if (link(canon, tmp) != 0) {
141 return -1; /* canonical evicted underneath us */
142 }
143 if (rename(tmp, obj) != 0) {
144 unlink(tmp);
145 return 0;
146 }
147
148 ngx_log_error(NGX_LOG_INFO, inst->log, 0,
149 "gcas: dedup \"%s\" onto canonical inode", path);
150 return 0;
151}

153/* ---- sd_posix_dedup_publish — driver->dedup_publish for posix stores ----
154 *
155 * WHAT: Binds the (content-verified) object at `path` and the canonical alias
156 * `canon` to one inode. NGX_OK on every completed/benign path (the
157 * per-name copy always stays correct); NGX_ERROR/errno only on invalid
158 * arguments.
159 *
160 * WHY: First appearance of a content hash registers the canonical; later
161 * byte-identical publishes adopt it, so N names for one content cost one
162 * inode. st_nlink carries the refcount — nothing to corrupt.
163 *
164 * HOW: 1. Resolve both absolute paths. 2. stat the object (gone = nothing to
165 * bind). 3. Two attempts: canonical present -> adopt (retry once when it
166 * is evicted mid-flight); absent -> mkdir -p + link it as the canonical
167 * (EEXIST = lost the register race — loop and adopt the winner).
168 */
169ngx_int_t
170sd_posix_dedup_publish(brix_sd_instance_t *inst, const char *path,
171 const char *canon)
172{
173 sd_posix_state_t *st = inst->state;
174 int attempt;
175 char cabs[PATH_MAX], oabs[PATH_MAX];
176 struct stat ost, cst;
177
178 if (path == NULL || canon == NULL
179 || posix_dedup_abs(st, canon, cabs, sizeof(cabs)) != 0
180 || posix_dedup_abs(st, path, oabs, sizeof(oabs)) != 0)
181 {
182 errno = EINVAL;
183 return NGX_ERROR;
184 }
185
186 if (stat(oabs, &ost) != 0) {
187 return NGX_OK; /* committed object already gone — nothing to bind */
188 }
189
190 for (attempt = 0; attempt < 2; attempt++) {
191
192 if (stat(cabs, &cst) == 0) {
193 if (posix_dedup_adopt(inst, path, cabs, oabs, &cst, &ost) == 0) {
194 return NGX_OK;
195 }
196 continue; /* canonical evicted underneath us — retry */
197 }
198
199 if (posix_dedup_ensure_parent(cabs) != 0) {
200 return NGX_OK; /* best-effort: keep the plain per-name copy */
201 }
202 if (link(oabs, cabs) == 0) {
203 ngx_log_error(NGX_LOG_INFO, inst->log, 0,
204 "gcas: registered canonical for \"%s\"", path);
205 return NGX_OK;
206 }
207 /* EEXIST: lost the race — loop once and adopt the winner. */
208 }
209 return NGX_OK;
210}

212/* ---- sd_posix_dedup_gc — driver->dedup_gc for posix stores ----
213 *
214 * WHAT: Unlinks the canonical alias `canon` once it is the last remaining name
215 * of its inode (st_nlink <= 1). NGX_OK always (best-effort; a canonical
216 * still shared, already gone, or unresolvable is a no-op).
217 *
218 * WHY: After the last data name referencing a content was evicted, the
219 * canonical would otherwise pin the inode forever.
220 *
221 * HOW: 1. Resolve the absolute canonical path. 2. stat; more links = still
222 * referenced, keep. 3. unlink + INFO-log the reap.
223 */
224ngx_int_t
225sd_posix_dedup_gc(brix_sd_instance_t *inst, const char *canon)
226{
227 sd_posix_state_t *st = inst->state;
228 char cabs[PATH_MAX];
229 struct stat cst;
230
231 if (canon == NULL
232 || posix_dedup_abs(st, canon, cabs, sizeof(cabs)) != 0)
233 {
234 return NGX_OK;
235 }
236
237 if (stat(cabs, &cst) != 0 || cst.st_nlink > 1) {
238 return NGX_OK; /* other names still hold the inode */
239 }
240
241 if (unlink(cabs) == 0) {
242 ngx_log_error(NGX_LOG_INFO, inst->log, 0,
243 "gcas: reaped canonical \"%s\" (last link)", canon);
244 }
245 return NGX_OK;
246}
