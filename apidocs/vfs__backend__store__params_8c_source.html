1/*
2 * vfs_backend_store_params.c - stamp the trailing params of a
3 * `brix_storage_backend` line onto the export's registry entry (phase-115).
4 *
5 * WHAT: one config-time call, made straight after brix_vfs_backend_config_str
6 * has registered the URL, that parses the line's trailing params through the
7 * tier vocabulary (verify_pages, mode=, prot=, credential=, block_size=) and
8 * copies the ones the live backend path honours onto brix_vfs_backend_entry_t.
9 *
10 * WHY it exists at all: BriX has two store-config paths. The tier composer
11 * (brix_cache_store / brix_cache_cold_store / brix_stage_store) parses its
12 * params and builds through brix_tier_build; the LIVE backend path parses only
13 * the URL, into the VFS backend registry, and builds through brix_vbr_build_*.
14 * No backend is ever built through the tier composer, so a param parsed only
15 * into a brix_tier_cfg_t is a promise nothing keeps — which is exactly how
16 * W4.3's `verify_pages` shipped unreachable (the directive was TAKE1, so
17 * nginx -t refused the token for arity before any parser saw it). This file
18 * is the one seam that carries a store-line param across to the path that
19 * actually builds the driver.
20 *
21 * HOW: parse into a scratch brix_tier_cfg_t (never retained — the tier cfg is
22 * the parser's vocabulary, not this path's storage), then copy the three fields
23 * the registry declares onto the entry. A line with no params returns before
24 * the URL is even looked at, so every configuration that works today is
25 * bit-identical through here.
26 */

34/* 2.0 F5: a forward:// export is configured by its params alone — the URL
35 * names no origin, so the permit list is the one thing deciding which origins
36 * the export may reach. An empty list is refused rather than defaulted: a
37 * forwarding proxy with no permit is an open relay to any host a client
38 * names. */
39static ngx_int_t
40vfs_store_params_check_forward(ngx_conf_t *cf,
41 const brix_vfs_backend_entry_t *e)
42{
43 /* A forward:// entry always carries at least one allowed protocol
44 * (vfs_parse_forward_origin refuses an empty list), so the flags are
45 * the entry's own identity — no driver-name branch (invariant 12). */
46 if (e == NULL
47 || !(e->origin_fwd_allow_root || e->origin_fwd_allow_roots))
48 {
49 return NGX_OK;
50 }
51 if (e->origin_fwd_permit[0] == '\0') {
52 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
53 "brix_storage_backend forward:// needs at least one "
54 "permit=<host|.suffix> (an empty permit list would relay to any "
55 "origin a client names)");
56 return NGX_ERROR;
57 }
58 return NGX_OK;
59}

70 /* A local export (no backend URL) has no registry entry: the params are
71 * still validated below, so a typo is an operator error either way, but
72 * there is no remote link for them to describe. */
73 e = (root_canon != NULL && root_canon[0] != '\0')
74 ? brix_vfs_backend_entry_find(root_canon) : NULL;
75
76 if (args == NULL || args->nelts == 0) {
77 /* no params: nothing to parse or stamp — unless the backend is one
78 * that only its params can configure */
79 return vfs_store_params_check_forward(cf, e);
80 }
81
82 err[0] = '\0';
83 if (brix_tier_parse_backend_params(&p, url, args) != NGX_OK) {
84 return NGX_ERROR; /* the parser already emitted the [emerg] */
85 }
86 if (e == NULL) {
87 return NGX_OK;
88 }
89 e->origin_verify_pages = (int) tier.verify_pages;
90 e->origin_ftp_mode_e = tier.ftp_mode_e ? 1 : 0;
91 e->origin_ftp_prot_p = tier.ftp_prot_p ? 1 : 0;
92 e->origin_ftp_streams = (int) tier.ftp_streams;
93 VFS_BE_STR(e, origin_fwd_permit, tier.fwd_permit);
