1/*
2 * directive.c — `brix_resolver` and `brix_dns_retry` setters (phase-116 W1).
3 *
4 * WHAT: Parses `brix_resolver auto|off [path=] [valid=] [min_ttl=] [max_ttl=]
5 * [negative_ttl=] [ipv4=] [ipv6=] [search=]` into a brix_dns_policy_t,
6 * builds the nginx resolver from the resolv.conf it names, and — the
7 * whole point — fills the enclosing scope's nginx resolver slot when
8 * the operator has not written `resolver` there. `brix_dns_retry`
9 * parses the target-registry backoff pair.
10 * WHY: Stock nginx creates an inert "dummy" resolver for every scope without
11 * an explicit `resolver`, so `proxy_pass $var`, `server … resolve` and
12 * OCSP all fail at runtime. Seeding the slot at parse time (not merge
13 * time) matters: the upstream module reads the http{}-level resolver in
14 * its init_main_conf, which runs BEFORE any merge hook.
15 * HOW: One setter serves both planes — `common` is member 0 of every brix
16 * module conf, and cf->module_type says which core module owns the
17 * slot. "Fills, never overrides": a slot that already holds a
18 * resolver is left alone with a notice. (The core-conf pokes are the
19 * R4-allowlisted seam of this file.)
20 */

29static char *
30dns_parse_time_arg(ngx_conf_t *cf, ngx_str_t *arg, size_t plen,
31 time_t *out)
32{
33 ngx_str_t v;
34 time_t t;
35
36 v.data = arg->data + plen;
37 v.len = arg->len - plen;
38 t = ngx_parse_time(&v, 1);
39 if (t == (time_t) NGX_ERROR) {
40 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
41 "brix_resolver: invalid time in \"%V\"", arg);
42 return NGX_CONF_ERROR;
43 }
44 *out = t;
45 return NGX_CONF_OK;
46}

49static char *
50dns_parse_flag_arg(ngx_conf_t *cf, ngx_str_t *arg, size_t plen,
51 ngx_flag_t *out)
52{
53 ngx_str_t v;
54
55 v.data = arg->data + plen;
56 v.len = arg->len - plen;
57 if (v.len == 2 && ngx_strncmp(v.data, "on", 2) == 0) {
58 *out = 1;
59 return NGX_CONF_OK;
60 }
61 if (v.len == 3 && ngx_strncmp(v.data, "off", 3) == 0) {
62 *out = 0;
63 return NGX_CONF_OK;
64 }
65 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
66 "brix_resolver: expected on|off in \"%V\"", arg);
67 return NGX_CONF_ERROR;
68}

72static char *
73dns_parse_option(ngx_conf_t *cf, brix_dns_policy_t *pol, ngx_str_t *arg)
74{
75 time_t t = 0;
76 char *rc;
77
78 if (ngx_strncmp(arg->data, "path=", 5) == 0) {
79 pol->path.data = arg->data + 5;
80 pol->path.len = arg->len - 5;
81 /* absolute only: a relative name would be read against the
82 * process cwd, which differs between `nginx -t` and the master */
83 if (pol->path.len == 0 || pol->path.data[0] != '/') {
84 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
85 "brix_resolver: path must be absolute in \"%V\"",
86 arg);
87 return NGX_CONF_ERROR;
88 }
89 return NGX_CONF_OK;
90 }
91 if (ngx_strncmp(arg->data, "valid=", 6) == 0) {
92 rc = dns_parse_time_arg(cf, arg, 6, &t);
93 pol->valid = (ngx_msec_t) t * BRIX_CMS_SEC_TO_MS_MULTIPLIER;
94 return rc;
95 }
96 if (ngx_strncmp(arg->data, "min_ttl=", 8) == 0) {
97 return dns_parse_time_arg(cf, arg, 8, &pol->min_ttl);
98 }
99 if (ngx_strncmp(arg->data, "max_ttl=", 8) == 0) {
100 return dns_parse_time_arg(cf, arg, 8, &pol->max_ttl);
101 }
102 if (ngx_strncmp(arg->data, "negative_ttl=", 13) == 0) {
103 return dns_parse_time_arg(cf, arg, 13, &pol->negative_ttl);
104 }
105 if (ngx_strncmp(arg->data, "ipv4=", 5) == 0) {
106 return dns_parse_flag_arg(cf, arg, 5, &pol->ipv4);
107 }
108 if (ngx_strncmp(arg->data, "ipv6=", 5) == 0) {
109 return dns_parse_flag_arg(cf, arg, 5, &pol->ipv6);
110 }
111 if (ngx_strncmp(arg->data, "search=", 7) == 0) {
112 return dns_parse_flag_arg(cf, arg, 7, &pol->search);
113 }
114 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
115 "brix_resolver: unknown parameter \"%V\"", arg);
116 return NGX_CONF_ERROR;
117}

120static brix_dns_policy_t *
121dns_policy_create(ngx_conf_t *cf)
122{
123 brix_dns_policy_t *pol;
124
125 pol = ngx_pcalloc(cf->pool, sizeof(brix_dns_policy_t));
126 if (pol == NULL) {
127 return NULL;
128 }
129 ngx_str_set(&pol->path, BRIX_RESOLV_DEFAULT_PATH);
130 pol->min_ttl = BRIX_DNS_MIN_TTL_DEFAULT;
131 pol->max_ttl = BRIX_DNS_MAX_TTL_DEFAULT;
132 pol->negative_ttl = BRIX_DNS_NEG_TTL_DEFAULT;
133 pol->ipv4 = 1;
134 pol->ipv6 = 1;
135 pol->search = 1;
136 return pol;
137}

140/* Fill the enclosing core scope's resolver slot when it is still empty. */
141static void
142dns_seed_core_slot(ngx_conf_t *cf, brix_dns_policy_t *pol)
143{
144 ngx_resolver_t **slot = NULL;
145 const char *plane = "?";
146
147 if (cf->module_type == NGX_HTTP_MODULE) {
148 ngx_http_core_loc_conf_t *clcf;
149 clcf = ngx_http_conf_get_module_loc_conf(cf, ngx_http_core_module);
150 slot = &clcf->resolver;
151 plane = "http";
152 } else if (cf->module_type == NGX_STREAM_MODULE) {
153 ngx_stream_core_srv_conf_t *cscf;
154 cscf = ngx_stream_conf_get_module_srv_conf(cf, ngx_stream_core_module);
155 slot = &cscf->resolver;
156 plane = "stream";
157 }
158 if (slot == NULL) {
159 return;
160 }
161 if (*slot != NULL) {
162 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
163 "brix_resolver: %s scope already has an explicit resolver; "
164 "keeping it (brix targets still use resolv.conf)", plane);
165 return;
166 }
167 *slot = pol->resolver;
168 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
169 "brix_resolver: %s resolver seeded from %V (%ui nameserver%s, "
170 "%ui search domain%s, ndots %ui)", plane, &pol->path,
171 (ngx_uint_t) pol->rc.nnameservers,
172 pol->rc.nnameservers == 1 ? "" : "s",
173 (ngx_uint_t) pol->rc.nsearch, pol->rc.nsearch == 1 ? "" : "s",
174 (ngx_uint_t) pol->rc.ndots);
175}

178static char *
179dns_policy_activate(ngx_conf_t *cf, brix_dns_policy_t *pol)
180{
181 u_char path[NGX_MAX_PATH];
182
183 ngx_cpystrn(path, pol->path.data,
184 ngx_min(pol->path.len + 1, sizeof(path)));
185 if (brix_resolv_conf_load(&pol->rc, (const char *) path) != 0) {
186 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
187 "brix_resolver: cannot read \"%V\" — using the libc defaults "
188 "(nameserver 127.0.0.1); the server still starts", &pol->path);
189 }
190 if (pol->max_ttl < pol->min_ttl) {
191 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
192 "brix_resolver: max_ttl is below min_ttl");
193 return NGX_CONF_ERROR;
194 }
195 pol->resolver = brix_dns_resolver_build(cf, pol);
196 if (pol->resolver != NULL) {
197 dns_seed_core_slot(cf, pol);
198 }
199#if (NGX_THREADS)
200 /* the libc fallback path needs the default pool to exist */
201 if (ngx_thread_pool_add(cf, NULL) == NULL) {
202 return NGX_CONF_ERROR;
203 }
204#endif
205 return NGX_CONF_OK;
206}

209char *
210brix_conf_set_resolver(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
211{
212 brix_shared_conf_t *c = conf; /* member 0 of every brix module conf */
213 ngx_str_t *value = cf->args->elts;
214 brix_dns_policy_t *pol;
215 ngx_uint_t i;
216 char *rc;
217
218 (void) cmd;
219 if (c->dns.policy != NULL) {
220 return "is duplicate";
221 }
222 pol = dns_policy_create(cf);
223 if (pol == NULL) {
224 return NGX_CONF_ERROR;
225 }
226 if (value[1].len == 3 && ngx_strncmp(value[1].data, "off", 3) == 0) {
227 if (cf->args->nelts != 2) {
228 return "off takes no parameters";
229 }
230 pol->mode = BRIX_DNS_MODE_OFF;
231 c->dns.policy = pol;
232 return NGX_CONF_OK;
233 }
234 if (value[1].len != 4 || ngx_strncmp(value[1].data, "auto", 4) != 0) {
235 return "expects auto|off";
236 }
237 pol->mode = BRIX_DNS_MODE_AUTO;
238 for (i = 2; i < cf->args->nelts; i++) {
239 rc = dns_parse_option(cf, pol, &value[i]);
240 if (rc != NGX_CONF_OK) {
241 return rc;
242 }
243 }
244 rc = dns_policy_activate(cf, pol);
245 if (rc != NGX_CONF_OK) {
246 return rc;
247 }
248 c->dns.policy = pol;
249 return NGX_CONF_OK;
250}

253char *
254brix_conf_set_dns_retry(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
255{
256 brix_shared_conf_t *c = conf;
257 ngx_str_t *value = cf->args->elts;
258 ngx_msec_t initial, max;
259
260 (void) cmd;
261 if (c->dns.retry_initial != NGX_CONF_UNSET_MSEC) {
262 return "is duplicate";
263 }
264 initial = ngx_parse_time(&value[1], 0);
265 max = ngx_parse_time(&value[2], 0);
266 if (initial == (ngx_msec_t) NGX_ERROR || max == (ngx_msec_t) NGX_ERROR) {
267 return "invalid time value";
268 }
269 if (initial == 0 || max < initial) {
270 return "expects <initial> <= <max>, both > 0";
271 }
272 c->dns.retry_initial = initial;
273 c->dns.retry_max = max;
274 return NGX_CONF_OK;
275}

278/*
279 * brix_dns_cache_max <n> — bound both per-worker DNS answer caches.
280 *
281 * WHAT: parse the entry cap into the block's DNS conf and apply it to the
282 * forward and reverse caches straight away.
283 * WHY: the caches are per-process singletons. Applying the bound when a
284 * runtime target is armed (the first shape of this knob) left it inert
285 * in a block that registers no hostname — including the reverse-DNS
286 * case, whose key space is the PEER's address and therefore the one
287 * that most needs a bound.
288 * HOW: parse time, in the master, so every forked worker inherits the value;
289 * one process holds one cache, so the last value parsed wins.
290 */
291char *
292brix_conf_set_dns_cache_max(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
293{
294 brix_shared_conf_t *c = conf;
295 ngx_str_t *value = cf->args->elts;
296 ngx_int_t n;
297
298 (void) cmd;
299 if (c->dns.cache_max != NGX_CONF_UNSET_UINT) {
300 return "is duplicate";
301 }
302 n = ngx_atoi(value[1].data, value[1].len);
303 if (n == NGX_ERROR || n <= 0) {
304 return "invalid number";
305 }
306 c->dns.cache_max = (ngx_uint_t) n;
307 brix_dns_cache_set_max(c->dns.cache_max);
308 brix_dns_reverse_cache_set_max(c->dns.cache_max);
309 return NGX_CONF_OK;
310}

313/*
314 * Config-time: guarantee that this configuration will have a runtime DNS
315 * backend, for a feature that resolves a name brix_dns_target_register() never
316 * saw — a peer's PTR (XrdAcc `h` rules, a protbind host template, `brix_auth
317 * host`), a pmark firefly destination, a proxy upstream or a WebDAV proxy-pool
318 * backend added through the admin API.
319 *
320 * WHY: with no `brix_resolver` the only backend either direction has is the
321 * thread-pool libc call, and ngx_thread_pool_get() declines when no pool
322 * was declared anywhere in the configuration. brix_dns_resolve() and
323 * brix_dns_reverse() then answer "no resolver and no thread pool
324 * available", the failure is negative-cached, and the consumer silently
325 * degrades — an `h` rule stops granting, a firefly destination is never
326 * reached, a proxy backend is marked dead. Before phase 116 the same
327 * lookups were blocking libc calls on the event loop, so they answered
328 * (and stalled the worker, which is why they had to go).
329 * HOW: the same ngx_thread_pool_add(cf, NULL) that brix_conf_set_resolver()
330 * and brix_dns_target_register() already perform, so a config that
331 * resolves anything behaves identically however the name got there. The
332 * pool is left at threads == 0 for ngx_thread_pool_init_conf() to fill
333 * with nginx's own defaults — pre-setting it would make a later
334 * `thread_pool default ...` directive fail as a duplicate. Idempotent.
335 * Threadless builds keep the pre-existing degrade behaviour: they have no
336 * non-blocking backend to offer.
337 */
338char *
339brix_dns_backend_prepare(ngx_conf_t *cf)
340{
341#if (NGX_THREADS)
342 if (ngx_thread_pool_add(cf, NULL) == NULL) {
343 return NGX_CONF_ERROR;
344 }
345#else
346 (void) cf;
347#endif
348 return NGX_CONF_OK;
349}
