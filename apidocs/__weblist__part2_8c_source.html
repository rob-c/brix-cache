6/*
7 * weblist.c — list the files under a WebDAV collection (for recursive web copy).
8 *
9 * WHAT: brix_webdav_list() issues one PROPFIND Depth: infinity against a davs/http
10 * collection and returns the absolute server paths of every FILE beneath it
11 * (collections themselves are skipped — subdirs are recreated locally from the
12 * file paths).
13 * WHY: The XRootD/WebDAV wire has no "give me the tree" transfer op, so `xrdcp -r`
14 * over davs:// must enumerate the collection itself and copy each file. Keeping
15 * the enumeration here (a new file) leaves the copy engine untouched.
16 * HOW: PROPFIND over the existing HTTP client; scan the multistatus body for each
17 * <D:response> block, take its <D:href> and treat it as a file unless the block
18 * carries <D:collection/>. hrefs are percent-decoded; an absolute-URL href is
19 * reduced to its path. Bounded entry count.
20 *
21 * Clean-room: parses the documented WebDAV multistatus shape this module emits
22 * (src/protocols/webdav/propfind.c), not any client library.
23 */

37/* ---- DELETE a WebDAV resource (file or empty collection) ----
38 *
39 * WHAT: Issues one WebDAV DELETE for `path` on endpoint `u`. Returns 0 on
40 * success (200/202/204), -1 with *st set otherwise — 404/410 mapped to
41 * XRDC_ENOENT, 401/403 to XRDC_EAUTH. bearer NULL ⇒ anonymous;
42 * client_cert is the X.509 proxy PEM for mutual TLS (or NULL).
43 *
44 * WHY: gives `xrdfs rm`/`rmdir` a WebDAV transport so write-metadata verbs work
45 * over http/https/dav/davs, not only root://. DELETE is NOT made idempotent
46 * on 404 (unlike MKCOL): removing a non-existent path is a real error the
47 * user should see, matching the binary protocol's kXR_NotFound.
48 *
49 * HOW: 1. Build the optional bearer header.
50 * 2. brix_http_req(..., "DELETE", ...).
51 * 3. Accept 200/202/204; otherwise set a mapped status and fail.
52 */
53int
54brix_webdav_delete(const brix_weburl *u, const char *path, const char *bearer,
55 int verify, const char *ca_dir, const char *client_cert,
56 brix_status *st)
57{
58 char headers[8192];
59 brix_http_resp r;
60 int ok;
61
62 webdav_bearer_header(bearer, headers, sizeof(headers));
63 if (brix_http_req(u->host, u->port, u->tls, "DELETE", path,
64 headers[0] ? headers : NULL, NULL, 0, XRDC_WEBLIST_TIMEOUT,
65 verify, ca_dir, client_cert, &r, st) != 0) {
66 return -1;
67 }
68 ok = (r.status == 200 || r.status == 202 || r.status == 204);
69 if (!ok) {
70 brix_status_set(st, webdav_status_to_xrdc(r.status), 0,
71 "DELETE returned HTTP %d", r.status);
72 }
73 brix_http_resp_free(&r);
74 return ok ? 0 : -1;
75}

79 *
80 * WHAT: Issues one WebDAV MOVE of `path` to `dest_abs` (an absolute-URL

120/* ---- Build the PROPFIND request headers for a Depth: infinity listing ----
121 *
122 * WHAT: Writes "Depth: infinity" plus an optional bearer Authorization line into
123 * headers[].
124 * WHY: The listing walks the whole collection (Depth: infinity) and may need a
125 * token; isolating the header build keeps the caller free of the auth branch.
126 * HOW: 1. With a non-empty bearer, emit both the Depth and Authorization lines.

129static void
130webdav_list_build_headers(const char *bearer, char *headers, size_t headerssz)

132 if (bearer != NULL && bearer[0] != '\0') {
133 snprintf(headers, headerssz,
134 "Depth: infinity\r\nAuthorization: Bearer %s\r\n", bearer);
135 } else {
136 snprintf(headers, headerssz, "Depth: infinity\r\n");
137 }

140/* ---- Append the file path of one <D:response> block, if it is a file ----
141 *
142 * WHAT: Given a single multistatus response block [p, end), extracts its <D:href>,
143 * skips it when the block carries <D:collection/>, and otherwise pushes the
144 * percent-decoded path onto *arr / *n / *cap. Returns 0 (including for skipped
145 * collections and hrefless/malformed blocks), or -1 (with *st set to
146 * XRDC_EPROTO) on an allocation failure.
147 * WHY: The response scan reduces to "per block, emit its file"; pulling the block
148 * logic out keeps the scan loop to control flow only.
149 * HOW: 1. Locate <D:href>..</D:href> inside the block; absent → nothing to emit.
150 * 2. If the block also holds <D:collection>, it is a directory → skip. 3. Else
151 * percent-decode the href (flags=0 keeps a literal '+'), reduce an absolute-URL
152 * href to its path component, and push it.
153 */
154static int
155webdav_list_emit_block(const char *p, const char *end, weblist_acc_t *acc,
156 brix_status *st)
157{
158 const char *h = strstr(p, "<D:href>");
159 const char *he;
160 const char *col;
161 char href[XRDC_PATH_MAX];
162 const char *path = href;
163
164 if (h == NULL || h >= end) {
165 return 0;
166 }
167 h += 8;
168 he = strstr(h, "</D:href>");
169 if (he == NULL || he >= end) {
170 return 0;
171 }
172 col = strstr(p, "<D:collection");
173 if (col != NULL && col < end) {
174 return 0; /* directory — subdirs are recreated from the file paths */
175 }
176 /* flags=0: keep a literal '+' (it is a real path byte in an href, not a
177 * form-encoded space). */
178 if (brix_http_urldecode((const unsigned char *) h, (size_t) (he - h),
179 href, sizeof(href), 0) != BRIX_URLDECODE_OK) {
180 href[0] = '\0'; /* overflow/malformed → skip content */
181 }
182 /* reduce an absolute-URL href to its path component */
183 if (strstr(href, "://") != NULL) {
184 char *sl = strstr(href, "://");
185 char *ps = strchr(sl + 3, '/');
186 path = ps ? ps : "/";
187 }
188 if (push(acc, path) != 0) {
189 brix_status_set(st, XRDC_EPROTO, 0, "webdav list: out of memory");
190 return -1;
191 }
192 return 0;
193}

195/* ---- Scan a PROPFIND multistatus body, collecting every file path ----
196 *
197 * WHAT: Walks each <D:response>..</D:response> block in body and appends the file
198 * paths onto *arr / *n / *cap, up to XRDC_WEBLIST_MAX. Returns 0, or -1 (with *st
199 * set) on an allocation failure inside a block.
200 * WHY: Separates the block-boundary iteration from the per-block extraction so the
201 * public entry stays a thin request/dispatch shell.
202 * HOW: 1. Find each "<D:response" and its closing "</D:response>"; stop at an
203 * unterminated block or the entry cap. 2. Hand the block to
204 * webdav_list_emit_block. 3. Advance past the close tag.
205 */
206static int
207webdav_list_scan_body(const char *body, weblist_acc_t *acc, brix_status *st)
208{
209 const char *p = body ? body : "";
210
211 while ((p = strstr(p, "<D:response")) != NULL && acc->n < XRDC_WEBLIST_MAX) {
212 const char *end = strstr(p, "</D:response>");
213 if (end == NULL) {
214 break;
215 }
216 if (webdav_list_emit_block(p, end, acc, st) != 0) {
217 return -1;
218 }
219 p = end + 12;
220 }
221 return 0;
222}

225brix_webdav_list(const brix_weburl *u, const char *bearer, int verify,

228{
229 char headers[8192];

232
233 *paths = NULL;
234 *n_out = 0;

238 }
239 webdav_list_build_headers(bearer, headers, sizeof(headers));
240 if (brix_http_req(u->host, u->port, u->tls, "PROPFIND", u->path, headers,
241 NULL, 0, XRDC_WEBLIST_TIMEOUT, verify, ca_dir, client_cert, &r, st) != 0) {
242 return -1;
243 }
244 if (r.status != 207 && r.status != 200) {
245 brix_status_set(st, XRDC_EPROTO, 0, "PROPFIND returned HTTP %d", r.status);
246 brix_http_resp_free(&r);
247 return -1;
248 }
249
250 if (webdav_list_scan_body(r.body, &acc, st) != 0) {
251 brix_strv_free(acc.arr, acc.n);
252 brix_http_resp_free(&r);
253 return -1;
254 }
255 brix_http_resp_free(&r);
256 *paths = acc.arr;
257 *n_out = acc.n;
