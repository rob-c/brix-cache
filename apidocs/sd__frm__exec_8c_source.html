28/* ===================== the "exec" MSS adapter (real HSM) =====================
29 * The classic FRM model: an operator-supplied stage command drives the real MSS
30 * (HPSS, CTA, dCache, an Enstore wrapper, ...). The local online buffer lives at
31 * <base>/.online/<key>; the recall/migrate/exists verbs shell out to:
32 * $BRIX_FRM_STAGECMD <verb> <key> <online-path>
33 * recall is expected to be ASYNC-SUBMIT (start the MSS recall and return promptly,
34 * not block until online); the driver then parks the open and polls until the
35 * online buffer appears. A `recall_poll` is the cheap local-buffer existence check,
36 * so no per-poll fork. */
37
38typedef struct {
39 frm_mss_head_t head; /* base + invoke — the shared-op seam */
40 char stagecmd[PATH_MAX]; /* brix_frm_stagecmd / $BRIX_FRM_STAGECMD */
41 ngx_msec_t timeout_ms; /* brix_frm_copy_timeout (0 = none) */
42 ngx_log_t *log;
43} exec_ctx_t;

45/* ===================== spawn hygiene (2.0, 2026-09-08) =====================
46 * The one program this adapter still spawns directly -- the `dread` listing in
47 * exec_list, whose stdout must be STREAMED (see its own note) -- is an operator
48 * binary executed from a worker. A bare posix_spawn hands it every descriptor the worker
49 * holds (the listen sockets, which nginx opens without CLOEXEC, live client
50 * connections, the epoll fd, the logs) and leaves it in the worker's process
51 * group. Two consequences a release cannot ship: a child of the program keeps
52 * the server's ports bound after a deadline kill or a restart, and the
53 * program can reach client traffic. exec_spawn therefore closes every fd above
54 * stderr in the child (after the caller's own dup2 actions) and starts it in a
55 * new session, so pid == pgid and the deadline kill reaches the whole group.
56 * glibc 2.34 has closefrom as a spawn action; older glibc gets one addclose
57 * per descriptor the worker holds right now (the same set, via /proc). */
58#ifndef __GLIBC_PREREQ
59/* Not glibc - assume old interface */
60#define __GLIBC_PREREQ(major, minor) 0
61#endif
62#if __GLIBC_PREREQ(2, 34)
63static int
64exec_fa_close_inherited(posix_spawn_file_actions_t *fa)
65{
66 return posix_spawn_file_actions_addclosefrom_np(fa, 3);
67}
68#else
69static int
70exec_fa_close_inherited(posix_spawn_file_actions_t *fa)
71{
72 DIR *d = opendir("/proc/self/fd");
73 struct dirent *e;
74 int fd, rc = 0;
75
76 if (d == NULL) {
77 return errno;
78 }
79 while (rc == 0 && (e = readdir(d)) != NULL) {
80 fd = atoi(e->d_name); /* "." and ".." read as 0: skipped */
81 if (fd >= 3 && fd != dirfd(d)) {
82 rc = posix_spawn_file_actions_addclose(fa, fd);
83 }
84 }
85 closedir(d);
86 return rc; /* a closed-by-then fd is ignored */
87}
88#endif

90/* posix_spawn with the hygiene above, for the streaming `dread` path only —
91 * every status-dependent verb goes through brix_subprocess_run instead (see
92 * exec_run). `fa` may carry the caller's dup2/close actions (NULL = none).
93 * 0 ok / -1 with errno. */
94int
95frm_exec_spawn(pid_t *pid, const char *prog, char *const argv[],
96 posix_spawn_file_actions_t *fa)
97{
98 posix_spawn_file_actions_t own;
99 posix_spawnattr_t attr;
100 int rc;
101
102 if (fa == NULL) {
103 posix_spawn_file_actions_init(&own);
104 fa = &own;
105 }
106 rc = exec_fa_close_inherited(fa);
107 if (rc == 0) {
108 posix_spawnattr_init(&attr);
109#ifdef POSIX_SPAWN_SETSID
110 rc = posix_spawnattr_setflags(&attr, POSIX_SPAWN_SETSID);
111#endif
112 if (rc == 0) {
113 rc = posix_spawn(pid, prog, fa, &attr, argv, environ);
114 }
115 posix_spawnattr_destroy(&attr);
116 }
117 if (fa == &own) {
118 posix_spawn_file_actions_destroy(&own);
119 }
120 if (rc != 0) {
121 errno = rc;
122 return -1;
123 }
124 return 0;
125}

127/* Run "<stagecmd> <verb> <key> <online>".
128 *
129 * WHAT: Return the child's exit code (0 ok, non-zero with EIO), or -1 on
130 * spawn/wait failure with the runner's errno. No shell: argv is passed directly.
131 *
132 * WHY: A normally exiting child reports only an exit code; errno left by the
133 * runner's tolerated intermediate-reaping race must not become an MSS error.
134 *
135 * The run goes through brix_subprocess_run (2.0, 2026-09-09), NOT a child of
136 * this worker: nginx's signal handler calls ngx_process_get_status() for
137 * SIGCHLD in workers too, so waitpid(-1, WNOHANG) there reaps ANY direct child
138 * before the feature's own wait reaches it — the adapter then saw ECHILD, lost
139 * the status and reported every verb as failed (the first live run of the
140 * rebuilt 2.0 binary: "unknown process NNNN exited with code 0" next to
141 * "stage command failed"). The shared runner puts the command under a
142 * double-forked agent the worker never had as a child, and that agent enforces
143 * the brix_frm_copy_timeout deadline and the SIGKILL of the whole process
144 * group. Spawn hygiene (closefrom(3) + its own session) is the runner's.
145 *
146 * HOW:
147 * 1. Pass the verb arguments and configured deadline to the shared runner.
148 * 2. Preserve runner failures, logging an enforced timeout.
149 * 3. Map a normal non-zero exit to EIO, independent of incidental errno.
150 */
151static int
152exec_run(const exec_ctx_t *c, const char *verb, const char *key,
153 const char *online)
154{
155 char *argv[5];
156 brix_subprocess_req_t req;
157 int exit_code = -1;
158
159 argv[0] = (char *) c->stagecmd;
160 argv[1] = (char *) verb;
161 argv[2] = (char *) key;
162 argv[3] = (char *) online;
163 argv[4] = NULL;
164
165 req.argv = argv;
166 req.out = NULL; /* the verbs report by exit code */
167 req.outsz = 0;
168 req.timeout_ms = (unsigned) c->timeout_ms;
169
170 if (brix_subprocess_run(&req, NULL, &exit_code) != 0) {
171 if (errno == ETIMEDOUT) {
172 ngx_log_error(NGX_LOG_ERR, c->log, 0,
173 "xrootd frm: stage command \"%s %s %s\" exceeded "
174 "brix_frm_copy_timeout (%M ms) and was killed",
175 c->stagecmd, verb, key, c->timeout_ms);
176 }
177 return -1;
178 }
179 errno = (exit_code == 0) ? 0 : EIO;
180 return exit_code;
181}

183/* The exec adapter's frm_mss_invoke_fn: every MSS verb is one child run of the
184 * stage command. The stage protocol has no purge verb — the MSS-side purge is

186static int
187exec_invoke(void *mss, const char *verb, const char *key, const char *online)
188{
189 if (strcmp(verb, "purge") == 0) {

201/* exec_list — §3.7 MSS namespace enumeration, the stock `rsscmd dread` analog:
202 * run "<stagecmd> dread <key> ''" with stdout captured; the command prints one
203 * entry name per line, a trailing '/' marking a directory. Nonzero exit (or a
204 * spawn failure) is -1 — the driver then reports the key as not enumerable.
205 * A stagecmd that does not know the dread verb simply exits nonzero, so
206 * existing recall-only stage commands keep today's ENOTSUP behaviour. */
207
208/* Normalize one dread line in place (strip CR/LF, a trailing '/' marks a dir)
209 * and hand a non-empty entry to `cb`. Returns cb's stop flag (0 to continue). */
210static int
211exec_emit_line(char *line, int (*cb)(void *ud, const char *name, int is_dir),
212 void *ud)
213{
214 size_t n = strlen(line);
215 int is_dir = 0;
216
217 while (n > 0 && (line[n - 1] == '\n' || line[n - 1] == '\r')) {
218 line[--n] = '\0';
219 }
220 if (n > 0 && line[n - 1] == '/') {
221 line[--n] = '\0';
222 is_dir = 1;
223 }
224 if (n == 0) {
225 return 0;
226 }
227 return cb(ud, line, is_dir);
228}

230static int
231exec_list(void *mss, const char *key,
232 int (*cb)(void *ud, const char *name, int is_dir), void *ud)
233{
234 exec_ctx_t *c = mss;
235 char *argv[5];
236 posix_spawn_file_actions_t fa;
237 int pfd[2];
238 pid_t pid;
239 int status;
240 FILE *out;
241 char line[NAME_MAX + 2];
242
243 argv[0] = (char *) c->stagecmd;
244 argv[1] = (char *) "dread";
245 argv[2] = (char *) key;
246 argv[3] = (char *) "";
247 argv[4] = NULL;
248
249 if (pipe(pfd) != 0) {
250 return -1;
251 }
252 posix_spawn_file_actions_init(&fa);
253 posix_spawn_file_actions_adddup2(&fa, pfd[1], STDOUT_FILENO);
254 posix_spawn_file_actions_addclose(&fa, pfd[0]);
255 posix_spawn_file_actions_addclose(&fa, pfd[1]);
256
257 if (frm_exec_spawn(&pid, c->stagecmd, argv, &fa) != 0) {
258 posix_spawn_file_actions_destroy(&fa);
259 close(pfd[0]);
260 close(pfd[1]);
261 return -1;
262 }
263 posix_spawn_file_actions_destroy(&fa);
264 close(pfd[1]);
265
266 out = fdopen(pfd[0], "r");
267 if (out == NULL) {
268 close(pfd[0]);
269 (void) waitpid(pid, &status, 0);
270 return -1;
271 }
272 while (fgets(line, sizeof(line), out) != NULL) {
273 if (exec_emit_line(line, cb, ud)) {
274 break;
275 }
276 }
277 fclose(out); /* closes pfd[0] */
278
279 while (waitpid(pid, &status, 0) < 0) {
280 if (errno == ECHILD) {
281 /* nginx's SIGCHLD handler reaps children in workers too, and this
282 * is the ONE adapter path that still spawns one directly: `dread`
283 * streams an unbounded listing through the pipe while the program
284 * runs, which the fixed-buffer capture in brix_subprocess_run
285 * cannot do without silently truncating a large directory. The
286 * stolen status is survivable HERE and nowhere else: the pipe
287 * reached EOF and the dread contract prints entries only on
288 * success, so what was read is the answer. Every verb whose result
289 * IS the exit status runs under the reparented agent (exec_run). */
290 return 0;
291 }
292 if (errno != EINTR) {
293 return -1;
294 }
295 }
296 return (WIFEXITED(status) && WEXITSTATUS(status) == 0) ? 0 : -1;
297}

312/* frm_mss_exchange — atomic swap of two ONLINE-BUFFER copies (phase-107 C6).
313 * Head-generic (frm_mss_head_t), shared by the exec and lib adapters; lives
314 * here because sd_frm_stub.c, home of the other frm_mss_* head ops, is at the
315 * 600-line cap. Raw SYS_renameat2 (glibc's wrapper postdates 2.28); a
316 * kernel/filesystem without RENAME_EXCHANGE answers ENOSYS/EINVAL, reported
317 * as ENOTSUP and never degraded to two renames (§3.5).
318 * WHAT: Return 0 after an atomic swap, or -1 with errno and no emulation.
319 * WHY: A plain rename destroys one online name instead of exchanging both.
320 * HOW: 1. Resolve both online paths. 2. Use Linux RENAME_EXCHANGE when available.
321 * 3. Report an unsupported primitive as ENOTSUP without mutating files. */
322#ifndef RENAME_EXCHANGE
323#define RENAME_EXCHANGE (1u << 1) /* <linux/fs.h>; avoided for its struct
324 * collisions, same as fs/path/beneath.c */
325#endif
326
327int
328frm_mss_exchange(void *mss, const char *a, const char *b)
329{
330 frm_mss_head_t *h = mss;
331 char pa[PATH_MAX];
332 char pb[PATH_MAX];
333
334 if (frm_online_path(h->base, a, pa, sizeof(pa)) != 0
335 || frm_online_path(h->base, b, pb, sizeof(pb)) != 0)
336 {
337 errno = ENAMETOOLONG;
338 return -1;
339 }
340#if defined(__linux__) && defined(SYS_renameat2)
341 if (syscall(SYS_renameat2, AT_FDCWD, pa, AT_FDCWD, pb,
342 (unsigned int) RENAME_EXCHANGE) != 0)
343 {
344 if (errno == ENOSYS || errno == EINVAL) {
345 errno = ENOTSUP;
346 }
347 return -1;
348 }
349 return 0;
350#else
351 errno = ENOTSUP;
352 return -1;
353#endif
354}
