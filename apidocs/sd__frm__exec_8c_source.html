24/* ===================== the "exec" MSS adapter (real HSM) =====================
25 * The classic FRM model: an operator-supplied stage command drives the real MSS
26 * (HPSS, CTA, dCache, an Enstore wrapper, ...). The local online buffer lives at
27 * <base>/.online/<key>; the recall/migrate/exists verbs shell out to:
28 * $BRIX_FRM_STAGECMD <verb> <key> <online-path>
29 * recall is expected to be ASYNC-SUBMIT (start the MSS recall and return promptly,
30 * not block until online); the driver then parks the open and polls until the
31 * online buffer appears. A `recall_poll` is the cheap local-buffer existence check,
32 * so no per-poll fork. */
33
34typedef struct {
35 frm_mss_head_t head; /* base + invoke — the shared-op seam */
36 char stagecmd[PATH_MAX]; /* $BRIX_FRM_STAGECMD */
37 ngx_log_t *log;
38} exec_ctx_t;

40/* Run "<stagecmd> <verb> <key> <online>"; returns the child's exit code (0 ok), or
41 * -1 on spawn/wait failure. No shell - argv is passed directly (no injection). */
42static int
43exec_run(const exec_ctx_t *c, const char *verb, const char *key,
44 const char *online)
45{
46 char *argv[5];
47 pid_t pid;
48 int status;
49
50 argv[0] = (char *) c->stagecmd;
51 argv[1] = (char *) verb;
52 argv[2] = (char *) key;
53 argv[3] = (char *) online;
54 argv[4] = NULL;
55
56 if (posix_spawn(&pid, c->stagecmd, NULL, NULL, argv, environ) != 0) {
57 return -1;
58 }
59 while (waitpid(pid, &status, 0) < 0) {
60 if (errno != EINTR) {
61 return -1;
62 }
63 }
64 return WIFEXITED(status) ? WEXITSTATUS(status) : -1;
65}

67/* The exec adapter's frm_mss_invoke_fn: every MSS verb is one child run of the
68 * stage command. The stage protocol has no purge verb — the MSS-side purge is
69 * a local no-op (the shared op already unlinked the online buffer). */
70static int
71exec_invoke(void *mss, const char *verb, const char *key, const char *online)
72{
73 if (strcmp(verb, "purge") == 0) {
74 return 0;
75 }
76 return exec_run(mss, verb, key, online);
77}

85/* exec_list — §3.7 MSS namespace enumeration, the stock `rsscmd dread` analog:
86 * run "<stagecmd> dread <key> ''" with stdout captured; the command prints one
87 * entry name per line, a trailing '/' marking a directory. Nonzero exit (or a
88 * spawn failure) is -1 — the driver then reports the key as not enumerable.
89 * A stagecmd that does not know the dread verb simply exits nonzero, so
90 * existing recall-only stage commands keep today's ENOTSUP behaviour. */
91
92/* Normalize one dread line in place (strip CR/LF, a trailing '/' marks a dir)
93 * and hand a non-empty entry to `cb`. Returns cb's stop flag (0 to continue). */
94static int
95exec_emit_line(char *line, int (*cb)(void *ud, const char *name, int is_dir),
96 void *ud)
97{
98 size_t n = strlen(line);
99 int is_dir = 0;
100
101 while (n > 0 && (line[n - 1] == '\n' || line[n - 1] == '\r')) {
102 line[--n] = '\0';
103 }
104 if (n > 0 && line[n - 1] == '/') {
105 line[--n] = '\0';
106 is_dir = 1;
107 }
108 if (n == 0) {
109 return 0;
110 }
111 return cb(ud, line, is_dir);
112}

114static int
115exec_list(void *mss, const char *key,
116 int (*cb)(void *ud, const char *name, int is_dir), void *ud)
117{
118 exec_ctx_t *c = mss;
119 char *argv[5];
120 posix_spawn_file_actions_t fa;
121 int pfd[2];
122 pid_t pid;
123 int status;
124 FILE *out;
125 char line[NAME_MAX + 2];
126
127 argv[0] = (char *) c->stagecmd;
128 argv[1] = (char *) "dread";
129 argv[2] = (char *) key;
130 argv[3] = (char *) "";
131 argv[4] = NULL;
132
133 if (pipe(pfd) != 0) {
134 return -1;
135 }
136 posix_spawn_file_actions_init(&fa);
137 posix_spawn_file_actions_adddup2(&fa, pfd[1], STDOUT_FILENO);
138 posix_spawn_file_actions_addclose(&fa, pfd[0]);
139 posix_spawn_file_actions_addclose(&fa, pfd[1]);
140
141 if (posix_spawn(&pid, c->stagecmd, &fa, NULL, argv, environ) != 0) {
142 posix_spawn_file_actions_destroy(&fa);
143 close(pfd[0]);
144 close(pfd[1]);
145 return -1;
146 }
147 posix_spawn_file_actions_destroy(&fa);
148 close(pfd[1]);
149
150 out = fdopen(pfd[0], "r");
151 if (out == NULL) {
152 close(pfd[0]);
153 (void) waitpid(pid, &status, 0);
154 return -1;
155 }
156 while (fgets(line, sizeof(line), out) != NULL) {
157 if (exec_emit_line(line, cb, ud)) {
158 break;
159 }
160 }
161 fclose(out); /* closes pfd[0] */
162
163 while (waitpid(pid, &status, 0) < 0) {
164 if (errno == ECHILD) {
165 /* In a worker, nginx's SIGCHLD handler reaps children — and unlike
166 * exec_run (which waits immediately and wins that race), draining
167 * the pipe to EOF gives the handler time to reap first. The exit
168 * status is lost, but the pipe reached EOF and the dread contract
169 * prints entries only on success — accept what was read. */
170 return 0;
171 }
172 if (errno != EINTR) {
173 return -1;
174 }
175 }
176 return (WIFEXITED(status) && WEXITSTATUS(status) == 0) ? 0 : -1;
177}

179/* exec_mkpath — §3.7 rcreate analog: "<stagecmd> rcreate <key> ''", exit 0 = the
180 * MSS created the directory (parents included). Uses exec_run's immediate
181 * waitpid, which wins the nginx SIGCHLD reap race (see exec_list). A stagecmd
182 * that does not know the verb exits nonzero ⇒ -1 (not creatable). */
183static int
184exec_mkpath(void *mss, const char *key, mode_t mode)
185{
186 exec_ctx_t *c = mss;
187
188 (void) mode; /* the MSS owns its modes; the verb carries none */
189 return (exec_run(c, "rcreate", key, "") == 0) ? 0 : -1;
190}

192/* frm_mss_exchange — atomic swap of two ONLINE-BUFFER copies (phase-107 C6).
193 * Head-generic (frm_mss_head_t), shared by the exec and lib adapters; lives
194 * here because sd_frm_stub.c, home of the other frm_mss_* head ops, is at the
195 * 600-line cap. Raw SYS_renameat2 (glibc's wrapper postdates 2.28); a
196 * kernel/filesystem without RENAME_EXCHANGE answers ENOSYS/EINVAL, reported
197 * as ENOTSUP and never degraded to two renames (§3.5). */
198#ifndef RENAME_EXCHANGE
199#define RENAME_EXCHANGE (1u << 1) /* <linux/fs.h>; avoided for its struct
200 * collisions, same as fs/path/beneath.c */
201#endif
202
203int
204frm_mss_exchange(void *mss, const char *a, const char *b)
205{
206 frm_mss_head_t *h = mss;
207 char pa[PATH_MAX];
208 char pb[PATH_MAX];
209
210 if (frm_online_path(h->base, a, pa, sizeof(pa)) != 0
211 || frm_online_path(h->base, b, pb, sizeof(pb)) != 0)
212 {
213 errno = ENAMETOOLONG;
214 return -1;
215 }
216 if (syscall(SYS_renameat2, AT_FDCWD, pa, AT_FDCWD, pb,
217 (unsigned int) RENAME_EXCHANGE) != 0)
218 {
219 if (errno == ENOSYS || errno == EINVAL) {
220 errno = ENOTSUP;
221 }
222 return -1;
223 }
224 return 0;
225}
