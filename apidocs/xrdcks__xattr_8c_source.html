1/*
2 * xrdcks_xattr.c — xrdcks: manage a checksum stored in a file's extended
3 * attribute (§7.20 client-app parity).
4 *
5 * WHAT: `xrdcks <path> <cksname> [<cksval>|delete]`. With no value, GET (print
6 * the stored checksum; compute+store it on a miss). With a hex value,
7 * SET it. With `delete`, remove it. The checksum lives in the
8 * `user.XrdCks.<cksname>` xattr as the XrdCks `XrdCksData` binary record.
9 * WHY: Files carrying `user.XrdCks.*` checksum xattrs (written by a stock
10 * XRootD server or the stock tool) are common in mixed deployments;
11 * BriX had no tool to read/write/verify them. The on-disk record layout
12 * is a stable, documented format (XrdCksData.hh) — matched byte-exactly
13 * here. NB: the STOCK `xrdcks` CLI on some builds is buggy (segfaults on
14 * get, drops the leading value byte on set); this is a correct
15 * implementation of the FORMAT, deliberately not bug-compatible.
16 * HOW: XrdCksData is a fixed 96-byte record:
17 * Name[16] fmTime[8 BE] csTime[4 BE] Rsvd[3] Length[1] Value[64]
18 * fmTime is the file mtime; Length is the value byte count; Value holds
19 * the binary digest (first Length bytes). We compute via the shared
20 * brix_cksum engine (hex) and hex-decode into Value.
21 */

44/* ---- hex <-> binary (even-length hex only) ---- */
45static int
46xrdcks_hex_to_bin(const char *hex, uint8_t *out, size_t outcap, size_t *outlen)
47{
48 size_t hlen = strlen(hex), i;
49
50 if (hlen == 0 || (hlen & 1) || hlen / 2 > outcap) {
51 return -1;
52 }
53 for (i = 0; i < hlen; i++) {
54 char ch = hex[i];
55 uint8_t nib;
56
57 if (ch >= '0' && ch <= '9') { nib = (uint8_t) (ch - '0'); }
58 else if (ch >= 'a' && ch <= 'f') { nib = (uint8_t) (ch - 'a' + 10); }
59 else if (ch >= 'A' && ch <= 'F') { nib = (uint8_t) (ch - 'A' + 10); }
60 else { return -1; }
61 if (i & 1) { out[i / 2] |= nib; }
62 else { out[i / 2] = (uint8_t) (nib << 4); }
63 }
64 *outlen = hlen / 2;
65 return 0;
66}

68static void
69xrdcks_bin_to_hex(const uint8_t *bin, size_t n, char *out)
70{
71 static const char hexd[] = "0123456789abcdef";
72 size_t i;
73
74 for (i = 0; i < n; i++) {
75 out[i * 2] = hexd[bin[i] >> 4];
76 out[i * 2 + 1] = hexd[bin[i] & 0x0f];
77 }
78 out[n * 2] = '\0';
79}

90/* ---- Encode an XrdCksData record (BIG-ENDIAN time fields, as stock stores) --- */
91static void
92xrdcks_encode(const char *cksname, int64_t fmtime, const uint8_t *val,
93 size_t vlen, uint8_t rec[XRDCKS_RECSZ])
94{
95 uint64_t fm = (uint64_t) fmtime;
96 int i;
97
98 memset(rec, 0, XRDCKS_RECSZ);
99 snprintf((char *) rec, XRDCKS_NAMESZ, "%s", cksname); /* Name[16] */
100 for (i = 0; i < 8; i++) { /* fmTime[8] BE */
101 rec[16 + i] = (uint8_t) (fm >> (56 - i * 8));
102 }
103 /* csTime[4]=0, Rsvd[3]=0 already zero. */
104 rec[31] = (uint8_t) vlen; /* Length[1] */
105 if (vlen > XRDCKS_VALSZ) { vlen = XRDCKS_VALSZ; }
106 memcpy(rec + 32, val, vlen); /* Value[64] */
107}

109/* ---- Decode a record's value length + bytes; -1 if malformed ---- */
110static int
111xrdcks_decode(const uint8_t *rec, size_t reclen, const uint8_t **val,
112 size_t *vlen)
113{
114 size_t len;
115
116 if (reclen != XRDCKS_RECSZ) {
117 return -1;
118 }
119 len = rec[31];
120 if (len > XRDCKS_VALSZ) {
121 return -1;
122 }
123 *val = rec + 32;
124 *vlen = len;
125 return 0;
126}

128/* ---- Compute the file's checksum into a binary value ---- */
129static int
130xrdcks_compute(const char *path, const char *cksname, uint8_t *val,
131 size_t valcap, size_t *vlen)
132{
133 brix_cksum_algo algo;
134 brix_status st;
135 char hex[2 * XRDCKS_VALSZ + 1];
136 int fd, rc;
137
138 if (brix_cksum_algo_parse(cksname, &algo) != 0) {
139 fprintf(stderr, "xrdcks: unknown checksum '%s'\n", cksname);
140 return -1;
141 }
142 fd = open(path, O_RDONLY);
143 if (fd < 0) {
144 fprintf(stderr, "xrdcks: open %s: %s\n", path, strerror(errno));
145 return -1;
146 }
147 brix_status_clear(&st);
148 rc = brix_cksum_fd(fd, algo, hex, sizeof(hex), &st);
149 close(fd);
150 if (rc != 0) {
151 fprintf(stderr, "xrdcks: %s: %s\n", path, st.msg);
152 return -1;
153 }
154 return xrdcks_hex_to_bin(hex, val, valcap, vlen);
155}

157/* ---- SET: write the given hex value into the xattr record ---- */
158static int
159xrdcks_set(const char *path, const char *cksname, const char *key,
160 const char *hexval)
161{
162 uint8_t val[XRDCKS_VALSZ], rec[XRDCKS_RECSZ];
163 size_t vlen;
164 struct stat sb;
165
166 if (xrdcks_hex_to_bin(hexval, val, sizeof(val), &vlen) != 0) {
167 fprintf(stderr, "xrdcks: bad hex value '%s' (even digits, <= %d bytes)\n",
168 hexval, XRDCKS_VALSZ);
169 return 4;
170 }
171 if (stat(path, &sb) != 0) {
172 fprintf(stderr, "xrdcks: stat %s: %s\n", path, strerror(errno));
173 return 4;
174 }
175 xrdcks_encode(cksname, (int64_t) sb.st_mtime, val, vlen, rec);
176 if (setxattr(path, key, rec, XRDCKS_RECSZ, 0) != 0) {
177 fprintf(stderr, "xrdcks: set %s on %s: %s\n", key, path,
178 strerror(errno));
179 return 4;
180 }
181 return 0;
182}

184/* ---- GET: print the stored value, or compute+store on a miss ---- */
185static int
186xrdcks_get(const char *path, const char *cksname, const char *key)
187{
188 uint8_t rec[XRDCKS_RECSZ];
189 ssize_t got = getxattr(path, key, rec, sizeof(rec));
190
191 if (got == XRDCKS_RECSZ) {
192 const uint8_t *val;
193 size_t vlen;
194 char hex[2 * XRDCKS_VALSZ + 1];
195
196 if (xrdcks_decode(rec, (size_t) got, &val, &vlen) != 0) {
197 fprintf(stderr, "xrdcks: %s: malformed %s record\n", path, key);
198 return 4;
199 }
200 xrdcks_bin_to_hex(val, vlen, hex);
201 printf("%s %s\n", cksname, hex);
202 return 0;
203 }
204 if (got < 0 && errno != ENODATA && errno != ENOATTR) {
205 fprintf(stderr, "xrdcks: get %s on %s: %s\n", key, path,
206 strerror(errno));
207 return 4;
208 }
209 /* No stored checksum: compute it, store it, and print (compute-on-miss). */
210 {
211 uint8_t val[XRDCKS_VALSZ], newrec[XRDCKS_RECSZ];
212 size_t vlen;
213 char hex[2 * XRDCKS_VALSZ + 1];
214 struct stat sb;
215
216 if (xrdcks_compute(path, cksname, val, sizeof(val), &vlen) != 0
217 || stat(path, &sb) != 0) {
218 return 4;
219 }
220 xrdcks_encode(cksname, (int64_t) sb.st_mtime, val, vlen, newrec);
221 (void) setxattr(path, key, newrec, XRDCKS_RECSZ, 0); /* best-effort */
222 xrdcks_bin_to_hex(val, vlen, hex);
223 printf("%s %s\n", cksname, hex);
224 return 0;
225 }
226}
