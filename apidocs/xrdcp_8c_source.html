130/* Append a strdup'd copy of `s` to a growable string array. 0 / -1. */
131int
132str_append(char ***list, size_t *n, size_t *cap, const char *s)
133{
134 if (*n == *cap) {
135 size_t nc = *cap ? *cap * 2 : 16;
136 char **na = (char **) realloc(*list, nc * sizeof(char *));
137 if (na == NULL) { return -1; }
138 *list = na;
139 *cap = nc;
140 }
141 (*list)[*n] = strdup(s);
142 if ((*list)[*n] == NULL) { return -1; }
143 (*n)++;
144 return 0;
145}

157/* Extract the alias name from `arg` (the part before the first ':') into name[sz].
158 * Returns 1 if `arg` looks like an alias reference, 0 otherwise (no colon, empty or
159 * oversized name, or a scheme:// URL). */
160static int
161alias_name_of(const char *arg, char *name, size_t sz)
162{
163 const char *colon = strchr(arg, ':');
164 size_t nlen;
165
166 if (colon == NULL) { return 0; }
167 nlen = (size_t) (colon - arg);
168 if (nlen == 0 || nlen >= sz) { return 0; }
169 if (colon[1] == '/' && colon[2] == '/') { return 0; } /* scheme:// — not an alias */
170 memcpy(name, arg, nlen);
171 name[nlen] = '\0';
172 return 1;
173}

176/* Fold the alias `info`'s per-endpoint credentials into `o` — a value already set
177 * (CLI flag or earlier alias) always wins. The opt pointers are backed by static
178 * storage for the process lifetime. PII: creds are never logged. */
179static void
180fold_alias_creds(const brix_alias_info *info, brix_copy_opts *o)
181{
182 static char s_bearer[8192], s_access[256], s_secret[256], s_region[64];
183
184 if (o->bearer == NULL && info->bearer[0] != '\0') {
185 snprintf(s_bearer, sizeof(s_bearer), "%s", info->bearer);
186 o->bearer = s_bearer;
187 }
188 /* Fold the S3 access/secret as ONE unit so a mismatched key pair can never be
189 * assembled from two different aliases. */
190 if (o->s3_access == NULL && o->s3_secret == NULL
191 && info->s3_access[0] != '\0' && info->s3_secret[0] != '\0') {
192 snprintf(s_access, sizeof(s_access), "%s", info->s3_access);
193 snprintf(s_secret, sizeof(s_secret), "%s", info->s3_secret);
194 o->s3_access = s_access;
195 o->s3_secret = s_secret;
196 }
197 if (o->s3_region == NULL && info->s3_region[0] != '\0') {
198 snprintf(s_region, sizeof(s_region), "%s", info->s3_region);
199 o->s3_region = s_region;
200 }
201 if (info->proxy[0] != '\0') {
202 setenv("X509_USER_PROXY", info->proxy, 0); /* 0 = don't clobber an existing env */
203 }
204}

207/* If `arg` names a ~/.xrdrc alias, fold its per-endpoint credentials into `o` — a
208 * value already set (CLI flag or earlier alias) always wins. The opt pointers are
209 * backed by static storage for the process lifetime. PII: creds are never logged. */
210void
211merge_alias_auth(const char *arg, brix_copy_opts *o)
212{
213 char name[256];
214 brix_alias_info info;
215
216 if (!alias_name_of(arg, name, sizeof(name))) { return; }
217 if (!brix_alias_lookup(name, &info)) { return; }
218
219 if (info.token_file_failed) {
220 fprintf(stderr, "xrdcp: alias %s: token_file %s missing or empty\n",
221 name, info.token_file);
222 }
223 fold_alias_creds(&info, o);
224}

227/* Copy the basename of a path/URL (after the last '/', ignoring trailing slashes)
228 * into out[sz]. out is empty if the input is all slashes. */
229void
230path_basename(const char *p, char *out, size_t sz)
231{
232 size_t len = strlen(p), start = 0, i, bl;
233 while (len > 0 && p[len - 1] == '/') { len--; } /* ignore trailing slashes */
234 for (i = 0; i < len; i++) {
235 if (p[i] == '/') { start = i + 1; }
236 }
237 bl = len - start;
238 if (bl >= sz) { bl = sz - 1; }
239 memcpy(out, p + start, bl);
240 out[bl] = '\0';
241}

244/* Trim leading/trailing whitespace (and CR/LF) from `line` in place and return a
245 * pointer to the first non-blank char. The returned string is "" for a blank line. */
246static char *
247manifest_trim(char *line)
248{
249 char *s = line, *e;
250 while (*s == ' ' || *s == '\t') { s++; }
251 e = s + strlen(s);
252 while (e > s && (e[-1] == '\n' || e[-1] == '\r' || e[-1] == ' ' || e[-1] == '\t')) {
253 *--e = '\0';
254 }
255 return s;
256}

259/* Read a manifest (one source per line; '#' comments + blank lines skipped) and
260 * append each entry to the source list. 0 / -1. */
261int
262read_manifest(const char *file, char ***list, size_t *n, size_t *cap)
263{
264 FILE *f = (strcmp(file, "-") == 0) ? stdin : fopen(file, "r");
265 char line[XRDC_PATH_MAX];
266
267 if (f == NULL) {
268 fprintf(stderr, "xrdcp: cannot open manifest %s: %s\n", file, strerror(errno));
269 return -1;
270 }
271 while (fgets(line, sizeof(line), f) != NULL) {
272 char *s = manifest_trim(line);
273 if (*s == '\0' || *s == '#') { continue; }
274 if (str_append(list, n, cap, s) != 0) {
275 if (f != stdin) { fclose(f); }
276 return -1;
277 }
278 }
279 if (f != stdin) { fclose(f); }
280 return 0;
281}

293/* An s3/s3s endpoint authenticates with AWS SigV4 keys, NOT a GSI proxy or
294 * bearer token — so the GSI/token credential pre-flight is irrelevant noise for

296int
297is_s3_url(const char *s)
298{
299 return strncasecmp(s, "s3://", 5) == 0 || strncasecmp(s, "s3s://", 6) == 0;

321/* Is `dst` an existing directory? 1=yes, 0=no, -1=can't determine (web/parse). */
322int
323dest_is_dir(const char *dst, const brix_opts *co)
324{
325 brix_url u;
326 brix_status st;
327 if (brix_is_web_url(dst)) { return -1; }
328 brix_status_clear(&st);
329 if (brix_url_parse(dst, &u, &st) != 0) { return -1; }
330 if (u.scheme == XRDC_SCHEME_LOCAL) {
331 struct stat sb;
332 return (stat(u.path, &sb) == 0 && S_ISDIR(sb.st_mode)) ? 1 : 0;
333 }
334 if (u.scheme == XRDC_SCHEME_ROOT || u.scheme == XRDC_SCHEME_ROOTS) {
335 brix_conn c;
336 brix_statinfo si;
337 int isdir;
338 if (brix_connect(&c, &u, co, &st) != 0) { return -1; }
339 isdir = (brix_stat(&c, u.path, &si, &st) == 0 && (si.flags & kXR_isDir)) ? 1 : 0;
340 brix_close(&c);
341 return isdir;
342 }
343 return -1;
344}

357
358/* brix_copy progress callback: a throttled \r-updated stderr bar with rate + ETA. */
359void
360xrdcp_progress(void *arg, long long done, long long total)
361{
362 xrdcp_prog *p = (xrdcp_prog *) arg;
363 uint64_t now = brix_mono_ns();
364 int final = (total >= 0 && done >= total);
365 double secs, mb, rate;
366
367 if (!final && (now - p->last_ns) < 200000000ULL) {
368 return; /* throttle intermediate updates to ~5 Hz */
369 }
370 p->last_ns = now;
371 secs = (double) (now - p->start_ns) / 1e9;
372 mb = (double) done / 1048576.0;
373 rate = (secs > 0.01) ? mb / secs : 0.0;
374 if (total > 0) {
375 int pct = (int) ((done * 100) / total);
376 double tmb = (double) total / 1048576.0;
377 double eta = (rate > 0.01) ? (tmb - mb) / rate : 0.0;
378 fprintf(stderr, "\r%-28s %3d%% %.1f/%.1f MiB %.1f MiB/s ETA %3.0fs ",
379 p->label, pct, mb, tmb, rate, eta);
380 } else {
381 fprintf(stderr, "\r%-28s %.1f MiB %.1f MiB/s ", p->label, mb, rate);
382 }
383 if (final) {
384 fprintf(stderr, "\n");
385 }
386 fflush(stderr);
387}

405static struct brix_cred_store *
406build_and_preflight_credentials(xrdcp_opts_t *o, xrdcp_lists_t *l)
407{
408 brix_copy_opts *opts = o->copt;
409 size_t i;
410
411 /* Fold any ~/.xrdrc per-endpoint credentials (the dst + every source alias) into
412 * opts so `xrdcp s3lab:/obj .` authenticates with no flags. */
413 merge_alias_auth(l->pos.items[l->pos.n - 1], opts);
414 for (i = 0; i < l->srcs.n; i++) {
415 merge_alias_auth(l->srcs.items[i], opts);
416 }
417
418 /* Expand globs (root:// + local) into the final source list. */
419 for (i = 0; i < l->srcs.n; i++) {
420 if (expand_source(l->srcs.items[i], o->conn,
421 &l->exp.items, &l->exp.n, &l->exp.cap) != 0) {
422 fprintf(stderr, "xrdcp: out of memory\n");
423 xrdcp_lists_free(l);
424 return NULL;
425 }
426 }
427 if (l->exp.n == 0) {
428 fprintf(stderr, "xrdcp: no sources after expansion\n");
429 xrdcp_lists_free(l);
430 return NULL;
431 }
432 /* --remove-source supports local and root:// sources only: web/S3 sources
433 * have no cheap post-transfer delete path and cannot be safely removed. */
434 if (opts->remove_source) {
435 for (i = 0; i < l->exp.n; i++) {
436 if (brix_is_web_url(l->exp.items[i])) {
437 fprintf(stderr, "xrdcp: --remove-source supports local and "
438 "root:// sources only\n");
439 xrdcp_lists_free(l);
440 return NULL;
441 }
442 }
443 }

445 /*
446 * Phase 40 (c): credential pre-flight. If a server endpoint is involved, warn
447 * the user INSTANTLY about a locally-detectable auth problem (expired/near-
448 * expiry bearer token or GSI proxy, or a read-only token on an upload) before
449 * the transfer fails with a bare "permission denied". Silent when creds look
450 * fine; never aborts (the server stays authoritative).
451 */
452 {
453 /* Only the GSI/token credential family is diagnosable here — an s3://
454 * endpoint authenticates with AWS SigV4 keys, so it must not trip the
455 * "GSI proxy expired" / "bearer token" pre-flight. */
456 int dst_cred = uses_cred_auth(o->dst);
457 int any_cred = dst_cred;
458 for (i = 0; i < l->exp.n && !any_cred; i++) {
459 if (uses_cred_auth(l->exp.items[i])) { any_cred = 1; }
460 }
461 if (any_cred) {
462 /* Phase 40 (b): if asked, proactively (re)acquire a stale token/proxy
463 * BEFORE diagnosing — so a healthy refresh leaves nothing to warn. */
464 if (o->auto_refresh) {
465 (void) brix_cred_autorefresh(dst_cred, o->oidc_account,
466 !opts->silent, stderr);
467 }
468 (void) brix_cred_diagnose(dst_cred, "xrdcp: ", stderr);
469 }
470 }
471
472 /* C1: build the credential store from CLI values and attach it to the
473 * connection options. The store is INERT here — nothing reads conn.cred
474 * yet; C2 will thread it through the auth/token handshake path. Building
475 * it now (before the transfer) means C2 only needs to consume conn.cred,
476 * not rebuild it. NULL/empty args fall back to per-handler env discovery. */
477 return brix_cli_cred_store_build(o->proxy, opts->bearer, NULL,
478 opts->s3_access, opts->s3_secret,
479 o->oidc_account, o->auto_refresh);
480}

483int
484main(int argc, char **argv)
485{
486 brix_copy_opts opts;
487 brix_opts conn;
488 xrdcp_opts_t o;
489 xrdcp_lists_t l;
490 int rc = 0;
491
492 memset(&opts, 0, sizeof(opts));
493 brix_opts_init(&conn); /* verify_host=1 + seed $XRDC_MAX_STALL_MS resilience */
494
495 memset(&o, 0, sizeof(o));
496 memset(&l, 0, sizeof(l));
497 o.copt = &opts;
498 o.conn = &conn;
499 o.jobs = 1; /* default: one file at a time (batch concurrency opt-in) */
500 /* Phase 94: data sub-streams ON by default (server default is also ON). An
501 * upload spreads write chunks across these secondaries (parallel upload); a
502 * server that does not service bound writes falls back to the primary safely.
503 * `-S N` overrides (N<=1 disables). Kept modest so tiny transfers pay only a
504 * few extra TCP setups. */
505 opts.streams = 4;
506
507 brix_crypto_init(); /* arm libxrdproto SHA-256/HMAC for GSI + sigver */
508 brix_copy_install_signal_handlers(); /* Phase 40 (a): drop partial dest on
509 * SIGINT/SIGTERM instead of leaving it */
510
511 /* Parse and validate command-line arguments. */
512 rc = parse_and_validate_args(argc, argv, &o, &l);
513 if (rc != 0) {
514 xrdcp_lists_free(&l);
515 if (rc == XRDCP_PARSE_EXIT_OK) {
516 return 0;
517 }
518 return rc;
519 }
520
521 /* Export --proxy into $X509_USER_PROXY so the davs/https leg's client-cert
522 * resolver (brix_web_proxy_pem) presents the same identity the root:// path
523 * already gets explicitly — the flag documents "overrides $X509_USER_PROXY",
524 * so it clobbers (overwrite=1) any inherited value. */
525 if (o.proxy != NULL && o.proxy[0] != '\0') {
526 setenv("X509_USER_PROXY", o.proxy, 1);
527 }
528
529 /* Build credential store with alias resolution, glob expansion, and pre-flight.
530 * C1: the store is INERT until C2 threads it through the auth path; NULL until
531 * brix_cli_cred_store_build runs; freed on every exit path after construction. */
