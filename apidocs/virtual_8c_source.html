1/* virtual.c — virtual / composed repos (phase-87 G16).
2 *
3 * WHAT: brix_cvmfs_virtual_repo <virtual-fqrn> <member-fqrn>... presents a
4 * read-only UNION of member repos under a repo name that need not exist
5 * anywhere upstream. A request naming the virtual fqrn is rewritten in
6 * place to member[0]; a definitive 404 advances to the next member
7 * (declaration order = precedence); the first non-404 answer is final.
8 * WHY: compose curated views (a site-wide umbrella over per-experiment
9 * repos, a staged repo shadowing production) purely at the HTTP plane —
10 * no catalog surgery, no origin changes, no second cache copy.
11 * HOW: the gate calls brix_cvmfs_virtual_enter() right after classification,
12 * BEFORE per-repo accounting, F3 repo authz, and class routing — so
13 * every member attempt is policed exactly as a direct request for that
14 * member would be (composition never elevates access), and the cache
15 * key IS the member path (an object cached via the virtual name and via
16 * direct member access is one entry). The handler's member loop and the
17 * off-loop fill-failure interceptor call brix_cvmfs_virtual_advance()
18 * on 404. A 401/403/5xx from a member is terminal: only "the object is
19 * not there" consults the next member — access denial must never be
20 * papered over by a more permissive sibling.
21 *
22 * HONEST LIMIT: this is HTTP-plane composition only. Signed metadata
23 * (.cvmfspublished et al.) comes from the first member that has it; no merged
24 * catalog is synthesized, so a FUSE client mounting the virtual name sees
25 * member[0]'s namespace plus whatever CAS objects the others contribute.
26 */
27#include "cvmfs.h"
28#include "cvmfs_module_internal.h"

30/* Rewrite r->uri to the current member: the fqrn span of the ORIGINAL
31 * (virtual-name) uri is replaced by members[virt_idx], then the result is
32 * re-classified so every downstream consumer of ctx->url sees the member.
33 * Only the repo span changes, so the traffic class is preserved — the
34 * classifier stays the sole truth all the same. */
35static ngx_int_t
36cvmfs_virt_apply(ngx_http_request_t *r, ngx_http_brix_cvmfs_ctx_t *ctx)
37{
38 const brix_cvmfs_virtual_t *v = ctx->virt;
39 const ngx_str_t *members = v->members->elts;
40 const ngx_str_t *member = &members[ctx->virt_idx];
41 size_t pre = ctx->virt_off;
42 size_t suf = pre + v->fqrn.len;
43 size_t len;
44 u_char *p;
45
46 len = ctx->virt_uri.len - v->fqrn.len + member->len;
47 p = ngx_pnalloc(r->pool, len);
48 if (p == NULL) {
49 return NGX_ERROR;
50 }
51 ngx_memcpy(p, ctx->virt_uri.data, pre);
52 ngx_memcpy(p + pre, member->data, member->len);
53 ngx_memcpy(p + pre + member->len, ctx->virt_uri.data + suf,
54 ctx->virt_uri.len - suf);
55 r->uri.data = p;
56 r->uri.len = len;
57
58 if (cvmfs_classify_url((const char *) p, len, &ctx->url) != 0
59 || ctx->url.repo == NULL)
60 {
61 return NGX_ERROR;
62 }
63 return NGX_OK;
64}

66ngx_int_t
67brix_cvmfs_virtual_enter(ngx_http_request_t *r,
68 ngx_http_brix_cvmfs_loc_conf_t *lcf)
69{
70 ngx_http_brix_cvmfs_ctx_t *ctx =
71 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
72 brix_cvmfs_virtual_t *e;
73 ngx_uint_t i;
74
75 /* ctx->virt set = a member pass of the advance loop (the uri already
76 * names a member — never rewrite twice, so composition cannot chain);
77 * repo == NULL = a REJECT shape the gate is about to refuse anyway. */
78 if (ctx == NULL || ctx->virt != NULL || ctx->url.repo == NULL) {
79 return NGX_DECLINED;
80 }
81
82 e = lcf->virtual_repos->elts;
83 for (i = 0; i < lcf->virtual_repos->nelts; i++) {
84 if (e[i].fqrn.len == ctx->url.repo_len
85 && ngx_strncmp(e[i].fqrn.data, ctx->url.repo,
86 ctx->url.repo_len) == 0)
87 {
88 break;
89 }
90 }
91 if (i == lcf->virtual_repos->nelts) {
92 return NGX_DECLINED; /* a direct (real) repo */
93 }
94
95 ctx->virt = &e[i];
96 ctx->virt_idx = 0;
97 ctx->virt_uri = r->uri;
98 ctx->virt_off = (size_t) ((u_char *) ctx->url.repo - r->uri.data);
99
100 if (cvmfs_virt_apply(r, ctx) != NGX_OK) {
101 return NGX_HTTP_INTERNAL_SERVER_ERROR;
102 }
103 return NGX_DECLINED;
104}

106ngx_int_t
107brix_cvmfs_virtual_advance(ngx_http_request_t *r)
108{
109 ngx_http_brix_cvmfs_ctx_t *ctx =
110 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
111
112 if (ctx == NULL || ctx->virt == NULL
113 || ctx->virt_idx + 1 >= ctx->virt->members->nelts)
114 {
115 return NGX_DECLINED; /* not composed / members exhausted */
116 }
117
118 ctx->virt_idx++;
119 if (cvmfs_virt_apply(r, ctx) != NGX_OK) {
120 return NGX_DECLINED; /* fail closed: the 404 stands */
121 }
122
123 /* each member attempt starts with a clean disposition — a failed
124 * member[0] fill must not label a member[1] cache hit as FILL */
125 ctx->cache_status = BRIX_CVMFS_CACHE_NONE;
126 ngx_str_null(&ctx->origin_used);
127 ctx->repo = NULL; /* re-mapped by the gate pass */
128
129 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
130 "cvmfs-virt: virtual=%V event=member-miss-advance member=%ui/%ui",
131 &ctx->virt->fqrn, ctx->virt_idx + 1,
132 (ngx_uint_t) ctx->virt->members->nelts);
133 return NGX_OK;
134}

136/* brix_cvmfs_virtual_repo <virtual-fqrn> <member-fqrn>... — one union entry
137 * per occurrence. Rejected at config time: a duplicate virtual name (would
138 * make precedence ambiguous), a duplicate member, and any nesting — a
139 * virtual name listed as a member of any entry, either direction. */
140
141static int
142cvmfs_virt_str_eq(const ngx_str_t *a, const ngx_str_t *b)
143{
144 return a->len == b->len
145 && ngx_strncmp(a->data, b->data, b->len) == 0;
146}

148/* Reject a new virtual name that is already a virtual (duplicate) or already
149 * a member of an earlier entry (nesting, member->virtual direction). */
150static char *
151cvmfs_virt_conf_check_name(ngx_conf_t *cf,
152 ngx_http_brix_cvmfs_loc_conf_t *c, const ngx_str_t *name)
153{
154 brix_cvmfs_virtual_t *e = c->virtual_repos->elts;
155 ngx_str_t *pm;
156 ngx_uint_t i, j;
157
158 for (i = 0; i < c->virtual_repos->nelts; i++) {
159 if (cvmfs_virt_str_eq(&e[i].fqrn, name)) {
160 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
161 "duplicate brix_cvmfs_virtual_repo \"%V\"", name);
162 return NGX_CONF_ERROR;
163 }
164 pm = e[i].members->elts;
165 for (j = 0; j < e[i].members->nelts; j++) {
166 if (cvmfs_virt_str_eq(&pm[j], name)) {
167 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
168 "brix_cvmfs_virtual_repo \"%V\" is already a member of "
169 "\"%V\" — virtual repos cannot nest", name,
170 &e[i].fqrn);
171 return NGX_CONF_ERROR;
172 }
173 }
174 }
175 return NGX_CONF_OK;
176}

178/* Validate one member name against the entry being built (self-reference,
179 * virtual->member nesting against every EARLIER entry, duplicate member)
180 * and append it. */
181static char *
182cvmfs_virt_conf_add_member(ngx_conf_t *cf,
183 ngx_http_brix_cvmfs_loc_conf_t *c, brix_cvmfs_virtual_t *entry,
184 const ngx_str_t *name)
185{
186 brix_cvmfs_virtual_t *e = c->virtual_repos->elts;
187 ngx_str_t *m;
188 ngx_uint_t j;
189
190 if (cvmfs_virt_str_eq(&entry->fqrn, name)) {
191 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
192 "brix_cvmfs_virtual_repo \"%V\" cannot be its own member",
193 &entry->fqrn);
194 return NGX_CONF_ERROR;
195 }
196 for (j = 0; j + 1 < c->virtual_repos->nelts; j++) {
197 if (cvmfs_virt_str_eq(&e[j].fqrn, name)) {
198 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
199 "brix_cvmfs_virtual_repo member \"%V\" is itself a "
200 "virtual repo — virtual repos cannot nest", name);
201 return NGX_CONF_ERROR;
202 }
203 }
204 m = entry->members->elts;
205 for (j = 0; j < entry->members->nelts; j++) {
206 if (cvmfs_virt_str_eq(&m[j], name)) {
207 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
208 "duplicate brix_cvmfs_virtual_repo member \"%V\"", name);
209 return NGX_CONF_ERROR;
210 }
211 }
212 m = ngx_array_push(entry->members);
213 if (m == NULL) {
214 return NGX_CONF_ERROR;
215 }
216 *m = *name;
217 return NGX_CONF_OK;
218}

222{
223 ngx_http_brix_cvmfs_loc_conf_t *c = conf;

237

243 if (entry == NULL) {

251 }
252
253 for (i = 2; i < cf->args->nelts; i++) {

257 return NGX_CONF_ERROR;
258 }
259 }
260
261 return NGX_CONF_OK;
