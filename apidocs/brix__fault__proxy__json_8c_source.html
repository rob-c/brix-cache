1/*
2 * brix_fault_proxy_json.c — optional JSON control-input front-end (A2).
3 *
4 * WHAT: lets a control-plane client send one command as a flat JSON object,
5 * e.g. {"cmd":"latency","args":"down 25"} or {"cmd":"status","args":"json"},
6 * instead of the bare "verb args" line grammar. The core recognises a
7 * leading '{' and hands the line here; this module reprojects it onto the
8 * exact same verb line and the core runs it through the ONE command parser.
9 *
10 * WHY: automation (and the A4 `status json` reply) is easier to drive from a
11 * JSON tool-chain when the request side is JSON too — without teaching the
12 * command grammar a second dialect. By translating to a verb line rather
13 * than duplicating any lever logic, every command stays defined in exactly
14 * one place (apply_command) and JSON input can never drift from it.
15 *
16 * HOW: a small dependency-free scanner extracts the string value of the object
17 * keys "cmd" (required) and "args" (optional), decoding standard JSON
18 * string escapes, and emits "<cmd>" or "<cmd> <args>". It parses only what
19 * it needs (two string fields of a flat object); anything malformed or a
20 * missing "cmd" returns -1 so the caller answers a structured parse error.
21 * No lever state is touched here — this is a pure string transform.
22 */

29/* Skip JSON insignificant whitespace. */
30static const char *
31json_skip_ws(const char *p)
32{
33 while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\r') {
34 p++;
35 }
36 return p;
37}

39/* Decode a single hex nibble, or -1 if not a hex digit. */
40static int
41hex_nibble(char c)
42{
43 if (c >= '0' && c <= '9') return c - '0';
44 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
45 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
46 return -1;
47}

49/* The two-character JSON escapes, as a table rather than a switch ladder
50 * (coding-standards §8.6). \u is handled separately — it is not two characters. */
51static const struct { char esc; char dec; } JSON_ESCAPES[] = {
52 { '"', '"' }, { '\\', '\\' }, { '/', '/' }, { 'b', '\b' },
53 { 'f', '\f' }, { 'n', '\n' }, { 'r', '\r' }, { 't', '\t' },
54};

56/* Decode a two-character escape; 1 = known (result in *dec), 0 = unknown. */
57static int
58json_escape_decode(char e, char *dec)
59{
60 for (size_t i = 0; i < sizeof(JSON_ESCAPES) / sizeof(JSON_ESCAPES[0]); i++) {
61 if (JSON_ESCAPES[i].esc == e) {
62 *dec = JSON_ESCAPES[i].dec;
63 return 1;
64 }
65 }
66 return 0;
67}

69/* Read exactly four hex digits as a code point, or -1 if any is not hex. A NUL
70 * is not a hex digit, so this cannot run off the end of the string. */
71static int
72json_parse_u4(const char *p)
73{
74 int cp = 0;
75 for (int i = 0; i < 4; i++) {
76 int h = hex_nibble(p[i]);
77 if (h < 0) {
78 return -1;
79 }
80 cp = (cp << 4) | h;
81 }
82 return cp;
83}

85/* Append `cp` to out[*o] as UTF-8. 0 = written, -1 = would overflow. */
86static int
87json_emit_utf8(unsigned cp, char *out, size_t *o, size_t outsz)
88{
89 if (cp < 0x80) {
90 if (*o + 1 >= outsz) return -1;
91 out[(*o)++] = (char) cp;
92 } else if (cp < 0x800) {
93 if (*o + 2 >= outsz) return -1;
94 out[(*o)++] = (char) (0xC0 | (cp >> 6));
95 out[(*o)++] = (char) (0x80 | (cp & 0x3F));
96 } else {
97 if (*o + 3 >= outsz) return -1;
98 out[(*o)++] = (char) (0xE0 | (cp >> 12));
99 out[(*o)++] = (char) (0x80 | ((cp >> 6) & 0x3F));
100 out[(*o)++] = (char) (0x80 | (cp & 0x3F));
101 }
102 return 0;
103}

105/* Decode the escape whose body starts at `*pp` (i.e. just past the backslash),
106 * appending to out[*o] and advancing `*pp` past it. 0 = ok, -1 = rejected.
107 * \uXXXX is emitted as UTF-8 (BMP only, which is all control-plane text needs). */
108static int
109json_parse_escape(const char **pp, char *out, size_t *o, size_t outsz)
110{
111 const char *p = *pp;
112 char dec;
113
114 if (*p == 'u') {
115 int cp = json_parse_u4(p + 1);
116 if (cp < 0) {
117 return -1;
118 }
119 *pp = p + 5; /* 'u' + four hex digits */
120 return json_emit_utf8((unsigned) cp, out, o, outsz);
121 }
122 if (!json_escape_decode(*p, &dec)) {
123 return -1; /* invalid escape */
124 }
125 if (*o + 1 >= outsz) {
126 return -1;
127 }
128 out[(*o)++] = dec;
129 *pp = p + 1;
130 return 0;
131}

133/*
134 * Parse a JSON string starting at `*pp` (which must point at the opening quote),
135 * decoding escapes into `out` (NUL-terminated, bounded by `outsz`). Advances
136 * `*pp` past the closing quote. Returns 0 on success, -1 on malformed input or
137 * overflow.
138 */
139static int
140json_parse_string(const char **pp, char *out, size_t outsz)
141{
142 const char *p = *pp;
143 size_t o = 0;
144
145 if (*p != '"') {
146 return -1;
147 }
148 p++;
149 for (;;) {
150 unsigned char c = (unsigned char) *p;
151 if (c == '\0') {
152 return -1; /* unterminated string */
153 }
154 if (c == '"') {
155 p++;
156 break; /* closing quote */
157 }
158 if (c == '\\') {
159 p++;
160 if (json_parse_escape(&p, out, &o, outsz) != 0) {
161 return -1;
162 }
163 continue;
164 }
165 if (o + 1 >= outsz) {
166 return -1; /* value would overflow */
167 }
168 out[o++] = (char) c;
169 p++;
170 }
171 out[o] = '\0';
172 *pp = p;
173 return 0;
174}

176/*
177 * Find the string value of top-level object key `key` in `json`. Returns 1 and
178 * fills `out` on success, 0 if the key is absent, -1 if present but malformed.
179 * Scans object members ("key" : value , ...) so a value string containing the
180 * key text cannot be mistaken for the key itself.
181 */
182/* Consume `"name" :` at `*pp`, leaving `*pp` on the value. 0 = ok, -1 = malformed. */
183static int
184json_member_key(const char **pp, char *kbuf, size_t ksz)
185{
186 const char *p = json_skip_ws(*pp);
187 if (json_parse_string(&p, kbuf, ksz) != 0) {
188 return -1;
189 }
190 p = json_skip_ws(p);
191 if (*p != ':') {
192 return -1;
193 }
194 *pp = json_skip_ws(p + 1);
195 return 0;
196}

198/* Consume the value at `*pp` — a string, or a bare token up to ',' or '}'. */
199static int
200json_skip_value(const char **pp)
201{
202 if (**pp != '"') {
203 while (**pp != '\0' && **pp != ',' && **pp != '}') {
204 (*pp)++;
205 }
206 return 0;
207 }
208 char skip[512];
209 return json_parse_string(pp, skip, sizeof skip);
210}

212/* ---- Copy a bare (non-string) JSON value verbatim ----
213 *
214 * WHAT: Copies the token at `*pp` up to ',' or '}' into `out`, advancing `*pp`
215 * past it. Returns 1 when a non-empty token was copied, -1 when the
216 * value was empty (a malformed member such as `{"k":}`).
217 *
218 * WHY: Copying verbatim is the point: 5242880 stays an integer and 0.02 stays
219 * 0.02, where reprojecting through a numeric parse and %g would not
220 * round-trip. An empty value is a parse error, not an empty string.
221 *
222 * HOW: 1. Copy bytes until the member terminator, truncating at `outsz` but
223 * still consuming the whole token so the caller's cursor stays sane.
224 * 2. Trim trailing space/tab left by the separator.
225 * 3. NUL-terminate and report whether anything survived.
226 */
227static int
228json_copy_bare(const char **pp, char *out, size_t outsz)
229{
230 const char *p = *pp;
231 size_t o = 0;
232
233 while (*p != '\0' && *p != ',' && *p != '}') {
234 if (o + 1 < outsz) { out[o++] = *p; }
235 p++;
236 }
237 while (o > 0 && (out[o - 1] == ' ' || out[o - 1] == '\t')) { o--; }
238
239 out[o] = '\0';
240 *pp = p;
241 return o > 0 ? 1 : -1;
242}

245/* ---- Decode the value of the member the cursor is sitting on ----
246 *
247 * WHAT: Returns 1 with `out` filled, or -1 when the value is malformed.
248 *
249 * WHY: A caller that has just matched a key should not have to know whether
250 * the value is quoted — that is the only difference between the two
251 * paths, and it belongs here.
252 *
253 * HOW: 1. A leading quote means a string: decode escapes.
254 * 2. Anything else is a bare token: copy it verbatim.
255 */
256static int
257json_take_member_value(const char **pp, char *out, size_t outsz)
258{
259 if (**pp == '"') {
260 return json_parse_string(pp, out, outsz) == 0 ? 1 : -1;
261 }
262
263 return json_copy_bare(pp, out, outsz);
264}

267/* ---- Step over an unwanted member ----
268 *
269 * WHAT: Skips the value at `*pp` and the separator after it. Returns 1 when
270 * another member follows, 0 at the end of the object, -1 on malformed
271 * input.
272 *
273 * WHY: Gives the search loop a single tri-state step, so "key absent" (0) and
274 * "object broken" (-1) stay distinguishable all the way out to the
275 * caller instead of collapsing into one failure.
276 *
277 * HOW: 1. Skip the value (nested objects/arrays included).
278 * 2. Classify the next non-space byte: ',' continues, '}' ends, anything
279 * else is malformed.
280 */
281static int
282json_advance_member(const char **pp)
283{
284 if (json_skip_value(pp) != 0) {
285 return -1;
286 }
287
288 *pp = json_skip_ws(*pp);
289 if (**pp == ',') { (*pp)++; return 1; }
290 if (**pp == '}') { return 0; }
291
292 return -1;
293}

296/* ---- Walk an object to a member's value ----
297 *
298 * WHAT: Leaves *val on the value of the member named `key`. Returns 1 when
299 * found, 0 if the key is absent, -1 if the object is malformed.
300 *
301 * HOW: walk the members once; the tri-state step is what keeps "absent" (0)
302 * and "malformed" (-1) distinct all the way to the caller.
303 */
304static int
305json_find_member(const char *json, const char *key, const char **val)
306{
307 const char *p = json_skip_ws(json);
308
309 if (*p != '{') {
310 return -1;
311 }
312 p = json_skip_ws(p + 1);
313 if (*p == '}') {
314 return 0; /* empty object */
315 }
316
317 for (;;) {
318 char kbuf[64];
319
320 if (json_member_key(&p, kbuf, sizeof kbuf) != 0) {
321 return -1;
322 }
323 if (strcmp(kbuf, key) == 0) {
324 *val = p;
325 return 1;
326 }
327
328 int rc = json_advance_member(&p);
329 if (rc <= 0) {
330 return rc; /* 0 = key absent, -1 = malformed */
331 }
332 }
333}

336/* String values only: 1 = found and decoded into `out`, 0 = key absent,
337 * -1 = malformed object or a non-string value. */
338static int
339json_object_get(const char *json, const char *key, char *out, size_t outsz)
340{
341 const char *p;
342 int rc = json_find_member(json, key, &p);
343
344 if (rc != 1) {
345 return rc;
346 }
347 if (*p != '"') {
348 return -1; /* we only accept string values */

369/*
370 * Reproject a flat JSON command object onto a verb line for apply_command().
371 * Emits "<cmd>" (no args) or "<cmd> <args>". Returns 0 on success, -1 on a
372 * malformed object or a missing "cmd".
373 */
374int
375fp_json_to_verb(const char *json, char *out, size_t outsz)
376{
377 char cmd[64] = "";
378 char args[1900] = "";
379
380 if (json_object_get(json, "cmd", cmd, sizeof cmd) != 1 || cmd[0] == '\0') {
381 return -1; /* "cmd" is required */
382 }
383 if (json_object_get(json, "args", args, sizeof args) < 0) {
384 return -1; /* present but malformed */
385 }
386
387 int n;
388 if (args[0] != '\0') {
389 n = snprintf(out, outsz, "%s %s", cmd, args);
390 } else {
391 n = snprintf(out, outsz, "%s", cmd);
392 }
393 if (n < 0 || (size_t) n >= outsz) {
394 return -1;
395 }
396 return 0;
397}
