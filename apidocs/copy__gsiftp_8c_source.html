1/*
2 * copy_gsiftp.c — gsiftp:// / ftp:// single-file transfers for brix_copy().
3 *
4 * WHAT: route one copy whose source or destination is a GridFTP URL: remote →
5 * local (RETR) and local → remote (STOR), with the local side going through
6 * the client VFS so a download commits atomically like every other scheme.
7 * WHY: GridFTP is still the transfer protocol of record at a large part of the
8 * WLCG estate, and until now xrdcp could not speak it at all — a gsiftp://
9 * argument fell through to the root:// URL parser and died as a usage error.
10 * HOW: brix_copy() intercepts the scheme before its root:// parse and calls
11 * copy_gsiftp(); the direction is decided here, the session engine
12 * (protocols/ftp) does the protocol work, and the payload moves through
13 * sink/source adapters over brix_vfs. Remote→remote and recursive copies are
14 * refused explicitly rather than half-supported.
15 */
16#include "copy_internal.h"
17
18#include "protocols/ftp/ftp_client.h"

20/* Sequential adapter over one VFS handle: the FTP data channel is a stream, so
21 * the engine hands over bytes in order and the offset is kept here. */
22typedef struct {
23 brix_vfs_file *vf;
24 int64_t off;
25} gftp_local_io;
26
27static int
28gftp_sink_local(void *ctx, const uint8_t *buf, size_t n, brix_status *st)
29{
30 gftp_local_io *lc = (gftp_local_io *) ctx;
31
32 if (brix_vfs_pwrite(lc->vf, lc->off, buf, n, st) != 0) {
33 return -1;
34 }
35 lc->off += (int64_t) n;
36 return 0;
37}

40static ssize_t
41gftp_src_local(void *ctx, uint8_t *buf, size_t cap, brix_status *st)
42{
43 gftp_local_io *lc = (gftp_local_io *) ctx;
44 ssize_t n = brix_vfs_pread(lc->vf, lc->off, buf, cap, st);
45
46 if (n > 0) {
47 lc->off += n;
48 }
49 return n;
50}

63/* Open a session for `u`, allocated on the heap: the session carries tens of
64 * kilobytes of control buffers and must not sit on a caller's stack. */
65static brix_ftp_sess *
66gftp_session(const brix_ftpurl *u, const brix_opts *co, brix_status *st)
67{
68 brix_ftp_sess *s = calloc(1, sizeof(*s));
69
70 if (s == NULL) {
71 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
72 return NULL;
73 }
74 if (brix_ftp_session_open(s, u, co, st) != 0) {
75 free(s);
76 return NULL;
77 }
78 return s;
79}

90/* "…/dir/" or an existing local directory takes the source's basename. */
91static void
92gftp_join_basename(const char *srcpath, const char *dst, char *out, size_t sz)
93{
94 const char *base = strrchr(srcpath, '/');
95 struct stat sb;
96 size_t dlen = strlen(dst);
97
98 base = (base != NULL) ? base + 1 : srcpath;
99 if (dlen > 0 && dst[dlen - 1] == '/') {
100 snprintf(out, sz, "%s%s", dst, base);
101 return;
102 }
103 if (stat(dst, &sb) == 0 && S_ISDIR(sb.st_mode)) {
104 snprintf(out, sz, "%s/%s", dst, base);
105 return;
106 }
107 snprintf(out, sz, "%s", dst);
108}

111static int
112gftp_download(const brix_ftpurl *su, const char *dst, const brix_copy_opts *o,
113 const brix_opts *co, brix_status *st)
114{
115 brix_ftp_sess *s;
116 brix_vfs_file *vf = NULL;
117 brix_vfs_open_opts vopts;
118 gftp_local_io lc;
119 char target[XRDC_PATH_MAX];
120 int64_t size = -1, mtime = -1;
121 int rc;
122
123 s = gftp_session(su, co, st);
124 if (s == NULL) {
125 return -1;
126 }
127 if (brix_ftp_stat(s, su->path, &size, &mtime, st) != 0) {
128 gftp_session_free(s);
129 return -1;
130 }
131 gftp_join_basename(su->path, dst, target, sizeof(target));
132 if (o != NULL && o->dry_run) {
133 if (!o->silent) {
134 fprintf(stderr, "xrdcp: (dry run) would copy %s -> %s\n", su->path,
135 target);
136 }
137 gftp_session_free(s);
138 return 0;
139 }
140
141 gftp_vfs_opts(o, size, &vopts);
142 if (brix_vfs_open(target, XRDC_VFS_WRITE | XRDC_VFS_FORCE, &vopts, &vf, st)
143 != 0) {
144 gftp_session_free(s);
145 return -1;
146 }
147 lc.vf = vf;
148 lc.off = 0;
149 rc = brix_ftp_retr(s, su->path, gftp_sink_local, &lc, o, size, st);
150 if (rc == 0 && size >= 0 && lc.off != size) {
151 brix_status_set(st, XRDC_EIO, 0,
152 "gsiftp: short transfer (%lld of %lld bytes)",
153 (long long) lc.off, (long long) size);
154 rc = -1;
155 }
156 if (rc == 0) {
157 rc = brix_vfs_commit(vf, st);
158 } else {
159 brix_vfs_abort(vf);
160 }
161 brix_vfs_close(vf);
162 gftp_session_free(s);
163 return rc;
164}

167static int
168gftp_upload(const char *src, const brix_ftpurl *du, const brix_copy_opts *o,
169 const brix_opts *co, brix_status *st)
170{
171 brix_ftp_sess *s;
172 brix_vfs_file *vf = NULL;
173 brix_vfs_open_opts vopts;
174 brix_vfs_stat vst;
175 brix_status tmp;
176 gftp_local_io lc;
177 char target[XRDC_PATH_MAX];
178 int64_t total = -1;
179 size_t plen;
180 int rc;
181
182 gftp_vfs_opts(o, -1, &vopts);
183 if (brix_vfs_open(src, XRDC_VFS_READ, &vopts, &vf, st) != 0) {
184 return -1;
185 }
186 brix_status_clear(&tmp);
187 if (brix_vfs_fstat(vf, &vst, &tmp) == 0) {
188 total = vst.size;
189 }
190
191 plen = strlen(du->path);
192 if (plen > 0 && du->path[plen - 1] == '/') {
193 const char *base = strrchr(src, '/');
194
195 snprintf(target, sizeof(target), "%s%s", du->path,
196 (base != NULL) ? base + 1 : src);
197 } else {
198 snprintf(target, sizeof(target), "%s", du->path);
199 }
200
201 if (o != NULL && o->dry_run) {
202 if (!o->silent) {
203 fprintf(stderr, "xrdcp: (dry run) would copy %s -> %s\n", src,
204 target);
205 }
206 brix_vfs_close(vf);
207 return 0;
208 }
209
210 s = gftp_session(du, co, st);
211 if (s == NULL) {
212 brix_vfs_close(vf);
213 return -1;
214 }
215 lc.vf = vf;
216 lc.off = 0;
217 rc = brix_ftp_stor(s, target, gftp_src_local, &lc, o, total, st);
218 gftp_session_free(s);
219 brix_vfs_close(vf);
220 return rc;
221}

224/*
225 * copy_gsiftp — the scheme's entry point from brix_copy().
226 *
227 * Exactly one endpoint may be a GridFTP URL: the client has no third-party-copy
228 * verb wired for this scheme, so a remote→remote request is a usage error rather
229 * than a silent double-hop through the local disk.
230 */
231int
232copy_gsiftp(const char *src, const char *dst, const brix_copy_opts *o,
233 const brix_opts *co, brix_status *st)
234{
235 brix_ftpurl u;
236 int src_ftp = brix_is_ftp_url(src);
237 int dst_ftp = brix_is_ftp_url(dst);
238
239 if (src_ftp && dst_ftp) {
240 brix_status_set(st, XRDC_EUSAGE, 0,
241 "gsiftp: third-party copy between GridFTP endpoints is "
242 "not supported");
243 return -1;
244 }
245 if ((src_ftp && brix_is_web_url(dst)) || (dst_ftp && brix_is_web_url(src))) {
246 brix_status_set(st, XRDC_EUSAGE, 0,
247 "gsiftp: copy between gsiftp:// and http/dav/s3 "
248 "endpoints is not supported");
249 return -1;
250 }
251 if (o != NULL && o->recursive) {
252 brix_status_set(st, XRDC_EUSAGE, 0,
253 "gsiftp: recursive copy is not supported");
254 return -1;
255 }
256 if (src_ftp) {
257 if (brix_ftpurl_parse(src, &u) != 0) {
258 brix_status_set(st, XRDC_EUSAGE, 0, "gsiftp: bad URL: %s", src);
259 return -1;
260 }
261 return gftp_download(&u, dst, o, co, st);
262 }
263 if (brix_ftpurl_parse(dst, &u) != 0) {
264 brix_status_set(st, XRDC_EUSAGE, 0, "gsiftp: bad URL: %s", dst);
265 return -1;
266 }
267 return gftp_upload(src, &u, o, co, st);
268}
