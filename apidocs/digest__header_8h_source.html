4/*
5 * digest_header.h — the RFC-3230 `Digest:` header grammar, in both directions.
6 *
7 * One header carries a comma-separated list of `token=value` pairs, where the
8 * token names a checksum algorithm and the encoding of the value depends on it:
9 * md5/sha-* are base64 (RFC 3230 + RFC 1864), adler32/crc32* are lowercase hex
10 * (the WLCG/dCache convention XRootD and dCache interoperate on). This module
11 * owns that grammar so the two directions cannot drift:
12 *
13 * inbound — a WebDAV PUT asserting what the client sent (put_body_digest.c)
14 * outbound — a `Want-Digest:` request to an HTTP origin, and the `Digest:`
15 * it answers with (the sd_http checksum-offload slot)
16 *
17 * Pool-free by design (a storage driver runs on AIO threads with no request
18 * pool), so every value is normalised through caller-provided buffers.
19 */
20
21#include <ngx_config.h>
22#include <ngx_core.h>

24/* Longest value the grammar yields: sha-512 = 128 hex chars + NUL. */
25#define BRIX_DIGEST_HEX_MAX 129

27typedef enum {
28 BRIX_DIGEST_NONE = 0, /* no algorithm we support was named */

30 BRIX_DIGEST_BAD /* supported algorithm named, value unusable */
31} brix_digest_kind_t;

33/*
34 * brix_digest_wire_token — the RFC-3230 token that names a canonical algorithm.
35 *
36 * WHAT: Maps a canonical brix algorithm name ("sha256", "adler32") to the token
37 * an origin expects in `Want-Digest:` ("sha-256", "adler32"); NULL when the
38 * algorithm has no registered token (the CRC-64 family).
39 * WHY: Canonical names and wire tokens differ exactly where RFC 3230 hyphenates
40 * the SHA family; asking an origin for "sha256" gets a shrug from a
41 * compliant one.
42 */
43const char *brix_digest_wire_token(const char *canon_alg);

45/*
46 * brix_digest_header_scan — find a usable digest in a `Digest:` header value.
47 *
48 * WHAT: Walks the comma-separated `token=value` pairs. With want_canon NULL the
49 * FIRST pair naming a supported algorithm decides; with want_canon set,
50 * only a pair naming exactly that canonical algorithm does (others are
51 * skipped, so an origin listing several digests still answers). Returns
52 * FOUND with hex_out holding the lowercase-hex value (and *alg_out, when
53 * non-NULL, pointing at the static canonical name), BAD when a supported
54 * algorithm carries an unusable value, or NONE.
55 * WHY: Both directions need "is the digest I can use in here, and what is it in
56 * hex" — the encoding split (base64 vs hex) is per-algorithm and is the
57 * part everyone gets wrong.
58 */
59brix_digest_kind_t brix_digest_header_scan(const u_char *val, size_t len,
60 const char *want_canon, const char **alg_out, char *hex_out, size_t hex_sz);

62/*
63 * brix_digest_value_hex — normalise one digest value into lowercase hex.
64 *
65 * WHAT: A base64 value (is_b64) is decoded then hex-encoded; a hex value is
66 * validated and lowercased. NGX_ERROR on an empty, malformed or over-long

73/*
74 * brix_digest_hex_pad — left-pad a hex value to its algorithm's fixed width.
75 *
76 * WHAT: Zero-extends `hex` in place to the canonical width of `canon_alg`
77 * (adler32/crc32* → 8, md5 → 32, sha-1 → 40, sha-256 → 64, sha-512 → 128);
78 * a no-op for an unknown algorithm or a value already at least that long.
79 * WHY: An origin may drop leading zeros from an adler32 ("1a2b3c"), but a
80 * checksum we hand back as authoritative is compared literally by clients
81 * against a zero-padded computation.
82 */
83void brix_digest_hex_pad(const char *canon_alg, char *hex, size_t hex_sz);
