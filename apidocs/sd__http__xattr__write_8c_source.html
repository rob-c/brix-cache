1/*
2 * sd_http_xattr_write.c — writing extended attributes on an http origin, as
3 * WebDAV dead properties.
4 *
5 * WHAT: The setxattr/removexattr vtable slots and their credential-scoped twins,
6 * over PROPPATCH (RFC 4918 §9.2).
7 *
8 * WHY: Split from the reader (sd_http_xattr.c) because one file carrying both
9 * halves passes the 600-line cap — not because they are two mappings. The
10 * shared wire spelling and the three helpers both halves need live in
11 * sd_http_xattr_internal.h.
12 *
13 * HOW: One PROPPATCH per mutation, sent through the driver's no-failover
14 * namespace sender: replaying a property write against a second endpoint
15 * could apply it twice. Two POSIX contracts that PROPPATCH does not have
16 * natively are bought with a preceding size-enquiry read — XATTR_CREATE /
17 * XATTR_REPLACE (PROPPATCH is an unconditional upsert) and "removing an
18 * absent attribute is ENODATA" (RFC 4918 §9.2 makes it a success).
19 */

32/*
33 * WHAT: The PROPPATCH status→errno verdict.
34 * WHY: A property write fails differently from a namespace mutation: 405/409
35 * mean "this origin does not keep dead properties here", which the VFS
36 * must see as ENOTSUP so it can fall back, not as EEXIST/ENOENT.
37 * HOW: Override the three codes that differ, delegate the rest to the shared
38 * mutation map so 401/403/404 keep one spelling across the driver.
39 */
40static int
41sd_http_xattr_errno(long status)
42{
43 switch (status) {
44 case 405:
45 case 409: return ENOTSUP;
46 case 507: return ENOSPC;
47 default: return sd_http_status_to_errno(status);
48 }
49}

52/*
53 * WHAT: Send one PROPPATCH body and judge the reply.
54 * WHY: set and remove differ only in the body; the credential path, the header
55 * block and the two-level verdict (transport status, then the per-property
56 * propstat status) belong to both.
57 * HOW: The shared no-failover namespace sender — replaying a property write
58 * against a second endpoint could apply it twice — then 200/204 accepted
59 * outright and a 207 accepted only if its propstat says 2xx.
60 */
61static ngx_int_t
62sd_http_xattr_patch(sd_http_inst_state *is, const char *path,
63 const char *body, size_t blen, const brix_sd_cred_t *cred)
64{
65 char open_auth[SD_HTTP_AUTH_MAX];
66 char hdrs[SD_HTTP_AUTH_MAX + 96];
67 char full[SD_HTTP_PATH_MAX];
68 const char *auth, *cert;
69 brix_s3_resp_t resp;
70 sd_http_ns_req_t nsr = { "PROPPATCH", NULL, hdrs, NULL, body, blen };
71 char errbuf[256];
72 const void *rbody;
73 size_t rlen = 0;
74 int ok;
75
76 if (sd_http_xattr_auth(is, cred, open_auth, sizeof(open_auth),
77 &auth, &cert) != 0)
78 {
79 return NGX_ERROR;
80 }
81 snprintf(hdrs, sizeof(hdrs), "Content-Type: application/xml\r\n%s",
82 (auth != NULL) ? auth : "");
83 sd_http_write_path(is, path, full, sizeof(full));
84 nsr.path = full;
85 nsr.cert_pem = (cert != NULL && cert[0] != '\0') ? cert : NULL;
86
87 if (sd_http_ns_send(is, &nsr, &resp, errbuf, sizeof(errbuf)) != 0) {
88 errno = EIO;
89 return NGX_ERROR;
90 }
91 if (resp.status != 200 && resp.status != 204 && resp.status != 207) {
92 errno = sd_http_xattr_errno(resp.status);
93 is->transport->resp_free(&resp);
94 return NGX_ERROR;
95 }
96 ok = 1;
97 if (resp.status == 207) {
98 rbody = is->transport->resp_body(&resp, &rlen);
99 ok = (rbody != NULL && rlen != 0 && rlen <= SD_HTTP_XATTR_XML_MAX)
100 && sd_http_xattr_status_ok((const char *) rbody,
101 (const char *) rbody + rlen);
102 }
103 is->transport->resp_free(&resp);
104 if (!ok) {
105 errno = ENOTSUP; /* the origin declined to keep this property */
106 return NGX_ERROR;
107 }
108 return NGX_OK;
109}

112/*
113 * WHAT: Enforce XATTR_CREATE / XATTR_REPLACE before a write.
114 * WHY: PROPPATCH has no conditional set — it is an upsert — so the exclusivity
115 * the POSIX flags promise has to be asked for separately or it is silently
116 * not enforced, which is worse than not offering the flags at all.
117 * HOW: One size-enquiry read; EEXIST when CREATE finds a value, ENODATA when
118 * REPLACE finds none, and no probe at all when neither flag is set.
119 * Inherently racy against a concurrent writer on the same origin, exactly
120 * as it is on any network filesystem; the flags are advisory there too.
121 */
122static int
123sd_http_xattr_flag_gate(brix_sd_instance_t *inst, const char *path,
124 const char *name, int flags, const brix_sd_cred_t *cred)
125{
126 ssize_t have;
127
128 if ((flags & (XATTR_CREATE | XATTR_REPLACE)) == 0) {
129 return 0;
130 }
131 have = sd_http_getxattr_common(inst, path, name, NULL, 0, cred);
132 if (have >= 0 && (flags & XATTR_CREATE) != 0) {
133 errno = EEXIST;
134 return -1;
135 }
136 if (have < 0 && errno == ENODATA && (flags & XATTR_REPLACE) != 0) {
137 return -1; /* errno already ENODATA */
138 }
139 if (have < 0 && errno != ENODATA) {
140 return -1; /* the probe itself failed */
141 }
142 return 0;
143}

146static ngx_int_t
147sd_http_setxattr_common(brix_sd_instance_t *inst, const char *path,
148 const char *name, const void *value, size_t len, int flags,
149 const brix_sd_cred_t *cred)
150{
151 char elem[SD_HTTP_XATTR_ELEM_MAX];
152 char *body;
153 size_t cap;
154 ngx_int_t rc;
155 int n;
156
157 if (inst == NULL || inst->state == NULL || path == NULL
158 || (value == NULL && len != 0))
159 {
160 errno = EINVAL;
161 return NGX_ERROR;
162 }
163 if (len > SD_HTTP_XATTR_VALUE_MAX) {
164 errno = E2BIG;
165 return NGX_ERROR;
166 }
167 if (sd_http_xattr_elem(name, elem, sizeof(elem)) != 0
168 || sd_http_xattr_flag_gate(inst, path, name, flags, cred) != 0)
169 {
170 return NGX_ERROR;
171 }
172 cap = sizeof(elem) * 2 + 2 * len + 256;
173 body = malloc(cap);
174 if (body == NULL) {
175 errno = ENOMEM;
176 return NGX_ERROR;
177 }
178 /* The hex of the value is written straight into the body buffer, so the
179 * only bytes between the tags are [0-9a-f] whatever the caller stored. */
180 n = snprintf(body, cap,
181 "<?xml version=\"1.0\" encoding=\"utf-8\"?>"
182 "<D:propertyupdate xmlns:D=\"DAV:\" xmlns:B=\"%s\">"
183 "<D:set><D:prop><B:%s>", SD_HTTP_XATTR_NS, elem);
184 brix_hex_encode((const uint8_t *) value, len, body + n);
185 n += (int) (2 * len);
186 n += snprintf(body + n, cap - (size_t) n,
187 "</B:%s></D:prop></D:set></D:propertyupdate>", elem);
188
189 rc = sd_http_xattr_patch(inst->state, path, body, (size_t) n, cred);
190 free(body);
191 return rc;
192}

195static ngx_int_t
196sd_http_removexattr_common(brix_sd_instance_t *inst, const char *path,
197 const char *name, const brix_sd_cred_t *cred)
198{
199 char elem[SD_HTTP_XATTR_ELEM_MAX];
200 char body[SD_HTTP_XATTR_ELEM_MAX + 256];
201 int n;
202
203 if (inst == NULL || inst->state == NULL || path == NULL) {
204 errno = EINVAL;
205 return NGX_ERROR;
206 }
207 if (sd_http_xattr_elem(name, elem, sizeof(elem)) != 0) {
208 return NGX_ERROR;
209 }
210 /* RFC 4918 §9.2 makes removing an absent property a SUCCESS, where POSIX
211 * demands ENODATA — so ask first rather than report a removal that removed
212 * nothing. */
213 if (sd_http_getxattr_common(inst, path, name, NULL, 0, cred) < 0) {
214 return NGX_ERROR; /* errno = ENODATA, or the probe's */
215 }
216 n = snprintf(body, sizeof(body),
217 "<?xml version=\"1.0\" encoding=\"utf-8\"?>"
218 "<D:propertyupdate xmlns:D=\"DAV:\" xmlns:B=\"%s\">"
219 "<D:remove><D:prop><B:%s/></D:prop></D:remove>"
220 "</D:propertyupdate>", SD_HTTP_XATTR_NS, elem);
221
222 return sd_http_xattr_patch(inst->state, path, body, (size_t) n, cred);
223}
