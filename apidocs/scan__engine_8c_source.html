19/* Render the walk's internal logical path (anchored at "." or a relative
20 * subdir) as a clean, root-relative display path: "." → "/", "./a/b" → "/a/b",
21 * "sub/x" → "/sub/x". The integrity layer keys on the fd, not this string. */
22static void
23scan_display_path(const char *logical, char *out, size_t outsz)
24{
25 const char *p = logical;
26
27 if (p[0] == '.' && p[1] == '/') {
28 p += 2;
29 } else if (p[0] == '.' && p[1] == '\0') {
30 p += 1;
31 }
32 snprintf(out, outsz, "/%s", p);
33}

58/* Append a ready NDJSON line (llen bytes) plus a newline to the growing heap
59 * buffer. Returns 1 ok, -1 OOM (caller owns *buf on failure). Mirrors
60 * brix_ckscan_append's exponential growth. Declared in scan_engine_internal.h
61 * (shared with the catalog-enumeration split). */
62int
63scan_append(u_char **buf, size_t *cap, size_t *used, const char *line, size_t llen)
64{
65 if (*used + llen + 2 > *cap) {
66 size_t new_cap = *cap * 2 + llen + 2;
67 u_char *nb = ngx_alloc(new_cap, ngx_cycle->log);
68
69 if (nb == NULL) {
70 return -1;
71 }
72 if (*buf != NULL) {
73 ngx_memcpy(nb, *buf, *used);
74 ngx_free(*buf);
75 }
76 *buf = nb;
77 *cap = new_cap;
78 }
79 ngx_memcpy(*buf + *used, line, llen);
80 *used += llen;
81 (*buf)[(*used)++] = '\n';
82 return 0;
83}

95/*
96 * WHAT: per-file action context — the walk cookie plus the one file the action
97 * is currently building a record for (its display path, stat facts, and
98 * read-only fd) and the caller's output line buffer.
99 * WHY: every mode action threads the identical (walk-state, this-file, output)
100 * triple; bundling it keeps each action at ≤ 2 params (ctx + info out)
101 * and makes the shared data flow explicit instead of a wide argument list.
102 * HOW: scan_walk_file fills one on the stack per file and hands its address to
103 * the selected action; the action never mutates it, only reads.
104 */
105typedef struct {
106 scan_walk_ctx_t *cc;
107 const char *logical; /* clean display path for this file */
108 off_t size;
109 time_t mtime;
110 int fd; /* read-only fd from the walk */
111 char *line; /* output record buffer */
112 size_t linesz;
113} scan_action_ctx_t;

115/* Look up the stored (cached) checksum without computing. Returns the hex on a
116 * cache hit (into `info`), or NULL on miss. */
117static const char *
118scan_stored(scan_action_ctx_t *a, brix_integrity_info_t *info)
119{
120 brix_integrity_opts_t io;
121
122 ngx_memzero(&io, sizeof(io));
123 io.allow_xattr_cache = 1;
124 io.no_compute = 1;
125 /* phase-105: a lookup, never a persist — the zeroed policy is READ_ONLY and
126 * that is the intended posture here, stated rather than left implicit. */
127 io.mutation_policy = BRIX_VFS_MUTATION_READ_ONLY;
128 io.proto = BRIX_PROTO_WEBDAV; /* the admin scan endpoint is an HTTP plane */
129 if (brix_integrity_get_fd(a->cc->log, a->fd, NULL, a->logical,
130 a->cc->opts->alg, &io, info) == NGX_OK)
131 {
132 return info->hex;
133 }
134 return NULL;
135}

137/* dump / compare: emit stored checksum (or null), never read bytes. */
138static int
139scan_action_dump(scan_action_ctx_t *a)
140{
141 brix_integrity_info_t info;
142 const char *stored = scan_stored(a, &info);
143 int n;
144
145 n = brix_scan_record_file(a->line, a->linesz, a->logical, (int64_t) a->size,
146 (int64_t) a->mtime, a->cc->opts->alg, stored,
147 NULL, "ok");
148 a->cc->sum->ok++;
149 return n;
150}

152/* verify: recompute and compare to stored. */
153static int
154scan_action_verify(scan_action_ctx_t *a)
155{
156 brix_integrity_info_t stored_info, comp_info;
157 const char *stored;
158 brix_integrity_opts_t io;
159 const char *status;
160 const char *computed = NULL;
161 char stored_copy[129];
162
163 stored = scan_stored(a, &stored_info);
164 if (stored != NULL) {
165 ngx_memcpy(stored_copy, stored, ngx_strlen(stored) + 1);
166 stored = stored_copy; /* survive the second get_fd's info reuse */
167 }
168
169 ngx_memzero(&io, sizeof(io));
170 io.allow_xattr_cache = 0; /* force a fresh compute over the bytes */
171 io.no_compute = 0;
172 /* phase-105: verify reports, it never repairs — READ_ONLY so a mismatch
173 * cannot silently overwrite the stored digest it just disagreed with. */
174 io.mutation_policy = BRIX_VFS_MUTATION_READ_ONLY;
175 io.proto = BRIX_PROTO_WEBDAV;
176 if (brix_integrity_get_fd(a->cc->log, a->fd, NULL, a->logical,
177 a->cc->opts->alg, &io, &comp_info) != NGX_OK)
178 {
179 status = "unreadable";
180 a->cc->sum->unreadable++;
181 } else {
182 computed = comp_info.hex;
183 a->cc->sum->bytes += (uint64_t) a->size;
184 if (stored == NULL) {
185 status = "missing";
186 a->cc->sum->missing++;
187 } else if (strcasecmp(stored, computed) == 0) {
188 status = "ok";
189 a->cc->sum->ok++;
190 } else {
191 status = "mismatch";
192 a->cc->sum->mismatch++;
193 }
194 }
195 return brix_scan_record_file(a->line, a->linesz, a->logical,
196 (int64_t) a->size, (int64_t) a->mtime,
197 a->cc->opts->alg, stored, computed, status);
198}

200/* fill: persist a checksum only when none is stored. */
201static int
202scan_action_fill(scan_action_ctx_t *a)
203{
204 brix_integrity_info_t info;
205 brix_integrity_opts_t io;
206 const char *status;
207 const char *stored;
208
209 stored = scan_stored(a, &info);
210 if (stored != NULL) {
211 status = "already";
212 a->cc->sum->already++;
213 return brix_scan_record_file(a->line, a->linesz, a->logical,
214 (int64_t) a->size, (int64_t) a->mtime,
215 a->cc->opts->alg, stored, NULL, status);
216 }
217
218 ngx_memzero(&io, sizeof(io));
219 io.allow_xattr_cache = 1; /* required alongside update for the persist path */
220 io.update_xattr_cache = 1;
221 io.no_compute = 0;
222 /* phase-105: `fill` IS the writable maintenance path Appendix H.2 reserves —
223 * an admin-authenticated, operator-invoked backfill over the separately
224 * configured brix_dashboard_scan_root, not a client request against an
225 * export. It is the only scan mode that may mutate, and it says so here. */
226 io.mutation_policy = BRIX_VFS_MUTATION_ALLOWED;
227 io.proto = BRIX_PROTO_WEBDAV;
228 if (brix_integrity_get_fd(a->cc->log, a->fd, NULL, a->logical,
229 a->cc->opts->alg, &io, &info) != NGX_OK)
230 {
231 status = "unreadable";
232 stored = NULL;
233 a->cc->sum->unreadable++;
234 } else {
235 status = "filled";
236 stored = info.hex;
237 a->cc->sum->filled++;
238 a->cc->sum->bytes += (uint64_t) a->size;
239 }
240 return brix_scan_record_file(a->line, a->linesz, a->logical,
241 (int64_t) a->size, (int64_t) a->mtime,
242 a->cc->opts->alg, stored, NULL, status);
243}

245/* inspect (A2): per-file backend introspection. The scan endpoint walks the
246 * export root via POSIX openat2 (not the SD-driver seam), so the backend is
247 * "posix" and namespace==backend by construction; the driver-bound view
248 * (Ceph object key, cluster facts) is a Phase-4 prerequisite. */
249static int
250scan_action_inspect(scan_action_ctx_t *a)
251{
252 brix_integrity_info_t info;
253 const char *stored = scan_stored(a, &info);
254
255 a->cc->sum->ok++;
256 return brix_scan_record_inspect(a->line, a->linesz, a->logical, "posix",
257 (int64_t) a->size, (int64_t) a->mtime,
258 stored != NULL ? "xattr" : "none",
259 1 /* posix: namespace == backend */);
260}

262/* inventory (E1): one "object" record per stored object. The scan endpoint walks
263 * the export via POSIX (namespace == catalog), so the logical path IS the backend
264 * key and every entry is backed (orphan=false). A catalog-native backend's
265 * inventory runs through brix_vfs_enumerate_catalog (the SD enumerate verb)
266 * instead — the seam is in place; threading a bound instance into the endpoint is
267 * the Ceph follow-on. */
268static int
269scan_action_inventory(scan_action_ctx_t *a)
270{
271 a->cc->sum->ok++;
272 return brix_scan_record_object(a->line, a->linesz, a->logical, a->logical,
273 (int64_t) a->size, (int64_t) a->mtime, 0);
274}

276/* drift (D2): one "drift" record per entry. Over a namespace walk the catalog and
277 * namespace coincide, so every entry is "in_both"; orphan_object / namespace_only
278 * arise only when a native catalog verb (brix_vfs_enumerate_catalog) supplies a
279 * backend-object set to reconcile against (scan_drift) — the Ceph follow-on. */
280static int
281scan_action_drift(scan_action_ctx_t *a)

283 a->cc->sum->ok++;

288/* Run the mode's per-file action against the assembled action context. Split out
289 * so scan_walk_file stays a thin fill-and-append shell around the dispatch. */
290static int
291scan_dispatch_action(scan_action_ctx_t *a)
292{
293 switch (a->cc->opts->mode) {
294 case BRIX_SCAN_VERIFY:
295 return scan_action_verify(a);
296 case BRIX_SCAN_FILL:
297 return scan_action_fill(a);
298 case BRIX_SCAN_INSPECT:
299 return scan_action_inspect(a);
300 case BRIX_SCAN_INVENTORY:
301 return scan_action_inventory(a);
302 case BRIX_SCAN_DRIFT:
303 return scan_action_drift(a);
304 case BRIX_SCAN_DUMP:
305 case BRIX_SCAN_COMPARE:
306 default:
307 return scan_action_dump(a);
308 }
309}

311static ngx_int_t
312scan_walk_file(void *cookie, const char *logical, const brix_vfs_stat_t *st,
313 int fd)
314{
315 scan_walk_ctx_t *cc = cookie;
316 scan_action_ctx_t a;
317 char line[SCAN_LINE_MAX];
318 char disp[PATH_MAX];
319 int n;
320
321 scan_display_path(logical, disp, sizeof(disp));
322
323 ngx_memzero(&a, sizeof(a));
324 a.cc = cc;
325 a.logical = disp;
326 a.size = st->size;
327 a.mtime = st->mtime;
328 a.fd = fd;
329 a.line = line;
330 a.linesz = sizeof(line);
331
332 n = scan_dispatch_action(&a);
333
334 if (n < 0) {
335 return NGX_OK; /* path too long to format — soft skip */
336 }
337 cc->sum->files++;
338 if (scan_append(cc->buf, cc->cap, cc->used, line, (size_t) n) < 0) {
339 cc->oom = 1;
340 return NGX_ERROR;
341 }
342 return NGX_OK;
343}

345ngx_int_t
346brix_scan_run(ngx_log_t *log, int rootfd, const char *logical,
347 const brix_scan_opts_t *opts, u_char **buf, size_t *cap, size_t *used,
348 brix_scan_summary_t *summary, uint16_t *err_code, char *err_msg,
349 size_t err_sz)
350{
351 scan_walk_ctx_t cc;
352 brix_vfs_walk_opts_t wopts;
353 brix_vfs_walk_target_t target = BRIX_VFS_WALK_NONE;
354 char walkmsg[128] = "";
355 ngx_int_t rc;
356
357 ngx_memzero(&cc, sizeof(cc));
358 cc.log = log;
359 cc.opts = opts;
360 cc.buf = buf;
361 cc.cap = cap;
362 cc.used = used;
363 cc.sum = summary;
364
365 ngx_memzero(&wopts, sizeof(wopts));
366 wopts.max_depth = opts->max_depth;
367 wopts.max_files = opts->max_files;
368 wopts.open_files = 1; /* the walk hands each regular file a read-only fd */
369
370 rc = brix_vfs_walk(log, rootfd, logical, &wopts, scan_walk_file, &cc,
371 &target, walkmsg, sizeof(walkmsg));
372
373 if (rc == NGX_DECLINED) {
374 *err_code = kXR_NotFound;
375 snprintf(err_msg, err_sz, "target not found");
376 return NGX_ERROR;
377 }
378 if (rc != NGX_OK) {
379 if (cc.oom) {
380 *err_code = kXR_NoMemory;
381 snprintf(err_msg, err_sz, "out of memory");
382 } else {
383 *err_code = kXR_IOError;
384 snprintf(err_msg, err_sz, "%s", walkmsg[0] ? walkmsg : "I/O error");
385 }
386 return NGX_ERROR;
387 }
388 if (target == BRIX_VFS_WALK_OTHER) {
389 *err_code = kXR_ArgInvalid;
390 snprintf(err_msg, err_sz, "not a file or directory");
391 return NGX_ERROR;
392 }
393 return NGX_OK;
394}
