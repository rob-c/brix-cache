1/*
2 * ftp_gsi_cred.c — X.509 credential handling for the GridFTP GSI initiator.
3 *
4 * WHAT: base64 codecs for the ADAT/ENC arguments, loading the user's X.509 proxy
5 * (leaf, chain, key) into the initiator's SSL_CTX, and issuing the
6 * delegated proxy the GridFTP server asks for mid-handshake.
7 * WHY: GSI's mandatory delegation round makes the client a certificate *issuer*,
8 * not just a TLS peer; keeping that crypto beside the credential loader —
9 * and away from the handshake driver — means the state machine reads as
10 * protocol and this file reads as PKI.
11 * HOW: the proxy file is opened through the shared hardened reader
12 * (brix_credfile_bio: no symlink, owner-only, 0600) exactly as the root://
13 * GSI driver does, and the request is signed by libxrdproto's
14 * brix_gsi_sign_pxyreq — the same primitive the server side uses, so the
15 * two halves cannot drift.
16 */

61char *
62brix_ftp_b64_encode(const uint8_t *data, size_t len)
63{
64 char *out;
65 size_t cap;
66 int n;
67
68 if (data == NULL || len == 0 || len > (size_t) INT_MAX / 2) {
69 return NULL;
70 }
71 cap = ((len + 2) / 3) * 4 + 1;
72 out = malloc(cap);
73 if (out == NULL) {
74 return NULL;
75 }
76 n = EVP_EncodeBlock((unsigned char *) out, data, (int) len);
77 if (n < 0) {
78 free(out);
79 return NULL;
80 }
81 out[n] = '\0';
82 return out;
83}

86uint8_t *
87brix_ftp_b64_decode(const char *b64, size_t *out_len)
88{
89 uint8_t *out;
90 size_t len, pad = 0;
91 int n;
92
93 if (b64 == NULL || out_len == NULL) {
94 return NULL;
95 }
96 len = strlen(b64);
97 if (len == 0 || (len % 4) != 0) {
98 return NULL;
99 }
100 if (b64[len - 1] == '=') { pad++; }
101 if (len >= 2 && b64[len - 2] == '=') { pad++; }
102
103 out = malloc(len / 4 * 3 + 1);
104 if (out == NULL) {
105 return NULL;
106 }
107 n = EVP_DecodeBlock(out, (const unsigned char *) b64, (int) len);
108 if (n < 0 || (size_t) n < pad) {
109 free(out);
110 return NULL;
111 }
112 /* EVP_DecodeBlock reports the padded length; the '=' bytes decoded to zero
113 * bytes the payload does not contain. */
114 *out_len = (size_t) n - pad;
115 return out;
116}

127/* Install leaf + issuer chain from the proxy file, and keep the certs-only PEM
128 * (the delegation signer identity) on the session. */
129static int
130load_certs(struct brix_ftp_gss *g, const char *proxy, brix_status *st)
131{
132 BIO *in, *mem;
133 X509 *cert;
134 BUF_MEM *bm;
135 int n = 0;
136
137 in = proxy_open(proxy);
138 if (in == NULL) {
139 brix_status_set(st, XRDC_EAUTH, 0,
140 "gsiftp: cannot read X.509 proxy %s", proxy);
141 return -1;
142 }
143 mem = BIO_new(BIO_s_mem());
144 if (mem == NULL) {
145 BIO_free(in);
146 return brix_ftp_gss_ssl_err(st, "BIO_new");
147 }
148 while ((cert = PEM_read_bio_X509(in, NULL, NULL, NULL)) != NULL) {
149 int ok = PEM_write_bio_X509(mem, cert);
150
151 if (ok) {
152 /* leaf first, the rest are issuers (SSL_CTX takes a reference). */
153 ok = (n == 0) ? SSL_CTX_use_certificate(g->ctx, cert)
154 : SSL_CTX_add1_chain_cert(g->ctx, cert);
155 }
156 X509_free(cert);
157 if (!ok) {
158 BIO_free(in);
159 BIO_free(mem);
160 return brix_ftp_gss_ssl_err(st, "install proxy certificate");
161 }
162 n++;
163 }
164 ERR_clear_error(); /* the loop's terminating NULL is a benign PEM EOF */
165 BIO_free(in);
166
167 if (n == 0) {
168 BIO_free(mem);
169 brix_status_set(st, XRDC_EAUTH, 0,
170 "gsiftp: no certificate in X.509 proxy %s", proxy);
171 return -1;
172 }
173 BIO_get_mem_ptr(mem, &bm);
174 g->pem = malloc(bm->length);
175 if (g->pem == NULL) {
176 BIO_free(mem);
177 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
178 return -1;
179 }
180 memcpy(g->pem, bm->data, bm->length);
181 g->pem_len = bm->length;
182 BIO_free(mem);
183 return 0;
184}

187int
188brix_ftp_gss_load_cred(struct brix_ftp_gss *g, const char *proxy,
189 brix_status *st)
190{
191 BIO *in;
192
193 if (load_certs(g, proxy, st) != 0) {
194 return -1;
195 }
196 in = proxy_open(proxy);
197 if (in != NULL) {
198 g->key = PEM_read_bio_PrivateKey(in, NULL, NULL, NULL);
199 BIO_free(in);
200 }
201 ERR_clear_error();
202 if (g->key == NULL) {
203 brix_status_set(st, XRDC_EAUTH, 0,
204 "gsiftp: no private key in X.509 proxy %s", proxy);
205 return -1;
206 }
207 if (SSL_CTX_use_PrivateKey(g->ctx, g->key) != 1) {
208 return brix_ftp_gss_ssl_err(st, "install proxy key");
209 }
210 return 0;
211}

214int
215brix_ftp_gss_sign_csr(struct brix_ftp_gss *g, const uint8_t *csr_der,
216 size_t csr_len, uint8_t **out_der, size_t *out_len,
217 brix_status *st)
218{
219 brix_gsi_blob_t signer = { g->pem, g->pem_len };
220 brix_gsi_blob_t req = { csr_der, csr_len };
221 brix_gsi_buf_t proxy = { NULL, 0 };
222 char errbuf[160];
223 brix_gsi_err_t err = { errbuf, sizeof(errbuf) };
224 X509 *issued;
225 unsigned char *der = NULL;
226 int dlen;
227
228 errbuf[0] = '\0';
229 if (brix_gsi_sign_pxyreq(&signer, g->key, &req, &proxy, &err) != 0) {
230 brix_status_set(st, XRDC_EAUTH, 0,
231 "gsiftp: delegation signing failed: %s", errbuf);
232 return -1;
233 }
234
235 /* The server wants the issued certificate as DER (it d2i_X509()s the first
236 * certificate of the token), so re-encode the PEM the signer returned. */
237 {
238 BIO *mem = BIO_new_mem_buf(proxy.data, (int) proxy.len);
239
240 issued = (mem != NULL) ? PEM_read_bio_X509(mem, NULL, NULL, NULL) : NULL;
241 BIO_free(mem);
242 }
243 free(proxy.data);
244 if (issued == NULL) {
245 return brix_ftp_gss_ssl_err(st, "re-read issued proxy");
246 }
247 dlen = i2d_X509(issued, &der);
248 X509_free(issued);
249 if (dlen <= 0 || der == NULL) {
250 return brix_ftp_gss_ssl_err(st, "encode issued proxy");
251 }
252 *out_der = malloc((size_t) dlen);
253 if (*out_der == NULL) {
254 OPENSSL_free(der);
255 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
256 return -1;
257 }
258 memcpy(*out_der, der, (size_t) dlen);
259 *out_len = (size_t) dlen;
260 OPENSSL_free(der);
261 return 0;
262}
