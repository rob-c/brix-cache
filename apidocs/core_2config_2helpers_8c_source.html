8/* Validate a configured path at startup: existence (stat), type
9 * (file/dir/either), and access mode (R/W/X_OK). A NULL/empty path is
10 * optional and skipped. Returns NGX_OK, or NGX_ERROR with an emerg log
11 * naming the failure. */
12ngx_int_t
13brix_validate_path(ngx_conf_t *cf, const char *label, const ngx_str_t *path,
14 brix_path_kind_t kind, int access_mode)
15{
16 struct stat st;
17
18 if (path == NULL || path->len == 0 || path->data == NULL) {
19 return NGX_OK;
20 }
21
22 if (stat((char *) path->data, &st) != 0) {
23 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
24 "brix: %s path \"%s\" is not accessible",
25 label, path->data);
26 return NGX_ERROR;
27 }
28
29 switch (kind) {
30 case BRIX_PATH_REGULAR_FILE:
31 if (!S_ISREG(st.st_mode)) {
32 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
33 "brix: %s path \"%s\" must be a regular file",
34 label, path->data);
35 return NGX_ERROR;
36 }
37 break;
38
39 case BRIX_PATH_DIRECTORY:
40 if (!S_ISDIR(st.st_mode)) {
41 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
42 "brix: %s path \"%s\" must be a directory",
43 label, path->data);
44 return NGX_ERROR;
45 }
46 break;
47
48 case BRIX_PATH_FILE_OR_DIRECTORY:
49 if (!S_ISREG(st.st_mode) && !S_ISDIR(st.st_mode)) {
50 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
51 "brix: %s path \"%s\" must be a file or directory",
52 label, path->data);
53 return NGX_ERROR;
54 }
55 break;
56 }
57
58 if (access_mode != 0 && access((char *) path->data, access_mode) != 0) {
59 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
60 "brix: %s path \"%s\" failed permission check",
61 label, path->data);
62 return NGX_ERROR;
63 }
64
65 return NGX_OK;
66}

68/* Copy an ngx_str_t into a NUL-terminated C string in cf->pool (ngx_str_t is
69 * not NUL-terminated, so strtol/strchr/etc. need this). Returns NGX_CONF_OK,
70 * or NGX_CONF_ERROR on allocation failure. */
71char *
72brix_copy_conf_string(ngx_conf_t *cf, const ngx_str_t *src, ngx_str_t *dst)
73{
74 dst->data = ngx_pnalloc(cf->pool, src->len + 1);
75 if (dst->data == NULL) {
76 return NGX_CONF_ERROR;
77 }
78
79 ngx_memcpy(dst->data, src->data, src->len);
80 dst->data[src->len] = '\0';
81 dst->len = src->len;
82 return NGX_CONF_OK;
83}

85/* Resolve a host:port directive argument into a pool-allocated ngx_addr_t
86 * (first resolved address; the port must be explicit — no default). Shared
87 * by the handoff/relay upstream directives. Returns NULL on failure; a
88 * resolve failure is logged with the directive's name. */
89ngx_addr_t *
90brix_conf_parse_addr(ngx_conf_t *cf, ngx_str_t *spec, const char *directive)
91{
92 ngx_url_t url;
93 ngx_addr_t *addr;
94
95 ngx_memzero(&url, sizeof(url));
96 url.url = *spec;
97 url.default_port = 0;
98
99 if (ngx_parse_url(cf->pool, &url) != NGX_OK || url.no_port
100 || url.naddrs == 0 || url.addrs == NULL)
101 {
102 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
103 "%s: could not resolve host:port in \"%V\"", directive, spec);
104 return NULL;
105 }
106
107 addr = ngx_pcalloc(cf->pool, sizeof(ngx_addr_t));
108 if (addr == NULL) {
109 return NULL;
110 }
111 addr->sockaddr = ngx_pnalloc(cf->pool, url.addrs[0].socklen);
112 if (addr->sockaddr == NULL) {
113 return NULL;
114 }
115 ngx_memcpy(addr->sockaddr, url.addrs[0].sockaddr, url.addrs[0].socklen);
116 addr->socklen = url.addrs[0].socklen;
117 addr->name = url.addrs[0].name;
118 return addr;
119}

121/* Whole-directive worker behind the single-argument upstream directives
122 * (`brix_http_handoff host:port`, `brix_transparent_proxy host:port`):
123 * duplicate check on *slot, record the raw spec in *name, resolve it via
124 * brix_conf_parse_addr, and log the configured target. The directive's own
125 * name (cmd->name) labels both messages. */
126char *
127brix_conf_upstream_directive(ngx_conf_t *cf, ngx_command_t *cmd,
128 ngx_str_t *name, ngx_addr_t **slot)
129{
130 ngx_str_t *value = cf->args->elts;
131
132 if (*slot != NULL) {
133 return "is duplicate";
134 }
135
136 *name = value[1];
137 *slot = brix_conf_parse_addr(cf, &value[1], (const char *) cmd->name.data);
138 if (*slot == NULL) {
139 return NGX_CONF_ERROR;
140 }
141
142 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
143 "brix: %V upstream configured: %V", &cmd->name, &value[1]);
144 return NGX_CONF_OK;
145}

147/* brix_conf_set_backend_sss_keytab — `brix_backend_sss_keytab <path>` setter.
148 *
149 * WHAT: Store the identity-injection keytab path (plain str slot) and
150 * load-validate it immediately with the same loader the SSS *auth*
151 * keytabs go through (existence, regular file, private permissions,
152 * parseable keys).
153 * WHY: The delegation gate signs per-caller SSS credentials with this keytab
154 * at request time; a missing/world-readable/garbage keytab must fail
155 * `nginx -t`, not turn into a fleet-wide runtime deny (fail-closed but
156 * diagnosed at config load). Phase-70 §5.6 / P90-70.3.
157 * HOW: ngx_conf_set_str_slot for storage (conf tokens are NUL-terminated, so
158 * the stored bytes are directly usable as a C path at request time),
159 * then brix_sss_load_keytab on the stored value; the parsed key array is
160 * discarded — it lives in cf->pool and only proves loadability. */
161char *
162brix_conf_set_backend_sss_keytab(ngx_conf_t *cf, ngx_command_t *cmd,
163 void *conf)
164{
165 ngx_str_t *path;
166 ngx_array_t *keys;
167 char *rv;
168
169 rv = ngx_conf_set_str_slot(cf, cmd, conf);
170 if (rv != NGX_CONF_OK) {
171 return rv;
172 }
173
174 path = (ngx_str_t *) ((char *) conf + cmd->offset);
175 if (brix_sss_load_keytab(cf, path, &keys) != NGX_OK) {
176 return NGX_CONF_ERROR;
177 }
178
179 return NGX_CONF_OK;
180}

182/* brix_conf_set_backend_sts_endpoint — `brix_backend_s3_sts_endpoint <url>`
183 * setter (phase-70 §5.5 origin leg; the §6 invariant that trust config is
184 * validated at LOAD, not first use — the residual the phase-88 audit flagged,
185 * mirroring the token-exchange endpoint).
186 *
187 * WHAT: Store the STS endpoint (plain str slot) and validate its shape at
188 * config time.
189 * WHY: The value is SigV4-signed then handed to libcurl's CURLOPT_URL verbatim
190 * (auth/s3/sts_http.c) and parsed for the "host" header by
191 * sts_host_from_url() (auth/s3/sts.c); a malformed endpoint would only
192 * surface as every S3 STS exchange fail-closing at first use.
193 * HOW: Require an http:// or https:// scheme with a non-empty host (the STS
194 * client pins CURLOPT_PROTOCOLS to http,https — SigV4 never transmits the
195 * secret, so a lab MinIO STS over http is legitimate, unlike the
196 * HTTPS-only exchange endpoint) and no whitespace/control bytes. */
197char *
198brix_conf_set_backend_sts_endpoint(ngx_conf_t *cf, ngx_command_t *cmd,
199 void *conf)
200{
201 char *rv;
202 ngx_str_t *ep;
203 size_t scheme;
204 ngx_uint_t i;
205
206 rv = ngx_conf_set_str_slot(cf, cmd, conf);
207 if (rv != NGX_CONF_OK) {
208 return rv;
209 }
210 ep = (ngx_str_t *) ((char *) conf + cmd->offset);
211
212 if (ep->len > sizeof("https://") - 1
213 && ngx_strncasecmp(ep->data, (u_char *) "https://", 8) == 0) {
214 scheme = sizeof("https://") - 1;
215 } else if (ep->len > sizeof("http://") - 1
216 && ngx_strncasecmp(ep->data, (u_char *) "http://", 7) == 0) {
217 scheme = sizeof("http://") - 1;
218 } else {
219 scheme = 0;
220 }
221
222 if (scheme == 0 || ep->data[scheme] == '/') {
223 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
224 "brix_backend_s3_sts_endpoint: \"%V\" is not an http:// or "
225 "https:// URL with a host — it is SigV4-signed and handed to the "
226 "STS client verbatim, so a malformed endpoint could never be "
227 "reached", ep);
228 return NGX_CONF_ERROR;
229 }
230 for (i = 0; i < ep->len; i++) {
231 if (ep->data[i] <= ' ' || ep->data[i] == 0x7f) {
232 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
233 "brix_backend_s3_sts_endpoint: whitespace or control byte at "
234 "offset %ui", i);
235 return NGX_CONF_ERROR;
236 }
237 }
238 return NGX_CONF_OK;
239}
