8/* Validate a configured path at startup: existence (stat), type
9 * (file/dir/either), and access mode (R/W/X_OK). A NULL/empty path is
10 * optional and skipped. Returns NGX_OK, or NGX_ERROR with an emerg log
11 * naming the failure. */
12ngx_int_t
13brix_validate_path(ngx_conf_t *cf, const char *label, const ngx_str_t *path,
14 brix_path_kind_t kind, int access_mode)
15{
16 struct stat st;
17
18 if (path == NULL || path->len == 0 || path->data == NULL) {
19 return NGX_OK;
20 }
21
22 if (stat((char *) path->data, &st) != 0) {
23 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
24 "brix: %s path \"%s\" is not accessible",
25 label, path->data);
26 return NGX_ERROR;
27 }
28
29 switch (kind) {
30 case BRIX_PATH_REGULAR_FILE:
31 if (!S_ISREG(st.st_mode)) {
32 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
33 "brix: %s path \"%s\" must be a regular file",
34 label, path->data);
35 return NGX_ERROR;
36 }
37 break;
38
39 case BRIX_PATH_DIRECTORY:
40 if (!S_ISDIR(st.st_mode)) {
41 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
42 "brix: %s path \"%s\" must be a directory",
43 label, path->data);
44 return NGX_ERROR;
45 }
46 break;
47
48 case BRIX_PATH_FILE_OR_DIRECTORY:
49 if (!S_ISREG(st.st_mode) && !S_ISDIR(st.st_mode)) {
50 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
51 "brix: %s path \"%s\" must be a file or directory",
52 label, path->data);
53 return NGX_ERROR;
54 }
55 break;
56 }
57
58 if (access_mode != 0 && access((char *) path->data, access_mode) != 0) {
59 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
60 "brix: %s path \"%s\" failed permission check",
61 label, path->data);
62 return NGX_ERROR;
63 }
64
65 return NGX_OK;
66}

68/* Copy an ngx_str_t into a NUL-terminated C string in cf->pool (ngx_str_t is
69 * not NUL-terminated, so strtol/strchr/etc. need this). Returns NGX_CONF_OK,
70 * or NGX_CONF_ERROR on allocation failure. */
71char *
72brix_copy_conf_string(ngx_conf_t *cf, const ngx_str_t *src, ngx_str_t *dst)
73{
74 dst->data = ngx_pnalloc(cf->pool, src->len + 1);
75 if (dst->data == NULL) {
76 return NGX_CONF_ERROR;
77 }
78
79 ngx_memcpy(dst->data, src->data, src->len);
80 dst->data[src->len] = '\0';
81 dst->len = src->len;
82 return NGX_CONF_OK;
83}

85/* brix_conf_set_backend_sss_keytab — `brix_backend_sss_keytab <path>` setter.
86 *
87 * WHAT: Store the identity-injection keytab path (plain str slot) and
88 * load-validate it immediately with the same loader the SSS *auth*
89 * keytabs go through (existence, regular file, private permissions,
90 * parseable keys).
91 * WHY: The delegation gate signs per-caller SSS credentials with this keytab
92 * at request time; a missing/world-readable/garbage keytab must fail
93 * `nginx -t`, not turn into a fleet-wide runtime deny (fail-closed but
94 * diagnosed at config load). Phase-70 §5.6 / P90-70.3.
95 * HOW: ngx_conf_set_str_slot for storage (conf tokens are NUL-terminated, so
96 * the stored bytes are directly usable as a C path at request time),
97 * then brix_sss_load_keytab on the stored value; the parsed key array is
98 * discarded — it lives in cf->pool and only proves loadability. */
99char *
100brix_conf_set_backend_sss_keytab(ngx_conf_t *cf, ngx_command_t *cmd,
101 void *conf)
102{
103 ngx_str_t *path;
104 ngx_array_t *keys;
105 char *rv;
106
107 rv = ngx_conf_set_str_slot(cf, cmd, conf);
108 if (rv != NGX_CONF_OK) {
109 return rv;
110 }
111
112 path = (ngx_str_t *) ((char *) conf + cmd->offset);
113 if (brix_sss_load_keytab(cf, path, &keys) != NGX_OK) {
114 return NGX_CONF_ERROR;
115 }
116
117 return NGX_CONF_OK;
118}

120/* brix_conf_set_backend_sts_endpoint — `brix_backend_s3_sts_endpoint <url>`
121 * setter (phase-70 §5.5 origin leg; the §6 invariant that trust config is
122 * validated at LOAD, not first use — the residual the phase-88 audit flagged,
123 * mirroring the token-exchange endpoint).
124 *
125 * WHAT: Store the STS endpoint (plain str slot) and validate its shape at
126 * config time.
127 * WHY: The value is SigV4-signed then handed to libcurl's CURLOPT_URL verbatim
128 * (auth/s3/sts_http.c) and parsed for the "host" header by
129 * sts_host_from_url() (auth/s3/sts.c); a malformed endpoint would only
130 * surface as every S3 STS exchange fail-closing at first use.
131 * HOW: Require an http:// or https:// scheme with a non-empty host (the STS
132 * client pins CURLOPT_PROTOCOLS to http,https — SigV4 never transmits the
133 * secret, so a lab MinIO STS over http is legitimate, unlike the
134 * HTTPS-only exchange endpoint) and no whitespace/control bytes. */
135char *
136brix_conf_set_backend_sts_endpoint(ngx_conf_t *cf, ngx_command_t *cmd,
137 void *conf)
138{
139 char *rv;
140 ngx_str_t *ep;
141 size_t scheme;
142 ngx_uint_t i;
143
144 rv = ngx_conf_set_str_slot(cf, cmd, conf);
145 if (rv != NGX_CONF_OK) {
146 return rv;
147 }
148 ep = (ngx_str_t *) ((char *) conf + cmd->offset);
149
150 if (ep->len > sizeof("https://") - 1
151 && ngx_strncasecmp(ep->data, (u_char *) "https://", 8) == 0) {
152 scheme = sizeof("https://") - 1;
153 } else if (ep->len > sizeof("http://") - 1
154 && ngx_strncasecmp(ep->data, (u_char *) "http://", 7) == 0) {
155 scheme = sizeof("http://") - 1;
156 } else {
157 scheme = 0;
158 }
159
160 if (scheme == 0 || ep->data[scheme] == '/') {
161 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
162 "brix_backend_s3_sts_endpoint: \"%V\" is not an http:// or "
163 "https:// URL with a host — it is SigV4-signed and handed to the "
164 "STS client verbatim, so a malformed endpoint could never be "
165 "reached", ep);
166 return NGX_CONF_ERROR;
167 }
168 for (i = 0; i < ep->len; i++) {
169 if (ep->data[i] <= ' ' || ep->data[i] == 0x7f) {
170 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
171 "brix_backend_s3_sts_endpoint: whitespace or control byte at "
172 "offset %ui", i);
173 return NGX_CONF_ERROR;
174 }
175 }
176 return NGX_CONF_OK;
177}
