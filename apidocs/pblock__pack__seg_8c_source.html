30/* pack_pread_full — read exactly len bytes at off (EINTR-safe); 0 or -1. */
31int
32pack_pread_full(int fd, void *buf, size_t len, off_t off)
33{
34 char *cursor = buf;
35 size_t got = 0;
36
37 while (got < len) {
38 ssize_t n = pread(fd, cursor + got, len - got, off + (off_t) got);
39
40 if (n < 0) {
41 if (errno == EINTR) { continue; }
42 return -1;
43 }
44 if (n == 0) {
45 errno = EIO; /* a record must never end early */
46 return -1;
47 }
48 got += (size_t) n;
49 }
50 return 0;
51}

82/* pack_lock — take the arena append/reap lock (pack/.lock, flock LOCK_EX).
83 * Returns the held fd, or -1/errno. */
84int
85pack_lock(const pblock_state_t *st)
86{
87 char lockp[PATH_MAX];
88 int fd;
89 int n = snprintf(lockp, sizeof(lockp), "%s/pack/.lock", st->root);
90
91 if (n <= 0 || (size_t) n >= sizeof(lockp)) {
92 errno = ENAMETOOLONG;
93 return -1;
94 }
95 fd = open(lockp, O_RDWR | O_CREAT | O_CLOEXEC, 0600);
96 if (fd < 0) {
97 return -1;
98 }
99 if (flock(fd, LOCK_EX) != 0) {
100 int err = errno;
101
102 close(fd);
103 errno = err;
104 return -1;
105 }
106 return fd;
107}

109/* pack_active_seg — highest existing segment number (0 when none). Scan the
110 * pack/ dir; called only under the arena lock, where the answer is stable. */
111int64_t
112pack_active_seg(const pblock_state_t *st)
113{
114 char dirp[PATH_MAX];
115 DIR *dir;
116 struct dirent *ent;
117 int64_t hi = 0;
118
119 if (snprintf(dirp, sizeof(dirp), "%s/pack", st->root) >= (int) sizeof(dirp)) {
120 return 0;
121 }
122 dir = opendir(dirp);
123 if (dir == NULL) {
124 return 0;
125 }
126 while ((ent = readdir(dir)) != NULL) {
127 long long seg;
128
129 if (sscanf(ent->d_name, "seg-%lld.dat", &seg) == 1
130 && (int64_t) seg > hi)
131 {
132 hi = (int64_t) seg;
133 }
134 }
135 closedir(dir);
136 return hi;
