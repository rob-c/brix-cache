1/*
2 * sd_http_readv.c — vectored-read coalescing for the HTTP-origin driver.
3 *
4 * WHAT: The `preadv` vtable slot: read one contiguous span with a single ranged
5 * GET and scatter it across the caller's iovecs, plus the loop
6 * (sd_http_pread_full) that covers a span larger than one GET.
7 *
8 * WHY: kXR_readv requests and pgread batches describe ONE contiguous span split
9 * into many page-sized iovecs. Without this slot brix_sd_obj_preadv falls
10 * back to one driver->pread per iovec — one HTTP round trip per 4 KiB —
11 * which turns a single 4 MiB vector read into a thousand requests against
12 * the origin and times the client out. sd_remote already coalesces this
13 * way over S3; a WebDAV/HTTP origin speaks the same Range grammar, so it
14 * can too. It lives in its own TU rather than in sd_http_read.c because
15 * that file is at the size cap and coalescing is its own concern: the
16 * single-request read path knows nothing about iovecs.
17 *
18 * HOW: Both functions are pure composition over sd_http_pread (sd_http_read.c)
19 * — no transport, credential, or endpoint knowledge lives here.
20 */

28/* sd_http_pread_full — fill `len` bytes at `off`, looping over the per-request
29 * SD_HTTP_PREAD_MAX cap.
30 *
31 * WHAT: Bytes read (short only at EOF), or -1 with errno set when the FIRST
32 * range GET failed.
33 * WHY: sd_http_pread returns at most SD_HTTP_PREAD_MAX per call and a 200-mode
34 * origin can return less still, so a caller that needs a whole span must
35 * loop; only preadv does, and only here.
36 * HOW: Repeat sd_http_pread until the span is covered. A 0 means EOF and ends
37 * the loop; a partial span already read is reported as a short read rather
38 * than discarded, matching pread(2). */
39static ssize_t
40sd_http_pread_full(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
41{
42 size_t done = 0;
43
44 while (done < len) {
45 ssize_t n = sd_http_pread(obj, (char *) buf + done, len - done,
46 off + (off_t) done);
47 if (n < 0) {
48 return (done > 0) ? (ssize_t) done : -1;
49 }
50 if (n == 0) {
51 break; /* EOF */
52 }
53 done += (size_t) n;
54 }
55 return (ssize_t) done;
56}

58/* sd_http_preadv — vectored read as ONE contiguous range GET, scattered.
59 *
60 * WHAT: Reads [off, off + sum(iov_len)) and scatters it across the iovecs.
61 * Total bytes read (short = EOF), or -1 with errno set.
62 * WHY: kXR_readv runs and pgread batches describe one contiguous span split
63 * into many page-sized iovecs. Without this slot brix_sd_obj_preadv falls
64 * back to one driver->pread per iovec — one HTTP round trip per 4 KiB —
65 * which turns a single 4 MiB vector read into a thousand requests and
66 * times the client out. sd_remote already coalesces this way over S3; a
67 * WebDAV origin speaks the same Range grammar, so it can too.
68 * HOW: A single iovec reads straight into the caller's buffer (no copy). A
69 * scattered call fills one malloc'd bounce buffer and memcpys out of it —
70 * the HTTP body is copied out of the response anyway, so the bounce costs
71 * one extra pass over bytes that have already crossed the socket. */
72ssize_t
73sd_http_preadv(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
74 off_t off)
75{
76 size_t total = 0, scattered;
77 ssize_t n;
78 char *bounce;
79 int i;
80
81 for (i = 0; i < iovcnt; i++) {
82 total += iov[i].iov_len;
83 }
84 if (total == 0) {
85 return 0;
86 }
87 if (iovcnt == 1) {
88 return sd_http_pread_full(obj, iov[0].iov_base, iov[0].iov_len, off);
89 }
90
91 bounce = malloc(total);
92 if (bounce == NULL) {
93 errno = ENOMEM;
94 return -1;
95 }
96 n = sd_http_pread_full(obj, bounce, total, off);
97 if (n < 0) {
98 free(bounce);
99 return -1;
100 }
101
102 scattered = 0;
103 for (i = 0; i < iovcnt && scattered < (size_t) n; i++) {
104 size_t take = iov[i].iov_len;
105
106 if (take > (size_t) n - scattered) {
107 take = (size_t) n - scattered;
108 }
109 memcpy(iov[i].iov_base, bounce + scattered, take);
110 scattered += take;
111 }
112 free(bounce);
113 return n;
114}
