1/*
2 * tpc_token_renew.c — the wire leg of mid-transfer delegated-credential
3 * renewal (W8.2).
4 *
5 * WHAT: The entry points around the pure decision kernel in
6 * tpc/common/cred_renew.c. tpc_cred_note_expiry() records the lifetime of
7 * whatever credential the pull ended up presenting, by reading the `exp`
8 * claim of the delegated token. tpc_cred_renew_if_due() is sampled once
9 * per streamed chunk: it asks for a verdict and, when renewal is due,
10 * permitted and useful, mints a fresh token and re-presents it to the
11 * source on the live connection with a second ztn kXR_auth.
12 *
13 * WHY: One-shot delegation fetches a credential at bootstrap and never looks at
14 * it again, so a copy that runs longer than the token's lifetime streams on
15 * a credential the source may stop honouring. Renewing has to happen from
16 * inside the pull thread, between drained read windows, because that is the
17 * only point at which the outbound socket is quiet enough to carry an
18 * unrelated request/response pair.
19 *
20 * HOW: Four gates stand between a chunk boundary and an issuer round trip, and
21 * each one exists because of a distinct way renewal could do harm:
22 *
23 * 1. cred_presented — renew only a credential this session actually
24 * presented. An anonymous or GSI pull has no ztn credential in play;
25 * sending it a kXR_auth would be a protocol error invented by a
26 * feature that was supposed to be invisible.
27 * 2. the verdict — pure, and NOT_NEEDED for every unknown input, so
28 * enabling renewal cannot by itself change a pull that works today.
29 * 3. mint authority — a forwarding mode holds the client's credential,
30 * not the right to re-mint it; it may warn, never fetch.
31 * 4. the retry window and cred_renew_off — a failed mint is retried at
32 * most once a minute so a broken issuer cannot turn one pull into a
33 * fork/exec storm, and an issuer that keeps returning a credential
34 * still inside the lead window disables renewal for this transfer
35 * rather than minting once per megabyte forever.
36 *
37 * The sample point reuses the stream loop's existing once-per-chunk clock
38 * read, so renewal costs one comparison per MiB and no syscall. Renewal is
39 * off until brix_tpc_outbound_renew_lead is set, like every other
40 * selection/pacing engine in the tree. The re-auth uses handshake sequence
41 * 4 — distinct from bootstrap's 1/2/3 and from the read loop's 3 — so a
42 * packet capture reads unambiguously.
43 */

70/*
71 * WHAT: Record the expiry of the credential now held in t->delegated_token.
72 * WHY: Renewal cannot be decided without a lifetime, and the only lifetime the
73 * destination can see is the one the token carries. A token that is not a
74 * JWS, or carries no positive `exp`, leaves the expiry at 0 — which the
75 * decision kernel reads as "nothing to decide", i.e. exactly the
76 * pre-renewal behaviour.
77 * HOW: brix_token_peek_exp() reads the claim WITHOUT verifying the signature.
78 * That is correct here and only here: this is the destination reading the
79 * lifetime of a credential it already holds and already presented, in
80 * order to decide when to ask for another one. Nothing is authorised on
81 * the basis of this value — the source re-validates the renewed token in
82 * full, exactly as it validated the first one.
83 */
84void
85tpc_cred_note_expiry(brix_tpc_pull_t *t)
86{
87 time_t exp = 0;
88
89 t->cred_expires_at = 0;
90
91 if (t->delegated_token[0] == '\0') {
92 return;
93 }
94 if (brix_token_peek_exp(t->delegated_token, strlen(t->delegated_token),
95 &exp) != 0)
96 {
97 return;
98 }
99 t->cred_expires_at = exp;
100}

102/*
103 * WHAT: The shared give-up path: fatal once the credential is genuinely past
104 * expiry, a one-line warning while it is merely due.
105 * WHY: Both the "cannot mint" and the "mint failed" arms end here, and both
106 * must answer the same question the same way — is the credential dead, or
107 * only close to it? A due credential is still one the source accepts, so
108 * the transfer continues; an expired one is not, and streaming on it
109 * produces a copy nobody can explain.
110 * HOW: EXPIRED sets err_msg/kXR_AuthFailed and returns -1; anything else warns
111 * and returns 0. `fatal_when_expired` lets the caller keep a pull alive on
112 * an expired credential it could never have renewed anyway, which is what
113 * brix_tpc_outbound_renew_strict off means.
114 */
115static int
116tpc_cred_renew_give_up(brix_tpc_pull_t *t, brix_tpc_renew_verdict_t verdict,
117 const char *why, ngx_flag_t fatal_when_expired)
118{
119 if (verdict == BRIX_TPC_RENEW_EXPIRED && fatal_when_expired) {
120 snprintf(t->err_msg, sizeof(t->err_msg),
121 "TPC delegated credential expired mid-transfer at offset "
122 "%llu: %s (token_mode \"%s\")",
123 (unsigned long long) t->bytes_written, why, t->token_mode);
124 t->xrd_error = kXR_AuthFailed;
125 return -1;
126 }
127
128 ngx_log_error(NGX_LOG_WARN, tpc_renew_log(t), 0,
129 "brix: TPC delegated credential %s mid-transfer: %s "
130 "(token_mode \"%s\") - continuing",
131 brix_tpc_renew_verdict_name(verdict), why, t->token_mode);
132 return 0;
133}

135/*
136 * WHAT: Mint a fresh delegated token and re-present it to the source.
137 * WHY: Split from the policy below so the policy stays a readable ladder and
138 * this half owns the buffer discipline: tpc_fetch_delegated_token()
139 * overwrites t->delegated_token in place, so a failed fetch leaves it
140 * unusable and it must be cleared. The recorded expiry is deliberately
141 * NOT cleared — it describes the credential the source is still holding,
142 * and forgetting it would blind every later verdict.
143 * HOW: fetch -> re-read the new expiry -> ztn kXR_auth on the live socket.
144 */
145static int
146tpc_cred_mint_and_present(brix_tpc_pull_t *t, int fd)
147{
148 if (tpc_fetch_delegated_token(t) != 0) {
149 t->delegated_token[0] = '\0';
150 return -1;
151 }
152
153 tpc_cred_note_expiry(t);
154
155 if (tpc_outbound_ztn_seq(t, fd, TPC_RENEW_AUTH_SEQ) != 0) {
156 return -1;
157 }
158
159 t->cred_renewals++;
160 return 0;
161}

163/*
164 * WHAT: Note that a renewal succeeded, and stop renewing when it did not help.
165 * WHY: An issuer configured with a lifetime shorter than the operator's lead
166 * window hands back a credential that is due the moment it arrives. Left
167 * alone that mints once per megabyte for the length of the copy — a
168 * fork/exec per MiB against the IdP, caused by a misconfiguration the
169 * operator cannot see. One warning and a per-transfer stop is the honest
170 * response: the credential IS fresh, it simply cannot satisfy this lead.
171 * HOW: Re-run the pure verdict against the token just obtained.
172 */
173static void
174tpc_cred_renew_note_success(brix_tpc_pull_t *t, time_t now)
175{
176 ngx_log_error(NGX_LOG_NOTICE, tpc_renew_log(t), 0,
177 "brix: TPC delegated credential renewed mid-transfer "
178 "(token_mode \"%s\", renewal %ui, offset %uz)",
179 t->token_mode, t->cred_renewals, t->bytes_written);
180
181 if (brix_tpc_renew_verdict(t->cred_expires_at, now,
182 t->conf->common.tpc_outbound_renew_lead)
183 == BRIX_TPC_RENEW_NOT_NEEDED)
184 {
185 return;
186 }
187
188 t->cred_renew_off = 1;
189 ngx_log_error(NGX_LOG_WARN, tpc_renew_log(t), 0,
190 "brix: TPC renewed credential is still inside the "
191 "brix_tpc_outbound_renew_lead window - the issuer's lifetime "
192 "is shorter than the lead; renewal disabled for this transfer");
193}

195/*
196 * WHAT: The once-per-chunk renewal sample. Returns 0 to continue streaming, -1
197 * to fail the pull with t->err_msg / t->xrd_error set.
198 * WHY: Four populations have to stay distinct — a session with no credential
199 * in play, a mode that can mint, a mode that cannot, and a mode that can
200 * but has just failed to. Only the second ever performs I/O here.
201 * HOW: Cheap gates first (no credential, renewal spent), then the pure
202 * verdict, then mint authority, then the retry window, then the mint.
203 * Every arm is an early return.
204 */
205int
206tpc_cred_renew_if_due(brix_tpc_pull_t *t, int fd)
207{
208 brix_tpc_renew_verdict_t verdict;
209 time_t now;
210
211 if (t->conf == NULL || !t->cred_presented || t->cred_renew_off) {
212 return 0;
213 }
214
215 now = time(NULL);
216 verdict = brix_tpc_renew_verdict(t->cred_expires_at, now,
217 t->conf->common.tpc_outbound_renew_lead);
218 if (verdict == BRIX_TPC_RENEW_NOT_NEEDED) {
219 return 0;
220 }
221
222 if (!brix_tpc_renew_mode_can_mint(t->token_mode)) {
223 return tpc_cred_renew_give_up(t, verdict,
224 "this token_mode forwards a credential it cannot re-mint",
225 t->conf->common.tpc_outbound_renew_strict);
226 }
227
228 if (now < t->cred_renew_next_try) {
229 return tpc_cred_renew_give_up(t, verdict,
230 "a previous renewal failed and the retry window is open", 1);
231 }
232
233 if (tpc_cred_mint_and_present(t, fd) != 0) {
234 t->cred_renew_next_try = now + TPC_RENEW_RETRY_SECS;
235 return tpc_cred_renew_give_up(t, verdict, "renewal failed", 1);
236 }
237
238 tpc_cred_renew_note_success(t, now);
239 return 0;
240}
