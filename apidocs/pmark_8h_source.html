4/*
5 * pmark.h — SciTags packet marking (network flow tagging) for nginx-xrootd.
6 *
7 * WHAT: Public types + API for the `src/pmark/` subsystem that labels each data
8 * flow with a SciTags (experiment, activity) pair, using the two SciTags
9 * techniques: (1) out-of-band "firefly" UDP reporting (firefly.c) and (2)
10 * in-band IPv6 Flow Label stamping (flowlabel.c). It mirrors XRootD's `pmark`
11 * directive family so a site keeps working with its SciTags/flowd dashboards
12 * after swapping an XRootD gateway for nginx-xrootd. See
13 * docs/refactor/phase-34-packet-marking-scitags.md for the full design.
14 *
15 * WHY: It is a SHARED subsystem (called from the root:// stream module, the
16 * WebDAV/S3 HTTP modules, and TPC) rather than a standalone nginx module, so
17 * the per-server config lives in the common preamble (config/shared_conf.h)
18 * and the runtime state is one per-worker object plus one per-flow object.
19 *
20 * HOW: A flow-id is a 16-bit value `(experiment << 6) | activity` (10-bit
21 * experiment 1..BRIX_PMARK_EXPERIMENT_MAX, 6-bit activity 1..BRIX_PMARK_ACTIVITY_MAX). Config is opt-in. Marking is
22 * ALWAYS fail-open: a misconfiguration, an unreachable collector, or a kernel
23 * that refuses a flow label degrades to "not marked" and NEVER fails, blocks,
24 * or slows a transfer.
25 */

191/* Pack (experiment, activity) into a 16-bit flow-id. */
192static ngx_inline ngx_uint_t
193brix_pmark_flowid(ngx_uint_t exp, ngx_uint_t act)
194{
195 return ((exp & 0x3FF) << BRIX_PMARK_ACT_BITS) | (act & BRIX_PMARK_ACT_MASK);
196}
197
198/* Split a 16-bit flow-id into (experiment, activity). Returns NGX_ERROR (and
199 * leaves the exp/act outputs untouched) for an out-of-range value — caller must treat that
200 * as "no tag", never as exp=act=0 being valid. */
201static ngx_inline ngx_int_t
202brix_pmark_flow_split(ngx_uint_t flow, ngx_uint_t *exp, ngx_uint_t *act)
203{
204 if (flow < BRIX_PMARK_FLOW_MIN || flow > BRIX_PMARK_FLOW_MAX) {
205 return NGX_ERROR;
206 }
207 *exp = flow >> BRIX_PMARK_ACT_BITS;
208 *act = flow & BRIX_PMARK_ACT_MASK;
209 return NGX_OK;
210}
211
212/* Reverse the low 9 bits of `v` (the SciTags community field is "used in reversed
213 * order" — draft-cc-v6ops-wlcg-flow-label-marking §4). */
214static ngx_inline uint32_t
215brix_pmark_reverse9(ngx_uint_t v)
216{
217 uint32_t r = 0;
218 int i;
219
220 v &= 0x1FF;
221 for (i = 0; i < 9; i++) {
222 r = (r << 1) | (uint32_t) (v & 1u);
223 v >>= 1;
224 }
225 return r;
226}
227
228/* Encode a SciTags (experiment, activity) pair into the 20-bit IPv6 Flow Label
229 * value (host order) per draft-cc-v6ops-wlcg-flow-label-marking. This is the
230 * STRUCTURAL value only (entropy bits zero) so it is deterministic and decodable;
231 * the caller ORs in random entropy (BRIX_PMARK_FL_ENTROPY_MASK) once per flow.
232 * `act` is the 6-bit activity (bits 2..7); `exp` is the 9-bit community placed in
233 * reversed order at bits 9..17. */
234static ngx_inline uint32_t
235brix_pmark_flowlabel_encode(ngx_uint_t exp, ngx_uint_t act)
236{
237 uint32_t community = brix_pmark_reverse9(exp);
238 uint32_t activity = (uint32_t) (act & BRIX_PMARK_ACT_MASK);
239
240 return ((community << 9) | (activity << 2)) & BRIX_PMARK_FL_MASK;
241}
