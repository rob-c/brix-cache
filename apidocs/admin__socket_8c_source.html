51/* Parse 32 lowercase/uppercase hex chars into a 16-byte sessid; 0 on error. */
52static int
53admin_parse_sessid(const u_char *hex, size_t len,
54 u_char out[BRIX_SESSION_ID_LEN])
55{
56 size_t i;
57
58 if (len != BRIX_SESSION_ID_LEN * 2) {
59 return 0;
60 }
61 for (i = 0; i < BRIX_SESSION_ID_LEN; i++) {
62 int hi = ngx_hextoi((u_char *) &hex[i * 2], 1);
63 int lo = ngx_hextoi((u_char *) &hex[i * 2 + 1], 1);
64 if (hi == NGX_ERROR || lo == NGX_ERROR) {
65 return 0;
66 }
67 out[i] = (u_char) ((hi << 4) | lo);
68 }
69 return 1;
70}

72static int
73admin_list_cb(void *ud, const unsigned char *sessid, unsigned pathid,
74 void *conn)
75{
76 admin_list_state_t *st = ud;
77 ngx_connection_t *tc = conn;
78 char dn[512];
79 char vo[512];
80 ngx_uint_t token_auth;
81 size_t i;
82 u_char *p;
83
84 if (pathid != BRIX_ADMIN_PATHID) {
85 return 0; /* a bound data channel, not a session's primary */
86 }
87 /* Room for: 32 hex + peer(<=64) + dn(<=128 shown) + separators + '\n'. */
88 if (st->cap - st->len < 32 + 1 + 64 + 1 + 128 + 2) {
89 return 1; /* reply buffer full — stop enumerating */
90 }
91
92 p = st->buf + st->len;
93 for (i = 0; i < BRIX_SESSION_ID_LEN; i++) {
94 p = ngx_sprintf(p, "%02xd", sessid[i]);
95 }
96 /* Peer address: the operator's handle for choosing a session to act on. */
97 if (tc->addr_text.len > 0) {
98 size_t alen = ngx_min(tc->addr_text.len, 64);
99 *p++ = ' ';
100 p = ngx_cpymem(p, tc->addr_text.data, alen);
101 } else {
102 p = ngx_sprintf(p, " -");
103 }
104 dn[0] = '\0';
105 if (brix_session_lookup(sessid, dn, sizeof(dn), vo, sizeof(vo),
106 &token_auth)
107 && dn[0] != '\0')
108 {
109 p = ngx_snprintf(p, 1 + 128, " %s", dn);
110 } else {
111 p = ngx_sprintf(p, " -");
112 }
113 *p++ = '\n';
114 st->len = (size_t) (p - st->buf);
115 st->n++;
116 return 0;
117}

119/* One-shot timed-pause resume: clears the pause flag and posts the read event
120 * so the recv loop drains whatever backed up. The connection is guaranteed
121 * alive while the timer is armed — disconnect deletes it (disconnect.c). */
122static void
123admin_pause_timeout(ngx_event_t *ev)
124{
125 ngx_connection_t *tc = ev->data;
126 brix_ctx_t *tctx;
127
128 tctx = ngx_stream_get_module_ctx((ngx_stream_session_t *) tc->data,
129 ngx_stream_brix_module);
130 if (tctx == NULL || tctx->destroyed) {
131 return;
132 }
133 tctx->admin_paused = 0;
134 ngx_post_event(tc->read, &ngx_posted_events);
135}

137/* Resolve a session id to its local ctx (NULL when unknown/not-local/destroyed);
138 * *target_out receives the connection when found. */
139static brix_ctx_t *
140admin_resolve_ctx(const u_char *sessid, ngx_connection_t **target_out)
141{
142 ngx_connection_t *target = brix_offload_lookup(sessid, BRIX_ADMIN_PATHID);
143 brix_ctx_t *tctx;
144
145 *target_out = target;
146 if (target == NULL) {
147 return NULL;
148 }
149 tctx = ngx_stream_get_module_ctx((ngx_stream_session_t *) target->data,
150 ngx_stream_brix_module);
151 return (tctx == NULL || tctx->destroyed) ? NULL : tctx;
152}

154/* "list" — enumerate the local offloaded sessions. */
155static void
156admin_cmd_list(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
157{
158 admin_list_state_t ls;
159 u_char head[32];
160 u_char *he;
161 size_t hlen;
162
163 (void) ud; (void) args; (void) alen;
164
165 ls.buf = rep->buf + 32; /* body after the "ok <n>\n" header */
166 ls.len = 0;
167 ls.cap = rep->cap - 64;
168 ls.n = 0;
169 (void) brix_offload_foreach(admin_list_cb, &ls);
170
171 he = ngx_snprintf(head, sizeof(head), "ok %uz\n", ls.n);
172 hlen = (size_t) (he - head);
173 /* The body was built past the largest possible header; slide the header in
174 * front of it so the reply is one contiguous run. */
175 ngx_memcpy(rep->buf + 32 - hlen, head, hlen);
176 rep->out = rep->buf + 32 - hlen;
177 rep->len = hlen + ls.len;
178}

180/* "disc <sid>" — graceful shutdown(2); the loop's EOF path does the teardown. */
181static void
182admin_cmd_disc(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
183{
184 u_char sessid[BRIX_SESSION_ID_LEN];
185 ngx_connection_t *target;
186
187 (void) ud;
188
189 if (!admin_parse_sessid(args, alen, sessid)) {
190 brix_admin_reply_set(rep, "err unknown-command\n");
191 return;
192 }
193 target = brix_offload_lookup(sessid, BRIX_ADMIN_PATHID);
194 if (target == NULL) {
195 brix_admin_reply_set(rep, "err unknown-or-not-local\n");
196 return;
197 }
198 (void) shutdown(target->fd, SHUT_RDWR);
199 brix_admin_reply_set(rep, "ok\n");
200}

202/* "msg <sid> <text>" — deliver an async attn message to the target session. */
203static void
204admin_cmd_msg(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
205{
206 u_char *sp = ngx_strlchr(args, args + alen, ' ');
207 u_char sessid[BRIX_SESSION_ID_LEN];
208 ngx_connection_t *target;
209 brix_ctx_t *tctx;
210 const char *text;
211 size_t tlen;
212
213 (void) ud;
214
215 if (sp == NULL
216 || !admin_parse_sessid(args, (size_t) (sp - args), sessid)) {
217 brix_admin_reply_set(rep, "err unknown-command\n");
218 return;
219 }
220 text = (const char *) sp + 1;
221 tlen = (size_t) (args + alen - (sp + 1));
222 tctx = admin_resolve_ctx(sessid, &target);
223 if (target == NULL) {
224 brix_admin_reply_set(rep, "err unknown-or-not-local\n");
225 } else if (tctx == NULL || tlen == 0) {
226 brix_admin_reply_set(rep, "err target-unusable\n");
227 } else if (brix_send_attn_asyncms(tctx, target, text, tlen) != NGX_OK) {
228 brix_admin_reply_set(rep, "err send-failed\n");
229 } else {
230 brix_admin_reply_set(rep, "ok\n");
231 }
232}

234/* "pause <sid> [secs]" — stop parsing the target; optional auto-resume timer. */
235static void
236admin_cmd_pause(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
237{
238 u_char *sp = ngx_strlchr(args, args + alen, ' ');
239 size_t hexlen = sp ? (size_t) (sp - args) : alen;
240 u_char sessid[BRIX_SESSION_ID_LEN];
241 ngx_connection_t *target;
242 brix_ctx_t *tctx;
243 ngx_int_t secs = 0;
244
245 (void) ud;
246
247 if (sp != NULL) {
248 secs = ngx_atoi(sp + 1, (size_t) (args + alen - (sp + 1)));
249 if (secs == NGX_ERROR || secs < 0) {
250 brix_admin_reply_set(rep, "err bad-seconds\n");
251 return;
252 }
253 }
254 if (!admin_parse_sessid(args, hexlen, sessid)) {
255 brix_admin_reply_set(rep, "err unknown-command\n");
256 return;
257 }
258 tctx = admin_resolve_ctx(sessid, &target);
259 if (tctx == NULL) {
260 brix_admin_reply_set(rep, "err unknown-or-not-local\n");
261 return;
262 }
263 tctx->admin_paused = 1;
264 if (secs > 0) {
265 tctx->admin_pause_ev.handler = admin_pause_timeout;
266 tctx->admin_pause_ev.data = target;
267 tctx->admin_pause_ev.log = target->log;
268 ngx_add_timer(&tctx->admin_pause_ev, (ngx_msec_t) secs * BRIX_ROOT_MS_PER_SEC);
269 }
270 brix_admin_reply_set(rep, "ok\n");
271}

273/* "cont <sid>" — clear a pause and repost the read event. */
274static void
275admin_cmd_cont(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
276{
277 u_char sessid[BRIX_SESSION_ID_LEN];
278 ngx_connection_t *target;
279 brix_ctx_t *tctx;
280
281 (void) ud;
282
283 if (!admin_parse_sessid(args, alen, sessid)) {

286 }

288 if (tctx == NULL) {
289 brix_admin_reply_set(rep, "err unknown-or-not-local\n");

294 }

298 ngx_post_event(target->read, &ngx_posted_events);
299 brix_admin_reply_set(rep, "ok\n");
300}

302/* "abort <sid>" — RST-on-close (SO_LINGER{1,0}) then shutdown(2). */
303static void
304admin_cmd_abort(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
305{
306 u_char sessid[BRIX_SESSION_ID_LEN];
307 ngx_connection_t *target;
308 struct linger lg;
309
310 (void) ud;
311
312 if (!admin_parse_sessid(args, alen, sessid)) {
313 brix_admin_reply_set(rep, "err unknown-command\n");
314 return;
315 }
316 target = brix_offload_lookup(sessid, BRIX_ADMIN_PATHID);
317 if (target == NULL) {
318 brix_admin_reply_set(rep, "err unknown-or-not-local\n");
319 return;
320 }
321 /* Abort = disconnect WITHOUT ceremony: SO_LINGER{1,0} makes the teardown's
322 * eventual close(2) send an RST instead of a FIN, so the client sees
323 * ECONNRESET — stock abort semantics. The teardown still runs through the
324 * normal event-loop EOF path (no re-entrancy), exactly like disc. */
325 lg.l_onoff = 1;
326 lg.l_linger = 0;
327 (void) setsockopt(target->fd, SOL_SOCKET, SO_LINGER, &lg, sizeof(lg));
328 (void) shutdown(target->fd, SHUT_RDWR);
329 brix_admin_reply_set(rep, "ok\n");
330}
