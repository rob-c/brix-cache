58/* Parse 32 lowercase/uppercase hex chars into a 16-byte sessid; 0 on error. */
59static int
60admin_parse_sessid(const u_char *hex, size_t len,
61 u_char out[BRIX_SESSION_ID_LEN])
62{
63 size_t i;
64
65 if (len != BRIX_SESSION_ID_LEN * 2) {
66 return 0;
67 }
68 for (i = 0; i < BRIX_SESSION_ID_LEN; i++) {
69 int hi = ngx_hextoi((u_char *) &hex[i * 2], 1);
70 int lo = ngx_hextoi((u_char *) &hex[i * 2 + 1], 1);
71 if (hi == NGX_ERROR || lo == NGX_ERROR) {
72 return 0;
73 }
74 out[i] = (u_char) ((hi << 4) | lo);
75 }
76 return 1;
77}

79static int
80admin_list_cb(void *ud, const unsigned char *sessid, unsigned pathid,
81 void *conn)
82{
83 admin_list_state_t *st = ud;
84 ngx_connection_t *tc = conn;
85 char dn[512];
86 char vo[512];
87 ngx_uint_t token_auth;
88 size_t i;
89 u_char *p;
90
91 if (pathid != BRIX_ADMIN_PATHID) {
92 return 0; /* a bound data channel, not a session's primary */
93 }
94 /* Room for: 32 hex + peer(<=64) + dn(<=128 shown) + separators + '\n'. */
95 if (st->cap - st->len < 32 + 1 + 64 + 1 + 128 + 2) {
96 return 1; /* reply buffer full — stop enumerating */
97 }
98
99 p = st->buf + st->len;
100 for (i = 0; i < BRIX_SESSION_ID_LEN; i++) {
101 p = ngx_sprintf(p, "%02xd", sessid[i]);
102 }
103 /* Peer address: the operator's handle for choosing a session to act on. */
104 if (tc->addr_text.len > 0) {
105 size_t alen = ngx_min(tc->addr_text.len, 64);
106 *p++ = ' ';
107 p = ngx_cpymem(p, tc->addr_text.data, alen);
108 } else {
109 p = ngx_sprintf(p, " -");
110 }
111 dn[0] = '\0';
112 if (brix_session_lookup(sessid, dn, sizeof(dn), vo, sizeof(vo),
113 &token_auth)
114 && dn[0] != '\0')
115 {
116 p = ngx_snprintf(p, 1 + 128, " %s", dn);
117 } else {
118 p = ngx_sprintf(p, " -");
119 }
120 *p++ = '\n';
121 st->len = (size_t) (p - st->buf);
122 st->n++;
123 return 0;
124}

126/* One-shot timed-pause resume: clears the pause flag and posts the read event
127 * so the recv loop drains whatever backed up. The connection is guaranteed
128 * alive while the timer is armed — disconnect deletes it (disconnect.c). */
129static void
130admin_pause_timeout(ngx_event_t *ev)
131{

134

137 if (tctx == NULL || tctx->destroyed) {
138 return;
139 }
140 tctx->admin_paused = 0;

166{

168 brix_ctx_t *tctx;

179/* "list" — enumerate the local offloaded sessions. */
180static void
181admin_cmd_list(brix_admin_conn_t *st, u_char *out)
182{
183 admin_list_state_t ls;
184 u_char head[32];
185 u_char *he;
186 size_t hlen;
187
188 ls.buf = out + 32; /* body after the "ok <n>\n" header */
189 ls.len = 0;
190 ls.cap = BRIX_ADMIN_REPLY_MAX - 64;
191 ls.n = 0;
192 (void) brix_offload_foreach(admin_list_cb, &ls);
193
194 he = ngx_snprintf(head, sizeof(head), "ok %uz\n", ls.n);
195 hlen = (size_t) (he - head);
196 /* The body was built past the largest possible header; slide the header in
197 * front of it so the reply is one contiguous run. */
198 ngx_memcpy(out + 32 - hlen, head, hlen);
199 st->reply = out + 32 - hlen;
200 st->reply_len = hlen + ls.len;
201}

203/* "disc <sid>" — graceful shutdown(2); the loop's EOF path does the teardown. */
204static void
205admin_cmd_disc(brix_admin_conn_t *st, u_char *out, u_char *line, size_t len)
206{
207 u_char sessid[BRIX_SESSION_ID_LEN];
208 ngx_connection_t *target;
209
210 if (!admin_parse_sessid(line + 5, len - 5, sessid)) {
211 admin_reply(st, out, "err unknown-command\n");
212 return;
213 }
214 target = brix_offload_lookup(sessid, BRIX_ADMIN_PATHID);
215 if (target == NULL) {
216 admin_reply(st, out, "err unknown-or-not-local\n");
217 return;
218 }
219 (void) shutdown(target->fd, SHUT_RDWR);
220 admin_reply(st, out, "ok\n");
221}

223/* "msg <sid> <text>" — deliver an async attn message to the target session. */
224static void
225admin_cmd_msg(brix_admin_conn_t *st, u_char *out, u_char *line, size_t len)
226{
227 u_char *sp = ngx_strlchr(line + 4, line + len, ' ');
228 u_char sessid[BRIX_SESSION_ID_LEN];
229 ngx_connection_t *target;
230 brix_ctx_t *tctx;
231 const char *text;
232 size_t tlen;
233
234 if (sp == NULL
235 || !admin_parse_sessid(line + 4, (size_t) (sp - (line + 4)), sessid)) {
236 admin_reply(st, out, "err unknown-command\n");
237 return;
238 }
239 text = (const char *) sp + 1;
240 tlen = (size_t) (line + len - (sp + 1));
241 tctx = admin_resolve_ctx(sessid, &target);
242 if (target == NULL) {
243 admin_reply(st, out, "err unknown-or-not-local\n");
244 } else if (tctx == NULL || tlen == 0) {
245 admin_reply(st, out, "err target-unusable\n");
246 } else if (brix_send_attn_asyncms(tctx, target, text, tlen) != NGX_OK) {
247 admin_reply(st, out, "err send-failed\n");
248 } else {
249 admin_reply(st, out, "ok\n");
250 }

253/* "pause <sid> [secs]" — stop parsing the target; optional auto-resume timer. */
254static void
255admin_cmd_pause(brix_admin_conn_t *st, u_char *out, u_char *line, size_t len)
256{
257 u_char *sp = ngx_strlchr(line + 6, line + len, ' ');
258 size_t hexlen = sp ? (size_t) (sp - (line + 6)) : len - 6;
259 u_char sessid[BRIX_SESSION_ID_LEN];
260 ngx_connection_t *target;
261 brix_ctx_t *tctx;
262 ngx_int_t secs = 0;
263
264 if (sp != NULL) {
265 secs = ngx_atoi(sp + 1, (size_t) (line + len - (sp + 1)));
266 if (secs == NGX_ERROR || secs < 0) {
267 admin_reply(st, out, "err bad-seconds\n");
268 return;
269 }
270 }
271 if (!admin_parse_sessid(line + 6, hexlen, sessid)) {
272 admin_reply(st, out, "err unknown-command\n");
273 return;
274 }
275 tctx = admin_resolve_ctx(sessid, &target);
276 if (tctx == NULL) {
277 admin_reply(st, out, "err unknown-or-not-local\n");
278 return;
279 }
280 tctx->admin_paused = 1;
281 if (secs > 0) {
282 tctx->admin_pause_ev.handler = admin_pause_timeout;
283 tctx->admin_pause_ev.data = target;
284 tctx->admin_pause_ev.log = target->log;
285 ngx_add_timer(&tctx->admin_pause_ev, (ngx_msec_t) secs * 1000);
286 }
287 admin_reply(st, out, "ok\n");
288}

290/* "cont <sid>" — clear a pause and repost the read event. */
291static void
292admin_cmd_cont(brix_admin_conn_t *st, u_char *out, u_char *line, size_t len)
293{
294 u_char sessid[BRIX_SESSION_ID_LEN];
295 ngx_connection_t *target;
296 brix_ctx_t *tctx;
297
298 if (!admin_parse_sessid(line + 5, len - 5, sessid)) {
299 admin_reply(st, out, "err unknown-command\n");
300 return;
301 }
302 tctx = admin_resolve_ctx(sessid, &target);
303 if (tctx == NULL) {
304 admin_reply(st, out, "err unknown-or-not-local\n");
305 return;

308 ngx_del_timer(&tctx->admin_pause_ev);

310 tctx->admin_paused = 0;

314 admin_reply(st, out, "ok\n");

317/* "abort <sid>" — RST-on-close (SO_LINGER{1,0}) then shutdown(2). */
318static void
319admin_cmd_abort(brix_admin_conn_t *st, u_char *out, u_char *line, size_t len)
320{
321 u_char sessid[BRIX_SESSION_ID_LEN];
322 ngx_connection_t *target;
323 struct linger lg;
324
325 if (!admin_parse_sessid(line + 6, len - 6, sessid)) {
326 admin_reply(st, out, "err unknown-command\n");
327 return;
328 }
329 target = brix_offload_lookup(sessid, BRIX_ADMIN_PATHID);
330 if (target == NULL) {
331 admin_reply(st, out, "err unknown-or-not-local\n");
332 return;
333 }
334 /* Abort = disconnect WITHOUT ceremony: SO_LINGER{1,0} makes the teardown's
335 * eventual close(2) send an RST instead of a FIN, so the client sees
336 * ECONNRESET — stock abort semantics. The teardown still runs through the
337 * normal event-loop EOF path (no re-entrancy), exactly like disc. */
338 lg.l_onoff = 1;
339 lg.l_linger = 0;
340 (void) setsockopt(target->fd, SOL_SOCKET, SO_LINGER, &lg, sizeof(lg));
341 (void) shutdown(target->fd, SHUT_RDWR);
342 admin_reply(st, out, "ok\n");
343}

345/* Dispatch one complete command line into a pool-allocated reply. */
346static void
347admin_dispatch(ngx_connection_t *c, brix_admin_conn_t *st)
348{
349 u_char *line = st->cmd;
350 size_t len = st->cmd_len;
351 u_char *out;
352
353 out = ngx_pnalloc(c->pool, BRIX_ADMIN_REPLY_MAX);
354 if (out == NULL) {
355 return;
356 }
357 st->reply = out;
358 st->reply_sent = 0;
359 len = admin_strip_eol(line, len);
360
361 /* Prefix-match the verb; each handler parses its own operands and owns its
362 * reply (falling back to unknown-command on a malformed session id). */
363 if (len == 4 && ngx_strncmp(line, "list", 4) == 0) {
364 admin_cmd_list(st, out);
365 } else if (len > 5 && ngx_strncmp(line, "disc ", 5) == 0) {
366 admin_cmd_disc(st, out, line, len);
367 } else if (len > 4 && ngx_strncmp(line, "msg ", 4) == 0) {
368 admin_cmd_msg(st, out, line, len);
369 } else if (len > 6 && ngx_strncmp(line, "pause ", 6) == 0) {
370 admin_cmd_pause(st, out, line, len);
371 } else if (len > 5 && ngx_strncmp(line, "cont ", 5) == 0) {
372 admin_cmd_cont(st, out, line, len);
373 } else if (len > 6 && ngx_strncmp(line, "abort ", 6) == 0) {
374 admin_cmd_abort(st, out, line, len);
375 } else {
376 admin_reply(st, out, "err unknown-command\n");
377 }
378}

393static void
394admin_write_handler(ngx_event_t *wev)
395{
396 ngx_connection_t *c = wev->data;
397 brix_admin_conn_t *st = c->data;
398 ssize_t n;
399
400 while (st->reply_sent < st->reply_len) {
401 n = c->send(c, st->reply + st->reply_sent,
402 st->reply_len - st->reply_sent);
403 if (n > 0) {
404 st->reply_sent += (size_t) n;
405 continue;
406 }
407 if (n == NGX_AGAIN) {
408 if (ngx_handle_write_event(wev, 0) != NGX_OK) {
409 admin_close(c);
410 }
411 return;
412 }
413 admin_close(c);
414 return;
415 }
416
417 /* Reply fully flushed — reset for the next command on the same conn. */
418 st->reply_len = 0;
419 st->reply_sent = 0;
420 st->cmd_len = 0;
421}

423static void
424admin_read_handler(ngx_event_t *rev)
425{
426 ngx_connection_t *c = rev->data;
427 brix_admin_conn_t *st = c->data;
428 ssize_t n;
429
430 for ( ;; ) {
431 if (st->cmd_len >= sizeof(st->cmd)) {
432 admin_close(c); /* oversized command line — refuse */
433 return;
434 }
435 n = c->recv(c, st->cmd + st->cmd_len, sizeof(st->cmd) - st->cmd_len);
436 if (n > 0) {
437 st->cmd_len += (size_t) n;
438 if (ngx_strlchr(st->cmd, st->cmd + st->cmd_len, '\n') != NULL) {
439 admin_dispatch(c, st);
440 admin_write_handler(c->write);
441 return; /* one command per read pass; next arrives as data */
442 }
443 continue;
444 }
445 if (n == NGX_AGAIN) {
446 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
447 admin_close(c);
448 }
449 return;
450 }
451 admin_close(c); /* EOF or error */
452 return;
453 }

456static void
457admin_accept_handler(ngx_event_t *rev)
458{
459 ngx_connection_t *lc = rev->data;
460 ngx_socket_t fd;
461 ngx_connection_t *c;
462 brix_admin_conn_t *st;
463
464 for ( ;; ) {
465 fd = accept(lc->fd, NULL, NULL);
466 if (fd == (ngx_socket_t) -1) {
467 return; /* EAGAIN or transient — nothing more to accept */
468 }
469 if (ngx_nonblocking(fd) == -1) {
470 (void) close(fd);
471 continue;
472 }
473 c = ngx_get_connection(fd, rev->log);
474 if (c == NULL) {
475 (void) close(fd);
476 continue;
477 }
478 /* ngx_get_connection() hands out a BARE connection: the I/O vtable is
479 * ngx_event_accept()'s job, so wire it here or c->recv/c->send are
480 * NULL function pointers. */
481 c->recv = ngx_recv;
482 c->send = ngx_send;
483 c->recv_chain = ngx_recv_chain;
484 c->send_chain = ngx_send_chain;
485 c->log = rev->log;
486 c->read->log = c->log;
487 c->write->log = c->log;
488 c->type = SOCK_STREAM;
489 c->pool = ngx_create_pool(2048, rev->log);
490 if (c->pool == NULL) {
491 ngx_close_connection(c);
492 continue;
493 }
494 st = ngx_pcalloc(c->pool, sizeof(*st));
495 if (st == NULL) {
496 admin_close(c);
497 continue;
498 }
499 c->data = st;
500 c->read->handler = admin_read_handler;
501 c->write->handler = admin_write_handler;
502 if (ngx_handle_read_event(c->read, 0) != NGX_OK) {
503 admin_close(c);
504 }
505 }

508void
509brix_admin_socket_init(ngx_cycle_t *cycle)
510{
511 struct sockaddr_un sa;
512 ngx_socket_t fd;
513 ngx_connection_t *lc;
514 char path[sizeof(sa.sun_path)];
515 int n;
516
517 if (s_admin_path[0] == '\0') {
518 return;
519 }
520
521 /*
522 * MULTI-WORKER REACH: every worker serves its OWN admin socket — worker 0
523 * at the configured <path>, worker n at "<path>.<n>". Sessions are per-
524 * worker (nginx process model), so each socket lists/controls exactly the
525 * sessions its worker owns; an admin tool sweeps the socket set. This is
526 * the natural mapping of stock's single-daemon adminpath onto nginx's
527 * process model — a documented divergence.
528 */
529 if (ngx_worker == 0) {
530 n = snprintf(path, sizeof(path), "%s", s_admin_path);
531 } else {
532 n = snprintf(path, sizeof(path), "%s.%lu", s_admin_path,
533 (unsigned long) ngx_worker);
534 }
535 if (n < 0 || (size_t) n >= sizeof(path)) {
536 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
537 "brix admin: per-worker path for \"%s\" exceeds unix "
538 "socket path limit — admin socket disabled on worker %ui",
539 s_admin_path, ngx_worker);
540 return;
541 }
542
543 (void) unlink(path); /* drop a stale socket from a prior run */ /* vfs-seam-allow: NOT_STORAGE — admin unix socket, not export namespace */
544
545 fd = socket(AF_UNIX, SOCK_STREAM, 0);
546 if (fd == (ngx_socket_t) -1) {
547 ngx_log_error(NGX_LOG_ERR, cycle->log, ngx_errno,
548 "brix admin: socket(%s) failed", path);
549 return;
550 }
551
552 ngx_memzero(&sa, sizeof(sa));
553 sa.sun_family = AF_UNIX;
554 ngx_memcpy(sa.sun_path, path, (size_t) n + 1);
555
556 if (bind(fd, (struct sockaddr *) &sa, sizeof(sa)) == -1
557 || chmod(path, 0600) == -1 /* vfs-seam-allow: NOT_STORAGE — admin socket 0600 privilege boundary */
558 || listen(fd, 8) == -1
559 || ngx_nonblocking(fd) == -1)
560 {
561 ngx_log_error(NGX_LOG_ERR, cycle->log, ngx_errno,
562 "brix admin: bind/listen(%s) failed", path);
563 (void) close(fd);
564 return;
565 }
566
567 lc = ngx_get_connection(fd, cycle->log);
568 if (lc == NULL) {
569 (void) close(fd);
570 return;
571 }
572 lc->read->handler = admin_accept_handler;
573 if (ngx_handle_read_event(lc->read, 0) != NGX_OK) {
574 ngx_close_connection(lc);
575 return;
576 }
577
578 ngx_log_error(NGX_LOG_INFO, cycle->log, 0,
579 "brix admin: listening on \"%s\" (worker %ui)", path,
580 ngx_worker);
581}
