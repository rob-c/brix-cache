12/* Return 1 if `path` is at or beneath `prefix` (component-aligned prefix match). */
13static ngx_flag_t
14brix_path_prefix_match(const char *prefix, const char *path)
15{
16 size_t prefix_len;
17
18 if (prefix == NULL || path == NULL) {
19 return 0;
20 }
21
22 prefix_len = strlen(prefix);
23 if (strncmp(prefix, path, prefix_len) != 0) {
24 return 0;
25 }
26
27 /* Require the match to fall on a path-component boundary so that prefix
28 * "/foo" matches "/foo" (exact) and "/foo/bar" (child) but NOT "/foobar".
29 * The char immediately after the prefix in path must therefore be either
30 * end-of-string or a '/' separator. */
31 return path[prefix_len] == '\0' || path[prefix_len] == '/';
32}
33/* Compare the first `bits` of two packed addresses — the CIDR prefix-match core. */
34static ngx_flag_t
35brix_authdb_addr_prefix_match(const u_char *a, const u_char *b,
36 ngx_uint_t bits)
37{
38 ngx_uint_t full;
39 ngx_uint_t rem;
40
41 /* Compare two raw addresses (4 bytes IPv4 / 16 bytes IPv6) under a /bits
42 * CIDR mask. Split the prefix length into whole bytes plus leftover bits:
43 * the first `full` bytes must match exactly, and the next byte must match
44 * only on its top `rem` bits. */
45 full = bits / 8;
46 rem = bits % 8;
47
48 if (full > 0 && ngx_memcmp(a, b, full) != 0) {
49 return 0;
50 }
51
52 if (rem != 0) {
53 /* Top `rem` bits set: e.g. rem=3 -> 0xff << 5 -> 0b11100000. Mask both
54 * addresses' partial byte and compare only the significant high bits. */
55 u_char mask = (u_char) (0xffu << (8 - rem));
56 if ((a[full] & mask) != (b[full] & mask)) {
57 return 0;
58 }
59 }
60
61 return 1;
62}
63/* Match a peer IP against a rule's host CIDR (rule_id is host/addr[/bits]). */
64static ngx_flag_t
65brix_authdb_host_cidr_match(const char *rule_id, const char *peer_ip)
66{
67 char cidr[128];
68 char *slash;
69 char *end;
70 long bits;
71 u_char rule_addr[16];
72 u_char peer_addr[16];
73 int family;
74 ngx_uint_t max_bits;
75
76 if (rule_id == NULL || peer_ip == NULL || peer_ip[0] == '\0') {
77 return 0;
78 }
79
80 /* Reject anything that would not fit (with NUL) in the local cidr buffer;
81 * this bounds the copy below and prevents reading a truncated CIDR. */
82 if (strlen(rule_id) >= sizeof(cidr)) {
83 return 0;
84 }
85
86 /* Work on a private mutable copy so we can split "addr/bits" in place. */
87 ngx_cpystrn((u_char *) cidr, (u_char *) rule_id, sizeof(cidr));
88 slash = strchr(cidr, '/');
89 if (slash == NULL) {
90 /* No '/': rule_id is a bare address -> require an exact string match
91 * against the peer IP (no masking). */
92 return strcmp(rule_id, peer_ip) == 0;
93 }
94
95 /* Split at '/': cidr now holds the address, slash holds the bit count. */
96 *slash++ = '\0';
97 /* Infer family from the address half; a ':' can only appear in IPv6. */
98 if (strchr(cidr, ':') != NULL) {
99 family = AF_INET6;
100 max_bits = 128;
101 } else {
102 family = AF_INET;
103 max_bits = 32;
104 }
105
106 /* Parse and validate the prefix length: must be a complete integer
107 * (end consumed entirely, slash actually had digits) within [0, max_bits].
108 * Any junk or out-of-range value rejects the rule rather than matching. */
109 bits = strtol(slash, &end, 10);
110 if (end == slash || *end != '\0' || bits < 0
111 || (ngx_uint_t) bits > max_bits)
112 {
113 return 0;
114 }
115
116 if (inet_pton(family, cidr, rule_addr) != 1
117 || inet_pton(family, peer_ip, peer_addr) != 1)
118 {
119 return 0;
120 }
121
122 return brix_authdb_addr_prefix_match(rule_addr, peer_addr,
123 (ngx_uint_t) bits);
124}

204/* WHAT: 1 if ONE selector of a rule applies to this subject; else 0.
205 * WHY: table-flat dispatch over the selector alphabet, so the rule-level
206 * matcher below is a plain AND-loop. Pure — no I/O, no mutation.
207 * HOW: `a` matches unconditionally; `u` compares the DN; `g` the VO list;
208 * `p` the peer address (exact, wildcard or CIDR); `v` and `l` the
209 * FQAN-derived vorg/role CSVs (2.0 F20). Any unknown selector does not
210 * match (default-deny) — the parser refuses those lines, so reaching the
211 * default arm means memory was corrupted and denying is the only safe
212 * answer. Semantics are frozen against the multi-user conformance suite;
213 * do not alter the per-selector rules. */
214static ngx_flag_t
215adb_selector_matches(brix_auth_type_t sel, const ngx_str_t *id,
216 const adb_subject_t *subj)
217{
218 switch (sel) {
219 case BRIX_AUTH_ALL:
220 return 1;
221 case BRIX_AUTH_USER:
222 return adb_user_matches(id, subj->dn);
223 case BRIX_AUTH_GROUP:
224 return adb_group_matches(id, subj->vo_list);
225 case BRIX_AUTHDB_HOST:
226 return brix_authdb_host_match(id, subj->peer_ip);
227 case BRIX_AUTH_VORG:
228 return adb_csv_matches(id, subj->vorg);
229 case BRIX_AUTH_ROLE:
230 return adb_csv_matches(id, subj->role);
231 default:
232 return 0;
233 }
234}

236/* WHAT: 1 when `a_id` and `b_id` are satisfied at the SAME index of the two
237 * index-aligned CSVs; else 0.
238 * WHY: 2.0 F20 — this is the security-load-bearing step the XrdAcc engine
239 * already performs on its (vorg, role, grup) tuples
240 * (acc/entity.c::acc_entity_fill_tuples): a credential holding
241 * cms/Role=NULL and atlas/Role=production must NOT satisfy
242 * `vl cms|production`. Testing the two CSVs independently would grant
243 * exactly that cross-product, which is a widening — the very shape F20
244 * exists to remove.
245 * HOW: walk both CSVs with one cursor each, comparing field i of one against
246 * field i of the other; brix_identity_derive_attrs emits one field per
247 * FQAN into each CSV (empty fields preserved), so the indices align by
248 * construction. Stop at the shorter list. */
249static ngx_flag_t
250adb_pair_matches(const char *a_csv, const ngx_str_t *a_id,
251 const char *b_csv, const ngx_str_t *b_id)
252{
253 const char *ap = a_csv;
254 const char *bp = b_csv;
255
256 if (a_csv == NULL || b_csv == NULL) {
257 return 0;
258 }
259
260 for (;;) {
261 const char *ae = strchr(ap, ',');
262 const char *be = strchr(bp, ',');
263 size_t alen = (ae != NULL) ? (size_t) (ae - ap) : strlen(ap);
264 size_t blen = (be != NULL) ? (size_t) (be - bp) : strlen(bp);
265
266 if (alen == a_id->len && ngx_strncmp(ap, a_id->data, alen) == 0
267 && blen == b_id->len && ngx_strncmp(bp, b_id->data, blen) == 0)
268 {
269 return 1;
270 }
271 if (ae == NULL || be == NULL) {
272 return 0;
273 }
274 ap = ae + 1;
275 bp = be + 1;
276 }
277}

279/* WHAT: 1 if EVERY selector of `rule` applies to this subject; else 0.
280 * WHY: 2.0 F20 — field 1 may carry up to BRIX_AUTHDB_MAX_SELECTORS letters and
281 * they are AND-ed, so a compound rule is strictly NARROWER than any of
282 * its selectors alone. Conjunction is the only safe reading: the pre-F20
283 * parser truncated the token and matched on the lead selector alone,
284 * which admitted subjects the operator had excluded.
285 * HOW: short-circuit AND over sel[0..nsel). `v` and `l` are held back and, when
286 * BOTH appear, resolved together by adb_pair_matches so the vorg and the
287 * role must come from the SAME FQAN — matching them independently would
288 * let two credentials be combined into one the operator never granted.
289 * Every other selector is scalar and matches independently. nsel is >= 1
290 * for every parsed rule; a zeroed rule (nsel == 0) matches nothing. */
291static ngx_flag_t
292adb_identity_matches(const brix_authdb_rule_t *rule,
293 const adb_subject_t *subj)
294{
295 const ngx_str_t *vorg_id = NULL;
296 const ngx_str_t *role_id = NULL;
297 ngx_uint_t i;
298
299 if (rule->nsel == 0) {
300 return 0;
301 }
302
303 for (i = 0; i < rule->nsel; i++) {
304 if (rule->sel[i] == BRIX_AUTH_VORG) {
305 vorg_id = &rule->sel_id[i];
306 } else if (rule->sel[i] == BRIX_AUTH_ROLE) {
307 role_id = &rule->sel_id[i];
308 } else if (!adb_selector_matches(rule->sel[i], &rule->sel_id[i], subj)) {
309 return 0;
310 }
311 }
312
313 if (vorg_id != NULL && role_id != NULL) {
314 return adb_pair_matches(subj->vorg, vorg_id, subj->role, role_id);
315 }
316 if (vorg_id != NULL) {
317 return adb_selector_matches(BRIX_AUTH_VORG, vorg_id, subj);
318 }
319 if (role_id != NULL) {
320 return adb_selector_matches(BRIX_AUTH_ROLE, role_id, subj);
321 }
322 return 1;
323}

325/* Find the authdb rule granting `needed_privs` on resolved_path for `identity`;
326 * returns the matching rule or NULL. */
327const brix_authdb_rule_t *
328brix_find_authdb_rule_identity(const char *resolved_path, ngx_array_t *rules,
329 const brix_identity_t *identity,
330 const char *peer_ip, uint32_t needed_privs)
331{
332 const brix_authdb_rule_t *best = NULL;
333 brix_authdb_rule_t *rule;
334 size_t best_len = 0;
335 ngx_uint_t i;
336 adb_subject_t subj;
337
338 if (resolved_path == NULL || rules == NULL) {
339 return NULL;
340 }
341
342 /* The `v`/`l` views come from brix_identity_derive_attrs, which splits the
343 * VOMS FQANs for EVERY identity (not only under the xrdacc engine), so a
344 * compound rule sees the same attributes the xrdacc engine would. */
345 subj.dn = brix_identity_dn_cstr(identity);
346 subj.vo_list = brix_identity_vo_csv_cstr(identity);
347 subj.vorg = brix_identity_acc_vorg_cstr(identity);
348 subj.role = brix_identity_acc_role_cstr(identity);
349 subj.peer_ip = peer_ip;
350
351 rule = rules->elts;
352 for (i = 0; i < rules->nelts; i++) {
353 size_t rule_len = strlen(rule[i].resolved);
354
355 if (!brix_path_prefix_match(rule[i].resolved, resolved_path)) {
356 continue;
357 }
358
359 if (!adb_identity_matches(&rule[i], &subj)) {
360 continue;
361 }
362
363 /* Path + identity match: now require the rule to grant ALL needed bits
364 * ((privs & needed) == needed). Only sufficient rules compete; among
365 * them the longest prefix wins, with >= letting a later equal-length
366 * rule override an earlier one (config last-wins on ties). This is the
367 * XRootD longest-prefix semantics — frozen against the conformance suite. */
368 if ((rule[i].privs & needed_privs) == needed_privs) {
369 if (rule_len >= best_len) {
370 best = &rule[i];
371 best_len = rule_len;
372 }
373 }
374 }
375
376 return best;
377}

409/* Authorize `query->identity` for query->needed_privs on query->resolved_path
410 * against query->rules. Returns NGX_OK (granted) or NGX_ERROR (denied). */
411ngx_int_t
412brix_check_authdb_identity(ngx_log_t *log, const brix_authdb_query_t *query)
413{
414 const brix_authdb_rule_t *rule;
415 char safe_path[512];
416 const char *dn;
417 const char *vo_list;
418
419 /* No rules loaded == no authdb policy == unrestricted (allow-all). */
420 if (query->rules == NULL || query->rules->nelts == 0) {
421 return NGX_OK;
422 }
423
424 rule = brix_find_authdb_rule_identity(query->resolved_path, query->rules,

429 }
430
431 dn = brix_identity_dn_cstr(query->identity);

435 ngx_log_error(NGX_LOG_WARN, log, 0,
436 "brix: authdb denied path=\"%s\" privs=0x%02xd "
437 "dn=\"%s\" vos=\"%s\" peer=\"%s\"",
438 safe_path, query->needed_privs, dn,
439 vo_list[0] ? vo_list : "-",
440 (query->peer_ip != NULL && query->peer_ip[0])
441 ? query->peer_ip : "-");

443 return NGX_ERROR;
444}
