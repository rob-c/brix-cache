1/*
2 * prepare_recall.c — the W6 (phase-107 C2) per-path arms of kXR_prepare:
3 * prestage through brix_vfs_recall() and evict through brix_vfs_evict().
4 *
5 * WHAT: brix_prepare_recall_one() wraps the VFS recall verb in the stage
6 * REGISTRY lifecycle the VFS deliberately does not own (vfs_recall.c
7 * banner): join an existing record for the same path (one record, one
8 * reqid, however many clients ask), else record BEFORE the driver call,
9 * and delete the record on a synchronous driver failure so a failed
10 * recall never leaves a reqid a client can poll forever.
11 * brix_prepare_evict_one() runs the FRM-1 ownership check (a path bound
12 * to a live stage record may only be evicted by that record's creator)
13 * and then drops the online copy at the top of the driver chain.
14 *
15 * WHY: Until W6, prestage ran only as the fork/exec prepare_command while
16 * every wire-native recall slot sat implemented and uncalled, and the
17 * kXR_prepare evict arm was a documented no-op. prepare_command stays
18 * supported as the FALLBACK for a driver with no recall slot (ENOTSUP),
19 * exactly like the durable FRM engine: both are stagers the registry
20 * record hands the work to. Only a nearline export with NEITHER fallback
21 * refuses (kXR_Unsupported) — a flat disk export keeps its historical
22 * advisory success, because there staging is a no-op, not a failure.
23 *
24 * HOW: Both arms build the same VFS ctx every root:// namespace mutation
25 * builds (op_vfs_ctx: identity + per-user backend credential policy +
26 * phase-70 bearer passthrough), so the recall_cred/evict_cred twins see
27 * the real requester. The read-only refusal (kXR_fsReadOnly, nothing
28 * recorded) fires in prepare_dispatch_special BEFORE the scan; the
29 * policy kernel inside the VFS verbs is the belt behind that edge gate.
30 */

73/* Record-before-driver-call: create the durable registry record for
74 * `out_resolved` (owner = the caller's stable key) so a concurrent second
75 * request JOINS instead of duplicating. Returns 1 with `rq` filled when a
76 * record was created, 0 otherwise (creation failure is logged best-effort,
77 * matching the pre-W6 enqueue). */
78static int
79prepare_recall_record(brix_ctx_t *ctx, ngx_connection_t *c,
80 prepare_scan_t *sc, brix_stage_registry_t *reg, char *out_resolved,
81 char rq[BRIX_STAGE_REQID_LEN])
82{
83 brix_stage_request_view_t v;
84 char owner_key[BRIX_PREPARE_OWNER_KEY_MAX];
85 const char *rdn = brix_prepare_owner_key(ctx, owner_key,
86 sizeof(owner_key));
87
88 ngx_memzero(&v, sizeof(v));
89 v.lfn = out_resolved;
90 v.requester_dn = (rdn != NULL && rdn[0] != '\0') ? rdn : NULL;
91 v.tod_expire = (int64_t) time(NULL)
92 + (int64_t) (sc->conf->frm.stage_ttl / 1000);
93
94 if (brix_stage_request_add(reg, &v, rq, BRIX_STAGE_REQID_LEN, c->log)
95 != NGX_OK)
96 {
97 ngx_log_error(NGX_LOG_ERR, c->log, 0,
98 "brix: stage request add failed for \"%s\"",
99 out_resolved);
100 rq[0] = '\0';
101 return 0;
102 }
103 prepare_adopt_reqid(sc, rq);
104 return 1;
105}

107/* The ENOTSUP arm: no nearline tier with a recall slot in the chain. The
108 * request still stages when a fallback stager exists — the legacy
109 * prepare_command (collect the path for it) or the durable FRM engine (the
110 * record just created IS its work queue). A nearline export with neither can
111 * never stage and says so now (kXR_Unsupported) instead of at the first read;
112 * a flat export keeps its historical advisory success. */
113static ngx_int_t
114prepare_recall_enotsup(brix_ctx_t *ctx, ngx_connection_t *c,
115 prepare_scan_t *sc, brix_vfs_ctx_t *vctx, char *out_resolved,
116 int have_record)
117{
118 if (sc->collect_stage && sc->cmd_count < sc->stage_max) {
119 sc->stage_paths[sc->cmd_count++] = out_resolved;
120 return NGX_OK; /* prepare_command is the stager */
121 }
122 if (have_record) {
123 return NGX_OK; /* the durable FRM engine is the stager */
124 }
125 if (brix_vfs_nearline_export(vctx)) {
126 (void) brix_prepare_send_fail(ctx, c, out_resolved, kXR_Unsupported,
127 "nearline export cannot stage: driver has no recall slot and "
128 "no prepare command is configured");
129 return NGX_DONE;
130 }
131 return NGX_OK; /* flat export: advisory success */
132}

134ngx_int_t
135brix_prepare_recall_one(brix_ctx_t *ctx, ngx_connection_t *c,
136 prepare_scan_t *sc, char *out_resolved)
137{
138 brix_vfs_ctx_t vctx;
139 brix_stage_registry_t *reg = NULL;
140 char rq[BRIX_STAGE_REQID_LEN];
141 char drv_rq[40];
142 ngx_int_t rc;
143 int have_record = 0, e;
144
145 rq[0] = '\0';
146 if (sc->do_enqueue) {
147 reg = brix_stage_registry_singleton();
148 }
149
150 if (reg != NULL
151 && brix_stage_request_find_by_path(reg, out_resolved, rq, sizeof(rq),
152 c->log) == NGX_OK)
153 {
154 brix_stage_request_t old;
155
156 /* JOIN (kXR_prepare idempotency): a live record for the same path
157 * absorbs the request — same reqid, one record, status unchanged.
158 * The joined outcome is booked HERE, not by the VFS (vfs_recall.c).
159 * A retired record (FAILED/CANCELLED — find_by_path matches any
160 * non-free slot) never absorbs: the fresh request must retry, so it
161 * falls through to record-and-recall and becomes the newest record. */
162 if (brix_stage_request_get(reg, rq, &old, c->log) == NGX_OK
163 && old.status != BRIX_STAGE_REQ_FAILED
164 && old.status != BRIX_STAGE_REQ_CANCELLED)
165 {
166 brix_metric_vfs_recall(BRIX_VFS_RECALL_JOINED);
167 prepare_adopt_reqid(sc, rq);
168 return NGX_OK;
169 }
170 rq[0] = '\0';
171 }
172 if (reg != NULL) {
173 have_record = prepare_recall_record(ctx, c, sc, reg, out_resolved, rq);
174 }
175
176 brix_prepare_vfs_ctx(ctx, c, sc->conf, out_resolved, &vctx);
177 rc = brix_vfs_recall(&vctx, drv_rq);
178
179 if (rc == NGX_OK) { /* already online — nothing to stage */
180 if (have_record) {
181 (void) brix_stage_request_set_status(reg, rq,
182 BRIX_STAGE_REQ_DONE, c->log);
183 }
184 return NGX_OK;
185 }
186 if (rc == NGX_AGAIN) { /* queued at the driver */
187 if (!have_record) {
188 prepare_adopt_reqid(sc, drv_rq); /* registry-less: the driver's
189 * own parking handle, if any */
190 }
191 return NGX_OK;
192 }
193
194 e = errno;
195 if (e == ENOTSUP) {
196 return prepare_recall_enotsup(ctx, c, sc, &vctx, out_resolved,
197 have_record);
198 }
199 /* Synchronous driver failure: delete the record so no orphan reqid is
200 * pollable (kXR_QPrep on it answers unknown, never "queued forever"). */
201 if (have_record) {
202 (void) brix_stage_request_delete(reg, rq, c->log);
203 }
204 ngx_log_error(NGX_LOG_ERR, c->log, e,
205 "brix: prestage recall failed for \"%s\"", out_resolved);
206 return NGX_OK; /* best-effort per path, like the
207 * pre-W6 staging-command launch */
208}

210ngx_int_t
211brix_prepare_evict_one(brix_ctx_t *ctx, ngx_connection_t *c,
212 prepare_scan_t *sc, char *out_resolved)
213{
214 brix_vfs_ctx_t vctx;
215 brix_stage_registry_t *reg = NULL;
216 uint64_t bytes = 0;
217 char rq[BRIX_STAGE_REQID_LEN];
218 char owner_key[BRIX_PREPARE_OWNER_KEY_MAX];
219 int have_record = 0;
220
221 if (sc->conf->frm.enable) {
222 reg = brix_stage_registry_singleton();
223 }
224
225 /* FRM-1 ownership: an evict of a path bound to a live stage record is an
226 * act on that record — only its creator may drop the copy it recalled.
227 * Runs AFTER the read-only edge gate (a read-only export answered EROFS
228 * before disclosing whose record exists) and BEFORE the driver. */
229 if (reg != NULL
230 && brix_stage_request_find_by_path(reg, out_resolved, rq, sizeof(rq),
231 c->log) == NGX_OK)
232 {
233 have_record = 1;
234 if (brix_stage_request_owner_check(reg, rq,
235 brix_prepare_owner_key(ctx, owner_key, sizeof(owner_key)),
236 c->log) != NGX_OK)
237 {
238 brix_log_access(ctx, c, "PREPARE", out_resolved, "evict-denied",
239 0, kXR_NotAuthorized, NULL, 0);
240 (void) brix_prepare_send_fail(ctx, c, out_resolved,
241 kXR_NotAuthorized, "not the owner of this staged path");
242 return NGX_DONE;
243 }
244 }
245
246 brix_prepare_vfs_ctx(ctx, c, sc->conf, out_resolved, &vctx);
247 if (brix_vfs_evict(&vctx, &bytes) != NGX_OK) {
248 int e = errno;
249
250 if (e == ENOTSUP) {
251 return NGX_OK; /* no evictable tier: the online copy is
252 * the only copy — advisory ok, exactly
253 * the pre-W6 noop (stock parity) */
254 }
255 (void) brix_prepare_send_fail(ctx, c, out_resolved,
256 brix_kxr_from_errno(e), "evict failed");
257 return NGX_DONE;
258 }
259
260 if (have_record) {
261 /* The online copy is gone: retire the record so kXR_QPrep answers from
262 * residency truth, not a stale DONE. */
263 (void) brix_stage_request_delete(reg, rq, c->log);
264 }
265 ngx_log_error(NGX_LOG_INFO, c->log, 0,
266 "brix: prepare evict \"%s\": %uL bytes released",
267 out_resolved, bytes);
268 return NGX_OK;
269}
