1/*
2 * prepare_recall.c — the W6 (phase-107 C2) per-path arms of kXR_prepare:
3 * prestage through brix_vfs_recall() and evict through brix_vfs_evict().
4 *
5 * WHAT: brix_prepare_recall_one() wraps the VFS recall verb in the stage
6 * REGISTRY lifecycle the VFS deliberately does not own (vfs_recall.c
7 * banner): join an existing record for the same path (one record, one
8 * reqid, however many clients ask), else record BEFORE the driver call,
9 * and delete the record on a synchronous driver failure so a failed
10 * recall never leaves a reqid a client can poll forever.
11 * brix_prepare_evict_one() runs the FRM-1 ownership check (a path bound
12 * to a live stage record may only be evicted by that record's creator)
13 * and then drops the online copy at the top of the driver chain.
14 *
15 * WHY: Until W6, prestage ran only as the fork/exec prepare_command while
16 * every wire-native recall slot sat implemented and uncalled, and the
17 * kXR_prepare evict arm was a documented no-op. prepare_command stays
18 * supported as the FALLBACK for a driver with no recall slot (ENOTSUP),
19 * exactly like the durable FRM engine: both are stagers the registry
20 * record hands the work to. Only a nearline export with NEITHER fallback
21 * refuses (kXR_Unsupported) — a flat disk export keeps its historical
22 * advisory success, because there staging is a no-op, not a failure.
23 *
24 * HOW: Both arms build the same VFS ctx every root:// namespace mutation
25 * builds (op_vfs_ctx: identity + per-user backend credential policy +
26 * phase-70 bearer passthrough), so the recall_cred/evict_cred twins see
27 * the real requester. The read-only refusal (kXR_fsReadOnly, nothing
28 * recorded) fires in prepare_dispatch_special BEFORE the scan; the
29 * policy kernel inside the VFS verbs is the belt behind that edge gate.
30 */

74/* Record-before-driver-call: create the durable registry record for
75 * `out_resolved` (owner = the caller's stable key) so a concurrent second
76 * request JOINS instead of duplicating. Returns 1 with `rq` filled when a
77 * record was created, 0 otherwise (creation failure is logged best-effort,
78 * matching the pre-W6 enqueue). */
79static int
80prepare_recall_record(brix_ctx_t *ctx, ngx_connection_t *c,
81 prepare_scan_t *sc, brix_stage_registry_t *reg, char *out_resolved,
82 char rq[BRIX_STAGE_REQID_LEN])
83{
84 brix_stage_request_view_t v;
85 char owner_key[BRIX_PREPARE_OWNER_KEY_MAX];
86 const char *rdn = brix_prepare_owner_key(ctx, owner_key,
87 sizeof(owner_key));
88
89 ngx_memzero(&v, sizeof(v));
90 v.lfn = out_resolved;
91 v.requester_dn = (rdn != NULL && rdn[0] != '\0') ? rdn : NULL;
92 v.tod_expire = (int64_t) time(NULL)
93 + (int64_t) (sc->conf->frm.stage_ttl / BRIX_ROOT_MS_TO_SEC);
94
95 if (brix_stage_request_add(reg, &v, rq, BRIX_STAGE_REQID_LEN, c->log)
96 != NGX_OK)
97 {
98 BRIX_LOG(NGX_LOG_ERR, c->log, 0,
99 "brix: stage request add failed for \"%s\"",
100 out_resolved);
101 rq[0] = '\0';
102 return 0;
103 }
104 prepare_adopt_reqid(sc, rq);
105 return 1;
106}

108/* The ENOTSUP arm: no nearline tier with a recall slot in the chain. The
109 * request still stages when a fallback stager exists — the legacy
110 * prepare_command (collect the path for it) or the durable FRM engine (the
111 * record just created IS its work queue). A nearline export with neither can
112 * never stage and says so now (kXR_Unsupported) instead of at the first read;
113 * a flat export keeps its historical advisory success. */
114static ngx_int_t
115prepare_recall_enotsup(brix_ctx_t *ctx, ngx_connection_t *c,
116 prepare_scan_t *sc, brix_vfs_ctx_t *vctx, char *out_resolved,
117 int have_record)
118{
119 if (sc->collect_stage && sc->cmd_count < sc->stage_max) {
120 sc->stage_paths[sc->cmd_count++] = out_resolved;
121 return NGX_OK; /* prepare_command is the stager */
122 }
123 if (have_record) {
124 return NGX_OK; /* the durable FRM engine is the stager */
125 }
126 if (brix_vfs_nearline_export(vctx)) {
127 (void) brix_prepare_send_fail(ctx, c, out_resolved, kXR_Unsupported,
128 "nearline export cannot stage: driver has no recall slot and "
129 "no prepare command is configured");
130 return NGX_DONE;
131 }
132 return NGX_OK; /* flat export: advisory success */
133}

135ngx_int_t
136brix_prepare_recall_one(brix_ctx_t *ctx, ngx_connection_t *c,
137 prepare_scan_t *sc, char *out_resolved)
138{
139 brix_vfs_ctx_t vctx;
140 brix_stage_registry_t *reg = NULL;
141 char rq[BRIX_STAGE_REQID_LEN];
142 char drv_rq[40];
143 ngx_int_t rc;
144 int have_record = 0, e;
145
146 rq[0] = '\0';
147 if (sc->do_enqueue) {
148 reg = brix_stage_registry_singleton();
149 }
150
151 if (reg != NULL
152 && brix_stage_request_find_by_path(reg, out_resolved, rq, sizeof(rq),
153 c->log) == NGX_OK)
154 {
155 brix_stage_request_t old;
156
157 /* JOIN (kXR_prepare idempotency): a live record for the same path
158 * absorbs the request — same reqid, one record, status unchanged.
159 * The joined outcome is booked HERE, not by the VFS (vfs_recall.c).
160 * A retired record (FAILED/CANCELLED — find_by_path matches any
161 * non-free slot) never absorbs: the fresh request must retry, so it
162 * falls through to record-and-recall and becomes the newest record. */
163 if (brix_stage_request_get(reg, rq, &old, c->log) == NGX_OK
164 && old.status != BRIX_STAGE_REQ_FAILED
165 && old.status != BRIX_STAGE_REQ_CANCELLED)
166 {
167 brix_metric_vfs_recall(BRIX_VFS_RECALL_JOINED);
168 prepare_adopt_reqid(sc, rq);
169 return NGX_OK;
170 }
171 rq[0] = '\0';
172 }
173 if (reg != NULL) {
174 have_record = prepare_recall_record(ctx, c, sc, reg, out_resolved, rq);
175 }
176
177 brix_prepare_vfs_ctx(ctx, c, sc->conf, out_resolved, &vctx);
178 rc = brix_vfs_recall(&vctx, drv_rq);
179
180 if (rc == NGX_OK) { /* already online — nothing to stage */
181 if (have_record) {
182 (void) brix_stage_request_set_status(reg, rq,
183 BRIX_STAGE_REQ_DONE, c->log);
184 }
185 return NGX_OK;
186 }
187 if (rc == NGX_AGAIN) { /* queued at the driver */
188 if (!have_record) {
189 prepare_adopt_reqid(sc, drv_rq); /* registry-less: the driver's
190 * own parking handle, if any */
191 }
192 return NGX_OK;
193 }
194
195 e = errno;
196 if (e == ENOTSUP) {
197 return prepare_recall_enotsup(ctx, c, sc, &vctx, out_resolved,
198 have_record);
199 }
200 /* Synchronous driver failure: delete the record so no orphan reqid is
201 * pollable (kXR_QPrep on it answers unknown, never "queued forever"). */
202 if (have_record) {
203 (void) brix_stage_request_delete(reg, rq, c->log);
204 }
205 BRIX_LOG(NGX_LOG_ERR, c->log, e,
206 "brix: prestage recall failed for \"%s\"", out_resolved);
207 return NGX_OK; /* best-effort per path, like the
208 * pre-W6 staging-command launch */
209}

211ngx_int_t
212brix_prepare_evict_one(brix_ctx_t *ctx, ngx_connection_t *c,
213 prepare_scan_t *sc, char *out_resolved)
214{
215 brix_vfs_ctx_t vctx;
216 brix_stage_registry_t *reg = NULL;
217 uint64_t bytes = 0;
218 char rq[BRIX_STAGE_REQID_LEN];
219 char owner_key[BRIX_PREPARE_OWNER_KEY_MAX];
220 int have_record = 0;
221
222 if (sc->conf->frm.enable) {
223 reg = brix_stage_registry_singleton();
224 }
225
226 /* FRM-1 ownership: an evict of a path bound to a live stage record is an
227 * act on that record — only its creator may drop the copy it recalled.
228 * Runs AFTER the read-only edge gate (a read-only export answered EROFS
229 * before disclosing whose record exists) and BEFORE the driver. */
230 if (reg != NULL
231 && brix_stage_request_find_by_path(reg, out_resolved, rq, sizeof(rq),
232 c->log) == NGX_OK)
233 {
234 have_record = 1;
235 if (brix_stage_request_owner_check(reg, rq,
236 brix_prepare_owner_key(ctx, owner_key, sizeof(owner_key)),
237 c->log) != NGX_OK)
238 {
239 brix_log_access(ctx, c, "PREPARE", out_resolved, "evict-denied",
240 0, kXR_NotAuthorized, NULL, 0);
241 (void) brix_prepare_send_fail(ctx, c, out_resolved,
242 kXR_NotAuthorized, "not the owner of this staged path");
243 return NGX_DONE;
244 }
245 }
246
247 brix_prepare_vfs_ctx(ctx, c, sc->conf, out_resolved, &vctx);
248 if (brix_vfs_evict(&vctx, &bytes) != NGX_OK) {
249 int e = errno;
250
251 if (e == ENOTSUP) {
252 return NGX_OK; /* no evictable tier: the online copy is
253 * the only copy — advisory ok, exactly
254 * the pre-W6 noop (stock parity) */
255 }
256 (void) brix_prepare_send_fail(ctx, c, out_resolved,
257 brix_kxr_from_errno(e), "evict failed");
258 return NGX_DONE;
259 }
260
261 if (have_record) {
262 /* The online copy is gone: retire the record so kXR_QPrep answers from
263 * residency truth, not a stale DONE. */
264 (void) brix_stage_request_delete(reg, rq, c->log);
265 }
266 BRIX_LOG(NGX_LOG_INFO, c->log, 0,
267 "brix: prepare evict \"%s\": %uL bytes released",
268 out_resolved, bytes);
269 return NGX_OK;
270}

272/*
273 * Accept one authorized, resolved prepare path through its selected VFS arms.
274 *
275 * WHAT: Runs evict, recall, or eviction followed by recall for stage+fresh;
276 * returns NGX_OK to continue the scan or the first terminal response result.
277 *
278 * WHY: Fresh must retire both the resident bytes and their live registry
279 * record before recall checks for a JOIN, otherwise a stale DONE record makes
280 * the supposedly fresh request an immediate no-op.
281 *
282 * HOW: Fresh composes the existing ownership-checked eviction and recall arms
283 * in order. Plain evict and stage dispatch to one arm; an advisory request with
284 * neither mutation succeeds without storage work.
285 */
286ngx_int_t
287brix_prepare_accept_resolved(brix_ctx_t *ctx, ngx_connection_t *c,
288 prepare_scan_t *sc, char *out_resolved)
289{
290 if (sc->do_fresh) {
291 ngx_int_t rc;
292
293 rc = brix_prepare_evict_one(ctx, c, sc, out_resolved);
294 if (rc != NGX_OK) {
295 return rc;
296 }
297 return brix_prepare_recall_one(ctx, c, sc, out_resolved);
298 }
299 if (sc->do_evict) {
300 return brix_prepare_evict_one(ctx, c, sc, out_resolved);
301 }
302 if (sc->do_stage) {
303 return brix_prepare_recall_one(ctx, c, sc, out_resolved);
304 }
305 return NGX_OK;
306}
