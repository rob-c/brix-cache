1/*
2 * config.c — the `brix_protbind` directive parser.
3 *
4 * WHAT: Turns one `brix_protbind <host-template> [none | [only] <proto>...]`
5 * directive into a brix_protbind_rule_t appended to a caller-owned rule
6 * array, or into an nginx configuration error string.
7 *
8 * WHY: Both the root:// stream server block and the HTTP location block accept
9 * the same grammar, so the parse — and the emerg diagnostic, which names
10 * the directive via cmd->name — lives once here and each module supplies
11 * only a one-line setter that hands over the address of its own array
12 * slot. Keeping the grammar in one file is also what makes the
13 * stream-vs-HTTP behaviour provably identical.
14 *
15 * HOW: Validates arity, records the template verbatim (matching happens at
16 * request time), then either records the `none` mode or walks the
17 * protocol tokens through brix_protbind_proto_id(), rejecting unknown
18 * names, duplicates and over-long lists at config time so a misconfigured
19 * server never starts.
20 */
21
22#include "core/ngx_brix_module.h"
23#include "protbind.h"

25/*
26 * brix_protbind_rules_array — lazily create the rule array.
27 *
28 * WHAT: Returns the existing array, creating a small one on first use; NULL on
29 * allocation failure.
30 *
31 * WHY: A NULL array is the "no protbind configured" sentinel the resolver and
32 * the config merge both rely on, so the array must not be pre-created by
33 * the module's create_conf. Four slots covers essentially every real
34 * stanza without a realloc.
35 *
36 * HOW: ngx_array_create from cf->pool on first call.
37 */
38static ngx_array_t *
39brix_protbind_rules_array(ngx_conf_t *cf, ngx_array_t **rules)
40{
41 if (*rules == NULL) {
42 *rules = ngx_array_create(cf->pool, 4, sizeof(brix_protbind_rule_t));
43 }
44
45 return *rules;
46}

48/*
49 * brix_protbind_parse_protos — parse the protocol-token tail of a directive.
50 *
51 * WHAT: Fills rule->protos/count from args[first..last]; returns NGX_CONF_OK
52 * or a static error string.
53 *
54 * WHY: Every rejection here is a misconfiguration that would otherwise fail
55 * open (an unknown name silently dropped) or produce a malformed sec
56 * token (a duplicated scheme). Failing at config parse turns all of them
57 * into a refusal to start, which is the only safe outcome for an
58 * authentication policy.
59 *
60 * HOW: 1. Map each token to its BRIX_AUTH_* id, rejecting unknown names.
61 * 2. Reject a repeat of an id already recorded for this rule.
62 * 3. Reject more than BRIX_PROTBIND_MAX_PROTOS entries.
63 */
64static char *
65brix_protbind_parse_protos(ngx_conf_t *cf, ngx_uint_t first,
66 brix_protbind_rule_t *rule)
67{
68 ngx_str_t *args = cf->args->elts;
69 ngx_uint_t arg, existing, proto;
70
71 for (arg = first; arg < cf->args->nelts; arg++) {
72 if (brix_protbind_proto_id(&args[arg], &proto) != NGX_OK) {
73 return "has an unknown protocol name "
74 "(expected gsi, ztn/token, sss, unix, krb5, host or pwd)";
75 }
76
77 for (existing = 0; existing < rule->count; existing++) {
78 if (rule->protos[existing] == proto) {
79 return "lists the same protocol twice";
80 }
81 }
82
83 if (rule->count >= BRIX_PROTBIND_MAX_PROTOS) {
84 return "lists more protocols than the server supports";
85 }
86
87 rule->protos[rule->count++] = proto;
88 }
89
90 return NGX_CONF_OK;
91}

93/* ---- Parse one brix_protbind directive ----
94 *
95 * WHAT: Appends a parsed rule to *rules (creating the array on first use).
96 * Returns NGX_CONF_OK, NGX_CONF_ERROR on allocation failure, or a static
97 * message describing the grammar violation.
98 *
99 * WHY: This is the whole configuration surface of the feature; expressing it
100 * as an ordinary nginx setter (cf->args in, error string out) lets both
101 * modules register it with no glue beyond computing the array address.
102 *
103 * HOW: 1. Require at least a template and one policy word.
104 * 2. Reject an empty template.
105 * 3. `none` must stand alone; `only` is an optional lead-in that must be
106 * followed by at least one protocol.
107 * 4. Delegate the protocol tail to brix_protbind_parse_protos.
108 */
109static char *
110brix_protbind_parse(ngx_conf_t *cf, ngx_array_t **rules)
111{
112 ngx_str_t *args = cf->args->elts;
113 ngx_array_t *array;
114 brix_protbind_rule_t *rule;
115 ngx_uint_t first_proto;
116
117 if (cf->args->nelts < 3) {
118 return "requires a host template and "
119 "\"none\" or a protocol list";
120 }
121
122 if (args[1].len == 0) {
123 return "has an empty host template";
124 }
125
126 array = brix_protbind_rules_array(cf, rules);
127 if (array == NULL) {
128 return NGX_CONF_ERROR;
129 }
130
131 rule = ngx_array_push(array);
132 if (rule == NULL) {
133 return NGX_CONF_ERROR;
134 }
135
136 ngx_memzero(rule, sizeof(*rule));
137 rule->host_tpl = args[1];
138
139 if (args[2].len == sizeof("none") - 1
140 && ngx_strncmp(args[2].data, "none", sizeof("none") - 1) == 0)
141 {
142 if (cf->args->nelts != 3) {
143 return "\"none\" takes no protocol list";
144 }
145 rule->mode = BRIX_PROTBIND_NONE;
146 return NGX_CONF_OK;
147 }
148
149 first_proto = 2;
150
151 if (args[2].len == sizeof("only") - 1
152 && ngx_strncmp(args[2].data, "only", sizeof("only") - 1) == 0)
153 {
154 if (cf->args->nelts < 4) {
155 return "\"only\" must be followed by at least one protocol";
156 }
157 rule->mode = BRIX_PROTBIND_ONLY;
158 first_proto = 3;
159
160 } else {
161 rule->mode = BRIX_PROTBIND_ALL;
162 }
163
164 return brix_protbind_parse_protos(cf, first_proto, rule);
165}

167/*
168 * brix_protbind_conf — the setter both frontends register.
169 *
170 * WHAT: Parses one directive into *rules and reports any grammar violation as
171 * an [emerg] prefixed with the directive's own name. Returns
172 * NGX_CONF_OK or NGX_CONF_ERROR.
173 *
174 * WHY: The stream and HTTP directives differ only in which array they fill and
175 * what they are called, and cmd->name already carries the latter — so the
176 * diagnostic belongs here rather than duplicated in two module wrappers
177 * that would then be free to drift.
178 *
179 * HOW: Delegate to brix_protbind_parse(); pass the two sentinel returns
180 * through untouched and turn anything else (a static message) into an
181 * emerg naming the directive.
182 */
183char *
184brix_protbind_conf(ngx_conf_t *cf, ngx_command_t *cmd, ngx_array_t **rules)
185{
186 char *rc = brix_protbind_parse(cf, rules);
187
188 if (rc == NGX_CONF_OK || rc == NGX_CONF_ERROR) {
189 return rc;
190 }
191
192 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "%V: %s", &cmd->name, rc);
193 return NGX_CONF_ERROR;
194}
