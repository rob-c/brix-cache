1/*
2 * sd_xroot.c — remote root:// origin storage driver: core.
3 *
4 * Wraps the in-process XRootD origin wire client (cache/origin_*.c) behind the SD
5 * vtable. This translation unit owns the shared origin-open machinery (the
6 * per-open object state, the connect → bootstrap → kXR_open orchestrator, and the
7 * credential-copy / teardown helpers), the driver vtable, the origin-checksum
8 * query, and instance lifecycle (create / create_origin / destroy). The object
9 * I/O + open + stat slots live in sd_xroot_io.c, the staged atomic-publish slots
10 * in sd_xroot_staged.c, and the namespace + metadata slots in sd_xroot_ns.c
11 * (phase-79 file-size split); all share driver-private state via
12 * sd_xroot_internal.h. Anonymous login is the wire client's native mode;
13 * authenticated origins use the cache's native-client delegation or a per-user
14 * credential copied into the fill task.
15 */

31/* Map a fill-task XRootD error (kXR_*) to an errno-style fact for *err_out. */
32int
33sd_xroot_errno(const brix_cache_fill_t *t)
34{
35 switch (t->xrd_error) {
36 case kXR_NotFound: return ENOENT;
37 case kXR_NotAuthorized: return EACCES;
38 case kXR_isDirectory: return EISDIR;
39 default: return EIO;
40 }
41}

43/* sd_xroot_copy_cred_into_task — copy a per-user credential into the fill task.
44 *
45 * WHAT: When `cred` is non-NULL, copy its x509 proxy path, bearer token, and
46 * principal into the fill task's cred_* fields (each only when non-empty).
47 * No-op when cred is NULL.
48 * WHY: The origin bootstrap is the sole auth decision point; a task-level copy
49 * lets each open carry independent identity without touching the shared
50 * conf. Both the plain-open and staged-open paths need the identical copy,
51 * so it lives in one place to prevent divergence.
52 * HOW: 1) return early on NULL cred (the fill task was calloc-zeroed, so the
53 * bootstrap sees the service credential / anonymous). 2) Exactly one of
54 * {x509_proxy, bearer, sss_keytab} is non-NULL for a credential-scoped
55 * open (mutually exclusive); copy whichever is present. 3) Copy principal
56 * when present (sss_keytab REQUIRES it — the asserted identity). */
57void
58sd_xroot_copy_cred_into_task(brix_cache_fill_t *t, const brix_sd_cred_t *cred)
59{
60 if (cred == NULL) {
61 return;
62 }
63 if (cred->x509_proxy != NULL && cred->x509_proxy[0] != '\0') {
64 ngx_cpystrn((u_char *) t->cred_x509_proxy, (u_char *) cred->x509_proxy,
65 sizeof(t->cred_x509_proxy));
66 }
67 if (cred->bearer != NULL && cred->bearer[0] != '\0') {
68 ngx_cpystrn((u_char *) t->cred_bearer, (u_char *) cred->bearer,
69 sizeof(t->cred_bearer));
70 }
71 if (cred->principal != NULL) {
72 ngx_cpystrn((u_char *) t->cred_principal, (u_char *) cred->principal,
73 sizeof(t->cred_principal));
74 }
75 if (cred->sss_keytab != NULL && cred->sss_keytab[0] != '\0') {
76 ngx_cpystrn((u_char *) t->cred_sss_keytab,
77 (u_char *) cred->sss_keytab, sizeof(t->cred_sss_keytab));
78 }
79 /* krb5 delegation carry (phase-70 §5.7): the FILE-ccache PATH holding the
80 * delegated TGT plus the origin service principal to authenticate against.
81 * Both are set together by the vfs_deleg krb5 gate (brix_vfs_deleg_krb5). */
82 if (cred->krb5_ccache != NULL && cred->krb5_ccache[0] != '\0') {
83 ngx_cpystrn((u_char *) t->cred_krb5_ccache,
84 (u_char *) cred->krb5_ccache, sizeof(t->cred_krb5_ccache));
85 }
86 if (cred->krb5_princ != NULL && cred->krb5_princ[0] != '\0') {
87 ngx_cpystrn((u_char *) t->cred_krb5_princ,
88 (u_char *) cred->krb5_princ, sizeof(t->cred_krb5_princ));
89 }
90}

108/* sd_xroot_origin_do_open — issue the kXR_open for a bootstrapped connection.
109 *
110 * WHAT: Perform the write- or read-open on an already connected+bootstrapped
111 * origin session, recording the fhandle and file size. Returns 0 on
112 * success (st->file_open set), or -1 with *req->err_out set on failure.
113 * WHY: Splitting the open branch off the connect/bootstrap sequence keeps
114 * sd_xroot_origin_open a flat orchestrator and isolates the write-vs-read
115 * distinction (mode bits, size source) in one place.
116 * HOW: 1) want_write → kXR_open(update) with mode bits, size 0 (truncated). 2)
117 * else → kXR_open(read|retstat), size from the returned stat. 3) on either
118 * failure map the fill-task kXR error to errno. */
119static int
120sd_xroot_origin_do_open(sd_xroot_obj_state *st,
121 const sd_xroot_origin_open_req_t *req)
122{
123 brix_cache_fill_t *t = st->t;
124
125 if (req->want_write) {
126 uint16_t mode_bits =
127 (uint16_t) ((req->mode != 0) ? (req->mode & 0777) : 0644);
128
129 if (brix_cache_origin_open_write(t, &st->oc, req->path, mode_bits,
130 st->fhandle) != 0)
131 {
132 if (req->err_out) { *req->err_out = sd_xroot_errno(t); }
133 return -1;
134 }
135 st->is_write = 1;
136 if (req->size_out) { *req->size_out = 0; } /* open_write truncates to empty */
137 } else {
138 if (brix_cache_origin_open(t, &st->oc, st->fhandle) != 0) {
139 if (req->err_out) { *req->err_out = sd_xroot_errno(t); }
140 return -1;
141 }
142 if (req->size_out) { *req->size_out = t->file_size; }
143 }
144 st->file_open = 1;
145 return 0;
146}

148/* Open the origin file: connect → bootstrap (handshake + anon or per-user login)
149 * → kXR_open. `req->want_write` selects kXR_open(update+delete+mkpath) (a fresh
150 * writable handle, size 0) over the read open (read|retstat for the size).
151 * `req->cred` carries a per-user x509 proxy path; when non-NULL and non-empty the
152 * bootstrap uses it instead of the conf's static service credential. Namespace
153 * callers (stat, sd_xroot_session) pass cred=NULL — they stay on the service
154 * credential in Phase 1. Returns the populated object state, or NULL with
155 * *req->err_out set.
156 *
157 * WHAT: Allocate and wire up the per-open fill task, copy the per-user proxy into
158 * it, then run connect → bootstrap → open as a flat sequence.
159 * WHY: The bootstrap is the sole auth decision point for every origin session; a
160 * task-level field lets each open carry independent identity without touching
161 * the shared conf or the connection pool.
162 * HOW: 1) calloc the state + fill task (zeroed cred fields ⇒ cred=NULL behaves
163 * like Phase-0). 2) copy the credential via the shared helper. 3) connect +
164 * bootstrap; on failure log the actual origin reason and map errno. 4)
165 * delegate the kXR_open branch to sd_xroot_origin_do_open. */
166sd_xroot_obj_state *
167sd_xroot_origin_open(const sd_xroot_origin_open_req_t *req)
168{
169 sd_xroot_obj_state *st = calloc(1, sizeof(*st));
170 brix_cache_fill_t *t = calloc(1, sizeof(*t));
171
172 if (st == NULL || t == NULL) {
173 free(st);
174 free(t);
175 if (req->err_out) { *req->err_out = ENOMEM; }
176 return NULL;
177 }
178 st->t = t;
179 st->oc.fd = -1;
180 t->conf = req->conf;
181 ngx_cpystrn((u_char *) t->clean_path, (u_char *) req->path,
182 sizeof(t->clean_path));
183
184 sd_xroot_copy_cred_into_task(t, req->cred);
185
186 if (brix_cache_origin_connect(t, &st->oc) != 0
187 || brix_cache_origin_bootstrap(t, &st->oc) != 0)
188 {
189 ngx_log_error(NGX_LOG_ERR, ngx_cycle->log, 0,
190 "brix sd_xroot: origin open \"%s\" failed: %s (kXR %d)",
191 req->path, (t->err_msg[0] != '\0') ? t->err_msg : "(no detail)",
192 t->xrd_error);
193 if (req->err_out) { *req->err_out = sd_xroot_errno(t); }
194 sd_xroot_obj_teardown(st);
195 return NULL;
196 }
197
198 if (sd_xroot_origin_do_open(st, req) != 0) {
199 sd_xroot_obj_teardown(st);
200 return NULL;
201 }
202 return st;
203}

205/* ---- sd_xroot_query_checksum — checksum-offload vtable slot ----
206 *
207 * WHAT: Asks the origin for its stored digest of the open object via kXR_Qcksum
208 * and returns NGX_OK with hex_out filled only when the origin's digest is
209 * in exactly the requested algorithm; NGX_DECLINED otherwise (no digest,
210 * different algorithm, or a wire fault — the caller computes instead).
211 *
212 * WHY: Without this slot a checksum request against a root:// primary pulls the
213 * entire object through kXR_read just to hash bytes the origin has already
214 * hashed; one metadata round trip on the already-open origin connection
215 * replaces the whole-object transfer.
216 *
217 * HOW: 1) Delegate to brix_sd_xroot_query_checksum (path-based kXR_Qcksum on
218 * the object's live origin connection). 2) An empty reply declines.
219 * 3) Match the origin's algorithm name against the requested canonical
220 * name case-insensitively — an alias mismatch just declines to the compute
221 * fallback, never mislabels a digest. 4) Copy the hex out on a match.
222 */
223static ngx_int_t
224sd_xroot_query_checksum(brix_sd_obj_t *obj, const char *algo,
225 char *hex_out, size_t hex_sz)
226{
227 char origin_alg[32];
228 char origin_hex[160];
229
230 brix_sd_xroot_query_checksum(obj, origin_alg, sizeof(origin_alg),
231 origin_hex, sizeof(origin_hex));
232 if (origin_alg[0] == '\0' || origin_hex[0] == '\0') {
233 return NGX_DECLINED;
234 }
235 if (strcasecmp(origin_alg, algo) != 0) {
236 return NGX_DECLINED; /* origin digest is in a different algorithm */
237 }
238 ngx_cpystrn((u_char *) hex_out, (u_char *) origin_hex, hex_sz);
239 return NGX_OK;
240}

400/* sd_xroot_origin_build_ca_store — build the GSI verify store for the origin.
401 *
402 * WHAT: When `ca_dir` is non-empty, record it on synth and build the per-worker
403 * X509 verify store (a hashed CA dir vs a bundle file) into synth->gsi_store;
404 * a build failure is logged and leaves gsi_store NULL. No-op on empty ca_dir.
405 * WHY: A NULL store makes the GSI handshake refuse the origin rather than trust
406 * an unverifiable one; isolating the stat-and-build keeps the create
407 * function flat.
408 * HOW: 1) copy ca_dir onto the instance and point synth at it. 2) stat to decide
409 * hashed-dir vs bundle-file. 3) call brix_build_ca_store with proxy certs
410 * allowed, signing-policy OFF and best-effort CRL (pre-existing outbound
411 * cache→origin trust behaviour — unchanged). 4) log on failure. */
412static void
413sd_xroot_origin_build_ca_store(sd_xroot_inst_state *is,
414 ngx_stream_brix_srv_conf_t *synth, const char *ca_dir, ngx_log_t *log)
415{
416 struct stat ca_st;
417 int is_dir;
418 int crl_count = 0;
419
420 if (ca_dir == NULL || ca_dir[0] == '\0') {
421 return;
422 }
423 ngx_cpystrn((u_char *) is->ca_dir, (u_char *) ca_dir, sizeof(is->ca_dir));
424 synth->cache_origin_ca_dir.data = (u_char *) is->ca_dir;
425 synth->cache_origin_ca_dir.len = ngx_strlen(is->ca_dir);
426 /* Also expose the origin CA as the generic trust anchor so the root:// TLS
427 * upgrade (brix_cache_origin_tls_upgrade → origin_tls_load_verify) verifies
428 * the origin's TLS cert against THIS store (CAfile or hashed CApath) instead
429 * of falling back to the system CAs. Same backing buffer as ca_dir. */
430 synth->common.trusted_ca.data = (u_char *) is->ca_dir;
431 synth->common.trusted_ca.len = ngx_strlen(is->ca_dir);
432
433 is_dir = (stat(is->ca_dir, &ca_st) == 0 && S_ISDIR(ca_st.st_mode));
434 synth->gsi_store = brix_build_ca_store(log,
435 is_dir ? is->ca_dir : NULL, is_dir ? NULL : is->ca_dir, NULL,
436 X509_V_FLAG_ALLOW_PROXY_CERTS, &crl_count,
437 BRIX_SP_MODE_OFF, BRIX_CRL_MODE_TRY);
438 if (synth->gsi_store == NULL) {
439 ngx_log_error(NGX_LOG_ERR, log, 0,
440 "brix: gsi origin CA store build failed for \"%s\" — GSI to this "
441 "origin will be refused", ca_dir);
442 }
443}

445brix_sd_instance_t *
446brix_sd_xroot_create_origin(const brix_sd_xroot_origin_cfg_t *cfg,
447 ngx_log_t *log)
448{
449 brix_sd_instance_t *inst;
450 sd_xroot_inst_state *is;
451 ngx_stream_brix_srv_conf_t *synth;
452
453 if (cfg == NULL || cfg->host == NULL || cfg->host[0] == '\0'
454 || cfg->port <= 0 || cfg->port > 65535)
455 {
456 errno = EINVAL;
457 return NULL;
458 }
459 inst = calloc(1, sizeof(*inst));
460 is = calloc(1, sizeof(*is));
461 synth = calloc(1, sizeof(*synth)); /* minimal: just the origin params */
462 if (inst == NULL || is == NULL || synth == NULL) {
463 free(inst);
464 free(is);
465 free(synth);
466 errno = ENOMEM;
467 return NULL;
468 }
469 ngx_cpystrn((u_char *) is->host, (u_char *) cfg->host, sizeof(is->host));
470 synth->cache_origin_host.data = (u_char *) is->host;
471 synth->cache_origin_host.len = ngx_strlen(is->host);
472 synth->cache_origin_port = (uint16_t) cfg->port;
473 synth->cache_origin_tls = cfg->tls ? 1 : 0;
474 synth->cache_origin_family = (ngx_uint_t) cfg->af_policy;
475
476 /* Credentials + CA store: each stores bytes on the instance so synth's
477 * ngx_str_t members reference storage valid for the instance lifetime. */
478 sd_xroot_origin_apply_creds(is, synth, cfg);
479 sd_xroot_origin_build_ca_store(is, synth, cfg->ca_dir, log);
480
481 is->conf = synth;
482 is->synth = synth; /* owned: free on destroy */
483 inst->driver = &brix_sd_xroot_driver;
484 inst->log = log;
485 inst->pool = NULL;
486 inst->state = is;
487 /* Effective caps start at the descriptor's, as brix_sd_instance_create does
488 * (this factory bypasses that path — it has no pool). CAP_NEARLINE is the one
489 * bit the OPERATOR adds: it makes the recall/residency slots reachable AND
490 * commits tier_build to requiring a cache tier in front, so it is never
491 * inferred. See sd_xroot_nearline.c. */
492 inst->caps = brix_sd_xroot_driver.caps;
493 if (cfg->nearline) {
494 inst->caps |= BRIX_SD_CAP_NEARLINE;
495 }
