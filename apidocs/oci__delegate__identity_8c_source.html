12/*
13 * oci_deleg_challenge_hdr — emit WWW-Authenticate: Basic challenge header.
14 *
15 * WHAT: Adds WWW-Authenticate: Basic realm="<realm>" response header to request.
16 * Allocates from request pool. Returns NGX_OK on success, NGX_ERROR on allocation
17 * failure.
18 *
19 * WHY: RFC 7235 requires servers to send WWW-Authenticate when rejecting with 401.
20 * Basic auth is used because clients already speak it (docker/podman login), and
21 * we're delegating the client's own registry credential, not issuing bearer tokens.
22 *
23 * HOW: Allocates header from ngx_list, formats realm string into pool memory,
24 * sets hash=1 for header lookup optimization.
25 */
26static ngx_int_t
27oci_deleg_challenge_hdr(ngx_http_request_t *r,
28 ngx_http_brix_oci_loc_conf_t *lcf)
29{
30 ngx_table_elt_t *h = ngx_list_push(&r->headers_out.headers);
31 u_char *v;
32 size_t n;
33
34 if (h == NULL) {
35 return NGX_ERROR;
36 }
37 n = sizeof("Basic realm=\"\"") + lcf->delegate_realm.len;
38 v = ngx_pnalloc(r->pool, n);
39 if (v == NULL) {
40 return NGX_ERROR;
41 }
42 h->hash = 1;
43 ngx_str_set(&h->key, "WWW-Authenticate");
44 h->value.data = v;
45 h->value.len = (size_t) (ngx_snprintf(v, n, "Basic realm=\"%V\"",
46 &lcf->delegate_realm) - v);
47
48 return NGX_OK;
49}

51/*
52 * brix_oci_delegate_refuse — uniform 401 refusal for all delegation failures.
53 *
54 * WHAT: Sends HTTP 401 Unauthorized with WWW-Authenticate: Basic challenge and
55 * OCI DENIED error envelope. Returns NGX_HTTP_UNAUTHORIZED. Used for all
56 * delegation failures: bad credential, no credential, repository not found,
57 * scope denied.
58 *
59 * WHY: Security through uniformity — any difference between "bad password",
60 * "no such repo", and "not authorized" creates an enumeration oracle that
61 * leaks information about private namespaces. RFC 7235 requires WWW-Authenticate
62 * on 401. OCI spec requires DENIED envelope for errors.
63 *
64 * HOW: Sets status=401, adds WWW-Authenticate header via oci_deleg_challenge_hdr(),
65 * builds OCI error JSON with DENIED code and generic message, marks request
66 * as completed.
67 */
68ngx_int_t
69brix_oci_delegate_refuse(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
70 ngx_http_brix_oci_ctx_t *ctx)
71{
72 ctx->disp = BRIX_OCI_OUT_REFUSED;
73 brix_oci_guard_emit(r, GUARD_R_AUTHFAIL, GUARD_OP_READ,
74 NGX_HTTP_UNAUTHORIZED);
75 if (oci_deleg_challenge_hdr(r, lcf) != NGX_OK) {
76 return NGX_HTTP_INTERNAL_SERVER_ERROR;
77 }
78 return brix_oci_error(r, NGX_HTTP_UNAUTHORIZED, BRIX_OCI_ERR_DENIED,
79 NULL);
80}

82/*
83 * oci_deleg_decode — decode Basic auth header and extract credential hash.
84 *
85 * WHAT: Decodes Base64 Basic authorization header into "user:pass" format,
86 * validates format (must contain colon, non-empty username), computes SHA256
87 * hash of credential for SHM keying. Stores hash in ctx->deleg_cred, plaintext
88 * in ctx->deleg_basic (request pool), username in ctx->deleg_user. Returns
89 * NGX_DECLINED on success (identity established), error code on failure.
90 *
91 * WHY: Basic auth credentials arrive Base64-encoded and must be decoded before
92 * use. The SHA256 hash is the only form that persists (as SHM key) — plaintext
93 * is used only for the upstream proof exchange then discarded. Validation prevents
94 * malformed credentials from reaching the upstream.
95 *
96 * HOW: Allocates decode buffer from request pool, decodes Base64, validates
97 * length and NUL-absence, finds colon separator, computes SHA256, extracts
98 * username for logging. Refuses with brix_oci_delegate_refuse() on any validation failure.
99 */
100static ngx_int_t
101oci_deleg_decode(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
102 ngx_http_brix_oci_ctx_t *ctx, ngx_str_t *b64)
103{
104 ngx_str_t dst;
105 u_char *colon;
106 char *user;
107
108 dst.data = ngx_pnalloc(r->pool, ngx_base64_decoded_length(b64->len) + 1);
109 if (dst.data == NULL) {
110 return NGX_HTTP_INTERNAL_SERVER_ERROR;
111 }
112 if (ngx_decode_base64(&dst, b64) != NGX_OK
113 || dst.len == 0 || dst.len >= BRIX_OCI_BASIC_MAX
114 || memchr(dst.data, '\0', dst.len) != NULL)
115 {
116 return brix_oci_delegate_refuse(r, lcf, ctx);
117 }
118 dst.data[dst.len] = '\0';
119
120 colon = (u_char *) strchr((char *) dst.data, ':');
121 if (colon == NULL || colon == dst.data) {
122 return brix_oci_delegate_refuse(r, lcf, ctx);
123 }
124
125 if (brix_oci_sha256_key(dst.data, dst.len, ctx->deleg_cred) != 0) {
126 return NGX_HTTP_INTERNAL_SERVER_ERROR;
127 }
128 ctx->deleg_basic = (const char *) dst.data;
129
130 /* The username half, for the dashboard/log identity. Never the pair. */
131 user = ngx_pnalloc(r->pool, (size_t) (colon - dst.data) + 1);
132 if (user == NULL) {
133 return NGX_HTTP_INTERNAL_SERVER_ERROR;
134 }
135 ngx_memcpy(user, dst.data, (size_t) (colon - dst.data));
136 user[colon - dst.data] = '\0';
137 ctx->deleg_user = user;
138
139 return NGX_DECLINED;
140}

142/*
143 * brix_oci_delegate_ident — extract downstream client identity from authorization header.
144 *
145 * WHAT: Main entry point for delegation identity extraction. Checks for Basic auth
146 * header, validates TLS requirement (unless deleg_insecure configured), decodes
147 * credential via oci_deleg_decode(). Returns NGX_DECLINED if delegation is off or
148 * anonymous access, NGX_HTTP_UNAUTHORIZED on credential errors, NGX_HTTP_BAD_REQUEST
149 * on TLS violation.
150 *
151 * WHY: Delegation mode requires every request to carry the client's own registry
152 * credential. This function is the gatekeeper that extracts and validates that
153 * credential before the authorization proof phase. TLS enforcement prevents
154 * credential interception on cleartext connections.
155 *
156 * HOW: Checks lcf->delegate flag, retrieves Authorization header, validates TLS
157 * (unless insecure mode), verifies Basic scheme prefix, delegates decoding to
158 * oci_deleg_decode(). Refuses non-Basic schemes (Bearer, Negotiate) to avoid
159 * guessing at credential formats.
160 */
161ngx_int_t
162brix_oci_delegate_ident(ngx_http_request_t *r,
163 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx)
164{
165 ngx_table_elt_t *auth = r->headers_in.authorization;
166 ngx_str_t b64;
167 int is_tls = 0;
168
169 if (!lcf->delegate || auth == NULL) {
170 return NGX_DECLINED; /* off, or anonymous: cred = 0s */
171 }
172
173#if (NGX_HTTP_SSL)
174 is_tls = (r->connection->ssl != NULL);
175#endif
176 /* A credential on a cleartext connection is already burned; the ONE
177 * thing still in our power is to refuse to act on it — before decoding,
178 * so the secret never even enters this process's data flow. */
179 if (!is_tls && !lcf->deleg_insecure) {
180 ctx->disp = BRIX_OCI_OUT_REFUSED;
181 brix_oci_guard_emit(r, GUARD_R_AUTHFAIL, GUARD_OP_READ,
182 NGX_HTTP_BAD_REQUEST);
183 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST, BRIX_OCI_ERR_DENIED,
184 "credentials require TLS on this mirror");
185 }
186
187 if (auth->value.len < sizeof("Basic ") - 1
188 || ngx_strncasecmp(auth->value.data, (u_char *) "Basic ",
189 sizeof("Basic ") - 1) != 0)
190 {
191 /* Bearer, Negotiate, junk: not a credential this surface can
192 * delegate, and guessing is how a secret goes somewhere wrong. */
193 return brix_oci_delegate_refuse(r, lcf, ctx);
194 }
195 b64.data = auth->value.data + sizeof("Basic ") - 1;
196 b64.len = auth->value.len - (sizeof("Basic ") - 1);
197
198 return oci_deleg_decode(r, lcf, ctx, &b64);
199}
