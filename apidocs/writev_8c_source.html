12/* brix_writev_body_extra — trailing segment-data length for kXR_writev
13 * WHAT: Validates a dlen-framed write_list descriptor block under the stock
14 * wire contract (dlen frames ONLY the descriptors: non-zero, 16-aligned,
15 * <= BRIX_WRITEV_MAXSEGS entries, descriptors + declared data within
16 * BRIX_MAX_WRITE_PAYLOAD) and reports sum(wlen) — the number of segment-data
17 * bytes the client streams after the request frame.
18 * WHY: Shared by the recv framing (to extend the read obligation so the
19 * streamed data lands in payload_buf before dispatch) and kept next to the
20 * handler that consumes the same layout. Also delegated to by
21 * brix_ckpxeq_body_extra (chkpoint_xeq.c): a kXR_ckpXeq-embedded writev
22 * tunnels this exact contract — its embedded header's dlen frames the
23 * descriptors only, and the segment data streams behind. Returns NGX_OK
24 * with *extra set, or NGX_DECLINED when the block violates the contract (the
25 * caller decides the error; the framing never rejects so the auth gates
26 * still run first). */
27ngx_int_t
28brix_writev_body_extra(const u_char *desc, uint32_t dlen, uint32_t *extra)
29{
30 const write_list *wl = (const write_list *) desc;
31 uint32_t n_segs, i;
32 uint64_t total = 0;
33
34 *extra = 0;
35
36 if (desc == NULL || dlen == 0 || dlen % BRIX_WRITEV_SEGSIZE != 0) {
37 return NGX_DECLINED;
38 }
39
40 n_segs = dlen / BRIX_WRITEV_SEGSIZE;
41 if (n_segs > BRIX_WRITEV_MAXSEGS) {
42 return NGX_DECLINED;
43 }
44
45 for (i = 0; i < n_segs; i++) {
46 total += (uint32_t) ntohl((uint32_t) wl[i].wlen);
47 }
48
49 if ((uint64_t) dlen + total > BRIX_MAX_WRITE_PAYLOAD) {
50 return NGX_DECLINED;
51 }
52
53 *extra = (uint32_t) total;
54 return NGX_OK;
55}

57/* writev_validate_framing — enforce the stock do_WriteV framing contract.
58 * WHAT: Validates the descriptor block that recv left in ctx->recv.payload —
59 * dlen present and a whole number of 16-byte descriptors, segment count within
60 * the vector cap, descriptors + declared data within the aggregate transfer
61 * cap, and the recv-appended trailing byte count matching sum(wlen). On
62 * success it publishes the segment count (*n_segs_out) and total data length
63 * (*total_wlen_out) to the caller.
64 * WHY: This is the FROZEN stock-framing gate — the same contract the recv path
65 * and the ckpXeq-embedded writev depend on. Concentrating it here keeps the
66 * handler flat and the wire-error parity (kXR error code + message + link drop)
67 * byte-identical to the reference server.
68 * HOW: Each violation emits its stock error via brix_send_error and returns
69 * NGX_ERROR (caller drops the link — a doubtful descriptor block makes the
70 * trailing byte count unknowable, so the connection cannot resynchronise). An
71 * all-zero-length vector is the stock immediate-success case: it sends OK and
72 * returns NGX_DONE. A clean, non-empty vector returns NGX_OK. */
73static ngx_int_t
74writev_validate_framing(brix_ctx_t *ctx, ngx_connection_t *c,
75 const write_list *wl, size_t *n_segs_out,
76 uint64_t *total_wlen_out)
77{
78 size_t n_segs, i;
79 uint64_t total_wlen = 0;
80
81 *n_segs_out = 0;
82 *total_wlen_out = 0;
83
84 /* Stock parity: a dlen that is zero or not a whole number of write_list
85 * descriptors is "Write vector is invalid" (kXR_ArgInvalid) + link drop. */
86 if (ctx->recv.payload == NULL || ctx->recv.cur_dlen == 0
87 || ctx->recv.cur_dlen % BRIX_WRITEV_SEGSIZE != 0)
88 {
89 BRIX_OP_ERR(ctx, BRIX_OP_WRITEV);
90 (void) brix_send_error(ctx, c, kXR_ArgInvalid,
91 "Write vector is invalid");
92 return NGX_ERROR;
93 }
94
95 n_segs = ctx->recv.cur_dlen / BRIX_WRITEV_SEGSIZE;
96
97 /* Stock parity: vector count cap (stock maxWvecsz == 1024 == ours). */
98 if (n_segs > BRIX_WRITEV_MAXSEGS) {
99 BRIX_OP_ERR(ctx, BRIX_OP_WRITEV);
100 (void) brix_send_error(ctx, c, kXR_ArgTooLong,
101 "Write vector is too long");
102 return NGX_ERROR;
103 }
104
105 for (i = 0; i < n_segs; i++) {
106 total_wlen += (uint32_t) ntohl((uint32_t) wl[i].wlen);
107 }
108
109 /* Our aggregate transfer cap (stock instead caps each element at its
110 * maxTransz); enforced by the framing BEFORE it reads the data, so a
111 * violation means the trailing bytes were never accepted — drop. */
112 if ((uint64_t) ctx->recv.cur_dlen + total_wlen > BRIX_MAX_WRITE_PAYLOAD) {
113 BRIX_OP_ERR(ctx, BRIX_OP_WRITEV);
114 (void) brix_send_error(ctx, c, kXR_NoMemory,
115 "writev transfer is too large");
116 return NGX_ERROR;
117 }
118
119 /* Defensive: the recv framing extends the body by exactly sum(wlen); a
120 * mismatch means the extension never ran for this frame (unreachable in
121 * normal operation) and the data bytes are not in the buffer. */
122 if (total_wlen != (uint64_t) ctx->recv.cur_body_extra) {
123 BRIX_OP_ERR(ctx, BRIX_OP_WRITEV);
124 (void) brix_send_error(ctx, c, kXR_ArgInvalid,
125 "Write vector is invalid");
126 return NGX_ERROR;
127 }
128
129 /* Stock parity: an all-zero-length vector succeeds immediately, before
130 * any handle validation (do_WriteV returns Send() when maxSZ == 0). */
131 if (total_wlen == 0) {
132 BRIX_OP_OK(ctx, BRIX_OP_WRITEV);
133 (void) brix_send_ok(ctx, c, NULL, 0);
134 return NGX_DONE;
135 }
136
137 *n_segs_out = n_segs;
138 *total_wlen_out = total_wlen;
139 return NGX_OK;
140}

142/* writev_validate_handles — all-or-nothing handle admission for the vector.
143 * WHAT: Verifies every segment's target handle is open and writable before any
144 * byte is written.
145 * WHY: INVARIANT — a bad handle in a later segment must never leave earlier
146 * segments already written; admitting the whole vector up front makes the write
147 * all-or-nothing. Only fhandle[0] is significant (handles are 0..255).
148 * HOW: Returns NGX_OK when every handle passes. On the first failure it emits
149 * the stock error (kXR_FileNotOpen for a stale/closed slot, kXR_NotAuthorized
150 * for a read-only handle) and returns NGX_DONE — the reply is already queued, so
151 * the caller must STOP (never fall through to the write). NGX_ERROR only when
152 * emitting that error itself failed (allocation). Note brix_send_error returns
153 * NGX_OK on a successfully queued reply, so its result cannot be propagated as-is
154 * here: an NGX_OK would be indistinguishable from "all handles passed" and let a
155 * rejected handle reach writev_try_aio (a stale fd<0 read / double response). */
156static ngx_int_t
157writev_validate_handles(brix_ctx_t *ctx, ngx_connection_t *c,
158 const write_list *wl, size_t n_segs)
159{
160 size_t i;
161
162 for (i = 0; i < n_segs; i++) {
163 int idx = (int)(unsigned char) wl[i].fhandle[0];
164 ngx_int_t rc = NGX_OK;
165
166 if (!brix_validate_write_handle(ctx, c, idx, "WRITEV",
167 BRIX_OP_WRITEV, &rc)) {
168 return rc == NGX_ERROR ? NGX_ERROR : NGX_DONE;
169 }
170 }
171
172 return NGX_OK;
173}

175/* writev_write_segment — write one segment through the VFS on this thread.
176 * WHAT: Handles a single (fd, offset, data, wlen) segment on the event-loop
177 * thread: replay neutralisation, the VFS write, dirty-extent marking, and the
178 * write-recovery journal update; accumulates the client-visible byte total and
179 * advances *data_ptr past this segment's data block.
180 * WHY: The synchronous fallback body per segment is a self-contained unit; the
181 * loop that drives it stays flat and the replay/short-write parity lives in one
182 * reviewable place.
183 * HOW: Decodes the wire fields, returns NGX_OK for an empty or replayed segment
184 * (already accounted, pointer advanced). On I/O error it emits the stock error
185 * via BRIX_RETURN_ERR (which returns brix_send_error's result) and propagates
186 * it as NGX_ERROR so the caller stops the loop and drops the link; on success
187 * it returns NGX_OK. */
188static ngx_int_t
189writev_write_segment(brix_ctx_t *ctx, ngx_connection_t *c,
190 const write_list *seg, const u_char **data_ptr,
191 size_t *bytes_written_total)
192{
193 int idx = (int)(unsigned char) seg->fhandle[0];
194 int64_t offset = (int64_t) brix_plat_be64toh((uint64_t) seg->offset); /* BE64 */
195 uint32_t wlen = (uint32_t) ntohl((uint32_t) seg->wlen); /* BE32 */
196 ssize_t nw;
197
198 /* Defense in depth (see writev_try_aio): never index the BRIX_MAX_FILES-slot
199 * ctx->files table with an out-of-range handle — an OOB read that segfaults
200 * the worker on a fuzzed kXR_writev whose fhandle[0] slipped past
201 * writev_validate_handles. */
202 if (idx < 0 || idx >= BRIX_MAX_FILES) {
203 return NGX_ERROR;
204 }
205
206 /* Empty segment carries no data block; nothing to advance over. */
207 if (wlen == 0) {
208 return NGX_OK;
209 }
210 if (ctx->files[idx].writer != NULL) {
211 return writev_staged_segment(ctx, c, seg, data_ptr,
212 bytes_written_total);
213 }
214
215 /* Same replay handling as the AIO path: a range already written on a prior
216 * (dropped) connection is acknowledged as success but not re-issued to
217 * disk. data_ptr must still skip this segment's bytes to stay aligned, and
218 * we count the bytes toward the client-visible total so the reply matches
219 * what the client thinks it sent. */
220 /* phase79-fp: security.ArrayBound — idx (0..255 from fhandle[0]) was already
221 * bounds-checked against BRIX_MAX_FILES by writev_validate_handles before the
222 * sync path runs; the analyzer cannot correlate the separate validation loop. */
223 if (ctx->files[idx].wrts_enabled &&
224 brix_wrts_is_replay(&ctx->files[idx], offset, wlen))
225 {
226 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, c->log, 0,
227 "brix: writev recovery replay skip offset=%lld len=%ud",
228 (long long) offset, wlen);
229 *bytes_written_total += (size_t) wlen;
230 *data_ptr += wlen;
231 return NGX_OK;
232 }
233
234 {
235 brix_vfs_job_t job;
236
237 brix_vfs_job_write_init(&job, ctx->files[idx].fd,
238 (off_t) offset, *data_ptr, (size_t) wlen);
239 brix_vfs_job_set_obj(&job, &ctx->files[idx].sd_obj);
240 brix_vfs_io_execute(&job);
241 nw = job.nio;
242 if (job.io_errno != 0) {
243 errno = job.io_errno;
244 }
245 }
246
247 if (nw < 0) {
248 BRIX_RETURN_ERR(ctx, c, BRIX_OP_WRITEV, "WRITEV",
249 ctx->files[idx].path, "-",
250 kXR_IOError, strerror(errno));
251 }
252 if ((uint32_t) nw < wlen) {
253 BRIX_RETURN_ERR(ctx, c, BRIX_OP_WRITEV, "WRITEV",
254 ctx->files[idx].path, "-",
255 kXR_IOError, "short write (disk full?)");
256 }
257
258 ctx->files[idx].bytes_written += (size_t) nw;
259 ctx->totals.bytes_written += (size_t) nw;
260 *bytes_written_total += (size_t) nw;
261 /* write-through cache: flag the just-written extent dirty so it gets
262 * flushed to the backing store (end offset is inclusive: off+nw-1). */
263 if (ctx->files[idx].wt_enabled) {
264 brix_wt_mark_dirty(ctx, idx, offset + (int64_t) nw - 1,
265 (size_t) nw);
266 }
267 /* write-recovery journal: remember this range so a later replay after
268 * reconnect is recognised by brix_wrts_is_replay() above. */
269 if (ctx->files[idx].wrts_enabled) {
270 brix_wrts_record(&ctx->files[idx], offset, (uint32_t) nw);
271 }
272 /* Advance by wlen (== nw here, short writes already errored out). */
273 *data_ptr += wlen;
274 return NGX_OK;
275}

422/* Handle kXR_writev — scatter the request's data segments to their per-handle
423 * (fd, offset) targets through the VFS write path, then reply with the aggregate
424 * status.
425 *
426 * Wire contract (stock XrdXrootdProtocol::do_WriteV): the header dlen frames
427 * ONLY the N*16-byte write_list descriptor block; the concatenated segment data
428 * (sum(wlen) bytes) streams after the frame. The recv framing has already
429 * appended that trailing data to ctx->recv.payload (ctx->recv.cur_body_extra bytes), so
430 * descriptors and data are contiguous here. Framing violations are rejected
431 * exactly like the reference server — send the error, then drop the link
432 * (return NGX_ERROR): once the descriptor block is in doubt the trailing byte
433 * count is unknowable and the connection cannot be resynchronised.
434 *
435 * The checkpoint-embedded form (kXR_ckpXeq carrying a writev — ckp_xeq_writev
436 * in chkpoint_xeq.c) tunnels this SAME layout one level down: the embedded
437 * sub-header's dlen frames the descriptors and the data streams behind, so
438 * the two paths must be kept contract-identical.
439 *
440 * Orchestrator: validate framing → admit every handle → try the AIO offload
441 * (returns early when posted) → synchronous per-segment write → optional
442 * doSync flush → access-log + reply. */
443ngx_int_t
444brix_handle_writev(brix_ctx_t *ctx, ngx_connection_t *c)
445{
446 xrdw_writev_req_t req;
447 write_list *wl;
448 size_t n_segs = 0;
449 uint64_t total_wlen = 0;
450 const u_char *data_ptr;
451 size_t bytes_written_total = 0;
452 int do_sync;
453 ngx_int_t rc;
454 ngx_flag_t posted = 0;
455 writev_run_t run = { 0 };
456
457 xrdw_writev_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
458 do_sync = (req.options & kXR_wv_doSync) ? 1 : 0;
459
460 wl = (write_list *) ctx->recv.payload;
461
462 rc = writev_validate_framing(ctx, c, wl, &n_segs, &total_wlen);
463 if (rc == NGX_DONE) {
464 return NGX_OK; /* all-zero-length vector: OK already sent. */
465 }
466 if (rc != NGX_OK) {
467 return rc; /* framing violation: error sent, drop the link. */
468 }
469
470 /* INVARIANT 1 / hostile-network integrity: kXR_writev carries NO per-segment
471 * CRC (like plain kXR_write, unlike kXR_pgwrite). brix_require_pgwrite refuses
472 * a data-carrying vector so clients must use the checksummed pgwrite path; the
473 * all-zero-length vector already returned NGX_DONE above, so any vector reaching
474 * here carries data. */
475 rc = writev_require_integrity(ctx, c);
476 if (rc == NGX_DONE) {
477 return NGX_OK;
478 }
479 if (rc != NGX_OK) {
480 return rc;
481 }
482
483 rc = writev_validate_handles(ctx, c, wl, n_segs);
484 if (rc == NGX_DONE) {
485 return NGX_OK; /* stock error already queued; handler is done */
486 }
487 if (rc != NGX_OK) {
488 return rc; /* send_error allocation failure → drop the link */
489 }
490
491 /* oss.maxsize (§3.9): refuse the whole vector if ANY segment's end offset
492 * would push its target past the cap. Computed here, once, before any
493 * segment is written, so a partial vector can never leak past the limit. */
494 rc = writev_validate_maxsize(ctx, c, wl, n_segs);
495 if (rc != NGX_OK) {
496 return rc;
497 }
498
499 /* All N descriptors occupy the first n_segs*16 bytes; the per-segment data
500 * blocks follow contiguously in segment order from here. */
501 data_ptr = ctx->recv.payload + n_segs * BRIX_WRITEV_SEGSIZE;
502
503 run.ctx = ctx;
504 run.c = c;
505 run.wl = wl;
506 run.n_segs = n_segs;
507 run.data_ptr = data_ptr;
508
509 if (!writev_has_staged(&run)) {
510 rc = writev_try_aio(&run, do_sync, &posted);
511 if (rc != NGX_OK) {
512 return rc; /* allocation failure: drop the link. */
513 }
514 if (posted) {
515 return NGX_OK; /* done callback owns the buffer + reply. */
516 }
517 }
518
519 /* Synchronous fallback: write each segment, then honour doSync. */
520 rc = writev_finish_sync(&run, do_sync, &bytes_written_total);
521 if (rc != NGX_OK) {
522 return rc;
523 }
524
525 return writev_reply(ctx, c, n_segs, bytes_written_total);
526}
