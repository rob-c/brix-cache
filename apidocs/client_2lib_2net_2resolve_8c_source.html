1/*
2 * resolve.c — the client's one getaddrinfo() (phase-116). See resolve.h.
3 *
4 * HOW: brix_resolve() = cache probe → getaddrinfo() → copy the chain into
5 * fixed sockaddr_storage slots → cache store. The cache is a small
6 * array of slots replaced round-robin under a mutex: cheap, bounded,
7 * and enough to turn the cvmfs per-object GET storm into one lookup a
8 * minute per mirror (libcurl's own default DNS cache TTL). Negative
9 * answers are not cached — a name that is not there yet (a fleet
10 * coming up) must not stay "not there" for a minute.
11 */

47resolve_now(void)
48{

50 clock_gettime(CLOCK_MONOTONIC, &ts);

55/* TTL in seconds, read once from the environment (0 disables the cache);
56 * called under g_cache_mu only. */
57static long
58resolve_cache_ttl(void)
59{
60 if (g_cache_ttl < 0) {
61 const char *e = getenv("XRDC_RESOLVE_CACHE_S");
62 g_cache_ttl = (e != NULL) ? strtol(e, NULL, 10) : BRIX_RESOLVE_CACHE_S;
63 if (g_cache_ttl < 0) {
64 g_cache_ttl = 0;
65 }
66 }
67 return g_cache_ttl;
68}

81/* copy a live cached answer into out; its count, or 0 on a miss */
82static int
83resolve_cache_get(const char *host, int port, int family, int socktype,
84 unsigned flags, brix_resolve_addr *out, int max)
85{
86 time_t now = resolve_now();
87 int n = 0, i;
88
89 pthread_mutex_lock(&g_cache_mu);
90 for (i = 0; i < RESOLVE_CACHE_SLOTS && n == 0; i++) {
91 const resolve_slot *s = &g_cache[i];
92 if (s->expires > now
93 && resolve_slot_matches(s, host, port, family, socktype, flags))
94 {
95 n = (s->n < max) ? s->n : max;
96 memcpy(out, s->addrs, (size_t) n * sizeof(*out));
97 }
98 }
99 pthread_mutex_unlock(&g_cache_mu);
100 return n;
101}

104static void
105resolve_cache_put(const char *host, int port, int family, int socktype,
106 unsigned flags, const brix_resolve_addr *addrs, int n)
107{
108 long ttl;
109
110 pthread_mutex_lock(&g_cache_mu);
111 ttl = resolve_cache_ttl();
112 if (ttl > 0 && strlen(host) < RESOLVE_HOST_MAX) {
113 resolve_slot *s = &g_cache[g_cache_next++ % RESOLVE_CACHE_SLOTS];
114 snprintf(s->host, sizeof(s->host), "%s", host);
115 s->port = port;
116 s->family = family;
117 s->socktype = socktype;
118 s->flags = flags;
119 s->n = n;
120 memcpy(s->addrs, addrs, (size_t) n * sizeof(*addrs));
121 s->expires = resolve_now() + ttl;
122 }
123 pthread_mutex_unlock(&g_cache_mu);
124}

129{

145static int
146resolve_copy_chain(const struct addrinfo *res, brix_resolve_addr *out, int max)
147{
148 const struct addrinfo *ai;
149 int n = 0;
150
151 for (ai = res; ai != NULL && n < max; ai = ai->ai_next) {
152 if (ai->ai_addr == NULL || ai->ai_addrlen > sizeof(out[n].ss)) {
153 continue;
154 }
155 memset(&out[n], 0, sizeof(out[n]));
156 memcpy(&out[n].ss, ai->ai_addr, ai->ai_addrlen);
157 out[n].len = ai->ai_addrlen;
158 out[n].family = ai->ai_family;
159 out[n].socktype = ai->ai_socktype;
160 out[n].protocol = ai->ai_protocol;
161 n++;
162 }
163 return n;
164}

167int
168brix_resolve(const char *host, int port, int family, int socktype,
169 unsigned flags, brix_resolve_addr *out, int max,
170 brix_resolve_err *err)
171{
172 struct addrinfo hints, *res = NULL;
173 char portstr[16];
174 const char *key = (host != NULL) ? host : "";
175 int gai, n;
176
177 if (err != NULL) {
178 err->transient = 0;
179 err->text = NULL;
180 }
181 if (out == NULL || max <= 0 || port < 0 || port > 65535
182 || (host == NULL && !(flags & BRIX_RESOLVE_PASSIVE)))
183 {
184 return -1;
185 }
186 n = resolve_cache_get(key, port, family, socktype, flags, out, max);
187 if (n > 0) {
188 return n;
189 }
190
191 memset(&hints, 0, sizeof(hints));
192 hints.ai_family = family;
193 hints.ai_socktype = socktype;
194 hints.ai_flags = resolve_hint_flags(flags);
195 snprintf(portstr, sizeof(portstr), "%d", port);
196 gai = getaddrinfo(host, portstr, &hints, &res);
197 if (gai != 0) {
198 if (err != NULL) {
199 err->transient = (gai == EAI_AGAIN);
200 err->text = gai_strerror(gai);
201 }
202 return 0;
203 }
204 n = resolve_copy_chain(res, out, max);
205 freeaddrinfo(res);
206 if (n == 0) {
207 if (err != NULL) {
208 err->text = "no usable address";
209 }
210 return 0;
211 }
212 if (!(flags & BRIX_RESOLVE_NUMERIC)) {
213 resolve_cache_put(key, port, family, socktype, flags, out, n);
214 }
215 return n;
216}

219const char *
220brix_resolve_ntop(const struct sockaddr *sa, socklen_t len, char *buf,
221 size_t buflen)
222{
223 const void *src = NULL;
224
225 if (buf == NULL || buflen == 0) {
226 return "";
227 }
228 buf[0] = '\0';
229 if (sa == NULL) {
230 return buf;
231 }
232 if (sa->sa_family == AF_INET && len >= sizeof(struct sockaddr_in)) {
233 src = &((const struct sockaddr_in *) sa)->sin_addr;
234 } else if (sa->sa_family == AF_INET6
235 && len >= sizeof(struct sockaddr_in6))
236 {
237 src = &((const struct sockaddr_in6 *) sa)->sin6_addr;
238 }
239 if (src == NULL
240 || inet_ntop(sa->sa_family, src, buf, (socklen_t) buflen) == NULL)
241 {
242 buf[0] = '\0';
243 }
244 return buf;
245}
