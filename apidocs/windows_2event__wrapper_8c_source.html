35 * EVENT FD EMULATION

38/*
39 * Windows doesn't have eventfd - emulate using pipe + counter
40 *
41 * Structure layout:
42 * - pipefd[0]: read end (returned as eventfd)
43 * - pipefd[1]: write end (internal)
44 * - counter: 64-bit event counter (Linux eventfd semantics)
45 * - lock: CRITICAL_SECTION protects counter
46 *
47 * Usage:
48 * 1. brix_plat_eventfd(initial_value, flags) - create
49 * 2. write(eventfd, &value, sizeof(value)) - signal
50 * 3. read(eventfd, &value, sizeof(value)) - wait/acknowledge
51 * 4. close(eventfd) - cleanup
52 */
53
54typedef struct {
55 int pipefd[2]; /* Pipe for signaling */
56 uint64_t counter; /* Event counter */
57 CRITICAL_SECTION lock; /* Protect counter */
58} brix_win_eventfd_t;
59
60static brix_win_eventfd_t *brix_win_eventfd_get(int fd);
61static void brix_win_eventfd_unregister(int fd);

63int
64brix_plat_eventfd_write(int efd, uint64_t value)
65{
66 /*
67 * Write to eventfd (signal event)
68 *
69 * Linux eventfd semantics:
70 * - Writes 8-byte value to counter
71 * - Counter wraps on overflow (unless EFD_SEMAPHORE)
72 * - Returns EINVAL if counter would overflow
73 *
74 * Windows implementation:
75 * - Acquire lock
76 * - Add value to counter
77 * - Write 8 bytes to pipe (signals readers)
78 * - Release lock
79 *
80 * @param efd Event fd (read end)
81 * @param value Value to add (typically 1)
82 * @return 0 on success, -1 on error
83 */
84 brix_win_eventfd_t *eventfd = brix_win_eventfd_get(efd);
85
86 if (eventfd == NULL) {
87 errno = EBADF;
88 return -1;
89 }
90
91 EnterCriticalSection(&eventfd->lock);
92
93 /* Check for overflow (Linux eventfd max is 2^64-1) */
94 if (eventfd->counter > UINT64_MAX - value) {
95 LeaveCriticalSection(&eventfd->lock);
96 errno = EINVAL;
97 return -1;
98 }
99
100 eventfd->counter += value;
101
102 /* Write to pipe to signal readers */
103 ssize_t written = _write(eventfd->pipefd[1], &value, sizeof(value));
104
105 LeaveCriticalSection(&eventfd->lock);
106
107 if (written != sizeof(value)) {
108 errno = EAGAIN;
109 return -1;
110 }
111
112 return 0;
113}

115int
116brix_plat_eventfd_read(int efd, uint64_t *value)
117{
118 /*
119 * Read from eventfd (wait for event)
120 *
121 * Linux eventfd semantics:
122 * - Reads 8-byte value from counter
123 * - Returns current counter value
124 * - Counter reset to 0 after read
125 * - Blocks if counter is 0 (unless EFD_NONBLOCK)
126 *
127 * Windows implementation:
128 * - Read from pipe (blocks if empty)
129 * - Accumulate all values read
130 * - Return total in *value
131 *
132 * @param efd Event fd (read end)
133 * @param value Output: accumulated counter value
134 * @return 0 on success, -1 on error
135 */
136 brix_win_eventfd_t *eventfd = brix_win_eventfd_get(efd);
137 uint64_t total = 0;
138 uint64_t chunk;
139 ssize_t nread;
140
141 if (eventfd == NULL || value == NULL) {
142 errno = EINVAL;
143 return -1;
144 }
145
146 /* Read all available data from pipe */
147 while ((nread = _read(eventfd->pipefd[0], &chunk, sizeof(chunk))) > 0) {
148 total += chunk;
149 }
150
151 if (nread < 0 && errno == EAGAIN) {
152 /* Non-blocking and no data */
153 if (total == 0) {
154 return -1;
155 }
156 }
157
158 EnterCriticalSection(&eventfd->lock);
159 eventfd->counter = 0; /* Reset counter after read */
160 LeaveCriticalSection(&eventfd->lock);
161
162 *value = total;
163 return 0;
164}

168{
169 /*
170 * Close eventfd and cleanup
171 *
172 * Implementation:
173 * 1. Unregister from fd table
174 * 2. Close both pipe ends
175 * 3. Delete CRITICAL_SECTION
176 * 4. Free structure
177 *
178 * @param efd Event fd (read end)
179 * @return 0 on success, -1 on error
180 */
181 brix_win_eventfd_t *eventfd = brix_win_eventfd_get(efd);
182
183 if (eventfd == NULL) {
184 errno = EBADF;
185 return -1;
186 }
187
188 /* Unregister first to prevent double-free */
189 brix_win_eventfd_unregister(efd);
190
191 /* Close pipe ends */
192 _close(eventfd->pipefd[0]);
193 _close(eventfd->pipefd[1]);
194
195 /* Cleanup synchronization */
196 DeleteCriticalSection(&eventfd->lock);
197
198 /* Free structure */
199 free(eventfd);
200
201 return 0;

204/*
205 * Windows eventfd emulation registry
206 * Maps fd to eventfd structure for proper cleanup
207 * Thread-safe using SRW lock
208 */
209#define BRIX_WIN_EVENTFD_MAX 256
210
211static struct {
212 brix_win_eventfd_t *events[BRIX_WIN_EVENTFD_MAX];
213 SRWLOCK lock;
214 int initialized;
215} g_eventfd_registry = { {0}, SRWLOCK_INIT, 0 };
216
217static int
218brix_win_eventfd_register(int fd, brix_win_eventfd_t *eventfd)
219{
220 if (!g_eventfd_registry.initialized) {
221 InitializeSRWLock(&g_eventfd_registry.lock);
222 g_eventfd_registry.initialized = 1;
223 }
224
225 AcquireSRWLockExclusive(&g_eventfd_registry.lock);
226
227 if (fd < 0 || fd >= BRIX_WIN_EVENTFD_MAX) {
228 ReleaseSRWLockExclusive(&g_eventfd_registry.lock);
229 errno = EBADF;
230 return -1;
231 }
232
233 g_eventfd_registry.events[fd] = eventfd;
234 ReleaseSRWLockExclusive(&g_eventfd_registry.lock);
235 return 0;
236}

238static brix_win_eventfd_t *
239brix_win_eventfd_get(int fd)
240{
241 brix_win_eventfd_t *eventfd = NULL;
242
243 if (!g_eventfd_registry.initialized) {
244 return NULL;
245 }
246
247 AcquireSRWLockShared(&g_eventfd_registry.lock);
248
249 if (fd >= 0 && fd < BRIX_WIN_EVENTFD_MAX) {
250 eventfd = g_eventfd_registry.events[fd];
251 }
252
253 ReleaseSRWLockShared(&g_eventfd_registry.lock);
254 return eventfd;
255}

257static void
258brix_win_eventfd_unregister(int fd)
259{
260 if (!g_eventfd_registry.initialized) {
261 return;
262 }
263
264 AcquireSRWLockExclusive(&g_eventfd_registry.lock);
265
266 if (fd >= 0 && fd < BRIX_WIN_EVENTFD_MAX) {
267 g_eventfd_registry.events[fd] = NULL;
268 }
269
270 ReleaseSRWLockExclusive(&g_eventfd_registry.lock);
271}

273int
274brix_plat_eventfd(unsigned int initial_value, int flags)
275{
276 /*
277 * Create eventfd emulation using pipe
278 *
279 * Implementation:
280 * 1. Create anonymous pipe with proper flags
281 * 2. Allocate and initialize eventfd structure
282 * 3. Register in fd table for cleanup
283 * 4. Return read end as "eventfd"
284 *
285 * Windows-specific:
286 * - BRIX_EVENTFD_CLOEXEC: HANDLE_FLAG_INHERIT=FALSE
287 * - BRIX_EVENTFD_NONBLOCK: Uses overlapped I/O
288 * - Counter protected by CRITICAL_SECTION
289 * - Proper cleanup on close
290 *
291 * @param initial_value Initial counter value (0 or 1 typically)
292 * @param flags BRIX_EVENTFD_CLOEXEC | BRIX_EVENTFD_NONBLOCK
293 * @return Read end fd, or -1 on error
294 */
295 brix_win_eventfd_t *eventfd;
296 int pipefd[2];
297
298 /* Create pipe with flags */
299 if (brix_plat_pipe2(pipefd, flags) < 0) {
300 return -1;
301 }
302
303 /* Allocate eventfd structure */
304 eventfd = (brix_win_eventfd_t *)calloc(1, sizeof(brix_win_eventfd_t));
305 if (eventfd == NULL) {
306 _close(pipefd[0]);
307 _close(pipefd[1]);
308 errno = ENOMEM;
309 return -1;
310 }
311
312 eventfd->pipefd[0] = pipefd[0];
313 eventfd->pipefd[1] = pipefd[1];
314 eventfd->counter = initial_value;
315
316 InitializeCriticalSection(&eventfd->lock);
317
318 /* Register in fd table for proper cleanup */
319 if (brix_win_eventfd_register(pipefd[0], eventfd) < 0) {
320 DeleteCriticalSection(&eventfd->lock);
321 free(eventfd);
322 _close(pipefd[0]);
323 _close(pipefd[1]);
324 return -1;
325 }
326
327 /* Return read end as eventfd */
328 return pipefd[0];
329}

335int
336brix_plat_pipe2(int pipefd[2], int flags)
337{
338 /*
339 * Windows: CreatePipe + SetHandleInformation
340 *
341 * Implements POSIX pipe2() semantics:
342 * - flags & BRIX_PIPE_CLOEXEC: Set HANDLE_FLAG_INHERIT=FALSE
343 * - flags & BRIX_PIPE_NONBLOCK: Use overlapped I/O
344 *
345 * Windows-specific:
346 * - Anonymous pipes don't support PIPE_NOWAIT
347 * - Non-blocking mode requires overlapped I/O
348 * - For simplicity, we document this limitation
349 *
350 * @param pipefd Output: [read_fd, write_fd]
351 * @param flags BRIX_PIPE_CLOEXEC | BRIX_PIPE_NONBLOCK
352 * @return 0 on success, -1 on error
353 */
354 HANDLE read_handle, write_handle;
355 SECURITY_ATTRIBUTES sa;
356
357 if (pipefd == NULL) {
358 errno = EINVAL;
359 return -1;
360 }
361
362 /* Setup security attributes */
363 sa.nLength = sizeof(SECURITY_ATTRIBUTES);
364 sa.bInheritHandle = (flags & BRIX_PIPE_CLOEXEC) ? FALSE : TRUE;
365 sa.lpSecurityDescriptor = NULL;
366
367 /* Create pipe */
368 if (!CreatePipe(&read_handle, &write_handle, &sa, 0)) {
369 brix_win32_set_errno(GetLastError());
370 return -1;
371 }
372
373 /*
374 * BRIX_PIPE_NONBLOCK note:
375 * Anonymous pipes on Windows don't support true non-blocking mode.
376 * PIPE_NOWAIT only works on named pipes.
377 * For eventfd emulation, the CRITICAL_SECTION protects against blocking.
378 * Applications should use select/WaitForMultipleObjects for polling.
379 */
380
381 /* Set CLOEXEC if requested */
382 if (flags & BRIX_PIPE_CLOEXEC) {
383 SetHandleInformation(read_handle, HANDLE_FLAG_INHERIT, 0);
384 SetHandleInformation(write_handle, HANDLE_FLAG_INHERIT, 0);
385 }
386
387 /* Convert handles to file descriptors */
388 pipefd[0] = _open_osfhandle((intptr_t)read_handle, _O_RDONLY);
389 pipefd[1] = _open_osfhandle((intptr_t)write_handle, _O_WRONLY);
390
391 if (pipefd[0] < 0 || pipefd[1] < 0) {
392 CloseHandle(read_handle);
393 CloseHandle(write_handle);
394 if (pipefd[0] >= 0) _close(pipefd[0]);
395 if (pipefd[1] >= 0) _close(pipefd[1]);
396 errno = EMFILE;
397 return -1;
398 }
399
400 return 0;
401}

514/*
515 * Windows Event Handling Limitations vs epoll/kqueue:
516 *
517 * 1. Scalability:
518 * - epoll: O(1) scalability, thousands of fds
519 * - kqueue: O(log n) scalability, thousands of fds
520 * - Windows select: O(n) + FD_SETSIZE limit (64 by default)
521 * - WaitForMultipleObjects: 64 handle limit
522 * - IOCP: O(1) scalability (Phase 2 solution)
523 *
524 * 2. Event Types:
525 * - epoll: EPOLLIN, EPOLLOUT, EPOLLET, EPOLLONESHOT
526 * - kqueue: EVFILT_READ, EVFILT_WRITE, EV_ADD, EV_DELETE
527 * - Windows: FD_READ, FD_WRITE, FD_ACCEPT, FD_CONNECT, FD_CLOSE
528 * - Missing: Edge-triggered mode, priority events
529 *
530 * 3. Interface:
531 * - epoll: fd-based (epoll_create, epoll_ctl, epoll_wait)
532 * - kqueue: fd-based (kqueue, kevent)
533 * - Windows: HANDLE-based (CreateEvent, WaitForMultipleObjects)
534 * - Incompatible: Requires abstraction layer
535 *
536 * 4. Performance:
537 * - epoll: Very efficient, kernel-side ready list
538 * - kqueue: Very efficient, kernel-side event queue
539 * - select: O(n) scan of all fds
540 * - IOCP: Very efficient, completion-based (Phase 2)
541 *
542 * 5. Integration:
543 * - epoll: Integrates with select/poll
544 * - kqueue: Integrates with select/poll
545 * - Windows: Separate event system, requires translation
546 *
547 * Phase 2 Improvements:
548 * - Full IOCP implementation
549 * - Better integration with nginx event loop
550 * - Remove 64-handle limitation
551 * - Improve scalability to thousands of connections
552 */
