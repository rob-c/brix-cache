6/* WHAT: Attach an authorized file handle to its inode. WHY: BSD GETATTR can omit
7 * the handle even for fstat after unlink. HOW: 1. Record uid/list ownership.
8 * 2. Retain the inode. 3. Apply caller-safe kernel cache policy. */
9void
10xfs_ll_handle_attach(xfs_node *node, struct fuse_file_info *fi, uid_t uid)
11{
12 afh *handle = (afh *) (uintptr_t) fi->fh;
13 handle->caller_uid = uid;
14 handle->node_next = node->open_handles;
15 node->open_handles = handle;
16 node->opens++;
17 fi->keep_cache = g_kernel_cache && !g_sss_ident;
18 fi->direct_io = g_sss_ident;
19}

21/* WHAT: Remove one exact open owner from its inode. WHY: No freed handle can be
22 * selected by a concurrent getattr. HOW: 1. Find pointer. 2. Unchain under lock. */
23static void
24xfs_ll_handle_detach(xfs_node *node, afh *handle)
25{
26 afh **cursor = &node->open_handles;
27 while (*cursor && *cursor != handle) {
28 cursor = &(*cursor)->node_next;
29 }
30 if (*cursor) {
31 *cursor = handle->node_next;
32 }
33}

35/* WHAT: Open an authorized inode and retain its identity. WHY: Handle outlives name.
36 * HOW: 1. Stat/open under caller identity. 2. Cache authorized handle attributes.
37 * 3. Add open owner; roll back if the reply is interrupted. */
38void
39xfs_ll_open(fuse_req_t req, fuse_ino_t ino, struct fuse_file_info *fi)
40{
41 xfs_session *session = xfs_req_lock(req);
42 char path[XRDC_PATH_MAX];
43 struct stat attr;
44 int rc = xfs_req_path(session, ino, path);
45 if (rc == 0) { rc = xfs_getattr(xfs_req_uid(req), path, &attr, NULL); }
46 if (rc == 0) { rc = xfs_open(xfs_req_uid(req), path, fi); }
47 if (rc != 0) {
48 xfs_req_finish(req, session, rc);
49 return;
50 }
51 xfs_node *node = xfs_nodes_find(&session->nodes, ino);
52 afh *handle = (afh *) (uintptr_t) fi->fh;
53 attr.st_ino = (ino_t) ino;
54 handle->attr = attr;
55 handle->wsize = fi->flags & O_TRUNC ? 0 : attr.st_size;
56 xfs_ll_handle_attach(node, fi, xfs_req_uid(req));
57 if (fuse_reply_open(req, fi) != 0) {
58 xfs_ll_handle_detach(node, handle);
59 xfs_release(path, fi);
60 xfs_nodes_close(&session->nodes, node);
61 }
62 pthread_mutex_unlock(&session->mutex);
63}

65/* WHAT: Return a bounded read buffer. WHY: A kernel request cannot exhaust memory.
66 * HOW: 1. Validate negotiated request bounds. 2. Use the existing handle read.
67 * 3. Reply exact bytes and free the request-local buffer. */
68void
69xfs_ll_read(fuse_req_t req, fuse_ino_t ino, size_t size, off_t offset,
70 struct fuse_file_info *fi)
71{
72 (void) ino;
73 if (size > 16U * 1024U * 1024U || offset < 0) {
74 fuse_reply_err(req, EINVAL);
75 return;
76 }
77 char *buffer = malloc(size ? size : 1);
78 if (!buffer) {
79 fuse_reply_err(req, ENOMEM);
80 return;
81 }
82 int rc = xfs_read(NULL, buffer, size, offset, fi);
83 if (rc < 0) {
84 fuse_reply_err(req, -rc);
85 } else {
86 fuse_reply_buf(req, buffer, (size_t) rc);
87 }
88 free(buffer);
89}

91/* WHAT: Forward a bounded write. WHY: Preserve maintained buffering/error handling.
92 * HOW: 1. Reject impossible offsets/sizes. 2. Write by authorized handle.
93 * 3. Reply with the actual count, never conceal a failed write. */
94void
95xfs_ll_write(fuse_req_t req, fuse_ino_t ino, const char *buffer, size_t size,

132/* WHAT: Close a handle and its inode owner. WHY: Forget may have happened earlier.
133 * HOW: 1. Close outside namespace lock. 2. Drop open owner under lock. 3. Reply. */
134void
135xfs_ll_release(fuse_req_t req, fuse_ino_t ino, struct fuse_file_info *fi)
136{
137 xfs_session *session = xfs_req_lock(req);
138 xfs_node *node = xfs_nodes_find(&session->nodes, ino);
139 if (node) {
140 xfs_ll_handle_detach(node, (afh *) (uintptr_t) fi->fh);
141 }
142 pthread_mutex_unlock(&session->mutex);
143 int rc = xfs_release(NULL, fi);
144 pthread_mutex_lock(&session->mutex);
145 if (node) {
146 xfs_nodes_close(&session->nodes, node);
147 }
148 xfs_req_finish(req, session, rc);
149}
