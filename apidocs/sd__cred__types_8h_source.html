16/* Delegation credential kinds a backend can consume (phase-71). A backend ORs
17 * the kinds it accepts into brix_sd_driver_s.cred_accept; the VFS denies (EACCES)
18 * before touching the origin when the live credential kind is not accepted. */
19typedef enum {
20 BRIX_SD_CRED_NONE = 0,
21 BRIX_SD_CRED_BEARER = 1u << 0, /* raw JWT bearer text */
22 BRIX_SD_CRED_PROXY_PEM = 1u << 1, /* full x509 proxy PEM */
23 /* Local identity: the backend consumes only WHO the client is (principal +
24 * VO list) for its own ownership/enforcement model — no forwardable secret
25 * is required or minted, so no credential directory needs configuring.
26 * pblock's catalog-internal ownership registry is the consumer. */
27 BRIX_SD_CRED_IDENTITY = 1u << 2,
28 /* SSS identity injection (phase-70 §5.6 / P90-70.3): the backend re-issues
29 * an XrdSecsss credential to its origin ASSERTING the inbound caller's
30 * principal, signed with a gateway-held shared keytab. No forwardable
31 * secret comes from the client; the keytab is the export's
32 * (brix_backend_sss_keytab). sd_xroot is the consumer. */
33 BRIX_SD_CRED_SSS = 1u << 3,
34 /* S3 STS EXCHANGE (phase-70 §5.5): the backend signs the origin leg with a
35 * short-lived SigV4 keypair + session token the gate obtained by trading the
36 * node's S3 service credential for one scoped to the inbound principal
37 * (AssumeRole / GetSessionToken). sd_remote/S3 is the consumer — it reads
38 * s3_ak/s3_sk/s3_region and, when set, s3_session (X-Amz-Security-Token). */
39 BRIX_SD_CRED_S3 = 1u << 4,
40 /* krb5 GSSAPI EXCHANGE (phase-70 §5.7): the backend authenticates its origin
41 * leg AS the inbound user by forwarding the user's delegated TGT through a
42 * multi-leg GSSAPI negotiation (brix_cache_origin_auth_krb5). The forwardable
43 * TGT is carried async-safely as a 0600 FILE ccache PATH (krb5_ccache) — a
44 * live gss_cred_id_t is request-scoped and cannot ride the async fill task —
45 * plus the origin service principal (krb5_princ). A root:// origin that
46 * advertises krb5 is the consumer; the origin leg re-imports the cred from
47 * the ccache path via brix_krb5_cred_from_ccache. */
48 BRIX_SD_CRED_GSS_KRB5 = 1u << 5
49} brix_sd_cred_kind_t;

51/* Per-open user credential passed from the protocol handler to the storage
52 * driver so a remote backend can authenticate AS the client user (Phase 1:
53 * x509 proxy; Phase 2 T2: WLCG bearer token). Extended with the fields needed
54 * to re-resolve the credential for async/deferred flushes.
55 *
56 * WHAT: Borrowed pointers valid for the duration of the open() / staged_open()
57 * call. Drivers that defer the open (thread-pool) MUST copy the strings
58 * internally before returning — the caller's buffers may be freed once
59 * the vtable function returns.
60 *
61 * WHY: A per-open cred lets the VFS pass identity down to the driver without
62 * threading it through every intermediate layer; the driver is the only
63 * entity that knows how to present it to a specific remote protocol.
64 * The extra fields (key, cred_dir, fallback_deny) let decorator layers
65 * (sd_stage, sd_cache) embed a re-resolvable identity into their durable
66 * state so an async flush — possibly after a crash and restart — can
67 * re-authenticate as the original user rather than the service account.
68 *
69 * HOW: The gate fills exactly ONE credential kind: {x509_proxy}, {bearer},
70 * {s3_ak + s3_sk (+ s3_region)}, or {ceph_keyring + ceph_user}
71 * depending on the kind selected by ucred_select; the other kinds'
72 * fields are NULL. Drivers check the kind they support (sd_xroot:
73 * x509_proxy then bearer; sd_remote/S3: s3_ak; sd_ceph: ceph_keyring) —
74 * only one kind is ever set for a given open.
75 * A NULL cred or a driver with no open_cred slot falls back to the plain
76 * open slot (service credential / anonymous).
77 * sd_xroot reads x509_proxy OR bearer + principal; sd_remote reads
78 * s3_ak/s3_sk/s3_region (phase-3 T3) to re-init its SigV4 signer per
79 * open instead of the export's static access_key/secret_key/region;
80 * sd_ceph reads ceph_keyring/ceph_user (ceph-peruser item) to open a
81 * per-user librados connection instead of the export's static
82 * user/keyring.
83 * The extra fields (key, cred_dir, fallback_deny) are consumed by
84 * sd_stage / sd_cache and are not required by sd_xroot, sd_remote, or
85 * sd_ceph. */
86
87/* How the per-open credential in brix_sd_cred_t was obtained — the strategy the
88 * VFS gate resolved for the backend leg (phase-70 §4). SELECT (the default, 0)
89 * is the pre-phase-70 directory-lookup behaviour, so every existing caller that
90 * leaves the struct zeroed keeps the same meaning. The other modes are set by
91 * the delegation gate (vfs_deleg.c) when the front door captured a forwardable
92 * credential:
93 * PASSTHROUGH — replay the exact credential the user presented (bearer bytes;
94 * a user-supplied full x509 proxy incl. private key);
95 * EXCHANGE — trade the inbound credential for a backend-valid one (RFC 8693
96 * token-exchange; S3 STS; GSSAPI krb5 forwarding);
97 * DELEGATE — obtain a fresh short-lived proxy via a GridSite handshake;
98 * MINT — mint a fresh short-lived proxy from a local CA;
99 * AUTO — dispatch by id->auth_method (§2 matrix).
100 * The field is advisory metadata for audit/metrics and for the async re-acquire
101 * record; the cred's byte/path fields still say WHICH credential to present. */
102enum brix_cred_mode {
103 BRIX_CRED_SELECT = 0,
104 BRIX_CRED_PASSTHROUGH,
105 BRIX_CRED_EXCHANGE,
106 BRIX_CRED_DELEGATE,
107 BRIX_CRED_MINT,
108 BRIX_CRED_AUTO
109};

111typedef struct {
112 const char *x509_proxy; /* path to per-user proxy PEM (NULL unless x509 cred) */
113 const char *bearer; /* WLCG bearer token text (NULL unless bearer cred) */
114 const char *s3_ak; /* S3 access key id (NULL unless s3 cred) */
115 const char *s3_sk; /* S3 secret key (NULL unless s3 cred; never log) */
116 const char *s3_region; /* S3 region (NULL unless s3 cred) */
117 const char *s3_session; /* STS session token → X-Amz-Security-Token (NULL for */
118 /* a static keypair; set only for EXCHANGE; never log)*/
119 const char *ceph_keyring; /* CephX keyring PATH (NULL unless ceph cred; never */
120 /* log its contents) */
121 const char *ceph_user; /* bare CephX user id, e.g. "bob" (NULL unless ceph */
122 /* cred) */
123 const char *sss_keytab; /* SSS identity-injection keytab PATH (NULL unless the
124 * delegation gate resolved SSS injection; the driver
125 * then asserts `principal` to the origin via SSS,
126 * signed with this keytab — phase-70 §5.6) */
127 const char *krb5_ccache; /* krb5 forwarded-TGT FILE ccache PATH (NULL unless the
128 * gate resolved krb5 GSSAPI EXCHANGE; the origin leg
129 * re-imports the delegated cred from it and negotiates
130 * AS the caller — phase-70 §5.7; never log) */
131 const char *krb5_princ; /* origin service principal for the krb5 leg, e.g.
132 * host/<backend-fqdn>@REALM (NULL unless krb5 cred) */
133 const char *key; /* credential-dir lookup key (audit + flush re-resolve) */
134 const char *principal; /* authenticated principal (audit/ledger; may be NULL) */
135 const char *vos; /* comma-separated VO/group names of the principal */
136 /* (NULL or "" when none; consumed by IDENTITY drivers) */
137 const char *cred_dir; /* export credential directory (flush re-resolve) */
138 enum brix_cred_mode mode; /* how this cred was obtained (phase-70; 0 = SELECT) */
139 unsigned fallback_deny:1; /* 1 = service-credential fallback forbidden */
140} brix_sd_cred_t;
