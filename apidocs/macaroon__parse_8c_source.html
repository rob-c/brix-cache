1/*
2 * Macaroon HMAC-SHA256 signature chain.
3 * Packet framing, per-type chain updates, third-party caveat capture,
4 * final signature verification.
5 *
6 * WHAT: Parses one macaroon binary as sequence of length-prefixed packets.
7 * Reconstructs HMAC-SHA256 chain (sig = HMAC(sig_prev, packet_data)) across:
8 * - identifier packets
9 * - location packets
10 * - cid (caveat identifier) packets
11 * - vid (verification identifier) packets
12 * - signature packets
13 * Captures third-party caveats (cid+vid+sig_before) into caller's tp_arr.
14 * Verifies final signature in constant time.
15 * Enforces fail-closed expiry.
16 * Finalizes extracted scopes into claims.
17 *
18 * WHY: Split out of macaroon.c (phase-79 file-size split).
19 * HMAC chain is integrity core of macaroon security model —
20 * any tampered or reordered caveat yields mismatched final signature.
21 * Keeping exact HMAC ordering in one file makes guarantee auditable:
22 * - identifier seeds the chain
23 * - cid/vid each fold in before capture
24 * - signature compared with CRYPTO_memcmp (constant-time)
25 * Separate from caveat interpretation (macaroon_caveats.c)
26 * and vid crypto (macaroon_crypto.c).
27 *
28 * HOW: Two-stage process:
29 * 1. brix_macaroon_packet_len() reads 4-char hex length prefix
30 * 2. macaroon_parse_core() loops packets, dispatching via macaroon_dispatch_packet():
31 * - identifier: seeds sig = HMAC(key, id)
32 * - cid: saves sig_before_cid, folds caveat,
33 * hands first-party to macaroon_parse_first_party_caveat()
34 * - vid: folds and records third-party triple (cid+vid+sig_before)
35 * - signature: compares with CRYPTO_memcmp
36 * 3. After loop: checks found_sig/expiry, finalizes scopes
37 */

58static ngx_int_t
59macaroon_packet_identifier(brix_macaroon_parse_state_t *state,
60 const u_char *data, size_t data_len)
61{
62 const u_char *identifier = data + 11;
63 size_t identifier_len = data_len - 11;
64
65 HMAC(EVP_sha256(), state->key, state->key_len, identifier,
66 identifier_len, state->sig, &state->sig_out_len);
67 ngx_memcpy(state->claims->sub, identifier,
68 identifier_len < sizeof(state->claims->sub)
69 ? identifier_len : sizeof(state->claims->sub) - 1);
70 state->found_id = 1;
71 state->have_last_cid = 0;
72 return NGX_OK;
73}

87static ngx_int_t
88macaroon_packet_cid(brix_macaroon_parse_state_t *state,
89 const u_char *data, size_t data_len)
90{
91 const u_char *caveat = data + 4;
92 size_t caveat_len = data_len - 4;
93 u_char next_sig[32];
94
95 if (!state->found_id) {
96 ngx_log_error(NGX_LOG_WARN, state->log, 0,
97 "brix_macaroon: caveat before identifier");
98 return NGX_ERROR;
99 }
100
101 ngx_memcpy(state->sig_before_cid, state->sig, 32);
102 HMAC(EVP_sha256(), state->sig, 32, caveat, caveat_len, next_sig,
103 &state->sig_out_len);
104 ngx_memcpy(state->sig, next_sig, 32);
105
106 state->last_cid_len = caveat_len < sizeof(state->last_cid)
107 ? caveat_len : sizeof(state->last_cid) - 1;
108 ngx_memcpy(state->last_cid, caveat, state->last_cid_len);
109 state->have_last_cid = 1;
110 macaroon_parse_first_party_caveat(state, caveat, caveat_len);
111 return NGX_OK;
112}

114static void
115macaroon_record_third_party_caveat(brix_macaroon_parse_state_t *state,
116 const u_char *vid_data, size_t vid_len)
117{
118 brix_macaroon_tp_t *tp;
119
120 if (!state->have_last_cid || state->tp_arr == NULL || state->n_tp == NULL
121 || *state->n_tp >= state->max_tp) {
122 return;
123 }
124
125 tp = &state->tp_arr[(*state->n_tp)++];
126 ngx_memcpy(tp->cid, state->last_cid, state->last_cid_len);
127 tp->cid_len = state->last_cid_len;
128 tp->vid_len = vid_len < sizeof(tp->vid) ? vid_len : sizeof(tp->vid) - 1;
129 ngx_memcpy(tp->vid, vid_data, tp->vid_len);
130 ngx_memcpy(tp->sig_before, state->sig_before_cid, 32);
131}

165static ngx_int_t
166macaroon_dispatch_packet(brix_macaroon_parse_state_t *state,
167 const u_char *data, size_t data_len)
168{
169 if (data_len >= 11 && memcmp(data, "identifier ", 11) == 0) {
170 return macaroon_packet_identifier(state, data, data_len);
171 }
172 if (data_len >= 9 && memcmp(data, "location ", 9) == 0) {
173 macaroon_packet_location(state, data, data_len);
174 return NGX_OK;
175 }
176 if (data_len >= 4 && memcmp(data, "cid ", 4) == 0) {
177 return macaroon_packet_cid(state, data, data_len);
178 }
179 if (data_len >= 4 && memcmp(data, "vid ", 4) == 0) {
180 return macaroon_packet_vid(state, data, data_len);
181 }
182 if (data_len >= 10 && memcmp(data, "signature ", 10) == 0) {
183 return macaroon_packet_signature(state, data, data_len);
184 }
185 return NGX_OK;
186}

188static ngx_int_t
189macaroon_check_expiry(brix_macaroon_parse_state_t *state)
190{
191 time_t now = time(NULL);
192
193 if (state->tp_arr != NULL && state->claims->exp <= 0) {
194 ngx_log_error(NGX_LOG_WARN, state->log, 0,
195 "brix_macaroon: rejected — no before: (expiry) caveat; "
196 "non-expiring macaroons are not accepted");
197 return NGX_ERROR;
198 }
199 if (state->claims->exp > 0 && now > (time_t)state->claims->exp) {
200 ngx_log_error(NGX_LOG_WARN, state->log, 0,
201 "brix_macaroon: token expired at %L (now=%L)",
202 (long long)state->claims->exp, (long long)now);
203 return NGX_ERROR;
204 }
205 return NGX_OK;
206}

238/*
239 * WHAT: Return the packet content length after its optional text delimiter.
240 * WHY: Signature packets contain raw HMAC bytes with no delimiter; a final
241 * 0x0a is part of the signature and must reach constant-time verification.
242 * HOW: 1. Preserve signature content. 2. Strip one newline from other packets.
243 */
244static size_t
245macaroon_packet_content_len(const u_char *data, size_t data_len)
246{
247 if (data_len >= 10 && memcmp(data, "signature ", 10) == 0) {
248 return data_len;
249 }
250 if (data_len > 0 && data[data_len - 1] == '\n') {
251 return data_len - 1;
252 }
253 return data_len;
254}

256/*
257 * WHAT: Parse one macaroon binary, reconstruct HMAC-SHA256 signature chain.
258 * Verify final signature and extract WLCG caveats into claims.
259 *
260 * WHY: Macaroon security model requires each caveat to deterministically
261 * modify HMAC chain: sig = HMAC(sig_prev, caveat_data).
262 * Ensures any tampered or reordered caveat produces mismatched final signature.
263 * Extracting activity:/path:/before: caveats converts raw binary authorization
264 * into structured claims for access control decisions.
265 *
266 * HOW: Multi-stage packet parsing:
267 *
268 * Initialization:
269 * - sig = HMAC(key, identifier)
270 * - scope_buf = ""
271 * - path_caveats[] = empty
272 * - last_cid = NULL, sig_before_cid = NULL
273 *
274 * Packet loop (p+4 ≤ end):
275 * 1. brix_macaroon_packet_len(p) → plen
276 * 2. data = p+4, dlen = plen-4
277 * 3. Strip trailing newline if present
278 * 4. Dispatch by packet type:
279 *
280 * "identifier ":
281 * - HMAC(EVP_sha256, key, identifier) → sig
282 * - Copy identifier to claims->sub
283 *
284 * "location ":
285 * - Copy to claims->iss
286 *
287 * "cid ":
288 * - Save sig_before_cid
289 * - HMAC(sig, cid) → next_sig → sig
290 * - Track last_cid for vid pairing
291 * - Parse first-party caveats within cid data:
292 * * activity: → scope mapping
293 * * before: → parse_iso8601 → claims->exp (minimum)
294 * * path: → path_caveats array
295 *
296 * "vid ":
297 * - HMAC(sig, vid_data) → sig
298 * - Record (cid+vid+sig_before) triple into tp_arr if available
299 *
300 * "signature ":
301 * - Compare provided 32-byte sig against computed sig
302 * - Reject mismatch via CRYPTO_memcmp
303 *
304 * Post-loop validation:
305 * - Check found_sig and found_id flags
306 * - Validate expiry: now ≤ claims->exp (fail-closed)
307 * - Finalize scopes from scope_buf via brix_token_parse_scopes()
308 * - Apply path caveats via macaroon_apply_path_caveats()
309 *
310 * Returns: 0 on success, -1 on failure
311 */
312int
313macaroon_parse_core(const macaroon_parse_input_t *in,
314 const u_char *bin, size_t bin_len)
315{
316 brix_macaroon_parse_state_t state;
317 ngx_log_t *log = in->log;
318 const u_char *p, *end;
319
320 macaroon_parse_state_init(&state, in);
321 p = bin;
322 end = bin + bin_len;
323
324 while (p + 4 <= end) {
325 int plen = brix_macaroon_packet_len(p);
326 u_char *data;
327 size_t dlen;
328
329 if (plen < 4 || p + plen > end) {
330 ngx_log_error(NGX_LOG_WARN, log, 0,
331 "brix_macaroon: malformed packet length: %d", plen);
332 return -1;
333 }
334
335 data = (u_char *)(p + 4);
336 dlen = (size_t)(plen - 4);
337
338 dlen = macaroon_packet_content_len(data, dlen);
339
340 if (macaroon_dispatch_packet(&state, data, dlen) != NGX_OK) {
341 return -1;
342 }
343
344 p += plen;
345 }
346
347 if (!state.found_sig) {
348 ngx_log_error(NGX_LOG_WARN, log, 0,
349 "brix_macaroon: no valid signature found");
350 return -1;
351 }
352
353 /*
354 * Expiry handling (fail-closed). A macaroon is a bearer credential: a root
355 * with no before: caveat leaves claims->exp == 0 and would otherwise be valid
356 * forever, so a single leak would never lapse. We therefore REQUIRE an
357 * expiry on the root/standalone macaroon (tp_arr != NULL identifies that
358 * context; discharges are validated with tp_arr == NULL). dCache/WLCG
359 * macaroons always carry a before: caveat on the root, so this rejects only
360 * malformed or deliberately-unbounded tokens. Discharge macaroons may
361 * legitimately omit before: — their lifetime is governed by the root and
362 * intersected by the caller — so we only enforce "not already expired" for
363 * them. claims->exp is the earliest before: caveat seen in the packet loop
364 * above (macaroons only narrow, never widen).
365 */
366 if (macaroon_check_expiry(&state) != NGX_OK) {
367 return -1;
368 }
369
370 macaroon_finalize_scopes(&state);
371 return 0;
372}
