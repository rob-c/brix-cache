1/*
2 * Macaroon HMAC-SHA256 signature chain.
3 * Packet framing, per-type chain updates, third-party caveat capture,
4 * final signature verification.
5 *
6 * WHAT: Parses one macaroon binary as sequence of length-prefixed packets.
7 * Reconstructs HMAC-SHA256 chain (sig = HMAC(sig_prev, packet_data)) across:
8 * - identifier packets
9 * - location packets
10 * - cid (caveat identifier) packets
11 * - vid (verification identifier) packets
12 * - cl (optional third-party caveat location) packets
13 * - signature packets
14 * Captures third-party caveats (cid+vid+sig_before) into caller's tp_arr.
15 * Verifies final signature in constant time.
16 * Enforces fail-closed expiry.
17 * Finalizes extracted scopes into claims.
18 *
19 * WHY: Split out of macaroon.c (phase-79 file-size split).
20 * HMAC chain is integrity core of macaroon security model —
21 * any tampered or reordered caveat yields mismatched final signature.
22 * Keeping exact HMAC ordering in one file makes guarantee auditable:
23 * - identifier seeds the chain
24 * - cid/vid each fold in before capture
25 * - signature compared with CRYPTO_memcmp (constant-time)
26 * Separate from caveat interpretation (macaroon_caveats.c)
27 * and vid crypto (macaroon_crypto.c).
28 *
29 * HOW: Two-stage process:
30 * 1. brix_macaroon_packet_len() reads 4-char hex length prefix
31 * 2. macaroon_parse_core() loops packets under a strict grammar:
32 * - identifier: seeds sig = HMAC(key, id)
33 * - cid: folds and remains pending until the next packet determines
34 * whether it is first-party or paired with an exact vid
35 * - vid: folds and records the third-party triple
36 * - signature: finalizes a pending first-party cid and compares
37 * 3. Reject residue after the terminal signature, then check expiry and
38 * finalize intersected scopes
39 */

61/* ---- Validate a text packet value before copying or interpreting it ----
62 *
63 * WHAT: Return NGX_OK for nonempty control-free bytes, else NGX_ERROR.
64 * WHY: Embedded NUL/newline bytes make signed identifiers differ from their
65 * C-string claims and can hide packet-like content in diagnostics.
66 * HOW: Scan the bounded value and reject C0 controls and DEL.
67 */
68static ngx_int_t
69macaroon_text_value_valid(const u_char *value, size_t value_len)
70{
71 size_t index;
72
73 if (value_len == 0) {
74 return NGX_ERROR;
75 }
76 for (index = 0; index < value_len; index++) {
77 if (value[index] < 0x20 || value[index] == 0x7f) {
78 return NGX_ERROR;
79 }
80 }
81 return NGX_OK;
82}

84/* ---- Seed the chain from the one required identifier packet ----
85 *
86 * WHAT: Validate and record a unique nonempty identifier; return NGX_OK or
87 * NGX_ERROR without truncation.
88 * WHY: Reseeding the chain with a duplicate identifier discards all earlier
89 * caveat integrity and lets later claims replace the authenticated subject.
90 * HOW: Enforce grammar/bounds, compute HMAC(root key, identifier), and copy the
91 * exact identifier into the zero-terminated subject claim.
92 */
93static ngx_int_t
94macaroon_packet_identifier(brix_macaroon_parse_state_t *state,
95 const u_char *data, size_t data_len)
96{
97 const u_char *identifier = data + 11;
98 size_t identifier_len = data_len - 11;
99
100 if (state->found_id || state->have_last_cid
101 || identifier_len >= sizeof(state->claims->sub)
102 || macaroon_text_value_valid(identifier, identifier_len) != NGX_OK) {
103 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
104 "brix_macaroon: invalid or duplicate identifier packet");
105 return NGX_ERROR;
106 }
107 if (HMAC(EVP_sha256(), state->key, state->key_len, identifier,
108 identifier_len, state->sig, &state->sig_out_len) == NULL
109 || state->sig_out_len != sizeof(state->sig)) {
110 return NGX_ERROR;
111 }
112 ngx_memcpy(state->claims->sub, identifier, identifier_len);
113 state->claims->sub[identifier_len] = '\0';
114 state->found_id = 1;
115 return NGX_OK;
116}

145/* ---- Commit a pending cid as a first-party caveat ----

163/* ---- Begin one pending caveat and extend the HMAC chain ----
164 * WHAT: Validate one cid, finalize any prior first-party cid, and retain this
165 * cid for first/third-party classification; return NGX_OK or error.
166 * WHY: Duplicate/truncated identifiers and reordered chain updates are unsafe.
167 * HOW: Check grammar/bounds, HMAC the exact bytes, and save exact pending data.
168 */
169static ngx_int_t
170macaroon_packet_cid(brix_macaroon_parse_state_t *state,
171 const u_char *data, size_t data_len)
172{
173 const u_char *caveat = data + 4;
174 size_t caveat_len = data_len - 4;
175 u_char next_sig[32];
176
177 if (!state->found_id) {
178 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
179 "brix_macaroon: caveat before identifier");
180 return NGX_ERROR;
181 }
182 if (caveat_len >= sizeof(state->last_cid)
183 || macaroon_text_value_valid(caveat, caveat_len) != NGX_OK) {
184 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
185 "brix_macaroon: invalid or oversized caveat identifier");
186 return NGX_ERROR;
187 }
188 if (macaroon_finalize_pending_cid(state) != NGX_OK) {
189 return NGX_ERROR;
190 }
191
192 ngx_memcpy(state->sig_before_cid, state->sig, 32);
193 if (HMAC(EVP_sha256(), state->sig, 32, caveat, caveat_len, next_sig,
194 &state->sig_out_len) == NULL
195 || state->sig_out_len != sizeof(next_sig)) {
196 OPENSSL_cleanse(next_sig, sizeof(next_sig));
197 return NGX_ERROR;
198 }
199 ngx_memcpy(state->sig, next_sig, 32);
200 OPENSSL_cleanse(next_sig, sizeof(next_sig));
201
202 state->last_cid_len = caveat_len;
203 ngx_memcpy(state->last_cid, caveat, state->last_cid_len);
204 state->last_cid[state->last_cid_len] = '\0';
205 state->have_last_cid = 1;
206 return NGX_OK;
207}

209/* ---- Capture one exact third-party caveat without truncation ----
210 *
211 * WHAT: Store pending cid, 48-byte vid, and pre-cid signature; return NGX_OK
212 * or NGX_ERROR for nested, excess, or misconfigured capture.
213 * WHY: Dropping an excess discharge requirement or truncating encrypted key
214 * material can turn an unsatisfied condition into accepted authority.
215 * HOW: Validate root capture state/capacity, reserve one slot, and copy exact
216 * bounded fields.
217 */
218static ngx_int_t
219macaroon_record_third_party_caveat(brix_macaroon_parse_state_t *state,
220 const u_char *vid_data, size_t vid_len)
221{
222 brix_macaroon_tp_t *tp;
223
224 if (!state->have_last_cid || state->tp_arr == NULL || state->n_tp == NULL
225 || state->max_tp <= 0 || *state->n_tp < 0
226 || *state->n_tp >= state->max_tp
227 || vid_len != BRIX_MACAROON_VID_LEN) {
228 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
229 "brix_macaroon: nested, excess, or malformed third-party caveat");
230 return NGX_ERROR;
231 }
232
233 tp = &state->tp_arr[(*state->n_tp)++];
234 ngx_memcpy(tp->cid, state->last_cid, state->last_cid_len);
235 tp->cid_len = state->last_cid_len;
236 tp->vid_len = vid_len;
237 ngx_memcpy(tp->vid, vid_data, tp->vid_len);
238 ngx_memcpy(tp->sig_before, state->sig_before_cid, 32);
239 return NGX_OK;
240}

243 * WHAT: Fold and capture an exact 48-byte vid; return NGX_OK or error.
244 * WHY: Orphan, nested, excess, and oversized vid records must fail closed.

309/* ---- Verify the one exact terminal signature packet ----
310 * WHAT: Verify a 32-byte HMAC in constant time; return NGX_OK or error.
311 * WHY: Short, duplicate, or pre-identifier signatures cannot authenticate.
312 * HOW: Finalize pending first-party cid, check exact size, and compare bytes.
313 */
314static ngx_int_t
315macaroon_packet_signature(brix_macaroon_parse_state_t *state,
316 const u_char *data, size_t data_len)
317{
318 const u_char *provided_sig = data + 10;
319
320 if (!state->found_id || state->found_sig || data_len != 10 + 32) {
321 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
322 "brix_macaroon: malformed or duplicate signature packet");
323 return NGX_ERROR;
324 }
325 if (macaroon_finalize_pending_cid(state) != NGX_OK) {
326 return NGX_ERROR;
327 }

329 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
330 "brix_macaroon: signature mismatch");
331 return NGX_ERROR;
332 }
333 state->found_sig = 1;
334 return NGX_OK;

337/* ---- Dispatch one recognized packet under the strict grammar ----
338 * WHAT: Route a packet to its typed handler; reject unknown/post-signature data.
339 * WHY: Ignored signed or unsigned packet types create semantic ambiguity.
340 * HOW: Match exact field prefixes in grammar order and fail closed otherwise.
341 */
342static ngx_int_t
343macaroon_dispatch_packet(brix_macaroon_parse_state_t *state,
344 const u_char *data, size_t data_len)
345{
346 if (state->found_sig) {
347 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
348 "brix_macaroon: packet follows terminal signature");
349 return NGX_ERROR;
350 }
351 if (data_len >= 3 && memcmp(data, "cl ", 3) == 0) {
352 return macaroon_packet_caveat_location(state, data, data_len);
353 }
354 state->caveat_location_allowed = 0;
355 if (data_len >= 11 && memcmp(data, "identifier ", 11) == 0) {
356 return macaroon_packet_identifier(state, data, data_len);
357 }
358 if (data_len >= 9 && memcmp(data, "location ", 9) == 0) {
359 return macaroon_packet_location(state, data, data_len);
360 }
361 if (data_len >= 4 && memcmp(data, "cid ", 4) == 0) {
362 return macaroon_packet_cid(state, data, data_len);
363 }
364 if (data_len >= 4 && memcmp(data, "vid ", 4) == 0) {
365 return macaroon_packet_vid(state, data, data_len);
366 }
367 if (data_len >= 10 && memcmp(data, "signature ", 10) == 0) {
368 return macaroon_packet_signature(state, data, data_len);
369 }
370 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
371 "brix_macaroon: unknown packet type");
372 return NGX_ERROR;
373}

375/* ---- Enforce fail-closed root expiry and elapsed deadlines ----
376 * WHAT: Return NGX_OK for a live credential, otherwise NGX_ERROR.
377 * WHY: Root bearer credentials without a deadline remain useful forever.
378 * HOW: Require explicit root expiry and reject every seen deadline at or before
379 * wall time, including epoch-zero and pre-epoch values.
380 */
381static ngx_int_t
382macaroon_check_expiry(brix_macaroon_parse_state_t *state)
383{
384 time_t now = time(NULL);
385
386 if (state->tp_arr != NULL && !state->before_seen) {
387 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
388 "brix_macaroon: rejected — no before: (expiry) caveat; "
389 "non-expiring macaroons are not accepted");
390 return NGX_ERROR;
391 }
392 if (state->before_seen
393 && (now == (time_t)-1 || state->claims->exp <= (int64_t)now)) {
394 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
395 "brix_macaroon: token expired at %L (now=%L)",
396 (long long)state->claims->exp, (long long)now);
397 return NGX_ERROR;
398 }
399 return NGX_OK;
400}

402/* ---- Finalize effective scopes and optional parse metadata ----
403 *
404 * WHAT: Materialize activity scopes, apply path restrictions, and copy
405 * caveat-presence metadata; return NGX_OK or NGX_ERROR on overflow.
406 * WHY: Authorization consumers need one coherent, fully narrowed claims view,
407 * while discharge orchestration must distinguish no restriction from an
408 * activity intersection that intentionally grants nothing.
409 * HOW: Parse the bounded scope string, apply every path caveat, then copy the
410 * validated metadata only after all narrowing succeeds.
411 */
412static ngx_int_t
413macaroon_finalize_scopes(brix_macaroon_parse_state_t *state,
414 brix_macaroon_parse_meta_t *meta)
415{
416 ngx_memcpy(state->claims->scope_raw, state->scope_buf, state->scope_off);
417 state->claims->scope_raw[state->scope_off] = '\0';
418 state->claims->scope_count = brix_token_parse_scopes(
419 state->claims->scope_raw, state->claims->scopes,
420 BRIX_MAX_TOKEN_SCOPES);
421
422 if (state->n_path_caveats > 0) {
423 if (macaroon_apply_path_caveats(state->log, state->claims,
424 state->path_caveats,
425 state->n_path_caveats) != NGX_OK) {
426 return NGX_ERROR;
427 }
428 }
429 if (meta != NULL) {
430 ngx_memzero(meta, sizeof(*meta));
431 meta->activity_seen = state->activity_seen;
432 meta->before_seen = state->before_seen;
433 meta->n_path_caveats = state->n_path_caveats;
434 if (state->n_path_caveats > 0) {
435 ngx_memcpy(meta->path_caveats, state->path_caveats,
436 (size_t)state->n_path_caveats
437 * sizeof(state->path_caveats[0]));
438 }
439 }
440 return NGX_OK;
441}

496/* WHAT: Dispatch every complete packet and require a terminal signature.
497 * WHY: Unknown, malformed, or trailing packets make authorization ambiguous.
498 * HOW: Decode bounded frames, preserve raw signature bytes, and stop on error. */
499static ngx_int_t
500macaroon_parse_packets(brix_macaroon_parse_state_t *state,
501 const u_char *bin, size_t bin_len)
502{
503 const u_char *p = bin;
504 const u_char *end = bin + bin_len;
505
506 while (p < end) {
507 size_t remaining = (size_t)(end - p);
508 int plen;
509 const u_char *data;
510 size_t raw_len, data_len;
511
512 if (remaining < 4) {
513 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
514 "brix_macaroon: trailing bytes after final packet");
515 return NGX_ERROR;
516 }
517 plen = brix_macaroon_packet_len(p);
518 if (plen < 4 || (size_t)plen > remaining) {
519 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
520 "brix_macaroon: malformed packet length: %d", plen);
521 return NGX_ERROR;
522 }
523
524 data = p + 4;
525 raw_len = (size_t)(plen - 4);
526 if (macaroon_packet_content_len(data, raw_len, &data_len) != NGX_OK) {
527 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
528 "brix_macaroon: malformed packet delimiter");
529 return NGX_ERROR;
530 }
531 if (macaroon_dispatch_packet(state, data, data_len) != NGX_OK) {
532 return NGX_ERROR;
533 }
534 if (state->found_sig && (size_t)plen != remaining) {
535 BRIX_LOG(NGX_LOG_WARN, state->log, 0,
536 "brix_macaroon: signature packet is not terminal");
537 return NGX_ERROR;
538 }
539 p += plen;
540 }
541 return NGX_OK;
542}

544/* WHAT: Strictly parse and authenticate one macaroon into claims.
545 * WHY: Every signed condition and framing byte must be enforced without
546 * truncation or reinterpretation. HOW: Parse packets, require identifier plus
547 * terminal signature, enforce expiry, then publish intersected scopes. */
548int
549macaroon_parse_core(const macaroon_parse_input_t *in,
550 const u_char *bin, size_t bin_len)
551{
552 brix_macaroon_parse_state_t state;
553
554 if (macaroon_parse_input_valid(in, bin, bin_len) != NGX_OK) {
555 return -1;
556 }
557 if (in->meta != NULL) {
558 ngx_memzero(in->meta, sizeof(*in->meta));
559 }
560 macaroon_parse_state_init(&state, in);
561 if (macaroon_parse_packets(&state, bin, bin_len) != NGX_OK) {
562 return -1;
563 }
564
565 if (!state.found_id || !state.found_sig) {
566 BRIX_LOG(NGX_LOG_WARN, state.log, 0,
567 "brix_macaroon: identifier or terminal signature missing");
568 return -1;
569 }
570
571 if (macaroon_check_expiry(&state) != NGX_OK) {
572 return -1;
573 }
574
575 if (macaroon_finalize_scopes(&state, in->meta) != NGX_OK) {
576 return -1;
577 }
578 return 0;
579}
