10int
11sd_gsiftp_copy(char *dst, size_t cap, const char *src)
12{
13 size_t len = src != NULL ? strlen(src) : 0;
14
15 if (len >= cap) {
16 errno = ENAMETOOLONG;
17 return -1;
18 }
19 if (len != 0) {
20 memcpy(dst, src, len);
21 }
22 dst[len] = '\0';
23 return 0;
24}

26static int
27sd_gsiftp_component_safe(const char *start, size_t len)
28{
29 if ((len == 1 && start[0] == '.')
30 || (len == 2 && start[0] == '.' && start[1] == '.')) {
31 return 0;
32 }
33 while (len-- != 0) {
34 unsigned char ch = (unsigned char) *start++;
35
36 if (ch < 0x20 || ch == 0x7f || ch == '\\') {
37 return 0;
38 }
39 }
40 return 1;
41}

43static int
44sd_gsiftp_logical_safe(const char *path)
45{
46 const char *part;
47 const char *cursor;
48
49 if (path == NULL || path[0] != '/') {
50 return 0;
51 }
52 part = path + 1;
53 for (cursor = part;; cursor++) {
54 if (*cursor != '/' && *cursor != '\0') {
55 continue;
56 }
57 if (!sd_gsiftp_component_safe(part, (size_t) (cursor - part))) {
58 return 0;
59 }
60 if (*cursor == '\0') {
61 return 1;
62 }
63 part = cursor + 1;
64 }
65}

67int
68sd_gsiftp_path(const sd_gsiftp_state *state, const char *logical,
69 char out[GSIFTP_PATH_CAP])
70{
71 const char *base = state->base_path;
72 int n;
73
74 if (!sd_gsiftp_logical_safe(logical)) {
75 errno = EACCES;
76 return -1;
77 }
78 if (base[0] == '\0' || strcmp(base, "/") == 0) {
79 n = snprintf(out, GSIFTP_PATH_CAP, "%s", logical);
80 } else {
81 n = snprintf(out, GSIFTP_PATH_CAP, "%s%s", base, logical);
82 }
83 if (n < 0 || n >= GSIFTP_PATH_CAP) {
84 errno = ENAMETOOLONG;
85 return -1;
86 }
87 return 0;

90const char *
91sd_gsiftp_proxy(const sd_gsiftp_state *state, const brix_sd_cred_t *cred,
92 int *err_out)
93{
94 if (cred != NULL && cred->x509_proxy != NULL
95 && cred->x509_proxy[0] != '\0') {
96 return cred->x509_proxy;
97 }
98 if (cred != NULL && cred->fallback_deny) {
99 errno = EACCES;
100 if (err_out != NULL) {
101 *err_out = EACCES;
102 }
103 return NULL;
104 }
105 if (state->require_gsi && state->x509_proxy[0] == '\0') {
106 errno = EACCES;
107 if (err_out != NULL) {
108 *err_out = EACCES;
109 }
110 return NULL;
111 }
112 return state->x509_proxy[0] != '\0' ? state->x509_proxy : NULL;
113}

187static int
188sd_gsiftp_fill_state(sd_gsiftp_state *state,
189 const brix_sd_gsiftp_cfg_t *cfg)
190{
191 if (sd_gsiftp_copy(state->host, sizeof(state->host), cfg->host) != 0
192 || sd_gsiftp_copy(state->base_path, sizeof(state->base_path),
193 cfg->base_path != NULL ? cfg->base_path : "") != 0
194 || sd_gsiftp_copy(state->x509_proxy, sizeof(state->x509_proxy),
195 cfg->x509_proxy) != 0
196 || sd_gsiftp_copy(state->ca_dir, sizeof(state->ca_dir),
197 cfg->ca_dir) != 0) {
198 return -1;
199 }
200 if (state->base_path[0] != '\0'
201 && !sd_gsiftp_logical_safe(state->base_path)) {
202 errno = EINVAL;
203 return -1;
204 }
205 while (state->base_path[1] != '\0'
206 && state->base_path[strlen(state->base_path) - 1] == '/') {
207 state->base_path[strlen(state->base_path) - 1] = '\0';
208 }
209 state->port = cfg->port;
210 state->require_gsi = cfg->require_gsi != 0;
211 state->timeout_ms = cfg->timeout_ms > 0 ? cfg->timeout_ms : 30000;
212 return 0;
213}

233{
234 brix_sd_instance_t *inst;
235 sd_gsiftp_state *state;
236

238 errno = EINVAL;

241 inst = sd_gsiftp_alloc_instance(&state);
