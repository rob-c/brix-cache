1/*
2 * envalias.c — shared env-var alias resolver (spec WS-1 change 1.1).
3 *
4 * WHAT: brix_env_resolve() walks a NULL-terminated alias chain, returns the
5 * first-set value, and fires a TTY-gated note when two names in the
6 * chain are set to DIFFERENT values.
7 * WHY: legacy env names (XrdSec*) remain accepted forever (C2 compat); this
8 * module is the single place that implements precedence so callers never
9 * hand-roll duplicate chains.
10 * HOW: Linear scan; remember first-set name+value; on each subsequent set
11 * entry compare values — on divergence call brix_cli_hint_once with the
12 * canonical key (chain[0]) and the two names. Values are NEVER printed
13 * (secrets protection). Return the winner (first-set).
14 */

16#include "cli/cli_hint.h"

22const char *
23brix_env_resolve(const char *const *chain, const char **which)
24{
25 /*
26 * WHAT: first-set-wins resolver with divergence detection.
27 * WHY: canonical name (chain[0]) takes precedence; any set legacy name is
28 * still accepted; a mismatch is flagged once per process per chain so
29 * users can correct their environment.
30 * HOW: two passes in one loop — winner_val/winner_name track the first set
31 * entry; differ_name tracks the first DIFFERING second entry so we can
32 * emit the note with both names.
33 */
34 const char *winner_val = NULL;
35 const char *winner_name = NULL;
36 const char *differ_name = NULL;
37 int i;
38
39 for (i = 0; chain[i] != NULL; i++) {
40 const char *val = getenv(chain[i]);
41
42 if (val == NULL) {
43 continue;
44 }
45
46 if (winner_val == NULL) {
47 /* First set entry — record as the winner. */
48 winner_val = val;
49 winner_name = chain[i];
50 continue;
51 }
52
53 /* A second set entry: check for divergence. */
54 if (differ_name == NULL && strcmp(winner_val, val) != 0) {
55 differ_name = chain[i];
56 }
57 }
58
59 /* Emit a TTY-gated note if two variables carry different values. */
60 if (differ_name != NULL) {
61 brix_cli_hint_once(chain[0],
62 "note: both %s and %s are set and differ; using %s (see brix-env(7))\n",
63 winner_name, differ_name, winner_name);
64 }
65
66 if (which != NULL) {
67 *which = winner_name;
68 }
69 return winner_val;
70}

83void
84brix_env_warn_stock_unsupported(void)
85{
86 /*
87 * WHAT: one-shot scan of environ for set-but-unsupported stock XRD_*
88 * variables; emits a single TTY-gated note listing up to six names.
89 * WHY: the stock client honors ~40 XRD_* keys; a drop-in user whose
90 * XRD_LOGLEVEL / XRD_CPRETRY / … silently does nothing gets exactly
91 * the surprise the parity audit (§7.10) flags. Making the ignore
92 * LOUD (on a TTY; scripts keep byte-identical output per the C3
93 * gate) turns a silent hazard into an actionable one.
94 * HOW: static once-guard, then a linear environ walk: names starting
95 * "XRD_" and not in brix_env_xrd_known are collected into one
96 * comma-separated note (names only, never values). Bounded copy per
97 * name; the note fires through brix_cli_hint_once under one key.
98 */
99 extern char **environ;
100 static int warned = 0;
101 char names[256];
102 size_t used = 0;
103 int listed = 0, extra = 0, i, j;
104
105 if (warned) {
106 return;
107 }
108 warned = 1;
109
110 names[0] = '\0';
111 for (i = 0; environ[i] != NULL; i++) {
112 const char *entry = environ[i];
113 const char *eq = strchr(entry, '=');
114 size_t nlen = (eq != NULL) ? (size_t) (eq - entry) : strlen(entry);
115 int known = 0;
116
117 if (nlen < 4 || strncmp(entry, "XRD_", 4) != 0) {
118 continue;
119 }
120 for (j = 0; brix_env_xrd_known[j] != NULL; j++) {
121 if (nlen == strlen(brix_env_xrd_known[j])
122 && strncmp(entry, brix_env_xrd_known[j], nlen) == 0)
123 {
124 known = 1;
125 break;
126 }
127 }
128 if (known) {
129 continue;
130 }
131 if (listed >= 6 || used + nlen + 3 >= sizeof(names)) {
132 extra++;
133 continue;
134 }
135 if (listed > 0) {
136 names[used++] = ',';
137 names[used++] = ' ';
138 }
139 memcpy(names + used, entry, nlen);
140 used += nlen;
141 names[used] = '\0';
142 listed++;
143 }
144
145 if (listed == 0) {
146 return;
147 }
148 if (extra > 0) {
149 brix_cli_hint_once("xrd-env",
150 "note: stock XRootD variable(s) %s (+%d more) are set but not "
151 "supported by brix-client (see brix-env(7))\n", names, extra);
152 } else {
153 brix_cli_hint_once("xrd-env",
154 "note: stock XRootD variable(s) %s are set but not supported by "
155 "brix-client (see brix-env(7))\n", names);
156 }
157}
