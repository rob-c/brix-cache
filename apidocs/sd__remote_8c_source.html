1/*
2 * sd_remote.c — remote-origin (s3://) storage driver. See the header.
3 *
4 * s3:// delegates entirely to the shared S3 driver (sd_s3): the SD object wraps
5 * an sd_s3_file*; pread/preadv are signed Range GETs; stat/fstat report the HEAD
6 * size. Writes are staged whole-object uploads (.staged_* → single PUT or MPU)
7 * plus .unlink (DELETE) — there is deliberately no .pwrite, so CAP_RANDOM_WRITE
8 * is NOT advertised and random in-place writes are rejected at the cap layer.
9 * .server_copy is a native S3 CopyObject (in-store, x-amz-copy-source).
10 *
11 * The namespace is a mutable object catalog (CAP_DIRS|CAP_DIRS_WRITE): a folder
12 * is a zero-byte "path/" marker object, so .mkdir PUTs one, .rename copies+deletes
13 * (files, and empty-directory markers), .stat recognises a marker as a directory,
14 * and .opendir pages ListObjectsV2. There is no atomic prefix move: renaming a
15 * NON-empty directory reports ENOTSUP rather than a partial in-store shuffle.
16 */

32/* Compose the sd_s3 object path "/bucket/key" from the instance bucket and the
33 * export-relative key (which already carries a leading '/'). Shared with the
34 * meta/write siblings via sd_remote_internal.h. */
35void
36sd_remote_s3_key(const brix_sd_remote_cfg_t *cfg, const char *key,
37 char *dst, size_t dstcap)
38{
39 snprintf(dst, dstcap, "/%s%s", cfg->bucket, (key != NULL) ? key : "/");
40}

42/* Compose the "/bucket/key/" DIRECTORY-MARKER object path: the same key as
43 * sd_remote_s3_key but with exactly one trailing '/'. S3 has no directories, so
44 * a folder is represented by a zero-byte object whose key ends in '/' (mkdir
45 * writes it, stat/opendir recognise it, rmdir deletes it). Shared with the meta
46 * sibling via sd_remote_internal.h. */
47void
48sd_remote_s3_dirkey(const brix_sd_remote_cfg_t *cfg, const char *key,
49 char *dst, size_t dstcap)
50{
51 size_t n;
52
53 sd_remote_s3_key(cfg, key, dst, dstcap);
54 n = strlen(dst);
55 if (n > 0 && dst[n - 1] != '/' && n + 1 < dstcap) {
56 dst[n] = '/';
57 dst[n + 1] = '\0';
58 }
59}

81/* ---- sd_remote_cred_gate — classify a per-user credential (P80.3) ---------
82 *
83 * WHAT: Decides how a cred-scoped slot must treat `cred`: 1 = a usable S3
84 * keypair (sign with the override), 0 = fall back to the instance's
85 * static service credential, -1 = refuse with EACCES.
86 *
87 * WHY: Every cred slot on this driver (open/staged_open/stat/unlink) shares
88 * the same three-way decision, including the fallback_deny rule: a cred
89 * of a kind this S3-only backend cannot use (x509/bearer) under deny
90 * mode must be refused rather than silently signed with the export's
91 * shared service credential — exactly the fallback the operator forbade.
92 *
93 * HOW: Usable means both s3_ak and s3_sk are present and non-empty; anything
94 * else is refused under fallback_deny and falls back otherwise.
95 *
96 * Shared with the meta/write siblings via sd_remote_internal.h. */
97int
98sd_remote_cred_gate(const brix_sd_cred_t *cred)
99{
100 if (cred != NULL
101 && cred->s3_ak != NULL && cred->s3_ak[0] != '\0'
102 && cred->s3_sk != NULL && cred->s3_sk[0] != '\0')
103 {
104 return 1;
105 }
106 if (cred != NULL && cred->fallback_deny) {
107 return -1;
108 }
109 return 0;
110}

112/* ---- sd_remote_open_impl (shared by the plain and cred-scoped open slots) --
113 *
114 * WHAT: Opens `path` for read against the S3 origin, using `ak`/`sk`/`region`
115 * as the SigV4 credential instead of the instance's static
116 * cfg->access_key/secret_key/region when they are non-NULL.
117 *
118 * WHY: Phase-3 T3 per-user S3 credentials: a request whose identity resolves
119 * to a `<key>.s3` file must sign against the ORIGIN with that user's
120 * keys, not the export's shared service credential. Sharing one open
121 * body between sd_remote_open and sd_remote_open_cred keeps the object-
122 * construction logic (size HEAD, obj/state alloc, snap fill) written
123 * exactly once.
124 *
125 * HOW: Builds the object path, then an sd_s3_open_params using the override
126 * ak/sk/region when given (falling back to the instance's static
127 * triple field-by-field when a pointer is NULL — region alone can be
128 * overridden without ak/sk, though callers always pass all three or
129 * none), and proceeds identically to the prior sd_remote_open body. */
130static brix_sd_obj_t *
131sd_remote_open_impl(brix_sd_instance_t *inst, const char *path, int sd_flags,
132 const char *ak, const char *sk, const char *region, const char *session,
133 int *err_out)
134{
135 const brix_sd_remote_cfg_t *cfg = inst->state;
136 sd_s3_open_params p;
137 char objpath[768];
138 char errbuf[256];
139 sd_s3_file *s3;
140 sd_remote_obj_state *st;
141 brix_sd_obj_t *obj;
142 int64_t size = 0;
143
144 /* Read-only origin: refuse any write/create/trunc intent up front. */
145 if (sd_flags & (BRIX_SD_O_WRITE | BRIX_SD_O_CREATE | BRIX_SD_O_TRUNC
146 | BRIX_SD_O_APPEND)) {
147 if (err_out) { *err_out = EROFS; }
148 return NULL;
149 }
150
151 sd_remote_s3_key(cfg, path, objpath, sizeof(objpath));
152 sd_remote_s3_params(cfg, objpath, &p);
153 if (ak != NULL) { p.ak = ak; }
154 if (sk != NULL) { p.sk = sk; }
155 if (region != NULL) { p.region = region; }
156 if (session != NULL) { p.session_token = session; }
157
158 s3 = sd_s3_open_read(&p, errbuf, sizeof(errbuf));
159 if (s3 == NULL) {
160 if (err_out) { *err_out = ENOMEM; } /* open_read only fails on bad args/OOM */
161 return NULL;
162 }
163 if (sd_s3_size(s3, &size, errbuf, sizeof(errbuf)) != 0) {
164 int e = errno; /* HEAD set errno (ENOENT on 404) */
165
166 sd_s3_close(s3);
167 if (err_out) { *err_out = e ? e : EIO; }
168 return NULL;
169 }
170
171 st = calloc(1, sizeof(*st));
172 obj = calloc(1, sizeof(*obj));
173 if (st == NULL || obj == NULL) {
174 free(st);
175 free(obj);
176 sd_s3_close(s3);
177 if (err_out) { *err_out = ENOMEM; }
178 return NULL;
179 }
180 st->s3 = s3;
181
182 obj->driver = inst->driver;
183 obj->inst = inst;
184 obj->fd = NGX_INVALID_FILE; /* no kernel fd — memory-served */
185 obj->state = st;
186 obj->heap_shell = 1; /* malloc'd shell; caller may free */
187 obj->snap.size = (off_t) size;
188 obj->snap.mode = S_IFREG | 0444;
189 obj->snap.is_reg = 1;
190
191 return obj;
192}

203/* ---- sd_remote_open_cred — per-user SigV4 credential override (phase-3 T3) -
204 *
205 * WHAT: Like sd_remote_open, but when cred->s3_ak (and s3_sk) are set, signs
206 * the request with the caller's per-user access key / secret key /
207 * region instead of the instance's static service credential.
208 *
209 * WHY: brix_sd_open_maybe_cred (sd.h) routes through this slot whenever the
210 * VFS credential gate (vfs_cred.c) resolved a `<key>.s3` file for the
211 * requesting identity — the origin must see THAT user's SigV4
212 * signature, not the export's shared key.
213 *
214 * HOW: Three-way sd_remote_cred_gate: a usable S3 keypair signs with the
215 * override; an unusable kind (x509/bearer) falls back to the static
216 * instance credential, unless the operator set fallback_deny — then it
217 * is refused with EACCES before any origin contact. */
218static brix_sd_obj_t *
219sd_remote_open_cred(brix_sd_instance_t *inst, const char *path, int sd_flags,
220 mode_t mode, const brix_sd_cred_t *cred, int *err_out)
221{
222 int gate = sd_remote_cred_gate(cred);
223
224 (void) mode;
225
226 if (gate > 0) {
227 return sd_remote_open_impl(inst, path, sd_flags,
228 cred->s3_ak, cred->s3_sk, cred->s3_region, cred->s3_session,
229 err_out);
230 }
231 if (gate < 0) {
232 if (err_out) { *err_out = EACCES; }
233 errno = EACCES;
234 return NULL;
235 }
236 return sd_remote_open_impl(inst, path, sd_flags, NULL, NULL, NULL, NULL,
237 err_out);
238}

270/*
271 * sd_remote_preadv — coalesced vectored read over the ranged-GET origin.
272 *
273 * WHAT: Reads the contiguous file span [off, off + sum(iov_len)) with as few
274 * ranged GETs as the transport allows and scatters the bytes into the
275 * iovecs. Returns total bytes read (short = EOF), or -1 with errno set.
276 *
277 * WHY: The vectored read paths (pgread batches, kXR_readv runs) describe one
278 * contiguous span split across many small page-sized iovecs. The generic
279 * per-iovec pread fallback would issue one signed HTTP round trip per
280 * 4 KiB page — thousands of requests for a large read, which times the
281 * client out. One GET per SD_S3_PREAD_MAX-capped chunk restores sane
282 * request counts.
283 *
284 * HOW: Single-iovec calls loop sd_remote_pread straight into the caller's
285 * buffer (no copy). Scattered calls fill a malloc'd bounce buffer with
286 * the same loop, then memcpy per iovec — the HTTP body is copied anyway,
287 * so the bounce adds one pass over bytes that already left the socket.
288 * A 0-byte pread means EOF; a short chunk keeps looping (sd_s3_pread
289 * caps each request, a cap hit is not EOF).
290 */
291static ssize_t
292sd_remote_pread_full(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
293{
294 size_t done = 0;
295
296 while (done < len) {
297 ssize_t n = sd_remote_pread(obj, (char *) buf + done, len - done,
298 off + (off_t) done);
299 if (n < 0) {
300 return (done > 0) ? (ssize_t) done : -1;
301 }
302 if (n == 0) {
303 break; /* EOF */
304 }
305 done += (size_t) n;
306 }
307 return (ssize_t) done;
308}

310static ssize_t
311sd_remote_preadv(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
312 off_t off)
313{
314 size_t total = 0;
315 size_t scattered;
316 ssize_t n;
317 char *bounce;
318 int i;
319
320 for (i = 0; i < iovcnt; i++) {
321 total += iov[i].iov_len;
322 }
323 if (total == 0) {
324 return 0;
325 }
326
327 if (iovcnt == 1) {
328 return sd_remote_pread_full(obj, iov[0].iov_base, iov[0].iov_len, off);
329 }
330
331 bounce = malloc(total);
332 if (bounce == NULL) {
333 errno = ENOMEM;
334 return -1;
335 }
336
337 n = sd_remote_pread_full(obj, bounce, total, off);
338 if (n < 0) {
339 free(bounce);
340 return -1;
341 }
342
343 scattered = 0;
344 for (i = 0; i < iovcnt && scattered < (size_t) n; i++) {
345 size_t take = iov[i].iov_len;
346 if (take > (size_t) n - scattered) {
347 take = (size_t) n - scattered;
348 }
349 memcpy(iov[i].iov_base, bounce + scattered, take);
350 scattered += take;
351 }
352
353 free(bounce);
354 return n;
355}
