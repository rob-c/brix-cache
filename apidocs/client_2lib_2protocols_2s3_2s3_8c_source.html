1/*
2 * s3.c — AWS Signature Version 4 (path-style) for the native client.
3 *
4 * WHAT: brix_s3_sign_v4() builds the x-amz-date / x-amz-content-sha256 /
5 * Authorization header block for any method+path+body-hash; the sha256
6 * helpers compute the payload hashes those headers carry.
7 * WHY: Both xrdcp's s3:// transfers and xrddiag's s3 probe need the identical
8 * signer — keeping one copy in the lib (rather than a private copy in each
9 * app) is the clean-room equivalent of XrdClS3's shared signing code.
10 * HOW: Canonical-request → string-to-sign → HMAC-SHA256 signing-key chain, using
11 * libxrdproto's SHA-256/HMAC kernels (compat/crypto.h). The CanonicalURI is
12 * percent-encoded (RFC-3986 unreserved + '/') to match the server's own
13 * canonicalization. Signed headers are host;x-amz-content-sha256;x-amz-date.
14 *
15 * Clean-room: implements the published SigV4 algorithm (AWS docs), not any SDK.
16 */
17#include "brix.h"
18#include "core/compat/crypto.h" /* brix_sha256 / brix_hmac_sha256 */
19#include "core/compat/hex.h" /* brix_hex_encode */
20#include "core/compat/uri.h" /* brix_http_urlencode (shared SigV4 canonical URI) */
21#include "core/compat/sigv4.h" /* brix_sigv4_signing_key (shared 4-round HMAC chain) */

28brix_s3_sha256_hex(const void *data, size_t len, char *out)
29{
30 uint8_t d[32];
31 brix_sha256((const uint8_t *) data, len, d);
32 brix_hex_encode(d, 32, out); /* lowercase 64 hex + NUL */

35/* ---- Build the SigV4 canonical request and return its SHA-256 hex ----
36 *
37 * WHAT: Percent-encodes the URI, assembles the canonical request
38 * (method, URI, canonical query, the three signed headers, the trailing
39 * SignedHeaders list, and the payload hash), and writes its lowercase
40 * SHA-256 hex (65 bytes incl. NUL) into canon_hex. Returns 0 on success,
41 * -1 if the URI cannot be encoded or the canonical request would overflow.
42 *
43 * WHY: The canonical-request hash is the first HMAC input; if this string
44 * differs by a single byte from what the server re-derives, the signature
45 * mismatches and the request 403s. Isolating its construction keeps the
46 * signer's stage boundaries explicit and each stage independently testable.
47 *
48 * HOW: 1. URL-encode uri with the shared RFC-3986 encoder (unreserved + '/'),
49 * byte-identical to the server's verify path.
50 * 2. snprintf the canonical request; treat truncation as a hard failure so
51 * a short buffer never silently signs a different string.
52 * 3. Hash the canonical request into canon_hex.
53 */
54static int
55brix_s3_sigv4_build_canon_hash(const char *method, const char *host,
56 const char *uri, const char *canon_qs,
57 const char *payload_hex, const char *amzdate,
58 char *canon_hex)
59{
60 char enc_uri[2048];
61 char canon[8192];
62 int cn;
63
64 /* Shared SigV4 CanonicalURI encoder (RFC-3986 unreserved + '/'), byte-identical
65 * to the server's verify path so client-signs == server-verifies. */
66 if (brix_http_urlencode((const unsigned char *) uri, strlen(uri),
67 enc_uri, sizeof(enc_uri), "/") < 0) {
68 return -1;
69 }
70 /* canonical request: method\nURI\nCANONICAL_QUERY\nheaders\nsignedHeaders\nhash.
71 * canon_qs must already be sorted + RFC-3986 encoded (the server re-derives the
72 * same string from the request query). */
73 cn = snprintf(canon, sizeof(canon),
74 "%s\n%s\n%s\nhost:%s\nx-amz-content-sha256:%s\nx-amz-date:%s\n\n"
75 "host;x-amz-content-sha256;x-amz-date\n%s",
76 method, enc_uri, canon_qs, host, payload_hex, amzdate, payload_hex);
77 /* A truncated canonical request would sign a different string than the server
78 * verifies → a confusing 403. Fail cleanly so the caller reports "can't sign". */
79 if (cn < 0 || (size_t) cn >= sizeof(canon)) {
80 return -1;
81 }
82 brix_s3_sha256_hex(canon, strlen(canon), canon_hex);
83 return 0;
84}

86/* ---- Build the credential scope and the string-to-sign ----
87 *
88 * WHAT: Formats the credential scope (datestamp/region/s3/aws4_request) into
89 * scope and the AWS4-HMAC-SHA256 string-to-sign (algorithm, amzdate, scope,
90 * canonical-request hash) into sts. Returns 0 on success, -1 if either
91 * snprintf would overflow its caller-supplied buffer.
92 *
93 * WHY: region is user/env-controlled (--s3-region / $AWS_DEFAULT_REGION); a
94 * truncated scope or string-to-sign would sign a different string than the
95 * Authorization header advertises → a confusing 403. Guarding both here
96 * keeps that contract in one place, matching the canonical-request guard.
97 *
98 * HOW: 1. snprintf the credential scope; fail on truncation.
99 * 2. snprintf the string-to-sign from amzdate, scope, and canon_hex; fail
100 * on truncation.
101 */
102static int
103brix_s3_sigv4_build_sts(const char *amzdate, const char *datestamp,
104 const char *region, const char *canon_hex,
105 char *scope, size_t scopesz, char *sts, size_t stssz)
106{
107 int cn;
108
109 cn = snprintf(scope, scopesz, "%s/%s/s3/aws4_request", datestamp, region);
110 if (cn < 0 || (size_t) cn >= scopesz) {
111 return -1;
112 }
113 cn = snprintf(sts, stssz, "AWS4-HMAC-SHA256\n%s\n%s\n%s",
114 amzdate, scope, canon_hex);
115 if (cn < 0 || (size_t) cn >= stssz) {
116 return -1;
117 }
118 return 0;
119}

121/* ---- Derive the signing key and sign the string-to-sign ----
122 *
123 * WHAT: Runs the 4-round HMAC signing-key derivation, HMAC-signs the
124 * string-to-sign, and writes its lowercase hexadecimal signature.
125 *
126 * WHY: Header and query authentication must share the exact same key chain.
127 *
128 * HOW: Derive with the shared four-round helper, HMAC, then hex-encode.
129 */
130static int
131brix_s3_sigv4_signature(const char *sk, const char *datestamp,
132 const char *region, const char *sts,
133 char sighex[65])
134{
135 uint8_t k4[32], sig[32];
136
137 if (!brix_sigv4_signing_key((const uint8_t *) sk, strlen(sk),
138 datestamp, region, "s3", k4)
139 || !brix_hmac_sha256(k4, 32, (const uint8_t *) sts, strlen(sts), sig))
140 return -1;
141 brix_hex_encode(sig, 32, sighex);
142 return 0;
143}

145/* Emit the ordinary SigV4 headers after signing the string-to-sign. */
146static int
147brix_s3_sigv4_sign_and_emit(const char *sk, const char *datestamp,
148 const char *region, const char *sts,
149 const char *ak, const char *payload_hex,
150 const char *amzdate, const char *scope,
151 char *hdrs, size_t hdrsz)
152{
153 char sighex[65];
154 int cn;
155
156 if (brix_s3_sigv4_signature(sk, datestamp, region, sts, sighex) != 0)
157 return -1;
158
159 cn = snprintf(hdrs, hdrsz,
160 "x-amz-date: %s\r\nx-amz-content-sha256: %s\r\n"
161 "Authorization: AWS4-HMAC-SHA256 Credential=%s/%s, "
162 "SignedHeaders=host;x-amz-content-sha256;x-amz-date, Signature=%s\r\n",
163 amzdate, payload_hex, ak, scope, sighex);
164 if (cn < 0 || (size_t) cn >= hdrsz) {
165 return -1;
166 }
167 return 0;
168}

170/* ---- Hash a host-only presigned canonical request ----
171 *
172 * WHAT: Writes the SHA-256 hex of a SigV4 presigned canonical request.
173 * WHY: Presigned requests sign only `host` and carry auth material in the query.
174 * HOW: Encode the URI, format the host-only canonical shape, then hash it.
175 */
176static int
177brix_s3_presign_canon_hash(const char *method, const char *host,
178 const char *uri, const char *canon_qs,
179 char canon_hex[65])
180{
181 char encoded_uri[2048];
182 char canonical[8192];
183 int written;
184
185 if (brix_http_urlencode((const unsigned char *) uri, strlen(uri),
186 encoded_uri, sizeof(encoded_uri), "/") < 0) return -1;
187 written = snprintf(canonical, sizeof(canonical),
188 "%s\n%s\n%s\nhost:%s\n\nhost\nUNSIGNED-PAYLOAD", method,
189 encoded_uri, canon_qs, host);
190 if (written < 0 || (size_t) written >= sizeof(canonical)) return -1;
191 brix_s3_sha256_hex(canonical, strlen(canonical), canon_hex);
192 return 0;
193}

195/* ---- Build a complete deterministic SigV4 presigned query ----
196 *
197 * WHAT: Returns zero with a canonical query including X-Amz-Signature.
198 * WHY: Conformance probes need expired and path-replay URLs without an SDK.
199 * HOW: Derive date/scope, encode Credential, hash host-only canonical request,
200 * sign its string-to-sign, and append the signature to the same query.
201 */
202static int
203brix_s3_presign_dates(time_t signed_at, unsigned expires,
204 char amzdate[20], char datestamp[12])
205{
206 struct tm tmv;
207
208 if (expires == 0 || expires > 604800) return -1;
209 if (gmtime_r(&signed_at, &tmv) == NULL) return -1;
210 if (strftime(amzdate, 20, "%Y%m%dT%H%M%SZ", &tmv) == 0) return -1;
211 return strftime(datestamp, 12, "%Y%m%d", &tmv) == 0 ? -1 : 0;
212}

223static int
224brix_s3_presign_credential(const char *ak, const char *region,
225 const char *datestamp, char scope[160], char encoded[1536])
226{
227 char credential[512];
228 int written;
229
230 written = snprintf(scope, 160, "%s/%s/s3/aws4_request",
231 datestamp, region);
232 if (written < 0 || written >= 160) return -1;
233 written = snprintf(credential, sizeof(credential), "%s/%s", ak, scope);
234 if (written < 0 || (size_t) written >= sizeof(credential)) return -1;
235 return brix_http_urlencode((const unsigned char *) credential,
236 strlen(credential), encoded, 1536, "") < 0 ? -1 : 0;
237}

251int
252brix_s3_presign_v4_at(const char *method, const char *host, const char *uri,
253 const char *ak, const char *sk, const char *region, time_t signed_at,
254 unsigned expires, char *query, size_t query_size)
255{
256 char amzdate[20], datestamp[12];
257 char scope[160], encoded_credential[1536];
258 char unsigned_query[3072], canon_hex[65], sts[640], sighex[65];
259 int written;
260
261 if (!brix_s3_presign_args_valid(method, host, uri, ak, sk, region, query))
262 return -1;
263 if (brix_s3_presign_dates(signed_at, expires, amzdate, datestamp) != 0)
264 return -1;
265 if (brix_s3_presign_credential(ak, region, datestamp, scope,
266 encoded_credential) != 0) return -1;
267 if (brix_s3_presign_unsigned_query(encoded_credential, amzdate, expires,
268 unsigned_query) != 0) return -1;
269 if (brix_s3_presign_canon_hash(method, host, uri, unsigned_query,
270 canon_hex) != 0) return -1;
271 if (brix_s3_sigv4_build_sts(amzdate, datestamp, region, canon_hex,
272 scope, sizeof(scope), sts, sizeof(sts)) != 0) return -1;
273 if (brix_s3_sigv4_signature(sk, datestamp, region, sts, sighex) != 0)
274 return -1;
275 written = snprintf(query, query_size, "%s&X-Amz-Signature=%s",
276 unsigned_query, sighex);
277 return written < 0 || (size_t) written >= query_size ? -1 : 0;
278}

280int
281brix_s3_sign_v4_q(const char *method, const char *host, const char *uri,
282 const char *canon_qs, const char *ak, const char *sk,
283 const char *region, const char *payload_hex,
284 char *hdrs, size_t hdrsz)
285{
286 char amzdate[20], datestamp[12];
287 char canon_hex[65];
288 char scope[160], sts[640];
289 time_t now = time(NULL);
290 struct tm tmv;
291
292 if (canon_qs == NULL) { canon_qs = ""; }
293 if (sk == NULL || ak == NULL || region == NULL || payload_hex == NULL) {
294 return -1;
295 }
296 if (gmtime_r(&now, &tmv) == NULL) {
297 return -1;
298 }
299 strftime(amzdate, sizeof(amzdate), "%Y%m%dT%H%M%SZ", &tmv);
300 strftime(datestamp, sizeof(datestamp), "%Y%m%d", &tmv);
301
302 if (brix_s3_sigv4_build_canon_hash(method, host, uri, canon_qs,
303 payload_hex, amzdate, canon_hex) != 0) {
304 return -1;
305 }
306 if (brix_s3_sigv4_build_sts(amzdate, datestamp, region, canon_hex,
307 scope, sizeof(scope), sts, sizeof(sts)) != 0) {
308 return -1;
309 }
310 return brix_s3_sigv4_sign_and_emit(sk, datestamp, region, sts, ak,
311 payload_hex, amzdate, scope, hdrs, hdrsz);
312}
