1/*
2 * reverse.c — the reverse (PTR) resolve driver (phase-116 W7).
3 *
4 * WHAT: brix_dns_reverse(): address-keyed cache probe, then either the
5 * policy's nginx resolver (ngx_resolve_addr, async) or a thread-pool
6 * getnameinfo(NI_NAMEREQD) — the ONLY such call in src/ (guard
7 * tools/ci/check_dns_seam.py). brix_dns_reverse_cached() is the
8 * never-blocking probe the event loop uses at decision time,
9 * brix_dns_reverse_prefetch() warms the cache from the accept/login
10 * hooks, and brix_dns_reverse_sync() is the blocking flavour for
11 * thread-pool code (cache -> loop resolver via the bridge -> libc).
12 * WHY: XrdAcc `h` rules, protbind hostname templates, `host` auth and the
13 * TPC origin id all need the peer's FQDN. Resolving it inline on the
14 * event loop stalled the whole worker on a slow nameserver (I-DNS-1);
15 * a per-connection thread hop would still serialize every decision
16 * behind a round trip. Warming an address-keyed cache from the accept
17 * path makes the common case a lookup, and consumers degrade to the
18 * peer IP — never to a stall — when the answer is not ready.
19 * HOW: Mirrors resolve.c/resolve_thread.c: the nginx ctx is kept in
20 * req->rctx so a cancel can ngx_resolve_addr_done() it; ngx_resolve_addr
21 * frees the ctx itself on NGX_ERROR and may run the handler inline for
22 * a cached node (ctx->name is then a resolver-owned copy released after
23 * the handler returns, so it is copied out by length). The thread task
24 * owns its allocation and the body reads only its copies. Every
25 * failure is negative-cached for the policy's negative TTL so a dead
26 * PTR zone is asked once per TTL, not once per connection; positive
27 * answers keep the record TTL clamped to the policy window.
28 */

53static time_t
54dns_rev_clamp_ttl(const brix_dns_policy_t *policy, time_t ttl)
55{
56 if (policy != NULL) {
57 if (ttl < policy->min_ttl) {
58 ttl = policy->min_ttl;
59 }
60 if (ttl > policy->max_ttl) {
61 ttl = policy->max_ttl;
62 }
63 return ttl;
64 }
65 return ttl < BRIX_DNS_MIN_TTL_DEFAULT ? BRIX_DNS_MIN_TTL_DEFAULT : ttl;

128static void
129dns_rev_ngx_handler(ngx_resolver_ctx_t *ctx)
130{
131 brix_dns_rev_req_t *req = ctx->data;
132 size_t n;
133
134 req->rctx = NULL;
135 if (ctx->state != NGX_OK || ctx->name.len == 0) {
136 ngx_int_t rc = ctx->state == NGX_RESOLVE_NXDOMAIN ? NGX_DECLINED
137 : NGX_ERROR;
138 const char *error = ngx_resolver_strerror(ctx->state);
139
140 ngx_resolve_addr_done(ctx);
141 dns_rev_fail(req, rc, error);
142 return;
143 }
144 n = ngx_min(ctx->name.len, sizeof(req->name) - 1);
145 ngx_memcpy(req->name, ctx->name.data, n);
146 req->name[n] = '\0';
147 req->ttl = dns_rev_clamp_ttl(req->policy, ctx->valid - ngx_time());
148 req->rc = NGX_OK;
149 req->error = NULL;
150 ngx_resolve_addr_done(ctx);
151 brix_dns_rev_req_finish(req);
152}

155static ngx_int_t
156dns_rev_ngx_start(brix_dns_rev_req_t *req, ngx_resolver_t *r)
157{
158 ngx_resolver_ctx_t *ctx;
159 ngx_msec_t timeout;
160
161 ctx = ngx_resolve_start(r, NULL);
162 if (ctx == NULL || ctx == NGX_NO_RESOLVER) {
163 dns_rev_fail(req, NGX_ERROR, "resolver unavailable");
164 return NGX_OK;
165 }
166 timeout = (ngx_msec_t) (req->policy ? req->policy->rc.timeout : 5) * BRIX_CMS_SEC_TO_MS_MULTIPLIER;
167 if (timeout < BRIX_DNS_TIMEOUT_FLOOR_MS) {
168 timeout = BRIX_DNS_TIMEOUT_FLOOR_MS;
169 }
170 ctx->addr.sockaddr = (struct sockaddr *) &req->ss;
171 ctx->addr.socklen = req->len;
172 ctx->handler = dns_rev_ngx_handler;
173 ctx->data = req;
174 ctx->timeout = timeout;
175 req->rctx = ctx;
176 if (ngx_resolve_addr(ctx) != NGX_OK) {
177 /* nginx freed the ctx and did not run the handler */
178 req->rctx = NULL;
179 dns_rev_fail(req, NGX_ERROR, "resolver could not be started");
180 }
181 return NGX_OK;
182}

185/* ---- public: async ------------------------------------------------------- */
186
187ngx_int_t
188brix_dns_reverse(brix_dns_rev_req_t *req)
189{
190 ngx_resolver_t *r;
191 ngx_int_t rc;
192
193 req->rc = NGX_ERROR;
194 req->error = NULL;
195 req->name[0] = '\0';
196 req->ttl = 0;
197 req->cached = 0;
198 req->rctx = NULL;
199 req->task = NULL;
200 if (req->log == NULL) {
201 req->log = ngx_cycle->log;
202 }
203 if (!dns_rev_addr_supported(&req->ss, req->len)) {
204 req->rc = NGX_DECLINED;
205 req->error = "peer is not an IP address";
206 brix_dns_rev_req_finish(req);
207 return NGX_OK;
208 }
209 rc = brix_dns_rcache_lookup((struct sockaddr *) &req->ss, req->len,
210 req->name, sizeof(req->name));
211 if (rc != NGX_AGAIN) {
212 req->cached = 1;
213 req->rc = rc;
214 req->error = rc == NGX_OK ? NULL : "no PTR record (negative cache)";
215 brix_dns_rev_req_finish(req);
216 return NGX_OK;
217 }
218 r = brix_dns_policy_resolver(req->policy);
219 if (r != NULL) {
220 return dns_rev_ngx_start(req, r);
221 }
222 rc = brix_dns_reverse_via_thread(req);
223 if (rc == NGX_OK) {
224 return NGX_OK;
225 }
226 if (rc == NGX_DECLINED) {
227 dns_rev_fail(req, NGX_ERROR, "no resolver and no thread pool available");
228 return NGX_OK;
229 }
230 return NGX_ERROR;
231}

267void
268brix_dns_reverse_prefetch(const brix_dns_policy_t *policy,
269 const struct sockaddr *sa, socklen_t len)
270{
271 brix_dns_rev_req_t *req;
272 time_t hold = BRIX_DNS_PENDING_HOLD;
273
274 if (sa == NULL || len == 0 || len > sizeof(req->ss)) {
275 return;
276 }
277 if (policy != NULL) {
278 time_t budget = (time_t) policy->rc.timeout * policy->rc.attempts + 1;
279
280 hold = ngx_max(hold, budget);
281 }
282 if (!brix_dns_rcache_mark_pending(sa, len, hold)) {
283 return; /* cached, negative or in flight */
284 }
285 req = ngx_calloc(sizeof(brix_dns_rev_req_t), ngx_cycle->log);
286 if (req == NULL) {
287 return;
288 }
289 ngx_memcpy(&req->ss, sa, len);
290 req->len = len;
291 req->policy = policy;
292 req->log = ngx_cycle->log; /* the fill may outlive its caller */
293 req->handler = dns_rev_prefetch_done;
294 if (brix_dns_reverse(req) != NGX_OK) {
295 ngx_free(req); /* handler did not run */
296 }
297}

300/* ---- public: blocking (thread-pool code) --------------------------------- */
301
302ngx_int_t
303brix_dns_reverse_sync(const brix_dns_policy_t *policy,
304 const struct sockaddr *sa, socklen_t len, char *buf, size_t buflen)
305{
306 ngx_int_t rc, answer = NGX_ERROR;
307 time_t ttl;
308
309 if (buf == NULL || buflen == 0) {
310 return NGX_ERROR;
311 }
312 buf[0] = '\0';
313 if (sa == NULL || !dns_rev_addr_supported((const struct sockaddr_storage *)
314 sa, len))
315 {
316 return NGX_DECLINED;
317 }
318 rc = brix_dns_rcache_lookup(sa, len, buf, buflen);
319 if (rc != NGX_AGAIN) {
320 return rc;
321 }
322 if (brix_dns_bridge_reverse(policy, sa, len, buf, buflen, &answer)
323 == NGX_OK)
324 {
325 return answer; /* the loop driver cached it */
326 }
327 rc = dns_libc_reverse(sa, len, buf, buflen);
328 ttl = rc == NGX_OK ? dns_rev_clamp_ttl(policy, BRIX_DNS_THREAD_TTL)
329 : dns_rev_negative_ttl(policy);
330 brix_dns_rcache_store(sa, len, rc == NGX_OK ? buf : NULL, ttl);
331 return rc;
332}

350static void
351dns_rev_thread_done(ngx_event_t *ev)
352{
353 ngx_thread_task_t *task = ev->data;
354 dns_rev_thread_ctx_t *tc = task->ctx;
355 brix_dns_rev_req_t *req = tc->req;
356 ngx_pool_t *pool = tc->pool;
357
358 if (req == NULL) {
359 ngx_destroy_pool(pool); /* cancelled while in flight */
360 return;
361 }
362 req->task = NULL;
363 if (tc->rc == NGX_OK) {
364 ngx_cpystrn((u_char *) req->name, (u_char *) tc->name,
365 sizeof(req->name));
366 req->rc = NGX_OK;
367 req->error = NULL;
368 req->ttl = dns_rev_clamp_ttl(req->policy, BRIX_DNS_THREAD_TTL);
369 brix_dns_rev_req_finish(req);
370 } else {
371 dns_rev_fail(req, tc->rc, tc->rc == NGX_DECLINED
372 ? "no PTR record" : "getnameinfo failed");
373 }
374 ngx_destroy_pool(pool);
375}

381brix_dns_reverse_via_thread(brix_dns_rev_req_t *req)
382{
383#if (NGX_THREADS)
384 ngx_thread_task_t *task;

392 if (rc != NGX_OK) {
393 return rc;
394 }
395 tc = task->ctx;
396 tc->req = req;
397 tc->pool = pool;
398 ngx_memcpy(&tc->ss, &req->ss, req->len);
399 tc->len = req->len;
400 brix_task_bind(task, dns_rev_thread_run, dns_rev_thread_done);
401 task->event.log = req->log;
402 return brix_dns_thread_task_post(tp, task, pool, &req->task);
403#else
404 (void) req;
405 return NGX_DECLINED;
406#endif
407}
