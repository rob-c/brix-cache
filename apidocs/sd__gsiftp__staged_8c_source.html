23static ssize_t
24sd_gsiftp_source_read(void *ctx, uint8_t *data, size_t cap)
25{
26 sd_gsiftp_source *source = ctx;
27 ssize_t n;
28
29 do {
30 n = pread(source->fd, data, cap, source->offset);
31 } while (n < 0 && errno == EINTR);
32 if (n > 0) {
33 source->offset += n;
34 }
35 return n;

146ssize_t
147sd_gsiftp_staged_write(brix_sd_staged_t *handle, const void *buf, size_t len,
148 off_t off)
149{
150 sd_gsiftp_staged_state *state = handle->state;
151 size_t done = 0;
152 int fd = fileno(state->scratch);
153
154 if (off < 0) {
155 errno = EINVAL;
156 return -1;
157 }
158 while (done < len) {
159 ssize_t n = pwrite(fd, (const uint8_t *) buf + done, len - done,
160 off + (off_t) done);
161
162 if (n > 0) {
163 done += (size_t) n;
164 continue;
165 }
166 if (n < 0 && errno == EINTR) {
167 continue;
168 }
169 if (n == 0) {
170 errno = EIO;
171 }
172 return -1;
173 }
174 return (ssize_t) done;
175}

200static int
201sd_gsiftp_precondition(brix_sd_staged_t *handle, brix_sd_precond_t *pre)
202{
203 sd_gsiftp_staged_state *state = handle->state;
204 sd_gsiftp_state *inst_state = handle->inst->state;
205 brix_sd_cred_t cred;
206 brix_sd_stat_t ignored;
207 const char *logical;
208
209 if (pre == NULL || pre->kind == BRIX_SD_PRECOND_NONE) {
210 return 0;
211 }
212 if (pre->kind != BRIX_SD_PRECOND_ABSENT) {
213 errno = ENOTSUP;
214 return -1;
215 }
216 logical = state->final_path;
217 if (inst_state->base_path[0] != '\0'
218 && strcmp(inst_state->base_path, "/") != 0) {
219 logical += strlen(inst_state->base_path);
220 }
221 memset(&cred, 0, sizeof(cred));
222 cred.x509_proxy = state->proxy[0] != '\0' ? state->proxy : NULL;
223 if (sd_gsiftp_stat_impl(handle->inst, logical, &ignored, &cred) == NGX_OK) {
224 errno = EEXIST;
225 return -1;
226 }
227 return errno == ENOENT ? 0 : -1;
228}

236ngx_int_t
237sd_gsiftp_staged_commit(brix_sd_staged_t *handle, brix_sd_precond_t *pre)
238{
239 sd_gsiftp_staged_state *state = handle->state;
240 sd_gsiftp_state *inst_state = handle->inst->state;
241 sd_gsiftp_source source;
242 gftp_session_t session;
243 char temp_path[GSIFTP_PATH_CAP];
244 int rc;
245
246 if (sd_gsiftp_precondition(handle, pre) != 0
247 || sd_gsiftp_temp_path(state->final_path, temp_path) != 0
248 || fflush(state->scratch) != 0
249 || sd_gsiftp_session(&session, inst_state,
250 state->proxy[0] != '\0' ? state->proxy : NULL) != 0) {
251 return NGX_ERROR;
252 }
253 source.fd = fileno(state->scratch);
254 source.offset = 0;
255 rc = gftp_store(&session, temp_path, sd_gsiftp_source_read, &source);
256 if (rc == 0) {
257 rc = gftp_expect(&session, 300, 399, "RNFR %s", temp_path);
258 }
259 if (rc == 0) {
260 rc = gftp_expect(&session, 200, 299, "RNTO %s", state->final_path);
261 }
262 if (rc != 0) {
263 sd_gsiftp_delete_temp(&session, temp_path);
264 gftp_session_close(&session);
265 return NGX_ERROR;
266 }
267 gftp_session_close(&session);
268 sd_gsiftp_staged_free(handle);
269 return NGX_OK;
270}
