28/* Minimal host:port / [v6]:port splitter (real enough for doctor_locate_classify;
29 * the shipped brix_split_host_port lives in libbrix, which this TU does not link). */
30int
31brix_split_host_port(const char *in, char *h, size_t hs, int *p, int d)
32{
33 const char *colon;
34 size_t hl;
35
36 *p = d;
37 if (in[0] == '[') { /* [v6]:port */
38 const char *close = strchr(in, ']');
39 if (close == NULL) { return -1; }
40 hl = (size_t) (close - in - 1);
41 if (hl == 0 || hl >= hs) { return -1; }
42 memcpy(h, in + 1, hl);
43 h[hl] = '\0';
44 if (close[1] == ':') { *p = atoi(close + 2); }
45 return 0;
46 }
47 colon = strrchr(in, ':');
48 if (colon == in) { return -1; } /* empty host */
49 if (colon != NULL) {
50 hl = (size_t) (colon - in);
51 if (hl >= hs) { return -1; }
52 memcpy(h, in, hl);
53 h[hl] = '\0';
54 *p = atoi(colon + 1);
55 } else if (in[0] == '\0') {
56 return -1;
57 } else {
58 snprintf(h, hs, "%s", in);
59 }
60 return 0;
61}

247static void
248test_cross_cluster(void)
249{
250 doctor_ep eps[4];
251 memset(eps, 0, sizeof(eps));
252
253 /* n<2 => nothing. */
254 doctor_cross_cluster(eps, 1, NULL);
255 CHECK(eps[0].ndx == 0);
256
257 /* [0] manager 5.6.0; DS [1] 5.6.0 90% free, [2] 5.5.0 10% free, [3] 5.6.0.
258 * => version skew (1) + exactly one manager + capacity outlier. */
259 eps[0].cfg.scraped = 1;
260 snprintf(eps[0].cfg.version, 48, "5.6.0");
261 snprintf(eps[0].cfg.role, 24, "manager");
262 eps[1].cfg.scraped = 1; snprintf(eps[1].cfg.version, 48, "5.6.0");
263 eps[1].cfg.space_total = 100; eps[1].cfg.space_free = 90;
264 eps[2].cfg.scraped = 1; snprintf(eps[2].cfg.version, 48, "5.5.0");
265 eps[2].cfg.space_total = 100; eps[2].cfg.space_free = 10;
266 eps[3].cfg.scraped = 1; snprintf(eps[3].cfg.version, 48, "5.6.0");
267 eps[3].cfg.space_total = 100; eps[3].cfg.space_free = 88;
268 doctor_cross_cluster(eps, 4, NULL);
269 CHECK(recorded(&eps[0], "config-version")); /* mixed versions */
270 CHECK(recorded(&eps[0], "cap-imbalance")); /* 10% vs ~89% mean */
271 CHECK(!recorded(&eps[0], "config-role")); /* exactly one manager */
272
273 /* zero managers => config-role FAIL. */

277 eps[1].cfg.scraped = 1; snprintf(eps[1].cfg.role, 24, "server");

284/* doctor_locate_classify types a kXR_locate token by its type+access prefix. */
285static void
286test_locate_classify(void)
287{
288 doctor_cmsloc m;
289 char host[256];
290 int port;
291
292 /* 'S''r' + bracketed IPv6 => online data server, read-only. */
293 memset(&m, 0, sizeof(m));
294 CHECK(doctor_locate_classify("Sr[2001:db8::1]:1094", &m,
295 host, sizeof(host), &port) == 0);
296 CHECK(m.role == DOC_CMS_SERVER && m.reported == 1);
297 CHECK(m.write == 0 && m.pending == 0);
298 CHECK(port == 1094 && strcmp(host, "2001:db8::1") == 0);
299
300 /* 'M''w' => manager/redirector, read/write. */
301 memset(&m, 0, sizeof(m));
302 CHECK(doctor_locate_classify("Mwmgr.example.org:2094", &m,
303 host, sizeof(host), &port) == 0);
304 CHECK(m.role == DOC_CMS_MANAGER && m.write == 1 && m.pending == 0);
305 CHECK(port == 2094 && strcmp(host, "mgr.example.org") == 0);
306
307 /* lowercase 's' => pending (queued/staging) data server. */
308 memset(&m, 0, sizeof(m));
309 CHECK(doctor_locate_classify("srds.example.org:1095", &m,
310 host, sizeof(host), &port) == 0);
311 CHECK(m.role == DOC_CMS_SERVER && m.pending == 1);
312
313 /* lowercase 'm' => pending manager. */
314 memset(&m, 0, sizeof(m));
315 CHECK(doctor_locate_classify("mrsub.example.org:1096", &m,
316 host, sizeof(host), &port) == 0);
317 CHECK(m.role == DOC_CMS_MANAGER && m.pending == 1);
318
319 /* malformed: unknown type byte, too short, or empty host => -1. */
320 memset(&m, 0, sizeof(m));
321 CHECK(doctor_locate_classify("Xrhost:1094", &m, host, sizeof(host), &port) == -1);
322 CHECK(doctor_locate_classify("S", &m, host, sizeof(host), &port) == -1);
323 CHECK(doctor_locate_classify("", &m, host, sizeof(host), &port) == -1);
324 CHECK(doctor_locate_classify(NULL, &m, host, sizeof(host), &port) == -1);
325}
