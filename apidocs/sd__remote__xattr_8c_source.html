1/*
2 * sd_remote_xattr.c — metadata-mutation slots for the remote-origin (s3://)
3 * storage driver: setxattr / removexattr (the user.<name> ↔ x-amz-meta-<name>
4 * surface, the write side of the getxattr/listxattr read slots in
5 * sd_remote_meta.c) and setattr (the reserved advisory unix-attr blob, so a
6 * chmod / kXR_setattr over an s3:// export "sticks" and round-trips — the
7 * approved advisory model, meta_advisory.h).
8 *
9 * S3 has NO in-place metadata edit: any write REPLACES the object's entire
10 * user-metadata set via a copy-onto-self (x-amz-metadata-directive: REPLACE).
11 * A single-attribute mutation must therefore READ the complete current set,
12 * apply the one change, and REWRITE the whole set — otherwise every other
13 * attribute (including the advisory blob) is silently dropped. Enumerating the
14 * current set needs the transport's raw-header slot; without it every mutation
15 * reports ENOTSUP, exactly as listxattr does (sd_remote_meta.c). The read runs
16 * as the requesting user for the *_cred variants, so a deny-mode request never
17 * touches the origin under the shared service credential.
18 */

32/* AWS caps total user metadata at 2 KiB; 32 attrs matches sd_s3_set_meta's hard
33 * limit. Names/values are held bare (no "x-amz-meta-" prefix) — sd_s3_set_meta
34 * lowercases and prefixes on the wire. */
35#define SD_REMOTE_XA_MAX 32
36#define SD_REMOTE_XA_NAME 160
37#define SD_REMOTE_XA_VALUE 2048

39typedef struct {
40 size_t n;
41 char name [SD_REMOTE_XA_MAX][SD_REMOTE_XA_NAME];
42 char value[SD_REMOTE_XA_MAX][SD_REMOTE_XA_VALUE];
43} sd_remote_meta_set;

45/* Lowercase copy of a bare attribute name into dst (AWS lowercases user-meta
46 * names). 0 / -1 (name too long). */
47static int
48sd_remote_xa_lname(const char *name, char *dst, size_t cap)
49{
50 size_t i, n = strlen(name);
51
52 if (n == 0 || n >= cap) {
53 return -1;
54 }
55 for (i = 0; i < n; i++) {
56 char c = name[i];
57 dst[i] = (c >= 'A' && c <= 'Z') ? (char) (c - 'A' + 'a') : c;
58 }
59 dst[n] = '\0';
60 return 0;
61}

63/* Index of `lname` in the set, or -1 if absent (names are already lowercased). */
64static int
65sd_remote_xa_find(const sd_remote_meta_set *ms, const char *lname)
66{
67 size_t i;
68
69 for (i = 0; i < ms->n; i++) {
70 if (strcmp(ms->name[i], lname) == 0) {

77/* Append one bare name=value pair (value may hold `vlen` bytes). NGX_OK, or
78 * NGX_ERROR with errno E2BIG (set full) / ERANGE (name or value too long). */
79static ngx_int_t
80sd_remote_xa_append(sd_remote_meta_set *ms, const char *lname,
81 const char *val, size_t vlen)
82{
83 if (ms->n >= SD_REMOTE_XA_MAX) {
84 errno = E2BIG;
85 return NGX_ERROR;
86 }
87 if (strlen(lname) >= SD_REMOTE_XA_NAME || vlen >= SD_REMOTE_XA_VALUE) {
88 errno = ERANGE;
89 return NGX_ERROR;
90 }
91 memcpy(ms->name[ms->n], lname, strlen(lname) + 1);
92 memcpy(ms->value[ms->n], val, vlen);
93 ms->value[ms->n][vlen] = '\0';
94 ms->n++;
95 return NGX_OK;
96}

98/* Load the object's COMPLETE current user-metadata set (every user.<name> plus
99 * the reserved advisory blob, which listxattr deliberately omits) so a single
100 * mutation can be rewritten whole. NGX_OK (ms filled), or NGX_ERROR with errno:
101 * ENOENT (object absent / 404), ENOTSUP (transport cannot enumerate headers),
102 * E2BIG (> 32 attrs), EIO otherwise. */
103static ngx_int_t
104sd_remote_meta_load(const brix_sd_remote_cfg_t *cfg, const char *objpath,
105 const char *ak, const char *sk, const char *region, const char *session,
106 sd_remote_meta_set *ms)
107{
108 sd_s3_open_params p;
109 char listbuf[4096]; /* NUL-separated "user.<name>" block */
110 char errbuf[256];
111 sd_s3_file *s3;
112 ssize_t ln;
113 const char *e;
114
115 ms->n = 0;
116 sd_remote_s3_params(cfg, objpath, &p);
117 sd_remote_params_cred(&p, ak, sk, region, session);
118
119 s3 = sd_s3_open_read(&p, errbuf, sizeof(errbuf));
120 if (s3 == NULL) {
121 errno = ENOMEM;
122 return NGX_ERROR;
123 }
124
125 errno = 0;
126 ln = sd_s3_list_meta(s3, listbuf, sizeof(listbuf), errbuf, sizeof(errbuf));
127 if (ln < 0) {
128 int saved = errno;
129 sd_s3_close(s3);
130 errno = saved ? saved : EIO; /* ENOTSUP / ENOENT / EIO */
131 return NGX_ERROR;
132 }
133
134 for (e = listbuf; (size_t) (e - listbuf) < (size_t) ln && *e != '\0'; ) {
135 const char *nm = e;
136 size_t elen = strlen(e);
137 char vbuf[SD_REMOTE_XA_VALUE];
138 sd_s3_meta_buf dst = { vbuf, sizeof(vbuf) };
139 ssize_t vn;
140
141 e += elen + 1;
142 if (strncmp(nm, "user.", 5) != 0 || nm[5] == '\0') {
143 continue; /* only the user. namespace maps */
144 }
145 nm += 5;
146 vn = sd_s3_get_meta(s3, nm, &dst, errbuf, sizeof(errbuf));
147 if (vn < 0) {
148 sd_s3_close(s3);
149 errno = EIO;
150 return NGX_ERROR;
151 }
152 if (sd_remote_xa_append(ms, nm, vbuf, (size_t) vn) != NGX_OK) {
153 int saved = errno;
154 sd_s3_close(s3);
155 errno = saved;
156 return NGX_ERROR;
157 }
158 }
159
160 /* The advisory blob surfaces as POSIX attrs, not a user xattr, so
161 * sd_s3_list_meta skips it; fetch it explicitly to preserve it on rewrite. */
162 {
163 char vbuf[SD_REMOTE_XA_VALUE];
164 sd_s3_meta_buf dst = { vbuf, sizeof(vbuf) };
165 ssize_t vn = sd_s3_get_meta(s3, BRIX_META_ADVISORY_S3META, &dst,
166 errbuf, sizeof(errbuf));
167
168 if (vn < 0) {
169 sd_s3_close(s3);
170 errno = EIO;
171 return NGX_ERROR;
172 }
173 if (vn > 0
174 && sd_remote_xa_append(ms, BRIX_META_ADVISORY_S3META, vbuf,
175 (size_t) vn) != NGX_OK)
176 {
177 int saved = errno;
178 sd_s3_close(s3);
179 errno = saved;
180 return NGX_ERROR;
181 }
182 }
183
184 sd_s3_close(s3);
185 return NGX_OK;
186}

188/* Rewrite `objpath`'s user metadata to exactly the pairs in `ms` (copy-onto-self
189 * REPLACE). NGX_OK / NGX_ERROR (errno set). */
190static ngx_int_t
191sd_remote_meta_store(const brix_sd_remote_cfg_t *cfg, const char *objpath,
192 const char *ak, const char *sk, const char *region, const char *session,
193 const sd_remote_meta_set *ms)
194{
195 sd_s3_open_params p;
196 sd_s3_meta_kv kv[SD_REMOTE_XA_MAX];
197 char errbuf[256];
198 size_t i;
199
200 for (i = 0; i < ms->n; i++) {
201 kv[i].name = ms->name[i];
202 kv[i].value = ms->value[i];
203 }
204 sd_remote_s3_params(cfg, objpath, &p);
205 sd_remote_params_cred(&p, ak, sk, region, session);
206
207 errno = 0;
208 if (sd_s3_set_meta(&p, kv, ms->n, errbuf, sizeof(errbuf)) != 0) {
209 if (errno == 0) { errno = EIO; }
210 return NGX_ERROR;
211 }
212 return NGX_OK;
213}

217/* Shared set/removexattr plumbing, run after the caller's own name/value
218 * validation: derive the lowercase header name from `name5` (the xattr name
219 * past "user."), compute the object key, load the current metadata set and
220 * locate the attribute (*idx < 0 when absent). NGX_OK, or NGX_ERROR with
221 * errno (ERANGE, or meta-load's ENOENT / ENOTSUP / EIO). */
222static ngx_int_t
223sd_remote_xa_open(const brix_sd_remote_cfg_t *cfg, const char *path,
224 const char *name5, const char *ak, const char *sk, const char *region,
225 const char *session, char *objpath, size_t objcap, char *lname,
226 size_t lcap, sd_remote_meta_set *ms, int *idx)
227{
228 if (sd_remote_xa_lname(name5, lname, lcap) != 0) {
229 errno = ERANGE;
230 return NGX_ERROR;
231 }
232
233 sd_remote_s3_key(cfg, path, objpath, objcap);
234 if (sd_remote_meta_load(cfg, objpath, ak, sk, region, session, ms)
235 != NGX_OK)
236 {
237 return NGX_ERROR; /* ENOENT / ENOTSUP / EIO */
238 }
239
240 *idx = sd_remote_xa_find(ms, lname);
241 return NGX_OK;
242}

244static ngx_int_t
245sd_remote_setxattr_impl(brix_sd_instance_t *inst, const char *path,
246 const char *name, const void *val, size_t len, int flags,
247 const char *ak, const char *sk, const char *region, const char *session)
248{
249 const brix_sd_remote_cfg_t *cfg = inst->state;
250 sd_remote_meta_set ms;
251 char objpath[768];
252 char lname[SD_REMOTE_XA_NAME];
253 int idx;
254
255 /* Only the user. namespace maps to x-amz-meta-*; a NUL/CR/LF in the value
256 * cannot ride in an HTTP header. */
257 if (strncmp(name, "user.", 5) != 0 || name[5] == '\0') {
258 errno = ENOTSUP;
259 return NGX_ERROR;
260 }
261 /* A value this mapping cannot REPRESENT is "unfit here", not "invalid":
262 * the xmeta carrier's fallback contract (xmeta_xattr_unfit) rides the
263 * sidecar on E2BIG/ENOTSUP but hard-fails on EINVAL — and the cache's own
264 * cinfo record is a binary blob, so EINVAL here broke every sidecar-backed
265 * cinfo store over a remote store the moment this driver gained a setxattr
266 * slot (the cachestore-sidecar scenario caught it). Oversize → E2BIG;
267 * bytes that cannot survive an HTTP header round-trip → ENOTSUP. A real
268 * client xattr with binary content gets the same honest answer: this
269 * transport cannot carry it. */
270 if (len >= SD_REMOTE_XA_VALUE) {
271 errno = E2BIG;
272 return NGX_ERROR;
273 }
274 if (memchr(val, '\0', len) != NULL
275 || memchr(val, '\r', len) != NULL
276 || memchr(val, '\n', len) != NULL)
277 {
278 errno = ENOTSUP;
279 return NGX_ERROR;
280 }
281 if (sd_remote_xa_open(cfg, path, name + 5, ak, sk, region, session,
282 objpath, sizeof(objpath), lname, sizeof(lname),
283 &ms, &idx) != NGX_OK)
284 {
285 return NGX_ERROR;
286 }
287
288 if (idx >= 0 && (flags & XATTR_CREATE)) {
289 errno = EEXIST;
290 return NGX_ERROR;
291 }
292 if (idx < 0 && (flags & XATTR_REPLACE)) {
293 errno = ENODATA;
294 return NGX_ERROR;
295 }
296 if (idx >= 0) {
297 memcpy(ms.value[idx], val, len);
298 ms.value[idx][len] = '\0';
299 } else if (sd_remote_xa_append(&ms, lname, val, len) != NGX_OK) {
300 return NGX_ERROR; /* E2BIG */
301 }
302
303 return sd_remote_meta_store(cfg, objpath, ak, sk, region, session, &ms);
304}

336sd_remote_removexattr_impl(brix_sd_instance_t *inst, const char *path,

339{

344 int idx;

354 return NGX_ERROR;

358 errno = ENODATA; /* attribute absent */
359 return NGX_ERROR;

362 * — S3 metadata is an unordered set). */

365 memcpy(ms.name[idx], ms.name[ms.n], sizeof(ms.name[idx]));
366 memcpy(ms.value[idx], ms.value[ms.n], sizeof(ms.value[idx]));
367 }
368
369 return sd_remote_meta_store(cfg, objpath, ak, sk, region, session, &ms);
370}

398/* ---- setattr (advisory unix-attr blob) ------------------------------------
399 *
400 * WHAT: Apply chmod / kXR_setattr (mode, mtime, owner) to `path` by patching the
401 * reserved advisory blob (x-amz-meta-xrd-unixattr) that overlays stat.
402 * WHY: finding #4 — WebDAV/xrdfs chmod/setattr over an s3:// backend hit the
403 * NULL setattr slot; the VFS treated that as a no-op so the change was
404 * silently lost.
405 * HOW: Resolve the target object (the "path" file key, else the "path/" folder
406 * marker for a directory), read its full metadata set, decode+patch the
407 * advisory blob with the present fields of *attr (atime is not tracked —
408 * S3 has no atime; UTIME_OMIT skips a field, UTIME_NOW stamps wall time),
409 * and rewrite the whole set so co-existing user xattrs survive.
410 */
411
412static ngx_int_t
413sd_remote_setattr_impl(brix_sd_instance_t *inst, const char *path,
414 const brix_sd_setattr_t *attr,
415 const char *ak, const char *sk, const char *region, const char *session)
416{
417 const brix_sd_remote_cfg_t *cfg = inst->state;
418 sd_remote_meta_set ms;
419 brix_meta_advisory_t delta;
420 char objpath[768];
421 char blob[SD_REMOTE_XA_VALUE];
422 int idx;
423
424 /* Nothing representable is success (matches the POSIX/no-op setattr
425 * contract) — and it is decided BEFORE the object round-trip, so an
426 * atime-only request costs no request pair. */
427 if (!brix_meta_advisory_from_setattr(attr, &delta)) {
428 return NGX_OK;
429 }
430
431 /* Target the file key; fall back to the "path/" directory marker. Either
432 * absent -> ENOENT (never fabricate an object to hang metadata on). */
433 sd_remote_s3_key(cfg, path, objpath, sizeof(objpath));
434 if (sd_remote_meta_load(cfg, objpath, ak, sk, region, session, &ms)
435 != NGX_OK)
436 {
437 if (errno != ENOENT) {
438 return NGX_ERROR; /* ENOTSUP / EIO */
439 }
440 sd_remote_s3_dirkey(cfg, path, objpath, sizeof(objpath));
441 if (sd_remote_meta_load(cfg, objpath, ak, sk, region, session, &ms)
442 != NGX_OK)
443 {
444 return NGX_ERROR; /* ENOENT (neither) / ENOTSUP / EIO */
445 }
446 }
447
448 idx = sd_remote_xa_find(&ms, BRIX_META_ADVISORY_S3META);
449 blob[0] = '\0';
450 if (idx >= 0) {
451 memcpy(blob, ms.value[idx], strlen(ms.value[idx]) + 1);
452 }
453 if (brix_meta_advisory_patch(blob, sizeof(blob), &delta) < 0) {
454 errno = EIO;
455 return NGX_ERROR;
456 }
457 if (idx >= 0) {
458 memcpy(ms.value[idx], blob, strlen(blob) + 1);
459 } else if (sd_remote_xa_append(&ms, BRIX_META_ADVISORY_S3META, blob,
460 strlen(blob)) != NGX_OK)
461 {
462 return NGX_ERROR; /* E2BIG */
463 }
464
465 return sd_remote_meta_store(cfg, objpath, ak, sk, region, session, &ms);
466}
