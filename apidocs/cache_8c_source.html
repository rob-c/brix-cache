1/*
2 * cache.c — per-worker DNS answer cache (phase-116 W4).
3 *
4 * WHAT: A bounded positive + negative cache keyed by (lowercased name, address
5 * family policy). Positive entries hold up to BRIX_DNS_MAX_ADDRS port-
6 * less addresses and an expiry; negative entries hold an expiry only.
7 * Least-recently-used entries are evicted once the cap is reached.
8 * WHY: The re-resolve paths (health checks, TPC pins, proxy pool adds) can
9 * ask for the same name many times per second; without a cache every
10 * ask is a resolver round-trip or a thread-pool hop. The negative
11 * cache stops a dead name from hammering the nameserver at retry pace.
12 * HOW: ngx_rbtree keyed by ngx_crc32_short(name) with a string compare on
13 * collision, plus an ngx_queue_t in LRU order — the ngx_resolver layout,
14 * without its per-name waiting lists. Per-process singleton allocated
15 * lazily with ngx_alloc (never from a request pool); a fork inherits an
16 * empty cache because entries are created only at runtime. A plain
17 * pthread mutex guards every entry point: the event loop and the
18 * thread-pool callers of brix_dns_resolve_sync() share one cache, and
19 * nothing under the lock blocks or re-enters (ngx_time() is a cached
20 * read, allocation is ngx_alloc). Not an SHM mutex — this cache is
21 * process-private (invariant 10 concerns shared memory only).
22 */
23#include "net/dns/dns.h"

84/* lowercase copy for a case-insensitive key; returns the crc */
85static uint32_t
86dns_cache_key(const ngx_str_t *name, u_char *lower, size_t *len)
87{
88 size_t i, n = ngx_min(name->len, BRIX_RESOLV_DOMAIN_LEN - 1);
89
90 for (i = 0; i < n; i++) {
91 lower[i] = ngx_tolower(name->data[i]);
92 }
93 if (n > 0 && lower[n - 1] == '.') {
94 n--; /* "host." == "host" */
95 }
96 *len = n;
97 return ngx_crc32_short(lower, n);
98}

101static dns_cache_entry_t *
102dns_cache_find(uint32_t key, const u_char *lower, size_t len,
103 brix_af_policy_t af)
104{
105 ngx_rbtree_node_t *node = dns_cache.tree.root;
106 ngx_rbtree_node_t *sentinel = dns_cache.tree.sentinel;
107
108 while (node != sentinel) {
109 dns_cache_entry_t *e;
110
111 if (key < node->key) {
112 node = node->left;
113 continue;
114 }
115 if (key > node->key) {
116 node = node->right;
117 continue;
118 }
119 e = (dns_cache_entry_t *) node;
120 if (e->len == len && e->af == af
121 && ngx_memcmp(e->name, lower, len) == 0)
122 {
123 return e;
124 }
125 /* same crc, different name: the tree is not a multimap, so walk
126 * the right subtree the way ngx_resolver does for its hash */
127 node = node->right;
128 }
129 return NULL;
130}

144static ngx_int_t
145dns_cache_lookup_locked(brix_dns_req_t *req)
146{
147 u_char lower[BRIX_RESOLV_DOMAIN_LEN];
148 size_t len;
149 uint32_t key;
150 dns_cache_entry_t *e;
151 ngx_uint_t i;
152
153 dns_cache_ready();
154 key = dns_cache_key(&req->name, lower, &len);
155 e = dns_cache_find(key, lower, len, req->af);
156 if (e == NULL) {
157 dns_cache.misses++;
158 return NGX_DECLINED;
159 }
160 if (e->expires <= ngx_time()) {
161 dns_cache_remove(e);
162 dns_cache.misses++;
163 return NGX_DECLINED;
164 }
165 if (e->pending) {
166 dns_cache.misses++; /* the async driver resolves as usual */
167 return NGX_DECLINED;
168 }
169
170 /* touch: move to the LRU head */
171 ngx_queue_remove(&e->queue);
172 ngx_queue_insert_head(&dns_cache.lru, &e->queue);
173
174 req->cached = 1;
175 req->ttl = e->expires - ngx_time();
176 if (e->negative) {
177 dns_cache.negative_hits++;
178 req->negative = 1;
179 req->naddrs = 0;
180 req->rc = NGX_ERROR;
181 req->error = "name not found (negative cache)";
182 return NGX_OK;
183 }
184 dns_cache.hits++;
185 req->naddrs = e->naddrs;
186 for (i = 0; i < e->naddrs; i++) {
187 req->addrs[i] = e->addrs[i];
188 ngx_inet_set_port((struct sockaddr *) &req->addrs[i].ss, req->port);
189 }
190 req->rc = NGX_OK;
191 req->error = NULL;
192 return NGX_OK;
193}

208static void
209dns_cache_evict_lru(void)
210{
211 ngx_queue_t *q;
212
213 while (dns_cache.entries >= dns_cache.max
214 && !ngx_queue_empty(&dns_cache.lru))
215 {
216 q = ngx_queue_last(&dns_cache.lru);
217 dns_cache_remove(ngx_queue_data(q, dns_cache_entry_t, queue));
218 }

222static void
223dns_cache_store_locked(const brix_dns_req_t *req)
224{
225 u_char lower[BRIX_RESOLV_DOMAIN_LEN];
226 size_t len;
227 uint32_t key;
228 dns_cache_entry_t *e;
229 ngx_uint_t i;
230
231 dns_cache_ready();
232 key = dns_cache_key(&req->name, lower, &len);
233 e = dns_cache_find(key, lower, len, req->af);
234 if (e != NULL) {
235 dns_cache_remove(e);
236 }
237 dns_cache_evict_lru();
238
239 e = ngx_calloc(sizeof(dns_cache_entry_t), req->log);
240 if (e == NULL) {
241 return;
242 }
243 e->name = ngx_alloc(len + 1, req->log);
244 if (e->name == NULL) {
245 ngx_free(e);
246 return;
247 }
248 ngx_memcpy(e->name, lower, len);
249 e->name[len] = '\0';
250 e->len = len;
251 e->af = req->af;
252 e->expires = ngx_time() + req->ttl;
253 e->negative = (req->rc != NGX_OK);
254 e->naddrs = e->negative ? 0 : req->naddrs;
255 for (i = 0; i < e->naddrs; i++) {
256 e->addrs[i] = req->addrs[i];
257 ngx_inet_set_port((struct sockaddr *) &e->addrs[i].ss, 0);
258 }
259 e->node.key = key;
260 ngx_rbtree_insert(&dns_cache.tree, &e->node);
261 ngx_queue_insert_head(&dns_cache.lru, &e->queue);
262 dns_cache.entries++;
263}

278/* WHAT: insert an in-flight marker for (name, af) unless a live entry exists.
279 * WHY: brix_dns_lookup_cached() must not start one background fill per probe
280 * while the first is still running; the marker makes the second probe
281 * an "in flight" answer instead. brix_dns_cache_store() replaces it.
282 * HOW: a pending entry with a short expiry; an expired marker (fill lost) is
283 * simply re-armed. */
284unsigned
285brix_dns_cache_mark_pending(const ngx_str_t *name, brix_af_policy_t af,
286 time_t hold)
287{
288 u_char lower[BRIX_RESOLV_DOMAIN_LEN];
289 size_t len;
290 uint32_t key;
291 dns_cache_entry_t *e;
292 unsigned marked = 0;
293
294 pthread_mutex_lock(&dns_cache_mu);
295 dns_cache_ready();
296 key = dns_cache_key(name, lower, &len);
297 e = dns_cache_find(key, lower, len, af);
298 if (e != NULL && e->expires > ngx_time()) {
299 pthread_mutex_unlock(&dns_cache_mu);
300 return 0;
301 }
302 if (e != NULL) {
303 dns_cache_remove(e);
304 }
305 dns_cache_evict_lru();
306 e = ngx_calloc(sizeof(dns_cache_entry_t), ngx_cycle->log);
307 if (e != NULL) {
308 e->name = ngx_alloc(len + 1, ngx_cycle->log);
309 if (e->name == NULL) {
310 ngx_free(e);
311 e = NULL;
312 }
313 }
314 if (e != NULL) {
315 ngx_memcpy(e->name, lower, len);
316 e->name[len] = '\0';
317 e->len = len;
318 e->af = af;
319 e->expires = ngx_time() + (hold > 0 ? hold : BRIX_DNS_PENDING_HOLD);
320 e->pending = 1;
321 e->node.key = key;
322 ngx_rbtree_insert(&dns_cache.tree, &e->node);
323 ngx_queue_insert_head(&dns_cache.lru, &e->queue);
324 dns_cache.entries++;
325 marked = 1;
326 }
327 pthread_mutex_unlock(&dns_cache_mu);
328 return marked;
329}
