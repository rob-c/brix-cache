3/*
4 * WHAT: Prometheus export for the OCI distribution plane (phase-104 D3.1).
5 * WHY: a registry mirror is judged on four numbers — how much of the pull
6 * traffic the cache absorbed, how many bytes still had to cross the WAN,
7 * whether the upstream token dance is working, and whether anything came
8 * back that did not hash to the digest it claimed. Everything here
9 * answers one of those.
10 * HOW: one three-label family (surface x class x outcome, all fixed enums —
11 * INVARIANT #8) written out row by row, plus four single-label/scalar
12 * families through the shared writer helpers. Image names, tags and
13 * digests never appear: they are unbounded wire data.

79/* The request family: 2 x 7 x 5 = 70 series, every index a compile-time enum.
80 * Zero rows are emitted too — a family that appears only once traffic of that
81 * shape arrives makes `rate()` on a fresh node return nothing rather than 0,
82 * which reads as "no data" instead of "no pushes attempted". */
83static void
84oci_emit_requests(metrics_writer_t *mw, const ngx_brix_oci_metrics_t *o)
85{
86 ngx_uint_t s, c, k;
87
88 mw_printf(mw,
89 "# HELP brix_oci_requests_total OCI distribution requests by surface, "
90 "traffic class and outcome\n"
91 "# TYPE brix_oci_requests_total counter\n");
92
93 for (s = 0; s < BRIX_OCI_SURFACE_COUNT; s++) {
94 for (c = 0; c < BRIX_OCI_MCLASS_COUNT; c++) {
95 for (k = 0; k < BRIX_OCI_OUT_COUNT; k++) {
96 mw_printf(mw,
97 "brix_oci_requests_total"
98 "{surface=\"%s\",class=\"%s\",outcome=\"%s\"} %lu\n",
99 brix_oci_surface_names[s], brix_oci_mclass_names[c],
100 brix_oci_outcome_names[k],
101 (unsigned long) o->requests_total[s][c][k]);
102 }
103 }
104 }
