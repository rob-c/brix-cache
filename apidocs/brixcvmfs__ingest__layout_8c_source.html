1/* brixcvmfs_ingest_layout.c — the shape of the published namespace: the tag
2 * symlink every layout ends with, and `--layout layered` (phase-104 D15.6) —
3 * per-layer publish roots and the ledgers that make them reusable. Tool
4 * surface only (G14).
5 *
6 * The flat layout publishes one merged rootfs per image, so two images off
7 * one base pay for that base twice in catalog rows, in scan time and in the
8 * bytes fetched to produce them (the CAS dedups the *stored* files, which is
9 * why the flat layout is still the default — it costs storage nothing). What
10 * it cannot dedup is the work: a rootfs has to be fetched, decompressed and
11 * scanned before the store can notice it already had every file.
12 *
13 * Layered publishes each layer at its own content-addressed root,
14 *
15 * <prefix>/.layers/<alg>/<layer-hex>/ one layer, verbatim
16 * <prefix>/.images/<alg>/<manifest-hex>/.layers the composition, in order
17 *
18 * so the second image off a shared base fetches, decompresses and scans only
19 * the layers that are actually new to the repository. The price is that the
20 * image root is no longer a runnable rootfs: composing the layer roots is
21 * the consumer's job (overlayfs lowerdirs, lowest last), which is why this
22 * is a flag and not the default.
23 *
24 * Ledgers, all under <repo>/.brix-ingest and all advisory (they are a cache
25 * of what the repository already holds; losing one costs a re-publish of
26 * content that was already there, never correctness):
27 * layers<pfx>/<layer-hex> "<prefix> <digest> <diffid|-> <rev> <utc>"
28 * imglayers<pfx>/<image-hex> the layer hexes of that image, one per line
29 * The second is what lets `ingest prune` tell an orphaned layer root from a
30 * shared one, exactly as the memo files do for image roots.
31 */

50int
51bci_tag_symlink(const char *upper, const char *tagrel, const char *root_rel)
52{
53 char lpath[ING_PATH_MAX + 8], dir[ING_PATH_MAX + 8], target[ING_PATH_MAX];
54 size_t off = 0;
55 char *slash;
56
57 if (snprintf(lpath, sizeof(lpath), "%s/%s", upper, tagrel)
58 >= (int) sizeof(lpath)) {
59 return bci_fail(ING_FAIL, "tag path too long", tagrel);
60 }
61 snprintf(dir, sizeof(dir), "%s", lpath);
62 slash = strrchr(dir, '/');
63 *slash = '\0'; /* an upper path always has one */
64 if (bci_mkdir_p(dir, 0755) != 0) {
65 return bci_fail(ING_FAIL, "cannot create tag dir", dir);
66 }
67 for (const char *p = tagrel; *p != '\0'; p++) { /* one ../ per level */
68 if (*p == '/' && off + 3 < sizeof(target)) {
69 memcpy(target + off, "../", 3);
70 off += 3;
71 }
72 }
73 if (off + strlen(root_rel) >= sizeof(target)) {
74 return bci_fail(ING_FAIL, "tag link target too long", root_rel);
75 }
76 snprintf(target + off, sizeof(target) - off, "%s", root_rel);
77 return symlink(target, lpath) == 0
78 ? ING_OK : bci_fail(ING_FAIL, "cannot create tag symlink", lpath);
79}

107/* The third field of a ledger line, NUL-terminated in place; NULL when the
108 * line is malformed or the field is the "-" that means "not captured". */
109static char *
110lay_line_diffid(char *line)
111{
112 char *p = strchr(line, ' '); /* past <prefix> */
113
114 p = p != NULL ? strchr(p + 1, ' ') : NULL; /* past <digest> */
115 if (p == NULL) {
116 return NULL;
117 }
118 p++;
119 char *end = strchr(p, ' ');
120 if (end != NULL) {
121 *end = '\0';
122 }
123 return p[0] == '-' && p[1] == '\0' ? NULL : p;
124}

127int
128bci_layer_known(const char *repo, const char *prefix, ing_layer_t *l)
129{
130 char path[ING_PATH_MAX], line[LAY_LINE_MAX], *diffid;
131
132 l->diffid[0] = '\0';
133 if (lay_ledger_path(path, sizeof(path), repo, "layers", prefix,
134 l->dig.hex) != 0
135 || bci_read_line(path, line, sizeof(line)) != 0)
136 {
137 return 0;
138 }
139 diffid = lay_line_diffid(line);
140 if (diffid != NULL) {
141 snprintf(l->diffid, sizeof(l->diffid), "%s", diffid);
142 }
143 return 1;

147int
148bci_layer_stage(const char *repo, const char *prefix, const char *upper,
149 const char *digest, int need_diffid, ing_layer_t *out,
150 char *updir, size_t updirlen)
151{
152 char rel[ING_PATH_MAX];
153
154 memset(out, 0, sizeof(*out));
155 if (brix_oci_digest_parse(digest, strlen(digest), &out->dig) != 0) {
156 return bci_fail(ING_FAIL, "manifest layer with an unusable digest",
157 digest);
158 }
159 /* A published layer whose ledger entry carries no diff_id predates
160 * --verify-diffids; materializing it is the only way to obtain one, and
161 * a flag that verified nothing would be worse than the cost. */
162 out->reused = bci_layer_known(repo, prefix, out)
163 && (!need_diffid || out->diffid[0] != '\0');
164
165 if (snprintf(rel, sizeof(rel), "%s/.layers/%s/%s", upper,
166 brix_oci_alg_name(out->dig.alg), out->dig.hex)
167 >= (int) sizeof(rel)
168 || snprintf(updir, updirlen, "%s", rel) >= (int) updirlen)
169 {
170 return bci_fail(ING_FAIL, "layer scratch path too long", out->dig.hex);
171 }
172 if (out->reused) {
173 return ING_OK; /* nothing enters the upper tree */
174 }
175 return bci_mkdir_p(updir, 0755) == 0 ? ING_OK
176 : bci_fail(ING_FAIL, "cannot create layer scratch", updir);
177}

180void
181bci_layer_record(const char *repo, const char *prefix, const ing_layer_t *v,
182 int n, long rev)
183{
184 char path[ING_PATH_MAX], line[LAY_LINE_MAX], utc[40], digest[BRIX_OCI_DIGEST_STRLEN];
185
186 bci_utc_now(utc, sizeof(utc));
187 for (int i = 0; i < n; i++) {
188 if (v[i].reused) {
189 continue; /* the ledger entry is why it was */
190 }
191 if (lay_ledger_path(path, sizeof(path), repo, "layers", prefix,
192 v[i].dig.hex) != 0
193 || brix_oci_digest_format(&v[i].dig, digest, sizeof(digest)) < 0)
194 {
195 continue;
196 }
197 snprintf(line, sizeof(line), "%s %s %s %ld %s\n", prefix, digest,
198 v[i].diffid[0] != '\0' ? v[i].diffid : "-", rev, utc);
199 if (bci_write_atomic(path, line) != 0) {
200 fprintf(stderr, "brixcvmfs ingest: warning: cannot write layer"
201 " ledger %s\n", path);
202 }
203 }

231int
232bci_layers_sidecar(const char *root_abs, const char *root_rel,
233 const ing_layer_t *v, int n)
234{
235 char path[ING_PATH_MAX + 16], up[ING_PATH_MAX];
236 size_t off = 0;
237 FILE *f;
238
239 /* One "../" per component of the image root, so the descriptor reads the
240 * same from a Stratum-0 tree and from a client mount. */
241 for (const char *p = root_rel; *p != '\0'; p++) {
242 if (p[0] == '/' && p[1] != '\0') {
243 off += 3;
244 }
245 }
246 off += 3; /* the last component itself */
247 if (off >= sizeof(up)) {
248 return bci_fail(ING_FAIL, "image root too deep for a descriptor",
249 root_rel);
250 }
251 memset(up, 0, off + 1);
252 for (size_t i = 0; i < off; i += 3) {
253 memcpy(up + i, "../", 3);
254 }
255
256 if (snprintf(path, sizeof(path), "%s/.layers", root_abs)
257 >= (int) sizeof(path)
258 || (f = fopen(path, "w")) == NULL)
259 {
260 return bci_fail(ING_FAIL, "cannot write layers descriptor", root_abs);
261 }
262 for (int i = 0; i < n; i++) {
263 fprintf(f, "%s.layers/%s/%s\n", up, brix_oci_alg_name(v[i].dig.alg),
264 v[i].dig.hex);
265 }
266 return fclose(f) == 0 ? ING_OK
267 : bci_fail(ING_FAIL, "cannot write layers descriptor", path);
268}

271/* ---- prune: the layer roots nothing composes any more ---------------------- */

273/* Does any surviving image list this layer? The imglayers records are the
274 * only place that knows, which is why prune removes an image's record only
275 * after the publish that removed the image itself. */
276static int
277lay_referenced(const char *dir, const char *hex)
278{
279 char path[ING_PATH_MAX + 160], line[BRIX_OCI_HEXLEN_MAX + 8];
280 DIR *d = opendir(dir);
281 struct dirent *e;
282 int found = 0;
283
284 if (d == NULL) {
285 return 0;
286 }
287 while (!found && (e = readdir(d)) != NULL) {
288 FILE *f;
289 if (e->d_name[0] == '.'
290 || snprintf(path, sizeof(path), "%s/%s", dir, e->d_name)
291 >= (int) sizeof(path)
292 || (f = fopen(path, "r")) == NULL)
293 {
294 continue;
295 }
296 while (!found && fgets(line, sizeof(line), f) != NULL) {
297 line[strcspn(line, "\r\n")] = '\0';
298 found = strcmp(line, hex) == 0;
299 }
300 fclose(f);
301 }
302 closedir(d);
303 return found;
304}

307int
308bci_layer_orphans(const char *repo, const char *prefix, cvmfs_changeset_t *cs,
309 char ***out_hex)
310{
311 char ledger[ING_PATH_MAX], imgdir[ING_PATH_MAX], path[ING_PATH_MAX];
312 char **victims = NULL;
313 int n = 0;
314 DIR *d;
315 struct dirent *e;
316
317 *out_hex = NULL;
318 if (lay_ledger_path(ledger, sizeof(ledger), repo, "layers", prefix, "") != 0
319 || lay_ledger_path(imgdir, sizeof(imgdir), repo, "imglayers", prefix,
320 "") != 0
321 || (d = opendir(ledger)) == NULL)
322 {
323 return 0; /* nothing layered was ever published */
324 }
325 while ((e = readdir(d)) != NULL) {
326 brix_oci_digest_t dig;
327 cvmfs_change_t *ch;
328 char **grown;
329
330 /* A ledger name is a bare hex whose width names its algorithm; a
331 * name that is not one is not this tool's to delete. */
332 if (brix_oci_digest_parse_hex(e->d_name, strlen(e->d_name), &dig) != 0
333 || lay_referenced(imgdir, e->d_name)
334 || bci_layer_path(path, sizeof(path), prefix, &dig) != ING_OK)
335 {
336 continue;
337 }
338 ch = bci_cs_append(cs);
339 grown = realloc(victims, (size_t) (n + 1) * sizeof(*victims));
340 if (grown != NULL) {
341 victims = grown; /* the old block is dangling now */
342 victims[n] = NULL;
343 }
344 if (ch == NULL || grown == NULL
345 || (ch->path = strdup(path)) == NULL
346 || (victims[n] = strdup(e->d_name)) == NULL)
347 {
348 closedir(d);
349 while (n-- > 0) {
350 free(victims[n]); /* the ledger stays: nothing was published */
351 }
352 free(victims);
353 bci_fail(ING_FAIL, "out of memory", NULL);
354 return -1;
355 }
356 ch->op = CVMFS_CH_DELETE;
357 n++;
358 }
359 closedir(d);
360 *out_hex = victims;
361 return n;
362}
