1/*
2 * vfs_writer_spill.c — SPILL mode of the unified write session (phase-107 C1).
3 *
4 * WHAT: brix_vfs_writer_spill_{enter,put,drain,discard} — a local reorder
5 * buffer that lets a staged-only backend (http, remote: the two of the
6 * twelve drivers with no CAP_RANDOM_WRITE) accept out-of-order extents:
7 * absorb any offset into a sparse POSIX temp under the configured spill
8 * root, then at commit drain it sequentially into the driver's staged
9 * upload.
10 * WHY: GridFTP mode E and XRootD clients reorder legitimately; before
11 * phase-107 three independent refusals (the writer's EINVAL, sd_http's
12 * ESPIPE, sd_s3's sequential checks) failed every reordered upload on
13 * exactly the two drivers most likely to sit under a WAN-facing export.
14 * Buffering ONCE in the VFS — not once per driver — is the honest
15 * version, and capacity is answered with ENOSPC at the earliest moment
16 * rather than with a slow path or a truncated object.
17 * HOW: One temp (named by brix_make_tmp_path, so the owned-temp reaper
18 * recognises an orphan) whose file offsets are object offsets minus
19 * spill.base — the already-staged prefix [0, base) stays in the driver
20 * session and the drain appends behind it. Coverage is tracked as a
21 * sorted, coalesced extent set: overlap is refused at the write (it
22 * would also poison the verify CRC) and a hole is refused at the drain,
23 * never zero-filled — the filesystem zero-fills sub-block holes
24 * silently, so SEEK_HOLE could not see them. No goto; early-return.
25 */
26#include "vfs_writer_internal.h"
27#include "vfs_backend_registry.h"
28#include "core/compat/tmp_path.h"

30#define BRIX_VFS_SPILL_EXT_INIT 64
31#define BRIX_VFS_SPILL_EXT_MAX 8192 /* matches the GridFTP EB range ceiling */
32#define BRIX_VFS_SPILL_DRAIN_CHUNK (256 * 1024)

34/* T4 — the scratch cannot hold the object: abort the staged session (nothing
35 * may be published), drop the scratch, park the session in FAILED, and account
36 * the refusal. `err` 0 means "capacity" (ENOSPC). Always NGX_ERROR. */
37static ngx_int_t
38spill_fail(brix_vfs_writer_t *w, int err, const char *what)
39{
40 ngx_log_error(NGX_LOG_ERR, w->log, err,
41 "brix: vfs writer: spill failed (%s) — upload refused",
42 what);
43 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
44 brix_vfs_writer_spill_discard(w);
45 if (w->st != NULL) {
46 brix_vfs_staged_abort(w->st, 1 /* remove temp */);
47 w->st = NULL;
48 }
49 w->mode = BRIX_VFS_WRITER_FAILED;
50 errno = err ? err : ENOSPC;
51 return NGX_ERROR;
52}

54/* Binary search: index of the first entry with .start > start (== the
55 * insertion slot that keeps the set sorted). */
56static ngx_uint_t
57spill_ext_slot(const brix_vfs_spill_t *sp, off_t start)
58{
59 ngx_uint_t lo = 0, hi = sp->n_ext;
60
61 while (lo < hi) {
62 ngx_uint_t mid = lo + (hi - lo) / 2;
63
64 if (sp->ext[mid].start <= start) {
65 lo = mid + 1;
66 } else {
67 hi = mid;
68 }
69 }
70 return lo;
71}

73/* Double the coverage-set capacity (pool realloc — the pool frees the old
74 * block with the request). NGX_OK, or NGX_ERROR with errno = ENOSPC at the
75 * BRIX_VFS_SPILL_EXT_MAX ceiling, ENOMEM on allocation failure. */
76static ngx_int_t
77spill_ext_grow(brix_vfs_writer_t *w, brix_vfs_spill_t *sp)
78{
79 brix_vfs_spill_ext_t *grown;
80 ngx_uint_t ncap = sp->ext_cap ? sp->ext_cap * 2
81 : BRIX_VFS_SPILL_EXT_INIT;
82
83 if (sp->n_ext >= BRIX_VFS_SPILL_EXT_MAX) {
84 errno = ENOSPC;
85 return NGX_ERROR;
86 }
87 grown = ngx_palloc(w->pool, ncap * sizeof(sp->ext[0]));
88 if (grown == NULL) {
89 errno = ENOMEM;
90 return NGX_ERROR;
91 }
92 if (sp->n_ext > 0) {
93 ngx_memcpy(grown, sp->ext, sp->n_ext * sizeof(sp->ext[0]));
94 }
95 sp->ext = grown;
96 sp->ext_cap = ncap;
97 return NGX_OK;
98}

100/* Record [start, end) in the sorted disjoint coverage set. NGX_OK, or
101 * NGX_ERROR with errno = EINVAL on overlap, ENOMEM on allocation failure, and
102 * ENOSPC when the set outgrows BRIX_VFS_SPILL_EXT_MAX. Exact-touch neighbours
103 * coalesce, so in-order runs inside a reordered stream stay one entry. */
104static ngx_int_t
105spill_ext_add(brix_vfs_writer_t *w, off_t start, off_t end)
106{
107 brix_vfs_spill_t *sp = &w->spill;
108 ngx_uint_t pos = spill_ext_slot(sp, start);
109
110 if ((pos > 0 && sp->ext[pos - 1].end > start)
111 || (pos < sp->n_ext && end > sp->ext[pos].start))
112 {
113 errno = EINVAL; /* overlaps bytes already landed */
114 return NGX_ERROR;
115 }
116 if (pos > 0 && sp->ext[pos - 1].end == start) {
117 sp->ext[pos - 1].end = end;
118 if (pos < sp->n_ext && sp->ext[pos].start == end) {
119 sp->ext[pos - 1].end = sp->ext[pos].end; /* bridged two runs */
120 ngx_memmove(&sp->ext[pos], &sp->ext[pos + 1],
121 (sp->n_ext - pos - 1) * sizeof(sp->ext[0]));
122 sp->n_ext--;
123 }
124 return NGX_OK;
125 }
126 if (pos < sp->n_ext && sp->ext[pos].start == end) {
127 sp->ext[pos].start = start;
128 return NGX_OK;
129 }
130
131 if (sp->n_ext == sp->ext_cap && spill_ext_grow(w, sp) != NGX_OK) {
132 return NGX_ERROR;
133 }
134 ngx_memmove(&sp->ext[pos + 1], &sp->ext[pos],
135 (sp->n_ext - pos) * sizeof(sp->ext[0]));
136 sp->ext[pos].start = start;
137 sp->ext[pos].end = end;
138 sp->n_ext++;
139 return NGX_OK;
140}

142/* Create the scratch temp under `root` with an exclusive owned-temp name, so
143 * a crashed worker's orphan is recognised (and reclaimed) by
144 * brix_tmp_reap_all() walking the registered spill root. */
145static ngx_int_t
146spill_create(brix_vfs_writer_t *w, const char *root)
147{
148 char base[PATH_MAX];
149 char path[PATH_MAX];
150 int attempts;
151
152 if ((size_t) snprintf(base, sizeof(base), "%s/spill", root)
153 >= sizeof(base))
154 {
155 errno = ENAMETOOLONG;
156 return NGX_ERROR;
157 }
158 for (attempts = 0; attempts < 16; attempts++) {
159 int fd;
160
161 if (brix_make_tmp_path(base, path, sizeof(path)) != NGX_OK) {
162 errno = ENAMETOOLONG;
163 return NGX_ERROR;
164 }
165 fd = open(path, O_CREAT | O_EXCL | O_RDWR | O_CLOEXEC, 0600); /* vfs-seam-allow: DOMAIN_STAGE — spill scratch under the service spill root, never export storage */
166 if (fd >= 0) {
167 size_t plen = ngx_strlen(path);
168
169 w->spill.path = ngx_pnalloc(w->pool, plen + 1);
170 if (w->spill.path == NULL) {
171 (void) close(fd); /* vfs-seam-allow: DOMAIN_STAGE — spill scratch teardown */
172 (void) unlink(path); /* vfs-seam-allow: DOMAIN_STAGE — spill scratch teardown */
173 errno = ENOMEM;
174 return NGX_ERROR;
175 }
176 ngx_memcpy(w->spill.path, path, plen + 1);
177 w->spill.fd = fd;
178 return NGX_OK;
179 }
180 if (errno != EEXIST) {
181 return NGX_ERROR;
182 }
183 }
184 errno = EEXIST;
185 return NGX_ERROR;
186}

188ngx_int_t
189brix_vfs_writer_spill_enter(brix_vfs_writer_t *w, off_t off, size_t len)
190{
191 const char *root = NULL;
192 off_t max = 0;
193 int ce;
194
195 if (off < w->staged_cursor) {
196 /* The bytes below the cursor already streamed into the driver's
197 * staged session (heap PUT body, multipart part) and cannot be read
198 * back or patched — no scratch can serve this write. Stay SEQUENTIAL:
199 * an in-order continuation still succeeds. */
200 ngx_log_error(NGX_LOG_ERR, w->log, 0,
201 "brix: vfs writer: write at %O rewinds below the %O bytes "
202 "already staged — a whole-object upload cannot be patched",
203 off, w->staged_cursor);
204 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
205 errno = EINVAL;
206 return NGX_ERROR;
207 }
208
209 (void) brix_vfs_backend_spill(w->ctx->root_canon, &root, &max);
210 if (root == NULL || root[0] == '\0') {
211 ngx_log_error(NGX_LOG_ERR, w->log, 0,
212 "brix: vfs writer: out-of-order write at %O (expected %O) on a "
213 "staged-only backend and no spill scratch is configured — set "
214 "brix_vfs_spill_path (or brix_stage_dir)",
215 off, w->staged_cursor);
216 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
217 errno = ENOSPC;
218 return NGX_ERROR; /* stays SEQUENTIAL */
219 }
220 if (max > 0 && off + (off_t) len - w->staged_cursor > max) {
221 /* §C1 "ceiling, stated up front": the triggering extent alone already
222 * exceeds brix_vfs_spill_max — refuse before creating anything. */
223 return spill_fail(w, ENOSPC, "brix_vfs_spill_max exceeded at entry");
224 }
225 if (spill_create(w, root) != NGX_OK) {
226 ce = errno;
227 ngx_log_error(NGX_LOG_ERR, w->log, ce,
228 "brix: vfs writer: cannot create spill scratch under \"%s\"",
229 root);
230 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
231 errno = ENOSPC; /* capacity answer for the client; ce is logged */
232 return NGX_ERROR; /* stays SEQUENTIAL */
233 }
234
235 w->spill.base = w->staged_cursor;
236 w->spill.high_water = w->staged_cursor;
237 w->spill.max = max;
238 w->mode = BRIX_VFS_WRITER_SPILL;
239 brix_metric_vfs_spill_active(1);
240 ngx_log_error(NGX_LOG_INFO, w->log, 0,
241 "brix: vfs writer: entering spill mode (write at %O, %O bytes "
242 "already staged) — scratch \"%s\"",
243 off, w->staged_cursor, w->spill.path);
244 return NGX_OK;
245}

247ngx_int_t
248brix_vfs_writer_spill_put(brix_vfs_writer_t *w, const void *buf, size_t len,
249 off_t off)
250{
251 off_t end = off + (off_t) len;
252
253 if (off < w->spill.base) {
254 ngx_log_error(NGX_LOG_ERR, w->log, 0,
255 "brix: vfs writer: spill write at %O rewinds below the %O bytes "
256 "already staged", off, w->spill.base);
257 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
258 errno = EINVAL;
259 return NGX_ERROR;
260 }
261 if (w->spill.max > 0 && end - w->spill.base > w->spill.max) {
262 return spill_fail(w, ENOSPC, "brix_vfs_spill_max exceeded");
263 }
264 if (spill_ext_add(w, off, end) != NGX_OK) {
265 if (errno == EINVAL) {
266 ngx_log_error(NGX_LOG_ERR, w->log, 0,
267 "brix: vfs writer: spill extent [%O,%O) overlaps bytes "
268 "already written — refused", off, end);
269 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
270 return NGX_ERROR; /* stays SPILL; not capacity */
271 }
272 return spill_fail(w, errno, "spill extent set exhausted");
273 }
274 if (brix_vfs_pwrite_full(w->spill.fd, buf, len,
275 off - w->spill.base) != NGX_OK)
276 {
277 return spill_fail(w, errno, "spill scratch write");
278 }
279 if (end > w->spill.high_water) {
280 w->spill.high_water = end;
281 }
282 w->spill.written += (off_t) len;
283 brix_metric_vfs_spill_bytes(brix_vfs_metrics_proto(w->ctx), len);
284 return NGX_OK;
285}

287ngx_int_t
288brix_vfs_writer_spill_drain(brix_vfs_writer_t *w)
289{
290 off_t size = w->spill.high_water - w->spill.base;
291 off_t done = 0;
292 u_char *chunk;
293
294 if (size == 0) {
295 return NGX_OK;
296 }
297 if (w->spill.n_ext != 1
298 || w->spill.ext[0].start != w->spill.base
299 || w->spill.ext[0].end != w->spill.high_water)
300 {
301 ngx_log_error(NGX_LOG_ERR, w->log, 0,
302 "brix: vfs writer: spill covers %O of %O bytes in %ui runs — the "
303 "client left holes; refusing to publish bytes it never sent",
304 w->spill.written, size, w->spill.n_ext);
305 brix_metric_vfs_spill_refused(brix_vfs_metrics_proto(w->ctx));
306 errno = EINVAL;
307 return NGX_ERROR;
308 }
309
310 chunk = ngx_palloc(w->pool, BRIX_VFS_SPILL_DRAIN_CHUNK);
311 if (chunk == NULL) {
312 errno = ENOMEM;
313 return NGX_ERROR;
314 }
315 while (done < size) {
316 size_t want = (size - done) < BRIX_VFS_SPILL_DRAIN_CHUNK
317 ? (size_t) (size - done) : BRIX_VFS_SPILL_DRAIN_CHUNK;
318 size_t got = 0;
319
320 if (brix_vfs_pread_full(w->spill.fd, chunk, want, done, &got) != NGX_OK
321 || got != want)
322 {
323 if (errno == 0) {
324 errno = EIO; /* scratch shorter than its own extents */
325 }
326 return NGX_ERROR;
327 }
328 if (brix_vfs_staged_write(w->st, chunk, want,
329 w->spill.base + done) != NGX_OK)
330 {
331 return NGX_ERROR;
332 }
333 done += (off_t) want;
334 }
335 return NGX_OK;
336}
