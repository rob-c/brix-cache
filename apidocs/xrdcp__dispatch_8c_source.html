1/*
2 * xrdcp_dispatch.c — xrdcp transfer-mode dispatch (Phase-38 split).
3 *
4 * WHAT: route one finalized job to its transfer path — recursive web
5 * up/download, a single copy, or a journalled batch — plus the failed-
6 * source hint and per-mode progress wiring.
7 * WHY: split from xrdcp.c to hold each TU within the Phase-38 size budget;
8 * "execute the transfer(s)" is one cohesive concern with a single public
9 * entry point (dispatch_transfer), distinct from parsing and credential
10 * build. Byte-frozen: same routing, same exit codes.
11 * HOW: dispatch_transfer fills an xrdcp_transfer_ctx and selects the mode; the
12 * actual copy/relay/recursive primitives live in xrdcp_transfer.c /
13 * xrdcp_recursive.c and the shared URL/path helpers in xrdcp.c, all
14 * reached through xrdcp_internal.h. No goto; early-return throughout.
15 */
16#include "xrdcp_internal.h"
17#include "core/version.h"
18#include "core/progname.h" /* brix_prog_base(): argv[0]-derived identity + footer */

62static int
63xrdcp_dispatch_recursive_web_download(const xrdcp_transfer_ctx *ctx)
64{
65 brix_status st;
66 size_t i;
67 size_t bad = 0;
68
69 brix_status_clear(&st);
70 for (i = 0; i < ctx->nexp; i++) {
71 if (brix_is_web_url(ctx->exp[i])) {
72 if (recursive_web_download(ctx->exp[i], ctx->dst, ctx->opts,
73 ctx->conn, ctx->retries) != 0) {
74 bad++;
75 }
76 } else if (copy_one_with_retry(ctx->exp[i], ctx->dst, ctx->opts,
77 ctx->conn, ctx->retries, &st) != 0) {
78 bad++;
79 fprintf(stderr, "xrdcp: %s: %s\n", ctx->exp[i], st.msg);
80 xrdcp_hint_failed_source(ctx->exp[i], ctx->dst, &st);
81 }
82 }
83 return (bad == 0) ? 0 : 1;
84}

115static int
116xrdcp_try_recursive_web(const xrdcp_transfer_ctx *ctx, int *handled)
117{
118 size_t i;
119 int any_web = 0;
120 int has_dir = 0;
121
122 *handled = 0;
123 if (!ctx->opts->recursive) {
124 return 0;
125 }
126 for (i = 0; i < ctx->nexp; i++) {
127 if (brix_is_web_url(ctx->exp[i])) { any_web = 1; }
128 if (is_local_dir(ctx->exp[i])) { has_dir = 1; }
129 }
130 if (any_web) {
131 *handled = 1;
132 return xrdcp_dispatch_recursive_web_download(ctx);
133 }
134 if (brix_is_web_url(ctx->dst) && has_dir) {
135 *handled = 1;
136 return xrdcp_dispatch_recursive_web_upload(ctx);
137 }
138 return 0;
139}

161static int
162xrdcp_dispatch_single(xrdcp_transfer_ctx *ctx)
163{
164 brix_status st;
165 char label[XRDC_NAME_MAX];
166 xrdcp_prog ps;
167 int one;
168
169 brix_status_clear(&st);
170 xrdcp_enable_progress_if_needed(ctx, &ps, label, sizeof(label));
171 one = transfer_one(ctx->exp[0], ctx->dst, ctx->opts, ctx->conn,
172 ctx->retries, ctx->sync_mode, &st);
173 if (one == 1 && !ctx->opts->silent) {
174 fprintf(stderr, "xrdcp: %s up-to-date, skipped\n", ctx->dst);
175 } else if (one < 0 && !ctx->opts->silent) {
176 fprintf(stderr, "xrdcp: %s\n", st.msg);
177 xrdcp_hint_failed_source(ctx->exp[0], ctx->dst, &st);
178 }
179 return (one >= 0) ? 0 : brix_shellcode(&st);
180}

203static void
204xrdcp_batch_one(const xrdcp_transfer_ctx *ctx, brix_journal *jrn,
205 xrdcp_tally_t *t, size_t idx)
206{
207 char dpath[XRDC_PATH_MAX];
208 int one;
209
210 if (jrn != NULL && brix_journal_has(jrn, ctx->exp[idx])) {
211 t->skip++;
212 if (!ctx->opts->silent) {
213 fprintf(stderr, "[%zu/%zu] %s (already transferred)\n",
214 t->ok + t->skip + t->fail, ctx->nexp, ctx->exp[idx]);
215 }
216 return;
217 }
218 one = batch_copy_one(ctx->exp[idx], ctx->dst, ctx->opts, ctx->conn,
219 ctx->retries, ctx->sync_mode, dpath, sizeof(dpath), &t->st);
220 if (one == 0) {
221 t->ok++;
222 if (!ctx->opts->silent) {
223 fprintf(stderr, "[%zu/%zu] %s -> %s\n",
224 t->ok + t->skip + t->fail, ctx->nexp, ctx->exp[idx], dpath);
225 }
226 if (jrn != NULL && !ctx->opts->dry_run) {
227 (void) brix_journal_mark(jrn, ctx->exp[idx]);
228 }
229 } else if (one == 1) {
230 t->skip++;
231 if (!ctx->opts->silent) {
232 fprintf(stderr, "[%zu/%zu] %s (up-to-date)\n",
233 t->ok + t->skip + t->fail, ctx->nexp, ctx->exp[idx]);
234 }
235 } else {
236 t->fail++;
237 fprintf(stderr, "xrdcp: %s: %s\n", ctx->exp[idx], t->st.msg);
238 xrdcp_hint_failed_source(ctx->exp[idx], ctx->dst, &t->st);
239 }
240}

243static int
244xrdcp_dispatch_batch(xrdcp_transfer_ctx *ctx)
245{
246 xrdcp_tally_t t;
247 brix_journal *jrn;
248 size_t i;
249 int rc;
250 int jobs = ctx->jobs;
251
252 memset(&t, 0, sizeof(t));
253 brix_status_clear(&t.st);
254 if (dest_is_dir(ctx->dst, ctx->conn) != 1) {
255 fprintf(stderr, "xrdcp: destination must be an existing directory for "
256 "multi-source copy: %s\n", ctx->dst);
257 return 50;
258 }
259 jrn = xrdcp_open_batch_journal(ctx, &t.st, &rc);
260 if (rc != 0) {
261 return rc;
262 }
263 if (jobs > (int) ctx->nexp) { jobs = (int) ctx->nexp; }
264 if (jobs > 1) {
265 batch_parallel(ctx->exp, ctx->nexp, ctx->dst, ctx->opts, ctx->conn,
266 ctx->retries, ctx->sync_mode, jobs, jrn,
267 &t.ok, &t.skip, &t.fail);
268 } else {
269 for (i = 0; i < ctx->nexp; i++) {
270 xrdcp_batch_one(ctx, jrn, &t, i);
271 }
272 }
273 if (!ctx->opts->silent) {
274 fprintf(stderr, "xrdcp: %zu copied, %zu skipped, %zu failed\n",
275 t.ok, t.skip, t.fail);
276 }
277 return (t.fail == 0) ? 0 : 1;
278}

281/*
282 * WHAT: Dispatch transfer based on mode (web recursive, single, or batch).
283 *
284 * WHY: Main() had CCN 187 with complex mode routing. Extracting the dispatch
285 * logic (web recursive download/upload, single transfer with progress,
286 * batch sequential/parallel, journal lifecycle) reduces complexity.
287 *
288 * HOW: Determine transfer mode from source types and flags. Route to:
289 * - Web recursive download/upload (davs/http collections)
290 * - Single transfer with progress bar (nexp==1)
291 * - Batch transfer with journal support (nexp>1)
292 * Returns 0 on success, 1 on partial failure, 50 on usage error, 51 on OOM.
293 */
294int
295dispatch_transfer(xrdcp_opts_t *o, xrdcp_lists_t *l)
296{
297 xrdcp_transfer_ctx ctx;
298 int handled;
299 int rc;
300
301 ctx.opts = o->copt;
302 ctx.conn = o->conn;
303 ctx.cred_store = o->cred_store;
304 ctx.exp = l->exp.items;
305 ctx.nexp = l->exp.n;
306 ctx.dst = o->dst;
307 ctx.from = o->from;
308 ctx.journal_path = o->journal_path;
309 ctx.retries = o->retries;
310 ctx.jobs = o->jobs;
311 ctx.sync_mode = o->sync_mode;
312 ctx.force_progress = o->force_progress;
313 ctx.no_progress = o->no_progress;
314
315 rc = xrdcp_try_recursive_web(&ctx, &handled);
316 if (handled) {
317 return rc;
318 }
319 if (l->exp.n == 1 && o->from == NULL) {
320 return xrdcp_dispatch_single(&ctx);
321 }
322
323 return xrdcp_dispatch_batch(&ctx);
324}
