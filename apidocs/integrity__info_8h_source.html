24/*
25 * brix_integrity_opts_t — caller-supplied policy for integrity lookup.
26 *
27 * allow_xattr_cache 1 → try reading a cached checksum from xattr first
28 * update_xattr_cache 1 → write computed checksum back to xattr on cache miss
29 * require_regular_file 1 → fail with NGX_DECLINED if fd is not a regular file
30 * no_compute 1 → cache-only: return NGX_DECLINED on a cache miss
31 * instead of computing (avoids a full-file read on a
32 * latency-sensitive path, e.g. S3 GET/HEAD echo)
33 * mutation_policy the ENDPOINT's phase-105 write posture. Persisting a
34 * computed checksum writes an xattr (or a §8.2 record)
35 * ONTO THE EXPORT OBJECT, so it is a mutation like any
36 * other and is refused on a read-only export — the
37 * request still gets its checksum, it is just recomputed
38 * next time. This unit has no request context (it is
39 * reached from thread-pool workers and the scanner), so
40 * the caller carries the policy here as a value.
41 * ZERO IS READ_ONLY: a caller that forgets it loses the
42 * cache write, never the other way round.
43 * proto protocol label for the refusal metric only.
44 */
45typedef struct {
46 ngx_flag_t allow_xattr_cache;
47 ngx_flag_t update_xattr_cache;
48 ngx_flag_t require_regular_file;
49 ngx_flag_t no_compute;

52} brix_integrity_opts_t;

54/*
55 * brix_integrity_get_fd — retrieve a checksum for an open file descriptor.
56 *
57 * If opts->allow_xattr_cache is set, tries reading a cached value from a
58 * "user.XrdCks.<alg>" extended attribute before computing. On a cache miss,
59 * computes the checksum and, when opts->update_xattr_cache is also set, writes
60 * the result back to the xattr for future lookups.
61 *
62 * When opts->require_regular_file is set the function calls fstat(2) on fd and
63 * returns NGX_DECLINED immediately if the file is not a regular file.
64 *
65 * Returns NGX_OK and fills *out on success.
66 * Returns NGX_DECLINED when the file is not regular and require_regular_file=1.
67 * Returns NGX_ERROR on algorithm parse failure or I/O error.
68 *
69 * opts may be NULL; NULL is treated as {allow_xattr_cache=1, update_xattr_cache=1,
70 * require_regular_file=0, mutation_policy=READ_ONLY} — a caller that supplies no
71 * policy gets the cache READ but not the cache WRITE (phase-105 fail-closed).
72 *
73 * Layer 3: `obj` is the open file's storage-driver object, or NULL. When non-NULL
74 * and obj->driver != NULL the checksum is COMPUTED by reading the whole logical
75 * object through the driver (block-striped / object backend) instead of the bare
76 * fd (which exposes only block 0). NULL keeps the default POSIX-fd compute. The
77 * xattr/sidecar cache layer is unchanged (keyed on fd/path).
78 */

84/*
85 * brix_integrity_seed_fd — record an ALREADY-PROVEN digest without reading a byte.
86 *
87 * The producer side of the cache layer brix_integrity_get_fd consults first. A
88 * caller that has just verified `hex` over exactly the bytes behind `fd` (a cache
89 * fill comparing the origin's advertised checksum against the recomputed part) can
90 * hand the value over here instead of leaving the first kXR_Qcksum / Want-Digest
91 * request on the cached copy to re-read the whole file.
92 *
93 * `alg_name` is canonicalised the same way get_fd canonicalises it, so a seed
94 * written as "SHA256" is found by a lookup for "sha256", and one written as
95 * "crc64xz" by a lookup for "crc64". (The same parser also REJECTS a spelling
96 * that is not purely alphanumeric, "sha-256" among them.) `hex` is validated as
97 * non-empty pure hex and lowercased before it is stored: this API records an
98 * assertion about file content, so a malformed or over-long value is REFUSED
99 * (NGX_ERROR) rather than truncated into a digest nobody can verify. `path` is the
100 * file's local path for the §8.2 record fallback on filesystems without user
101 * xattrs, or NULL to attempt the xattr only.
102 *
103 * Best-effort by contract: NGX_OK means the value was accepted and the write was
104 * attempted, never that a particular cache layer holds it. A lost seed costs a
105 * recompute, never a wrong answer.
106 */
107ngx_int_t brix_integrity_seed_fd(int fd, const char *path,
108 const char *alg_name, const char *hex);
