9/*
10 * Per-worker RFC-8693 token-exchange result cache (phase-70 §5.4, P90-70.9).
11 *
12 * An EXCHANGE-mode export mints one backend-audienced token per (subject
13 * token, audience) pair; without a cache every VFS op re-POSTs to the issuer.
14 * This is the direct-mapped, lock-free, event-loop-only cache the exchange
15 * integration layer owns (exchange.c deliberately keeps none): the key is
16 * SHA-256(subject-token ‖ audience) so a different client token — even for
17 * the same user — can NEVER be served another subject's minted token, and the
18 * entry TTL is min(minted `exp`, store-time + BRIX_TX_CACHE_MAX_TTL_SECS) so
19 * revocation at the issuer is picked up within the clamp window.
20 *
21 * `now` is injected by the caller (time(NULL) in production) so expiry is
22 * deterministic under test. Lookup returns a BORROWED view of the cached
23 * bytes — copy before the next store could evict the slot (in the
24 * single-threaded event loop: before returning to it).
25 */
26
27typedef struct brix_tx_cache_s brix_tx_cache_t;
