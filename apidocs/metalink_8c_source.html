1/*
2 * metalink.c — pure metalink v4 (RFC 5854) / v3 document parser (phase-100).
3 *
4 * WHAT: brix_metalink_is_name() suffix detection and brix_metalink_parse(),
5 * which turns one metalink XML document into a ranked mirror list plus
6 * the optional size and strongest client-supported digest.
7 * WHY: Metalink is the client's virtual-redirector input: copy_metalink.c
8 * fails over across the ranked mirrors and hands them to the extreme-copy
9 * engine. Keeping the parser pure (bytes in, struct out, no I/O and no
10 * allocation) makes hostile-input behavior unit-testable in isolation
11 * (client/tests/c/metalink_unit.c).
12 * HOW: A single bounded forward scan over the lexing primitives in
13 * metalink_lex.c (see metalink_internal.h) — no libxml dependency, no
14 * recursion. Only the FIRST <file> element is resolved (a copy resolves
15 * one logical file, matching XrdCl's virtual redirector). Sibling
16 * copy_metalink.c owns fetching + failover.

25/* ---- Is this mirror URL a scheme the copy engine may pull from? ----
26 *
27 * WHAT: 1 for the remote pull schemes (root/roots/xroot/xroots + http/https/
28 * dav/davs, case-insensitive), 0 for everything else.
29 *
30 * WHY: SECURITY — a hostile remote metalink must not steer the copy at a local
31 * file (file:///etc/… exfiltrates via an upload destination) or at a
32 * credentialed transport the user never selected (s3 SigV4). Skipped
33 * mirrors are counted, never fatal, so a mixed document still works.
34 *
35 * HOW: Prefix table over the accepted scheme spellings.
36 */
37static int
38ml_scheme_allowed(const char *mirror_url)
39{
40 static const char *const allowed[] = {
41 "root://", "roots://", "xroot://", "xroots://",
42 "http://", "https://", "dav://", "davs://",
43 };
44 size_t i;
45
46 for (i = 0; i < sizeof(allowed) / sizeof(allowed[0]); i++) {
47 if (strncasecmp(mirror_url, allowed[i], strlen(allowed[i])) == 0) {
48 return 1;
49 }
50 }
51 return 0;
52}

55/* ---- Insert one mirror into the ranked list (stable, bounded) ----
56 *
57 * WHAT: Insertion-sort `url` (rank ascending, ties keep arrival order) into
58 * ml->urls. When full, a better-than-worst rank evicts the worst entry
59 * (counted in n_skipped); otherwise the newcomer is skipped.
60 *
61 * WHY: Consumers walk urls[] best-first; sorting on insert keeps the parser
62 * single-pass and the eviction rule keeps a 1000-mirror hostile document
63 * bounded without dropping a best-ranked mirror listed last.
64 *
65 * HOW: 1. Full + not-better → skip. 2. Full + better → drop the tail. 3. Shift
66 * strictly-worse entries up one and place the newcomer.
67 */
68static void
69ml_insert_url(brix_metalink *ml, const char *url, int rank)
70{
71 size_t pos, k;
72
73 if (ml->n_urls == XRDC_METALINK_MAX_URLS) {
74 if (rank >= ml->urls[ml->n_urls - 1].rank) {
75 ml->n_skipped++;
76 return;
77 }
78 ml->n_urls--; /* evict the current worst */
79 ml->n_skipped++;
80 }
81 for (pos = 0; pos < ml->n_urls; pos++) {
82 if (ml->urls[pos].rank > rank) {
83 break;
84 }
85 }
86 for (k = ml->n_urls; k > pos; k--) {
87 ml->urls[k] = ml->urls[k - 1];
88 }
89 snprintf(ml->urls[pos].rank_url, sizeof(ml->urls[pos].rank_url), "%s", url);
90 ml->urls[pos].rank = rank;
91 ml->n_urls++;
92}

95/* ---- Fold one <hash type="...">hex</hash> into the strongest supported ----
96 *
97 * WHAT: If `algo`/`hex` name a digest the --cksum machinery can verify
98 * (md5 > crc32c > adler32) and beat the currently-held one, record them
99 * (hex validated and lowercased). Unsupported/malformed digests are
100 * ignored.
101 *
102 * WHY: The metalink digest rides into the transfer as "--cksum algo:hex"
103 * (literal compare); a non-hex "digest" from a hostile document must not
104 * reach that path.
105 *
106 * HOW: 1. Strength table lookup (0 = unsupported). 2. Validate 8..128 hex
107 * chars. 3. Keep only a strict upgrade.
108 */
109static void
110ml_fold_hash(brix_metalink *ml, const char *algo, const char *hex)
111{
112 static const struct { const char *name; int strength; } table[] = {
113 { "md5", 3 }, { "crc32c", 2 }, { "adler32", 1 },
114 };
115 int strength = 0, held = 0;
116 size_t i, hlen = strlen(hex);
117
118 for (i = 0; i < sizeof(table) / sizeof(table[0]); i++) {
119 if (strcasecmp(algo, table[i].name) == 0) {
120 strength = table[i].strength;
121 }
122 if (strcasecmp(ml->hash_algo, table[i].name) == 0) {
123 held = table[i].strength;
124 }
125 }
126 if (strength == 0 || strength <= held) {
127 return;
128 }
129 if (hlen < 8 || hlen >= XRDC_METALINK_HEX_MAX) {
130 return;
131 }
132 for (i = 0; i < hlen; i++) {
133 if (!isxdigit((unsigned char) hex[i])) {
134 return;
135 }
136 }
137 snprintf(ml->hash_algo, sizeof(ml->hash_algo), "%s", algo);
138 for (i = 0; i < hlen; i++) {
139 ml->hash_hex[i] = (char) tolower((unsigned char) hex[i]);
140 }
141 ml->hash_hex[hlen] = '\0';
142}

145/* ---- Rank one <url> element from its attributes ----
146 *
147 * WHAT: Map v4 priority (ascending, 1 = best) and v3 preference (descending,
148 * 100 = best) onto one ascending rank; absent both → 500 (after every
149 * explicitly-ranked mirror, before nothing).
150 *
151 * WHY: One comparable rank lets v3 and v4 documents share the sorted list.
152 *
153 * HOW: priority wins when both appear; values are clamped to [1,100] before
154 * mapping (v3: rank = 101 - preference).
155 */
156static int
157ml_rank_of(const char *attrs, const char *gt)
158{
159 char val[16];
160
161 if (ml_attr(attrs, gt, "priority", val, sizeof(val)) == 0) {
162 int prio = atoi(val);
163 if (prio < 1) { prio = 1; }
164 if (prio > 100) { prio = 100; }
165 return prio;
166 }
167 if (ml_attr(attrs, gt, "preference", val, sizeof(val)) == 0) {
168 int pref = atoi(val);
169 if (pref < 1) { pref = 1; }
170 if (pref > 100) { pref = 100; }
171 return 101 - pref;
172 }
173 return 500;
174}

177/* ---- Earlier of two scan candidates ----
178 *
179 * WHAT: Returns whichever of `a`/`b` points earlier into the scope, or the
180 * non-NULL one, or NULL when neither matched.
181 *
182 * WHY: The scan must stay single-pass, so each iteration has to consume the
183 * earliest element regardless of kind. Folding the NULL cases in here
184 * is what lets the caller pick a winner with one expression instead of
185 * the nested "is it before the others" ladder this replaces.
186 *
187 * HOW: NULL means "this kind does not occur again", so it always loses.
188 */
189static const char *
190ml_earliest(const char *a, const char *b)
191{
192 if (a == NULL) {
193 return b;
194 }
195 if (b == NULL) {
196 return a;
197 }
198
199 return (a < b) ? a : b;
200}

203/* ---- Fold one <size> element into the parse ----
204 *
205 * WHAT: Sets ml->size from the element at `hit` when it is the first valid one
206 * seen; malformed or repeated elements are ignored.
207 *
208 * WHY: First valid wins: a metalink that declares the size twice is
209 * inconsistent, and honouring the later value would let a trailing
210 * element silently resize a transfer already sized from the first.
211 *
212 * HOW: 1. Re-open the tag to get its '>' and read the element text.
213 * 2. Ignore it if a size is already set or the text is negative.
214 */
215static void
216ml_take_size(brix_metalink *ml, const char *hit, const char *fend)
217{
218 const char *sgt;
219 char text[XRDC_METALINK_URL_MAX];
220 long long parsed;
221
222 if (ml_tag_open(hit, fend, "size", NULL, &sgt) == NULL
223 || ml_elem_text(sgt, fend, "size", text, sizeof(text)) != 0
224 || ml->size >= 0)
225 {
226 return;
227 }
228
229 parsed = atoll(text);
230 if (parsed >= 0) {
231 ml->size = (int64_t) parsed;
232 }
233}

236/* ---- Fold one <hash> element into the parse ----
237 *
238 * WHAT: Offers the element's type + hex digest to ml_fold_hash, which keeps
239 * the strongest supported algorithm.
240 *
241 * WHY: v3's <verification> wrapper needs no special casing — its <hash>
242 * children match the same scan, so the wrapper is simply scanned through.
243 *
244 * HOW: 1. Re-open the tag for its attribute window and '>'.
245 * 2. Require both a type attribute and element text; a hash missing
246 * either is unusable, not a weaker hash.
247 */
248static void
249ml_take_hash(brix_metalink *ml, const char *hit, const char *fend)
250{
251 const char *hattrs, *hgt;
252 char algo[XRDC_METALINK_ALGO_MAX];
253 char hex[XRDC_METALINK_HEX_MAX];
254
255 if (ml_tag_open(hit, fend, "hash", &hattrs, &hgt) != NULL
256 && ml_attr(hattrs, hgt, "type", algo, sizeof(algo)) == 0
257 && ml_elem_text(hgt, fend, "hash", hex, sizeof(hex)) == 0)
258 {
259 ml_fold_hash(ml, algo, hex);
260 }
261}

264/* ---- Fold one <url ...>text</url> element into the parse ----
265 *
266 * WHAT: Inserts the mirror at its computed rank, or counts it as skipped.
267 *
268 * WHY: A rejected mirror is counted rather than dropped silently, so a
269 * metalink whose mirrors are all unusable is distinguishable from one
270 * that listed none. The scheme gate is a security boundary: local and
271 * credentialed URLs must never reach the transport (§ml_scheme_allowed).
272 *
273 * HOW: 1. Read the element text; unterminated/oversized/empty is a skip.
274 * 2. Reject a disallowed scheme.
275 * 3. Otherwise insert at the priority/preference-derived rank.
276 */
277static void
278ml_take_url(brix_metalink *ml, const char *attrs, const char *gt,
279 const char *fend)
280{
281 char text[XRDC_METALINK_URL_MAX];
282
283 if (ml_elem_text(gt, fend, "url", text, sizeof(text)) != 0
284 || text[0] == '\0') {
285 ml->n_skipped++; /* unterminated/oversized/empty URL */
286 } else if (!ml_scheme_allowed(text)) {
287 ml->n_skipped++; /* local/credentialed/unknown scheme */
288 } else {
289 ml_insert_url(ml, text, ml_rank_of(attrs, gt));
290 }
291}

294/* ---- Collect every <url>, <size> and <hash> inside the <file> scope ----
295 *
296 * WHAT: Walk [p,fend) once, folding each element kind into *ml.
297 *
298 * WHY: Keeps brix_metalink_parse a flat sequence: locate scope, collect,
299 * validate. All three element kinds share the open-tag/text machinery.
300 *
301 * HOW: 1. Find the next '<'-led candidate for each kind from the cursor;
302 * advance by the earliest hit so the scan stays single-pass. 2. <url>:
303 * rank + scheme-gate + insert. 3. <size>: first valid wins. 4. <hash>:
304 * strongest supported wins (v3's <verification> wrapper needs no
305 * special casing — its <hash> children match the same scan).
306 */
307static void
308ml_collect_file_scope(brix_metalink *ml, const char *p, const char *fend)
309{
310 for (;;) {
311 const char *attrs = NULL, *gt = NULL;
312 const char *uhit = ml_tag_open(p, fend, "url", &attrs, &gt);
313 const char *shit = ml_tag_open(p, fend, "size", NULL, NULL);
314 const char *hhit = ml_tag_open(p, fend, "hash", NULL, NULL);
315
316 /* earliest of the three candidates drives this iteration */
317 const char *first = ml_earliest(ml_earliest(uhit, shit), hhit);
318
319 if (first == NULL) {
320 return; /* no candidates left in scope */
321 }
322
323 if (first == shit) {
324 ml_take_size(ml, shit, fend);
325 p = shit + 1;
326
327 } else if (first == hhit) {
328 ml_take_hash(ml, hhit, fend);
329 p = hhit + 1;
330
331 } else {
332 ml_take_url(ml, attrs, gt, fend);
333 p = gt + 1;
334 }
335 }
336}

339int
340brix_metalink_is_name(const char *s)
341{
342 const char *q, *path_end, *dot;
343 size_t plen;
344
345 if (s == NULL) {
346 return 0;
347 }
348 q = strchr(s, '?');
349 path_end = (q != NULL) ? q : s + strlen(s);
350 for (dot = path_end; dot > s && dot[-1] != '.' && dot[-1] != '/'; dot--) { }
351 if (dot == s || dot[-1] != '.') {
352 return 0;
353 }
354 plen = (size_t) (path_end - dot);
355 return (plen == 5 && strncasecmp(dot, "meta4", 5) == 0)
356 || (plen == 8 && strncasecmp(dot, "metalink", 8) == 0);
357}

360/*
361 * WHAT: Parse one metalink v4/v3 document into *out — ranked mirrors, optional
362 * size, strongest supported digest. 0 with >=1 usable mirror, else -1
363 * (st set, XRDC_EPROTO class).
364 *
365 * WHY: Single entry point for both dialects keeps copy_metalink.c free of any
366 * XML knowledge and the hostile-input surface behind one tested door.
367 *
368 * HOW: 1. Bound the document (empty / >4 MiB refused). 2. Require a <metalink
369 * root tag. 3. Scope to the FIRST <file> element (self-closing or
370 * missing → error; an unterminated scope clamps to end-of-document,
371 * lenient). 4. Collect url/size/hash. 5. Zero usable mirrors → error
372 * naming how many candidates were skipped.
373 */
374int
375brix_metalink_parse(const char *xml, size_t len, brix_metalink *out,
376 brix_status *st)
377{
378 const char *end, *fgt = NULL, *fscope, *fend;
379
380 memset(out, 0, sizeof(*out));
381 out->size = -1;
382
383 if (xml == NULL || len == 0 || len > XRDC_METALINK_MAX_BYTES) {
384 brix_status_set(st, XRDC_EPROTO, 0,
385 "metalink: empty or oversized document (%zu bytes, cap %u)",
386 len, XRDC_METALINK_MAX_BYTES);
387 return -1;
388 }
389 end = xml + len;
390
391 if (ml_tag_open(xml, end, "metalink", NULL, NULL) == NULL) {
392 brix_status_set(st, XRDC_EPROTO, 0, "not a metalink document");
393 return -1;
394 }
395 if (ml_tag_open(xml, end, "file", NULL, &fgt) == NULL || fgt == NULL
396 || fgt[-1] == '/') {
397 brix_status_set(st, XRDC_EPROTO, 0, "metalink: no <file> entry");
398 return -1;
399 }
400 fscope = fgt + 1;
401 fend = ml_find_ci(fscope, end, "</file");
402 if (fend == NULL) {
403 fend = end; /* unterminated scope: clamp, stay lenient */
404 }
405
406 ml_collect_file_scope(out, fscope, fend);
407
408 if (out->n_urls == 0) {
409 brix_status_set(st, XRDC_EPROTO, 0,
410 "metalink: no usable mirror URLs "
411 "(%zu candidate(s) skipped: bad scheme/length)",
412 out->n_skipped);
413 return -1;
414 }
415 return 0;
416}
