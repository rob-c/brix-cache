1/* scrub.c — phase-87 G17: background CAS integrity scrubbing (self-healing).
2 *
3 * WHAT: a repeating worker-0 timer walks the cvmfs cache store in bounded
4 * windows and re-runs the cvmfs-cas verify (name-hash == byte-hash) on
5 * resident CAS objects; a mismatch is evicted so the next access
6 * re-fills verified from the origin.
7 * WHY: the fill-time verify proves the bytes ONCE; disk bitrot after commit
8 * would then be served forever (the CAS client rejects it, but every
9 * reader pays the corrupt transfer and the object never heals). The
10 * scrub is the F1 verify path run proactively, off the hot path.
11 * HOW: registration mirrors the T19 RTT probe (config-time per-process
12 * statics, per-worker timer at init_process — worker 0 only, the store
13 * is shared and duplicate hashing is pure waste). Each pass the event
14 * loop collects one cursor-bounded window of CAS keys via
15 * brix_cstore_scan, a thread-pool task hashes them (the expensive
16 * part), and the completion evicts mismatches on the event loop — the
17 * same loop the watermark reaper mutates the store from.
18 *
19 * A mismatch here is LOCAL corruption (this host's disk): the origin
20 * proved these bytes at fill time. It is logged as such and NEVER
21 * raises signal=cvmfs_tamper — that signal names a lying origin and
22 * feeds an instant-ban jail (the cold_tier lesson). If the origin's
23 * copy is ALSO bad, the re-fill's own verify gate rejects, quarantines
24 * and raises the tamper signal — the scrub needs no second gate.
25 */

55void
56brix_cvmfs_scrub_register(const char *root_canon, time_t interval,
57 ngx_uint_t rate, const ngx_str_t *pool_name)
58{
59 ngx_uint_t i;
60 cvmfs_scrub_reg_t *reg = NULL;
61
62 for (i = 0; i < cvmfs_scrub_regs_n; i++) {
63 if (ngx_strcmp(cvmfs_scrub_regs[i].root, root_canon) == 0) {
64 reg = &cvmfs_scrub_regs[i]; /* reload: update in place */
65 break;
66 }
67 }
68 if (reg == NULL) {
69 if (cvmfs_scrub_regs_n >= CVMFS_SCRUB_MAX_EXPORTS
70 || ngx_strlen(root_canon) >= sizeof(reg->root))
71 {
72 return;
73 }
74 reg = &cvmfs_scrub_regs[cvmfs_scrub_regs_n++];
75 }
76 ngx_cpystrn((u_char *) reg->root, (u_char *) root_canon,
77 sizeof(reg->root));
78 reg->interval = (interval > 0) ? interval : 60;
79 reg->rate = (rate == 0) ? CVMFS_SCRUB_RATE_DEF
80 : (rate > CVMFS_SCRUB_RATE_MAX) ? CVMFS_SCRUB_RATE_MAX : rate;
81 reg->pool[0] = '\0';
82 if (pool_name != NULL && pool_name->len > 0
83 && pool_name->len < sizeof(reg->pool))
84 {
85 ngx_memcpy(reg->pool, pool_name->data, pool_name->len);
86 reg->pool[pool_name->len] = '\0';
87 }
88}

96/* Per-worker scrub context (one per export; the task is allocated once and
97 * re-posted — timer → collect → post → done → re-arm is strictly sequential). */
98typedef struct {
99 ngx_event_t timer;
100 ngx_thread_task_t *task;
101 const cvmfs_scrub_reg_t *reg;
102 brix_cstore_t *cs; /* resolved on each fire */
103 char root_abs[PATH_MAX]; /* store local root, trimmed */
104 size_t root_len;
105 ngx_uint_t cursor; /* CAS objects to skip this pass */
106 ngx_uint_t seen_cas; /* CAS objects seen by the walk */
107 ngx_uint_t n; /* window entries collected */
108 unsigned wrapped:1; /* walk reached end-of-store */
109 cvmfs_scrub_item_t *items; /* reg->rate entries */
110} cvmfs_scrub_ctx_t;

112/* Scan visitor (event loop): collect the pass's window of CAS keys.
113 * Non-CAS entries (manifests, whitelists, /.gcas canonical names) are
114 * skipped — they are either mutable or reachable through a CAS key. */
115static ngx_int_t
116cvmfs_scrub_collect(const char *key, const brix_cache_cinfo_t *ci,
117 const brix_sd_stat_t *stx, void *ctx)
118{
119 cvmfs_scrub_ctx_t *sc = ctx;
120 cvmfs_url_info_t info;
121
122 (void) ci;
123 (void) stx;
124
125 if (cvmfs_classify_url(key, strlen(key), &info) != 0
126 || info.cls != CVMFS_URL_CAS)
127 {
128 return NGX_OK;
129 }
130 if (sc->seen_cas++ < sc->cursor) {
131 return NGX_OK; /* before this pass's window */
132 }
133 if (sc->n >= sc->reg->rate) {
134 return NGX_DONE; /* window full — stop the walk */
135 }
136 if (strlen(key) < CVMFS_SCRUB_KEY_MAX) {
137 ngx_cpystrn((u_char *) sc->items[sc->n].key, (u_char *) key,
138 CVMFS_SCRUB_KEY_MAX);
139 sc->items[sc->n].verdict = BRIX_CACHE_VERIFY_UNVERIFIED;
140 sc->n++;
141 }
142 return NGX_OK;
143}

145/* Thread-pool side: hash every window entry against its CAS name. The log is
146 * deliberately NOT passed down — the fill-path mismatch wording blames the
147 * origin transfer, which is the wrong actor here; the completion handler
148 * emits the accurate local-corruption line instead. */
149static void
150cvmfs_scrub_thread(void *data, ngx_log_t *log)
151{
152 cvmfs_scrub_ctx_t *sc = data;
153 char path[PATH_MAX];
154 ngx_uint_t i;
155 int n;
156
157 (void) log;
158 for (i = 0; i < sc->n; i++) {
159 n = snprintf(path, sizeof(path), "%.*s%s",
160 (int) sc->root_len, sc->root_abs, sc->items[i].key);
161 if (n < 0 || (size_t) n >= sizeof(path)) {
162 sc->items[i].verdict = BRIX_CACHE_VERIFY_ERROR;
163 continue;
164 }
165 sc->items[i].verdict =
166 brix_cache_verify_cvmfs_cas(path, sc->items[i].key, NULL,
167 NULL, NULL);
168 }
169}

171/* Event-loop side: evict mismatches, advance the cursor, re-arm. An ERROR
172 * verdict (object evicted/raced away mid-pass) is a silent skip. */
173static void
174cvmfs_scrub_done(ngx_event_t *ev)
175{
176 ngx_thread_task_t *task = ev->data;
177 cvmfs_scrub_ctx_t *sc = task->ctx;
178 ngx_uint_t i, corrupt = 0;
179
180 for (i = 0; i < sc->n; i++) {
181 if (sc->items[i].verdict != BRIX_CACHE_VERIFY_MISMATCH) {
182 continue;
183 }
184 corrupt++;
185 ngx_log_error(NGX_LOG_WARN, ev->log, 0,
186 "cvmfs scrub: LOCAL corruption in cached object \"%s\" "
187 "(stored bytes no longer match their CAS name — disk bitrot "
188 "or on-host modification, NOT an origin event); evicting so "
189 "the next access re-fills verified from the origin",
190 sc->items[i].key);
191 brix_cstore_evict(sc->cs, sc->items[i].key);
192 }
193
194 ngx_log_error(NGX_LOG_INFO, ev->log, 0,
195 "cvmfs scrub pass (export %s): checked %ui CAS object(s) "
196 "at cursor %ui, corrupt=%ui",
197 sc->reg->root, sc->n, sc->cursor, corrupt);
198
199 sc->cursor = sc->wrapped ? 0 : sc->cursor + sc->n;
200
201 if (!ngx_exiting) {
202 ngx_add_timer(&sc->timer, (ngx_msec_t) sc->reg->interval * 1000
203 + (ngx_msec_t) (ngx_random() % 500));
204 }
205}

207/* Timer side (event loop): resolve the store, collect one window, post the
208 * hashing task. Any miss (non-cache backend, non-local store, empty window,
209 * no pool) just re-arms — the scrub is best-effort by design. */
210static void
211cvmfs_scrub_fire(ngx_event_t *ev)
212{
213 cvmfs_scrub_ctx_t *sc = ev->data;
214 brix_sd_instance_t *inst;
215 ngx_thread_pool_t *pool;
216 ngx_str_t pname;
217 const char *root;
218 ngx_int_t rc;
219
220 inst = brix_vfs_backend_resolve(sc->reg->root, ev->log);
221 sc->cs = (inst != NULL) ? brix_sd_cache_cstore(inst) : NULL;
222 root = (sc->cs != NULL) ? brix_cstore_local_root(sc->cs) : NULL;
223
224 if (root != NULL) {
225 sc->root_len = strlen(root);
226 while (sc->root_len > 0 && root[sc->root_len - 1] == '/') {
227 sc->root_len--;
228 }
229 if (sc->root_len < sizeof(sc->root_abs)) {
230 ngx_memcpy(sc->root_abs, root, sc->root_len);
231 } else {
232 root = NULL;
233 }
234 }
235
236 sc->n = 0;
237 sc->seen_cas = 0;
238 sc->wrapped = 0;
239
240 if (root != NULL) {
241 rc = brix_cstore_scan(sc->cs, cvmfs_scrub_collect, sc);
242 sc->wrapped = (rc == NGX_OK); /* full walk: window hit the end */
243 }
244
245 if (sc->n > 0) {
246 if (sc->reg->pool[0] != '\0') {
247 pname.data = (u_char *) sc->reg->pool;
248 pname.len = ngx_strlen(sc->reg->pool);
249 } else {
250 ngx_str_set(&pname, "default");
251 }
252 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname);
253 if (pool != NULL && ngx_thread_task_post(pool, sc->task) == NGX_OK) {
254 return; /* re-armed by cvmfs_scrub_done */
255 }
256 }
257
258 sc->cursor = 0; /* empty/failed pass: restart */
259 if (!ngx_exiting) {
260 ngx_add_timer(ev, (ngx_msec_t) sc->reg->interval * 1000);
261 }
262}

264ngx_int_t
265brix_cvmfs_scrub_init_worker(ngx_cycle_t *cycle)
266{
267 ngx_uint_t i;
268
269 /* The store is shared across workers: one scrubber is enough, N of them
270 * is N× the hashing I/O for the same coverage. Worker 0 always exists
271 * (and IS the process in single-process mode). */
272 if (ngx_worker > 0) {
273 return NGX_OK;
274 }
275
276 for (i = 0; i < cvmfs_scrub_regs_n; i++) {
277 const cvmfs_scrub_reg_t *reg = &cvmfs_scrub_regs[i];
278 ngx_thread_task_t *task;
279 cvmfs_scrub_ctx_t *sc;
280
281 task = ngx_thread_task_alloc(cycle->pool, sizeof(cvmfs_scrub_ctx_t));
282 if (task == NULL) {
283 return NGX_ERROR;
284 }
285 sc = task->ctx;
286 sc->task = task;
287 sc->reg = reg;
288 sc->items = ngx_pcalloc(cycle->pool,
289 reg->rate * sizeof(cvmfs_scrub_item_t));
290 if (sc->items == NULL) {
291 return NGX_ERROR;
292 }
293
294 brix_task_bind(task, cvmfs_scrub_thread, cvmfs_scrub_done);
295 task->event.log = cycle->log;
296
297 sc->timer.handler = cvmfs_scrub_fire;
298 sc->timer.data = sc;
299 sc->timer.log = cycle->log;
300 ngx_add_timer(&sc->timer, (ngx_msec_t) reg->interval * 1000);
301 }
302 return NGX_OK;
303}
