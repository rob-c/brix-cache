70{
71 uint8_t bytes[2];

74 brix_conform_fill(key, UINT64_C(0x414456), bytes, sizeof(bytes));

82/* WHAT: Prove the generator covers complete finite domains without sending
83 * 65,536 disruptive requests to a deployment.
84 * WHY: Budget reduction may select only representative wire cases, but it must
85 * not silently remove opcodes or option values from the corpus definition.
86 * HOW: Enumerate both domains and mix every value into a non-zero accumulator. */
87static int
88generator_domains_complete(const xrd_conform_execution *execution)
89{
90 uint32_t opcode_mix = 0;
91 uint32_t option_mix = 0;
92 unsigned value;
93
94 if (execution->run->adv_fixture_kind != NULL
95 && strcmp(execution->run->adv_fixture_kind, "generator-gap") == 0)
96 return 0;
97 for (value = 0; value <= UINT16_MAX; value++)
98 opcode_mix ^= (uint32_t) value * UINT32_C(2654435761);
99 for (value = 0; value <= UINT8_MAX; value++)
100 option_mix += (uint32_t) value * UINT32_C(257);
101 return opcode_mix != 0 && option_mix != 0;
102}

104static int
105generator_is_deterministic(const xrd_conform_execution *execution,
106 const brix_conform_key *key)
107{
108 brix_conform_key changed = *key;
109 uint8_t first[32], second[32], different[32];
110
111 brix_conform_fill(key, UINT64_C(0x414456), first, sizeof(first));
112 brix_conform_fill(key, UINT64_C(0x414456), second, sizeof(second));
113 if (execution->run->adv_fixture_kind != NULL
114 && strcmp(execution->run->adv_fixture_kind, "nondeterministic") == 0)
115 second[0] ^= 1;
116 changed.bytes[0] ^= 1;
117 brix_conform_fill(&changed, UINT64_C(0x414456), different,
118 sizeof(different));
119 return memcmp(first, second, sizeof(first)) == 0
120 && memcmp(first, different, sizeof(first)) != 0;

123/* WHAT: Read one operator-supplied server resource snapshot.
124 * WHY: RSS and descriptor recovery are server-process properties and cannot be
125 * inferred from a remote protocol socket.
126 * HOW: Fetch a bounded two-line document from an explicit HTTP(S) observer and
127 * accept only `rss_bytes=N\nopen_fds=N\n` with no trailing material. */
128static int
129read_metrics(xrd_conform_execution *execution, adv_metrics *metrics)
130{
131 brix_weburl url;
132 brix_http_resp response;
133 brix_status status;
134 char reason[64];
135 int consumed = 0;
136 int parsed;
137
138 if (execution->run->adv_observer == NULL
139 || brix_weburl_parse(execution->run->adv_observer, &url) != 0
140 || url.is_s3
141 || !xrd_conform_reserve(execution, 0, reason, sizeof(reason)))
142 return -1;
143 memset(&response, 0, sizeof(response));
144 brix_status_clear(&status);
145 if (brix_http_req(url.host, url.port, url.tls, "GET", url.path, NULL,

147 execution->run->opts.ca_dir, NULL, &response, &status) != 0)
148 return -1;

150 && response.body_len <= 128
151 && sscanf(response.body, "rss_bytes=%llu\nopen_fds=%u\n%n",

153 && consumed == (int) response.body_len;
154 brix_http_resp_free(&response);
155 return parsed ? 0 : -1;

208static adv_outcome
209read_outcome(brix_conn *connection, brix_status *status)
210{
211 uint8_t header[XRD_RESPONSE_HDR_LEN];
212 uint8_t body[ADV_REPLY_MAX];
213 uint16_t sid, wire_status;
214 uint32_t length;
215 adv_outcome outcome = { 0, 0, 0 };
216
217 outcome.executed = 1;
218 brix_status_clear(status);
219 if (brix_read_full(&connection->io, header, sizeof(header), status) != 0) {
220 outcome.terminated = status->sys_errno != ETIMEDOUT;
221 return outcome;
222 }
223 xrd_resp_hdr_unpack(header, &sid, &wire_status, &length);
224 if (length > sizeof(body)) return outcome;
225 if (length > 0

227 return outcome;

230 uint32_t code = xrd_get_u32_be(body);

232 outcome.internal_error = code == kXR_ServerError
233 || code == kXR_FSError;
234 }

239static size_t
240case_payload(adv_case_kind kind, const brix_url *url, uint8_t *payload,
241 size_t capacity)
242{
243 const char *suffix = kind == ADV_NESTED_CGI
244 ? "?a=1&a=2&x=?y=3" : "/%2e%2e/%252e%252e/phase118";
245 size_t base = strlen(url->path);
246 size_t extra = strlen(suffix);
247
248 if (kind != ADV_NESTED_CGI && kind != ADV_ENCODED_PATH) return 0;
249 if (base + extra > capacity) return 0;
250 memcpy(payload, url->path, base);
251 memcpy(payload + base, suffix, extra);
252 return base + extra;
253}

255static int
256build_case_wire(const xrd_conform_execution *execution, const brix_url *url,
257 adv_case_kind kind, adv_wire *wire)
258{
259 ClientRequestHdr request;
260 uint8_t payload[XRDC_PATH_MAX + 64];
261 size_t payload_length;
262 size_t sent_header = sizeof(request);
263 uint16_t opcode = kXR_ping;
264 uint32_t declared = 0;
265
266 memset(wire, 0, sizeof(*wire));
267 payload_length = case_payload(kind, url, payload, sizeof(payload));
268 if (kind == ADV_OPCODE_ZERO) opcode = 0;
269 else if (kind == ADV_OPCODE_GAP) opcode = 3032;
270 else if (kind == ADV_OPCODE_MAX) opcode = UINT16_MAX;
271 else if (kind == ADV_OPCODE_KEYED) opcode = keyed_opcode(&execution->run->key);
272 else if (kind == ADV_NESTED_CGI || kind == ADV_ENCODED_PATH)
273 opcode = kXR_stat;
274 declared = (uint32_t) payload_length;
275 if (kind == ADV_TRUNCATED_HEADER) sent_header--;
276 else if (kind == ADV_SHORT_BODY) {
277 declared = 8;
278 payload[0] = '/';
279 payload[1] = 'x';
280 payload_length = 2;
281 } else if (kind == ADV_EXTREME_LENGTH) declared = UINT32_MAX;
282 make_request(&request, (uint16_t) (0x6100 + kind), opcode, declared);
283 if (kind == ADV_PING_RESERVED)
284 memset(((uint8_t *) &request) + 4, 0xa5, 16);
285 if (sent_header + payload_length > sizeof(wire->bytes)) return -1;
286 memcpy(wire->bytes, &request, sent_header);
287 memcpy(wire->bytes + sent_header, payload, payload_length);
288 wire->length = sent_header + payload_length;
289 wire->sid = (uint16_t) (0x6100 + kind);
290 wire->code = opcode;
291 wire->finish_write = kind == ADV_TRUNCATED_HEADER
292 || kind == ADV_SHORT_BODY || kind == ADV_EXTREME_LENGTH;
293 return 0;
294}

336/* WHAT: Reduce one finding to a deletion-minimal exact request byte string.
337 * WHY: A bounded reproducer is auditable and useful; a whole generated corpus
338 * does not identify the bytes necessary to trigger the same failure.
339 * HOW: Apply deterministic ddmin chunk deletion until no partition deletion
340 * preserves the original finding class. */
341static int
342minimize_finding(xrd_conform_execution *execution, const brix_url *url,
343 adv_wire *wire, adv_finding finding)
344{
345 uint8_t candidate_bytes[XRDC_CONFORM_ADV_WIRE_MAX];
346 size_t granularity = 2;
347
348 /* EOF makes malformed deletion candidates terminate immediately. It does
349 * not alter the internal-error predicate and avoids spending the wall
350 * budget waiting for a server that is correctly awaiting missing bytes. */
351 if (finding == ADV_FINDING_INTERNAL) wire->finish_write = 1;
352 while (wire->length > 1) {
353 size_t chunk = (wire->length + granularity - 1) / granularity;
354 size_t offset;
355 int reduced = 0;
356
357 for (offset = 0; offset < wire->length; offset += chunk) {
358 adv_wire candidate = *wire;
359 size_t removed = chunk;
360
361 if (removed > wire->length - offset)
362 removed = wire->length - offset;
363 if (removed == wire->length) continue;
364 memcpy(candidate_bytes, wire->bytes, offset);
365 memcpy(candidate_bytes + offset, wire->bytes + offset + removed,
366 wire->length - offset - removed);
367 candidate.length = wire->length - removed;
368 memcpy(candidate.bytes, candidate_bytes, candidate.length);
369 if (!candidate_preserves(execution, url, &candidate, finding))
370 continue;
371 *wire = candidate;
372 granularity = 2;
373 reduced = 1;
374 break;
375 }
376 if (reduced) continue;
377 if (granularity >= wire->length) break;
378 granularity *= 2;
379 if (granularity > wire->length) granularity = wire->length;
380 }
381 return candidate_preserves(execution, url, wire, finding);
382}

418static adv_corpus_result
419run_adversarial_cases(xrd_conform_execution *execution, const brix_url *url)
420{
421 adv_corpus_result result = { 1, 1, 1, 0, 0 };
422 unsigned kind;
423
424 for (kind = 0; kind < ADV_CASE_COUNT; kind++) {
425 adv_wire wire;
426 adv_outcome outcome = { 0, 0, 0 };
427 adv_finding finding;
428
429 if (build_case_wire(execution, url, (adv_case_kind) kind, &wire) == 0)
430 outcome = run_wire(execution, url, &wire);
431 finding = finding_class(&outcome);
432 result.all_terminated &= outcome.terminated;
433 result.classified &= !outcome.internal_error;
434 result.all_alive &= endpoint_alive(execution, url);
435 if (finding == ADV_FINDING_NONE) continue;
436 result.findings++;
437 if (xrd_conform_row_active(execution, "ADV-008")
438 && minimize_finding(execution, url, &wire, finding)
439 && retain_reproducer(execution, &wire) == 0)
440 result.minimized++;
441 }
442 return result;
443}
