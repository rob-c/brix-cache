1/*
2 * stage_events.c — 2.0 F2: the StageEvents notification file
3 * (brix_frm_stagemsg), BriX's counterpart of xrootd's oss.stagemsg /
4 * XRDOFSEVENTS hand-off to an external stager or tape monitor.
5 *
6 * WHAT: One append-only text line per stage transition — the durable engine
7 * (queued / started / done / failed / deadletter / replayed / dropped),
8 * the kXR_prepare registry (queued / staging / online / failed /
9 * cancelled / deleted / expired) and the MSS adapter (recall-begin /
10 * recall-online / recall-failed / migrate-done / migrate-failed) — so a
11 * site hook can tail ONE file instead of parsing error.log or the
12 * journal directory. Line grammar (space separated, LF terminated):
13 *
14 * <utc-iso8601> <source> <event> <reqid|-> <key|-> [name=value ...]
15 *
16 * The key and every value are %-escaped with ngx_escape_uri (space,
17 * '%', '#', '?', '"', controls, non-ASCII), so a line always splits on
18 * spaces and a name=value pair on its first '='.
19 *
20 * WHY: Stock xrootd sites drive tape bookkeeping from the stager's event
21 * stream. The in-band kXR_wait / HTTP 202 park informs the CLIENT and
22 * the transfer ledger books BYTES; neither is a per-request transition
23 * feed an operator can hook without reading engine internals.
24 *
25 * HOW: brix_stage_engine_conf_apply() opens the file once per worker
26 * (O_APPEND|O_CLOEXEC, created 0600). Every emit is ONE write(2) of a
27 * whole line, so lines from several workers interleave but never tear.
28 * The feed is best-effort by design: a failed open or write is logged
29 * once per worker, the feed goes quiet, and the next emit re-opens
30 * lazily — an operator who repairs the directory needs no reload, and a
31 * broken feed never fails a stage. No credential reaches the file:
32 * callers pass keys, principals and numbers, never a brix_stage_cred_t.
33 */

55static void
56stage_events_fail(const char *what)
57{
58 if (stage_events.reported || stage_events.log == NULL) {
59 return;
60 }
61 stage_events.reported = 1;
62 ngx_log_error(NGX_LOG_ERR, stage_events.log, errno,
63 "brix: brix_frm_stagemsg \"%s\": %s failed; stage notifications are "
64 "off in this worker until the file is writable again (staging "
65 "itself is unaffected)", stage_events.path, what);
66}

68/* 1 = an fd is ready, 0 = the feed is off or its file is still unopenable. */
69static int
70stage_events_reopen(void)
71{
72 if (stage_events.fd >= 0) {
73 return 1;
74 }
75 if (stage_events.path[0] == '\0') {
76 return 0;
77 }
78 stage_events.fd = open(stage_events.path,
79 O_WRONLY | O_APPEND | O_CREAT | O_CLOEXEC, 0600);
80 if (stage_events.fd < 0) {
81 stage_events_fail("open");
82 return 0;
83 }
84 return 1;
85}

124/* Append `src` %-escaped ("-" when empty). Returns the new length, or 0 when
125 * the field would not fit — the caller then drops the rest of the line. */
126static size_t
127stage_events_put(u_char *line, size_t len, const char *src)
128{
129 size_t n, esc;
130
131 if (src == NULL || src[0] == '\0') {
132 src = "-";
133 }
134 n = strlen(src);
135 esc = ngx_escape_uri(NULL, (u_char *) src, n, NGX_ESCAPE_URI);
136 if (len + n + 2 * esc + 2 > STAGE_EVENTS_LINE_MAX) { /* + ' ' + '\n' */
137 return 0;
138 }
139 (void) ngx_escape_uri(line + len, (u_char *) src, n, NGX_ESCAPE_URI);
140 return len + n + 2 * esc;
141}

143static size_t
144stage_events_put_pair(u_char *line, size_t len, const char *name,
145 const char *value)
146{
147 size_t n = strlen(name);
148
149 if (len + n + 3 > STAGE_EVENTS_LINE_MAX) {
150 return 0;
151 }
152 line[len++] = ' ';
153 memcpy(line + len, name, n);
154 len += n;
155 line[len++] = '=';
156 return stage_events_put(line, len, value);
157}

159void
160brix_stage_events_emit(const char *source, const char *event,
161 const char *reqid, const char *key, ...)
162{
163 u_char line[STAGE_EVENTS_LINE_MAX];
164 size_t len, next;
165 time_t now;
166 ngx_tm_t tm;
167 va_list ap;
168 const char *name, *value;
169
170 if (!brix_stage_events_enabled() || !stage_events_reopen()) {
171 return;
172 }
173 now = time(NULL);
174 ngx_gmtime(now, &tm);
175 len = (size_t) snprintf((char *) line, sizeof(line),
176 "%04d-%02d-%02dT%02d:%02d:%02dZ %s %s ",
177 tm.ngx_tm_year, tm.ngx_tm_mon, tm.ngx_tm_mday,
178 tm.ngx_tm_hour, tm.ngx_tm_min, tm.ngx_tm_sec,
179 source, event);
180 len = stage_events_put(line, len, reqid);
181 if (len == 0) {
182 return;
183 }
184 line[len++] = ' ';
185 len = stage_events_put(line, len, key);
186 if (len == 0) {
187 return;
188 }
189
190 va_start(ap, key);
191 for (name = va_arg(ap, const char *); name != NULL;
192 name = va_arg(ap, const char *))
193 {
194 value = va_arg(ap, const char *);
195 next = stage_events_put_pair(line, len, name, value);
196 if (next == 0) {
197 break; /* keep what fits; the line stays whole */
198 }
199 len = next;
200 }
201 va_end(ap);
202
203 line[len++] = '\n';
204 if (write(stage_events.fd, line, len) == (ssize_t) len) {
205 stage_events.reported = 0; /* a later outage is news again */
206 return;
207 }
208 stage_events_fail("write");
209 (void) close(stage_events.fd);
210 stage_events.fd = -1;
211}
