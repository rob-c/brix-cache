30/* Store a `replace <find> <repl>` (or clear on "off") into the named direction. */
31int
32ext_set_replace(int d, char *findtok, char *repltok, char *reply, size_t rsz)
33{
34 if (!findtok || strcmp(findtok, "off") == 0) {
35 pthread_mutex_lock(&g_ext_lock);
36 if (d != 2) { g_up_mut.find_len = 0; }
37 if (d != 1) { g_down_mut.find_len = 0; }
38 pthread_mutex_unlock(&g_ext_lock);
39 return 0;
40 }
41 unsigned char fb[128], rb[256];
42 int fl = fp_ext_parse_payload(findtok, fb, sizeof(fb));
43 int rl = repltok ? fp_ext_parse_payload(repltok, rb, sizeof(rb)) : 0;
44 if (fl <= 0 || rl < 0) {
45 snprintf(reply, rsz, "err: bad replace payload (use hex:.. or str:..)\n");
46 return -1;
47 }
48 pthread_mutex_lock(&g_ext_lock);
49 if (d != 2) {
50 memcpy(g_up_mut.find, fb, (size_t) fl); g_up_mut.find_len = fl;
51 memcpy(g_up_mut.repl, rb, (size_t) rl); g_up_mut.repl_len = rl;
52 }
53 if (d != 1) {
54 memcpy(g_down_mut.find, fb, (size_t) fl); g_down_mut.find_len = fl;
55 memcpy(g_down_mut.repl, rb, (size_t) rl); g_down_mut.repl_len = rl;
56 }
57 pthread_mutex_unlock(&g_ext_lock);
58 return 0;
59}

78/* Configure the forged PROXY-protocol header: `proxy-header v1|v2 SRC [DST]`. */
79int
80ext_set_proxy(char *args, char *reply, size_t rsz)
81{
82 char *mode = strtok(args, " ");
83 char *src = strtok(NULL, " ");
84 char *dst = strtok(NULL, " ");
85 if (!mode || strcmp(mode, "off") == 0) {
86 g_proxy_mode = 0;
87 return 0;
88 }
89 int m = (strcmp(mode, "v1") == 0) ? 1 : (strcmp(mode, "v2") == 0) ? 2 : 0;
90 if (m == 0 || !src) {
91 snprintf(reply, rsz, "err: proxy-header needs v1|v2 SRC [DST]\n");
92 return -1;
93 }
94 pthread_mutex_lock(&g_ext_lock);
95 snprintf(g_proxy_src, sizeof(g_proxy_src), "%s", src);
96 snprintf(g_proxy_dst, sizeof(g_proxy_dst), "%s", dst ? dst : "");
97 pthread_mutex_unlock(&g_ext_lock);
98 g_proxy_mode = m;
99 return 0;
100}

111/* `chaos [<ms>|off]` — a background thread flipping levers at random. Every arm
112 * bumps g_chaos_gen so the previously-detached thread retires itself. */
113static void
114ext_set_chaos(const char *args)
115{
116 if (strcmp(args, "off") == 0 || args[0] == '\0') {
117 __atomic_add_fetch(&g_chaos_gen, 1, __ATOMIC_SEQ_CST);
118 g_chaos_on = 0;
119 return;
120 }
121 int ms = atoi(args);
122 g_chaos_ms = ms > 10 ? ms : 100;
123 unsigned gen = __atomic_add_fetch(&g_chaos_gen, 1, __ATOMIC_SEQ_CST);
124 pthread_t th;
125 if (pthread_create(&th, NULL, chaos_thread, (void *) (uintptr_t) gen) == 0) {
126 pthread_detach(th);
127 g_chaos_on = 1;
128 }
129}

131/* Extended, still-root-free levers: socket-level stress (mss/rcvbuf/sndbuf),
132 * backpressure (stall/unstall), connection lifetime, payload MITM (inject/
133 * replace), PROXY-header forgery, and the chaos monkey. Returns 1 if handled. */
134int
135cmd_set_ext(const char *verb, char *args, char *reply, size_t rsz)
136{
137 if (strcmp(verb, "mss") == 0) {
138 g_mss = atoi(args);
139 } else if (strcmp(verb, "rcvbuf") == 0) {
140 g_rcvbuf = atoi(args);
141 } else if (strcmp(verb, "sndbuf") == 0) {
142 g_sndbuf = atoi(args);
143 } else if (strcmp(verb, "max-lifetime") == 0) {
144 g_max_life_ms = atol(args);
145 } else if (strcmp(verb, "stall") == 0) {
146 ext_set_stall(args, 1);
147 } else if (strcmp(verb, "unstall") == 0) {
148 ext_set_stall(args, 0);
149 } else if (strcmp(verb, "inject") == 0) {
150 int d = dir_of(args);
151 ext_set_inject(d, args, reply, rsz);
152 } else if (strcmp(verb, "replace") == 0) {
153 int d = dir_of(args);
154 char *f = strtok(args, " ");
155 char *r = strtok(NULL, " ");
156 ext_set_replace(d, f, r, reply, rsz);
157 } else if (strcmp(verb, "proxy-header") == 0) {
158 ext_set_proxy(args, reply, rsz);
159 } else if (strcmp(verb, "chaos") == 0) {
160 ext_set_chaos(args);
161 } else {
162 return 0;
163 }
164 return 1;
165}

167/* Flap the listener in/out of service (a load balancer removing/adding the
168 * backend): block for down_ms, unblock for up_ms, repeat until superseded. */
169void *
170flap_thread(void *arg)
171{
172 unsigned gen = (unsigned) (uintptr_t) arg;
173 while (__atomic_load_n(&g_flap_gen, __ATOMIC_SEQ_CST) == gen) {
174 apply_command((char[]){"unblock"}, NULL, 0);
175 usleep((useconds_t) (g_flap_up_ms > 0 ? g_flap_up_ms : 500) * 1000);
176 if (__atomic_load_n(&g_flap_gen, __ATOMIC_SEQ_CST) != gen) {
177 break;
178 }
179 apply_command((char[]){"block"}, NULL, 0);
180 usleep((useconds_t) (g_flap_down_ms > 0 ? g_flap_down_ms : 500) * 1000);
181 }
182 apply_command((char[]){"unblock"}, NULL, 0); /* leave service restored */
183 return NULL;
184}

186/* Sweep a numeric lever from start to end over ms (a degrading link / a server
187 * warming up under load). Stops early if superseded (clear / new ramp epoch). */
188void *
189ramp_thread(void *arg)
190{
191 struct ramp_arg *r = arg;
192 const int steps = 20;
193 int per = r->ms / steps;
194 if (per < 10) {
195 per = 10;
196 }
197 for (int s = 0; s <= steps; s++) {
198 if (__atomic_load_n(&g_ramp_gen, __ATOMIC_SEQ_CST) != r->gen) {
199 break;
200 }
201 double v = r->start + (r->end - r->start) * s / steps;
202 char cmd[64];
203 snprintf(cmd, sizeof(cmd), "%s %g", r->lever, v);
204 apply_command(cmd, NULL, 0);
205 usleep((useconds_t) per * 1000);
206 }
207 free(r);
208 return NULL;
209}
