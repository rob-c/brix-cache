1/*
2 * diag_doctor_recon.c — deep read-only reconnaissance of one endpoint (--deep-recon).
3 *
4 * WHAT: over one already-open root:// connection, surface the whole read-only
5 * picture an operator might want to eyeball on a server: the operational
6 * counters from `query stats a` (link traffic, xrootd op mix, login
7 * auth/fail, OFS TPC accounting, OSS capacity/inodes, XrdHttp totals), a
8 * full Qconfig key sweep (how many advertised keys the server answers),
9 * the decoded kXR_protocol capability bits, and a bounded list of the
10 * top-level namespace roots our identity is allowed to see.
11 * WHY: --config-audit classifies a handful of VALUES into advice; --deep-recon
12 * is the reconnaissance counterpart — it reports everything, diagnostic
13 * not advisory, so an operator can inspect a remote server's live state in
14 * one pass without shelling in. It reads only what the server volunteers.
15 * HOW: pure composition of the public libbrix API — kXR_QStats + kXR_Qconfig +
16 * a dirlist("/"). The XML/flag parsers are pure over caller buffers and
17 * unit-tested off recorded server output. PII-free: only server-advertised
18 * scalars and top-level path *names* the server itself returns; never a
19 * token, credential, or user path. No goto; early-return.
20 */
21#include "diag_internal.h"

24/* ---- bounded, scoped XML field extraction (pure; unit-tested) ---- */

26/*
27 * Return a pointer to the inner text of `<stats id="ID" ...>` and its length up
28 * to the next `</stats>`, or NULL when the block is absent. The open tag is
29 * matched on the exact id (the closing quote is part of the needle, so "oss"
30 * never matches "oss.0"), then we advance past the first '>' so any attributes
31 * after the id are skipped. No allocation; the span is a view into `xml`.
32 */
33static const char *
34recon_stats_block(const char *xml, const char *id, size_t *blen)
35{
36 char open[48];
37 const char *p, *gt, *end;
38
39 snprintf(open, sizeof(open), "<stats id=\"%s\"", id);
40 p = strstr(xml, open);
41 if (p == NULL) {
42 return NULL;
43 }
44 gt = strchr(p, '>');
45 if (gt == NULL) {
46 return NULL;
47 }
48 p = gt + 1;
49 end = strstr(p, "</stats>");
50 if (end == NULL) {
51 return NULL;
52 }
53 *blen = (size_t) (end - p);
54 return p;
55}

58/*
59 * Parse the signed int64 inside the first `<tag>…</tag>` found within the
60 * `blen`-bounded block `blk`, or -1 when the tag is absent/empty/non-numeric.
61 * strstr scans the whole NUL-terminated buffer, so an offset past `blen` means
62 * the match belongs to a later block and is rejected — that is what bounds us.
63 */
64static int64_t
65recon_tag_i64(const char *blk, size_t blen, const char *tag)
66{
67 char open[24];
68 const char *p, *q;
69
70 snprintf(open, sizeof(open), "<%s>", tag);
71 p = strstr(blk, open);
72 if (p == NULL || (size_t) (p - blk) >= blen) {
73 return -1;
74 }
75 q = p + strlen(open);
76 while (*q == ' ' || *q == '\t') {
77 q++;
78 }
79 if (*q != '-' && (*q < '0' || *q > '9')) {
80 return -1;
81 }
82 return (int64_t) strtoll(q, NULL, 10);
83}

86/* Return the inner text of a child `<tag>…</tag>` within the bounded block
87 * `blk`, or NULL when absent. Used to scope into nested groups (ops/lgn/tpc). */
88static const char *
89recon_child(const char *blk, size_t blen, const char *tag, size_t *clen)
90{
91 char open[24], close[24];
92 const char *p, *end;
93
94 snprintf(open, sizeof(open), "<%s>", tag);
95 snprintf(close, sizeof(close), "</%s>", tag);
96 p = strstr(blk, open);
97 if (p == NULL || (size_t) (p - blk) >= blen) {
98 return NULL;
99 }
100 p = p + strlen(open);
101 end = strstr(p, close);
102 if (end == NULL) {
103 return NULL;
104 }
105 *clen = (size_t) (end - p);
106 return p;
107}

110/*
111 * doctor_recon_xml_i64 — scoped int64 read: the value of top-level <tag> inside
112 * `<stats id="ID">`. Exposed for offline unit testing of the extractor spine.
113 */
114int64_t
115doctor_recon_xml_i64(const char *xml, const char *id, const char *tag)
116{
117 const char *blk;
118 size_t blen;
119
120 if (xml == NULL || id == NULL || tag == NULL) {
121 return -1;
122 }
123 blk = recon_stats_block(xml, id, &blen);
124 if (blk == NULL) {
125 return -1;
126 }
127 return recon_tag_i64(blk, blen, tag);
128}

251/*
252 * doctor_recon_caps_str — render the set kXR_protocol capability bits of `f`
253 * as a comma-separated name list into out[osz]; returns the count named. The
254 * bit→name table is the same vocabulary conn_explain.c narrates, compacted for
255 * a single reconnaissance line and machine consumption.
256 */
257int
258doctor_recon_caps_str(unsigned f, char *out, size_t osz)
259{
260 static const struct { unsigned bit; const char *name; } M[] = {
261 { kXR_isServer, "server" },
262 { kXR_isManager, "manager" },
263 { kXR_attrCache, "cache" },
264 { kXR_attrMeta, "meta" },
265 { kXR_attrProxy, "proxy" },
266 { kXR_attrSuper, "supervisor" },
267 { kXR_attrVirtRdr, "virt-redirector" },
268 { kXR_recoverWrts, "recover-writes" },
269 { kXR_collapseRedir, "collapse-redir" },
270 { kXR_ecRedir, "ec-redir" },
271 { kXR_haveTLS, "tls-available" },
272 { kXR_gotoTLS, "tls-required" },
273 { kXR_tlsLogin, "tls-login" },
274 { kXR_anongpf, "anon-gpf" },
275 { kXR_supgpf, "gpf" },
276 { kXR_suppgrw, "pgrw" },
277 { kXR_supposc, "posc" },
278 };
279 size_t i, used = 0;
280 int cnt = 0;
281
282 if (osz > 0) {
283 out[0] = '\0';
284 }
285 for (i = 0; i < sizeof(M) / sizeof(M[0]); i++) {
286 if (!(f & M[i].bit)) {
287 continue;
288 }
289 used += (size_t) snprintf(out + used, used < osz ? osz - used : 0,
290 "%s%s", cnt ? "," : "", M[i].name);
291 cnt++;
292 }
293 return cnt;
294}

297/* ---- wire scrape (kXR_QStats / kXR_Qconfig / dirlist) ---- */
298
299/*
300 * Fetch one Qconfig key's value-line into out (may be NULL to only test
301 * support). Returns 1 when the server answered with a real value, 0 when the
302 * key is unsupported (echoed verbatim) or unanswered — the echo-suppression
303 * rule doctor_scrape_config/xrd_probe_caps use.
304 */
305static int
306recon_qkey(brix_conn *c, const char *key, char *out, size_t osz)
307{
308 char reply[256], *nl, *eq;
309 const char *val;
310 brix_status st;
311
312 if (out != NULL) {
313 out[0] = '\0';
314 }
315 brix_status_clear(&st);
316 if (brix_query(c, kXR_Qconfig, key, reply, sizeof(reply), &st) != 0) {
317 return 0;
318 }
319 if ((nl = strchr(reply, '\n')) != NULL) {
320 *nl = '\0';
321 }
322 eq = strchr(reply, '=');
323 val = (eq != NULL) ? eq + 1 : reply;
324 if (strcmp(val, key) == 0) { /* echo == key ⇒ unsupported/absent */
325 return 0;
326 }
327 if (out != NULL && osz > 0) {
328 snprintf(out, osz, "%.*s", (int) osz - 1, val);
329 }
330 return 1;
331}

334/* The full Qconfig key sweep. Counts how many advertised keys this server
335 * actually answers (a fingerprint of build/config surface) and captures the two
336 * cluster-identity values worth surfacing directly. PII-free. */
337static void
338recon_sweep_config(brix_conn *c, doctor_recon *r)
339{
340 static const char *KEYS[] = {
341 "version", "role", "sitename", "chksum", "tpc", "tpcdlg", "pgread",
342 "bind_max", "pio_max", "poscmax", "readv_ior_max", "readv_iov_max",
343 "window", "wan_port", "wan_window", "cid", "cms", "start", "dlgctm",
344 "fslib", "ofslib", "oidc", "xrdfs.ext", "oss.cgroup", "tpcprot",
345 };
346 size_t i;
347
348 r->cfg_probed = (int) (sizeof(KEYS) / sizeof(KEYS[0]));
349 for (i = 0; i < sizeof(KEYS) / sizeof(KEYS[0]); i++) {
350 char *dst = NULL; /* capture cid/cms directly; test-only otherwise */
351 size_t dsz = 0;
352
353 if (strcmp(KEYS[i], "cid") == 0) {
354 dst = r->cid; dsz = sizeof(r->cid);
355 } else if (strcmp(KEYS[i], "cms") == 0) {
356 dst = r->cms; dsz = sizeof(r->cms);
357 }
358 if (recon_qkey(c, KEYS[i], dst, dsz)) {
359 r->cfg_supported++;
360 }
361 }
362}

365/* Bounded authorized-root discovery: a single dirlist of "/". A refusal (not
366 * authorized) simply leaves roots_listed=0 — never an error. Only top-level
367 * entry NAMES are recorded (never a full path), capped at RECON_MAX_ROOTS. */
368static void
369recon_list_roots(brix_conn *c, doctor_recon *r)
370{
371 brix_dirent *ents = NULL;
372 size_t count = 0, i;
373 brix_status st;
374
375 brix_status_clear(&st);
376 if (brix_dirlist(c, "/", 0 /*no stat*/, &ents, &count, &st) != 0) {
377 return;
378 }
379 r->roots_listed = 1;
380 for (i = 0; i < count; i++) {
381 if (ents[i].name[0] == '\0' || strcmp(ents[i].name, ".") == 0
382 || strcmp(ents[i].name, "..") == 0) {
383 continue;
384 }
385 if (r->nroots >= RECON_MAX_ROOTS) {
386 r->roots_more = 1;
387 break;
388 }
389 snprintf(r->roots[r->nroots], sizeof(r->roots[0]), "%.*s",
390 (int) sizeof(r->roots[0]) - 1, ents[i].name);
391 r->nroots++;
392 }
393 free(ents);
394}

415/*
416 * doctor_recon_probe — fill e->recon over the live connection `c`. Best-effort:
417 * each leg (stats / config sweep / root list) is independent, so a server that
418 * denies one still yields the others. PII-free and read-only throughout.
419 */
420void
421doctor_recon_probe(const diag_args *a, brix_conn *c, doctor_ep *e)
422{
423 doctor_recon *r = &e->recon;
424 char xml[16384];
425 brix_status st;
426
427 (void) a;
428 recon_init(r, e->caps);
429
430 brix_status_clear(&st);
431 if (brix_query(c, kXR_QStats, "a", xml, sizeof(xml), &st) == 0
432 && xml[0] != '\0') {
433 doctor_recon_parse_stats(xml, r);
434 }
435 recon_sweep_config(c, r);
436 recon_list_roots(c, r);
437}
