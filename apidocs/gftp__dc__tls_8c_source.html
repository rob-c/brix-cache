1/*
2 * gftp_dc_tls.c — PROT P (protected) outbound GridFTP data channel.
3 *
4 * WHAT: raise one already-connected data socket to a TLS session that presents
5 * the same X.509 proxy the control channel authenticated with, then verify the
6 * chain the origin answers with and pin its subject DN to the credential this
7 * session delegated to that origin.
8 *
9 * WHY: with PROT C the data leg is cleartext on a port the origin nominated in
10 * a reply — the bytes are unprotected and nothing but the address pin ties the
11 * connection to the authenticated session. PROT P closes both halves: the
12 * payload is encrypted, and the DN pin proves the machine that answered on that
13 * port is the machine we authenticated to. A protected channel therefore never
14 * degrades: if any step here fails the transfer fails (gftp_dc_open closes the
15 * connection rather than returning a clear one).
16 *
17 * HOW: GridFTP's protected data channel is a straight TLS session on the data
18 * socket — a client ClientHello, no globus token framing (see
19 * src/protocols/gridftp/ftp_dc_sec.h for the inbound mirror). The credential is
20 * built by gftp_gsi_setup(), the same function the control channel uses, so the
21 * two cannot drift on which certificate BriX presents or which roots it trusts.
22 * The handshake is synchronous: this runs on a VFS worker thread, not the event
23 * loop, and SO_RCVTIMEO/SO_SNDTIMEO bound it. The DN pin is the shared
24 * header-only predicate in protocols/gridftp/ftp_dc_dn.h — applied here against
25 * OUR OWN subject, not the origin's, because a GridFTP server runs its data
26 * channel on the credential the client delegated to it. See gftp_dc_tls_pin().
27 */

45/* The identity this session delegated: the subject of the proxy the data-channel
46 * context presents, spelled the way brix_gsi_verify_chain spells the peer's so
47 * the two can be compared at all. Returns 0 / -1. */
48static int
49gftp_dc_tls_self_dn(gftp_gsi_t *gsi, char *out, size_t cap)
50{
51 X509 *self = SSL_get_certificate(gsi->ssl);
52
53 out[0] = '\0';
54 if (self == NULL) {
55 return -1;
56 }
57 return X509_NAME_oneline(X509_get_subject_name(self), out, (int) cap) == NULL
58 ? -1 : 0;
59}

62/* Post-handshake gate: hold the chain the origin answered with to BriX's proxy
63 * conformance policy, then require its leaf DN to name the credential we
64 * delegated to that origin. Returns 0 / -1 with the session error set.
65 *
66 * The base of the pin is OUR subject, and that is not a slip. A GridFTP server
67 * does not run its data channel on its host certificate — it runs it on the
68 * proxy the client delegated over the control channel, so the DN that comes
69 * back is ours plus one `/CN=<serial>` per delegation step. Pinning to the
70 * origin's control DN instead compares the client's identity against the
71 * server's and can never match: it refused every correct transfer with
72 * data-channel DN ".../CN=Test User/CN=12345/CN=12346/CN=3228637842"
73 * != control DN "/DC=test/DC=xrootd/CN=localhost"
74 * which reads like a mismatch to be fixed at the origin and is in fact this
75 * predicate asking the wrong question.
76 *
77 * The security property is unchanged in strength and clearer to state: the peer
78 * holds the PRIVATE KEY of the credential we handed to the origin we
79 * authenticated, proven by the handshake. A third party that reaches the
80 * passive data port has the public proxy at most, and cannot answer. The
81 * control channel's own identity is still required — an unauthenticated control
82 * channel delegates nothing, and the first branch below refuses it. */
83static int
84gftp_dc_tls_pin(gftp_session_t *session, gftp_gsi_t *gsi)
85{
86 brix_gsi_verify_result_t res;
87 X509 *leaf;
88 STACK_OF(X509) *chain;
89 ngx_log_t *log;
90 ngx_int_t rc;
91 char base[1024];
92
93 if (session->peer_dn[0] == '\0') {
94 gftp_set_error(session, EACCES,
95 "GridFTP PROT P has no control-channel identity to pin to");
96 return -1;
97 }
98 if (gftp_dc_tls_self_dn(gsi, base, sizeof(base)) != 0) {
99 gftp_set_error(session, EACCES,
100 "GridFTP PROT P cannot read its own delegated identity");
101 return -1;
102 }
103 leaf = SSL_get_peer_certificate(gsi->ssl); /* +1 ref */
104 if (leaf == NULL) {
105 gftp_set_error(session, EACCES,
106 "GridFTP data channel presented no certificate");
107 return -1;
108 }
109 chain = SSL_get_peer_cert_chain(gsi->ssl); /* borrowed */
110 log = (ngx_cycle != NULL) ? ngx_cycle->log : NULL;
111 ngx_memzero(&res, sizeof(res));
112 rc = brix_gsi_verify_chain(log, SSL_CTX_get_cert_store(gsi->ctx), leaf,
113 chain, 0, &res, 0);
114 X509_free(leaf);
115 if (rc != NGX_OK) {
116 gftp_set_error(session, EACCES,
117 "GridFTP data-channel proxy chain verify failed");
118 return -1;
119 }
120 if (!brix_ftp_dc_dn_matches(res.dn_buf, (const u_char *) base, strlen(base)))
121 {
122 gftp_set_error(session, EACCES,
123 "GridFTP data-channel DN \"%s\" is not the credential delegated "
124 "beneath \"%s\"", res.dn_buf, base);
125 return -1;
126 }
127 return 0;
128}

131/* TLS-layer verdict for the data-channel peer: accept, and let the post-handshake
132 * gate below decide. This is not a relaxation — it is where the decision lives.
133 *
134 * The peer presents the credential the control channel DELEGATED to it: an RFC
135 * 3820 proxy, which is not a TLS server certificate and never claims to be.
136 * OpenSSL's in-handshake check applies the sslServer purpose to it and rejects
137 * the chain with X509_V_ERR_INVALID_PURPOSE (26) before a single byte moves, so
138 * a correct origin serving a correctly delegated credential could not complete
139 * one protected transfer. The inbound half of the same feature
140 * (protocols/gridftp/ftp_dc_sec.c) has always done exactly this, for exactly this
141 * reason; the connect role was the half that still asked OpenSSL.
142 *
143 * What replaces it is stricter, not weaker: gftp_dc_tls_pin() runs BriX's proxy
144 * conformance verifier over the presented chain against the SAME CA store, and
145 * then requires the leaf DN to name the control-channel peer — a test OpenSSL's
146 * generic path does not make at all. */
147static int
148gftp_dc_tls_accept_cb(int preverify, X509_STORE_CTX *store)
149{
150 (void) preverify;
151 (void) store;
152 return 1;
153}

156/* SSL_connect failure detail for a data channel. OpenSSL's queue says only
157 * "certificate verify failed"; the reason the chain was rejected lives in the
158 * verify result and is the only part an operator can act on. */
159static int
160gftp_dc_tls_connect_error(gftp_session_t *session, SSL *ssl)
161{
162 long verdict = SSL_get_verify_result(ssl);
163
164 if (verdict != X509_V_OK) {
165 ERR_clear_error();
166 gftp_set_error(session, EACCES,
167 "GridFTP PROT P data-channel handshake: %s (%ld)",
168 X509_verify_cert_error_string(verdict), verdict);
169 return -1;
170 }
171 return gftp_ssl_error(session, "PROT P data-channel handshake");
172}

175int
176gftp_dc_tls_start(gftp_session_t *session, gftp_dc_t *dc)
177{
178 gftp_gsi_t *gsi;
179 long seconds = session->timeout_ms / 1000;
180
181 gsi = calloc(1, sizeof(*gsi));
182 if (gsi == NULL) {
183 gftp_set_error(session, ENOMEM, "cannot allocate GridFTP data channel");
184 return -1;
185 }
186 if (gftp_gsi_setup(gsi, session->proxy_path, session->ca_dir, session)
187 != 0)
188 {
189 gftp_gsi_free(gsi);
190 return -1;
191 }
192 gsi->ssl = SSL_new(gsi->ctx);
193 if (gsi->ssl == NULL) {
194 gftp_gsi_free(gsi);
195 return gftp_ssl_error(session, "create data-channel TLS session");
196 }
197 /* GridFTP senders close the data connection without close_notify once the
198 * last block is on the wire; that is EOD, not a truncation attack. */
199 SSL_set_options(gsi->ssl, SSL_OP_IGNORE_UNEXPECTED_EOF);
200 /* Data channel only — the control channel keeps OpenSSL's own verdict. */
201 SSL_set_verify(gsi->ssl, SSL_VERIFY_PEER, gftp_dc_tls_accept_cb);
202 if (SSL_set_fd(gsi->ssl, dc->fd) != 1) {
203 gftp_gsi_free(gsi);
204 return gftp_ssl_error(session, "bind data-channel socket");
205 }
206 brix_apply_socket_io_timeouts(dc->fd, seconds > 0 ? seconds : 1);
207 if (SSL_connect(gsi->ssl) != 1) {
208 int rc = gftp_dc_tls_connect_error(session, gsi->ssl);
209
210 gftp_gsi_free(gsi);
211 return rc;
212 }
213 if (gftp_dc_tls_pin(session, gsi) != 0) {
214 gftp_gsi_free(gsi);
215 return -1;
216 }
217 dc->tls = gsi;
218 return 0;
219}

222ssize_t
223gftp_dc_tls_read(gftp_session_t *session, gftp_dc_t *dc, void *buf, size_t cap)
224{
225 gftp_gsi_t *gsi = dc->tls;
226 int n;
227 int reason;
228
229 ERR_clear_error();
230 n = SSL_read(gsi->ssl, buf, (int) (cap > INT_MAX ? INT_MAX : cap));
231 if (n > 0) {
232 return n;
233 }
234 reason = SSL_get_error(gsi->ssl, n);
235 if (reason == SSL_ERROR_ZERO_RETURN) {
236 return 0; /* orderly end of data */
237 }
238 if (reason == SSL_ERROR_SYSCALL && ERR_peek_error() == 0 && errno == 0) {
239 return 0; /* peer closed after EOD */
240 }
241 return gftp_ssl_error(session, "read protected data channel");
242}

245int
246gftp_dc_tls_write_all(gftp_session_t *session, gftp_dc_t *dc, const void *buf,
247 size_t len)
248{
249 gftp_gsi_t *gsi = dc->tls;
250 const uint8_t *cursor = buf;
251 size_t written = 0;
252
253 while (written < len) {
254 size_t want = len - written;
255 int n;
256
257 if (want > INT_MAX) {
258 want = INT_MAX;
259 }
260 ERR_clear_error();
261 n = SSL_write(gsi->ssl, cursor + written, (int) want);
262 if (n <= 0) {
263 return gftp_ssl_error(session, "write protected data channel");
264 }
265 written += (size_t) n;
266 }
267 return 0;
268}
