1/*
2 * sd_ceph_ns_cred.c — the credential-scoped NAMESPACE slots of the flat RADOS
3 * driver: stat / unlink / getxattr / listxattr / setxattr / removexattr /
4 * setattr / truncate_path / opendir, each executed on the requesting user's own
5 * CephX connection rather than on the export's service connection.
6 *
7 * WHY THIS FILE EXISTS
8 * The identity a RADOS operation asserts at the OSDs is the ioctx it runs on
9 * and nothing else. Before this file the driver published open_cred alone, so
10 * a request carrying a per-user keyring got its data plane checked as that user
11 * while every metadata op — the probe stat, the listing, the xattr read, the
12 * delete — still reached st->ioctx and executed as the export service account.
13 * That is the classic confused deputy: the caller supplies the path, the deputy
14 * supplies the authority. It is the same defect closed on sd_remote, and it is
15 * only visible in ALLOW mode: brix_sd_<op>_maybe_cred already refuses with
16 * EACCES when a cred with fallback_deny meets a driver that has the plain slot
17 * and no _cred twin, so deny mode was always safe and the permitted path was
18 * the hole.
19 *
20 * SHAPE
21 * Every slot is the same three steps — resolve the caller's ioctx with
22 * sd_ceph_cred_ioctx_get(), run the ioctx-explicit core (the *_io functions the
23 * plain slots also call, so there is exactly one implementation of each op),
24 * release with sd_ceph_cred_ioctx_put() on EVERY exit path — so those steps
25 * live once, in sd_ceph_ns_cred_run(), and each slot is only the argument
26 * bundle plus its op tag. The _put releases a TRANSIENT connection only; a
27 * cached one belongs to the LRU. No pin is taken: a namespace op leaves no
28 * handle behind, unlike open_cred, whose object keeps reading from the
29 * connection after the slot returns. opendir_cred is the one slot written out
30 * longhand: it returns a handle rather than a count.
31 *
32 * NOT HERE, and deliberately so:
33 * - rename_cred: sd_ceph_rename threads sd_ceph_state_t through eight helpers
34 * and copies bytes through st->striper, which is bound to the EXPORT's
35 * connection. A cred-shaped wrapper over that would assert the wrong identity
36 * for the copy while looking correct at the call site.
37 * - staged_open_cred: sd_ceph_staged_t carries only the final oid, so a
38 * cred-scoped stage would have to hold the ioctx and pin the connection
39 * across commit AND abort.
40 * - mkdir_cred: directories are synthetic (ADR-1) and mkdir touches no object,
41 * so there is no cluster-side authority to scope.
42 */

49/* sd_ceph_ns_op_e / sd_ceph_ns_args_t — which namespace core to run, and the
50 * union of everything the cores take. One tagged call site instead of eight
51 * copies of "acquire the caller's ioctx, run the op, release": the acquire and
52 * the release are the security-relevant half, and they must not exist in eight
53 * places where one of them can quietly lose an early return. */

79/* sd_ceph_ns_apply — run one ioctx-explicit namespace core.
80 * The cores are shared verbatim with the plain (service-credential) slots, so
81 * there is exactly one implementation of each op and the credential decides
82 * nothing but WHICH ioctx it runs on. Returns the core's own value widened to
83 * ssize_t: NGX_OK (0) / NGX_ERROR (-1) for the ngx_int_t cores, a byte count or
84 * -1 for the two that return ssize_t — the two encodings agree on both 0 and
85 * -1, which is why one return type serves. */
86static ssize_t
87sd_ceph_ns_apply(sd_ceph_state_t *st, rados_ioctx_t io, sd_ceph_ns_op_e op,
88 const sd_ceph_ns_args_t *a)
89{
90 switch (op) {
91 case SD_CEPH_NS_STAT:
92 return sd_ceph_stat_io(st, io, a->path, a->stat_out);
93 case SD_CEPH_NS_UNLINK:
94 return sd_ceph_unlink_io(st, io, a->path, a->is_dir);
95 case SD_CEPH_NS_UNLINK_MANY:
96 return sd_ceph_unlink_many_io(st, io, a->batch);
97 case SD_CEPH_NS_GETXATTR:
98 return sd_ceph_getxattr_io(st, io, a->path, a->name, a->buf, a->cap);
99 case SD_CEPH_NS_LISTXATTR:
100 return sd_ceph_listxattr_io(st, io, a->path, a->buf, a->cap);
101 case SD_CEPH_NS_SETXATTR:
102 return sd_ceph_setxattr_io(st, io, a->path, a->name, a->val, a->cap);
103 case SD_CEPH_NS_REMOVEXATTR:
104 return sd_ceph_removexattr_io(st, io, a->path, a->name);
105 case SD_CEPH_NS_TRUNCATE_PATH:
106 return sd_ceph_truncate_path_io(st, io, a->path, a->len);
107 case SD_CEPH_NS_SETATTR:
108 return sd_ceph_setattr_io(st, io, a->path, a->attr);
109 }
110 errno = ENOSYS; /* unreachable: the enum is closed */
111 return -1;
112}

114/* sd_ceph_ns_cred_run — the whole shape of a credential-scoped namespace slot:
115 * resolve the caller's ioctx, run the core on it, release on EVERY exit path.
116 * The release frees a TRANSIENT connection only — a cached one belongs to the
117 * LRU — and no pin is needed because a namespace op leaves no handle behind.
118 * Returns the core's value, or -1 with errno set if the credential was refused
119 * (deny mode) or could not be connected. */
120static ssize_t
121sd_ceph_ns_cred_run(brix_sd_instance_t *inst, const brix_sd_cred_t *cred,
122 sd_ceph_ns_op_e op, const sd_ceph_ns_args_t *a)
123{
124 sd_ceph_state_t *st = inst->state;
125 sd_ceph_cred_ioctx_t ci;
126 ssize_t rc;
127
128 if (sd_ceph_cred_ioctx_get(st, inst->pool, cred, &ci) != 0) {
129 return -1;
130 }
131 rc = sd_ceph_ns_apply(st, ci.ioctx, op, a);
132 sd_ceph_cred_ioctx_put(&ci);
133 return rc;
134}

247/* sd_ceph_opendir_cred — list as the caller. Stands apart from the tagged
248 * dispatch above because it alone returns a handle rather than a count, and
249 * reports its failure through *err_out. Safe to release the connection at
250 * return: sd_ceph_opendir_io is EAGER — it snapshots the whole listing into the
251 * handle before it returns, so no later readdir touches the cluster. */
252brix_sd_dir_t *
253sd_ceph_opendir_cred(brix_sd_instance_t *inst, const char *path, int *err_out,
254 const brix_sd_cred_t *cred)
255{
256 sd_ceph_state_t *st = inst->state;
257 sd_ceph_cred_ioctx_t ci;
258 brix_sd_dir_t *d;
259
260 if (sd_ceph_cred_ioctx_get(st, inst->pool, cred, &ci) != 0) {
261 if (err_out != NULL) {
262 *err_out = errno;
263 }
264 return NULL;
265 }
266 d = sd_ceph_opendir_io(inst, ci.ioctx, path, err_out);
267 sd_ceph_cred_ioctx_put(&ci);
268 return d;
269}
