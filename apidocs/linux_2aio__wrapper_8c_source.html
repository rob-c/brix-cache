86/* ---- Prepare a typed read or write submission ----
87 * WHAT: Queue the request with its callback, or report validation/capacity errors.
88 * WHY: Read/write preparation has identical ownership and pending accounting.
89 * HOW: 1. Validate inputs and acquire a submission slot. 2. Attach the callback.
90 * 3. Prepare the chosen opcode and increment pending operations.
91 */
92static int
93brix_aio_prepare(brix_aio_ctx_t *ctx, int fd, const void *buf, size_t count,
94 off_t offset, void (*callback)(int, ssize_t, void *), void *user_data, unsigned int opcode)
95{
96 struct io_uring_sqe *sqe;
97 brix_aio_callback_t *cb;
98
99 if (ctx == NULL || buf == NULL || callback == NULL) {
100 errno = EINVAL;
101 return -1;
102 }
103
104 sqe = io_uring_get_sqe(&ctx->ring);
105 if (sqe == NULL) {
106 errno = EAGAIN;
107 return -1;
108 }
109
110 cb = calloc(1, sizeof(brix_aio_callback_t));
111 if (cb == NULL) {
112 errno = ENOMEM;
113 return -1;
114 }
115
116 cb->callback = callback;
117 cb->user_data = user_data;
118 cb->fd = fd;
119
120 io_uring_prep_rw(opcode, sqe, fd, buf, count, offset);
121 io_uring_sqe_set_data(sqe, cb);
122
123 ngx_atomic_fetch_add(&ctx->pending_ops, 1);
124
125 return 0;
126}

144/* ---- Submit queued Linux AIO requests and wait for their completions ----
145 *
146 * WHAT: Return the number completed, or -1 on a submission/wait error.
147 * WHY: liburing requires an actual timespec pointer for finite timeouts.
148 * HOW: 1. Submit prepared requests and check the submission result.
149 * 2. Wait with a converted timeout, or without a deadline when negative.
150 * 3. Dispatch and consume each completed request.
151 */
152int
153brix_aio_wait(brix_aio_ctx_t *ctx, int timeout_ms)
154{
155 struct io_uring_cqe *cqe;
156 struct __kernel_timespec timeout;
157 int ret;
158 int completed = 0;
159
160 if (ctx == NULL) {
161 errno = EINVAL;
162 return -1;
163 }
164
165 ret = io_uring_submit(&ctx->ring);
166 if (ret < 0) {
167 errno = -ret;
168 return -1;
169 }
170
171 timeout.tv_sec = timeout_ms / 1000;
172 timeout.tv_nsec = (timeout_ms % 1000) * 1000000L;
173
174 while (ctx->pending_ops > 0) {
175 if (timeout_ms < 0) {
176 ret = io_uring_wait_cqe(&ctx->ring, &cqe);
177 } else {
178 ret = io_uring_wait_cqe_timeout(&ctx->ring, &cqe, &timeout);
179 }
180
181 if (ret < 0) {
182 if (ret == -ETIME) {
183 return completed; /* Timeout */
184 }
185 if (ret == -EINTR) {
186 continue; /* Interrupted, retry */
187 }
188 errno = -ret;
189 return -1;
190 }
191
192 /* Process completion */
193 brix_aio_io_uring_callback(cqe);
194 io_uring_cqe_seen(&ctx->ring, cqe);
195 ctx->pending_ops--;
196 completed++;
197 }
198
199 return completed;
200}
