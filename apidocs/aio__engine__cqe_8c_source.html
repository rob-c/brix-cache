9/* ---- Translate one slot-poll CQE into a readiness event ----
10 *
11 * WHAT: Given a poll CQE whose user_data is `ud` (slot-generation in the high
12 * 32 bits, slot index in the low 32), validate it against the live slot table
13 * and, if current, fill *ev with the aconn pointer + EPOLL* mask and re-arm the
14 * multishot poll if it auto-disarmed. Returns 1 if *ev was written, 0 if the
15 * CQE was stale/cancelled and dropped. Does NOT mark the CQE seen (caller does).
16 *
17 * WHY: Preserves the UAF discipline shared with the server ring — a poll that
18 * completes after its aconn was reconnected or freed carries an outdated
19 * generation and must be dropped. Isolating the guard + translation keeps the
20 * wait loop within complexity limits without altering the drop semantics.
21 *
22 * HOW:
23 * 1. Split `ud` into slot index and generation.
24 * 2. Drop (return 0) unless the slot is in range, in use, generation-matched,
25 * still bound to an aconn, and the CQE result is non-negative.
26 * 3. Map poll bits to EPOLL* bits: POLLIN|POLLHUP|POLLERR -> EPOLLIN,
27 * POLLOUT -> EPOLLOUT; write aconn pointer + mask into *ev.
28 * 4. If the multishot auto-disarmed (no IORING_CQE_F_MORE), re-arm it.
29 * 5. Return 1.
30 */
31static int
32uring_translate_slot_cqe(brix_loop *l, struct io_uring_cqe *cqe,
33 struct epoll_event *ev, uint64_t ud)
34{
35 uint32_t slot = AIO_UD_SLOT(ud);
36 uint32_t gen = AIO_UD_GEN(ud);
37 brix_aconn *ac;
38 uint32_t ev_mask = 0;
39
40 if (!(slot < AIO_URING_SLOTS && l->uslots[slot].in_use
41 && l->uslots[slot].gen == gen
42 && l->uslots[slot].ac != NULL && cqe->res >= 0)) {
43 return 0; /* stale/cancelled CQE for a recycled slot — dropped. */
44 }
45
46 ac = l->uslots[slot].ac;
47 if (cqe->res & (POLLIN | POLLHUP | POLLERR)) { ev_mask |= EPOLLIN; }
48 if (cqe->res & POLLOUT) { ev_mask |= EPOLLOUT; }
49 ev->data.ptr = ac;
50 ev->events = ev_mask;
51 /* Re-arm if the multishot auto-disarmed (no F_MORE). */
52 if (!(cqe->flags & IORING_CQE_F_MORE)) {
53 (void) uring_poll_submit(l, ac, ac->epoll_events);
54 }
55 return 1;
56}

59/* ---- P44-C: complete one multishot RECV CQE ----
60 *
61 * WHAT: Validate the CQE against the slot table, append the provided buffer's
62 * bytes to ac->rbuf, recycle the buffer, and parse — the CQE *is* the read, so
63 * no readiness event is produced (always returns 0 events).
64 *
65 * WHY: This is the ii-b replacement for the plaintext arm of aconn_do_read:
66 * completions carry the bytes, so calling read(2) as well would race the
67 * kernel for the socket's byte stream. Error/EOF handling mirrors
68 * aconn_read_plain exactly (parse buffered frames first, then fail).
69 *
70 * HOW: stale gen -> recycle buffer + drop. res > 0 -> copy into rbuf, recycle,
71 * parse. res == 0 -> peer close. -ENOBUFS -> transient (buffers were all in
72 * flight); just re-arm. Other negatives -> transport error. If the multishot
73 * auto-disarmed (!F_MORE) and the conn survived, re-arm it.
74 */
75/* Validate a RECV CQE against the slot table and return the live conn, or NULL
76 * (recycling the rx buffer) when the slot is stale — the conn re-armed or was
77 * torn down while this completion sat in the CQ ring. */
78static brix_aconn *
79uring_rxtx_recv_resolve(brix_loop *l, struct io_uring_cqe *cqe, uint64_t ud)
80{
81 uint32_t slot = AIO_UD_SLOT(ud);
82
83 if (slot < AIO_URING_SLOTS && l->uslots[slot].in_use
84 && l->uslots[slot].gen == AIO_UD_GEN(ud)
85 && l->uslots[slot].ac != NULL) {
86 return l->uslots[slot].ac;
87 }
88 uring_rxtx_recycle(l, cqe); /* stale: never leak the rx buffer */
89 return NULL;
90}

92/* Append a positive-length RECV CQE's provided-buffer bytes to ac->rbuf,
93 * recycle the buffer, and parse. Returns -1 (transport error already posted)
94 * if the read buffer can't grow, 0 otherwise. */
95static int
96uring_rxtx_recv_data(brix_loop *l, brix_aconn *ac, struct io_uring_cqe *cqe)
97{
98 brix_status st;
99
100 if (xbuf_reserve(&ac->rbuf, (size_t) cqe->res) != 0) {
101 uring_rxtx_recycle(l, cqe);
102 brix_status_set(&st, XRDC_EPROTO, 0, "out of memory (read buffer)");
103 aconn_on_transport_error(ac, &st);
104 return -1;
105 }
106 if (cqe->flags & IORING_CQE_F_BUFFER) {
107 uint16_t bid = (uint16_t) (cqe->flags >> IORING_CQE_BUFFER_SHIFT);
108 memcpy(ac->rbuf.buf + ac->rbuf.len,
109 l->brx_pool + (size_t) bid * AIO_URING_RXBUF_SZ,
110 (size_t) cqe->res);
111 ac->rbuf.len += (size_t) cqe->res;
112 }
113 uring_rxtx_recycle(l, cqe); /* copy done — buffer back to kernel */
114 aconn_parse(ac);
115 return 0;
116}

118static void
119uring_rxtx_recv_cqe(brix_loop *l, struct io_uring_cqe *cqe, uint64_t ud)
120{
121 uint32_t slot = AIO_UD_SLOT(ud);
122 brix_aconn *ac;
123 brix_status st;
124
125 ac = uring_rxtx_recv_resolve(l, cqe, ud);
126 if (ac == NULL) {
127 return;
128 }
129
130 if (cqe->res > 0) {
131 if (uring_rxtx_recv_data(l, ac, cqe) != 0) {
132 return;
133 }
134 } else if (cqe->res == 0) {
135 brix_status_set(&st, XRDC_ESOCK, 0, "connection closed by peer");
136 aconn_parse(ac); /* deliver any final complete frame */
137 if (!ac->dead) {
138 aconn_on_transport_error(ac, &st);
139 }
140 return;
141 } else if (cqe->res != -ENOBUFS && cqe->res != -EAGAIN
142 && cqe->res != -EINTR) {
143 brix_status_set(&st, XRDC_ESOCK, -cqe->res, "read: %s",
144 strerror(-cqe->res));
145 aconn_on_transport_error(ac, &st);
146 return;
147 }
148
149 /* Re-arm if the multishot auto-disarmed and the conn is still with us
150 * (parse may have killed it). */
151 if (!(cqe->flags & IORING_CQE_F_MORE)
152 && slot < AIO_URING_SLOTS && l->uslots[slot].in_use
153 && l->uslots[slot].gen == AIO_UD_GEN(ud) && !ac->dead) {
154 l->uslots[slot].recv_armed = 0;
155 (void) uring_recv_submit(l, ac);
156 }
157}

160/* ---- P44-C: complete one one-shot SEND CQE ----
161 *
162 * WHAT: Advance ac->wbuf.start by the sent byte count — exactly what
163 * aconn_do_write does on a send(2) return — and submit the next staged slice
164 * if the queue is still non-empty (short send, or frames appended while this
165 * SEND was in flight).
166 *
167 * HOW: stale gen -> drop (the cancel path already cleared send_inflight).
168 * res < 0: EINTR/EAGAIN retries the same slice; anything else is a transport
169 * error (EPIPE from a dead peer lands here — MSG_NOSIGNAL semantics).
170 */
171static void
172uring_rxtx_send_cqe(brix_loop *l, struct io_uring_cqe *cqe, uint64_t ud)
173{
174 uint32_t slot = AIO_UD_SLOT(ud);
175 brix_aconn *ac;
176 brix_status st;
177
178 if (!(slot < AIO_URING_SLOTS && l->uslots[slot].in_use
179 && l->uslots[slot].gen == AIO_UD_GEN(ud)
180 && l->uslots[slot].ac != NULL)) {
181 return; /* stale: conn re-armed or gone */
182 }
183 ac = l->uslots[slot].ac;
184 l->uslots[slot].send_inflight = 0;
185 l->uslots[slot].stage_len = 0;
186
187 if (cqe->res < 0) {
188 if (cqe->res == -EINTR || cqe->res == -EAGAIN) {
189 (void) uring_rxtx_send_submit(l, ac); /* retry the slice */
190 return;
191 }
192 brix_status_set(&st, XRDC_ESOCK, -cqe->res, "write: %s",
193 strerror(-cqe->res));
194 aconn_on_transport_error(ac, &st);
195 return;
196 }
197
198 ac->wbuf.start += (size_t) cqe->res;
199 if (ac->wbuf.start >= ac->wbuf.len) {
200 ac->wbuf.start = ac->wbuf.len = 0; /* fully flushed */
201 } else {
202 (void) uring_rxtx_send_submit(l, ac); /* short send / more queued */
203 }
204}

207/* ---- Translate one CQE into at most one readiness event ----
208 *
209 * WHAT: Classify a single CQE by its user_data and, for the evfd wake and live
210 * slot polls, fill *ev with the corresponding readiness event. Returns 1 if
211 * *ev was written, 0 otherwise. Always marks the CQE seen before returning.
212 *
213 * WHY: Factors the user_data classification (ignore / evfd / slot-op) out of
214 * the wait loop so each case is a single linear path; the loop then just
215 * accumulates the return count. The seen-marking is centralised here so every
216 * consumed CQE is retired exactly once regardless of classification.
217 *
218 * HOW:
219 * 1. Read the CQE's user_data.
220 * 2. AIO_URING_IGNORE_UD (a cancel echo): mark seen, return 0.
221 * 3. AIO_URING_EVFD_UD (cross-thread wake): write a loop-tagged EPOLLIN event,
222 * mark seen, return 1.
223 * 4. Otherwise dispatch on the packed op kind: RECV/SEND CQEs (P44-C) carry
224 * the bytes and are completed in place (0 events); a poll CQE goes through
225 * uring_translate_slot_cqe. Mark seen, return the result.
226 */
227static int
228uring_translate_cqe(brix_loop *l, struct io_uring_cqe *cqe,
229 struct epoll_event *ev)
230{
231 uint64_t ud = io_uring_cqe_get_data64(cqe);
232 int produced = 0;
233
234 if (ud == AIO_URING_IGNORE_UD) {
235 io_uring_cqe_seen(&l->uring, cqe);
236 return 0;
237 }
238 if (ud == AIO_URING_EVFD_UD) {
239 ev->data.ptr = l;
240 ev->events = EPOLLIN;
241 io_uring_cqe_seen(&l->uring, cqe);
242 return 1;
243 }
244
245 switch (AIO_UD_KIND(ud)) {
246 case AIO_UD_RECV:
247 uring_rxtx_recv_cqe(l, cqe, ud);
248 break;
249 case AIO_UD_SEND:
250 uring_rxtx_send_cqe(l, cqe, ud);
251 break;
252 default:
253 produced = uring_translate_slot_cqe(l, cqe, ev, ud);
254 break;
255 }
256 io_uring_cqe_seen(&l->uring, cqe);
257 return produced;
258}

261/* ---- Drain ready io_uring poll CQEs into evs[] ----
262 *
263 * WHAT: Block up to `timeout_ms` for at least one CQE, then peek and translate
264 * up to `max` CQEs into evs[], returning the number of readiness events written
265 * (never negative — a timeout yields 0).
266 *
267 * WHY: Mirrors epoll_wait's contract for the io_uring engine so io_engine_wait
268 * can select an engine and return, staying a flat two-branch function.
269 *
270 * HOW:
271 * 1. Convert timeout_ms to a __kernel_timespec and wait for one CQE; on any
272 * wait error (timeout / -ETIME / -EINTR) return 0.
273 * 2. While there is room (n < max) and a CQE can be peeked, translate it and
274 * add its 0/1 event contribution to n.
275 * 3. Return n.
276 */
277static int
278uring_drain_cqes(brix_loop *l, struct epoll_event *evs, int max, int timeout_ms)
279{
280 struct io_uring_cqe *cqe;
281 struct __kernel_timespec ts;
282 int n = 0;
283
284 ts.tv_sec = timeout_ms / 1000;
285 ts.tv_nsec = (long) (timeout_ms % 1000) * 1000000L;
286
287 /* Block until at least one CQE or the timeout. */
288 if (io_uring_wait_cqe_timeout(&l->uring, &cqe, &ts) < 0) {
289 return 0; /* timeout / -ETIME / -EINTR: no events this tick */
290 }
291
292 while (n < max && io_uring_peek_cqe(&l->uring, &cqe) == 0) {
293 n += uring_translate_cqe(l, cqe, &evs[n]);
294 }
295 return n;
296}

301/* Fill evs[] with up to `max` readiness events (data.ptr + EPOLL* mask) and
302 * return the count, or -1 on a hard wait error. For io_uring, translate poll
303 * CQEs (generation-guarded) and re-arm any multishot that auto-disarmed. */
304int
305io_engine_wait(brix_loop *l, struct epoll_event *evs, int max, int timeout_ms)
306{
307#if (BRIX_HAVE_LIBURING)
308 if (l->use_uring) {
309 return uring_drain_cqes(l, evs, max, timeout_ms);
310 }
311#endif
312 return epoll_wait(l->epfd, evs, max, timeout_ms);
