1/*
2 * oci_module.c — nginx wiring for the OCI Distribution plane.
3 *
4 * WHAT: the config lifecycle head (create_loc_conf + the UNSET sentinels the
5 * merge reads), the four directive setters that cannot be a stock slot
6 * writer, the `$oci_class` log variable, the SHM-zone
7 * postconfiguration, the directive table and the module record.
8 * WHY: the shape is deliberately the cvmfs module's, because the lifecycle
9 * question is the same one: a location becomes a protocol endpoint the
10 * moment its enabling directive is parsed, and everything else — the
11 * metrics zone, the dashboard zones, the fill thread pool — must exist
12 * for that endpoint whether or not a stream{} block was ever configured.
13 * HOW: create sets sentinels and nothing else (the merge owns every default),
14 * the merge lives in oci_merge.c, and $oci_class reads the request ctx
15 * the handler fills. The disposition ($oci_cache) moved to the shared
16 * $brix_cache_status in phase 112; its five-value vocabulary survives as
17 * the brix_oci_requests_total{outcome} labels (Appendix J.6), so the
18 * access log and the metric family still cannot disagree.
19 */

76/* "brix_oci_mirror <base-url>" — record the upstream AND make this location
77 * the /v2/ endpoint. The URL is only stored here; parsing (and every refusal
78 * that depends on it) happens at merge time, when inheritance has settled. */
79static char *
80oci_conf_mirror(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
81{
82 ngx_http_brix_oci_loc_conf_t *lcf = conf;
83 ngx_http_core_loc_conf_t *clcf;
84 ngx_str_t *value = cf->args->elts;
85
86 (void) cmd;
87 if (lcf->mirror != NGX_CONF_UNSET) {
88 return "is duplicate";
89 }
90
91 lcf->mirror = 1;
92 lcf->mirror_url = value[1];
93
94 clcf = ngx_http_conf_get_module_loc_conf(cf, ngx_http_core_module);
95 clcf->handler = ngx_http_brix_oci_handler;
96
97 return NGX_CONF_OK;
98}

101/* "brix_oci_mirror_auth <user> <password-file>" — both halves land in the
102 * loc conf as plain strings; the file is opened, permission-checked and read
103 * at merge time so a bad credential is an nginx -t failure. */
104static char *
105oci_conf_mirror_auth(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
106{
107 ngx_http_brix_oci_loc_conf_t *lcf = conf;
108 ngx_str_t *value = cf->args->elts;
109
110 (void) cmd;
111 if (lcf->mirror_user.len > 0) {
112 return "is duplicate";
113 }
114 if (value[1].len == 0 || value[2].len == 0) {
115 return "requires a non-empty user and password file";
116 }
117
118 lcf->mirror_user = value[1];
119 lcf->mirror_pwfile = value[2];
120
121 return NGX_CONF_OK;
122}

125/* "brix_oci_upstream_auth_realm <host>" — widen the realm trust boundary by
126 * exactly one host, repeatably (§D15.11).
127 *
128 * The derived rule (upstream host, its registrable parent, a sibling under
129 * that parent) covers every registry that hosts its own token service, and a
130 * site whose registry delegates to an unrelated identity host would otherwise
131 * be unmirrorable — which is how a check like this gets deleted rather than
132 * configured. Each entry is validated HERE so a typo is an nginx -t failure
133 * and not a refusal at 3am, and validated by the same authority parser a
134 * realm goes through, so an entry can only name something a realm could
135 * spell. There is no wildcard form on purpose. */
136static char *
137oci_conf_auth_realm(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
138{
139 ngx_http_brix_oci_loc_conf_t *lcf = conf;
140 ngx_str_t *value = cf->args->elts;
141 int rc;
142
143 (void) cmd;
144 if (lcf->auth_realms == NULL) {
145 BRIX_PCALLOC_OR_RETURN(lcf->auth_realms, cf->pool,
146 sizeof(*lcf->auth_realms), NGX_CONF_ERROR);
147 }
148
149 rc = brix_oci_realm_list_add(lcf->auth_realms,
150 (const char *) value[1].data, value[1].len);
151 if (rc == -2) {
152 return "has more entries than the allowlist holds";
153 }
154 if (rc == -3) {
155 return "is a duplicate";
156 }
157 if (rc != 0) {
158 return "must be one bare host - no scheme, no port, no wildcard";
159 }
160
161 return NGX_CONF_OK;
162}

184/* "brix_oci_token_zone <name> <size>" — sugar over the generic KV plane with
185 * the D1.3 geometry pinned: a 32-byte sha256 key, a 4 KiB value. Pinning them
186 * here rather than exposing key=/val= is the point of the sugar — a zone whose
187 * value cap is under a real DockerHub JWT would silently cache nothing.
188 *
189 * The name is also written into the location config so it inherits down the
190 * location tree by nginx's own rules: written in http{}, every mirror below
191 * uses it; written in a location, only that subtree does. */
192static char *
193oci_conf_token_zone(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
194{
195 ngx_http_brix_oci_loc_conf_t *lcf = conf;
196 ngx_str_t *value = cf->args->elts;
197 ngx_str_t name = value[1];
198 brix_kv_t *kv;
199 ssize_t size;
200
201 (void) cmd;
202
203 lcf->token_zone_name = name;
204 lcf->token_zone_set = 1;
205
206 size = ngx_parse_size(&value[2]);
207 if (size == NGX_ERROR || size <= 0) {
208 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
209 "invalid brix_oci_token_zone size \"%V\"",
210 &value[2]);
211 return NGX_CONF_ERROR;
212 }
213
214 if (brix_kv_find(&name) != NULL) {
215 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
216 "duplicate brix_oci_token_zone \"%V\"", &name);
217 return NGX_CONF_ERROR;
218 }
219
220 BRIX_PCALLOC_OR_RETURN(kv, cf->pool, sizeof(*kv), NGX_CONF_ERROR);
221
222 if (brix_kv_configure(cf, kv, &name, (size_t) size,
223 BRIX_OCI_TOKEN_KEYLEN, BRIX_OCI_TOKEN_MAX,
224 &ngx_http_brix_oci_module) != NGX_OK)
225 {
226 return NGX_CONF_ERROR;
227 }
228
229 return NGX_CONF_OK;
230}

235static ngx_int_t
236oci_var_set(ngx_http_request_t *r, ngx_http_variable_value_t *v,
237 const char *val)
238{
239 size_t n = ngx_strlen(val);
240 u_char *p = ngx_pnalloc(r->pool, n);
241
242 if (p == NULL) {
243 return NGX_ERROR;
244 }
245 ngx_memcpy(p, val, n);
246 v->data = p;
247 v->len = n;
248 v->valid = 1;
249 v->no_cacheable = 0;
250 v->not_found = 0;
251 return NGX_OK;
252}

290{
291 ngx_http_variable_t *v, *nv;
292
293 for (v = ngx_http_brix_oci_vars; v->name.len; v++) {
294 nv = ngx_http_add_variable(cf, &v->name, v->flags);

298 nv->get_handler = v->get_handler;
