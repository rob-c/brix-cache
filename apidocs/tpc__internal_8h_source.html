1/* tpc_internal.h — native TPC destination-side pull types and internal API.
2 * Requires nginx and BriX types provided by the includes below.
3 * WHAT: Share parsed opaque fields and the heap-owned pull task across stages.
4 * WHY: Event-loop setup, blocking pool work and completion need one ownership
5 * contract. Allocate before posting; free only from the completion callback.
6 * HOW: Cached policy checks precede asynchronous DNS and connection setup.
7 * Source-open resolution bounds wait/waitresp retries, body sizes and hops.
8 * The implementation files below own parsing, transport, authentication,
9 * delegated-token fetching and the final publish callback.

39/* Async kXR_open resolution (phase-57 §F8): a real source (EOS/dCache, or any
40 * server still completing the TPC rendezvous) may answer the open with kXR_wait
41 * (retry-after) or kXR_waitresp (a deferred kXR_attn asynresp will follow) before
42 * it settles. tpc_open_resolve() honours that flow, bounded on every axis so a
43 * source that never resolves fails cleanly instead of hanging the pull thread. */
44#define TPC_OPEN_RESOLVE_MAX_SEC 120 /* total wall-clock cap for the negotiation */
45#define TPC_OPEN_WAIT_CAP_SEC 15 /* clamp the per-recv idle timeout during open
46 * resolution, so a silent (no-attn) source
47 * fails fast enough for the client's --tpc
48 * fallback to run */
49#define TPC_OPEN_WAIT_RETRY_SEC 1 /* how long to sleep before RESENDing an open
50 * after a kXR_wait. A real xrootd source can
51 * request a multi-second wait while the TPC
52 * grant is only briefly visible to our racing
53 * dest-open; sleeping the server's full hint
54 * (up to CAP_SEC) misses the window and the
55 * grant lapses ("tpc authorization expired").
56 * Stock XrdCl re-polls sub-second, so cap our
57 * retry sleep here (MAX_ITERS/MAX_SEC still
58 * bound the total) and re-poll while valid. */
59#define TPC_OPEN_RESOLVE_MAX_ITERS 16 /* max wait/waitresp/attn rounds */

473/*
474 * A queued response is not a return value. brix_send_error() returns NGX_OK once
475 * the kXR_error is on the wire, so a function that ends `return
476 * brix_send_error(...)` reports SUCCESS to whoever called it. At a protocol entry
477 * point that is exactly right: NGX_OK there means "request handled" and nothing
478 * runs afterwards. One call deep it is a gate bypass — the caller reads NGX_OK as
479 * "the check passed, carry on" and carries on past a refusal it has already
480 * answered, allocating the handle and queueing a second, contradictory response.
481 * The TPC prepare path therefore says TPC_ANSWERED — "refused, and the client has
482 * been told" — and only brix_tpc_prepare_pull, the entry point where the
483 * distinction stops mattering, folds it back into the wire contract's NGX_OK.
484 */
