25/* The kXR_fattr protocol handler stores a user attribute "X" under the on-disk key
26 * "user.U.X" (BRIX_FATTR_XKEY_PFX, applied ABOVE the VFS). When we forward to
27 * ANOTHER xrootd server it re-applies the SAME mapping, so handing it the already-
28 * mapped key would double-prefix it on the origin ("user.U.user.U.X") and break
29 * direct-origin interop. So strip one "user.U." before forwarding get/set/remove,
30 * and re-add it on list — the origin then carries a single, standard "user.U.X".
31 * (Kept in sync with src/fattr/ngx_brix_fattr.h rather than #included, to avoid a
32 * backend→protocol-handler dependency. Names from other consumers — webdav locks/
33 * dead-props, s3 tags — have no "user.U." prefix and pass through unchanged.) */
34/* SD_XROOT_FATTR_PFX / _LEN live in sd_xroot_internal.h — shared with the
35 * credential-scoped wrappers (sd_xroot_ns_cred.c). */
36const char *
37sd_xroot_fattr_unmap(const char *name)
38{
39 if (strncmp(name, SD_XROOT_FATTR_PFX, SD_XROOT_FATTR_PFX_LEN) == 0) {
40 return name + SD_XROOT_FATTR_PFX_LEN;
41 }
42 return name;
43}

45/* WHAT: Report whether a fallback_deny credential must be refused outright.
46 * WHY: A fallback_deny cred whose selected kind (e.g. S3 ak/sk) this driver
47 * cannot present carries neither x509_proxy nor bearer. Falling back to
48 * the static service credential would silently serve the request under
49 * the wrong identity, so it must be refused before any session is opened.
50 * Every *_cred namespace wrapper (unlink_cred, rename_cred, opendir_cred,
51 * ...) reaches this via the shared sd_xroot_session choke point.
52 * HOW: True only when cred is non-NULL, has fallback_deny set, and carries
53 * none of the driver's presentable kinds (x509_proxy, bearer, sss_keytab,
54 * krb5 forwarded-TGT ccache). NULL cred and creds with a presentable
55 * kind return false (allowed). */
56static int
57sd_xroot_cred_must_deny(const brix_sd_cred_t *cred)
58{
59 return cred != NULL && cred->fallback_deny
60 && (cred->x509_proxy == NULL || cred->x509_proxy[0] == '\0')
61 && (cred->bearer == NULL || cred->bearer[0] == '\0')
62 && (cred->sss_keytab == NULL || cred->sss_keytab[0] == '\0')
63 && (cred->krb5_ccache == NULL || cred->krb5_ccache[0] == '\0');
64}

66/* WHAT: Copy a per-user credential into a fill task so the origin bootstrap
67 * presents it at authentication instead of the static service cred.
68 * WHY: Namespace ops that carry a user credential must NOT fall back to the
69 * static service credential; this mirrors sd_xroot_origin_open.
70 * HOW: Exactly one of {x509_proxy, bearer, sss_keytab, krb5 ccache} is
71 * non-NULL for a credential-scoped session; copy whichever is set, plus
72 * principal (sss_keytab REQUIRES it — it is the asserted identity — and
73 * the krb5 leg carries krb5_princ, the origin service principal). Caller
74 * guarantees the
75 * task's cred_* fields are already zeroed (calloc), so a NULL cred is a
76 * no-op and the service-cred path is left unchanged. */
77static void
78sd_xroot_cred_copy(brix_cache_fill_t *t, const brix_sd_cred_t *cred)
79{
80 if (cred == NULL) {
81 return;
82 }
83 if (cred->x509_proxy != NULL && cred->x509_proxy[0] != '\0') {
84 ngx_cpystrn((u_char *) t->cred_x509_proxy, (u_char *) cred->x509_proxy,
85 sizeof(t->cred_x509_proxy));
86 }
87 if (cred->bearer != NULL && cred->bearer[0] != '\0') {
88 ngx_cpystrn((u_char *) t->cred_bearer, (u_char *) cred->bearer,
89 sizeof(t->cred_bearer));
90 }
91 if (cred->principal != NULL) {
92 ngx_cpystrn((u_char *) t->cred_principal,
93 (u_char *) cred->principal, sizeof(t->cred_principal));
94 }
95 if (cred->sss_keytab != NULL && cred->sss_keytab[0] != '\0') {
96 ngx_cpystrn((u_char *) t->cred_sss_keytab,
97 (u_char *) cred->sss_keytab, sizeof(t->cred_sss_keytab));
98 }
99 if (cred->krb5_ccache != NULL && cred->krb5_ccache[0] != '\0') {
100 ngx_cpystrn((u_char *) t->cred_krb5_ccache,
101 (u_char *) cred->krb5_ccache, sizeof(t->cred_krb5_ccache));
102 }
103 if (cred->krb5_princ != NULL && cred->krb5_princ[0] != '\0') {
104 ngx_cpystrn((u_char *) t->cred_krb5_princ,
105 (u_char *) cred->krb5_princ, sizeof(t->cred_krb5_princ));
106 }
107}

109/* Connect + bootstrap a fresh origin session (no file open) for a path-based op.
110 * When cred is non-NULL the bootstrap uses the per-user credential instead of
111 * the static service credential. Supports both x509 proxies (cred->x509_proxy)
112 * and WLCG bearer tokens (cred->bearer); they are mutually exclusive per the
113 * brix_sd_cred_t contract. NULL cred → service credential / anonymous.
114 * On success fills *oc + *t_out (caller closes oc + frees t); returns -1 with
115 * *err_out on failure.
116 *
117 * WHAT: Refuse an unpresentable fallback_deny cred, then allocate and wire a
118 * fill task, copying any per-user credential, and connect+bootstrap.
119 * WHY: This function opens its own origin session independent of
120 * sd_xroot_open_common, so the wrong-kind cred leak is reachable via
121 * every *_cred namespace wrapper unless checked at this shared point.
122 * HOW: sd_xroot_cred_must_deny gates unusable creds up front; calloc zeroes
123 * all cred_* fields so sd_xroot_cred_copy is a no-op for NULL cred.
124 * Prototype in sd_xroot_internal.h — shared with the credential-scoped
125 * wrappers (sd_xroot_ns_cred.c) and the directory-listing path
126 * (sd_xroot_ns_dir.c). */
127int
128sd_xroot_session(ngx_stream_brix_srv_conf_t *conf,
129 const brix_sd_cred_t *cred,
130 brix_cache_origin_conn_t *oc, brix_cache_fill_t **t_out, int *err_out)
131{
132 brix_cache_fill_t *t;
133
134 if (sd_xroot_cred_must_deny(cred)) {
135 if (err_out) { *err_out = EACCES; }
136 errno = EACCES;
137 return -1;
138 }
139
140 t = calloc(1, sizeof(*t));
141 if (t == NULL) {
142 if (err_out) { *err_out = ENOMEM; }
143 return -1;
144 }
145 oc->fd = -1;
146 t->conf = conf;
147
148 sd_xroot_cred_copy(t, cred);
149
150 if (brix_cache_origin_connect(t, oc) != 0
151 || brix_cache_origin_bootstrap(t, oc) != 0)
152 {
153 if (err_out) { *err_out = sd_xroot_errno(t); }
154 brix_cache_origin_close(oc);
155 free(t);
156 return -1;
157 }
158 *t_out = t;
159 return 0;
160}

246/* Export capacity from the ORIGIN (optional `space` vtable slot).
247 *
248 * WHAT: kXR_query/kXR_Qspace on a fresh anonymous session, decoded by the
249 * shared oss.* grammar into a brix_sd_space_t.
250 * WHY: A root:// gateway's statvfs/Qspace/QFSinfo/SRR answers described the
251 * LOCAL export directory's filesystem — meaningless for capacity planning
252 * when the bytes actually live on the origin. Any failure returns
253 * NGX_ERROR so brix_query_backend_usage / statvfs consumers fall back to
254 * the local statvfs rather than erroring the client op.
255 * HOW: sd_xroot_session (bounded by the origin socket timeouts, same cost
256 * class as sd_xroot_stat) → brix_cache_origin_query_space → close; used
257 * is derived as total − free since the oss.* report carries both. */
258ngx_int_t
259sd_xroot_space(brix_sd_instance_t *inst, brix_sd_space_t *out)
260{
261 sd_xroot_inst_state *is = inst->state;
262 brix_cache_origin_conn_t oc;
263 brix_cache_fill_t *t;
264 uint64_t total = 0, freeb = 0;
265 int rc, e = 0;
266
267 if (sd_xroot_session(is->conf, NULL, &oc, &t, &e) != 0) {
268 errno = e;
269 return NGX_ERROR;
270 }
271 rc = brix_cache_origin_query_space(t, &oc, &total, &freeb);
272 e = errno;
273 brix_cache_origin_close(&oc);
274 free(t);
275 if (rc != 0) {
276 errno = e;
277 return NGX_ERROR;
278 }
279 out->total_bytes = total;
280 out->free_bytes = freeb;
281 out->used_bytes = total > freeb ? total - freeb : 0;
282 return NGX_OK;
283}

285/* Copy src→dst byte stream on an open session (read each chunk from src_fh, write
286 * to dst_fh), then truncate+sync dst. Returns NGX_OK + *bytes_out, or NGX_ERROR.
287 * Prototype in sd_xroot_internal.h — shared with the credential-scoped
288 * server_copy_cred wrapper (sd_xroot_ns_cred.c). */
289ngx_int_t
290sd_xroot_copy_body(brix_cache_fill_t *t, brix_cache_origin_conn_t *oc,
291 const u_char *src_fh, const u_char *dst_fh, off_t *bytes_out)
292{
293 const size_t cap = 1u << 20;
294 u_char *buf = malloc(cap);
295 off_t off = 0;
296
297 if (buf == NULL) { errno = ENOMEM; return NGX_ERROR; }
298
299 for (;;) {
300 brix_cache_sink_t sink;
301 brix_cache_read_range_t rng;
302
303 ngx_memzero(&sink, sizeof(sink));
304 sink.fd = -1;
305 sink.mem = buf;
306 sink.mem_cap = cap;
307
308 ngx_memzero(&rng, sizeof(rng));
309 rng.read_off = (uint64_t) off;
310 rng.want = cap;
311
312 if (brix_cache_origin_read_chunk(t, oc, src_fh, &sink, &rng) != 0) {
313 free(buf); errno = EIO; return NGX_ERROR;
314 }
315 if (rng.got == 0) {
316 break;
317 }
318 if (brix_cache_origin_write_chunk(t, oc, dst_fh, (uint64_t) off, buf,
319 rng.got) != 0)
320 {
321 free(buf); errno = EIO; return NGX_ERROR;
322 }
323 off += (off_t) rng.got;
324 if (rng.got < cap) {
325 break; /* short read = EOF */
326 }
327 }
328 free(buf);
329
330 if (brix_cache_origin_truncate(t, oc, dst_fh, (uint64_t) off) != 0
331 || brix_cache_origin_sync(t, oc, dst_fh) != 0)
332 {
333 errno = EIO;
334 return NGX_ERROR;
335 }
336 if (bytes_out) { *bytes_out = off; }
337 return NGX_OK;
338}
