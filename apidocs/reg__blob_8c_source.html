1/* reg_blob.c — blob data plane: redirect-following digest-verified fetch
2 * and the D4 upload-session push (see reg_client.h).
3 *
4 * The fetch streams over the http_internal seams (zero new socket code):
5 * hand-built GET → read_resp_headers → Location on 3xx / dance on 401 /
6 * httpx_download_body on 2xx. Authorization is stripped on every redirect
7 * leg (the classic token-leak bug class, pinned by the D5.5 negative).
8 * Verification is a sequential read-back of the just-written temp (one
9 * page-cache-warm pass) — a true socket-side tee would mean re-implementing
10 * the chunked/CL framing the seams already own. */

27/* Hash the temp back and compare; mismatch truncates the fd (no partial
28 * bytes survive) — the caller owns the path and unlinks on failure. */
29static int
30blob_verify(int fd, const brix_oci_digest_t *want, char *err, size_t errlen)
31{
32 brix_oci_hash_ctx_t c;
33 brix_oci_digest_t got;
34 const char *alg = brix_oci_alg_name(want->alg);
35 unsigned char buf[65536];
36 ssize_t n;
37
38 /* Hash under the algorithm the descriptor names, never a fixed one:
39 * a sha512 layer summed with sha256 fails every honest fetch. */
40 if (lseek(fd, 0, SEEK_SET) != 0
41 || brix_oci_hash_init(&c, want->alg) != 0)
42 {
43 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
44 "blob verify: temp not seekable / %s init", alg);
45 }
46 while ((n = read(fd, buf, sizeof(buf))) > 0) {
47 if (brix_oci_hash_update(&c, buf, (size_t) n) != 0) {
48 brix_oci_hash_abort(&c);
49 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
50 "blob verify: %s update failed", alg);
51 }
52 }
53 if (n < 0) {
54 brix_oci_hash_abort(&c);
55 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
56 "blob verify: read-back failed");
57 }
58 if (brix_oci_hash_final(&c, &got) != 0) {
59 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
60 "blob verify: %s final failed", alg);
61 }
62 if (!brix_oci_digest_eq(&got, want)) {
63 if (ftruncate(fd, 0) != 0) {
64 /* best-effort scrub: the caller unlinks the temp anyway */
65 }
66 return regc_fail(err, errlen, BRIX_OCI_REG_EVERIFY,
67 "blob digest mismatch: got %s:%s want %s:%s",
68 brix_oci_alg_name(got.alg), got.hex,
69 alg, want->hex);
70 }
71 return BRIX_OCI_REG_OK;
72}

81/* Classify a received response before deciding which hop action owns it. */
82static blob_reply_t
83blob_reply_action(int status, int at_origin, int danced)
84{
85 if (status == 401 && at_origin && !danced) {
86 return BLOB_REPLY_AUTH;
87 }
88 if (status >= 300 && status < 400) {
89 return BLOB_REPLY_REDIRECT;
90 }
91 if (status >= 200 && status < 300) {
92 return BLOB_REPLY_DOWNLOAD;
93 }
94 return BLOB_REPLY_ERROR;
95}

106static int
107blob_get_headers(brix_oci_reg_t *r, const char *host, int port, int tls,
108 const char *path, const char *scope, brix_io *io,
109 void **tls_ctx, char **hdr, size_t *total, size_t *body_off,
110 int *status, brix_status *st, char *err, size_t errlen)
111{
112 char auth[4200], req[8192], host_port[300];
113 int req_len;
114
115 memset(st, 0, sizeof(*st));
116 if (httpx_connect(io, host, port, tls, r->verify, r->ca_dir,
117 r->client_cert, r->timeout_ms, tls_ctx, st) != 0) {
118 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
119 "connect %s: %s", host, st->msg);
120 }
121 auth[0] = '\0';
122 if (strcmp(host, r->host) == 0) {
123 regc_auth_header(r, scope, auth, sizeof(auth));
124 }
125 brix_format_host_port(host, (uint16_t) port, host_port, sizeof(host_port));
126 req_len = snprintf(req, sizeof(req),
127 "GET %s HTTP/1.1\r\nHost: %s\r\n"
128 "User-Agent: brixoci\r\n"
129 "Accept: application/octet-stream\r\n"
130 "Connection: close\r\n%s\r\n", path, host_port, auth);
131 if (req_len < 0 || (size_t) req_len >= sizeof(req)) {
132 hop_close(io, tls, *tls_ctx);
133 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
134 "blob request too long");
135 }
136 if (brix_write_full(io, req, (size_t) req_len, st) != 0) {
137 hop_close(io, tls, *tls_ctx);
138 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
139 "blob GET send: %s", st->msg);
140 }
141 *hdr = malloc(XRDC_HDR_CAP);
142 if (*hdr == NULL) {
143 hop_close(io, tls, *tls_ctx);
144 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT, "out of memory");
145 }
146 if (read_resp_headers(io, *hdr, XRDC_HDR_CAP, r->timeout_ms, status,
147 total, body_off, st) != 0) {
148 blob_hop_free(io, tls, *tls_ctx, *hdr);
149 *hdr = NULL;
150 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
151 "blob GET: %s", st->msg);
152 }
153 return BRIX_OCI_REG_OK;
154}

216int
217brix_oci_reg_blob_fetch(brix_oci_reg_t *r, const char *name,
218 const char *digest, int out_fd, char *err,
219 size_t errlen)
220{
221 brix_oci_digest_t want;
222 char ename[512], path[2048], scope[600], host[256];
223 int port, tls, hop, danced = 0, rc;
224
225 if (brix_oci_digest_parse(digest, strlen(digest), &want) != 0) {
226 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
227 "invalid blob digest \"%s\"", digest);
228 }
229 if (regc_eff_name(r, name, ename, sizeof(ename)) != 0) {
230 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
231 "repository name too long");
232 }
233 snprintf(path, sizeof(path), "/v2/%s/blobs/%s:%s", ename,
234 brix_oci_alg_name(want.alg), want.hex);
235 snprintf(scope, sizeof(scope), "repository:%s:pull", ename);
236 snprintf(host, sizeof(host), "%s", r->host);
237 port = r->port;
238 tls = !r->plain_http;
239
240 for (hop = 0; hop <= 4; hop++) {
241 brix_io io;
242 brix_status st;
243 void *tls_ctx = NULL;
244 char *hdr = NULL;
245 size_t total = 0, body_off = 0;
246 int status = 0;
247 int at_origin = strcmp(host, r->host) == 0;
248 blob_reply_t action;
249
250 rc = blob_get_headers(r, host, port, tls, path, scope, &io,
251 &tls_ctx, &hdr, &total, &body_off, &status,
252 &st, err, errlen);
253 if (rc != BRIX_OCI_REG_OK) {
254 return rc;
255 }
256 action = blob_reply_action(status, at_origin, danced);
257 if (action == BLOB_REPLY_AUTH) {
258 rc = blob_auth_retry(r, hdr, scope, &danced, err, errlen);
259 blob_hop_free(&io, tls, tls_ctx, hdr);
260 if (rc != BRIX_OCI_REG_OK) {
261 return rc;
262 }
263 hop--; /* the auth retry is not a redirect hop */
264 continue;
265 }
266 if (action == BLOB_REPLY_REDIRECT) {
267 char location[2048];
268 int have_location = raw_header(hdr, "Location", location,
269 sizeof(location));
270
271 blob_hop_free(&io, tls, tls_ctx, hdr);
272 if (!have_location) {
273 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
274 "redirect without Location");
275 }
276 rc = blob_redirect(r, location, host, sizeof(host), &port, &tls,
277 path, sizeof(path), err, errlen);
278 if (rc != BRIX_OCI_REG_OK) {
279 return rc;
280 }
281 continue;
282 }
283 if (action == BLOB_REPLY_ERROR) {
284 blob_hop_free(&io, tls, tls_ctx, hdr);
285 return regc_status_fail(status, "blob GET", err, errlen);
286 }
287 return blob_download(r, &io, tls, tls_ctx, hdr, total, body_off,
288 out_fd, &want, &st, err, errlen);
289 }
290 return regc_fail(err, errlen, BRIX_OCI_REG_ETRANSPORT,
291 "too many redirects fetching blob");
292}

413int
414brix_oci_reg_blob_push(brix_oci_reg_t *r, const char *name,
415 const char *digest, int in_fd, size_t len,
416 char *err, size_t errlen)
417{
418 brix_oci_digest_t d;
419 char ename[512], scope[600], loc[2048];
420 char up[2600];
421 int exists, ltls, rc;
422
423 if (brix_oci_digest_parse(digest, strlen(digest), &d) != 0) {
424 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
425 "invalid blob digest \"%s\"", digest);
426 }
427 if (regc_eff_name(r, name, ename, sizeof(ename)) != 0) {
428 return regc_fail(err, errlen, BRIX_OCI_REG_EPROTO,
429 "repository name too long");
430 }
431 snprintf(scope, sizeof(scope), "repository:%s:push,pull", ename);
432
433 /* dedupe probe — content addressing makes re-pushing pointless */
434 rc = blob_push_probe(r, ename, &d, scope, &exists, err, errlen);
435 if (rc != BRIX_OCI_REG_OK) {
436 return rc;
437 }
438 if (exists) {
439 return BRIX_OCI_REG_OK;
440 }
441
442 /* open an upload session */
443 rc = blob_push_session(r, ename, scope, loc, sizeof(loc), err, errlen);
444 if (rc != BRIX_OCI_REG_OK) {
445 return rc;
446 }
447 /* single-shot seal: PUT the whole body with ?digest= (':' encoded) */
448 rc = blob_push_target(r, loc, &d, up, sizeof(up), &ltls, err, errlen);
449 if (rc != BRIX_OCI_REG_OK) {
450 return rc;
451 }
452 return blob_push_upload(r, scope, ltls, up, in_fd, len, err, errlen);
453}
