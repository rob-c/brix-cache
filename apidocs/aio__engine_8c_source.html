8/* ============ Phase 44: pluggable loop I/O engine (epoll | io_uring) ============
9 *
10 * Two readiness engines share one interface. The epoll branch is the historical
11 * code, preserved verbatim. The io_uring branch (multishot IORING_OP_POLL_ADD,
12 * default OFF — gated by XRDC_IO_URING_LOOP=on + a runtime probe, best-effort
13 * fallback to epoll) is a drop-in readiness source: the loop still runs
14 * aconn_do_read/aconn_do_write unchanged, so TLS (which drives the fd through
15 * OpenSSL itself) is safe. Cross-thread wake is write(evfd) for BOTH engines —
16 * the io_uring engine arms a multishot poll on the same evfd.
17 *
18 * UAF safety: a poll's user_data carries (slot-generation<<32 | slot); the slot
19 * table maps back to the aconn and the reaper drops any CQE whose generation no
20 * longer matches (poll completing after its aconn was reconnected/freed) — the
21 * same discipline as the server ring. fd changes (reconnect) cancel the old
22 * poll and bump the slot generation before re-arming the new fd. */

29/* epoll interest mask -> poll(2) mask for io_uring_prep_poll_*. */
30unsigned
31uring_pollmask(int want)
32{
33 unsigned m = 0;
34 if (want & EPOLLIN) { m |= POLLIN; }
35 if (want & EPOLLOUT) { m |= POLLOUT; }
36 return m;
37}

40/* Claim a free poll slot for ac (loop-thread only; no lock). Returns 0 / -1. */
41int
42uring_slot_alloc(brix_loop *l, brix_aconn *ac)
43{
44 unsigned i;
45 for (i = 0; i < AIO_URING_SLOTS; i++) {
46 if (!l->uslots[i].in_use) {
47 l->uslots[i].in_use = 1;
48 l->uslots[i].ac = ac;
49 ac->uring_slot = (int) i;
50 return 0;
51 }
52 }
53 return -1; /* table full — pool has more conns than slots (shouldn't happen) */
54}

57/* Submit a multishot poll for ac with the current slot generation as user_data. */
58int
59uring_poll_submit(brix_loop *l, brix_aconn *ac, int want)
60{
61 struct io_uring_sqe *sqe = io_uring_get_sqe(&l->uring);
62 uint32_t slot = (uint32_t) ac->uring_slot;
63
64 if (sqe == NULL) {
65 return -1;
66 }
67 io_uring_prep_poll_multishot(sqe, ac->fd, uring_pollmask(want));
68 io_uring_sqe_set_data64(sqe,
69 AIO_URING_UD(AIO_UD_POLL, l->uslots[slot].gen, slot));
70 return io_uring_submit(&l->uring) < 0 ? -1 : 0;
71}

76/* Return one provided buffer to the RX ring (kernel may refill it after this —
77 * always copy out of it first). */
78void
79uring_rxtx_recycle(brix_loop *l, struct io_uring_cqe *cqe)
80{
81 uint16_t bid;
82
83 if (!(cqe->flags & IORING_CQE_F_BUFFER)) {
84 return;
85 }
86 bid = (uint16_t) (cqe->flags >> IORING_CQE_BUFFER_SHIFT);
87 io_uring_buf_ring_add(l->brx,
88 l->brx_pool + (size_t) bid * AIO_URING_RXBUF_SZ,
89 AIO_URING_RXBUF_SZ, bid,
90 io_uring_buf_ring_mask(AIO_URING_RXBUFS), 0);
91 io_uring_buf_ring_advance(l->brx, 1);
92}

95/* Arm (or re-arm) the persistent multishot RECV for a cleartext conn. Buffers
96 * are kernel-selected from the BGID_RX provided ring. */
97int
98uring_recv_submit(brix_loop *l, brix_aconn *ac)
99{
100 struct io_uring_sqe *sqe = io_uring_get_sqe(&l->uring);
101 uint32_t slot = (uint32_t) ac->uring_slot;
102
103 if (sqe == NULL) {
104 return -1;
105 }
106 io_uring_prep_recv_multishot(sqe, ac->fd, NULL, 0, 0);
107 sqe->flags |= IOSQE_BUFFER_SELECT;
108 sqe->buf_group = AIO_URING_BGID_RX;
109 io_uring_sqe_set_data64(sqe,
110 AIO_URING_UD(AIO_UD_RECV, l->uslots[slot].gen, slot));
111 if (io_uring_submit(&l->uring) < 0) {
112 return -1;
113 }
114 l->uslots[slot].recv_armed = 1;
115 return 0;
116}

119/* Stage + submit one one-shot SEND for the current wbuf tail. The bytes are
120 * copied into the slot's staging slice first: wbuf may be realloc'd by new
121 * frames while the kernel still reads the send buffer, so the SQE must never
122 * point into wbuf itself. Returns 0 with the SEND in flight (or nothing to
123 * do), -1 when the ring can't take it (caller may fall back to send(2) —
124 * safe, as nothing is in flight then). */
125int
126uring_rxtx_send_submit(brix_loop *l, brix_aconn *ac)
127{
128 struct io_uring_sqe *sqe;
129 struct brix_poll_slot *s;
130 size_t avail;
131 uint32_t n;
132
133 if (ac->uring_slot < 0 && io_engine_arm(l, ac, EPOLLIN) != 0) {
134 return -1;
135 }
136 s = &l->uslots[ac->uring_slot];
137
138 if (s->send_inflight || ac->wbuf.start >= ac->wbuf.len) {
139 return 0;
140 }
141 if (s->stage == NULL) {
142 s->stage = (uint8_t *) malloc(AIO_URING_STAGE_SZ);
143 if (s->stage == NULL) {
144 return -1;
145 }
146 }
147 avail = ac->wbuf.len - ac->wbuf.start;
148 n = (uint32_t) (avail > AIO_URING_STAGE_SZ ? AIO_URING_STAGE_SZ : avail);
149 memcpy(s->stage, ac->wbuf.buf + ac->wbuf.start, n);
150
151 sqe = io_uring_get_sqe(&l->uring);
152 if (sqe == NULL) {
153 return -1;
154 }
155 io_uring_prep_send(sqe, ac->fd, s->stage, n, MSG_NOSIGNAL);
156 io_uring_sqe_set_data64(sqe,
157 AIO_URING_UD(AIO_UD_SEND, s->gen, (uint32_t) ac->uring_slot));
158 if (io_uring_submit(&l->uring) < 0) {
159 return -1;
160 }
161 s->send_inflight = 1;
162 s->stage_len = n;
163 return 0;
164}

167/* Cancel every outstanding rxtx op on ac's fd (multishot RECV + any SEND) and
168 * invalidate their CQEs via the generation bump. The cancel acks come back
169 * with AIO_URING_IGNORE_UD-free cookies but stale gens, so they are dropped. */
170static void
171uring_rxtx_cancel(brix_loop *l, brix_aconn *ac, int freeing)
172{
173 uint32_t slot = (uint32_t) ac->uring_slot;
174 struct brix_poll_slot *s = &l->uslots[slot];
175 struct io_uring_sqe *sqe;
176
177 sqe = io_uring_get_sqe(&l->uring);
178 if (sqe != NULL) {
179 io_uring_prep_cancel_fd(sqe, ac->fd, IORING_ASYNC_CANCEL_ALL);
180 io_uring_sqe_set_data64(sqe, AIO_URING_IGNORE_UD);
181 (void) io_uring_submit(&l->uring);
182 }
183 s->gen++; /* stale CQEs (incl. in-flight SEND) drop */
184 s->recv_armed = 0;
185 s->send_inflight = 0;
186 s->stage_len = 0;
187 if (freeing) {
188 free(s->stage);
189 s->stage = NULL;
190 s->in_use = 0;
191 s->ac = NULL;
192 s->rxtx = 0;
193 ac->uring_slot = -1;
194 }
195}

198/* Cancel ac's outstanding poll (by current user_data) and bump the slot gen so
199 * any late CQE for it is dropped. Keeps the slot allocated (caller re-arms) or
200 * frees it when freeing == 1. */
201void
202uring_poll_cancel(brix_loop *l, brix_aconn *ac, int freeing)
203{
204 uint32_t slot = (uint32_t) ac->uring_slot;
205 struct io_uring_sqe *sqe;
206
207 if (ac->uring_slot < 0) {
208 return;
209 }
210 sqe = io_uring_get_sqe(&l->uring);
211 if (sqe != NULL) {
212 io_uring_prep_poll_remove(sqe,
213 ((uint64_t) l->uslots[slot].gen << 32) | slot);
214 io_uring_sqe_set_data64(sqe, AIO_URING_IGNORE_UD);
215 (void) io_uring_submit(&l->uring);
216 }
217 l->uslots[slot].gen++; /* invalidate the old poll's user_data */
218 if (freeing) {
219 l->uslots[slot].in_use = 0;
220 l->uslots[slot].ac = NULL;
221 ac->uring_slot = -1;
222 }
223}

229/* Create the readiness set + the wake eventfd. evfd is used by both engines;
230 * epoll registers it in the set, io_uring arms a multishot poll on it. */
231int
232io_engine_setup(brix_loop *l, brix_status *st)
233{
234 l->evfd = eventfd(0, EFD_NONBLOCK | EFD_CLOEXEC);
235 if (l->evfd < 0) {
236 brix_status_set(st, XRDC_ESOCK, errno, "eventfd: %s", strerror(errno));
237 return -1;
238 }
239
240#if (BRIX_HAVE_LIBURING)
241 if (l->use_uring) {
242 struct io_uring_sqe *sqe;
243 if (io_uring_queue_init(256, &l->uring, 0) < 0) {
244 brix_status_set(st, XRDC_ESOCK, errno, "io_uring_queue_init");
245 return -1;
246 }
247 l->uring_ok = 1;
248
249 if (l->use_rxtx) {
250 /* P44-C: provided-buffer ring for the cleartext RECV tier.
251 * Best-effort — on an old kernel (needs 5.19+) degrade to the
252 * POLL_ADD-only engine instead of failing the loop. */
253 int rr = 0;
254 l->brx_pool = (uint8_t *)
255 malloc((size_t) AIO_URING_RXBUFS * AIO_URING_RXBUF_SZ);
256 if (l->brx_pool != NULL) {
257 l->brx = io_uring_setup_buf_ring(&l->uring, AIO_URING_RXBUFS,
258 AIO_URING_BGID_RX, 0, &rr);
259 }
260 if (l->brx != NULL) {
261 unsigned i;
262 for (i = 0; i < AIO_URING_RXBUFS; i++) {
263 io_uring_buf_ring_add(l->brx,
264 l->brx_pool + (size_t) i * AIO_URING_RXBUF_SZ,
265 AIO_URING_RXBUF_SZ, (unsigned short) i,
266 io_uring_buf_ring_mask(AIO_URING_RXBUFS), (int) i);
267 }
268 io_uring_buf_ring_advance(l->brx, AIO_URING_RXBUFS);
269 } else {
270 free(l->brx_pool);
271 l->brx_pool = NULL;
272 l->use_rxtx = 0;
273 }
274 }
275
276 sqe = io_uring_get_sqe(&l->uring); /* multishot poll on the evfd */
277 if (sqe == NULL) {
278 brix_status_set(st, XRDC_ESOCK, 0, "io_uring evfd arm");
279 return -1;
280 }
281 io_uring_prep_poll_multishot(sqe, l->evfd, POLLIN);
282 io_uring_sqe_set_data64(sqe, AIO_URING_EVFD_UD);
283 if (io_uring_submit(&l->uring) < 0) {
284 brix_status_set(st, XRDC_ESOCK, errno, "io_uring evfd submit");
285 return -1;
286 }
287 return 0;
288 }
289#endif
290
291 l->epfd = epoll_create1(EPOLL_CLOEXEC);
292 if (l->epfd < 0) {
293 brix_status_set(st, XRDC_ESOCK, errno, "epoll_create1: %s",
294 strerror(errno));
295 return -1;
296 }
297 {
298 struct epoll_event ev;
299 memset(&ev, 0, sizeof(ev));
300 ev.events = EPOLLIN;
301 ev.data.ptr = l; /* loop pointer tags the eventfd */
302 if (epoll_ctl(l->epfd, EPOLL_CTL_ADD, l->evfd, &ev) != 0) {
303 brix_status_set(st, XRDC_ESOCK, errno, "epoll_ctl(evfd): %s",
304 strerror(errno));
305 return -1;
306 }
307 }
308 return 0;
309}

312void
313io_engine_teardown(brix_loop *l)
314{
315#if (BRIX_HAVE_LIBURING)
316 if (l->use_uring) {
317 if (l->uring_ok) {
318 unsigned i;
319 for (i = 0; i < AIO_URING_SLOTS; i++) { /* rxtx send stages */
320 free(l->uslots[i].stage);
321 l->uslots[i].stage = NULL;
322 }
323 if (l->brx != NULL) {
324 io_uring_free_buf_ring(&l->uring, l->brx, AIO_URING_RXBUFS,
325 AIO_URING_BGID_RX);
326 l->brx = NULL;
327 }
328 free(l->brx_pool);
329 l->brx_pool = NULL;
330 io_uring_queue_exit(&l->uring);
331 l->uring_ok = 0;
332 }
333 if (l->evfd >= 0) { close(l->evfd); l->evfd = -1; }
334 return;
335 }
336#endif
337 if (l->epfd >= 0) { close(l->epfd); l->epfd = -1; }
338 if (l->evfd >= 0) { close(l->evfd); l->evfd = -1; }
339}

342/* Arm/modify interest for ac to `want`; sets ac->epoll_events on success.
343 * Returns 0 / -1 (caller keeps its own failure handling). */
344int
345io_engine_arm(brix_loop *l, brix_aconn *ac, int want)
346{
347#if (BRIX_HAVE_LIBURING)
348 if (l->use_uring) {
349 if (ac->uring_slot < 0) {
350 if (uring_slot_alloc(l, ac) != 0) { return -1; }
351 l->uslots[ac->uring_slot].rxtx = aconn_is_rxtx(ac) ? 1 : 0;
352 }
353
354 if (l->uslots[ac->uring_slot].rxtx) {
355 /* P44-C: the multishot RECV is the only persistent op — POLLOUT
356 * interest is meaningless (SENDs are submitted directly by
357 * aconn_do_write), so mask changes are no-ops once armed. */
358 if (!l->uslots[ac->uring_slot].recv_armed) {
359 ac->fd_gen++;
360 if (uring_recv_submit(l, ac) != 0) {
361 l->uslots[ac->uring_slot].rxtx = 0; /* degrade to poll */
362 } else {
363 ac->epoll_events = want;
364 return 0;
365 }
366 } else {
367 ac->epoll_events = want;
368 return 0;
369 }
370 } else if (ac->epoll_events != 0) {
371 uring_poll_cancel(l, ac, 0); /* drop the old mask's poll, keep slot */
372 }
373
374 ac->fd_gen++;
375 if (uring_poll_submit(l, ac, want) != 0) { return -1; }
376 ac->epoll_events = want;
377 return 0;
378 }
379#endif
380 {
381 struct epoll_event ev;
382 int op = (ac->epoll_events == 0) ? EPOLL_CTL_ADD : EPOLL_CTL_MOD;
383 memset(&ev, 0, sizeof(ev));
384 ev.events = (uint32_t) want;
385 ev.data.ptr = ac;
386 if (epoll_ctl(l->epfd, op, ac->fd, &ev) != 0) {
387 return -1;
388 }
389 ac->epoll_events = want;
390 return 0;
391 }
392}

395void
396io_engine_del(brix_loop *l, brix_aconn *ac)
397{
398#if (BRIX_HAVE_LIBURING)
399 if (l->use_uring) {
400 if (ac->uring_slot >= 0 && l->uslots[ac->uring_slot].rxtx) {
401 uring_rxtx_cancel(l, ac, 1); /* cancel RECV/SEND + free the slot */
402 } else {
403 uring_poll_cancel(l, ac, 1); /* cancel + free the slot */
404 }
405 ac->epoll_events = 0;
406 return;
407 }
408#endif
409 if (ac->fd >= 0) {
410 epoll_ctl(l->epfd, EPOLL_CTL_DEL, ac->fd, NULL);
411 }
412 ac->epoll_events = 0;
413}
