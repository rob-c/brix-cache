9/* WHAT: When the endpoint is an http/https/dav/davs URL, xrdfs speaks */
10/* WebDAV PROPFIND for metadata (the webfile.c transport, shared */
11/* with the FUSE driver) instead of the binary root:// protocol — */
12/* so `xrdfs https://host/path ls` works like the official client.*/
13/* WHY: A WebDAV endpoint has no root:// session; only the read-only */

18/* Map a cwd-relative arg onto the absolute server path under the URL base. */
19void
20web_build_path(const char *base, const char *cwd, const char *arg,
21 char *out, size_t outsz)
22{
23 char rel[XRDC_PATH_MAX];
24 build_path(cwd, (arg != NULL && arg[0] != '\0') ? arg : ".", rel, sizeof(rel));

28 snprintf(out, outsz, "%s%s", base, rel);

33/* ---- Print one directory entry in xrdfs ls format ----
34 *
35 * WHAT: Emits a single line for one PROPFIND-derived dirent to stdout. In long
36 * mode (want_long) with stat data present, prints the drwx-style flag
37 * triplet, a right-justified size (human-readable when human is set), and
38 * the full "prefix + name" path; otherwise prints just "prefix + name".
39 * No return value; output-only.
40 *
41 * WHY: Factored out of web_ls_print_dir so the per-entry formatting (with its
42 * flag/size ternaries) is a single-purpose helper, keeping the directory
43 * walker's cyclomatic complexity within the project cap and matching the
44 * exact one-line-per-entry format of the root:// ls_print_dir().
45 *
46 * HOW: 1. If long output is requested and the entry carries stat data, derive
47 * the dir/readable/writable flag characters and format the size.
48 * 2. Print the long-form line with flags, size, prefix, and name.
49 * 3. Otherwise print the short form: prefix followed by name.
50 */
51static void
52web_print_one_entry(const brix_dirent *ent, const char *prefix,
53 int want_long, int human)
54{
55 if (want_long && ent->have_stat) {
56 int f = ent->st.flags;
57 char td = (f & kXR_isDir) ? 'd' : '-';
58 char r = (f & kXR_readable) ? 'r' : '-';
59 char wc = (f & kXR_writable) ? 'w' : '-';
60 char szs[32];
61 fmt_size(ent->st.size, szs, sizeof(szs), human);
62 printf("%c%c%c %12s %s%s\n", td, r, wc, szs, prefix, ent->name);
63 } else {
64 printf("%s%s\n", prefix, ent->name);
65 }
66}

69/* ---- Recurse into subdirectories for xrdfs ls -R over WebDAV ----
70 *
71 * WHAT: For each subdirectory in a directory listing, prints the "full:" header
72 * and recursively lists its contents via web_ls_print_dir(). Returns 0 on
73 * success, -1 if any recursive listing fails (with st describing the
74 * failure). Skips "." / ".." entries, non-directories, and any child whose
75 * joined path would overflow.
76 *
77 * WHY: Separated from web_ls_print_dir so the recursion loop is an independent,
78 * single-purpose step. This keeps the walker's complexity within the cap
79 * while preserving the exact skip rules and recursive descent order.
80 *
81 * HOW: 1. Iterate the entries; skip dot entries and anything that is not a
82 * stat-bearing directory.
83 * 2. Build the child's full path; skip entries that do not fit.
84 * 3. Print the "\nfull:\n" section header, then recurse; on the first
85 * recursion failure return -1, otherwise return 0.
86 */
87static int
88web_ls_recurse_children(const web_ctx *w, const char *path,
89 const brix_dirent *ents, size_t n,
90 int want_long, int human, brix_status *st)
91{
92 size_t k;
93
94 for (k = 0; k < n; k++) {
95 char full[XRDC_PATH_MAX];
96 if (is_dot(ents[k].name)
97 || !(ents[k].have_stat && (ents[k].st.flags & kXR_isDir))) {
98 continue;
99 }
100 if (join_path(path, ents[k].name, full, sizeof(full)) != 0) {
101 continue;
102 }
103 printf("\n%s:\n", full);
104 if (web_ls_print_dir(w, full, want_long, 1, human, st) != 0) {
105 return -1;
106 }
107 }
108 return 0;
109}

112/* WebDAV equivalent of ls_print_dir() — same output format, PROPFIND source. */
113int
114web_ls_print_dir(const web_ctx *w, const char *path, int want_long,
115 int recursive, int human, brix_status *st)
116{
117 brix_dirent *ents = NULL;
118 size_t n = 0, k, plen;
119 char prefix[XRDC_PATH_MAX + 2];
120 const char *sep;
121 char proxybuf[512];
122 const char *pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
123
124 if (brix_web_readdir(w->u, path, w->bearer, w->verify, w->ca_dir, pcert,
125 &ents, &n, st) != 0) {
126 return -1;
127 }
128 plen = strlen(path);
129 sep = (plen > 0 && path[plen - 1] == '/') ? "" : "/";
130 snprintf(prefix, sizeof(prefix), "%s%s", path, sep);
131
132 for (k = 0; k < n; k++) {
133 web_print_one_entry(&ents[k], prefix, want_long, human);
134 }
135 if (recursive
136 && web_ls_recurse_children(w, path, ents, n, want_long, human, st) != 0) {
137 free(ents);
138 return -1;
139 }
140 free(ents);
141 return 0;
142}

145int
146web_ls(const web_ctx *w, const char *cwd, int argc, char **argv)
147{
148 brix_status st;
149 char path[XRDC_PATH_MAX];
150 const char *arg = NULL;
151 int want_long = 0, recursive = 0, human = 0, i;
152
153 for (i = 1; i < argc; i++) {
154 if (strcmp(argv[i], "-l") == 0) { want_long = 1; }
155 else if (strcmp(argv[i], "-R") == 0) { recursive = 1; }
156 else if (strcmp(argv[i], "-h") == 0) { human = 1; }
157 else if (strcmp(argv[i], "-lR") == 0 || strcmp(argv[i], "-Rl") == 0) {
158 want_long = 1; recursive = 1;
159 } else { arg = argv[i]; }
160 }
161 web_build_path(w->base, cwd, arg, path, sizeof(path));
162 brix_status_clear(&st);
163 if (web_ls_print_dir(w, path, want_long, recursive, human, &st) != 0) {
164 return xrdfs_web_report_err("ls", path, &st, 0, w);
165 }
166 return 0;
167}

191/* ---- Resolve an mv/rm-position arg to an absolute server path ----
192 *
193 * WHAT: Writes the absolute server path named by `arg` into out[outsz]. If `arg`
194 * is itself a WebDAV URL on the same endpoint (the shape the `xrd` wrapper
195 * forwards for a second operand, e.g. `xrdfs davs://h/a mv davs://h/b`),
196 * its URL path component is used; otherwise `arg` is a cwd-relative path
197 * resolved against the endpoint's export base.
198 *
199 * WHY: the write-metadata verbs accept both a bare path and a full URL for each
200 * operand — the same duality the root:// path enjoys — so `mv`/`rm` behave
201 * identically whether invoked directly or via the `xrd` wrapper.
202 *
203 * HOW: try brix_weburl_parse; on a WebDAV (non-S3) match copy its path, else
204 * fall back to web_build_path over the export base.
205 */
206static void
207web_resolve_arg(const web_ctx *w, const char *cwd, const char *arg,
208 char *out, size_t outsz)
209{
210 brix_weburl wu;
211
212 if (arg != NULL && brix_weburl_parse(arg, &wu) == 0 && !wu.is_s3) {
213 snprintf(out, outsz, "%s", wu.path[0] != '\0' ? wu.path : "/");
214 return;
215 }
216 web_build_path(w->base, cwd, arg, out, outsz);
217}

220/* ---- Format the endpoint-absolute URL for a WebDAV MOVE Destination ----
221 *
222 * WHAT: Builds "<scheme>://<host>[:port]<path>" for `path` on the connection's
223 * endpoint into out[outsz]; scheme is https/http by the endpoint's TLS bit
224 * and an IPv6 host is bracketed. No return value.
225 *
226 * WHY: RFC 4918 requires MOVE's Destination to be an absolute URI on the same
227 * authority; the transport helper stays scheme-agnostic and lets the caller
228 * (which owns the connection) name it.
229 *
230 * HOW: pick the scheme from w->u->tls, bracket a ':'-bearing host, printf.
231 */
232static void
233web_dest_url(const web_ctx *w, const char *path, char *out, size_t outsz)
234{
235 const char *scheme = w->u->tls ? "https" : "http";
236 int v6 = (strchr(w->u->host, ':') != NULL);
237
238 snprintf(out, outsz, "%s://%s%s%s:%d%s", scheme,
239 v6 ? "[" : "", w->u->host, v6 ? "]" : "", w->u->port, path);
240}

287/* Rename a WebDAV resource (mv) — MOVE with an absolute-URL Destination. */
288static int
289web_mv(const web_ctx *w, const char *cwd, int argc, char **argv)
290{
291 brix_status st;
292 char src[XRDC_PATH_MAX], dst[XRDC_PATH_MAX], dest_url[XRDC_PATH_MAX + 320];
293 char proxybuf[512];
294 const char *pcert;
295
296 if (argc < 2) {
297 fprintf(stderr, "usage: mv <src> <dst>\n");
298 return 50;
299 }
300 pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
301 if (argc >= 3) {
302 web_resolve_arg(w, cwd, argv[1], src, sizeof(src));
303 web_resolve_arg(w, cwd, argv[2], dst, sizeof(dst));
304 } else {
305 /* single operand: src is the endpoint's export base, dst the operand
306 * (the `xrdfs davs://h/src mv davs://h/dst` direct form). */
307 web_resolve_arg(w, cwd, NULL, src, sizeof(src));
308 web_resolve_arg(w, cwd, argv[1], dst, sizeof(dst));
309 }
310 web_dest_url(w, dst, dest_url, sizeof(dest_url));
311 brix_status_clear(&st);
312 if (brix_webdav_move(w->u, src, dest_url, w->bearer, w->verify, w->ca_dir,
313 pcert, &st) != 0) {
314 /* two-path op: mirror xrdfs's mv error line (src -> dst) then hint. */
315 fprintf(stderr, "xrdfs: mv %s -> %s failed: %s\n", src, dst, st.msg);
316 xrdfs_web_hints(&st, 1, w);
317 return 50;
318 }
319 return 0;
320}

323/* Dispatch one command against a WebDAV endpoint: read-only metadata (ls, stat)
324 * plus write-metadata (mkdir, rm, rmdir, mv). Data/byte ops and the interactive
325 * shell remain root:// only. */
326int
327web_dispatch(const web_ctx *w, int argc, char **argv)
328{
329 const char *cmd = argv[0];
330 if (strcmp(cmd, "ls") == 0) { return web_ls(w, "/", argc, argv); }
331 if (strcmp(cmd, "stat") == 0) { return web_stat(w, "/", argc, argv); }
332 if (strcmp(cmd, "mkdir") == 0) { return web_mkdir(w, "/", argc, argv); }
333 if (strcmp(cmd, "rm") == 0) { return web_delete(w, "/", "rm", argc, argv); }
334 if (strcmp(cmd, "rmdir") == 0) { return web_delete(w, "/", "rmdir", argc, argv); }
335 if (strcmp(cmd, "mv") == 0) { return web_mv(w, "/", argc, argv); }
336 fprintf(stderr, "xrdfs: '%s' is not supported over an http(s)/WebDAV endpoint "
337 "(metadata only: ls, stat, mkdir, rm, rmdir, mv). Use a root:// "
338 "endpoint for the full command set.\n", cmd);
339 return 50;
340}
