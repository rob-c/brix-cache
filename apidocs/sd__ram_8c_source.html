1/*
2 * sd_ram.c — the RAM object store (parity audit §3.6 row 6, §4.12 row 12: the
3 * oss.memfile / XrdRmc analog), spelled `ram:<size>`.
4 *
5 * WHAT: a complete, bounded, heap-resident object store — objects live in this
6 * worker's own memory and nowhere else. Its intended role is a CACHE
7 * store (`brix_cache_store ram:512m`): a RAM tier in front of the origin
8 * that never touches a disk, for nodes whose working set fits in memory
9 * and whose latency budget cannot absorb a page-cache miss.
10 *
11 * WHY: stock xrootd has two RAM caches — oss.memfile (mmap/mlock a file into
12 * RAM) and XrdRmc (an in-process block cache). Both are per-process, and
13 * both exist because the OS page cache is advisory: it can be reclaimed
14 * under pressure and gives no capacity guarantee. A store with its OWN
15 * hard cap gives the guarantee the page cache cannot.
16 *
17 * HOW: a hash table of refcounted entries under one pthread mutex — the store
18 * is read from the event loop and written from cache-fill worker threads,
19 * the exact mix that cost cinfo_l1 a TSan-found corruption before it grew
20 * the same lock. No fd, so no sendfile: CAP_MEMFILE says bytes are served
21 * memory-backed. The cap is enforced where the bytes are actually claimed
22 * (fill reservation and buffer growth), so a full store returns ENOSPC to
23 * the fill and the cache decorator degrades to a plain source read — a
24 * full cache slows reads down, it never fails them.
25 *
26 * SCOPE: the store is PER WORKER, like both of the stock caches it mirrors, so
27 * an N-worker server holds up to N copies and N x <size> bytes. That is a
28 * property of the design, not an oversight; docs/03-configuration states
29 * it and tests/test_phase115_ram_tier.py pins it.
30 */

41/* Bucket count: fixed and modest. The store is capacity-bounded, not count-
42 * bounded, and a cache store's population is dominated by object SIZE, so a
43 * few thousand chains cost 8 KB and keep every lookup short. */
44#define SD_RAM_BUCKETS 1024

48static ngx_int_t
49sd_ram_init(brix_sd_instance_t *inst, void *driver_conf)
50{
51 const brix_sd_ram_conf_t *conf = driver_conf;
52 sd_ram_state_t *st;
53
54 if (conf == NULL || conf->capacity == 0) {
55 errno = EINVAL;
56 return NGX_ERROR;
57 }
58 st = calloc(1, sizeof(*st));
59 if (st == NULL) {
60 errno = ENOMEM;
61 return NGX_ERROR;
62 }
63 st->buckets = calloc(SD_RAM_BUCKETS, sizeof(*st->buckets));
64 if (st->buckets == NULL) {
65 free(st);
66 errno = ENOMEM;
67 return NGX_ERROR;
68 }
69 if (pthread_mutex_init(&st->mtx, NULL) != 0) {
70 free(st->buckets);
71 free(st);
72 errno = ENOMEM;
73 return NGX_ERROR;
74 }
75 st->nbuckets = SD_RAM_BUCKETS;
76 st->capacity = conf->capacity;
77 st->log = inst->log;
78 inst->state = st;
79
80 /* Say PER WORKER out loud. `ram:8g` on a 16-worker server is 128 GB of
81 * resident memory, and the size in the directive is the one number an
82 * operator reads as the answer — the log is where that gets corrected. */
83 ngx_log_error(NGX_LOG_NOTICE, inst->log, 0,
84 "brix: ram cache store ready, capacity=%uL bytes PER WORKER",
85 (uint64_t) st->capacity);
86 return NGX_OK;
87}

89static void
90sd_ram_cleanup(brix_sd_instance_t *inst)
91{
92 sd_ram_state_t *st = inst->state;
93 size_t i;
94
95 if (st == NULL) {
96 return;
97 }
98 for (i = 0; i < st->nbuckets; i++) {
99 sd_ram_ent_t *e = st->buckets[i];
100
101 while (e != NULL) {
102 sd_ram_ent_t *next = e->hnext;
103
104 e->refs = 1; /* worker teardown: drop every holder */
105 brix_sd_ram_unref(st, e);
106 e = next;
107 }
108 }
109 (void) pthread_mutex_destroy(&st->mtx);
110 free(st->buckets);
111 free(st);
112 inst->state = NULL;
113}

117/* A write/create open is refused: the ONLY writer of this store is the staged
118 * fill spine (staged_open/write/commit), which is what a cache store uses and
119 * what keeps a partially written object from ever being visible under its final
120 * name. A direct write open would defeat that, so it is ENOTSUP, never a silent
121 * degradation. */

123/* Report `err` through both channels the open contract uses, and decline. */
124static brix_sd_obj_t *
125sd_ram_open_fail(int err, int *err_out)
126{
127 if (err_out != NULL) {
128 *err_out = err;
129 }
130 errno = err;
131 return NULL;
132}

134/* Attach a handle to the entry at `path`, taking its reference. Returns 0, or
135 * the errno the open must report. Caller holds no lock; this takes it. */
136static int
137sd_ram_open_attach(sd_ram_state_t *st, const char *path, sd_ram_open_t *oh)
138{
139 sd_ram_ent_t *ent;
140 int err = 0;
141
142 (void) pthread_mutex_lock(&st->mtx);
143 ent = brix_sd_ram_find(st, path);
144 if (ent == NULL) {
145 err = ENOENT;
146 } else if (ent->is_dir) {
147 err = EISDIR;
148 } else {
149 ent->refs++; /* the handle's own reference */
150 brix_sd_ram_lru_touch(st, ent); /* an open is a use */
151 oh->ent = ent;
152 }
153 (void) pthread_mutex_unlock(&st->mtx);
154 return err;
155}

157static brix_sd_obj_t *
158sd_ram_open(brix_sd_instance_t *inst, const char *path, int sd_flags,
159 mode_t mode, int *err_out)
160{
161 sd_ram_state_t *st = inst->state;
162 brix_sd_obj_t *obj;
163 sd_ram_open_t *oh;
164 int err;
165
166 (void) mode;
167 if (!brix_sd_ram_path_ok(path)) {
168 return sd_ram_open_fail(EINVAL, err_out);
169 }
170 if (sd_flags & (BRIX_SD_O_WRITE | BRIX_SD_O_CREATE | BRIX_SD_O_TRUNC
171 | BRIX_SD_O_APPEND))
172 {
173 return sd_ram_open_fail(ENOTSUP, err_out);
174 }
175
176 /* Heap shell (open may run off the event loop; inst->pool is not
177 * thread-safe) with the handle state in its OWN allocation, never inline
178 * behind the shell: the VFS adopts the obj BY VALUE and frees a heap_shell
179 * (open_resolved_file_open.c), so anything allocated behind the shell dies
180 * with it while the copy's `state` still points there. Inlining it was the
181 * 2.0 root:// read crash — sd_ram_pread on a freed handle. */
182 obj = ngx_calloc(sizeof(*obj), inst->log);
183 if (obj == NULL) {
184 return sd_ram_open_fail(ENOMEM, err_out);
185 }
186 oh = ngx_calloc(sizeof(*oh), inst->log);
187 if (oh == NULL) {
188 ngx_free(obj);
189 return sd_ram_open_fail(ENOMEM, err_out);
190 }
191
192 err = sd_ram_open_attach(st, path, oh);
193 if (err != 0) {
194 ngx_free(oh);
195 ngx_free(obj);
196 return sd_ram_open_fail(err, err_out);
197 }
198
199 obj->driver = inst->driver;
200 obj->inst = inst;
201 obj->fd = NGX_INVALID_FILE; /* memory-served: no fd, no sendfile */
202 obj->state = oh;
203 obj->heap_shell = 1;
204 return obj;
205}

207static ngx_int_t
208sd_ram_close(brix_sd_obj_t *obj)
209{
210 sd_ram_state_t *st = obj->inst->state;
211 sd_ram_open_t *oh = obj->state;
212
213 if (oh == NULL) {
214 return NGX_OK;
215 }
216 if (oh->ent != NULL) {
217 (void) pthread_mutex_lock(&st->mtx);
218 brix_sd_ram_unref(st, oh->ent);
219 (void) pthread_mutex_unlock(&st->mtx);
220 }
221 ngx_free(oh); /* ours; the shell (or the adopter's
222 * by-value copy) is the caller's */
223 obj->state = NULL;
224 return NGX_OK;
225}

229static ssize_t
230sd_ram_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
231{
232 sd_ram_state_t *st = obj->inst->state;
233 sd_ram_open_t *oh = obj->state;
234 sd_ram_ent_t *e;
235 size_t n;
236
237 if (off < 0) {
238 errno = EINVAL;
239 return -1;
240 }
241 (void) pthread_mutex_lock(&st->mtx);
242 e = oh->ent;
243 if (e == NULL) {
244 (void) pthread_mutex_unlock(&st->mtx);
245 errno = EBADF;
246 return -1;
247 }
248 if ((uint64_t) off >= (uint64_t) e->size) {
249 (void) pthread_mutex_unlock(&st->mtx);
250 return 0; /* EOF */
251 }
252 n = e->size - (size_t) off;
253 if (n > len) {
254 n = len;
255 }
256 memcpy(buf, e->data + off, n);
257 (void) pthread_mutex_unlock(&st->mtx);
258 return (ssize_t) n;
259}

261static ssize_t
262sd_ram_preadv(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
263 off_t off)
264{
265 ssize_t total = 0;
266 int i;
267
268 for (i = 0; i < iovcnt; i++) {
269 ssize_t n = sd_ram_pread(obj, iov[i].iov_base, iov[i].iov_len,
270 off + total);
271
272 if (n < 0) {
273 return total > 0 ? total : n;
274 }
275 total += n;
276 if ((size_t) n < iov[i].iov_len) {
277 break; /* EOF inside this segment */
278 }
279 }
280 return total;
281}

390/*
391 * The store's own view of its capacity — the ONLY honest source, since there is
392 * no filesystem under it to statvfs. brix_cstore_freespace consults this for a
393 * non-LOCAL store, so occupancy reporting (kXR_statvfs, the cache metrics) is
394 * truthful for a RAM tier. It is NOT what bounds the store: the watermark
395 * reaper needs a physical state root this store has not got, so the cap is
396 * enforced where the bytes are, by brix_sd_ram_make_room.
397 *
398 * In-flight fill reservations count as USED: a reader told there is room for
399 * bytes another fill has already promised would be reading a number that is
400 * about to be false.
401 */
402static ngx_int_t
403sd_ram_space(brix_sd_instance_t *inst, brix_sd_space_t *out)
404{
405 sd_ram_state_t *st = inst->state;
406 uint64_t used;
407
408 (void) pthread_mutex_lock(&st->mtx);
409 used = st->used + st->reserved;
410 (void) pthread_mutex_unlock(&st->mtx);
411
412 out->total_bytes = st->capacity;
413 out->used_bytes = used;
414 out->free_bytes = (used < st->capacity) ? st->capacity - used : 0;
415 return NGX_OK;
416}
