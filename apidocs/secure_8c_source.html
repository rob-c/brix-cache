1/* secure.c — scvmfs:// security preamble (EXPERIMENTAL, phase-68 T22).
2 *
3 * WHAT: transport + client-authz gate that runs before the cvmfs gate on
4 * locations with `brix_scvmfs on`.
5 * WHY: "secure CVMFS" repositories are credential-protected; the site
6 * cache must enforce the same boundary or it becomes a leak. Layering
7 * it as a preamble keeps ONE protocol core — scvmfs can never drift
8 * behaviorally from cvmfs because it IS cvmfs after this function.
9 * HOW: TLS presence comes from the connection (r->connection->ssl); bearer
10 * mode delegates to the shared SciTokens issuer registry
11 * (brix_token_validate_registry — the same engine the WebDAV and
12 * stream token paths use; READ scope suffices for a read-only
13 * protocol). This file contains POLICY GLUE ONLY — zero crypto.
14 * x509 mode authenticates the TLS-verified peer by its end-entity (EEC)
15 * subject DN (RFC 3820 proxy certs skipped via brix_px_classify — a GSI
16 * proxy authenticates as its issuing EEC) against an optional DN
17 * allow-glob list; it is still POLICY GLUE — the crypto is nginx's own
18 * ssl_verify_client chain validation plus the shared brix_x509_oneline /
19 * brix_sp_glob_match helpers. voms mode layers a VOMS-VO authorisation
20 * gate on top of x509 via the shared brix_extract_voms_info engine
21 * (per-VO LSC vomsdir + VOMS signing-CA trust); still POLICY GLUE.
22 */

41/* Extract "Authorization: Bearer <token>" into a NUL-terminated pool copy. */
42static ngx_int_t
43scvmfs_bearer_token(ngx_http_request_t *r, const char **token, size_t *len)
44{
45 ngx_str_t *v;
46 u_char *p;
47 size_t n;
48
49 if (r->headers_in.authorization == NULL) {
50 return NGX_DECLINED;
51 }
52 v = &r->headers_in.authorization->value;
53 if (v->len <= sizeof("Bearer ") - 1
54 || ngx_strncasecmp(v->data, (u_char *) "Bearer ",
55 sizeof("Bearer ") - 1) != 0)
56 {
57 return NGX_DECLINED;
58 }
59 n = v->len - (sizeof("Bearer ") - 1);
60 p = ngx_pnalloc(r->pool, n + 1);
61 if (p == NULL) {
62 return NGX_ERROR;
63 }
64 ngx_memcpy(p, v->data + sizeof("Bearer ") - 1, n);
65 p[n] = '\0';
66 *token = (const char *) p;
67 *len = n;
68 return NGX_OK;
69}

71/* scvmfs_registry_check — shared bearer gate behind both the scvmfs bearer
72 * mode and the F3 repo-authz plane: fetch the Bearer credential, canonicalise
73 * r->uri, and validate against `reg` with READ scope (a read-only protocol).
74 * Returns NGX_DECLINED on success with *claims filled, otherwise the HTTP
75 * status to answer with. reg == NULL fails CLOSED — merge-time validation
76 * makes that unreachable, but never open up. */
77static ngx_int_t
78scvmfs_registry_check(ngx_http_request_t *r, const void *reg,
79 brix_token_claims_t *claims)
80{
81 const char *token;
82 size_t token_len;
83 char uri_path[PATH_MAX];
84 int bucket = 0;
85 ngx_int_t rc;
86
87 if (reg == NULL) {
88 return NGX_HTTP_UNAUTHORIZED;
89 }
90
91 rc = scvmfs_bearer_token(r, &token, &token_len);
92 if (rc == NGX_DECLINED) {
93 return NGX_HTTP_UNAUTHORIZED; /* no Bearer credential */
94 }
95 if (rc != NGX_OK) {
96 return NGX_HTTP_INTERNAL_SERVER_ERROR;
97 }
98
99 if (brix_str_cbuf(uri_path, sizeof(uri_path), &r->uri) == NULL) {
100 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
101 }
102
103 {
104 brix_token_registry_args_t ra;
105
106 ra.log = r->connection->log;
107 ra.token = token;
108 ra.token_len = token_len;
109 ra.reg = (const brix_token_registry_t *) reg;
110 ra.macaroon_secret = NULL;
111 ra.secret_len = 0;
112 ra.clock_skew = BRIX_TOKEN_CLOCK_SKEW_SECS;
113 ra.claims = claims;
114
115 if (brix_token_validate_registry(&ra, uri_path, BRIX_TOKEN_OP_READ,
116 &bucket) != 0)
117 {
118 return NGX_HTTP_UNAUTHORIZED; /* invalid/expired/out-of-scope */
119 }
120 }
121 return NGX_DECLINED;
122}

188/* The gate entry matching this repo, or NULL = repo not gated. First match
189 * wins; "*" is a catch-all so an exact entry listed before it can pin a
190 * specific registry while "*" sweeps the rest. */
191static brix_cvmfs_repo_authz_t *
192cvmfs_repo_authz_match(ngx_http_brix_cvmfs_loc_conf_t *lcf,
193 const char *repo, size_t repo_len)
194{
195 brix_cvmfs_repo_authz_t *e = lcf->repo_authz->elts;
196 ngx_uint_t i;
197
198 for (i = 0; i < lcf->repo_authz->nelts; i++) {
199 if (e[i].repo.len == 1 && e[i].repo.data[0] == '*') {
200 return &e[i];
201 }
202 if (e[i].repo.len == repo_len
203 && ngx_strncmp(e[i].repo.data, repo, repo_len) == 0)
204 {
205 return &e[i];
206 }
207 }
208 return NULL;
209}

308/* Recurse the static (prefix/exact) location tree looking for an scvmfs
309 * x509/voms location. By module postconfiguration nginx has already folded the
310 * config-time location queue into this tree (built in ngx_http_merge_servers),
311 * so the queue is empty and the tree is the live structure to walk. */
312static ngx_int_t
313scvmfs_tree_wants_proxy_certs(ngx_http_location_tree_node_t *node)
314{
315 if (node == NULL) {
316 return 0;
317 }
318 if (node->exact != NULL && scvmfs_loc_wants_proxy_certs(node->exact)) {
319 return 1;
320 }
321 if (node->inclusive != NULL
322 && scvmfs_loc_wants_proxy_certs(node->inclusive))
323 {
324 return 1;
325 }
326 return scvmfs_tree_wants_proxy_certs(node->left)
327 || scvmfs_tree_wants_proxy_certs(node->right)
328 || scvmfs_tree_wants_proxy_certs(node->tree);
329}

368/* Enable client GSI-proxy-cert verification on an scvmfs x509/voms server's TLS
369 * context. nginx core rejects RFC 3820 proxy certs during ssl_verify_client
370 * chain validation unless X509_V_FLAG_ALLOW_PROXY_CERTS is set; x509 mode
371 * authenticates a proxy as its issuing EEC and voms mode lifts the proxy's VOMS
372 * AC, so both need a proxy chain to VERIFY (X509_V_OK) before the preamble sees
373 * it. Mirrors webdav's proxy_certs postconfig hook. Bearer/none modes (and
374 * servers with no scvmfs x509/voms location) are left untouched. cscf is an
375 * ngx_http_core_srv_conf_t *. */
376ngx_int_t
377brix_scvmfs_postconf_proxy_certs(ngx_conf_t *cf, void *cscf)
378{
379#if (NGX_HTTP_SSL)
380 ngx_http_core_srv_conf_t *core_srv = cscf;
381 ngx_http_conf_ctx_t *ctx = core_srv->ctx;
382 ngx_http_core_loc_conf_t *clcf;
383 ngx_http_brix_cvmfs_loc_conf_t *lcf;
384 ngx_http_ssl_srv_conf_t *sslcf;
385 X509_VERIFY_PARAM *param;
386 ngx_int_t wants;
387
388 /* Server-level directive (ctx->loc_conf is valid here even though the core
389 * loc conf's own ->loc_conf field is not), then the nested location tree. */
390 lcf = ctx->loc_conf[ngx_http_brix_cvmfs_module.ctx_index];
391 wants = (lcf != NULL && lcf->scvmfs
392 && (lcf->scvmfs_authz == BRIX_SCVMFS_AUTHZ_X509
393 || lcf->scvmfs_authz == BRIX_SCVMFS_AUTHZ_VOMS));
394 if (!wants) {
395 clcf = ctx->loc_conf[ngx_http_core_module.ctx_index];
396 wants = (clcf != NULL && scvmfs_loc_wants_proxy_certs(clcf));
397 }
398 if (!wants) {
399 return NGX_OK;
400 }
401 sslcf = ctx->srv_conf[ngx_http_ssl_module.ctx_index];
402 if (sslcf == NULL || sslcf->ssl.ctx == NULL) {
403 return NGX_OK; /* no TLS on this server */

405 param = SSL_CTX_get0_param(sslcf->ssl.ctx);
406 if (param != NULL) {
407 X509_VERIFY_PARAM_set_flags(param, X509_V_FLAG_ALLOW_PROXY_CERTS);
408 ngx_log_error(NGX_LOG_INFO, cf->log, 0,
409 "scvmfs: enabled X509_V_FLAG_ALLOW_PROXY_CERTS on server %V",
410 &core_srv->server_name);
411 }
412 return NGX_OK;

415 return NGX_OK;

419/* ---- per-VO/per-job QoS fill throttling (phase-85 F9) ---------------------
420 * brix_cvmfs_qos <class> sub=<subject>|default fills=<n> — multi-occurrence;
421 * each entry maps ONE validated token subject (or `default` = everything
422 * unclassified) to a fills-per-second budget. Only ORIGIN FILLS are charged
423 * (the caller gates on the remote-miss predicate) — cache hits always flow,
424 * so a throttled class is bounded at the shared WAN/origin resource, which
425 * is exactly the noisy-neighbor surface. fills=0 = unlimited (parity). */
426
427char *
428cvmfs_conf_qos(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
429{
430 ngx_http_brix_cvmfs_loc_conf_t *c = conf;
431 ngx_str_t *value = cf->args->elts;
432 brix_cvmfs_qos_t *entry;
433 ngx_int_t fills;
434
435 (void) cmd;
436
437 if (value[3].len <= sizeof("fills=") - 1
438 || ngx_strncmp(value[3].data, "fills=", sizeof("fills=") - 1) != 0)
439 {
440 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
441 "brix_cvmfs_qos needs <class> sub=<subject>|default fills=<n>");
442 return NGX_CONF_ERROR;
443 }
444 fills = ngx_atoi(value[3].data + sizeof("fills=") - 1,
445 value[3].len - (sizeof("fills=") - 1));
446 if (fills == NGX_ERROR) {
447 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
448 "brix_cvmfs_qos: \"%V\" is not a fill rate", &value[3]);
449 return NGX_CONF_ERROR;
450 }
451
452 if (c->qos == NGX_CONF_UNSET_PTR) {
453 c->qos = ngx_array_create(cf->pool, 4, sizeof(brix_cvmfs_qos_t));
454 if (c->qos == NULL) {
455 return NGX_CONF_ERROR;
456 }
457 }
458 entry = ngx_array_push(c->qos);
459 if (entry == NULL) {
460 return NGX_CONF_ERROR;
461 }
462 ngx_memzero(entry, sizeof(*entry));
463 entry->name = value[1];
464 entry->fills = (ngx_uint_t) fills;
465
466 if (value[2].len == sizeof("default") - 1
467 && ngx_strncmp(value[2].data, "default", value[2].len) == 0)
468 {
469 /* sub stays empty = the unclassified catch-all */
470 return NGX_CONF_OK;
471 }
472 if (value[2].len <= sizeof("sub=") - 1
473 || ngx_strncmp(value[2].data, "sub=", sizeof("sub=") - 1) != 0)
474 {
475 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
476 "brix_cvmfs_qos: match must be sub=<subject> or default");
477 return NGX_CONF_ERROR;
478 }
479 entry->sub.data = value[2].data + (sizeof("sub=") - 1);
480 entry->sub.len = value[2].len - (sizeof("sub=") - 1);
481 return NGX_CONF_OK;
482}

484/* The class for `sub` ("" = anonymous): first sub= match wins, else the
485 * first `default` entry, else NULL (identity unthrottled). */
486static brix_cvmfs_qos_t *
487cvmfs_qos_class(ngx_http_brix_cvmfs_loc_conf_t *lcf, const char *sub)
488{
489 brix_cvmfs_qos_t *e = lcf->qos->elts;
490 brix_cvmfs_qos_t *def = NULL;
491 size_t sub_len = ngx_strlen(sub);
492 ngx_uint_t i;
493
494 for (i = 0; i < lcf->qos->nelts; i++) {
495 if (e[i].sub.len == 0) {
496 if (def == NULL) {
497 def = &e[i];
498 }
499 continue;
500 }
501 if (e[i].sub.len == sub_len
502 && ngx_strncmp(e[i].sub.data, sub, sub_len) == 0)
503 {
504 return &e[i];
505 }
506 }
507 return def;
508}

510ngx_int_t
511brix_cvmfs_qos_check(ngx_http_request_t *r,
512 ngx_http_brix_cvmfs_loc_conf_t *lcf)
513{
514 ngx_http_brix_cvmfs_ctx_t *ctx =
515 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
516 brix_cvmfs_qos_t *cls;
517 ngx_msec_t now;
518
519 if (lcf->qos == NULL || ctx == NULL) {
520 return NGX_DECLINED;
521 }
522 cls = cvmfs_qos_class(lcf, ctx->token_sub);
523 if (cls == NULL || cls->fills == 0) {
524 return NGX_DECLINED; /* unthrottled / 0 = parity */
525 }
526
527 /* Token bucket in milli-fills: capacity fills*1000, refill fills/ms-
528 * scaled, one fill costs 1000. Worker-local (COW conf memory), event-
529 * loop only — each worker bounds its own share. */
530 now = ngx_current_msec;
531 if (cls->last == 0) {
532 cls->tokens = (ngx_int_t) (cls->fills * 1000); /* first sight: full */
533 } else if (now != cls->last) {
534 ngx_int_t cap = (ngx_int_t) (cls->fills * 1000);
535
536 cls->tokens += (ngx_int_t) ((now - cls->last) * cls->fills);
537 if (cls->tokens > cap) {
538 cls->tokens = cap;
539 }
540 }
541 cls->last = now;
542
543 if (cls->tokens < 1000) {
544 ngx_log_error(NGX_LOG_NOTICE, r->connection->log, 0,
545 "cvmfs: qos class \"%V\" fill budget exhausted "
546 "(sub \"%s\", %ui fills/s) - 429", &cls->name,
547 ctx->token_sub[0] != '\0' ? ctx->token_sub : "anonymous",
548 cls->fills);
549 return NGX_HTTP_TOO_MANY_REQUESTS;
550 }
551 cls->tokens -= 1000;
552 return NGX_DECLINED;
553}
