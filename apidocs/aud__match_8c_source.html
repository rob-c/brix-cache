1/*
2 * aud_match.c — backend audience gate for bearer passthrough (see aud_match.h).
3 *
4 * WHAT: brix_token_backend_aud_ok() — does the client token's `aud` claim
5 * accept the backend, per the `brix_backend_token_audience_ok` list?
6 *
7 * WHY: Phase-70 §5.2: the PASSTHROUGH zero-provisioning path forwards the
8 * client's own JWT to the origin, which is only correct when the token
9 * was actually audienced for that origin (or carries the WLCG
10 * any-endpoint wildcard). Without this gate the directive was parsed but
11 * never enforced — a silent fail-open (P90-70.9).
12 *
13 * HOW: Split the compact JWS with xrdjwt_split(), b64url-decode the payload
14 * into a bounded stack buffer, then test membership with
15 * json_string_or_array_contains() so both the string and array `aud`
16 * forms are honoured exactly (same helper the front-door aud pin uses).
17 * No signature work — the front door validated the token before capture.
18 * Fail-closed on any decode failure once a gate is configured.
19 */
20#include "aud_match.h"
21#include "b64url.h"
22#include "json.h"
23
24/* WLCG Common JWT Profile rules 104/105: a token audienced
25 * 'https://wlcg.cern.ch/jwt/v1/any' is valid at ANY WLCG endpoint, so it is
26 * forwardable regardless of the operator's backend allow-list. Mirrors the
27 * front-door pin in validate.c. */
28#define BRIX_WLCG_ANY_AUD "https://wlcg.cern.ch/jwt/v1/any"
29
30/* A JWT payload larger than this cannot be audience-checked and is treated as
31 * not forwardable (fail-closed). Generous: real WLCG/OIDC payloads are <2k. */
32#define BRIX_AUD_PAYLOAD_MAX 8192
33
34int
35brix_token_backend_aud_ok(const ngx_str_t *bearer,
36 const ngx_array_t *ok_list, ngx_log_t *log)
37{
38 xrdjwt_seg seg[3];
39 u_char payload[BRIX_AUD_PAYLOAD_MAX];
40 ssize_t plen;
41 ngx_uint_t i;
42 const ngx_str_t *ent;
43
44 if (ok_list == NULL || ok_list->nelts == 0) {
45 return 1; /* no gate configured — passthrough unrestricted */
46 }
47
48 if (bearer == NULL || bearer->len == 0 || bearer->data == NULL) {
49 return 0;
50 }
51
52 if (xrdjwt_split((const char *) bearer->data, bearer->len, seg) != 0) {
53 ngx_log_error(NGX_LOG_INFO, log, 0,
54 "brix: backend audience gate: bearer is not a compact JWS - "
55 "not forwarding");
56 return 0;
57 }
58
59 plen = b64url_decode(seg[1].p, seg[1].n, payload, sizeof(payload));
60 if (plen <= 0) {
61 ngx_log_error(NGX_LOG_INFO, log, 0,
62 "brix: backend audience gate: token payload undecodable - "
63 "not forwarding");
64 return 0;
65 }
66
67 if (json_string_or_array_contains((const char *) payload, (size_t) plen,
68 "aud", BRIX_WLCG_ANY_AUD)) {
69 return 1;
70 }
71
72 ent = ok_list->elts;
73 for (i = 0; i < ok_list->nelts; i++) {
74 if (ent[i].len == 0 || ent[i].data == NULL) {
75 continue;
76 }
77 if (json_string_or_array_contains((const char *) payload,
78 (size_t) plen, "aud", (const char *) ent[i].data)) {
79 return 1;
80 }
81 }
82
83 /* Configured gate, no match (or no aud claim at all): fail closed. The
84 * claim values are not logged — they can carry user-identifying URLs. */
85 ngx_log_error(NGX_LOG_INFO, log, 0,
86 "brix: backend audience gate: token aud does not accept this "
87 "backend - not forwarding");
88 return 0;
89}
