1/*
2 * delete_objects.c — S3 DeleteObjects handler (POST /bucket/?delete).
3 *
4 * Parses a list of <Object><Key>…</Key></Object> entries from the request
5 * body, deletes them, and returns a <DeleteResult> XML response with
6 * <Deleted> or <Error> entries for each object.
7 *
8 * The body XML parser uses libxml2 with network access disabled, so XML
9 * entities in <Key> values are decoded correctly before filesystem resolution.
10 * ENOENT is treated as success per S3 idempotency rules.
11 */
12/* WHY: S3 DeleteObjects supports batch deletion in a single POST request —
13 * far more efficient than individual DELETE calls for pipelines that need to
14 * remove hundreds or thousands of objects. Phase-107 C4 makes the server side
15 * match: every key is collected and CONFINED first, then disposed by ONE
16 * brix_vfs_delete_many() call (one write gate, one OP_DELETE observation, and
17 * over a remote backend one signed DeleteObjects round trip per 1,000-key
18 * window instead of 1,000). The per-object stages live in
19 * delete_objects_batch.c; the XML rendering in delete_objects_xml.c. */

59 * HOW: memcpy trailer → create temp buf → ngx_memcpy copy → last_buf=1 → send header

94 * s3_delete_objects_body — body callback invoked by nginx after the POST body
95 * has been fully buffered.
96 */
97/* WHAT: Parse the POST body XML to extract <Object><Key>…</Key></Object> entries,
98 * delete each key from the filesystem, and build a DeleteResult XML response with
99 * <Deleted> or <Error> entries per object.
100 *
101 * WHY: S3 clients send DeleteObjects as a single POST with an XML body listing all
102 * keys to remove. libxml2 is used because real clients may XML-escape key
103 * names; string scanning would delete the wrong key and emit malformed XML.
104 *
105 * Per S3 semantics:
106 * - ENOENT → success (idempotent: deleting something that doesn't exist is fine)
107 * - EISDIR/EPERM on unlink → retry with rmdir flag for directories
108 * - AccessDenied → path escaped outside root_canon via s3_resolve_key()
109 *
110 * HOW:
111 * 1. Read body via brix_http_body_read_all() (cap: S3_DEL_MAX_BODY)
112 * 2. Allocate XML output buffer (S3_DEL_XML_MAX)
113 * 3. Write XML header
114 * 4. Parse <Delete><Object><Key> entries with libxml2, resolve fs_path
115 * 5. Attempt unlink → fallback rmdir on EISDIR/EPERM
116 * 6. Emit <Deleted> (success/ENOENT) or <Error> (AccessDenied/BucketNotEmpty/InternalError)
117 * 7. Finalise response via s3_delete_objects_finish()
118 *
119 * Constraints:
120 * - Max keys: S3_DEL_MAX_KEYS (1000) — rejects XML that exceeds the limit
121 * - Body cap: S3_DEL_MAX_BODY (1 MiB) — rejects oversized XML as MalformedXML
122 * - Key length: < S3_MAX_KEY — reports InvalidArgument per offending object
123 */

127/*
128 * s3_delete_parse_body — read the buffered POST body and parse it into a
129 * validated <Delete> document.
130 *
131 * WHAT: Reads the body (cap S3_DEL_MAX_BODY), parses it with the hardened
132 * libxml2 posture, and confirms the root element is <Delete>; on success
133 * *doc_out owns the parsed document (caller must xmlFreeDoc it).
134 * WHY: Body read and XML validation are one cohesive precondition stage; the
135 * client-facing MalformedXML / body-too-large errors and the 500 paths are
136 * all resolved here so the batch loop only runs on a good document.
137 * HOW: Returns NGX_OK with a document, or NGX_DONE after finalising the
138 * response with the appropriate error (nothing left for the caller to do).
139 */
140static ngx_int_t
141s3_delete_parse_body(ngx_http_request_t *r, ngx_uint_t method_slot,
142 xmlDocPtr *doc_out)
143{
144 u_char *body_buf;
145 size_t body_len;
146 xmlDocPtr doc;
147 xmlNodePtr root;
148 ngx_int_t rc;
149
150 *doc_out = NULL;
151
152 if (r->request_body == NULL) {
153 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_INTERNAL_ERROR]);
154 s3_metrics_finalize_request_method(r, method_slot,
155 NGX_HTTP_INTERNAL_SERVER_ERROR);
156 return NGX_DONE;
157 }
158
159 rc = brix_http_body_read_all(r, S3_DEL_MAX_BODY, &body_buf, &body_len);
160 if (rc == NGX_DECLINED) {
161 s3_metrics_finalize_request_method(r, method_slot,
162 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
163 "MalformedXML",
164 "Request body too large."));
165 return NGX_DONE;
166 }
167 if (rc != NGX_OK) {
168 s3_metrics_finalize_request_method(r, method_slot,
169 NGX_HTTP_INTERNAL_SERVER_ERROR);
170 return NGX_DONE;
171 }
172
173 {
174 /* Match the hardened libxml2 posture used by the WebDAV parsers
175 * (propfind/proppatch/search/lockinfo): NONET blocks network entity
176 * fetches; NO_XXE (libxml2 >= 2.13, when available) refuses external
177 * entity loading outright. NOENT/DTDLOAD are deliberately NOT set, so
178 * external entities are never substituted (no file:// XXE), and HUGE is
179 * omitted so libxml2 keeps its billion-laughs amplification cap. */
180 int opts = XML_PARSE_NONET | XML_PARSE_NOERROR | XML_PARSE_NOWARNING;
181#if defined(XML_PARSE_NO_XXE)
182 opts |= XML_PARSE_NO_XXE;
183#endif
184 doc = xmlReadMemory((const char *) body_buf, (int) body_len,
185 "delete_objects.xml", NULL, opts);
186 }
187 if (doc == NULL) {
188 s3_metrics_finalize_request_method(r, method_slot,
189 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
190 "MalformedXML",
191 "Request body is not valid DeleteObjects XML."));
192 return NGX_DONE;
193 }
194
195 root = xmlDocGetRootElement(doc);
196 if (!s3_delete_xml_name_is(root, "Delete")) {
197 xmlFreeDoc(doc);
198 s3_metrics_finalize_request_method(r, method_slot,
199 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
200 "MalformedXML",
201 "DeleteObjects root element must be Delete."));
202 return NGX_DONE;
203 }
204
205 *doc_out = doc;
206 return NGX_OK;
207}

209/*
210 * s3_delete_result_init — allocate the output XML buffer and write the
211 * <DeleteResult> preamble.
212 *
213 * WHAT: Allocates S3_DEL_XML_MAX from the request pool, points xml_buf at it,
214 * and seeds *xml_len with the length of the XML header prefix.
215 * WHY: The batch loop appends per-key elements into a fixed, pre-sized buffer
216 * (no reallocation mid-stream); this stage establishes that buffer once.
217 * HOW: Returns NGX_OK on success or NGX_ERROR on allocation failure (the
218 * caller maps that to a 500).
219 */
220static ngx_int_t
221s3_delete_result_init(ngx_http_request_t *r, ngx_buf_t *xml_buf,
222 size_t *xml_len)
223{
224 static const char header[] =
225 "<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
226 "<DeleteResult "
227 "xmlns=\"http://s3.amazonaws.com/doc/2006-03-01/\">";
228 u_char *xml_out;
229
230 xml_out = ngx_palloc(r->pool, S3_DEL_XML_MAX);
231 if (xml_out == NULL) {
232 return NGX_ERROR;
233 }
234
235 xml_buf->start = xml_out;
236 xml_buf->end = xml_out + S3_DEL_XML_MAX;
237
238 *xml_len = sizeof(header) - 1;
239 memcpy(xml_out, header, *xml_len);
240 return NGX_OK;
241}

260/*
261 * s3_delete_run_batch — collect the <Object> children of <Delete>, dispose
262 * them through ONE VFS batch call, and render a per-key result for each.
263 *
264 * WHAT: Enforces the S3_DEL_MAX_KEYS cap, requires at least one <Object>,
265 * collects every key via s3_delete_collect_one() (validate + confine — no
266 * deletes yet), then s3_delete_execute() (one brix_vfs_delete_many) and
267 * s3_delete_render() (client-ordered <Deleted>/<Error> elements).
268 * WHY: Collect-before-execute means a malformed <Object> anywhere in the body
269 * now rejects the request BEFORE anything is deleted (the old
270 * parse-and-delete loop had already removed the keys ahead of it), and the
271 * batch runs under exactly one write gate and one metric observation
272 * (phase-107 C4).
273 * HOW: Returns NGX_OK once every result element is appended (caller sends the
274 * buffered response). On any error it finalises the response here and
275 * returns NGX_DONE — the malformed sub-cases pick the exact message.
276 */
277static ngx_int_t
278s3_delete_run_batch(s3_del_ctx_t *dc, ngx_uint_t method_slot, xmlNodePtr root)
279{
280 ngx_http_request_t *r = dc->r;
281 s3_del_item_t *items;
282 xmlNodePtr obj;
283 ngx_uint_t nkeys = 0;
284
285 items = ngx_palloc(r->pool, S3_DEL_MAX_KEYS * sizeof(s3_del_item_t));
286 if (items == NULL) {
287 s3_metrics_finalize_request_method(r, method_slot,
288 NGX_HTTP_INTERNAL_SERVER_ERROR);
289 return NGX_DONE;
290 }
291
292 for (obj = root->children; obj != NULL; obj = obj->next) {
293 int malformed_500 = 0;
294
295 if (!s3_delete_xml_name_is(obj, "Object")) {
296 continue;
297 }
298
299 if (nkeys >= S3_DEL_MAX_KEYS) {
300 s3_metrics_finalize_request_method(r, method_slot,
301 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
302 "MalformedXML",
303 "DeleteObjects request contains too "
304 "many keys."));
305 return NGX_DONE;
306 }
307
308 if (s3_delete_collect_one(dc, obj, &items[nkeys], &malformed_500)
309 != NGX_OK)
310 {
311 if (malformed_500) {
312 s3_metrics_finalize_request_method(r, method_slot,
313 NGX_HTTP_INTERNAL_SERVER_ERROR);
314 } else {
315 s3_metrics_finalize_request_method(r, method_slot,
316 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
317 "MalformedXML",
318 "DeleteObjects object is missing Key."));
319 }
320 return NGX_DONE;
321 }
322 nkeys++;
323 }
324
325 if (nkeys == 0) {
326 s3_metrics_finalize_request_method(r, method_slot,
327 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
328 "MalformedXML",
329 "DeleteObjects request has no Object entries."));
330 return NGX_DONE;
331 }
332
333 if (s3_delete_execute(dc, items, nkeys) != NGX_OK) {
334 return NGX_DONE; /* response finalised (EROFS 403 or alloc 500) */
335 }
336
337 if (s3_delete_render(dc, items, nkeys) != NGX_OK) {
338 s3_metrics_finalize_request_method(r, method_slot,
339 NGX_HTTP_INTERNAL_SERVER_ERROR);
340 return NGX_DONE; /* output buffer overflow */
341 }
342
343 return NGX_OK;
344}

349 ngx_http_s3_loc_conf_t *cf;

371 dc.r = r;
372 dc.cf = cf;
373 dc.method_slot = method_slot;
374 dc.xml_buf = xml_buf;
375 dc.xml_len = &xml_len;
376
377 if (s3_delete_run_batch(&dc, method_slot, xmlDocGetRootElement(doc))
378 != NGX_OK)
379 {
380 xmlFreeDoc(doc);
381 return; /* response already finalised */
382 }
383
384 xmlFreeDoc(doc);
385 s3_delete_objects_finish(r, method_slot, xml_buf, xml_len);
386}
