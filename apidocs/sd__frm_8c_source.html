1/*
2 * sd_frm.c - the nearline (tape/MSS) backend FS driver (SP5). See header.
3 *
4 * The driver is a thin residency layer over a pluggable MSS adapter. A read of an
5 * offline object RECALLS it into the MSS online buffer (synchronously in this
6 * increment - the stub recall is a local copy), then serves from the buffer via a
7 * real fd; a staged write lands in the online buffer and is MIGRATED to tape on
8 * commit. The composing registry requires a cache tier in front (G8), so the
9 * recall is in practice the cache miss-fill sourced from tape.
10 *
11 * The built-in "stub" MSS adapter simulates tape with two local directories under
12 * a base: <base>/<key> is "on tape" (offline) and <base>/.online/<key> is the
13 * online buffer. It is what the SP5 tests drive; a real MSS (exec stagecmd, HPSS,
14 * CTA) is another adapter selected on the tape:// store-URL.
15 */

39/* Bounded synchronous recall: ensure `key` is online in the MSS buffer. Returns 0
40 * (online), or -1 (errno: ENOENT absent, EAGAIN still in-flight, EIO error). A
41 * genuinely slow MSS would return EAGAIN and the cache tier would park the open on
42 * the stage_engine waiter (the deferred async path); the stub completes at once.
43 *
44 * `*recalled` (when non-NULL) is set to 1 iff a tape->cache recall was actually
45 * initiated on this call (a real nearline miss-fill), so the caller can book the
46 * one unified `kind=tape` ledger line only for genuine recalls — never for an
47 * already-online cache hit (residency ONLINE) or an absent object. */
48static int
49frm_ensure_online(sd_frm_state *st, const char *key, int *recalled)
50{
51 off_t sz = 0;
52 time_t mt = 0;
53 int res = st->mss->residency(st->mss_ctx, key, &sz, &mt);
54
55 if (recalled != NULL) { *recalled = 0; }
56 if (res == BRIX_RESIDENCY_ONLINE) {
57 return 0;
58 }
59 if (res == BRIX_RESIDENCY_ABSENT) {
60 errno = ENOENT;
61 return -1;
62 }
63 if (recalled != NULL) { *recalled = 1; }
64 if (st->mss->recall_begin(st->mss_ctx, key) != 0) {
65 errno = EIO;
66 return -1;
67 }
68 /* One poll: a synchronous adapter (begin copied) is online now; an async MSS is
69 * still staging, so return EAGAIN and let the HTTP plane answer 202 (the open
70 * "parks" via client retry, §9.2). A later retry re-polls and completes. */
71 {
72 int p = st->mss->recall_poll(st->mss_ctx, key);
73
74 if (p == 1) {
75 return 0;
76 }
77 if (p < 0) {
78 errno = EIO;
79 return -1;
80 }
81 }
82 errno = EAGAIN;
83 return -1;
84}

86/* Fail an open: book ONE terminal tape-recall error line when this open really
87 * did trigger a recall (the caller filters EAGAIN, which is not terminal — the
88 * open parks and the client retries, §9.2 — so it is never double-counted),
89 * publish errno to the caller, and return the NULL the driver contract wants. */
90static brix_sd_obj_t *
91sd_frm_open_fail(const char *path, int book, int e, int *err_out, ngx_log_t *log)
92{
93 if (book) {
94 brix_xfer_finish(BRIX_XFER_TAPE, "in", path, NULL, 0,
95 BRIX_XFER_SRC_ERR, e, log);
96 }
97 if (err_out) {
98 *err_out = e;
99 }
100 return NULL;
101}

128static brix_sd_obj_t *
129sd_frm_open(brix_sd_instance_t *inst, const char *path, int sd_flags,
130 mode_t mode, int *err_out)
131{
132 sd_frm_state *st = SD_FRM_ST(inst);
133 brix_sd_obj_t *o;
134 int fd;
135 int e;
136 int recalled = 0;
137 ngx_log_t *log = (ngx_cycle != NULL) ? ngx_cycle->log : NULL;
138
139 (void) mode;
140 if ((sd_flags & BRIX_SD_O_WRITE) != 0) {
141 /* writes go through the staged path (migrate); a direct write-open is not
142 * supported on a nearline backend. */
143 return sd_frm_open_fail(path, 0, EROFS, err_out, log);
144 }
145 if (frm_ensure_online(st, path, &recalled) != 0) {
146 /* An already-online cache hit (recalled==0) is no recall at all. */
147 e = errno;
148 return sd_frm_open_fail(path, recalled && e != EAGAIN, e, err_out, log);
149 }
150 fd = st->mss->open_online(st->mss_ctx, path);
151 if (fd < 0) {
152 e = errno ? errno : EIO;
153 return sd_frm_open_fail(path, recalled, e, err_out, log);
154 }
155 o = sd_frm_obj_new(inst, fd);
156 if (o == NULL) {
157 (void) close(fd);
158 return sd_frm_open_fail(path, recalled, ENOMEM, err_out, log);
159 }
160 /* Terminal success of a genuine tape->cache recall: one unified ledger line
161 * (kind=tape, dir=in), byte count from the now-online object. Mirrors the
162 * async stage_engine RECALL emit so the sync read-fault path is auditable in
163 * the same schema (finding #12). */
164 if (recalled) {
165 brix_xfer_finish(BRIX_XFER_TAPE, "in", path, NULL,
166 (size_t) o->snap.size, BRIX_XFER_OK, 0, log);
167 }
168 return o;
169}

171static ngx_int_t
172sd_frm_close(brix_sd_obj_t *obj)
173{
174 if (obj == NULL) {
175 return NGX_OK;
176 }
177 if (obj->fd >= 0) {
178 (void) close(obj->fd);
179 obj->fd = -1;
180 }
181 /* The shell is NOT freed here. driver->close closes the fd/state only; the
182 * malloc'd shell (heap_shell) is owned by the caller that holds the object
183 * by pointer — brix_sd_obj_release() and the VFS adopt paths both do
184 * `close(o); if (o->heap_shell) free(o);`, and brix_vfs_adopt_obj() frees
185 * the original after copying it by value. Freeing it here double-frees the
186 * shell the moment the object is released or adopt fails (matches the posix
187 * / http / remote / pblock drivers, which all leave the shell to the owner).
188 */
189 return NGX_OK;
190}

235/* Residency (the VFS residency seam, phase-64 §9) — classify `key` via the MSS
236 * adapter WITHOUT initiating a recall, mapping the adapter's residency model onto
237 * the SD residency enum the protocol handlers consume. ABSENT ⇒ LOST (errno ENOENT
238 * so the seam can surface a missing object); ONLINE/NEARLINE/OFFLINE pass through. */
239static ngx_int_t
240sd_frm_residency(brix_sd_instance_t *inst, const char *key,
241 brix_sd_residency_t *out)
242{
243 sd_frm_state *st = SD_FRM_ST(inst);
244 off_t sz = 0;
245 time_t mt = 0;
246 int res = st->mss->residency(st->mss_ctx, key, &sz, &mt);
247
248 switch (res) {
249 case BRIX_RESIDENCY_ONLINE: *out = BRIX_SD_RES_ONLINE; break;
250 case BRIX_RESIDENCY_NEARLINE: *out = BRIX_SD_RES_NEARLINE; break;
251 case BRIX_RESIDENCY_OFFLINE: *out = BRIX_SD_RES_OFFLINE; break;
252 default: errno = ENOENT; return NGX_ERROR; /* ABSENT */
253 }
254 return NGX_OK;
255}

257/* Evict (phase-107 C2) — release the MSS ONLINE-BUFFER copy of `key`; the tape
258 * copy remains the durable one (a later recall restages it). Safe because the
259 * staged-write lifecycle guarantees ONLINE ⇒ a tape copy exists: staged_commit
260 * fails when the migrate fails, and abort purges the buffer copy, so outside a
261 * mid-commit crash window nothing is ever ONLINE without its tape counterpart.
262 * Idempotent: NEARLINE/OFFLINE (already released) is NGX_OK with 0 bytes;
263 * ABSENT is ENOENT (the logical object does not exist — distinct from "already
264 * evicted"). Bytes reclaimed = the residency-reported size of the buffer copy.
265 * No *_cred twin: the MSS adapter has no per-user leg (every verb runs as the
266 * service), so a cred-accepting slot here would BE the confused deputy — the
267 * *_maybe_cred forwarder instead refuses per-user evicts in DENY mode. */
268static ngx_int_t
269sd_frm_evict(brix_sd_instance_t *inst, const char *path, uint64_t *bytes_out)
270{
271 sd_frm_state *st = SD_FRM_ST(inst);
272 off_t sz = 0;
273 time_t mt = 0;
274 int res = st->mss->residency(st->mss_ctx, path, &sz, &mt);
275
276 if (bytes_out != NULL) {
277 *bytes_out = 0;
278 }
279 if (res == BRIX_RESIDENCY_ABSENT) {
280 errno = ENOENT;
281 return NGX_ERROR;
282 }
283 if (res != BRIX_RESIDENCY_ONLINE) {
284 return NGX_OK; /* already released — idempotent */
285 }
286 if (st->mss->purge == NULL || st->mss->purge(st->mss_ctx, path) != 0) {
287 errno = EIO;
288 return NGX_ERROR;
289 }
290 if (bytes_out != NULL) {
291 *bytes_out = (uint64_t) (sz > 0 ? sz : 0);
292 }
293 return NGX_OK;
294}

296static ngx_int_t
297sd_frm_recall_common(brix_sd_instance_t *inst, const char *key,
298 char reqid_out[40], const char *principal)
299{
300 sd_frm_state *st = SD_FRM_ST(inst);
301 ngx_log_t *log = (ngx_cycle != NULL) ? ngx_cycle->log : NULL;
302 int recalled = 0;
303
304 if (reqid_out != NULL) {
305 reqid_out[0] = '\0'; /* synchronous recall: no parking handle */
306 }
307 if (frm_ensure_online(st, key, &recalled) == 0) {
308 /* The cache-fill path (sd_cache) drives every nearline miss through this
309 * verb, so a genuine tape->cache recall books its one unified ledger
310 * line here (kind=tape, dir=in) — the sync counterpart to the async
311 * stage_engine RECALL emit (finding #12). Byte count = the now-online
312 * object size. An already-online object (recalled==0) is a plain fill. */
313 if (recalled) {
314 off_t sz = 0;
315 time_t mt = 0;
316
317 (void) st->mss->residency(st->mss_ctx, key, &sz, &mt);
318 brix_xfer_finish(BRIX_XFER_TAPE, "in", key, principal,
319 (size_t) (sz > 0 ? sz : 0), BRIX_XFER_OK, 0, log);
320 }
321 return NGX_OK; /* online now - the cache tier does a normal fill */
322 }
323 {
324 int e = errno;
325
326 /* Terminal recall failure books a kind=tape/error line; EAGAIN (async
327 * still in flight) is non-terminal and must not be recorded. */
328 if (recalled && e != EAGAIN) {
329 brix_xfer_finish(BRIX_XFER_TAPE, "in", key, principal, 0,
330 BRIX_XFER_SRC_ERR, e, log);
331 }
332 return (e == EAGAIN) ? NGX_AGAIN : NGX_ERROR;
333 }
334}

342/* recall_cred (phase-107 C2) — SAME recall, attributed. The MSS adapter has no
343 * per-user execution leg (every verb runs as the service — the reason evict has
344 * no twin), but the tape LEDGER does: this twin books the kind=tape line under
345 * the requesting principal instead of anonymously, so a per-user kXR_prepare
346 * on an frm export is auditable to who asked for the tape mount. The recall is

358/* §3.7 pure-tape enumeration: the dir cursor snapshots the MSS listing at
359 * opendir via the adapter's `list` verb (see sd_frm.h); no verb ⇒ ENOTSUP.
360 * Malloc-owned throughout (no pool); closedir frees snapshot+cursor+dir. */
361
362typedef struct {
363 brix_sd_dirent_t *ents;
364 size_t n;
365 size_t cap;
366 size_t next;
367} frm_dir_state;
368
369static int
370sd_frm_list_cb(void *ud, const char *name, int is_dir)
371{
372 frm_dir_state *ds = ud;
373
374 if (ds->n == ds->cap) {
375 size_t ncap = ds->cap ? ds->cap * 2 : 64;
376 brix_sd_dirent_t *ne = realloc(ds->ents, ncap * sizeof(*ne));
377
378 if (ne == NULL) {
379 return 1; /* stop early; serve what we have */
380 }
381 ds->ents = ne;
382 ds->cap = ncap;
383 }
384 snprintf(ds->ents[ds->n].name, sizeof(ds->ents[ds->n].name), "%s", name);
385 ds->ents[ds->n].d_type = is_dir ? DT_DIR : DT_REG;
386 ds->n++;
387 return 0;
388}

390static brix_sd_dir_t *
391sd_frm_opendir(brix_sd_instance_t *inst, const char *path, int *err_out)
392{
393 sd_frm_state *st = inst->state;
394 frm_dir_state *ds;
395 brix_sd_dir_t *dir;
396
397 if (st->mss->list == NULL) {
398 if (err_out != NULL) { *err_out = ENOTSUP; }
399 return NULL;
400 }
401 ds = calloc(1, sizeof(*ds));
402 dir = calloc(1, sizeof(*dir));
403 if (ds == NULL || dir == NULL) {
404 free(ds);
405 free(dir);
406 if (err_out != NULL) { *err_out = ENOMEM; }
407 return NULL;
408 }
409 if (st->mss->list(st->mss_ctx, path, sd_frm_list_cb, ds) != 0) {
410 int e = errno != 0 ? errno : EIO;
411
412 free(ds->ents);
413 free(ds);
414 free(dir);
415 if (err_out != NULL) { *err_out = e; }
416 return NULL;
417 }
418 dir->inst = inst;
419 dir->state = ds;
420 return dir;
421}

496/* ---- Construct an sd_frm nearline backend instance ----
497 *
498 * WHAT: Allocates the sd_frm instance and state, selects the MSS adapter (exec if
499 * requested and usable, otherwise the built-in stub), and wires the state to the
500 * frm driver. Returns the instance, or NULL with errno set (EINVAL for an empty
501 * location, ENOMEM for any allocation failure) after freeing partial state.
502 *
503 * WHY: The single public entry point for the nearline (tape/MSS) backend; the two
504 * adapter choices are delegated to helpers so this stays a linear early-return
505 * sequence below the complexity cap.
506 *
507 * HOW:
508 * 1. Reject an empty `location` with EINVAL.
509 * 2. Allocate inst + state; on failure free both and return ENOMEM.
510 * 3. Try the exec adapter; on hard failure free both and return NULL.
511 * 4. If no adapter is set yet, fall back to the stub; on failure free and return.
512 * 5. Publish the driver and state onto the instance and return it.
513 */
514brix_sd_instance_t *
515brix_sd_frm_create(const char *adapter, const char *location, ngx_log_t *log)
516{
517 brix_sd_instance_t *inst;
518 sd_frm_state *st;
519
520 if (location == NULL || location[0] == '\0') {
521 errno = EINVAL;
522 return NULL;
523 }
524 inst = calloc(1, sizeof(*inst));
525 st = calloc(1, sizeof(*st));
526 if (inst == NULL || st == NULL) {
527 free(inst);
528 free(st);
529 errno = ENOMEM;
530 return NULL;
531 }
532 st->log = log;
533
534 /* Adapter precedence: library-native (dlopen) → exec (stagecmd) → stub. The
535 * lib select never hard-fails (an absent vendor .so degrades gracefully), so
536 * only the exec/stub selects can abort the create on a real allocation error. */
537 (void) frm_select_lib_adapter(st, adapter, location, log);
538
539 if (st->mss == NULL
540 && frm_select_exec_adapter(st, adapter, location, log) != 0)
541 {
542 free(inst);
543 free(st);
544 return NULL;
545 }
546 if (st->mss == NULL
547 && frm_select_stub_adapter(st, adapter, location, log) != 0)
548 {
549 free(inst);
550 free(st);
551 return NULL;
552 }
553
554 inst->driver = &brix_sd_frm_driver;
555 inst->caps = brix_sd_frm_driver.caps; /* effective caps default = descriptor
556 * caps (matches brix_sd_instance_create);
557 * without this brix_sd_caps() reports 0 and
558 * the CAP_NEARLINE gates never see tape. */
559 inst->log = log;
560 inst->pool = NULL;
561 inst->state = st;
562 inst->domain = BRIX_VFS_DOMAIN_EXPORT; /* strict default (C9) */
563 return inst;
564}
