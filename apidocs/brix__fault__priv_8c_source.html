1/*
2 * brix_fault_priv.c — privileged ("root-ful") fault levers for brix-fault-proxy.
3 *
4 * See brix_fault_priv.h for the rationale. This unit is the ONLY place the proxy
5 * touches host-global network state, and it does so through three subsystems, each
6 * fully owned so teardown is a single clean removal:
7 *
8 * netem — a `tc qdisc ... netem` on the configured NIC's egress. Implemented
9 * in brix_fault_priv_netem.c (see brix_fault_priv_internal.h). We keep a
10 * per-feature fragment table (delay, loss, corrupt, …) and re-emit the
11 * whole qdisc on every change (`tc qdisc replace`), so features compose
12 * and any one can be cleared independently. This is REAL below-TCP
13 * impairment: loss/corrupt force genuine retransmits or checksum drops,
14 * reorder/duplicate deliver genuine out-of-order / doubled PACKETS —
15 * none of which the userland byte-stream levers can honestly produce.
16 * cut — an `nft` table (inet brix_fault_proxy) whose input-hook rules match
17 * THIS proxy's own 4-tuples (listen port / target ports) and hand the
18 * kernel a verdict: `reject with tcp reset` (a correctly-sequenced RST,
19 * i.e. an on-path reset attack), `reject with icmpx ...` (a forged ICMP
20 * unreachable), or `drop` (a true silent black hole — the peer keeps
21 * retransmitting into the void, unlike the userland `hang` which still
22 * completes the TCP handshake and ACKs).
23 * mtu — shrink the NIC MTU to wedge large transfers behind a next-hop MTU
24 * black hole / forced fragmentation, a below-TCP effect no relay can fake.
25 *
26 * NON-GOAL: forging spoofed-source ICMP "fragmentation needed" via a raw socket.
27 * Hardened stacks (RFC 5927) validate the quoted sequence number, which an off-path
28 * injector cannot know, so it is unreliable; `priv mtu` achieves the same wedged-
29 * transfer outcome deterministically.
30 *
31 * NO SHELL: every external command is fork()+execvp() with an explicit argv; the
32 * nft ruleset is piped to `nft -f -` over a pipe. Interface names and numeric
33 * operands are validated before use.
34 */

71/* Map a cut mode to an nft verdict clause, or NULL if the mode is unknown. */
72static const char *
73cut_verdict(const char *mode)
74{
75 if (strcmp(mode, "rst") == 0) return "reject with tcp reset";
76 if (strcmp(mode, "drop") == 0) return "drop";
77 if (strcmp(mode, "icmp-admin") == 0) return "reject with icmpx type admin-prohibited";
78 if (strcmp(mode, "icmp-host") == 0) return "reject with icmpx type host-unreachable";
79 if (strcmp(mode, "icmp-net") == 0) return "reject with icmpx type no-route";
80 if (strcmp(mode, "icmp-port") == 0) return "reject with icmpx type port-unreachable";
81 return NULL;
82}

84/* Install (or replace) the nft table cutting `dir` traffic with `mode`. dir:
85 * 0 both / 1 up (proxy<->upstream) / 2 down (proxy<->client). Caller holds
86 * g_lock. */
87static int
88cut_apply(const char *mode, int dir, char *reply, size_t rsz)
89{
90 const char *v = cut_verdict(mode);
91 if (!v) {
92 snprintf(reply, rsz, "err: unknown cut mode '%s'\n", mode);
93 return -1;
94 }
95
96 /* Build a target-port set "{ p1, p2 }" for the upstream direction. */
97 char pset[128] = "";
98 int po = snprintf(pset, sizeof(pset), "{ ");
99 for (int i = 0; i < g_ntports; i++) {
100 po += snprintf(pset + po, sizeof(pset) - po, "%s%d",
101 i ? ", " : "", g_tports[i]);
102 }
103 snprintf(pset + po, sizeof(pset) - po, " }");
104
105 char rs[768];
106 int n = snprintf(rs, sizeof(rs),
107 "add table inet brix_fault_proxy\n"
108 "delete table inet brix_fault_proxy\n"
109 "add table inet brix_fault_proxy\n"
110 "add chain inet brix_fault_proxy in "
111 "{ type filter hook input priority -150; policy accept; }\n");
112 if (dir != 1) { /* down: packets arriving from the client (dport=listen) */
113 n += snprintf(rs + n, sizeof(rs) - n,
114 "add rule inet brix_fault_proxy in tcp dport %d %s\n",
115 g_listen_port, v);
116 }
117 if (dir != 2 && g_ntports > 0) { /* up: replies from upstream (sport=target) */
118 n += snprintf(rs + n, sizeof(rs) - n,
119 "add rule inet brix_fault_proxy in tcp sport %s %s\n", pset, v);
120 }
121
122 char *argv[] = { "nft", "-f", "-", NULL };
123 if (priv_run_stdin(argv, rs) != 0) {
124 snprintf(reply, rsz, "err: nft failed (see proxy stderr)\n");
125 return -1;
126 }
127 g_nft_on = 1;
128 return 0;
129}

172/* Handle `priv mtu <bytes>|restore`. Caller holds g_lock. */
173static int
174cmd_mtu(char *args, char *reply, size_t rsz)
175{
176 if (g_iface[0] == '\0') {
177 snprintf(reply, rsz, "err: no --priv-iface configured for mtu\n");
178 return -1;
179 }
180 char *tok = strtok(args, " ");
181 if (!tok) {
182 snprintf(reply, rsz, "err: mtu needs <bytes> or 'restore'\n");
183 return -1;
184 }
185 if (strcmp(tok, "restore") == 0) {
186 if (g_mtu_saved <= 0) {
187 snprintf(reply, rsz, "ok (nothing to restore)\n");
188 return 0;
189 }
190 int rc = mtu_set(g_mtu_saved);
191 g_mtu_saved = -1;
192 if (rc != 0) {
193 snprintf(reply, rsz, "err: ip link set mtu failed\n");
194 }
195 return rc;
196 }
197 char *end;
198 long b = strtol(tok, &end, 10);
199 if (end == tok || b < 68 || b > 65535) { /* IPv4 minimum link MTU is 68 */
200 snprintf(reply, rsz, "err: mtu out of range (68..65535)\n");
201 return -1;
202 }
203 if (g_mtu_saved < 0) {
204 g_mtu_saved = mtu_current(); /* remember the real MTU once */
205 }
206 if (mtu_set((int) b) != 0) {
207 snprintf(reply, rsz, "err: ip link set mtu failed\n");
208 return -1;
209 }
210 return 0;
211}

215int
216fp_priv_enable(const char *iface, int listen_port,
217 const int *target_ports, int nports, const char **err)
218{
219 if (geteuid() != 0) {
220 *err = "privileged levers require root (euid 0)";
221 return -1;
222 }
223 if (iface && iface[0] && !valid_iface(iface)) {
224 *err = "invalid or nonexistent --priv-iface";
225 return -1;
226 }
227 pthread_mutex_lock(&g_lock);
228 memset(g_iface, 0, sizeof(g_iface));
229 if (iface) {
230 snprintf(g_iface, sizeof(g_iface), "%s", iface);
231 }
232 g_listen_port = listen_port;
233 g_ntports = 0;
234 for (int i = 0; i < nports && g_ntports < (int) (sizeof(g_tports) / sizeof(g_tports[0])); i++) {
235 g_tports[g_ntports++] = target_ports[i];
236 }
237 g_on = 1;
238 pthread_mutex_unlock(&g_lock);
239 (void) err;
240 return 0;
241}

249/* `priv frag-drop` / `priv first-not-syn-drop` — kernel-crafted verdicts a
250 * mis-configured DPI imposes that TCP-flow-centric proxies cannot fake:
251 * frag-drop — drop every IP fragment (fragmented large UDP / EDNS0 DNS
252 * dies while small requests succeed).
253 * first-not-syn — after state loss (failover / ECMP rehash) drop any packet
254 * to the listen port that isn't a fresh SYN (mid-stream
255 * freeze). Uses conntrack `invalid`.
256 * Each (re)creates the shared table with just its rule, so it is mutually
257 * exclusive with `cut` (last writer wins); torn down with the table on exit. */
258static int
259special_apply(const char *what, char *reply, size_t rsz)
260{
261 char rs[768];
262 int n = snprintf(rs, sizeof(rs),
263 "add table inet brix_fault_proxy\n"
264 "delete table inet brix_fault_proxy\n"
265 "add table inet brix_fault_proxy\n"
266 "add chain inet brix_fault_proxy in "
267 "{ type filter hook input priority -150; policy accept; }\n");
268 if (strcmp(what, "frag-drop") == 0) {
269 n += snprintf(rs + n, sizeof(rs) - n,
270 "add rule inet brix_fault_proxy in ip frag-off & 0x1fff != 0 drop\n");
271 } else if (strcmp(what, "first-not-syn-drop") == 0) {
272 n += snprintf(rs + n, sizeof(rs) - n,
273 "add rule inet brix_fault_proxy in tcp dport %d ct state invalid drop\n",
274 g_listen_port);
275 } else {
276 snprintf(reply, rsz, "err: unknown priv verdict '%s'\n", what);
277 return -1;
278 }
279 char *argv[] = { "nft", "-f", "-", NULL };
280 if (priv_run_stdin(argv, rs) != 0) {
281 snprintf(reply, rsz, "err: nft failed (see proxy stderr)\n");
282 return -1;
283 }
284 g_nft_on = 1;
285 return 0;
286}

290priv_cut(char *rest, char *reply, size_t rsz)
291{

296 return;

316/* Route one privileged sub-verb. Caller holds g_lock. */
317static void
318priv_dispatch(const char *verb, char *rest, char *reply, size_t rsz)
319{
320 if (strcmp(verb, "netem") == 0) {
321 netem_command(rest, reply, rsz);
322 } else if (strcmp(verb, "cut") == 0) {
323 priv_cut(rest, reply, rsz);
324 } else if (strcmp(verb, "uncut") == 0) {
325 if (cut_clear() != 0) {
326 snprintf(reply, rsz, "err: nft delete failed\n");
327 }
328 } else if (strcmp(verb, "frag-drop") == 0 ||
329 strcmp(verb, "first-not-syn-drop") == 0) {
330 special_apply(verb, reply, rsz);
331 } else if (strcmp(verb, "strip-opt") == 0) {
332 /* TCP-option stripping (WScale/SACK) needs an NFQUEUE/eBPF option-rewrite
333 * that isn't portable via nft alone — degrade with a clear reason rather
334 * than pretend. The rcvbuf/sndbuf clamps approximate the window effect. */
335 snprintf(reply, rsz, "err: strip-opt requires NFQUEUE/eBPF option "
336 "surgery (unavailable); approximate with rcvbuf\n");
337 } else if (strcmp(verb, "mtu") == 0) {
338 cmd_mtu(rest, reply, rsz);
339 } else if (strcmp(verb, "clear") == 0) {
340 priv_clear_all();
341 } else if (strcmp(verb, "status") == 0) {
342 snprintf(reply, rsz, "priv on iface=%s nft=%s mtu_saved=%d\n",
343 g_iface[0] ? g_iface : "(none)", g_nft_on ? "yes" : "no",
344 g_mtu_saved);
345 } else {
346 snprintf(reply, rsz, "err: unknown priv sub-command '%s'\n", verb);
347 }
348}

350int
351fp_priv_command(char *args, char *reply, size_t rsz)
352{
353 snprintf(reply, rsz, "ok\n");
354 if (!g_on) {
355 snprintf(reply, rsz,
356 "err: privileged mode not enabled (need root + --privileged)\n");
357 return 1;
358 }
359
360 char *verb = strtok(args, " ");
361 char *rest = strtok(NULL, "");
362 char empty[1] = "";
363 if (!verb) {
364 snprintf(reply, rsz, "err: priv needs a sub-command\n");
365 return 1;
366 }
367 if (!rest) {
368 rest = empty;
369 }
370
371 pthread_mutex_lock(&g_lock);
372 priv_dispatch(verb, rest, reply, rsz);
373 pthread_mutex_unlock(&g_lock);
374 return 1;
375}
