6/*
7 * unified.c — cross-protocol metric label tables and classification helpers.
8 *
9 * WHAT: Owns the shared label vocabulary for the unified metrics API declared in
10 * unified.h — the static name tables (proto/op/err/auth/tpc-direction) and
11 * the latency-bucket bound table — plus the name-lookup and classification
12 * helpers: brix_metric_proto_name / _op_name / _err_name, the auth-slot
13 * mapper (brix_metric_auth_slot) and its name form, and the errno→err /
14 * http-status→err classifiers. The record-side mutators live in
15 * unified_record.c and the Prometheus exporter in unified_export*.c.
16 * WHY: One vocabulary behind one exporter keeps every protocol plane's dashboard
17 * directly comparable and keeps labels low-cardinality (INVARIANT #8).
18 * Concentrating the tables and the pure classifiers here — separate from
19 * the SHM mutators and the exporter — keeps every file single-concern and
20 * under the file-size budget (coding-standards §1).
21 * HOW: Static name tables map each enum to its label string; three of them
22 * (auth names, tpc-direction names, latency bounds) are shared with the
23 * exporter/record files and so carry external linkage via
24 * unified_internal.h. The classification helpers are pure functions —
25 * table lookups and small switch/if ladders — with no SHM or I/O side
26 * effects, so they are trivially reusable across protocols.
27 */

247/*
248 * brix_metric_auth_slot — map an identity auth_method bitmask to one
249 * BRIX_METRIC_AUTH_* slot. Tested in priority order (GSI, TOKEN, SSS, S3KEY,
250 * UNIX, KRB5); returns BRIX_METRIC_AUTH_NONE when no known bit is set.
251 */
252ngx_uint_t
253brix_metric_auth_slot(ngx_uint_t auth_method)
254{
255 if (auth_method & BRIX_AUTHN_GSI) {
256 return BRIX_METRIC_AUTH_GSI;
257 }
258 if (auth_method & BRIX_AUTHN_TOKEN) {
259 return BRIX_METRIC_AUTH_TOKEN;
260 }
261 if (auth_method & BRIX_AUTHN_SSS) {
262 return BRIX_METRIC_AUTH_SSS;
263 }
264 if (auth_method & BRIX_AUTHN_S3KEY) {
265 return BRIX_METRIC_AUTH_S3KEY;
266 }
267 if (auth_method & BRIX_AUTHN_UNIX) {
268 return BRIX_METRIC_AUTH_UNIX;
269 }
270 if (auth_method & BRIX_AUTHN_KRB5) {
271 return BRIX_METRIC_AUTH_KRB5;
272 }
273 if (auth_method & BRIX_AUTHN_HOST) {
274 return BRIX_METRIC_AUTH_HOST;
275 }
276 if (auth_method & BRIX_AUTHN_PWD) {
277 return BRIX_METRIC_AUTH_PWD;
278 }
279
280 return BRIX_METRIC_AUTH_NONE;
281}

289/*
290 * brix_metric_err_from_errno — classify a POSIX errno into an
291 * brix_err_class_t bucket (0→NONE, ENOENT/ENOTDIR→NOT_FOUND,
292 * EACCES/EPERM→FORBIDDEN, EIO/ENOMEM/ENOSPC→IO, else OTHER).
293 */
294brix_err_class_t
295brix_metric_err_from_errno(int sys_errno)
296{
297 switch (sys_errno) {
298 case 0:
299 return BRIX_ERR_NONE;
300 case ENOENT:
301 case ENOTDIR:
302 return BRIX_ERR_NOT_FOUND;
303 case EACCES:
304 case EPERM:
305 case EROFS:
306 /* phase-110 W4: a read-only-export refusal (the phase-105 typed
307 * mutation gate's EROFS) is a permission-class outcome, not "other":
308 * one $brix_status / JSON "status" / io_ops{status} word for it on
309 * every plane. */
310 return BRIX_ERR_FORBIDDEN;
311 case EIO:
312 case ENOMEM:
313 case ENOSPC:
314 return BRIX_ERR_IO;
315 default:
316 return BRIX_ERR_OTHER;
317 }
318}

320/*
321 * brix_metric_err_from_http_status — classify an HTTP status code into an
322 * brix_err_class_t bucket (2xx/3xx→NONE, 404→NOT_FOUND, 401/403→FORBIDDEN,
323 * 5xx→IO, else OTHER), so WebDAV/S3 outcomes share the unified error vocabulary.
324 */
325brix_err_class_t
326brix_metric_err_from_http_status(ngx_uint_t status)
327{
328 if (status >= 200 && status < 400) {
329 return BRIX_ERR_NONE;
330 }
331 if (status == NGX_HTTP_NOT_FOUND) {
332 return BRIX_ERR_NOT_FOUND;
333 }
334 if (status == NGX_HTTP_FORBIDDEN || status == NGX_HTTP_UNAUTHORIZED) {
335 return BRIX_ERR_FORBIDDEN;
336 }
337 if (status >= 500 && status < 600) {
338 return BRIX_ERR_IO;
339 }
340
341 return BRIX_ERR_OTHER;
342}
