1/*
2 * health_check.c — Phase 22 active stream health checks (see health_check.h).
3 *
4 * The probe is a self-contained async XRootD client connection. Its wire
5 * framing mirrors the proven upstream bootstrap (src/upstream/bootstrap.c):
6 * the bootstrap write buffer is built by the shared
7 * brix_upstream_build_bootstrap(), and responses are read as uniform 8-byte
8 * ServerResponseHdr + dlen-byte body frames, exactly as the upstream read loop
9 * does. Unlike the upstream path it carries no client context — on bootstrap
10 * completion it sends kXR_ping (or kXR_stat "/") and reports the verdict to the
11 * registry rather than forwarding a saved client request.
12 *
13 * A single deadline timer (hc_timeout_ms) bounds the whole probe; a hung server
14 * that accepts the TCP connection but never answers is failed on timeout.
15 */

59/*
60 * Single termination point for a probe (called from every error/success path).
61 * Reports the verdict to the shared registry, bumps metrics, and tears down all
62 * probe state. MUST be the last thing any caller does with hc: it destroys the
63 * pool that backs hc itself, so hc is dangling on return.
64 */
65void
66brix_hc_finish(brix_hc_ctx_t *hc, int passed)
67{
68 /* Stash the pool before any registry call: hc lives inside it and is freed
69 * by the ngx_destroy_pool() at the end. */
70 ngx_pool_t *pool = hc->pool;
71
72 /* Cancel the deadline timer and close the socket before reporting, so a
73 * verdict callback can never observe a half-torn-down probe. */
74 if (hc->tev.timer_set) {
75 ngx_del_timer(&hc->tev);
76 }
77 if (hc->conn != NULL) {
78 ngx_close_connection(hc->conn);
79 hc->conn = NULL;
80 }
81
82 if (passed) {
83 brix_srv_hc_pass(hc->host, hc->port);
84 BRIX_HC_METRIC_INC(hc_pass_total);
85 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, hc->log, 0,
86 "brix: health check: %s:%d passed",
87 hc->host, (int) hc->port);
88 } else {
89 if (brix_srv_hc_fail(hc->host, hc->port,
90 hc->threshold, hc->blacklist_ms))
91 {
92 BRIX_HC_METRIC_INC(hc_blacklist_total);
93 ngx_log_error(NGX_LOG_WARN, hc->log, 0,
94 "brix: health check: %s:%d blacklisted after"
95 " %ui consecutive failures",
96 hc->host, (int) hc->port, (ngx_uint_t) hc->threshold);
97 }
98 BRIX_HC_METRIC_INC(hc_fail_total);
99 }
100
101 ngx_destroy_pool(pool); /* frees hc itself (last use above) */
102}

118/*
119 * Allocate the probe pool + ctx and seed the identity/config fields for
120 * brix_hc_start(). Returns the ctx, or NULL after releasing the claim.
121 * Allocation failures before the probe is viable count as neither pass nor
122 * fail toward blacklisting (threshold/blacklist_ms = 0); they only clear
123 * hc_in_progress so the slot is re-probed next interval.
124 */
125static brix_hc_ctx_t *
126brix_hc_ctx_create(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *conf,
127 const char *host, uint16_t port)
128{
129 ngx_pool_t *pool;
130 brix_hc_ctx_t *hc;
131
132 pool = ngx_create_pool(BRIX_HC_POOL_SIZE, cycle->log);
133 if (pool == NULL) {
134 brix_srv_hc_fail(host, port, 0, 0); /* release the claim */
135 return NULL;
136 }
137 hc = ngx_pcalloc(pool, sizeof(*hc));
138 if (hc == NULL) {
139 ngx_destroy_pool(pool);
140 brix_srv_hc_fail(host, port, 0, 0);
141 return NULL;
142 }
143 hc->pool = pool;
144 hc->log = cycle->log;
145 hc->conf = conf;
146 hc->phase = XRD_HC_HANDSHAKE;
147 hc->probe_type = conf->hc.type;
148 hc->threshold = (uint32_t) conf->hc.threshold;
149 hc->blacklist_ms = conf->hc.blacklist_ms;
150 hc->port = port;
151 ngx_cpystrn((u_char *) hc->host, (u_char *) host, sizeof(hc->host));
152 return hc;
153}

155/*
156 * Create the non-blocking probe socket and wire it into an nginx connection
157 * for brix_hc_start(). Returns NGX_OK with hc->conn owning the fd, or
158 * NGX_ERROR after full pre-connection cleanup (socket closed by hand,
159 * brix_hc_pre_connect_fail() frees res, records the failure, destroys the
160 * pool).
161 */
162static ngx_int_t
163brix_hc_open_conn(brix_hc_ctx_t *hc, const struct sockaddr *sa)
164{
165 ngx_connection_t *c;
166 ngx_socket_t fd;
167
168 /* Until hc->conn is wired up, the socket is owned manually and must be
169 * closed by hand on error (the pool does not know about a bare fd). These
170 * pre-connection failures count as a real probe failure (they use the
171 * configured threshold/blacklist). */
172 fd = ngx_socket(sa->sa_family, SOCK_STREAM, 0);
173 if (fd == (ngx_socket_t) -1 || ngx_nonblocking(fd) == -1) {
174 if (fd != (ngx_socket_t) -1) { ngx_close_socket(fd); }
175 brix_hc_pre_connect_fail(hc->host, hc->port, hc->conf, hc->pool);
176 return NGX_ERROR;
177 }
178
179 c = ngx_get_connection(fd, hc->log);
180 if (c == NULL) {
181 ngx_close_socket(fd);
182 brix_hc_pre_connect_fail(hc->host, hc->port, hc->conf, hc->pool);
183 return NGX_ERROR;
184 }
185 /* From here the connection (and therefore the fd) is owned by hc->conn and
186 * released through brix_hc_finish() -> ngx_close_connection(). */
187 c->pool = hc->pool;
188 c->data = hc;
189 c->recv = ngx_recv;
190 c->send = ngx_send;
191 c->read->handler = brix_hc_read_handler;
192 c->write->handler = brix_hc_write_handler;
193 c->read->log = c->write->log = hc->log;
194 hc->conn = c;
195 return NGX_OK;
196}

198/*
199 * Build the pipelined bootstrap write buffer for brix_hc_start(). Returns
200 * NGX_OK with hc->wbuf/wbuf_len/wbuf_pos queued for sending, or NGX_ERROR on
201 * allocation failure (no cleanup here; the caller reports the verdict).
202 */
203static ngx_int_t
204brix_hc_queue_bootstrap(brix_hc_ctx_t *hc)
205{
206 size_t bslen;
207
208 /* Step F: with an outbound TLS ctx the probe can complete a kXR_gotoTLS
209 * upgrade, so advertise kXR_ableTLS — a brix server only answers gotoTLS
210 * to TLS-capable clients (it lets others finish in cleartext, which would
211 * silently skip the deep-probe path). A TLS-capable probe must NOT
212 * pipeline the plaintext login behind the protocol request: on gotoTLS
213 * the server hands every pending cleartext byte to the TLS layer, and a
214 * pre-sent login corrupts the handshake ("packet length too long"). So
215 * it sends handshake + protocol only and defers login until the protocol
216 * verdict. Without a ctx the flag byte stays 0 and the full pipelined
217 * bootstrap is byte-identical to pre-Step-F probes. */
218#if (NGX_SSL)
219 hc->tls_capable = (hc->conf->upstream_tls
220 && hc->conf->upstream_tls_ctx != NULL);
221#endif
222 /* Always allocate (and build) all three frames; wbuf_len truncates the
223 * SEND to handshake + protocol for a TLS-capable probe. */
224 bslen = XRD_HANDSHAKE_LEN + sizeof(ClientProtocolRequest)
225 + sizeof(ClientLoginRequest);
226 hc->wbuf = ngx_palloc(hc->pool, bslen);
227 if (hc->wbuf == NULL) {
228 return NGX_ERROR;
229 }
230 brix_upstream_build_bootstrap_flags(hc->wbuf,
231 hc->tls_capable ? kXR_ableTLS : 0);
232 hc->wbuf_len = hc->tls_capable
233 ? bslen - sizeof(ClientLoginRequest) : bslen;
234 hc->wbuf_pos = 0;
235 return NGX_OK;
236}

238/*
239 * Kick off the connect() for brix_hc_start() and hand the probe to the event
240 * loop. Every failure path ends in brix_hc_finish(), so the claim is always
241 * released.
242 */
243static void
244brix_hc_begin_connect(brix_hc_ctx_t *hc, const struct sockaddr *sa,
245 socklen_t salen)
246{
247 int rc;
248
249 /* Non-blocking connect: rc == 0 means it completed immediately (typical for
250 * localhost/loopback); EINPROGRESS means it is in flight; any other error
251 * is a hard failure. */
252 rc = connect(hc->conn->fd, sa, salen);
253
254 if (rc == -1 && ngx_socket_errno != NGX_EINPROGRESS) {
255 brix_hc_finish(hc, 0);
256 return;
257 }
258 if (ngx_handle_write_event(hc->conn->write, 0) != NGX_OK) {
259 brix_hc_finish(hc, 0);
260 return;
261 }
262
263 if (rc == 0) {
264 /* Connected synchronously: send the bootstrap now; the reply arrives
265 * via the read handler. */
266 hc->connecting = 0;
267 if (brix_hc_flush(hc) == NGX_ERROR) {
268 brix_hc_finish(hc, 0);
269 }
270 } else {
271 /* In progress: the first writable event runs brix_hc_write_handler,
272 * which confirms SO_ERROR and then flushes the bootstrap. */
273 hc->connecting = 1; /* finish in the write handler */
274 }
275}

277/*
278 * Continuation of brix_hc_start() once the target's address is known
279 * (phase-116: the resolution is async — a literal or cache hit lands
280 * synchronously, a hostname from the driver). Sets up a non-blocking
281 * connection, queues the pipelined bootstrap, arms the deadline timer, and
282 * starts connect(). Control then returns to the event loop; the probe
283 * completes asynchronously in the read/write handlers, always ending at
284 * brix_hc_finish().
285 *
286 * Ownership note: every early-exit path here MUST release the claim — either
287 * via brix_hc_finish() (once hc reports a verdict) or via a direct
288 * brix_srv_hc_fail() (before hc is viable). The helpers above uphold this on
289 * their own failure paths. Otherwise the slot stays hc_in_progress=1 forever
290 * and is never probed again.
291 */
292static void
293brix_hc_resolved(brix_dns_req_t *req)
294{
295 brix_hc_ctx_t *hc = req->data;
296 const struct sockaddr *sa;
297 socklen_t salen;
298
299 if (req->rc != NGX_OK || req->naddrs == 0) {
300 ngx_log_error(NGX_LOG_WARN, hc->log, 0,
301 "brix: health check: cannot resolve %s:%d: %s",
302 hc->host, (int) hc->port,
303 req->error ? req->error : "unknown");
304 brix_hc_pre_connect_fail(hc->host, hc->port, hc->conf, hc->pool);
305 return;
306 }
307 sa = (const struct sockaddr *) &req->addrs[0].ss;
308 salen = req->addrs[0].len;
309
310 if (brix_hc_open_conn(hc, sa) != NGX_OK) {
311 return;
312 }
313
314 if (brix_hc_queue_bootstrap(hc) != NGX_OK) {
315 brix_hc_finish(hc, 0);
316 return;
317 }
318
319 /* Single deadline for the whole probe. */
320 hc->tev.handler = brix_hc_timeout_handler;
321 hc->tev.data = hc;
322 hc->tev.log = hc->log;
323 ngx_add_timer(&hc->tev, hc->conf->hc.timeout_ms);
324
325 BRIX_HC_METRIC_INC(hc_probes_total);
326
327 brix_hc_begin_connect(hc, sa, salen);
328}

330/*
331 * Launch one probe against host:port (the slot was already claimed by
332 * brix_srv_hc_claim()): allocate the probe and start the async resolution;
333 * brix_hc_resolved() carries on. A resolution that cannot even start counts
334 * as a real probe failure (the claim is released there).
335 */
336static void
337brix_hc_start(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *conf,
338 const char *host, uint16_t port)
339{
340 brix_hc_ctx_t *hc;
341
342 hc = brix_hc_ctx_create(cycle, conf, host, port);
343 if (hc == NULL) {
344 return;
345 }
346
347 hc->dns.name.data = (u_char *) hc->host;
348 hc->dns.name.len = ngx_strlen(hc->host);
349 hc->dns.port = port;
350 hc->dns.af = BRIX_AF_AUTO;
351 hc->dns.socktype = SOCK_STREAM;
352 hc->dns.policy = conf->common.dns.policy;
353 hc->dns.log = hc->log;
354 hc->dns.handler = brix_hc_resolved;
355 hc->dns.data = hc;
356
357 if (brix_dns_resolve(&hc->dns) != NGX_OK) {
358 ngx_log_error(NGX_LOG_WARN, hc->log, 0,
359 "brix: health check: cannot resolve %s:%d: %s",
360 host, (int) port,
361 hc->dns.error ? hc->dns.error : "unknown");
362 brix_hc_pre_connect_fail(host, port, conf, hc->pool);
363 }
364}

366/*
367 * Recurring scan tick. Re-arms itself first (so a probe-launch failure can't
368 * stop future scans), then tries to claim at most one due server from the
369 * shared registry and probe it. Claiming exactly one per tick is what spreads
370 * load and guarantees a single worker probes each server per interval.
371 */
372static void
373brix_hc_timer_handler(ngx_event_t *ev)
374{
375 brix_hc_mgr_t *mgr = ev->data;
376 ngx_stream_brix_srv_conf_t *conf = mgr->conf;
377 char host[256];
378 uint16_t port;
379 ngx_msec_t next_due = conf->hc.interval_ms;
380 ngx_msec_t delay;
381
382 /* Claim is atomic under the registry spinlock and self-rate-limits via
383 * hc_next_check, so concurrent workers never double-probe one server. */
384 if (brix_srv_hc_claim(host, sizeof(host), &port,
385 conf->hc.interval_ms, &next_due))
386 {
387 brix_hc_start(mgr->cycle, conf, host, port);
388 /* More servers may be due now — spread the remaining probes at the
389 * per-slot cadence rather than firing them all this tick. */
390 delay = mgr->scan_interval_ms;
391 } else {
392 /* Nothing due: sleep until the soonest server becomes due instead of
393 * waking at the sub-second floor to find nothing (the old idle drain).
394 * Floored by the spread cadence (burst guard) and capped by the probe
395 * interval (so a freshly-registered server is probed within one). */
396 delay = next_due;
397 if (delay < mgr->scan_interval_ms) {
398 delay = mgr->scan_interval_ms;
399 }
400 if (delay > conf->hc.interval_ms) {
401 delay = conf->hc.interval_ms;
402 }
403 }
404
405 /* Re-arm (bounded to [scan_interval, hc_interval]) — a probe-launch
406 * failure or empty registry can never stop future scans. Suppressed once
407 * the worker is exiting so the scan loop releases the draining worker. */
408 if (!ngx_exiting) {
409 ngx_add_timer(ev, delay);
410 }
411}

413void
414brix_hc_manager_start(ngx_cycle_t *cycle,
415 ngx_stream_brix_srv_conf_t *conf)
416{
417 brix_hc_mgr_t *mgr;
418 ngx_msec_t scan;
419

422 }
423

425 if (mgr == NULL) {

427 }
428 mgr->cycle = cycle;
429 mgr->conf = conf;
430
431 /* Spread probes across the interval: one claim attempt every
432 * interval/slots ms, floored at 100 ms. */
433 scan = conf->hc.interval_ms / BRIX_SRV_REGISTRY_SLOTS;
434 mgr->scan_interval_ms = scan < 100 ? 100 : scan;
435
436 mgr->timer.handler = brix_hc_timer_handler;
437 mgr->timer.data = mgr;
438 mgr->timer.log = cycle->log;
439
440 ngx_add_timer(&mgr->timer, 2000); /* let CMS connections settle first */
441
442 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
443 "brix: health check manager started "
444 "(interval=%Ms timeout=%Ms scan=%Ms)",
445 conf->hc.interval_ms, conf->hc.timeout_ms,
446 mgr->scan_interval_ms);
