22/* ============ shared online-buffer vtable ops (exec + lib adapters) ============
23 *
24 * WHAT: The whole brix_mss_adapter_t surface except destroy, implemented once
25 * against the frm_mss_head_t every real-HSM adapter context starts with.
26 * WHY: The exec and lib adapters were line-for-line clones that differed only
27 * in HOW an MSS verb runs (posix_spawn of the stage command vs a dlsym'd
28 * in-process call). That one difference is now the head's `invoke`
29 * callback; everything else — resolve <base>/.online/<key>, the local
30 * stat/access/open/unlink discipline, mkparents before a recall — is
31 * stated once here so the two transports cannot drift.
32 * HOW: Each op resolves the online path, does its local-buffer work, and
33 * calls the MSS only through head->invoke(verb, key, online).
34 */

36int
37frm_online_path(const char *base, const char *key, char *out, size_t cap)
38{
39 int n = snprintf(out, cap, "%s/.online/%s", base,
40 (key[0] == '/') ? key + 1 : key);
41
42 return (n > 0 && (size_t) n < cap) ? 0 : -1;
43}

45int
46frm_mss_residency(void *mss, const char *key, off_t *size_out, time_t *mtime_out)
47{
48 frm_mss_head_t *h = mss;
49 char online[PATH_MAX];
50 struct stat sb;
51
52 if (frm_online_path(h->base, key, online, sizeof(online)) == 0
53 && stat(online, &sb) == 0)
54 {
55 if (size_out) { *size_out = sb.st_size; }
56 if (mtime_out) { *mtime_out = sb.st_mtime; }
57 return BRIX_RESIDENCY_ONLINE;
58 }
59 /* Ask the MSS: 0 = on tape (offline), non-zero = absent. The size is
60 * unknown until recalled; the cache fill restats the online buffer. */
61 if (h->invoke(mss, "exists", key, "") == 0) {
62 if (size_out) { *size_out = 0; }
63 if (mtime_out) { *mtime_out = time(NULL); }
64 return BRIX_RESIDENCY_OFFLINE;
65 }
66 return BRIX_RESIDENCY_ABSENT;
67}

69/* The six path-resolving ops share one body: resolve <base>/.online/<key>,
70 * then do the op's local-buffer work + MSS invoke. A resolve failure is
71 * ENAMETOOLONG/-1 for every op except purge, which stays best-effort (it still
72 * offers the MSS-side drop and reports success, matching the old adapters). */
73typedef enum {
74 MSS_OP_RECALL_BEGIN,
75 MSS_OP_RECALL_POLL,
76 MSS_OP_MIGRATE,
77 MSS_OP_PURGE,
78 MSS_OP_OPEN,
79 MSS_OP_CREATE,
80} mss_op_t;
81
82/* One MSS verb, with a DETERMINISTIC errno on failure (2.0, 2026-09-09).
83 *
84 * The invoke contract is "0 or non-zero" -- a stage command that exits 1 sets
85 * no errno at all, and the runner it goes through may leave one of its own
86 * behind (the reparented agent's waitpid loses the race with nginx's SIGCHLD
87 * handler and leaves ECHILD; an internal open leaves ENOENT). Callers up the
88 * stack DO read errno: brix_sd_frm_seal's caller maps it through
89 * stage_seal_drop_reason(), where a stale ENOENT means "the marker was
90 * withdrawn" and DROPS a journaled archive record that should have been
91 * retried and finally dead-lettered. A verb that fails without a reason of its
92 * own therefore reports EIO: an MSS-side error, retryable, never a drop. */
93static int
94mss_invoke_verb(frm_mss_head_t *h, void *mss, const char *verb,
95 const char *key, const char *online)
96{
97 errno = 0;
98 if (h->invoke(mss, verb, key, online) == 0) {
99 return 0;
100 }
101 if (errno == 0) {
102 errno = EIO;
103 }
104 return -1;
105}

107static int
108mss_online_op(void *mss, const char *key, mss_op_t op, mode_t mode)
109{
110 frm_mss_head_t *h = mss;
111 char online[PATH_MAX];
112
113 if (frm_online_path(h->base, key, online, sizeof(online)) != 0) {
114 if (op == MSS_OP_PURGE) {
115 (void) h->invoke(mss, "purge", key, "");
116 return 0;
117 }
118 errno = ENAMETOOLONG;
119 return -1;
120 }
121
122 switch (op) {
123 case MSS_OP_RECALL_BEGIN:
124 if (access(online, F_OK) == 0) {
125 return 0; /* already online */
126 }
127 frm_mkparents(online); /* the MSS writes the online buffer */
128 return mss_invoke_verb(h, mss, "recall", key, online);
129 case MSS_OP_RECALL_POLL:
130 return (access(online, F_OK) == 0) ? 1 : 0;
131 case MSS_OP_MIGRATE:
132 return mss_invoke_verb(h, mss, "migrate", key, online);
133 case MSS_OP_PURGE:
134 (void) unlink(online);
135 (void) h->invoke(mss, "purge", key, ""); /* best-effort MSS-side drop */
136 return 0;
137 case MSS_OP_OPEN:
138 return open(online, O_RDONLY | O_CLOEXEC);
139 case MSS_OP_CREATE:
140 frm_mkparents(online);
141 return open(online, O_RDWR | O_CREAT | O_TRUNC | O_CLOEXEC,
142 mode ? mode : 0644);
143 }
144 return -1;
145}
