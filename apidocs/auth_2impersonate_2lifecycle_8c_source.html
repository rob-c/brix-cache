1/*
2 * lifecycle.c — nginx integration glue for per-request UNIX impersonation
3 * (phase 40). See lifecycle.h for the contract.
4 *
5 * This is the ONLY impersonation file that knows about nginx config/lifecycle
6 * types. It owns one process-global settings block (there is at most one broker
7 * per nginx instance), turns the `brix_idmap*` directives into that
8 * block, validates the chosen mode, spawns the privileged broker in the master
9 * (FRM double-fork, reparented to init), connects the worker client, and sets the
10 * broker's target principal per request. Everything is inert unless the mode is
11 * `map`; `off` and `single` add no process, socket, or capability.
12 */

67/* Duplicate a conf token into a NUL-terminated string on cf->pool. */
68static char *
69imp_dup(ngx_conf_t *cf, ngx_str_t *src, ngx_str_t *dst)
70{
71 dst->data = ngx_pnalloc(cf->pool, src->len + 1);
72 if (dst->data == NULL) {
73 return NGX_CONF_ERROR;
74 }
75 ngx_memcpy(dst->data, src->data, src->len);
76 dst->data[src->len] = '\0';
77 dst->len = src->len;
78 return NGX_CONF_OK;
79}

81char *
82brix_imp_conf_mode(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
83{
84 ngx_str_t *v = cf->args->elts;
85
86 (void) cmd;
87 (void) conf;
88 imp_settings.configured = 1;
89
90 if (ngx_strcmp(v[1].data, "off") == 0) {
91 imp_settings.mode = BRIX_IMP_OFF;
92 } else if (ngx_strcmp(v[1].data, "single") == 0) {
93 imp_settings.mode = BRIX_IMP_SINGLE;
94 } else if (ngx_strcmp(v[1].data, "map") == 0) {
95 imp_settings.mode = BRIX_IMP_MAP;
96 } else {
97 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
98 "invalid brix_idmap mode \"%V\" "
99 "(expected off|single|map)", &v[1]);
100 return NGX_CONF_ERROR;
101 }
102 return NGX_CONF_OK;
103}

105char *
106brix_imp_conf_str(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
107{
108 ngx_str_t *v = cf->args->elts;
109 ngx_str_t *dst;
110
111 (void) conf;
112 imp_settings.configured = 1;
113
114 switch (cmd->offset) {
115 case BRIX_IMP_F_SOCKET: dst = &imp_settings.socket; break;
116 case BRIX_IMP_F_EXPORT_ROOT: dst = &imp_settings.export_root; break;
117 case BRIX_IMP_F_GRIDMAP: dst = &imp_settings.gridmap; break;
118 case BRIX_IMP_F_DEFAULT_USER: dst = &imp_settings.default_user; break;
119 case BRIX_IMP_F_SINGLE_USER: dst = &imp_settings.single_user; break;
120 case BRIX_IMP_F_BROKER_USER: dst = &imp_settings.broker_user; break;
121 case BRIX_IMP_F_FORBIDDEN_USERS: dst = &imp_settings.forbidden_users; break;
122 case BRIX_IMP_F_FORBIDDEN_GROUPS: dst = &imp_settings.forbidden_groups; break;
123 default:
124 return "has an unknown target field";
125 }
126 return imp_dup(cf, &v[1], dst);
127}

154ngx_int_t
155brix_imp_validate(ngx_conf_t *cf, const char *derived_export_root)
156{
157 if (!imp_settings.configured || imp_settings.mode == BRIX_IMP_OFF) {
158 imp_settings.mode = imp_settings.configured ? imp_settings.mode
159 : BRIX_IMP_OFF;
160 return NGX_OK;
161 }
162
163 if (imp_settings.mode == BRIX_IMP_SINGLE) {
164 if (imp_settings.single_user.len == 0) {
165 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
166 "brix_idmap single requires "
167 "brix_idmap_user <name>");
168 return NGX_ERROR;
169 }
170 if (imp_settings.gridmap.len || imp_settings.default_user.len) {
171 BRIX_CONF_LOG(NGX_LOG_WARN, cf, 0,
172 "brix_idmap single ignores gridmap/default_user "
173 "(all identities squash to \"%V\")", &imp_settings.single_user);
174 }
175 return NGX_OK;
176 }
177
178 /* mode == map */
179 if (geteuid() != 0) {
180 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
181 "brix_idmap map requires the nginx master to run as root "
182 "(needed to spawn the privileged identity broker)");
183 return NGX_ERROR;
184 }
185 if (imp_settings.socket.len == 0) {
186 ngx_str_t def = ngx_string(IMP_DEFAULT_SOCKET);
187 if (imp_dup(cf, &def, &imp_settings.socket) != NGX_CONF_OK) {
188 return NGX_ERROR;
189 }
190 }
191 if (imp_settings.export_root.len == 0) {
192 if (derived_export_root == NULL || derived_export_root[0] == '\0') {
193 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
194 "brix_idmap map needs an export root: set "
195 "brix_idmap_export <path> (no data server with a "
196 "local root was found to derive it from)");
197 return NGX_ERROR;
198 }
199 {
200 ngx_str_t s;
201 s.data = (u_char *) derived_export_root;
202 s.len = ngx_strlen(derived_export_root);
203 if (imp_dup(cf, &s, &imp_settings.export_root) != NGX_CONF_OK) {
204 return NGX_ERROR;
205 }
206 }
207 }
208 return NGX_OK;
209}

211/* ---- brix_worker_user: the confined account the worker is forced down to ----
212 *
213 * Process-global (there is one worker uid per process; the strictest posture is
214 * shared across every brix server, stream + http). Empty => unset => default to
215 * "nobody" with a startup warning. Set by the `brix_worker_user` directive. */
216char brix_worker_user[64] = "";
217
218/* ---- Store the configured confined worker account ----
219 *
220 * WHAT: Accept a bounded nonempty account name or return a configuration error.
221 * WHY: Configuration-time identity selection is separate from privilege drops.
222 * HOW: 1. Validate length. 2. Copy and terminate the existing process setting.
223 */
224char *
225brix_conf_set_worker_user(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
226{
227 ngx_str_t *value = cf->args->elts;
228
229 (void) cmd;
230 (void) conf;
231
232 if (value[1].len == 0 || value[1].len >= sizeof(brix_worker_user)) {
233 return "invalid brix_worker_user account name";
234 }
235 /* Last non-empty setting wins; the account is resolved at worker init. */
236 ngx_memcpy(brix_worker_user, value[1].data, value[1].len);
237 brix_worker_user[value[1].len] = '\0';
238 return NGX_CONF_OK;
239}

241/* ---- brix_imp_worker_runtime_ids ----
242 *
243 * WHAT: Resolve the uid/gid the workers actually serve as (see lifecycle.h).
244 * WHY: Config-time provisioning of worker-writable directories (the default
245 * credential store, the default gateway stage spool) must chown to the
246 * POST-de-escalation identity: with the always-on drop, `user root` (or
247 * no `user` at all under a root master) yields workers running as
248 * brix_worker_user/nobody, and a root:0700 directory would EACCES them.
249 * HOW: A real unprivileged `user` account wins (the drop no-ops for it);
250 * otherwise resolve brix_worker_user (default "nobody") exactly as
251 * brix_imp_worker_deescalate will at worker init. */
252ngx_int_t
253brix_imp_worker_runtime_ids(ngx_uid_t conf_uid, ngx_gid_t conf_gid,
254 uid_t *uid_out, gid_t *gid_out)
255{
256 const char *acct;
257 struct passwd *pw;
258
259 if (conf_uid != (ngx_uid_t) NGX_CONF_UNSET_UINT && conf_uid != 0) {
260 *uid_out = (uid_t) conf_uid;
261 *gid_out = (gid_t) conf_gid;
262 return NGX_OK;
263 }
264
265 /* nginx fills an omitted `user` directive from its compile-time
266 * --user/--group defaults only in the core module's init_conf, after
267 * location merges have already asked us to provision stage/credential
268 * stores. At that earlier point ccf->user is still UNSET. Resolve the
269 * same compiled account here so distro builds such as Fedora's
270 * --user=nginx do not get a nobody-owned 0700 store for an nginx worker.
271 * An explicitly root-capable worker must still take the fail-closed
272 * brix_worker_user/nobody path below. */
273 if (conf_uid == (ngx_uid_t) NGX_CONF_UNSET_UINT) {
274 pw = getpwnam(NGX_USER);
275 if (pw != NULL && pw->pw_uid != 0) {
276 *uid_out = pw->pw_uid;
277 *gid_out = pw->pw_gid;
278 return NGX_OK;
279 }
280 }
281 acct = brix_worker_user[0] ? brix_worker_user : "nobody";
282 pw = getpwnam(acct);
283 if (pw == NULL || pw->pw_uid == 0) {
284 return NGX_ERROR;
285 }
286 *uid_out = pw->pw_uid;
287 *gid_out = pw->pw_gid;
288 return NGX_OK;
289}
