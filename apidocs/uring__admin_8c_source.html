6/* File: src/aio/uring_admin.c — io_uring runtime kill switch (Phase 44)
7 * WHAT: The no-rebuild, no-reload kill switch for the optional io_uring backend
8 * (§8.1/§14.3): a single cross-worker ngx_atomic_t in shared memory that,
9 * when set, makes every worker's selector skip the io_uring tier on its
10 * next op (in-flight CQEs still drain). Three ways to flip it — the admin
11 * API (api_admin.c), a watched panic-file (per-worker poll timer here), or
12 * the runtime probe never bringing a ring up in the first place.
13 *
14 * WHY: io_uring has been a major kernel attack surface; operators need to
15 * neutralise it fleet-wide in seconds during an incident without a rebuild
16 * or reload. The flag lives in SHM so one admin call / one dropped file is
17 * observed by every worker on its next op. Build-independent (no liburing
18 * types) so the HTTP dashboard module can drive it.
19 *
20 * HOW: A tiny SHM zone holds one ngx_atomic_t. The hot path reads it lock-free
21 * (brix_uring_disabled in uring.h). The panic-file poll mirrors file
22 * existence into the flag. Everything degrades to a clean no-op when the
23 * zone was never registered (stub build / io_uring off everywhere). */
24

42/*
43 * brix_uring_ks_zone_init — SHM zone init callback. Slab-allocates the single
44 * ngx_atomic_t flag and stashes it in sp->data (so a reload re-attaches to the
45 * same flag). Mirrors the minimal single-object SHM pattern used elsewhere in
46 * the module.
47 */
48static ngx_int_t
49brix_uring_ks_zone_init(ngx_shm_zone_t *zone, void *data)
50{
51 ngx_slab_pool_t *sp;
52 ngx_atomic_t *flag;
53
54 if (data) { /* reload: inherit the previous flag */
55 zone->data = data;
56 return NGX_OK;
57 }
58
59 sp = (ngx_slab_pool_t *) zone->shm.addr;
60
61 if (zone->shm.exists) { /* re-attach to an existing segment */
62 zone->data = sp->data;
63 return NGX_OK;
64 }
65
66 flag = ngx_slab_alloc(sp, sizeof(ngx_atomic_t));
67 if (flag == NULL) {
68 return NGX_ERROR;
69 }
70 *flag = 0; /* enabled by default */
71 sp->data = (void *) flag;
72 zone->data = (void *) flag;
73 return NGX_OK;
74}

136/*
137 * brix_uring_panic_mirror — mirror the panic-file's existence into the SHM
138 * disable flag. A NOTICE is logged only on a state change so the flip is
139 * visible without log spam. No-op when no panic path was configured.
140 */
141static void
142brix_uring_panic_mirror(ngx_log_t *log)
143{
144 struct stat st;
145 ngx_uint_t exists;
146 ngx_int_t prev;
147
148 if (brix_uring_panic_path[0] == '\0') {
149 return;
150 }
151 exists = (stat((char *) brix_uring_panic_path, &st) == 0) ? 1 : 0;
152
153 prev = brix_uring_killswitch_get();
154 if (prev != (ngx_int_t) exists) {
155 brix_uring_killswitch_set(exists);
156 ngx_log_error(NGX_LOG_NOTICE, log, 0,
157 "brix: io_uring %s via panic-file \"%s\"",
158 exists ? "DISABLED" : "re-enabled",
159 (char *) brix_uring_panic_path);
160 }
161}

163/*
164 * brix_uring_maint_handler — the per-worker maintenance timer: mirror the
165 * panic file (if configured) into the kill switch, then run the P44-A
166 * quiesce/re-enable tick (teardown once drained while disabled; re-create the
167 * ring when the switch clears), then re-arm.
168 */
169static void
170brix_uring_maint_handler(ngx_event_t *ev)
171{
172 brix_uring_panic_mirror(ev->log);
173 brix_uring_quiesce_tick();
174 ngx_add_timer(ev, BRIX_URING_MAINT_POLL_MS);

177ngx_int_t
178brix_uring_maint_arm(ngx_cycle_t *cycle)
179{
180 /* Shared dummy connection (src/core/config/process.c): nginx's --with-debug
181 * timer-expiry log reads ngx_event_ident(ev->data)->fd, so a connection-less
182 * timer must not leave ev->data NULL or it SIGSEGVs the worker. */
183 extern ngx_connection_t brix_maint_timer_conn;
184
185 if (brix_uring_maint_timer.timer_set) {
186 return NGX_OK; /* already armed (panicfile_arm ran) */
187 }
188
189 brix_uring_maint_timer.handler = brix_uring_maint_handler;
190 brix_uring_maint_timer.log = cycle->log;
191 brix_uring_maint_timer.data = &brix_maint_timer_conn;
192 brix_uring_maint_timer.cancelable = 1; /* don't hold up worker shutdown */
193
194 /* Fire once promptly so a panic file already present at start disables on
195 * boot (and a kill switch inherited across reload quiesces immediately). */
196 ngx_add_timer(&brix_uring_maint_timer, 1);
197 return NGX_OK;
198}
