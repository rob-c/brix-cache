27/* TTL-gated refresh + pin-drift audit: when the mount is pinned and a verified
28 * upstream manifest has moved past the pin, log ONE audit line per drift
29 * transition (re-armed if upstream returns to the pin), keep serving the pin. */
30static void brix_refresh(void) {
31 static int drift_logged = 0;
32 cvmfs_client_refresh(g_cl, mono_now());
33 char up[48];
34 if (cvmfs_client_pin_drift(g_cl, up, sizeof(up))) {
35 if (!drift_logged) {
36 char pin[48];
37 cvmfs_hash_to_hex(&g_cl->pin_root, 0, pin, sizeof(pin));
38 fprintf(stderr, "brixcvmfs: audit signal=pindrift repo=%s pinned=%s "
39 "upstream=%s serving=pinned\n", g_cl->config.name, pin, up);
40 drift_logged = 1;
41 }
42 } else {
43 drift_logged = 0;
44 }
45}

49int brixcvmfs_op_getattr(const char *path, struct stat *st, struct fuse_file_info *fi) {
50 (void) fi;
51 brix_refresh(); /* TTL-gated: no-op until due */
52 cvmfs_client_reap_tick(g_cl, mono_now()); /* quota-gated: no-op until due */
53 cvmfs_dirent_t e;
54 int rc = cvmfs_client_resolve(g_cl, cat_path(path), &e, mono_now());
55 if (rc < 0) return -EIO;
56 if (rc == 0) return -ENOENT;
57
58 memset(st, 0, sizeof(*st));
59 st->st_mode = e.mode;
60 st->st_size = (off_t) e.size;
61 st->st_mtime = (time_t) e.mtime;
62 st->st_nlink = (e.flags & CVMFS_FLAG_DIR) ? 2 : e.linkcount;
63 st->st_uid = e.uid ? e.uid : getuid();
64 st->st_gid = e.gid ? e.gid : getgid();
65 return 0;
66}

76int brixcvmfs_op_readdir(const char *path, void *buf, fuse_fill_dir_t filler,
77 off_t off, struct fuse_file_info *fi, enum fuse_readdir_flags fl) {
78 (void) off; (void) fi; (void) fl;
79 brix_refresh(); /* TTL-gated: no-op until due */
80 filler(buf, ".", NULL, 0, 0);
81 filler(buf, "..", NULL, 0, 0);
82 readdir_ctx_t r = { buf, filler };
83 int n = cvmfs_client_readdir(g_cl, cat_path(path), readdir_emit, &r, mono_now());
84 if (n >= 0) pf_enqueue(cat_path(path)); /* first listing = prefetch signal */
85 return n < 0 ? -EIO : 0;
86}

88int brixcvmfs_op_open(const char *path, struct fuse_file_info *fi) {
89 if ((fi->flags & O_ACCMODE) != O_RDONLY) return -EROFS; /* read-only fs */
90 cvmfs_dirent_t e;
91 int rc = cvmfs_client_resolve(g_cl, cat_path(path), &e, mono_now());
92 if (rc < 0) return -EIO;
93 if (rc == 0) return -ENOENT;
94 if (!(e.flags & CVMFS_FLAG_FILE)) return -EISDIR;
95 return 0;
96}

107int brixcvmfs_op_readlink(const char *path, char *buf, size_t size) {
108 cvmfs_dirent_t e;
109 int rc = cvmfs_client_resolve(g_cl, cat_path(path), &e, mono_now());
110 if (rc <= 0) return rc < 0 ? -EIO : -ENOENT;
111 if (!(e.flags & CVMFS_FLAG_LINK)) return -EINVAL;
112 snprintf(buf, size, "%s", e.symlink);
113 return 0;
114}

139/* access(2): the fs is read-only, so any W_OK is refused up front; X_OK is
140 * honored against the node's mode bits (a 0644 file has no execute bit). R_OK /
141 * F_OK on a world-readable RO tree always succeed. Mirrors official cvmfs. */
142int brixcvmfs_op_access(const char *path, int mask) {
143 cvmfs_dirent_t e;
144 int rc = cvmfs_client_resolve(g_cl, cat_path(path), &e, mono_now());
145 if (rc < 0) return -EIO;
146 if (rc == 0) return -ENOENT;
147 if (mask & W_OK) return -EROFS; /* read-only filesystem */
148 if ((mask & X_OK) && !(e.mode & (S_IXUSR | S_IXGRP | S_IXOTH))) return -EACCES;
149 return 0;
150}
