1/*
2 * sd_pblock_unittest_pack.c — phase-88 W2 packed small-blob arena. Legs:
3 * SUCCESS — a small staged commit comes to rest as ONE segment record
4 * (no per-object block file), reads back byte-identical
5 * through the driver (memfd serving), and unlinking every
6 * packed blob reaps the segment file itself.
7 * ERROR — a write-intent open materialises the blob back to the
8 * striped layout (row gone, block file back, content
9 * preserved, writable); an oversized commit never packs.
10 * SECURITY-NEG — a bit-flipped record fails the open with EIO — a packed
11 * blob is served crc-verified or not at all.
12 * INTERPLAY — pack=1&dedup=1: the second identical commit folds onto the
13 * first PACKED blob (pack-aware byte-verify) and the packed
14 * copy survives until its last reference goes.
15 */

35/* pk_staged_put — full staged write of `data` under `path` (the admission
36 * road: only staged commits pack). Returns 0 or -1. */
37static int
38pk_staged_put(brix_sd_instance_t *inst, const char *path, const char *data,
39 size_t len)
40{
41 brix_sd_staged_t *h;
42 int err = 0;
43
44 h = D->staged_open(inst, path, 0644, 0, &err);
45 if (h == NULL) {
46 return -1;
47 }
48 if (len > 0 && D->staged_write(h, data, len, 0) != (ssize_t) len) {
49 D->staged_abort(h);
50 return -1;
51 }
52 if (D->staged_commit(h, NULL) != NGX_OK) {
53 D->staged_abort(h);
54 return -1;
55 }
56 return 0;
57}

59/* pk_pack_row — the pack row for `path`'s blob: 1 found (the seg, off and len
60 * outparams set), 0 absent, -1 no object row. */
61static int
62pk_pack_row(const char *root, const char *path, long long *seg,
63 long long *off, long long *len)
64{
65 char db[PATH_MAX];
66 sqlite3 *h = NULL;
67 sqlite3_stmt *q = NULL;
68 char blob[PBLOCK_BLOB_ID_CAP];
69 int found = -1;
70
71 blob[0] = '\0';
72 snprintf(db, sizeof(db), "%s/catalog.db", root);
73 CHECK(sqlite3_open(db, &h) == SQLITE_OK, "pack db open");
74 if (sqlite3_prepare_v2(h,
75 "SELECT blob_id FROM objects WHERE path = ?1;", -1, &q, NULL)
76 == SQLITE_OK)
77 {
78 sqlite3_bind_text(q, 1, path, -1, SQLITE_STATIC);
79 if (sqlite3_step(q) == SQLITE_ROW) {
80 const unsigned char *b = sqlite3_column_text(q, 0);
81
82 snprintf(blob, sizeof(blob), "%s", b ? (const char *) b : "");
83 }
84 }
85 sqlite3_finalize(q);
86 q = NULL;
87 if (blob[0] != '\0') {
88 found = 0;
89 if (sqlite3_prepare_v2(h,
90 "SELECT seg, off, len FROM pack WHERE blob_id = ?1;", -1, &q,
91 NULL) == SQLITE_OK)
92 {
93 sqlite3_bind_text(q, 1, blob, -1, SQLITE_STATIC);
94 if (sqlite3_step(q) == SQLITE_ROW) {
95 found = 1;
96 if (seg != NULL) { *seg = sqlite3_column_int64(q, 0); }
97 if (off != NULL) { *off = sqlite3_column_int64(q, 1); }
98 if (len != NULL) { *len = sqlite3_column_int64(q, 2); }
99 }
100 }
101 sqlite3_finalize(q);
102 }
103 sqlite3_close(h);
104 return found;
105}

107/* pk_data_files — regular files under <root>/data (block files remaining). */
108static int
109pk_data_files_walk(const char *dir)
110{
111 DIR *d = opendir(dir);
112 struct dirent *ent;
113 int files = 0;
114
115 if (d == NULL) {
116 return 0;
117 }
118 while ((ent = readdir(d)) != NULL) {
119 char sub[PATH_MAX];
120 struct stat stx;
121
122 if (strcmp(ent->d_name, ".") == 0 || strcmp(ent->d_name, "..") == 0) {
123 continue;
124 }
125 if (snprintf(sub, sizeof(sub), "%s/%s", dir, ent->d_name)
126 >= (int) sizeof(sub))
127 {
128 continue;
129 }
130 if (lstat(sub, &stx) != 0) {
131 continue;
132 }
133 if (S_ISDIR(stx.st_mode)) {
134 files += pk_data_files_walk(sub);
135 } else if (S_ISREG(stx.st_mode)) {
136 files++;
137 }
138 }
139 closedir(d);
140 return files;
141}

152/* pk_read_back — open read-only + pread + close; bytes read or -1. */
153static ssize_t
154pk_read_back(brix_sd_instance_t *inst, const char *path, char *buf, size_t cap)
155{
156 int err = 0;
157 brix_sd_obj_t *o = D->open(inst, path, BRIX_SD_O_READ, 0, &err);
158 ssize_t n;
159
160 if (o == NULL) {
161 errno = err;
162 return -1;
163 }
164 n = D->pread(o, buf, cap, 0);
165 if (pb_close(o) != NGX_OK) {
166 return -1;
167 }
168 return n;
169}

171void
172test_pack_arena(void)
173{
174 char root[] = "/tmp/pb_pack.XXXXXX";
175 brix_sd_instance_t inst = {0};
176 brix_sd_pblock_conf_t conf = {0};
177 char buf[256], segp[PATH_MAX];
178 long long seg = 0, off = 0, len = 0;
179 const char *DATA = "packed-arena-payload-0123456789";
180 const size_t DLEN = strlen("packed-arena-payload-0123456789");
181
182 CHECK(mkdtemp(root) != NULL, "mkdtemp");
183 lab_write_sidecar(root, "pack=1&pack_max=1k");
184 conf.root = root;
185 conf.busy_timeout_ms = 2000;
186 conf.block_size = 4096;
187 inst.driver = D;
188 inst.caps = D->caps;
189 CHECK(D->init(&inst, &conf) == NGX_OK, "pack init");
190
191 /* SUCCESS: two small staged commits pack — no block files remain, both
192 * records land in segment 1, both read back byte-identical (memfd). */
193 CHECK(pk_staged_put(&inst, "/p1", DATA, DLEN) == 0, "staged put p1");
194 CHECK(pk_staged_put(&inst, "/p2", "second-object", 13) == 0,
195 "staged put p2");
196 CHECK(pk_pack_row(root, "/p1", &seg, &off, &len) == 1
197 && len == (long long) DLEN, "p1 packed (seg %lld len %lld)",
198 seg, len);
199 CHECK(pk_pack_row(root, "/p2", NULL, NULL, NULL) == 1, "p2 packed");
200 CHECK(pk_data_files(root) == 0,
201 "no block files remain (got %d)", pk_data_files(root));
202 snprintf(segp, sizeof(segp), "%s/pack/seg-%lld.dat", root, seg);
203 CHECK(access(segp, F_OK) == 0, "segment file exists");
204 CHECK(pk_read_back(&inst, "/p1", buf, sizeof(buf)) == (ssize_t) DLEN
205 && memcmp(buf, DATA, DLEN) == 0, "p1 reads back from the arena");
206 CHECK(pk_read_back(&inst, "/p2", buf, sizeof(buf)) == 13
207 && memcmp(buf, "second-object", 13) == 0, "p2 reads back");
208
209 /* ERROR (oversize): above pack_max stays striped. */
210 {
211 char big[2048];
212
213 memset(big, 'B', sizeof(big));
214 CHECK(pk_staged_put(&inst, "/big", big, sizeof(big)) == 0,
215 "staged put big");
216 CHECK(pk_pack_row(root, "/big", NULL, NULL, NULL) == 0,
217 "oversize object stayed striped");
218 CHECK(pk_data_files(root) == 1, "big's block file exists");
219 CHECK(pk_read_back(&inst, "/big", buf, 8) == 8
220 && memcmp(buf, "BBBBBBBB", 8) == 0, "big reads back");
221 }
222
223 /* ERROR (materialise): a write-intent open brings p1 back to the striped
224 * layout — row gone, block file back, content preserved and writable. */
225 {
226 int err = 0;
227 brix_sd_obj_t *o = D->open(&inst, "/p1",
228 BRIX_SD_O_READ | BRIX_SD_O_WRITE, 0,
229 &err);
230
231 CHECK(o != NULL, "write open of packed p1 (err %d)", err);
232 if (o != NULL) {
233 CHECK(pk_pack_row(root, "/p1", NULL, NULL, NULL) == 0,
234 "write open materialised p1 out of the arena");
235 CHECK(D->pread(o, buf, sizeof(buf), 0) == (ssize_t) DLEN
236 && memcmp(buf, DATA, DLEN) == 0,
237 "materialised content preserved");
238 CHECK(D->pwrite(o, "X", 1, 0) == 1, "materialised blob writable");
239 CHECK(pb_close(o) == NGX_OK, "close materialised p1");
240 }
241 CHECK(pk_read_back(&inst, "/p1", buf, sizeof(buf)) == (ssize_t) DLEN
242 && buf[0] == 'X', "overwrite persisted");
243 }
244
245 /* SUCCESS (reap): dropping every packed blob reaps the segment file. */
246 CHECK(D->unlink(&inst, "/p2", 0) == NGX_OK, "unlink p2");
247 CHECK(access(segp, F_OK) != 0,
248 "segment reaped once its last record died");
249
250 /* SECURITY-NEG: re-pack an object, flip one payload byte in the segment,
251 * and the read-open must fail EIO — never serve unverified bytes. */
252 CHECK(pk_staged_put(&inst, "/evil", DATA, DLEN) == 0, "staged put evil");
253 CHECK(pk_pack_row(root, "/evil", &seg, &off, &len) == 1, "evil packed");
254 snprintf(segp, sizeof(segp), "%s/pack/seg-%lld.dat", root, seg);
255 {
256 int fd = open(segp, O_RDWR);
257 /* record: 28-byte header + 32-char blob-id key, then the payload */
258 off_t flip = (off_t) off + 28 + 32 + 3;
259 char byte;
260
261 CHECK(fd >= 0, "open segment for corruption");
262 CHECK(pread(fd, &byte, 1, flip) == 1, "read corruption target");
263 byte ^= 0x40;
264 CHECK(pwrite(fd, &byte, 1, flip) == 1, "flip payload byte");
265 close(fd);
266 }
267 {
268 int err = 0;
269 brix_sd_obj_t *o = D->open(&inst, "/evil", BRIX_SD_O_READ, 0, &err);
270
271 CHECK(o == NULL && err == EIO,
272 "corrupt record refused with EIO (obj %p err %d)",
273 (void *) o, err);
274 if (o != NULL) {
275 pb_close(o);
276 }
277 }
278 D->cleanup(&inst);
279
280 /* INTERPLAY: pack + dedup — the second identical commit byte-verifies
281 * against the PACKED first copy and folds; the survivor stays packed and
282 * outlives the first name. */
283 {
284 char root2[] = "/tmp/pb_packdd.XXXXXX";
285 brix_sd_instance_t inst2 = {0};
286 brix_sd_pblock_conf_t conf2 = {0};
287
288 CHECK(mkdtemp(root2) != NULL, "mkdtemp dd");
289 lab_write_sidecar(root2, "pack=1&dedup=1");
290 conf2.root = root2;
291 conf2.busy_timeout_ms = 2000;
292 conf2.block_size = 4096;
293 inst2.driver = D;
294 inst2.caps = D->caps;
295 CHECK(D->init(&inst2, &conf2) == NGX_OK, "dd init");
296
297 CHECK(pk_staged_put(&inst2, "/a", DATA, DLEN) == 0, "dd put a");
298 CHECK(pk_staged_put(&inst2, "/b", DATA, DLEN) == 0, "dd put b");
299 CHECK(pk_pack_row(root2, "/a", NULL, NULL, NULL) == 1, "a packed");
300 CHECK(pk_pack_row(root2, "/b", NULL, NULL, NULL) == 1,
301 "b folded onto the packed blob");
302 CHECK(pk_data_files(root2) == 0, "dd left no block files");
303 CHECK(D->unlink(&inst2, "/a", 0) == NGX_OK, "dd unlink a");
304 CHECK(pk_read_back(&inst2, "/b", buf, sizeof(buf)) == (ssize_t) DLEN
305 && memcmp(buf, DATA, DLEN) == 0,
306 "survivor b still reads from the arena");
307 D->cleanup(&inst2);
308 }
