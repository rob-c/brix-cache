65/*
66 * Parse "<host> <port>" and, if present, the remaining text. Returns 0 when
67 * the host is missing/oversized or the port is absent, non-numeric, or outside
68 * 1-BRIX_CMS_MAX_PORT — a rejected target is never passed to a registry helper,
69 * so a typo cannot drain a node the operator did not name.
70 */
71static int
72cms_admin_parse_target(u_char *args, size_t alen, cms_admin_target_t *t)
73{
74 u_char *sp = ngx_strlchr(args, args + alen, ' ');
75 u_char *pend;
76 size_t hlen;
77 ngx_int_t port;
78
79 if (sp == NULL) {
80 return 0;
81 }
82 hlen = (size_t) (sp - args);
83 if (hlen == 0 || hlen >= sizeof(t->host)) {
84 return 0;
85 }
86 ngx_memcpy(t->host, args, hlen);
87 t->host[hlen] = '\0';
88
89 pend = ngx_strlchr(sp + 1, args + alen, ' ');
90 if (pend == NULL) {
91 pend = args + alen;
92 t->tail = NULL;
93 t->tail_len = 0;
94 } else {
95 t->tail = pend + 1;
96 t->tail_len = (size_t) (args + alen - (pend + 1));
97 }
98 port = ngx_atoi(sp + 1, (size_t) (pend - (sp + 1)));
99 if (port == NGX_ERROR || port <= 0 || port > BRIX_CMS_MAX_PORT) {
100 return 0;
101 }
102 t->port = (uint16_t) port;
103 return 1;
104}

121/* "nodes" — list the SHM registry. Node-wide: any worker's socket sees all. */
122static void
123cms_admin_nodes(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
124{
125 brix_srv_snapshot_entry_t *snap;
126 ngx_msec_t now = ngx_current_msec;
127 u_char *body = rep->buf + 32; /* header slid in front afterwards */
128 u_char *last = rep->buf + rep->cap;
129 u_char *pos, head[32], *he;
130 ngx_uint_t n, i, listed = 0;
131 size_t hlen;
132
133 (void) ud; (void) args; (void) alen;
134
135 snap = ngx_alloc(sizeof(*snap) * BRIX_SRV_REGISTRY_SLOTS, ngx_cycle->log);
136 if (snap == NULL) {
137 brix_admin_reply_set(rep, "err out-of-memory\n");
138 return;
139 }
140 n = brix_srv_snapshot(snap, BRIX_SRV_REGISTRY_SLOTS, now);
141
142 pos = body;
143 for (i = 0; i < n; i++) {
144 /* Stop cleanly rather than truncating a line: a half-written entry
145 * would parse as a DIFFERENT node to whatever reads this. */
146 if ((size_t) (last - pos) < BRIX_CMS_STATE_SAFE_BUF) {
147 break;
148 }
149 pos = ngx_snprintf(pos, (size_t) (last - pos),
150 "%s:%d role=%s free_mb=%uD util_pct=%uD state=%s\n",
151 snap[i].host, (int) snap[i].port,
152 snap[i].role[0] ? snap[i].role : "-",
153 snap[i].free_mb, snap[i].util_pct,
154 cms_admin_state(&snap[i], now));
155 listed++;
156 }
157 ngx_free(snap);
158
159 he = ngx_snprintf(head, sizeof(head), "ok %ui\n", listed);
160 hlen = (size_t) (he - head);
161 ngx_memcpy(rep->buf + 32 - hlen, head, hlen);
162 rep->out = rep->buf + 32 - hlen;
163 rep->len = hlen + (size_t) (pos - body);
164}

166/* "drain <host> <port> [secs]" — blacklist so the router stops selecting it. */
167static void
168cms_admin_drain(void *ud, u_char *args, size_t alen, brix_admin_reply_t *rep)
169{
170 cms_admin_target_t t;
171 ngx_int_t secs = CMS_ADMIN_DRAIN_DEFAULT_S;
172
173 (void) ud;
174
175 if (!cms_admin_parse_target(args, alen, &t)) {
176 brix_admin_reply_set(rep, "err bad-target\n");
177 return;
178 }
179 if (t.tail != NULL) {
180 secs = ngx_atoi(t.tail, t.tail_len);
181 if (secs == NGX_ERROR || secs <= 0) {
182 cms_admin_audit("cluster/drain", t.host, t.port, "bad_request");
183 brix_admin_reply_set(rep, "err bad-seconds\n");
184 return;
185 }
186 }
187 brix_srv_blacklist(t.host, t.port, (ngx_msec_t) secs * BRIX_CMS_SEC_TO_MS_MULTIPLIER);
188 cms_admin_audit("cluster/drain", t.host, t.port, "drained");
189 brix_admin_reply_set(rep, "ok\n");
190}

192/*
193 * The three plain target verbs — undrain, forget, reset — are one shape:
194 * parse "<host> <port>", call one registry helper, audit the outcome, answer.
195 * Written once so a change to the refusal wording or the audit fields cannot
196 * land on two of the three.
197 */
198typedef int (*cms_admin_act_pt)(const char *host, uint16_t port);
199
200typedef struct {
201 const char *action; /* audit action field */
202 cms_admin_act_pt act; /* 0 = no such node */
203 const char *ok_result; /* audit result field on success */
204} cms_admin_op_t;
205
206static void
207cms_admin_target_verb(u_char *args, size_t alen, brix_admin_reply_t *rep,
208 const cms_admin_op_t *op)
209{
210 cms_admin_target_t t;
211
212 if (!cms_admin_parse_target(args, alen, &t)) {
213 brix_admin_reply_set(rep, "err bad-target\n");
214 return;
215 }
216 if (!op->act(t.host, t.port)) {
217 cms_admin_audit(op->action, t.host, t.port, "not_found");
218 brix_admin_reply_set(rep, "err not-found\n");
219 return;
220 }
221 cms_admin_audit(op->action, t.host, t.port, op->ok_result);
222 brix_admin_reply_set(rep, "ok\n");
223}
