16 *
17 * - length/extract/forbidden-component pre-checks

25static ngx_flag_t
26brix_prepare_has_forbidden_component(const char *path)
27{
28 const char *p = path;
29
30 while (*p != '\0') {
31 const char *seg;
32 size_t len;
33
34 while (*p == '/') {
35 p++;
36 }
37
38 seg = p;
39 while (*p != '\0' && *p != '/') {
40 p++;
41 }
42
43 len = (size_t) (p - seg);
44 if ((len == 1 && seg[0] == '.')
45 || (len == 2 && seg[0] == '.' && seg[1] == '.'))
46 {
47 return 1;
48 }
49 }
50
51 return 0;
52}
53/*
54 * WHY: kXR_prepare rejects paths containing dot (.) or double-dot (..)
55 * components to prevent directory traversal into parent exports. Used as
56 * a fast pre-check before full path resolution — avoids expensive
57 * resolve_path() calls on obviously invalid paths.
58 *
59 * HOW: Scans path character-by-character, skipping leading '/' separators;
60 * extracts each segment between slashes via seg→p pointer arithmetic.
61 * For each segment checks len==1 && seg[0]=='.' or len==2 &&
62 * seg[0]=='.' && seg[1]=='.' — if match returns 1 (forbidden).
63 * Returns 0 if no forbidden components found after full scan. Static
64 * helper used exclusively by brix_prepare_check_path().
65 */

90/*
91 * WHAT: run the three prepare authorization tiers (authdb VO/ACL, VO
92 * identity ACL, token scope) on one resolved path.
93 *
94 * WHY: every prepare outcome — present, absent-with-noerrs, absent-without —
95 * must be reached through the SAME gate on the SAME path, because
96 * authorization is a property of the identity + logical path and not of
97 * on-disk existence. Its single call site sits ahead of the stat in
98 * check_path for that reason; see the SECURITY note there.
99 *
100 * HOW: each tier that denies sends its specific error via check_fail and its
101 * rc (NGX_DONE/error) is returned; NGX_OK only when all three pass.
102 */
103static ngx_int_t
104prepare_path_authz(brix_ctx_t *ctx, ngx_connection_t *c,
105 const prepare_scan_t *sc, const char *pathbuf, const char *full_path)
106{
107 ngx_stream_brix_srv_conf_t *conf = sc->conf;
108 ngx_flag_t recalls = (sc->do_stage || sc->do_evict);
109 uint32_t level = BRIX_AUTH_READ;
110 brix_acc_op_t aop = BRIX_AOP_READ;
111
112 /* 2.0 F20: a bare kXR_prepare only browses the namespace, so READ is the
113 * whole requirement. The kXR_stage / kXR_evict arms actually drive a tape
114 * or nearline recall (or drop an online copy) on the operator's storage, so
115 * they additionally demand the `x` privilege — the native authdb's new
116 * BRIX_AUTH_STAGE bit. Both bits are required together, never either-or.
117 *
118 * The XrdAcc op has to move with the level, not independently of it: the
119 * two engines answer the SAME question and brix_authz_check picks between
120 * them purely on `brix_authdb_engine`. Pinning the op at AOP_STAGE while
121 * the native level said READ made a bare prepare demand `x` under xrdacc
122 * and `r` under the native parser, so one authdb granting `rl` admitted the
123 * browse and its faithful translation denied it — F20's rule implemented on
124 * one engine only. */
125 if (recalls) {
126 level |= BRIX_AUTH_STAGE;
127 aop = BRIX_AOP_STAGE;
128 }
129
130 if (brix_authz_check(ctx, c, conf, pathbuf, full_path, "PREPARE",
131 level, aop) != NGX_OK) {
132 return brix_prepare_check_fail(ctx, c, full_path, kXR_NotAuthorized,
133 "not authorized");
134 }
135 if (brix_check_vo_acl_identity(c->log, full_path, conf->common.vo_rules,
136 ctx->identity) != NGX_OK) {
137 return brix_prepare_check_fail(ctx, c, full_path, kXR_NotAuthorized,
138 "VO not authorized");
139 }
140 if (brix_check_token_scope(ctx, pathbuf, 0) != NGX_OK) {
141 return brix_prepare_check_fail(ctx, c, pathbuf, kXR_NotAuthorized,
142 "token scope denied");
143 }
144 return NGX_OK;
145}

147/*
148 * Map a confined-stat failure (non-noerrs, or non-ENOENT errno) to the wire
149 * error:
150 *
151 * - ENOENT/ENOTDIR → kXR_NotFound
152 * - EACCES/EPERM → kXR_NotAuthorized
153 * - any other errno → kXR_IOError
154 *
155 * Returns the check_fail rc (NGX_DONE/error).
156 */
157static ngx_int_t
158prepare_stat_error(brix_ctx_t *ctx, ngx_connection_t *c,
159 const char *pathbuf, const char *full_path)
160{
161 if (errno == ENOENT || errno == ENOTDIR) {
162 return brix_prepare_check_fail(ctx, c, pathbuf, kXR_NotFound,
163 "file not found");
164 }
165 if (errno == EACCES || errno == EPERM) {
166 return brix_prepare_check_fail(ctx, c, full_path, kXR_NotAuthorized,
167 "not authorized");
168 }
169 return brix_prepare_check_fail(ctx, c, full_path, kXR_IOError,
170 "prepare stat failed");
171}

173ngx_int_t
174brix_prepare_check_path(brix_ctx_t *ctx, ngx_connection_t *c,
175 prepare_scan_t *sc, const u_char *line, size_t line_len,
176 char *out_resolved) /* PATH_MAX buffer filled with absolute path on
177 auth-pass paths; '\0' if path cannot be resolved.
178 Pass NULL when staging collection is not needed. */
179{
180 ngx_stream_brix_srv_conf_t *conf = sc->conf;
181 ngx_flag_t noerrs = (sc->options & kXR_noerrs) != 0;
182 char pathbuf[BRIX_MAX_PATH + 1];
183 char full_path[PATH_MAX];
184 struct stat st;
185
186 if (line_len > BRIX_MAX_PATH) {
187 return brix_prepare_check_fail(ctx, c, "-", kXR_ArgTooLong,
188 "prepare path too long");
189 }
190
191 if (!brix_extract_path(c->log, line, line_len, pathbuf,
192 sizeof(pathbuf), 1)) {
193 return brix_prepare_check_fail(ctx, c, "-", kXR_ArgInvalid,
194 "invalid prepare path");
195 }
196
197 if (brix_prepare_has_forbidden_component(pathbuf)) {
198 return brix_prepare_check_fail(ctx, c, pathbuf, kXR_ArgInvalid,
199 "invalid prepare path");
200 }
201
202 /* phase74-fp: pathbuf is the request path, full_path the output buf. */
203 brix_beneath_full_path(conf->common.root_canon, pathbuf, /* NOLINT(readability-suspicious-call-argument) */
204 full_path, sizeof(full_path));
205
206 /*
207 * SECURITY: authorize BEFORE the existence probe, never after.
208 *
209 * Authorization is a property of the IDENTITY + LOGICAL PATH, not of
210 * on-disk existence — so the answer for a principal who may not read this
211 * namespace path is kXR_NotAuthorized whether or not the object is there.
212 * Checking existence first leaks the namespace: the caller distinguishes
213 * "absent" (kXR_NotFound) from "present but forbidden" (kXR_NotAuthorized)
214 * purely from the error code, which is a directory-enumeration oracle over
215 * paths they hold no privilege on — the same oracle open_request.c closed on
216 * the read-open path. It also matters operationally: a prepare/stage drives
217 * a tape or nearline recall on the operator's storage, and an unauthorized
218 * principal must not be able to schedule that work, nor to have the recalled
219 * bytes land in a shared cache they can later be served from.
220 *
221 * The noerrs branch already ran the three tiers for exactly this reason; the
222 * plain branch reached kXR_NotFound with no tier run at all, so the two
223 * branches disagreed about what a prepare reveals. One call here, before the
224 * stat, is what makes them agree by construction.
225 */
226 {
227 ngx_int_t arc = prepare_path_authz(ctx, c, sc, pathbuf, full_path);
228 if (arc != NGX_OK) {
229 return arc;
230 }
231 }
232
233 /*
234 * CONTRACT: the same "file is absent" condition (ENOENT/ENOTDIR from the
235 * confined stat) has two outcomes selected by the kXR_noerrs flag:
236 * - noerrs set → not an error. The path is counted in sc->missing and the
237 * request still succeeds, so a client can prepare/stage files that do
238 * not exist on disk yet (tape nearline recall, not-yet-cached objects).
239 * - noerrs clear → kXR_NotFound, failing the request on the first miss.
240 * EACCES/EPERM and any other errno always fail regardless of noerrs.
241 */
242 if (brix_stat_beneath(conf->rootfd, pathbuf, &st) != 0) {
243 if ((errno == ENOENT || errno == ENOTDIR) && noerrs) {
244 sc->missing++;
245 /* For staging: supply absolute path even if file doesn't exist yet
246 * (tape nearline / not-yet-created). */
247 if (out_resolved != NULL) {
248 ngx_cpystrn((u_char *) out_resolved, (u_char *) full_path,
249 PATH_MAX);
250 }
251 return NGX_OK;
252 }
253 return prepare_stat_error(ctx, c, pathbuf, full_path);
254 }
255
256 /* Copy the absolute export path for the staging command; only authorized
257 * paths reach here so the staging hook can trust the value. */
258 if (out_resolved != NULL) {
259 ngx_cpystrn((u_char *) out_resolved, (u_char *) full_path, PATH_MAX);
260 }
261
262 if (S_ISDIR(st.st_mode)) {
263 if (noerrs) {
264 sc->missing++;
265 return NGX_OK;
266 }
267 return brix_prepare_check_fail(ctx, c, pathbuf, kXR_isDirectory,
268 "prepare target is a directory");
269 }
270
271 return NGX_OK;
272}
