6static int
7reserve_probe(xrd_conform_execution *execution, brix_status *status)
8{
9 char reason[64];
10 unsigned index;
11
12 for (index = 0; index < 20; index++) {
13 size_t bytes = index == 0 ? SCOPED_TPC_BYTES * 3U : 0;
14
15 if (!xrd_conform_reserve(execution, bytes, reason, sizeof(reason))) {
16 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
17 return -1;
18 }
19 }
20 return 0;
21}

23static int
24read_source(brix_conn *source, const char *path,
25 uint8_t payload[SCOPED_TPC_BYTES], brix_status *status)
26{
27 brix_file file;
28 ssize_t length;
29 brix_status close_status;
30
31 if (brix_file_open_read(source, path, &file, status) != 0) return -1;
32 length = brix_file_read(source, &file, 0, payload, SCOPED_TPC_BYTES,
33 status);
34 brix_status_clear(&close_status);
35 if (brix_file_close(source, &file, &close_status) != 0 && length >= 0) {
36 *status = close_status;
37 return -1;
38 }
39 return length == SCOPED_TPC_BYTES ? 0 : -1;
40}

65static int
66prepare_scoped_pair(xrd_conform_execution *execution, scoped_pair *pair,
67 brix_status *status)
68{
69 if (execution->run->also_count == 0
70 || brix_endpoint_parse(execution->run->config.target,
71 &pair->source_url, status) != 0
72 || brix_endpoint_parse(execution->run->also[0],
73 &pair->destination_url, status) != 0) return -1;
74 if (brix_conform_scratch_path(execution->run->config.scratch,
75 "brix-conform-tpc-source", pair->allowed_source.path,
76 sizeof(pair->allowed_source.path)) != 0
77 || brix_conform_scratch_path(execution->run->config.scratch,
78 "brix-conform-tpc-outside-scope", pair->denied_source.path,
79 sizeof(pair->denied_source.path)) != 0
80 || brix_conform_scratch_path(execution->run->config.scratch,
81 "brix-conform-tpc-destination", pair->allowed_destination.path,
82 sizeof(pair->allowed_destination.path)) != 0
83 || brix_conform_scratch_path(execution->run->config.scratch,
84 "brix-conform-tpc-denied-destination",
85 pair->denied_destination.path,
86 sizeof(pair->denied_destination.path)) != 0) return -1;
87 return xrd_conform_tpc_root_url(&pair->source_url,
88 pair->allowed_source.path, pair->allowed_source_url,
89 sizeof(pair->allowed_source_url)) == 0
90 && xrd_conform_tpc_root_url(&pair->source_url,
91 pair->denied_source.path, pair->denied_source_url,
92 sizeof(pair->denied_source_url)) == 0
93 && xrd_conform_tpc_root_url(&pair->destination_url,
94 pair->allowed_destination.path, pair->allowed_destination_url,
95 sizeof(pair->allowed_destination_url)) == 0
96 && xrd_conform_tpc_root_url(&pair->destination_url,
97 pair->denied_destination.path, pair->denied_destination_url,
98 sizeof(pair->denied_destination_url)) == 0
99 && reserve_probe(execution, status) == 0 ? 0 : -1;
100}

102static void
103run_scoped_copies(scoped_pair *pair, brix_conn *source,
104 brix_conn *destination, brix_opts *opts, uint8_t *payload,
105 int *adopted, int *allowed, int *denied, brix_status *status)
106{
107 brix_copy_opts copy = { 0 };
108
109 *adopted = read_source(source, pair->allowed_source.path, payload,
110 status) == 0;
111 remove_destination(destination, &pair->allowed_destination);
112 remove_destination(destination, &pair->denied_destination);
113 copy.tpc_mode = XRDC_TPC_DELEGATE;
114 copy.tpc_token_mode = "passthrough";
115 copy.force = 1;
116 copy.posc = 1;
117 copy.silent = 1;
118 if (*adopted)
119 *allowed = brix_copy(pair->allowed_source_url,
120 pair->allowed_destination_url, &copy, opts, status) == 0;
121 brix_status_clear(status);
122 *denied = brix_copy(pair->denied_source_url, pair->denied_destination_url,
123 &copy, opts, status) != 0 && status->kxr == kXR_NotAuthorized;
124}

126static void
127grade_scoped_copy(xrd_conform_execution *execution, int connected_source,
128 int connected_destination, int adopted, int allowed, int denied,
129 const brix_status *status)
130{
131 int passed = adopted && allowed && denied;

133 xrd_conform_result_set(execution, "TPC-005",
134 passed ? BRIX_CONFORM_VERDICT_CONFORMANT
135 : adopted ? BRIX_CONFORM_VERDICT_DIVERGENT
136 : BRIX_CONFORM_VERDICT_INCONCLUSIVE,
137 passed ? "delegated-credential-confined-to-transfer-scope"
138 : !connected_source || !connected_destination
139 ? "scoped-tpc-endpoint-unavailable"
140 : !adopted ? "fixture-required:scoped-source-objects"
141 : !allowed ? "in-scope-delegated-tpc-failed"
142 : "delegated-credential-escaped-transfer-scope",
143 "in-scope TPC succeeds; sibling returns kXR_NotAuthorized",
144 status->msg);

159/* WHAT: Prove a delegated bearer cannot read outside its transfer scope.
160 * WHY: A successful in-scope TPC says nothing about confused-deputy expansion.
161 * HOW: Read and copy an externally prepared scoped object with explicit
162 * delegation, then reuse the same credential for an out-of-scope source
163 * and require the exact authorization refusal. TPC-001/003/004 separately
164 * own byte-exact and failed-destination cleanup assertions. */
165void
166xrd_conform_probe_tpc_scoped(xrd_conform_execution *execution)
167{
168 scoped_pair pair;
169 brix_conn source;
170 brix_conn destination;
171 brix_opts opts = execution->run->opts;
172 brix_status status;
173 uint8_t payload[SCOPED_TPC_BYTES];
174 int connected_source = 0;
175 int connected_destination = 0;
176 int adopted = 0;
177 int allowed = 0;
178 int denied = 0;
179
180 brix_status_clear(&status);
181 if (prepare_scoped_pair(execution, &pair, &status) != 0) {
182 xrd_conform_result_set(execution, "TPC-005",
183 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
184 "fixture-required:scoped-delegated-credential", NULL, status.msg);
185 return;
186 }
187 opts.capture = NULL;
188 if (brix_connect(&source, &pair.source_url, &opts, &status) == 0)
189 connected_source = 1;
190 if (connected_source
191 && brix_connect(&destination, &pair.destination_url, &opts,
192 &status) == 0)
193 connected_destination = 1;
194 if (connected_destination) {
195 run_scoped_copies(&pair, &source, &destination, &opts, payload,
196 &adopted, &allowed, &denied, &status);
197 }
198 grade_scoped_copy(execution, connected_source, connected_destination,
199 adopted, allowed, denied, &status);
200 cleanup_scoped_pair(&pair, &source, &destination, connected_source,
201 connected_destination);
202}
