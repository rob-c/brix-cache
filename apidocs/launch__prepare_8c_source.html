47/*
48 * A queued response is not a return value. brix_send_error() returns NGX_OK once
49 * the kXR_error is on the wire, so a function that ends `return
50 * brix_send_error(...)` reports SUCCESS to whoever called it. At a protocol entry
51 * point that is exactly right: NGX_OK there means "request handled" and nothing
52 * runs afterwards. One call deep it is a gate bypass - the caller reads NGX_OK as
53 * "the check passed, carry on" and carries on past a refusal it has already
54 * answered, allocating the handle and queueing a second, contradictory response.
55 * The gates below therefore return TPC_ANSWERED, "refused, and the client has
56 * been told", which only brix_tpc_prepare_pull - the entry point, where the
57 * distinction stops mattering - folds back into the wire contract's NGX_OK.
58 */
59/* Log, count and answer one refusal. Never reports success; a failed write is
60 * still NGX_ERROR, so connection teardown is unchanged. */
61ngx_int_t
62brix_tpc_refuse(brix_ctx_t *ctx, ngx_connection_t *c, const char *dst_path,
63 uint16_t code, const char *msg)
64{
65 brix_log_access(ctx, c, "OPEN", dst_path, "tpc-pull", 0, code, msg, 0);
66 BRIX_OP_ERR(ctx, BRIX_OP_OPEN_WR);
67
68 return brix_send_error(ctx, c, code, msg) == NGX_OK
69 ? TPC_ANSWERED : NGX_ERROR;
70}

124/* WHAT: Construct origin ID string from ctx->login.user+ngx_pid+getnameinfo host via snprintf("%s.%u@host") — falls back to "xrd" for empty user, ngx_pid for zero pid, addr_text.len then "unknown" for unresolved host. Caller: brix_tpc_prepare_pull (origin ID storage step). */

126static void
127tpc_build_origin_id(brix_ctx_t *ctx, ngx_connection_t *c, char *dst,
128 size_t dst_size)
129{
130 char host[NI_MAXHOST];
131 const char *user;
132 uint32_t pid;
133
134 user = ctx->login.user[0] != '\0' ? ctx->login.user : "xrd";
135 pid = ctx->login.pid != 0 ? ctx->login.pid : (uint32_t) ngx_pid;
136
137 host[0] = '\0';
138
139 /* Present the client host EXACTLY as the XRootD TPC source stores it in the
140 * rendezvous grant (XrdOfsTPC::genOrg -> XrdNetAddr::Name): the source pairs
141 * the destination's tpc.org against the grant with a raw strcmp
142 * (XrdOfsTPCInfo::Match), so any textual divergence leaves the grant
143 * unredeemed until it TTL-expires — the destination's pull-open then hangs on
144 * kXR_waitresp and finally fails "[3012] TPC kXR_open recv failed". Every
145 * XRootD server is IPv6 dual-stack, so it sees an IPv4 client as the
146 * IPv4-MAPPED address ::ffff:a.b.c.d and reverse-resolves THAT (which fails
147 * for non-DNS hosts like loopback -> numeric bracketed literal). brix's TPC
148 * listener is IPv4-only, so a naive getnameinfo() on the bare IPv4 resolves
149 * 127.0.0.1 -> "localhost" while the grant holds "[::ffff:127.0.0.1]". Match
150 * the source: map an IPv4 peer into ::ffff:a.b.c.d, reverse-resolve the mapped
151 * form, and fall back to its bracketed numeric literal. */
152 if (c->sockaddr != NULL) {
153 struct sockaddr_in6 mapped;
154 struct sockaddr *sa = c->sockaddr;
155 socklen_t salen = c->socklen;
156
157 if (c->sockaddr->sa_family == AF_INET) {
158 const struct sockaddr_in *s4 =
159 (const struct sockaddr_in *) c->sockaddr;
160
161 ngx_memzero(&mapped, sizeof(mapped));
162 mapped.sin6_family = AF_INET6;
163 mapped.sin6_port = s4->sin_port;
164 mapped.sin6_addr.s6_addr[10] = 0xff;
165 mapped.sin6_addr.s6_addr[11] = 0xff;
166 ngx_memcpy(&mapped.sin6_addr.s6_addr[12], &s4->sin_addr, 4);
167 sa = (struct sockaddr *) &mapped;
168 salen = (socklen_t) sizeof(mapped);
169 }
170
171 if (getnameinfo(sa, salen, host, sizeof(host), NULL, 0,
172 NI_NAMEREQD) != 0) {
173 host[0] = '\0';
174 }
175
176 if (host[0] == '\0' && sa->sa_family == AF_INET6) {
177 char numeric[INET6_ADDRSTRLEN];
178 const struct sockaddr_in6 *s6 = (const struct sockaddr_in6 *) sa;
179
180 if (inet_ntop(AF_INET6, &s6->sin6_addr, numeric,
181 sizeof(numeric)) != NULL) {
182 (void) snprintf(host, sizeof(host), "[%s]", numeric);
183 }
184 }
185 }
186
187 if (host[0] == '\0' && c->addr_text.len > 0) {
188 size_t host_len = c->addr_text.len;
189
190 if (host_len >= sizeof(host)) {
191 host_len = sizeof(host) - 1;
192 }
193 ngx_memcpy(host, c->addr_text.data, host_len);
194 host[host_len] = '\0';
195 }
196
197 if (host[0] == '\0') {
198 ngx_cpystrn((u_char *) host, (u_char *) "unknown", sizeof(host));
199 }
200
201 {
202 int prefix_len;
203
204 prefix_len = snprintf(dst, dst_size, "%s.%u@", user, (unsigned) pid);
205 if (prefix_len < 0 || (size_t) prefix_len >= dst_size) {
206 dst[0] = '\0';
207 return;
208 }
209
210 ngx_cpystrn((u_char *) dst + prefix_len, (u_char *) host,
211 dst_size - (size_t) prefix_len);
212 }
213}

340/* WHAT: Validate the pull preconditions before any state is allocated — a
341 * configured thread pool, a non-empty TPC source host+path, the operator's
342 * source-name allowlist, and an SSRF-safe source address per the
343 * loopback/private allow flags. NGX_OK when the request may proceed;
344 * TPC_ANSWERED once a refusal has been answered; NGX_ERROR if it could not.
345 * WHY: this is the security-load-bearing front door, SSRF gate included;
346 * isolating it keeps the orchestrator flat and the checks independently
347 * reviewable.
348 * HOW: thread_pool NULL → kXR_ServerError; empty src host/path → kXR_ArgInvalid;
349 * brix_tpc_source_guard_check != 0 → kXR_NotAuthorized + guard signal;
350 * brix_tpc_check_src_policy != 0 → kXR_NotAuthorized. */
351static ngx_int_t
352tpc_prepare_check_preconditions(brix_ctx_t *ctx, ngx_connection_t *c,
353 ngx_stream_brix_srv_conf_t *conf, const brix_tpc_params_t *tpc,
354 const char *dst_path)
355{
356 char policy_err[512];
357 char egress_err[512];
358 uint16_t sport;
359
360 if (conf->common.thread_pool == NULL) {
361 return brix_tpc_refuse(ctx, c, dst_path, kXR_ServerError,
362 "TPC pull requires brix_thread_pool to be configured");
363 }
364
365 if (tpc->src_host[0] == '\0' || tpc->src_path[0] == '\0') {
366 return brix_tpc_refuse(ctx, c, dst_path, kXR_ArgInvalid,

368 }

370 /*
371 * Source-egress allowlist gate (naming policy): when the operator enabled
372 * brix_tpc_source_guard, the requested source host must appear on
373 * brix_tpc_source_allow or we refuse to originate — before the address-range
374 * check below, so a non-permitted host is rejected without a DNS lookup.
375 * This is the server-side request-forgery control the client egress
376 * self-test verifies; a refusal is banworthy, so it emits the guard signal
377 * and bumps the labelless refusal counter.

379 if (brix_tpc_source_guard_check(conf->common.tpc_source_guard,
380 conf->common.tpc_source_allow, tpc->src_host,

382 != 0)

390 * Source policy gate (SSRF defence): before we ever connect outbound, the
391 * resolved source host/port is checked against the loopback/private-range
392 * allow flags. A destination server must not be coercible into pulling from
393 * internal addresses unless the operator explicitly permits it.
394 */
395 sport = tpc->src_port ? tpc->src_port : 1094;
396 if (brix_tpc_check_src_policy(tpc->src_host, sport,
397 conf->common.tpc_allow_local, conf->common.tpc_allow_private,
398 policy_err, sizeof(policy_err))
399 != 0)
400 {
401 return brix_tpc_refuse(ctx, c, dst_path, kXR_NotAuthorized, policy_err);
402 }
403
404 return NGX_OK;
405}

408ngx_int_t
409brix_tpc_prepare_pull(brix_ctx_t *ctx, ngx_connection_t *c,
410 ngx_stream_brix_srv_conf_t *conf, const brix_tpc_params_t *tpc,
411 const char *dst_path, uint16_t options, uint16_t mode_bits)
412{
413 brix_file_t *file;
414 struct stat st;

416 int idx;
417
418 /* TPC_ANSWERED means the gate refused and already said so on the wire; the
419 * wire contract spells "handled, nothing more to send" NGX_OK. Collapsing the
420 * two here — and only here — is what keeps a refusal from reading as consent
421 * one frame up while still ending the request. */
422 pre = tpc_prepare_check_preconditions(ctx, c, conf, tpc, dst_path);
423 if (pre != NGX_OK) {
424 return pre == TPC_ANSWERED ? NGX_OK : pre;
425 }
426

428 if (idx < 0) {
429 pre = brix_tpc_refuse(ctx, c, dst_path, kXR_ServerError,
430 "too many open files");
431 return pre == TPC_ANSWERED ? NGX_OK : pre;
432 }
433
434 file = &ctx->files[idx];
435 pre = brix_tpc_open_destination(ctx, c, conf, dst_path, options, mode_bits,
436 idx, file, &st);
437 if (pre != NGX_OK) {
438 return pre == TPC_ANSWERED ? NGX_OK : NGX_ERROR;
439 }
440
441 tpc_init_dst_file(ctx, c, conf, file, tpc, &st);
442
443 if (brix_set_fhandle_path(ctx, c, idx, dst_path) != NGX_OK) {
444 brix_free_fhandle(ctx, idx);
445 return NGX_ERROR;
446 }
447
448 if (!ctx->is_bound) {
449 brix_session_handle_publish(ctx->login.sessid, idx, file);
450 }
451
452 brix_log_access(ctx, c, "OPEN", dst_path, "tpc-pull", 1, 0, NULL, 0);
453 BRIX_OP_OK(ctx, BRIX_OP_OPEN_WR);
454
455 return tpc_send_open_response(ctx, c, idx, &st, options);
