46/*
47 * A queued response is not a return value. brix_send_error() returns NGX_OK once
48 * the kXR_error is on the wire, so a function that ends `return
49 * brix_send_error(...)` reports SUCCESS to whoever called it. At a protocol entry
50 * point that is exactly right: NGX_OK there means "request handled" and nothing
51 * runs afterwards. One call deep it is a gate bypass - the caller reads NGX_OK as
52 * "the check passed, carry on" and carries on past a refusal it has already
53 * answered, allocating the handle and queueing a second, contradictory response.
54 * The gates below therefore return TPC_ANSWERED, "refused, and the client has
55 * been told", which only brix_tpc_prepare_pull - the entry point, where the
56 * distinction stops mattering - folds back into the wire contract's NGX_OK.
57 */
58/* Log, count and answer one refusal. Never reports success; a failed write is
59 * still NGX_ERROR, so connection teardown is unchanged. */
60ngx_int_t
61brix_tpc_refuse(brix_ctx_t *ctx, ngx_connection_t *c, const char *dst_path,
62 uint16_t code, const char *msg)
63{
64 brix_log_access(ctx, c, "OPEN", dst_path, "tpc-pull", 0, code, msg, 0);
65 BRIX_OP_ERR(ctx, BRIX_OP_OPEN_WR);
66
67 return brix_send_error(ctx, c, code, msg) == NGX_OK
68 ? TPC_ANSWERED : NGX_ERROR;
69}

71/* WHAT: Build kXR_ok open response body.
72 * - Includes fhandle + optional statbuf from fstat
73 * - Calls brix_build_resp_hdr then brix_queue_response
74 * - Returns NGX_OK or NGX_ERROR on alloc failure
75 * - Caller: brix_tpc_prepare_pull (end of pull prep) */
76static ngx_int_t
77tpc_send_open_response(brix_ctx_t *ctx, ngx_connection_t *c, int idx,
78 const struct stat *st, uint16_t options)
79{
80 ServerOpenBody body;
81 char statbuf[256];
82 size_t bodylen;
83 size_t total;
84 u_char *buf;
85 ngx_flag_t want_stat;
86
87 want_stat = (options & kXR_retstat) ? 1 : 0;
88 statbuf[0] = '\0';
89 bodylen = sizeof(ServerOpenBody);
90
91 if (want_stat && st != NULL) {
92 int stat_flags = 0;
93
94 if (st->st_mode & (S_IRUSR | S_IRGRP | S_IROTH)) {
95 stat_flags |= kXR_readable;
96 }
97 if (st->st_mode & (S_IWUSR | S_IWGRP | S_IWOTH)) {
98 stat_flags |= kXR_writable;
99 }
100
101 snprintf(statbuf, sizeof(statbuf), "%llu %lld %d %ld",
102 (unsigned long long) st->st_ino,
103 (long long) st->st_size,
104 stat_flags,
105 (long) st->st_mtime);
106 bodylen += strlen(statbuf) + 1;
107 }
108
109 total = XRD_RESPONSE_HDR_LEN + bodylen;
110 BRIX_PALLOC_OR_RETURN(buf, c->pool, total, NGX_ERROR);
111
112 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok, (uint32_t) bodylen,
113 (ServerResponseHdr *) buf);
114
115 ngx_memzero(&body, sizeof(body));
116 body.fhandle[0] = (u_char) idx;
117 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, &body, sizeof(body));
118
119 if (statbuf[0] != '\0') {
120 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN + sizeof(ServerOpenBody),
121 statbuf, strlen(statbuf) + 1);
122 }
123
124 return brix_queue_response(ctx, c, buf, total);
125}

213 * destination's cap) and TPC_STREAMS_MAX — never trusted raw. */

220 * per-location audit file). The attacker-chosen source host rides the sanitized
221 * path field; the labelless metric carries the count (INVARIANT 8).
222 * WHY: an un-guarded gateway is a request-forgery primitive; a client probing

224 * so the refusal shares the guard/fail2ban contract (signal=tpc_egress).
225 * HOW: copy the cached ISO-8601 clock + client IP into NUL-terminated stacks,
226 * sanitize the host into the path field, format via the shared guard formatter. */

228tpc_egress_emit_signal(brix_ctx_t *ctx, ngx_connection_t *c, const char *host)

261/* WHAT: Validate the pull preconditions before any state is allocated — a
262 * configured thread pool, a non-empty TPC source host+path, the operator's
263 * source-name allowlist, and an SSRF-safe source address per the
264 * loopback/private allow flags. NGX_OK when the request may proceed;
265 * TPC_ANSWERED once a refusal has been answered; NGX_ERROR if it could not.
266 * WHY: this is the security-load-bearing front door, SSRF gate included;
267 * isolating it keeps the orchestrator flat and the checks independently
268 * reviewable.
269 * HOW: thread_pool NULL → kXR_ServerError; empty src host/path → kXR_ArgInvalid;
270 * brix_tpc_source_guard_check != 0 → kXR_NotAuthorized + guard signal;
271 * brix_tpc_check_src_policy < 0 → kXR_NotAuthorized, > 0 (no cached
272 * answer) → brix_tpc_prepare_park_dns parks the open on an async resolve
273 * and the verdict is delivered by launch_dns.c (TPC_ANSWERED here). */
274static ngx_int_t
275tpc_prepare_check_preconditions(brix_ctx_t *ctx, ngx_connection_t *c,
276 ngx_stream_brix_srv_conf_t *conf, const brix_tpc_params_t *tpc,
277 const char *dst_path, uint16_t options, uint16_t mode_bits)
278{
279 char policy_err[512];
280 char egress_err[512];
281 uint16_t sport;
282 int verdict;
283
284 if (conf->common.thread_pool == NULL) {
285 return brix_tpc_refuse(ctx, c, dst_path, kXR_ServerError,
286 "TPC pull requires brix_thread_pool to be configured");
287 }
288
289 if (tpc->src_host[0] == '\0' || tpc->src_path[0] == '\0') {
290 return brix_tpc_refuse(ctx, c, dst_path, kXR_ArgInvalid,
291 "invalid or incomplete TPC source");
292 }
293
294 /*
295 * Source-egress allowlist gate (naming policy): when the operator enabled
296 * brix_tpc_source_guard, the requested source host must appear on
297 * brix_tpc_source_allow or we refuse to originate — before the address-range
298 * check below, so a non-permitted host is rejected without a DNS lookup.
299 * This is the server-side request-forgery control the client egress
300 * self-test verifies; a refusal is banworthy, so it emits the guard signal
301 * and bumps the labelless refusal counter.
302 */
303 if (brix_tpc_source_guard_check(conf->common.tpc_source_guard,
304 conf->common.tpc_source_allow, tpc->src_host,
305 egress_err, sizeof(egress_err))
306 != 0)
307 {
308 BRIX_SRV_METRIC_INC(ctx, tpc_egress_refused_total);
309 tpc_egress_emit_signal(ctx, c, tpc->src_host);
310 return brix_tpc_refuse(ctx, c, dst_path, kXR_NotAuthorized, egress_err);
311 }
312
313 /*
314 * Source policy gate (SSRF defence): before we ever connect outbound, the
315 * source host's addresses are checked against the loopback/private-range
316 * allow flags. A destination server must not be coercible into pulling from
317 * internal addresses unless the operator explicitly permits it. The verdict
318 * is answered from an IP literal or the per-worker DNS cache; an unknown
319 * name parks this open on an async resolve (launch_dns.c) and the verdict is
320 * delivered when the answer arrives — the event loop never blocks on DNS
321 * (I-DNS-1), and the pull thread re-checks every candidate it actually
322 * dials (I-DNS-3).
323 */
324 sport = tpc->src_port ? tpc->src_port : TPC_DEFAULT_PORT;
325 verdict = brix_tpc_check_src_policy(conf, tpc->src_host, sport,
326 policy_err, sizeof(policy_err));
327 if (verdict < 0) {
328 return brix_tpc_refuse(ctx, c, dst_path, kXR_NotAuthorized, policy_err);
329 }
330 if (verdict > 0) {
331 return brix_tpc_prepare_park_dns(ctx, c, conf, tpc, dst_path, options,
332 mode_bits);
333 }
334
335 return NGX_OK;
336}
