4/*
5 * tier.h — the composable storage-tier model (phase-64 §2).
6 *
7 * WHAT: The POD types that describe a node's storage stack as up to three
8 * orthogonal TIERS — backend, cache_store, stage_store — each a complete
9 * self-describing filesystem addressed by { driver, location, credential }.
10 * Declares the parsed per-tier config (brix_tier_cfg_t), the per-export
11 * stack (brix_tier_stack_t) that the registry memoises, the cache/stage
12 * policy PODs, the capability-gap report, and the four public entry points
13 * (parse_store / status / build / build_stack).
14 *
15 * WHY: Before phase-64 the registry carried flat backend[16] / origin_x / staging
16 * fields and the cache was a separate, scheme-dispatched subsystem. The tier
17 * model unifies them: every tier is one brix_sd_instance_t, the registry
18 * composes them top-down (cache → stage → backend), and the VFS consumes the
19 * composed top BLIND — no handler, VFS op, or cache module branches on a
20 * driver/protocol/tier (P3, G4/G5). The cache and stage tiers are OPTIONAL,
21 * except a nearline (tape) backend REQUIRES a cache tier as its recall
22 * target (P4, §9.4).
23 *
24 * HOW: tier_config.c parses a "<scheme>:<location> [credential=][block_size=]"
25 * store URL into an brix_tier_cfg_t and reports each tier's readiness
26 * against a per-role capability contract (§2.2) — a missing slot is a
27 * tracked "needs development" task, never a blocker (P1). tier_build.c is
28 * the single driver-name dispatch that turns one cfg into a bound SD
29 * instance, and composes the three into the stack's top via the sd_cache /
30 * sd_stage decorators. See docs/refactor/phase-64-fully-tiered-composable-
31 * storage.md (§2, §4, §5, §8, Appendix I/L).
32 */

259/* Parse ONLY the trailing params of a `brix_storage_backend` line into p->out,
260 * with role BACKEND (phase-115 W5.1).
261 *
262 * WHY a second entry point rather than brix_tier_parse_store: the BACKEND tier
263 * is not built through the tier composer — brix_vfs_backend_config_str parses
264 * its URL into the VFS backend registry, and that dispatcher accepts schemes
265 * the tier table does not carry (mirage:, cephfs-ro:, a bare driver name).
266 * Running the full store-URL parser over every backend line would newly refuse
267 * configurations that work today. This entry resolves the scheme ONLY when
268 * params are present — a line with no params is untouched — so an operator pays
269 * the tier table's vocabulary exactly when they ask for one of its params.
270 *
271 * `args` may be NULL/empty: that is the no-params fast path and always NGX_OK
272 * with p->out zeroed. Otherwise the scheme is resolved (driver + tls) and the
273 * params validated exactly as on a cache/stage line, so `verify_pages`,
274 * `mode=`, `prot=`, `credential=`, `block_size=` and `nearline` speak one
275 * vocabulary on every store directive. NGX_ERROR (message in p->err, [emerg]
276 * through p->cf) on an unknown scheme or a refused param.
277 *
278 * An EMPTY `url` WITH params is the `posix:<path>` spelling that
279 * brix_storage_backend_posix_root has already folded into the export root: the
280 * driver is the default POSIX one, and it is named as such so the refusal
281 * points at the param rather than at a store line the operator can see. */
282ngx_int_t brix_tier_parse_backend_params(brix_tier_parse_t *p, ngx_str_t *url,
283 ngx_array_t *args);
