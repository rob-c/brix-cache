1/*
2 * sd_cache_maint.c — the decorator's public maintenance + async-fill seam.
3 *
4 * WHAT: The non-driver entrypoints callers use to introspect and maintain a
5 * composed cache decorator from outside its own vtable: the
6 * caller-driven evict + cached-size probe (eviction accounting), the
7 * SP2 async-fill offload trio (fill_needs_offload / fill_key /
8 * fill_key_ex) and the registry-driven tier attachment (cold store,
9 * sibling ring, swarm ring swap).
10 *
11 * WHY: Split from sd_cache.c for the file-size cap when the eviction
12 * accounting probes landed; the tier setters followed in 2.0 when the
13 * NOFILL source-miss branch grew the open tree. They share no open-path
14 * state —
15 * each re-enters through the public instance handle — so they isolate
16 * cleanly from the interposed read-open decision tree that sd_cache.c
17 * owns.
18 *
19 * HOW: Every entrypoint gates on brix_sd_cache_instance_is() — which stays in
20 * sd_cache.c beside the static driver vtable it compares against — and
21 * reaches the decorator state via SD_CACHE_ST from sd_cache_internal.h;
22 * the fill entrypoints delegate to the fill spine (sd_cache_fill.c).
23 */

32/* Evict `key` from this decorator's cache store, if `inst` is a cache decorator
33 * (a no-op otherwise). The namespace VFS ops (delete/rename) dispatch on the
34 * unwrapped leaf instance so per-user credentials thread to the origin driver,
35 * which skips the decorator's own unlink/rename and therefore its evict side
36 * effect; the caller invokes this after a successful leaf op to keep the store
37 * coherent. Mirrors the evict in sd_cache_unlink()/sd_cache_rename(). Returns
38 * the logical bytes evicted (0 if nothing was cached / not a decorator) so the
39 * caller can feed brix_metric_cache_evicted. */
40uint64_t
41brix_sd_cache_evict(brix_sd_instance_t *inst, const char *key)
42{
43 sd_cache_inst_state *st;
44
45 if (!brix_sd_cache_instance_is(inst) || key == NULL) {
46 return 0;
47 }
48 st = SD_CACHE_ST(inst);
49 return brix_cstore_evict_sized(&st->cstore, key);
50}

52/* Vtable evict pair (phase-107 C2) — the promoted form of brix_sd_cache_evict
53 * above: drop THIS decorator's cached copy of `path`, then RELAY downward when
54 * the source also carries the slot, summing the reclaimed bytes. The relay is
55 * what keeps a nearline release reachable on a cache-fronted export — a cache
56 * that only dropped its own copy would leave the frm online buffer full
57 * forever — while a source without the slot (posix/http) simply contributes
58 * nothing. INSTANCE-keyed cred twin: the decorator's own store is
59 * service-owned so the cred adds nothing to the local drop; it exists so the
60 * *_maybe_cred forwarder threads the per-user credential into the relay
61 * instead of refusing in DENY mode (the truncate_path_cred rationale). The
62 * local drop always happens; a relay failure propagates with *bytes_out still
63 * carrying everything actually reclaimed. */
64static ngx_int_t
65sd_cache_evict_common(brix_sd_instance_t *inst, const char *path,
66 uint64_t *bytes_out, const brix_sd_cred_t *cred)
67{
68 sd_cache_inst_state *st = SD_CACHE_ST(inst);
69 uint64_t bytes, below = 0;
70 ngx_int_t rc = NGX_OK;
71
72 bytes = brix_cstore_evict_sized(&st->cstore, path);
73 if (st->source->driver->evict != NULL
74 || st->source->driver->evict_cred != NULL)
75 {
76 rc = brix_sd_evict_maybe_cred(st->source, path, &below, cred);
77 }
78 if (bytes_out != NULL) {
79 *bytes_out = bytes + below;
80 }
81 return rc;
82}

84ngx_int_t
85sd_cache_evict_op(brix_sd_instance_t *inst, const char *path,
86 uint64_t *bytes_out)

118/* ---- async-fill seam (SP2 "shell -> full") --------------------------------
119 * The decorator's open() runs the miss-fill INLINE - correct on a worker thread
120 * (the stream fill task, a WebDAV/S3 PUT worker) but a stall on the event loop
121 * when the fill reads a remote source or writes a remote store (a socket wire
122 * client cannot do blocking I/O on the un-pumped loop; an in-process store just
123 * freezes the worker for the transfer). The HTTP read plane therefore probes
124 * whether an inline open would block, runs the fill on the thread pool, and
125 * re-enters. These three entrypoints expose exactly that - without making the
126 * SD open() contract asynchronous. See src/shared/http_cache_fill.c. */
127
128/* Would a read-open of `key` block the calling thread on slow (remote) I/O? 1
129 * only for a cache MISS whose whole-file fill would touch a non-local tier - the
130 * source exposes no local fd (a remote read: xroot/http/s3/ceph) or the cache
131 * store is not a local POSIX dir (a remote write: e.g. a rados store). A COMPLETE
132 * hit, a slice-mode object (open returns without filling), a local->local copy,
133 * or a non-cache instance all return 0 (serve inline). No blocking call - the
134 * cinfo probe hits the per-worker L1 / a local sidecar. */
135int
136brix_sd_cache_fill_needs_offload(brix_sd_instance_t *inst, const char *key)
137{
138 sd_cache_inst_state *st;
139 brix_cache_cinfo_t ci;
140 int src_slow;
141 int store_slow;
142
143 if (!brix_sd_cache_instance_is(inst) || key == NULL) {
144 return 0;
145 }
146 st = SD_CACHE_ST(inst);
147
148 /* A COMPLETE cached object is served from the store with no fill —
149 * unless its phase-68 TTL has passed (an expired manifest refills; the
150 * failed-refill path serves it stale within the 10x-TTL bound). */
151 if (brix_cstore_cinfo_load(&st->cstore, key, &ci) == NGX_OK
152 && (ci.flags & BRIX_CINFO_F_COMPLETE))
153 {
154 if (!(st->policy.cvmfs_manifest_ttl > 0
155 && brix_cache_cinfo_expired(&ci, time(NULL)) == 1))
156 {
157 return 0;
158 }
159 /* expired: fall through to the miss logic (refill if a slow tier) */
160 }
161 /* Slice/partial mode (LOCAL store): open() returns a partial object without a
162 * whole-file fill, so the open call itself does not block. */
163 if (st->policy.slice_size > 0
164 && st->cstore.meta_mode == BRIX_CMETA_LOCAL)
165 {
166 return 0;
167 }
168 /* A miss: the inline open would run the whole-file fill. Offload iff a slow
169 * tier is involved - a remote source read or a remote store write. */
170 src_slow = (brix_sd_caps(st->source) & BRIX_SD_CAP_FD) == 0;
171 store_slow = (st->cstore.meta_mode != BRIX_CMETA_LOCAL);
172 return (src_slow || store_slow) ? 1 : 0;
173}

175/* Run the whole-file fill for `key` (source -> cache store + cinfo) on the

178 * Safe off the event loop: pure driver pread/pwrite + cstore ops, no nginx pool. */

182{

186/* As brix_sd_cache_fill_key, but with the phase-92 store-then-evict passthrough
187 * opt-in threaded through (see sd_cache_fill / sd_cache.h). When *out_pt is set
188 * to 1 on an NGX_OK return, the object was filled ONLY under the passthrough
189 * policy: the caller must evict `key` (brix_sd_cache_evict) after serving it. */
190ngx_int_t
191brix_sd_cache_fill_key_ex(brix_sd_instance_t *inst, const char *key,
192 const brix_sd_cred_t *cred, int allow_pt, int *out_pt)
193{
194 if (out_pt != NULL) {
195 *out_pt = 0;
196 }
197 if (!brix_sd_cache_instance_is(inst) || key == NULL) {
198 errno = EINVAL;
199 return NGX_ERROR;
200 }
201 return sd_cache_fill(SD_CACHE_ST(inst), key, cred, allow_pt, out_pt);
202}

205/* Attach/detach the OPTIONAL cold store tier (phase-85 F7). `cold` is BORROWED

228/* Attach/detach the sibling-mesh ring (phase-85 F8). The member instances are
229 * BORROWED (registry-owned, worker lifetime) so brix_sd_cache_destroy never
230 * frees them. No-op for a non-cache instance; n == 0 (or an out-of-range self)
231 * detaches. With a ring set, sd_cache_fill tries one verified fill from the
232 * key's rendezvous-owning sibling before the origin. */
233void
234brix_sd_cache_set_peers(brix_sd_instance_t *inst,
235 const brix_sd_cache_peer_t *peers, int n, int self)
236{
237 sd_cache_inst_state *st;
238 int i;
239
240 if (!brix_sd_cache_instance_is(inst)) {
241 return;
242 }
243 st = SD_CACHE_ST(inst);
244 if (peers == NULL || n <= 0 || n > BRIX_SD_CACHE_MAX_PEERS
245 || self < 0 || self >= n)
246 {
247 st->n_peers = 0;
248 return;
249 }
250 for (i = 0; i < n; i++) {
251 st->peers[i] = peers[i];
252 }
253 st->n_peers = n;
254 st->peer_self = self;
255}

257/* Publish a swarm-built ring (phase-87 G12). Event loop only; the barrier
258 * orders the ring's contents before the pointer store so a worker-thread
259 * fill that loads the new pointer sees a fully built ring. */
260void
261brix_sd_cache_ring_swap(brix_sd_instance_t *inst,
262 const brix_sd_cache_ring_t *ring)
263{
264 sd_cache_inst_state *st;
265
266 if (!brix_sd_cache_instance_is(inst)) {
267 return;
268 }
269 if (ring != NULL
270 && (ring->n <= 0 || ring->n > BRIX_SD_CACHE_MAX_PEERS
271 || ring->self < 0 || ring->self >= ring->n))
272 {
273 return;
274 }
275 st = SD_CACHE_ST(inst);
276 ngx_memory_barrier();
277 st->dyn_ring = ring;
278}
