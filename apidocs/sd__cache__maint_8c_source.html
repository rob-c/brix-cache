1/*
2 * sd_cache_maint.c — the decorator's public maintenance + async-fill seam.
3 *
4 * WHAT: The non-driver entrypoints callers use to introspect and maintain a
5 * composed cache decorator from outside its own vtable: the
6 * caller-driven evict + cached-size probe (eviction accounting) and the
7 * SP2 async-fill offload trio (fill_needs_offload / fill_key /
8 * fill_key_ex).
9 *
10 * WHY: Split from sd_cache.c for the file-size cap when the eviction
11 * accounting probes landed. These functions share no open-path state —
12 * each re-enters through the public instance handle — so they isolate
13 * cleanly from the interposed read-open decision tree that sd_cache.c
14 * owns.
15 *
16 * HOW: Every entrypoint gates on brix_sd_cache_instance_is() — which stays in
17 * sd_cache.c beside the static driver vtable it compares against — and

19 * the fill entrypoints delegate to the fill spine (sd_cache_fill.c).

29/* Evict `key` from this decorator's cache store, if `inst` is a cache decorator
30 * (a no-op otherwise). The namespace VFS ops (delete/rename) dispatch on the
31 * unwrapped leaf instance so per-user credentials thread to the origin driver,
32 * which skips the decorator's own unlink/rename and therefore its evict side
33 * effect; the caller invokes this after a successful leaf op to keep the store
34 * coherent. Mirrors the evict in sd_cache_unlink()/sd_cache_rename(). Returns
35 * the logical bytes evicted (0 if nothing was cached / not a decorator) so the
36 * caller can feed brix_metric_cache_evicted. */
37uint64_t
38brix_sd_cache_evict(brix_sd_instance_t *inst, const char *key)
39{
40 sd_cache_inst_state *st;
41
42 if (!brix_sd_cache_instance_is(inst) || key == NULL) {
43 return 0;
44 }
45 st = SD_CACHE_ST(inst);
46 return brix_cstore_evict_sized(&st->cstore, key);
47}

49/* Vtable evict pair (phase-107 C2) — the promoted form of brix_sd_cache_evict
50 * above: drop THIS decorator's cached copy of `path`, then RELAY downward when
51 * the source also carries the slot, summing the reclaimed bytes. The relay is
52 * what keeps a nearline release reachable on a cache-fronted export — a cache
53 * that only dropped its own copy would leave the frm online buffer full
54 * forever — while a source without the slot (posix/http) simply contributes
55 * nothing. INSTANCE-keyed cred twin: the decorator's own store is
56 * service-owned so the cred adds nothing to the local drop; it exists so the
57 * *_maybe_cred forwarder threads the per-user credential into the relay
58 * instead of refusing in DENY mode (the truncate_path_cred rationale). The
59 * local drop always happens; a relay failure propagates with *bytes_out still
60 * carrying everything actually reclaimed. */
61static ngx_int_t
62sd_cache_evict_common(brix_sd_instance_t *inst, const char *path,
63 uint64_t *bytes_out, const brix_sd_cred_t *cred)
64{
65 sd_cache_inst_state *st = SD_CACHE_ST(inst);
66 uint64_t bytes, below = 0;
67 ngx_int_t rc = NGX_OK;
68
69 bytes = brix_cstore_evict_sized(&st->cstore, path);
70 if (st->source->driver->evict != NULL
71 || st->source->driver->evict_cred != NULL)
72 {
73 rc = brix_sd_evict_maybe_cred(st->source, path, &below, cred);
74 }
75 if (bytes_out != NULL) {
76 *bytes_out = bytes + below;
77 }
78 return rc;
79}

115/* ---- async-fill seam (SP2 "shell -> full") --------------------------------
116 * The decorator's open() runs the miss-fill INLINE - correct on a worker thread
117 * (the stream fill task, a WebDAV/S3 PUT worker) but a stall on the event loop
118 * when the fill reads a remote source or writes a remote store (a socket wire
119 * client cannot do blocking I/O on the un-pumped loop; an in-process store just
120 * freezes the worker for the transfer). The HTTP read plane therefore probes
121 * whether an inline open would block, runs the fill on the thread pool, and
122 * re-enters. These three entrypoints expose exactly that - without making the
123 * SD open() contract asynchronous. See src/shared/http_cache_fill.c. */
124
125/* Would a read-open of `key` block the calling thread on slow (remote) I/O? 1
126 * only for a cache MISS whose whole-file fill would touch a non-local tier - the
127 * source exposes no local fd (a remote read: xroot/http/s3/ceph) or the cache
128 * store is not a local POSIX dir (a remote write: e.g. a rados store). A COMPLETE
129 * hit, a slice-mode object (open returns without filling), a local->local copy,
130 * or a non-cache instance all return 0 (serve inline). No blocking call - the
131 * cinfo probe hits the per-worker L1 / a local sidecar. */
132int
133brix_sd_cache_fill_needs_offload(brix_sd_instance_t *inst, const char *key)
134{
135 sd_cache_inst_state *st;
136 brix_cache_cinfo_t ci;
137 int src_slow;
138 int store_slow;
139
140 if (!brix_sd_cache_instance_is(inst) || key == NULL) {
141 return 0;
142 }
143 st = SD_CACHE_ST(inst);
144
145 /* A COMPLETE cached object is served from the store with no fill —
146 * unless its phase-68 TTL has passed (an expired manifest refills; the
147 * failed-refill path serves it stale within the 10x-TTL bound). */
148 if (brix_cstore_cinfo_load(&st->cstore, key, &ci) == NGX_OK
149 && (ci.flags & BRIX_CINFO_F_COMPLETE))
150 {
151 if (!(st->policy.cvmfs_manifest_ttl > 0
152 && brix_cache_cinfo_expired(&ci, time(NULL)) == 1))
153 {
154 return 0;
155 }
156 /* expired: fall through to the miss logic (refill if a slow tier) */
157 }
158 /* Slice/partial mode (LOCAL store): open() returns a partial object without a
159 * whole-file fill, so the open call itself does not block. */
160 if (st->policy.slice_size > 0
161 && st->cstore.meta_mode == BRIX_CMETA_LOCAL)
162 {
163 return 0;
164 }
165 /* A miss: the inline open would run the whole-file fill. Offload iff a slow
166 * tier is involved - a remote source read or a remote store write. */
167 src_slow = (brix_sd_caps(st->source) & BRIX_SD_CAP_FD) == 0;
168 store_slow = (st->cstore.meta_mode != BRIX_CMETA_LOCAL);
169 return (src_slow || store_slow) ? 1 : 0;
170}

172/* Run the whole-file fill for `key` (source -> cache store + cinfo) on the

175 * Safe off the event loop: pure driver pread/pwrite + cstore ops, no nginx pool. */

179{
180 return brix_sd_cache_fill_key_ex(inst, key, cred, 0, NULL);
181}

183/* As brix_sd_cache_fill_key, but with the phase-92 store-then-evict passthrough
184 * opt-in threaded through (see sd_cache_fill / sd_cache.h). When *out_pt is set
185 * to 1 on an NGX_OK return, the object was filled ONLY under the passthrough
186 * policy: the caller must evict `key` (brix_sd_cache_evict) after serving it. */
187ngx_int_t
188brix_sd_cache_fill_key_ex(brix_sd_instance_t *inst, const char *key,

190{

197 }
