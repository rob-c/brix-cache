1/*
2 * curl_pin.h — CURLOPT_RESOLVE pinning for libcurl callers (phase-116 App. A).
3 *
4 * WHAT: brix_dns_curl_pin() resolves the host of `url` through
5 * brix_dns_resolve_sync() and pins the answer into the easy handle
6 * with CURLOPT_RESOLVE, so libcurl never runs its own resolver.
7 * brix_dns_curl_pin_entry() is the shared "host:port:ip1,ip2" formatter
8 * (also used by the WebDAV TPC pin, which has its own policy check).
9 * WHY: libcurl resolves with getaddrinfo (or c-ares) on whatever thread
10 * performs the transfer — a second resolver path with its own cache and
11 * its own idea of resolv.conf. Pinning keeps every lookup in the one
12 * driver (I-DNS-2/3) and makes the seam guard's rule mechanical: a
13 * CURLOPT_URL without a pin call is a defect.
14 * HOW: Thread-pool code only (the sync resolver blocks). The slist the
15 * caller receives must outlive curl_easy_perform() and be freed after
16 * it; every pin call first clears CURLOPT_RESOLVE, so a warm per-thread
17 * handle never keeps a pointer to a list its previous request freed,
18 * and a literal host after a named one pins nothing.
19 * brix_dns_curl_perform_pinned() is the transfer loop for callers that
20 * used to set CURLOPT_FOLLOWLOCATION: libcurl would resolve each
21 * Location host itself, so the helper follows redirects by hand and
22 * re-pins every hop through the same driver.
23 */

47/* Total wall-clock ceiling a pinned transfer gets when the caller names none.
48 * Never 0: libcurl reads 0 as "no timeout", and an unbounded blocking perform

50 * the S3 origin's own fallback ceiling. Left overridable at compile time for
51 * one caller only: tests/c/curl_pin_budget_test.c proves the fallback bounds a
52 * black hole without the unit waiting a minute to do it. */

57/* One pinned transfer that may redirect. `on_hop` (optional) runs before
58 * every attempt so the caller can rewind its response sink; `err` receives
59 * the resolution failure text when the result is CURLE_COULDNT_RESOLVE_HOST. */
60typedef struct {
61 const brix_dns_policy_t *policy;
62 const char *url;
63 ngx_uint_t max_redirects;
64 /* 0 is fail-closed: redirects must retain scheme, host and effective
65 * port. Set only for a request whose method/body/headers carry no reusable
66 * authority; HTTPS still cannot downgrade. */
67 ngx_flag_t allow_cross_origin;
68 /* Total ms for the whole transfer, redirects included. <= 0 takes
69 * BRIX_DNS_CURL_TIMEOUT_MS_DEFAULT; the loop spends it down hop by hop, so
70 * a chain of N redirects cannot cost N times the caller's timeout. */
71 long timeout_ms;
72 void (*on_hop)(void *data);
73 void *hop_data;
74 char *err;
75 size_t errsz;
76} brix_dns_curl_transfer_t;

78/* Perform `t->url` on `curl`, re-pinning each redirect hop (Location taken
79 * from CURLINFO_REDIRECT_URL) through the driver. HTTPS can never downgrade;
80 * origin changes require allow_cross_origin. Method and body options stay as
81 * the caller set them, so a POST is re-issued as a POST (307/308 semantics).
82 * Returns the last curl_easy_perform() result, or
83 * CURLE_TOO_MANY_REDIRECTS / CURLE_URL_MALFORMAT / CURLE_COULDNT_RESOLVE_HOST
84 * / CURLE_OPERATION_TIMEDOUT from the loop itself. The caller must NOT set
85 * CURLOPT_FOLLOWLOCATION or CURLOPT_URL, and need not set a timeout: the loop
86 * owns the budget and overrides CURLOPT_TIMEOUT_MS / CURLOPT_CONNECTTIMEOUT_MS
87 * on every hop with what is left of `timeout_ms`. */
88CURLcode brix_dns_curl_perform_pinned(CURL *curl,
89 const brix_dns_curl_transfer_t *t);
