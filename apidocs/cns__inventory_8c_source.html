31static brix_cns_entry_t *
32inv_find(const brix_cns_inv_t *inv, const char *path)
33{
34 uint32_t i;
35
36 for (i = 0; i < inv->capacity; i++) {
37 const brix_cns_entry_t *e = &inv->slots[i];
38 if (e->used && strcmp(e->path, path) == 0) {
39 return (brix_cns_entry_t *) e;
40 }
41 }
42 return NULL;
43}

58int
59brix_cns_inv_apply(brix_cns_inv_t *inv, uint8_t op, const char *path,
60 uint64_t size, uint64_t mtime, uint32_t server_id)
61{
62 brix_cns_entry_t *e;
63 size_t plen;
64
65 if (inv == NULL || path == NULL) {
66 return -1;
67 }
68 plen = strlen(path);
69 if (plen == 0 || plen > BRIX_CNS_PATH_MAX) {
70 return -1;
71 }
72
73 if (op == BRIX_CNS_DEL || op == BRIX_CNS_RMDIR) {
74 e = inv_find(inv, path);
75 if (e != NULL) {
76 e->used = 0;
77 if (inv->count > 0) { inv->count--; }
78 }
79 return 0; /* removing an absent path is a no-op */
80 }
81
82 if (op != BRIX_CNS_ADD && op != BRIX_CNS_MKDIR) {
83 return -1;
84 }
85
86 e = inv_find(inv, path);
87 if (e == NULL) {
88 e = inv_free_slot(inv);
89 if (e == NULL) {
90 return -1; /* inventory full (fixed cap) */
91 }
92 memcpy(e->path, path, plen);
93 e->path[plen] = '\0';
94 e->used = 1;
95 inv->count++;
96 }
97 e->size = size;
98 e->mtime = mtime;
99 e->server_id = server_id;
100 e->is_dir = (op == BRIX_CNS_MKDIR) ? 1 : 0;
101 return 0;
102}

104/*
105 * inv_is_under — is `path` a strict descendant of directory `dir`?
106 *
107 * Component-boundary aware on purpose: "/a/bc" is NOT under "/a/b" even though
108 * it shares the prefix, so a rename of /a/b must leave /a/bc alone. The check is
109 * therefore prefix + an explicit '/' at the boundary.
110 */
111static int
112inv_is_under(const char *path, const char *dir, size_t dlen)
113{
114 return strncmp(path, dir, dlen) == 0 && path[dlen] == '/';
115}

117/*
118 * inv_reprefix — rewrite one child entry from the old directory to the new one.
119 *
120 * Returns 0 when the entry was rewritten, -1 when the resulting path would not
121 * fit — the caller then drops the entry rather than storing a truncated (and so
122 * fabricated) path.
123 */
124static int
125inv_reprefix(brix_cns_entry_t *e, size_t oldlen, const char *newp, size_t newlen)
126{
127 const char *tail = e->path + oldlen; /* starts at the '/' boundary */
128 size_t tlen = strlen(tail);
129
130 if (newlen + tlen > BRIX_CNS_PATH_MAX) {
131 return -1;
132 }
133 /* Build in place back-to-front: the tail may overlap its destination, so
134 * memmove it clear of the new prefix before writing the prefix in. */
135 memmove(e->path + newlen, tail, tlen + 1);
136 memcpy(e->path, newp, newlen);
137 return 0;
138}

140/*
141 * inv_rename_subtree — carry every recorded descendant of `oldp` across to
142 * `newp`.
143 *
144 * Runs BEFORE the parent entry moves: the parent itself never matches
145 * inv_is_under (an exact match has no '/' boundary at oldlen), so that ordering
146 * keeps this a single pass with no entry visited twice. An entry whose new path
147 * would not fit is dropped rather than truncated — a truncated path is a
148 * fabricated one, and the next inventory sync re-learns the real entry.
149 */
150static void
151inv_rename_subtree(brix_cns_inv_t *inv, const char *oldp, size_t oldlen,
152 const char *newp, size_t newlen)
153{
154 uint32_t i;
155
156 for (i = 0; i < inv->capacity; i++) {
157 brix_cns_entry_t *e = &inv->slots[i];
158
159 if (!e->used || !inv_is_under(e->path, oldp, oldlen)) {
160 continue;
161 }
162 if (inv_reprefix(e, oldlen, newp, newlen) != 0) {
163 e->used = 0;
164 if (inv->count > 0) { inv->count--; }
165 }
166 }
167}

169/*
170 * inv_rename_dest_slot — the entry that will hold `newp` once the rename lands.
171 *
172 * An existing destination is overwritten and the now-stale source entry retired
173 * — exactly what the data server's rename(2) already did on disk. Otherwise the
174 * source entry is reused in place, or a free slot claimed when the source was
175 * never recorded. NULL only when the fixed-capacity inventory is full.
176 */
177static brix_cns_entry_t *
178inv_rename_dest_slot(brix_cns_inv_t *inv, const char *oldp, const char *newp)
179{
180 brix_cns_entry_t *dst = inv_find(inv, newp);
181 brix_cns_entry_t *src = inv_find(inv, oldp);
182
183 if (dst != NULL) {
184 if (src != NULL && src != dst) {
185 src->used = 0;
186 if (inv->count > 0) { inv->count--; }
187 }
188 return dst;
189 }
190 if (src != NULL) {
191 return src;
192 }
193
194 dst = inv_free_slot(inv);
195 if (dst == NULL) {
196 return NULL;
197 }
198 dst->used = 1;
199 inv->count++;
200 return dst;
201}

203int
204brix_cns_inv_rename(brix_cns_inv_t *inv, const char *oldp, const char *newp,
205 uint64_t size, uint64_t mtime, int is_dir, uint32_t server_id)
206{
207 brix_cns_entry_t *e;
208 size_t oldlen, newlen;
209
210 if (inv == NULL || oldp == NULL || newp == NULL) {
211 return -1;
212 }
213 oldlen = strlen(oldp);
214 newlen = strlen(newp);
215 if (oldlen == 0 || oldlen > BRIX_CNS_PATH_MAX
216 || newlen == 0 || newlen > BRIX_CNS_PATH_MAX)
217 {
218 return -1;
219 }
220
221 inv_rename_subtree(inv, oldp, oldlen, newp, newlen);
222
223 e = inv_rename_dest_slot(inv, oldp, newp);
224 if (e == NULL) {
225 return -1; /* inventory full (fixed cap) */
226 }
227 memcpy(e->path, newp, newlen);
228 e->path[newlen] = '\0';
229 e->size = size;
230 e->mtime = mtime;
231 e->server_id = server_id;
232 e->is_dir = is_dir ? 1 : 0;
233 return 0;
234}
