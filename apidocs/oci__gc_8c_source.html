1/*
2 * oci_gc.c — the registry's own garbage collection, on a maintenance timer
3 * (phase-104 §D15.5).
4 *
5 * WHAT: a config-time registry of store roots plus one worker-0 timer that
6 * runs the shared mark-and-sweep pass (shared/oci/gc.h) over each of
7 * them at its configured interval.
8 * WHY: `brixoci gc` already answers the whole-store question correctly, but
9 * it answers it only when somebody remembers to ask. A registry that
10 * accepts DELETEs and never reclaims is a registry whose disk usage
11 * only ever goes up, so the deployment that cannot run a cron job —
12 * a container, a read-only host, an operator who has one config file
13 * and no shell — needs the proxy to ask on its own.
14 * HOW: the SAME kernel, deliberately: a divergence between what the tool
15 * sweeps and what the server sweeps would be a divergence nobody could
16 * see from either side. This file contributes only *when*: one store at
17 * a time, on the thread pool so the walk stays off the event loop, on
18 * worker 0 alone so N workers do not sweep one store N times, and never
19 * while a previous pass is still running. The grace window is what makes
20 * that safe against a push in flight — see shared/oci/gc.h — and the
21 * pass is off unless `brix_oci_gc_interval` says otherwise.
22 */

32/* Registered store roots. One registry per host is what a config file can
33 * plausibly describe; a deployment with more distinct registry roots than
34 * this is one that wants a cron job, not a bigger array. */
35#define OCI_GC_MAX_STORES 16

37typedef struct {
38 char root[PATH_MAX];
39 time_t grace;
40 ngx_msec_t interval;
41 ngx_msec_t due; /* ngx_current_msec of the next pass */
42} oci_gc_store_t;

44/* Config-time state, read by worker 0's timer. Written before the fork, so
45 * every worker sees the same table and only one of them acts on it. */

47static ngx_uint_t oci_gc_store_count;

55void
56brix_oci_gc_register(const char *root, ngx_msec_t interval, time_t grace)
57{
58 oci_gc_store_t *s;
59 ngx_uint_t i;
60 size_t n;
61
62 if (root == NULL || root[0] != '/' || interval == 0) {
63 return;
64 }
65 n = ngx_strlen(root);
66 if (n == 0 || n >= PATH_MAX) {
67 return;
68 }
69 /* Every location that inherits the directives registers the same root, so
70 * the dedup is the normal case rather than an operator mistake. First
71 * registration wins: two locations describing one store with different
72 * cadences are describing one store, and the store can only have one. */
73 for (i = 0; i < oci_gc_store_count; i++) {
74 if (ngx_strcmp(oci_gc_stores[i].root, root) == 0) {
75 return;
76 }
77 }
78 if (oci_gc_store_count >= OCI_GC_MAX_STORES) {
79 return;
80 }
81 s = &oci_gc_stores[oci_gc_store_count++];
82 ngx_memcpy(s->root, root, n + 1);
83 s->interval = interval;
84 s->grace = grace;
85}

95/* What one finished pass has to say. A store with nothing to reclaim says
96 * nothing at all: a maintenance timer that logs every time it runs teaches
97 * operators to filter it out, and then it cannot tell them anything. */
98static void
99oci_gc_report(const brix_oci_gc_t *c, int rc, const char *err, ngx_log_t *log)
100{
101 if (rc != BRIX_OCI_GC_OK) {
102 ngx_log_error(NGX_LOG_ERR, log, 0,
103 "brix oci: registry gc over \"%s\" failed: %s",
104 c->root, err);
105 return;
106 }
107 if (c->st.blobs_swept == 0 && c->st.marks == 0 && c->st.refs == 0) {
108 return;
109 }
110 ngx_log_error(NGX_LOG_NOTICE, log, 0,
111 "brix oci: registry gc over \"%s\" swept %ui blob(s) "
112 "(%uL bytes), %ui layer mark(s), %ui referrer(s); "
113 "kept %ui live, %ui within grace",
114 c->root, (ngx_uint_t) c->st.blobs_swept,
115 (uint64_t) c->st.bytes, (ngx_uint_t) c->st.marks,
116 (ngx_uint_t) c->st.refs, (ngx_uint_t) c->st.blobs_live,
117 (ngx_uint_t) c->st.blobs_young);
118}

152static void
153oci_gc_done(ngx_event_t *ev)
154{
155 ngx_thread_task_t *task = ev->data;
156 oci_gc_task_t *t = task->ctx;
157 ngx_pool_t *pool = t->pool;
158

167/* Post one store's pass to the thread pool. NGX_DECLINED = no pool, or the
168 * post failed; the caller then runs it inline. */
169static ngx_int_t
170oci_gc_offload(const oci_gc_store_t *s, ngx_log_t *log)
171{
172 static ngx_str_t pname = ngx_string("default");
173 ngx_thread_pool_t *pool;
174 ngx_thread_task_t *task;
175 oci_gc_task_t *t;
176 ngx_pool_t *tp;
177
178 pool = (ngx_cycle != NULL)
179 ? ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname) : NULL;
180 if (pool == NULL) {
181 return NGX_DECLINED;
182 }
183 tp = ngx_create_pool(4096, log);
184 if (tp == NULL) {
185 return NGX_DECLINED;
186 }
187 task = ngx_thread_task_alloc(tp, sizeof(oci_gc_task_t));
188 if (task == NULL) {
189 ngx_destroy_pool(tp);
190 return NGX_DECLINED;
191 }
192 t = task->ctx;
193 t->pool = tp;
194 t->log = log;
195 t->rc = BRIX_OCI_GC_OK;
196 t->err[0] = '\0';
197 oci_gc_fill(&t->gc, s);
198
199 brix_task_bind(task, oci_gc_thread, oci_gc_done);
200 task->event.log = log;
201 if (ngx_thread_task_post(pool, task) != NGX_OK) {
202 ngx_destroy_pool(tp);
203 return NGX_DECLINED;
204 }
205 oci_gc_inflight++;
206 return NGX_OK;
207}

212/* Without a thread pool the walk runs here, on the event loop. That is the
213 * tree's standing fallback (stage_engine_scheduler.c makes the same choice),
214 * and it is the honest one: a pass that silently does not happen is worse
215 * than a pass that costs worker 0 a stall it can be configured away from with
216 * `aio threads`. */
217static void
218oci_gc_inline(const oci_gc_store_t *s, ngx_log_t *log)
219{
220 brix_oci_gc_t c;
221 char err[256] = "";
222 int rc;
223
224 oci_gc_fill(&c, s);
225 rc = brix_oci_gc_run(&c, err, sizeof(err));
226 oci_gc_report(&c, rc, err, log);
227}

230/* The store whose next pass is due, or NULL. Every store carries its own due
231 * time, so no rotation is needed: one that has just run is not due again
232 * until its own interval has elapsed. */
233static oci_gc_store_t *
234oci_gc_due(void)
235{
236 ngx_uint_t i;
237
238 for (i = 0; i < oci_gc_store_count; i++) {
239 if ((ngx_msec_int_t) (ngx_current_msec - oci_gc_stores[i].due) >= 0) {
240 return &oci_gc_stores[i];
241 }
242 }
243 return NULL;
244}

247/* Timer callback: at most ONE store per tick, and never while a previous pass
248 * is still walking — a registry under GC should cost the disk one sweep, not
249 * as many as the timer can start. */
250static void
251oci_gc_tick(ngx_event_t *ev)
252{
253 oci_gc_store_t *s;
254
255 if (oci_gc_inflight == 0 && (s = oci_gc_due()) != NULL) {
256 s->due = ngx_current_msec + s->interval;
257#if (NGX_THREADS)
258 if (oci_gc_offload(s, ev->log) != NGX_OK) {
259 oci_gc_inline(s, ev->log);
260 }
261#else
262 oci_gc_inline(s, ev->log);
263#endif
264 }
265 if (!ngx_exiting) {
266 ngx_add_timer(ev, brix_oci_gc_tick_ms());
267 }
268}

271ngx_msec_t
272brix_oci_gc_tick_ms(void)
273{
274 ngx_msec_t tick = 0;
275 ngx_uint_t i;
276
277 /* The tick is the shortest configured interval: a store asking to be
278 * swept every 2s cannot be served by a timer that wakes every minute,
279 * and a longer-interval store is held back by its own due time anyway. */
280 for (i = 0; i < oci_gc_store_count; i++) {
281 if (tick == 0 || oci_gc_stores[i].interval < tick) {
282 tick = oci_gc_stores[i].interval;
283 }
284 }
285 return tick < BRIX_OCI_GC_MIN_INTERVAL ? BRIX_OCI_GC_MIN_INTERVAL : tick;
286}

289void
290brix_oci_gc_arm_timer(ngx_cycle_t *cycle)
291{
292 /* Shared dummy connection (src/core/config/process_timers.c): nginx's
293 * --with-debug timer-expiry log reads ngx_event_ident(ev->data)->fd, so a
294 * connection-less timer must not leave ev->data NULL (worker SIGSEGV). */
295 extern ngx_connection_t brix_maint_timer_conn;
296 ngx_uint_t i;
297
298 if (ngx_worker != 0 || oci_gc_store_count == 0) {
299 return;
300 }
301 /* A pass costs a full walk of the store, so the first one waits out an
302 * interval rather than landing on a worker that has just started. */
303 for (i = 0; i < oci_gc_store_count; i++) {
304 oci_gc_stores[i].due = ngx_current_msec + oci_gc_stores[i].interval;
305 }
306 oci_gc_timer.handler = oci_gc_tick;
307 oci_gc_timer.data = &brix_maint_timer_conn;
308 oci_gc_timer.log = cycle->log;
309 oci_gc_timer.cancelable = 1; /* never delay a graceful shutdown */
310 ngx_add_timer(&oci_gc_timer, brix_oci_gc_tick_ms());
311}
