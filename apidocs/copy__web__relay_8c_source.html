4static void
5copy_retry_pause(const brix_copy_opts *opts, int attempt,
6 const char *src, const brix_status *status)
7{
8 int backoff = 1 << (attempt < 5 ? attempt : 5);
9 unsigned half_ms;
10 unsigned wait_ms;
11 struct timespec delay;
12
13 if (backoff > 30) {
14 backoff = 30;
15 }
16 half_ms = (unsigned) backoff * 500u;
17 wait_ms = half_ms + brix_jitter_ms(half_ms + 1u);
18 if (opts == NULL || !opts->silent) {
19 fprintf(stderr, "xrdcp: %s failed (%s); retry %d/%d in %.1fs\n",
20 src, status->msg, attempt + 1, opts->retry_count,
21 wait_ms / 1000.0);
22 }
23 delay.tv_sec = wait_ms / 1000u;
24 delay.tv_nsec = (long) (wait_ms % 1000u) * 1000000L;
25 (void) nanosleep(&delay, NULL);
26}
