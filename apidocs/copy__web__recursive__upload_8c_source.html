8/* Shared state for one recursive web-upload walk (local tree → web collection).
9 * `base` is the dst URL's path with trailing slashes trimmed ("" for the root);
10 * the bucket/collection root, into which the source directory's CONTENTS go
11 * (symmetric with recursive download). */
12
13/* Build "<base>/<rel>" (or "/<rel>" when base is the root "") into out.
14 * 0 on success, -1 if it would not fit. */
15static int
16web_join(const char *base, const char *rel, char *out, size_t outsz)
17{
18 int w = snprintf(out, outsz, "%s/%s", base, rel);
19 return (w < 0 || (size_t) w >= outsz) ? -1 : 0;
20}

23/* WHAT: one directory entry's local path + upload-root-relative path.
24 * WHY: every walk step derives this pair once and threads it through the
25 * dir/file handlers; a struct keeps those helpers under the parameter cap.
26 * HOW: filled by walk_child_paths on the walk frame's stack (same footprint
27 * as the two buffers the walk loop previously held). */
28typedef struct {
29 char local[XRDC_PATH_MAX]; /* localdir/<name> */
30 char rel[XRDC_PATH_MAX]; /* path relative to the upload root */
31} walk_child_t;
32
33
34/* WHAT: build a directory entry's local and root-relative paths.
35 * WHY: isolates the overflow-guarded path joins (with the "" top-of-tree rel
36 * special case) from the walk's classify/dispatch loop.
37 * HOW: snprintf both joins with truncation checks. 0 on success, -1 if either
38 * would not fit (caller reports and skips the entry). */
39static int
40walk_child_paths(const char *localdir, const char *rel, const char *name,
41 walk_child_t *out)
42{
43 if ((size_t) snprintf(out->local, sizeof(out->local), "%s/%s",
44 localdir, name) >= sizeof(out->local)) {
45 return -1;
46 }
47 if (rel[0] == '\0') {
48 if ((size_t) snprintf(out->rel, sizeof(out->rel), "%s", name)
49 >= sizeof(out->rel)) {
50 return -1;
51 }
52 return 0;
53 }
54 if ((size_t) snprintf(out->rel, sizeof(out->rel), "%s/%s", rel, name)
55 >= sizeof(out->rel)) {
56 return -1;
57 }
58 return 0;
59}

62/* WHAT: handle one subdirectory during the upload walk: MKCOL it, then recurse.
63 * WHY: creating the remote collection top-down before descending keeps the
64 * child PUTs/MKCOLs from hitting 409 Conflict; S3 has no real dirs.
65 * HOW: MKCOL <base>/<rel> (davs/http only; skipped under --dry-run, which
66 * still recurses so files get printed); on MKCOL failure the subtree is
67 * skipped — its files would 409 — and `fail` is bumped. */
68static void
69walk_handle_dir(web_upload_ctx_t *c, const walk_child_t *ch)
70{
71 if (!c->endpoint->is_s3
72 && (c->opts == NULL || !c->opts->dry_run)) {
73 char rpath[XRDC_PATH_MAX * 2];
74 brix_status mst;
75 char proxybuf[512];
76 const char *pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
77
78 brix_status_clear(&mst);
79 if (web_join(c->base, ch->rel, rpath, sizeof(rpath)) != 0
80 || brix_webdav_mkcol(c->endpoint, rpath, c->bearer,
81 c->conn ? c->conn->verify_host : 1,
82 c->conn ? c->conn->ca_dir : NULL,
83 pcert, &mst) != 0) {
84 fprintf(stderr, "xrdcp: mkcol %s: %s\n", ch->rel, mst.msg);
85 c->failed++;
86 return; /* skip this subtree — its files would 409 */
87 }
88 }
89 copy_web_upload_walk(c, ch->local, ch->rel);
90}

93/* WHAT: handle one regular file during the upload walk: filter, then PUT.
94 * WHY: keeps the per-file filter/dry-run/URL-build/transfer/accounting chain
95 * out of the walk's directory-scan loop.
96 * HOW: apply --exclude/--include and --dry-run before any I/O; build the
97 * remote URL (overflow-guarded); PUT via copy_one_with_retry; bump `ok`
98 * (progress line unless silent) or `fail` (error line). */
99static void
100walk_handle_file(web_upload_ctx_t *c, const walk_child_t *ch)
101{
102 char rurl[XRDC_PATH_MAX * 2 + 320];
103 char rpath[XRDC_PATH_MAX * 2];
104 brix_status cst;
105
106 if (c->opts != NULL && !brix_copy_filter_match(c->opts, ch->rel)) {
107 return; /* filtered — skip, not a failure */
108 }
109 if (web_join(c->base, ch->rel, rpath, sizeof(rpath)) != 0
110 || (size_t) snprintf(rurl, sizeof(rurl), "%s://%s:%d%s",
111 c->scheme, c->endpoint->host,
112 c->endpoint->port, rpath)
113 >= sizeof(rurl)) {
114 fprintf(stderr, "xrdcp: remote path too long for %s\n", ch->rel);
115 c->failed++;
116 return;
117 }
118 if (c->opts != NULL && c->opts->dry_run) {
119 printf("[dry-run] copy %s -> %s\n", ch->local, rurl);
120 c->copied++;
121 return;
122 }
123 brix_status_clear(&cst);
124 if (copy_web_leaf_retry(ch->local, rurl, c->opts, c->conn, &cst) == 0) {
125 c->copied++;
126 if (c->opts == NULL || !c->opts->silent) {
127 fprintf(stderr, "[%zu] %s -> %s\n", c->copied + c->failed,
128 ch->local, rurl);
129 }
130 } else {
131 c->failed++;
132 fprintf(stderr, "xrdcp: %s: %s\n", rurl, cst.msg);
133 }
134}

137/* Recursively walk a local directory, MKCOL'ing each WebDAV collection (davs/http
138 * only — S3 keys are flat) and PUT'ing each regular file. `rel` is the path of the
139 * current directory relative to the upload root ("" at the top). Symlinks and
140 * special files are skipped (only real dirs + regular files are uploaded). */
141void
142copy_web_upload_walk(web_upload_ctx_t *c, const char *localdir,
143 const char *rel)
144{
145 DIR *d = opendir(localdir);
146 struct dirent *de;
147
148 if (d == NULL) {
149 fprintf(stderr, "xrdcp: cannot open %s: %s\n", localdir, strerror(errno));
150 c->failed++;
151 return;
152 }
153 while ((de = readdir(d)) != NULL) {
154 walk_child_t ch;
155 struct stat sb;
156
157 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0) {
158 continue;
159 }
160 if (walk_child_paths(localdir, rel, de->d_name, &ch) != 0) {
161 fprintf(stderr, "xrdcp: path too long under %s\n", localdir);
162 c->failed++;
163 continue;
164 }
165 if (lstat(ch.local, &sb) != 0) {
166 fprintf(stderr, "xrdcp: stat %s: %s\n", ch.local, strerror(errno));
167 c->failed++;
168 continue;
169 }
170 if (S_ISDIR(sb.st_mode)) {
171 walk_handle_dir(c, &ch);
172 } else if (S_ISREG(sb.st_mode)) {
173 walk_handle_file(c, &ch);
174 }
175 /* else: symlink / fifo / device — skip (not uploaded) */
176 }
177 closedir(d);
178}

202/* Recursively upload a local directory's CONTENTS into a web (davs/http/s3)
203 * collection: `xrdcp -r ./dir davs://h/coll/` → coll/<files-under-dir>. The wire
204 * has no recursive transfer op, so walk locally + MKCOL + per-file PUT. Returns
205 * 0 if every file uploaded, 1 otherwise. */
206int
207copy_web_recursive_upload(const char *localdir, const char *dst,
208 const brix_copy_opts *o, const brix_opts *co,
209 brix_status *status)
210{
211 brix_weburl u;
212 brix_copy_opts fo;
213 char base[XRDC_PATH_MAX];
214 web_upload_ctx_t c;
215 size_t blen;
216
217 if (brix_weburl_parse(dst, &u) != 0) {
218 fprintf(stderr, "xrdcp: bad web URL %s\n", dst);
219 brix_status_set(status, XRDC_EUSAGE, 0,
220 "bad web destination URL: %s", dst);
221 return -1;
222 }
223 /* Each file is a plain (non-recursive) copy so brix_copy's "no recursive web"
224 * guard doesn't trip. */
225 if (o == NULL) {
226 memset(&fo, 0, sizeof(fo));
227 } else {
228 fo = *o;
229 }
230 fo.recursive = 0;
231
232 snprintf(base, sizeof(base), "%s", u.path);
233 blen = strlen(base);
234 while (blen > 0 && base[blen - 1] == '/') { base[--blen] = '\0'; }
235
236 c.endpoint = &u;
237 c.base = base;
238 c.scheme = copy_web_scheme(u.proto);
239 c.bearer = (o != NULL && o->bearer != NULL) ? o->bearer : getenv("BEARER_TOKEN");
240 c.opts = &fo;
241 c.conn = co;
242 c.copied = 0;
243 c.failed = 0;
244
245 /* Ensure the destination collection itself exists (idempotent). Root ("")
246 * and S3 buckets need no MKCOL. Under --dry-run, skip to avoid creating
247 * the base collection on the remote. */
248 web_upload_create_base(&u, base, &c);
249
250 copy_web_upload_walk(&c, localdir, "");
251 if (o == NULL || !o->silent) {
252 fprintf(stderr, "xrdcp: %zu copied, %zu failed (recursive web upload)\n",
253 c.copied, c.failed);
254 }
255 if (c.failed != 0) {
256 brix_status_set(status, XRDC_ESOCK, 0,
257 "%zu recursive web upload(s) failed", c.failed);
258 return -1;
259 }
260 return 0;
261}
