1/*
2 * sd_http_redirect.c — the origin 3xx hop policy (phase-104 D1.4).
3 *
4 * WHAT: given a `Location:` from an origin redirect, decide the next hop: the
5 * host/port/tls/path to request, and the header block to request it
6 * with. Pure policy — this file performs no I/O and touches no curl
7 * handle; the caller (sd_http_select.c) does the perform.
8 * WHY: container registries answer a blob GET with a 302 to a CDN, so a fill
9 * that will not follow one cannot mirror DockerHub at all. Following is
10 * the easy half; the hard half is that the redirect is chosen by the
11 * RESPONSE, i.e. by whoever answered — and our request carries a bearer
12 * minted for the registry and, on a GSI leg, the end user's proxy
13 * certificate. Handing either to the host a 302 names is the classic
14 * token-leak bug class, and it is exactly what libcurl's FOLLOWLOCATION
15 * would do here: custom `Authorization:` set via CURLOPT_HTTPHEADER is
16 * NOT stripped across hosts the way libcurl's own auth is. So the hop is
17 * taken by hand, and the credential decision is made in one auditable
18 * place instead of being a curl default nobody re-reads.
19 * HOW: parse an absolute Location with the shared URL kernel (shared/oci/url.c
20 * — the same parser that vets a challenge realm, and the only one in the
21 * tree that refuses userinfo authorities and `..` path components);
22 * accept a root-relative Location as the same endpoint; refuse everything
23 * else. Same host AND port AND scheme ⇒ the header block rides along
24 * unchanged. Anything else is a different peer: `Authorization` is
25 * dropped from the block and the caller is told to drop the client cert.
26 */

36/* Resolve a Location into the hop's peer and path. 1 = the hop stays on the
37 * endpoint that sent it (root-relative, or an absolute URL naming the same
38 * host/port/scheme), 0 = it moves to another peer, -1 = the Location is not a
39 * form this client resolves.
40 *
41 * Anything that is neither root-relative nor an absolute http(s) URL — a bare
42 * relative path, a "//host/p" scheme-relative form, file:, gopher:, a data:
43 * URI — is refused rather than resolved: no registry emits one, and resolving
44 * is where the interesting escapes live. */
45static int
46sd_http_redirect_peer(const char *location, const sd_http_endpoint *from,
47 sd_http_redirect_t *hop)
48{
49 brix_oci_url_t url;
50 const char *query;
51
52 if (location[0] == '/') {
53 if (location[1] == '/') {
54 return -1;
55 }
56 if ((size_t) snprintf(hop->path, sizeof(hop->path), "%s", location)
57 >= sizeof(hop->path))
58 {
59 return -1;
60 }
61 snprintf(hop->host, sizeof(hop->host), "%s", from->host);
62 hop->port = from->port;
63 hop->tls = from->tls;
64 return 1;
65 }
66
67 if (brix_oci_url_parse(location, strlen(location), &url) != 0) {
68 return -1;
69 }
70 /* url.path is the prefix with no trailing slash and STOPS at '?', so the
71 * query is carried across by hand: a CDN blob URL is a SIGNED url — the
72 * whole authorization is `?Expires=…&Signature=…` — and a hop that drops
73 * it arrives unsigned and is refused, which reads back here as the origin
74 * denying the object. A registry CDN redirect always carries a path, and
75 * an empty one is "/". */
76 query = strchr(location, '?');
77 if ((size_t) snprintf(hop->path, sizeof(hop->path), "%s%s",
78 url.path[0] ? url.path : "/",
79 (query != NULL) ? query : "") >= sizeof(hop->path))
80 {
81 return -1;
82 }
83 snprintf(hop->host, sizeof(hop->host), "%s", url.host);
84 hop->port = url.port;
85 hop->tls = url.tls;
86
87 return (strcasecmp(url.host, from->host) == 0
88 && url.port == from->port
89 && url.tls == from->tls) ? 1 : 0;
90}

93/* Copy the header block, dropping every `Authorization:` line. Header blocks
94 * here are CRLF-joined and always CRLF-terminated (composed by
95 * sd_http_read.c/sd_http_fo_challenge_retry), so a line is [start, crlf+2). */
96static int
97sd_http_redirect_strip_auth(const char *hdrs, char *out, size_t cap)
98{
99 const char *p = hdrs;
100 size_t used = 0;
101
102 out[0] = '\0';
103 if (hdrs == NULL) {
104 return 0;
105 }
106
107 while (*p != '\0') {
108 const char *eol = strstr(p, "\r\n");
109 size_t len = (eol != NULL) ? (size_t) (eol - p) + 2 : strlen(p);
110
111 if (strncasecmp(p, "Authorization:", 14) != 0) {
112 if (used + len >= cap) {
113 return -1;
114 }
115 memcpy(out + used, p, len);
116 used += len;
117 out[used] = '\0';
118 }
119 if (eol == NULL) {
120 break;
121 }
122 p = eol + 2;
123 }
124 return 0;
125}

134int
135sd_http_redirect_next(const char *location, const sd_http_endpoint *from,
136 const char *extra_hdrs, sd_http_redirect_t *hop)
137{
138 int same_peer;
139
140 if (location == NULL || location[0] == '\0') {
141 return -1;
142 }
143
144 same_peer = sd_http_redirect_peer(location, from, hop);
145 if (same_peer < 0) {
146 return -1;
147 }
148
149 /* A cleartext hop off a TLS origin is a downgrade: the bearer would be
150 * dropped anyway, but the OBJECT would then arrive over a channel an
151 * on-path attacker can rewrite. Refuse rather than fill from it. */
152 if (from->tls && !hop->tls) {
153 return -1;
154 }
155
156 hop->carries_credential = same_peer;
157 if (same_peer) {
158 if ((size_t) snprintf(hop->hdrs, sizeof(hop->hdrs), "%s",
159 (extra_hdrs != NULL) ? extra_hdrs : "")
160 >= sizeof(hop->hdrs))
161 {
162 return -1;
163 }
164 return 0;
165 }
166
167 return sd_http_redirect_strip_auth(extra_hdrs, hop->hdrs,
168 sizeof(hop->hdrs));
169}
