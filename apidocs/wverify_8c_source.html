1/*
2 * wverify.c — self-computed write-integrity accumulator (see wverify.h).
3 *
4 * Offset-sorted extent list; each update CRC-32's its buffer and coalesces with
5 * any extent it is now adjacent to, combining CRCs with zlib crc32_combine. A
6 * complete write leaves exactly one extent {0, size, crc}; anything else (gap,
7 * overlap, degraded) makes brix_wverify_expected fail closed.
8 *
9 * Phase-88 W3: a SHA-256 rides the IN-ORDER prefix — extents that land exactly
10 * at the current prefix end feed an incremental EVP digest; the first
11 * out-of-order arrival kills the SHA (no combine exists) while the CRC keeps
12 * its arrival-order independence.
13 */
14#include "core/compat/wverify.h"
15
16#include <openssl/evp.h>
17#include <stdlib.h>
18#include <string.h>
19#include <zlib.h>

21/* Cap the live extent count. An in-order stream stays at 1; an out-of-order
22 * writer (MODE E) holds at most one extent per not-yet-adjacent stream, far
23 * below this. Exceeding it marks the accumulator degraded (fails closed) rather
24 * than growing unbounded on a pathological arrival pattern. */
25#define BRIX_WVERIFY_MAX_EXTENTS 4096
26
27typedef struct {
28 off_t lo;
29 off_t hi; /* exclusive */
30 uint32_t crc; /* CRC-32 of bytes [lo, hi) */
31} wv_extent_t;
32
33struct brix_wverify_s {
34 wv_extent_t *ext;
35 size_t n;
36 size_t cap;
37 int degraded; /* an update failed; expected() must fail closed */
38 EVP_MD_CTX *sha; /* W3: incremental SHA-256 of the in-order prefix
39 * (NULL once dead — OOM or out-of-order arrival) */
40 off_t sha_off; /* prefix end the next in-order extent must hit */
41};

43brix_wverify_t *
44brix_wverify_begin(void)
45{
46 brix_wverify_t *w = calloc(1, sizeof(*w));
47
48 if (w == NULL) {
49 return NULL; /* OOM propagates to the caller */
50 }
51 /* A dead SHA never degrades the CRC contract — it only forfeits the
52 * cryptographic identity (callers fall back to byte-verified dedup). */
53 w->sha = EVP_MD_CTX_new();
54 if (w->sha != NULL
55 && EVP_DigestInit_ex(w->sha, EVP_sha256(), NULL) != 1)
56 {
57 EVP_MD_CTX_free(w->sha);
58 w->sha = NULL;
59 }
60 return w;
61}

74/* wv_sha_feed — advance the in-order SHA prefix, or kill the SHA on the first
75 * out-of-order arrival / digest failure (the CRC path is untouched). */
76static void
77wv_sha_feed(brix_wverify_t *w, const void *buf, off_t off, size_t len)
78{
79 if (w->sha == NULL) {
80 return;
81 }
82 if (off != w->sha_off
83 || EVP_DigestUpdate(w->sha, buf, len) != 1)
84 {
85 EVP_MD_CTX_free(w->sha);
86 w->sha = NULL;
87 return;
88 }
89 w->sha_off = off + (off_t) len;
90}

92/* Ensure room for one more extent. 0 / -1 (cap or OOM). */
93static int
94wv_reserve(brix_wverify_t *w)
95{
96 size_t want;
97 wv_extent_t *grown;
98
99 if (w->n < w->cap) {
100 return 0;
101 }
102 if (w->n >= BRIX_WVERIFY_MAX_EXTENTS) {
103 return -1;
104 }
105 want = w->cap ? w->cap * 2 : 16;
106 if (want > BRIX_WVERIFY_MAX_EXTENTS) {
107 want = BRIX_WVERIFY_MAX_EXTENTS;
108 }
109 grown = realloc(w->ext, want * sizeof(*grown));
110 if (grown == NULL) {
111 return -1;
112 }
113 w->ext = grown;
114 w->cap = want;
115 return 0;
116}

118/* Merge extent i with i+1 when they abut (ext[i].hi == ext[i+1].lo). */
119static void
120wv_merge_at(brix_wverify_t *w, size_t i)
121{
122 wv_extent_t *a = &w->ext[i];
123 wv_extent_t *b = &w->ext[i + 1];
124
125 a->crc = (uint32_t) crc32_combine((uLong) a->crc, (uLong) b->crc,
126 (z_off_t) (b->hi - b->lo));
127 a->hi = b->hi;
128 memmove(b, b + 1, (w->n - i - 2) * sizeof(*b));
129 w->n--;
130}

132/*
133 * WHAT: Validate one write extent and calculate its exclusive upper bound.
134 * WHY: Invalid updates must consistently degrade the accumulator fail-closed.
135 * HOW: Check pointers, emptiness, and overflow before forming off + len.
136 */
137static int
138wv_update_bounds(brix_wverify_t *w, const void *buf, off_t off, size_t len,
139 off_t *hi)
140{
141 if (w == NULL || off < 0 || (buf == NULL && len != 0)) {
142 if (w != NULL)
143 w->degraded = 1;
144 return -1;
145 }
146 if (len == 0) {
147 *hi = off;
148 return 1;
149 }
150 if ((off_t) len < 0
151 || off > (off_t) (((uint64_t) 1 << 62)) - (off_t) len)
152 {
153 w->degraded = 1;
154 return -1;
155 }
156 *hi = off + (off_t) len;
157 return 0;
158}

160/*
161 * WHAT: Find the sorted insertion point for a non-overlapping write extent.
162 * WHY: Duplicate or intersecting bytes make the expected checksum ambiguous.
163 * HOW: Scan by lower bound and reject intersection with either neighbour.
164 */
165static int
166wv_insertion_point(brix_wverify_t *w, off_t off, off_t hi, size_t *position)
167{
168 size_t pos;
169
170 for (pos = 0; pos < w->n && w->ext[pos].lo < off; pos++) {
171 /* scan */
172 }
173 if (pos > 0 && w->ext[pos - 1].hi > off) {
174 w->degraded = 1;
175 return -1;
176 }
177 if (pos < w->n && w->ext[pos].lo < hi) {
178 w->degraded = 1;
179 return -1;
180 }
181 *position = pos;
182 return 0;
183}

185int
186brix_wverify_update(brix_wverify_t *w, const void *buf, off_t off, size_t len)
187{
188 off_t hi;
189 size_t pos;
190 int bounds;
191
192 bounds = wv_update_bounds(w, buf, off, len, &hi);
193 if (bounds != 0)
194 return bounds < 0 ? -1 : 0;
195
196 /* Insertion point: first extent whose lo >= off. Reject any overlap with the
197 * neighbours on either side (a fresh write must not restate committed bytes). */
198 if (wv_insertion_point(w, off, hi, &pos) != 0)
199 return -1;
200
201 if (wv_reserve(w) != 0) {
202 w->degraded = 1;
203 return -1;
204 }
205 memmove(&w->ext[pos + 1], &w->ext[pos], (w->n - pos) * sizeof(w->ext[0]));
206 w->ext[pos].lo = off;
207 w->ext[pos].hi = hi;
208 w->ext[pos].crc = (uint32_t) crc32(crc32(0L, Z_NULL, 0),
209 (const Bytef *) buf, (uInt) len);
210 w->n++;
211 wv_sha_feed(w, buf, off, len); /* W3: after acceptance, never on reject */
212
213 /* Coalesce with an abutting right neighbour, then the left, so the list stays
214 * minimal (a contiguous stream collapses to a single extent). */
215 if (pos + 1 < w->n && w->ext[pos].hi == w->ext[pos + 1].lo) {
216 wv_merge_at(w, pos);
217 }
218 if (pos > 0 && w->ext[pos - 1].hi == w->ext[pos].lo) {
219 wv_merge_at(w, pos - 1);
220 }
221
222 return 0;
223}

239int
240brix_wverify_expected_sha256(const brix_wverify_t *w, unsigned char out32[32])
241{
242 EVP_MD_CTX *fin;
243 unsigned int mdlen = 0;
244 int ok;
245
246 if (w == NULL || out32 == NULL || w->sha == NULL) {
247 return -1;
248 }
249 /* Same completeness contract as expected(), plus: the in-order prefix must
250 * BE the whole single run (an appending stream that covered everything). */
251 if (w->degraded || w->n != 1 || w->ext[0].lo != 0
252 || w->sha_off != w->ext[0].hi)
253 {
254 return -1;
255 }
256 /* Finalise a copy so the accumulator survives (close paths may consult the
257 * CRC afterwards). */
258 fin = EVP_MD_CTX_new();
259 if (fin == NULL) {
260 return -1;
261 }
262 ok = EVP_MD_CTX_copy_ex(fin, w->sha) == 1
263 && EVP_DigestFinal_ex(fin, out32, &mdlen) == 1
264 && mdlen == 32;
265 EVP_MD_CTX_free(fin);
266 return ok ? 0 : -1;
267}
