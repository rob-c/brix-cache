1/*
2 * upstream/tls.c — Outbound TLS upgrade for transparent proxy upstream redirector
3 *
4 * WHAT:
5 * Two functions for upgrading the upstream TCP connection to TLS after a
6 * remote XRootD server signals kXR_gotoTLS in its kXR_protocol response:
7 *
8 * 1. brix_upstream_start_tls()
9 * - Wraps up->conn in SSL
10 * - Sets SNI hostname
11 * - Starts TLS handshake
12 *
13 * 2. brix_upstream_tls_handshake_done()
14 * - nginx ssl->handler callback
15 * - Resends kXR_login over TLS once handshake completes
16 *
17 * WHY:
18 * Transparent proxy mode requires the upstream connection to support TLS
19 * when backend servers demand it (kXR_gotoTLS).
20 * The client's kXR_login must be sent over the upgraded TLS channel,
21 * not cleartext TCP.
22 * This file handles the transition:
23 * raw TCP → SSL-wrapped connection → re-login over TLS
24 *
25 * HOW:
26 * start_tls():
27 * - Calls ngx_ssl_create_connection() with upstream TLS context
28 * - Sets SNI (directive override wins, falls back to configured host)
29 * - Assigns ssl->handler callback
30 * - Calls ngx_ssl_handshake()
31 * handshake_done() on completion:
32 * - Checks ctx validity and handshaked flag
33 * - Restores normal event handlers
34 * - Allocates fresh kXR_login frame via brix_upstream_build_login()
35 * - Flushes over TLS connection
36 * - Arms read event for login response
37 */

45/*
46 * brix_upstream_start_tls() wraps the upstream TCP connection in SSL and initiates the TLS handshake.
47 *
48 * WHAT:
49 * Wraps the upstream TCP connection in SSL and initiates the TLS handshake.
50 *
51 * WHY:
52 * When backend XRootD server signals kXR_gotoTLS, the existing cleartext TCP
53 * connection must be upgraded to TLS before sending kXR_login.
54 * The SSL context comes from conf->upstream_tls_ctx;
55 * SNI is set from either an explicit override directive or the configured
56 * upstream host name.
57 *
58 * HOW:
59 * 1. Sets bs_phase = XRD_UP_BS_TLS
60 * 2. Calls ngx_ssl_create_connection() with NGX_SSL_BUFFER | NGX_SSL_CLIENT flags
61 * 3. Sets SNI via SSL_set_tlsext_host_name()
62 * (conf->upstream_tls_name wins if non-empty, falls back to conf->upstream_host)
63 * 4. Assigns ssl->handler callback
64 * 5. Calls ngx_ssl_handshake()
65 * 6. If handshake completes synchronously (unlikely), immediately fires
66 * brix_upstream_tls_handshake_done()
67 *
68 * RETURNS:
69 * NGX_OK on success, NGX_ERROR on SSL creation failure
70 */
71ngx_int_t
72brix_outbound_start_tls(ngx_ssl_t *ssl_ctx, ngx_connection_t *c,
73 const char *sni, void (*handler)(ngx_connection_t *c))
74{
75 if (ngx_ssl_create_connection(ssl_ctx, c,
76 NGX_SSL_BUFFER | NGX_SSL_CLIENT)
77 != NGX_OK)
78 {
79 return NGX_ERROR;
80 }
81
82 SSL_set_tlsext_host_name(c->ssl->connection, sni);
83 c->ssl->handler = handler;
84
85 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
86 "brix: outbound starting TLS handshake (sni=%s)", sni);
87
88 if (ngx_ssl_handshake(c) != NGX_AGAIN) {
89 /* Completed synchronously (unlikely but handle it). */
90 handler(c);
91 }
92
93 return NGX_OK;
94}

113/*
114 * brix_upstream_tls_handshake_done() is the nginx ssl->handler callback that fires once the TLS handshake completes.
115 *
116 * WHAT:
117 * nginx ssl->handler callback fired once TLS handshake completes
118 * (success or failure).
119 * On success: restores normal event handlers, allocates fresh kXR_login frame,
120 * flushes it over the now-TLS connection.
121 *
122 * WHY:
123 * The SSL handshake replaces the normal upstream read/write event handlers.
124 * After completion, those handlers must be restored so subsequent
125 * request/response traffic flows normally.
126 * A fresh kXR_login is needed because credentials sent over cleartext before
127 * TLS upgrade are no longer valid on the encrypted channel.
128 *
129 * HOW:
130 * 1. Checks ctx validity and destroyed flag → calls cleanup if invalid
131 * 2. Checks uconn->ssl->handshaked → calls abort if failed
132 * 3. On success:
133 * a) Restores read handler to brix_upstream_read_handler
134 * b) Restores write handler to brix_upstream_write_handler
135 * c) Allocates ClientLoginRequest from pool via ngx_palloc()
136 * d) Builds login frame via brix_upstream_build_login()
137 * e) Sets wbuf/wbuf_len/wbuf_pos for flush → calls brix_upstream_flush()
138 * f) If partial write: arms write event
139 * g) If fully written: arms read event to wait for login response
140 * 4. Sets bs_phase = XRD_UP_BS_LOGIN
141 */
142static void
143brix_upstream_tls_handshake_done(ngx_connection_t *uconn)
144{
145 brix_upstream_t *up = uconn->data;
146 brix_ctx_t *ctx = up->client_ctx;
147 ClientLoginRequest *req;
148
149 if (ctx == NULL || ctx->destroyed) {
150 brix_upstream_cleanup(up);
151 return;
152 }
153
154 if (!uconn->ssl->handshaked) {
155 brix_upstream_abort(up, "upstream: TLS handshake failed");
156 return;
157 }
158
159 /* Belt-and-braces: when verification is enabled on the CTX a bad chain/host
160 * already fails the handshake above; this makes the trust decision explicit
161 * and survives refactors. Harmless (X509_V_OK) when verification is off. */
162 if (SSL_get_verify_result(uconn->ssl->connection) != X509_V_OK) {
163 brix_upstream_abort(up, "upstream: TLS peer verification failed");
164 return;
165 }
166
167 ngx_log_debug0(NGX_LOG_DEBUG_STREAM, up->client_conn->log, 0,
168 "brix: upstream TLS handshake done; resending kXR_login");
169
170 /* Restore normal event handlers (TLS handshake replaces them). */
171 uconn->read->handler = brix_upstream_read_handler;
172 uconn->write->handler = brix_upstream_write_handler;
173
174 /* Build a fresh kXR_login for the TLS channel. */
175 req = ngx_palloc(uconn->pool, sizeof(*req));
176 if (req == NULL) {
177 brix_upstream_abort(up, "upstream: TLS re-login OOM");
178 return;
179 }
180 brix_upstream_build_login(req);
181
182 up->wbuf = (u_char *) req;
183 up->wbuf_len = sizeof(*req);
184 up->wbuf_pos = 0;
185 up->bs_phase = XRD_UP_BS_LOGIN;
186
187 /* Reset response accumulator for the login response. */
188 up->rhdr_pos = 0;
189 up->resp_dlen = 0;
190 up->resp_body = NULL;
191 up->resp_body_pos = 0;
192
193 if (brix_upstream_flush(up) == NGX_ERROR) {
194 brix_upstream_abort(up, "upstream: TLS re-login send failed");
195 return;
196 }
197
198 if (up->wbuf_pos < up->wbuf_len) {
199 /* Not all bytes written yet; write handler will finish and arm read. */
200 if (ngx_handle_write_event(uconn->write, 0) != NGX_OK) {
201 brix_upstream_abort(up, "upstream: TLS write event arm failed");
202 }
203 return;
204 }
205
206 /* All sent; wait for login response. */
207 if (ngx_handle_read_event(uconn->read, 0) != NGX_OK) {
208 brix_upstream_abort(up, "upstream: TLS read event arm failed");
209 }
210}
