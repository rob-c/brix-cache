1/*
2 * http_variables.c — registration and get_handlers for the $brix_* HTTP
3 * variable surface (phase 106 W1).
4 *
5 * WHAT: brix_http_add_variables() registers every $brix_* variable the HTTP
6 * planes expose, and implements the handlers whose state is not owned by
7 * a single protocol.
8 *
9 * WHY: See http_variables.h. Registration is owned by the COMMON module so
10 * one registration serves every HTTP protocol and the variable's
11 * existence does not depend on which protocol module happens to be
12 * loaded.
13 *
14 * HOW: Handlers follow three rules, which apply to every variable added here
15 * (phase 106, "variable-handler trap"):
16 *
17 * 1. Never allocate from a pool that may already be gone. A handler
18 * can run in the log phase, so every value below is either a static
19 * string or memory owned by the request that outlives logging.
20 * 2. Always tolerate a missing request ctx. A request may never have
21 * reached brix at all (rejected by `deny`, served by another
22 * module), in which case the handler reports "-" rather than
23 * dereferencing NULL. ngx_http_brix_cvmfs_ctx_t may legitimately be
24 * absent even inside a cvmfs location.
25 * 3. State the cacheability. Anything derived from the connection or
26 * the matched location is per-request cacheable; anything derived
27 * from the data plane changes as the request proceeds and is marked
28 * NOCACHEABLE.
29 *
30 * SECURITY: no variable here may expose credential material — a token,
31 * a macaroon, a private key, or a raw Authorization value. Variables are
32 * an exfiltration surface because an operator can log them or copy them
33 * into a proxied header. Identity variables expose the SUBJECT, never
34 * the credential that proved it. ($brix_delegated_cred, registered
35 * elsewhere, predates this rule and is the single reviewed exception.)
36 */

59/*
60 * brix_var_set_static — hand nginx a constant string as a variable value.
61 *
62 * Every handler in this file resolves to a static string, so this is the one
63 * place that fills the value struct. `no_cacheable` is the caller's decision
64 * and is passed in rather than assumed.
65 */
66ngx_int_t
67brix_var_set_static(ngx_http_variable_value_t *v, const char *s,
68 ngx_uint_t no_cacheable)
69{
70 v->len = (unsigned) ngx_strlen(s);
71 v->valid = 1;
72 v->no_cacheable = no_cacheable ? 1 : 0;
73 v->not_found = 0;
74 v->data = (u_char *) s;
75 return NGX_OK;
76}

79/*
80 * cvmfs_cache_status — translate the cvmfs plane's own disposition into the
81 * shared vocabulary.
82 *
83 * The cvmfs plane tracked a cache disposition (T16) before the shared surface
84 * existed, using its own spelling, and it keeps that spelling internally:
85 * ctx->cache_status is read by cvmfs_var_origin() and the cvmfs metrics.
86 * Mapping here rather than renaming the plane's enum keeps the translation in
87 * ONE place, so a new disposition is one arm to add, not a plane-wide edit.
88 * FILL is nginx's MISS: went to the origin and populated. (Phase 112 removed
89 * the plane's own $cvmfs_cache variable; this is now the only reader.)
90 */
91static brix_cache_status_e
92cvmfs_cache_status(ngx_uint_t cvmfs_status)
93{
94 switch (cvmfs_status) {
95 case BRIX_CVMFS_CACHE_HIT: return BRIX_CACHE_STATUS_HIT;
96 case BRIX_CVMFS_CACHE_FILL: return BRIX_CACHE_STATUS_MISS;
97 case BRIX_CVMFS_CACHE_NEG: return BRIX_CACHE_STATUS_NEGHIT;
98 default: return BRIX_CACHE_STATUS_NONE;
99 }
100}

103/*
104 * $brix_cache_status — HIT / MISS / BYPASS / NEGHIT / "-".
105 *
106 * Reports the cache disposition in nginx's own $upstream_cache_status
107 * vocabulary so existing dashboards and log parsers work unchanged.
108 *
109 * Today only the cvmfs plane tracks a per-request disposition, so other planes
110 * report "-" (no cache decision was reached). That is honest rather than
111 * misleading: "-" cannot be mistaken for a MISS, so a hit-rate computed from
112 * this variable is never silently wrong — it is simply empty for planes that
113 * do not yet report. Extending a plane means giving it a disposition and
114 * adding its arm here; the variable's name, vocabulary and log_format do not
115 * change when that happens, which is the whole point of registering it now.
116 *
117 * NOCACHEABLE: the disposition is a data-plane outcome and is not known until
118 * the request has been served.
119 */
120/*
121 * oci_rpm_cache_status — the oci/rpm outcome enums share one layout
122 * (hit/fill/local/refused/error) and one mapping: "local" is a HIT (served
123 * from the local store with no origin contact); "refused"/"error" are not
124 * cache dispositions at all and report the sentinel.
125 */
126static brix_cache_status_e
127oci_rpm_cache_status(ngx_uint_t disp)
128{
129 switch (disp) {
130 case 0: /* HIT */ return BRIX_CACHE_STATUS_HIT;
131 case 1: /* FILL */ return BRIX_CACHE_STATUS_MISS;
132 case 2: /* LOCAL */ return BRIX_CACHE_STATUS_HIT;
133 default: return BRIX_CACHE_STATUS_NONE;
134 }
135}

138static brix_cache_status_e
139brix_request_cache_status(ngx_http_request_t *r)
140{
141 ngx_http_brix_cvmfs_ctx_t *cctx;
142 ngx_http_brix_oci_ctx_t *octx;
143 ngx_http_brix_rpm_ctx_t *rctx;
144 brix_io_monitor_t *m;
145
146 /* phase-110 W1: the data planes (WebDAV/S3) record their disposition on
147 * the request's I/O monitor at the VFS cache decision
148 * (brix_vfs_observe_cache_result). First arm, so a cache-enabled export
149 * answers HIT/MISS/BYPASS here; the plane-specific probes below stay for
150 * the planes that keep their own disposition. */
151 m = ngx_http_get_module_ctx(r, ngx_http_brix_common_module);
152 if (m != NULL && m->cache != BRIX_CACHE_STATUS_NONE) {
153 return m->cache;
154 }
155
156 cctx = ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
157 if (cctx != NULL) {
158 return cvmfs_cache_status(cctx->cache_status);
159 }
160 octx = ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
161 if (octx != NULL) {
162 return oci_rpm_cache_status((ngx_uint_t) octx->disp);
163 }
164 rctx = ngx_http_get_module_ctx(r, ngx_http_brix_rpm_module);
165 if (rctx != NULL) {
166 return oci_rpm_cache_status((ngx_uint_t) rctx->disp);
167 }
168 return BRIX_CACHE_STATUS_NONE;
169}

182/*
183 * $brix_tls — "on" when the request arrived over TLS, "off" otherwise.
184 *
185 * nginx's own $https covers the plain case, but brix serves planes where TLS
186 * can be established by brix itself (the stream-side upgrade has no $https
187 * equivalent), and a single spelling that means the same thing on every plane
188 * is what lets one log_format serve them all.
189 *
190 * Per-request cacheable: a connection cannot change transport mid-request.
191 */
192static ngx_int_t
193brix_var_tls(ngx_http_request_t *r, ngx_http_variable_value_t *v,
194 uintptr_t data)
195{
196 (void) data;
197 return brix_var_set_static(v,
198 brix_http_request_is_tls(r) ? "on" : "off", 0);
199}

250ngx_int_t
251brix_http_add_variables(ngx_conf_t *cf)
252{
253 ngx_http_variable_t *v, *nv;
254
255 for (v = brix_http_variables; v->name.len; v++) {
256 nv = ngx_http_add_variable(cf, &v->name, v->flags);
257 if (nv == NULL) {
258 return NGX_ERROR;
259 }
260 nv->get_handler = v->get_handler;
261 nv->data = v->data;
262 }
263
264 return NGX_OK;
265}
