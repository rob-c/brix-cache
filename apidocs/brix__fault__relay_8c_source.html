32/* Blocking connect to host:port (best-effort, first address that works). */
33int
34dial(const char *host, int port)
35{
36 struct addrinfo hints, *res = NULL, *ai;
37 char portstr[16];
38 int fd = -1;
39
40 memset(&hints, 0, sizeof(hints));
41 hints.ai_family = AF_UNSPEC;
42 hints.ai_socktype = SOCK_STREAM;
43 snprintf(portstr, sizeof(portstr), "%d", port);
44 if (getaddrinfo(host, portstr, &hints, &res) != 0) {
45 return -1;
46 }
47 for (ai = res; ai != NULL; ai = ai->ai_next) {
48 fd = socket(ai->ai_family, ai->ai_socktype, ai->ai_protocol);
49 if (fd < 0) {
50 continue;
51 }
52 if (connect(fd, ai->ai_addr, ai->ai_addrlen) == 0) {
53 break;
54 }
55 close(fd);
56 fd = -1;
57 }
58 freeaddrinfo(res);
59 return fd;
60}

62int
63write_all(int fd, const char *buf, ssize_t n)
64{
65 ssize_t off = 0;
66 while (off < n) {
67 ssize_t w = write(fd, buf + off, (size_t) (n - off));
68 if (w < 0) {
69 if (errno == EINTR) {
70 continue;
71 }
72 return -1;
73 }
74 off += w;
75 }
76 return 0;

111/* Clamp the outgoing segment to the chunk `piece` and, if truncation is armed,
112 * to the exact remaining distance to the cut. Returns the clamped length, or -1
113 * when the truncation point is already reached (caller must sever). */
114ssize_t
115fault_clamp_seg(ssize_t seg, int piece, long trunc, unsigned long conn_ctr)
116{
117 if (seg > piece) {
118 seg = piece;
119 }
120 /* Cap at the truncation boundary so the cut is exact rather than firing
121 * only after the whole (possibly 64 KiB) read is delivered. */
122 if (trunc > 0) {
123 long remaining = trunc - (long) conn_ctr;
124 if (remaining <= 0) {
125 return -1;
126 }
127 if (seg > remaining) {
128 seg = remaining;
129 }
130 }
131 return seg;
132}

134/* Apply direction `L`'s pacing/jitter/reorder delays before a segment write.
135 * `sent` is the bytes already forwarded this direction, so a token-bucket
136 * `burst` budget can be spent up front (no pacing until the budget is used). */
137void
138fault_delays(ssize_t seg, unsigned *seed, const lever_t *L, unsigned long sent)
139{
140 if (L->jitter_ms > 0) {
141 long ms;
142 if (L->lat_dist == 1) {
143 /* Normal(mean=jitter_ms, sigma=lat_sigma_ms) via Irwin-Hall (sum of
144 * 12 uniforms - 6 ≈ N(0,1)); no libm. Clamped to [0, ...]. */
145 double u = 0.0;
146 for (int j = 0; j < 12; j++) {
147 u += (double) (rand_r(seed) % 100000u) / 100000.0;
148 }
149 double z = u - 6.0;
150 ms = (long) ((double) L->jitter_ms + z * (double) L->lat_sigma_ms);
151 if (ms < 0) { ms = 0; }
152 } else {
153 ms = rand_r(seed) % (unsigned) (L->jitter_ms + 1);
154 }
155 usleep((useconds_t) ms * 1000);
156 }
157 if (L->reorder_ppm > 0 && (int) (rand_r(seed) % 1000000u) < L->reorder_ppm) {
158 usleep((useconds_t) L->reorder_ms * 1000);
159 }
160 /* rate: pace to the ceiling, but spend any `burst` credit up front — while
161 * the bytes forwarded so far are within the burst budget, forward unpaced. */
162 if (L->rate_kbps > 0 && (L->burst_bytes <= 0 || (long) sent >= L->burst_bytes)) {
163 usleep((useconds_t) ((long long) seg * 1000000
164 / ((long long) L->rate_kbps * 1024)));
165 }
166}

168/* Flip random bits in buf[off, off+seg) at the configured per-byte rate. */
169void
170fault_corrupt(char *buf, ssize_t off, ssize_t seg, unsigned *seed, int cor)
171{
172 if (cor <= 0) {
173 return;
174 }
175 long flips = 0;
176 for (ssize_t k = 0; k < seg; k++) {
177 if ((int) (rand_r(seed) % 1000000u) < cor) {
178 buf[off + k] ^= (char) (1u << (rand_r(seed) % 8u));
179 CBUMP(corrupt, 1);
180 flips++;
181 }
182 }
183 /* One batched corrupt event per segment — the COUNT only, never a byte. */
184 if (flips > 0) {
185 brix_fp_event(t_conn_id, t_fwd_up ? "up" : "down", "corrupt", NULL,
186 "count", flips);
187 }
188}

190/* Deliver one segment starting at buf+off, applying `L`'s active faults. On
191 * success writes the delivered length to *wrote and returns 0; returns -1 when
192 * the stream should be severed (truncate cut, lossy drop, write error) or
193 * silently dropped (block/epoch change). */
194int
195forward_segment(int to, char *buf, ssize_t off, ssize_t n, unsigned epoch,
196 const lever_t *L, unsigned *seed, unsigned long *conn_ctr,
197 unsigned long *glob_ctr, int piece, ssize_t *wrote)
198{
199 ssize_t seg = fault_clamp_seg(n - off, piece, L->truncate_at, *conn_ctr);
200 if (seg < 0) {
201 /* Emit a discrete truncate event (with the exact cut offset) in addition
202 * to the sever that follows — no payload bytes, only the boundary. */
203 brix_fp_event(t_conn_id, t_fwd_up ? "up" : "down", "truncate", NULL, "at", L->truncate_at);
204 return fault_sever("truncate");
205 }
206
207 fault_delays(seg, seed, L, *conn_ctr);
208
209 if (L->lossy_ppm > 0 && (int) (rand_r(seed) % 1000000u) < L->lossy_ppm) {
210 return fault_sever("lossy"); /* application-visible "loss" = sever the stream */
211 }
212 if (g_blocked || g_drop_epoch != epoch) {
213 return -1;
214 }
215
216 fault_corrupt(buf, off, seg, seed, L->corrupt_ppm);
217 global_rate_gate(seg);
218
219 if (write_all(to, buf + off, seg) != 0) {
220 return -1;
221 }
222 if (L->dup_ppm > 0 && (int) (rand_r(seed) % 1000000u) < L->dup_ppm) {
223 (void) write_all(to, buf + off, seg); /* duplicate delivery */
224 CBUMP(dups, 1);
225 }
226
227 CBUMP2(conn_ctr, glob_ctr, seg);
228 *wrote = seg;
229
230 if (L->truncate_at > 0 && (long) *conn_ctr >= L->truncate_at) {
231 brix_fp_event(t_conn_id, t_fwd_up ? "up" : "down", "truncate", NULL, "at", L->truncate_at);
232 return fault_sever("truncate"); /* deterministic mid-transfer cut */
233 }
234 return 0;
235}

237/*
238 * Forward n bytes to `to`, applying direction `L`'s active fault levers. Bytes
239 * may be mutated in place (corruption). Returns 0 on success, -1 if the
240 * connection should be severed (write error, a lossy drop, or a truncate cut).
241 * `seed` is this thread's private rand_r() state; `conn_ctr` the per-connection
242 * per-direction running byte total (for truncate-at); `glob_ctr` the process-wide
243 * traffic counter for this direction.
244 */
245int
246forward_faulted(int to, char *buf, ssize_t n, unsigned epoch, volatile lever_t *L,
247 unsigned *seed, unsigned long *conn_ctr, unsigned long *glob_ctr)
248{
249 /* Snapshot the levers once so a mid-buffer control-plane change can't split
250 * this read across two fault configurations (matches the original). */
251 lever_t snap = *L;
252 /* Fold any named toxics for this direction onto the snapshot (never the live
253 * levers). The lock-free gate keeps the zero-toxic path free of the lock. */
254 int is_up = (L == &g_up);
255 if (fp_toxic_active(is_up)) {
256 fp_toxic_compose(is_up, &snap);
257 }
258 int piece = (snap.drip_bytes > 0) ? snap.drip_bytes
259 : (snap.chunk_bytes > 0 ? snap.chunk_bytes : (int) n);
260 ssize_t off = 0;
261
262 if (piece <= 0) {
263 piece = (int) n; /* n >= 1 (read returned >0) */
264 }
265 if (snap.latency_ms > 0) {
266 usleep((useconds_t) snap.latency_ms * 1000);
267 }
268
269 while (off < n) {
270 ssize_t wrote = 0;
271 if (forward_segment(to, buf, off, n, epoch, &snap, seed,
272 conn_ctr, glob_ctr, piece, &wrote) != 0) {
273 return -1;
274 }
275 off += wrote;
276 if (snap.drip_bytes > 0 && off < n) {
277 usleep((useconds_t) snap.drip_ms * 1000);
278 }
279 }
280 return 0;
281}

304/* Pre-dial dispositions that answer the client without ever reaching upstream:
305 * the fail-nth sever and the hang/black-hole hold. Returns 1 if the connection
306 * was handled (closed + CDEC), 0 to proceed with dialling upstream. */
307int
308relay_predial(int cfd, unsigned epoch, unsigned long conn_id)
309{
310 /* fail-nth: sever exactly the Nth accepted connection, then pass the rest. */
311 if (g_fail_nth > 0 && conn_id == (unsigned long) g_fail_nth) {
312 CBUMP(severs, 1);
313 sever(cfd, -1, g_abortive);
314 CDEC(active);
315 return 1;
316 }
317
318 /* hang / black hole: accept but never relay — hold the client open. */
319 if (g_hang) {
320 struct pollfd hp = { cfd, POLLIN, 0 };
321 while (g_hang && g_drop_epoch == epoch && !g_blocked) {
322 if (poll(&hp, 1, 100) > 0 && (hp.revents & (POLLHUP | POLLERR))) {
323 break; /* client gave up */
324 }
325 }
326 close(cfd);
327 CDEC(active);
328 return 1;
329 }
330 return 0;
331}

382/* Length-prefix lie: if the armed big-endian uint32 offset lies wholly within
383 * this buffer, rewrite it (set/add/sub) — a framing attack on binary protocols. */
384void
385mangle_apply(int is_up, char *buf, ssize_t nr, unsigned long base)
386{
387 struct fp_mangle *M = is_up ? &g_mangle_up : &g_mangle_down;
388 if (!M->active) {
389 return;
390 }
391 long off = M->offset;
392 if (off < (long) base || off + 4 > (long) base + nr) {
393 return; /* the 4-byte field is not fully contained in this read */
394 }
395 unsigned char *p = (unsigned char *) buf + (off - (long) base);
396 unsigned long v = ((unsigned long) p[0] << 24) | ((unsigned long) p[1] << 16)
397 | ((unsigned long) p[2] << 8) | (unsigned long) p[3];
398 long nv = (M->op == 0) ? M->val
399 : (M->op == 1) ? (long) v + M->val
400 : (long) v - M->val;
401 unsigned long u = (unsigned long) nv & 0xFFFFFFFFUL;
402 p[0] = (unsigned char) (u >> 24); p[1] = (unsigned char) (u >> 16);
403 p[2] = (unsigned char) (u >> 8); p[3] = (unsigned char) u;
404 CBUMP(mangled, 1);
405}

458/* Replay mode: act as a synthetic peer, feeding the client the recorded byte
459 * timeline (the g_replay_updir direction) with the original inter-segment
460 * timing. No live upstream is dialled. The store is immutable while replay is
461 * active, so it is read lock-free. */
462void
463replay_to_client(int cfd, unsigned epoch)
464{
465 int want = g_replay_updir ? 1 : 0;
466 unsigned long long last = 0;
467 int started = 0;
468 for (size_t k = 0; k < g_replay_store.n; k++) {
469 fp_replay_rec *r = &g_replay_store.recs[k];
470 if (r->is_up != want) {
471 continue;
472 }
473 if (started) {
474 long long d = (long long) r->ts_ms - (long long) last;
475 if (d > 0 && d < 60000) { /* honour original gaps, cap at 60s */
476 usleep((useconds_t) d * 1000);
477 }
478 }
479 last = r->ts_ms;
480 started = 1;
481 if (g_blocked || g_drop_epoch != epoch) {
482 break;
483 }
484 if (r->len > 0 &&
485 write_all(cfd, (const char *) r->bytes, (ssize_t) r->len) != 0) {
486 break;
487 }
488 CBUMP(replayed, (unsigned long) r->len);
489 }

492void *
493relay_thread(void *arg)
494{
495 relay_arg *ra = (relay_arg *) arg;
496 int cfd = ra->client_fd;
497 unsigned epoch = ra->epoch;
498 unsigned long conn_id = ra->conn_id;
499 fp_route *route = ra->route;
500 free(ra);
501
502 t_conn_id = conn_id; /* attribute deep-path fault events to this connection */
503 fp_event_set_route(fp_route_name(route)); /* tag this thread's events */
504
505 unsigned seed = g_seed + (unsigned) conn_id * 2654435761u;
506
507 /* toxicity: decide once per connection which directions this relay afflicts.
508 * A separate RNG stream (tseed) so the afflicted-set is reproducible under
509 * --seed without perturbing the byte-fault stream. */
510 unsigned tseed = seed ^ 0x5bd1e995u;
511 t_afflict_up = (int) (rand_r(&tseed) % 1000000u) < g_toxicity_up_ppm;
512 t_afflict_down = (int) (rand_r(&tseed) % 1000000u) < g_toxicity_down_ppm;
513
514 if (relay_predial(cfd, epoch, conn_id)) {
515 return NULL;
516 }
517
518 /* connect-delay: model slow upstream dial / SYN-ACK latency. */
519 if (g_connect_delay_ms > 0) {
520 usleep((useconds_t) g_connect_delay_ms * 1000);
521 }
522
523 /* Replay: synthesise the response from a recorded session, no upstream. */
524 if (g_replay_active) {
525 replay_to_client(cfd, epoch);
526 close(cfd);
527 CDEC(active);
528 return NULL;
529 }
530
531 int ufd = dial_route(route);
532 if (ufd < 0) {
533 close(cfd);
534 CDEC(active);
535 return NULL;
536 }
537 /* Egress NODELAY on BOTH ends so chunk/drip pieces are delivered as separate
538 * segments (otherwise the kernel coalesces them and the peer never sees a
539 * partial PDU). The accept side already set NODELAY on cfd. */
540 { int one = 1;
541 setsockopt(ufd, IPPROTO_TCP, TCP_NODELAY, &one, sizeof(one));
542 setsockopt(cfd, IPPROTO_TCP, TCP_NODELAY, &one, sizeof(one)); }
543
544 apply_conn_tuning(cfd, ufd);
545 send_proxy_header(ufd, cfd); /* forged PROXY header, before any client bytes */
546
547 /* fanout: open (and hold) extra upstream connections per client connection to
548 * drain the server's accept/worker pool — a connection-amplification attack. */
549 int extra[16];
550 int nextra = 0;
551 int fo = g_fanout;
552 if (fo > 0) {
553 if (fo > 16) {
554 fo = 16;
555 }
556 for (int k = 0; k < fo; k++) {
557 int f = dial_route(route);
558 if (f >= 0) {
559 extra[nextra++] = f;
560 CBUMP(fanout_conns, 1);
561 }
562 }
563 }
564
565 /* Own the per-direction byte totals here so per-route accounting survives
566 * every relay_pump exit path (sever / max-life / EOF). */
567 unsigned long up_ctr = 0, down_ctr = 0;
568 relay_pump(cfd, ufd, epoch, seed, &up_ctr, &down_ctr);
569 fp_route_add_bytes(route, up_ctr, down_ctr);
570
571 for (int k = 0; k < nextra; k++) {
572 close(extra[k]);
573 }
574 return NULL;
575}
