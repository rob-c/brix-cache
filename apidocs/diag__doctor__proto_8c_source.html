4 * WHAT: the per-protocol endpoint doctors for the HTTP family (http/https/davs),

19/*
20 * WHAT: file-local probe context for the HTTP-family deep-dive batteries.
21 * WHY: the static request-build/verdict-render helpers below would otherwise
22 * each need 6-7 parameters (host/port/tls/path/timeout/verify); one ctx
23 * struct keeps them under the parameter gate with explicit data flow.
24 * HOW: filled once by doctor_http/doctor_s3 from their (frozen, extern)
25 * signatures and passed by const pointer to every helper.
26 */
27typedef struct {
28 int tls; /* 1 = TLS transport */
29 const char *host;
30 int port;
31 const char *path; /* request path (S3: the URI) */
32 int tmo; /* per-probe timeout, ms */
33 int verify; /* verify TLS certificates */
34} http_probe_ctx;

37/*
38 * WHAT: record the TLS session facts + the "tls" OK verdict on the endpoint.
39 * WHY: the identical stanza was inlined in both doctor_http and doctor_s3.
40 * HOW: no-op on cleartext; otherwise copy version/cipher off the response
41 * and record the handshake/certificate verdict.

43static void
44http_record_tls(int tls, const brix_http_resp *r, doctor_ep *e)
45{
46 if (!tls) {
47 return;
48 }

50 snprintf(e->tls_ver, sizeof(e->tls_ver), "%s", r->tls_ver);
51 snprintf(e->tls_cipher, sizeof(e->tls_cipher), "%s", r->tls_cipher);
52 dx_record(e, &(dx_note){ "tls", DX_OK, 0, "TLS handshake completed + certificate accepted", "" });
53}

56/*
57 * WHAT: verdict-render half of the generic HTTP battery — classify the
58 * transfer-relevant headers of the already-received response.
59 * WHY: extracted from doctor_http for the complexity gate (request-build vs
60 * verdict-render split); stage numbering/order preserved.
61 * HOW: each stage inspects one response header and records one finding.
62 */
63static void
64http_check_common_headers(doctor_ep *e, const brix_http_resp *r)
65{
66 char val[512];
67
68 /* Stage 3: byte ranges (partial reads / multi-stream transfers depend on this). */
69 if (brix_http_header(r, "Accept-Ranges", val, sizeof(val))
70 && strstr(val, "bytes") != NULL) {
71 dx_record(e, &(dx_note){ "ranges", DX_OK, 0, "byte-range reads supported (Accept-Ranges: bytes)", "" });
72 } else {
73 dx_record(e, &(dx_note){ "ranges", DX_WARN, 0,
74 "server did not advertise Accept-Ranges (partial/parallel reads may not work)",
75 "enable range support if clients use partial reads" });
76 }
77
78 /* Stage 4: checksum advertisement (RFC-3230 Digest, WLCG transfers rely on it). */
79 if (brix_http_header(r, "Digest", val, sizeof(val)) && strchr(val, '=') != NULL) {
80 /* RFC-3230 form is "algo=value"; require the '=' so a malformed header
81 * isn't counted as a working checksum. */
82 dx_record(e, &(dx_note){ "checksum", DX_OK, 0, "server advertises a content Digest (checksum)", "" });
83 } else {
84 dx_record(e, &(dx_note){ "checksum", DX_WARN, 0,
85 "no Digest header (checksum verification unavailable over HTTP)",
86 "enable Want-Digest/Digest if integrity checks are required" });
87 }
88
89 /* Stage 5: content-length present (sized transfers). */
90 if (brix_http_header(r, "Content-Length", val, sizeof(val))) {
91 dx_record(e, &(dx_note){ "content-length", DX_OK, 0, "response is sized (Content-Length present)", "" });
92 }
93}

96/*
97 * WHAT: davs deep-dive extras — OPTIONS (WebDAV class / TPC) + PROPFIND.
98 * WHY: extracted from doctor_http for the complexity gate.
99 * HOW: two bounded requests; each response is rendered into davs-class /
100 * davs-tpc / davs-listing verdicts in the original emission order.
101 */
102static void
103http_davs_extras(const http_probe_ctx *pc, doctor_ep *e)
104{
105 brix_http_resp r;
106 brix_status st;
107 char val[512];
108
109 brix_status_clear(&st);
110 if (brix_http_req(pc->host, pc->port, pc->tls, "OPTIONS", pc->path, NULL,
111 NULL, 0, pc->tmo, pc->verify, NULL, NULL, &r, &st) == 0) {
112 if (brix_http_header(&r, "DAV", val, sizeof(val))) {
113 int class2 = (strstr(val, "2") != NULL);
114 dx_record(e, &(dx_note){ "davs-class", DX_OK, r.status,
115 class2 ? "WebDAV class 2 advertised (LOCK supported)"
116 : "WebDAV advertised (DAV header present)", "" });
117 } else {
118 dx_record(e, &(dx_note){ "davs-class", DX_WARN, r.status,
119 "OPTIONS returned no DAV header (WebDAV may be disabled)",
120 "confirm brix_webdav is on for this location" });
121 }
122 if (brix_http_header(&r, "Allow", val, sizeof(val))
123 && strstr(val, "COPY") != NULL) {
124 dx_record(e, &(dx_note){ "davs-tpc", DX_OK, 0,
125 "COPY method allowed (third-party-copy capable)", "" });
126 }
127 brix_http_resp_free(&r);
128 } else {
129 dx_record(e, &(dx_note){ "davs-class", DX_WARN, st.kxr,
130 "OPTIONS request failed", "" });
131 }
132 brix_status_clear(&st);
133 if (brix_http_req(pc->host, pc->port, pc->tls, "PROPFIND", pc->path,
134 "Depth: 1\r\n", NULL, 0, pc->tmo, pc->verify, NULL, NULL, &r,
135 &st) == 0) {
136 if (r.status == 207) {
137 dx_record(e, &(dx_note){ "davs-listing", DX_OK, 207,
138 "PROPFIND multistatus listing works", "" });
139 } else if (r.status == 401 || r.status == 403) {
140 dx_record(e, &(dx_note){ "davs-listing", DX_WARN, r.status,
141 "PROPFIND requires authentication", "provide a credential" });
142 } else {
143 dx_record(e, &(dx_note){ "davs-listing", DX_WARN, r.status,
144 "PROPFIND did not return 207 multistatus", "" });
145 }
146 brix_http_resp_free(&r);
147 }

151/*
152 * http / https deep-dive: connect (+TLS cert/cipher), HEAD/GET the path, classify
153 * the HTTP status, byte-range support, the Digest (checksum) header, and 401 auth
154 * posture. For davs, also OPTIONS (WebDAV class) and PROPFIND (listing). Every probe
155 * is bounded by the per-probe timeout. PII-free: only statuses/header-names/sizes.
156 */
157void
158doctor_http(const diag_args *a, const dx_url_t *u, doctor_ep *e)
159{
160 brix_http_resp r;
161 brix_status st;
162 http_probe_ctx pc;
163
164 pc.tls = u->tls;
165 pc.host = u->host;
166 pc.port = u->port;
167 pc.path = u->path;
168 pc.tmo = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
169 pc.verify = a->verify_tls;
170
171 memset(e, 0, sizeof(*e));
172 e->proto = u->proto;
173 e->status = DOC_GREEN;
174 snprintf(e->host, sizeof(e->host), "%s", u->host);
175 e->port = u->port;
176
177 /* Stage 1: reachability + (TLS handshake/cert). Try HEAD; fall back to a 1-byte
178 * ranged GET if HEAD is refused, so we still measure connect/TLS. */
179 brix_status_clear(&st);
180 if (brix_http_req(u->host, u->port, u->tls, "HEAD", u->path, NULL, NULL, 0,
181 pc.tmo, pc.verify, NULL, NULL, &r, &st) != 0) {
182 brix_status_clear(&st);
183 if (brix_http_req(u->host, u->port, u->tls, "GET", u->path,
184 "Range: bytes=0-0\r\n", NULL, 0, pc.tmo, pc.verify,
185 NULL, NULL, &r, &st) != 0) {
186 dx_http_fail(e, u->tls, &st);
187 return;
188 }
189 }
190 e->connected = 1;
191 http_record_tls(u->tls, &r, e);
192
193 /* Stage 2: HTTP status. */
194 dx_http_status(e, "http", r.status);
195
196 /* Stages 3-5: transfer-relevant response headers. */
197 http_check_common_headers(e, &r);
198 brix_http_resp_free(&r);
199
200 /* davs extras: OPTIONS (WebDAV class) + PROPFIND (collection listing). */
201 if (u->proto == DXP_DAVS) {
202 http_davs_extras(&pc, e);
203 }
204}

207/*
208 * WHAT: verdict-render half of the anonymous S3 reachability probe.
209 * WHY: extracted from doctor_s3 for the complexity gate.
210 * HOW: map the unauthenticated GET's HTTP status onto the S3 auth-posture
211 * verdicts (enforced / missing / public / other).
212 */
213static void
214s3_classify_anon(doctor_ep *e, const brix_http_resp *r)
215{
216 if (r->status == 403) {
217 dx_record(e, &(dx_note){ "s3-auth", DX_OK, 403,
218 "endpoint enforces S3 authentication (anonymous request denied)", "" });
219 } else if (r->status == 404) {
220 dx_record(e, &(dx_note){ "s3-bucket", DX_WARN, 404,
221 "bucket/key not found (NoSuchBucket/NoSuchKey)",
222 "verify the bucket and key path" });
223 } else if (r->status >= 200 && r->status < 300) {
224 dx_record(e, &(dx_note){ "s3-auth", DX_WARN, r->status,
225 "anonymous S3 request SUCCEEDED — the resource is public",
226 "confirm public access is intended; otherwise restrict it" });
227 } else {
228 dx_http_status(e, "s3-req", r->status);
229 }
230}

233/*
234 * WHAT: verdict-render half of the SigV4 probe — classify the signed
235 * request's response.
236 * WHY: extracted from s3_sigv4_probe for the complexity gate.
237 * HOW: 2xx = accepted; 403 splits into signature fault vs policy deny by
238 * reading the S3 <Code> element; anything else falls back to the
239 * generic HTTP status classifier.
240 */
241static void
242s3_classify_signed(doctor_ep *e, const brix_http_resp *r)
243{
244 if (r->status >= 200 && r->status < 300) {
245 dx_record(e, &(dx_note){ "s3-sigv4", DX_OK, r->status,
246 "SigV4-signed request accepted (signature/clock/region OK)", "" });
247 } else if (r->status == 403) {
248 /* read the S3 <Code> element (not a body-wide substring, which
249 * could false-match) to tell a signature fault from a policy deny. */
250 const char *cs = r->body ? strstr(r->body, "<Code>") : NULL;
251 int sig_fault = (cs != NULL
252 && strncmp(cs + 6, "SignatureDoesNotMatch", 21) == 0);
253 if (sig_fault) {
254 dx_record(e, &(dx_note){ "s3-sigv4", DX_FAIL, 403,
255 "SigV4 signature rejected (SignatureDoesNotMatch — clock skew / region / key mismatch)",
256 "check client clock vs server, the region, and the access key/secret" });
257 } else {
258 dx_record(e, &(dx_note){ "s3-sigv4", DX_WARN, 403,
259 "signed request denied (access policy, not a signature fault)",
260 "check the bucket/object policy for this identity" });
261 }
262 } else {
263 dx_http_status(e, "s3-sigv4", r->status);
264 }
265}

268/*
269 * WHAT: request-build half of the authenticated SigV4 probe.
270 * WHY: extracted from doctor_s3 for the complexity gate; only called when
271 * AWS credentials are present in the environment.
272 * HOW: sign the exact Host header we send, replay the GET with the SigV4
273 * Authorization headers, and hand the response to s3_classify_signed.
274 * PII-free: never emits the key or signature.
275 */
276static void
277s3_sigv4_probe(const http_probe_ctx *pc, const char *ak, const char *sk,
278 doctor_ep *e)
279{
280 brix_http_resp r;
281 brix_status st;
282 char hdrs[1024];
283 char hostport[300];
284 const char *region = getenv("AWS_DEFAULT_REGION");
285 s3_sign_req sq;
286
287 if (region == NULL || region[0] == '\0') { region = "us-east-1"; }
288 /* Sign the exact Host header we send (host:port) — the server canonicalises
289 * the Host value verbatim, so signing the bare host would mismatch. */
290 snprintf(hostport, sizeof(hostport), "%s:%d", pc->host, pc->port);
291 sq.method = "GET";
292 sq.host = hostport;
293 sq.uri = pc->path;
294 sq.ak = ak;
295 sq.sk = sk;
296 sq.region = region;
297 if (s3_sign(&sq, hdrs, sizeof(hdrs)) != 0) {
298 dx_record(e, &(dx_note){ "s3-sigv4", DX_WARN, 0, "could not build a SigV4 signature (client)", "" });
299 return;
300 }
301 brix_status_clear(&st);
302 if (brix_http_req(pc->host, pc->port, pc->tls, "GET", pc->path, hdrs, NULL,
303 0, pc->tmo, pc->verify, NULL, NULL, &r, &st) != 0) {
304 dx_record(e, &(dx_note){ "s3-sigv4", DX_WARN, st.kxr, "signed request failed to complete", "" });
305 return;
306 }
307 s3_classify_signed(e, &r);
308 brix_http_resp_free(&r);
309}

312/*
313 * s3 deep-dive: connect (+TLS), an UNAUTHENTICATED GET to confirm reachability +
314 * that the server enforces auth (403/AccessDenied) vs is public (200) vs missing
315 * (404/NoSuchBucket). If AWS_ACCESS_KEY_ID/SECRET are in the environment, also send
316 * a SigV4-signed HEAD and confirm the signature is accepted (catches signer/clock/
317 * region faults). PII-free: never emits the key or signature.
318 */
319void
320doctor_s3(const diag_args *a, const dx_url_t *u, doctor_ep *e)
321{
322 brix_http_resp r;
323 brix_status st;
324 http_probe_ctx pc;
325 const char *ak = getenv("AWS_ACCESS_KEY_ID");
326 const char *sk = getenv("AWS_SECRET_ACCESS_KEY");
327
328 pc.tls = u->tls;
329 pc.host = u->host;
330 pc.port = u->port;
331 pc.path = u->path;
332 pc.tmo = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
333 pc.verify = a->verify_tls;
334
335 memset(e, 0, sizeof(*e));
336 e->proto = DXP_S3;
337 e->status = DOC_GREEN;
338 snprintf(e->host, sizeof(e->host), "%s", u->host);
339 e->port = u->port;
340
341 /* Stage 1: reachability + TLS via an unauthenticated GET. */
342 brix_status_clear(&st);
343 if (brix_http_req(u->host, u->port, u->tls, "GET", u->path, NULL, NULL, 0,
344 pc.tmo, pc.verify, NULL, NULL, &r, &st) != 0) {
345 dx_http_fail(e, u->tls, &st);
346 return;
347 }
348 e->connected = 1;
349 http_record_tls(u->tls, &r, e);
350 s3_classify_anon(e, &r);
351 brix_http_resp_free(&r);
352
353 /* Stage 2: authenticated SigV4 probe (only if AWS creds are present). */
354 if (ak != NULL && sk != NULL && ak[0] != '\0' && sk[0] != '\0') {
355 s3_sigv4_probe(&pc, ak, sk, e);
356 } else {
357 dx_record(e, &(dx_note){ "s3-sigv4", DX_OK, 0,
358 "no AWS credentials in environment — signed-request check skipped (posture only)",
359 "set AWS_ACCESS_KEY_ID/AWS_SECRET_ACCESS_KEY to test SigV4 acceptance" });
360 }
361}

364/*
365 * §5.4 cns-stat-drift: compare the manager's (CNS) view of a path against the
366 * data server that holds it. A size/mtime disagreement ⇒ stale/unconverged
367 * inventory; a DS that will not connect/stat at all ⇒ a ghost registration.
368 * PII-free: records classified cause/remedy only, never the path. */
369static void
370cms_stat_drift_check(const diag_args *a, const char *ds_auth, const char *path,
371 const brix_statinfo *si_mgr, doctor_ep *e)
372{
373 brix_url du;
374 brix_conn dc;
375 brix_status dst;
376 brix_statinfo si_ds;
377 char host[256], dsurl[320];
378 int port, v6;
379
380 if (brix_split_host_port(ds_auth, host, sizeof(host), &port, 1094) != 0) {
381 return;
382 }
383 v6 = (strchr(host, ':') != NULL && host[0] != '[');
384 snprintf(dsurl, sizeof(dsurl), "root://%s%s%s:%d/",
385 v6 ? "[" : "", host, v6 ? "]" : "", port);
386 brix_status_clear(&dst);
387 if (brix_endpoint_parse(dsurl, &du, &dst) != 0) {
388 return;
389 }
390 snprintf(du.path, sizeof(du.path), "%s", path[0] ? path : "/");
391 if (brix_connect(&dc, &du, &a->conn, &dst) != 0) {
392 dx_record(e, &(dx_note){ "cms-ghost", DX_FAIL, dst.kxr,
393 "located holder does not serve the path (ghost registration)",
394 "remove the stale CMS registry entry / restart the data server" });
395 return;
396 }
397 dc.io.timeout_ms = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
398 brix_status_clear(&dst);
399 if (brix_stat(&dc, du.path, &si_ds, &dst) == 0) {
400 long dm = si_ds.mtime - si_mgr->mtime;
401 if (dm < 0) { dm = -dm; }
402 if (si_ds.size != si_mgr->size || dm > 2) {
403 dx_record(e, &(dx_note){ "cns-stat-drift", DX_WARN, 0,
404 "manager (CNS) metadata disagrees with the data server",
405 "check the DS emit path / manager inventory convergence" });
406 } else {
407 dx_record(e, &(dx_note){ "cns-stat-drift", DX_OK, 0,
408 "manager (CNS) metadata agrees with the data server", "" });
409 }
410 } else {
411 dx_record(e, &(dx_note){ "cms-ghost", DX_FAIL, dst.kxr,
412 "located holder present but does not stat the path (ghost/stale entry)",
413 "remove the stale CMS registry entry / restart the data server" });
414 }
415 brix_close(&dc);
416}

419/*
420 * cms deep-dive: a cluster manager (cmsd/redirector) is a root:// endpoint that
421 * answers kXR_locate with the data server(s) holding a path and issues kXR_redirect.
422 * Connect to the manager, locate the path, and confirm the redirect resolution to a
423 * reachable data server; flag no-holder / unreachable-DS (ghost) / redirect issues.
424 * Reuses the libbrix locate + reconnect machinery (the redirect loop-guard applies).
425 */
426void
427doctor_cms(const diag_args *a, const char *host, int port, const char *path,
428 doctor_ep *e)
429{
430 brix_url u;
431 brix_conn c;
432 brix_status st;
433 char ds_auth[288];
434
435 ds_auth[0] = '\0';
436 memset(e, 0, sizeof(*e));
437 e->proto = DXP_CMS;
438 e->status = DOC_GREEN;
439 snprintf(e->host, sizeof(e->host), "%s", host);
440 e->port = port;
441
442 memset(&u, 0, sizeof(u));
443 u.scheme = XRDC_SCHEME_ROOT;
444 snprintf(u.host, sizeof(u.host), "%s", host);
445 u.port = port;
446 snprintf(u.path, sizeof(u.path), "%s", path[0] ? path : "/");
447
448 brix_status_clear(&st);
449 if (brix_connect(&c, &u, &a->conn, &st) != 0) {
450 dx_record(e, &(dx_note){ "cms-connect", DX_FAIL, st.kxr,
451 "could not connect to the cluster manager",
452 "check the manager (cmsd/redirector) is up on this host:port" });
453 doc_issue(e, DOC_RED, "manager connect failed");
454 return;
455 }
456 e->connected = 1;
457 c.io.timeout_ms = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
458 dx_record(e, &(dx_note){ "cms-connect", DX_OK, 0, "manager reachable + login completed", "" });
459
460 {
461 char loc[2048];
462 brix_status lst;
463 brix_status_clear(&lst);
464 if (brix_locate(&c, u.path, loc, sizeof(loc), &lst) != 0) {
465 /* a manager returns NotFound when no server holds the path. */
466 dx_record_status(e, "cms-locate", &lst);
467 } else {
468 char *t, *save;
469 for (t = strtok_r(loc, " \t\r\n", &save); t != NULL;
470 t = strtok_r(NULL, " \t\r\n", &save)) {
471 if (t[0] == 'S') {
472 e->holders++;
473 if (ds_auth[0] == '\0' && t[1] != '\0') {
474 snprintf(ds_auth, sizeof(ds_auth), "%s", t + 2);
475 }
476 }
477 }
478 if (e->holders == 0) {
479 dx_record(e, &(dx_note){ "cms-locate", DX_FAIL, 0,
480 "manager located no data server for the path (no holder)",
481 "check data-server registration and the CMS registry" });
482 } else {
483 dx_record(e, &(dx_note){ "cms-locate", DX_OK, 0,
484 "manager resolved the path to data server(s)", "" });
485 }
486 }
487 }
488
489 /* Resolution: a stat through the manager must follow the redirect to a live DS.
490 * A redirect loop or dead DS surfaces here (the loop-guard returns an error). */
491 {
492 brix_statinfo si;
493 brix_status rst;
494 brix_status_clear(&rst);
495 if (brix_stat(&c, u.path, &si, &rst) == 0) {
496 dx_record(e, &(dx_note){ "cms-redirect", DX_OK, 0,
497 "manager→data-server redirect resolved to a live server", "" });
498 /* §5.4: with --config-audit, cross-check the manager's (CNS) stat
499 * against the data server that holds the path (drift / ghost). */
500 e->mgr_stat_size = si.size;
501 e->mgr_stat_mtime = si.mtime;
502 e->mgr_stat_have = 1;
503 if ((a->config_audit || a->all_servers) && ds_auth[0] != '\0') {
504 cms_stat_drift_check(a, ds_auth, u.path, &si, e);
505 }
506 } else if (rst.kxr == kXR_NotFound) {
507 dx_record(e, &(dx_note){ "cms-redirect", DX_WARN, rst.kxr,
508 "path not found via the manager (redirect resolved, file absent)",
509 "verify the path exists on a registered data server" });
510 } else {
511 dx_record(e, &(dx_note){ "cms-redirect", DX_FAIL, rst.kxr,
512 "manager redirect did not resolve to a reachable data server (dead DS / redirect loop)",
513 "check data-server health and the CMS registry for stale entries" });
514 }
515 }
516 brix_close(&c);
517}
