1/*
2 * oci_referrers.h — the referrers graph internal to the registry surface
3 * (D15.1, distribution-spec §"Listing Referrers").
4 *
5 * WHAT: index a pushed manifest under the subject it declares, forget that
6 * edge when the manifest is deleted, and answer
7 * GET /v2/<name>/referrers/<digest> from what the index holds.
8 * WHY: the referrers API is how a signature, an SBOM or a provenance
9 * attestation becomes DISCOVERABLE: the artifact is an ordinary
10 * manifest carrying `subject`, and without the reverse index a client
11 * holding an image digest has no way to ask "what did anyone say about
12 * this?" other than enumerating the whole repository. The index has to
13 * be written by the push that creates the edge, because that is the
14 * only moment the registry sees the referrer and the subject together.
15 * HOW: one empty-directory-per-subject fanout under the repository, holding
16 * one small JSON descriptor per referrer, plus a `.subject` back-
17 * pointer beside the manifest so a DELETE knows which edge to cut
18 * without scanning every subject in the repo. The descriptors are
19 * written once, at push time, so the read path is a bounded directory
20 * walk and a concatenation rather than a re-parse of every manifest.
21 */

29/* One subject may collect this many referrers before the listing truncates.
30 * A subject with hundreds of signatures is already pathological; the cap is
31 * here so that a directory somebody filled cannot turn one GET into an
32 * unbounded allocation. */
33#define BRIX_OCI_REFERRERS_MAX 512

35/* A stored descriptor is mediaType + digest + size + artifactType + a few
36 * annotations. 4 KiB is roomy for that and small enough that 512 of them
37 * still bound the assembled index well under the response cap. */
38#define BRIX_OCI_REFERRER_DESC_MAX 4096

40/* Is this document's `subject`, if it declares one, a descriptor this
41 * registry can index? 0 = yes, or the manifest declares none; -1 with `*why`
42 * set to the refusal detail. A manifest whose subject we cannot read must be
43 * REFUSED rather than stored: accepting it would publish an artifact whose
44 * edge is invisible, which is indistinguishable to the pusher from a
45 * signature that was recorded. */
46int brix_oci_referrers_subject_ok(json_t *doc, const char **why);

48/* Record the edge `d` --subject--> doc["subject"], if the document declares
49 * one. `doc` is the already-parsed manifest, `size` its stored byte count and
50 * `ctype` its media type; both ride into the descriptor because the listing
51 * must answer with them without opening the referrer.
52 *
53 * `subj` receives the subject's digest string ("<alg>:<hex>") for the
54 * OCI-Subject response header, or "" when the manifest declares no subject.
55 * NGX_OK either way; NGX_ERROR only when an edge was declared and could not
56 * be recorded — a push whose discoverability silently failed is worse than a
57 * push that failed. */
58ngx_int_t brix_oci_referrers_index(const brix_oci_store_t *st,
59 const brix_oci_req_t *req, json_t *doc, const brix_oci_digest_t *d,
60 const char *ctype, off_t size, char *subj, size_t subjsz, ngx_log_t *log);

62/* Cut the edge a manifest DELETE invalidates: the descriptor under its

64 * manifest is going away regardless, and a stale descriptor is a listing
