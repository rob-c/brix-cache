1/*
2 * sd_s3_list_flat.c — S3 ListObjectsV2 without a delimiter: the flat, recursive
3 * object listing behind the backend-catalog verb.
4 *
5 * WHAT: sd_s3_list_flat_page() — one signed GET returning up to 1000 keys under
6 * a prefix at ANY depth, each with its <Size> and <LastModified>.
7 * WHY: driver->enumerate asks what objects an export holds, not what a
8 * directory contains. Building that from the delimited lister costs one
9 * signed request per pseudo-directory, which is precisely the
10 * amplification the catalog verb exists to avoid; and because S3 reports
11 * size and mtime in the listing itself, a stat-bearing enumeration costs
12 * no extra request at all.
13 * HOW: the same canonical query as the delimited page minus `delimiter`, then a
14 * single <Contents> tag walk. Every request-side primitive is shared with
15 * sd_s3_list.c through sd_s3_list_scan.c.
16 */

27/* Parse an S3 <LastModified> ("2009-10-12T17:50:30.000Z", always UTC) to a
28 * time_t. Returns 0 on any malformed value — an unparsable timestamp must not
29 * become a plausible-looking epoch date that an inventory would then treat as a
30 * real mtime, and the caller reports it as "no stat" instead. */
31static time_t
32s3l_parse_iso8601(const char *s, const char *e)
33{
34 char buf[40];
35 struct tm tm;
36 int y, mo, d, h, mi, sec;
37 size_t n = (size_t) (e - s);
38
39 if (n < 19 || n >= sizeof(buf)) {
40 return 0;
41 }
42 memcpy(buf, s, n);
43 buf[n] = '\0';
44 if (sscanf(buf, "%4d-%2d-%2dT%2d:%2d:%2d", &y, &mo, &d, &h, &mi, &sec) != 6) {
45 return 0;
46 }
47 if (y < 1970 || mo < 1 || mo > 12 || d < 1 || d > 31
48 || h < 0 || h > 23 || mi < 0 || mi > 59 || sec < 0 || sec > 60)
49 {
50 return 0; /* field-validate before timegm, which would normalise */
51 }
52 memset(&tm, 0, sizeof(tm));
53 tm.tm_year = y - 1900;
54 tm.tm_mon = mo - 1;
55 tm.tm_mday = d;
56 tm.tm_hour = h;
57 tm.tm_min = mi;
58 tm.tm_sec = sec;
59 tm.tm_isdst = 0;
60 return timegm(&tm);
61}

63/* Read one <Contents> block's <Size>, rejecting a non-numeric or negative body.
64 * Returns 1 with *out set, or 0. */
65static int
66s3l_block_size(const char *block, const char *end, uint64_t *out)
67{
68 const char *ss;
69 const char *se;
70 char num[32];
71 char *endp;
72 unsigned long long v;
73 size_t n;
74
75 if (!sd_s3l_first_text(block, end, "<Size>", "</Size>", &ss, &se)) {
76 return 0;
77 }
78 n = (size_t) (se - ss);
79 if (n == 0 || n >= sizeof(num)) {
80 return 0;
81 }
82 memcpy(num, ss, n);
83 num[n] = '\0';
84 errno = 0;
85 v = strtoull(num, &endp, 10);
86 if (errno != 0 || endp != num + n) {
87 return 0;
88 }
89 *out = (uint64_t) v;
90 return 1;
91}

93/* Emit one <Contents> block as a catalog entry. `bend` bounds the block so a
94 * missing <Size>/<LastModified> cannot be satisfied from the NEXT object's
95 * fields — the scanner is positional, and an unbounded sd_s3l_first_text would
96 * happily attribute a neighbour's size to this key. Returns cb's stop signal. */
97static int
98s3l_emit_flat(const char *block, const char *bend, sd_s3_list_flat_cb cb,
99 void *ud)
100{
101 char key[SD_S3_KEY_MAX];
102 const char *ks;
103 const char *ke;
104 uint64_t size = 0;
105 time_t mtime = 0;
106 int n;
107
108 if (!sd_s3l_first_text(block, bend, "<Key>", "</Key>", &ks, &ke)) {
109 return 0;
110 }
111 n = sd_s3l_xml_unescape(ks, ke, key, sizeof(key));
112 if (n <= 0) {
113 return 0; /* unrepresentable or empty key — skip */
114 }
115 if (!s3l_block_size(block, bend, &size)) {
116 size = 0;
117 }
118 if (sd_s3l_first_text(block, bend, "<LastModified>", "</LastModified>",
119 &ks, &ke))
120 {
121 mtime = s3l_parse_iso8601(ks, ke);
122 }
123 return cb(ud, key, size, mtime);
124}

126/* Walk every <Contents> block in the page. There are no <CommonPrefixes> in an
127 * undelimited listing, so the flat walk is a single tag scan. */
128static void
129s3l_walk_flat(const char *body, const char *end, sd_s3_list_flat_cb cb,
130 void *ud)
131{
132 const char *scan = body;
133
134 while (scan < end) {
135 const char *c = sd_s3l_find(scan, (size_t) (end - scan), "<Contents>");
136 const char *bend;
137
138 if (c == NULL) {
139 break;
140 }
141 c += sizeof("<Contents>") - 1;
142 bend = sd_s3l_find(c, (size_t) (end - c), "</Contents>");
143 if (bend == NULL) {
144 bend = end; /* truncated final block — scan what we have */
145 }
146 if (s3l_emit_flat(c, bend, cb, ud) != 0) {
147 return; /* caller asked to stop */
148 }
149 scan = bend;
150 }
151}

153int
154sd_s3_list_flat_page(const sd_s3_open_params *p, const char *prefix,
155 const char *cont_in, sd_s3_list_flat_cb cb, void *ud, int *truncated,
156 char *cont_out, size_t cont_cap, char *errbuf, size_t errcap)
157{
158 char qs[S3L_QS_CAP];
159 brix_s3_resp_t resp;
160 sd_s3_file *f = NULL;
161 const char *body;
162 size_t blen = 0;
163 int rc;
164
165 if (p == NULL || prefix == NULL || cb == NULL || truncated == NULL
166 || cont_out == NULL || cont_cap == 0)
167 {
168 errno = EINVAL;
169 sd_s3_set_err(errbuf, errcap, "s3 list-flat: bad parameters");
170 return -1;
171 }
172 *truncated = 0;
173 cont_out[0] = '\0';
174
175 if (sd_s3l_build_query(prefix, strlen(prefix), cont_in, 0 /* flat */, qs,
176 sizeof(qs), errbuf, errcap) != 0) {
177 return -1;
178 }
179 rc = sd_s3l_fetch(p, qs, &f, &resp, errbuf, errcap);
180 if (rc != 0) {
181 return rc;
182 }
183 if (f == NULL) {
184 /* Same defensive check the delimited page makes: sd_s3l_fetch's contract is
185 * rc == 0 ⇒ *f_out set, and a future arm breaking it would land here as
186 * a null deref rather than an error. */
187 errno = EIO;
188 sd_s3_set_err(errbuf, errcap, "s3 list-flat: fetch returned no handle");
189 return -1;
190 }
191
192 body = f->transport->resp_body(&resp, &blen);
193 if (body == NULL) {
194 blen = 0;
195 body = "";
196 }
197 sd_s3l_page_meta(body, body + blen, truncated, cont_out, cont_cap);
198 s3l_walk_flat(body, body + blen, cb, ud);
199
200 f->transport->resp_free(&resp);
201 sd_s3_close(f);
202 return 0;
203}
