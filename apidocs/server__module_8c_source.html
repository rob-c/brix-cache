36/* ---- Merge CMS connection limits and deadlines ----
37 *
38 * WHAT: Inherit the CMS listener's timeout and connection-limit controls,
39 * deriving idle and TCP user timeouts when they are not explicit.
40 *
41 * WHY: Keeping the dependent timeout defaults together prevents the main
42 * configuration merge from becoming a branch-heavy policy accumulator.
43 *
44 * HOW: 1. Merge the login timeout.
45 * 2. Derive the idle timeout from the ping interval when unset.
46 * 3. Merge connection caps and keepalive, then default TCP user timeout
47 * to the effective idle timeout.
48 */
49static void
50brix_cms_srv_merge_connection_policy(ngx_stream_brix_cms_srv_conf_t *prev,
51 ngx_stream_brix_cms_srv_conf_t *conf)
52{
53 ngx_msec_t derived_idle;
54
55 ngx_conf_merge_msec_value(conf->login_timeout, prev->login_timeout,
56 BRIX_CMS_FSXEQ_TIMEOUT_DEFAULT_MS);
57 if (conf->idle_timeout == NGX_CONF_UNSET_MSEC) {
58 if (prev->idle_timeout != NGX_CONF_UNSET_MSEC) {
59 conf->idle_timeout = prev->idle_timeout;
60 } else {
61 derived_idle = (ngx_msec_t) conf->interval
62 * BRIX_CMS_IP_CHECK_INTERVAL_MS;
63 conf->idle_timeout =
64 (derived_idle > BRIX_CMS_READ_TIMEOUT_MAX_MS)
65 ? derived_idle : BRIX_CMS_READ_TIMEOUT_MAX_MS;
66 }
67 }
68
69 ngx_conf_merge_value(conf->max_connections, prev->max_connections,
70 BRIX_CMS_MAX_CONNECTIONS);
71 ngx_conf_merge_value(conf->max_connections_per_ip,
72 prev->max_connections_per_ip,
73 BRIX_CMS_MAX_CONNECTIONS_PER_IP);
74 ngx_conf_merge_value(conf->tcp_keepalive, prev->tcp_keepalive, 1);
75 if (conf->tcp_user_timeout == NGX_CONF_UNSET_MSEC) {
76 conf->tcp_user_timeout =
77 (prev->tcp_user_timeout != NGX_CONF_UNSET_MSEC)
78 ? prev->tcp_user_timeout : conf->idle_timeout;
79 }
80}

83/* ---- Merge and validate CMS peer-admission policy ----
84 *
85 * WHAT: Inherit CIDR/SSS admission controls and return NGX_CONF_OK only when
86 * an enabled listener has at least one non-empty effective mechanism.
87 *
88 * WHY: A policy-free CMS registration listener lets an arbitrary reachable
89 * peer publish routing claims, so configuration must fail closed.
90 *
91 * HOW: 1. Inherit omitted CIDR and SSS settings from the parent.
92 * 2. Accept disabled listeners without an admission policy.
93 * 3. Accept non-empty CIDR or parsed SSS keys; otherwise log and reject.
94 */
95static char *
96brix_cms_srv_merge_admission(ngx_conf_t *cf,
97 ngx_stream_brix_cms_srv_conf_t *prev,
98 ngx_stream_brix_cms_srv_conf_t *conf)
99{
100 if (conf->allow == NULL) {
101 conf->allow = prev->allow;
102 }
103 if (conf->sss_keys == NULL) {
104 conf->sss_keys = prev->sss_keys;
105 conf->sss_keytab = prev->sss_keytab;
106 }
107
108 if (!conf->enable) {
109 return NGX_CONF_OK;
110 }
111 if (conf->allow != NULL && conf->allow->nelts > 0) {
112 return NGX_CONF_OK;
113 }
114 if (conf->sss_keys != NULL && conf->sss_keys->nelts > 0) {
115 return NGX_CONF_OK;
116 }
117
118 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
119 "brix_cms_server on requires brix_cms_server_allow and/or "
120 "brix_cms_server_sss_keytab; refusing an unauthenticated "
121 "registration listener");
122 return NGX_CONF_ERROR;
123}

126/* ---- Merge mutually exclusive CMS operator-list files ----
127 *
128 * WHAT: Merge blacklist and whitelist paths and reject a configuration that
129 * enables both, returning NGX_CONF_OK or NGX_CONF_ERROR.
130 *
131 * WHY: One file must remain the sole operator authority; combining them can
132 * create contradictory decisions on every polling pass.
133 *
134 * HOW: 1. Merge both path strings parent-to-child.
135 * 2. Log and reject when both effective paths are non-empty.
136 */
137static char *
138brix_cms_srv_merge_operator_lists(ngx_conf_t *cf,
139 ngx_stream_brix_cms_srv_conf_t *prev,
140 ngx_stream_brix_cms_srv_conf_t *conf)
141{
142 ngx_conf_merge_str_value(conf->blacklist_file, prev->blacklist_file, "");
143 ngx_conf_merge_str_value(conf->whitelist_file, prev->whitelist_file, "");
144 if (conf->whitelist_file.len == 0 || conf->blacklist_file.len == 0) {
145 return NGX_CONF_OK;
146 }
147
148 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
149 "brix_cms_whitelist_file and brix_cms_blacklist_file are "
150 "mutually exclusive");
151 return NGX_CONF_ERROR;
152}

155/* ---- Merge the complete CMS-server configuration ----
156 *
157 * WHAT: Merge parent CMS listener settings into the child and return
158 * NGX_CONF_OK, or NGX_CONF_ERROR for an unsafe combination.
159 *
160 * WHY: The orchestrator keeps merge ordering explicit while focused helpers
161 * own dependent timeouts, admission policy, and operator-list policy.
162 *
163 * HOW: 1. Merge enable/interval and clamp the ping interval.
164 * 2. Merge connection policy and validate peer admission.
165 * 3. Merge operator lists, then merge the supervisor threshold.
166 */
167
168static char *
169brix_cms_srv_merge_conf(ngx_conf_t *cf, void *parent, void *child)
170{
171 ngx_stream_brix_cms_srv_conf_t *prev = parent;
172 ngx_stream_brix_cms_srv_conf_t *conf = child;
173 char *rv;
174
175 ngx_conf_merge_value(conf->enable, prev->enable, 0);
176 ngx_conf_merge_value(conf->interval, prev->interval, 60);
177 if (conf->interval < 1) {
178 conf->interval = 1; /* never derive a 0ms self-rearming ping timer */
179 }
180
181 brix_cms_srv_merge_connection_policy(prev, conf);
182
183 rv = brix_cms_srv_merge_admission(cf, prev, conf);
184 if (rv != NGX_CONF_OK) {
185 return rv;
186 }
187
188 rv = brix_cms_srv_merge_operator_lists(cf, prev, conf);
189 if (rv != NGX_CONF_OK) {
190 return rv;
191 }
192
193 /* §2.9: ManTree-style supervisor offload (0 = off). */
194 ngx_conf_merge_value(conf->max_direct, prev->max_direct, 0);
195
196 return NGX_CONF_OK;
197}

211
212 rv = ngx_conf_set_flag_slot(cf, cmd, conf_ptr);

224 /*
225 * Auto-role derivation: a block that runs `brix_cms_server on` accepts
226 * downstream cmsd nodes, i.e. it IS a manager. Turn on the main module's
227 * manager_mode for this same server block so the upstream CMS client
228 * (send.c) advertises the kYR_Manager bit and aggregates downstream space
229 * upward — that is precisely what makes it a MANAGER (no upstream) or, when
230 * `brix_cms_manager` is also set here, a SUB-MANAGER (registers UP). Role
231 * is then read purely off the main conf at worker init (connect.c
232 * brix_cms_role_worker_init). An explicit `brix_manager_mode off` in the
233 * same block still wins: only flip the flag while it is UNSET so the
234 * operator can always override the auto-derivation.
235 */
236 bcf = ngx_stream_conf_get_module_srv_conf(cf, ngx_stream_brix_module);
237 if (bcf != NULL && bcf->manager_mode == NGX_CONF_UNSET) {
238 bcf->manager_mode = 1;
239 BRIX_CONF_LOG(NGX_LOG_NOTICE, cf, 0,
240 "brix: brix_cms_server on -> auto-enabling manager "
241 "mode for this block (cmsd manager/sub-manager)");
242 }

244 return NGX_CONF_OK;

248/* brix_cms_srv_set_allow — parse a CIDR allowlist of permitted
249 * data-node IPs. Mirrors brix_admin_set_allow (dashboard/api_admin.c):
250 * accumulate ngx_cidr_t entries that the accept-time gate matches the peer
251 * address against. Compatible with vanilla nodes — a node from a trusted
252 * subnet connects unchanged. */
253static char *
254brix_cms_srv_set_allow(ngx_conf_t *cf, ngx_command_t *cmd, void *conf_ptr)
255{
256 ngx_stream_brix_cms_srv_conf_t *conf = conf_ptr;
257 ngx_str_t *value = cf->args->elts;
258 ngx_uint_t i;
259
260 (void) cmd;
261
262 if (conf->allow == NULL) {
263 conf->allow = ngx_array_create(cf->pool, cf->args->nelts - 1,
264 sizeof(ngx_cidr_t));
265 if (conf->allow == NULL) {
266 return NGX_CONF_ERROR;
267 }
268 }
269
270 for (i = 1; i < cf->args->nelts; i++) {
271 ngx_cidr_t *cidr = ngx_array_push(conf->allow);
272 ngx_int_t rc;
273 if (cidr == NULL) {
274 return NGX_CONF_ERROR;
275 }
276 rc = ngx_ptocidr(&value[i], cidr);
277 if (rc == NGX_ERROR) {
278 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
279 "invalid CIDR \"%V\" in brix_cms_server_allow", &value[i]);
280 return NGX_CONF_ERROR;
281 }
282 if (rc == NGX_DONE) {
283 BRIX_CONF_LOG(NGX_LOG_WARN, cf, 0,
284 "low address bits of \"%V\" in brix_cms_server_allow "
285 "were ignored", &value[i]);
286 }
287 }
288 return NGX_CONF_OK;
289}
