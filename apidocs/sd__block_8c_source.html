1/*
2 * sd_block.c — block-device storage driver (shared by the nginx server and the
3 * userland clients).
4 *
5 * WHAT: brix_sd_block_driver — a backend for writing/reading a raw block
6 * device (or a file used as one) in place. The raw byte I/O is identical to
7 * POSIX, so the pread/pwrite/preadv/fsync slots delegate to the POSIX
8 * driver; the only block-specific behaviour is `fstat`, which reports the
9 * true device capacity via BLKGETSIZE64 (a block device's struct.st_size is
10 * 0), and `open`, which never creates or truncates the device.
11 *
12 * SERVER PLANE (module build only): a block device has no directory
13 * namespace, so the export presents a FIXED-EXTENT namespace — the device
14 * capacity is divided into equal-size extents, each a logical object named
15 * by its 0-based index ("/0", "/1", ...). "/" is the namespace root that
16 * lists them. Opening "/N" returns the device fd WINDOWED to the extent's
17 * byte range [N*extent_size, (N+1)*extent_size); every read/write is
18 * shifted by the extent base and clamped to the extent length, so one
19 * object can never read or scribble into its neighbour. extent_size == 0
20 * (the default — the directive carries no block_size) makes the whole
21 * device a single extent "/0".
22 * WHY: block was previously implemented only in client/lib/vfs_block.c — a
23 * second storage driver outside src/. This is the single home: both the
24 * client (block:// copy endpoints) and a block-backed server export use the
25 * same driver. ngx-free (dual-build via sd.h's XRDPROTO_NO_NGX fallback) so
26 * the client links the raw-I/O surface from libxrdproto; the namespace /
27 * instance ops (which need an nginx pool + log) compile only in the module.
28 * HOW: flat, POD-pointer-only vtable. A per-open object carries its extent
29 * window (sd_block_obj_t base/len); a NULL window (the client/unconfined
30 * path, which wraps a bare fd) means "absolute offset, no clamp" so the raw
31 * ops serve both worlds from one implementation.
32 */

46/* Per-open extent window (obj->state). base/len confine an opened extent to its
47 * slice of the device; a NULL obj->state is the client/unconfined path (a bare
48 * fd wrapped for raw I/O) where offsets are absolute and unclamped. */
49typedef struct {
50 off_t base; /* absolute device offset of this extent */
51 off_t len; /* extent length in bytes (device tail may be short) */
52} sd_block_obj_t;
53
54/* Trailing-cap on a single windowed preadv so the iovec trim uses a bounded
55 * stack copy; a larger iovcnt is honoured as a (legal) short read. */
56#define BRIX_BLOCK_IOV_MAX 64
57
58/* sd_block_read_window — translate a logical read [*off, *len) within an extent
59 * to an absolute device range. Returns 0 when the request starts at/after the
60 * extent end (the caller returns a 0-byte EOF read), else clamps *len to the
61 * extent tail and shifts *off by the extent base. os == NULL (client path) is a
62 * no-op pass-through: the offset stays absolute. */
63static int
64sd_block_read_window(const sd_block_obj_t *os, off_t *off, size_t *len)
65{
66 off_t avail;
67
68 if (os == NULL) {
69 return 1; /* unconfined: absolute offset */
70 }
71 if (*off < 0 || *off >= os->len) {
72 return 0; /* past the extent → EOF */
73 }
74 avail = os->len - *off;
75 if ((off_t) *len > avail) {
76 *len = (size_t) avail;
77 }
78 *off += os->base;
79 return 1;
80}
81
82/* Raw byte I/O == POSIX: delegate to the POSIX driver's slots so the syscall
83 * loop policy stays single-sourced (vfs_core) and there is no second copy. For a
84 * server extent the offset is first windowed; the client path (state == NULL)
85 * passes straight through. */
86static ssize_t
87sd_block_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
88{
89 if (!sd_block_read_window(obj->state, &off, &len)) {
90 return 0; /* read wholly past the extent → EOF */
91 }
92 return brix_sd_posix_driver.pread(obj, buf, len, off);
93}

95/* sd_block_pwrite — a fixed extent cannot grow: a write that starts at/after the
96 * extent end, or that would cross the extent boundary, is refused with ENOSPC
97 * (never silently truncated into the neighbouring extent). */
98static ssize_t
99sd_block_pwrite(brix_sd_obj_t *obj, const void *buf, size_t len, off_t off)
100{
101 const sd_block_obj_t *os = obj->state;
102
103 if (os != NULL) {
104 if (off < 0 || off >= os->len
105 || (off_t) len > os->len - off)
106 {
107 errno = ENOSPC;
108 return -1;
109 }
110 off += os->base;
111 }
112 return brix_sd_posix_driver.pwrite(obj, buf, len, off);
113}

115/* sd_block_preadv_window — the vectored counterpart of sd_block_read_window:
116 * shift the base offset and trim the iovec run so the total never crosses the
117 * extent end. The client path (state == NULL) delegates unmodified. */
118static ssize_t
119sd_block_preadv_window(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
120 off_t off, int flags, int has_flags)
121{
122 const sd_block_obj_t *os = obj->state;
123 struct iovec tmp[BRIX_BLOCK_IOV_MAX];
124 off_t remaining;
125 int i, n;
126
127 if (os == NULL) {
128 return has_flags
129 ? brix_sd_posix_driver.preadv2(obj, iov, iovcnt, off, flags)
130 : brix_sd_posix_driver.preadv(obj, iov, iovcnt, off);
131 }
132 if (iovcnt < 0) {
133 errno = EINVAL;
134 return -1;
135 }
136 if (off < 0 || off >= os->len) {
137 return 0; /* past the extent → EOF */
138 }
139 if (iovcnt > BRIX_BLOCK_IOV_MAX) {
140 iovcnt = BRIX_BLOCK_IOV_MAX; /* bounded; the tail is a short read */
141 }
142 remaining = os->len - off;
143 for (i = 0, n = 0; i < iovcnt && remaining > 0; i++) {
144 size_t l = iov[i].iov_len;
145
146 if ((off_t) l > remaining) {
147 l = (size_t) remaining;
148 }
149 tmp[n].iov_base = iov[i].iov_base;
150 tmp[n].iov_len = l;
151 remaining -= l;
152 n++;
153 }
154 off += os->base;
155 return has_flags
156 ? brix_sd_posix_driver.preadv2(obj, tmp, n, off, flags)
157 : brix_sd_posix_driver.preadv(obj, tmp, n, off);
158}

180/* sd_block_fstat — a server extent reports as a fixed-size regular object (its
181 * window length); the client/unconfined path reports the real file, and for a
182 * true block device (st_size == 0) queries the capacity via BLKGETSIZE64. */
183static ngx_int_t
184sd_block_fstat(brix_sd_obj_t *obj, brix_sd_stat_t *out)
185{
186 const sd_block_obj_t *os = obj->state;
187 struct stat sb;
188
189 if (fstat(obj->fd, &sb) != 0) {
190 return NGX_ERROR;
191 }
192 ngx_memzero(out, sizeof(*out));
193 out->mtime = sb.st_mtime;
194 out->ctime = sb.st_ctime;
195 out->ino = sb.st_ino;
196
197 if (os != NULL) {
198 out->size = os->len;
199 out->mode = S_IFREG | 0644;
200 out->is_reg = 1;
201 return NGX_OK;
202 }
203
204 out->size = sb.st_size;
205 out->mode = sb.st_mode;
206 out->is_dir = S_ISDIR(sb.st_mode) ? 1 : 0;
207 out->is_reg = S_ISREG(sb.st_mode) ? 1 : 0;
208
209#ifdef BLKGETSIZE64
210 if (S_ISBLK(sb.st_mode)) {
211 uint64_t sz = 0;
212 if (ioctl(obj->fd, BLKGETSIZE64, &sz) == 0) {
213 out->size = (off_t) sz;
214 }
215 }
216#endif
217 return NGX_OK;
218}

247/* sd_block_is_root — the empty path or a lone "/" is the namespace root. */
248static int
249sd_block_is_root(const char *path)
250{
251 return path == NULL || path[0] == '\0'
252 || (path[0] == '/' && path[1] == '\0');
253}

255/* sd_block_parse_index — an extent name is a leading-'/'-optional run of
256 * decimal digits ("/0", "12"). Anything else — a non-numeric component, an
257 * embedded '/', a path-escape attempt — is not an extent: return -1 so the
258 * caller reports ENOENT (the namespace exposes ONLY the fixed extents). */
259static int64_t
260sd_block_parse_index(const char *path)
261{
262 const char *p = path;
263 int64_t idx = 0;
264
265 if (p == NULL) {
266 return -1;
267 }
268 if (*p == '/') {
269 p++;
270 }
271 if (*p == '\0') {
272 return -1; /* the root dir, not an extent */
273 }
274 for (; *p != '\0'; p++) {
275 if (*p < '0' || *p > '9') {
276 return -1;
277 }
278 idx = idx * 10 + (*p - '0');
279 if (idx > 0x7fffffff) {
280 return -1;
281 }
282 }
283 return idx;
284}

286/* sd_block_init — probe the device capacity (BLKGETSIZE64 for a real block
287 * device, st_size for a regular file used as one) and derive the fixed extent
288 * geometry. extent_size == 0 makes the whole device one extent. */
289static ngx_int_t
290sd_block_init(brix_sd_instance_t *inst, void *driver_conf)
291{
292 const brix_sd_block_conf_t *conf = driver_conf;
293 sd_block_state_t *st;
294 struct stat sb;
295 off_t cap;
296 size_t dlen;
297 int fd;
298
299 if (conf == NULL || conf->device == NULL || conf->device[0] == '\0') {
300 errno = EINVAL;
301 return NGX_ERROR;
302 }
303
304 fd = open(conf->device, O_RDONLY);
305 if (fd < 0) {
306 return NGX_ERROR; /* errno set by open(2) */
307 }
308 if (fstat(fd, &sb) != 0) {
309 int e = errno;
310 close(fd);
311 errno = e;
312 return NGX_ERROR;
313 }
314 cap = sb.st_size;
315#ifdef BLKGETSIZE64
316 if (S_ISBLK(sb.st_mode)) {
317 uint64_t sz = 0;
318 if (ioctl(fd, BLKGETSIZE64, &sz) == 0) {
319 cap = (off_t) sz;
320 }
321 }
322#endif
323 close(fd);
324
325 if (cap <= 0) {
326 errno = ENODEV;
327 return NGX_ERROR; /* an empty/zero-length device */
328 }
329
330 st = ngx_pcalloc(inst->pool, sizeof(*st));
331 if (st == NULL) {
332 errno = ENOMEM;
333 return NGX_ERROR;
334 }
335 dlen = ngx_strlen(conf->device);
336 st->device = ngx_pnalloc(inst->pool, dlen + 1);
337 if (st->device == NULL) {
338 errno = ENOMEM;
339 return NGX_ERROR;
340 }
341 ngx_memcpy(st->device, conf->device, dlen + 1);
342
343 st->capacity = cap;
344 st->extent_size = (conf->extent_size > 0) ? (off_t) conf->extent_size : cap;
345 st->extents = (uint32_t) ((cap + st->extent_size - 1) / st->extent_size);
346 inst->state = st;
347 return NGX_OK;
348}

350/* sd_block_open — resolve an extent name to a windowed handle on the device. A
351 * non-extent name is ENOENT; the root is EISDIR; a valid extent opens the device
352 * (never create/truncate) and binds the extent window into obj->state. */
353static brix_sd_obj_t *
354sd_block_open(brix_sd_instance_t *inst, const char *path, int sd_flags,
355 mode_t mode, int *err_out)
356{
357 sd_block_state_t *st = inst->state;
358 brix_sd_obj_t *obj;
359 sd_block_obj_t *os;
360 int64_t idx;
361 int fd;
362
363 if (sd_block_is_root(path)) {
364 if (err_out != NULL) { *err_out = EISDIR; }
365 return NULL;
366 }
367 idx = sd_block_parse_index(path);
368 if (idx < 0 || (uint32_t) idx >= st->extents) {
369 if (err_out != NULL) { *err_out = ENOENT; }
370 return NULL;
371 }
372
373 /* A device is opened in place: create/truncate/exclusive are meaningless. */
374 sd_flags &= ~(BRIX_SD_O_CREATE | BRIX_SD_O_TRUNC | BRIX_SD_O_EXCL);
375 fd = brix_sd_block_open_unconfined(st->device, sd_flags, mode);
376 if (fd < 0) {
377 if (err_out != NULL) { *err_out = errno; }
378 return NULL;
379 }
380
381 /* Heap-allocate the shell + window (ngx_calloc, NOT inst->pool): open can run
382 * on a cache-fill worker thread and inst->pool is the thread-unsafe cycle
383 * pool. heap_shell frees the shell in the adopting layer; sd_block_close
384 * frees the window. */
385 obj = ngx_calloc(sizeof(*obj), inst->log);
386 os = ngx_calloc(sizeof(*os), inst->log);
387 if (obj == NULL || os == NULL) {
388 if (obj != NULL) { ngx_free(obj); }
389 if (os != NULL) { ngx_free(os); }
390 close(fd);
391 if (err_out != NULL) { *err_out = ENOMEM; }
392 return NULL;
393 }
394
395 os->base = (off_t) idx * st->extent_size;
396 os->len = st->extent_size;
397 if (os->base + os->len > st->capacity) {
398 os->len = st->capacity - os->base; /* short device tail */
399 }
400
401 obj->driver = inst->driver;
402 obj->inst = inst;
403 obj->fd = fd;
404 obj->state = os;
405 obj->heap_shell = 1;
406 return obj;
407}

409/* sd_block_close — close the handle fd and free the extent window. */
410static ngx_int_t
411sd_block_close(brix_sd_obj_t *obj)
412{
413 ngx_int_t rc = NGX_OK;
414
415 if (obj == NULL) {
416 return NGX_OK;
417 }
418 if (obj->fd != NGX_INVALID_FILE) {
419 if (close(obj->fd) != 0) {
420 rc = NGX_ERROR;
421 }

426 obj->state = NULL;
427 }
428 return rc;
429}

431/* sd_block_stat — the root is a directory; a valid extent index is a fixed-size
432 * regular object (the last extent may be a short device tail); anything else is
433 * ENOENT. */
434static ngx_int_t
435sd_block_stat(brix_sd_instance_t *inst, const char *path, brix_sd_stat_t *out)
436{
437 sd_block_state_t *st = inst->state;
438 int64_t idx;
439
440 ngx_memzero(out, sizeof(*out));
441 if (sd_block_is_root(path)) {
442 out->mode = S_IFDIR | 0755;
443 out->is_dir = 1;
444 return NGX_OK;
445 }
446 idx = sd_block_parse_index(path);
447 if (idx < 0 || (uint32_t) idx >= st->extents) {
448 errno = ENOENT;
449 return NGX_ERROR;
450 }
451 out->mode = S_IFREG | 0644;
452 out->is_reg = 1;
453 out->size = ((uint32_t) idx == st->extents - 1)
454 ? st->capacity - (off_t) idx * st->extent_size
455 : st->extent_size;
456 return NGX_OK;
457}
