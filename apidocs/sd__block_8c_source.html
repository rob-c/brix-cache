1/*
2 * sd_block.c — block-device storage driver (shared by the nginx server and the
3 * userland clients).
4 *
5 * WHAT: brix_sd_block_driver — a backend for writing/reading a raw block
6 * device (or a file used as one) in place, plus the driver descriptor. The
7 * raw byte I/O is identical to POSIX, so the pread/pwrite/preadv/fsync
8 * slots delegate to the POSIX driver; the block-specific behaviour is
9 * `fstat`, which reports the true device capacity via BLKGETSIZE64 (a
10 * block device's struct.st_size is 0), `open`, which never creates or
11 * truncates the device, and the EXTENT WINDOW every offset is translated
12 * through.
13 *
14 * The server plane — the per-export instance and the fixed-extent
15 * namespace it synthesizes — lives in sd_block_ns.c and compiles only in
16 * the module build.
17 * WHY: block was previously implemented only in client/lib/vfs_block.c — a
18 * second storage driver outside src/. This is the single home: both the
19 * client (block:// copy endpoints) and a block-backed server export use the
20 * same driver. ngx-free (dual-build via sd.h's XRDPROTO_NO_NGX fallback) so
21 * the client links the raw-I/O surface from libxrdproto; the namespace /
22 * instance ops (which need an nginx pool + log) compile only in the module.
23 * HOW: flat, POD-pointer-only vtable. A per-open object carries its extent
24 * window (sd_block_obj_t base/len); a NULL window (the client/unconfined
25 * path, which wraps a bare fd) means "absolute offset, no clamp" so the raw
26 * ops serve both worlds from one implementation.
27 */

44/* Declared in sd_block_internal.h — sd_block_ns.c has no use for it, but the
45 * advisory slot below and the byte ops share the one translation. */
46int
47sd_block_read_window(const sd_block_obj_t *os, off_t *off, size_t *len)
48{
49 off_t avail;
50
51 if (os == NULL) {
52 return 1; /* unconfined: absolute offset */
53 }
54 if (*off < 0 || *off >= os->len) {
55 return 0; /* past the extent → EOF */
56 }
57 avail = os->len - *off;
58 if ((off_t) *len > avail) {
59 *len = (size_t) avail;
60 }
61 *off += os->base;
62 return 1;
63}

65/* Raw byte I/O == POSIX: delegate to the POSIX driver's slots so the syscall
66 * loop policy stays single-sourced (vfs_core) and there is no second copy. For a
67 * server extent the offset is first windowed; the client path (state == NULL)
68 * passes straight through. */
69static ssize_t
70sd_block_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
71{
72 if (!sd_block_read_window(obj->state, &off, &len)) {
73 return 0; /* read wholly past the extent → EOF */
74 }
75 return brix_sd_posix_driver.pread(obj, buf, len, off);
76}

78/* sd_block_pwrite — a fixed extent cannot grow: a write that starts at/after the
79 * extent end, or that would cross the extent boundary, is refused with ENOSPC
80 * (never silently truncated into the neighbouring extent). */
81static ssize_t
82sd_block_pwrite(brix_sd_obj_t *obj, const void *buf, size_t len, off_t off)
83{
84 const sd_block_obj_t *os = obj->state;
85
86 if (os != NULL) {
87 if (off < 0 || off >= os->len
88 || (off_t) len > os->len - off)
89 {
90 errno = ENOSPC;
91 return -1;
92 }
93 off += os->base;
94 }
95 return brix_sd_posix_driver.pwrite(obj, buf, len, off);
96}

98/* sd_block_preadv_window — the vectored counterpart of sd_block_read_window:
99 * shift the base offset and trim the iovec run so the total never crosses the
100 * extent end. The client path (state == NULL) delegates unmodified. */
101static ssize_t
102sd_block_preadv_window(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
103 off_t off, int flags, int has_flags)
104{
105 const sd_block_obj_t *os = obj->state;
106 struct iovec tmp[BRIX_BLOCK_IOV_MAX];
107 off_t remaining;
108 int i, n;
109
110 if (os == NULL) {
111 return has_flags
112 ? brix_sd_posix_driver.preadv2(obj, iov, iovcnt, off, flags)
113 : brix_sd_posix_driver.preadv(obj, iov, iovcnt, off);
114 }
115 if (iovcnt < 0) {
116 errno = EINVAL;
117 return -1;
118 }
119 if (off < 0 || off >= os->len) {
120 return 0; /* past the extent → EOF */
121 }
122 if (iovcnt > BRIX_BLOCK_IOV_MAX) {
123 iovcnt = BRIX_BLOCK_IOV_MAX; /* bounded; the tail is a short read */
124 }
125 remaining = os->len - off;
126 for (i = 0, n = 0; i < iovcnt && remaining > 0; i++) {
127 size_t l = iov[i].iov_len;
128
129 if ((off_t) l > remaining) {
130 l = (size_t) remaining;
131 }
132 tmp[n].iov_base = iov[i].iov_base;
133 tmp[n].iov_len = l;
134 remaining -= l;
135 n++;
136 }
137 off += os->base;
138 return has_flags
139 ? brix_sd_posix_driver.preadv2(obj, tmp, n, off, flags)
140 : brix_sd_posix_driver.preadv(obj, tmp, n, off);
141}

163/* sd_block_fstat — a server extent reports as a fixed-size regular object (its
164 * window length); the client/unconfined path reports the real file, and for a
165 * true block device (st_size == 0) queries the capacity via BLKGETSIZE64. */
166static ngx_int_t
167sd_block_fstat(brix_sd_obj_t *obj, brix_sd_stat_t *out)
168{
169 const sd_block_obj_t *os = obj->state;
170 struct stat sb;
171
172 if (fstat(obj->fd, &sb) != 0) {
173 return NGX_ERROR;
174 }
175 ngx_memzero(out, sizeof(*out));
176 out->mtime = sb.st_mtime;
177 out->ctime = sb.st_ctime;
178 out->ino = sb.st_ino;
179
180 if (os != NULL) {
181 out->size = os->len;
182 out->mode = S_IFREG | 0644;
183 out->is_reg = 1;
184 return NGX_OK;
185 }
186
187 out->size = sb.st_size;
188 out->mode = sb.st_mode;
189 out->is_dir = S_ISDIR(sb.st_mode) ? 1 : 0;
190 out->is_reg = S_ISREG(sb.st_mode) ? 1 : 0;
191
192#ifdef BLKGETSIZE64
193 if (S_ISBLK(sb.st_mode)) {
194 uint64_t sz = 0;
195 if (ioctl(obj->fd, BLKGETSIZE64, &sz) == 0) {
196 out->size = (off_t) sz;
197 }
198 }
199#endif
200 return NGX_OK;
201}

203/* sd_block_reserve — phase-107 C5: a fixed extent IS the reservation, so this
204 * is a pure capacity check. A declared final size that cannot fit the extent
205 * refuses the OPEN with ENOSPC (kXR_NoSpace / 507) before any byte lands,
206 * instead of the write that eventually crosses the boundary. A server extent
207 * checks os->len; the unconfined client handle (state == NULL) queries the
208 * device capacity via fstat, and reports nothing on a probe failure (the
209 * caller treats a non-ENOSPC errno as advisory). */
210static ngx_int_t
211sd_block_reserve(brix_sd_obj_t *obj, off_t size)
212{
213 const sd_block_obj_t *os = obj->state;
214 brix_sd_stat_t st;
215
216 if (os != NULL) {
217 if (size > os->len) {
218 errno = ENOSPC;
219 return NGX_ERROR;
220 }
221 return NGX_OK;
222 }
223 if (sd_block_fstat(obj, &st) != NGX_OK) {
224 return NGX_ERROR; /* errno from the probe: advisory */
225 }
226 if (st.size > 0 && size > st.size) {
227 errno = ENOSPC;
228 return NGX_ERROR;
229 }
230 return NGX_OK;
231}

234/* sd_block_read_sendfile_fd — hand back the device fd for zero-copy ONLY when
235 * the object's extent starts at device offset 0.
236 *
237 * WHAT: Returns obj->fd for the unconfined (client) handle and for extent 0;
238 * NGX_INVALID_FILE for every extent with a non-zero base.
239 * WHY: The fd this slot returns is consumed by callers that address it with
240 * LOGICAL offsets — brix_vfs_file_sendfile_fd asks for the whole object and
241 * the protocol handler then builds its own range from the object size. The
242 * driver never sees that range, so it cannot clamp it after the fact; a
243 * base-shifted extent handed out as a bare fd would serve the START of the
244 * device under every object's name and let a ranged GET walk into the
245 * neighbouring extent. base == 0 is exactly the condition under which
246 * logical and physical offsets coincide, so it is the whole gate. This is
247 * the common single-extent export (extent_size == 0 makes the device one
248 * object "/0"), which is where the zero-copy win actually matters.
249 * HOW: want_zerocopy is the VFS's transport verdict and is honoured first; a
250 * NULL window is the client path, whose offsets are already absolute.
251 */
252static ngx_fd_t
253sd_block_read_sendfile_fd(brix_sd_obj_t *obj, off_t off, size_t len,
254 unsigned want_zerocopy)
255{
256 const sd_block_obj_t *os = obj->state;
257
258 if (!want_zerocopy || obj->fd == NGX_INVALID_FILE) {
259 return NGX_INVALID_FILE;
260 }
261 if (os != NULL) {
262 if (os->base != 0) {
263 return NGX_INVALID_FILE; /* logical 0 is not physical 0 */
264 }
265 if (off < 0 || (off_t) len > os->len - off) {
266 return NGX_INVALID_FILE; /* the ask already leaves the extent */
267 }
268 }
269 return obj->fd;
270}

272/* sd_block_read_advise — map the backend-neutral advice onto posix_fadvise(2)
273 * over the extent's absolute device range.
274 *
275 * WHAT: Windows [off, off+len) through the extent (len == 0 ⇒ from off to the
276 * extent end) and advises the device fd over the resulting absolute range.
277 * WHY: Without this slot brix_vfs_file_read_advise is a no-op on a block
278 * export, so the sequential-serve and WILLNEED prefetch engines lose their
279 * read-ahead entirely. Advising the UNWINDOWED offset would be worse than
280 * nothing: it would warm the wrong extent's pages and evict the right
281 * ones.
282 * HOW: Advisory throughout, per the slot contract — NGX_OK whether or not the
283 * kernel honoured the hint, NGX_ERROR (errno set) only on a hard failure.
284 * posix_fadvise RETURNS the error number rather than setting errno, so it
285 * is copied into errno to keep the seam's contract. Position, size and
286 * contents are untouched.
287 */
288static ngx_int_t
289sd_block_read_advise(brix_sd_obj_t *obj, off_t off, size_t len, int advice)
290{
291 const sd_block_obj_t *os = obj->state;
292#if defined(POSIX_FADV_SEQUENTIAL)
293 int a, rc;
294#endif
295
296 if (os != NULL && len == 0) {
297 if (off < 0 || off >= os->len) {
298 return NGX_OK; /* nothing at/after the extent end */
299 }
300 len = (size_t) (os->len - off); /* "to EOF" is the EXTENT's end */
301 }
302 if (!sd_block_read_window(os, &off, &len)) {
303 return NGX_OK; /* wholly past the extent */
304 }
305
306#if defined(POSIX_FADV_SEQUENTIAL)
307 a = advice == BRIX_SD_ADV_WILLNEED ? POSIX_FADV_WILLNEED
308 : advice == BRIX_SD_ADV_RANDOM ? POSIX_FADV_RANDOM
309 : POSIX_FADV_SEQUENTIAL;
310
311 rc = posix_fadvise(obj->fd, off, (off_t) len, a);
312 if (rc != 0) {
313 errno = rc;
314 return NGX_ERROR;
315 }
316#else
317 (void) advice;
318#endif
319 return NGX_OK;
320}
