100/*
101 * WHAT: the symptom→cause rule table. Each row maps a (probe, kXR-code) pair to a
102 * verdict + a PII-free root cause + a remediation.
103 * WHY: turns a raw server status ("kXR_NotAuthorized on write") into an actionable
104 * diagnosis ("export is read-only → enable allow_write"). The mapping is keyed on
105 * BOTH the probe and the code because the same code means different things per
106 * subsystem (NotAuthorized on read = ACL; on write = read-only/scope).
107 * HOW: dx_classify scans top-to-bottom for the first row whose probe matches (or is the
108 * NULL wildcard) and whose kxr matches (or is the DX_ANY wildcard). Unmatched codes
109 * fall back to a generic finding so we never invent guidance we are unsure of.
110 */

114/* Append a classified finding; escalate the endpoint's status to its severity. */
115void
116dx_record(doctor_ep *e, const dx_note *n)
117{
118 dx_finding *f;
119 if (e->ndx >= DOC_MAXDX) {
120 return;
121 }
122 f = &e->dx[e->ndx++];
123 snprintf(f->probe, sizeof(f->probe), "%s", n->probe);
124 f->verdict = n->verdict;
125 f->kxr = n->kxr;
126 snprintf(f->cause, sizeof(f->cause), "%s", n->cause ? n->cause : "");
127 snprintf(f->remedy, sizeof(f->remedy), "%s", n->remedy ? n->remedy : "");
128 if (n->verdict > e->status) {
129 e->status = n->verdict;
130 }
131}

134/*
135 * Classify a failed probe (st->kxr) into a finding via DX_RULES, with a graceful
136 * generic fallback for codes we have no specific rule for. A successful probe
137 * (kxr==0, return 0 → caller records DX_OK) is handled by the caller.
138 */
139void
140dx_record_status(doctor_ep *e, const char *probe, const brix_status *st)
141{
142 size_t i;
143 int kxr = st ? st->kxr : 0;
144 for (i = 0; i < sizeof(DX_RULES) / sizeof(DX_RULES[0]); i++) {
145 const dx_rule *r = &DX_RULES[i];
146 if (r->probe != NULL && strcmp(r->probe, probe) != 0) {
147 continue;
148 }
149 if (r->kxr != DX_ANY && r->kxr != kxr) {
150 continue;
151 }
152 dx_record(e, &(dx_note){ probe, r->sev, kxr, r->cause, r->remedy });
153 return;
154 }
155 {
156 /* generic fallback — name the code, give conservative guidance. */
157 char cause[160];
158 if (kxr > 0) {
159 snprintf(cause, sizeof(cause), "%s probe failed: server returned %s",
160 probe, brix_kxr_name(kxr));
161 } else {
162 /* PII: never echo st->msg — server wire text may carry a path. */
163 snprintf(cause, sizeof(cause), "%s probe failed (local/transport error)",
164 probe);
165 }
166 dx_record(e, &(dx_note){ probe, DX_FAIL, kxr, cause,
167 "inspect the server logs for this operation" });
168 }
169}

172/* Is the target a loopback host? Write-mutation probes run unconditionally only here;
173 * for any other host they additionally require the explicit --i-am-authorized gate. */
174int
175dx_is_loopback(const char *host)
176{
177 /* Exact match only — a prefix like "127." would match "127.attacker.com" and
178 * wrongly enable mutating probes on a remote host. Anything not exactly a
179 * loopback literal must pass --i-am-authorized instead. */
180 return host != NULL && (strcmp(host, "127.0.0.1") == 0
181 || strcmp(host, "localhost") == 0 || strcmp(host, "::1") == 0);
182}

253/*
254 * WHAT: the xrddiag option table — every non-connection flag/option as a
255 * {name, kind, diag_args-field-offset} descriptor.
256 * WHY: replaces main()'s 19-arm strcmp ladder; adding an option is one row.
257 * HOW: dx_opt_parse strcmp-matches a row, dx_opt_apply writes the field:
258 * FLAG sets the int to 1, CLEAR to 0, INT/STR consume the next argv
259 * word (a missing value fails the match, so the caller reports the
260 * same "unknown option" error the old fall-through ladder produced).
261 */
262typedef enum {
263 DXO_FLAG, /* presence flag: int field = 1 */
264 DXO_CLEAR, /* negative flag: int field = 0 */
265 DXO_INT, /* atoi(next argv) into an int field */
266 DXO_STR, /* next argv into a const char * field */
267} dx_opt_kind;
268

270 const char *name; /* exact option spelling */
271 dx_opt_kind kind; /* how the value is taken and stored */
272 size_t off; /* offsetof(diag_args, <target field>) */
273} dx_opt_t;

308/*
309 * WHAT: apply one matched option descriptor to the args struct.
310 * WHY: keeps the field write (and value consumption) in one place so the
311 * descriptor table stays declarative.
312 * HOW: flags write 1/0 in place; INT/STR advance *i and consume argv[*i].
313 * Returns 0, or -1 when a value-taking option has no next word (the
314 * caller then reports it exactly like an unknown option — the same
315 * behavior the original fall-through ladder had).
316 */
317static int
318dx_opt_apply(diag_args *a, const dx_opt_t *o, int argc, char **argv, int *i)
319{
320 char *base = (char *) a;
321
322 if (o->kind == DXO_FLAG) {
323 *(int *) (base + o->off) = 1;
324 return 0;
325 }
326 if (o->kind == DXO_CLEAR) {
327 *(int *) (base + o->off) = 0;
328 return 0;
329 }
330 if (*i + 1 >= argc) {
331 return -1;
332 }
333 ++*i;
334 if (o->kind == DXO_INT) {
335 *(int *) (base + o->off) = atoi(argv[*i]);
336 } else {
337 *(const char **) (base + o->off) = argv[*i];
338 }
339 return 0;
340}

343/*
344 * WHAT: parse one xrddiag-specific option at argv[*i].
345 * WHY: main()'s per-option knowledge lives in DX_OPTS; this is just lookup.
346 * HOW: exact match against the table, then the one prefix-form option
347 * (--prometheus=PATH). Returns 0 on success (advancing *i past any
348 * consumed value), -1 for an unknown option or a missing value.
349 */
350static int
351dx_opt_parse(diag_args *a, int argc, char **argv, int *i)
352{
353 const char *p = argv[*i];
354 size_t k;
355
356 for (k = 0; k < sizeof(DX_OPTS) / sizeof(DX_OPTS[0]); k++) {
357 if (strcmp(p, DX_OPTS[k].name) == 0) {
358 return dx_opt_apply(a, &DX_OPTS[k], argc, argv, i);
359 }
360 }
361 if (strncmp(p, "--prometheus=", 13) == 0) {
362 a->watch_prom = 1;
363 a->prom_path = p + 13;
364 return 0;
365 }
366 return -1;
367}

370/*
371 * WHAT: collected positional (non-option) arguments — the URL list.
372 * WHY: bundles the array + count so the argv walker stays under the
373 * parameter cap while main keeps ownership of the storage.
374 */
375typedef struct {
376 const char *v[8]; /* positional args in order (URL cap) */
377 int n; /* how many of v[] are filled */
378} dx_pos_t;
379
380
381/*
382 * WHAT: walk argv[2..] splitting options from positional URLs.
383 * WHY: isolates the whole option/positional scan (including the shared
384 * connection-option parser and its --help path) out of main.
385 * HOW: option-shaped words go to brix_opts_parse_arg first, then the
386 * DX_OPTS table; everything else lands in pos (capped). Returns 0 to
387 * continue, or 1 with *rc set when main should exit immediately
388 * (--help → 0, unknown option / too many URLs → 50).
389 */
390static int
391dx_parse_argv(diag_args *a, int argc, char **argv, dx_pos_t *pos, int *rc)
392{
393 int i;
394
395 for (i = 2; i < argc; i++) {
396 const char *p = argv[i];
397 if (p[0] == '-' && p[1] != '\0' && strcmp(p, "-") != 0) {
398 int pr = brix_opts_parse_arg(&a->conn, argc, argv, &i);
399 if (pr == 2) { /* --help */
400 usage_fp(stdout, argv[0]);
401 *rc = 0;
402 return 1;
403 }
404 if (pr) {
405 continue;
406 }
407 if (dx_opt_parse(a, argc, argv, &i) != 0) {
408 fprintf(stderr, "xrddiag: unknown option '%s'\n", p);
409 usage(argv[0]);
410 *rc = 50;
411 return 1;
412 }
413 } else if (pos->n < (int) (sizeof(pos->v) / sizeof(pos->v[0]))) {
414 pos->v[pos->n++] = p;
415 } else {
416 fprintf(stderr, "xrddiag: too many arguments (max %zu URLs)\n",
417 sizeof(pos->v) / sizeof(pos->v[0]));
418 *rc = 50;
419 return 1;
420 }
421 }
422 return 0;
423}

426/*
427 * WHAT: route the multi-call personalities before normal parsing.
428 * WHY: the absorbed micro-tools stay invocable via symlinks (argv[0]) and
429 * double as xrddiag subcommands (argv[1]). The installed link names
430 * carry a -brix suffix (wait41-brix, mpxstats-brix) so the client RPM
431 * co-installs with the stock xrootd server RPMs, which own the bare
432 * names; the bare spellings remain reachable as subcommands.
433 * HOW: match argv[0]'s basename against the installed link names, then
434 * argv[1] against the subcommand aliases (shifting argv by one).
435 * Returns 1 with *rc set when a personality ran, 0 to continue as
436 * xrddiag.
437 */
438static int
439dx_personality_route(int argc, char **argv, int *rc)
440{
441 /* strip any co-install "brix-" prefix so brix-xrdqstats still routes to the
442 * xrdqstats personality (busybox dispatch on basename(argv[0])). */
443 const char *base = brix_prog_strip_compat(brix_prog_base(argv[0]));
444
445 /* The bare-name personalities carry a -brix suffix in the stock package
446 * (wait41-brix); the compat package spells them brix-wait41, which strips to
447 * the bare "wait41". Accept both so either install layout routes correctly. */
448 if (strcmp(base, "xrdqstats") == 0) { *rc = brix_qstats_main(argc, argv); return 1; }
449 if (strcmp(base, "wait41-brix") == 0 || strcmp(base, "wait41") == 0) { *rc = brix_wait41_main(argc, argv); return 1; }
450 if (strcmp(base, "mpxstats-brix") == 0 || strcmp(base, "mpxstats") == 0) { *rc = brix_mpxstats_main(argc, argv); return 1; }
451 if (argc >= 2) {
452 if (strcmp(argv[1], "qstats") == 0) { *rc = brix_qstats_main(argc - 1, argv + 1); return 1; }
453 if (strcmp(argv[1], "wait41") == 0) { *rc = brix_wait41_main(argc - 1, argv + 1); return 1; }
454 if (strcmp(argv[1], "mpxstats") == 0) { *rc = brix_mpxstats_main(argc - 1, argv + 1); return 1; }
455 }
456 return 0;
457}

488/*
489 * WHAT: report an unrecognised subcommand and exit-code 50.
490 * WHY: spec WS-7: every unknown-command site must suggest a close match
491 * for interactive users (TTY-gated, pipeline-silent per C3).
492 * HOW: static NULL-terminated table of all xrddiag subcommands feeds
493 * brix_suggest for the did-you-mean hint, then usage to stderr.
494 */
495static int
496dx_unknown_sub(const char *sub, const char *prog)
497{
498 static const char *const XRDDIAG_CMDS[] = {
499 "remote-doctor", "tpc-egress", "watch", "check", "bench", "metabench",
500 "topology", "status", "compare", "probe-robustness", "replay",
501 "srr", "tape", "qstats", "wait41", "mpxstats", NULL
502 };
503 const char *suggestion = brix_suggest(sub, XRDDIAG_CMDS);
504
505 fprintf(stderr, "xrddiag: unknown subcommand '%s'\n", sub);
506 if (suggestion != NULL) {
507 brix_cli_hint("hint: did you mean '%s'?\n", suggestion);
508 }
509 usage(prog);
510 return 50;
511}

514/*
515 * WHAT: run the named subcommand against the parsed args.
516 * WHY: table lookup keeps main a straight parse→dispatch pipeline.
517 * HOW: first exact DX_CMDS match wins; a miss goes through the
518 * did-you-mean reporter.
519 */
520static int
521dx_dispatch(const char *sub, const diag_args *a, const char *prog)
522{
523 size_t k;
524
525 for (k = 0; k < sizeof(DX_CMDS) / sizeof(DX_CMDS[0]); k++) {
526 if (strcmp(sub, DX_CMDS[k].name) == 0) {
527 return DX_CMDS[k].fn(a);
528 }
529 }
530 return dx_unknown_sub(sub, prog);
531}
