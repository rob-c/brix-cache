1/*
2 * sd_cache_forward.c — namespace / xattr / dir / staged-write forwarders.
3 *
4 * WHAT: The read-cache decorator's delegating vtable slots — every write /
5 * namespace / xattr / directory / staged-write op forwards straight to the
6 * wrapped source (a write also invalidates the cached copy). The one
7 * non-trivial slot is sd_cache_stat, which answers from a COMPLETE cinfo
8 * to keep a warm-object stat off the source.
9 *
10 * WHY: Split from sd_cache.c (phase-79) to keep every cache file under the
11 * ~500-line, one-concept-per-file cap. These forwarders are the "cache is
12 * transport-transparent above the seam" half of the driver — reviewable
13 * apart from the interposed read-open decision tree and the fill / slice
14 * machinery. The read cache only interposes READ-open; everything here is
15 * pass-through-plus-evict.
16 *
17 * HOW: Each slot reaches the source instance through SD_CACHE_SRC / the state's
18 * ->source (sd_cache_internal.h) and dispatches through the source
19 * driver's matching slot, returning NGX_ERROR / ENOTSUP when the source
20 * lacks it. Every slot that mutates the object the store holds a copy of —
21 * unlink, rename (both keys), truncate_path, server_copy (the DESTINATION),
22 * setattr, setxattr, removexattr, staged-open — additionally
23 * brix_cstore_evict the affected key(s) on success, and only on success.
24 * All slots are non-static — the driver vtable in
25 * sd_cache.c wires them by name through sd_cache_internal.h. ZERO
26 * behaviour change from the pre-split file.
27 */

45/* ---- namespace / xattr / dir forwarders (delegate to the source) ----------
46 *
47 * Every op below exists in a PLAIN and a CRED-scoped form, and both are one
48 * `_common` body taking a (possibly NULL) credential.
49 *
50 * WHY the cred form has to exist at all: brix_sd_<op>_maybe_cred decides
51 * cred-slot vs plain-slot vs deny-refusal by looking at THE INSTANCE IT IS
52 * CALLED ON. A decorator publishing `.mkdir` and no `.mkdir_cred` therefore
53 * reads, one tier up, exactly like a driver with no per-user support — whatever
54 * the source can actually do. The VFS namespace sites work around that today by
55 * unwrapping to the LEAF (brix_vfs_ns_leaf) and dispatching there, which costs
56 * them the decorator itself: the cache's stat-from-cinfo shortcut, and — the
57 * part that bites — its automatic eviction, which each bypassing site then has
58 * to re-add by hand as a brix_sd_cache_evict call. Restoring the twins removes
59 * the reason for the bypass, and makes a decorator that is dispatched on
60 * DIRECTLY (any caller that is not the ns gate) carry the credential correctly.
61 *
62 * HOW: each cred slot re-dispatches through brix_sd_<op>_maybe_cred against the
63 * SOURCE, so the source's own cred/plain/deny decision is the one that governs
64 * — the decorator adds no policy, it only stops erasing the credential. The
65 * plain slots pass cred=NULL and land on the identical path they always took.
66 * ENOTSUP (not the forwarder's ENOSYS) is preserved for the xattr ops, whose
67 * callers read it as "this filesystem has no extended attributes". */

69static ngx_int_t
70sd_cache_stat_common(brix_sd_instance_t *inst, const char *path,
71 brix_sd_stat_t *out, const brix_sd_cred_t *cred)
72{
73 sd_cache_inst_state *st = SD_CACHE_ST(inst);
74 brix_sd_instance_t *s = SD_CACHE_SRC(inst);
75 brix_cache_cinfo_t ci;
76
77 /* A COMPLETE cached object answers stat from its cinfo — the same
78 * authoritative-hit doctrine as open() (section 6.4), and it keeps a stat of
79 * a warm object off the source (a remote source would otherwise take a
80 * blocking wire round-trip on the caller's thread — the event loop for the
81 * kXR_open pre-flight probe). A miss/partial falls through to the source.
82 * A credential does not change that: sd_cache_open_common serves a complete
83 * hit from the store under a cred too, so diverging here would make stat
84 * stricter than the open it is a pre-flight for. */
85 if (brix_cstore_cinfo_load(&st->cstore, path, &ci) == NGX_OK
86 && (ci.flags & BRIX_CINFO_F_COMPLETE))
87 {
88 ngx_memzero(out, sizeof(*out));
89 out->size = (off_t) ci.size;
90 out->mtime = (time_t) ci.mtime;
91 out->ctime = (time_t) ci.mtime;
92 out->mode = (mode_t) S_IFREG
93 | (mode_t) ((ci.mode != 0) ? (ci.mode & 0777) : 0644);
94 out->is_reg = 1;
95 return NGX_OK;
96 }
97
98 return brix_sd_stat_maybe_cred(s, path, out, cred);
99}

222/* Phase-107 C4: the batch is a relay, never a loop the decorator invents — a
223 * source without the slot pair is ENOTSUP so brix_vfs_rmtree_dispatch keeps
224 * the per-key walk (and with it this decorator's own unlink+evict). On a
225 * relayed batch the store's copies of the removed keys go the same way
226 * sd_cache_unlink_common's does, over the leading `done` entries a failed
227 * batch actually attempted. */
228static ngx_int_t
229sd_cache_unlink_many_common(brix_sd_instance_t *inst,
230 brix_sd_unlink_batch_t *b, const brix_sd_cred_t *cred)
231{
232 sd_cache_inst_state *st = SD_CACHE_ST(inst);
233 ngx_int_t rc;
234 size_t i, attempted;
235
236 if (st->source->driver->unlink_many == NULL
237 && st->source->driver->unlink_many_cred == NULL)
238 {
239 errno = ENOTSUP;
240 return NGX_ERROR;
241 }
242 rc = brix_sd_unlink_many_maybe_cred(st->source, b, cred);
243 attempted = (rc == NGX_OK) ? b->n : b->done;
244 for (i = 0; i < attempted; i++) {
245 if (b->errs[i] == 0) {
246 (void) brix_cstore_evict(&st->cstore, b->paths[i]);
247 }
248 }
249 return rc;
250}
