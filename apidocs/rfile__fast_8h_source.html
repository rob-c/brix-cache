1/*
2 * rfile_fast.h — private contract between the two halves of the fast path.
3 *
4 * WHAT: The ring/job/state types and the four entry points rfile_fast.c uses to
5 * drive the pipeline that lives in rfile_fast_ring.c.
6 * WHY: rfile_fast.c had grown past the size limit carrying both the session
7 * (eligibility, attach/detach, socket mode, the env tunables) and the
8 * engine (the slot ring and the fill/wait/deliver loop over it). They
9 * share only these types, so the seam costs one header and no behaviour.
10 * HOW: Private to client/lib/fs — nothing outside the fast path includes it;
11 * the public entry point is still brix_rfile_stream_fast() in brix.h.
12 */

22/* Slot lifecycle: IDLE (no request) → BUSY (submitted) → READY (reply parked by
23 * the loop thread) → IDLE again once the calling thread has drained it. */
24enum { FAST_SLOT_IDLE = 0, FAST_SLOT_BUSY, FAST_SLOT_READY };

26typedef struct fast_ring fast_ring;

28/* One outstanding kXR_read plus the reply the loop thread parks in it. `buf`
29 * is the slot's own landing buffer: the engine reads the reply body straight
30 * into it (brix_aio_opts.dst), so a byte is copied out of the socket once and
31 * then handed to the sink — no per-chunk malloc and no intermediate staging. */
32typedef struct {
33 fast_ring *ring; /* owner, for lock/condvar (set once at init) */
34 uint8_t *buf; /* landing buffer, `chunk` bytes, owned by the ring */
35 int64_t offset; /* absolute file offset this request covers */
36 uint32_t want; /* bytes requested */
37 uint32_t blen; /* bytes the reply actually delivered into buf */
38 int state; /* FAST_SLOT_* */
39 int failed; /* 1 ⇒ st below explains why */
40 brix_status st;
41} fast_slot;
42
43struct fast_ring {
44 pthread_mutex_t lock;
45 pthread_cond_t cv;
46 fast_slot *slots;
47 unsigned depth;
48 uint8_t *pool; /* depth × chunk bytes, carved into the slots */
49};

51/* The immutable description of one pipelined stream, bundled so the driver
52 * helpers stay under the five-parameter gate. */
53typedef struct {
54 brix_rfile *rf;
55 brix_rfile_sink_fn sink;
56 void *arg;
57 int64_t offset; /* first byte of the range */
58 int64_t limit; /* bytes to stream, or -1 = to EOF */
59 uint32_t chunk;
60 unsigned depth;
61 int deadline_ms;
62} fast_job;
63
64/* Mutable progress of one pipelined stream. */
65typedef struct {
66 int64_t issued; /* bytes covered by requests already submitted */
67 int64_t moved; /* bytes handed to the sink */
68 unsigned head; /* ring index of the next chunk to deliver */
69 unsigned tail; /* ring index of the next chunk to submit */
70 unsigned inflight; /* slots between head and tail */
71 int eof; /* a short reply proved end-of-file */
72 int stop; /* the sink asked to stop (success) */
73 int rc; /* 0 so far, -1 once *st is set */
74} fast_state;
