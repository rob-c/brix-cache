199/*
200 * Flow-id parsing fields and IPv6 flow-label masks.
201 * The compact flow ID stores experiment and activity contiguously; packet
202 * labels use the separate SciTags layout in brix_pmark_flowlabel_encode().
203 * BRIX_IPV6_FL_MASK: 20-bit field mask (0x000FFFFF)
204 * BRIX_IPV6_FL_ENTROPY_MASK: Entropy bit positions (bits 0,1,8,18,19)
205 * BRIX_IPV6_FL_ACTIVITY_BITS: Number of activity bits (6)
206 * BRIX_IPV6_FL_ACTIVITY_MASK: Activity field mask (0x3F)
207 * BRIX_IPV6_FL_EXP_SHIFT: Experiment ID shift (6 bits)
208 * BRIX_IPV6_FL_EXP_MASK: Experiment field mask (0x3FF)
209 */
210#define BRIX_IPV6_FL_MASK 0x000FFFFFu /* 20-bit flow label field */
211#define BRIX_IPV6_FL_ENTROPY_MASK 0x000C0103u /* E bits 0,1,8,18,19 */
212#define BRIX_IPV6_FL_ACTIVITY_BITS 6 /* low 6 bits = activity */
213#define BRIX_IPV6_FL_ACTIVITY_MASK 0x3F /* activity mask */
214#define BRIX_IPV6_FL_EXP_SHIFT 6 /* exp starts at bit 6 */
215#define BRIX_IPV6_FL_EXP_MASK 0x3FF /* 10-bit experiment mask */
