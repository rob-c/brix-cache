1/* WHAT: Parses "brix_cms_manager host:port [host:port ...]" (repeatable) and */
2/* stores every resolved address so the heartbeat client can log into ALL */
3/* redundant CMS managers concurrently (stock ManList parity). */
4/* WHY: The CMS heartbeat subsystem needs the full redundant manager set: each */
5/* endpoint gets its own login + heartbeat link and locate requests rotate */
6/* across the live links, so one dead manager no longer stalls discovery. */

11/* cms_manager_add — resolve ONE host:port argument and append it to the
12 * managers array. Returns an nginx conf-parse verdict string. */
13static char *
14cms_manager_add(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *xcf,
15 ngx_str_t *value)
16{
17 ngx_url_t url;
18 ngx_addr_t *addr;
19 ngx_uint_t i;
20 brix_cms_manager_ent_t *ent, *ents;
21
22 if (xcf->cms.managers->nelts >= NGX_BRIX_CMS_MAX_MANAGERS) {
23 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
24 "brix_cms_manager: more than %d managers", NGX_BRIX_CMS_MAX_MANAGERS);
25 return NGX_CONF_ERROR;
26 }
27
28 ent = ngx_array_push(xcf->cms.managers);
29 if (ent == NULL) {
30 return NGX_CONF_ERROR;
31 }
32
33 if (brix_copy_conf_string(cf, value, &ent->raw) != NGX_CONF_OK) {
34 return NGX_CONF_ERROR;
35 }
36
37 ngx_memzero(&url, sizeof(url));
38 url.url = ent->raw;
39 url.default_port = 0;
40
41 if (ngx_parse_url(cf->pool, &url) != NGX_OK) {
42 if (url.err != NULL) {
43 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
44 "brix_cms_manager: %s in \"%V\"", url.err, value);
45 }
46 return NGX_CONF_ERROR;
47 }
48
49 if (url.no_port) {
50 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
51 "brix_cms_manager: missing port in \"%V\"", value);
52 return NGX_CONF_ERROR;
53 }
54
55 if (url.naddrs == 0 || url.addrs == NULL) {
56 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
57 "brix_cms_manager: could not resolve \"%V\"", value);
58 return NGX_CONF_ERROR;
59 }
60
61 if (ngx_inet_get_port(url.addrs[0].sockaddr) == 0) {
62 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
63 "brix_cms_manager: invalid port in \"%V\"", value);
64 return NGX_CONF_ERROR;
65 }
66
67 /* Reject the same resolved endpoint twice: a duplicate would double-login
68 * (a stock manager 30s-blacklists the second identity) and skew rotation. */
69 ents = xcf->cms.managers->elts;
70 for (i = 0; i + 1 < xcf->cms.managers->nelts; i++) {
71 if (ents[i].addr->socklen == url.addrs[0].socklen
72 && ngx_memcmp(ents[i].addr->sockaddr, url.addrs[0].sockaddr,
73 url.addrs[0].socklen) == 0)
74 {
75 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
76 "brix_cms_manager: duplicate manager \"%V\"", value);
77 return NGX_CONF_ERROR;
78 }
79 }
80
81 BRIX_PCALLOC_OR_RETURN(addr, cf->pool, sizeof(ngx_addr_t), NGX_CONF_ERROR);
82
83 addr->sockaddr = ngx_pnalloc(cf->pool, url.addrs[0].socklen);
84 if (addr->sockaddr == NULL) {
85 return NGX_CONF_ERROR;
86 }
87
88 ngx_memcpy(addr->sockaddr, url.addrs[0].sockaddr, url.addrs[0].socklen);
89 addr->socklen = url.addrs[0].socklen;
90 addr->name = url.addrs[0].name;
91 ent->addr = addr;
92
93 return NGX_CONF_OK;
94}

97brix_conf_set_cms_manager(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
98{
99 ngx_stream_brix_srv_conf_t *xcf = conf;
100 ngx_str_t *value;
101 ngx_uint_t i;
102 brix_cms_manager_ent_t *ents;
103 char *rc;
104
105 value = cf->args->elts;
106 (void) cmd;
107
108 if (xcf->cms.managers == NULL) {
109 xcf->cms.managers = ngx_array_create(cf->pool, 4,
110 sizeof(brix_cms_manager_ent_t));
111 if (xcf->cms.managers == NULL) {
112 return NGX_CONF_ERROR;
113 }
114 }
115
116 for (i = 1; i < cf->args->nelts; i++) {
117 rc = cms_manager_add(cf, xcf, &value[i]);
118 if (rc != NGX_CONF_OK) {
119 return rc;
120 }
121 }
122
123 /* First entry stays mirrored into manager/addr: those are the "has
124 * upstream" gate and the role-log identity across the tree. */
125 ents = xcf->cms.managers->elts;
126 xcf->cms.manager = ents[0].raw;
127 xcf->cms.addr = ents[0].addr;
128
129 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
130 "brix: CMS manager configured: %V (%ui total)",
131 &xcf->cms.manager, xcf->cms.managers->nelts);
132
133 return NGX_CONF_OK;
134}
