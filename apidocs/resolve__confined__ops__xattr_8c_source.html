21/* Confined extended-attribute ops — broker-routed under impersonation.
22 *
23 * WHAT: set/get/remove/list extended attributes on a resolved (already
24 * lexically-confined) path. When impersonation map mode is active the op
25 * is routed to the privileged broker, which re-confines under its own
26 * export rootfd (openat2 RESOLVE_BENEATH) and performs the f*xattr as the
27 * mapped user — so the lock/dead-property xattr lands with, and is
28 * DAC-checked for, the real user (not the unprivileged worker). Without
29 * impersonation, the same confined-open boundary pins a descriptor and
30 * every operation uses f*xattr; a lexical prefix or an earlier stat is
31 * not sufficient protection against traversal or symlink replacement.
32 *
33 * WHY: WebDAV LOCK tokens and PROPPATCH dead-properties are stored as `user.*`
34 * xattrs on the resource. Without broker routing the worker (svc) would
35 * attempt setxattr on a file owned 0644 by the mapped user and fail EACCES
36 * — i.e. LOCK/PROPPATCH were broken under impersonation. Routing fixes
37 * that and keeps the on-disk metadata owned by the right identity.
38 *
39 * Return values mirror the POSIX *xattr contract (get/list: byte count, or the
40 * size when bufsz==0; -1/ERANGE when the caller buffer is too small).
41 */
42typedef enum {
43 XATTR_OP_SET,
44 XATTR_OP_GET,
45 XATTR_OP_REMOVE,
46 XATTR_OP_LIST,
47} xattr_confined_op_t;

58/* ---- Apply an attribute operation to an already confined descriptor ----
59 * WHAT: Return the native f*xattr result without reopening a pathname.
60 * WHY: A path-based syscall could follow a link swapped in after resolution.
61 * HOW: 1. Select the operation. 2. Pass the borrowed buffers to its fd API.
62 */
63static ssize_t
64xattr_descriptor_op(xattr_confined_op_t op, int descriptor, const xattr_io_t *io)
65{
66 switch (op) {
67 case XATTR_OP_SET:
68 return fsetxattr(descriptor, io->name, io->setval, io->len, io->flags);
69 case XATTR_OP_GET:
70 return fgetxattr(descriptor, io->name, io->buf, io->len);
71 case XATTR_OP_REMOVE:
72 return fremovexattr(descriptor, io->name);
73 default:
74 return flistxattr(descriptor, io->buf, io->len);
75 }
76}

78/* ---- Open the attribute target through the existing confinement boundary ----
79 * WHAT: Return a confined descriptor, or -1 with the authoritative open errno.
80 * WHY: Attribute writes on write-only files must not require data-read rights;
81 * directories still need the read-side open that their kernel permits.
82 * HOW: 1. Try a nonblocking read open. 2. On EACCES only, mutators may try a
83 * nonblocking write open. 3. Leave all other confinement errors unchanged.
84 */
85static int
86xattr_open_target(xattr_confined_op_t op, ngx_log_t *log,
87 const char *root_canon, const char *resolved)
88{
89 int descriptor = brix_open_confined_canon(log, root_canon, resolved,
90 O_RDONLY | O_NONBLOCK | O_CLOEXEC, 0);
91
92 if (descriptor < 0 && errno == EACCES
93 && (op == XATTR_OP_SET || op == XATTR_OP_REMOVE)) {
94 return brix_open_confined_canon(log, root_canon, resolved,
95 O_WRONLY | O_NONBLOCK | O_CLOEXEC, 0);
96 }
97 return descriptor;
98}

100/* ---- Own a confined descriptor for one non-impersonated attribute operation ----
101 * WHAT: Return the fd operation's result and preserve its failure errno.
102 * WHY: Every local attribute caller needs the same race-resistant boundary.
103 * HOW: 1. Open beneath the root. 2. Operate on that fd. 3. Close on every path.
104 */
105static ssize_t
106xattr_local_op(xattr_confined_op_t op, ngx_log_t *log, const char *root_canon,
107 const char *resolved, const xattr_io_t *io)
108{
109 int descriptor = xattr_open_target(op, log, root_canon, resolved);
110 ssize_t result;
111 int saved_errno;
112
113 if (descriptor < 0) {
114 return -1;
115 }
116 result = xattr_descriptor_op(op, descriptor, io);
117 saved_errno = errno;
118 close(descriptor);
119 if (result < 0) {
120 errno = saved_errno;
121 }
122 return result;
123}
