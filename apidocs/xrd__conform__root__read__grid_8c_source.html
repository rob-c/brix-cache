30{
31 uint64_t index;

34 if (size == 0) return 1;
35 if (pieces > size) pieces = size;
36 for (index = 0; index < pieces; index++) {
37 uint64_t begin = index * size / pieces;
38 uint64_t end = (index + 1) * size / pieces;
39 size_t length = (size_t) (end - begin);

41 brix_status status;
42 ssize_t got;

44 if (!xrd_conform_reserve(execution, length, reason, sizeof(reason)))

46 buffer = malloc(length == 0 ? 1 : length);
47 if (buffer == NULL) return -1;

49 got = brix_file_read(connection, file, (int64_t) begin, buffer, length,
50 &status);
51 if (got != (ssize_t) length
