1/* brixrpm_createrepo.c — the `brixrpm createrepo` verb (phase-104 D12.1).
2 *
3 * WHAT: scan a repo directory for *.rpm, parse each through shared/rpm and
4 * emit repodata/ via the repomd writer; keep a per-package fragment
5 * memo (`repodata/.brixrpm-cache`) so `--update` re-parses only
6 * packages whose (size, mtime) changed.
7 * WHY: parsing + sha256-ing every package is the whole cost of a rebuild —
8 * the memo turns the steady-state republish into a stat sweep.
9 * HOW: the memo is length-framed: one `pkg <size> <mtime> <prilen> <fillen>
10 * <othlen> <pkgid> <href>` line per package followed by exactly that
11 * many fragment bytes, staged + renamed like every other artifact.
12 * The memo is always rewritten (a non---update run still primes it).
13 */

50static int
51strv_push(strv_t *s, const char *str)
52{
53 if (s->n == s->cap) {
54 size_t cap = s->cap > 0 ? s->cap * 2 : 64;
55 char **nv = realloc(s->v, cap * sizeof(*nv));
56
57 if (nv == NULL) {
58 return -1;
59 }
60 s->v = nv;
61 s->cap = cap;
62 }
63 s->v[s->n] = strdup(str);
64 if (s->v[s->n] == NULL) {
65 return -1;
66 }
67 s->n++;
68 return 0;
69}

144/* One directory entry: skip the noise, recurse into subdirectories, collect
145 * the `.rpm`s. An entry that cannot be reached is a warning and a skip — a
146 * repository built from what IS readable beats no repository at all — so the
147 * only failure this returns is an allocation or a failed recursion. */
148static int
149cr_scan_child(const char *root, const char *rel, const char *dirpath,
150 const char *name, strv_t *out)
151{
152 char sub[CR_PATH_MAX], full[CR_PATH_MAX];
153 struct stat st;
154 size_t nlen = strlen(name);
155
156 if (name[0] == '.' || strcmp(name, "repodata") == 0) {
157 return 0;
158 }
159 if (snprintf(sub, sizeof(sub), "%s%s%s", rel,
160 rel[0] != '\0' ? "/" : "", name) >= (int) sizeof(sub) ||
161 snprintf(full, sizeof(full), "%s/%s", root, sub) >=
162 (int) sizeof(full)) {
163 fprintf(stderr, "brixrpm: warning: path too long under %s "
164 "(skipped)\n", dirpath);
165 return 0;
166 }
167 if (lstat(full, &st) != 0) {
168 fprintf(stderr, "brixrpm: warning: %s: %s (skipped)\n", full,
169 strerror(errno));
170 return 0;
171 }
172 if (S_ISDIR(st.st_mode)) {
173 return cr_scan(root, sub, out);
174 }
175 if (S_ISREG(st.st_mode) && nlen > 4 &&
176 strcmp(name + nlen - 4, ".rpm") == 0) {
177 return strv_push(out, sub);
178 }
179 return 0;
180}

183/* Recursive *.rpm scan. Dot entries, repodata/ and symlinked dirs are
184 * skipped; unreadable subtrees warn and continue (only the root is fatal,
185 * checked by the caller). rel is "" at the root. */
186static int
187cr_scan(const char *root, const char *rel, strv_t *out)
188{
189 char dirpath[CR_PATH_MAX];
190 DIR *d;
191 struct dirent *de;
192 int rc = 0;
193
194 if (snprintf(dirpath, sizeof(dirpath), "%s%s%s", root,
195 rel[0] != '\0' ? "/" : "", rel) >= (int) sizeof(dirpath)) {
196 return 0;
197 }
198 d = opendir(dirpath);
199 if (d == NULL) {
200 if (rel[0] != '\0') {
201 fprintf(stderr, "brixrpm: warning: %s: %s (skipped)\n", dirpath,
202 strerror(errno));
203 return 0;
204 }
205 fprintf(stderr, "brixrpm: %s: %s\n", dirpath, strerror(errno));
206 return -1;
207 }
208 while (rc == 0 && (de = readdir(d)) != NULL) {
209 rc = cr_scan_child(root, rel, dirpath, de->d_name, out);
210 }
211 closedir(d);
212 return rc;
213}

216static char *
217cr_read_block(FILE *f, size_t len)
218{
219 char *b = malloc(len + 1);
220
221 if (b == NULL || fread(b, 1, len, f) != len) {
222 free(b);
223 return NULL;
224 }
225 b[len] = '\0';

229/* One memo record: the "pkg" header line already in `line`, then the three
230 * length-counted XML fragments that follow it. 0 = `out` grew by one usable
231 * entry; -1 = the memo is malformed from here on and the caller must discard
232 * the whole thing (a partially-trusted memo would republish stale metadata
233 * for packages that have since changed). */
234static int
235cr_cache_read_ent(FILE *f, char *line, entv_t *out)
236{
237 cr_ent_t *e = entv_push(out);
238 size_t pl, fl, ol;
239 int off = 0;
240 char *nl;
241
242 if (e == NULL ||
243 sscanf(line, "pkg %lld %lld %zu %zu %zu %64s %n", &e->size,
244 &e->mtime, &pl, &fl, &ol, e->pkgid, &off) != 6 ||
245 off == 0 || line[off] == '\0') {
246 if (e != NULL) {
247 out->n--;
248 }
249 return -1;
250 }
251 nl = strchr(line + off, '\n');
252 if (nl != NULL) {
253 *nl = '\0';
254 }
255 e->href = strdup(line + off);
256 e->pri = e->href != NULL ? cr_read_block(f, pl) : NULL;
257 e->fil = e->pri != NULL ? cr_read_block(f, fl) : NULL;
258 e->oth = e->fil != NULL ? cr_read_block(f, ol) : NULL;
259
260 return e->oth != NULL ? 0 : -1;
261}

264/* Load the memo. Absent file = empty memo; a malformed memo is discarded
265 * whole (it is only a cache) with a warning. */
266static void
267cr_cache_load(const char *dir, entv_t *out)
268{
269 char path[CR_PATH_MAX], line[CR_PATH_MAX + 256];
270 FILE *f;
271 int bad = 0;
272
273 if (snprintf(path, sizeof(path), "%s/repodata/" CR_CACHE_NAME, dir) >=
274 (int) sizeof(path)) {
275 return;
276 }
277 f = fopen(path, "r");
278 if (f == NULL) {
279 return;
280 }
281 if (fgets(line, sizeof(line), f) == NULL ||
282 strcmp(line, CR_CACHE_MAGIC) != 0) {
283 bad = 1;
284 }
285 while (!bad && fgets(line, sizeof(line), f) != NULL) {
286 bad = (cr_cache_read_ent(f, line, out) != 0);
287 }
288 fclose(f);
289 if (bad) {
290 fprintf(stderr, "brixrpm: warning: %s malformed — ignoring memo\n",
291 path);
292 entv_free(out);
293 memset(out, 0, sizeof(*out));
294 }
295 qsort(out->v, out->n, sizeof(*out->v), ent_cmp);
296}

298/* Rewrite the memo (staged + rename; a torn memo must never be loadable). */
299static int
300cr_cache_save(const char *dir, const entv_t *ev)
301{
302 char fin[CR_PATH_MAX], tmp[CR_PATH_MAX], rd[CR_PATH_MAX];
303 FILE *f;
304 size_t i;
305 int ok = 1;
306
307 if (snprintf(rd, sizeof(rd), "%s/repodata", dir) >= (int) sizeof(rd) ||
308 snprintf(fin, sizeof(fin), "%s/" CR_CACHE_NAME, rd) >=
309 (int) sizeof(fin) ||
310 snprintf(tmp, sizeof(tmp), "%s/" CR_CACHE_NAME ".tmp", rd) >=
311 (int) sizeof(tmp)) {
312 return -1;
313 }
314 if (mkdir(rd, 0755) != 0 && errno != EEXIST) {
315 return -1;
316 }
317 f = fopen(tmp, "w");
318 if (f == NULL) {
319 return -1;
320 }
321 ok = fputs(CR_CACHE_MAGIC, f) >= 0;
322 for (i = 0; ok && i < ev->n; i++) {
323 const cr_ent_t *e = &ev->v[i];
324
325 ok = fprintf(f, "pkg %lld %lld %zu %zu %zu %s %s\n", e->size,
326 e->mtime, strlen(e->pri), strlen(e->fil),
327 strlen(e->oth), e->pkgid, e->href) > 0 &&
328 fputs(e->pri, f) >= 0 && fputs(e->fil, f) >= 0 &&
329 fputs(e->oth, f) >= 0;
330 }
331 if (fclose(f) != 0) {
332 ok = 0;
333 }
334 if (!ok || rename(tmp, fin) != 0) {
335 unlink(tmp);
336 return -1;
337 }
338 return 0;
339}

341/* Parse + render one package into a fresh memo entry. 0 ok / 1 skip (bad
342 * package, non-strict) / -1 fatal. */
343static int
344cr_parse_one(const brixrpm_cr_opts_t *o, const char *href, cr_ent_t *e,
345 long long mtime, cr_stats_t *st, char *err, size_t errlen)
346{
347 char full[CR_PATH_MAX];
348 brix_rpm_pkg_t *p;
349 int rc;
350
351 if (snprintf(full, sizeof(full), "%s/%s", o->dir, href) >=
352 (int) sizeof(full)) {
353 snprintf(err, errlen, "%s: path too long", href);
354 return o->strict ? -1 : 1;
355 }
356 p = brix_rpm_open(full, err, errlen);
357 if (p == NULL) {
358 return o->strict ? -1 : 1;
359 }
360 rc = brix_repomd_render(p, href, mtime, &e->pri, &e->fil, &e->oth,
361 &st->sanitized, err, errlen);
362 if (rc == 0) {
363 snprintf(e->pkgid, sizeof(e->pkgid), "%s", brix_rpm_pkgid(p));
364 }
365 brix_rpm_close(p);
366 /* A render failure is OOM *or* a malformed file list, and the file list is
367 * attacker-controlled: a DIRINDEXES entry pointing past DIRNAMES arrives
368 * here, not at brix_rpm_open(). That is the same answer as an unparsable
369 * header — warn and skip, and let --strict decide whether one bad package
370 * is allowed to fail the whole repo. */
371 return rc == 0 ? 0 : (o->strict ? -1 : 1);
372}

374/* Republish a package straight from the memo: emit its three stored fragments
375 * and carry the record into this run's memo. The strings are copied rather
376 * than aliased — the old memo is freed as a unit at the end of the run, and a
377 * fresh entry pointing into it would dangle at the next republish. */
378static int
379cr_reuse_cached(brix_repomd_t *w, const cr_ent_t *hit, entv_t *fresh,
380 char *err, size_t errlen)
381{
382 cr_ent_t *ke = entv_push(fresh);
383
384 if (ke == NULL ||
385 brix_repomd_add_fragments(w, hit->pri, hit->fil, hit->oth, err,
386 errlen) != 0) {
387 return -1;
388 }
389 *ke = *hit;
390 ke->href = strdup(hit->href);
391 ke->pri = strdup(hit->pri);
392 ke->fil = strdup(hit->fil);
393 ke->oth = strdup(hit->oth);
394
395 return (ke->href == NULL || ke->pri == NULL || ke->fil == NULL ||
396 ke->oth == NULL) ? -1 : 0;
397}

400/* Is the memo entry still the truth about this file? Size must match either
401 * way — it is the free discriminator. Then --paranoid re-reads the package
402 * and compares its sha256 against the pkgid the memo recorded, where the
403 * default trusts the mtime instead.
404 *
405 * The gap that buys: a package REWRITTEN IN PLACE at the same length with the
406 * timestamp preserved (a rebuild copied with `cp -p`, an rsync without
407 * --checksum, a tampered mirror leg) is invisible to (size, mtime) and would
408 * be republished under its old checksum — metadata that names bytes the file
409 * no longer holds. Under --paranoid that is caught, counted and re-parsed;
410 * an unchanged package still skips the header walk and the render, so the
411 * flag costs one read pass, not a rebuild. */
412static int
413cr_memo_trusts(const brixrpm_cr_opts_t *o, const cr_ent_t *hit,
414 const struct stat *fs, const char *full, cr_stats_t *st)
415{
416 char hex[65];
417 char err[512];
418
419 if (hit->size != (long long) fs->st_size) {
420 return 0;
421 }
422 if (!o->paranoid) {
423 return hit->mtime == (long long) fs->st_mtime;
424 }
425 if (brix_rpm_file_sha256(full, hex, sizeof(hex), err, sizeof(err)) != 0) {
426 fprintf(stderr, "brixrpm: warning: %s (re-parsing)\n", err);
427 return 0;
428 }
429 if (strcmp(hex, hit->pkgid) == 0) {
430 return 1;
431 }
432 fprintf(stderr, "brixrpm: warning: %s changed in place under an "
433 "unchanged (size, mtime) — re-parsing\n", hit->href);
434 st->drifted++;
435 return 0;
436}

439/* One href through the memo-or-parse decision into the writer + new memo. */
440static int
441cr_add_one(const brixrpm_cr_opts_t *o, brix_repomd_t *w, const char *href,
442 const entv_t *old, entv_t *fresh, cr_stats_t *st, char *err,
443 size_t errlen)
444{
445 char full[CR_PATH_MAX];
446 struct stat fs;
447 const cr_ent_t *hit;
448 cr_ent_t ne;
449 int rc;
450
451 if (snprintf(full, sizeof(full), "%s/%s", o->dir, href) >=
452 (int) sizeof(full) || stat(full, &fs) != 0) {
453 snprintf(err, errlen, "%s: %s", href, strerror(errno));
454 return o->strict ? -1 : 1;
455 }
456
457 hit = o->update ? ent_find(old, href) : NULL;
458 if (hit != NULL && cr_memo_trusts(o, hit, &fs, full, st)) {
459 if (cr_reuse_cached(w, hit, fresh, err, errlen) != 0) {
460 return -1;
461 }
462 st->cached++;
463 return 0;
464 }
465
466 memset(&ne, 0, sizeof(ne));
467 ne.size = (long long) fs.st_size;
468 ne.mtime = (long long) fs.st_mtime;
469 rc = cr_parse_one(o, href, &ne, (long long) fs.st_mtime, st, err,
470 errlen);
471 if (rc != 0) {
472 return rc;
473 }
474 ne.href = strdup(href);
475 if (ne.href == NULL ||
476 brix_repomd_add_fragments(w, ne.pri, ne.fil, ne.oth, err,
477 errlen) != 0) {
478 ent_free_payload(&ne);
479 return -1;
480 }
481 {
482 cr_ent_t *ke = entv_push(fresh);
483
484 if (ke == NULL) {
485 ent_free_payload(&ne);
486 return -1;
487 }
488 *ke = ne;
489 }
490 st->parsed++;
491 return 0;
492}

494/* The pump: every scanned href into the writer, memoizing as we go. */
495static int
496cr_pump(const brixrpm_cr_opts_t *o, brix_repomd_t *w, const strv_t *hrefs,
497 const entv_t *old, entv_t *fresh, cr_stats_t *st, char *err,
498 size_t errlen)
499{
500 size_t i;
501
502 for (i = 0; i < hrefs->n; i++) {
503 int rc = cr_add_one(o, w, hrefs->v[i], old, fresh, st, err, errlen);
504
505 if (rc < 0) {
506 return -1;
507 }
508 if (rc > 0) {
509 fprintf(stderr, "brixrpm: warning: %s (skipped)\n", err);
510 st->skipped++;
511 }
512 }
513 return 0;
514}

517 * packages whose bytes moved under a still-plausible (size, mtime); it is

519 * never re-hashed has nothing to say at all. */

521cr_drift_note(const cr_stats_t *st, char *buf, size_t buflen)
522{
523 if (st->drifted == 0) {
524 return "";
525 }
526 snprintf(buf, buflen, ", %u changed-in-place", st->drifted);
527 return buf;

531int
532brixrpm_createrepo(const brixrpm_cr_opts_t *o)
533{
534 char err[512] = "";
535 char note[64];
536 strv_t hrefs = {0};
537 entv_t old = {0}, fresh = {0};
538 cr_stats_t st = {0, 0, 0, 0, 0};
539 brix_repomd_t *w = NULL;
540 int rc = -1;
541
542 if (cr_scan(o->dir, "", &hrefs) == 0) {
543 qsort(hrefs.v, hrefs.n, sizeof(*hrefs.v), strp_cmp);
544 if (o->update) {
545 cr_cache_load(o->dir, &old);
546 }
547 w = brix_repomd_begin(o->dir, err, sizeof(err));
548 if (w != NULL) {
549 rc = cr_pump(o, w, &hrefs, &old, &fresh, &st, err, sizeof(err));
550 }
551 }
552
553 if (rc == 0) {
554 if (cr_cache_save(o->dir, &fresh) != 0) {
555 fprintf(stderr, "brixrpm: warning: could not write %s memo\n",
556 CR_CACHE_NAME);
557 }
558 rc = brix_repomd_finish(w, err, sizeof(err));
559 if (rc == 0) {
560 w = NULL; /* finish consumed it */
561 printf("brixrpm: createrepo %s: %u packages (%u parsed, "
562 "%u cached, %u skipped, %u paths-sanitized%s)\n", o->dir,
563 st.parsed + st.cached, st.parsed, st.cached, st.skipped,
564 st.sanitized, cr_drift_note(&st, note, sizeof(note)));
565 }
566 }
567 if (rc != 0 && err[0] != '\0') {
568 fprintf(stderr, "brixrpm: %s\n", err);
569 }
570 brix_repomd_abort(w); /* NULL-safe; live only on failure */
571 strv_free(&hrefs);
572 entv_free(&old);
573 entv_free(&fresh);
574 return rc == 0 ? 0 : 1;
575}
