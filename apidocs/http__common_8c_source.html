373/*
374 * brix_http_common_create_loc_conf() — allocate the common module's location
375 * conf and seed the embedded preamble with UNSET sentinels so parent->child
376 * inheritance (below) can tell "not configured" from an explicit value.
377 */
378static void *
379brix_http_common_create_loc_conf(ngx_conf_t *cf)
380{
381 ngx_http_brix_common_conf_t *c;
382
383 c = ngx_pcalloc(cf->pool, sizeof(*c));
384 if (c == NULL) {
385 return NULL;
386 }
387 ngx_http_brix_shared_init(&c->common);
388 return c;

391/*
392 * brix_http_common_merge_loc_conf() — inheritance-only merge: propagate parent
393 * values into unset child slots and apply NO defaults. Per-protocol defaults
394 * still come from ngx_http_brix_shared_merge() in each protocol's merge, so a
395 * field left unset here stays UNSET and lets each protocol pick its own
396 * default after adopting the unified value.
397 */
398static char *
399brix_http_common_merge_loc_conf(ngx_conf_t *cf, void *parent, void *child)
400{
401 ngx_http_brix_common_conf_t *prev = parent;
402 ngx_http_brix_common_conf_t *conf = child;
403
404 (void) cf;
405 brix_shared_adopt_unified(&conf->common, &prev->common);
406 return NGX_CONF_OK;
407}
