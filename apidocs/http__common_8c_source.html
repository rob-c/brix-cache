467/* brix_tpc_source_allow <host>... on the HTTP planes (phase-101 W4): append EVERY
468 * argument to common.tpc_source_allow. A custom setter (not the stock str_array
469 * slot, which silently keeps only the first arg) because this is a SECURITY
470 * allowlist — dropping hosts after the first would widen egress. `common` is
471 * member 0 of the common-module conf, so the cast to the preamble type is valid.
472 * Mirrors the stream-side brix_tpc_conf_source_allow. */

474brix_http_conf_tpc_source_allow(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
475{

477 ngx_str_t *value, *slot;
478 ngx_uint_t i;
479
480 (void) cmd;
481
482 if (sc->tpc_source_allow == NULL) {
483 sc->tpc_source_allow = ngx_array_create(cf->pool, 4, sizeof(ngx_str_t));
484 if (sc->tpc_source_allow == NULL) {
485 return NGX_CONF_ERROR;
486 }
487 }
488
489 value = cf->args->elts;
