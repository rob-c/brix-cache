1/*
2 * write_stream.c — streaming large plain kXR_write.
3 *
4 * WHAT: A single kXR_write whose dlen exceeds BRIX_WRITE_STREAM_CHUNK is
5 * delivered to the open file / staged writer in bounded, chunk-sized
6 * installments instead of being buffered whole and written once. The recv
7 * framing loop (recv_process.c) drives the chunking: brix_write_stream_begin
8 * arms the streaming state at the request-header phase, each filled chunk is
9 * applied by brix_write_stream_apply_chunk, and after the last chunk
10 * brix_write_stream_finish emits exactly one reply for the logical write.
11 *
12 * WHY: BriX previously buffered the entire write payload before the pwrite, so
13 * the per-write cap (BRIX_MAX_WRITE_PAYLOAD, 16 MiB) doubled as a hard
14 * ceiling on a single kXR_write. A client that legitimately sends one very
15 * large inline write (go-hep's 64 MiB WriteAtContext, some Rust clients)
16 * was rejected outright. Streaming lifts the ceiling to
17 * BRIX_MAX_WRITE_STREAM while keeping resident memory bounded to a single
18 * chunk, and it is the shared consumer the kXR_bind data-path feeds too.
19 *
20 * HOW: The chunk buffer is the existing recv payload buffer (allocated one chunk
21 * wide by the shared REQ_PAYLOAD setup); ctx->recv.cur_dlen is repurposed as
22 * the CURRENT chunk length while sw_active is set, and sw_done tracks how
23 * much of sw_total has been applied. Chunks are applied SYNCHRONOUSLY (no
24 * AIO), in offset order — a staged handle appends via the reply-free
25 * brix_staged_append_raw, a plain fd/driver handle writes via a VFS job,
26 * exactly like the synchronous kXR_write fallback. The kXR_write protocol
27 * acknowledges the whole write once: the first failure is latched into
28 * sw_err, later chunks are discarded (their bytes still drained off the wire
29 * so framing stays aligned), and finish() sends one kXR_IOError instead of
30 * the kXR_ok. A handle that is invalid / not writable at begin() arms a
31 * drain: bytes are consumed and discarded, and the latched validation error
32 * is sent at the end — never a mid-stream reply that would desync framing.
33 *
34 * SSI, inline-decompression (write_codec), and require_pgwrite handles are
35 * NOT streamed here — begin() declines them so their specialised buffered
36 * paths keep their exact behaviour and per-page-CRC invariants.
37 */

44/*
45 * brix_write_stream_arm_drain — mark the streaming write as a drain: consume and
46 * discard sw_total bytes, then reply with the latched (code,msg) at finish. Used
47 * when the handle is unusable (invalid / not writable) but the payload is already
48 * inbound, so replying now would desync the framing.
49 */
50static void
51brix_write_stream_arm_drain(brix_ctx_t *ctx, int code, const char *msg)
52{
53 ctx->recv.sw_drain = 1;
54 ctx->recv.sw_err = code;
55 ctx->recv.sw_errmsg = msg;
56}

58ngx_flag_t
59brix_write_stream_begin(brix_ctx_t *ctx, ngx_connection_t *c,
60 ngx_stream_brix_srv_conf_t *conf)
61{
62 const u_char *body = ((ClientRequestHdr *) ctx->recv.hdr_buf)->body;
63 xrdw_write_req_t req;
64 brix_file_t *file;
65 int idx;
66 uint32_t total = ctx->recv.cur_dlen; /* whole logical write length */
67
68 (void) c;
69
70 /* A deployment forcing the checksummed pgwrite path refuses cleartext writes;
71 * leave that to the buffered handler (which emits the kXR_Unsupported reply). */
72 if (conf->common.require_pgwrite) {
73 return 0;
74 }
75
76 xrdw_write_req_unpack(body, &req);
77 idx = (int) (unsigned char) req.fhandle[0];
78
79 ctx->recv.sw_active = 1;
80 ctx->recv.sw_staged = 0;
81 ctx->recv.sw_drain = 0;
82 ctx->recv.sw_idx = idx;
83 ctx->recv.sw_base_off = req.offset;
84 ctx->recv.sw_total = total;
85 ctx->recv.sw_done = 0;
86 ctx->recv.sw_err = 0;
87 ctx->recv.sw_errmsg = NULL;
88
89 if (idx < 0 || idx >= BRIX_MAX_FILES) {
90 brix_write_stream_arm_drain(ctx, kXR_FileNotOpen, "invalid file handle");
91 } else {
92 file = &ctx->files[idx];
93
94 /* "open" mirrors brix_validate_file_handle: a kernel fd, a driver-backed
95 * object handle, or a whole-object staged writer all count as open. */
96 if (file->fd < 0 && file->sd_obj.driver == NULL && file->writer == NULL) {
97 brix_write_stream_arm_drain(ctx, kXR_FileNotOpen,
98 "invalid file handle");
99 } else if (!file->writable) {
100 brix_write_stream_arm_drain(ctx, kXR_NotAuthorized,
101 "file not open for writing");
102 } else if (file->ssi != NULL || file->write_codec != 0) {
103 /* SSI accumulation and inline write-decompression have their own
104 * buffered handlers — do not stream them. */
105 ctx->recv.sw_active = 0;
106 return 0;
107 } else {
108 ctx->recv.sw_staged = (file->writer != NULL) ? 1 : 0;
109 }
110 }
111
112 /* Repurpose cur_dlen as the FIRST chunk length; streaming is only entered when
113 * total > BRIX_WRITE_STREAM_CHUNK, so the first chunk is always a full chunk.
114 * cur_body_extra stays 0 (writes carry no trailing streamed body). */
115 ctx->recv.cur_dlen = BRIX_WRITE_STREAM_CHUNK;
116 ctx->recv.cur_body_extra = 0;
117 return 1;
118}

120/*
121 * brix_write_stream_apply_direct — synchronously write one chunk to a plain fd /
122 * driver-backed handle via a VFS job (mirrors the synchronous kXR_write
123 * fallback), latching the first error and committing the success-path accounting.
124 */
125static void
126brix_write_stream_apply_direct(brix_ctx_t *ctx, brix_file_t *file,
127 int64_t off, const u_char *buf, size_t len)
128{
129 brix_vfs_job_t job;
130 ssize_t n;
131
132 brix_vfs_job_write_init(&job, file->fd, (off_t) off, buf, len);
133 job.csi = file->csi; /* phase-59 W2: page tags */
134 brix_vfs_job_set_obj(&job, &file->sd_obj);/* route via driver if bound */
135 brix_vfs_io_execute(&job);
136 n = job.nio;
137 if (job.io_errno != 0) {
138 errno = job.io_errno;
139 }
140
141 if (n < 0) {
142 ctx->recv.sw_err = kXR_IOError;
143 ctx->recv.sw_errmsg = "async write error";
144 return;
145 }
146 if ((size_t) n < len) {
147 ctx->recv.sw_err = kXR_IOError;
148 ctx->recv.sw_errmsg = "short write (disk full?)";
149 return;
150 }
151
152 file->bytes_written += (size_t) n;
153 ctx->totals.bytes_written += (size_t) n;
154 brix_rl_charge_ctx(ctx, (size_t) n);
155
156 if (file->wt_enabled) {
157 brix_wt_mark_dirty(ctx, ctx->recv.sw_idx,
158 off + (int64_t) n - 1, (size_t) n);
159 }
160 if (file->wrts_enabled) {
161 brix_wrts_record(file, off, (uint32_t) n);
162 }
163}

165void
166brix_write_stream_apply_chunk(brix_ctx_t *ctx, ngx_connection_t *c)
167{
168 const u_char *buf = ctx->recv.payload;
169 size_t len = ctx->recv.cur_dlen;
170 int64_t off = ctx->recv.sw_base_off + (int64_t) ctx->recv.sw_done;
171 brix_file_t *file;
172
173 (void) c;
174
175 /* Draining an unusable handle, or a prior chunk already failed: keep the wire
176 * aligned by consuming these bytes without applying them (all-or-nothing). */
177 if (ctx->recv.sw_drain || ctx->recv.sw_err != 0) {
178 return;
179 }
180
181 file = &ctx->files[ctx->recv.sw_idx];
182
183 if (ctx->recv.sw_staged) {
184 int ar = brix_staged_append_raw(ctx, ctx->recv.sw_idx, off, buf, len);
185 if (ar == BRIX_STAGED_APPEND_IO) {
186 /* errno → kXR (ENOSPC from an unspillable reorder → kXR_NoSpace);
187 * the message must be a literal — strerror's buffer would not
188 * survive until brix_write_stream_finish sends the reply. */
189 ctx->recv.sw_err = brix_kxr_from_errno(errno);
190 ctx->recv.sw_errmsg = "staged write I/O error";
191 }
192 return;
193 }
194
195 brix_write_stream_apply_direct(ctx, file, off, buf, len);
196}

198ngx_int_t
199brix_write_stream_finish(brix_ctx_t *ctx, ngx_connection_t *c)
200{
201 int idx = ctx->recv.sw_idx;
202 int64_t base = ctx->recv.sw_base_off;
203 uint32_t total = ctx->recv.sw_total;
204 int err = ctx->recv.sw_err;
205 const char *errmsg = ctx->recv.sw_errmsg;
206 int drain = ctx->recv.sw_drain;
207 const char *path;
208 char detail[64];
209
210 /* Clear streaming state before replying (the reply may re-enter recv). */
211 ctx->recv.sw_active = 0;
212 ctx->recv.sw_staged = 0;
213 ctx->recv.sw_drain = 0;
214 ctx->recv.sw_err = 0;
215 ctx->recv.sw_errmsg = NULL;
216
217 path = (!drain && idx >= 0 && idx < BRIX_MAX_FILES && ctx->files[idx].path)
218 ? ctx->files[idx].path : "-";
219
220 snprintf(detail, sizeof(detail), "%lld+%u", (long long) base,
221 (unsigned) total);
222
223 if (err != 0) {
224 brix_log_access(ctx, c, "WRITE", path, detail, 0, err,
225 errmsg ? errmsg : "write failed", 0);
226 BRIX_OP_ERR(ctx, BRIX_OP_WRITE);
227 return brix_send_error(ctx, c, (uint16_t) err,
228 errmsg ? errmsg : "write failed");
229 }
230
231 brix_log_access(ctx, c, "WRITE", path, detail, 1, 0, NULL, total);
232 BRIX_OP_OK(ctx, BRIX_OP_WRITE);
233 return brix_send_ok(ctx, c, NULL, 0);
234}
