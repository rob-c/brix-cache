3 *
4 * WHAT: The four merge concerns invoked by the ngx_http_s3_merge_loc_conf

35/*
36 * s3_merge_preamble() — adopt the unified brix_* directives, then run the
37 * shared common.* preamble merge.
38 *
39 * WHAT: Pulls the location's merged values for the unified directives
40 * (brix_export, brix_cache_store, ...) out of the common module into the
41 * embedded common preamble, then applies the shared common.* merge (hard
42 * read-only enforcement + pmark) with parent-inheritance.
43 *
44 * WHY: These two steps must run first and in this order — the adopt seeds
45 * conf->common from the common module before the protocol merge applies its
46 * own defaults, and the shared merge folds parent→child on the same struct.
47 * Isolating them keeps the ordering dependency visible and the orchestrator
48 * flat. Merge order + defaults are byte-frozen.
49 *
50 * HOW: brix_http_common_adopt() copies the common-module values in place;
51 * ngx_http_brix_shared_merge() returns NGX_CONF_OK/ERROR — propagated 1:1.
52 */
53char *
54s3_merge_preamble(ngx_conf_t *cf, ngx_http_s3_loc_conf_t *prev,
55 ngx_http_s3_loc_conf_t *conf)
56{
57 /* Unified directives (brix_export, brix_cache_store, ...) live in the
58 * common module; pull the merged values for this location into our
59 * embedded preamble before protocol merge applies defaults. */
60 brix_http_common_adopt(cf, &conf->common);
61
62 /* Shared common.* preamble (incl. hard read-only enforcement + pmark). */
63 if (ngx_http_brix_shared_merge(cf, &prev->common, &conf->common, "")
64 != NGX_CONF_OK)
65 {
66 return NGX_CONF_ERROR;
67 }
68
69 return NGX_CONF_OK;
70}

72/*
73 * s3_merge_scalars() — merge the protocol-local scalar/string directive group.
74 *
75 * WHAT: Folds parent→child for the S3 flags, ints, msec, size and string
76 * directives (session-token/chunk-sig/list-cache/max-keys/mpu/zip/bucket/
77 * keys/region), plus the XrdAcc engine merge.
78 *
79 * WHY: These are independent value merges with fixed defaults and no failure
80 * path; grouping them separates the pure-merge concern from the token and
81 * export-build concerns that can fail. Defaults are byte-frozen.
82 *
83 * HOW: ngx_conf_merge_* macros apply the frozen default when a value is unset;
84 * brix_acc_http_merge_conf() folds the embedded XrdAcc conf.
85 */
86void
87s3_merge_scalars(ngx_http_s3_loc_conf_t *prev, ngx_http_s3_loc_conf_t *conf)
88{
89 ngx_conf_merge_value(conf->allow_unsigned_session_token,
90 prev->allow_unsigned_session_token, 0);
91 ngx_conf_merge_value(conf->verify_chunk_signatures,
92 prev->verify_chunk_signatures, 0);
93 ngx_conf_merge_value(conf->list_cache, prev->list_cache, 0);
94 ngx_conf_merge_msec_value(conf->list_cache_ttl, prev->list_cache_ttl,
95 10000); /* 10s default staleness bound */
96 ngx_conf_merge_value(conf->max_keys, prev->max_keys, 1000);
97 ngx_conf_merge_value(conf->mpu_max_age, prev->mpu_max_age, 0);
98 /* XrdAcc merge moved into ngx_http_brix_shared_merge (common.acc), W2. */
99 /* cache_root merge -> shared_merge (common.cache_root), phase-101 W8. */
100 ngx_conf_merge_str_value(conf->bucket, prev->bucket, "");
101 ngx_conf_merge_str_value(conf->access_key, prev->access_key, "");
102 ngx_conf_merge_str_value(conf->secret_key, prev->secret_key, "");
103 ngx_conf_merge_str_value(conf->region, prev->region, "us-east-1");
104}

106/*
107 * s3_merge_token() — merge the WLCG bearer-token group and load the JWKS.
108 *
109 * WHAT: Folds parent→child for the token enable/jwks/issuer/audience/skew
110 * directives, requires a JWKS path when token auth is on, and when a path is
111 * present loads its keys and registers a pool cleanup for them.
112 *
113 * WHY: Token auth carries config-time validation and side effects (key load +
114 * cleanup registration) that the pure scalar merge does not; keeping it
115 * separate keeps each concern under the complexity gate. Error strings and
116 * defaults are byte-frozen.
117 *
118 * HOW: The merges apply frozen defaults; the required-check and key-load emit
119 * the same NGX_LOG_EMERG diagnostics as before and early-return
120 * NGX_CONF_ERROR on the first failure.
121 */
122char *
123s3_merge_token(ngx_conf_t *cf, ngx_http_s3_loc_conf_t *prev,
124 ngx_http_s3_loc_conf_t *conf)
125{
126 /* WLCG bearer-token auth merge */
127 ngx_conf_merge_value(conf->token_enable, prev->token_enable, 0);
128 /* token quartet merges -> shared_merge (W4; unified skew 30). */
129
130 if (conf->token_enable && conf->common.token_jwks.len == 0) {
131 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
132 "brix_s3_token: brix_token_jwks is required when "
133 "brix_s3_token is on");
134 return NGX_CONF_ERROR;
135 }
136
137 if (conf->token_enable && conf->common.token_jwks.len > 0) {
138 int key_rc;
139
140 key_rc = brix_jwks_load(cf->log,
141 (const char *) conf->common.token_jwks.data,
142 conf->jwks_keys, BRIX_MAX_JWKS_KEYS);
143 if (key_rc <= 0) {
144 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
145 "brix_token_jwks: no usable keys in \"%V\"",
146 &conf->common.token_jwks);
147 return NGX_CONF_ERROR;
148 }
149 conf->jwks_key_count = key_rc;
150
151 if (brix_jwks_register_cleanup(cf->pool, conf->jwks_keys,
152 &conf->jwks_key_count) != NGX_OK)
153 {
154 return NGX_CONF_ERROR;
155 }
156 if (brix_http_common_register_jwks_refresh(cf,
157 &conf->common.token_jwks, conf->jwks_keys,
158 &conf->jwks_key_count,
159 conf->common.token_jwks_refresh_interval) != NGX_OK)
160 {
161 return NGX_CONF_ERROR;
162 }
163 }
164
165 return NGX_CONF_OK;
166}

168/*
169 * s3_export_attach_credential() — attach a named brix_credential's secrets to
170 * the export's source backend (§14).
171 *
172 * WHAT: When the export names a brix_credential, resolves it, derives its
173 * bearer token, and hands the bearer plus any x509/CA/S3/SSS secrets to the
174 * VFS backend registered for this export root.
175 *
176 * WHY: This is the densest sub-concern of the enabled-export build — a lookup,
177 * a bearer derivation, and a fully-populated backend-cred struct. Extracting
178 * it keeps the export orchestrator readable while preserving the exact
179 * deny/error diagnostics.
180 *
181 * HOW: Copies the credential name into a bounded buffer, looks it up
182 * (missing → NGX_LOG_EMERG + NGX_CONF_ERROR), maps it through the shared
183 * brix_credential_to_backend_cred() (P80.1 — the ONE mapper), then calls
184 * brix_vfs_backend_set_credential().
185 */
186static char *
187s3_export_attach_credential(ngx_conf_t *cf, ngx_http_s3_loc_conf_t *conf)
188{
189 char cred_z[256];
190 char bearer[4096];
191 const brix_credential_t *cred;
192 brix_vfs_backend_cred_t bcred;
193
194 ngx_cpystrn((u_char *) cred_z, conf->common.storage_credential.data,
195 ngx_min(conf->common.storage_credential.len + 1,
196 sizeof(cred_z)));
197 cred = brix_credential_lookup(cred_z);
198 if (cred == NULL) {
199 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
200 "brix_s3_storage_credential: no brix_credential \"%V\"",
201 &conf->common.storage_credential);
202 return NGX_CONF_ERROR;
203 }
204 if (brix_credential_to_backend_cred(cred, bearer, sizeof(bearer),
205 &bcred, cf->log) != NGX_OK)
206 {
207 return NGX_CONF_ERROR;
208 }
209 brix_vfs_backend_set_credential(conf->common.root_canon, &bcred);
210
211 return NGX_CONF_OK;
212}

214/*
215 * s3_merge_export() — the enabled-export config build block.
216 *
217 * WHAT: For an S3-enabled location: pins the posix backend to the export tree,
218 * canonicalizes the export root, registers the temp reaper + confinement
219 * rootfd, registers the composable storage backend, attaches a named
220 * credential (if any), registers the cache/stage tiers, canonicalizes the
221 * read-through cache root, and enforces the cache-outside-export guard.
222 *
223 * WHY: This whole block only runs when common.enable is true and chains a set
224 * of config-time build steps that each canonicalize/register/guard, any of
225 * which can fail. Grouping it under one helper keeps the orchestrator flat
226 * and the step order — which is byte-frozen — in one place.
227 *
228 * HOW: Each step early-returns NGX_CONF_ERROR on failure; the credential
229 * attach is delegated when storage_credential is named; the final guard runs
230 * unconditionally (cache_root_canon empty when no cache root was set).
231 */
232char *
233s3_merge_export(ngx_conf_t *cf, ngx_http_s3_loc_conf_t *conf)
234{
235 brix_export_root_opts_t root_opts;
236
237 /* posix:<path> backend → the local export tree (composable brix_export). */
238 brix_storage_backend_posix_root(&conf->common);
239
240 root_opts.directive_name = "brix_export";
241 root_opts.allow_write = conf->common.allow_write
242 && !brix_storage_backend_is_remote(&conf->common);
243 root_opts.required = 0;
244 root_opts.canon_size = sizeof(conf->common.root_canon);
245 if (brix_prepare_export_root(cf, &conf->common.root, &root_opts,
246 conf->common.root_canon) != NGX_CONF_OK)
247 {
248 return NGX_CONF_ERROR;
249 }
250 /* SP4: reap interrupted NON-staged direct-write temps under this root. */
251 if (conf->common.root_canon[0] != '\0') {
252 brix_tmp_reap_register(conf->common.root_canon);
253 }
254
255 /* phase-101 W5.2c: finalize this export's OWN copy of the native authdb READ
256 * rules against its root so the S3 access phase can enforce them (a deep copy
257 * — the shared common.authdb_rules is finalized by webdav against a possibly
258 * different root, so an in-place finalize here would corrupt it). */
259 if (brix_authdb_rules_finalize_copy(cf, &conf->common.root,
260 conf->common.authdb_rules, &conf->authdb_final) != NGX_OK)
261 {
262 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
263 "brix_authdb: failed to finalize rules for root \"%V\"",
264 &conf->common.root);
265 return NGX_CONF_ERROR;
266 }
267
268 /* Open the persistent confinement rootfd (kernel openat2
269 * RESOLVE_BENEATH anchor); no-op when no brix_s3_root is set. */
270 if (brix_http_open_rootfd(cf, &conf->common) != NGX_CONF_OK) {
271 return NGX_CONF_ERROR;
272 }
273
274 /* Register the export's composable storage backend (phase-63): a
275 * "root://"/"http://" URL or a driver name routes every VFS op (S3 GET
276 * goes through brix_vfs_open) to the source backend; default POSIX is a
277 * no-op. Mirrors the stream/webdav config paths. */
278 if (brix_http_register_storage_backend(cf, &conf->common)
279 != NGX_CONF_OK)
280 {
281 return NGX_CONF_ERROR;
282 }
283
284 /* §14: attach the named brix_credential's bearer to the source backend. */
285 if (conf->common.storage_credential.len > 0) {
286 if (s3_export_attach_credential(cf, conf) != NGX_CONF_OK) {
287 return NGX_CONF_ERROR;
288 }
289 }
290
291 if (brix_prepare_cache_root(cf, &conf->common) != NGX_CONF_OK) {
292 return NGX_CONF_ERROR;
293 }
294
295 /* Phase-64/W9: cache_root has now been lowered to cache_store, so there is
296 * exactly one composable cache registration path. */
297 if (brix_tier_register_stores(cf, &conf->common) != NGX_OK) {
298 return NGX_CONF_ERROR;
299 }
300
301 /* phase-107 C1: writer reorder-spill scratch (no brix_stage_dir on this
302 * surface, so only an explicit brix_vfs_spill_path provisions one). */
303 if (brix_prepare_spill_scratch(cf, &conf->common, NULL) != NGX_CONF_OK) {
304 return NGX_CONF_ERROR;
305 }
306
307 return NGX_CONF_OK;
308}
