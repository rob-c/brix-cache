1/*
2 * rfile_fast_ring.c — the slot ring and the pipeline that drives it.
3 *
4 * WHAT: Everything between "a request is submitted" and "its bytes reach the
5 * sink": the completion callback the loop thread parks replies with,
6 * kXR_read submission, the fill -> wait -> deliver loop, the drain that
7 * consumes replies still on the wire after an abort, and the ring's own
8 * allocation and release.
9 * WHY: Split out of rfile_fast.c, which had grown past the size limit holding
10 * both this and the session around it (eligibility, attach/detach, socket
11 * mode, the env tunables). The session half now reaches the engine only
12 * through the four entry points in rfile_fast.h - behaviour-identical.
13 * HOW: The calling thread owns ring order strictly: it submits at `tail`,
14 * consumes at `head`, and the loop thread only ever writes the slot it
15 * was handed. All shared state changes under ring->lock with a
16 * broadcast on ring->cv.
17 */

30/* ---- Completion callback: park one reply in its slot ----
31 *
32 * WHAT: Records the reply length (or the failure) for the slot this request was
33 * submitted against, marks it READY and wakes the consuming thread. The
34 * bytes are already in slot->buf — the engine read them there — so `body`
35 * aliases it and is never freed.
36 *
37 * WHY: The async engine invokes this inline on its loop thread, so it must not
38 * block or do I/O — parking the pointer and signalling is the whole job.
39 * Any status other than a plain kXR_ok (redirect, wait, authmore) belongs
40 * to the synchronous roundtrip layer, not here, so it is recorded as a
41 * failure and the caller falls back rather than mis-framing the stream.
42 *
43 * HOW: 1. Take the ring lock.
44 * 2. Store body/blen; on rc != 0 copy the transport status, on an
45 * unexpected kXR_* build one.
46 * 3. Mark the slot READY, broadcast, unlock.
47 */
48static void
49fast_on_reply(void *ctx, int rc, uint16_t kxr, uint8_t *body, uint32_t blen,
50 const brix_status *st)
51{
52 fast_slot *slot = (fast_slot *) ctx;
53 fast_ring *ring = slot->ring;
54
55 (void) body; /* aliases slot->buf: the engine wrote straight into it */
56 pthread_mutex_lock(&ring->lock);
57 slot->blen = blen;
58 slot->failed = 0;
59 if (rc != 0) {
60 slot->failed = 1;
61 if (st != NULL) {
62 slot->st = *st;
63 } else {
64 brix_status_set(&slot->st, XRDC_ESOCK, 0, "pipelined read failed");
65 }
66 } else if (kxr != kXR_ok) {
67 slot->failed = 1;
68 brix_status_set(&slot->st, XRDC_EPROTO, 0,
69 "unexpected read reply status %u", (unsigned) kxr);
70 }
71 slot->state = FAST_SLOT_READY;
72 pthread_cond_broadcast(&ring->cv);
73 pthread_mutex_unlock(&ring->lock);
74}

77/* ---- Submit one kXR_read for a slot ----
78 *
79 * WHAT: Builds the 24-byte kXR_read request for [slot->offset, +slot->want)
80 * against `fhandle` and hands it to the async engine with fast_on_reply
81 * bound to `slot`. Returns 0 (queued) or -1 (st set, slot left IDLE).
82 *
83 * WHY: kXR_read is self-addressing (fhandle + offset + length), so a reply can
84 * be matched purely by streamid and the requests need no ordering on the
85 * wire — that is exactly what lets several ride at once. Requests are NOT
86 * marked retry_safe: they are handle-bound, so a transparent re-issue
87 * after a reconnect would quote a dead fhandle.
88 *
89 * HOW: 1. Zero a ClientReadRequest, set requestid = kXR_read.
90 * 2. Pack fhandle/offset/rlen into its body with the shared packer.
91 * 3. Reset the slot's reply fields, mark it BUSY and submit.
92 */
93static int
94fast_submit(brix_aconn *ac, const uint8_t *fhandle, fast_slot *slot,
95 int deadline_ms, brix_status *st)
96{
97 ClientReadRequest req;
98 xrdw_read_req_t body = { .offset = slot->offset,
99 .rlen = (int32_t) slot->want };
100 brix_aio_opts opts = { deadline_ms, -1, 0, slot->buf, slot->want };
101
102 memset(&req, 0, sizeof(req));
103 req.requestid = htons(kXR_read);
104 memcpy(body.fhandle, fhandle, XRD_FHANDLE_LEN);
105 xrdw_read_req_pack(&body, ((ClientRequestHdr *) &req)->body);
106
107 slot->blen = 0;
108 slot->failed = 0;
109 slot->state = FAST_SLOT_BUSY;
110 if (brix_aio_submit_ex(ac, &req, NULL, 0, &opts, fast_on_reply, slot,
111 st) != 0) {
112 slot->state = FAST_SLOT_IDLE;
113 return -1;
114 }
115 return 0;
116}

119/* ---- Bytes of the requested range still unrequested ----
120 *
121 * WHAT: Returns how many bytes of the job's range have not yet been covered by
122 * a submitted request — LLONG-large for an EOF-driven (limit < 0) job.
123 *
124 * WHY: The fill loop needs one predicate for both the sized case (stop exactly
125 * at limit) and the EOF-driven case (keep reading past the end and let a
126 * short reply prove EOF), and expressing "unbounded" as a huge remainder
127 * keeps that loop free of a second branch.
128 *
129 * HOW: limit < 0 → INT64_MAX; else limit - issued clamped at 0.
130 */
131static int64_t
132fast_remaining(const fast_job *job, int64_t issued)
133{
134 if (job->limit < 0) {
135 return INT64_MAX;
136 }
137 return (issued >= job->limit) ? 0 : job->limit - issued;
138}

141/* ---- Keep the pipeline full ----
142 *
143 * WHAT: Submits reads into free ring slots until `depth` are outstanding, the
144 * range is fully requested, or EOF was proven. Returns 0, or -1 with
145 * *st set when a submit failed.
146 *
147 * WHY: Refilling from the same place that primes the ring keeps exactly one
148 * copy of the "next offset, next slot" arithmetic, so a chunk can never be
149 * issued twice or skipped — the ordering the in-order delivery depends on.
150 *
151 * HOW: While there is room and work: size the chunk against the remainder,
152 * fill the tail slot's offset/want, submit, advance issued/tail/inflight.
153 */
154static int
155fast_fill(brix_aconn *ac, const fast_job *job, fast_ring *ring,
156 fast_state *state, brix_status *st)
157{
158 const uint8_t *fhandle = job->rf->f.fhandle;
159
160 while (state->inflight < job->depth && !state->eof) {
161 int64_t left = fast_remaining(job, state->issued);
162 fast_slot *slot = &ring->slots[state->tail];
163
164 if (left <= 0) {
165 return 0;
166 }
167 slot->offset = job->offset + state->issued;
168 slot->want = (left < (int64_t) job->chunk) ? (uint32_t) left
169 : job->chunk;
170 if (fast_submit(ac, fhandle, slot, job->deadline_ms, st) != 0) {
171 return -1;
172 }
173 state->issued += slot->want;
174 state->tail = (state->tail + 1) % job->depth;
175 state->inflight += 1;
176 }
177 return 0;
178}

181/* ---- Wait until the head slot's reply has arrived ----
182 *
183 * WHAT: Blocks the calling thread until the slot at `index` is READY, then
184 * returns it. Never fails: every submitted request completes exactly
185 * once, including with an error when the connection is torn down.
186 *
187 * WHY: Delivery must be in file order even though replies may complete in any
188 * order, so the consumer always waits for one specific slot rather than
189 * "any completion" — that single rule is what keeps the sink seeing a
190 * strictly ascending, gap-free byte stream.
191 *
192 * HOW: Take the lock, cond_wait while the state is not READY, unlock, return.
193 */
194static fast_slot *
195fast_wait_head(fast_ring *ring, unsigned index)
196{
197 fast_slot *slot = &ring->slots[index];
198
199 pthread_mutex_lock(&ring->lock);
200 while (slot->state != FAST_SLOT_READY) {
201 pthread_cond_wait(&ring->cv, &ring->lock);
202 }
203 pthread_mutex_unlock(&ring->lock);
204 return slot;
205}

208/* ---- Hand one completed slot to the sink ----
209 *
210 * WHAT: Validates and delivers the head slot's reply, frees its body and
211 * returns the slot to IDLE. Updates state->moved / eof / stop / rc and
212 * sets *st on failure.
213 *
214 * WHY: This is the only place a byte leaves the pipeline, so the short-reply
215 * EOF rule, the over-long-reply guard and the sink's stop/fail contract
216 * all live together where they can be read as one policy.
217 *
218 * HOW: 1. A failed slot copies its status and fails the stream.
219 * 2. blen > want is a protocol violation (a server returning more than
220 * asked) and fails the stream.
221 * 3. blen < want proves EOF; blen > 0 is passed to the sink.
222 * 4. A positive sink return stops the stream successfully, a negative one
223 * fails it (the sink set *st).
224 */
225static void
226fast_deliver(const fast_job *job, fast_slot *slot, fast_state *state,
227 brix_status *st)
228{
229 int sink_rc = 0;
230
231 if (slot->failed) {
232 *st = slot->st;
233 state->rc = -1;
234 } else if (slot->blen > slot->want) {
235 brix_status_set(st, XRDC_EPROTO, 0,
236 "read returned more than requested (%u > %u)",
237 slot->blen, slot->want);
238 state->rc = -1;
239 } else {
240 if (slot->blen < slot->want) {
241 state->eof = 1;
242 }
243 if (slot->blen > 0) {
244 sink_rc = job->sink(slot->buf, slot->blen, slot->offset,
245 job->arg, st);
246 state->moved += slot->blen;
247 }
248 if (sink_rc < 0) {
249 state->rc = -1;
250 } else if (sink_rc > 0) {
251 state->stop = 1;
252 }
253 }
254 slot->blen = 0;
255 slot->state = FAST_SLOT_IDLE;
256}

259/* ---- Drive the pipeline to completion ----
260 *
261 * WHAT: Runs fill → wait → deliver until the range is done, EOF is proven, the
262 * sink stops it, the operator cancels, or a failure sets *st. Returns 0
263 * or -1; state->moved always holds the bytes delivered.
264 *
265 * WHY: Keeping the loop itself tiny (four named steps) is what lets the
266 * ordering invariant — deliver slot `head`, then advance — be checked by
267 * eye; every other concern is in one of the helpers it calls.
268 *
269 * HOW: 1. Honour the file's cancel predicate between chunks.
270 * 2. Refill the pipeline; stop when nothing is outstanding.
271 * 3. Wait for the head slot, deliver it, advance head, drop inflight.
272 */
273int
274fast_run(brix_aconn *ac, const fast_job *job, fast_ring *ring,
275 fast_state *state, brix_status *st)
276{
277 brix_rfile *rf = job->rf;
278
279 while (state->rc == 0 && !state->stop) {
280 if (rf->cancel != NULL && rf->cancel()) {
281 brix_status_set(st, XRDC_EUSAGE, ECANCELED, "transfer canceled");
282 return -1;
283 }
284 if (fast_fill(ac, job, ring, state, st) != 0) {
285 return -1;
286 }
287 if (state->inflight == 0) {
288 return 0;
289 }
290 fast_deliver(job, fast_wait_head(ring, state->head), state, st);
291 state->head = (state->head + 1) % job->depth;
292 state->inflight -= 1;
293 }
294 return state->rc;
295}

298/* ---- Consume the replies still on the wire ----
299 *
300 * WHAT: Waits for and discards every still-outstanding reply, returning 1 when
301 * they all arrived cleanly and 0 when any of them failed.
302 *
303 * WHY: The connection is handed back to the SYNCHRONOUS helpers (close, a
304 * checksum query, the serial resume pump) the moment this file is done
305 * with it, and those read the socket frame by frame with no streamid
306 * demultiplexing. A reply still in the pipe would be read as the answer
307 * to the next request and corrupt the session, so a stream that ends while
308 * requests are outstanding (EOF proven early, or the sink asking to stop)
309 * must first pull those replies off the wire. A failed reply means the
310 * alignment is already lost, which the caller repairs by reopening.
311 *
312 * HOW: Walk the ring from head, waiting for each slot, freeing its body and
313 * returning it to IDLE until nothing is in flight.
314 */
315int
316fast_drain(const fast_job *job, fast_ring *ring, fast_state *state)
317{
318 int clean = 1;
319
320 while (state->inflight > 0) {
321 fast_slot *slot = fast_wait_head(ring, state->head);
322
323 if (slot->failed) {
324 clean = 0;
325 }
326 slot->blen = 0;
327 slot->state = FAST_SLOT_IDLE;
328 state->head = (state->head + 1) % job->depth;
329 state->inflight -= 1;
330 }
331 return clean;
332}

335/* ---- Allocate and initialise the slot ring ----
336 *
337 * WHAT: Allocates `depth` zeroed slots plus their mutex/condvar, each slot
338 * pointing back at the ring. Returns 0 or -1 (st set).
339 *
340 * WHY: Every slot must know its ring before any request is submitted, because
341 * the loop thread reaches the lock only through the slot it was handed.
342 *
343 * HOW: calloc the slots, init the mutex and condvar, back-link each slot.
344 */
345int
346fast_ring_init(fast_ring *ring, unsigned depth, uint32_t chunk, brix_status *st)
347{
348 unsigned i;
349
350 ring->slots = (fast_slot *) calloc(depth, sizeof(*ring->slots));
351 ring->pool = (uint8_t *) malloc((size_t) depth * chunk);
352 if (ring->slots == NULL || ring->pool == NULL) {
353 free(ring->slots);
354 free(ring->pool);
355 ring->slots = NULL;
356 ring->pool = NULL;
357 brix_status_set(st, XRDC_EIO, ENOMEM, "pipelined read: out of memory");
358 return -1;
359 }
360 ring->depth = depth;
361 pthread_mutex_init(&ring->lock, NULL);
362 pthread_cond_init(&ring->cv, NULL);
363 for (i = 0; i < depth; i++) {
364 ring->slots[i].ring = ring;
365 ring->slots[i].buf = ring->pool + (size_t) i * chunk;
366 }
367 return 0;
368}

371/* ---- Release the ring ----
372 *
373 * WHAT: Frees the slot array, the landing-buffer pool and the synchronisation
374 * primitives.
375 *
376 * WHY: The slot buffers are the landing zone the async engine writes replies
377 * into, so they MUST outlive every outstanding request: call this only
378 * after the aconn is closed, when no callback can still be running.
379 *
380 * HOW: Destroy the condvar + mutex, then free the slot array and the pool.
381 */
382void
383fast_ring_free(fast_ring *ring)
384{
385 pthread_cond_destroy(&ring->cv);
386 pthread_mutex_destroy(&ring->lock);
387 free(ring->slots);
388 free(ring->pool);
389 ring->slots = NULL;
390 ring->pool = NULL;
391 ring->depth = 0;
392}
