12/*
13 * sss_cred_t — outputs of a successful SSS credential decode.
14 *
15 * WHAT: the verified key, the decoded identity, and the decrypted-buffer
16 * span that sss_recv_cred() hands to the mapping/reply stages.
17 * WHY: the wire-verify stage produces several coupled results that the later
18 * stages consume together; grouping them keeps data flow explicit without new
19 * globals.
20 * HOW: zero-initialised by the caller; populated only on the NGX_OK path.
21 */
22typedef struct {
23 const brix_sss_key_t *key;
24 brix_sss_identity_t id;
25 u_char *clear; /* decrypted credential buffer */
26 size_t clear_span; /* bytes to OPENSSL_cleanse (== cipher_len) */
27 ngx_int_t replied_rc; /* wire result when recv returns NGX_DECLINED/NGX_DONE */
28 const u_char *cipher; /* ciphertext start (past the outer header) */
29 size_t cipher_len; /* ciphertext length */
30 size_t hdr_len; /* outer-header length (== cipher offset) */
31} sss_cred_t;

37/*
38 * sss_deny — funnel every SSS verify-chain rejection through one exit.
39 *
40 * WHAT: queues the kXR_error (via brix_sss_auth_failed) and returns NGX_DONE,
41 * stashing the actual wire-send result in out->replied_rc.
42 * WHY: brix_sss_auth_failed returns NGX_OK on a successful send (the error was
43 * queued), which is indistinguishable from "credential verified" to a caller
44 * that gates on `rc != NGX_OK`. A deny that reports NGX_OK falls through to
45 * sss_map_identity/sss_reply — a NULL-key dereference for an unknown key id and,
46 * for a non-NULL key that fails the CRC/lifetime check, a spurious kXR_ok with a
47 * registered session (auth bypass). Returning a dedicated stop sentinel keeps a
48 * deny terminal while preserving the exact wire result the handler reports back.
49 * HOW: NGX_DONE is caught by every `rc != NGX_OK` guard in the chain; the handler
50 * maps NGX_DONE (and the existing NGX_DECLINED challenge sentinel) back to
51 * out->replied_rc so the top-level return value is unchanged.
52 */
53static ngx_int_t
54sss_deny(brix_ctx_t *ctx, ngx_connection_t *c, sss_cred_t *out)
55{
56 out->replied_rc = brix_sss_auth_failed(ctx, c);
57 return NGX_DONE;
58}

60/*
61 * sss_parse_header — validate the SSS outer header and locate the key+cipher.
62 *
63 * WHAT: checks the "sss\0"+BF32 magic, the key-name-size field, the header
64 * length/NUL-termination, looks up the configured key by wire id, and computes
65 * the trailing ciphertext span.
66 * WHY: the untrusted outer framing has several coupled bounds checks that must
67 * all pass before any decryption; keeping them together makes the deny surface
68 * auditable in one place.
69 * HOW: on success sets out->key/cipher/cipher_len/hdr_len and returns NGX_OK.
70 * Any malformed field or unknown key returns sss_deny() (NGX_DONE, error queued).
71 */
72static ngx_int_t
73sss_parse_header(brix_ctx_t *ctx, ngx_connection_t *c,
74 ngx_stream_brix_srv_conf_t *conf, sss_cred_t *out)
75{
76 const u_char *payload = ctx->recv.payload;
77 size_t hdr_len = 0;
78 int64_t key_id;
79
80 if (!brix_sss_header_framing_ok(payload, ctx->recv.cur_dlen, &hdr_len)) {
81 return sss_deny(ctx, c, out);
82 }
83
84 key_id = (int64_t) brix_sss_read_be64(payload + 8);
85 out->key = brix_sss_find_key(conf, key_id);
86 if (out->key == NULL) {
87 return sss_deny(ctx, c, out);
88 }
89
90 out->cipher = payload + hdr_len;
91 out->cipher_len = ctx->recv.cur_dlen - hdr_len;
92 if (out->cipher_len <= 4) {
93 return sss_deny(ctx, c, out);
94 }
95
96 out->hdr_len = hdr_len;
97 return NGX_OK;
98}

100/*
101 * sss_verify_keytab — decrypt the credential and check its integrity/freshness.
102 *
103 * WHAT: Blowfish-CFB64 decrypts the ciphertext with the matched key, verifies
104 * the trailing CRC32 (wrong-key/tamper guard), enforces the lifetime replay
105 * window, handles the interactive SNDLID challenge form, and parses the
106 * identity TLV.
107 * WHY: this is the keytab-bound trust decision; isolating it freezes the
108 * verification semantics as one reviewable unit.
109 * HOW: allocates and fills out->clear/out->clear_span/out->id. Returns NGX_OK
110 * when a self-contained credential verifies; NGX_DECLINED for SNDLID with the
111 * authmore result in out->replied_rc; sss_deny() (NGX_DONE, error queued) on any
112 * wrong-key/expired/malformed credential.
113 */
114static ngx_int_t
115sss_verify_keytab(brix_ctx_t *ctx, ngx_connection_t *c,
116 ngx_stream_brix_srv_conf_t *conf, sss_cred_t *out)
117{
118 const u_char *payload = ctx->recv.payload;
119 size_t out_len, clear_len;
120 uint8_t options;
121 uint32_t got_crc, want_crc, gen_time, now;
122
123 /* out->key is set non-NULL by sss_parse_header (which denies on a key miss)
124 * and sss_recv_cred only reaches here on its NGX_OK; guard defensively so a
125 * future caller reordering cannot turn a NULL key into a crash. sss_deny
126 * returns NGX_DONE, so this genuinely stops the chain (an earlier NGX_OK
127 * deny return here did not — CVE-class NULL deref in sss_map_identity). */
128 if (out->key == NULL) {
129 return sss_deny(ctx, c, out);
130 }
131
132 BRIX_PALLOC_OR_RETURN(out->clear, c->pool, out->cipher_len, NGX_ERROR);
133 out->clear_span = out->cipher_len;
134
135 if (brix_sss_bf32_crypt(0, out->key->key, out->key->key_len,
136 out->cipher, out->cipher_len,
137 out->clear, out->cipher_len, &out_len)
138 != NGX_OK)
139 {
140 return sss_deny(ctx, c, out);
141 }
142
143 if (out_len <= 4) {
144 return sss_deny(ctx, c, out);
145 }
146
147 clear_len = out_len - 4;
148 got_crc = brix_sss_read_be32(out->clear + clear_len);
149 want_crc = brix_sss_crc32(out->clear, clear_len);
150 /* Wrong-key detection: a CRC mismatch means either the wrong key was
151 * used for decryption or the ciphertext was tampered with. */
152 if (got_crc != want_crc || clear_len < BRIX_SSS_DATA_HDR_LEN) {
153 return sss_deny(ctx, c, out);
154 }
155
156 gen_time = brix_sss_read_be32(out->clear + 32);
157 now = (uint32_t) (ngx_time() - BRIX_SSS_BASE_TIME);
158 /* Credential replay prevention: reject tokens older than sss_lifetime. */
159 if (gen_time + (uint32_t) conf->sss_lifetime <= now) {
160 return sss_deny(ctx, c, out);
161 }
162
163 options = out->clear[39];
164 if (options == BRIX_SSS_OPT_SNDLID) {
165 out->replied_rc = brix_sss_send_authmore(ctx, c, out->key,
166 payload, out->hdr_len);
167 return NGX_DECLINED;
168 }
169
170 if (brix_sss_parse_identity(out->clear + BRIX_SSS_DATA_HDR_LEN,
171 clear_len - BRIX_SSS_DATA_HDR_LEN,
172 &out->id)
173 != NGX_OK)
174 {
175 return sss_deny(ctx, c, out);
176 }
177
178 return NGX_OK;
179}

181/*
182 * sss_recv_cred — receive and verify one SSS credential off the wire.
183 *
184 * WHAT: orchestrates the outer-header parse then the keytab decrypt/verify.
185 * WHY: keeps the two-stage verification chain as one linear call the handler
186 * consumes; the security decisions live in the two step helpers.
187 * HOW: returns NGX_OK with *out filled, NGX_DECLINED for a sent SNDLID
188 * challenge (out->replied_rc), NGX_DONE for a deny (error already queued,
189 * out->replied_rc), or the raw error result of a step.
190 */
191static ngx_int_t
192sss_recv_cred(brix_ctx_t *ctx, ngx_connection_t *c,
193 ngx_stream_brix_srv_conf_t *conf, sss_cred_t *out)
194{
195 ngx_int_t rc;
196
197 rc = sss_parse_header(ctx, c, conf, out);
198 if (rc != NGX_OK) {
199 return rc;
200 }
201
202 return sss_verify_keytab(ctx, c, conf, out);
203}

205/*
206 * sss_map_identity — resolve the effective user and group for a credential.
207 *
208 * WHAT: applies the key's option flags to the decoded identity to pick the
209 * authenticated user name and group string.
210 * WHY: the ANYUSR/ALLUSR/USRGRP/ANYGRP mapping policy is a pure decision that
211 * belongs apart from the I/O so it is trivially auditable.
212 * HOW: pure — reads key opts + id, writes the user/group out-pointers
213 * (pointers into id or the key, or static literals); no side effects.
214 */
215static void
216sss_map_identity(const brix_sss_key_t *key, const brix_sss_identity_t *id,
217 const char **user, const char **group)
218{
219 *user = key->user;
220 if (key->opts & (BRIX_SSS_OPT_ANYUSR | BRIX_SSS_OPT_ALLUSR)) {
221 *user = id->name[0] ? id->name : "nobody";
222 }
223
224 *group = "";
225 if (!(key->opts & BRIX_SSS_OPT_USRGRP)) {
226 if (key->opts & BRIX_SSS_OPT_ANYGRP) {
227 *group = id->grps[0] ? id->grps : "nogroup";
228 } else {
229 *group = key->group;
230 }
231 }

234/*
235 * sss_entity_policy — apply the keytab's trust policy to the v2 entity fields.
236 *
237 * WHAT: drops the client-asserted VORG/ROLE/ENDO when the matched key pins
238 * the identity (no ANYUSR/ALLUSR option), and drops a proxied CRED blob
239 * unless brix_sss_getcreds is on.
240 * WHY: the keytab decides how much of what the peer asserts is trusted. A
241 * key that maps every holder to one fixed user must not let that holder
242 * decorate itself with a VO or role the operator never granted, and a
243 * proxied credential is only kept where an operator opted in
244 * (release-2.0-readiness F9).
245 * HOW: pure edit of `id` plus one INFO line per drop; the NAME/GRPS mapping
246 * stays with sss_map_identity.
247 */
248static void
249sss_entity_policy(const ngx_stream_brix_srv_conf_t *conf,
250 const brix_sss_key_t *key, brix_sss_identity_t *id, ngx_log_t *log)
251{
252 int pinned = !(key->opts & (BRIX_SSS_OPT_ANYUSR | BRIX_SSS_OPT_ALLUSR));
253
254 if (pinned && (id->vorg[0] || id->role[0] || id->endo[0])) {
255 id->vorg[0] = '\0';
256 id->role[0] = '\0';
257 id->endo[0] = '\0';
258 BRIX_LOG(NGX_LOG_INFO, log, 0,
259 "brix: SSS entity fields dropped: keytab pins the identity");
260 }
261
262 if (id->creds_len > 0 && !conf->sss_getcreds) {
263 OPENSSL_cleanse(id->creds, id->creds_len);
264 id->creds_len = 0;
265 BRIX_LOG(NGX_LOG_INFO, log, 0,
266 "brix: SSS proxied credential dropped: "
267 "brix_sss_getcreds is off");
268 }
269}

302 *
303 * WHAT: stamps the login/identity state (including the v2 entity fields the

348/* Handle the kXR_auth SSS (XrdSecsss shared-secret) credential: verify the
349 * client token against the keytab, set the identity/session, and return kXR_ok
350 * or kXR_error. */
351ngx_int_t
352brix_handle_sss_auth(brix_ctx_t *ctx, ngx_connection_t *c,
353 ngx_stream_brix_srv_conf_t *conf)
354{
355 sss_cred_t cred;
356 const char *user, *group;
357 ngx_int_t rc;
358
359 ngx_memzero(&cred, sizeof(cred));
360
361 rc = sss_recv_cred(ctx, c, conf, &cred);
362 if (rc != NGX_OK) {
363 /* NGX_DECLINED: SNDLID authmore already sent. NGX_DONE: the verify chain
364 * denied and already queued the kXR_error. Both stash the wire-send
365 * result in cred.replied_rc; report it so a deny never reaches
366 * sss_map_identity/sss_reply. Any other rc is a raw error to propagate. */
367 return (rc == NGX_DECLINED || rc == NGX_DONE) ? cred.replied_rc : rc;
368 }
369
370 sss_entity_policy(conf, cred.key, &cred.id, c->log);
371 sss_map_identity(cred.key, &cred.id, &user, &group);
372
373 /* Zero the decrypted credential buffer before it ages in the pool.
374 * Prevents plaintext identity data from lingering across later requests. */
375 OPENSSL_cleanse(cred.clear, cred.clear_span);
376
377 rc = sss_reply(ctx, c, &cred.id, user, group);
378
379 /* The decoded identity (incl. any proxied credential) lived on this
380 * stack frame; scrub it the same way as the cleartext buffer. */
381 OPENSSL_cleanse(&cred.id, sizeof(cred.id));
382 return rc;
383}
