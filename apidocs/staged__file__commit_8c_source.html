2 * staged_file_commit.c — Cross-device / cross-backend staged-commit publish.
3 *
4 * WHAT: brix_commit_staged() and its helpers — publish a completed staged
5 * partial onto its final path when a plain rename(2) cannot: a
6 * cross-device fast-cache stage dir (EXDEV → copy + fsync + rename) or a
7 * non-POSIX final export (upload INTO the driver's staged write→commit
8 * state machine).
9 *
10 * WHY: Split from staged_file.c (phase-79 file-size split). The same-filesystem
11 * rename lifecycle (open/commit/abort) stays in staged_file.c; this file
12 * owns the heavier byte-moving publish paths that reach the storage-driver
13 * (SD) seam and the per-export backend registry. Isolating them keeps each
14 * file single-concept and under the size budget.
15 *
16 * HOW: brix_commit_staged() flushes+chmods the open fd (commit_flush_and_chmod;
17 * the fsync is skipped under BRIX_COMMIT_RELAXED / brix_durable_commit off),
18 * resolves the final export's backend, and dispatches: non-POSIX backend →
19 * commit_staged_to_backend (cstb_open_source + cstb_pump_and_commit);
20 * same-FS → rename(2); EXDEV → commit_cross_device. All byte I/O routes
21 * through the SD seam (brix_sd_posix_wrap / driver vtable), never raw
22 * pread/pwrite here. Zero behavior change from the pre-split code.
23 */

39/* VFS↔VFS (backend↔backend) byte move: copy the whole source object into the
40 * destination object by reading through the SOURCE object's storage driver and
41 * writing through the DESTINATION object's driver. The two objects may live on
42 * different backends — this is the cross-mount staging commit, so it is exactly
43 * the place where one side could later be a block/object (S3) store while the
44 * other stays POSIX.
45 *
46 * Today both ends are POSIX (brix_sd_posix_wrap over a kernel fd), so the only
47 * raw pread/pwrite happen INSIDE the POSIX backend (src/fs/backend/sd_posix.c) —
48 * never here. When a stage or final mount becomes a non-POSIX backend, only how
49 * the object is obtained changes (that driver's open() instead of a bare-fd
50 * wrap); this positional copy loop is unchanged. 0 / NGX_ERROR (errno set). */

60/* commit_be_logical — the export-root-relative ("/sub/file", or "/" for the root)
61 * key form a non-POSIX backend's namespace expects, matching
62 * brix_vfs_export_relative_root used by every other driver-backed op. */
63static const char *
64commit_be_logical(const char *abs, const char *root)
65{
66 size_t rl = (root != NULL) ? strlen(root) : 0;
67
68 if (rl > 0 && strncmp(abs, root, rl) == 0) {
69 if (abs[rl] == '/') { return abs + rl; } /* "/sub/file" */
70 if (abs[rl] == '\0') { return "/"; } /* the export root itself */
71 }
72 return abs;
73}

75/*
76 * WHAT: Resolve the read side of a backend commit — reuse the caller's still-open
77 * staged fd, or open the partial O_RDONLY, and report the mode to publish.
78 *
79 * WHY: commit_staged_to_backend can be reached with the partial fd still held
80 * (synchronous close) or already closed (crash-recovery reap), and the
81 * ownership decision governs whether the read fd must be closed on exit. A
82 * single resolver keeps that decision out of the copy loop.
83 *
84 * HOW: If fd is valid, borrow it (*owned=0); else open stage_path O_RDONLY
85 * (*owned=1). Derive the publish mode from fstat (default 0644). Returns the
86 * read fd, or NGX_INVALID_FILE with errno set on open failure.
87 *
88 * Parameters:
89 * fd — caller's staged fd, or NGX_INVALID_FILE
90 * stage_path — partial path to open when fd is closed
91 * owned — output: 1 if the returned fd must be closed by the caller
92 * mode — output: file mode to publish into the backend
93 */
94static ngx_fd_t
95cstb_open_source(ngx_fd_t fd, const char *stage_path, int *owned, mode_t *mode)
96{
97 struct stat sb;
98 ngx_fd_t rfd;
99
100 *owned = 0;
101 *mode = 0644;
102
103 if (fd != NGX_INVALID_FILE) {
104 rfd = fd;
105 } else {
106 rfd = open(stage_path, O_RDONLY | O_CLOEXEC);
107 if (rfd < 0) {
108 return NGX_INVALID_FILE;
109 }
110 *owned = 1;
111 }
112
113 if (fstat(rfd, &sb) == 0) {
114 *mode = sb.st_mode & 07777;
115 }
116 return rfd;
117}

119/*
120 * WHAT: Stream a POSIX-read partial (via the SD seam) into an open driver-staged
121 * blob, whole-object, and atomically publish it.
122 *
123 * WHY: The read→write pump is the body of the cross-backend commit; isolating it
124 * keeps commit_staged_to_backend a flat acquire/pump/cleanup orchestrator
125 * and confines the per-chunk error handling to one place.
126 *
127 * HOW: Wrap rfd as a POSIX source object, positional-read 64 KiB chunks, and
128 * staged_write each fully before advancing the offset. On any read/write
129 * error, or a failing staged_commit, abort the driver blob (releasing it)
130 * and return NGX_ERROR with the originating errno preserved. On success the
131 * blob is committed (partial NOT unlinked here — the caller owns that).
132 *
133 * Parameters:
134 * rfd — open read fd for the staged partial (positional reads)
135 * dst — destination backend instance (driver vtable)
136 * st — open driver-staged blob to fill and commit
137 */
138static ngx_int_t
139cstb_pump_and_commit(ngx_fd_t rfd, brix_sd_instance_t *dst,
140 brix_sd_staged_t *st)
141{
142 brix_sd_obj_t src_obj;
143 off_t off = 0;
144
145 brix_sd_posix_wrap(&src_obj, rfd); /* read the partial via the SD seam */
146 for ( ;; ) {
147 char buf[65536];
148 ssize_t r = src_obj.driver->pread(&src_obj, buf, sizeof(buf), off);
149 ssize_t w = 0;
150
151 if (r < 0) {
152 int e = errno;
153 dst->driver->staged_abort(st);
154 errno = e;
155 return NGX_ERROR;
156 }
157 if (r == 0) {
158 break; /* EOF: whole partial consumed */
159 }
160 while (w < r) {
161 ssize_t k = dst->driver->staged_write(st, buf + w,
162 (size_t) (r - w), off + w);
163 if (k <= 0) {
164 int e = errno;
165 dst->driver->staged_abort(st);
166 errno = e ? e : EIO;
167 return NGX_ERROR;
168 }
169 w += k;
170 }
171 off += r;
172 }
173
174 /* Atomic publish. On failure staged_commit leaves the handle valid, so abort
175 * to release it; KEEP the POSIX partial so a resume retry can republish. */
176 if (dst->driver->staged_commit(st, NULL /* replace allowed */) != NGX_OK) {
177 int e = errno;
178 dst->driver->staged_abort(st);
179 errno = e ? e : EIO;
180 return NGX_ERROR;
181 }
182 return NGX_OK;
183}

185/*
186 * commit_staged_to_backend — publish a POSIX-staged partial INTO a non-POSIX
187 * storage backend (e.g. pblock) by streaming it through that driver's staged
188 * write→commit state machine. This is the cross-FILESYSTEM atomic commit that
189 * rename(2) cannot do: the partial lives on an independent POSIX staging mount
190 * (brix_stage_dir) and the final export is a different, driver-owned namespace,
191 * so we read the partial through the (POSIX) source backend and write it through
192 * the destination driver's staged_write, then staged_commit (which publishes
193 * atomically — for pblock, a single catalog row insert pointing at the freshly
194 * written blocks). On success the POSIX partial is unlinked; on failure it is
195 * KEPT (a resume client can retry the close) and the driver's staged blob is
196 * aborted. Returns NGX_OK / NGX_ERROR (errno set).
197 */
198static ngx_int_t
199commit_staged_to_backend(ngx_fd_t fd, const char *stage_path,
200 const char *final_path, brix_sd_instance_t *dst, const char *root_canon,
201 ngx_log_t *log)
202{
203 const char *logical = commit_be_logical(final_path, root_canon);
204 brix_sd_staged_t *st;
205 struct stat sb;
206 mode_t mode = 0644;
207 off_t dsz;
208 int rfd, owned = 0, serr = 0;
209 ngx_int_t rc;
210
211 (void) log;
212
213 if (dst->driver->staged_open == NULL || dst->driver->staged_write == NULL
214 || dst->driver->staged_commit == NULL
215 || dst->driver->staged_abort == NULL)
216 {
217 errno = ENOTSUP;
218 return NGX_ERROR;
219 }
220
221 /* Source: the open staged partial if we still hold it, else open it O_RDONLY.
222 * Reads are positional (pread), so the fd's file position is irrelevant. */
223 rfd = cstb_open_source(fd, stage_path, &owned, &mode);
224 if (rfd == NGX_INVALID_FILE) {
225 return NGX_ERROR;
226 }
227
228 /* Phase-107 C5: the partial's size is exact — declare it so the backend
229 * driver can reserve (pblock quota-admits the bytes at open). */
230 dsz = (fstat(rfd, &sb) == 0) ? sb.st_size : 0;
231
232 st = dst->driver->staged_open(dst, logical, mode, dsz, &serr);
233 if (st == NULL) {
234 if (owned) { (void) close(rfd); }
235 errno = serr ? serr : EIO;
236 return NGX_ERROR;
237 }
238
239 rc = cstb_pump_and_commit(rfd, dst, st); /* aborts st itself on failure */
240 if (owned) { (void) close(rfd); }
241 if (rc != NGX_OK) {
242 return NGX_ERROR; /* errno set by the pump */
243 }
244
245 (void) unlink(stage_path); /* published → drop the partial */
246 return NGX_OK;
247}

249/*
250 * WHAT: Make the still-open staged fd durable and publish the caller's intended
251 * mode on it, just before any commit path reads it.
252 *
253 * WHY: close() alone does not flush, and a crash / power loss / ENOSPC mid-write
254 * must not expose a torn object — so the data must be fsync'd BEFORE the
255 * commit publishes it. The temp is written 0600 (private); the final mode is
256 * restored here so every downstream commit path derives the object's mode
257 * from the fd.
258 *
259 * HOW: Wrap fd as a POSIX object and fsync via the backend; a failing fsync means
260 * "not durable" → NGX_ERROR (do not publish). Then fchmod to final_mode when
261 * non-zero (0 means "leave the temp's current mode", e.g. root:// POSC temps
262 * that already carry the client's create mode). No-op for a closed fd.
263 *
264 * Parameters:
265 * fd — open staged fd, or NGX_INVALID_FILE (crash-recovery reap)
266 * final_mode — mode to publish, or 0 to keep the temp's mode
267 */
268static ngx_int_t
269commit_flush_and_chmod(ngx_fd_t fd, mode_t final_mode, int relaxed)
270{
271 brix_sd_obj_t sobj;
272
273 if (fd == NGX_INVALID_FILE) {
274 return NGX_OK;
275 }
276
277 /* `relaxed` (brix_durable_commit off): skip the data fsync — the operator
278 * chose stock-XRootD close semantics. The commit rename still orders after
279 * every completed write, so a CLEAN close never publishes torn data; only a
280 * host crash inside the writeback window can lose the tail. */
281 brix_sd_posix_wrap(&sobj, fd); /* durability flush via the backend */
282 if (!relaxed && sobj.driver->fsync(&sobj) != NGX_OK) {
283 return NGX_ERROR; /* not durable — do not publish */
284 }
285 /* SECURITY: publish the caller's intended mode on the still-open fd before any
286 * commit path reads it. */
287 if (final_mode != 0) {
288 (void) fchmod(fd, final_mode);
289 }
290 return NGX_OK;
291}

293/*
294 * WHAT: Cross-device commit — copy the staged partial to a temp ON THE FINAL
295 * filesystem, fsync it, atomically rename it onto the final path, and drop
296 * the staged copy.
297 *
298 * WHY: When the staged partial sits on a configured fast-cache device different
299 * from the storage (e.g. CEPHFS), rename(2) returns EXDEV and cannot publish
300 * it, so the bytes must be moved before the (now same-FS) atomic rename.
301 *
302 * HOW: Derive a temp adjacent to final_path, open the partial O_RDONLY, create
303 * the temp preserving the partial's mode, VFS↔VFS move + fsync, close, then
304 * rename temp→final and unlink the staged copy. Every error path unlinks the
305 * temp and restores the originating errno.
306 *
307 * Parameters:
308 * stage_path — the staged partial (cross-device source)
309 * final_path — destination path (drives the adjacent temp + final rename)
310 * log — for the phase-107 C7 lock-carry refusal report
311 */
312static ngx_int_t
313commit_cross_device(const char *stage_path, const char *final_path,
314 ngx_log_t *log)
315{
316 char tmp[PATH_MAX];
317 int rfd, dfd, e;
318 struct stat sb;
319 mode_t mode;
320
321 if (brix_make_tmp_path(final_path, tmp, sizeof(tmp)) != NGX_OK) {
322 errno = ENAMETOOLONG;
323 return NGX_ERROR;
324 }
325 rfd = open(stage_path, O_RDONLY | O_CLOEXEC);
326 if (rfd < 0) {
327 return NGX_ERROR;
328 }
329 /* Preserve the staged file's mode on the committed object. */
330 mode = (fstat(rfd, &sb) == 0) ? (sb.st_mode & 07777) : 0644;
331 dfd = open(tmp, O_WRONLY | O_CREAT | O_EXCL | O_NOFOLLOW | O_CLOEXEC, mode);
332 if (dfd < 0) {
333 e = errno; close(rfd); errno = e; return NGX_ERROR;
334 }
335 {
336 /* VFS↔VFS move: stage object (source backend) → temp object (final
337 * backend). Both POSIX today; the loop is driver-mediated so a non-POSIX
338 * mount on either side needs no change here. */
339 brix_sd_obj_t src_obj, dst_obj;
340 brix_sd_posix_wrap(&src_obj, rfd);
341 brix_sd_posix_wrap(&dst_obj, dfd);
342 if (stage_move_objects(&src_obj, &dst_obj) != NGX_OK
343 || dst_obj.driver->fsync(&dst_obj) != NGX_OK) {
344 e = errno; close(rfd); close(dfd); (void) unlink(tmp); errno = e;
345 return NGX_ERROR;
346 }
347 }
348 close(rfd);
349 if (close(dfd) != 0) {
350 e = errno; (void) unlink(tmp); errno = e; return NGX_ERROR;
351 }
352 /* Phase-107 C7: the data copy above moved bytes only — the lock record
353 * carried onto the STAGED partial (brix_commit_staged) did not follow it,
354 * so carry from the destination onto this adjacent temp before it
355 * publishes (brix_staged_lock_carry). */
356 if (brix_staged_lock_carry(log, final_path, tmp) != NGX_OK) {
357 e = errno; (void) unlink(tmp); errno = e; return NGX_ERROR;
358 }
359 if (rename(tmp, final_path) != 0) {
360 e = errno; (void) unlink(tmp); errno = e; return NGX_ERROR;
361 }
362 (void) unlink(stage_path); /* drop the cache copy; commit succeeded */
363 return NGX_OK;
364}

366/*
367 * WHAT: Commit a staged file onto its final path — atomically and across
368 * filesystems. When the final export uses a NON-POSIX storage backend
369 * (e.g. pblock), the partial is uploaded INTO that backend via the driver's
370 * staged write→commit state machine (commit_staged_to_backend). Otherwise
371 * it tries rename(2) first (atomic, same-FS, the fast path); on
372 * EXDEV (the staged file is on a configured fast-cache device different from
373 * the storage, e.g. CEPHFS) it copies the data to a temp ON THE FINAL
374 * filesystem, fsync()s, atomically renames that temp onto the final path,
375 * and unlinks the staged copy. Either way a concurrent reader sees only the
376 * old object or the complete new one — never a partial.
377 *
378 * WHY: Upload staging on a fast device + final storage on a POSIX mount means the
379 * commit can be a cross-device move, which rename(2) cannot do. Centralising
380 * this lets both root:// (close) and WebDAV (PUT) commit identically.
381 *
382 * fd: the open staged-file descriptor (for the durability fsync before commit), or
383 * NGX_INVALID_FILE if already closed. Returns NGX_OK / NGX_ERROR (errno set).
384 */
385ngx_int_t
386brix_commit_staged(ngx_fd_t fd, const char *stage_path, const char *final_path,
387 mode_t final_mode, unsigned flags, ngx_log_t *log)
388{
389 const char *be_root = NULL;
390 brix_sd_instance_t *dst;
391
392 if (commit_flush_and_chmod(fd, final_mode,
393 (flags & BRIX_COMMIT_RELAXED) != 0) != NGX_OK) {
394 return NGX_ERROR;
395 }
396
397 /* Non-POSIX final export (e.g. pblock): the staged POSIX partial lives on an
398 * independent staging mount and the final namespace is driver-owned, so
399 * rename(2) cannot publish it. Upload the partial INTO the backend via its
400 * staged write→commit state machine (a cross-filesystem atomic publish).
401 * POSIX exports (the common case) skip this and take the rename path below. */
402 dst = brix_vfs_backend_resolve_for_path(final_path, &be_root, log);
403 if (dst != NULL && dst->driver != NULL
404 && dst->driver != brix_sd_default_driver())
405 {
406 return commit_staged_to_backend(fd, stage_path, final_path, dst,
407 be_root, log);
408 }
409
410 /* Phase-107 C7: every rename route below REPLACES the destination inode —
411 * carry a live lock record onto the staged partial first, so an ADMITTED
412 * write under a lock (advisory enforcement; root:// strict refuses at open)
413 * does not silently discharge it (brix_staged_lock_carry). The backend
414 * route above publishes through the driver's own staged commit, which
415 * carries inside staged_commit_internal. */
416 if (brix_staged_lock_carry(log, final_path, stage_path) != NGX_OK) {
417 return NGX_ERROR;
418 }
419
420 /*
421 * Phase 40: under impersonation the final path is the mapped user's private
422 * tree (e.g. a 0700 home the unprivileged worker cannot enter), so the publish
423 * rename MUST run as that user via the broker — the raw rename(2) below runs as
424 * the worker and EACCESes. Because the native xrootd client retries the close,
425 * that EACCES manifests as a hung root:// upload rather than a clean error.
426 * brix_rename_confined_canon routes to the broker when a request principal is
427 * active (and confines under be_root otherwise); the fd fsync above stays
428 * worker-side (fd ops ignore fsuid). EXDEV (cross-device stage mount) can't go
429 * through the broker, so it falls through to the cross-device mover.
430 */
431 if (brix_imp_client_active() && be_root != NULL) {
432 /* The fd fchmod in commit_flush_and_chmod ran as the unprivileged worker
433 * and was a silent EPERM no-op on the mapped user's file, so the staged
434 * 0600 (created PRIVATE) would survive the publish and block group DAC on a
435 * file uploaded into a shared/setgid dir. Re-apply the intended final mode
436 * AS THE MAPPED USER via the broker, on the staged temp before the rename
437 * (a 0700 private dir still isolates it; a 0770 shared dir now permits the
438 * group read the mode intends). Best-effort: a cross-device stage temp is
439 * not under be_root, so the confine fails and the mode stays staged — the
440 * same EXDEV edge the rename below hands to the cross-device mover. */
441 if (final_mode != 0) {
442 (void) brix_chmod_confined_canon(log, be_root, stage_path, final_mode);
443 }
444 if (brix_rename_confined_canon(log, be_root, stage_path,
445 final_path) == 0)
446 {
447 return NGX_OK;
448 }
449 if (errno != EXDEV) {
450 return NGX_ERROR;
451 }
452 /* EXDEV: fall through to the cross-device mover below. */
453 }
454
455 if (rename(stage_path, final_path) == 0) {
456 return NGX_OK; /* same filesystem: atomic, zero-copy */
457 }
458 if (errno != EXDEV) {
459 return NGX_ERROR;
460 }
461
462 /* Cross-device commit: copy the staged data to a temp adjacent to (and on the
463 * same filesystem as) the final path, then atomically rename it into place. */
464 return commit_cross_device(stage_path, final_path, log);
465}
