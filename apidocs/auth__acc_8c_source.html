1/* auth_acc.c — the XrdAcc authorization tier for the S3 plane.
2 * WHAT: Map an S3 method to an XrdAcc operation, then ask the acc engine
3 * whether this identity may perform it on this URI.
4 * WHY: Sits beside its auth siblings (auth_bearer.c, auth_sigv4_*.c) rather
5 * than inside handler.c, which owns classification and dispatch and had grown
6 * past the file-size contract carrying this gate as well.
7 * HOW: Two entry points declared in s3_handler_internal.h; the deny path emits
8 * the same XML error vocabulary every other S3 refusal uses.
9 */

16/* Map an S3 request method to the XrdAcc operation it requires. */
17brix_acc_op_t
18s3_method_aop(ngx_http_request_t *r)
19{
20 switch (r->method) {
21 case NGX_HTTP_GET: return BRIX_AOP_READ; /* GetObject / ListObjects */
22 case NGX_HTTP_HEAD: return BRIX_AOP_STAT;
23 case NGX_HTTP_PUT: return BRIX_AOP_CREATE;
24 case NGX_HTTP_POST: return BRIX_AOP_CREATE; /* multipart upload */
25 case NGX_HTTP_DELETE: return BRIX_AOP_DELETE;
26 default: return BRIX_AOP_STAT;
27 }
28}

30/* ---- Identify the body-addressed DeleteObjects operation ----
31 *
32 * WHAT: Returns non-zero only for POST ?delete on the configured bucket root,
33 * either /bucket or /bucket/. Object-key POST requests always return zero.
34 *
35 * WHY: DeleteObjects cannot be authorized at the ordinary request gate because
36 * its resources arrive later in XML. The exemption must be just as narrow as
37 * the dispatcher route so an object-key query spelling cannot skip XrdAcc.
38 *
39 * HOW: Require POST and the parsed delete query flag, then byte-compare r->uri
40 * with the two bucket-root spellings accepted by the path-style URI parser.
41 */
42ngx_flag_t
43s3_is_delete_objects_request(ngx_http_request_t *r,
44 const ngx_http_s3_loc_conf_t *cf)
45{
46 size_t root_len;
47
48 if (r->method != NGX_HTTP_POST || !s3_has_query_flag(r, "delete")) {
49 return 0;
50 }
51
52 if (cf->bucket.len > (size_t) -2) {
53 return 0;
54 }
55 root_len = cf->bucket.len + 1;
56 if (r->uri.len != root_len && r->uri.len != root_len + 1) {
57 return 0;
58 }
59 if (r->uri.data[0] != '/') {
60 return 0;
61 }
62 if (cf->bucket.len != 0
63 && ngx_strncmp(r->uri.data + 1, cf->bucket.data, cf->bucket.len) != 0)
64 {
65 return 0;
66 }
67 return r->uri.len == root_len || r->uri.data[root_len] == '/';
68}

70/* ---- Authorize one exact S3 logical resource through XrdAcc ----
71 *
72 * WHAT: Returns NGX_OK when XrdAcc is not selected or grants `op` on `path`;
73 * returns NGX_ERROR on denial. It never writes an HTTP response.
74 *
75 * WHY: Body- and header-addressed S3 operations need authorization after their
76 * resource is decoded. Sharing the identity/peer construction with the request
77 * gate prevents those late decisions from drifting in host or VO semantics.
78 *
79 * HOW: Build the same entity and peer tuple used by the request gate, replace
80 * the peer with its cached reverse name when configured, and call the shared
81 * XrdAcc HTTP decision engine with the caller's exact operation and path.
82 */
83ngx_int_t
84s3_acc_authorize_path(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
85 brix_identity_t *id, brix_acc_op_t op, const char *path)
86{
87 const char *name = "", *vorg = "", *role = "", *grp = "";
88 char host[BRIX_S3_HANDLER_HOST_BUF];
89 size_t host_len;
90
91 if (cf->common.acc.format != BRIX_AUTHDB_FORMAT_XRDACC) {
92 return NGX_OK;
93 }
94 if (id != NULL) {
95 name = brix_identity_dn_cstr(id);
96 vorg = brix_identity_acc_vorg_cstr(id);
97 role = brix_identity_acc_role_cstr(id);
98 grp = brix_identity_acc_group_cstr(id);
99 }
100
101 host_len = ngx_min(r->connection->addr_text.len, sizeof(host) - 1);
102 ngx_memcpy(host, r->connection->addr_text.data, host_len);
103 host[host_len] = '\0';
104
105 if (cf->common.acc.resolve_hosts) {
106 char reverse_name[BRIX_DNS_REVERSE_NAME_LEN];
107
108 if (brix_acc_resolve_peer(cf->common.dns.policy,
109 r->connection->sockaddr,
110 r->connection->socklen,
111 reverse_name, sizeof(reverse_name)) == NGX_OK)
112 {
113 host_len = ngx_min(ngx_strlen(reverse_name), sizeof(host) - 1);
114 ngx_memcpy(host, reverse_name, host_len);
115 host[host_len] = '\0';
116 }
117 }
118
119 return brix_acc_http_authorize(r->pool, r->connection->log,
120 &cf->common.acc, name, host, vorg, role, grp,
121 op, path);
122}

124/*
125 * s3_acc_check — XrdAcc tier for S3 (when `brix_acc_format xrdacc`).
126 *
127 * Returns NGX_OK when the request may proceed (allowed, or the engine is not
128 * selected) — and, on a DENY, the result of the XML error it has already sent,
129 * which is NGX_OK too. The caller distinguishes them by r->header_sent, the
130 * same contract s3_verify_sigv4 has; see the comment at the call site.
131 *
132 * Denying by RETURNING NGX_HTTP_FORBIDDEN, as this gate used to, is what every
133 * other refusal in this module deliberately does not do: nginx then renders its
134 * own 403 page, so the response carries no <Error><Code> at all. An S3 client
135 * reads the code, not the prose — boto3 raises ClientError off it, and the
136 * multi-user oracle infers the deciding TIER from it — so a bodyless 403 is
137 * indistinguishable from a proxy's, and this tier's verdict could not be told
138 * apart from an authentication failure. Every sibling gate (bearer, SigV4,
139 * token scope, write-disabled, the resolver) answers with AccessDenied; this
140 * one now does as well.
141 */
142ngx_int_t
143s3_acc_check(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
144 brix_identity_t *id)
145{
146 char path[BRIX_S3_HANDLER_PATH_BUF];
147 size_t n;
148 ngx_int_t rc;
149
150 if (cf->common.acc.format != BRIX_AUTHDB_FORMAT_XRDACC) {
151 return NGX_OK;
152 }
153
154 /* The body parser performs one exact DELETE decision per resolved key
155 * before the VFS batch. Authorizing the bucket URI here would either ask
156 * for the wrong CREATE privilege or reject legitimate delete-only grants. */
157 if (s3_is_delete_objects_request(r, cf)) {
158 return NGX_OK;
159 }
160
161 n = ngx_min(r->uri.len, sizeof(path) - 1);
162 ngx_memcpy(path, r->uri.data, n);
163 path[n] = '\0';
164
165 rc = s3_acc_authorize_path(r, cf, id, s3_method_aop(r), path);
166 if (rc != NGX_ERROR) {
167 return NGX_OK;
168 }
169
170 return s3_fail(r, NGX_HTTP_FORBIDDEN, "AccessDenied",
171 "xrdacc denied", BRIX_S3_EVENT_ACCESS_DENIED);
172}
