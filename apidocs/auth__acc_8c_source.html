1/* auth_acc.c — the XrdAcc authorization tier for the S3 plane.
2 * WHAT: Map an S3 method to an XrdAcc operation, then ask the acc engine
3 * whether this identity may perform it on this URI.
4 * WHY: Sits beside its auth siblings (auth_bearer.c, auth_sigv4_*.c) rather
5 * than inside handler.c, which owns classification and dispatch and had grown
6 * past the file-size contract carrying this gate as well.

17brix_acc_op_t
18s3_method_aop(ngx_http_request_t *r)
19{
20 switch (r->method) {
21 case NGX_HTTP_GET: return BRIX_AOP_READ; /* GetObject / ListObjects */
22 case NGX_HTTP_HEAD: return BRIX_AOP_STAT;
23 case NGX_HTTP_PUT: return BRIX_AOP_CREATE;
24 case NGX_HTTP_POST: return BRIX_AOP_CREATE; /* multipart upload */
25 case NGX_HTTP_DELETE: return BRIX_AOP_DELETE;
26 default: return BRIX_AOP_STAT;
27 }
28}

30/*
31 * s3_acc_check — XrdAcc tier for S3 (when `brix_acc_format xrdacc`).
32 *
33 * Returns NGX_OK when the request may proceed (allowed, or the engine is not
34 * selected) — and, on a DENY, the result of the XML error it has already sent,
35 * which is NGX_OK too. The caller distinguishes them by r->header_sent, the
36 * same contract s3_verify_sigv4 has; see the comment at the call site.
37 *
38 * Denying by RETURNING NGX_HTTP_FORBIDDEN, as this gate used to, is what every
39 * other refusal in this module deliberately does not do: nginx then renders its
40 * own 403 page, so the response carries no <Error><Code> at all. An S3 client
41 * reads the code, not the prose — boto3 raises ClientError off it, and the
42 * multi-user oracle infers the deciding TIER from it — so a bodyless 403 is
43 * indistinguishable from a proxy's, and this tier's verdict could not be told
44 * apart from an authentication failure. Every sibling gate (bearer, SigV4,
45 * token scope, write-disabled, the resolver) answers with AccessDenied; this
46 * one now does as well.
47 */
48ngx_int_t
49s3_acc_check(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
50 brix_identity_t *id)
51{
52 const char *name = "", *vorg = "", *role = "", *grp = "";
53 char host[BRIX_S3_HANDLER_HOST_BUF], path[BRIX_S3_HANDLER_PATH_BUF];
54 size_t n;
55 ngx_int_t rc;
56
57 if (cf->common.acc.format != BRIX_AUTHDB_FORMAT_XRDACC) {
58 return NGX_OK;
59 }
60 if (id != NULL) {
61 name = brix_identity_dn_cstr(id); /* S3 access key (or subject) */
62 vorg = brix_identity_acc_vorg_cstr(id);
63 role = brix_identity_acc_role_cstr(id);
64 grp = brix_identity_acc_group_cstr(id);
65 }
66 n = ngx_min(r->connection->addr_text.len, sizeof(host) - 1);
67 ngx_memcpy(host, r->connection->addr_text.data, n);
68 host[n] = '\0';
69
70 /* Opt-in reverse DNS for `h <host>`/`h .domain` rules: a cache probe —
71 * the PREACCESS wait (core/http/http_peer_name.c) fetched the answer. */
72 if (cf->common.acc.resolve_hosts) {
73 char hbuf[BRIX_DNS_REVERSE_NAME_LEN];
74
75 if (brix_acc_resolve_peer(cf->common.dns.policy,
76 r->connection->sockaddr,
77 r->connection->socklen,
78 hbuf, sizeof(hbuf)) == NGX_OK)
79 {
80 n = ngx_min(ngx_strlen(hbuf), sizeof(host) - 1);
81 ngx_memcpy(host, hbuf, n);
82 host[n] = '\0';
83 }
84 }
85
86 n = ngx_min(r->uri.len, sizeof(path) - 1);
87 ngx_memcpy(path, r->uri.data, n);
88 path[n] = '\0';
89
90 rc = brix_acc_http_authorize(r->pool, r->connection->log,
91 &cf->common.acc, name, host, vorg, role, grp,
92 s3_method_aop(r), path);
93 if (rc != NGX_ERROR) {
94 return NGX_OK;
95 }
96
97 return s3_fail(r, NGX_HTTP_FORBIDDEN, "AccessDenied",
98 "xrdacc denied", BRIX_S3_EVENT_ACCESS_DENIED);
99}
