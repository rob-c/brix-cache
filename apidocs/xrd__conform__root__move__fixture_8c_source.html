78/* WHAT: Attempt one rename between two explicitly asserted export roots.
79 * WHY: A server must not implement cross-export rename as an unreported copy;
80 * the source and destination state are part of the security oracle.
81 * HOW: Mint one run-keyed child beneath each scratch root, create the source,
82 * require a defined refusal, verify source-present/destination-absent,
83 * and clean both paths even after a divergent response. */
84void
85xrd_conform_probe_cross_export_move(xrd_conform_execution *execution)
86{
87 brix_conform_result *row = xrd_conform_result_find(execution, "MV-005");
88 xrd_conform_scratch_object source, destination;
89 brix_url source_url, destination_url;
90 brix_conn connection;
91 brix_status status;
92 char leaf[80];
93 char received[160];
94 int move_rc = -1;
95 int move_kxr = 0;
96 int source_present = 0;
97 int destination_present = 0;
98 int pass;
99
100 if (!move_row_active(row)) return;
101 if (execution->run->move_fixture_target == NULL) return;
102 brix_status_clear(&status);
103 if (brix_endpoint_parse(execution->run->config.target, &source_url,
104 &status) != 0
105 || brix_endpoint_parse(execution->run->move_fixture_target,
106 &destination_url, &status) != 0
107 || !same_authority(&source_url, &destination_url)) {
108 xrd_conform_result_set(execution, "MV-005",
109 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
110 "move-fixture-authority-mismatch", NULL, status.msg);
111 return;
112 }
113 snprintf(leaf, sizeof(leaf), "brix-conform-mv-cross-%s",
114 execution->run->key_hex);
115 if (brix_conform_scratch_path(execution->run->config.scratch, leaf,
116 source.path, sizeof(source.path)) != 0
117 || brix_conform_scratch_path(destination_url.path, leaf,
118 destination.path, sizeof(destination.path)) != 0) return;
119 if (brix_connect(&connection, &source_url, &execution->run->opts,
120 &status) != 0) {
121 xrd_conform_result_set(execution, "MV-005",
122 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
123 "two-export-move-fixture-unavailable", NULL, status.msg);
124 return;
125 }
126 if (create_source(execution, &connection, &source, &status) != 0
127 || object_exists(execution, &connection, &destination, &status)) {
128 cleanup_objects(execution, &connection, &source, &destination);
129 brix_close(&connection);
130 xrd_conform_result_set(execution, "MV-005",
131 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
132 "two-export-move-fixture-invalid-state", NULL, status.msg);
133 return;
134 }
135 brix_status_clear(&status);
136 if (xrd_conform_reserve_zero_status(execution, &status))
137 move_rc = brix_mv(&connection, source.path, destination.path, &status);
138 move_kxr = status.kxr;
139 source_present = object_exists(execution, &connection, &source, &status);
140 destination_present = object_exists(execution, &connection, &destination,
141 &status);
142 snprintf(received, sizeof(received),
143 "mv=%d kxr=%d source=%d destination=%d", move_rc, move_kxr,
144 source_present, destination_present);
145 pass = move_refused_safely(move_rc, move_kxr, source_present,
146 destination_present);
147 xrd_conform_result_set(execution, "MV-005",
148 pass ? BRIX_CONFORM_VERDICT_CONFORMANT
149 : BRIX_CONFORM_VERDICT_DIVERGENT,
150 pass ? "cross-export-move-refused-without-copy"
151 : "cross-export-move-was-undefined-or-mutated-state",
152 "defined refusal; source=present destination=absent", received);
153 cleanup_objects(execution, &connection, &source, &destination);
154 brix_close(&connection);
155}
