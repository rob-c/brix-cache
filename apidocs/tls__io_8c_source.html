2 * tls_io.c — TLS byte transfer for the native client.
3 *
4 * WHAT: The read/write half of the client's TLS support — brix_tls_read (fills
5 * exactly n, for root:// framing), brix_tls_read_some (one record, for the
6 * HTTP client) and brix_tls_write, plus the retry/deadline helpers they
7 * share.
8 * WHY: Split from tls.c when that file crossed the 600-line cap
9 * (coding-standards §1). Session setup and byte transfer are separate
10 * concerns, and the slow-drip deadline logic is the part worth reading
11 * without the SSL_CTX plumbing around it.
12 * HOW: The fd is non-blocking, so every operation drives OpenSSL's
13 * WANT_READ/WANT_WRITE through brix_tls_wait_io() under one whole-operation
14 * deadline; a stalled peer trips the deadline rather than the idle timeout.
15 * Setup, teardown and introspection stay in tls.c; see tls_internal.h.

29/* Clamp the idle timeout to a whole-operation deadline; see read_wait_ms in
30 * sock.c for the rationale (slow-drip guard). -1 = deadline passed. */
31static int
32tls_read_wait_ms(int timeout_ms, uint64_t deadline_ns)
33{
34 if (deadline_ns == 0) {
35 return timeout_ms;
36 }
37 {
38 uint64_t now = brix_mono_ns();
39 int64_t rem_ms;
40
41 if (now >= deadline_ns) {
42 return -1;
43 }
44 rem_ms = (int64_t) ((deadline_ns - now) / 1000000ULL);
45 if (rem_ms <= 0) {
46 return -1;
47 }
48 if (timeout_ms > 0 && (int64_t) timeout_ms < rem_ms) {
49 return timeout_ms;
50 }
51 return (int) rem_ms;
52 }
53}

55/*
56 * WHAT: Wait for the socket readiness requested by a retryable TLS read.
57 * WHY: WANT_READ/WANT_WRITE must share one slow-drip deadline and diagnostics.
58 * HOW: Clamp poll time to the operation deadline and translate timeout/error.
59 */
60static int
61tls_read_retry_wait(brix_io *io, int ssl_error, uint64_t deadline_ns,
62 size_t got, size_t wanted, brix_status *status)
63{
64 short events = ssl_error == SSL_ERROR_WANT_READ ? POLLIN : POLLOUT;
65 int wait_ms = tls_read_wait_ms(io->timeout_ms, deadline_ns);
66 int ready;
67
68 if (wait_ms < 0) {
69 brix_status_set(status, XRDC_ESOCK, ETIMEDOUT,
70 "TLS read exceeded slow-drip deadline "
71 "(%d ms, got %zu/%zu bytes)",
72 io->stall_deadline_ms, got, wanted);
73 return -1;
74 }
75 ready = brix_tls_wait_io(io->fd, events, wait_ms);
76 if (ready == 0 && deadline_ns != 0 && brix_mono_ns() >= deadline_ns) {
77 brix_status_set(status, XRDC_ESOCK, ETIMEDOUT,
78 "TLS read exceeded slow-drip deadline "
79 "(%d ms, got %zu/%zu bytes)",
80 io->stall_deadline_ms, got, wanted);
81 return -1;
82 }
83 if (ready == 0) {
84 brix_status_set(status, XRDC_ESOCK, ETIMEDOUT, "TLS read timed out");
85 return -1;
86 }
87 if (ready < 0) {
88 brix_status_set(status, XRDC_ESOCK, errno, "poll(tls read): %s",
89 strerror(errno));
90 return -1;
91 }
92 return 0;
93}

95/*
96 * WHAT: Translate a terminal SSL_read_ex failure into the client status model.
97 * WHY: Peer close, syscall failure, and TLS protocol errors need distinct reports.
98 * HOW: Map SSL error classes, preserving retryable errno only for the caller.
99 */
100static int
101tls_read_failure(int ssl_error, size_t got, size_t wanted, brix_status *status)
102{
103 if (ssl_error == SSL_ERROR_ZERO_RETURN) {
104 brix_status_set(status, XRDC_ESOCK, 0,
105 "TLS connection closed by peer (read %zu/%zu)",
106 got, wanted);
107 return -1;
108 }
109 if (ssl_error == SSL_ERROR_SYSCALL) {
110 if (errno == EINTR || errno == EAGAIN)
111 return 1;
112 brix_status_set(status, XRDC_ESOCK, errno, "TLS read: %s",
113 errno ? strerror(errno) : "unexpected EOF");
114 return -1;
115 }
116 brix_tls_err(status, XRDC_ESOCK, "TLS read");
117 return -1;
118}

120int
121brix_tls_read(brix_io *io, void *buf, size_t n, brix_status *st)
122{
123 SSL *ssl = (SSL *) io->ssl;
124 uint8_t *p = (uint8_t *) buf;
125 size_t got = 0;
126 /* Shared absolute cutoff for the whole logical operation (armed once by
127 * brix_io_stall_arm), so a multi-record TLS read shares one budget rather
128 * than re-arming per record. 0 = disabled. */
129 uint64_t deadline_ns = io->stall_deadline_ns;
130
131 while (got < n) {
132 size_t nread = 0;
133 int ok, err;
134
135 if (brix_copy_quit_requested()) { /* Phase 40 (a): prompt cancel */
136 brix_status_set(st, XRDC_ESOCK, EINTR, "transfer cancelled (signal)");
137 return -1;
138 }
139 ERR_clear_error(); /* so SSL_get_error reflects THIS op, not a stale queue */
140 ok = SSL_read_ex(ssl, p + got, n - got, &nread);
141
142 if (ok == 1) {
143 got += nread;
144 continue;
145 }
146 err = SSL_get_error(ssl, 0);
147 if (err == SSL_ERROR_WANT_READ || err == SSL_ERROR_WANT_WRITE) {
148 if (tls_read_retry_wait(io, err, deadline_ns, got, n, st) != 0)
149 return -1;
150 continue;
151 }
152 if (tls_read_failure(err, got, n, st) < 0)
153 return -1;
154 }
155 return 0;
156}

158int
159brix_tls_write(brix_io *io, const void *buf, size_t n, brix_status *st)
160{
161 SSL *ssl = (SSL *) io->ssl;
162 const uint8_t *p = (const uint8_t *) buf;
163 size_t sent = 0;
164
165 while (sent < n) {
166 size_t nw = 0;
167 int ok, err;
168
169 if (brix_copy_quit_requested()) { /* Phase 40 (a): prompt cancel */
170 brix_status_set(st, XRDC_ESOCK, EINTR, "transfer cancelled (signal)");
171 return -1;
172 }
173 ERR_clear_error();
174 ok = SSL_write_ex(ssl, p + sent, n - sent, &nw);
175
176 if (ok == 1) {
177 sent += nw;
178 continue;
179 }
180 err = SSL_get_error(ssl, 0);
181 if (err == SSL_ERROR_WANT_READ || err == SSL_ERROR_WANT_WRITE) {
182 short ev = (err == SSL_ERROR_WANT_READ) ? POLLIN : POLLOUT;
183 int pr = brix_tls_wait_io(io->fd, ev, io->timeout_ms);
184 if (pr == 0) {
185 brix_status_set(st, XRDC_ESOCK, ETIMEDOUT, "TLS write timed out");
186 return -1;
187 }
188 if (pr < 0) {
189 brix_status_set(st, XRDC_ESOCK, errno, "poll(tls write): %s",
190 strerror(errno));
191 return -1;
192 }
193 continue;
194 }
195 if (err == SSL_ERROR_SYSCALL) {
196 if (errno == EINTR || errno == EAGAIN) {
197 continue;
198 }
199 brix_status_set(st, XRDC_ESOCK, errno, "TLS write: %s",
200 errno ? strerror(errno) : "unexpected EOF");
201 return -1;
202 }
203 brix_tls_err(st, XRDC_ESOCK, "TLS write");
204 return -1;
205 }
206 return 0;
207}

209/*
210 * Read UP TO n bytes (one record) over TLS — unlike brix_tls_read (which fills
211 * exactly n for root:// framing), this is a stream read for the HTTP client: sets
212 * *got to the bytes read (0 = clean EOF / peer close) and returns 0, or -1 on error.
213 */
214int
215brix_tls_read_some(brix_io *io, void *buf, size_t n, size_t *got, brix_status *st)
216{
217 SSL *ssl = (SSL *) io->ssl;
218 size_t nread = 0;
219
220 *got = 0;
221 for (;;) {
222 int ok, err;
223 ERR_clear_error();
224 ok = SSL_read_ex(ssl, buf, n, &nread);
225 if (ok == 1) {
226 *got = nread;
227 return 0;
228 }
229 err = SSL_get_error(ssl, 0);
230 if (err == SSL_ERROR_ZERO_RETURN) {
231 return 0; /* clean close → *got stays 0 (EOF) */
232 }
233 if (err == SSL_ERROR_WANT_READ || err == SSL_ERROR_WANT_WRITE) {
234 short ev = (err == SSL_ERROR_WANT_READ) ? POLLIN : POLLOUT;
235 int pr = brix_tls_wait_io(io->fd, ev, io->timeout_ms);
236 if (pr <= 0) {
237 brix_status_set(st, XRDC_ESOCK, pr == 0 ? ETIMEDOUT : errno,
238 "TLS read %s", pr == 0 ? "timed out" : "poll failed");
239 return -1;
240 }
241 continue;
242 }
243 if (err == SSL_ERROR_SYSCALL && errno == 0) {
244 return 0; /* unclean EOF — treat as end of body */
245 }
246 if (err == SSL_ERROR_SYSCALL && (errno == EINTR || errno == EAGAIN)) {
247 continue;
248 }
249 brix_tls_err(st, XRDC_ESOCK, "TLS read");
250 return -1;
251 }
