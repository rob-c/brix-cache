11/* Resolve the S3 access/secret key pair for SigV4 signing.
12 *
13 * WHY: precedence is frozen — explicit opts first, then the credential store
14 * (co->cred) when set, then the AWS_* environment, so env-sourced credentials
15 * behave identically to today. A store-acquire failure is not an error: the
16 * status is cleared and resolution falls through to the environment. Either
17 * pointer may come back NULL (anonymous access — the caller decides). */
18static void
19s3_resolve_keys(const brix_copy_opts *o, const brix_opts *co,
20 const char **ak, const char **sk, brix_status *st)
21{
22 brix_cred_view sv;
23
24 *ak = (o && o->s3_access) ? o->s3_access : NULL;
25 *sk = (o && o->s3_secret) ? o->s3_secret : NULL;
26
27 /* Prefer the cred store for S3 keys when no explicit opts override. */
28 if ((*ak == NULL || *sk == NULL) && co != NULL && co->cred != NULL) {
29 if (brix_cred_acquire(co->cred, XRDC_CRED_S3KEYS, 0, &sv, st) == 0) {
30 if (*ak == NULL) { *ak = sv.s3_access; }
31 if (*sk == NULL) { *sk = sv.s3_secret; }
32 } else {
33 brix_status_clear(st);
34 }
35 }
36 /* Fall through to env when store not set or acquire failed. */
37 if (*ak == NULL) { *ak = getenv("AWS_ACCESS_KEY_ID"); }
38 if (*sk == NULL) { *sk = getenv("AWS_SECRET_ACCESS_KEY"); }
39}

42/* Build the S3 SigV4 Authorization block for a->u into hdrs[].
43 *
44 * HOW: resolve keys (opts → store → env; both missing = anonymous, empty
45 * hdrs); host is signed as "host:port" to match the wire Host header
46 * byte-for-byte (brackets IPv6 literals); UNSIGNED-PAYLOAD for every method
47 * because the body streams and is not folded into the signature (both
48 * nginx-xrootd's S3 and real AWS accept that). Returns 0/-1 (a->st set). */
49static int
50auth_hdr_s3(const web_auth_ctx *a, char *hdrs, size_t hdrsz)
51{
52 const brix_weburl *u = a->u;
53 const brix_copy_opts *o = a->o;
54 brix_status *st = a->st;
55 const char *ak, *sk;
56 const char *rg = (o && o->s3_region) ? o->s3_region
57 : getenv("AWS_DEFAULT_REGION");
58 char host[300], payhash[65];
59
60 s3_resolve_keys(o, a->co, &ak, &sk, st);
61 if (ak == NULL || sk == NULL) {
62 return 0; /* anonymous — server may permit unsigned access */
63 }
64 if (rg == NULL) { rg = "us-east-1"; }
65 /* A '?' would split path vs query in the server's canonical request but
66 * we sign the whole path as CanonicalURI — reject rather than mis-sign. */
67 if (strchr(u->path, '?') != NULL) {
68 brix_status_set(st, XRDC_EUSAGE, 0,
69 "s3: query strings in the URL are not supported");
70 return -1;
71 }
72 /* The SigV4 signed host MUST match the wire Host header byte-for-byte; that
73 * header brackets IPv6 literals ([::1]:9000), so sign the same form. */
74 brix_format_host_port(u->host, (uint16_t) u->port, host, sizeof(host));
75 /* UNSIGNED-PAYLOAD for every method: the body isn't folded into the
76 * signature (it streams), which both nginx-xrootd's S3 and real AWS accept. */
77 snprintf(payhash, sizeof(payhash), "UNSIGNED-PAYLOAD");
78 if (brix_s3_sign_v4(a->method, host, u->path, ak, sk, rg, payhash,
79 hdrs, hdrsz) != 0) {
80 brix_status_set(st, XRDC_EAUTH, 0, "s3: failed to build SigV4 signature");
81 return -1;
82 }
83 return 0;
84}

87/* Build the WebDAV/HTTP bearer Authorization header into hdrs[] (left empty
88 * for an anonymous endpoint).
89 *
90 * WHY: precedence is frozen — explicit opts first, then the credential store
91 * (co->cred) when set, then $BEARER_TOKEN; a store-acquire failure clears the
92 * status and falls through to the environment. Returns 0/-1 (st set). */
93static int
94auth_hdr_bearer(const brix_copy_opts *o, const brix_opts *co,
95 char *hdrs, size_t hdrsz, brix_status *st)
96{
97 const char *tok = (o && o->bearer) ? o->bearer : NULL;
98 brix_cred_view bv;
99
100 /* Prefer the cred store for the bearer token when no explicit opt override. */
101 if (tok == NULL && co != NULL && co->cred != NULL) {
102 if (brix_cred_acquire(co->cred, XRDC_CRED_BEARER, 0, &bv, st) == 0
103 && bv.token != NULL) {
104 tok = bv.token;
105 } else {
106 brix_status_clear(st);
107 }
108 }
109 /* Fall through to env when store not set or acquire failed. */
110 if (tok == NULL) { tok = getenv("BEARER_TOKEN"); }
111
112 if (tok != NULL && tok[0] != '\0') {
113 int n = snprintf(hdrs, hdrsz, "Authorization: Bearer %s\r\n", tok);
114 if (n < 0 || (size_t) n >= hdrsz) {
115 brix_status_set(st, XRDC_EUSAGE, 0, "bearer token too long");
116 return -1;
117 }
118 }
119 return 0;
120}

123/* Build the auth header block for a web request into hdrs[] (may be empty for an
124 * anonymous endpoint). S3 → SigV4 (host signed as "host:port" to match the Host
125 * header we send); WebDAV/HTTP → Authorization: Bearer if a token is available.
126 *
127 * a->co carries the credential store (co->cred); when set the store is tried
128 * first for both the bearer token and S3 keys, falling back to opts/env on
129 * failure so env-sourced credentials behave identically to today. */
130int
131web_auth_headers(const web_auth_ctx *a, char *hdrs, size_t hdrsz)
132{
133 hdrs[0] = '\0';
134 if (a->u->is_s3) {
135 return auth_hdr_s3(a, hdrs, hdrsz);
136 }
137 return auth_hdr_bearer(a->o, a->co, hdrs, hdrsz, a->st);
138}

141/* Transfer result of one streaming HTTP GET (web_dl_fetch out-params).
142 * Bundled so the fetch wrapper stays under the 5-parameter gate. */
143typedef struct {
144 int outfd; /* IN: destination fd (file or STDOUT_FILENO) */
145 int status; /* OUT: HTTP status */
146 long long blen; /* OUT: body bytes written */
147} web_dl_io;

149/* Run one streaming HTTP GET of su into io->outfd.
150 *
151 * WHY: both the stdout and to-file branches of copy_web_download issue the
152 * same brix_http_download call (empty hdrs → NULL, co-optional verify/CA
153 * defaults); centralizing it keeps the two call sites literally identical. */
154static int
155web_dl_fetch(const brix_weburl *su, const brix_opts *co, const char *hdrs,
156 web_dl_io *io, brix_status *st)
157{
158 char proxybuf[512];
159 const char *pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
160
161 return brix_http_download(su->host, su->port, su->tls, su->path,
162 hdrs[0] ? hdrs : NULL, co ? co->verify_host : 1,
163 co ? co->ca_dir : NULL, pcert, io->outfd,
164 brix_tmo_io_ms(), &io->status, &io->blen, st);
165}

168int
169copy_web_download(const web_dl_req *rq, brix_status *st)
170{
171 const brix_weburl *su = rq->su;
172 const brix_url *du = rq->du;
173 const brix_copy_opts *o = rq->o;
174 const brix_opts *co = rq->co;
175 char hdrs[8192];
176 char tmp[XRDC_PATH_MAX];
177 int rc;
178 web_dl_io io = { -1, 0, 0 };
179 web_auth_ctx a = { su, "GET", o, co, st };
180
181 if (web_auth_headers(&a, hdrs, sizeof(hdrs)) != 0) {
182 return -1;
183 }
184 if (rq->to_stdout) {
185 io.outfd = STDOUT_FILENO;
186 return web_dl_fetch(su, co, hdrs, &io, st);
187 }
188 /* Refuse to overwrite an existing destination unless -f. */
189 if (!(o && o->force) && access(du->path, F_OK) == 0) {
190 brix_status_set(st, XRDC_EUSAGE, 0,
191 "destination exists (use -f to overwrite): %s", du->path);
192 return -1;
193 }
194 /* Download to a temp sibling and atomically rename on success: a failed
195 * transfer must never truncate or delete a pre-existing destination. */
196 io.outfd = open_download_temp(du->path, tmp, sizeof(tmp), st);
197 if (io.outfd < 0) {
198 return -1;
199 }
200 rc = web_dl_fetch(su, co, hdrs, &io, st);
201 close(io.outfd);
202 rc = atomic_dest_finish(tmp, du->path, rc, st);
203 if (rc != 0) {
204 return rc;
205 }
206 if (o && !o->silent) {
207 fprintf(stderr, "xrdcp: downloaded %lld bytes (HTTP %d)\n",
208 io.blen, io.status);
209 }
210 return 0;
211}
