31/* Load the CVMFS repo master public key PEM named by `path` into the cache
32 * policy (phase-85 F1). Config-time, cf->pool-owned (cycle lifetime — workers
33 * inherit the pointer through the registered policy copy). The file may hold
34 * several concatenated PEM keys (CVMFS key rotation); content is validated by
35 * the OpenSSL PEM parser at verify time, here only shape-checked. [emerg] on
36 * any failure — a verifying proxy with an unloadable trust anchor must not
37 * start. */
38static ngx_int_t
39brix_tier_load_master_key(ngx_conf_t *cf, const ngx_str_t *path,
40 brix_cache_policy_t *pol)
41{
42 char pathz[1024];
43 u_char *buf;
44 ssize_t n;
45 off_t size;
46 ngx_fd_t fd;
47 ngx_file_info_t fi;
48
49 if (path->len >= sizeof(pathz)) {
50 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
51 "brix_cvmfs_verify_manifest: key path too long");
52 return NGX_ERROR;
53 }
54 ngx_cpystrn((u_char *) pathz, path->data, path->len + 1);
55
56 fd = open(pathz, O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: DOMAIN_CONFIG — config-domain trust-anchor PEM (not export storage) */
57 if (fd == -1) {
58 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
59 "brix_cvmfs_verify_manifest: cannot open \"%s\"", pathz);
60 return NGX_ERROR;
61 }
62 if (ngx_fd_info(fd, &fi) == -1
63 || (size = ngx_file_size(&fi)) <= 0 || size > 65536)
64 {
65 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
66 "brix_cvmfs_verify_manifest: \"%s\" is empty, unreadable "
67 "or larger than 64KB", pathz);
68 (void) close(fd);
69 return NGX_ERROR;
70 }
71 buf = ngx_palloc(cf->pool, (size_t) size + 1);
72 if (buf == NULL) {
73 (void) close(fd);
74 return NGX_ERROR;
75 }
76 n = read(fd, buf, (size_t) size); /* vfs-seam-allow: DOMAIN_CONFIG — config-domain trust-anchor PEM (not export storage) */
77 (void) close(fd);
78 if (n != (ssize_t) size) {
79 ngx_conf_log_error(NGX_LOG_EMERG, cf, ngx_errno,
80 "brix_cvmfs_verify_manifest: short read on \"%s\"", pathz);
81 return NGX_ERROR;
82 }
83 buf[size] = '\0';
84 if (ngx_strstr(buf, "BEGIN PUBLIC KEY") == NULL) {
85 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
86 "brix_cvmfs_verify_manifest: \"%s\" holds no PEM public key",
87 pathz);
88 return NGX_ERROR;
89 }
90 pol->cvmfs_master_pub = buf;
91 pol->cvmfs_master_pub_len = (size_t) size;
92 return NGX_OK;
93}

95/* Parse + validate the brix_cache_peers ring (phase-85 F8) and record it on
96 * the backend registry. Each member token is "host:port"; this node's own slot
97 * is "self=host:port" (the mesh needs every node to carry the IDENTICAL list so
98 * rendezvous ownership agrees, so self is marked, never omitted). Operator
99 * errors — malformed authority, no/duplicate self, fewer than 2 members, more
100 * than 16 — are [emerg], failing nginx -t. */
101static ngx_int_t
102brix_tier_register_cache_peers(ngx_conf_t *cf,
103 ngx_http_brix_shared_conf_t *common)
104{
105 char hosts[16][256];
106 int ports[16];
107 int self = -1;
108 ngx_str_t *tok = common->cache_peers->elts;
109 ngx_uint_t i;
110
111 if (common->cache_peers->nelts < 2) {
112 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
113 "brix_cache_peers: a mesh needs at least 2 ring members "
114 "(self=host:port plus one sibling)");
115 return NGX_ERROR;
116 }
117 if (common->cache_peers->nelts > 16) {
118 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
119 "brix_cache_peers: at most 16 ring members supported");
120 return NGX_ERROR;
121 }
122
123 for (i = 0; i < common->cache_peers->nelts; i++) {
124 char buf[300];
125 char *auth = buf;
126 char *colon;
127 long port;
128
129 if (tok[i].len >= sizeof(buf)) {
130 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
131 "brix_cache_peers: member \"%V\" is too long", &tok[i]);
132 return NGX_ERROR;
133 }
134 ngx_memcpy(buf, tok[i].data, tok[i].len);
135 buf[tok[i].len] = '\0';
136
137 if (ngx_strncmp(auth, "self=", sizeof("self=") - 1) == 0) {
138 if (self >= 0) {
139 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
140 "brix_cache_peers: more than one self= member");
141 return NGX_ERROR;
142 }
143 self = (int) i;
144 auth += sizeof("self=") - 1;
145 }
146 colon = strrchr(auth, ':');
147 if (colon == NULL || colon == auth || colon[1] == '\0') {
148 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
149 "brix_cache_peers: member \"%V\" is not host:port", &tok[i]);
150 return NGX_ERROR;
151 }
152 port = strtol(colon + 1, NULL, 10);
153 if (port < 1 || port > 65535) {
154 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
155 "brix_cache_peers: member \"%V\" has an invalid port",
156 &tok[i]);
157 return NGX_ERROR;
158 }
159 *colon = '\0';
160 if (ngx_strlen(auth) >= sizeof(hosts[0])) {
161 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
162 "brix_cache_peers: member \"%V\" host is too long", &tok[i]);
163 return NGX_ERROR;
164 }
165 ngx_cpystrn((u_char *) hosts[i], (u_char *) auth, sizeof(hosts[i]));
166 ports[i] = (int) port;
167 }
168
169 if (self < 0) {
170 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
171 "brix_cache_peers: mark this node's own ring slot with "
172 "self=host:port");
173 return NGX_ERROR;
174 }
175
176 brix_vfs_backend_config_cache_peers(common->root_canon,
177 (const char (*)[256]) hosts, ports,
178 (int) common->cache_peers->nelts, self);
179 return NGX_OK;
180}

182/* Fill the read-through cache policy from the merged srv-conf knobs. Split out
183 * of brix_tier_register_cache_store so each function's branching stays within
184 * the readability gate — pure knob-to-policy plumbing, no validation here. */
185static void
186brix_tier_fill_cache_policy(ngx_http_brix_shared_conf_t *common,
187 brix_cache_policy_t *polp)
188{
189 brix_cache_policy_t pol;
190
191 ngx_memzero(&pol, sizeof(pol));
192 pol.enabled = 1;
193 pol.max_file_size = common->cache_max_object;
194 /* Read-fill admission (bridged from the srv conf at finalisation): deny/allow
195 * prefixes + include regex gate the composable sd_cache fill for parity with
196 * write-through and the legacy cache_origin admit. NULL => no filter. */
197 pol.deny_prefixes = common->cache_deny_prefixes;
198 pol.allow_prefixes = common->cache_allow_prefixes;
199 pol.include_regex = common->cache_include_re;
200 /* Documented defaults (90/80) applied here: the merge keeps the pair UNSET
201 * so the stream reaper merge can detect an explicit setting. */
202 pol.evict_at = (common->cache_evict_at == NGX_CONF_UNSET_UINT)
203 ? 90 : common->cache_evict_at;
204 pol.evict_to = (common->cache_evict_to == NGX_CONF_UNSET_UINT)
205 ? 80 : common->cache_evict_to;
206 pol.meta_mode = (int) common->cache_meta_mode;
207 pol.batch_cinfo = (common->cache_batch_cinfo == 2)
208 ? -1 : (int) common->cache_batch_cinfo;
209 pol.l1_entries = common->cache_index_cache;
210 pol.slice_size = common->cache_slice_size;
211 /* Background block prefetch (audit §4.1): merged defaults are 0 (off) and
212 * 8 MiB, but guard UNSET for a conf that reaches registration unmerged. */
213 pol.prefetch_jobs = (common->cache_prefetch == NGX_CONF_UNSET
214 || common->cache_prefetch <= 0)
215 ? 0 : (ngx_uint_t) common->cache_prefetch;
216 pol.prefetch_window = (common->cache_prefetch_window == NGX_CONF_UNSET_SIZE)
217 ? 8 * 1024 * 1024 : common->cache_prefetch_window;
218 /* Cache-only serving (audit §4.4): a miss must NOT reach the origin. */
219 pol.only_if_cached = (common->cache_only_if_cached == 1);
220 /* phase-68 digest-verification mode — the posix-store constraint it
221 * carries is validated by the caller against the parsed store driver. */
222 pol.verify = (common->cache_verify_mode == NGX_CONF_UNSET_UINT)
223 ? BRIX_CACHE_VERIFY_OFF
224 : (brix_cache_verify_mode_e) common->cache_verify_mode;
225 pol.cvmfs_manifest_ttl = common->cache_manifest_ttl;
226 pol.cvmfs_offline_ttl = common->cache_offline_ttl;
227 /* §4.3 uvkeep: unset/negative → 0 (off). */
228 pol.uvkeep = (common->cache_uvkeep == NGX_CONF_UNSET
229 || common->cache_uvkeep < 0)
230 ? 0 : common->cache_uvkeep;
231 pol.global_cas = (common->cache_global_cas == 1);
232 pol.passthrough = (common->cache_passthrough == 1);
233 pol.passthrough_max = common->cache_passthrough_max;
234 if (common->cache_quarantine_dir.len > 0) {
235 ngx_cpystrn((u_char *) pol.quarantine_dir,
236 common->cache_quarantine_dir.data,
237 ngx_min(common->cache_quarantine_dir.len + 1,
238 sizeof(pol.quarantine_dir)));
239 }
240 *polp = pol;
241}

243/* Verify-chain constraints on the parsed cache store: the self-addressing
244 * digest verifications (cvmfs-cas, oci-digest, rpm-repodata) and the phase-85
245 * F1 manifest-signature verify all read the staged part's physical path BEFORE
246 * commit, so the store's driver must expose staged_path (posix .part files, or
247 * pblock single-block staged blobs — phase-88 W1); reject other stores loudly.
248 * Split out of brix_tier_register_cache_store for the readability gate. */
249static ngx_int_t
250brix_tier_check_cache_verify(ngx_conf_t *cf,
251 ngx_http_brix_shared_conf_t *common, const brix_tier_cfg_t *cfg,
252 brix_cache_policy_t *pol, const brix_sd_driver_t *sdrv)
253{
254 if (brix_cache_verify_is_selfaddr(pol->verify)
255 && (sdrv == NULL || sdrv->staged_path == NULL))
256 {
257 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
258 "brix_cache_verify %s requires a cache store whose driver exposes "
259 "staged fill paths (posix or pblock; got \"%s\")",
260 brix_cache_verify_mode_str(pol->verify), cfg->driver);
261 return NGX_ERROR;
262 }
263 /* A pblock store only yields a verifiable staged path while a fill fits
264 * one block: warn when the stripe is below the CVMFS publisher's object
265 * ceiling, so an oversized (multi-block) object cannot silently turn into
266 * a fail-closed fill at runtime. Applies to every self-addressing verify
267 * (cvmfs-cas, oci-digest, rpm-repodata) — they share the staged-path read. */
268 if (brix_cache_verify_is_selfaddr(pol->verify)
269 && ngx_strcmp(cfg->driver, "pblock") == 0)
270 {
271 int64_t ebs = cfg->block_size > 0
272 ? (int64_t) cfg->block_size : PBLOCK_DEFAULT_BLOCK_SIZE;
273
274 if (ebs < (int64_t) CVMFS_OBJECT_MAX_BYTES) {
275 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
276 "brix_cache_verify %s on a pblock store: block_size "
277 "(%L) is below the CVMFS object ceiling (%uz) — an object "
278 "larger than one block cannot be verified and its fill will "
279 "fail closed; set \"block_size=256M\" on brix_cache_store",
280 brix_cache_verify_mode_str(pol->verify),
281 ebs, (size_t) CVMFS_OBJECT_MAX_BYTES);
282 }
283 }
284 /* phase-85 F1: brix_cvmfs_verify_manifest — load the repo master public
285 * key(s) once at config time; the fill spine verifies every MANIFEST-class
286 * fill's signature chain against it before publish. Same staged-path
287 * constraint as cvmfs-cas (the verify reads the staged part path). */
288 if (common->cache_cvmfs_master_key.len > 0) {
289 if (sdrv == NULL || sdrv->staged_path == NULL) {
290 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
291 "brix_cvmfs_verify_manifest requires a cache store whose "
292 "driver exposes staged fill paths (posix or pblock; got "
293 "\"%s\")", cfg->driver);
294 return NGX_ERROR;
295 }
296 if (brix_tier_load_master_key(cf, &common->cache_cvmfs_master_key,
297 pol) != NGX_OK)
298 {
299 return NGX_ERROR; /* [emerg] already logged */
300 }
301 }
302 return NGX_OK;
303}

305/* phase-87 G13 (phase-88 W1) constraints for brix_cache_global_cas: cross-repo
306 * dedup of verified CAS objects is a driver verb — posix collapses names onto
307 * one inode via hardlinks, pblock folds byte-identical blobs via F10 refs (and
308 * must have that gate ARMED, or global_cas would silently ENOTSUP every fill).
309 * Split out of brix_tier_register_cache_store for the readability gate. */
310static ngx_int_t
311brix_tier_check_cache_cas(ngx_conf_t *cf, const brix_tier_cfg_t *cfg,
312 const brix_cache_policy_t *pol, const brix_sd_driver_t *sdrv)
313{
314 if (pol->global_cas && (sdrv == NULL || sdrv->dedup_publish == NULL)) {
315 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
316 "brix_cache_global_cas requires a cache store whose driver "
317 "supports commit-time dedup (posix or pblock; got \"%s\")",
318 cfg->driver);
319 return NGX_ERROR;
320 }
321#if BRIX_HAVE_SQLITE
322 if (pol->global_cas && ngx_strcmp(cfg->driver, "pblock") == 0) {
323 pblock_opts_t popts;
324
325 if (pblock_opts_load_sidecar(cfg->path, &popts) != 0 || !popts.dedup) {
326 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
327 "brix_cache_global_cas on a pblock cache store requires its "
328 "refs gate: add \"?dedup=1\" to the store URL "
329 "(brix_cache_store pblock:%s?dedup=1)", cfg->path);
330 return NGX_ERROR;
331 }
332 }
333#endif
334 return NGX_OK;
335}

337/* Parse the cache_store URL and record its tier cfg + read-through policy on the
338 * backend registry. Split out of brix_tier_register_stores so each function's
339 * branching stays within the readability gate. Operator errors are [emerg]. */
340ngx_int_t
341brix_tier_register_cache_store(ngx_conf_t *cf,
342 ngx_http_brix_shared_conf_t *common)
343{
344 char err[256];
345 brix_tier_cfg_t cfg;
346 brix_cache_policy_t pol;
347 const brix_sd_driver_t *sdrv;
348 brix_tier_parse_t tp = { cf, &cfg, err, sizeof(err) };
349
350 if (brix_tier_parse_store(&tp, &common->cache_store,
351 common->cache_store_args, BRIX_TIER_CACHE) != NGX_OK)
352 {
353 return NGX_ERROR; /* [emerg] already logged */
354 }
355 brix_tier_fill_cache_policy(common, &pol);
356 sdrv = brix_sd_driver_find(cfg.driver);
357
358 /* phase-88 W1: persist a pblock cache store's `?tail` static opts (captured
359 * by the tier parse) as its <store>/pblock.opts sidecar, exactly like the
360 * primary-backend pblock:// flow — the driver reads it at instance init. */
361 if (ngx_strcmp(cfg.driver, "pblock") == 0 && cfg.opts[0] != '\0') {
362 brix_pblock_write_opts_sidecar(cfg.path, cfg.opts);
363 }
364
365 if (brix_tier_check_cache_verify(cf, common, &cfg, &pol, sdrv) != NGX_OK
366 || brix_tier_check_cache_cas(cf, &cfg, &pol, sdrv) != NGX_OK)
367 {
368 return NGX_ERROR; /* [emerg] already logged */
369 }
370 brix_vfs_backend_config_cache_store(common->root_canon, &cfg, &pol);
371
372 /* Phase-85 F7: the optional cold tier under the cache — its own store URL,
373 * governed by the hot cache's policy (no separate knobs). */
374 if (common->cache_cold_store.len > 0) {
375 brix_tier_cfg_t ccfg;
376 brix_tier_parse_t ctp = { cf, &ccfg, err, sizeof(err) };
377
378 if (brix_tier_parse_store(&ctp, &common->cache_cold_store,
379 common->cache_cold_store_args, BRIX_TIER_CACHE) != NGX_OK)
380 {
381 return NGX_ERROR; /* [emerg] already logged */
382 }
383 brix_vfs_backend_config_cache_cold_store(common->root_canon, &ccfg);
384 }
385
386 /* Phase-85 F8: the sibling-mesh ring under the cache tier. */
387 if (common->cache_peers != NULL
388 && brix_tier_register_cache_peers(cf, common) != NGX_OK)
389 {
390 return NGX_ERROR; /* [emerg] already logged */
391 }
392 return NGX_OK;
393}
