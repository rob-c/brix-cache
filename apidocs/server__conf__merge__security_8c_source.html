2 * server_conf_merge_security.c — server-block merge for the identity/crypto
3 * configuration area (storage-plane merge split to server_conf_merge_storage.c).
4 *
5 * WHAT: Owns brix_merge_srv_security() (auth scheme + GSI/pwd, XrdAcc engine,
6 * X.509/CRL, tokens + L1/L2 caches, sss/krb5/unix/host, TLS toggles) and
7 * brix_merge_srv_storage() (compression, ZIP, the read-through cache
8 * origin/sizing/eviction/verify, memory budget, readv sizing, io_uring),
9 * together with the file-local per-concern helpers each delegates to.
10 * WHY: Split (phase-79 file-size cap) out of the former 1249-line
11 * server_conf.c; the two entry points are non-static (declared in
12 * server_conf_internal.h) for server_conf.c's linear orchestrator, every
13 * sub-helper file-local.
14 * HOW: Standard ngx_conf_merge_* / BRIX_MERGE_* inheritance, one helper per
15 * concern group, invoked in the original order so cross-group derivations
16 * (staging LOW from HIGH, reaper watermarks from the eviction threshold)
17 * observe already-merged inputs. No behaviour change from the split.
18 */

31/*
32 * WHAT: merge the GSI/pwd + XrdAcc engine group and validate the native-authdb
33 * auth-scheme requirement.
34 * WHY: the native-authdb rule couples the merged `auth` scheme and `acc.format`
35 * so it must run after both settle; grouping keeps that dependency local.
36 * HOW: inherit the GSI/pwd/acc scalars child<-parent, then reject a native
37 * authdb without an authenticating scheme (xrdacc is exempt: it authorizes
38 * anonymous `u *` rules).
39 */
40static char *
41brix_merge_srv_gsi_acc(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
42 ngx_stream_brix_srv_conf_t *prev)
43{
44 ngx_conf_merge_uint_value(conf->auth, prev->auth, BRIX_AUTH_NONE);
45 ngx_conf_merge_value(conf->auth_maxfail, prev->auth_maxfail, 0); /* §5.7 */
46
47 /* protbind rules are inherited whole: a server block that states none of
48 * its own keeps the outer block's host policy, but the moment it states one
49 * it owns the ordering, and silently prepending the parent's rules would
50 * change which template matches first. */
51 if (conf->protbind == NULL) {
52 conf->protbind = prev->protbind;
53 }
54 ngx_conf_merge_uint_value(conf->gsi_signed_dh, prev->gsi_signed_dh,
55 BRIX_GSI_SDH_OFF);
56 ngx_conf_merge_value(conf->gsi_max_inflight, prev->gsi_max_inflight, 256);
57 /* §5.10: GSI client-cert chain depth cap; 0 = unlimited (default). */
58 ngx_conf_merge_value(conf->gsi_verify_depth, prev->gsi_verify_depth, 0);
59 ngx_conf_merge_uint_value(conf->gsi_keypool_size, prev->gsi_keypool_size,
60 BRIX_GSI_KEYPOOL_SIZE_DEFAULT);
61 ngx_conf_merge_uint_value(conf->gsi_keypool_seed, prev->gsi_keypool_seed,
62 BRIX_GSI_KEYPOOL_SEED_DEFAULT);
63 ngx_conf_merge_str_value(conf->gsi_ciphers, prev->gsi_ciphers, "");
64 ngx_conf_merge_str_value(conf->pwd_file, prev->pwd_file, "");
65
66 /*
67 * The native authdb engine matches by DN/VO, so it needs an authenticating
68 * scheme — but ANY of them will do: sss/krb5/pwd/host/unix all stamp
69 * ctx->login.dn (and pwd/sss also fill the VO list) exactly as gsi and
70 * token do, so u/g rules bind behind them too. Only anonymous servers are
71 * rejected; the xrdacc engine is exempt even there, because it also
72 * authorizes anonymous `u *` rules. Validated here, where both directives
73 * have settled.
74 */
75 if (conf->common.acc.authdb.len > 0
76 && conf->common.acc.format == BRIX_AUTHDB_FORMAT_NATIVE
77 && conf->auth == BRIX_AUTH_NONE)
78 {
79 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
80 "brix_authdb (native format) requires an authenticating brix_auth "
81 "scheme; use `brix_authdb_engine xrdacc` for anonymous rules");
82 return NGX_CONF_ERROR;
83 }
84
85 /* 2.0 F20: a native authdb line the grammar could not parse is refused
86 * here, not at the directive — `format` only settles now, and an
87 * XrdAcc-format file is legitimately unparseable by the native grammar. */
88 return brix_authdb_defect_refuse(cf, conf->common.acc.format,
89 &conf->common.acc.authdb_defect);
90}

92/*
93 * WHAT: merge the FRM prepare command + the X.509 material (cert/key/CA, VOMS)
94 * and CRL/signing-policy toggles, plus the access/session logging fields.
95 * WHY: the fallible FRM merge (2.0 F1: it publishes the process-wide engine
96 * values and refuses a disagreement) stays local to one helper.
97 * HOW: merge prepare_command, delegate to brix_frm_conf_merge(), then inherit
98 * the X.509/CRL/log scalars child<-parent.
99 */
100static char *
101brix_merge_srv_x509(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
102 ngx_stream_brix_srv_conf_t *prev)
103{
104 ngx_conf_merge_str_value(conf->prepare_command, prev->prepare_command, "");
105 if (brix_frm_conf_merge(cf, &conf->frm, &prev->frm) != NGX_CONF_OK) {
106 return NGX_CONF_ERROR;
107 }
108 /* §5.10: root:// TLS cipher list; empty = OpenSSL defaults (default). */
109 ngx_conf_merge_str_value(conf->tls_ciphers, prev->tls_ciphers, "");
110 /* §5.10: root:// TLSv1.3 cipher-suite list; empty = OpenSSL defaults. */
111 ngx_conf_merge_str_value(conf->tls_ciphersuites, prev->tls_ciphersuites, "");
112 ngx_conf_merge_str_value(conf->crl, prev->crl, "");
113 ngx_conf_merge_value(conf->crl_reload, prev->crl_reload, 0);
114 ngx_conf_merge_uint_value(conf->signing_policy_mode,
115 prev->signing_policy_mode, BRIX_SP_MODE_ON);
116 ngx_conf_merge_uint_value(conf->crl_mode, prev->crl_mode, BRIX_CRL_MODE_TRY);
117 /* 2.0 F19: same defaults as the http plane — widest CRL reach, silent log. */
118 ngx_conf_merge_uint_value(conf->crl_scope, prev->crl_scope,
119 BRIX_CRL_SCOPE_ALL);
120 ngx_conf_merge_uint_value(conf->tls_verify_log, prev->tls_verify_log,
121 BRIX_TLS_VERIFY_LOG_OFF);
122 ngx_conf_merge_str_value(conf->access_log, prev->access_log, "");
123 ngx_conf_merge_value(conf->session_log, prev->session_log, 1);
124
125 return NGX_CONF_OK;
126}

128/*
129 * WHAT: merge the token group (JWKS, issuer/audience, config/registry, macaroon
130 * secrets) and the throttle group (limits + named rate-limit zone), and
131 * validate the clock-skew bound and throttle-zone reference.
132 * WHY: both carry config-time validation (clock-skew range, zone existence);
133 * keeping them together isolates the two failure paths.
134 * HOW: inherit the token/throttle scalars, clamp-check clock skew to [0,300],
135 * then resolve the named rate-limit zone, failing if it was not declared.
136 */
137static char *
138brix_merge_srv_tokens(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
139 ngx_stream_brix_srv_conf_t *prev)
140{
141 ngx_conf_merge_ptr_value(conf->token_registry, prev->token_registry, NULL);
142 ngx_conf_merge_str_value(conf->throttle.zone_name,
143 prev->throttle.zone_name, "");
144 ngx_conf_merge_ptr_value(conf->throttle.zone, prev->throttle.zone, NULL);
145 ngx_conf_merge_uint_value(conf->throttle.max_open_files,
146 prev->throttle.max_open_files, 0);
147 ngx_conf_merge_str_value(conf->throttle.bwm_zone_name,
148 prev->throttle.bwm_zone_name, "");
149 ngx_conf_merge_size_value(conf->throttle.bwm_budget,
150 prev->throttle.bwm_budget, 0);
151
152 /* phase-59 W3a: resolve the named rate-limit zone the throttle keys its
153 * per-user counters into (declared via brix_rate_limit_zone). */
154 if (conf->throttle.zone == NULL && conf->throttle.zone_name.len > 0) {
155 conf->throttle.zone = brix_rl_zone_get(&conf->throttle.zone_name);
156 if (conf->throttle.zone == NULL) {
157 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,

160 return NGX_CONF_ERROR;
161 }
162 }

164 /* F3/P90-28.1: the macaroon root-secret hex lives in conf memory for the
165 * process lifetime — keep its pages out of core dumps and off swap.
166 * Best-effort, never fatal (per-request binary copies are stack + F1-
167 * cleansed; this guards the only long-lived form of the key). */
168 if (conf->common.token_macaroon_secret.len > 0
169 && brix_secret_page_guard(conf->common.token_macaroon_secret.data,
170 conf->common.token_macaroon_secret.len) != 0)
171 {
172 ngx_conf_log_error(NGX_LOG_WARN, cf, ngx_errno,
173 "brix: could not fully page-guard the macaroon secret "
174 "(madvise/mlock); continuing unguarded");
175 }
176 if (conf->common.token_macaroon_secret_old.len > 0
177 && brix_secret_page_guard(conf->common.token_macaroon_secret_old.data,
178 conf->common.token_macaroon_secret_old.len) != 0)
179 {
180 ngx_conf_log_error(NGX_LOG_WARN, cf, ngx_errno,
181 "brix: could not fully page-guard the old macaroon secret "
182 "(madvise/mlock); continuing unguarded");
183 }
184 return NGX_CONF_OK;
185}

187/*
188 * WHAT: merge the CSI record cache, the Phase-20 L1/L2 caches (token/auth/rate),
189 * the sss/krb5/unix/host schemes, the security level, and the TLS toggles.
190 * WHY: a plain inheritance tail with no validation; a void helper trims the
191 * orchestrator without splitting a decision.
192 * HOW: whole-config inherit for the kv-backed caches (NULL == disabled), then
193 * child<-parent for the remaining scheme/TLS scalars.
194 */
195static void
196brix_merge_srv_authtail(ngx_stream_brix_srv_conf_t *conf,
197 ngx_stream_brix_srv_conf_t *prev)
198{
199 brix_csi_conf_merge(&conf->csi, &prev->csi);
200
201 /* Phase 20 caches/limits: inherit the parent's whole config when this
202 * block did not declare its own (kv still NULL). */
203 if (conf->token_cache_kv == NULL) {
204 conf->token_cache_kv = prev->token_cache_kv;
205 }
206 if (conf->auth_cache.kv == NULL) {
207 conf->auth_cache = prev->auth_cache;
208 }
209 if (conf->rate_limit.kv == NULL) {
210 conf->rate_limit = prev->rate_limit;
211 }
212 ngx_conf_merge_str_value(conf->sss_keytab, prev->sss_keytab, "");
213 ngx_conf_merge_value(conf->sss_lifetime, prev->sss_lifetime, 13);
214 ngx_conf_merge_value(conf->sss_getcreds, prev->sss_getcreds, 0);
215 ngx_conf_merge_str_value(conf->krb5.principal, prev->krb5.principal, "");
216 ngx_conf_merge_str_value(conf->krb5.keytab, prev->krb5.keytab, "");
217 ngx_conf_merge_value(conf->krb5.ip_check, prev->krb5.ip_check, 0);
218 ngx_conf_merge_value(conf->krb5.delegate, prev->krb5.delegate, 0);
219 ngx_conf_merge_value(conf->unix_trust_remote, prev->unix_trust_remote, 0);
220 ngx_conf_merge_ptr_value(conf->host_allow, prev->host_allow, NULL);
221 if (conf->oss_spaces == NULL) { /* phase-115 W3.3: inherit the table */
222 conf->oss_spaces = prev->oss_spaces;
223 }
224 ngx_conf_merge_uint_value(conf->security_level, prev->security_level, 0);
225 /* Off by default: fail-closed signing refuses every client whose auth
226 * protocol cannot sign (all stock non-GSI clients), so it is opt-in. */
227 ngx_conf_merge_value(conf->signing_required, prev->signing_required, 0);
228 ngx_conf_merge_uint_value(conf->min_sec_level, prev->min_sec_level, 0);
229 ngx_conf_merge_value(conf->ztn_cleartext, prev->ztn_cleartext, 0);
230 /* ztn -maxsz analog; 0 = no extra cap (compatibility default). */
231 ngx_conf_merge_size_value(conf->ztn_maxsz, prev->ztn_maxsz, 0);
232 ngx_conf_merge_value(conf->opaque_strict, prev->opaque_strict, 0);
233 ngx_conf_merge_value(conf->tls, prev->tls, 0);
234 /* kTLS default OFF (phase-33 P5): software kTLS regresses vs OpenSSL's
235 * userspace AES-GCM on AES-NI hosts and is broken on some kernels (WSL2), so
236 * it is opt-in and documented HW-offload-only (`brix_ktls on`).
237 * SSL_OP_ENABLE_KTLS is a transparent no-op when the cipher/kernel cannot
238 * offload, so this default is byte-exact vs userspace TLS. */
239 ngx_conf_merge_value(conf->tls_ktls, prev->tls_ktls, 0);
240 /* §5.10: TLS session resumption on by default (unchanged); off full-handshakes. */
241 ngx_conf_merge_value(conf->tls_reuse, prev->tls_reuse, 1);
242}

244/* Identity & crypto: auth scheme + GSI/pwd, XrdAcc engine (+ native-authdb
245 * validation), SciTags/FRM, X.509 material + CRL, access log, tokens + L1/L2
246 * caches, sss/krb5/unix/host, security level, and TLS toggles. */
247char *
248brix_merge_srv_security(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
249 ngx_stream_brix_srv_conf_t *prev)
250{
251 /*
252 * Standard nginx inheritance rules: values set on the current server
253 * override the parent, otherwise we fall back to the parent or the hard
254 * coded module default. Each concern group is delegated to a helper below,
255 * invoked in the original linear order so cross-group derivations still see
256 * their already-merged inputs.
257 */
258 /* Shared common.* preamble (root defaults to "/": a pure cache node may
259 * omit brix_root and serve the whole namespace). Also covers the tier
260 * grammar + pmark + hard read-only enforcement — do not re-merge those. */
261 if (ngx_http_brix_shared_merge(cf, &prev->common, &conf->common, "/")
262 != NGX_CONF_OK)
263 {
264 return NGX_CONF_ERROR;
265 }
266 if (brix_merge_srv_gsi_acc(cf, conf, prev) != NGX_CONF_OK) {
267 return NGX_CONF_ERROR;
268 }
269 if (brix_merge_srv_x509(cf, conf, prev) != NGX_CONF_OK) {
270 return NGX_CONF_ERROR;
271 }
272 if (brix_merge_srv_tokens(cf, conf, prev) != NGX_CONF_OK) {
273 return NGX_CONF_ERROR;
274 }
275 brix_merge_srv_authtail(conf, prev);
276 /* 2.0 F4: the purge rules name brix_oss_space groups; the table is
277 * inherited just above, so only now can a rule be checked against it. */
278 if (brix_frm_purge_policy_check(cf, conf->oss_spaces, &conf->frm)
279 != NGX_CONF_OK)
280 {
281 return NGX_CONF_ERROR;
282 }
283
284 /* phase-116: `h` rules, a protbind host template or `brix_auth host` all
285 * decide on the peer's PTR name — guarantee a reverse backend exists even
286 * when the configuration declares no brix_resolver. */
287 if (conf->auth == BRIX_AUTH_HOST
288 || conf->common.acc.resolve_hosts > 0
289 || brix_protbind_needs_hostname(conf->protbind))
290 {
291 return brix_dns_backend_prepare(cf);
292 }
293
294 return NGX_CONF_OK;
295}
