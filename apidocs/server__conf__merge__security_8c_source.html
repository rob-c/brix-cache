3 * configuration area (storage-plane merge split to server_conf_merge_storage.c).
4 *
5 * WHAT: Owns brix_merge_srv_security() (auth scheme + GSI/pwd, XrdAcc engine,
6 * X.509/CRL, tokens + L1/L2 caches, sss/krb5/unix/host, TLS toggles) and

9 * together with the file-local per-concern helpers each delegates to.
10 * WHY: Split (phase-79 file-size cap) out of the former 1249-line
11 * server_conf.c; the two entry points are non-static (declared in
12 * server_conf_internal.h) for server_conf.c's linear orchestrator, every
13 * sub-helper file-local.
14 * HOW: Standard ngx_conf_merge_* / BRIX_MERGE_* inheritance, one helper per
15 * concern group, invoked in the original order so cross-group derivations
16 * (staging LOW from HIGH, reaper watermarks from the eviction threshold)

29/*
30 * WHAT: merge the GSI/pwd + XrdAcc engine group and validate the native-authdb
31 * auth-scheme requirement.
32 * WHY: the native-authdb rule couples the merged `auth` scheme and `acc.format`
33 * so it must run after both settle; grouping keeps that dependency local.
34 * HOW: inherit the GSI/pwd/acc scalars child<-parent, then reject a native
35 * authdb without an authenticating scheme (xrdacc is exempt: it authorizes
36 * anonymous `u *` rules).
37 */
38static char *
39brix_merge_srv_gsi_acc(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
40 ngx_stream_brix_srv_conf_t *prev)
41{
42 ngx_conf_merge_uint_value(conf->auth, prev->auth, BRIX_AUTH_NONE);
43 ngx_conf_merge_value(conf->auth_maxfail, prev->auth_maxfail, 0); /* §5.7 */
44
45 /* protbind rules are inherited whole: a server block that states none of
46 * its own keeps the outer block's host policy, but the moment it states one
47 * it owns the ordering, and silently prepending the parent's rules would
48 * change which template matches first. */
49 if (conf->protbind == NULL) {
50 conf->protbind = prev->protbind;
51 }
52 ngx_conf_merge_uint_value(conf->gsi_signed_dh, prev->gsi_signed_dh,
53 BRIX_GSI_SDH_OFF);
54 ngx_conf_merge_value(conf->gsi_max_inflight, prev->gsi_max_inflight, 256);
55 /* §5.10: GSI client-cert chain depth cap; 0 = unlimited (default). */
56 ngx_conf_merge_value(conf->gsi_verify_depth, prev->gsi_verify_depth, 0);
57 ngx_conf_merge_uint_value(conf->gsi_keypool_size, prev->gsi_keypool_size,
58 BRIX_GSI_KEYPOOL_SIZE_DEFAULT);
59 ngx_conf_merge_uint_value(conf->gsi_keypool_seed, prev->gsi_keypool_seed,
60 BRIX_GSI_KEYPOOL_SEED_DEFAULT);
61 ngx_conf_merge_str_value(conf->gsi_ciphers, prev->gsi_ciphers, "");
62 ngx_conf_merge_str_value(conf->pwd_file, prev->pwd_file, "");
63
64 /*
65 * The native authdb engine matches by DN/VO, so it needs an authenticating
66 * scheme — but ANY of them will do: sss/krb5/pwd/host/unix all stamp
67 * ctx->login.dn (and pwd/sss also fill the VO list) exactly as gsi and
68 * token do, so u/g rules bind behind them too. Only anonymous servers are
69 * rejected; the xrdacc engine is exempt even there, because it also
70 * authorizes anonymous `u *` rules. Validated here, where both directives
71 * have settled.
72 */
73 if (conf->common.acc.authdb.len > 0
74 && conf->common.acc.format == BRIX_AUTHDB_FORMAT_NATIVE
75 && conf->auth == BRIX_AUTH_NONE)
76 {
77 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
78 "brix_authdb (native format) requires an authenticating brix_auth "
79 "scheme; use `brix_authdb_engine xrdacc` for anonymous rules");
80 return NGX_CONF_ERROR;
81 }
82
83 return NGX_CONF_OK;
84}

86/*
87 * WHAT: merge the FRM prepare command + the X.509 material (cert/key/CA, VOMS)
88 * and CRL/signing-policy toggles, plus the access/session logging fields.
89 * WHY: brix_frm_conf_merge() depends on the merged prepare_command; grouping
90 * makes that ordering explicit and keeps the fallible FRM merge local.
91 * HOW: merge prepare_command, delegate to brix_frm_conf_merge(), then inherit
92 * the X.509/CRL/log scalars child<-parent.
93 */
94static char *
95brix_merge_srv_x509(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
96 ngx_stream_brix_srv_conf_t *prev)
97{
98 ngx_conf_merge_str_value(conf->prepare_command, prev->prepare_command, "");
99 if (brix_frm_conf_merge(cf, &conf->frm, &prev->frm, &conf->prepare_command)
100 != NGX_CONF_OK)
101 {
102 return NGX_CONF_ERROR;
103 }
104 /* §5.10: root:// TLS cipher list; empty = OpenSSL defaults (default). */
105 ngx_conf_merge_str_value(conf->tls_ciphers, prev->tls_ciphers, "");
106 /* §5.10: root:// TLSv1.3 cipher-suite list; empty = OpenSSL defaults. */
107 ngx_conf_merge_str_value(conf->tls_ciphersuites, prev->tls_ciphersuites, "");
108 ngx_conf_merge_str_value(conf->crl, prev->crl, "");
109 ngx_conf_merge_value(conf->crl_reload, prev->crl_reload, 0);
110 ngx_conf_merge_uint_value(conf->signing_policy_mode,
111 prev->signing_policy_mode, BRIX_SP_MODE_ON);
112 ngx_conf_merge_uint_value(conf->crl_mode, prev->crl_mode, BRIX_CRL_MODE_TRY);
113 ngx_conf_merge_str_value(conf->access_log, prev->access_log, "");
114 ngx_conf_merge_value(conf->session_log, prev->session_log, 1);
115
116 return NGX_CONF_OK;
117}

119/*
120 * WHAT: merge the token group (JWKS, issuer/audience, config/registry, macaroon
121 * secrets) and the throttle group (limits + named rate-limit zone), and
122 * validate the clock-skew bound and throttle-zone reference.
123 * WHY: both carry config-time validation (clock-skew range, zone existence);
124 * keeping them together isolates the two failure paths.
125 * HOW: inherit the token/throttle scalars, clamp-check clock skew to [0,300],
126 * then resolve the named rate-limit zone, failing if it was not declared.
127 */
128static char *
129brix_merge_srv_tokens(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
130 ngx_stream_brix_srv_conf_t *prev)
131{
132 ngx_conf_merge_ptr_value(conf->token_registry, prev->token_registry, NULL);
133 ngx_conf_merge_str_value(conf->throttle.zone_name,
134 prev->throttle.zone_name, "");
135 ngx_conf_merge_ptr_value(conf->throttle.zone, prev->throttle.zone, NULL);
136 ngx_conf_merge_uint_value(conf->throttle.max_open_files,
137 prev->throttle.max_open_files, 0);
138 ngx_conf_merge_str_value(conf->throttle.bwm_zone_name,
139 prev->throttle.bwm_zone_name, "");
140 ngx_conf_merge_size_value(conf->throttle.bwm_budget,
141 prev->throttle.bwm_budget, 0);
142
143 /* phase-59 W3a: resolve the named rate-limit zone the throttle keys its
144 * per-user counters into (declared via brix_rate_limit_zone). */
145 if (conf->throttle.zone == NULL && conf->throttle.zone_name.len > 0) {
146 conf->throttle.zone = brix_rl_zone_get(&conf->throttle.zone_name);
147 if (conf->throttle.zone == NULL) {
148 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
149 "brix_throttle_zone \"%V\" is not a declared "
150 "brix_rate_limit_zone", &conf->throttle.zone_name);
151 return NGX_CONF_ERROR;
152 }
153 }

155 /* F3/P90-28.1: the macaroon root-secret hex lives in conf memory for the
156 * process lifetime — keep its pages out of core dumps and off swap.
157 * Best-effort, never fatal (per-request binary copies are stack + F1-
158 * cleansed; this guards the only long-lived form of the key). */
159 if (conf->common.token_macaroon_secret.len > 0
160 && brix_secret_page_guard(conf->common.token_macaroon_secret.data,
161 conf->common.token_macaroon_secret.len) != 0)
162 {
163 ngx_conf_log_error(NGX_LOG_WARN, cf, ngx_errno,
164 "brix: could not fully page-guard the macaroon secret "
165 "(madvise/mlock); continuing unguarded");
166 }
167 if (conf->common.token_macaroon_secret_old.len > 0
168 && brix_secret_page_guard(conf->common.token_macaroon_secret_old.data,
169 conf->common.token_macaroon_secret_old.len) != 0)
170 {
171 ngx_conf_log_error(NGX_LOG_WARN, cf, ngx_errno,
172 "brix: could not fully page-guard the old macaroon secret "
173 "(madvise/mlock); continuing unguarded");
174 }
175 return NGX_CONF_OK;
176}

178/*
179 * WHAT: merge the CSI record cache, the Phase-20 L1/L2 caches (token/auth/rate),
180 * the sss/krb5/unix/host schemes, the security level, and the TLS toggles.
181 * WHY: a plain inheritance tail with no validation; a void helper trims the
182 * orchestrator without splitting a decision.
183 * HOW: whole-config inherit for the kv-backed caches (NULL == disabled), then
184 * child<-parent for the remaining scheme/TLS scalars.
185 */
186static void
187brix_merge_srv_authtail(ngx_stream_brix_srv_conf_t *conf,
188 ngx_stream_brix_srv_conf_t *prev)
189{
190 brix_csi_conf_merge(&conf->csi, &prev->csi);
191
192 /* Phase 20 caches/limits: inherit the parent's whole config when this
193 * block did not declare its own (kv still NULL). */
194 if (conf->token_cache_kv == NULL) {
195 conf->token_cache_kv = prev->token_cache_kv;
196 }
197 if (conf->auth_cache.kv == NULL) {
198 conf->auth_cache = prev->auth_cache;
199 }
200 if (conf->rate_limit.kv == NULL) {
201 conf->rate_limit = prev->rate_limit;
202 }
203 ngx_conf_merge_str_value(conf->sss_keytab, prev->sss_keytab, "");
204 ngx_conf_merge_value(conf->sss_lifetime, prev->sss_lifetime, 13);
205 ngx_conf_merge_str_value(conf->krb5.principal, prev->krb5.principal, "");
206 ngx_conf_merge_str_value(conf->krb5.keytab, prev->krb5.keytab, "");
207 ngx_conf_merge_value(conf->krb5.ip_check, prev->krb5.ip_check, 0);
208 ngx_conf_merge_value(conf->krb5.delegate, prev->krb5.delegate, 0);
209 ngx_conf_merge_value(conf->unix_trust_remote, prev->unix_trust_remote, 0);
210 ngx_conf_merge_ptr_value(conf->host_allow, prev->host_allow, NULL);
211 ngx_conf_merge_uint_value(conf->security_level, prev->security_level, 0);
212 /* Off by default: fail-closed signing refuses every client whose auth
213 * protocol cannot sign (all stock non-GSI clients), so it is opt-in. */
214 ngx_conf_merge_value(conf->signing_required, prev->signing_required, 0);
215 ngx_conf_merge_uint_value(conf->min_sec_level, prev->min_sec_level, 0);
216 ngx_conf_merge_value(conf->ztn_cleartext, prev->ztn_cleartext, 0);
217 /* ztn -maxsz analog; 0 = no extra cap (compatibility default). */
218 ngx_conf_merge_size_value(conf->ztn_maxsz, prev->ztn_maxsz, 0);
219 ngx_conf_merge_value(conf->opaque_strict, prev->opaque_strict, 0);
220 ngx_conf_merge_value(conf->tls, prev->tls, 0);
221 /* kTLS default OFF (phase-33 P5): software kTLS regresses vs OpenSSL's
222 * userspace AES-GCM on AES-NI hosts and is broken on some kernels (WSL2), so
223 * it is opt-in and documented HW-offload-only (`brix_ktls on`).
224 * SSL_OP_ENABLE_KTLS is a transparent no-op when the cipher/kernel cannot
225 * offload, so this default is byte-exact vs userspace TLS. */
226 ngx_conf_merge_value(conf->tls_ktls, prev->tls_ktls, 0);
227 /* §5.10: TLS session resumption on by default (unchanged); off full-handshakes. */
228 ngx_conf_merge_value(conf->tls_reuse, prev->tls_reuse, 1);
229}

231/* Identity & crypto: auth scheme + GSI/pwd, XrdAcc engine (+ native-authdb
232 * validation), SciTags/FRM, X.509 material + CRL, access log, tokens + L1/L2
233 * caches, sss/krb5/unix/host, security level, and TLS toggles. */
234char *
235brix_merge_srv_security(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
236 ngx_stream_brix_srv_conf_t *prev)
237{
238 /*
239 * Standard nginx inheritance rules: values set on the current server
240 * override the parent, otherwise we fall back to the parent or the hard
241 * coded module default. Each concern group is delegated to a helper below,
242 * invoked in the original linear order so cross-group derivations still see
243 * their already-merged inputs.
244 */
245 /* Shared common.* preamble (root defaults to "/": a pure cache node may
246 * omit brix_root and serve the whole namespace). Also covers the tier
247 * grammar + pmark + hard read-only enforcement — do not re-merge those. */
248 if (ngx_http_brix_shared_merge(cf, &prev->common, &conf->common, "/")
249 != NGX_CONF_OK)
250 {
251 return NGX_CONF_ERROR;
252 }
253 if (brix_merge_srv_gsi_acc(cf, conf, prev) != NGX_CONF_OK) {
254 return NGX_CONF_ERROR;
255 }
256 if (brix_merge_srv_x509(cf, conf, prev) != NGX_CONF_OK) {
257 return NGX_CONF_ERROR;
258 }
259 if (brix_merge_srv_tokens(cf, conf, prev) != NGX_CONF_OK) {
260 return NGX_CONF_ERROR;
261 }
262 brix_merge_srv_authtail(conf, prev);
263
264 return NGX_CONF_OK;
265}
