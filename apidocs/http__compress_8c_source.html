2 * http_compress.c — outbound (GET) response compression (see http_compress.h).
3 *
4 * Negotiation picks the best AVAILABLE codec named in Accept-Encoding by a fixed
5 * preference (zstd > br > gzip > xz > deflate > bzip2 > lz4), skipping ranges, HEAD,
6 * tiny files, and already-compressed content types. The compressed sender forces
7 * chunked output (compressed length is unknown up front) and streams the file

26/* Content types that are already compressed — never re-compress these. */
27static int
28content_type_incompressible(ngx_http_request_t *r)
29{
30 static const char *deny[] = {
31 "application/gzip", "application/zstd", "application/x-xz",
32 "application/x-bzip2", "application/zip", "application/x-7z-compressed",
33 "application/x-br", "application/x-brotli", /* already brotli-encoded */
34 "application/x-rar-compressed", "application/x-zstd",
35 "image/", "video/", "audio/", "application/octet-stream-compressed",
36 };
37 ngx_str_t *ct = &r->headers_out.content_type;
38 size_t i;
39
40 if (ct->len == 0 || ct->data == NULL) {
41 return 0; /* unknown type: allow compression */
42 }
43 for (i = 0; i < sizeof(deny) / sizeof(deny[0]); i++) {
44 size_t dl = ngx_strlen(deny[i]);
45 if (ct->len >= dl && ngx_strncasecmp(ct->data, (u_char *) deny[i], dl) == 0) {
46 return 1;
47 }
48 }
49 return 0;
50}

52/*
53 * WHAT: decide whether the ";q=..." parameter starting at `semi` explicitly
54 * disables the coding (an exactly-zero q-value).
55 * WHY: RFC 7231 5.3.1 lets a client veto a coding with "q=0" in any of its
56 * zero spellings ("q=0", "q=0.", "q=0.0", "q=0.00", "q=0.000"); a nonzero
57 * fractional digit (q=0.x) still accepts. Pulled out of the segment scan so
58 * the caller stays a flat name-match with no nested q-parser.
59 * HOW: find "q=0" within [semi, seg_end); a match is a real veto UNLESS it is
60 * followed by ".<digits>" containing a non-'0' digit (which makes q>0).
61 */
62static int
63accept_encoding_q_is_zero(const u_char *semi, const u_char *seg_end)
64{
65 const u_char *q = ngx_strlcasestrn((u_char *) semi, (u_char *) seg_end,
66 (u_char *) "q=0", 3 - 1);
67 const u_char *after;
68
69 if (q == NULL) {
70 return 0; /* no explicit q=0 => not vetoed */
71 }
72 after = q + 3; /* first char past "q=0" */
73 if (after < seg_end && *after == '.') {
74 const u_char *d = after + 1;
75 /* scan the fractional digits; any non-'0' => q>0 => accepted */
76 while (d < seg_end && *d >= '0' && *d <= '9') {
77 if (*d != '0') { return 0; }
78 d++;
79 }
80 }
81 return 1; /* q=0 / q=0. / q=0.0[00] => vetoed */
82}

84/*
85 * WHAT: test whether a single Accept-Encoding list member [seg, seg_end) names
86 * `token` (len `tlen`) as an acceptable coding.
87 * WHY: isolates the per-member name comparison + q-value veto from the list
88 * iteration so neither carries the other's branching.
89 * HOW: the coding name runs up to ';' (its q-value) or the member end, trimmed
90 * of trailing OWS; a case-insensitive length-exact match that is not
91 * vetoed by an explicit q=0 means the coding is acceptable.
92 */
93static int
94accept_encoding_segment_has(const u_char *seg, const u_char *seg_end,
95 const char *token, size_t tlen)
96{
97 const u_char *semi = ngx_strlchr((u_char *) seg, (u_char *) seg_end, ';');
98 const u_char *name_end = semi ? semi : seg_end;
99
100 while (name_end > seg && (name_end[-1] == ' ' || name_end[-1] == '\t')) {
101 name_end--;
102 }
103 if ((size_t) (name_end - seg) != tlen
104 || ngx_strncasecmp((u_char *) seg, (u_char *) token, tlen) != 0)
105 {
106 return 0;
107 }
108 if (semi != NULL && accept_encoding_q_is_zero(semi, seg_end)) {
109 return 0;
110 }
111 return 1;
112}

114/* Is `token` (len) present as an acceptable coding in the Accept-Encoding value
115 * (case-insensitive, ignoring an explicit ";q=0")? */
116static int
117accept_encoding_has(const u_char *ae, size_t ae_len, const char *token)
118{
119 size_t tlen = ngx_strlen(token);
120 const u_char *p = ae, *end = ae + ae_len;
121
122 while (p < end) {
123 const u_char *comma, *seg_end;
124
125 /* skip inter-member OWS — RFC 7230 permits SP and HTAB around list members */
126 while (p < end && (*p == ' ' || *p == '\t' || *p == ',')) { p++; }
127 comma = ngx_strlchr((u_char *) p, (u_char *) end, ',');
128 seg_end = comma ? comma : end;
129 if (accept_encoding_segment_has(p, seg_end, token, tlen)) {
130 return 1;
131 }
132 p = seg_end;
133 }
134 return 0;
135}

168/*
169 * WHAT: Request-owned cursor, codec and two fixed-size buffers for a send.
170 * WHY: Socket backpressure is not a codec failure. Preserve partially consumed
171 * input and pending output without retaining caller stack storage or
172 * allocating one more buffer for every chunk of a slow reader's file.
173 * HOW: A request cleanup owns fd/codec; its data also locates the write handler's
174 * cursor without borrowing any protocol module's request-context slot.
175 */
176typedef struct {
177 ngx_http_request_t *r;
178 ngx_fd_t fd;
179 off_t size;
180 off_t off;
181 brix_codec_stream_t *stream;
182 u_char *inbuf;
183 ngx_buf_t *output;
184 size_t fill;
185 size_t ip;
186 int finish;
187 int ended;
188 off_t bytes;
189 brix_http_compress_done_fn done;
190 void *data;
191} compress_send_t;
192
193/* WHAT: Release owned native resources once, including abandoned requests.
194 * WHY: Neither a disconnected reader nor a failed output filter may leak an fd
195 * or codec. HOW: Disarm each resource after release; pool storage remains. */
196static void
197compress_cleanup(void *data)
198{
199 compress_send_t *st = data;
200
201 if (st->stream != NULL) {
202 brix_codec_close(st->stream);
203 st->stream = NULL;
204 }
205 if (st->fd != NGX_INVALID_FILE) {
206 ngx_close_file(st->fd);
207 st->fd = NGX_INVALID_FILE;
208 }
209}

211/* WHAT: Find this request's send cursor, or NULL. WHY: Keep the shared sender
212 * independent of protocol contexts. HOW: Match our own cleanup identity. */
213static compress_send_t *
214compress_state(ngx_http_request_t *r)
215{
216 ngx_http_cleanup_t *cln;
217
218 for (cln = r->cleanup; cln != NULL; cln = cln->next) {
219 if (cln->handler == compress_cleanup) {
220 return cln->data;
221 }
222 }
223 return NULL;
224}

226/* WHAT: Queue the current memory buffer, marking only codec END as response EOF.
227 * WHY: Cleartext and TLS both send generated bytes, never a file-backed buffer.
228 * HOW: Reuse only after all downstream filters have drained the previous bytes. */
229static ngx_int_t
230compress_emit(compress_send_t *st, size_t len)
231{
232 ngx_buf_t *b = st->output;
233 ngx_chain_t out;
234
235 b->pos = b->start;
236 b->last = b->start + len;
237 b->temporary = 1;
238 b->memory = 1;
239 b->flush = 1;
240 b->last_buf = st->ended && st->r == st->r->main;
241 b->last_in_chain = st->ended;
242 out.buf = b;
243 out.next = NULL;
244 return ngx_http_output_filter(st->r, &out);
245}

247/*
248 * WHAT: emit the response headers for a compressed body — 200 + ETag/Last-Modified,
249 * forced chunked, Content-Encoding + Vary.
250 * WHY: keeps all header mutation (and its allocation failure paths) out of the
251 * streaming orchestrator so that reads as a flat init-then-stream sequence.
252 * HOW: set the file headers, drop Content-Length (unknown => chunked), then push
253 * the Content-Encoding (desc->http_token) and Vary list elements.
254 */
255static ngx_int_t
256compress_set_headers(ngx_http_request_t *r, const brix_codec_desc_t *desc,
257 off_t size, time_t mtime, unsigned etag_flags)
258{
259 ngx_table_elt_t *ce, *vary;
260
261 if (brix_http_set_file_headers(r, mtime, size, size, NULL, etag_flags,
262 0, 0, size > 0 ? size - 1 : 0) != NGX_OK)
263 {
264 return NGX_ERROR;
265 }
266 r->headers_out.content_length_n = -1; /* unknown -> chunked */
267 if (r->headers_out.content_length != NULL) {
268 r->headers_out.content_length->hash = 0;
269 r->headers_out.content_length = NULL;
270 }
271
272 ce = ngx_list_push(&r->headers_out.headers);
273 vary = ngx_list_push(&r->headers_out.headers);
274 if (ce == NULL || vary == NULL) {
275 return NGX_ERROR;
276 }
277 ce->hash = 1;
278 ngx_str_set(&ce->key, "Content-Encoding");
279 ce->value.data = (u_char *) desc->http_token;
280 ce->value.len = ngx_strlen(desc->http_token);
281 vary->hash = 1;
282 ngx_str_set(&vary->key, "Vary");
283 ngx_str_set(&vary->value, "Accept-Encoding");
284 return NGX_OK;
285}

287/* WHAT: Whether downstream filters still own output. WHY: A codec buffer may
288 * not be overwritten merely because a filter returned NGX_OK. HOW: Inspect the
289 * buffer cursor and both nginx buffering masks (including TLS output). */
290static int
291compress_pending(const compress_send_t *st)
292{
293 return st->output->pos != st->output->last
294 || st->r->buffered || st->r->connection->buffered;
295}

297/* WHAT: Read the next bounded storage chunk; reject premature EOF.
298 * WHY: Do not leak bytes appended past the stat'd representation or bless a
299 * truncated file as a complete compressed object. HOW: Read through the VFS
300 * seam, limiting each read to the original remaining size. */
301static ngx_int_t
302compress_read_chunk(compress_send_t *st)
303{
304 off_t remaining = st->size - st->off;
305 size_t want = remaining > BRIX_COMPRESS_CHUNK
306 ? BRIX_COMPRESS_CHUNK : (size_t) remaining;
307 size_t got = 0;
308
309 if (brix_vfs_pread_full(st->fd, st->inbuf, want, st->off, &got) != NGX_OK
310 || got != want)
311 {
312 return NGX_ERROR;
313 }
314 st->off += (off_t) got;
315 st->fill = got;
316 st->ip = 0;
317 st->finish = st->off == st->size;
318 return NGX_OK;
319}

321/* WHAT: Advance the codec once, emitting only initialized output bytes.
322 * WHY: Preserve partial input across backpressure and refuse a stuck codec.
323 * HOW: Keep its cursor in request storage, count produced bytes and mark EOF
324 * only when the codec reports END, not just when it consumed the final input. */
325static ngx_int_t
326compress_codec_step(compress_send_t *st)
327{
328 size_t op = 0, before = st->ip;
329 brix_codec_rc_t rc;
330
331 rc = brix_codec_step(st->stream, st->inbuf, st->fill, &st->ip,
332 st->output->start, BRIX_COMPRESS_CHUNK, &op, st->finish);
333 if (rc < 0 || (rc != BRIX_CODEC_END && op == 0 && before == st->ip)) {
334 return NGX_ERROR;
335 }
336 st->bytes += (off_t) op;
337 st->ended = rc == BRIX_CODEC_END;
338 if (op > 0 || st->ended) {
339 return compress_emit(st, op);
340 }
341 return NGX_OK;
342}

344/* WHAT: Pump until complete, backpressured, or failed. WHY: Bound live output
345 * memory independently of file size and reader speed. HOW: Drain the one pending
346 * buffer before reading/encoding another; NGX_AGAIN preserves all cursors. */
347static ngx_int_t
348compress_stream_body(compress_send_t *st)
349{
350 ngx_int_t rc;
351
352 for (;;) {
353 if (compress_pending(st)) {
354 rc = ngx_http_output_filter(st->r, NULL);
355 if (rc == NGX_ERROR) {
356 return NGX_ERROR;
357 }
358 if (rc == NGX_AGAIN || compress_pending(st)) {
359 return NGX_AGAIN;
360 }
361 }
362 if (st->ended) {
363 return NGX_OK;
364 }
365 if (st->ip == st->fill && !st->finish
366 && compress_read_chunk(st) != NGX_OK)
367 {
368 return NGX_ERROR;
369 }
370 rc = compress_codec_step(st);
371 if (rc != NGX_OK) {
372 return rc;
373 }
374 }
375}

377/* WHAT: Rearm native write readiness and the configured send timeout.
378 * WHY: Slow readers must neither truncate a response nor keep a worker spinning.
379 * HOW: Respect an output filter's existing rate-limit delay/timer. */
380static ngx_int_t
381compress_arm_write(ngx_http_request_t *r)
382{
383 ngx_event_t *wev = r->connection->write;
384 ngx_http_core_loc_conf_t *cf;
385
386 cf = ngx_http_get_module_loc_conf(r, ngx_http_core_module);
387 if (ngx_handle_write_event(wev, cf->send_lowat) != NGX_OK) {
388 return NGX_ERROR;
389 }
390 if (!wev->ready && !wev->delayed) {
391 ngx_add_timer(wev, cf->send_timeout);
392 }
393 return NGX_OK;
394}

396/* WHAT: Complete or resume a parked compressed response on the event loop.
397 * WHY: NGX_AGAIN means the socket will become writable, not a fatal send error.
398 * HOW: Resume the request-owned pump; one completion balances its held request
399 * reference, while request cleanup also covers disconnect/error finalization. */
400static void
401compress_write_handler(ngx_http_request_t *r)
402{
403 compress_send_t *st = compress_state(r);
404 ngx_event_t *wev = r->connection->write;
405 ngx_int_t rc;
406
407 if (st == NULL) {
408 ngx_http_finalize_request(r, NGX_ERROR);
409 return;
410 }
411 if (wev->timedout && !wev->delayed) {
412 r->connection->timedout = 1;
413 rc = NGX_HTTP_REQUEST_TIME_OUT;
414 } else {
415 if (wev->timedout) {
416 wev->timedout = 0;
417 wev->delayed = 0;
418 }
419 if (wev->delayed) {
420 return;
421 }
422 rc = compress_stream_body(st);
423 }
424 if (rc == NGX_AGAIN && compress_arm_write(r) == NGX_OK) {
425 return;
426 }
427 if (rc == NGX_AGAIN) {
428 rc = NGX_ERROR;
429 }
430 compress_cleanup(st);
431 r->write_event_handler = ngx_http_request_empty_handler;
432 if (st->done != NULL) {
433 st->done(r, st->data, st->bytes, rc);
434 }
435 ngx_http_finalize_request(r, rc);
436}

438/* WHAT: Create a request-owned sender and its cleanup. WHY: Never retain the
439 * caller's stack across asynchronous output. HOW: Allocate just two 64KiB
440 * buffers; once the cleanup is armed it owns the fd on every return path. */
441static compress_send_t *
442compress_create(ngx_http_request_t *r, ngx_fd_t fd, off_t size,
443 brix_codec_id_t codec)
444{
445 ngx_http_cleanup_t *cln = ngx_http_cleanup_add(r, sizeof(compress_send_t));
446 compress_send_t *st;
447
448 if (cln == NULL) {
449 ngx_close_file(fd);
450 return NULL;
451 }
452 st = cln->data;
453 ngx_memzero(st, sizeof(*st));
454 st->r = r;
455 st->fd = fd;
456 st->size = size;
457 cln->handler = compress_cleanup;
458 st->stream = brix_codec_open(codec, BRIX_CODEC_DIR_COMPRESS, -1, NULL);
459 st->inbuf = ngx_palloc(r->pool, BRIX_COMPRESS_CHUNK);
460 st->output = ngx_create_temp_buf(r->pool, BRIX_COMPRESS_CHUNK);
461 if (st->stream == NULL || st->inbuf == NULL || st->output == NULL) {
462 compress_cleanup(st);
463 return NULL;
464 }
465 return st;
466}
