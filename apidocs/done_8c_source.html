1/* File: done.c — Main-thread completion callback for native TPC pull
2 * WHAT: Handles the thread-pool-to-event-loop handoff when a TPC pull operation completes. Restores the deferred request context, then dispatches based on reply_kind: SYNC mode sends kXR_open success response with fhandle+stat data (or error if pull failed), async mode sends open success response with file handle and optional stat block including tpc.key for xrdcp extraction. On connection close during in-flight pull, releases dst_fd, unlinks dst_path, frees fhandle slot.
3 *
4 * WHY: Native TPC pull runs blocking socket I/O inside a detached thread-pool worker; the nginx event loop must resume when the thread finishes because it deferred the kXR_open response. This callback bridges the thread→event-loop boundary — restoring streamid context, updating file metadata (tpc_done flag, bytes_written, cached stat info), and sending the wire response that the client is waiting for. Sync vs async reply_kind determines whether to build a full ServerOpenBody with stat data or just send an ok status code.
5 *
6 * HOW: ev->data = ngx_thread_task_t → extract t (brix_tpc_pull_t) from task→ctx → restore request context via brix_aio_restore_request(streamid) → if connection closed, close dst_fd + unlink dst_path + free fhandle slot → reply_kind == BRIX_TPC_REPLY_SYNC: build ServerOpenBody with fhandle[idx] + optional statbuf (inode/size/permissions/mtime + tpc.key) via snprintf, allocate response buffer with ngx_palloc, build wire header + body via brix_build_resp_hdr + ngx_memcpy, queue response via brix_queue_response → async reply_kind: send ok status code only → brix_aio_resume(c) in all paths. Returns void (callback).
7 * */

19static void
20tpc_sess_finish_pull(brix_tpc_pull_t *t, int ok, brix_sess_end_t why)
21{
22 char errscratch[BRIX_SESSLOG_ERR_MAX];
23 const char *err;
24
25 if (t == NULL) {
26 return;
27 }
28
29 err = ok ? NULL : brix_sesslog_err_from_kxr(
30 t->xrd_error ? t->xrd_error : kXR_ServerError,
31 errscratch, sizeof(errscratch));
32 brix_sess_xfer_add(&t->sess_xfer, t->bytes_written);
33 brix_sess_result(t->sess, ok, t->src_path, BRIX_SESS_MODE_READ, err);
34 brix_sess_xfer_end(t->sess, &t->sess_xfer,
35 ok ? BRIX_SESS_XFER_COMPLETE
36 : BRIX_SESS_XFER_ABORTED);
37 brix_sess_end(t->sess, why);
38 t->sess = NULL;
39}

41/* WHAT: Finish transfer accounting — registry state update, TPC metric, registry remove.
42 * WHY: Every completion path (orphaned, connection-gone, sync ok/fail, async ok/fail)
43 * performs exactly this triple with the same log target; centralizing it keeps the
44 * registry and metrics in lockstep across all outcomes.
45 * HOW: ok selects DONE+SUCCESS vs ERROR+ERROR; bytes_written is reported both to the
46 * registry (as offset) and to the metric, then the registry entry is removed. */
47static void
48tpc_done_account(brix_tpc_pull_t *t, int ok, ngx_log_t *log)
49{
50 (void) brix_tpc_registry_update(t->transfer_id,
51 (off_t) t->bytes_written,
52 ok ? BRIX_TPC_STATE_DONE
53 : BRIX_TPC_STATE_ERROR, log);
54 brix_tpc_metric_transfer(BRIX_TPC_PROTO_STREAM,
55 BRIX_TPC_DIR_PULL,
56 ok ? BRIX_TPC_METRIC_SUCCESS

62/* WHAT: Reclaim the destination of a failed/abandoned pull — close fd, unlink the
63 * partial file, release the fhandle slot when a live ctx still owns one.
64 * WHY: A failed transfer must never leave a truncated object or a leaked fd/fhandle
65 * behind; the same close→unlink→slot-free sequence is shared by the orphaned,
66 * connection-gone and async-failure paths.
67 * HOW: close(dst_fd) then confined unlink of dst_path under root_canon; if ctx is
68 * still present and idx is a valid slot, clear fd/transfer_id and free the
69 * fhandle (fd already -1, so the free cannot double-close). Pass ctx=NULL when
70 * there is no fhandle table to touch. */
71/* Phase-105 export authority for the destination side of this pull: the TPC
72 * endpoint's own write posture. The destination is a real export object (TPC
73 * writes the final name directly, it is not a service-owned temp), so even the
74 * failure cleanup goes through the gated form — a read-only endpoint could not
75 * have created it, and must not be able to remove one either. */
76static void

78 brix_vfs_export_op_ctx_t *opctx)
79{
80 brix_vfs_export_op_ctx_init(opctx, log, t->conf->common.root_canon,
81 brix_vfs_policy_from_write_enable(t->conf->common.allow_write),

85static void
86tpc_done_teardown_dst(brix_tpc_pull_t *t, brix_ctx_t *ctx, int idx,
87 ngx_log_t *log)
88{
89 int remove_final = 1;
90
91 if (t->dst_writer != NULL) {
92 remove_final = 0;
93 brix_vfs_writer_abort(t->dst_writer);
94 }
95 if (ctx != NULL && ctx->files != NULL && idx >= 0 && idx < BRIX_MAX_FILES
96 && ctx->files[idx].writer != NULL)
97 {
98 remove_final = 0;
99 }
100 if (t->dst_writer == NULL && t->dst_fd >= 0) {
101 close(t->dst_fd);
102 }
103 if (remove_final) {
104 brix_vfs_export_op_ctx_t opctx;
105
106 tpc_dst_opctx(t, log, &opctx);
107 (void) brix_vfs_export_unlink(&opctx, t->dst_path);
108 }
109 if (ctx != NULL && ctx->files != NULL && idx >= 0 && idx < BRIX_MAX_FILES) {
110 ctx->files[idx].fd = -1;
111 ctx->files[idx].tpc_transfer_id = 0;
112 brix_free_fhandle(ctx, idx);
113 }
114}

190/* WHAT: SYNC-mode failure reply — discard the half-written destination, account the
191 * error, and answer the waiting kXR_sync with the pull's error code.
192 * WHY: In SYNC mode the client already holds the open handle; on failure we must
193 * free the slot (or close the raw fd if the slot was reaped) and unlink the
194 * partial file so a failed transfer never leaves a truncated object behind.
195 * HOW: slot-free XOR raw close depending on whether the handle survived, confined
196 * unlink, then registry/metric accounting and the error response + aio resume.
197 * c is never NULL here: brix_tpc_pull_done gates the no-connection case before
198 * dispatching to any reply helper. */
199static void
200tpc_done_sync_fail(brix_tpc_pull_t *t, brix_ctx_t *ctx, ngx_connection_t *c,
201 int idx)
202{
203 brix_file_t *file;
204 int err = t->xrd_error ? t->xrd_error : kXR_ServerError;
205
206 /* idx may be out of range if the handle was reaped; guard before use. */
207 file = (ctx->files != NULL && idx >= 0 && idx < BRIX_MAX_FILES)
208 ? &ctx->files[idx] : NULL;
209
210 tpc_sess_finish_pull(t, 0, BRIX_SESS_END_ERROR);
211
212 /* Failed copy: discard the half-written destination. If the slot is
213 * still valid, free it (which also closes the fd); otherwise close
214 * the raw fd directly. Either way unlink the partial file so a
215 * failed transfer never leaves a truncated object behind. */
216 if (file != NULL) {
217 int had_writer = file->writer != NULL;
218
219 file->fd = -1;
220 file->tpc_transfer_id = 0;
221 brix_free_fhandle(ctx, idx);
222 if (!had_writer) {
223 brix_vfs_export_op_ctx_t opctx;
224
225 tpc_dst_opctx(t, c->log, &opctx);
226 (void) brix_vfs_export_unlink(&opctx, t->dst_path);
227 }
228 } else {
229 if (t->dst_writer == NULL && t->dst_fd >= 0) {
230 close(t->dst_fd);
231 }
232 if (t->dst_writer == NULL) {
233 brix_vfs_export_op_ctx_t opctx;
234
235 tpc_dst_opctx(t, c->log, &opctx);
236 (void) brix_vfs_export_unlink(&opctx, t->dst_path);
237 }
238 }
239
240 tpc_done_account(t, 0, c->log);
241
242 brix_log_access(ctx, c, "TPC-PULL", t->dst_path, "error",
243 0, (uint16_t) err,
244 t->err_msg[0] ? t->err_msg : "TPC pull failed",
245 0);
246 BRIX_OP_ERR(ctx, BRIX_OP_SYNC);
247 brix_send_error(ctx, c, (uint16_t) err,
248 t->err_msg[0] ? t->err_msg : "TPC pull failed");
249 brix_aio_resume(c);
250}

252/* WHAT: SYNC-mode reply — the waiting request is a kXR_sync on an already-open
253 * destination handle; answer it with ok/error and keep the handle open.
254 * WHY: The pull was triggered by a kXR_sync (the client opened the file, then
255 * sync'd to start the copy), so the client closes the handle itself; on
256 * success we refresh the handle's cached metadata from the completed file.
257 * HOW: failure delegates to tpc_done_sync_fail; success marks the handle done,
258 * accumulates bytes_written, fstat-refreshes cached size/inode so subsequent
259 * stat/read on the handle needs no path syscall (INVARIANT 7), then accounts
260 * the transfer and sends ok + aio resume. */
261static void
262tpc_done_reply_sync(brix_tpc_pull_t *t, brix_ctx_t *ctx, ngx_connection_t *c,
263 int idx)
264{
265 brix_file_t *file;
266
267 if (t->result != NGX_OK) {
268 tpc_done_sync_fail(t, ctx, c, idx);
269 return;
270 }
271
272 /* idx may be out of range if the handle was reaped; guard before use. */
273 file = (ctx->files != NULL && idx >= 0 && idx < BRIX_MAX_FILES)
274 ? &ctx->files[idx] : NULL;
275
276 if (tpc_done_commit_dst(t, ctx, idx) != NGX_OK) {
277 t->result = NGX_ERROR;
278 tpc_done_sync_fail(t, ctx, c, idx);
279 return;
280 }
281
282 /* Success: mark the handle done and refresh its cached metadata from
283 * the now-complete file so a subsequent stat/read on the same handle
284 * sees the real size/inode without another path syscall (INVARIANT 7). */
285 tpc_sess_finish_pull(t, 1, BRIX_SESS_END_SERVER);
286 if (file != NULL) {
287 file->tpc_done = 1;
288 file->tpc_started = 0;
289 file->tpc_transfer_id = 0;
290 file->bytes_written += t->bytes_written;
291
292 (void) tpc_done_refresh_stat(t, ctx, c, idx);
293 }
294
295 brix_log_access(ctx, c, "TPC-PULL", t->dst_path, "ok",
296 1, 0, NULL, t->bytes_written);
297 BRIX_OP_OK(ctx, BRIX_OP_SYNC);
298 tpc_done_account(t, 1, c->log);
299 brix_send_ok(ctx, c, NULL, 0);
300 brix_aio_resume(c);
301}

303/* WHAT: Format the optional ASCII stat/tpc.key trailer for the kXR_open ok body.
304 * WHY: xrdcp scrapes tpc.key from the open response, and kXR_retstat clients need
305 * the "ino size flags mtime" stat line; the trailer shape depends on which of
306 * the two the client can consume.
307 * HOW: fstat(dst_fd) → derive xrootd readable/writable flags from POSIX mode bits →
308 * snprintf one of three trailer shapes:
309 * retstat + key -> "ino size flags mtime\ntpc.key=KEY"
310 * retstat only -> "ino size flags mtime"
311 * key only -> "\ntpc.key=KEY" (so xrdcp can still scrape the key even
312 * though it didn't request a stat)
313 * On fstat failure or neither-wanted, statbuf stays empty. */
314static void
315tpc_done_stat_trailer(brix_tpc_pull_t *t, char *statbuf, size_t buflen)
316{
317 struct stat st;
318 ngx_flag_t want_stat = (t->options & kXR_retstat) ? 1 : 0;
319 int sf = 0;
320
321 statbuf[0] = '\0';
322
323 if (!t->dst_stat_valid) {
324 return;
325 }
326 st = t->dst_stat;
327
328 /* xrootd stat flags are derived from POSIX mode bits. */
329 if (st.st_mode & (S_IRUSR | S_IRGRP | S_IROTH)) { sf |= kXR_readable; }
330 if (st.st_mode & (S_IWUSR | S_IWGRP | S_IWOTH)) { sf |= kXR_writable; }
331
332 if (want_stat && t->tpc_key[0] != '\0') {
333 snprintf(statbuf, buflen, "%llu %lld %d %ld\ntpc.key=%s",
334 (unsigned long long) st.st_ino, (long long) st.st_size,
335 sf, (long) st.st_mtime, t->tpc_key);
336 } else if (want_stat) {
337 snprintf(statbuf, buflen, "%llu %lld %d %ld",
338 (unsigned long long) st.st_ino, (long long) st.st_size,
339 sf, (long) st.st_mtime);
340 } else if (t->tpc_key[0] != '\0') {
341 /* Include key even without kXR_retstat so xrdcp can extract it. */
342 snprintf(statbuf, buflen, "\ntpc.key=%s", t->tpc_key);
343 }
344}

346/* WHAT: Build and queue the deferred kXR_open ok response for the async path.
347 * WHY: The client's OPEN was deferred while the pull ran in the worker thread;
348 * it now needs the ServerOpenBody carrying its fhandle index, plus the
349 * optional stat/tpc.key trailer so xrdcp can extract the key.
350 * HOW: Body layout: fixed ServerOpenBody (carries the 1-byte fhandle index)
351 * optionally followed by a NUL-terminated ASCII stat/tpc.key trailer.
352 * `bodylen` is grown to cover the trailer before allocating; the wire
353 * header's dlen must equal bodylen (header length is added separately).
354 * Assemble buf in wire order: [response header][ServerOpenBody][trailer] —
355 * header first with dlen=bodylen, body at the fixed header offset, optional
356 * trailer (with its NUL) appended immediately after the body — then queue
357 * it and resume the connection. Alloc failure replies kXR_NoMemory. */
358static void
359tpc_done_send_open_ok(brix_tpc_pull_t *t, brix_ctx_t *ctx,
360 ngx_connection_t *c, int idx)
361{
362 ServerOpenBody body_s;
363 char statbuf[256];
364 size_t bodylen, total;
365 u_char *buf;
366
367 tpc_done_stat_trailer(t, statbuf, sizeof(statbuf));
368
369 bodylen = sizeof(ServerOpenBody);
370 /* +1 for the NUL terminator that the trailer carries on the wire. */
371 if (statbuf[0] != '\0') {
372 bodylen += strlen(statbuf) + 1;
373 }
374
375 total = XRD_RESPONSE_HDR_LEN + bodylen;
376 buf = ngx_palloc(c->pool, total);
377 if (buf == NULL) {
378 brix_send_error(ctx, c, kXR_NoMemory, "alloc failed");
379 brix_aio_resume(c);
380 return;
381 }
382
383 brix_build_resp_hdr(t->streamid, kXR_ok, (uint32_t) bodylen,
384 (ServerResponseHdr *) buf);
385
386 ngx_memzero(&body_s, sizeof(body_s));
387 body_s.fhandle[0] = (u_char) idx;
388 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, &body_s, sizeof(body_s));
389
390 if (statbuf[0]) {
391 size_t slen = strlen(statbuf) + 1; /* include NUL terminator */
392 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN + sizeof(ServerOpenBody),
393 statbuf, slen);
394 }
395
396 brix_queue_response(ctx, c, buf, total);
397 brix_aio_resume(c);
398}

400/* WHAT: Async-mode reply — the deferred request is the kXR_open itself; answer
401 * it with the pull result, tearing the handle down ourselves on failure.
402 * WHY: The pull was kicked off directly from the kXR_open, so on failure the
403 * client never got a usable fhandle — we must close+unlink the partial
404 * destination and release the slot before replying with the error.
405 * HOW: failure: session finish, destination teardown, accounting, access log,
406 * error response, aio resume. success: session finish, access log, clear
407 * the slot's transfer id, accounting, then build/queue the open ok
408 * response (with stat/tpc.key trailer) via tpc_done_send_open_ok. */
409static void
410tpc_done_reply_open(brix_tpc_pull_t *t, brix_ctx_t *ctx, ngx_connection_t *c,
411 int idx)
412{
413 if (t->result != NGX_OK) {
414 int err = t->xrd_error ? t->xrd_error : kXR_ServerError;
415
416 tpc_sess_finish_pull(t, 0, BRIX_SESS_END_ERROR);
417
418 /* Failed copy: close+unlink the partial destination and release the
419 * fhandle slot before replying with the error to the OPEN. (c is never
420 * NULL here — brix_tpc_pull_done gates the no-connection case.) */
421 tpc_done_teardown_dst(t, ctx, idx, c->log);
422
423 tpc_done_account(t, 0, c->log);
424
425 brix_log_access(ctx, c, "TPC-PULL", t->dst_path, "error",
426 0, (uint16_t) err,
427 t->err_msg[0] ? t->err_msg : "TPC pull failed", 0);
428 BRIX_OP_ERR(ctx, BRIX_OP_OPEN_WR);
429 brix_send_error(ctx, c, (uint16_t) err,
430 t->err_msg[0] ? t->err_msg : "TPC pull failed");
431 brix_aio_resume(c);
432 return;
433 }
434
435 if (tpc_done_commit_dst(t, ctx, idx) != NGX_OK) {
436 t->result = NGX_ERROR;
437 tpc_done_sync_fail(t, ctx, c, idx);
438 return;
439 }
440 (void) tpc_done_refresh_stat(t, ctx, c, idx);
441
442 tpc_sess_finish_pull(t, 1, BRIX_SESS_END_SERVER);
443 brix_log_access(ctx, c, "TPC-PULL", t->dst_path, "ok", 1, 0, NULL, 0);
444 BRIX_OP_OK(ctx, BRIX_OP_OPEN_WR);
445 if (ctx->files != NULL && idx >= 0 && idx < BRIX_MAX_FILES) {
446 ctx->files[idx].tpc_transfer_id = 0;
447 }
448 tpc_done_account(t, 1, c->log);
449
450 tpc_done_send_open_ok(t, ctx, c, idx);
451}

453/* WHAT: Main-thread completion callback — restore deferred request, dispatch SYNC vs async reply, build/send kXR_open response or error. */
454void
455brix_tpc_pull_done(ngx_event_t *ev)
456{
457 ngx_thread_task_t *task;
458 brix_tpc_pull_t *t;
459 brix_ctx_t *ctx;
460 ngx_connection_t *c;
461 int idx;
462
463 /*
464 * Re-entry point from the thread pool: nginx delivers the completion as an
465 * event whose ->data is the original ngx_thread_task_t. The pull state
466 * (brix_tpc_pull_t) lives in task->ctx; we recover ctx/connection/handle
467 * from it. From here on we are back on the event thread — no blocking I/O.
468 */
469 task = ev->data;
470 t = task->ctx;
471 ctx = t->ctx;
472 c = t->c;
473 idx = t->fhandle_idx;
474
475 /*
476 * Defensive: the pull state can outlive its session/connection (the worker
477 * thread completes after a teardown). With no ctx/connection there is
478 * nobody to reply to and no fhandle table to touch — finish the transfer
479 * bookkeeping against the cycle log and reclaim the destination.
480 */
481 if (ctx == NULL || c == NULL) {
482 ngx_log_t *log = (c != NULL) ? c->log : ngx_cycle->log;
483
484 tpc_sess_finish_pull(t, 0, BRIX_SESS_END_ERROR);
485 tpc_done_account(t, 0, log);
486 tpc_done_teardown_dst(t, NULL, idx, log);
487 return;
488 }
489
490 /*
491 * restore_request rebinds the deferred kXR_open streamid context. If it
492 * fails the client connection went away while the pull was running in the
493 * worker thread: there is nobody to reply to, so just reclaim everything
494 * (fd, partially-written file, fhandle slot, registry entry) and bail —
495 * no brix_aio_resume(), because there is no connection to resume.
496 */
497 if (!brix_aio_restore_request(ctx, t->streamid)) {
498 tpc_sess_finish_pull(t, t->result == NGX_OK,
499 t->result == NGX_OK ? BRIX_SESS_END_SERVER
500 : BRIX_SESS_END_ERROR);
501 /* Connection closed while pull was in-flight — release resources. */
502 tpc_done_account(t, 0, c->log);
503 tpc_done_teardown_dst(t, ctx, idx, c->log);
504 return;
505 }
506
507 /*
508 * SYNC reply path: the pull was triggered by a kXR_sync on an already-open
509 * destination handle (the client opened the file, then sync'd to start the
510 * copy). The waiting request is the sync, so we reply with ok/error for
511 * SYNC and keep the file handle open for the client to close itself.
512 *
513 * Async reply path (reply_kind != SYNC): the pull was kicked off directly
