100/* Render `format` and check each expectation token in the NULL-terminated
101 * list: a leading '!' means the token must be ABSENT from the rendering.
102 * Frees the rendering; failures name the offending token. */
103static void
104render_expect(const doctor_ep *eps, int n, const char *format, ...)
105{
106 char *s = render(eps, n, format);
107 va_list ap;
108 const char *tok;
109
110 va_start(ap, format);
111 while ((tok = va_arg(ap, const char *)) != NULL) {
112 int absent = tok[0] == '!';
113 if (absent ? has(s, tok + 1) : !has(s, tok)) {
114 printf("FAIL %s render: %s '%s'\n", format,
115 absent ? "unexpected" : "missing", tok + absent);
116 g_fail++;
117 }
118 }
119 va_end(ap);
120 free(s);
121}
