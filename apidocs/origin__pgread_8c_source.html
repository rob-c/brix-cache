2/*
3 * origin_pgread.c — page-verified origin reads (kXR_pgread) for the cache fill.
4 *
5 * WHAT: brix_cache_origin_pgread_chunk() — the integrity-checking sibling of
6 * brix_cache_origin_read_chunk(). It asks the origin for the SAME byte
7 * range with kXR_pgread instead of kXR_read, so the reply carries a
8 * CRC32c for every 4 KiB page, and it verifies each of those CRCs before
9 * one byte reaches the sink.
10 *
11 * WHY: a cache fill over plain kXR_read trusts the wire completely. TCP's
12 * checksum is 16 bits and covers only what the endpoints agree on; a
13 * flipped bit in a switch, a mis-behaving on-path box, or an active MITM
14 * on a cleartext `root://` link all produce bytes the fill would commit
15 * and then serve — permanently, to every later client, as a good hit.
16 * The whole-file digest check (verify.h) closes that hole ONLY for a
17 * fill that reads a complete file AND an origin that can answer
18 * kXR_Qcksum; a slice/block fill, a range read, and a digest-less origin
19 * are all unprotected. Per-page CRC32c protects every read, verified
20 * BEFORE the bytes are written, and it is the mechanism XRootD itself
21 * standardised for exactly this (INVARIANT #1).
22 *
23 * HOW: one kXR_pgread request, then a train of kXR_status(4007) frames:
24 * [ServerResponseHdr 8B status=kXR_status, dlen=24]
25 * [ServerResponseBody_Status 16B crc32c, streamID, requestid,
26 * resptype, reserved, dlen]
27 * [ServerResponseBody_pgRead 8B file offset]
28 * ---- bdy.dlen page bytes follow, NOT counted in hdr.dlen ----
29 * The body's own crc32c covers the 20 bytes streamID..offset; the page
30 * bytes are units of [CRC32c-BE 4][data <= 4096] aligned to the FILE
31 * offset. Decoding + per-page verification is NOT reimplemented here:
32 * it is xrdp_pg_decode() from core/compat/pgio.h, the same kernel the
33 * server encodes with and the native client decodes with, so producer
34 * and consumer agree by construction. resptype kXR_PartialResult keeps
35 * the train going; kXR_FinalResult ends it.
36 *
37 * Capability, not trial and error: the bootstrap records the origin's
38 * kXR_protocol advert in oc->srv_flags, so an origin that never claimed
39 * kXR_suppgrw is answered from that flag with no wasted round trip. An
40 * origin that advertises the capability and then refuses the request
41 * (kXR_Unsupported / kXR_InvalidRequest) is treated identically. Both
42 * return BRIX_CACHE_PGREAD_UNSUPPORTED and set NO task error — the
43 * caller owns the fall-back-or-fail decision, because only it knows
44 * whether the operator asked for best-effort or require.
45 */

64/* The fixed status body of a pgread frame: ServerResponseBody_Status (16) +
65 * ServerResponseBody_pgRead (8). hdr.dlen is exactly this — the page data is
66 * NOT counted in it (see the wire contract in root/response/status.c). */
67#define PG_BODY_LEN 24u
68/* The body's crc32c covers everything after itself: streamID..offset = 20 B. */
69#define PG_BODY_CRC_LEN (PG_BODY_LEN - 4u)
70/* Byte offsets inside that body (the layout above, read unaligned-safe). */
71#define PG_OFF_RESPTYPE 7u
72#define PG_OFF_DLEN 12u
73#define PG_OFF_FILEOFF 16u
74/* resptype values: the train continues on PARTIAL, ends on FINAL. */
75#define PG_RESP_FINAL 0u
76#define PG_RESP_PARTIAL 1u
77/* A kXR_error body is a message, not a status frame: read up to this much of
78 * it so the origin's own text survives into the task error. */
79#define PG_ERR_BODY_MAX 512u
80
81/* One parsed kXR_status frame head. */
82typedef struct {
83 uint8_t resptype; /* PG_RESP_FINAL | PG_RESP_PARTIAL */
84 uint32_t pgdlen; /* page bytes that follow this frame's body */
85 uint64_t foff; /* FILE offset those pages are aligned to */
86} pg_frame_t;

88/* The largest page-data blob a reply to `want` bytes can legitimately carry:
89 * the data plus one 4-byte CRC per page, plus one extra page's worth of CRC
90 * for an unaligned first fragment. */
91static uint32_t
92pg_max_pgdlen(size_t want)
93{
94 size_t pages = want / kXR_pgPageSZ + 2;
95
96 return (uint32_t) (want + pages * 4);
97}

100static int
101pg_send_request(brix_cache_fill_t *t, brix_cache_origin_conn_t *oc,
102 const u_char fhandle[XRD_FHANDLE_LEN], const brix_cache_read_range_t *rng)
103{
104 ClientPgReadRequest req;
105
106 /* slots: 1 bootstrap, 2 open, 3 read, 4 pgread */
107 xrd_creq_begin(&req, sizeof(req), 4, kXR_pgread);
108 ngx_memcpy(req.fhandle, fhandle, XRD_FHANDLE_LEN);
109 req.offset = (kXR_int64) brix_plat_htobe64(rng->read_off);
110 req.rlen = htonl((kXR_int32) rng->want);
111 req.dlen = 0; /* no request args (no pathid/retry) */
112
113 if (brix_cache_io_send(oc, &req, sizeof(req)) != 0) {
114 brix_cache_set_error(t, kXR_ServerError, errno,
115 "cache origin pgread write failed");
116 return -1;
117 }
118 return 0;
119}

121/* 1 iff a kXR_error body says "I do not implement this request" — the two

124static int

137/* Validate the 24-byte status body and fill fr. Returns 0 / -1 (error set). */
138static int
139pg_parse_body(brix_cache_fill_t *t, u_char *body, pg_frame_t *fr)
140{
141 uint32_t want_crc, got_crc;
142
143 want_crc = xrd_get_u32_be(body);
144 got_crc = brix_crc32c_value(body + 4, PG_BODY_CRC_LEN);
145 if (want_crc != got_crc) {
146 brix_cache_set_error(t, kXR_ChkSumErr, 0,
147 "cache origin pgread status header CRC mismatch");
148 return -1;
149 }
150
151 fr->resptype = body[PG_OFF_RESPTYPE];
152 fr->pgdlen = xrd_get_u32_be(body + PG_OFF_DLEN);
153 fr->foff = xrd_get_u64_be(body + PG_OFF_FILEOFF);
154
155 if (fr->resptype != PG_RESP_FINAL && fr->resptype != PG_RESP_PARTIAL) {
156 brix_cache_set_error(t, kXR_ServerError, 0,
157 "cache origin pgread returned an unknown resptype");
158 return -1;
159 }
160 return 0;
161}

163/* Read one frame of the kXR_status train.
164 * Returns 0 (fr valid), -1 (task error set), or BRIX_CACHE_PGREAD_UNSUPPORTED. */
165static int
166pg_read_frame(brix_cache_fill_t *t, brix_cache_origin_conn_t *oc,
167 pg_frame_t *fr)
168{
169 uint16_t status;
170 uint32_t dlen;
171 u_char *body = NULL;
172 int rc;
173
174 if (brix_cache_read_response(t, oc, &status, &body, &dlen,
175 PG_ERR_BODY_MAX) != 0) {
176 return -1;
177 }
178
179 if (status == kXR_error) {
180 rc = pg_error_is_unsupported(body, dlen)
181 ? BRIX_CACHE_PGREAD_UNSUPPORTED : -1;
182 if (rc == -1) {
183 brix_cache_set_origin_error(t, body, dlen,
184 "cache origin pgread failed");
185 }
186 free(body);
187 return rc;
188 }
189
190 if (status != kXR_status || dlen != PG_BODY_LEN) {
191 free(body);
192 brix_cache_set_error(t, kXR_ServerError, 0,
193 "cache origin pgread returned an invalid status frame");
194 return -1;
195 }
196
197 rc = pg_parse_body(t, body, fr);
198 free(body);
199 return rc;
200}

202/* Bounds-check one frame against the request before a byte is read:
203 * the pages must belong to the offset we are next expecting, and they must fit
204 * in what is left of the request. An origin that answers with a DIFFERENT
205 * file offset would otherwise scatter its bytes into parts of the object the
206 * caller never asked about — a write-anywhere primitive handed to the origin.
207 * Returns 0 / -1 (task error set). */
208static int
209pg_check_frame(brix_cache_fill_t *t, const brix_cache_read_range_t *rng,
210 const pg_frame_t *fr)
211{
212 if (fr->foff != rng->read_off + rng->got) {
213 brix_cache_set_error(t, kXR_ServerError, 0,
214 "cache origin pgread returned pages for an unrequested offset");
215 return -1;
216 }
217 if (fr->pgdlen > pg_max_pgdlen(rng->want - rng->got)) {
218 brix_cache_set_error(t, kXR_ServerError, 0,
219 "cache origin pgread returned too much data");
220 return -1;
221 }
222 /* A PARTIAL frame promises more to come, so an empty one advances nothing:
223 * an origin could hold a fill thread on this socket forever by sending them.
224 * The train has to make progress or end. */
225 if (fr->pgdlen == 0 && fr->resptype == PG_RESP_PARTIAL) {
226 brix_cache_set_error(t, kXR_ServerError, 0,
227 "cache origin pgread sent an empty partial frame");
228 return -1;
229 }
230 return 0;
231}

233/* Decode one frame's page buffer, verifying every page's CRC32c. Sets *n_out to
234 * the decoded byte count. Returns 0 / -1 (task error set: kXR_ChkSumErr naming
235 * the offending FILE offset, or kXR_ServerError for malformed framing). */
236static int
237pg_decode_verify(brix_cache_fill_t *t, const pg_frame_t *fr,
238 const u_char *pg, u_char *dst, ssize_t *n_out)
239{
240 int64_t bad = 0;
241 ssize_t n;
242 char msg[128];
243
244 n = xrdp_pg_decode(pg, fr->pgdlen, (int64_t) fr->foff, dst,
245 fr->pgdlen, &bad);
246 if (n == -1) {
247 ngx_snprintf((u_char *) msg, sizeof(msg),
248 "cache origin pgread CRC mismatch at file offset %L%Z", bad);
249 brix_cache_set_error(t, kXR_ChkSumErr, 0, msg);
250 return -1;
251 }
252 if (n < 0) {
253 brix_cache_set_error(t, kXR_ServerError, 0,
254 "cache origin pgread returned malformed page framing");
255 return -1;
256 }
257 *n_out = n;
258 return 0;
259}

261/* Write n verified bytes into the sink at the range's next write position.
262 * Returns 0 / -1 (task error set). */
263static int
264pg_commit(brix_cache_fill_t *t, brix_cache_sink_t *sink,
265 brix_cache_read_range_t *rng, const u_char *data, size_t n)
266{
267 if (n > rng->want - rng->got) {
268 brix_cache_set_error(t, kXR_ServerError, 0,
269 "cache origin pgread returned too much data");
270 return -1;
271 }
272 if (n == 0) {
273 return 0;
274 }
275 if (brix_cache_sink_pwrite(sink, data, n,
276 (off_t) (rng->dst_off + rng->got)) != 0)
277 {
278 brix_cache_set_syserror(t, kXR_IOError, "cache file write failed");
279 return -1;
280 }
281 rng->got += n;
282 return 0;
283}

285/* Receive, verify and write one frame's page data. Returns 0 / -1 (error set).
286 * The two buffers are allocated and freed on ONE straight line here so the
287 * verify/commit steps above can stay pure decisions. */
288static int
289pg_consume_pages(brix_cache_fill_t *t, brix_cache_origin_conn_t *oc,
290 brix_cache_sink_t *sink, brix_cache_read_range_t *rng,
291 const pg_frame_t *fr)
292{
293 u_char *pg, *dst;
294 ssize_t n = 0;
295 int rc = -1;
296
297 pg = malloc(fr->pgdlen);
298 dst = malloc(fr->pgdlen); /* decoded is always < encoded */
299 if (pg == NULL || dst == NULL) {
300 free(pg);
301 free(dst);
302 brix_cache_set_error(t, kXR_NoMemory, 0,
303 "cache origin pgread allocation failed");
304 return -1;
305 }
306
307 if (brix_cache_io_recv_exact(oc, pg, fr->pgdlen) != 0) {
308 brix_cache_set_error(t, kXR_ServerError, errno,
309 "cache origin pgread page read failed");
310 } else if (pg_decode_verify(t, fr, pg, dst, &n) == 0) {
311 rc = pg_commit(t, sink, rng, dst, (size_t) n);
312 }
313
314 free(pg);
315 free(dst);
316 return rc;
317}

319int
320brix_cache_origin_pgread_chunk(brix_cache_fill_t *t,
321 brix_cache_origin_conn_t *oc, const u_char fhandle[XRD_FHANDLE_LEN],
322 brix_cache_sink_t *sink, brix_cache_read_range_t *rng)
323{
324 pg_frame_t fr;
325 int rc;
326
327 rng->got = 0;
328
329 /* The advert decides before the wire does: an origin that never claimed
330 * kXR_suppgrw is not asked, so a pre-5.x federation costs one bootstrap
331 * flag test per read instead of a refused round trip per chunk. */
332 if (!(oc->srv_flags & kXR_suppgrw)) {
333 return BRIX_CACHE_PGREAD_UNSUPPORTED;
334 }
335
336 if (pg_send_request(t, oc, fhandle, rng) != 0) {
337 return -1;
338 }
339
340 for ( ;; ) {
341 rc = pg_read_frame(t, oc, &fr);
342 if (rc != 0) {
343 /* An UNSUPPORTED verdict is only honest while nothing has been
344 * written: past the first frame the caller cannot re-read the
345 * range with kXR_read without double-writing, so a late refusal
346 * is a protocol error, not a capability answer. */
347 if (rc == BRIX_CACHE_PGREAD_UNSUPPORTED && rng->got > 0) {
348 brix_cache_set_error(t, kXR_ServerError, 0,
349 "cache origin abandoned pgread mid-train");
350 return -1;
351 }
352 return rc;
353 }
354
355 if (pg_check_frame(t, rng, &fr) != 0) {
356 return -1;
357 }
358
359 if (fr.pgdlen > 0 && pg_consume_pages(t, oc, sink, rng, &fr) != 0) {
360 return -1;
361 }
362
363 if (fr.resptype == PG_RESP_FINAL) {
364 return 0;
365 }
366 }
