12/*
13 * ftp_ev_cmd.c — namespace and metadata verbs that complete on the control
14 * channel (no data connection): CWD, SIZE, MKD/DELE/RMD, MDTM/MLST/STAT/CKSM,
15 * and the RNFR/RNTO rename pair.
16 *
17 * WHAT: each handler resolves its path argument under the export, performs one
18 * VFS operation, and queues a single reply.
19 *
20 * WHY: these are pure request→reply verbs — no socket I/O beyond the reply — so
21 * they carry over from the sync engine essentially unchanged, now writing through
22 * the event reply queue instead of a blocking send. Keeping them out of the
23 * dispatcher keeps that file focused on routing.
24 *
25 * HOW: brix_ftp_ev_resolve() confines the path (INVARIANT 4), brix_ftp_ev_vfs_ctx()
26 * builds the identity-bearing context, and the brix_vfs_* seam performs the
27 * operation; write verbs gate on conf->allow_write first (INVARIANT 3-adjacent).
28 */

31/* Format a VFS entry's RFC 3659 facts into `out`; returns the byte count. */
32static size_t
33ev_fmt_facts(char *out, size_t outsz, const brix_vfs_stat_t *st)
34{
35 struct tm tm;
36 char mtime[32];
37 time_t t = st->mtime;
38
39 mtime[0] = '\0';
40 if (gmtime_r(&t, &tm) == NULL
41 || strftime(mtime, sizeof(mtime), "%Y%m%d%H%M%S", &tm) == 0)
42 {
43 /* strftime leaves the buffer contents indeterminate when it returns 0,
44 * so a year gmtime_r accepts but the format cannot render would put
45 * uninitialised bytes straight into an MLST fact line. Re-clamp to
46 * empty and emit "modify=" with no value rather than garbage. */
47 mtime[0] = '\0';
48 }
49 return (size_t) snprintf(out, outsz,
50 "type=%s;size=%lld;modify=%s;perm=%s;",
51 st->is_directory ? "dir" : "file",
52 (long long) st->size, mtime,
53 st->is_directory ? "el" : "r");
54}

133/*
134 * MKD, DELE and RMD are one transaction with three sets of words: refuse on a
135 * read-only export, resolve the argument inside it, run a single namespace op,
136 * report the result to the CNS manager, reply. Table-driven rather than written
137 * out three times (coding-standards §8) — only `op`, the CNS op code and the
138 * four replies differ.
139 *
140 * `ok` is applied to `arg` so MKD can echo the created name back per RFC 959
141 * §4.2; the two fixed 250s simply consume none of it. Every string below is a
142 * literal — `arg` is client-controlled and must never reach a format position.
143 */
144typedef struct {
145 ngx_int_t (*op)(brix_vfs_ctx_t *vctx);
146 uint32_t cns_op;
147 const char *denied; /* export is read-only */
148 const char *unresolved; /* argument would leave the export */
149 const char *op_failed; /* the namespace op itself refused */
150 const char *ok; /* success reply, formatted against `arg` */
151} ftp_ev_ns_verb_t;
152
153
154/* Thin adapters so the three ops share one pointer type; the flags are the
155 * verbs' historical semantics — MKD does not create parents (RFC 959 has no
156 * such verb) and RMD is non-recursive (that is what DELE-per-entry is for). */
157static ngx_int_t
158ftp_ev_ns_mkdir(brix_vfs_ctx_t *vctx)
159{
160 return brix_vfs_mkdir(vctx, 0755, 0 /* no parents */);
161}

171static ngx_int_t
172ftp_ev_ns_mutate(ftp_ev_t *fc, const char *arg, const ftp_ev_ns_verb_t *v)
173{
174 char abs[PATH_MAX];
175 brix_vfs_ctx_t vctx;
176
177 if (!fc->conf->allow_write) {
178 return brix_ftp_ev_reply(fc, "%s", v->denied);
179 }
180 if (brix_ftp_ev_resolve(fc, arg, abs, sizeof(abs)) != 0) {
181 return brix_ftp_ev_reply(fc, "%s", v->unresolved);
182 }
183 brix_ftp_ev_vfs_ctx(fc, abs, &vctx);
184 if (v->op(&vctx) != NGX_OK) {
185 return brix_ftp_ev_reply(fc, "%s", v->op_failed);
186 }
187
188 /* phase-97 §5: success path only — a refused mutation may neither seed a
189 * phantom entry nor retract a live one. */
190 brix_cns_emit_at(fc->conf->root_canon, v->cns_op, abs, 0, 0);
191 return brix_ftp_ev_reply(fc, v->ok, arg);
192}

317/* Map a GridFTP CKSM algorithm token to a BRIX_CK_* kind, or -1 if unknown. */
318static int
319ev_cksm_kind(const char *algo)
320{
321 if (strcasecmp(algo, "ADLER32") == 0) { return BRIX_CK_ADLER32; }
322 if (strcasecmp(algo, "CRC32") == 0) { return BRIX_CK_CRC32; }
323 if (strcasecmp(algo, "CRC32C") == 0) { return BRIX_CK_CRC32C; }
324 if (strcasecmp(algo, "MD5") == 0) { return BRIX_CK_MD5; }
325 if (strcasecmp(algo, "SHA1") == 0) { return BRIX_CK_SHA1; }
326 if (strcasecmp(algo, "SHA256") == 0) { return BRIX_CK_SHA256; }
327 return -1;
328}

331ngx_int_t
332brix_ftp_ev_cmd_cksm(ftp_ev_t *fc, const char *arg)
333{
334 char algo[32], abs[PATH_MAX];
335 long long offset, length;
336 int consumed = 0;
337 int kind;
338 off_t start, len;
339 const char *path;
340 brix_vfs_ctx_t vctx;
341 brix_vfs_file_t *fh;
342 brix_sd_obj_t obj;
343 int verr = 0;
344
345 if (sscanf(arg, "%31s %lld %lld %n", algo, &offset, &length, &consumed) < 3
346 || consumed == 0 || arg[consumed] == '\0')
347 {
348 return brix_ftp_ev_reply(fc,
349 "501 Usage: CKSM <algo> <offset> <length> <path>\r\n");
350 }
351 path = arg + consumed;
352
353 kind = ev_cksm_kind(algo);
354 if (kind < 0) {
355 return brix_ftp_ev_reply(fc, "504 Unsupported checksum algorithm %s\r\n",
356 algo);
357 }
358 if (offset < 0 || length < -1) {
359 return brix_ftp_ev_reply(fc, "501 Invalid CKSM range\r\n");
360 }
361 start = (off_t) offset;
362 len = (length < 0) ? -1 : (off_t) length;
363
364 if (brix_ftp_ev_resolve(fc, path, abs, sizeof(abs)) != 0) {
365 return brix_ftp_ev_reply(fc, "550 No such file\r\n");
366 }
367 brix_ftp_ev_vfs_ctx(fc, abs, &vctx);
368 fh = brix_vfs_open(&vctx, BRIX_VFS_O_READ, &verr);
369 if (fh == NULL) {
370 return brix_ftp_ev_reply(fc, "550 Cannot open file\r\n");
371 }
372 brix_vfs_file_sd_obj(fh, &obj);
373
374 if (kind == BRIX_CK_MD5 || kind == BRIX_CK_SHA1 || kind == BRIX_CK_SHA256) {
375 unsigned char raw[EVP_MAX_MD_SIZE];
376 u_char hex[2 * EVP_MAX_MD_SIZE + 1];
377 unsigned int rawlen = 0;
378 int ok = brix_cksum_digest_obj_range(kind, &obj, start, len,
379 raw, &rawlen);
380 brix_vfs_close(fh, fc->c->log);
381 if (ok != 0) {
382 return brix_ftp_ev_reply(fc, "550 Checksum failed\r\n");
383 }
384 ngx_hex_dump(hex, raw, rawlen);
385 hex[2 * rawlen] = '\0';
386 return brix_ftp_ev_reply(fc, "213 %s\r\n", hex);
387 } else {
388 uint32_t v = 0;
389 char hex[16];
390 int ok = brix_cksum_u32_obj_range(kind, &obj, start, len, &v);
391 brix_vfs_close(fh, fc->c->log);
392 if (ok != 0) {
393 return brix_ftp_ev_reply(fc, "550 Checksum failed\r\n");
394 }
395 snprintf(hex, sizeof(hex), "%08x", (unsigned) v);
396 return brix_ftp_ev_reply(fc, "213 %s\r\n", hex);
397 }
398}
