1/*
2 * credinfo_voms.c — VOMS attribute-certificate narration for `explain` (§15.2).
3 *
4 * WHAT: Renders the VOMS extension of a GSI proxy for the credential dump: one
5 * "VOMS:" line per FQAN, one summary line per VO (server URI, issuer,
6 * validity) and one verdict line per VO from the shared native verifier.
7 * WHY: credinfo.c used to byte-scan the DER for FQAN-looking strings — that
8 * could neither tell a forged AC from a genuine one nor say why a server
9 * would reject it. shared/voms/ is the single VOMS decoder/verifier for
10 * the nginx module and the native client, so the client's diagnostic now
11 * states exactly what the module will conclude about the same proxy.
12 * HOW: brix_voms_find_extension over the proxy leaf + chain, brix_voms_decode,
13 * then brix_voms_verify with a trust built from the environment
14 * (X509_CERT_DIR → X509_STORE, X509_VOMS_DIR → LSC directory). A missing
15 * trust directory downgrades to "unverified (no <dir>)" rather than a
16 * false failure. Fail-soft: every path prints a line and returns.
17 */

27#define VOMS_LINE " VOMS: "
28#define VOMS_SKEW_SECS 300 /* notBefore tolerance: the usual 5-minute clock allowance */
29#define DEFAULT_CERT_DIR "/etc/grid-security/certificates"
30#define DEFAULT_VOMS_DIR "/etc/grid-security/vomsdir"

32/* Trust for one explain run: a usable store + vomsdir, or the directory whose
33 * absence made verification impossible (then store is NULL). */
34typedef struct {
35 X509_STORE *store;
36 const char *vomsdir;
37 const char *missing_dir;
38} credinfo_voms_trust_t;

40/* ---- Environment directory with a grid-security default ----
41 *
42 * WHAT: returns getenv(var) when set and non-empty, else `fallback`.
43 * WHY: X509_CERT_DIR / X509_VOMS_DIR are the conventional overrides every
44 * grid tool honours; the defaults are the EGI/OSG install locations.
45 * HOW: 1. getenv; 2. empty counts as unset.
46 */
47static const char *
48voms_env_dir(const char *var, const char *fallback)
49{
50 const char *value = getenv(var);
51
52 return (value != NULL && value[0] != '\0') ? value : fallback;
53}

55/* ---- Is `path` an existing directory? ----
56 *
57 * WHAT: 1 when stat succeeds and the inode is a directory, else 0.
58 * WHY: a host without grid trust material must read "unverified", never
59 * "NOT verified" — the AC may be perfectly good.
60 * HOW: 1. stat; 2. S_ISDIR.
61 */
62static int
63voms_dir_exists(const char *path)
64{
65 struct stat st;
66
67 return stat(path, &st) == 0 && S_ISDIR(st.st_mode);
68}

70/* ---- Build the verification trust from the environment ----
71 *
72 * WHAT: fills *trust with an X509_STORE over X509_CERT_DIR and the
73 * X509_VOMS_DIR path, or records the first missing directory in
74 * trust->missing_dir (store NULL) so the caller reports "unverified".
75 * WHY: verification is only meaningful with both the CA bundle (signer chain)
76 * and the vomsdir (LSC binding); one without the other would produce a
77 * misleading verdict.
78 * HOW: 1. resolve both directories; 2. require each to exist; 3. X509_STORE_new
79 * + X509_STORE_load_locations(NULL, cert_dir); 4. a store failure is
80 * reported as the cert dir being unusable.
81 */
82static void
83voms_trust_open(credinfo_voms_trust_t *trust)
84{
85 const char *cert_dir = voms_env_dir("X509_CERT_DIR", DEFAULT_CERT_DIR);
86
87 memset(trust, 0, sizeof(*trust));
88 trust->vomsdir = voms_env_dir("X509_VOMS_DIR", DEFAULT_VOMS_DIR);
89 if (!voms_dir_exists(cert_dir)) {
90 trust->missing_dir = cert_dir;
91 return;
92 }
93 if (!voms_dir_exists(trust->vomsdir)) {
94 trust->missing_dir = trust->vomsdir;
95 return;
96 }
97 trust->store = X509_STORE_new();
98 if (trust->store == NULL
99 || !X509_STORE_load_locations(trust->store, NULL, cert_dir)) {
100 X509_STORE_free(trust->store);
101 trust->store = NULL;
102 trust->missing_dir = cert_dir;
103 }
104}

106/* ---- Release the trust store ----
107 *
108 * WHAT: frees the X509_STORE (NULL-safe).
109 * WHY: the store is the only owned resource of a trust; one release point.
110 * HOW: 1. X509_STORE_free.
111 */
112static void
113voms_trust_close(credinfo_voms_trust_t *trust)
114{
115 X509_STORE_free(trust->store);
116 trust->store = NULL;
117}

119/* ---- Format an epoch as ISO-8601 UTC ----
120 *
121 * WHAT: writes "YYYY-MM-DDTHH:MM:SSZ" into buf; "?" when the time is not
122 * representable.
123 * WHY: AC validity is what a server compares against its own clock; a fixed,
124 * zone-free rendering is comparable across hosts and in tests.
125 * HOW: 1. gmtime_r; 2. strftime.
126 */
127static void
128voms_fmt_time(time_t when, char *buf, size_t cap)
129{
130 struct tm utc;
131
132 if (gmtime_r(&when, &utc) == NULL
133 || strftime(buf, cap, "%Y-%m-%dT%H:%M:%SZ", &utc) == 0) {
134 snprintf(buf, cap, "?");
135 }
136}

138/* ---- Print one VO's decoded facts ----
139 *
140 * WHAT: one "VOMS: <fqan>" line per FQAN, then the VO summary line
141 * (vo, server URI, issuer DN, validity window).
142 * WHY: the FQAN lines keep the prefix the existing diagnostics and tests
143 * match on; the summary is what an operator needs to check vomsdir.
144 * HOW: 1. loop fqans; 2. format both validity bounds; 3. summary line.
145 */
146static void
147voms_print_entry(const brix_voms_entry_t *entry, FILE *out)
148{
149 char not_before[32], not_after[32];
150 int i;
151
152 for (i = 0; i < entry->nfqans; i++) {
153 fprintf(out, VOMS_LINE "%s\n", entry->fqans[i]);
154 }
155 voms_fmt_time(entry->not_before, not_before, sizeof(not_before));
156 voms_fmt_time(entry->not_after, not_after, sizeof(not_after));
157 fprintf(out, VOMS_LINE "vo=%s server=%s issuer=%s valid %s..%s\n",
158 entry->vo, entry->uri, entry->issuer_dn, not_before, not_after);
159}

161/* ---- Print one VO's verification verdict ----
162 *
163 * WHAT: "VOMS: verified", "VOMS: NOT verified: <reason>" or, when no trust
164 * directory was available, "VOMS: unverified (no <dir>)".
165 * WHY: three distinct words for three distinct situations — a verified AC, a
166 * rejected AC, and a host that cannot judge — so nobody reads an absent
167 * CA bundle as a broken proxy.
168 * HOW: 1. missing_dir wins; 2. else map the entry's verdict.
169 */
170static void
171voms_print_verdict(const brix_voms_entry_t *entry,
172 const credinfo_voms_trust_t *trust, FILE *out)
173{
174 if (trust->missing_dir != NULL) {
175 fprintf(out, VOMS_LINE "unverified (no %s)\n", trust->missing_dir);
176 return;
177 }
178 if (entry->verdict == BRIX_VOMS_OK) {
179 fprintf(out, VOMS_LINE "verified\n");
180 return;
181 }
182 fprintf(out, VOMS_LINE "NOT verified: %s\n", brix_voms_strerror(entry->verdict));
183}

185/* ---- Verify (when trust exists) and print every decoded VO ----
186 *
187 * WHAT: runs brix_voms_verify over the result under `trust` when a store is
188 * available, then prints each entry's facts and verdict.
189 * WHY: verification is one call over all entries (each gets its own verdict);
190 * printing per entry afterwards keeps facts and verdict adjacent.
191 * HOW: 1. build brix_voms_trust_t {store, vomsdir, now, skew}; 2. verify
192 * when store != NULL (the return value is per-entry in verdict, so it is
193 * not needed here); 3. print entry + verdict for each VO.
194 */
195static void
196voms_verify_and_print(brix_voms_result_t *res, X509 *holder, STACK_OF(X509) *chain,
197 const credinfo_voms_trust_t *trust, FILE *out)
198{
199 brix_voms_trust_t lib_trust = { trust->store, trust->vomsdir, 0, VOMS_SKEW_SECS };
200 int i;
201
202 if (trust->store != NULL) {
203 (void) brix_voms_verify(res, holder, chain, &lib_trust);
204 }
205 for (i = 0; i < res->n; i++) {
206 voms_print_entry(&res->entries[i], out);
207 voms_print_verdict(&res->entries[i], trust, out);
208 }
209}

211/* ---- Narrate the VOMS extension of a proxy ----
212 *
213 * WHAT: prints "VOMS: none" when neither leaf nor chain carries the
214 * extension, "VOMS: present (undecodable: <reason>)" when it will not
215 * decode, else the decoded FQANs, summary and verdict per VO.
216 * WHY: the entry point brix_gsi_cert_explain hands the proxy to; it owns the
217 * decode result and the trust for the duration of one print.
218 * HOW: 1. brix_voms_find_extension; 2. brix_voms_decode; 3. voms_trust_open;
219 * 4. voms_verify_and_print; 5. release trust and result.
220 */
221void
222brix_credinfo_voms_explain(X509 *leaf, STACK_OF(X509) *chain, FILE *out)
223{
224 X509 *holder = NULL;
225 const unsigned char *der = NULL;
226 int len = 0;
227 brix_voms_result_t *res = NULL;
228 brix_voms_status_t rc;
229 credinfo_voms_trust_t trust;
230
231 if (brix_voms_find_extension(leaf, chain, &holder, &der, &len) != BRIX_VOMS_OK) {
232 fprintf(out, VOMS_LINE "none\n");
233 return;
234 }
235 rc = brix_voms_decode(der, len, &res);
236 if (rc != BRIX_VOMS_OK) {
237 fprintf(out, VOMS_LINE "present (undecodable: %s)\n", brix_voms_strerror(rc));
238 return;
239 }
240 voms_trust_open(&trust);
241 voms_verify_and_print(res, holder, chain, &trust, out);
242 voms_trust_close(&trust);
243 brix_voms_result_free(res);
244}
