8/*
9 * ftp_ev_dispatch.c — the command dispatcher and the small stateless verbs.
10 *
11 * WHAT: brix_ftp_ev_dispatch() frames a single command line into verb+arg,
12 * enforces the pre-auth gate, and routes to the right handler; the transfer-
13 * parameter and session verbs whose entire implementation is a single reply
14 * (TYPE/MODE/STRU/ALLO/PWD/CDUP/NOOP/SYST/OPTS/REST/ABOR/SITE/PBSZ/PROT/DCAU/
15 * USER/PASS/QUIT/FEAT) are handled inline here; the heavier namespace, security,
16 * and transfer verbs delegate to their own files.
17 *
18 * WHY: this mirrors the sync engine's ftp_dispatch() one-to-one so the two engines
19 * accept exactly the same grammar and emit the same reply codes — the invariant
20 * the transition depends on. Keeping the one-line verbs here (rather than one
21 * trivial function each) keeps the dispatcher readable without inflating the file
22 * count with near-empty handlers.
23 *
24 * HOW: the verbs are split into per-section group routers (ev_grp_*), each a thin
25 * strcasecmp chain that returns the handler's result for a verb it owns or
26 * NGX_DECLINED for one it does not; brix_ftp_ev_dispatch() tries them in grammar
27 * order and falls through to "500 Unknown command". Only the genuinely branchy
28 * inline verbs (PROT/DCAU/OPTS/MODE/ALLO/REST) get their own static helper so no
29 * single function carries the whole grammar's branch weight. Each handler queues
30 * its reply via brix_ftp_ev_reply() and returns NGX_OK to continue, NGX_DONE on
31 * QUIT (flush then close), or NGX_ERROR on a fatal reply failure. The dispatcher
32 * never performs I/O itself — the engine flushes.
33 */

36/* Verbs accepted before the client has authenticated: the login verbs, the RFC

143/* ALLO <bytes> [R <recsize>] (RFC 959): the leading integer is the size of the
144 * file about to be stored. Record it (one-shot, consumed by the next STOR) so
145 * brix_gridftp_require_allo_size can hold the transfer to exactly that many
146 * bytes — the only completeness signal a stream-mode STOR has, since a bare
147 * connection close is indistinguishable from a mid-flight truncation. A
148 * malformed/negative size leaves it unset (the ACK stays lenient, matching
149 * historical behaviour). */
150static ngx_int_t
151ev_cmd_allo(ftp_ev_t *fc, const char *arg)
152{
153 char *endp = NULL;
154 long long sz = strtoll(arg, &endp, 10);
155
156 fc->allo_size = (arg[0] != '\0' && endp != arg && sz >= 0)
157 ? (off_t) sz : -1;
158 return brix_ftp_ev_reply(fc, "200 ALLO ok\r\n");
159}

255/*
256 * ev_site_mutates — does this SITE subcommand ask to change the export?
257 *
258 * WHAT: Non-zero for the five SITE subcommands the GridFTP grammar defines as
259 * mutations — CHMOD, CHGRP, UTIME, SYMLINK and RDEL (recursive delete).
260 *
261 * WHY: Phase 105 W4 — SITE is answered here with a blanket "200 OK", so on a
262 * read-only export `SITE CHMOD 777 /file` reported success for a mode change
263 * that no code in this tree performs. A client cannot tell that apart from a
264 * chmod that worked, which is the one thing a read-only endpoint must never
265 * say. These five now get the same 550 every other write verb answers with
266 * (ftp_ev_cmd.c), leaving every other SITE subcommand exactly as it was.
267 */
268static int
269ev_site_mutates(const char *arg)
270{
271 static const char *const mutating[] = {
272 "CHMOD", "CHGRP", "UTIME", "SYMLINK", "RDEL"
273 };
274 size_t i;
275
276 for (i = 0; i < sizeof(mutating) / sizeof(mutating[0]); i++) {
277 size_t n = strlen(mutating[i]);
278
279 if (strncasecmp(arg, mutating[i], n) == 0
280 && (arg[n] == '\0' || arg[n] == ' '))
281 {
282 return 1;
283 }
284 }
285 return 0;
286}

427/* Handle one framed command line. */
428ngx_int_t
429brix_ftp_ev_dispatch(ftp_ev_t *fc, char *line)
430{
431 char *verb;
432 char *arg = brix_ftp_ev_split(line, &verb);
433 ngx_int_t rc;
434
435 /* Pre-auth gate: reject file/namespace/transfer verbs until login. */
436 if (!fc->authed && !ev_verb_preauth(verb)) {
437 return brix_ftp_ev_reply(fc, "530 Please login with USER and PASS\r\n");
438 }
439
440 if ((rc = ev_grp_security(fc, verb, arg)) != NGX_DECLINED) { return rc; }
441 if ((rc = ev_grp_login(fc, verb, arg)) != NGX_DECLINED) { return rc; }
442 if ((rc = ev_grp_dcprot(fc, verb, arg)) != NGX_DECLINED) { return rc; }
443 if ((rc = ev_grp_session(fc, verb, arg)) != NGX_DECLINED) { return rc; }
444 if ((rc = ev_grp_xferparam(fc, verb, arg)) != NGX_DECLINED) { return rc; }
445 if ((rc = ev_grp_navigation(fc, verb, arg)) != NGX_DECLINED) { return rc; }
446 if ((rc = ev_grp_namespace(fc, verb, arg)) != NGX_DECLINED) { return rc; }
447 if ((rc = ev_grp_transfer(fc, verb, arg)) != NGX_DECLINED) { return rc; }
448 if ((rc = ev_grp_ctl(fc, verb)) != NGX_DECLINED) { return rc; }
449
450 return brix_ftp_ev_reply(fc, "500 Unknown command\r\n");
451}
