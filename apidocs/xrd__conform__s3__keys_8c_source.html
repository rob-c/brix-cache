6/* WHAT: Encode only the S3 URI while preserving an already-canonical query.
7 * WHY: SigV4 hashes the encoded URI, and raw spaces must never enter HTTP/1.1.
8 * HOW: RFC-3986 encode path bytes with slash safe, then append the query. */
9int
10xrd_conform_s3_wire_path(const brix_weburl *url, char *out, size_t out_size)
11{
12 const char *query = strchr(url->path, '?');
13 size_t uri_length = query == NULL ? strlen(url->path)
14 : (size_t) (query - url->path);
15 ssize_t encoded;
16
17 encoded = brix_http_urlencode((const unsigned char *) url->path,
18 uri_length, out, out_size, "/");
19 if (encoded < 0) return -1;
20 if (query == NULL) return 0;
21 if ((size_t) encoded + strlen(query) >= out_size) return -1;
22 memcpy(out + encoded, query, strlen(query) + 1);
23 return 0;
24}

135/* ---- Compare a ListObjectsV2 page with the exact unusual-key set ----
136 *
137 * WHAT: Returns one only when all four expected keys occur exactly once and no
138 * other key is present; malformed XML fragments return zero.
139 * WHY: Membership-only checks can accept duplicated, rewritten or injected
140 * names and would not establish a lossless S3 listing round trip.
141 * HOW: Walk bounded Key elements, map each value to an expected object, reject
142 * unknown/duplicate entries, then require all four bits and four entries.
143 */
144static int
145list_has_special(xrd_conform_execution *execution, const brix_weburl *base,
146 brix_weburl objects[4], const char *prefix)
147{
148 brix_weburl listing;
149 brix_http_resp response;
150 brix_status status;
151 const char *cursor;
152 unsigned seen = 0;
153 size_t found = 0;
154 int rc;
155
156 if (listing_url(base, prefix, &listing) != 0) return 0;
157 memset(&response, 0, sizeof(response));
158 brix_status_clear(&status);
159 rc = xrd_conform_http_request_fields(execution, &listing, "GET", NULL,
160 NULL, 32768, &response, &status);
161 if (rc != 0 || response.status != 200 || response.body == NULL) {
162 if (rc == 0) brix_http_resp_free(&response);
163 return 0;
164 }
165 cursor = response.body;
166 while ((cursor = strstr(cursor, "<Key>")) != NULL) {
167 const char *value = cursor + strlen("<Key>");
168 const char *end = strstr(value, "</Key>");
169 int index;
170
171 if (end == NULL) {
172 brix_http_resp_free(&response);
173 return 0;
174 }
175 index = key_index(objects, value, (size_t) (end - value));
176 if (index < 0 || (seen & (1U << (unsigned) index)) != 0) {
177 brix_http_resp_free(&response);
178 return 0;
179 }
180 seen |= 1U << (unsigned) index;
181 found++;
182 cursor = end + strlen("</Key>");
183 }
184 brix_http_resp_free(&response);
185 return found == 4 && seen == 0x0fU;
186}
