12/*
13 * brix_request_identity — the verified identity of this request, or NULL.
14 *
15 * There is no shared per-request identity record (phase-106 W1 as-built note):
16 * each protocol keeps its own request ctx, so this probes them the same way
17 * $brix_protocol probes loc confs. cvmfs/oci/rpm carry no brix_identity_t —
18 * their planes are anonymous-or-token-subject-only — and report NULL here.
19 */
20brix_identity_t *
21brix_request_identity(ngx_http_request_t *r)
22{
23 ngx_http_brix_webdav_req_ctx_t *wctx;
24 ngx_http_s3_req_ctx_t *sctx;
25
26 wctx = ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
27 if (wctx != NULL && wctx->identity != NULL) {
28 return wctx->identity;
29 }
30 sctx = ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
31 if (sctx != NULL && sctx->identity != NULL) {
32 return sctx->identity;
33 }
34 return NULL;
35}

38/*
39 * brix_var_set_str — publish an ngx_str_t owned by the request pool.
40 *
41 * No copy: identity strings are pool-allocated for the request and the pool
42 * outlives the log phase. An empty value reports the sentinel so "no VO" and
43 * "VO named the empty string" cannot be conflated in a log field.
44 */
45static ngx_int_t
46brix_var_set_str(ngx_http_variable_value_t *v, const ngx_str_t *val)
47{
48 if (val == NULL || val->len == 0) {
49 return brix_var_set_static(v, "-", 0);
50 }
51 v->len = (unsigned) val->len;
52 v->valid = 1;
53 v->no_cacheable = 0;
54 v->not_found = 0;
55 v->data = val->data;
56 return NGX_OK;
57}

65static const ngx_str_t *
66brix_identity_field(brix_identity_t *id, brix_http_idvar_e which)
67{
68 ngx_str_t *first;
69
70 switch (which) {
71 case BRIX_HV_DN: return &id->dn;
72 case BRIX_HV_VO: return &id->vo_csv;
73 case BRIX_HV_SUB: return &id->subject;
74 case BRIX_HV_ISSUER: return &id->issuer;
75 case BRIX_HV_FQAN:
76 /* The PRIMARY FQAN — the first entry of the verified list, matching
77 * the VOMS convention that the first FQAN is the operative one. The
78 * full list is $brix_vo. */
79 if (id->vo_list == NULL || id->vo_list->nelts == 0) {
80 return NULL;
81 }
82 first = id->vo_list->elts;
83 return &first[0];
84 }
85 return NULL;
86}

146/* $brix_tier — the storage-driver family serving this location ("posix",
147 * "s3", "http", "xroot", ...): the resolved instance's own name, so config
148 * sugar and the tier grammar cannot make the label drift from what runs. */
149ngx_int_t
150brix_var_tier(ngx_http_request_t *r, ngx_http_variable_value_t *v,
151 uintptr_t data)
152{
153 ngx_http_brix_shared_conf_t *common = brix_request_shared_conf(r);
154
155 (void) data;
156 if (common == NULL) {
157 return brix_var_set_static(v, "-", 0);
158 }
159 if (common->storage_instance != NULL) {
160 return brix_var_set_static(v,
161 brix_sd_backend_name(common->storage_instance), 0);
162 }
163 /* NULL instance = the default POSIX backend (vfs.h contract). */
164 return brix_var_set_static(v, "posix", 0);
165}

168/*
169 * $brix_origin — the configured origin, with any userinfo stripped.
170 *
171 * SECURITY: a remote storage_backend URL may carry user:pass@ userinfo, and a
172 * variable is loggable — so everything up to and including a '@' in the
173 * authority is removed before publishing. Never the raw config string.
174 */
175ngx_int_t
176brix_var_origin(ngx_http_request_t *r, ngx_http_variable_value_t *v,
177 uintptr_t data)
178{
179 ngx_http_brix_shared_conf_t *common = brix_request_shared_conf(r);
180 ngx_str_t shown;
181 u_char *at, *end, *slash;
182
183 (void) data;
184 if (common == NULL || common->storage_backend.len == 0) {
185 return brix_var_set_static(v, "-", 0);
186 }
187
188 shown = common->storage_backend;
189 end = shown.data + shown.len;
190 /* Userinfo can only appear before the first path slash after "//". */
191 slash = ngx_strlchr(shown.data, end, '/');
192 while (slash != NULL && slash + 1 < end && slash[1] == '/') {
193 slash = ngx_strlchr(slash + 2, end, '/');
194 break;
195 }
196 at = ngx_strlchr(shown.data, slash != NULL ? slash : end, '@');
197 if (at != NULL) {
198 shown.len -= (size_t) (at + 1 - shown.data);
199 shown.data = at + 1;
200 }
201 return brix_var_set_str(v, &shown);
202}
