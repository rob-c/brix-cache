1/*
2 * vfs_authz.c — authorization backstop shared by every VFS entry point.
3 *
4 * The protocol edge remains the primary authorization boundary and owns its
5 * wire response. This file independently re-evaluates the same rule engines
6 * after the read-only mutation policy and before locks or storage. OBSERVE is
7 * deliberately non-refusing; ENFORCE maps every undecidable/denied result to
8 * EACCES. Metrics use only the bounded protocol/result axes.
9 */

27void
28brix_vfs_ctx_bind_authz(brix_vfs_ctx_t *vctx,
29 ngx_array_t *authdb_rules, ngx_array_t *vo_rules,
30 void *acc_tables, ngx_uint_t acc_format,
31 const char *peer, brix_authz_backstop_mode_t mode)
32{
33 size_t peer_len;
34
35 if (vctx == NULL) {
36 return;
37 }
38
39 vctx->authz.authdb_rules = authdb_rules;
40 vctx->authz.vo_rules = vo_rules;
41 vctx->authz.acc_tables = acc_tables;
42 vctx->authz.acc_format = acc_format;
43 if (peer != NULL) {
44 peer_len = ngx_min(ngx_strlen(peer), sizeof(vctx->authz.peer) - 1);
45 ngx_memcpy(vctx->authz.peer, peer, peer_len);
46 vctx->authz.peer[peer_len] = '\0';
47 } else {
48 vctx->authz.peer[0] = '\0';
49 }
50 vctx->authz.mode = mode <= BRIX_AUTHZ_BACKSTOP_ENFORCE
51 ? mode : BRIX_AUTHZ_BACKSTOP_OBSERVE;
52 vctx->authz.acc_entity = acc_format == BRIX_AUTHDB_FORMAT_XRDACC
53 ? brix_authz_acc_entity(vctx->pool, vctx->identity,
54 vctx->authz.peer[0] != '\0' ? vctx->authz.peer : NULL)
55 : NULL;
56 vctx->authz.bound = 1;
57}

67uint32_t
68brix_vfs_authz_level_for_op(brix_vfs_mutation_op_t op, int *also_delete)
69{
70 if (also_delete != NULL) {
71 *also_delete = 0;
72 }
73 if ((ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT) {
74 return 0;
75 }
76
77 if (op <= BRIX_VFS_MUTATE_SYNC
78 || (op >= BRIX_VFS_MUTATE_SETATTR && op <= BRIX_VFS_MUTATE_LOCK))
79 {
80 return BRIX_AUTH_UPDATE;
81 }
82
83 switch (op) {
84 case BRIX_VFS_MUTATE_MKDIR:
85 return BRIX_AUTH_MKDIR;
86 case BRIX_VFS_MUTATE_REMOVE:
87 return BRIX_AUTH_DELETE;
88 case BRIX_VFS_MUTATE_RENAME:
89 case BRIX_VFS_MUTATE_COPY:
90 if (also_delete != NULL) {
91 *also_delete = 1;
92 }
93 return BRIX_AUTH_UPDATE;
94 default: return 0;
95 }
96}

108static brix_acc_op_t
109brix_vfs_authz_acc_op(brix_vfs_mutation_op_t op, int destination)
110{
111 if (op == BRIX_VFS_MUTATE_RENAME || op == BRIX_VFS_MUTATE_COPY) {
112 return brix_vfs_authz_two_name_acc(op, destination);
113 }
114 if ((op >= BRIX_VFS_MUTATE_WRITE && op <= BRIX_VFS_MUTATE_SYNC)
115 || (op >= BRIX_VFS_MUTATE_SETATTR && op <= BRIX_VFS_MUTATE_PUBLISH))
116 {
117 return BRIX_AOP_UPDATE;
118 }
119 if (op >= BRIX_VFS_MUTATE_STAGE && op <= BRIX_VFS_MUTATE_EVICT) {
120 return BRIX_AOP_STAGE;
121 }
122 switch (op) {
123 case BRIX_VFS_MUTATE_MKDIR: return BRIX_AOP_MKDIR;
124 case BRIX_VFS_MUTATE_REMOVE: return BRIX_AOP_DELETE;
125 case BRIX_VFS_MUTATE_LOCK: return BRIX_AOP_LOCK;
126 case BRIX_VFS_MUTATE_OPEN: return BRIX_AOP_UPDATE;
127 default: return BRIX_AOP_ANY;
128 }
129}

131static int
132brix_vfs_authz_has_rules(const brix_vfs_ctx_t *ctx)
133{
134 const brix_vfs_authz_t *a = &ctx->authz;
135
136 return (a->authdb_rules != NULL && a->authdb_rules->nelts != 0)
137 || (a->vo_rules != NULL && a->vo_rules->nelts != 0)
138 || (a->acc_format == BRIX_AUTHDB_FORMAT_XRDACC
139 && a->acc_tables != NULL
140 && ((brix_acc_tables_t *) a->acc_tables)->rule_count != 0)
141 || (ctx->identity != NULL
142 && (ctx->identity->auth_method & BRIX_AUTHN_TOKEN));
143}

145static ngx_int_t
146brix_vfs_authz_outcome(const brix_vfs_ctx_t *ctx,
147 brix_authz_backstop_result_t result, const char *operation)
148{
149 brix_metric_vfs_authz_backstop(brix_vfs_metrics_proto(ctx), result);
150
151 if (result == BRIX_AUTHZ_BACKSTOP_AGREE
152 || result == BRIX_AUTHZ_BACKSTOP_NO_RULES)
153 {
154 return NGX_OK;
155 }
156
157 if (ctx->log != NULL) {
158 ngx_log_error(NGX_LOG_WARN, ctx->log, 0,
159 "brix: VFS authorization backstop result=%s op=%s",
160 brix_metric_vfs_authz_backstop_result_name(result),
161 operation != NULL ? operation : "read");
162 }
163 if (ctx->authz.mode != BRIX_AUTHZ_BACKSTOP_ENFORCE) {
164 return NGX_OK;
165 }
166
167 brix_io_monitor_record_err(ctx->io_monitor, BRIX_ERR_FORBIDDEN);
168 errno = EACCES;
169 return NGX_ERROR;
170}

172static brix_vfs_authz_need_t
173brix_vfs_authz_need(brix_vfs_mutation_op_t op, int destination)
174{
175 brix_vfs_authz_need_t need;
176 int also_delete;
177
178 need.privilege = brix_vfs_authz_level_for_op(op, &also_delete);
179 if (also_delete && !destination) {
180 need.privilege = BRIX_AUTH_DELETE;
181 }
182 need.acc_op = brix_vfs_authz_acc_op(op, destination);
183 return need;
184}

204static ngx_int_t
205brix_vfs_require_authorized_at(const brix_vfs_ctx_t *ctx, const char *path,
206 brix_vfs_mutation_op_t op, int destination)
207{
208 brix_authz_identity_query_t query;
209 brix_vfs_authz_need_t need;
210
211 if (ctx == NULL || path == NULL || path[0] == '\0') {
212 errno = EINVAL;
213 return NGX_ERROR;
214 }
215 if (!ctx->authz.bound) {
216 return brix_vfs_authz_outcome(ctx, BRIX_AUTHZ_BACKSTOP_UNBOUND,
217 brix_vfs_mutation_op_name(op));
218 }
219 if (ctx->authz.mode == BRIX_AUTHZ_BACKSTOP_OFF) {
220 return NGX_OK;
221 }
222 if (!brix_vfs_authz_has_rules(ctx)) {
223 return brix_vfs_authz_outcome(ctx, BRIX_AUTHZ_BACKSTOP_NO_RULES,
224 brix_vfs_mutation_op_name(op));
225 }
226
227 need = brix_vfs_authz_need(op, destination);
228 if (need.privilege == 0 || need.acc_op == BRIX_AOP_ANY) {
229 return brix_vfs_authz_outcome(ctx, BRIX_AUTHZ_BACKSTOP_EDGE_MISSING,
230 brix_vfs_mutation_op_name(op));
231 }
232 brix_vfs_authz_query_init(ctx, path, &query);
233 query.needed_privs = need.privilege;
234 query.acc_op = need.acc_op;
235 query.need_write = 1;
236
237 return brix_vfs_authz_outcome(ctx,
238 brix_authz_check_identity(&query) == NGX_OK
239 ? BRIX_AUTHZ_BACKSTOP_AGREE
240 : BRIX_AUTHZ_BACKSTOP_EDGE_MISSING,
241 brix_vfs_mutation_op_name(op));
242}
