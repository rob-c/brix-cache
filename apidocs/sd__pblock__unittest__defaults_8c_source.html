1/*
2 * sd_pblock_unittest_defaults.c — phase-88 W5: the standard-on features.
3 *
4 * A BARE export (no opts sidecar at all) now arms the always-safe integrity/
5 * performance features: F3 per-block CRC32c (csi) and the W4 shared namespace
6 * cache (nsidx). Everything else stays opt-in. Legs:
7 * SUCCESS/integrity — a bare export advertises CAP_FSCS, and a byte of
8 * on-disk rot is EIO at read time, never served.
9 * SUCCESS/perf — a bare export creates catalog.bxi, and a lookup is
10 * served from the shared table (proven: a raw-SQL
11 * mutation behind the API is not seen).
12 * ERROR/opt-out — csi=0&nsidx=0 restores the legacy behaviour exactly:
13 * no CAP_FSCS, rot is served silently, no catalog.bxi
14 * (the defaults are overridable, not hard-wired).

41/* def_raw_set_size — mutate a row BEHIND the catalog API so shared-cache
42 * serving becomes observable (the nsidx proof from the catalog unittest). */
43static void
44def_raw_set_size(const char *root, const char *path, long long size)
45{
46 char db[PATH_MAX];
47 sqlite3 *h = NULL;
48 sqlite3_stmt *q = NULL;
49
50 snprintf(db, sizeof(db), "%s/catalog.db", root);
51 CHECK(sqlite3_open(db, &h) == SQLITE_OK, "def raw db open");
52 if (sqlite3_prepare_v2(h,
53 "UPDATE objects SET size = ?2 WHERE path = ?1;", -1, &q, NULL)
54 == SQLITE_OK)
55 {
56 sqlite3_bind_text(q, 1, path, -1, SQLITE_STATIC);
57 sqlite3_bind_int64(q, 2, size);
58 CHECK(sqlite3_step(q) == SQLITE_DONE, "def raw update");
59 }
60 sqlite3_finalize(q);
61 sqlite3_close(h);
62}

64/* def_flip_block0 — flip one byte of a blob's block-0 file (on-disk rot). */
65static void
66def_flip_block0(const char *root, const char *blob, long off)
67{
68 char p[PATH_MAX];
69 int fd;
70 char byte;
71
72 snprintf(p, sizeof(p), "%s/data/%c%c/%c%c/%s/0", root,
73 blob[0], blob[1], blob[2], blob[3], blob);
74 fd = open(p, O_RDWR);
75 CHECK(fd >= 0, "open block 0 for rot (%s)", p);
76 CHECK(pread(fd, &byte, 1, off) == 1, "read rot target");
77 byte = (char) (byte ^ 0x20);
78 CHECK(pwrite(fd, &byte, 1, off) == 1, "write rot");
79 close(fd);
80}

82void
83test_standard_defaults(void)
84{
85 const char *DATA = "standard-defaults-payload-0123456789";
86 const size_t DLEN = strlen("standard-defaults-payload-0123456789");
87 char buf[128], blob[PBLOCK_BLOB_ID_CAP], bxi[PATH_MAX];
88 ssize_t n;
89
90 /* SUCCESS: a bare export (NO sidecar) arms csi + nsidx. */
91 {
92 char root[] = "/tmp/pb_defs.XXXXXX";
93 brix_sd_instance_t inst = {0};
94 brix_sd_pblock_conf_t conf = {0};
95
96 CHECK(mkdtemp(root) != NULL, "mkdtemp");
97 conf.root = root; /* deliberately no pblock.opts */
98 conf.busy_timeout_ms = 2000;
99 conf.block_size = 4096;
100 inst.driver = D;
101 inst.caps = D->caps;
102 CHECK(D->init(&inst, &conf) == NGX_OK, "bare init");
103
104 CHECK((inst.caps & BRIX_SD_CAP_FSCS) != 0,
105 "bare export advertises CAP_FSCS (csi standard)");
106 snprintf(bxi, sizeof(bxi), "%s/catalog.bxi", root);
107 CHECK(access(bxi, F_OK) == 0,
108 "bare export created the shared namespace cache sidecar");
109
110 /* Integrity: on-disk rot is EIO at read, never served. */
111 CHECK(write_file(&inst, "/f", DATA, DLEN) == 0, "seed f");
112 def_blob_id(root, "/f", blob, sizeof(blob));
113 CHECK(blob[0] != '\0', "blob resolved");
114 def_flip_block0(root, blob, 5);
115 errno = 0;
116 n = read_file(&inst, "/f", buf, sizeof(buf));
117 CHECK(n < 0 && errno == EIO,
118 "rot must be EIO, got n=%zd errno=%d", n, errno);
119
120 /* Performance/coherence: a lookup is served from the shared table —
121 * a mutation smuggled past the API is (correctly) not seen. */
122 CHECK(write_file(&inst, "/g", DATA, DLEN) == 0, "seed g");
123 {
124 brix_sd_stat_t stx;
125
126 CHECK(D->stat(&inst, "/g", &stx) == NGX_OK
127 && stx.size == (off_t) DLEN, "stat fills the cache");
128 def_raw_set_size(root, "/g", 999);
129 CHECK(D->stat(&inst, "/g", &stx) == NGX_OK
130 && stx.size == (off_t) DLEN,
131 "lookup served from the shared cache (size %lld)",
132 (long long) stx.size);
133 }
134 D->cleanup(&inst);
135 }
136
137 /* ERROR/opt-out: csi=0&nsidx=0 restores the legacy behaviour exactly. */
138 {
139 char root[] = "/tmp/pb_defsoff.XXXXXX";
140 brix_sd_instance_t inst = {0};
141 brix_sd_pblock_conf_t conf = {0};
142
143 CHECK(mkdtemp(root) != NULL, "mkdtemp off");
144 lab_write_sidecar(root, "csi=0&nsidx=0");
145 conf.root = root;
146 conf.busy_timeout_ms = 2000;
147 conf.block_size = 4096;
148 inst.driver = D;
149 inst.caps = D->caps;
150 CHECK(D->init(&inst, &conf) == NGX_OK, "opt-out init");
151
152 CHECK((inst.caps & BRIX_SD_CAP_FSCS) == 0,
153 "csi=0 must not advertise CAP_FSCS");
154 snprintf(bxi, sizeof(bxi), "%s/catalog.bxi", root);
155 CHECK(access(bxi, F_OK) != 0,
156 "nsidx=0 must not create the shared cache sidecar");
157
158 CHECK(write_file(&inst, "/f", DATA, DLEN) == 0, "seed f (off)");
159 def_blob_id(root, "/f", blob, sizeof(blob));
160 def_flip_block0(root, blob, 5);
161 n = read_file(&inst, "/f", buf, sizeof(buf));
162 CHECK(n == (ssize_t) DLEN && memcmp(buf, DATA, DLEN) != 0,
163 "opt-out serves the (rotten) bytes unverified — the legacy "
164 "contract (n=%zd)", n);
165 D->cleanup(&inst);
166 }
167}
