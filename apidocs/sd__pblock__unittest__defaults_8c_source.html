1/*
2 * sd_pblock_unittest_defaults.c — phase-88 W5: the standard-on features.
3 *
4 * A BARE export (no opts sidecar at all) now arms the always-safe integrity/
5 * performance features: F3 per-block CRC32c (csi) and the W4 shared namespace
6 * cache (nsidx). Everything else stays opt-in. Legs:
7 * SUCCESS/integrity — a bare export advertises CAP_FSCS, and a byte of
8 * on-disk rot is EIO at read time, never served.
9 * SUCCESS/perf — a bare export creates catalog.bxi, and a lookup is
10 * served from the shared table (proven: a raw-SQL
11 * mutation behind the API is not seen).
12 * ERROR/opt-out — csi=0&nsidx=0 restores the legacy behaviour exactly:
13 * no CAP_FSCS, rot is served silently, no catalog.bxi
14 * (the defaults are overridable, not hard-wired).
15 */

42/* def_raw_set_size — mutate a row BEHIND the catalog API so shared-cache
43 * serving becomes observable (the nsidx proof from the catalog unittest). */
44static void
45def_raw_set_size(const char *root, const char *path, long long size)
46{
47 char db[PATH_MAX];
48 sqlite3 *h = NULL;
49 sqlite3_stmt *q = NULL;
50
51 snprintf(db, sizeof(db), "%s/catalog.db", root);
52 CHECK(sqlite3_open(db, &h) == SQLITE_OK, "def raw db open");
53 if (sqlite3_prepare_v2(h,
54 "UPDATE objects SET size = ?2 WHERE path = ?1;", -1, &q, NULL)
55 == SQLITE_OK)
56 {
57 sqlite3_bind_text(q, 1, path, -1, SQLITE_STATIC);
58 sqlite3_bind_int64(q, 2, size);
59 CHECK(sqlite3_step(q) == SQLITE_DONE, "def raw update");
60 }
61 sqlite3_finalize(q);
62 sqlite3_close(h);
63}

65/* def_flip_block0 — flip one byte of a blob's block-0 file (on-disk rot). */
66static void
67def_flip_block0(const char *root, const char *blob, long off)
68{
69 char p[PATH_MAX];
70 int fd;
71 char byte;
72
73 snprintf(p, sizeof(p), "%s/data/%c%c/%c%c/%s/0", root,
74 blob[0], blob[1], blob[2], blob[3], blob);
75 fd = open(p, O_RDWR);
76 CHECK(fd >= 0, "open block 0 for rot (%s)", p);
77 CHECK(pread(fd, &byte, 1, off) == 1, "read rot target");
78 byte = (char) (byte ^ 0x20);
79 CHECK(pwrite(fd, &byte, 1, off) == 1, "write rot");
80 close(fd);
81}

83/* Exercise the sidecar reader independently of driver startup: a valid line
84 * parses, an absent sidecar yields defaults, and an overlong first line is
85 * rejected whole rather than being applied as a truncated configuration. */
86static void
87def_test_sidecar_read_contract(void)
88{
89 char root[] = "/tmp/pb_sidecar.XXXXXX";
90 char path[PATH_MAX];
91 pblock_opts_t opts;
92 FILE *f;
93 size_t i;
94
95 CHECK(mkdtemp(root) != NULL, "sidecar mkdtemp");
96 CHECK(pblock_opts_load_sidecar(root, &opts) == 0,
97 "absent sidecar must yield defaults");
98 CHECK(opts.csi == -1 && opts.nsidx == -1,
99 "absent sidecar lost standard defaults");
100
101 lab_write_sidecar(root, "csi=0&nsidx=0");
102 CHECK(pblock_opts_load_sidecar(root, &opts) == 0,
103 "valid sidecar parse");
104 CHECK(opts.csi == 0 && opts.nsidx == 0,
105 "valid sidecar values not applied");
106
107 snprintf(path, sizeof(path), "%s/pblock.opts", root);
108 f = fopen(path, "w");
109 CHECK(f != NULL, "open overlong sidecar");
110 if (f != NULL) {
111 for (i = 0; i < 1100; i++) {
112 CHECK(fputc('x', f) != EOF, "write overlong sidecar");
113 }
114 CHECK(fclose(f) == 0, "close overlong sidecar");
115 errno = 0;
116 CHECK(pblock_opts_load_sidecar(root, &opts) == -1
117 && errno == EOVERFLOW,
118 "overlong sidecar must fail whole (errno=%d)", errno);
119 }
120}

122void
123test_standard_defaults(void)
124{
125 const char *DATA = "standard-defaults-payload-0123456789";
126 const size_t DLEN = strlen("standard-defaults-payload-0123456789");
127 char buf[128], blob[PBLOCK_BLOB_ID_CAP], bxi[PATH_MAX];
128 ssize_t n;
129
130 def_test_sidecar_read_contract();
131
132 /* SUCCESS: a bare export (NO sidecar) arms csi + nsidx. */
133 {
134 char root[] = "/tmp/pb_defs.XXXXXX";
135 brix_sd_instance_t inst = {0};
136 brix_sd_pblock_conf_t conf = {0};
137
138 CHECK(mkdtemp(root) != NULL, "mkdtemp");
139 conf.root = root; /* deliberately no pblock.opts */
140 conf.busy_timeout_ms = 2000;
141 conf.block_size = 4096;
142 inst.driver = D;
143 inst.caps = D->caps;
144 CHECK(D->init(&inst, &conf) == NGX_OK, "bare init");
145
146 CHECK((inst.caps & BRIX_SD_CAP_FSCS) != 0,
147 "bare export advertises CAP_FSCS (csi standard)");
148 snprintf(bxi, sizeof(bxi), "%s/catalog.bxi", root);
149 CHECK(access(bxi, F_OK) == 0,
150 "bare export created the shared namespace cache sidecar");
151
152 /* Integrity: on-disk rot is EIO at read, never served. */
153 CHECK(write_file(&inst, "/f", DATA, DLEN) == 0, "seed f");
154 def_blob_id(root, "/f", blob, sizeof(blob));
155 CHECK(blob[0] != '\0', "blob resolved");
156 def_flip_block0(root, blob, 5);
157 errno = 0;
158 n = read_file(&inst, "/f", buf, sizeof(buf));
159 CHECK(n < 0 && errno == EIO,
160 "rot must be EIO, got n=%zd errno=%d", n, errno);
161
162 /* Performance/coherence: a lookup is served from the shared table —
163 * a mutation smuggled past the API is (correctly) not seen. */
164 CHECK(write_file(&inst, "/g", DATA, DLEN) == 0, "seed g");
165 {
166 brix_sd_stat_t stx;
167
168 CHECK(D->stat(&inst, "/g", &stx) == NGX_OK
169 && stx.size == (off_t) DLEN, "stat fills the cache");
170 def_raw_set_size(root, "/g", 999);
171 CHECK(D->stat(&inst, "/g", &stx) == NGX_OK
172 && stx.size == (off_t) DLEN,
173 "lookup served from the shared cache (size %lld)",
174 (long long) stx.size);
175 }
176 D->cleanup(&inst);
177 }
178
179 /* ERROR/opt-out: csi=0&nsidx=0 restores the legacy behaviour exactly. */
180 {
181 char root[] = "/tmp/pb_defsoff.XXXXXX";
182 brix_sd_instance_t inst = {0};
183 brix_sd_pblock_conf_t conf = {0};
184
185 CHECK(mkdtemp(root) != NULL, "mkdtemp off");
186 lab_write_sidecar(root, "csi=0&nsidx=0");
187 conf.root = root;
188 conf.busy_timeout_ms = 2000;
189 conf.block_size = 4096;
190 inst.driver = D;
191 inst.caps = D->caps;
192 CHECK(D->init(&inst, &conf) == NGX_OK, "opt-out init");
193
194 CHECK((inst.caps & BRIX_SD_CAP_FSCS) == 0,
195 "csi=0 must not advertise CAP_FSCS");
196 snprintf(bxi, sizeof(bxi), "%s/catalog.bxi", root);
197 CHECK(access(bxi, F_OK) != 0,
198 "nsidx=0 must not create the shared cache sidecar");
199
200 CHECK(write_file(&inst, "/f", DATA, DLEN) == 0, "seed f (off)");
201 def_blob_id(root, "/f", blob, sizeof(blob));
202 def_flip_block0(root, blob, 5);
203 n = read_file(&inst, "/f", buf, sizeof(buf));
204 CHECK(n == (ssize_t) DLEN && memcmp(buf, DATA, DLEN) != 0,
205 "opt-out serves the (rotten) bytes unverified — the legacy "
206 "contract (n=%zd)", n);
207 D->cleanup(&inst);
208 }
209}
