14/* brix_fhandle_slot_live — 1 iff the slot is currently occupied by an open handle.
15 * A slot is live when it has ANY of: a kernel fd (fd >= 0), a driver-backed object
16 * (sd_obj.driver != NULL — remote/object backends are memory-served with fd < 0),
17 * or a whole-object staged writer (writer != NULL). This is the SAME liveness
18 * predicate brix_validate_file_handle uses; allocation and validation must agree,
19 * else a driver-backed handle (fd stays -1) reads as "free" and a second open on
20 * the same connection re-allocates its slot, collapsing distinct handles onto one
21 * index (breaks clients that hold multiple concurrent opens — e.g. uproot). */
22static ngx_inline int
23brix_fhandle_slot_live(const brix_ctx_t *ctx, int handle_index)
24{
25 return ctx->files[handle_index].fd >= 0
26 || ctx->files[handle_index].sd_obj.driver != NULL
27 || ctx->files[handle_index].writer != NULL;
28}

30/* brix_files_ensure — lazily allocate the fixed handle table (see fd_table.h).
31 * pcalloc zeroes the block, so only the two -1 sentinels need explicit init;
32 * once allocated the address never changes (AIO tasks hold brix_file_t
33 * pointers across worker threads, so a growable table would dangle them). */
34ngx_int_t
35brix_files_ensure(brix_ctx_t *ctx, ngx_connection_t *c)
36{
37 int i;
38
39 if (ctx->files != NULL) {
40 return NGX_OK;
41 }
42
43 ctx->files = ngx_pcalloc(c->pool, BRIX_MAX_FILES * sizeof(brix_file_t));
44 if (ctx->files == NULL) {
45 return NGX_ERROR;
46 }
47
48 for (i = 0; i < BRIX_MAX_FILES; i++) {
49 ctx->files[i].fd = -1; /* fd < 0 == slot free */
50 ctx->files[i].shared_handle_slot_hint = -1; /* Phase 33 C2: no cache yet */
51 }
52
53 return NGX_OK;
54}

56/* brix_alloc_fhandle — return the first free slot in ctx->files; the index becomes
57 * the one-byte on-wire fhandle, so it is bounded by BRIX_MAX_FILES. -1 when all
58 * slots are occupied. Single-owner per connection (event thread, no locking); a
59 * slot is only reused once its handle is fully torn down (brix_free_fhandle). */
60int
61brix_alloc_fhandle(brix_ctx_t *ctx)
62{
63 int handle_index;
64
65 if (brix_files_ensure(ctx, ctx->session->connection) != NGX_OK) {
66 return -1;
67 }
68
69 /*
70 * The XRootD wire handle stores this slot number in one byte. Keep all
71 * allocation/validation paths using the same bounded table.
72 */
73 for (handle_index = 0; handle_index < BRIX_MAX_FILES; handle_index++) {
74 if (!brix_fhandle_slot_live(ctx, handle_index)) {
75 return handle_index;
76 }
77 }
78
79 return -1;
80}

82/* brix_ctx_has_open_file — 1 if any handle slot is occupied (fd >= 0), else 0.
83 * Used by the recv-loop drain gate: a connection with an open file is mid-transfer
84 * (a streaming read parked between kXR_read chunks), so a draining worker must let
85 * it finish rather than fast-teardown at the request boundary — a forced mid-stream
86 * reconnect loses the in-flight fill. Single-owner per connection (event thread). */
87int
88brix_ctx_has_open_file(const brix_ctx_t *ctx)
89{
90 int handle_index;
91
92 if (ctx->files == NULL) {
93 return 0;
94 }
95
96 for (handle_index = 0; handle_index < BRIX_MAX_FILES; handle_index++) {
97 if (brix_fhandle_slot_live(ctx, handle_index)) {
98 return 1;
99 }
100 }
101
102 return 0;
103}

105/* brix_set_fhandle_path — store a heap copy (ngx_alloc, NOT pool, so it outlives
106 * the kXR_open request) of the canonical path in the slot, freeing any prior path
107 * first; brix_free_fhandle owns the free. NGX_OK, or NGX_ERROR on bad bounds or
108 * allocation failure. Heap (not pool) avoids fragmentation across open/close cycles. */
109ngx_int_t
110brix_set_fhandle_path(brix_ctx_t *ctx, ngx_connection_t *c,
111 int handle_index, const char *path)
112{
113 char *path_copy;
114 size_t path_bytes;
115
116 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES
117 || path == NULL || ctx->files == NULL)
118 {
119 return NGX_ERROR;
120 }
121
122 path_bytes = ngx_strlen(path) + 1;
123 path_copy = ngx_alloc(path_bytes, c->log);
124 if (path_copy == NULL) {
125 return NGX_ERROR;
126 }
127
128 ngx_memcpy(path_copy, path, path_bytes);
129
130 if (ctx->files[handle_index].path != NULL) {
131 ngx_free(ctx->files[handle_index].path);
132 }
133
134 ctx->files[handle_index].path = path_copy;
135 return NGX_OK;
136}

146/* brix_validate_file_handle — 1 iff handle_index is in bounds and the slot has an
147 * active fd; otherwise logs + sends kXR_FileNotOpen and returns 0. The prerequisite
148 * check before any read/write capability check. */
149
150ngx_flag_t
151brix_validate_file_handle(brix_ctx_t *ctx, ngx_connection_t *c,
152 int handle_index, const char *verb, ngx_uint_t op, ngx_int_t *rc)
153{
154 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES
155 || ctx->files == NULL
156 || (ctx->files[handle_index].fd < 0
157 && ctx->files[handle_index].sd_obj.driver == NULL
158 && ctx->files[handle_index].writer == NULL))
159 {
160 /* A driver-backed handle (object/remote backend) is "open" with no kernel
161 * fd — data I/O routes through sd_obj.driver, so fd < 0 is normal there.
162 * A whole-object staged write handle (writer != NULL) is likewise "open"
163 * with no fd — byte I/O routes through the write session (phase-70). */
164 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-",
165 kXR_FileNotOpen, "invalid file handle", rc);
166 }
167
168 return 1;
169}

171/* brix_validate_read_handle — two phases: (1) brix_ensure_read_handle confirms
172 * the fd and refreshes bound secondaries; (2) the readable capability bit. 1 iff
173 * both pass, else logs + sends kXR_FileNotOpen/kXR_ServerError (phase 1) or
174 * kXR_NotAuthorized (phase 2). Re-checking the bit per read guards against drift. */
175
176ngx_flag_t
177brix_validate_read_handle(brix_ctx_t *ctx, ngx_connection_t *c,
178 int handle_index, const char *verb, ngx_uint_t op, ngx_int_t *rc)
179{
180 ngx_int_t ensure_rc;
181
182 ensure_rc = brix_ensure_read_handle(ctx, c, handle_index);
183 if (ensure_rc != NGX_OK) {
184 if (ensure_rc == NGX_ERROR) {
185 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_ServerError,
186 "could not prepare file handle", rc);
187 }
188
189 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_FileNotOpen,
190 "invalid file handle", rc);
191 }
192
193 /*
194 * Authorization and path checks happened at open time. Later handle I/O
195 * verifies the capability bit recorded on the handle rather than resolving
196 * the path again on every read.
197 */
198 if (!ctx->files[handle_index].readable) {
199 BRIX_BAIL_ERR(ctx, c, op, verb, ctx->files[handle_index].path, "-",
200 kXR_NotAuthorized, "file not open for reading", rc);
201 }
202
203 return 1;
204}

206/* brix_validate_write_handle — brix_validate_file_handle plus the writable
207 * capability bit. 1 iff both pass, else logs + sends kXR_NotAuthorized on
208 * writable=0. Re-checking the bit per write guards against drift, as for reads. */
209
210ngx_flag_t
211brix_validate_write_handle(brix_ctx_t *ctx, ngx_connection_t *c,
212 int handle_index, const char *verb, ngx_uint_t op, ngx_int_t *rc)
213{
214 /* Phase 94: a bound secondary carrying a write reopens the primary's
215 * published writable handle in this worker (mirror of the read path). A
216 * primary session short-circuits (its handle is already open locally). */
217 if (ctx->is_bound) {
218 ngx_int_t ensure_rc = brix_ensure_write_handle(ctx, c, handle_index);
219 if (ensure_rc != NGX_OK) {
220 if (ensure_rc == NGX_ERROR) {
221 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_ServerError,
222 "could not prepare write handle", rc);
223 }
224 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_FileNotOpen,
225 "invalid file handle", rc);
226 }
227 }
228
229 if (!brix_validate_file_handle(ctx, c, handle_index, verb, op, rc)) {
230 return 0;
231 }
232
233 if (!ctx->files[handle_index].writable) {
234 BRIX_BAIL_ERR(ctx, c, op, verb, ctx->files[handle_index].path, "-",
235 kXR_NotAuthorized, "file not open for writing", rc);
236 }
237
238 /*
239 * Phase-105: the endpoint gate for every handle-based mutation — kXR_write,
240 * writev, pgwrite, truncate, sync, chkpoint and the CLONE destination all
241 * reach the I/O core through this one validator, so gating here refuses
242 * BEFORE a job is posted (W2) rather than after a worker thread has already
243 * touched the file. The posture is the handle's own copy: a reload that
244 * turns the export read-only under an open write handle is answered
245 * kXR_fsReadOnly on the next op, and never with kXR_NotAuthorized, which
246 * would be indistinguishable from an authorization failure.
247 */
248 if (brix_vfs_require_carried_mutation(
249 ctx->files[handle_index].mutation_policy, BRIX_PROTO_ROOT,
250 BRIX_VFS_MUTATE_WRITE) != NGX_OK)
251 {
252 BRIX_BAIL_ERR(ctx, c, op, verb, ctx->files[handle_index].path, "-",
253 kXR_fsReadOnly, "read-only export", rc);
254 }
255
256 return 1;
257}
