14/* brix_fhandle_slot_live — 1 iff the slot is currently occupied by an open handle.
15 * A slot is live when it has ANY of: a kernel fd (fd >= 0), a driver-backed object
16 * (sd_obj.driver != NULL — remote/object backends are memory-served with fd < 0),
17 * or a whole-object staged writer (writer != NULL). This is the SAME liveness
18 * predicate brix_validate_file_handle uses; allocation and validation must agree,
19 * else a driver-backed handle (fd stays -1) reads as "free" and a second open on
20 * the same connection re-allocates its slot, collapsing distinct handles onto one
21 * index (breaks clients that hold multiple concurrent opens — e.g. uproot). */
22static ngx_inline int
23brix_fhandle_slot_live(const brix_ctx_t *ctx, int handle_index)
24{
25 return ctx->files[handle_index].fd >= 0
26 || ctx->files[handle_index].sd_obj.driver != NULL
27 || ctx->files[handle_index].writer != NULL;
28}

30/* brix_alloc_fhandle — return the first free slot in ctx->files; the index becomes
31 * the one-byte on-wire fhandle, so it is bounded by BRIX_MAX_FILES. -1 when all
32 * slots are occupied. Single-owner per connection (event thread, no locking); a
33 * slot is only reused once its handle is fully torn down (brix_free_fhandle). */
34int
35brix_alloc_fhandle(brix_ctx_t *ctx)
36{
37 int handle_index;
38
39 /*
40 * The XRootD wire handle stores this slot number in one byte. Keep all
41 * allocation/validation paths using the same bounded table.
42 */
43 for (handle_index = 0; handle_index < BRIX_MAX_FILES; handle_index++) {
44 if (!brix_fhandle_slot_live(ctx, handle_index)) {
45 return handle_index;
46 }
47 }
48
49 return -1;
50}

52/* brix_ctx_has_open_file — 1 if any handle slot is occupied (fd >= 0), else 0.
53 * Used by the recv-loop drain gate: a connection with an open file is mid-transfer
54 * (a streaming read parked between kXR_read chunks), so a draining worker must let
55 * it finish rather than fast-teardown at the request boundary — a forced mid-stream
56 * reconnect loses the in-flight fill. Single-owner per connection (event thread). */
57int
58brix_ctx_has_open_file(const brix_ctx_t *ctx)
59{
60 int handle_index;
61
62 for (handle_index = 0; handle_index < BRIX_MAX_FILES; handle_index++) {
63 if (brix_fhandle_slot_live(ctx, handle_index)) {
64 return 1;
65 }
66 }
67
68 return 0;

71/* brix_set_fhandle_path — store a heap copy (ngx_alloc, NOT pool, so it outlives
72 * the kXR_open request) of the canonical path in the slot, freeing any prior path
73 * first; brix_free_fhandle owns the free. NGX_OK, or NGX_ERROR on bad bounds or
74 * allocation failure. Heap (not pool) avoids fragmentation across open/close cycles. */
75ngx_int_t
76brix_set_fhandle_path(brix_ctx_t *ctx, ngx_connection_t *c,
77 int handle_index, const char *path)
78{
79 char *path_copy;
80 size_t path_bytes;
81
82 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES
83 || path == NULL)
84 {
85 return NGX_ERROR;
86 }
87
88 path_bytes = ngx_strlen(path) + 1;
89 path_copy = ngx_alloc(path_bytes, c->log);
90 if (path_copy == NULL) {
91 return NGX_ERROR;
92 }
93
94 ngx_memcpy(path_copy, path, path_bytes);
95
96 if (ctx->files[handle_index].path != NULL) {
97 ngx_free(ctx->files[handle_index].path);
98 }
99
100 ctx->files[handle_index].path = path_copy;
101 return NGX_OK;
102}

123/* brix_bound_confined_open — cache-aware confined (re)open of a primary-published
124 * shared entry's path with the given flags, validating the reopened object's
125 * device/inode against the published tuple. Returns a fresh caller-owned fd, or
126 * -1 on open/fstat failure or a device/inode mismatch (path replaced since publish;
127 * the fd is closed before returning). Shared by the bound READ (O_RDONLY) and
128 * bound WRITE (O_WRONLY) reopen paths so the confinement + stale-reference check
129 * lives in exactly one place. */
130static int
131brix_bound_confined_open(brix_ctx_t *ctx, ngx_connection_t *c,
132 const brix_shared_handle_entry_t *shared, int open_flags, struct stat *st)
133{
134 ngx_stream_brix_srv_conf_t *conf;
135 int fd;
136
137 conf = ngx_stream_get_module_srv_conf(ctx->session,
138 ngx_stream_brix_module);
139 if (conf == NULL) {
140 return -1;
141 }
142
143 /*
144 * The primary published a canonical, ACL-checked path. The secondary
145 * still opens it with the same confinement helper so a stale or corrupted
146 * shared-memory entry cannot escape the configured export root.
147 */
148 if (shared->from_cache) {
149 fd = open(shared->path, open_flags | O_NOFOLLOW); /* vfs-seam-allow: separate server-managed cache-root domain (from_cache), opened as worker; O_NOFOLLOW so a stray symlink in the svc-owned cache tree is not followed */
150 } else {
151 /* shared->path is the absolute path; strip root_canon to get the
152 * path relative to rootfd for openat2 RESOLVE_BENEATH. */
153 const char *rel = shared->path;
154 size_t root_len = strlen(conf->common.root_canon);
155 if (root_len > 0
156 && ngx_strncmp((u_char *) shared->path,
157 (u_char *) conf->common.root_canon,
158 root_len) == 0
159 && shared->path[root_len] == '/')
160 {
161 rel = shared->path + root_len;
162 }
163 fd = brix_vfs_open_fd_at(conf->rootfd, rel, open_flags, 0);
164 }
165
166 if (fd < 0) {
167 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, errno,
168 "brix: bound handle confined open failed path=%s flags=%d",
169 shared->path, open_flags);
170 return -1;
171 }
172
173 if (fstat(fd, st) != 0) {
174 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, errno,
175 "brix: fstat failed on bound reopen path=%s", shared->path);
176 close(fd);
177 return -1;
178 }
179
180 if (st->st_dev != shared->device || st->st_ino != shared->inode) {
181 close(fd);
182 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
183 "brix: bound handle path changed before reopen path=%s",
184 shared->path);
185 return -1;
186 }
187
188 return fd;
189}

191/* brix_reopen_bound_read_handle — reopen a bound secondary's stale fd O_RDONLY to
192 * match the primary's shared entry (via brix_bound_confined_open), refreshing all
193 * file metadata from the fresh fstat; NGX_DECLINED (revoke) if the object's
194 * device/inode changed. */
195
196static ngx_int_t
197brix_reopen_bound_read_handle(brix_ctx_t *ctx, ngx_connection_t *c,
198 int handle_index, const brix_shared_handle_entry_t *shared)
199{
200 brix_file_t *file;
201 struct stat st;
202 int fd;
203
204 if (!shared->readable || shared->path[0] == '\0') {
205 return NGX_DECLINED;
206 }
207
208 fd = brix_bound_confined_open(ctx, c, shared,
209 O_RDONLY | O_NOCTTY | O_CLOEXEC, &st);
210 if (fd < 0) {
211 return NGX_DECLINED;
212 }
213
214 file = &ctx->files[handle_index];
215 file->fd = fd;
216 file->readable = 1;
217 file->writable = 0;
218 file->from_cache = shared->from_cache ? 1 : 0;
219 file->is_regular = S_ISREG(st.st_mode) ? 1 : 0;
220 file->device = st.st_dev;
221 file->inode = st.st_ino;
222 file->cached_size = (off_t) st.st_size;
223 file->read_last_end = -1;
224 file->read_ahead_end = 0;
225 file->bytes_read = 0;
226 file->bytes_written = 0;
227 file->open_time = ngx_current_msec;
228
229 if (brix_set_fhandle_path(ctx, c, handle_index, shared->path)
230 != NGX_OK)
231 {
232 brix_free_fhandle(ctx, handle_index);
233 return NGX_ERROR;
234 }
235
236 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
237 "brix: bound handle=%d reopened shared path=%s",
238 handle_index, shared->path);
239 return NGX_OK;
240}

314/* brix_ensure_read_handle — for an unbound session, NGX_OK iff the fd exists. For
315 * a bound secondary, re-validate against the primary's shared entry every time: if
316 * the primary closed/reused the slot, revoke (free + NGX_DECLINED); if stale but
317 * still published, reopen a fresh matching fd. Invariant: a bound secondary never
318 * reads from a handle no longer in the primary's published table. */
319
320ngx_int_t
321brix_ensure_read_handle(brix_ctx_t *ctx, ngx_connection_t *c,
322 int handle_index)
323{
324 brix_shared_handle_entry_t shared;
325
326 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES) {
327 return NGX_DECLINED;
328 }
329
330 if (!ctx->is_bound) {
331 /* fd >= 0 (POSIX) OR a driver-backed object/remote handle (no kernel fd;
332 * reads route through sd_obj.driver via the buffered serve path). */
333 return (ctx->files[handle_index].fd >= 0
334 || ctx->files[handle_index].sd_obj.driver != NULL)
335 ? NGX_OK : NGX_DECLINED;
336 }
337
338 /*
339 * A bound stream only follows the primary session's current handle table.
340 * Re-check the shared slot on every read request so a primary close, reuse,
341 * or session teardown immediately revokes the secondary's local fd.
342 */
343 if (!brix_session_handle_lookup_hint(ctx->bound_sessid, handle_index,
344 &ctx->files[handle_index]
345 .shared_handle_slot_hint,
346 &shared))
347 {
348 if (ctx->files[handle_index].fd >= 0) {
349 brix_free_fhandle(ctx, handle_index);
350 }
351 return NGX_DECLINED;
352 }
353
354 if (brix_local_file_matches_shared_handle(&ctx->files[handle_index],
355 &shared))
356 {
357 return NGX_OK;
358 }
359
360 if (ctx->files[handle_index].fd >= 0
361 || ctx->files[handle_index].path != NULL)
362 {
363 brix_free_fhandle(ctx, handle_index);
364 }
365
366 return brix_reopen_bound_read_handle(ctx, c, handle_index, &shared);
367}

369/* brix_ensure_write_handle — Phase 94 write-side mirror of brix_ensure_read_handle.
370 * For an unbound (primary) session the write handle is already open locally, so
371 * NGX_OK iff a kernel fd / driver object / staged writer exists. For a bound
372 * secondary, re-validate against the primary's shared WRITABLE entry every write:
373 * a published-read-only or absent entry → revoke (NGX_DECLINED); a stale-but-still-
374 * published entry → reopen a fresh matching O_WRONLY fd. Invariant: a bound
375 * secondary never writes a handle the primary did not publish as writable. */
376ngx_int_t
377brix_ensure_write_handle(brix_ctx_t *ctx, ngx_connection_t *c,
378 int handle_index)
379{
380 brix_shared_handle_entry_t shared;
381
382 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES) {
383 return NGX_DECLINED;
384 }
385
386 if (!ctx->is_bound) {
387 return (ctx->files[handle_index].fd >= 0
388 || ctx->files[handle_index].sd_obj.driver != NULL
389 || ctx->files[handle_index].writer != NULL)
390 ? NGX_OK : NGX_DECLINED;
391 }
392
393 if (!brix_session_handle_lookup_hint(ctx->bound_sessid, handle_index,
394 &ctx->files[handle_index]
395 .shared_handle_slot_hint,
396 &shared))
397 {
398 if (ctx->files[handle_index].fd >= 0) {
399 brix_free_fhandle(ctx, handle_index);
400 }
401 return NGX_DECLINED;
402 }
403
404 /* A published read-only handle is a data channel a bound conn may only read. */
405 if (!shared.writable) {
406 if (ctx->files[handle_index].fd >= 0
407 && ctx->files[handle_index].writable)
408 {

410 }
411 return NGX_DECLINED;
412 }
413
414 if (brix_local_file_matches_shared_write_handle(&ctx->files[handle_index],
415 &shared))
416 {
417 return NGX_OK;
418 }
419
420 if (ctx->files[handle_index].fd >= 0
421 || ctx->files[handle_index].path != NULL)
422 {
423 brix_free_fhandle(ctx, handle_index);
424 }
425
426 return brix_reopen_bound_write_handle(ctx, c, handle_index, &shared);
427}

432/* brix_validate_file_handle — 1 iff handle_index is in bounds and the slot has an
433 * active fd; otherwise logs + sends kXR_FileNotOpen and returns 0. The prerequisite
434 * check before any read/write capability check. */
435
436ngx_flag_t
437brix_validate_file_handle(brix_ctx_t *ctx, ngx_connection_t *c,
438 int handle_index, const char *verb, ngx_uint_t op, ngx_int_t *rc)
439{
440 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES
441 || (ctx->files[handle_index].fd < 0
442 && ctx->files[handle_index].sd_obj.driver == NULL
443 && ctx->files[handle_index].writer == NULL))
444 {
445 /* A driver-backed handle (object/remote backend) is "open" with no kernel
446 * fd — data I/O routes through sd_obj.driver, so fd < 0 is normal there.
447 * A whole-object staged write handle (writer != NULL) is likewise "open"
448 * with no fd — byte I/O routes through the write session (phase-70). */
449 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-",
450 kXR_FileNotOpen, "invalid file handle", rc);
451 }
452
453 return 1;

456/* brix_validate_read_handle — two phases: (1) brix_ensure_read_handle confirms
457 * the fd and refreshes bound secondaries; (2) the readable capability bit. 1 iff
458 * both pass, else logs + sends kXR_FileNotOpen/kXR_ServerError (phase 1) or
459 * kXR_NotAuthorized (phase 2). Re-checking the bit per read guards against drift. */
460
461ngx_flag_t
462brix_validate_read_handle(brix_ctx_t *ctx, ngx_connection_t *c,
463 int handle_index, const char *verb, ngx_uint_t op, ngx_int_t *rc)
464{
465 ngx_int_t ensure_rc;
466
467 ensure_rc = brix_ensure_read_handle(ctx, c, handle_index);
468 if (ensure_rc != NGX_OK) {
469 if (ensure_rc == NGX_ERROR) {
470 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_ServerError,
471 "could not prepare file handle", rc);
472 }
473
474 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_FileNotOpen,
475 "invalid file handle", rc);
476 }
477
478 /*
479 * Authorization and path checks happened at open time. Later handle I/O
480 * verifies the capability bit recorded on the handle rather than resolving
481 * the path again on every read.
482 */
483 if (!ctx->files[handle_index].readable) {
484 BRIX_BAIL_ERR(ctx, c, op, verb, ctx->files[handle_index].path, "-",
485 kXR_NotAuthorized, "file not open for reading", rc);
486 }
487
488 return 1;
489}

491/* brix_validate_write_handle — brix_validate_file_handle plus the writable
492 * capability bit. 1 iff both pass, else logs + sends kXR_NotAuthorized on
493 * writable=0. Re-checking the bit per write guards against drift, as for reads. */
494

496brix_validate_write_handle(brix_ctx_t *ctx, ngx_connection_t *c,
497 int handle_index, const char *verb, ngx_uint_t op, ngx_int_t *rc)
498{

500 * published writable handle in this worker (mirror of the read path). A
501 * primary session short-circuits (its handle is already open locally). */
502 if (ctx->is_bound) {

504 if (ensure_rc != NGX_OK) {
505 if (ensure_rc == NGX_ERROR) {
506 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_ServerError,
507 "could not prepare write handle", rc);
508 }
509 BRIX_BAIL_ERR(ctx, c, op, verb, "-", "-", kXR_FileNotOpen,
510 "invalid file handle", rc);
511 }
512 }
513
514 if (!brix_validate_file_handle(ctx, c, handle_index, verb, op, rc)) {
515 return 0;
516 }
517
518 if (!ctx->files[handle_index].writable) {
519 BRIX_BAIL_ERR(ctx, c, op, verb, ctx->files[handle_index].path, "-",
520 kXR_NotAuthorized, "file not open for writing", rc);
521 }

523 return 1;
