8/* Arm S3 STS EXCHANGE (phase-70 §5.5) when the export configured a full STS
9 * service triple (endpoint + service ak + service sk). An S3 SigV4 secret is
10 * never on the wire, so this is the only way to authenticate the S3 origin leg
11 * AS the caller: the node exchanges its own service credential for temporary
12 * creds scoped to the caller's identity. The conf is built on the request pool
13 * (its ngx_str_t fields borrow conf-owned bytes) and handed to set_sts, which
14 * allocates the live-cred bag when no forwardable bytes were captured — the STS
15 * case, exactly like SSS injection. A missing ak/sk (or endpoint) leaves STS
16 * disarmed; region defaults inside the STS client, ttl inside the conf merge. */
17static void
18deleg_wire_stamp_sts(brix_vfs_ctx_t *vctx, ngx_http_brix_shared_conf_t *cc)
19{
20 brix_s3_sts_conf_t *cf;
21
22 if (cc->backend_sts_endpoint.len == 0
23 || cc->backend_sts_access_key.len == 0
24 || cc->backend_sts_secret_key.len == 0)
25 {
26 return;
27 }
28
29 cf = ngx_pcalloc(vctx->pool, sizeof(*cf));
30 if (cf == NULL) {
31 return; /* degrade to SELECT — deny is the gate's call */
32 }
33 cf->endpoint = cc->backend_sts_endpoint;
34 cf->region = cc->backend_sts_region;
35 cf->role_arn = cc->backend_sts_role;
36 cf->svc_ak = cc->backend_sts_access_key;
37 cf->svc_sk = cc->backend_sts_secret_key;
38 cf->ttl_secs = (cc->backend_sts_ttl != NGX_CONF_UNSET)
39 ? (int) cc->backend_sts_ttl : 3600;
40 cf->flavor = (cc->backend_sts_flavor != NGX_CONF_UNSET_UINT)
41 ? (int) cc->backend_sts_flavor : BRIX_STS_FLAVOR_AWS;
42
43 brix_vfs_deleg_set_sts(vctx,
44 (enum brix_cred_mode) cc->backend_delegation, cf);
45}

47/* EXCHANGE-with-endpoint is the one mode that does NOT forward the client
48 * bearer verbatim: the exchange mints a backend-audienced replacement. */
49static int
50deleg_wire_exchanges(const ngx_http_brix_shared_conf_t *cc)
51{
52 return cc->backend_delegation == BRIX_CRED_EXCHANGE
53 && cc->backend_tx_endpoint.len > 0;
54}

72{
73 const ngx_str_t *aud = NULL;
74
75 /* SSS injection first: it allocates the bag when no bytes were captured
76 * (§5.6); the exchange stamp below is a piggyback-only setter. */
77 brix_vfs_deleg_set_sss(vctx,
78 (enum brix_cred_mode) cc->backend_delegation,
79 &cc->backend_sss_keytab);
80
81 /* S3 STS EXCHANGE (§5.5): like SSS, arms on the no-captured-bytes path and
82 * allocates the bag if needed. Independent of the bearer token-exchange
83 * below (that is the OAuth RFC-8693 leg; this is the S3 SigV4 leg). */
84 deleg_wire_stamp_sts(vctx, cc);
85
86 if (!deleg_wire_exchanges(cc)) {
87 return;
88 }
89 if (cc->backend_token_aud != NULL && cc->backend_token_aud->nelts > 0) {
90 aud = &((ngx_str_t *) cc->backend_token_aud->elts)[0];
91 }
92 brix_vfs_deleg_set_exchange(vctx, &cc->backend_tx_endpoint,
93 &cc->backend_tx_client_id, &cc->backend_tx_client_secret,
94 aud, &cc->backend_tx_cache);
95}
