2 * metalink_lex.c — bounded XML lexing for the metalink parser (phase-100).
3 *
4 * WHAT: The tag / attribute / element-text extractors and the XML character
5 * entity decoder that metalink.c builds its document semantics on.
6 * WHY: Split from metalink.c on the 600-line file gate (see
7 * metalink_internal.h): the lexer knows nothing about mirrors, ranks or
8 * digests, so it is one concept with its own hostile-input rules — the
9 * 11-byte entity cap, the ASCII-only numeric range, and the name-boundary
10 * check that stops "<url" matching "<urlfoo".
11 * HOW: Every routine is bounded by an explicit end/gt pointer, allocates

22/* ---- Case-insensitive bounded substring search ----
23 *
24 * WHAT: Return the first occurrence of NUL-terminated `needle` in [p,end),
25 * comparing ASCII case-insensitively, or NULL when absent.
26 *
27 * WHY: XML tag/attribute names are matched case-insensitively so hand-written
28 * v3 documents ("<Metalink>", "MD5") parse the same as canonical ones.
29 *
30 * HOW: 1. Walk each start position with enough room for the needle.
31 * 2. Compare byte-wise through tolower().
32 */
33const char *
34ml_find_ci(const char *p, const char *end, const char *needle)
35{
36 size_t nlen = strlen(needle);
37
38 if (nlen == 0 || p == NULL || end - p < (ptrdiff_t) nlen) {
39 return NULL;
40 }
41 for (; p + nlen <= end; p++) {
42 size_t i;
43 for (i = 0; i < nlen; i++) {
44 if (tolower((unsigned char) p[i])
45 != tolower((unsigned char) needle[i])) {
46 break;
47 }
48 }
49 if (i == nlen) {
50 return p;
51 }
52 }
53 return NULL;
54}

57/* ---- Locate the next opening tag `<name` with a real name boundary ----
58 *
59 * WHAT: Find the next `<name` in [p,end) whose following byte terminates the
60 * tag name (whitespace, '>', or '/'), so "<url" never matches "<urlfoo".
61 * Returns the position of '<', or NULL. attrs_out / gt_out (may be NULL)
62 * get the attribute region start (after the name) and the closing '>'.
63 *
64 * WHY: The attribute extractor and text extractor both need the exact
65 * attribute window and element-body start; computing them here keeps every
66 * caller a one-liner and the bounds checked in one place.
67 *
68 * HOW: 1. Compose "<name" (bounded) and ml_find_ci it, retrying on false
69 * name-prefix hits. 2. Scan to the tag's '>' (bounded by `end`).
70 */
71const char *
72ml_tag_open(const char *p, const char *end, const char *name,
73 const char **attrs_out, const char **gt_out)
74{
75 char tag[32];
76 size_t nlen = strlen(name);
77
78 if (nlen + 2 > sizeof(tag)) {
79 return NULL;
80 }
81 tag[0] = '<';
82 memcpy(tag + 1, name, nlen + 1);
83
84 for (;;) {
85 const char *hit = ml_find_ci(p, end, tag);
86 const char *after, *gt;
87
88 if (hit == NULL) {
89 return NULL;
90 }
91 after = hit + 1 + nlen;
92 if (after >= end
93 || !(isspace((unsigned char) *after) || *after == '>'
94 || *after == '/')) {
95 p = hit + 1; /* name-prefix false positive, e.g. <urls> */
96 continue;
97 }
98 for (gt = after; gt < end && *gt != '>'; gt++) { }
99 if (gt >= end) {
100 return NULL; /* unterminated tag */
101 }
102 if (attrs_out != NULL) {
103 *attrs_out = after;
104 }
105 if (gt_out != NULL) {
106 *gt_out = gt;
107 }
108 return hit;
109 }
110}

113/*
114 * WHAT: One named XML entity — its literal spelling (including '&' and ';')
115 * and the byte it decodes to.
116 * WHY: The five predefined entities differ only in spelling and result, so
117 * they belong in a table rather than a strncasecmp ladder
118 * (coding-standards §8.6).
119 * HOW: `len` is precomputed so the match is a length test before the compare.
120 */
121typedef struct {
122 const char *name;
123 size_t len;
124 char ch;
125} ml_entity_t;
126
127static const ml_entity_t ml_named_entities[] = {
128 { "&amp;", 5, '&' },
129 { "&lt;", 4, '<' },
130 { "&gt;", 4, '>' },
131 { "&quot;", 6, '"' },
132 { "&apos;", 6, '\'' },
133};

136/* ---- Match one of the five predefined entities ----
137 *
138 * WHAT: Returns the number of bytes consumed and stores the decoded byte, or 0
139 * when `s` (an '&'…';' run of `elen` bytes) is not a named entity.
140 *
141 * WHY: Length-first matching means a truncated or over-long run can never be
142 * mistaken for a shorter entity that happens to share a prefix.
143 *
144 * HOW: 1. Walk the table, requiring an exact length match.
145 * 2. Compare case-insensitively — real-world metalinks emit &AMP;.
146 */
147static size_t
148ml_entity_named(const char *s, size_t elen, char *out)
149{
150 size_t n;
151
152 for (n = 0; n < sizeof(ml_named_entities) / sizeof(ml_named_entities[0]); n++) {
153 if (elen == ml_named_entities[n].len
154 && strncasecmp(s, ml_named_entities[n].name, elen) == 0)
155 {
156 *out = ml_named_entities[n].ch;
157 return elen;
158 }
159 }
160
161 return 0;
162}

165/* ---- Match a numeric &#NN; / &#xNN; entity ----
166 *
167 * WHAT: Returns the number of bytes consumed and stores the decoded byte, or 0
168 * when the run is not a numeric entity in the representable range.
169 *
170 * WHY: Only ASCII is decoded: this buffer is a byte string handed to the
171 * transport, so emitting a multi-byte codepoint here would produce a URL
172 * the server never issued. Anything else is left verbatim rather than
173 * being lossily folded.
174 *
175 * HOW: 1. Require "&#" and at least one digit before the ';'.
176 * 2. Pick base 16 for an x/X prefix, else base 10.
177 * 3. Accept only 0 < cp < 128.
178 */
179static size_t
180ml_entity_numeric(const char *s, size_t elen, char *out)
181{
182 int base;
183 long cp;
184
185 if (elen <= 3 || s[1] != '#') {
186 return 0;
187 }
188
189 base = (s[2] == 'x' || s[2] == 'X') ? 16 : 10;
190 cp = strtol(s + (base == 16 ? 3 : 2), NULL, base);
191 if (cp <= 0 || cp >= 128) { /* ASCII only; else verbatim */
192 return 0;
193 }
194
195 *out = (char) cp;
196 return elen;
197}

200/* ---- Decode the entity starting at `s`, if there is one ----
201 *
202 * WHAT: Returns the bytes consumed and stores the decoded byte, or 0 when the
203 * run is not a recognised entity.
204 *
205 * WHY: An entity is at most 11 bytes to the ';'; capping the search means a
206 * stray '&' in a long URL costs a bounded scan instead of walking to the
207 * end of the value.
208 *
209 * HOW: 1. Find the ';' within the cap; no ';' means no entity.
210 * 2. Try the named table, then the numeric forms.
211 */
212static size_t
213ml_entity_at(const char *s, size_t avail, char *out)
214{
215 const char *semi = memchr(s, ';', avail < 12 ? avail : 12);
216 size_t elen, adv;
217
218 if (semi == NULL) {
219 return 0;
220 }
221
222 elen = (size_t) (semi - s) + 1;
223
224 adv = ml_entity_named(s, elen, out);
225 if (adv != 0) {
226 return adv;
227 }
228
229 return ml_entity_numeric(s, elen, out);
230}

233/* ---- Decode XML character entities into a bounded buffer ----
234 *
235 * WHAT: Copy [src,src+len) into out[outsz], decoding &amp; &lt; &gt; &quot;
236 * &apos; and numeric &#NN;/&#xNN; forms. Returns 0, or -1 when the
237 * decoded text would not fit (out is always NUL-terminated).
238 *
239 * WHY: Metalink URLs legally carry '&' query separators as &amp;; skipping
240 * decoding would hand the transport a URL the server never issued.
241 *
242 * HOW: 1. Byte copy, branching on '&'. 2. An entity is at most 11 bytes to the
243 * ';'; unknown/malformed entities are copied verbatim (lenient, like
244 * browsers) so a stray '&' does not kill the mirror.
245 */
246static int
247ml_entity_decode(const char *src, size_t len, char *out, size_t outsz)
248{
249 size_t si = 0, oi = 0;
250
251 while (si < len) {
252 char decoded = 0;
253 size_t adv = 0;
254
255 if (src[si] == '&') {
256 adv = ml_entity_at(src + si, len - si, &decoded);
257 }
258 if (adv == 0) { /* plain byte or unrecognized entity */
259 decoded = src[si];
260 adv = 1;
261 }
262 if (oi + 1 >= outsz) {
263 out[oi] = '\0';
264 return -1;
265 }
266 out[oi++] = decoded;
267 si += adv;
268 }
269 out[oi] = '\0';
270 return 0;
271}

274/* ---- Extract one attribute value from a tag's attribute window ----
275 *
276 * WHAT: Find `name="value"` (or '...') inside [attrs,gt) and entity-decode the
277 * value into out[outsz]. Returns 0 when found and fits, -1 otherwise.
278 *
279 * WHY: priority/preference/type live in attributes; a bounded scanner beats a
280 * full XML attribute grammar for these three fixed names.
281 *
282 * HOW: 1. ml_find_ci the name with a boundary check (preceded by whitespace).
283 * 2. Skip spaces, require '=', skip spaces, require a quote. 3. The
284 * value runs to the matching quote (bounded by gt).
285 */
286int
287ml_attr(const char *attrs, const char *gt, const char *name,
288 char *out, size_t outsz)
289{
290 const char *p = attrs;
291 size_t nlen = strlen(name);
292
293 while ((p = ml_find_ci(p, gt, name)) != NULL) {
294 const char *v = p + nlen;
295 char quote;
296 const char *vend;
297
298 if (p > attrs && !isspace((unsigned char) p[-1])) {
299 p += 1; /* substring of a longer attribute name */
300 continue;
301 }
302 while (v < gt && isspace((unsigned char) *v)) { v++; }
303 if (v >= gt || *v != '=') {
304 p += 1;
305 continue;
306 }
307 v++;
308 while (v < gt && isspace((unsigned char) *v)) { v++; }
309 if (v >= gt || (*v != '"' && *v != '\'')) {
310 p += 1;
311 continue;
312 }
313 quote = *v++;
314 vend = memchr(v, quote, (size_t) (gt - v));
315 if (vend == NULL) {
316 return -1;
317 }
318 return ml_entity_decode(v, (size_t) (vend - v), out, outsz);
319 }
320 return -1;
321}

324/* ---- Extract an element's text body ----
325 *
326 * WHAT: Given the '>' of an opening <name ...> tag, find `</name` and
327 * entity-decode the text between (whitespace-trimmed) into out[outsz].
328 * Returns 0, or -1 (missing close tag / does not fit).
329 *
330 * WHY: <url>, <size> and <hash> all carry their payload as element text.
331 *
332 * HOW: 1. Compose "</name" and ml_find_ci from just after '>'. 2. Trim ASCII
333 * whitespace off both ends. 3. Entity-decode the middle.
334 */
335int
336ml_elem_text(const char *gt, const char *end, const char *name,
337 char *out, size_t outsz)
338{
339 char close_tag[32];
340 const char *text = gt + 1, *close, *tend;
341 size_t nlen = strlen(name);
342
343 if (nlen + 3 > sizeof(close_tag)) {
344 return -1;
345 }
346 close_tag[0] = '<';
347 close_tag[1] = '/';
348 memcpy(close_tag + 2, name, nlen + 1);
349
350 close = ml_find_ci(text, end, close_tag);
351 if (close == NULL) {
352 return -1;
353 }
354 tend = close;
355 while (text < tend && isspace((unsigned char) *text)) { text++; }
356 while (tend > text && isspace((unsigned char) tend[-1])) { tend--; }
357 return ml_entity_decode(text, (size_t) (tend - text), out, outsz);
358}
