37int
38doctor_cfg_capacity_pct(int64_t total, int64_t freeb)

96/*
97 * WHAT: fetch one Qconfig key's value-line, "" if unsupported/unanswered.
98 * HOW: strip at '=' or '\n' exactly as xrd_probe_caps does; a server that
99 * echoes the key name verbatim does NOT support it (→ "").
100 */
101static void
102qc_val(brix_conn *c, const char *key, char *out, size_t osz)
103{
104 char reply[256], *nl, *eq;
105 const char *val;
106 brix_status st;
107
108 out[0] = '\0';
109 brix_status_clear(&st);
110 if (brix_query(c, kXR_Qconfig, key, reply, sizeof(reply), &st) != 0) {
111 return;
112 }
113 if ((nl = strchr(reply, '\n')) != NULL) {
114 *nl = '\0';
115 }
116 eq = strchr(reply, '=');
117 val = (eq != NULL) ? eq + 1 : reply;
118 if (strcmp(val, key) == 0) { /* echo == key ⇒ unsupported/absent */
119 return;
120 }
121 snprintf(out, osz, "%s", val);
122}

136/* Populate e->cfg from a live root:// connection. Best-effort + PII-free: an
137 * unanswered/unsupported key leaves its field at the "absent" sentinel. */
138void
139doctor_scrape_config(brix_conn *c, doctor_ep *e)
140{
141 doctor_cfg *g = &e->cfg;
142 char csv[256];
143 unsigned long long tot = 0, freeb = 0;
144 char sp[512];
145 brix_status st;
146
147 g->scraped = 1;
148 g->bind_max = g->pio_max = g->readv_iov_max = g->readv_ior_max = -1;
149 g->space_total = g->space_free = -1;
150
151 qc_val(c, "version", g->version, sizeof(g->version));
152 qc_val(c, "role", g->role, sizeof(g->role));
153 qc_val(c, "sitename", g->sitename, sizeof(g->sitename));
154 qc_val(c, "chksum", csv, sizeof(csv));
155 doctor_cfg_parse_chksum(csv, &g->have_adler32, &g->have_crc32c);
156 g->tpc = qc_int(c, "tpc") > 0 ? 1 : 0;
157 g->tpcdlg = qc_int(c, "tpcdlg") > 0 ? 1 : 0;
158 g->pgread = qc_int(c, "pgread") > 0 ? 1 : 0;
159 g->bind_max = qc_int(c, "bind_max");
160 g->pio_max = qc_int(c, "pio_max");
161 g->readv_iov_max = qc_int(c, "readv_iov_max");
162 g->readv_ior_max = qc_int(c, "readv_ior_max");
163
164 /* capacity: kXR_Qspace is rpCheck'd server-side — pass an absolute path. */
165 brix_status_clear(&st);
166 if (brix_query(c, kXR_Qspace, "/", sp, sizeof(sp), &st) == 0 && sp[0] != '\0') {
167 brix_parse_qspace(sp, &tot, &freeb);
168 if (tot > 0) {
169 g->space_total = (int64_t) tot;
170 g->space_free = (int64_t) freeb;
171 }
172 }
173}

178void
179doctor_audit_rules(const diag_args *a, doctor_ep *e)
180{
181 doctor_cfg *g = &e->cfg;
182 int pct;
183
184 if (!g->scraped) {
185 return;
186 }
187 if (!g->have_adler32 && !g->have_crc32c) {
188 dx_record(e, &(dx_note){ "config-chksum", DX_WARN, 0,
189 "no common checksum algorithm advertised (no adler32/crc32c)",
190 "enable adler32 or crc32c so clients can verify data integrity" });
191 }
192 if (!g->tpc || !g->tpcdlg) {
193 dx_record(e, &(dx_note){ "config-tpc", DX_WARN, 0,
194 "third-party copy not fully advertised (tpc/tpcdlg)",
195 "enable tpc (and tpcdlg delegation) if this endpoint takes part in TPC" });
196 }
197 if (g->sitename[0] == '\0') {
198 dx_record(e, &(dx_note){ "config-sitename", DX_WARN, 0,
199 "sitename is unset",
200 "set a sitename for monitoring/attribution across the federation" });
201 }
202 if (g->bind_max >= 0 && g->bind_max < 4) {
203 dx_record(e, &(dx_note){ "config-parallel", DX_WARN, 0,
204 "server-side parallelism is capped low (bind_max < 4)",
205 "raise bind_max / readv limits if clients need more parallel streams" });
206 }
207 pct = doctor_cfg_capacity_pct(g->space_total, g->space_free);
208 if (pct >= 0 && pct < doctor_cfg_cap_threshold(a)) {
209 dx_record(e, &(dx_note){ "capacity-low", DX_WARN, 0,
210 "export filesystem is nearly full",
211 "free space or add capacity before writes fail with kXR_NoSpace" });
212 }

241
242/* Capacity-balance: flag a DS whose free%% deviates > 25 points from the fleet
243 * mean (DS nodes are index >= 1). Records one capacity-imbalance WARN. */
244static void
245doctor_cluster_balance(doctor_ep *eps, int n, FILE *out)
246{
247 int i, cnt = 0, sum = 0, mean, pct, outliers = 0;
248
249 for (i = 1; i < n; i++) {
250 pct = doctor_cfg_capacity_pct(eps[i].cfg.space_total, eps[i].cfg.space_free);
251 if (pct >= 0) { sum += pct; cnt++; }
252 }
253 if (cnt < 2) {
254 return;
255 }
256 mean = sum / cnt;
257 for (i = 1; i < n; i++) {
258 pct = doctor_cfg_capacity_pct(eps[i].cfg.space_total, eps[i].cfg.space_free);
259 if (pct >= 0 && (pct - mean > 25 || mean - pct > 25)) { outliers++; }
260 }
261 if (out != NULL) {
262 fprintf(out, " capacity balance: mean=%d%% free, %d outlier(s)\n",
263 mean, outliers);
264 }
265 if (outliers > 0) {
266 dx_record(&eps[0], &(dx_note){ "cap-imbalance", DX_WARN, 0,
267 "data-server free space is unbalanced across the cluster",
268 "rebalance data or check a stuck/degraded data server" });
269 }
270}

273/* Cross-node fleet diff: version skew, manager-role count, capacity balance.

275void
276doctor_cross_cluster(doctor_ep *eps, int n, FILE *out)
277{

297 "cluster does not advertise exactly one manager role",

299 }

363/* ---- §5.3 manager fan-out (manager → N located data servers) ---- */
364
365#define DOCTOR_FANOUT_CAP 33 /* [0]=manager + up to 32 data servers */
366
367/* Classify one kXR_locate token "<type><access>host:port" into its CMS-plane
368 * role/access and split out host/port. The type byte is the manager's own word
369 * on what the node is: 'S'/'s' data server, 'M'/'m' subordinate manager
370 * (redirector); lowercase means pending (queued/staging). The access byte is
371 * 'r' (read-only) or 'w' (read/write). Returns 0 on a recognised location, -1
372 * when the token is not one (unknown type byte or unparseable host). */
373static int
374doctor_locate_classify(const char *tok, doctor_cmsloc *cms,
375 char *host, size_t hsz, int *port)
376{
377 char t0;
378
379 if (tok == NULL || tok[0] == '\0' || tok[1] == '\0') {
380 return -1;
381 }
382 t0 = tok[0];
383 if (t0 == 'S' || t0 == 's') {
384 cms->role = DOC_CMS_SERVER;
385 } else if (t0 == 'M' || t0 == 'm') {
386 cms->role = DOC_CMS_MANAGER;
387 } else {
388 return -1; /* not a location entry */
389 }
390 cms->pending = (t0 >= 'a' && t0 <= 'z'); /* lowercase => queued/staging */
391 cms->write = (tok[1] == 'w');
392 cms->reported = 1;
393 if (brix_split_host_port(tok + 2, host, hsz, port, 1094) != 0) {
394 return -1;
395 }
396 return 0;
397}
398
399/* Turn one CMS locate token into endpoint slot `slot`. Returns 0 when the slot
400 * was filled, -1 when the token names nothing we can classify.
401 *
402 * The classification comes from the locate token itself — data server vs
403 * redirector, read-only vs read/write, online vs pending. That is the authority
404 * for what the node IS even when we can never open an inbound connection to it,
405 * so an IPv6-only holder on a v4-only host is recorded SKIPPED (not a red
406 * connect failure) with its CMS verdict intact. */
407static int
408fanout_add_ds(const diag_args *a, char *tok, doctor_ep *slot, int lv6)
409{
410 doctor_cmsloc cms;
411 char host[256], dsurl[320];
412 int port, v6;
413
414 memset(&cms, 0, sizeof(cms));
415 if (doctor_locate_classify(tok, &cms, host, sizeof(host), &port) != 0) {
416 return -1;
417 }
418 v6 = (strchr(host, ':') != NULL && host[0] != '[');
419 if (!lv6 && (v6 || doctor_host_ipv6_only(host))) {
420 slot->proto = DXP_ROOT;
421 snprintf(slot->host, sizeof(slot->host), "%s", host);
422 slot->port = port;
423 slot->skipped = 1;
424 slot->cms = cms;
425 return 0;
426 }
427 snprintf(dsurl, sizeof(dsurl), "root://%s%s%s:%d/",
428 v6 ? "[" : "", host, v6 ? "]" : "", port);
429 doctor_one(a, dsurl, slot);
430 slot->cms = cms; /* re-apply: doctor_one zeroes the endpoint */
431 return 0;
432}

435/* Fill arr[1..] from a fresh locate on "/" against the manager. Returns the new
436 * endpoint count (1 when the manager cannot be re-connected or locates nothing)
437 * and sets *more when the fleet overflowed DOCTOR_FANOUT_CAP. */
438static int
439fanout_discover(const diag_args *a, doctor_ep *arr, int lv6, int *more)
440{
441 brix_url u;
442 brix_conn c;
443 brix_status st;
444 char loc[4096];
445 char *t, *save;
446 int n = 1;
447
448 brix_status_clear(&st);
449 if (brix_endpoint_parse(a->urls[0], &u, &st) != 0
450 || brix_connect(&c, &u, &a->conn, &st) != 0) {
451 return n;
452 }
453 c.io.timeout_ms = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
454 brix_status_clear(&st);
455 if (brix_locate(&c, "/", loc, sizeof(loc), &st) == 0) {
456 for (t = strtok_r(loc, " \t\r\n", &save); t != NULL;
457 t = strtok_r(NULL, " \t\r\n", &save)) {
458 if (n >= DOCTOR_FANOUT_CAP) {
459 *more = 1;
460 break;
461 }
462 if (fanout_add_ds(a, t, &arr[n], lv6) == 0) {
463 n++;
464 }
465 }
466 }
467 brix_close(&c);
468 return n;
469}

472/* Turn a manager URL (a->urls[0]) into eps[0]=manager + eps[1..) data servers.
473 * Heap-allocates the endpoint array (caller frees). Returns 0 on success, -1 on
474 * allocation failure; *truncated is set when the DS list exceeded the cap. */
475int
476doctor_fanout(const diag_args *a, doctor_ep **eps_out, int *n_out, int *truncated)
477{
478 doctor_ep *arr;
479 int n = 1, more = 0;
480 int lv6 = doctor_have_ipv6(); /* skip IPv6-only DSs if local v6 is down */
481
482 *truncated = 0;
483 *eps_out = NULL;
484 *n_out = 0;
485 arr = calloc(DOCTOR_FANOUT_CAP, sizeof(*arr));
486 if (arr == NULL) {
487 return -1;
488 }
489
490 /* [0] = the manager itself — full battery + config scrape via dispatch. */
491 doctor_dispatch(a, a->urls[0], &arr[0]);
492
493 /* discover data servers from a fresh locate on "/". */
494 n = fanout_discover(a, arr, lv6, &more);
495
496 /* The node we connected to answered a locate with other hosts — it is, by
497 * that act, acting as a manager/redirector. Record that on the CMS plane so
498 * the map types the root correctly even when its Qconfig role is opaque
499 * (e.g. EOS reports role=none). A lone node that located only itself stays
500 * unclassified and renders as a standalone server. */
501 if (n > 1) {
502 arr[0].cms.role = DOC_CMS_MANAGER;
503 arr[0].cms.reported = 1;
504 }
505
506 /* EOS dialect: if urls[0] is an EOS MGM, speak its /proc command channel to
507 * replace the locate-blind self-node with the real FST farm (or mark the MGM
508 * admin-gated when enumeration is not permitted for this identity). Runs only
509 * for --map and leaves a non-EOS mesh untouched. */
510 if (doctor_eos_map(a, arr, DOCTOR_FANOUT_CAP, &n)) {
511 more = 1;
512 }
513
514 *truncated = more;
515 *eps_out = arr;
516 *n_out = n;
517 return 0;
518}
