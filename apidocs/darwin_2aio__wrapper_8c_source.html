40aio_run_job(brix_aio_job_t *job)

75static void *
76aio_worker(void *data)
77{
78 brix_aio_ctx_t *ctx = data;
79
80 for (;;) {

82
83 pthread_mutex_lock(&ctx->mutex);
84 while (ctx->queued_head == NULL && !ctx->stopping) {
85 pthread_cond_wait(&ctx->queued_cond, &ctx->mutex);
86 }
87 if (ctx->queued_head == NULL && ctx->stopping) {
88 pthread_mutex_unlock(&ctx->mutex);

91 job = aio_pop_job(&ctx->queued_head, &ctx->queued_tail);
92 pthread_mutex_unlock(&ctx->mutex);

94 job->error = job->result < 0 ? errno : 0;
95 pthread_mutex_lock(&ctx->mutex);
96 aio_push_job(&ctx->complete_head, &ctx->complete_tail, job);
97 pthread_cond_signal(&ctx->complete_cond);

112/* Create a bounded pool; at most four threads service max_entries requests. */
113brix_aio_ctx_t *
114brix_aio_create(size_t max_entries)
115{
116 brix_aio_ctx_t *ctx;
117 size_t i;
118
119 if (max_entries == 0) {
120 errno = EINVAL;
121 return NULL;
122 }
123 ctx = calloc(1, sizeof(*ctx));
124 if (ctx == NULL) return NULL;
125 ctx->max_entries = max_entries;
126 ctx->worker_count = max_entries < 4 ? max_entries : 4;
127 ctx->workers = calloc(ctx->worker_count, sizeof(ctx->workers[0]));
128 if (ctx->workers == NULL) {
129 free(ctx);
130 return NULL;
131 }
132 if (pthread_mutex_init(&ctx->mutex, NULL) != 0) {
133 free(ctx->workers);
134 free(ctx);
135 errno = ENOMEM;
136 return NULL;
137 }
138 if (pthread_cond_init(&ctx->queued_cond, NULL) != 0) {
139 pthread_mutex_destroy(&ctx->mutex);
140 free(ctx->workers);
141 free(ctx);
142 errno = ENOMEM;
143 return NULL;
144 }
145 if (pthread_cond_init(&ctx->complete_cond, NULL) != 0) {
146 pthread_cond_destroy(&ctx->queued_cond);
147 pthread_mutex_destroy(&ctx->mutex);
148 free(ctx->workers);
149 free(ctx);
150 errno = ENOMEM;
151 return NULL;
152 }
153 for (i = 0; i < ctx->worker_count; i++) {
154 if (pthread_create(&ctx->workers[i], NULL, aio_worker, ctx) != 0) {
155 ctx->worker_count = i;
156 pthread_mutex_lock(&ctx->mutex);
157 ctx->stopping = 1;
158 pthread_cond_broadcast(&ctx->queued_cond);
159 pthread_mutex_unlock(&ctx->mutex);
160 while (i > 0) pthread_join(ctx->workers[--i], NULL);
161 aio_release_ctx(ctx);
162 errno = EAGAIN;
163 return NULL;
164 }
165 }
166 return ctx;
167}

169static int
170aio_submit(brix_aio_ctx_t *ctx, int fd, void *buf, size_t count, off_t offset,
171 void (*callback)(int, ssize_t, void *), void *user_data, int write_op)
172{
173 brix_aio_job_t *job;
174
175 if (ctx == NULL || fd < 0 || buf == NULL || callback == NULL || offset < 0) {
176 errno = EINVAL;
177 return -1;
178 }
179 job = calloc(1, sizeof(*job));
180 if (job == NULL) return -1;
181 job->fd = fd;
182 job->buf = buf;
183 job->count = count;
184 job->offset = offset;
185 job->callback = callback;
186 job->user_data = user_data;
187 job->write_op = write_op != 0;
188 pthread_mutex_lock(&ctx->mutex);
189 if (ctx->stopping || ctx->outstanding >= ctx->max_entries) {
190 pthread_mutex_unlock(&ctx->mutex);
191 free(job);
192 errno = ctx->stopping ? ECANCELED : EAGAIN;
193 return -1;
194 }
195 ctx->outstanding++;
196 aio_push_job(&ctx->queued_head, &ctx->queued_tail, job);
197 pthread_cond_signal(&ctx->queued_cond);
198 pthread_mutex_unlock(&ctx->mutex);
199 return 0;
200}

231/* Wait for available completions and dispatch callbacks, respecting timeout. */
232int
233brix_aio_wait(brix_aio_ctx_t *ctx, int timeout_ms)
234{
235 struct timespec deadline;
236 int completed = 0;
237 int wait_rc = 0;
238
239 if (ctx == NULL) {
240 errno = EINVAL;
241 return -1;
242 }
243 if (timeout_ms >= 0) aio_deadline(&deadline, timeout_ms);
244 pthread_mutex_lock(&ctx->mutex);
245 while (ctx->complete_head == NULL && ctx->outstanding != 0 && wait_rc == 0) {
246 wait_rc = timeout_ms < 0
247 ? pthread_cond_wait(&ctx->complete_cond, &ctx->mutex)
248 : pthread_cond_timedwait(&ctx->complete_cond, &ctx->mutex,
249 &deadline);
250 }
251 while (ctx->complete_head != NULL) {
252 brix_aio_job_t *job = aio_pop_job(&ctx->complete_head,
253 &ctx->complete_tail);
254
255 ctx->outstanding--;
256 pthread_mutex_unlock(&ctx->mutex);
257 errno = job->error;
258 job->callback(job->fd, job->result, job->user_data);
259 free(job);
260 completed++;
261 pthread_mutex_lock(&ctx->mutex);
262 }
263 pthread_mutex_unlock(&ctx->mutex);
264 if (wait_rc != 0 && wait_rc != ETIMEDOUT) {
265 errno = wait_rc;
266 return -1;
267 }
268 return completed;
269}

274{
275 size_t i;
276
