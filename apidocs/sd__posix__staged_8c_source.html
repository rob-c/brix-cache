1/*
2 * sd_posix_staged.c — the POSIX Storage Driver's staged-write family.
3 *
4 * WHAT: The temp + atomic-rename staged upload (staged_open/_write/_commit/
5 * _abort/_path), split VERBATIM out of sd_posix_ns.c when the phase-107
6 * C6 typed publish precondition grew the commit past that file's size
7 * cap. The driver descriptor stays in sd_posix.c.
8 *
9 * WHY: Same reason as the sd_posix.c → sd_posix_ns.c cut: keep every unit
10 * under the file-size cap with zero behaviour change. The commit is the
11 * one body with new behaviour (C6): ABSENT publishes atomically via
12 * RENAME_NOREPLACE; MATCH_* is an honest advisory stat-compare that
13 * reports atomic = 0 (a posix rename cannot fold a compare into itself).
14 *
15 * HOW: Delegates to the shared brix_staged_* compat primitives exactly as
16 * before; the precondition evaluation reuses the one shared evaluator
17 * (brix_sd_precond_eval_stat, sd_batch_types.h) so the etag grammar
18 * never forks from the VFS compat arm.
19 */

48/* Create the target's parent-directory chain inside the store root before the
49 * O_EXCL temp is opened. A staged upload into a fresh subdirectory (e.g. a
50 * write-stage tier keyed on "/sub/file" whose store is a dedicated dir that
51 * has never seen "/sub") would otherwise ENOENT: brix_staged_open opens the
52 * temp adjacent to abspath and does NOT create parents. This mirrors the
53 * O_MKDIRPATH mkpath a direct write runs, so the store side of a subdirectory
54 * commit succeeds (the source-side origin creates its own chain via mkpath). */
55static void
56sd_posix_staged_mkparents(brix_sd_instance_t *inst, sd_posix_state_t *st,
57 const char *abspath)
58{
59 char parent[PATH_MAX];
60 char *slash;
61 size_t alen = ngx_strlen(abspath);
62
63 if (alen >= sizeof(parent)) {
64 return;
65 }
66 ngx_memcpy(parent, abspath, alen + 1);
67 slash = strrchr(parent, '/');
68 if (slash != NULL && slash > parent) {
69 *slash = '\0';
70 (void) brix_mkdir_recursive_confined_canon(inst->log,
71 st->root_canon, parent, 0755, NULL);
72 }
73}

75/* Phase-107 C5 admission: a declared final size preallocates the temp up
76 * front, so a store that cannot hold the object refuses the OPEN
77 * (ENOSPC/EDQUOT), not the commit hours of streaming later. Anything else
78 * (EOPNOTSUPP on an odd filesystem) is advisory per the seam contract - the
79 * open proceeds. Returns 0 to admit, or the errno that must fail the open. */
80static int
81sd_posix_staged_admit(brix_sd_instance_t *inst, sd_posix_staged_t *ps,
82 off_t declared_size)
83{
84 brix_sd_obj_t shell;
85
86 if (declared_size <= 0) {
87 return 0;
88 }
89 ngx_memzero(&shell, sizeof(shell));
90 shell.driver = inst->driver;
91 shell.inst = inst;
92 shell.fd = ps->staged.fd;
93 if (sd_posix_reserve(&shell, declared_size) != NGX_OK
94 && (errno == ENOSPC || errno == EDQUOT))
95 {
96 return errno;
97 }
98 return 0;
99}

101brix_sd_staged_t *
102sd_posix_staged_open(brix_sd_instance_t *inst, const char *final_path,
103 mode_t mode, off_t declared_size, int *err_out)
104{
105 sd_posix_state_t *st = inst->state;
106 brix_sd_staged_t *handle;
107 sd_posix_staged_t *ps;
108 char abspath[PATH_MAX];
109
110 /* Allocate the staged handle on the heap (ngx_calloc), NOT from inst->pool.
111 * staged_open runs in a cache-fill thread-pool thread (brix_cache_fill_*),
112 * but inst->pool is the shared, thread-UNSAFE backend pool the main thread
113 * also allocates from (sd_posix_open/opendir). Concurrent fills racing on
114 * inst->pool corrupt its `last` pointer -> a bad allocation whose memzero
115 * SIGSEGVs. The handle + ps are freed explicitly in staged_commit /
116 * staged_abort (the terminal ops — the driver vtable has no close). */
117 handle = ngx_calloc(sizeof(*handle), inst->log);
118 ps = ngx_calloc(sizeof(*ps), inst->log);
119 if (handle == NULL || ps == NULL) {
120 if (handle != NULL) { ngx_free(handle); }
121 if (ps != NULL) { ngx_free(ps); }
122 if (err_out != NULL) { *err_out = ENOMEM; }
123 return NULL;
124 }
125
126 /* The vtable contract is a root-RELATIVE key (leading slash), matching
127 * sd_posix_open and the non-POSIX drivers' staged_open. brix_staged_open
128 * (and _commit/_abort) work in ABSOLUTE paths under root_canon, so build the
129 * absolute final path here and store it for commit/abort. */
130 if ((size_t) snprintf(abspath, sizeof(abspath), "%s%s",
131 st->root_canon, final_path) >= sizeof(abspath))
132 {
133 ngx_free(handle);
134 ngx_free(ps);
135 if (err_out != NULL) { *err_out = ENAMETOOLONG; }
136 return NULL;
137 }
138
139 sd_posix_staged_mkparents(inst, st, abspath);
140
141 {
142 brix_staged_open_req_t oreq = {
143 .root_canon = st->root_canon,
144 .final_path = abspath,
145 .open_flags = O_WRONLY | O_CREAT | O_EXCL,
146 .mode = mode,
147 .attempts = 8,
148 };
149 if (brix_staged_open(inst->log, &oreq, &ps->staged) != NGX_OK) {
150 ngx_free(handle);
151 ngx_free(ps);
152 if (err_out != NULL) { *err_out = errno; }
153 return NULL;
154 }
155 }
156
157 {
158 int err = sd_posix_staged_admit(inst, ps, declared_size);
159
160 if (err != 0) {
161 brix_staged_abort(inst->log, st->root_canon, &ps->staged, 1);
162 ngx_free(handle);
163 ngx_free(ps);
164 if (err_out != NULL) { *err_out = err; }
165 errno = err;
166 return NULL;
167 }
168 }
169
170 ngx_cpystrn((u_char *) ps->final_path, (u_char *) abspath,
171 sizeof(ps->final_path));
172 handle->inst = inst;
173 handle->state = ps;
174 return handle;
175}

189ngx_int_t
190sd_posix_staged_commit(brix_sd_staged_t *st, brix_sd_precond_t *pre)
191{
192 sd_posix_staged_t *ps = st->state;
193 sd_posix_state_t *inst_st = st->inst->state;
194 ngx_int_t rc;
195
196 /* MATCH_* (phase-107 C6): advisory stat-compare against the final path
197 * before a plain rename. A posix rename cannot fold the compare into
198 * itself, so this is check-then-publish; pre->atomic stays 0 and the
199 * protocol layer must not claim RFC 7232 semantics for it (§3.5 — honest
200 * advisory over an emulation that lies). Evaluated BEFORE the rename so
201 * a refused precondition leaves the target untouched (the temp is then
202 * released by the caller's staged_abort). */
203 if (pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE
204 && pre->kind != BRIX_SD_PRECOND_ABSENT)
205 {
206 struct stat sb;
207
208 if (lstat(ps->final_path, &sb) != 0) {
209 errno = ECANCELED; /* MATCH_* against a missing target
210 * is a failed match, not ENOENT */
211 return NGX_ERROR;
212 }
213 if (brix_sd_precond_eval_stat(pre, sb.st_size, sb.st_mtime) != 0) {
214 return NGX_ERROR; /* errno = ECANCELED / ENOTSUP */
215 }
216 }
217
218 rc = brix_sd_precond_absent(pre)
219 ? brix_staged_commit_excl(st->inst->log, inst_st->root_canon,
220 &ps->staged, ps->final_path)
221 : brix_staged_commit(st->inst->log, inst_st->root_canon,
222 &ps->staged, ps->final_path);
223 /* Ownership contract (brix_vfs_staged_commit / sd_remote_staged_commit):
224 * free the heap-allocated handle ONLY on success. On failure the handle
225 * stays valid and the caller (stage_engine / brix_vfs) invokes
226 * staged_abort to release it — freeing here would double-free. */
227 if (rc != NGX_OK) {
228 if (brix_sd_precond_absent(pre) && errno == EEXIST) {
229 /* RENAME_NOREPLACE said no — a storage-decided (atomic) refusal,
230 * unless this host ever degraded (beneath.c latch); the C6
231 * advisory metric keys on this verdict after a failed commit. */
232 pre->atomic = !brix_renameat_noreplace_degraded();
233 }
234 return rc;
235 }
236 if (brix_sd_precond_absent(pre)) {
237 /* RENAME_NOREPLACE decided at the filesystem — unless this host has
238 * ever degraded to the check-then-act fallback (beneath.c latch). */
239 pre->atomic = !brix_renameat_noreplace_degraded();
240 }
241 ngx_free(ps);
242 ngx_free(st);
243 return NGX_OK;
244}

246void
247sd_posix_staged_abort(brix_sd_staged_t *st)
248{
249 sd_posix_staged_t *ps = st->state;
250 sd_posix_state_t *inst_st = st->inst->state;
251
252 brix_staged_abort(st->inst->log, inst_st->root_canon, &ps->staged, 1);
253 /* Terminal op — release the heap-allocated handle (see staged_open). */
254 ngx_free(ps);
255 ngx_free(st);
256}
