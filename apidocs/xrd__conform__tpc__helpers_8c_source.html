52/* ---- Verify a copied destination ----
53 * WHAT: Return one only for exact size and byte equality after a clean close.
54 * WHY: A successful transfer status alone does not prove destination integrity.
55 * HOW: 1. Stat and open. 2. Read bounded bytes. 3. Close and compare.
56 */
57int
58xrd_conform_tpc_destination_exact(brix_conn *connection, const char *path,
59 const uint8_t *expected, size_t expected_length, brix_status *status)
60{
61 uint8_t received[TPC_PAYLOAD_SIZE];
62 brix_statinfo info;
63 brix_file file;
64 ssize_t length;
65
66 if (brix_stat(connection, path, &info, status) != 0
67 || expected_length > sizeof(received)
68 || info.size != (long long) expected_length
69 || brix_file_open_read(connection, path, &file, status) != 0)
70 return 0;
71 length = brix_file_read(connection, &file, 0, received,
72 sizeof(received), status);
73 if (brix_file_close(connection, &file, status) != 0) return 0;
74 return length == (ssize_t) expected_length
75 && memcmp(received, expected, expected_length) == 0;
76}

79/* ---- Parse an unsigned TPC query reply ----
80 * WHAT: Return one for a complete decimal value, otherwise zero.
81 * WHY: Partial or negative capability values must not be accepted as evidence.
82 * HOW: 1. Trim whitespace. 2. Parse with overflow checks. 3. Require the end.
83 */
84int
85xrd_conform_tpc_unsigned(const char *value, unsigned long long *number)
86{
87 char *end = NULL;
88 unsigned long long parsed;
89
90 while (isspace((unsigned char) *value)) value++;
91 if (!isdigit((unsigned char) *value)) return 0;
92 errno = 0;
93 parsed = strtoull(value, &end, 10);
94 if (errno != 0 || end == value) return 0;
95 while (isspace((unsigned char) *end)) end++;
96 if (*end != '\0') return 0;
97 *number = parsed;
98 return 1;
99}
