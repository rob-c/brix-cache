1/*
2 * refresh.c — JWKS hot refresh via mtime-polling timer.
3 *
4 * WHAT: Implements a per-worker nginx timer that polls the JWKS file for mtime changes
5 * and reloads public keys in-place when the file has been updated. Old keys are
6 * freed only after a successful reload so that in-flight token validations are
7 * never left with a half-updated key set.
8 *
9 * WHY: Key rotation is a routine operation for WLCG/OIDC deployments — new signing
10 * keys are published to the JWKS file while nginx is running. Requiring a full
11 * nginx reload to pick up new keys would break token auth during the reload
12 * window. Mtime polling is the lightest-weight approach (no inotify dependency)
13 * and safe in nginx's single-threaded event loop: each worker has its own copy
14 * of the config structure, so the key array swap (memcpy + count update) is
15 * atomic within a worker with no cross-process locking needed.
16 *
17 * HOW: brix_token_jwks_schedule() copies a protocol-neutral refresh descriptor
18 * into the worker cycle and arms one event. Stream and HTTP adapters point
19 * the descriptor at the key arrays their validators already read. The
20 * handler stats the file, reloads only after an mtime change, swaps the key
21 * array after a complete parse, and always preserves the old set on error.
22 */

30/* Handler: called by the nginx event loop at each refresh interval */
31/*
32 * Poll the JWKS file once: reload and swap in a new key set only if the file
33 * exists, its mtime changed since the last poll, and it parses to >=1 key. Any
34 * miss is a no-op that preserves the current keys (a WARN is logged). Does not
35 * touch the refresh timer — the caller always re-arms it afterwards.
36 */
37typedef struct {
38 brix_jwks_refresh_spec_t spec;
39 ngx_event_t event;
40} brix_jwks_refresh_runtime_t;

42static void
43brix_token_jwks_try_reload(brix_jwks_refresh_runtime_t *runtime,
44 ngx_log_t *log)
45{
46 brix_jwks_refresh_spec_t *spec = &runtime->spec;
47 struct stat st;
48 brix_jwks_key_t new_keys[BRIX_MAX_JWKS_KEYS];
49 int new_count;
50
51 if (stat((const char *) spec->path.data, &st) != 0) {
52 ngx_log_error(NGX_LOG_WARN, log, ngx_errno,
53 "brix: JWKS stat failed for \"%s\" — will retry",
54 spec->path.data);
55 return;
56 }
57
58 if (st.st_mtime == spec->mtime) {
59 return; /* file unchanged — no reload needed */
60 }
61
62 new_count = brix_jwks_load(log,
63 (const char *) spec->path.data,
64 new_keys, BRIX_MAX_JWKS_KEYS);
65 if (new_count <= 0) {
66 ngx_log_error(NGX_LOG_WARN, log, 0,
67 "brix: JWKS reload from \"%s\" returned %d keys "
68 "— keeping old keys",
69 spec->path.data, new_count);
70 return;
71 }
72
73 /* Swap: free the old keys then install the new set */
74 brix_jwks_free(spec->keys, *spec->key_count);
75 ngx_memcpy(spec->keys, new_keys,
76 (size_t) new_count * sizeof(brix_jwks_key_t));
77 *spec->key_count = new_count;
78 spec->mtime = st.st_mtime;
79
80 ngx_log_error(NGX_LOG_NOTICE, log, 0,
81 "brix: JWKS refreshed from \"%s\" — %d key(s) loaded",
82 spec->path.data, new_count);
83}

85static void
86brix_token_jwks_refresh_handler(ngx_event_t *ev)
87{
88 brix_jwks_refresh_runtime_t *runtime = ev->data;

90 brix_token_jwks_try_reload(runtime, ev->log);

93 * never keep a draining worker alive (mirrors the FRM reaper pattern). */

99/* Public: schedule the refresh timer for one server block */
100/*
101 * WHAT: Allocates and arms a per-worker JWKS mtime-poll timer for the given server
102 * block config. Does nothing if token_jwks is empty or refresh interval is
103 * disabled (NGX_CONF_UNSET_MSEC) or zero.
104 *
105 * WHY: Must be called from init_process (after fork) so each worker gets its own
106 * ngx_event_t allocated from that worker's cycle pool and armed in that
107 * worker's event loop. Calling before fork would result in shared state across
108 * workers which is incorrect for nginx's process model.
109 *
110 * HOW: Allocates ngx_event_t via ngx_pcalloc, sets handler/data/log, stores pointer
111 * in worker-cycle storage, then calls ngx_add_timer with the configured
112 * interval in milliseconds.
113 */
114ngx_int_t
115brix_token_jwks_schedule(ngx_cycle_t *cycle,
116 const brix_jwks_refresh_spec_t *spec)
117{
118 brix_jwks_refresh_runtime_t *runtime;
119
120 if (cycle == NULL || spec == NULL || spec->path.len == 0
121 || spec->keys == NULL
122 || spec->key_count == NULL
123 || spec->interval == (ngx_msec_t) NGX_CONF_UNSET_MSEC
124 || spec->interval == 0)
125 {
126 return NGX_OK;
127 }
128
129 runtime = ngx_pcalloc(cycle->pool, sizeof(*runtime));
130 if (runtime == NULL) {
131 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
132 "brix: failed to allocate JWKS refresh timer");
133 return NGX_ERROR;
134 }
135
136 runtime->spec = *spec;
137 runtime->event.handler = brix_token_jwks_refresh_handler;
138 runtime->event.data = runtime;
139 runtime->event.log = cycle->log;
140 runtime->event.cancelable = 1;
141
142 ngx_add_timer(&runtime->event, runtime->spec.interval);
143
144 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
145 "brix: JWKS refresh timer started — interval=%Mms path=\"%s\"",
146 runtime->spec.interval, runtime->spec.path.data);
147 return NGX_OK;
148}
