1/*
2 * refresh.c — JWKS hot refresh via mtime-polling timer.
3 *
4 * WHAT: Implements a per-worker nginx timer that polls the JWKS file for mtime changes
5 * and reloads public keys in-place when the file has been updated. Old keys are
6 * freed only after a successful reload so that in-flight token validations are
7 * never left with a half-updated key set.
8 *
9 * WHY: Key rotation is a routine operation for WLCG/OIDC deployments — new signing
10 * keys are published to the JWKS file while nginx is running. Requiring a full
11 * nginx reload to pick up new keys would break token auth during the reload
12 * window. Mtime polling is the lightest-weight approach (no inotify dependency)
13 * and safe in nginx's single-threaded event loop: each worker has its own copy
14 * of the config structure, so the key array swap (memcpy + count update) is
15 * atomic within a worker with no cross-process locking needed.
16 *
17 * HOW: brix_token_jwks_schedule() copies a protocol-neutral refresh descriptor
18 * into the worker cycle and arms one event. Stream and HTTP adapters point
19 * the descriptor at the key arrays their validators already read. The
20 * handler stats the file, reloads only after an mtime change, swaps the key
21 * array after a complete parse, and always preserves the old set on error.
22 */

31/* Handler: called by the nginx event loop at each refresh interval */
32/*
33 * Poll the JWKS file once: reload and swap in a new key set only if the file
34 * exists, its mtime changed since the last poll, and it parses to >=1 key. Any
35 * miss is a no-op that preserves the current keys (a WARN is logged). Does not
36 * touch the refresh timer — the caller always re-arms it afterwards.
37 */
38typedef struct {
39 brix_jwks_refresh_spec_t spec;
40 ngx_event_t event;
41} brix_jwks_refresh_runtime_t;

43static void
44brix_token_jwks_try_reload(brix_jwks_refresh_runtime_t *runtime,
45 ngx_log_t *log)
46{
47 brix_jwks_refresh_spec_t *spec = &runtime->spec;
48 struct stat st;
49 brix_jwks_key_t new_keys[BRIX_MAX_JWKS_KEYS];
50 int new_count;
51
52 if (stat((const char *) spec->path.data, &st) != 0) {
53 BRIX_LOG(NGX_LOG_WARN, log, ngx_errno,
54 "brix: JWKS stat failed for \"%s\" — will retry",
55 spec->path.data);
56 return;
57 }
58
59 if (st.st_mtime == spec->mtime) {
60 return; /* file unchanged — no reload needed */
61 }
62
63 new_count = brix_jwks_load(log,
64 (const char *) spec->path.data,
65 new_keys, BRIX_MAX_JWKS_KEYS);
66 if (new_count <= 0) {
67 BRIX_LOG(NGX_LOG_WARN, log, 0,
68 "brix: JWKS reload from \"%s\" returned %d keys "
69 "— keeping old keys",
70 spec->path.data, new_count);
71 return;
72 }
73
74 /* Swap: free the old keys then install the new set */
75 brix_jwks_free(spec->keys, *spec->key_count);
76 ngx_memcpy(spec->keys, new_keys,
77 (size_t) new_count * sizeof(brix_jwks_key_t));
78 *spec->key_count = new_count;
79 spec->mtime = st.st_mtime;
80
81 BRIX_LOG(NGX_LOG_NOTICE, log, 0,
82 "brix: JWKS refreshed from \"%s\" — %d key(s) loaded",
83 spec->path.data, new_count);
84}

86static void
87brix_token_jwks_refresh_handler(ngx_event_t *ev)
88{
89 brix_jwks_refresh_runtime_t *runtime = ev->data;
90
91 brix_token_jwks_try_reload(runtime, ev->log);
92
93 /* Stop re-arming once the worker is shutting down so the poll timer can
94 * never keep a draining worker alive (mirrors the FRM reaper pattern). */
95 if (!ngx_exiting) {
96 ngx_add_timer(ev, runtime->spec.interval);
97 }
98}

100/* Public: schedule the refresh timer for one server block */
101/*
102 * WHAT: Allocates and arms a per-worker JWKS mtime-poll timer for the given server
103 * block config. Does nothing if token_jwks is empty or refresh interval is
104 * disabled (NGX_CONF_UNSET_MSEC) or zero.
105 *
106 * WHY: Must be called from init_process (after fork) so each worker gets its own
107 * ngx_event_t allocated from that worker's cycle pool and armed in that
108 * worker's event loop. Calling before fork would result in shared state across
109 * workers which is incorrect for nginx's process model.
110 *
111 * HOW: Allocates ngx_event_t via ngx_pcalloc, sets handler/data/log, stores pointer
112 * in worker-cycle storage, then calls ngx_add_timer with the configured
113 * interval in milliseconds.
114 */
115ngx_int_t
116brix_token_jwks_schedule(ngx_cycle_t *cycle,
117 const brix_jwks_refresh_spec_t *spec)
118{
119 brix_jwks_refresh_runtime_t *runtime;
120
121 if (cycle == NULL || spec == NULL || spec->path.len == 0
122 || spec->keys == NULL
123 || spec->key_count == NULL
124 || spec->interval == (ngx_msec_t) NGX_CONF_UNSET_MSEC
125 || spec->interval == 0)
126 {
127 return NGX_OK;
128 }
129
130 runtime = ngx_pcalloc(cycle->pool, sizeof(*runtime));
131 if (runtime == NULL) {
132 BRIX_LOG(NGX_LOG_ERR, cycle->log, 0,
133 "brix: failed to allocate JWKS refresh timer");
134 return NGX_ERROR;
135 }
136
137 runtime->spec = *spec;
138 runtime->event.handler = brix_token_jwks_refresh_handler;
139 runtime->event.data = runtime;
140 runtime->event.log = cycle->log;
141 runtime->event.cancelable = 1;
142
143 ngx_add_timer(&runtime->event, runtime->spec.interval);
144
145 BRIX_LOG(NGX_LOG_NOTICE, cycle->log, 0,
146 "brix: JWKS refresh timer started — interval=%Mms path=\"%s\"",
147 runtime->spec.interval, runtime->spec.path.data);
148 return NGX_OK;
149}

152brix_token_jwks_schedule_refresh(ngx_cycle_t *cycle,

154{

156
157 ngx_memzero(&spec, sizeof(spec));
158 spec.path = conf->common.token_jwks;
159 spec.keys = conf->jwks_keys;
160 spec.key_count = &conf->jwks_key_count;
161 spec.mtime = conf->jwks_mtime;
162 spec.interval = conf->common.token_jwks_refresh_interval;
163 if (brix_token_jwks_schedule(cycle, &spec) != NGX_OK) {
164 BRIX_LOG(NGX_LOG_ERR, cycle->log, 0,
165 "brix: JWKS refresh timer was not started");
166 }
