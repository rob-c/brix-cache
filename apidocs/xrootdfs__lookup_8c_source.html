7/* WHAT: Look up one child with a stable node id. WHY: Paths are only aliases.
8 * HOW: 1. Resolve and stat as caller. 2. Reserve bookkeeping. 3. Reply once. */
9void
10xfs_ll_lookup(fuse_req_t req, fuse_ino_t parent, const char *name)
11{
12 xfs_session *session = xfs_req_lock(req);
13 char path[XRDC_PATH_MAX];
14 xfs_node_ticket ticket = {0};
15 int rc = xfs_req_child(session, parent, name, path);
16 if (rc == 0) {
17 rc = xfs_nodes_prepare(&session->nodes, path, &ticket);
18 }
19 if (rc == 0) {
20 rc = xfs_ll_entry(req, session, path, NULL, &ticket, NULL);
21 }
22 xfs_nodes_cancel(&session->nodes, &ticket);
23 pthread_mutex_unlock(&session->mutex);
24 if (rc != 0) {
25 fuse_reply_err(req, -rc);
26 }
27}

43/* WHAT: Stat a specific open handle, returning negative errno.
44 * WHY: Open-time metadata cannot observe later writes or staged-file size.
45 * HOW: 1. Lock handle. 2. Stat its actual remote fd. 3. Include pending writes.
46 * 4. Preserve the kernel identity and unlock on every outcome. */
47static int
48xfs_ll_handle_stat(afh *handle, fuse_ino_t ino, struct stat *attr)
49{
50 pthread_mutex_lock(&handle->lock);
51 *attr = handle->attr;
52 int rc = 0;
53 if (handle->mf) {
54 brix_statinfo info;
55 brix_status status;
56 brix_status_clear(&status);
57 rc = brix_mfile_stat(handle->mf, &info, &status);
58 if (rc == 0) {
59 xfs_fill_stat(&info, attr);
60 } else {
61 rc = xfs_err(&status);
62 }
63 }
64 off_t pending_end = handle->io.wbuf_off + (off_t) handle->io.wbuf_len;
65 if (rc == 0 && handle->io.wbuf_len && attr->st_size < pending_end) {
66 attr->st_size = pending_end;
67 }
68 attr->st_ino = (ino_t) ino;
69 pthread_mutex_unlock(&handle->lock);
70 return rc;
71}

73/* WHAT: Select only this caller's live handle, or NULL.
74 * WHY: BSD GETATTR can omit fh, but another user's open must not lend authority.
75 * HOW: 1. Walk the mount-locked inode owner list. 2. Require the exact caller uid. */
76static afh *
77xfs_ll_caller_handle(fuse_req_t req, xfs_node *node)
78{
79 for (afh *handle = node->open_handles; handle; handle = handle->node_next) {
80 if (handle->caller_uid == xfs_req_uid(req)) {
81 return handle;
82 }
83 }
84 return NULL;
85}

87/* WHAT: Fetch authorized attributes while preserving the kernel node id.
88 * WHY: An unlinked open handle remains usable without reopening a stale name.
89 * HOW: 1. Resolve inode. 2. Use handle metadata or caller-authorized stat.
90 * 3. Replace only the transport inode with the mount identity. */
91static int
92xfs_ll_stat(fuse_req_t req, xfs_session *session, fuse_ino_t ino,
93 struct fuse_file_info *fi, struct stat *attr)
94{
95 xfs_node *node = xfs_nodes_find(&session->nodes, ino);
96 if (!node) {
97 return -ESTALE;
98 }
99 afh *handle = fi && fi->fh ? (afh *) (uintptr_t) fi->fh
100 : xfs_ll_caller_handle(req, node);
101 if (handle) {
102 return xfs_ll_handle_stat(handle, ino, attr);
103 }
104 const char *path = xfs_node_path(node);
105 if (!path) {
106 return -ENOENT;
107 }
108 int rc = xfs_getattr(xfs_req_uid(req), path, attr, NULL);
109 if (rc == 0 && node->backend && (uint64_t) attr->st_ino != node->backend) {
110 return -ESTALE; /* the path now names a different backing inode */
111 }
112 attr->st_ino = (ino_t) ino;
113 return rc;
114}

132/* WHAT: Apply requested metadata fields, returning first error.
133 * WHY: Low-level setattr combines the existing high-level operations.
134 * HOW: 1. Mode. 2. Owner fields independently. 3. Size. 4. Requested times. */
135static int
136xfs_ll_apply_attr(uid_t uid, const char *path, struct stat *attr, int fields,
137 struct fuse_file_info *fi)
138{
139 int rc = 0;
140 if (fields & FUSE_SET_ATTR_MODE) {
141 rc = xfs_chmod(uid, path, attr->st_mode, fi);
142 }
143 if (rc == 0 && (fields & (FUSE_SET_ATTR_UID | FUSE_SET_ATTR_GID))) {
144 rc = xfs_chown(uid, path, fields & FUSE_SET_ATTR_UID ? attr->st_uid : (uid_t) -1,
145 fields & FUSE_SET_ATTR_GID ? attr->st_gid : (gid_t) -1, fi);
146 }
147 if (rc == 0 && (fields & FUSE_SET_ATTR_SIZE)) {
148 rc = xfs_truncate(uid, path, attr->st_size, fi);
149 }
150 if (rc == 0 && (fields & (FUSE_SET_ATTR_ATIME | FUSE_SET_ATTR_MTIME
151 | FUSE_SET_ATTR_ATIME_NOW | FUSE_SET_ATTR_MTIME_NOW))) {
152 /* The root:// extension carries whole seconds, not host-private stat
153 * nanosecond members (which differ on Darwin). */
154 struct timespec times[2] = {{attr->st_atime, 0}, {attr->st_mtime, 0}};
155 if (!(fields & FUSE_SET_ATTR_ATIME)) { times[0].tv_nsec = UTIME_OMIT; }
156 if (!(fields & FUSE_SET_ATTR_MTIME)) { times[1].tv_nsec = UTIME_OMIT; }
157 if (fields & FUSE_SET_ATTR_ATIME_NOW) { times[0].tv_nsec = UTIME_NOW; }
158 if (fields & FUSE_SET_ATTR_MTIME_NOW) { times[1].tv_nsec = UTIME_NOW; }
159 rc = xfs_utimens(uid, path, times, fi);
160 }
161 return rc;
162}
