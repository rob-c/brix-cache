89void
90xrd_conform_probe_concurrent_write(xrd_conform_execution *execution,
91 brix_conn *connection, const xrd_conform_scratch_object *object)
92{
93 uint8_t left[4096], right[4096], actual[8192];
94 conform_write_job jobs[2];
95 pthread_t threads[2];
96 brix_streamset streams;
97 brix_file file;
98 brix_status status;
99 char reason[64];
100 int created = 0;
101 int ok;
102
103 memset(left, 'L', sizeof(left));
104 memset(right, 'R', sizeof(right));
105 brix_status_clear(&status);
106 if (open_write(execution, connection, object, 1, &file, &status) != 0) {
107 xrd_conform_grade_status(execution, "WRT-011", 0, "concurrent-extents-landed",
108 "concurrent-write-open-failed", &status);
109 return;
110 }
111 if (brix_streams_open(&streams, connection, 3, &status) < 2) {
112 close_file(execution, connection, &file, &status);
113 xrd_conform_result_set(execution, "WRT-011",
114 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
115 "two-bound-write-streams-unavailable", NULL, status.msg);
116 return;
117 }
118 if (!xrd_conform_reserve(execution, sizeof(actual), reason,
119 sizeof(reason))) {
120 brix_streams_close(&streams);
121 close_file(execution, connection, &file, &status);
122 xrd_conform_result_set(execution, "WRT-011",
123 BRIX_CONFORM_VERDICT_INCONCLUSIVE, reason, NULL, NULL);
124 return;
125 }
126 memset(jobs, 0, sizeof(jobs));
127 jobs[0] = (conform_write_job) {
128 &streams.sec[0], &file, 0, left, sizeof(left), -1, { 0 }
129 };
130 jobs[1] = (conform_write_job) {
131 &streams.sec[1], &file, sizeof(left), right, sizeof(right), -1, { 0 }
132 };
133 while (created < 2 && pthread_create(&threads[created], NULL,
134 run_write_job, &jobs[created]) == 0) created++;
135 for (int index = 0; index < created; index++)
136 pthread_join(threads[index], NULL);
137 close_file(execution, connection, &file, &status);
138 brix_streams_close(&streams);
139 ok = concurrent_write_matches(execution, connection, object, jobs,
140 created, left, sizeof(left), right, sizeof(right), actual, &status);
141 xrd_conform_grade_status(execution, "WRT-011", ok, "concurrent-extents-landed",
142 created == 2 ? "concurrent-write-readback-mismatch"
143 : "concurrent-write-thread-failed",
144 concurrent_write_status(jobs, created, &status));
145}

156xrd_conform_probe_truncated_write(xrd_conform_execution *execution,
157 brix_conn *connection, const xrd_conform_scratch_object *object)
158{
159 static const uint8_t baseline[] = "complete-baseline";
160 static const uint8_t prefix[] = "partial";
161 uint8_t actual[sizeof(baseline)];
162 ClientWriteRequest request;
163 xrdw_write_req_t fields = { .offset = 0, .pathid = 0 };
164 brix_payload_ext payload = {
165 .data = prefix, .len = sizeof(prefix), .dlen = sizeof(prefix) + 17U
166 };
167 brix_streamset streams;
168 brix_file file;
169 brix_statinfo info;
170 brix_status status;
171 uint16_t stream_id;
172 int rc;
173
174 brix_status_clear(&status);
175 rc = open_write(execution, connection, object, 1, &file, &status);
176 if (rc == 0 && xrd_conform_reserve_status(execution, sizeof(baseline), &status))
177 rc = brix_file_write(connection, &file, 0, baseline, sizeof(baseline),
178 &status);
179 if (rc == 0) rc = close_file(execution, connection, &file, &status);
180 if (rc != 0) {
181 xrd_conform_grade_status(execution, "WRT-010", 0, "truncated-write-discarded",
182 "truncated-write-baseline-failed", &status);
183 return;
184 }
185 if (open_write(execution, connection, object, 2, &file, &status) != 0) {
186 xrd_conform_grade_status(execution, "WRT-010", 0, "truncated-write-discarded",
187 "truncated-write-update-open-failed", &status);
188 return;
189 }
190 if (brix_streams_open(&streams, connection, 2, &status) < 1) {
191 close_file(execution, connection, &file, &status);
192 xrd_conform_result_set(execution, "WRT-010",
193 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
194 "bound-disposable-stream-unavailable", NULL, status.msg);
195 return;
196 }
197 if (!xrd_conform_reserve_status(execution, payload.dlen, &status)) {
198 brix_streams_close(&streams);
199 close_file(execution, connection, &file, &status);
200 xrd_conform_result_set(execution, "WRT-010",
201 BRIX_CONFORM_VERDICT_INCONCLUSIVE, status.msg, NULL, NULL);
202 return;
203 }
204 memcpy(fields.fhandle, file.fhandle, sizeof(fields.fhandle));
205 memset(&request, 0, sizeof(request));
206 request.requestid = htons(kXR_write);
207 xrdw_write_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
208 rc = brix_send_ext(&streams.sec[0], &request, &payload, &stream_id,
209 &status);
210 brix_streams_close(&streams);
211 if (close_file(execution, connection, &file, &status) != 0) rc = -1;
212 if (rc == 0 && read_exact(execution, connection, object, actual,
213 sizeof(actual), &status) != 0) rc = -1;
214 if (rc == 0 && (!xrd_conform_reserve_status(execution, 0, &status)
215 || brix_stat(connection, object->path, &info, &status) != 0))
216 rc = -1;
217 xrd_conform_grade_status(execution, "WRT-010",
218 truncated_write_intact(rc, &info, actual, baseline, sizeof(baseline)),
219 "truncated-write-discarded", "truncated-write-partially-applied",
220 &status);
221}

223/* WHAT: Advertise a write one byte beyond the server's streaming ceiling.
224 * WHY: A path-sized framing gate or a silent reset does not define the write
225 * payload limit and leaves clients unable to classify the refusal.
226 * HOW: Use a disposable authenticated connection, send only the request header,
227 * and require kXR_ArgTooLong before the server closes the unsynchronizable
228 * stream. */
229void
230xrd_conform_probe_oversized_write(xrd_conform_execution *execution)
231{
232 ClientWriteRequest request;
233 brix_payload_ext payload = {
234 .data = NULL, .len = 0, .dlen = BRIX_MAX_WRITE_STREAM + 1U
235 };
236 brix_resp_out response;
237 brix_conn connection;
238 brix_opts opts = execution->run->opts;
239 brix_url url;
240 brix_status status;
241 uint8_t *body = NULL;
242 uint32_t body_length = 0;
243 uint16_t stream_id;
244 uint16_t wire_status = 0;
245 int rc = -1;
246
247 brix_status_clear(&status);
248 opts.capture = NULL;
249 if (brix_endpoint_parse(execution->target, &url, &status) == 0
250 && brix_connect(&connection, &url, &opts, &status) == 0) {
251 memset(&request, 0, sizeof(request));
252 request.requestid = htons(kXR_write);
253 response.status = &wire_status;
254 response.body = &body;
255 response.blen = &body_length;
256 if (xrd_conform_reserve_status(execution, 0, &status)
257 && brix_send_ext(&connection, &request, &payload, &stream_id,
258 &status) == 0)
259 rc = brix_recv(&connection, stream_id, &response, &status);
260 brix_close(&connection);
261 }
262 free(body);
263 xrd_conform_grade_status(execution, "WRT-008", rc != 0 && status.kxr == kXR_ArgTooLong,
264 "oversized-write-refused-with-ArgTooLong",
265 "oversized-write-lacked-defined-error", &status);
266}
