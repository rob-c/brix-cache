34/* WHAT: Enforce/query native no-new-privileges, refusing every other request.
35 * WHY: Successful no-op prctl emulation would falsely attest confinement.
36 * HOW: Map the two supported operations to procctl and propagate errors. */
37int
38brix_plat_prctl(int operation, unsigned long argument)
39{
40 int value;
41
42 if (operation == PR_SET_NO_NEW_PRIVS && argument == 1) {
43 value = PROC_NO_NEW_PRIVS_ENABLE;
44 return procctl(P_PID, 0, PROC_NO_NEW_PRIVS_CTL, &value);
45 }
46 if (operation == PR_GET_NO_NEW_PRIVS) {
47 if (procctl(P_PID, 0, PROC_NO_NEW_PRIVS_STATUS, &value) != 0) {
48 return -1;
49 }
50 return value == PROC_NO_NEW_PRIVS_ENABLE;
51 }
52 errno = ENOTSUP;
53 return -1;
54}

95 * HOW: Read back all uid slots and fail closed on root or partial drops. */

135/* WHAT: Restrict helper IPC, stdio and optional logging descriptors.
136 * WHY: They carry bytes only, never ioctl, exec or new network authority.
137 * HOW: Apply the bounded rights sets through the common limit helper. */
138static int
139broker_limit_channels(int channel, int rootfd, int logfd)
140{
141 cap_rights_t rights;
142
143 cap_rights_init(&rights, CAP_READ, CAP_WRITE, CAP_EVENT, CAP_FSTAT);
144 if (broker_limit_fd(channel, &rights, 0) != 0) {
145 return -1;
146 }
147 cap_rights_init(&rights, CAP_READ, CAP_WRITE, CAP_FSTAT);
148 for (int fd = 0; fd <= 2; fd++) {
149 if (broker_limit_fd(fd, &rights, 0) != 0) {
150 return -1;
151 }
152 }
153 if (logfd > 2 && logfd != channel && logfd != rootfd) {
154 cap_rights_init(&rights, CAP_WRITE, CAP_FSTAT);
155 if (broker_limit_fd(logfd, &rights, 0) != 0) {
156 return -1;
157 }
158 }
159 return 0;

162/* WHAT: Seal a fixed-identity helper into descriptor-only capability mode.
163 * WHY: It must neither regain root nor reach ambient paths, exec or devices.
164 * HOW: Verify identity, restrict descriptor rights, then enter and read back
165 * the irreversible native capability-mode flag. */
166int
167brix_plat_broker_confine(int channel, int rootfd, int logfd)

181 CAP_FCHMODAT, CAP_FCHOWNAT, CAP_FUTIMESAT, CAP_LINKAT_SOURCE,
