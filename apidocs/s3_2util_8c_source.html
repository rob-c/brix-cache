22/*
23 * Build a transient S3 VFS request descriptor for an already-resolved confined
24 * path. Shared by the PUT (put.c) and POST-object (post_object.c) write paths,
25 * which previously carried byte-identical private copies. Pulls pool/log, the
26 * TLS flag, and the authenticated identity from the request; roots and the
27 * write gate from the loc-conf. rootfd stays -1 (transient confined open).
28 */
29void
30s3_build_vfs_ctx(ngx_http_request_t *r, const char *fs_path,
31 ngx_http_s3_loc_conf_t *cf, brix_vfs_ctx_t *vctx)
32{
33 ngx_http_s3_req_ctx_t *s3ctx;
34 int is_tls = 0;
35
36 s3ctx = ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
37
38 is_tls = brix_http_request_is_tls(r);
39
40 brix_vfs_ctx_init(vctx, r->pool, r->connection->log, BRIX_PROTO_S3,
41 cf->common.root_canon, cf->common.cache_root_canon,
42 brix_vfs_policy_from_write_enable(cf->common.allow_write),
43 is_tls, (s3ctx != NULL) ? s3ctx->identity : NULL, fs_path);
44 brix_vfs_ctx_bind_backend_cred(vctx,
45 &cf->common.storage_credential_dir,
46 cf->common.storage_credential_fallback);
47 /* Phase-2 T9: S3 access keys have no native x509 — this is the primary
48 * beneficiary of opt-in minting (see fs/backend/cred_mint.h). No-op
49 * unless brix_storage_credential_mint_ca is configured. */
50 brix_vfs_ctx_bind_backend_mint(vctx,
51 &cf->common.storage_credential_mint_ca_cert,
52 &cf->common.storage_credential_mint_ca_key,
53 cf->common.storage_credential_mint_ttl);
54 s3_vfs_bind_deleg(r, cf, vctx);
55 /* phase-110 W3: S3 write/multipart ctx builder (event-loop handlers) —
56 * bind the request's I/O monitor so $brix_bytes_received / $brix_op /
57 * $brix_status report on S3 PUT/POST exactly as on WebDAV PUT. */
58 brix_http_monitor_bind(r, vctx);
59}

61/* ---- s3_vfs_bind_deleg -----------------------------------------------------
62 *
63 * WHAT: Bind the request's captured forwardable credential (bearer JWT and/or
64 * user-supplied full x509 proxy PEM) onto a cred-bound VFS ctx, using the
65 * export's resolved delegation mode. See s3.h for the contract.
66 *
67 * WHY: Called at every S3 brix_vfs_ctx_bind_backend_cred site so a delegated
68 * export authenticates the backend leg AS the inbound user rather than the
69 * shared service credential. The bytes were captured once at the auth gate
70 * (bearer in s3_verify_bearer, proxy in handler.c's header capture) and
71 * stashed on the S3 req ctx; here they are handed to the VFS.
72 *
73 * HOW: Reads cf->common.backend_delegation as the mode and the req ctx's
74 * bearer_token / deleg_proxy_pem as the bytes; brix_vfs_deleg_bind is a
75 * no-op for SELECT mode or when nothing was captured. Mirrors
76 * webdav_vfs_bind_deleg: a bearer forwarded VERBATIM (anything except
77 * EXCHANGE-with-endpoint) must pass the backend audience gate first
78 * (phase-70 §5.2 / P90-70.9), and EXCHANGE conf + the per-conf
79 * minted-token cache slot are stamped after the bind. */
80void
81s3_vfs_bind_deleg(ngx_http_request_t *r,
82 ngx_http_s3_loc_conf_t *cf, brix_vfs_ctx_t *vctx)
83{
84 ngx_http_s3_req_ctx_t *s3ctx;
85 const ngx_str_t *bearer;
86
87 brix_http_vfs_bind_authz(r, &cf->common, cf->authdb_final, NULL, vctx);
88 if (cf->common.backend_delegation == BRIX_CRED_SELECT) {
89 return;
90 }
91
92 s3ctx = ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
93 bearer = brix_proto_deleg_gate_bearer(
94 (s3ctx != NULL) ? &s3ctx->bearer_token : NULL,
95 &cf->common, r->connection->log);
96
97 (void) brix_vfs_deleg_bind(r->pool, vctx,
98 (enum brix_cred_mode) cf->common.backend_delegation,
99 bearer,
100 (s3ctx != NULL) ? &s3ctx->deleg_proxy_pem : NULL);
101
102 brix_proto_deleg_stamp_conf(vctx, &cf->common);
103}

125/*
126 * s3_resolve_key_ex — resolve an S3 object key to a filesystem path, keeping
127 * the resolver's own status.
128 *
129 * Steps:
130 * 1. Strip leading slashes from the key (S3 URLs have /bucket/key).
131 * 2. Prepend exactly one slash so decoded_path starts with /.
132 * 3. Call brix_http_resolve_path_ex() which canonicalizes, URL-decodes,
133 * and confines the path to root.
134 *
135 * allow_internal: pass cf->common.cache_store_endpoint. Non-zero on a trusted
136 * cache-store endpoint permits internal sidecar keys (<key>.cinfo etc.); 0
137 * everywhere else keeps them invisible (default-deny).
138 *
139 * Returns 0 when the path is confined, else the resolver's HTTP status: 403 for
140 * an escape/malformed component, 404 for a reserved internal name, 414 for an
141 * overflow. Those three MUST stay apart: compat/path.c chose 404 for the
142 * reserved name precisely "so the response does not distinguish an internal name
143 * from a genuinely absent one", and collapsing it into a 403 hands back exactly
144 * the inference the 404 was chosen to prevent.
145 */
146int
147s3_resolve_key_ex(const char *root, const char *key, char *out, size_t outsz,
148 unsigned allow_internal)
149{
150 char key_path[PATH_MAX];
151 int n;
152
153 /* Strip leading slashes; prepend exactly one so decoded_path starts with /. */
154 while (*key == '/')
155 key++;
156
157 n = snprintf(key_path, sizeof(key_path), "/%s", key);
158 if (n < 0 || (size_t) n >= sizeof(key_path))
159 return 414;
160
161 return brix_http_resolve_path_ex(root, key_path, out, outsz,
162 allow_internal ? 1u : 0u);
163}

177/*
178 * s3_resolve_key_error — the single mapping from a resolver status to the S3
179 * error triple plus the event it is booked against.
180 *
181 * The 404 arm carries "NoSuchKey" and BRIX_S3_EVENT_NO_SUCH_KEY because that is
182 * what a genuinely absent key answers: an operator watching brix_s3_events_total
183 * must not be able to separate the two either (invariant 8 forbids a label that
184 * would let them, and the whole point is that nothing should).
185 */
186void
187s3_resolve_key_error(int rc, s3_key_error_t *err)
188{
189 if (rc == 404) {
190 err->status = NGX_HTTP_NOT_FOUND;
191 err->code = "NoSuchKey";
192 err->message = "The specified key does not exist.";
193 err->event = BRIX_S3_EVENT_NO_SUCH_KEY;
194 return;
195 }
196 if (rc == 414) {
197 err->status = NGX_HTTP_BAD_REQUEST;
198 err->code = "KeyTooLongError";
199 err->message = "Your key is too long.";
200 err->event = BRIX_S3_EVENT_INVALID_URI;
201 return;
202 }
203 err->status = NGX_HTTP_FORBIDDEN;
204 err->code = "AccessDenied";
205 err->message = "Access Denied.";
206 err->event = BRIX_S3_EVENT_ACCESS_DENIED;
207}

269/*
270 * s3_object_crc64nvme_b64 — compute (or cache-read) an object's CRC-64/NVME and
271 * base64-encode it in the AWS x-amz-checksum-crc64nvme wire form.
272 *
273 * WHAT: Produces the 12-char base64 of the 8 big-endian CRC-64/NVME bytes for the
274 * open fd into out (needs >= 13 bytes). Uses the xattr integrity cache, so
275 * the value computed at upload time is reused on later reads.
276 * WHY: AWS S3 clients (SDK/CLI default integrity) send and expect this checksum;
277 * it is base64-of-bytes, NOT hex like the root:///WebDAV digest forms.
278 * HOW: brix_integrity_get_fd("crc64nvme") returns the 16-hex value (from cache
279 * or freshly computed); strtoull → uint64 → 8 big-endian bytes →
280 * ngx_encode_base64. cache_only=1 declines (NGX_DECLINED) on a cache miss
281 * instead of paying a full-file read — used on the GET/HEAD echo path.
282 * Returns NGX_OK (out filled), NGX_DECLINED (cache_only miss), or NGX_ERROR.
283 */
284ngx_int_t
285s3_object_crc64nvme_b64(ngx_http_request_t *r, int fd, const char *path,
286 ngx_flag_t cache_only, char *out, size_t outsz)
287{
288 brix_integrity_info_t info;
289 brix_integrity_opts_t iopts;
290 uint64_t crc;
291 unsigned char be[8];
292 ngx_str_t src, dst;
293 ngx_int_t rc;
294 int i;
295
296 if (out == NULL || outsz < (size_t) (ngx_base64_encoded_length(8) + 1)) {
297 return NGX_ERROR;
298 }
299
300 ngx_memzero(&iopts, sizeof(iopts));
301 iopts.allow_xattr_cache = 1;
302 iopts.update_xattr_cache = cache_only ? 0 : 1;
303 iopts.require_regular_file = 1;
304 iopts.no_compute = cache_only ? 1 : 0;
305 /* phase-105: see s3_checksum_b64 — the cache write is an export mutation. */
306 iopts.mutation_policy = brix_vfs_policy_from_write_enable(
307 ((ngx_http_s3_loc_conf_t *)
308 ngx_http_get_module_loc_conf(r, ngx_http_brix_s3_module))
309 ->common.allow_write);
310 iopts.proto = BRIX_PROTO_S3;
311
312 rc = brix_integrity_get_fd(r->connection->log, fd, NULL, path, "crc64nvme",
313 &iopts, &info);
314 if (rc != NGX_OK) {
315 return rc; /* NGX_DECLINED (cache-only miss) or NGX_ERROR */
316 }
317
318 /* info.hex is the 16-char lowercase crc64nvme; convert to the raw u64 and
319 * emit the 8 big-endian bytes base64-encoded (the AWS wire format). */
320 crc = (uint64_t) strtoull(info.hex, NULL, 16);
321 for (i = 0; i < 8; i++) {
322 be[i] = (unsigned char) (crc >> (56 - 8 * i));
323 }
324 src.data = be;
325 src.len = sizeof(be);
326 dst.data = (u_char *) out;
327 ngx_encode_base64(&dst, &src);
328 out[dst.len] = '\0';
329 return NGX_OK;
330}
