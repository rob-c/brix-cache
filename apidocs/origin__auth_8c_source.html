36/* Frame one kXR_auth request (ClientAuthRequest header, credtype = the 4-byte
37 * protocol id, + credential payload) on the connector stream. Shared by the ztn
38 * and sss auth helpers here and the gsi helpers in origin_auth_gsi.c (prototype in
39 * cache_internal.h). Returns 0, or -1 (errno set). */
40int
41cache_origin_send_kxr_auth(brix_cache_origin_conn_t *oc, const char credtype[4],
42 const u_char *payload, uint32_t plen)
43{
44 ClientAuthRequest req;
45
46 ngx_memzero(&req, sizeof(req));
47 req.streamid[1] = 1; /* the connector stream */
48 req.requestid = htons(kXR_auth);
49 ngx_memcpy(req.credtype, credtype, 4);
50 req.dlen = htonl((kXR_int32) plen);
51
52 if (brix_cache_io_send(oc, &req, sizeof(req)) != 0
53 || (plen > 0 && brix_cache_io_send(oc, payload, plen) != 0))
54 {
55 return -1;
56 }
57 return 0;
58}

60/* XrdSecProtocolztn credential wire format (stock XRootD, byte-frozen).
61 *
62 * The ztn credential is an XrdSecProtocolztn::TokenResp: an 8-byte TokenHdr, a
63 * 2-byte big-endian length, then the token followed by one NUL terminator:
64 *
65 * off 0..3 id[4] = "ztn\0" (NUL-terminated protocol id)
66 * off 4 ver = 0 (XrdSecProtocolztn::ztnVersion)
67 * off 5 opr = 'T' (TokenHdr::IsTkn — "here is a token")
68 * off 6..7 rsvd[2] = 0, 0
69 * off 8..9 len = htons(tsz + 1) (token length INCLUDING its trailing NUL)
70 * off 10.. tkn[tsz] + one NUL byte
71 *
72 * The stock server (Authenticate) reads opr at off 5 to route the request,
73 * ntohs(len) at off 8, and the token at off 10 (pfxLen = sizeof(TokenHdr) +
74 * sizeof(uint16_t) = 10). It requires len >= 1, len's byte at (10 + len - 1) to
75 * be NUL, and (10 + len) <= credential size. The whole credential is
76 * (10 + tsz + 1) bytes. */
77#define BRIX_ZTN_VERSION 0
78#define BRIX_ZTN_OPR_ISTKN 'T'
79#define BRIX_ZTN_PREFIX_LEN 10 /* 8-byte TokenHdr + 2-byte length */
80
81/* cache_origin_build_ztn_credential — frame a bearer token as a stock XrdSecztn
82 * TokenResp credential.
83 *
84 * WHAT: allocate and fill the (10 + token->len + 1)-byte TokenResp blob for the
85 * given token and return it via *out / *outlen.
86 * WHY : the earlier code sent a raw "ztn\0" + token blob (the format OUR own
87 * parser tolerates); a stock XrdSecProtocolztn reads byte 5 as the opr
88 * code, sees a token character instead of 'T', and rejects the exchange
89 * with "Invalid ztn response code". This produces the exact bytes stock
90 * expects.
91 * HOW : write the 8-byte header (id/ver/opr/rsvd), the big-endian token length
92 * (token->len + 1, the trailing NUL counts), the token, and one NUL.
93 * Returns 0 with *out malloc'd (caller frees), -1 on allocation failure. */
94static int
95cache_origin_build_ztn_credential(const ngx_str_t *token, u_char **out,
96 size_t *outlen)
97{
98 u_char *blob;
99 size_t blen;
100 uint16_t tlen_be;
101
102 blen = BRIX_ZTN_PREFIX_LEN + token->len + 1; /* hdr + len + token + NUL */
103 blob = malloc(blen);
104 if (blob == NULL) {
105 return -1;
106 }
107
108 ngx_memcpy(blob, "ztn", 4); /* id[4], incl. trailing NUL */
109 blob[4] = BRIX_ZTN_VERSION; /* ver */
110 blob[5] = BRIX_ZTN_OPR_ISTKN; /* opr = IsTkn */
111 blob[6] = 0; /* rsvd[0] */
112 blob[7] = 0; /* rsvd[1] */
113
114 tlen_be = htons((uint16_t) (token->len + 1)); /* length includes the NUL */
115 ngx_memcpy(blob + 8, &tlen_be, sizeof(tlen_be));
116
117 ngx_memcpy(blob + BRIX_ZTN_PREFIX_LEN, token->data, token->len);
118 blob[BRIX_ZTN_PREFIX_LEN + token->len] = 0; /* required trailing NUL */
119
120 *out = blob;
121 *outlen = blen;
122 return 0;
123}

125/* brix_cache_origin_auth_ztn — present a WLCG/SciToken bearer to the origin via
126 * the XrdSecztn protocol after a kXR_login advertised "&P=ztn". The exchange is a
127 * single-round kXR_auth: credtype "ztn\0", payload = the stock XrdSecztn
128 * TokenResp (see cache_origin_build_ztn_credential for the exact framing). The
129 * server (which advertised its version/maxtsz in the login parms) validates the
130 * token and replies kXR_ok. Returns 0 on a kXR_ok auth, -1 otherwise (t error
131 * set). §14/C-3. */
132int
133brix_cache_origin_auth_ztn(brix_cache_fill_t *t,
134 brix_cache_origin_conn_t *oc, const ngx_str_t *token)
135{
136 u_char *blob;
137 size_t blen;
138 uint16_t status;
139 uint32_t dlen;
140 u_char *body;
141
142 if (cache_origin_build_ztn_credential(token, &blob, &blen) != 0) {
143 brix_cache_set_error(t, kXR_NoMemory, 0,
144 "cache origin ztn payload allocation failed");
145 return -1;
146 }
147
148 if (cache_origin_send_kxr_auth(oc, "ztn", blob, (uint32_t) blen) != 0) {
149 free(blob);
150 brix_cache_set_error(t, kXR_ServerError, errno,
151 "cache origin ztn auth write failed");
152 return -1;
153 }
154 free(blob);
155
156 body = NULL;
157 if (brix_cache_read_response(t, oc, &status, &body, &dlen, 4096) != 0) {
158 return -1;
159 }
160 if (status == kXR_error) {
161 brix_cache_set_origin_error(t, body, dlen,
162 "cache origin token auth rejected");
163 free(body);
164 return -1;
165 }
166 free(body);
167
168 if (status != kXR_ok) {
169 /* ztn is single-round; a second authmore (or anything else) is a failure. */
170 brix_cache_set_error(t, kXR_AuthFailed, 0,
171 "cache origin token auth incomplete");
172 return -1;
173 }
174 return 0;
175}

177/* cache_origin_load_sss_key — load the first usable key from an SSS keytab file into
178 * *out. The keytab is an operator-configured, trusted path (opened O_NOFOLLOW so a
179 * planted symlink cannot redirect it) parsed with the SHARED keytab line grammar
180 * (sss_keytab_parse_line) — the exact tokenisation the server's loader uses, so a key
181 * that works one side works the other. Returns 0 with *out filled, or -1 (unreadable /
182 * malformed / no usable key). */
183static int
184cache_origin_load_sss_key(const char *path, brix_sss_key_t *out)
185{
186 int fd;
187 FILE *fp;
188 char line[1024];
189 int found = 0;
190
191 fd = open(path, O_RDONLY | O_NOFOLLOW | O_CLOEXEC); /* vfs-seam-allow: DOMAIN_CONFIG — config-domain SSS keytab (not export storage) */
192 if (fd < 0) {
193 return -1;
194 }
195 fp = fdopen(fd, "r");
196 if (fp == NULL) {
197 close(fd);
198 return -1;
199 }
200 ngx_memzero(out, sizeof(*out));
201 while (!found && fgets(line, sizeof(line), fp) != NULL) {
202 sss_keytab_entry_t entry;
203 int rc = sss_keytab_parse_line(line, &entry,
204 (int64_t) ngx_time());
205
206 if (rc < 0) { /* malformed ⇒ fail closed */
207 (void) fclose(fp); /* read-only stream — nothing buffered to lose */
208 return -1;
209 }
210 if (rc == 0) { /* blank / comment / expired */
211 continue;
212 }
213 out->id = entry.id;
214 out->exp = (time_t) entry.exp;
215 out->key_len = entry.key_len;
216 ngx_memcpy(out->key, entry.key, entry.key_len);
217 ngx_cpystrn((u_char *) out->user, (u_char *) entry.user,
218 sizeof(out->user));
219 ngx_cpystrn((u_char *) out->group, (u_char *) entry.group,
220 sizeof(out->group));
221 ngx_cpystrn((u_char *) out->name, (u_char *) entry.name,
222 sizeof(out->name));
223 found = 1;
224 }
225 (void) fclose(fp); /* read-only stream — nothing buffered to lose */
226 return found ? 0 : -1;
227}

229/* brix_cache_origin_auth_sss — present an SSS (Simple Shared Secret) credential to
230 * the origin via the XrdSecsss protocol after a login advertised "&P=sss". Mints the
231 * SAME kXR_auth blob the proxy path sends (brix_sss_build_proxy_credential): a
232 * Blowfish-CFB block over a nonce + gen-time + an asserted username, keyed by the
233 * shared secret. as_user == NULL asserts the keytab's own principal (the static
234 * service leg); non-NULL is identity injection (phase-70 §5.6 / P90-70.3): assert
235 * the CALLER, failing closed on an empty or over-long principal — the credential
236 * builder would otherwise silently substitute "xrd" or truncate to the 63-byte
237 * NAME TLV bound, either of which changes WHO reaches the origin. Single-round:
238 * expect kXR_ok. Returns 0, or -1 (t error set). §14. */
239int
240brix_cache_origin_auth_sss(brix_cache_fill_t *t,
241 brix_cache_origin_conn_t *oc, const char *keytab_path,
242 const char *as_user)
243{
244 brix_sss_key_t key;
245 u_char cred[2048];
246 size_t cred_len = 0;
247 uint16_t status;
248 uint32_t dlen;
249 u_char *body = NULL;
250 const char *assert_user;
251
252 if (as_user != NULL
253 && (as_user[0] == '\0' || ngx_strlen(as_user) > 63))
254 {
255 brix_cache_set_error(t, kXR_AuthFailed, 0,
256 "cache origin SSS identity injection: caller principal is "
257 "missing or exceeds the SSS name bound - refusing to assert "
258 "a substituted or truncated identity");
259 return -1;
260 }
261
262 if (cache_origin_load_sss_key(keytab_path, &key) != 0) {
263 brix_cache_set_error(t, kXR_AuthFailed, 0,
264 "cache origin SSS keytab unreadable or has no usable key");
265 return -1;
266 }
267 assert_user = (as_user != NULL) ? as_user : key.user;
268 if (brix_sss_build_proxy_credential(&key, assert_user, cred, sizeof(cred),
269 &cred_len) != NGX_OK)
270 {
271 brix_cache_set_error(t, kXR_AuthFailed, 0,
272 "cache origin SSS credential build failed");
273 return -1;
274 }
275 if (cache_origin_send_kxr_auth(oc, "sss", cred, (uint32_t) cred_len) != 0) {
276 brix_cache_set_error(t, kXR_ServerError, errno,
277 "cache origin SSS auth write failed");
278 return -1;
279 }
280 if (brix_cache_read_response(t, oc, &status, &body, &dlen, 4096) != 0) {
281 return -1;
282 }
283 if (status == kXR_error) {
284 brix_cache_set_origin_error(t, body, dlen,
285 "cache origin SSS auth rejected");
286 free(body);
287 return -1;
288 }
289 free(body);
290 if (status != kXR_ok) { /* SSS is single-round */
291 brix_cache_set_error(t, kXR_AuthFailed, 0,
292 "cache origin SSS auth incomplete");
293 return -1;
294 }
295 return 0;
296}

320/* brix_cache_origin_auth_krb5 — re-authenticate to the origin AS the inbound
321 * user over GSSAPI/krb5 after a kXR_login advertised "&P=krb5" (phase-70 §5.7).
322 * Drives the production multi-leg engine brix_krb5_deleg_negotiate(): every
323 * initiator token is framed as a kXR_auth(credtype "krb5") request and the
324 * origin's kXR_authmore reply is fed back through gss_init_sec_context() until
325 * the context completes (kXR_ok) with mutual auth. deleg_gss_cred is the
326 * captured gss_cred_id_t (NULL ⇒ the process default cred); origin_service_princ
327 * is "host/<fqdn>@REALM" (see brix_krb5_origin_princ_from_host). Returns 0 on a
328 * completed exchange, -1 otherwise (t error set).
329 *
330 * The kXR frame codec (brix_krb5_kxr_wire) and the engine are exercised live
331 * over real GSS bytes by tests/test_krb5_forward_live.py (mode "kxrwire", a real
332 * KDC + a kXR-framed acceptor over a socket).
333 *
334 * RETAINED REFERENCE DIALECT — SUPERSEDED, not on the production path. The live
335 * krb5 origin leg is brix_cache_origin_auth_krb5_raw() below (dispatched from
336 * origin_protocol_bootstrap.c), a RAW AP-REQ exchange — stock XRootD krb5 speaks
337 * raw krb5_rd_req, NOT this GSSAPI gss_init_sec_context init-token negotiation
338 * (phase-88 UPDATE (iv); phase-92 §5). This GSSAPI variant has zero production
339 * callers and is kept only as a reference implementation of the GSSAPI dialect,
340 * with its live-wire unit. Its lack of a caller is deliberate, NOT infra-blocked. */
341int
342brix_cache_origin_auth_krb5(brix_cache_fill_t *t, brix_cache_origin_conn_t *oc,
343 void *deleg_gss_cred, const char *origin_service_princ)
344{
345 brix_krb5_kxr_wire_t w;
346 ngx_pool_t *pool;
347 ngx_int_t rc;
348
349 if (!brix_krb5_forward_available()) {
350 brix_cache_set_error(t, kXR_AuthFailed, 0,
351 "cache origin krb5 auth unavailable (no krb5/GSSAPI support)");
352 return -1;
353 }
354
355 /* A private pool for the transient token copies the engine hands to the
356 * codec — thread-safe (a fresh malloc-backed pool, no shared nginx state). */
357 pool = ngx_create_pool(NGX_DEFAULT_POOL_SIZE, t->c->log);
358 if (pool == NULL) {
359 brix_cache_set_error(t, kXR_NoMemory, 0,
360 "cache origin krb5 pool allocation failed");
361 return -1;
362 }
363
364 ngx_memzero(&w, sizeof w);
365 w.send = cache_origin_krb5_send;
366 w.recv = cache_origin_krb5_recv;
367 w.io = oc;
368 w.max_body = 1 << 16; /* a GSS/SPNEGO token is a few KiB; 64K is generous */
369
370 rc = brix_krb5_deleg_negotiate(pool, deleg_gss_cred, origin_service_princ,
371 brix_krb5_kxr_wire, &w, t->c->log);
372
373 if (w.reply != NULL) { /* free the final leg's borrowed reply token */
374 free(w.reply);
375 w.reply = NULL;
376 }
377 ngx_destroy_pool(pool);
378
379 if (rc != NGX_OK) {
380 brix_cache_set_error(t, kXR_AuthFailed, 0,
381 "cache origin krb5 negotiation failed");
382 return -1;
383 }
384 return 0;
385}

387/* brix_cache_origin_auth_krb5_raw — re-authenticate to the origin AS the inbound
388 * user with a RAW krb5 AP-REQ (phase-70 §5.7). This is the dialect stock XRootD
389 * (libXrdSeckrb5) and brix's own acceptor (src/auth/krb5/auth.c) actually speak —
390 * krb5_rd_req over an AP-REQ — as opposed to the GSSAPI init-context tokens the
391 * multi-leg engine above emits, which no real "&P=krb5" origin can consume.
392 *
393 * The delegated user's TGT is already carried onto the fill task as a ccache PATH
394 * (ccache_path), so brix_krb5_apreq_from_ccache builds the "krb5\0"+AP-REQ payload
395 * — byte-for-byte the native client's wire — directly, with no GSS re-import.
396 * origin_spn is the origin's advertised service principal ("&P=krb5,<princ>").
397 *
398 * A single leg suffices: the AP-REQ is self-contained, so a correct origin answers
399 * kXR_ok. A kXR_authmore ("fwdtgt") — the origin asking us to chain-delegate onward
400 * to ITS backend — is not chained here and fails closed. Returns 0 on kXR_ok, -1
401 * otherwise (t error set). Per-user only: never falls back to a service credential.
402 * Live-verified against a real KDC by tests/test_krb5_forward_live.py mode "apreq"
403 * (the produced AP-REQ is accepted by krb5_rd_req against the origin keytab). */

462 brix_cache_set_error(t, kXR_AuthFailed, 0,
463 "cache origin krb5 AP-REQ rejected by origin");
464 return -1;
465 }
466 return 0;
467}
