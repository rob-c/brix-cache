1/*
2 * brixposix_stdio.c — §7.8 stdio interposition for the POSIX preload shim.
3 *
4 * WHAT: fopen/freopen over the shim's shadow descriptors (the glibc fopen64 /
5 * freopen64 spellings forward here from lib/platform/linux/preload_lfs.c),
6 * so a program that reaches for FILE* instead of a raw fd still sees the
7 * remote namespace.
8 * WHY: The fd family alone leaves out most of the software this shim exists
9 * for. Analysis code, config readers and every script language's file
10 * object go through fopen; without it the shim covered `dd` and not
11 * `awk`.
12 * HOW: The host's cookie stdio (glibc fopencookie, BSD funopen; the body is
13 * brixposix_stream_open in lib/platform/<host>/preload_stream.c): we
14 * hand back a REAL FILE* whose backing store is our shadow fd, and libc
15 * keeps doing the buffering, fgets, fscanf, ungetc and feof. So this TU
16 * interposes exactly the entry points that CREATE a stream and touches
17 * nothing that reads one — no fake FILE*, no reimplemented buffering,
18 * and no risk of a stream we made being handed to a libc function we
19 * did not.
20 *
21 * The cookie's read/write/seek/close are the shim's OWN interposed
22 * read/write/lseek/close, not the libc ones, which is what keeps this file
23 * short and keeps it correct as the fd layer grows.
24 */

42/*
43 * Translate an fopen mode to open(2) flags.
44 *
45 * Returns 0 on success. A mode this shim cannot honour is REFUSED (-1) rather
46 * than passed to the real fopen: the path is under the remote prefix, so a
47 * fallback would open — or create — a LOCAL file wearing a remote path's name,
48 * and the caller would read bytes that are not the ones it asked for. A
49 * refusal is legible; a silent local shadow is not.
50 *
51 * "r" read -> supported
52 * "w" truncate/create -> supported (the fd layer's write handle)
53 * "a" append -> refused: a remote write handle starts at offset 0
54 * and there is no "seek to end" on it, so an append
55 * would overwrite from the front.
56 * "+" read+write -> refused: one handle is read OR write, never both.
57 */
58static int
59mode_to_flags(const char *mode, int *out)
60{
61 if (mode == NULL || mode[0] == '\0' || strchr(mode, '+') != NULL) {
62 return -1;
63 }
64 if (mode[0] == 'r') {
65 *out = O_RDONLY;
66 return 0;
67 }
68 if (mode[0] == 'w') {
69 *out = O_WRONLY | O_CREAT | O_TRUNC;
70 return 0;
71 }
72 return -1; /* 'a', and anything else glibc might grow */
73}

75/* Open `path` remotely if it is ours; *handled stays 0 when it is not. */
76static FILE *
77remote_fopen(const char *path, const char *mode, int *handled)
78{
79 char remote[XRDC_PATH_MAX];
80 int flags, fd;
81
82 *handled = 0;
83 if (path == NULL || !map_path(path, remote, sizeof(remote))) {
84 return NULL;
85 }
86 *handled = 1;
87 if (mode_to_flags(mode, &flags) != 0) {
88 errno = ENOTSUP;
89 return NULL;
90 }
91 fd = BRIXPOSIX_WRAP(open)(path, flags, 0644); /* our open(): maps the path again */
92 if (fd < 0) {
93 return NULL;
94 }
95 return brixposix_stream_open(fd, mode);
96}

113/*
114 * freopen's contract is "close `stream`, reopen it on `path`" — the CALLER
115 * keeps its FILE*, and the classic use is redirecting stdout. fopencookie
116 * cannot rebind an existing stream, so a remote target is refused rather than
117 * half-served: returning a different FILE* would leave every caller that
118 * relies on the identity (`freopen(p, "r", stdin)`) reading the old stream.
119 */
120FILE *
121BRIXPOSIX_WRAP(freopen)(const char *path, const char *mode, FILE *stream)
122{
123 char remote[XRDC_PATH_MAX];
124
125 REAL(freopen);
126 if (path != NULL && map_path(path, remote, sizeof(remote))) {
127 errno = ENOTSUP;
128 return NULL;
129 }
130 return real_freopen(path, mode, stream);
131}
