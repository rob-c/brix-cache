1/*
2 * brixposix_stdio.c — §7.8 stdio interposition for the POSIX preload shim.
3 *
4 * WHAT: fopen/fopen64/freopen over the shim's shadow descriptors, so a program
5 * that reaches for FILE* instead of a raw fd still sees the remote
6 * namespace.
7 * WHY: The fd family alone leaves out most of the software this shim exists
8 * for. Analysis code, config readers and every script language's file
9 * object go through fopen; without it the shim covered `dd` and not
10 * `awk`.
11 * HOW: glibc's fopencookie: we hand back a REAL FILE* whose backing store is
12 * our shadow fd, and glibc keeps doing the buffering, fgets, fscanf,
13 * ungetc and feof. So this TU interposes exactly the three entry points
14 * that CREATE a stream and touches nothing that reads one — no fake
15 * FILE*, no reimplemented buffering, and no risk of a stream we made
16 * being handed to a libc function we did not.
17 *
18 * The cookie's read/write/seek/close are the shim's OWN interposed
19 * read/write/lseek/close, not the libc ones, which is what keeps this file
20 * short and keeps it correct as the fd layer grows.
21 */

35 * precedes libc in the global symbol search and every one of those calls

41/*
42 * Translate an fopen mode to open(2) flags.
43 *
44 * Returns 0 on success. A mode this shim cannot honour is REFUSED (-1) rather
45 * than passed to the real fopen: the path is under the remote prefix, so a
46 * fallback would open — or create — a LOCAL file wearing a remote path's name,
47 * and the caller would read bytes that are not the ones it asked for. A
48 * refusal is legible; a silent local shadow is not.
49 *
50 * "r" read -> supported
51 * "w" truncate/create -> supported (the fd layer's write handle)
52 * "a" append -> refused: a remote write handle starts at offset 0
53 * and there is no "seek to end" on it, so an append
54 * would overwrite from the front.
55 * "+" read+write -> refused: one handle is read OR write, never both.
56 */
57static int
58mode_to_flags(const char *mode, int *out)
59{
60 if (mode == NULL || mode[0] == '\0' || strchr(mode, '+') != NULL) {
61 return -1;
62 }
63 if (mode[0] == 'r') {
64 *out = O_RDONLY;
65 return 0;
66 }
67 if (mode[0] == 'w') {
68 *out = O_WRONLY | O_CREAT | O_TRUNC;
69 return 0;
70 }
71 return -1; /* 'a', and anything else glibc might grow */
72}

111/* Wrap an already-open shadow fd in a FILE*. On failure the fd is closed
112 * here: the caller only ever saw the FILE*, so nothing else can. */
113static FILE *
114stream_over(int fd, const char *mode)
115{
116 FILE *fp = fopencookie((void *) (intptr_t) fd, mode, g_cookie_fns);
117
118 if (fp == NULL) {
119 int saved = errno;
120 close(fd);
121 errno = saved;
122 return NULL;
123 }
124 return fp;
125}

127/* Open `path` remotely if it is ours; *handled stays 0 when it is not. */
128static FILE *
129remote_fopen(const char *path, const char *mode, int *handled)
130{
131 char remote[XRDC_PATH_MAX];
132 int flags, fd;
133
134 *handled = 0;
135 if (path == NULL || !map_path(path, remote, sizeof(remote))) {
136 return NULL;
137 }
138 *handled = 1;
139 if (mode_to_flags(mode, &flags) != 0) {
140 errno = ENOTSUP;
141 return NULL;
142 }
143 fd = open(path, flags, 0644); /* our open(): maps the path again */
144 if (fd < 0) {
145 return NULL;
146 }
147 return stream_over(fd, mode);
148}

166/*
167 * freopen's contract is "close `stream`, reopen it on `path`" — the CALLER
168 * keeps its FILE*, and the classic use is redirecting stdout. fopencookie
169 * cannot rebind an existing stream, so a remote target is refused rather than
170 * half-served: returning a different FILE* would leave every caller that
171 * relies on the identity (`freopen(p, "r", stdin)`) reading the old stream.
172 */
173FILE *
174freopen(const char *path, const char *mode, FILE *stream)
175{
176 char remote[XRDC_PATH_MAX];
177
178 REAL(freopen);
179 if (path != NULL && map_path(path, remote, sizeof(remote))) {
180 errno = ENOTSUP;
181 return NULL;
182 }
183 return real_freopen(path, mode, stream);
184}
