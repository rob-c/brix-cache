2 * xrdcp_parse_transport.c — xrdcp's transport-posture options (Phase-38 split).
3 *
4 * WHAT: the `--zip`/`--parallel`/`--no-metalink`/`--io-uring-direct` valueless
5 * flag table plus the three value-taking options that shape how bytes
6 * move: -S/--streams, --sources, and --io-uring.
7 * WHY: split from xrdcp_parse.c on the 600-line file gate. The family is one
8 * concept — nothing here touches credentials, manifests or filters — and
9 * it is the one option group expressed as data (coding-standards §8.6),

11 * HOW: xrdcp_parse_transport_option() is the fan-out entry, reached from

13 * target directly rather than the CLI scratch state: every option here

24/*
25 * WHAT: One valueless transport spelling — an exact argv match that assigns a
26 * constant to one `int` field of brix_copy_opts.
27 * WHY: These flags differ only in name, field and value, so express the family
28 * as data rather than a branch ladder (coding-standards §8.6). A new
29 * spelling is a row; the matcher below never changes.
30 * HOW: `field` is an offsetof into brix_copy_opts, applied by
31 * xrdcp_set_transport_flag.
32 */
33typedef struct {
34 const char *spelling;
35 size_t field;
36 int value;
37} xrdcp_transport_flag_t;

41static const xrdcp_transport_flag_t xrdcp_transport_flags[] = {

74/* ---- Apply a valueless transport flag ----
75 *
76 * WHAT: Assigns the table row whose spelling equals `a`, returning 1 when one
77 * matched and 0 when the argument is not a valueless transport flag.
78 *
79 * WHY: Keeps the family's growth out of the parser's control flow — the
80 * ladder this replaces was the bulk of the caller's branch count.
81 *
82 * HOW: 1. Walk the table comparing spellings exactly.
83 * 2. On a hit, write the row's value through its offsetof and report the
84 * match; otherwise report no match.
85 */
86static int
87xrdcp_set_transport_flag(brix_copy_opts *o, const char *a)
88{
89 size_t n;
90
91 for (n = 0; n < sizeof(xrdcp_transport_flags)
92 / sizeof(xrdcp_transport_flags[0]); n++) {
93 if (strcmp(a, xrdcp_transport_flags[n].spelling) == 0) {
94 *(int *) ((char *) o + xrdcp_transport_flags[n].field) =
95 xrdcp_transport_flags[n].value;
96 return 1;
97 }
98 }
99 return 0;
100}

103/* ---- Parse --sources N ----
104 *
105 * WHAT: Consumes `--sources N` (phase-100 extreme copy), returning 1 on
106 * success, 50 on a value outside 1..16 (usage printed), and 0 when the
107 * argument is not --sources or its value is missing.
108 *
109 * WHY: Block-stealing download from up to N replicas (metalink mirrors /
110 * locate discovery); 1 = plain single source, and the engine caps at 16
111 * distinct connections, so a count outside that range is a CLI error
112 * rather than something to silently clamp.
113 *
114 * HOW: 1. Reject anything that is not --sources with a following value.
115 * 2. atoi the value and bound it to 1..16.
116 * 3. Store it and report the match.
117 */
118static int
119xrdcp_parse_sources_opt(brix_copy_opts *o, int argc, char **argv, size_t *i)
120{
121 int n;
122
123 if (strcmp(argv[*i], "--sources") != 0 || *i + 1 >= (size_t) argc) {
124 return 0;
125 }
126
127 n = atoi(argv[++(*i)]);
128 if (n < 1 || n > 16) {
129 fprintf(stderr, "xrdcp: --sources: expected a count in 1..16, "
130 "got '%s'\n", argv[*i]);
131 usage(argv[0]);
132 return 50;
133 }
134
135 o->sources = n;
136 return 1;
137}

140/* ---- Parse the --io-uring mode, either spelling ----
141 *
142 * WHAT: Consumes `--io-uring=<mode>` or `--io-uring <mode>`, returning 1 on
143 * success, 50 on an unrecognised mode (usage printed), and 0 when the
144 * argument is neither spelling.
145 *
146 * WHY: Both spellings validate the same way and must reject the same set, so
147 * one helper owns the mode vocabulary — a second copy would be a second
148 * source of truth for what "auto" means.
149 *
150 * HOW: 1. Take the mode text from after the '=' or from the next argv slot.
151 * 2. Map it with brix_cli_parse_io_uring; a negative result is a clean
152 * CLI error, not a silent fallback.
153 * 3. Store the mode and report the match.
154 */
155static int
156xrdcp_parse_io_uring_opt(brix_copy_opts *o, int argc, char **argv, size_t *i)
157{
158 const char *a = argv[*i];
159 const char *mode;
160 int v;
161
162 if (strncmp(a, "--io-uring=", 11) == 0) {
163 mode = a + 11;
164
165 } else if (strcmp(a, "--io-uring") == 0 && *i + 1 < (size_t) argc) {
166 mode = argv[++(*i)];
167
168 } else {
169 return 0;
170 }
171
172 v = brix_cli_parse_io_uring(mode);
173 if (v < 0) {
174 fprintf(stderr, "xrdcp: --io-uring: invalid mode '%s' (use on|off|auto)\n",
175 mode);
176 usage(argv[0]);
177 return 50;
178 }
179
180 o->io_uring = v;
181 return 1;
182}

185int
186xrdcp_parse_transport_option(brix_copy_opts *o, int argc, char **argv, size_t *i)
187{
188 const char *a = argv[*i];
189 int rc;
190
191 if (xrdcp_set_transport_flag(o, a)) {
192 return 1;
193 }
194
195 if ((strcmp(a, "-S") == 0 || strcmp(a, "--streams") == 0)
196 && *i + 1 < (size_t) argc) {
197 o->streams = atoi(argv[++(*i)]);
198 return 1;
199 }
200
201 rc = xrdcp_parse_sources_opt(o, argc, argv, i);
202 if (rc != 0) {
203 return rc;
204 }
205
206 /* --max-stall / --no-retry are parsed by brix_opts_parse_arg into the shared
207 * brix_opts (s->o->conn) — which runs first in xrdcp_parse_option — and are
208 * folded into copt by finalize_resilience_posture(). Do NOT duplicate the
209 * flag here: a second handler is unreachable dead code and a second source of
210 * truth for the give-up window. */
211 return xrdcp_parse_io_uring_opt(o, argc, argv, i);
212}
