22/* The session's write-mechanics state machine (phase-107 C1 §4). RANDOM vs
23 * SEQUENTIAL is chosen at open from the backend's CAP_RANDOM_WRITE;
24 * SEQUENTIAL promotes ONE WAY to SPILL on the first out-of-order extent (or
25 * on BRIX_VFS_WRITER_O_UNORDERED at open), because once bytes land in the
26 * scratch it is the authority on the object's contents; SPILL degrades to
27 * FAILED when the scratch cannot hold the object (ENOSPC / spill_max). No
28 * state is ever re-entered, and RANDOM never spills — it has no ordering
29 * constraint to violate. */
30typedef enum {
31 BRIX_VFS_WRITER_RANDOM = 0, /* in-place handle, any offset */
32 BRIX_VFS_WRITER_SEQUENTIAL, /* staged upload, cursor-ordered */
33 BRIX_VFS_WRITER_SPILL, /* local scratch absorbs any offset */
34 BRIX_VFS_WRITER_FAILED /* scratch exhausted; ENOSPC until finish */
35} brix_vfs_writer_mode_t;

45/* Spill scratch (phase-107 C1, Appendix A.9): one sparse POSIX temp under the
46 * export's spill root — service storage, never the export namespace. File
47 * offsets are object offsets minus `base`: bytes below `base` were already
48 * streamed into the driver's staged session before the promotion and can
49 * never be patched again. The extent set is what makes the drain honest — a
50 * byte the client never sent is refused, not zero-filled (the filesystem
51 * zero-fills sub-block holes silently, so SEEK_HOLE cannot detect them). */
52typedef struct {
53 ngx_fd_t fd; /* scratch temp (NGX_INVALID_FILE = none) */
54 u_char *path; /* pool-owned; unlinked on discard */
55 off_t base; /* first object byte the spill owns */
56 off_t high_water; /* highest object offset+len seen */
57 off_t written; /* bytes landed (metrics/diagnostics) */
58 off_t max; /* cap on (high_water - base); 0 = none */
59 brix_vfs_spill_ext_t *ext; /* sorted disjoint coverage set */
60 ngx_uint_t n_ext;
61 ngx_uint_t ext_cap;
62} brix_vfs_spill_t;

83/* T1 — promote a SEQUENTIAL session to SPILL for a write of `len` bytes at
84 * object offset `off` (len 0 = the BRIX_VFS_WRITER_O_UNORDERED declaration at
85 * open). Creates the scratch under the export's spill root (brix_vfs_spill_path,
86 * falling back to brix_stage_dir). NGX_OK with w->mode == SPILL on success.
87 * NGX_ERROR with errno = EINVAL when `off` rewinds below the already-staged
88 * prefix (unservable — the session STAYS SEQUENTIAL so an in-order
89 * continuation still succeeds), or ENOSPC when no spill root is configured /
90 * scratch creation fails (stays SEQUENTIAL) or the triggering extent already
91 * exceeds spill_max (T4: mode -> FAILED, staged session aborted). */
92ngx_int_t brix_vfs_writer_spill_enter(brix_vfs_writer_t *w, off_t off,
93 size_t len);

95/* Land one extent in the scratch. EINVAL for an offset below spill.base or an

97 * the verify CRC); ENOSPC past spill_max, on a failed scratch write, or when
98 * the extent set outgrows its ceiling — the capacity cases are T4 and move
99 * the session to FAILED. */
100ngx_int_t brix_vfs_writer_spill_put(brix_vfs_writer_t *w, const void *buf,

103/* T2 — stream the scratch [base, high_water) sequentially into the session's
