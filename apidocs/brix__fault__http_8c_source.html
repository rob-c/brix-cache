20/* Case-insensitive: does the header line start with `name`? */
21static int
22hdr_is(const unsigned char *line, size_t len, const char *name)
23{
24 size_t nl = strlen(name);
25 if (len < nl) {
26 return 0;
27 }
28 for (size_t i = 0; i < nl; i++) {
29 if (tolower(line[i]) != tolower((unsigned char) name[i])) {
30 return 0;
31 }
32 }
33 return 1;
34}

36/* True when `line` is a header whose name matches the configured strip target
37 * (case-insensitive, the name terminated by optional spaces then a colon). */
38static int
39hdr_stripped(const unsigned char *line, size_t llen, const fp_http_cfg *c)
40{
41 if (c->strip_len <= 0 || llen < (size_t) c->strip_len) {
42 return 0;
43 }
44 for (int i = 0; i < c->strip_len; i++) {
45 if (tolower(line[i]) != tolower(c->strip_name[i])) {
46 return 0;
47 }
48 }
49 size_t j = (size_t) c->strip_len;
50 while (j < llen && (line[j] == ' ' || line[j] == '\t')) {
51 j++;
52 }
53 return j < llen && line[j] == ':';
54}

65/* Emit a Transfer-Encoding header line with its name obfuscated per `mode`. */
66static void
67emit_te_obfuscated(unsigned char *out, size_t cap, size_t *o,
68 const unsigned char *line, size_t llen, int mode, int lf)
69{
70 const unsigned char *colon = memchr(line, ':', llen);
71 if (!colon) {
72 emit_line(out, cap, o, line, llen, lf);
73 return;
74 }
75 size_t namelen = (size_t) (colon - line);
76 size_t rest = llen - namelen; /* colon + value */
77 if (mode == 1) { /* space before the colon */
78 fp_bufcat(out, cap, o, line, namelen);
79 fp_bufputc(out, cap, o, ' ');
80 fp_bufcat(out, cap, o, colon, rest);
81 } else if (mode == 2) { /* leading tab (line folding) */
82 fp_bufputc(out, cap, o, '\t');
83 fp_bufcat(out, cap, o, line, llen);
84 } else { /* '-' -> '_' inside the name */
85 for (size_t k = 0; k < namelen; k++) {
86 fp_bufputc(out, cap, o, line[k] == '-' ? '_' : line[k]);
87 }
88 fp_bufcat(out, cap, o, colon, rest);
89 }
90 fp_buf_eol(out, cap, o, lf);
91}

94static void
95emit_added_headers(unsigned char *out, size_t cap, size_t *o,
96 const fp_http_cfg *c, fp_http_stats *st, int lf)
97{

123/* Offset of the CRLFCRLF that ends the header block, or `n` when the buffer
124 * holds no complete header block (the caller then leaves the bytes alone). */
125size_t
126fp_http_header_len(const unsigned char *in, size_t n)
127{
128 for (size_t i = 0; i + 3 < n; i++) {
129 if (in[i] == '\r' && in[i + 1] == '\n' &&
130 in[i + 2] == '\r' && in[i + 3] == '\n') {
131 return i;
132 }
133 }
134 return n;
135}

143int
144fp_http_hold_decide(const fp_http_cfg *c, const unsigned char *in, size_t n,
145 size_t *release)
146{
147 if (c->hold_thresh <= 0) {
148 return 0;
149 }
150 size_t hlen = fp_http_header_len(in, n);
151 if (hlen == n) {
152 return 0; /* no complete header block -> pass */
153 }
154 if (hlen < (size_t) c->hold_thresh) {
155 return 0; /* header under the threshold -> pass */
156 }
157 if (c->hold_partial) {
158 *release = (size_t) c->hold_thresh < n ? (size_t) c->hold_thresh : n;
159 } else {
160 *release = 0; /* whole-message hold */
161 }
162 return 1;
163}

171int
172fp_http_body_hold_decide(const fp_http_cfg *c, const unsigned char *in, size_t n,
173 size_t *release)
174{
175 if (c->body_hold_thresh <= 0) {
176 return 0;
177 }
178 size_t hlen = fp_http_header_len(in, n);
179 /* Body bytes = those after CRLFCRLF; a segment with no header block is treated
180 * as a pure body continuation (its whole length counts). */
181 size_t body = (hlen == n) ? n : n - (hlen + 4);
182 if (body < (size_t) c->body_hold_thresh) {
183 return 0;
184 }
185 if (c->body_hold_partial) {
186 /* Release everything up to and including the threshold-th body byte. */
187 size_t hdr = (hlen == n) ? 0 : hlen + 4;
188 size_t rel = hdr + (size_t) c->body_hold_thresh;
189 *release = rel < n ? rel : n;
190 } else {
191 *release = 0;
192 }
193 return 1;
194}

196/* Copy every header line through, rewriting Transfer-Encoding on the way when
197 * obfuscation is armed. `hlen` counts header bytes, excluding the terminator. */
198static void
199http_emit_headers(const unsigned char *in, size_t hlen,
200 unsigned char *out, size_t outcap, size_t *o,
201 const fp_http_cfg *c, fp_http_stats *st, int lf)
202{
203 size_t ls = 0;
204 while (ls <= hlen) {
205 size_t le = ls;
206 while (le + 1 <= hlen && !(in[le] == '\r' && in[le + 1] == '\n')) {
207 le++;
208 }
209 if (le > hlen) {
210 le = hlen;
211 }
212 const unsigned char *line = in + ls;
213 size_t llen = le - ls;
214 if (llen > 0 && hdr_stripped(line, llen, c)) {
215 st->headers_added++; /* count the drop as a header edit */
216 } else if (llen > 0) {
217 if (c->obfuscate_te && hdr_is(line, llen, "transfer-encoding")) {
218 emit_te_obfuscated(out, outcap, o, line, llen, c->obfuscate_te, lf);
219 st->te_obf++;
220 } else {
221 emit_line(out, outcap, o, line, llen, lf);
222 }
223 }
224 if (le >= hlen) {
225 break;
226 }
227 ls = le + 2;
228 }
229}

231size_t
232fp_http_rewrite(const unsigned char *in, size_t n,
233 unsigned char *out, size_t outcap,
234 const fp_http_cfg *c, fp_http_stats *st, int *applied)
235{
236 *applied = 0;
237 size_t hlen = fp_http_header_len(in, n); /* header bytes, no terminator */
238 if (hlen == n) {
239 return 0; /* no header block here */
240 }
241 *applied = 1;
242 st->msgs++;
243 int lf = c->naked_lf;
244 size_t o = 0;
245
246 http_emit_headers(in, hlen, out, outcap, &o, c, st, lf);
247
248 emit_added_headers(out, outcap, &o, c, st, lf);
249 fp_buf_eol(out, outcap, &o, lf); /* blank line: end of headers */
250 if (lf) {
251 st->lf_converted++;
252 }
253
254 const unsigned char *body = in + hlen + 4; /* body follows CRLFCRLF */
255 fp_bufcat(out, outcap, &o, body, (size_t) (in + n - body));
256
257 if (c->append_len > 0) {
258 fp_bufcat(out, outcap, &o, c->append, (size_t) c->append_len);
259 st->appended++;
260 }
261 return o;
262}
