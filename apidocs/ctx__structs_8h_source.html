1/*
2 * core/types/ctx_structs.h
3 *
4 * Per-connection context sub-struct helper types, grouped out of the main
5 * brix_ctx_t definition (context.h) so the connection state reads as a set of
6 * named concern groups rather than a flat wall of ~100 fields. Every field is
7 * reached as ctx-><group>.<field> (e.g. ctx->gsi.sess_key, ctx->recv.payload,
8 * ctx->out.pipeline_depth).
9 *
10 * NOT self-contained: included by context.h at the point right before the
11 * brix_ctx_t definition, AFTER the helper slot structs (brix_resp_slot_t,
12 * brix_read_slot_t) and after the translation unit has pulled in state.h,
13 * file.h, tunables.h, token/token.h, metrics/metrics.h, and nginx/OpenSSL
14 * headers — exactly where these fields previously lived inline. Do not include
15 * it directly — include context.h.
16 */

24/* XrdSecpwd (Phase 52 WS-B) multi-round handshake state. Round 1 derives the
25 * DH session key from the client's kXRS_puk; round 2 decrypts the credential
26 * with it, so the ephemeral DH keypair need not survive between rounds. */
27typedef struct {
28 uint8_t session_key[16]; /* aes-128 DH session key (round 1) */
29 unsigned round; /* rounds seen (1 = puk-exchange done, awaiting creds) */
30 char user[64]; /* username asserted in round 1, verified in round 2 */
31} brix_ctx_pwd_t;

33/* XrdSeckrb5 forwarded-TGT delegation-capture state (phase-70 §5.7, inbound
34 * two-round exchange). Round 1 verifies the AP_REQ and — when brix_krb5_delegate
35 * is on — replies kXR_authmore "fwdtgt" instead of finalizing, parking the
36 * round-1 auth context (holds the session subkey the forwarded KRB_CRED is
37 * encrypted under) and the mapped client principal so round 2 can decrypt the
38 * client's krb5_fwd_tgt_creds() blob. Handles are opaque (void*) so krb5.h never
39 * leaks into this widely-included header; deleg_capture.c owns the casts and the
40 * pool-cleanup that frees them. On capture the forwarded TGT is serialised to a
41 * 0600 FILE ccache whose path is stashed in `ccache` for the request-time VFS
42 * delegation bind (brix_root_vfs_bind_deleg → brix_vfs_deleg_set_krb5). */
43typedef struct {
44 unsigned round; /* 0 = fresh; 1 = fwdtgt challenge sent, awaiting KRB_CRED */
45 void *auth_ctx; /* round-1 krb5_auth_context (session subkey), freed at round 2/cleanup */
46 void *client; /* copied krb5_principal of the verified client */
47 char cname[512]; /* mapped local name, promoted to login.dn at finalize */
48 char ccache[1024]; /* captured forwarded-TGT 0600 FILE ccache path (no "FILE:" prefix) */
49} brix_ctx_krb5_t;

111/* kXR_sigver request-signing state (GSI sessions). The client wraps each
112 * request in a kXR_sigver envelope carrying the stock XrdSecProtect secver-0
113 * signature: the GSI session cipher's encryption of SHA-256(seqno_be(8) ||
114 * header(24) || payload-unless-nodata), IV-prepended on the signed-DH path.
115 * The replay guard requires seqno > last_seqno. The cipher material is a COPY
116 * (armed at kXGC_cert) so delegation's ctx->gsi.sess_key cleanse cannot disarm
117 * signing mid-session. */
118typedef struct {
119 char sig_cipher[24]; /* session cipher name (kXRS_cipher_alg) */
120 u_char sig_key[32]; /* session key (first key_len of DH secret) */
121 int sig_keylen; /* valid bytes in sig_key (0 = unset) */
122 int sig_use_iv; /* 1 = IV-prepended blobs (signed-DH peer) */
123 int signing_active; /* 1 = cipher/key are valid and in use */
124 uint64_t last_seqno; /* highest seqno accepted so far */
125 int pending; /* 1 = next dispatch must verify the signature */
126 int verified; /* 1 = current request passed sigver verification */
127 uint16_t expectrid; /* the opcode the sigver envelope covers */
128 u_char sid[2]; /* streamid of the sigver frame (must match) */
129 uint64_t seqno; /* seqno from the kXR_sigver frame */
130 int nodata; /* 1 = payload was excluded from the hash */
131 u_char sig[64]; /* signature blob (BRIX_GSI_SIGVER_SIG_MAX) */
132 int sig_len; /* valid bytes in sig */
133} brix_ctx_sigver_t;

135/* GSI handshake + X.509 proxy delegation state. dh_key is the ephemeral DH key
136 * (kXGS_cert), freed after the shared secret is derived. signed_dh selects the
137 * signed vs unsigned DH wire form (phase-48). The deleg_* fields capture the
138 * client's proxy during login for a later TPC pull (phase-57 §F6). */
139typedef struct {
140 EVP_PKEY *dh_key; /* DH key (kXGS_cert), freed after secret derived */
141 int signed_dh; /* 0 = unsigned DH (default), 1 = signed DH (>=10400) */
142 char sess_cipher[24]; /* negotiated session cipher name */
143 u_char sess_key[32]; /* session AES key (DH-secret derived) */
144 int sess_keylen; /* valid bytes in sess_key (0 = unset) */
145 int sess_use_iv; /* 1 = IV-prepended main (signed-DH path) */
146 EVP_PKEY *deleg_reqkey; /* fresh proxy key (build_pxyreq), or NULL */
147 uint32_t clnt_opts; /* client's kXRS_clnt_opts (kOpts*) delegation mode */
148 int deleg_await; /* 1 = sent kXGS_pxyreq, awaiting kXGC_sigpxy */
149 u_char *deleg_chain_pem; /* client chain PEM (for assemble), heap */
150 size_t deleg_chain_len;
151 u_char *deleg_proxy_pem; /* captured delegated proxy credential (PEM) */
152 size_t deleg_proxy_len;
153 u_char deleg_client_rtag[64]; /* client's kXGC_cert random tag */
154 int deleg_client_rtag_len;
155 /* phase-70 §5.1: raw client-pushed FULL proxy PEM (kXRS_x509_fullproxy)
156 * captured from the decrypted kXGC_cert inner buffer, BEFORE DN validation.
157 * auth.c validates (chain+key parse, leaf DN == authenticated DN) then
158 * promotes the bytes to ctx->deleg_proxy_pem; heap-owned, freed at cleanup. */
159 u_char *client_fullproxy_pem;
160 size_t client_fullproxy_len;
161} brix_ctx_gsi_t;

163/* Per-connection rate-limit state (Phase 25/33). bw_* = the current request's

166 * *_rule pointers are brix_rl_rule_t* (void to keep ratelimit.h out of this

174 uint32_t key_cache_valid; /* bitmask: bit i ⇒ key_cache[i] holds rule i's key */
175} brix_ctx_rl_t;

177/* Session login + authenticated-identity state. Two-step XRootD login: kXR_login
178 * sets logged_in and issues sessid; kXR_auth sets auth_done and fills the identity
179 * strings (dn/primary_vo/vo_list/peer_ip). acc_host is the opt-in XrdAcc
180 * reverse-DNS cache; gsi_counted tracks an in-flight GSI handshake slot. */
181typedef struct {
182 u_char sessid[BRIX_SESSION_ID_LEN]; /* opaque ID we issued at login */
183 ngx_flag_t logged_in; /* set when kXR_login is accepted */
184 ngx_flag_t auth_done; /* set when authentication is complete */
185 char user[9]; /* fixed-width kXR_login username, NUL-terminated */
186 uint32_t pid; /* client pid from kXR_login, host byte order */
187 uint8_t auth_fail_count; /* failed kXR_auth attempts; capped */
188 size_t pool_bytes_used; /* cumulative ngx_palloc bytes; capped */
189 char dn[512]; /* GSI subject DN (literal proxy-leaf DN) */
190 char eec_dn[512]; /* P80.11: stable End-Entity Cert DN (proxy
191 * serial stripped); "" for non-proxy/non-GSI
192 * auth. The authorization identity — see
193 * brix_gsi_complete_auth. */
194 char primary_vo[128]; /* first VO from the VOMS attribute cert */
195 char vo_list[512]; /* space-separated list of all VOs */
196 char peer_ip[64]; /* remote peer address for authdb HOST ('p') rules */
197 const char *acc_host; /* XrdAcc reverse-DNS host cache (points into c->pool) */
198 unsigned acc_host_done:1;
199 unsigned gsi_counted:1; /* holds a GSI in-flight handshake slot (Phase 51 E4) */
200} brix_ctx_login_t;

202/* Request receive/framing state. Read in two stages: the fixed 24-byte header
203 * into hdr_buf, then (if dlen>0) the payload into the reusable payload_buf.
204 * cur_* are the parsed header fields; cur_body_extra/extended cover the trailing
205 * body streamed after the dlen-framed payload (kXR_writev / kXR_chkpoint). */
206typedef struct {
207 u_char hdr_buf[24]; /* raw bytes of the current request header */
208 size_t hdr_pos; /* header bytes received so far */
209 u_char cur_streamid[2]; /* echoed back unchanged in every response */
210 uint16_t cur_reqid; /* opcode, host byte order */
211 u_char cur_body[16]; /* request-specific parameter bytes (wire.h) */
212 uint32_t cur_dlen; /* payload length that follows the header */
213 uint32_t cur_body_extra; /* streamed data bytes beyond cur_dlen */
214 unsigned cur_body_extended:2;/* completed extension stages (2 = done) */
215 u_char *payload; /* current request payload, NULL if none */
216 size_t payload_pos; /* bytes accumulated so far */
217 u_char *payload_buf; /* reusable receive buffer */
218 size_t payload_buf_size; /* allocated size of payload_buf */
219
220 /* Streaming large plain kXR_write. When sw_active is set the payload is a
221 * SINGLE bounded chunk (payload_buf holds BRIX_WRITE_STREAM_CHUNK bytes) and
222 * cur_dlen is repurposed as the CURRENT chunk length rather than the whole
223 * write; each filled chunk is applied to the fd / staged writer at
224 * sw_base_off+sw_done and one ack is sent after the last chunk. See
225 * write_stream.c. */
226 unsigned sw_active:1; /* a chunked streaming write is in progress */
227 unsigned sw_staged:1; /* route: staged append (else direct pwrite) */
228 unsigned sw_drain:1; /* handle unusable: discard bytes, err at end */
229 int sw_idx; /* file-handle slot being written */
230 int64_t sw_base_off; /* wire offset of the logical write */
231 uint32_t sw_total; /* total logical write length (original dlen) */
232 uint32_t sw_done; /* bytes already applied */
233 int sw_err; /* latched kXR_* error code (0 = none yet) */
234 const char *sw_errmsg; /* latched error detail string */
235} brix_ctx_recv_t;

237/* Output response ring + write-pipelining state (Phase 29/32). A response is
238 * built into ring[tail] and drained from ring[head]; count = slots in use; the
239 * ring holds pipeline_depth slots (all arithmetic modulo pipeline_depth).
240 * wr_inflight bounds in-flight plain-write pwrites; the recv loop keeps reading
241 * while count + wr_inflight < pipeline_depth. */
242typedef struct {
243 ngx_uint_t pipeline_depth; /* ring capacity = configured in-flight bound */
244 brix_resp_slot_t *ring; /* [pipeline_depth] response slots */
245 ngx_uint_t head; /* slot being drained to the socket */
246 ngx_uint_t tail; /* slot currently being built */
247 ngx_uint_t count; /* number of slots in use (responses queued) */
248 unsigned recv_deferred:1; /* drain barrier: parked non-pipelinable req awaits count==0 */
249 u_char deferred_streamid[2]; /* parked request's sid: a pipelined AIO ack completing
250 * during the park clobbers recv.cur_streamid, so the
251 * deferred dispatch must reinstall its own sid */
252 unsigned resp_pipelinable:1; /* current response is a single-chunk sendfile read */
253 ngx_uint_t wr_inflight; /* plain-write pwrites posted, not yet acked */
254 unsigned resp_async:1; /* ack drains without disturbing recv */
255 unsigned finalize_pending:1; /* deferred teardown while wr_inflight > 0 */
256 ngx_int_t finalize_status; /* ngx_stream status to finalize with */
257} brix_ctx_out_t;

262 * pool[pipeline_depth] backs concurrent single-shot memory reads (Phase 32 WS3);
263 * win_* is the windowed-read continuation for large TLS/non-regular reads. */
264typedef struct {

269 u_char *write_scratch; /* pgwrite decode buffer */
270 size_t write_scratch_size;
271 u_char *cmp_scratch; /* inline read-compression codec output (Phase-42 W4) */
272 size_t cmp_scratch_size;
273 ngx_thread_task_t *read_aio_task;
274 ngx_thread_task_t *pgread_aio_task;
275 ngx_thread_task_t *readv_aio_task;
276 brix_read_slot_t *pool; /* [pipeline_depth] in-flight read buffers */
277 ngx_uint_t inflight; /* pool entries currently in use */
278 ngx_uint_t aio_inflight; /* phase-32 WS3: single-shot read AIO tasks

283 unsigned backpressured:1; /* recv stopped admitting reads (pool full) */
284 unsigned win_active:1; /* windowed memory-read in flight */
285 int win_idx;
286 int win_fd;
287 off_t win_offset; /* next file offset to read */
288 size_t win_remaining; /* bytes still to send */
289 u_char win_streamid[2];
290} brix_ctx_rd_t;
