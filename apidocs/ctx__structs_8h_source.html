1/*
2 * core/types/ctx_structs.h
3 *
4 * Per-connection context sub-struct helper types, grouped out of the main
5 * brix_ctx_t definition (context.h) so the connection state reads as a set of
6 * named concern groups rather than a flat wall of ~100 fields. Every field is
7 * reached as ctx-><group>.<field> (e.g. ctx->gsi.sess_key, ctx->recv.payload,
8 * ctx->out.pipeline_depth).
9 *
10 * NOT self-contained: included by context.h at the point right before the
11 * brix_ctx_t definition, AFTER the helper slot structs (brix_resp_slot_t,
12 * brix_read_slot_t) and after the translation unit has pulled in state.h,
13 * file.h, tunables.h, token/token.h, metrics/metrics.h, and nginx/OpenSSL
14 * headers — exactly where these fields previously lived inline. Do not include
15 * it directly — include context.h.
16 */

33/* XrdSeckrb5 forwarded-TGT delegation-capture state (phase-70 §5.7, inbound
34 * two-round exchange). Round 1 verifies the AP_REQ and — when brix_krb5_delegate
35 * is on — replies kXR_authmore "fwdtgt" instead of finalizing, parking the
36 * round-1 auth context (holds the session subkey the forwarded KRB_CRED is
37 * encrypted under) and the mapped client principal so round 2 can decrypt the
38 * client's krb5_fwd_tgt_creds() blob. Handles are opaque (void*) so krb5.h never
39 * leaks into this widely-included header; deleg_capture.c owns the casts and the
40 * pool-cleanup that frees them. On capture the forwarded TGT is serialised to a
41 * 0600 FILE ccache whose path is stashed in `ccache` for the request-time VFS
42 * delegation bind (brix_root_vfs_bind_session → brix_vfs_deleg_set_krb5). */
43typedef struct {
44 unsigned round; /* 0 = fresh; 1 = fwdtgt challenge sent, awaiting KRB_CRED */
45 void *auth_ctx; /* round-1 krb5_auth_context (session subkey), freed at round 2/cleanup */
46 void *client; /* copied krb5_principal of the verified client */
47 char cname[512]; /* mapped local name, promoted to login.dn at finalize */
48 char ccache[1024]; /* captured forwarded-TGT 0600 FILE ccache path (no "FILE:" prefix) */
49} brix_ctx_krb5_t;

111/* kXR_sigver request-signing state (GSI sessions). The client wraps each
112 * request in a kXR_sigver envelope carrying the stock XrdSecProtect secver-0
113 * signature: the GSI session cipher's encryption of SHA-256(seqno_be(8) ||
114 * header(24) || payload-unless-nodata), IV-prepended on the signed-DH path.
115 * The replay guard requires seqno > last_seqno. The cipher material is a COPY
116 * (armed at kXGC_cert) so delegation's ctx->gsi.sess_key cleanse cannot disarm
117 * signing mid-session. */

119 char sig_cipher[24]; /* session cipher name (kXRS_cipher_alg) */
120 u_char sig_key[32]; /* session key (first key_len of DH secret) */
121 int sig_keylen; /* valid bytes in sig_key (0 = unset) */
122 int sig_use_iv; /* 1 = IV-prepended blobs (signed-DH peer) */
123 int signing_active; /* 1 = cipher/key are valid and in use */
124 uint64_t last_seqno; /* highest seqno accepted so far */
125 int pending; /* 1 = next dispatch must verify the signature */
126 int verified; /* 1 = current request passed sigver verification */
127 uint16_t expectrid; /* the opcode the sigver envelope covers */
128 u_char sid[2]; /* streamid of the sigver frame (must match) */
129 uint64_t seqno; /* seqno from the kXR_sigver frame */
130 int nodata; /* 1 = payload was excluded from the hash */
131 u_char sig[64]; /* signature blob (BRIX_GSI_SIGVER_SIG_MAX) */
132 int sig_len; /* valid bytes in sig */
133 int unsignable_logged; /* 1 = this session already logged that it
134 * cannot sign while a signing level is
135 * configured (audit §5.2). One line per
136 * session, not per request: the condition is
137 * a property of the session's auth protocol,
138 * so repeating it per opcode would be a log
139 * flood with no added information. */
140} brix_ctx_sigver_t;

219/* Request receive/framing state. Read in two stages: the fixed 24-byte header
220 * into hdr_buf, then (if dlen>0) the payload into the reusable payload_buf.
221 * cur_* are the parsed header fields; cur_body_extra/extended cover the trailing
222 * body streamed after the dlen-framed payload (kXR_writev / kXR_chkpoint). */
223typedef struct {
224 u_char hdr_buf[24]; /* raw bytes of the current request header */
225 size_t hdr_pos; /* header bytes received so far */
226 u_char cur_streamid[2]; /* echoed back unchanged in every response */
227 uint16_t cur_reqid; /* opcode, host byte order */
228 u_char cur_body[16]; /* request-specific parameter bytes (wire.h) */
229 uint32_t cur_dlen; /* payload length that follows the header */
230 uint32_t cur_body_extra; /* streamed data bytes beyond cur_dlen */
231 unsigned cur_body_extended:2;/* completed extension stages (2 = done) */
232 u_char *payload; /* current request payload, NULL if none */
233 size_t payload_pos; /* bytes accumulated so far */
234 u_char *payload_buf; /* reusable receive buffer */
235 size_t payload_buf_size; /* allocated size of payload_buf */

237 /* Post-auth read-ahead stash (see BRIX_RECV_STASH_SIZE): one big recv
238 * feeds many exact-size frame reads with zero further syscalls. Only
239 * engaged once no other subsystem can ever read this socket raw
240 * (auth done; no upstream/relay conf), so buffered bytes are always
241 * consumed by the framing loop. stash_len > stash_head means pipelined
242 * request bytes are buffered — a kXR_bind must then refuse migration
243 * (the bytes would not travel with the fd). */
244 u_char *stash; /* lazily allocated from c->pool */
245 uint32_t stash_head; /* next unconsumed byte */
246 uint32_t stash_len; /* bytes buffered (0 = empty) */

248 /* Streaming large plain kXR_write. When sw_active is set the payload is a
249 * SINGLE bounded chunk (payload_buf holds BRIX_WRITE_STREAM_CHUNK bytes) and
250 * cur_dlen is repurposed as the CURRENT chunk length rather than the whole
251 * write; each filled chunk is applied to the fd / staged writer at
252 * sw_base_off+sw_done and one ack is sent after the last chunk. See
253 * write_stream.c. */
254 unsigned sw_active:1; /* a chunked streaming write is in progress */
255 unsigned sw_staged:1; /* route: staged append (else direct pwrite) */
256 unsigned sw_drain:1; /* handle unusable: discard bytes, err at end */
257 int sw_idx; /* file-handle slot being written */
258 int64_t sw_base_off; /* wire offset of the logical write */
259 uint32_t sw_total; /* total logical write length (original dlen) */
260 uint32_t sw_done; /* bytes already applied */
261 int sw_err; /* latched kXR_* error code (0 = none yet) */
262 const char *sw_errmsg; /* latched error detail string */
263} brix_ctx_recv_t;

265/* Output response ring + write-pipelining state (Phase 29/32). A response is
266 * built into ring[tail] and drained from ring[head]; count = slots in use; the
267 * ring holds pipeline_depth slots (all arithmetic modulo pipeline_depth).
268 * wr_inflight bounds in-flight plain-write pwrites; the recv loop keeps reading
269 * while count + wr_inflight < pipeline_depth. */
270typedef struct {
271 ngx_uint_t pipeline_depth; /* ring capacity = configured in-flight bound */
272 brix_resp_slot_t *ring; /* [pipeline_depth] response slots */
273 ngx_uint_t head; /* slot being drained to the socket */
274 ngx_uint_t tail; /* slot currently being built */
275 ngx_uint_t count; /* number of slots in use (responses queued) */
276 unsigned recv_deferred:1; /* drain barrier: parked non-pipelinable req awaits count==0 */
277 u_char deferred_streamid[2]; /* parked request's sid: a pipelined AIO ack completing
278 * during the park clobbers recv.cur_streamid, so the
279 * deferred dispatch must reinstall its own sid */
280 unsigned resp_pipelinable:1; /* current response is a single-chunk sendfile read */
281 ngx_uint_t wr_inflight; /* plain-write pwrites posted, not yet acked */
282 unsigned resp_async:1; /* ack drains without disturbing recv */
283 unsigned finalize_pending:1; /* deferred teardown while wr_inflight > 0 */
284 ngx_int_t finalize_status; /* ngx_stream status to finalize with */

286 /* §1.2 pool-send (offload-AIO): a worker thread that finished a large
287 * pgread may send the frame on this connection's cleartext socket itself,
288 * so the socket has two potential writers. send_token is the ownership
289 * CAS (0 free / 1 held) every socket-touching path takes once
290 * pool_send_active is set; send_busy mirrors "the out-ring holds parked
291 * frames" for the worker thread (the head one may be mid-frame, so it
292 * must decline and preserve wire order). Whole frames may reorder freely
293 * — responses correlate by streamid — but bytes within a frame may not. */
294 ngx_atomic_t send_token; /* socket ownership: 0 free, 1 held */
295 ngx_atomic_t send_busy; /* 1 while ring holds parked frames */
296 unsigned pool_send_active:1; /* pool-send discipline engaged */
297} brix_ctx_out_t;

314 /* used-since-last-trim marks, one per BRIX_GET_SCRATCH slot (the macro
315 * sets <slot>_hot on every fetch). brix_trim_scratch only shrinks a slot

317 * the marks as it passes. This keeps a streaming transfer's buffer warm
318 * across back-to-back large requests (no per-request free/mmap churn)
