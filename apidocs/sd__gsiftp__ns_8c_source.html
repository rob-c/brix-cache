32static ngx_int_t
33sd_gsiftp_command_one(brix_sd_instance_t *inst, const char *verb,
34 const char *path, const brix_sd_cred_t *cred)
35{
36 sd_gsiftp_state *state = inst->state;
37 gftp_session_t session;
38 char remote[GSIFTP_PATH_CAP];
39 const char *proxy;
40 int err = 0;
41 int rc;
42
43 if (sd_gsiftp_select_proxy(state, cred, &proxy, &err) != 0) {
44 return NGX_ERROR;
45 }
46 if (sd_gsiftp_path(state, path, remote) != 0
47 || sd_gsiftp_session(&session, state, proxy) != 0) {
48 return NGX_ERROR;
49 }
50 rc = gftp_command(&session, "%s %s", verb, remote);
51 if (rc == 0 && (session.code < 200 || session.code >= 300)) {
52 errno = session.code == 550 ? ENOENT : EIO;
53 rc = -1;
54 }
55 gftp_session_close(&session);
56 return rc == 0 ? NGX_OK : NGX_ERROR;
57}

114static ngx_int_t
115sd_gsiftp_rename_impl(brix_sd_instance_t *inst, const char *src,
116 const char *dst, int noreplace, const brix_sd_cred_t *cred)
117{
118 sd_gsiftp_state *state = inst->state;
119 gftp_session_t session;
120 char remote_src[GSIFTP_PATH_CAP];
121 char remote_dst[GSIFTP_PATH_CAP];
122 const char *proxy;
123 int err = 0;
124 int rc;
125
126 if (noreplace && !sd_gsiftp_destination_absent(inst, dst, cred)) {
127 return NGX_ERROR;
128 }
129 if (sd_gsiftp_select_proxy(state, cred, &proxy, &err) != 0) {
130 return NGX_ERROR;
131 }
132 if (sd_gsiftp_path(state, src, remote_src) != 0
133 || sd_gsiftp_path(state, dst, remote_dst) != 0
134 || sd_gsiftp_session(&session, state, proxy) != 0) {
135 return NGX_ERROR;
136 }
137 rc = gftp_expect(&session, 300, 399, "RNFR %s", remote_src);
138 if (rc == 0) {
139 rc = gftp_expect(&session, 200, 299, "RNTO %s", remote_dst);
140 }
141 gftp_session_close(&session);
142 return rc == 0 ? NGX_OK : NGX_ERROR;
143}

189static brix_sd_dir_t *
190sd_gsiftp_opendir_impl(brix_sd_instance_t *inst, const char *path,
191 const brix_sd_cred_t *cred, int *err_out)
192{
193 sd_gsiftp_state *state = inst->state;
194 sd_gsiftp_dir_state *dir_state;
195 brix_sd_dir_t *dir;
196 gftp_session_t session;
197 char remote[GSIFTP_PATH_CAP];
198 char command[GFTP_COMMAND_CAP];
199 const char *proxy;
200 int err = 0;
201
202 if (sd_gsiftp_select_proxy(state, cred, &proxy, &err) != 0) {
203 if (err_out != NULL) {
204 *err_out = EACCES;
205 }
206 return NULL;
207 }
208 if (sd_gsiftp_path(state, path, remote) != 0
209 || snprintf(command, sizeof(command), "MLSD %s", remote)
210 >= (int) sizeof(command)
211 || sd_gsiftp_session(&session, state, proxy) != 0) {
212 if (err_out != NULL) {
213 *err_out = errno != 0 ? errno : EIO;
214 }
215 return NULL;
216 }
217 dir = sd_gsiftp_dir_alloc(&dir_state, err_out);
218 if (dir == NULL || gftp_slurp(&session, command, &dir_state->listing,
219 &dir_state->length) != 0) {
220 if (err_out != NULL) {
221 *err_out = errno != 0 ? errno : ENOMEM;
222 }
223 sd_gsiftp_dir_free(dir, dir_state);
224 gftp_session_close(&session);
225 return NULL;
226 }
227 gftp_session_close(&session);
228 dir->inst = inst;
229 dir->state = dir_state;
230 return dir;
231}

246static int
247sd_gsiftp_next_line(sd_gsiftp_dir_state *state, const char **line,
248 size_t *length)
249{
250 size_t start;
251
252 while (state->cursor < state->length
253 && (state->listing[state->cursor] == '\r'
254 || state->listing[state->cursor] == '\n')) {
255 state->cursor++;
256 }
257 if (state->cursor == state->length) {
258 return 0;
259 }
260 start = state->cursor;
261 while (state->cursor < state->length
262 && state->listing[state->cursor] != '\r'
263 && state->listing[state->cursor] != '\n') {
264 state->cursor++;
265 }
266 *line = state->listing + start;
267 *length = state->cursor - start;
268 return 1;
269}

271ngx_int_t
272sd_gsiftp_readdir(brix_sd_dir_t *dir, brix_sd_dirent_t *out)
273{
274 sd_gsiftp_dir_state *state = dir->state;
275 const char *line;
276 size_t length;
277
278 while (sd_gsiftp_next_line(state, &line, &length)) {
279 gftp_mlsx_ent_t entry;
280
281 if (gftp_mlsx_parse(line, length, &entry) != 0
282 || (entry.name_len == 1 && entry.name[0] == '.')
283 || (entry.name_len == 2 && entry.name[0] == '.'
284 && entry.name[1] == '.')) {
285 continue;
286 }
287 if (entry.name_len >= sizeof(out->name)) {
288 errno = ENAMETOOLONG;
289 return NGX_ERROR;
290 }
291 memset(out, 0, sizeof(*out));
292 memcpy(out->name, entry.name, entry.name_len);
293 out->name[entry.name_len] = '\0';
294 out->d_type = entry.is_dir ? DT_DIR : DT_REG;
295 return NGX_OK;
296 }
297 return NGX_DONE;
298}
