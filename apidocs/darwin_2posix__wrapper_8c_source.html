29int
30brix_plat_anon_fd(const char *name, const char *dir)
31{
32 char template[PATH_MAX];
33 const char *tmpdir = dir ? dir : "/tmp";
34
35 if (name != NULL && name[0] != '\0') {
36 snprintf(template, sizeof(template), "%s/%s_XXXXXX", tmpdir, name);
37 } else {
38 snprintf(template, sizeof(template), "%s/brix_anon_XXXXXX", tmpdir);
39 }
40
41 int fd = mkstemp(template);
42 if (fd >= 0) {
43 int flags = fcntl(fd, F_GETFD);
44 if (flags >= 0) {
45 fcntl(fd, F_SETFD, flags | FD_CLOEXEC);
46 }
47 /* Unlink immediately so file is deleted when closed */
48 unlink(template); /* vfs-seam-allow: SEAM_CORRECT - PAL anonymous descriptor lifetime */
49 }
50 return fd;
51}

61/* Whether to force the drive's write cache to media (F_FULLFSYNC, the Linux
62 * fdatasync equivalent) or stop at the cache-level fsync(2). F_FULLFSYNC is
63 * the default: durability parity with Linux. BRIX_PLAT_FSYNC_FULL=0 is a
64 * development/test-host knob only — it costs ~28 ms per file on a laptop SSD
65 * (50x fsync) and serialises every writer on the host. */
66static int
67darwin_fsync_full(void)
68{
69 static int cached = -1;
70
71 if (cached < 0) {
72 const char *knob = getenv("BRIX_PLAT_FSYNC_FULL");
73 cached = (knob != NULL && knob[0] == '0' && knob[1] == '\0') ? 0 : 1;
74 }
75 return cached;
76}

138int
139brix_plat_eventfd(unsigned int initial_value, int flags)
140{
141 int pipefd[2];
142
143 /* macOS lacks eventfd - use pipe as fallback */
144 if (pipe(pipefd) < 0) {
145 return -1;
146 }
147
148 /* Apply flags via fcntl */
149 if (flags & BRIX_EVENTFD_CLOEXEC) {
150 fcntl(pipefd[0], F_SETFD, FD_CLOEXEC);
151 fcntl(pipefd[1], F_SETFD, FD_CLOEXEC);
152 }
153
154 if (flags & BRIX_EVENTFD_NONBLOCK) {
155 int fflags;
156 fflags = fcntl(pipefd[0], F_GETFL, 0);
157 fcntl(pipefd[0], F_SETFL, fflags | O_NONBLOCK);
158 fflags = fcntl(pipefd[1], F_GETFL, 0);
159 fcntl(pipefd[1], F_SETFL, fflags | O_NONBLOCK);
160 }
161
162 /* Return read end - caller manages both ends */
163 return pipefd[0];
164}

188int
189brix_plat_random(void *buf, size_t len)
190{
191 if (SecRandomCopyBytes(kSecRandomDefault, len, (uint8_t *)buf) == errSecSuccess) {
192 return 0;
193 }
194
195 /* Fallback to /dev/urandom */
196 int fd = open("/dev/urandom", O_RDONLY); /* vfs-seam-allow: NOT_STORAGE - PAL entropy device */
197 if (fd < 0) {
198 return -1;
199 }
200
201 ssize_t n = read(fd, buf, len);
202 close(fd);
203
204 return (n == (ssize_t)len) ? 0 : -1;
205}

211/* Darwin reports a missing attribute as ENOATTR (93); Linux, and every
212 * ENODATA check in the tree, expects ENODATA (96). Normalise on failure. */
213static ssize_t
214darwin_xattr_rc(ssize_t rc)
215{
216 if (rc < 0 && errno == ENOATTR) {
217 errno = ENODATA;
218 }
219 return rc;
220}

222/* Read-side (get/list/remove): Darwin limits names to XATTR_MAXNAMELEN (127)
223 * where Linux allows 255 and reports ENAMETOOLONG; such a name cannot exist
224 * here, so readers see it as absent. Writes keep ENAMETOOLONG. */
225static ssize_t
226darwin_xattr_read_rc(ssize_t rc)
227{
228 if (rc < 0 && errno == ENAMETOOLONG) {
229 errno = ENODATA;
230 }
231 return darwin_xattr_rc(rc);
232}

292int
293brix_plat_execvpe(const char *file, char *const argv[], char *const envp[])
294{
295 /* macOS lacks execvpe - use posix_spawn for PATH search */
296 posix_spawn_file_actions_t fa;
297 posix_spawnattr_t attr;
298 pid_t pid;
299 int status;
300 extern char **environ;
301
302 if (posix_spawn_file_actions_init(&fa) != 0) {
303 return -1;
304 }
305
306 if (posix_spawnattr_init(&attr) != 0) {
307 posix_spawn_file_actions_destroy(&fa);
308 return -1;
309 }
310
311 status = posix_spawn(&pid, file, &fa, &attr, argv, envp ? envp : environ);
312
313 posix_spawnattr_destroy(&attr);
314 posix_spawn_file_actions_destroy(&fa);
315
316 if (status != 0) {
317 errno = status;
318 return -1;
319 }
320
321 /* Wait for child - this blocks, but execvpe doesn't return either */
322 waitpid(pid, &status, 0);
323
324 /* If we get here, child exited - mimic execvpe behavior */
325 _exit(WEXITSTATUS(status));
326}
