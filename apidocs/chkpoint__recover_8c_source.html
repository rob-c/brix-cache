35/*
36 * ckp_validate_path — extract original path from checkpoint path.
37 *
38 * WHAT: Validates .ckp path format and extracts original path by removing
39 * the .ckp suffix.
40 * WHY: Checkpoint files must end in .ckp; the original path is everything
41 * before that suffix. Invalid lengths are rejected early.
42 * HOW: Checks length bounds, copies all but last 4 chars, null-terminates.
43 * Returns NGX_OK on success, NGX_ERROR on invalid format.
44 */
45static ngx_int_t
46ckp_validate_path(const char *ckp_path, char *orig_path, size_t orig_size)
47{
48 size_t len = strlen(ckp_path);
49
50 if (len <= 4 || len >= orig_size) {
51 return NGX_ERROR;
52 }
53
54 ngx_memcpy(orig_path, ckp_path, len - 4);
55 orig_path[len - 4] = '\0';
56 return NGX_OK;
57}

187/*
188 * ckp_cleanup_and_log — close fd, remove checkpoint, log success.
189 *
190 * WHAT: Cleans up checkpoint fd, removes .ckp journal, logs recovery.
191 * WHY: Journal must be removed after successful recovery to prevent replay;
192 * success logging provides audit trail.
193 * HOW: Closes ckp_fd, unlinks .ckp path (with vfs-mutation-gate-allow
194 * comment explaining why this bypasses normal mutation gates), logs
195 * recovery notice. Returns NGX_OK on success, NGX_ERROR if unlink fails.
196 */
197static ngx_int_t
198ckp_cleanup_and_log(ngx_log_t *log, const char *root_canon,
199 int ckp_fd, const char *ckp_path)
200{
201 ngx_close_file(ckp_fd);
202
203 /* vfs-mutation-gate-allow: startup recovery of THIS server's own .ckp
204 * journal, after its snapshot has been committed. The journal's existence
205 * is the durable proof that a writable endpoint authorised the checkpoint;
206 * no request and no endpoint configuration are in scope here to re-decide
207 * against, and leaving the record behind would replay it forever. */
208 if (brix_vfs_unlink_path(log, root_canon, ckp_path) != 0) {
209 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
210 "brix: checkpoint recovery cannot remove \"%s\"",
211 ckp_path);
212 return NGX_ERROR;
213 }
214
215 brix_log_safe_path(log, NGX_LOG_NOTICE, 0,
216 "brix: recovered abandoned checkpoint \"%s\"",
217 ckp_path);
218 return NGX_OK;
219}

221/*
222 * ckp_recover_one — recover one abandoned checkpoint file.
223 *
224 * WHAT: Recovers a single .ckp snapshot by copying it over the original file.
225 * WHY: Worker crashes leave .ckp files behind; recovery restores them at
226 * startup to prevent data loss from abandoned checkpoints.
227 * HOW: Orchestrates 7 single-responsibility helpers:
228 * 1. ckp_validate_path — extract original path
229 * 2. ckp_open_snapshot — open .ckp read-only
230 * 3. ckp_validate_snapshot — verify regular file
231 * 4. ckp_stage_original — open original for atomic write
232 * 5. ckp_copy_data — zero-copy transfer
233 * 6. ckp_sync_and_commit — fsync + atomic rename
234 * 7. ckp_cleanup_and_log — remove journal, log success
235 */
236static ngx_int_t
237ckp_recover_one(ngx_log_t *log, const char *root_canon, const char *ckp_path)
238{
239 char orig_path[PATH_MAX];
240 struct stat st;
241 int ckp_fd;
242 brix_staged_file_t staged;
243
244 /* Step 1: Validate and extract original path */
245 if (ckp_validate_path(ckp_path, orig_path, sizeof(orig_path)) != NGX_OK) {
246 return NGX_ERROR;
247 }
248
249 /* Step 2: Open checkpoint snapshot */
250 ckp_fd = ckp_open_snapshot(log, root_canon, ckp_path);
251 if (ckp_fd < 0) {
252 return NGX_ERROR;
253 }
254
255 /* Step 3: Validate snapshot is regular file */
256 if (ckp_validate_snapshot(log, ckp_fd, ckp_path, &st) != NGX_OK) {
257 return NGX_ERROR;
258 }
259
260 /* Step 4: Stage original file for atomic write */
261 if (ckp_stage_original(log, root_canon, orig_path, &staged) != NGX_OK) {
262 ngx_close_file(ckp_fd);
263 return NGX_ERROR;
264 }
265
266 /* Step 5: Copy data from checkpoint to staged file */
267 if (ckp_copy_data(log, ckp_fd, &staged, st.st_size, ckp_path,
268 root_canon, orig_path) != NGX_OK) {
269 ngx_close_file(ckp_fd);
270 return NGX_ERROR;
271 }
272
273 /* Step 6: Sync and commit atomically */
274 if (ckp_sync_and_commit(log, root_canon, &staged, orig_path) != NGX_OK) {
275 ngx_close_file(ckp_fd);
276 return NGX_ERROR;
277 }
278
279 /* Step 7: Cleanup and log success */
280 return ckp_cleanup_and_log(log, root_canon, ckp_fd, ckp_path);
281}

289/*
290 * Open one recovery-scan directory confined to the export root and hand back
291 * the DIR stream plus its validated directory fd (used for fstatat). Returns
292 * NGX_OK (caller owns out->dp), NGX_DECLINED for a skippable subdirectory
293 * (private/racing-removed — recovery must continue past it), or NGX_ERROR for
294 * a fatal condition (inaccessible export root or stream setup failure).
295 */
296static ngx_int_t
297ckp_recover_open_dir(ngx_log_t *log, const char *root_canon, const char *dir,
298 ngx_uint_t depth, ckp_scan_dir_t *out)
299{
300 DIR *dp;
301 int dfd, scan_fd;
302
303 dfd = brix_vfs_open_fd(log, root_canon, dir,
304 O_RDONLY | O_DIRECTORY | O_CLOEXEC
305 | O_NOFOLLOW, 0);
306 if (dfd < 0) {
307 /*
308 * A SUBDIRECTORY we cannot enter must NOT abort recovery: under per-request
309 * impersonation the export legitimately contains per-user PRIVATE dirs
310 * (e.g. 0700) the worker uid cannot read, and a dir can be removed mid-scan.
311 * Skip those (recovery only concerns this server's own .ckp temp files,
312 * which live in dirs the worker can reach). Only an inaccessible EXPORT
313 * ROOT (depth 0) or an unexpected errno is fatal.
314 */
315 if (depth > 0 && (ngx_errno == EACCES || ngx_errno == ENOENT
316 || ngx_errno == ENOTDIR || ngx_errno == ELOOP))
317 {
318 brix_log_safe_path(log, NGX_LOG_INFO, ngx_errno,
319 "brix: checkpoint recovery skipping "
320 "inaccessible dir \"%s\"", dir);
321 return NGX_DECLINED;
322 }
323 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
324 "brix: checkpoint recovery cannot scan \"%s\"",
325 dir);
326 return NGX_ERROR;
327 }
328
329 dp = fdopendir(dfd);
330 if (dp == NULL) {
331 ngx_close_file(dfd);
332 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
333 "brix: checkpoint recovery cannot scan \"%s\"",
334 dir);
335 return NGX_ERROR;
336 }
337
338 /* fdopendir owns dfd now; re-derive the scan fd once and refuse to walk on
339 * a failed dirfd() so fstatat never sees a negative directory fd. */
340 scan_fd = dirfd(dp);
341 if (scan_fd < 0) {
342 closedir(dp);
343 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
344 "brix: checkpoint recovery cannot scan \"%s\"",
345 dir);
346 return NGX_ERROR;
347 }
348
349 out->dp = dp;
350 out->scan_fd = scan_fd;
351 return NGX_OK;
352}

354static ngx_int_t
355ckp_recover_scan(ngx_log_t *log, const char *root_canon, const char *dir,
356 ngx_uint_t depth)
357{
358 ckp_scan_dir_t sd;
359 struct dirent *de;
360 ngx_int_t rc;
361
362 if (depth > 128) {
363 brix_log_safe_path(log, NGX_LOG_ERR, 0,
364 "brix: checkpoint recovery depth exceeded at "
365 "\"%s\"", dir);
366 return NGX_ERROR;
367 }
368
369 rc = ckp_recover_open_dir(log, root_canon, dir, depth, &sd);
370 if (rc == NGX_DECLINED) {
371 return NGX_OK; /* skippable subdirectory — logged by the helper */
372 }
373 if (rc != NGX_OK) {
374 return NGX_ERROR;
375 }
376
377 while ((de = readdir(sd.dp)) != NULL) {
378 char path[PATH_MAX];
379 size_t dlen, nlen;
380 struct stat st;
381
382 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0) {
383 continue;
384 }
385
386 dlen = strlen(dir);
387 nlen = strlen(de->d_name);
388 if (dlen + 1 + nlen >= sizeof(path)) {
389 closedir(sd.dp);
390 return NGX_ERROR;
391 }
392
393 ngx_memcpy(path, dir, dlen);
394 path[dlen] = '/';
395 ngx_memcpy(path + dlen + 1, de->d_name, nlen + 1);
396
397 if (fstatat(sd.scan_fd, de->d_name, &st, AT_SYMLINK_NOFOLLOW) != 0) {
398 /* A transiently-removed or inaccessible entry: skip it, don't abort
399 * the whole recovery (and thus the worker). */
400 brix_log_safe_path(log, NGX_LOG_INFO, ngx_errno,
401 "brix: checkpoint recovery skipping entry "
402 "\"%s\"", path);
403 continue;
404 }
405
406 if (S_ISDIR(st.st_mode)) {
407 if (ckp_recover_scan(log, root_canon, path, depth + 1) != NGX_OK) {
408 closedir(sd.dp);
409 return NGX_ERROR;
410 }
411 continue;
412 }
413
414 if (S_ISREG(st.st_mode) && ckp_name_has_suffix(de->d_name)) {
415 if (ckp_recover_one(log, root_canon, path) != NGX_OK) {
416 closedir(sd.dp);
417 return NGX_ERROR;
418 }
419 }
420 }
421
422 closedir(sd.dp);
423 return NGX_OK;
424}

426ngx_int_t
427brix_chkpoint_recover_root(ngx_log_t *log, const char *root_canon)
428{
429 char lock_path[PATH_MAX];
430 size_t root_len;
431 int lock_fd;
432 ngx_int_t rc;
433
434 if (root_canon == NULL || root_canon[0] == '\0') {
435 return NGX_OK;
436 }
437 /* A pure cache node (no brix_root) anchors at the "/" namespace: there is no
438 * local export tree, so no checkpoint journal to recover (and "/" is not a
439 * writable place to drop a recovery lock). Nothing to do. */
440 if (root_canon[0] == '/' && root_canon[1] == '\0') {
441 return NGX_OK;
442 }
443
444 root_len = strlen(root_canon);
445 if (root_len + sizeof("/.nginx-xrootd-ckp-recovery.lock")
446 > sizeof(lock_path))
447 {
448 return NGX_ERROR;
449 }
450
451 ngx_memcpy(lock_path, root_canon, root_len);
452 ngx_memcpy(lock_path + root_len, "/.nginx-xrootd-ckp-recovery.lock",
453 sizeof("/.nginx-xrootd-ckp-recovery.lock"));
454
455 lock_fd = open(lock_path, O_CREAT | O_RDWR | O_CLOEXEC | O_NOFOLLOW, BRIX_ROOT_PRIVATE_FILE_MODE);
456 if (lock_fd < 0) {
457 /* An export this worker cannot write is not a recovery failure: a .ckp
458 * snapshot is only ever produced by a worker writing INTO this root, so
459 * a root that refuses our writes cannot hold a journal to roll back.
460 * Read-only exports (EROFS) and permission-restricted ones (EACCES /
461 * EPERM — e.g. a root-master fleet whose workers drop to an
462 * unprivileged user) are legitimate read-serving deployments, so skip
463 * recovery and carry on rather than failing worker init and
464 * crash-looping the server.
465 * Every other errno still fails loudly — it signals a genuinely broken
466 * export (ENOTDIR, ENAMETOOLONG, ELOOP from a hostile symlink, …). */
467 if (ngx_errno == EACCES || ngx_errno == EPERM || ngx_errno == EROFS) {
468 brix_log_safe_path(log, NGX_LOG_WARN, ngx_errno,
469 "brix: checkpoint recovery skipped — export "
470 "root is not writable by this worker "
471 "\"%s\"", lock_path);
472 return NGX_OK;
473 }
474 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
475 "brix: checkpoint recovery lock failed "
476 "\"%s\"", lock_path);
477 return NGX_ERROR;
478 }
479
480 if (flock(lock_fd, LOCK_EX) != 0) {
481 ngx_close_file(lock_fd);
482 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
483 "brix: checkpoint recovery cannot lock "
484 "\"%s\"", lock_path);
485 return NGX_ERROR;
486 }
487
488 rc = ckp_recover_scan(log, root_canon, root_canon, 0);
489
490 (void) flock(lock_fd, LOCK_UN);
491 ngx_close_file(lock_fd);
492
493 return rc;
494}
