1/*
2 * policy.c — protbind rule evaluation: peer → ordered protocol set.
3 *
4 * WHAT: Owns the decision half of the engine — building a frontend's base set,
5 * picking the first rule whose template matches the peer, and composing
6 * the rule's protocol list with the base set into the ordered result the
7 * caller advertises and enforces.
8 *
9 * WHY: Kept apart from match.c (pure lookups) and config.c (parsing) so the
10 * precedence rules — first match wins, `none` short-circuits, `only`
11 * truncates, the default form appends the base — live in one small file
12 * that can be read end-to-end while auditing an access decision.
13 *
14 * HOW: Every function here is pure over its arguments: no config lookups, no
15 * DNS, no logging. Callers supply the already-resolved peer identity,
16 * which is what lets the root:// stream path and the HTTP path share
17 * this code unchanged.
18 */

23/*
24 * brix_protbind_append — add one protocol to a set, ignoring duplicates.
25 *
26 * WHAT: Appends `proto` unless it is already present or the set is full.
27 *
28 * WHY: The default (non-`only`) rule form concatenates the rule's list with
29 * the base set, and the two routinely overlap ("gsi" listed first, base
30 * {ztn,gsi}); silently de-duplicating keeps the advertised sec token from
31 * offering the same scheme twice, which stock clients treat as a
32 * malformed protocol list.
33 *
34 * HOW: Linear membership scan (the set holds at most 8 ids), then append.
35 */
36static void
37brix_protbind_append(brix_protbind_set_t *set, ngx_uint_t proto)
38{
39 ngx_uint_t index;
40
41 for (index = 0; index < set->count; index++) {
42 if (set->protos[index] == proto) {
43 return;
44 }
45 }
46
47 if (set->count < BRIX_PROTBIND_MAX_PROTOS) {
48 set->protos[set->count++] = proto;
49 }
50}

52/* ---- Base protocol set implied by a stream listener's brix_auth mode ----
53 *
54 * WHAT: Fills *out with the protocols `brix_auth <mode>` allows on its own:
55 * none → empty and require_auth 0, both → {ztn, gsi}, anything else →
56 * that single scheme.
57 *
58 * WHY: protbind refines an existing listener policy rather than replacing it,
59 * so the pre-protbind behaviour has to be expressible as a set. Encoding
60 * it here — including `both`'s token-then-GSI order, which is the order
61 * the sec token has always advertised — is what guarantees that a config
62 * with no brix_protbind line behaves exactly as it did before.
63 *
64 * HOW: 1. Zero the set. 2. Return early for the anonymous mode. 3. Expand
65 * `both` into its two members; otherwise append the mode itself.
66 */
67void
68brix_protbind_base_set(ngx_uint_t base_auth, brix_protbind_set_t *out)
69{
70 ngx_memzero(out, sizeof(*out));
71
72 if (base_auth == BRIX_AUTH_NONE) {
73 return;
74 }
75
76 out->require_auth = 1;
77
78 if (base_auth == BRIX_AUTH_BOTH) {
79 brix_protbind_append(out, BRIX_AUTH_TOKEN);
80 brix_protbind_append(out, BRIX_AUTH_GSI);
81 return;
82 }
83
84 brix_protbind_append(out, base_auth);
85}

87/* ---- Base protocol set for an HTTP frontend ----
88 *
89 * WHAT: Fills *out with {gsi, ztn, pwd} — the three credential sources a
90 * WebDAV/HTTP location can actually run — and stamps `require_auth` from
91 * the location's own auth=none/optional/required decision.
92 *
93 * WHY: HTTP has no login negotiation: a request arrives carrying whatever
94 * credentials it has, and the server tries the sources in descending
95 * strength. That historical order (client cert, then bearer token, then
96 * Basic) IS the base ordering, so a location with no protbind rule keeps
97 * byte-identical behaviour. The remaining schemes (sss/unix/krb5/host)
98 * have no HTTP transport and are therefore absent by construction.
99 *
100 * HOW: Zero, append the three ids in order, record require_auth.
101 */
102void
103brix_protbind_http_base(ngx_flag_t require_auth, brix_protbind_set_t *out)
104{
105 ngx_memzero(out, sizeof(*out));
106
107 brix_protbind_append(out, BRIX_AUTH_GSI);
108 brix_protbind_append(out, BRIX_AUTH_TOKEN);
109 brix_protbind_append(out, BRIX_AUTH_PWD);
110
111 out->require_auth = require_auth;
112}

114/*
115 * brix_protbind_rule_matches — does either peer identity satisfy a template?
116 *
117 * WHAT: Returns 1 when the template matches the peer's hostname or, failing
118 * that, its textual IP address.
119 *
120 * WHY: Sites write templates against names ("*.cern.ch") but reverse DNS can
121 * be absent or broken, and admins also legitimately write literal
122 * addresses or address prefixes ("192.168.1.*"). Trying both identities
123 * covers each intent without a separate directive; a bare "*" matches via
124 * the hostname arm even when both identities are unknown.
125 *
126 * HOW: Match the hostname first (the intended identity), then the IP.
127 */
128static ngx_flag_t
129brix_protbind_rule_matches(const brix_protbind_rule_t *rule,
130 const char *peer_host, const char *peer_ip)
131{
132 if (brix_protbind_host_match(&rule->host_tpl, peer_host)) {
133 return 1;
134 }
135
136 return brix_protbind_host_match(&rule->host_tpl, peer_ip);
137}

139/*
140 * brix_protbind_apply — turn the winning rule into the resolved set.
141 *
142 * WHAT: Fills *out from `rule`, folding in `base` for the default rule form.
143 *
144 * WHY: The three rule modes differ only in what surrounds the rule's own
145 * list, so expressing them as one function keeps the precedence in a
146 * single readable place instead of spread across the search loop.
147 *
148 * HOW: 1. `none` → the empty, auth-less set. 2. Copy the rule's protocols in
149 * written order (that order is the advertisement order). 3. For the
150 * default (non-`only`) form append the base protocols not already named.
151 * 4. require_auth follows from the set being non-empty.
152 */
153static void
154brix_protbind_apply(const brix_protbind_rule_t *rule,
155 const brix_protbind_set_t *base, brix_protbind_set_t *out)
156{
157 ngx_uint_t index;
158
159 ngx_memzero(out, sizeof(*out));
160
161 if (rule->mode == BRIX_PROTBIND_NONE) {
162 return;
163 }
164
165 for (index = 0; index < rule->count; index++) {
166 brix_protbind_append(out, rule->protos[index]);
167 }
168
169 if (rule->mode != BRIX_PROTBIND_ONLY) {
170 for (index = 0; index < base->count; index++) {
171 brix_protbind_append(out, base->protos[index]);
172 }
173 }
174
175 out->require_auth = (out->count > 0);
176}

178/* ---- Resolve the effective protocol set for one peer ----
179 *
180 * WHAT: Fills *out with the ordered protocols this peer may authenticate with;
181 * with no rules, or no rule matching, *out is a copy of *base.
182 *
183 * WHY: This is the single entry point every frontend calls, so "first
184 * matching rule wins" is enforced in exactly one place. First-match (not
185 * most-specific-match) is stock XRootD's rule and the reason admins put
186 * their `*` catch-all last.
187 *
188 * HOW: 1. Copy the base as the default outcome. 2. Walk the rules in
189 * configuration order. 3. On the first template that matches either
190 * peer identity, apply it and return.
191 */
192void
193brix_protbind_resolve(ngx_array_t *rules, const brix_protbind_set_t *base,
194 const char *peer_host, const char *peer_ip, brix_protbind_set_t *out)
195{
196 brix_protbind_rule_t *rule;
197 ngx_uint_t index;
198
199 *out = *base;
200
201 if (rules == NULL || rules->nelts == 0) {
202 return;
203 }
204
205 rule = rules->elts;
206 for (index = 0; index < rules->nelts; index++) {
207 if (brix_protbind_rule_matches(&rule[index], peer_host, peer_ip)) {
208 brix_protbind_apply(&rule[index], base, out);
209 return;
210 }
211 }
212}

214/* ---- Membership test for an offered credential ----
215 *
216 * WHAT: Returns 1 when `proto` is in the resolved set.
217 *
218 * WHY: Advertising a protocol list is not enforcement: a client can send any
219 * credential type it likes regardless of what the sec token offered, so
220 * the kXR_auth dispatcher must re-check membership before running a
221 * scheme's handler. One shared predicate keeps the advertised set and
222 * the enforced set from drifting apart.
223 *
224 * HOW: Linear scan over at most BRIX_PROTBIND_MAX_PROTOS ids.
225 */
226ngx_flag_t
227brix_protbind_allows(const brix_protbind_set_t *set, ngx_uint_t proto)
228{
229 ngx_uint_t index;
230
231 for (index = 0; index < set->count; index++) {
232 if (set->protos[index] == proto) {
233 return 1;
234 }
235 }
236
237 return 0;
238}
