1/*
2 * registry.c — SHM-backed session registry for cross-worker coordination.
3 *
4 * Stores each session's metadata (sessid, DN, VO list, token_auth) in a shared-
5 * memory table so any worker can resolve a session it did not itself create —
6 * needed because kXR_bind secondaries and proxied requests may land on a
7 * different worker than the primary login. All slot access is mutex-protected
8 * (spin+yield; see shm_slots). The published-handle table lives in handles.c.
9 */

27/* Per-process pointer to the SHM session table (NULL until the zone is set up). */
28static brix_session_table_t *
29session_table(void)
30{
31 /* Single read of zone->data: the checked value IS the returned value. */
32 void *table = brix_session_shm_zone ? brix_session_shm_zone->data : NULL;
33
34 if (table == NULL || table == (void *) 1) {
35 return NULL;
36 }
37 return (brix_session_table_t *) table;
38}

40/* Shared-memory zone init callback: lay the session table out in the zone and
41 * create its spin+yield mutex. Returns NGX_OK / NGX_ERROR. */
42ngx_int_t
43brix_session_shm_init_zone(ngx_shm_zone_t *shm_zone, void *data)
44{
45 ngx_flag_t fresh;
46 brix_session_table_t *tbl;
47
48 /*
49 * Allocate the table FROM the slab pool (never lay it over shm.addr) so
50 * nginx's ngx_unlock_mutexes() — which treats every zone's shm.addr as an
51 * ngx_slab_pool_t header on every child death — does not get clobbered and
52 * SIGSEGV the master. The helper handles fresh-alloc, reload (data != NULL),
53 * and re-attach, creates brix_session_mutex from the table's leading
54 * ngx_shmtx_sh_t lock, and publishes the table via shm_zone->data.
55 */
56 tbl = brix_shm_table_alloc(shm_zone, data,
57 sizeof(brix_session_table_t)
58 + (size_t) brix_session_registry_nslots
59 * sizeof(brix_session_entry_t),
60 &brix_session_mutex, &fresh);
61 if (tbl == NULL) {
62 return NGX_ERROR;
63 }
64
65 if (fresh) {
66 tbl->capacity = brix_session_registry_nslots;
67 }
68
69 return NGX_OK;
70}

75brix_configure_session_registry(ngx_conf_t *cf, ngx_uint_t slots)
76{

88 brix_session_shm_zone = ngx_shared_memory_add(cf, &zone_name,

91 if (brix_session_shm_zone == NULL) {
92 return NGX_ERROR;
93 }
94
95 brix_shm_zone_warn_on_resize(cf, brix_session_shm_zone,
96 "brix_session_slots");
97
98 brix_session_shm_zone->init = brix_session_shm_init_zone;
99 brix_session_shm_zone->data = (void *) 1;
100
101 zone_size = brix_shm_zone_size(sizeof(brix_shared_handle_table_t));
102 brix_handle_shm_zone = ngx_shared_memory_add(cf, &handle_zone_name,
103 zone_size,
104 &ngx_stream_brix_module);
105 if (brix_handle_shm_zone == NULL) {
106 return NGX_ERROR;
107 }
108
109 brix_handle_shm_zone->init = brix_handle_shm_init_zone;
110 brix_handle_shm_zone->data = (void *) 1;
111
112 return NGX_OK;
113}

115/* ---- §1.4: which worker owns a session's primary connection ----
116 *
117 * WHAT: Returns the owner ngx_worker slot recorded at registration, or -1
118 * for an unknown session.
119 *
120 * WHY: A kXR_bind that lands on a different worker than the session's
121 * primary cannot be offloaded there (the offload conn map is
122 * per-worker); bind_migrate.c uses this to hand the secondary's fd to
123 * the owning worker instead of falling back to inline responses.
124 *
125 * HOW: Same lock→find→read→unlock shape as the pathid-bitmap operations.
126 */
127ngx_int_t
128brix_session_owner_worker(const u_char sessid[BRIX_SESSION_ID_LEN])
129{
130 brix_session_table_t *tbl = session_table();
131 brix_session_entry_t *e;
132 ngx_int_t owner = -1;
133
134 if (tbl == NULL) {
135 return -1;
136 }
137 ngx_shmtx_lock(&brix_session_mutex);
138 e = brix_session_find_locked(tbl, sessid);
139 if (e != NULL) {
140 owner = e->owner_worker;
141 }
142 ngx_shmtx_unlock(&brix_session_mutex);
143 return owner;
144}

146/* ---- Set / clear / test a session's bound-pathid bit ----
147 *
148 * WHAT: brix_session_pathid_bind marks `pathid` bound for `sessid`;
149 * _unbind clears it; _bound returns 1 when it is currently set.
150 * All are no-ops (or 0) for pathid outside 1-253 or an unknown session.
151 *
152 * WHY: kXR_bind assigns pathids on the SECONDARY connection's worker while
153 * pathid-tagged requests arrive on the primary (any worker), so the
154 * validation truth must live in the shared session registry. Stock
155 * refuses an unbound pathid with kXR_ArgInvalid "invalid path ID"
156 * (verified live against 5.6.9); this bitmap is what makes that answer
157 * possible (parity-audit §1.2) and the groundwork for response
158 * offloading (§1.1).
159 *
160 * HOW: Lock the registry mutex, find the slot by sessid, set/clear/test bit
161 * pathid in the 32-byte map, unlock. Callers never hold the mutex.
162 */
163/* Set (`set` != 0) or clear the pathid bit in a session's map under the
164 * registry mutex; both bind and unbind are this one guarded slot-mutation. */
165static void
166brix_session_pathid_set(const u_char sessid[BRIX_SESSION_ID_LEN],
167 unsigned pathid, int set)
168{
169 brix_session_table_t *tbl = session_table();
170 brix_session_entry_t *e;
171
172 /* Separate ifs: gcc 11's -fanalyzer loses the non-NULL constraint on an
173 * accessor-returned pointer inside a compound || guard and reports an
174 * infeasible NULL deref in brix_session_find_locked. */
175 if (tbl == NULL) {
176 return;
177 }
178 if (pathid < 1 || pathid > 253) {
179 return;
180 }
181 ngx_shmtx_lock(&brix_session_mutex);
182 e = brix_session_find_locked(tbl, sessid);
183 if (e != NULL) {
184 u_char bit = (u_char) (1u << (pathid % 8));
185
186 if (set) {
187 e->pathid_map[pathid / 8] |= bit;
188 } else {
189 e->pathid_map[pathid / 8] &= (u_char) ~bit;
190 }
191 }
192 ngx_shmtx_unlock(&brix_session_mutex);
193}

209int
210brix_session_pathid_bound(const u_char sessid[BRIX_SESSION_ID_LEN],
211 unsigned pathid)
212{
213 brix_session_table_t *tbl = session_table();
214 brix_session_entry_t *e;
215 int bound = 0;
216
217 /* Separate ifs — same gcc 11 compound-guard analyzer limitation. */
218 if (tbl == NULL) {
219 return 0;
220 }
221 if (pathid < 1 || pathid > 253) {
222 return 0;
223 }
224 ngx_shmtx_lock(&brix_session_mutex);
225 e = brix_session_find_locked(tbl, sessid);
226 if (e != NULL) {
227 bound = (e->pathid_map[pathid / 8] >> (pathid % 8)) & 1u;
228 }
229 ngx_shmtx_unlock(&brix_session_mutex);

233/* Store a session's metadata (sessid, DN, VO list, token_auth) in the first free
234 * SHM slot at login completion; a no-op if the sessid is already present.
235 * W5/P90-27.2: an identity at its per-source soft cap recycles its OWN LRU slot
236 * first (self-eviction), so the F4 global reap only ever fires for genuinely
237 * diverse load. Mutex-protected (cross-worker). */
238int
239brix_session_register(const u_char sessid[BRIX_SESSION_ID_LEN],
240 const char *dn, const char *vo_list, ngx_uint_t token_auth)
241{
242 brix_session_table_t *tbl;
243 brix_session_scan_t sc;
244 ngx_msec_t now;
245 int found, reaped = 0;
246 int slot = -1;
247 char src_key[BRIX_SESSION_SRC_KEY_LEN];
248 u_char victim[BRIX_SESSION_ID_LEN];
249
250 tbl = session_table();
251 if (tbl == NULL) {
252 return -1;
253 }
254
255 brix_session_src_key(dn, token_auth, src_key);
256 now = ngx_current_msec;
257
258 ngx_shmtx_lock(&brix_session_mutex);
259
260 found = brix_session_scan(tbl, sessid, now, src_key, &sc);
261
262 /* Per-source soft cap (W5): over-quota identities recycle their own LRU
263 * slot BEFORE consuming a free one or invoking the global reap. */
264 if (!found && sc.src_count >= BRIX_SESSION_PER_SOURCE_SOFT_CAP) {
265 reaped = brix_session_src_cap_evict(tbl, sc.src_lru_slot,
266 &sc.free_slot, victim);
267 }
268
269 /* Table full and no match: try the Phase 27 F4 reap-on-full defence, which
270 * may free the LRU slot (reaped) or reject and count the attempt. */
271 if (!found && !reaped && sc.free_slot == tbl->capacity) {
272 reaped = brix_session_reap_lru(tbl, now, sc.lru_slot, sc.lru_seen,
273 &sc.free_slot, victim);
274 }
275
276 if (found) {
277 slot = (int) sc.match_slot; /* already registered, same slot */
278 } else if (sc.free_slot < tbl->capacity) {
279 brix_session_fill_slot(tbl, sc.free_slot, sessid, dn, vo_list,
280 token_auth, src_key, now);
281 slot = (int) sc.free_slot;
282 }
283
284 ngx_shmtx_unlock(&brix_session_mutex);
285
286 /* Unpublish the reaped victim's handles AFTER releasing the session mutex
287 * (mirrors brix_session_unregister's lock order: session then handle). */
288 if (reaped) {
289 brix_session_finish_eviction(victim);
290 }
291
292 /* Round 15: the slot is the caller's teardown hint; -1 when the table was
293 * unavailable or full (a rejected registration has nothing to tear down). */
294 return slot;
295}

297/* Look a session up by sessid, copying its DN / VO list / token_auth out (used
298 * by kXR_bind secondaries and proxy mode). Returns 1 on hit, 0 on miss.
299 * Mutex-protected. */
300int
301brix_session_lookup(const u_char sessid[BRIX_SESSION_ID_LEN],
302 char *dn_out, size_t dn_size,
303 char *vo_out, size_t vo_size,
304 ngx_uint_t *token_auth_out)
305{
306 brix_session_table_t *tbl;
307 brix_session_entry_t *e;
308 ngx_uint_t i;
309 int found = 0;
310
311 tbl = session_table();
312 if (tbl == NULL) {
313 return 0;
314 }
315
316 ngx_shmtx_lock(&brix_session_mutex);
317
318 /* Live prefix only: high_water bounds every occupied slot. */
319 for (i = 0; i < tbl->high_water; i++) {
320 e = &tbl->slots[i];
321 if (!e->in_use) {
322 continue;
323 }
324 if (ngx_memcmp(e->sessid, sessid, BRIX_SESSION_ID_LEN) == 0) {
325 ngx_cpystrn((u_char *) dn_out, (u_char *) e->dn, dn_size);
326 ngx_cpystrn((u_char *) vo_out, (u_char *) e->vo_list, vo_size);
327 *token_auth_out = e->token_auth;
328 e->last_seen = ngx_current_msec; /* F4: activity keeps it off the LRU */
329 found = 1;
330 break;
331 }
332 }
333
334 ngx_shmtx_unlock(&brix_session_mutex);
335 return found;
336}

346/*
347 *
348 * WHAT: As brix_session_unregister(), but clears the registry slot that
349 * brix_session_register() reported at login (round 15).
350 *
351 * WHY: The scan below is O(live sessions) under the single cross-worker
352 * brix_session_mutex, and it runs on EVERY disconnect. Under an ultra-parallel
353 * storm the live prefix is at its longest exactly when every session is also
354 * disconnecting, so the cost per teardown grew with concurrency instead of
355 * backing off — the same shape round 14 removed from the handle table.
356 *
357 * HOW: A session occupies exactly one slot, and register() reports it. The
358 * sessid is re-checked at the hinted slot under the lock, so a hint made stale
359 * by the F4 reap or the W5 self-eviction simply does not match and clears
360 * nothing — which is correct, because an evicted session has no entry left to
361 * clear. That re-check also makes the hinted path STRICTER than the scan: if
362 * this session was evicted and then re-registered by a NEW connection into a
363 * different slot, the scan would find and destroy that live registration, while
364 * the hint refuses to touch a slot it does not own. hint < 0 (never
365 * registered, or a caller with no ctx such as kXR_endsess for a foreign
366 * sessid) keeps the original scan.
367 */
368void
369brix_session_unregister_hinted(const u_char sessid[BRIX_SESSION_ID_LEN],
370 int slot_hint)
371{
372 brix_session_table_t *tbl;
373 brix_session_entry_t *e;
374 ngx_uint_t i;
375
376 tbl = session_table();
377 if (tbl == NULL) {
378 return;
379 }
380
381 ngx_shmtx_lock(&brix_session_mutex);
382
383 if (slot_hint >= 0 && (ngx_uint_t) slot_hint < tbl->high_water) {
384 e = &tbl->slots[slot_hint];
385 if (e->in_use
386 && ngx_memcmp(e->sessid, sessid, BRIX_SESSION_ID_LEN) == 0)
387 {
388 ngx_memzero(e, sizeof(*e));
389 brix_session_shrink(tbl);
390 }
391
392 } else if (slot_hint < 0) {
393 /* Live prefix only: high_water bounds every occupied slot. */
394 for (i = 0; i < tbl->high_water; i++) {
395 e = &tbl->slots[i];
396 if (e->in_use
397 && ngx_memcmp(e->sessid, sessid, BRIX_SESSION_ID_LEN) == 0)
398 {
399 ngx_memzero(e, sizeof(*e));
400 /* This is the only clear that is not immediately refilled by
401 * its caller (reap and self-eviction both hand the slot
402 * straight to fill_slot), so it alone retires a freed top
403 * run. */
404 brix_session_shrink(tbl);
405 break;
406 }
407 }
408 }
409
410 ngx_shmtx_unlock(&brix_session_mutex);
411 brix_session_handle_unpublish_all(sessid);
412}
