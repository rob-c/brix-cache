9/*
10 * WHAT: A parsed --cksum spec: the resolved algorithm, its lowercase name (for
11 * kXR_Qcksum + messages), the optional mode suffix, and the silent flag.
12 * WHY: cksum_verify's spec grammar ("<type>[:source|:print|:<value>]") is parsed
13 * once up front; carrying the result (plus the caller's silent flag) in a
14 * small struct lets the digest and per-mode helpers take one argument
15 * instead of several loose values.
16 * HOW: `mode` points into the caller's `spec` (after the colon) or is NULL when
17 * no suffix was given; `name` is a NUL-terminated copy of the type token;
18 * `silent` suppresses the success prints.
19 */
20typedef struct {
21 brix_cksum_algo algo;
22 char name[32];
23 const char *mode;
24 int silent;
25} cksum_spec_t;

28/*
29 * WHAT: Parse `spec` into a resolved algorithm + mode (the "resolve-algo" step).
30 * Returns 0 on success; on a usage error sets `st` and returns -1.
31 * WHY: Splitting the grammar parse out of cksum_verify keeps the orchestrator a
32 * flat sequence (parse → digest → compare) with no nested branching.
33 * HOW: Split at the first ':' — the token before it is the algorithm name, the
34 * remainder (or NULL) is the mode. Reject an empty or over-long name, then
35 * resolve the name to an algo id via the shared parser.
36 */
37static int
38cksum_parse_spec(const char *spec, cksum_spec_t *out, brix_status *st)
39{
40 const char *colon = strchr(spec, ':');
41 size_t alen = colon ? (size_t) (colon - spec) : strlen(spec);
42
43 if (alen == 0 || alen >= sizeof(out->name)) {
44 brix_status_set(st, XRDC_EUSAGE, 0, "bad --cksum type");
45 return -1;
46 }
47 memcpy(out->name, spec, alen);
48 out->name[alen] = '\0';
49 out->mode = colon ? colon + 1 : NULL;
50
51 if (brix_cksum_algo_parse(out->name, &out->algo) != 0) {
52 brix_status_set(st, XRDC_EUSAGE, 0,
53 "unsupported --cksum type \"%s\"", out->name);
54 return -1;
55 }
56 return 0;
57}

60/*
61 * WHAT: Compute `spec->algo` over the local file `local_path` into `hex` (the
62 * "compute-local" step). Returns 0 on success, -1 (st set) otherwise.
63 * WHY: The open/digest/close sequence is a single self-contained unit with one
64 * cleanup path; isolating it removes an fd lifetime from the orchestrator.
65 * HOW: open() the file read-only, run the shared fd digest, and close on every
66 * path (including the digest-failure path).
67 */
68static int
69cksum_compute_local(const char *local_path, const cksum_spec_t *spec,
70 char *hex, size_t hexsz, brix_status *st)
71{
72 int lfd = open(local_path, O_RDONLY);
73
74 if (lfd < 0) {
75 brix_status_set(st, XRDC_EUSAGE, errno,
76 "open %s for checksum: %s", local_path, strerror(errno));
77 return -1;
78 }
79 if (brix_cksum_fd(lfd, spec->algo, hex, hexsz, st) != 0) {
80 close(lfd);
81 return -1;
82 }
83 close(lfd);
84 return 0;
85}

88/*
89 * WHAT: :source / :end2end mode — fetch the server's digest for `remote_path` and
90 * compare it to `local_hex` (the "fetch-remote + compare" step).
91 * WHY: The server round-trip and the mismatch classification are one logical
92 * branch of cksum_verify; extracting it keeps the orchestrator's mode
93 * dispatch a flat ladder of single-line calls.
94 * HOW: Require a remote path, query kXR_Qcksum, then case-insensitively compare.
95 * A query/transport failure is UNVERIFIED (digest UNKNOWN); a definite
96 * inequality is a non-retryable MISMATCH (EINTEGRITY).
97 */
98static int
99cksum_verify_source(brix_conn *c, const char *remote_path, const char *local_hex,
100 const cksum_spec_t *spec, brix_status *st)
101{
102 char server_hex[129];
103
104 if (remote_path == NULL) {
105 brix_status_set(st, XRDC_EUSAGE, 0, "--cksum:source has no remote");
106 return XRDC_CK_UNVERIFIED;
107 }
108 if (brix_query_cksum(c, remote_path, spec->name,
109 server_hex, sizeof(server_hex), st) != 0) {
110 return XRDC_CK_UNVERIFIED; /* server digest UNKNOWN, not WRONG */
111 }
112 if (strcasecmp(local_hex, server_hex) != 0) {
113 /* A checksum mismatch is a data-integrity failure, not a transient
114 * framing fault — classify it non-retryable so no resilient loop
115 * spins re-verifying the same bytes. */
116 brix_status_set(st, XRDC_EINTEGRITY, 0,
117 "%s mismatch: local %s != server %s",
118 spec->name, local_hex, server_hex);
119 return XRDC_CK_MISMATCH;
120 }
121 if (!spec->silent) {
122 printf("%s %s OK (matches server)\n", spec->name, local_hex);
123 }
124 return XRDC_CK_OK;
125}

128/*
129 * WHAT: :<value> mode — compare `local_hex` to the literal expected digest carried
130 * in `spec->mode` (the "compare-literal" step).
131 * WHY: The literal branch mirrors the :source branch but has no round-trip;
132 * splitting it keeps both compare paths at the same altitude.
133 * HOW: Case-insensitive compare; inequality is a non-retryable MISMATCH.
134 */
135static int
136cksum_verify_literal(const char *local_hex, const cksum_spec_t *spec,
137 brix_status *st)
138{
139 if (strcasecmp(local_hex, spec->mode) != 0) {
140 brix_status_set(st, XRDC_EINTEGRITY, 0,
141 "%s mismatch: got %s expected %s",
142 spec->name, local_hex, spec->mode);
143 return XRDC_CK_MISMATCH;
144 }
145 if (!spec->silent) {
146 printf("%s %s OK\n", spec->name, local_hex);
147 }
148 return XRDC_CK_OK;
149}

152/*
153 * --cksum handling. `spec` is "<type>[:source|:print|:<value>]". Computes the
154 * named checksum over the local file `local_path` (the bytes we actually moved)
155 * and then, by mode:
156 * :source / :end2end → query the server (kXR_Qcksum) for `remote_path` on the
157 * already-open connection `c` and require they match;
158 * :<value> → require the local digest equals the given hex;
159 * :print / (none) → print "<type> <digest>" (unless silent).
160 * Returns XRDC_CK_OK (matched/printed), XRDC_CK_MISMATCH (digest known and WRONG
161 * — caller drops the destination), or XRDC_CK_UNVERIFIED (query/transport/usage
162 * error — the digest is UNKNOWN, so the caller keeps the file and only warns;
163 * deleting a byte-perfect download because a control-plane query hiccupped would
164 * be the inverse footgun).
165 */
166int
167cksum_verify(brix_conn *c, const char *remote_path, const char *local_path,
168 const char *spec, int silent, brix_status *st)
169{
170 cksum_spec_t parsed = { 0 };
171 char local_hex[129];
172
173 if (cksum_parse_spec(spec, &parsed, st) != 0) {
174 return XRDC_CK_UNVERIFIED;
175 }
176 parsed.silent = silent;
177
178 if (local_path == NULL) {
179 /* stdio endpoint — nothing on disk to digest; skip rather than lie. */
180 if (!silent) {
181 fprintf(stderr, "xrdcp: --cksum skipped for stdin/stdout\n");
182 }
183 return XRDC_CK_OK;
184 }
185
186 if (cksum_compute_local(local_path, &parsed,
187 local_hex, sizeof(local_hex), st) != 0) {
188 return XRDC_CK_UNVERIFIED;
189 }
190
191 if (parsed.mode != NULL
192 && (strcmp(parsed.mode, "source") == 0
193 || strcmp(parsed.mode, "end2end") == 0)) {
194 return cksum_verify_source(c, remote_path, local_hex, &parsed, st);
195 }
196
197 if (parsed.mode != NULL && strcmp(parsed.mode, "print") != 0) {
198 return cksum_verify_literal(local_hex, &parsed, st);
199 }
200
201 /* print / no mode */
202 if (!silent) {
203 printf("%s %s\n", parsed.name, local_hex);
204 }
205 return XRDC_CK_OK;
206}
