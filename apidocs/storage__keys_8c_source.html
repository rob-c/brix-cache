15/* ---- Choose a physical fattr key ----
16 * WHAT: Return zero with a direct or wrapped key, otherwise -1 with errno.
17 * WHY: Long or reserved logical names need collision-checked portable storage.
18 * HOW: 1. Validate length. 2. Choose direct or digest form. 3. Bound the output.
19 */
20int
21fattr_storage_key(const char *name, size_t nlen, char *key, size_t keycap,
22 int *wrapped)
23{
24 uint8_t digest[32];
25 size_t direct_len;
26
27 if (name == NULL || nlen == 0 || nlen > kXR_faMaxNlen) {
28 errno = EINVAL;
29 return -1;
30 }
31 direct_len = BRIX_FATTR_XKEY_PFX_LEN + nlen;
32 *wrapped = direct_len > BRIX_XATTR_NAME_MAX
33 || (nlen >= sizeof("$brix$") - 1U
34 && memcmp(name, "$brix$", sizeof("$brix$") - 1U) == 0);
35 if (!*wrapped) {
36 if (direct_len + 1U > keycap) {
37 errno = ENAMETOOLONG;
38 return -1;
39 }
40 memcpy(key, BRIX_FATTR_XKEY_PFX, BRIX_FATTR_XKEY_PFX_LEN);
41 memcpy(key + BRIX_FATTR_XKEY_PFX_LEN, name, nlen);
42 key[direct_len] = '\0';
43 return 0;
44 }
45 if (FATTR_STORED_KEY_LEN + 1U > keycap
46 || !brix_sha256((const uint8_t *) name, nlen, digest))
47 {
48 errno = EIO;
49 return -1;
50 }
51 memcpy(key, FATTR_STORED_PREFIX, FATTR_STORED_PREFIX_LEN);
52 brix_hex_encode(digest, sizeof(digest), key + FATTR_STORED_PREFIX_LEN);
53 return 0;
54}

58 * WHAT: Return zero with the SHA-256-derived key, otherwise -1 with errno.
59 * WHY: Fallback reads must use exactly the same reserved key namespace as writes.

81/* ---- Derive a bounded chunk key ----
82 * WHAT: Return zero with a chunk key, otherwise -1 with EINVAL.
83 * WHY: Chunk values must remain tied to the logical name and value digest.
84 * HOW: 1. Encode the value digest. 2. Format the key. 3. Reject overflow or bad index.
85 */
86int
87fattr_storage_chunk_key(char *out, size_t cap, const char *base,
88 const uint8_t value_digest[32], size_t index)
89{
90 char value_hex[33];
91 int n;
92
93 brix_hex_encode(value_digest, 16, value_hex);
94 value_hex[32] = '\0';
95 n = snprintf(out, cap, "%s%s.%s.%02x", FATTR_CHUNK_PREFIX,
96 base + FATTR_STORED_PREFIX_LEN, value_hex,
97 (unsigned) index);
98 if (n < 0 || (size_t) n >= cap || index >= FATTR_CHUNK_MAX) {
99 errno = EINVAL;
100 return -1;
101 }
102 return 0;
103}
