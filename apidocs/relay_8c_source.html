7/*
8 * relay.c — transparent pass-through relay: a bidirectional buffered TCP relay
9 * from the stream xrootd listener to an upstream XRootD server, with a
10 * non-consuming tap that decodes the cleartext frames in flight. See relay.h
11 * for WHAT/WHY. Modeled on src/handoff/handoff.c — the difference is the
12 * upstream is the configured XRootD server, the relay engages before any frame
13 * is read, and each freshly-recv'd chunk is fed to a per-direction tap decoder.
14 */
15

93/* Tear the relay down once: free the upstream connection (best effort) and
94 * finalize the stream session (which closes the client and frees its pool). */
95static void
96relay_close(brix_relay_t *r)
97{
98 if (r->closed) {
99 return;
100 }
101 r->closed = 1;
102
103 if (r->metrics != NULL) {
104 ngx_atomic_fetch_add(&r->metrics->connections_active,
105 (ngx_atomic_int_t) -1);
106 }
107 if (r->upstream != NULL) {
108 ngx_close_connection(r->upstream);
109 r->upstream = NULL;
110 }

112}

115/*
116 * Move bytes one direction: drain the pending window [off,end) into `to`, then
117 * read more from `from` and loop. Each freshly-read chunk is fed to `dec` (the
118 * direction's tap decoder) exactly once — before it is queued for send, never on
119 * a re-send after NGX_AGAIN — so the tap observes every byte once, in order.
120 * When the guard sink flags a frame in that chunk, the chunk is NOT forwarded
121 * and the whole relay comes down (bounce = connection drop).
122 * Arms the appropriate event (and an idle timer) when either side would block;
123 * returns NGX_ERROR on EOF/error so the caller closes the whole relay.
124 */
125static ngx_int_t
126relay_pump(brix_relay_t *r, ngx_connection_t *from, ngx_connection_t *to,
127 u_char *buf, size_t *off, size_t *end, brix_tap_stream_t *dec)
128{
129 ssize_t n;
130
131 for ( ;; ) {
132 while (*off < *end) {
133 n = to->send(to, buf + *off, *end - *off);
134 if (n > 0) {
135 *off += (size_t) n;
136 continue;
137 }
138 if (n == NGX_AGAIN) {
139 if (ngx_handle_write_event(to->write, 0) != NGX_OK) {
140 return NGX_ERROR;
141 }
142 ngx_add_timer(to->write, BRIX_RELAY_IDLE);
143 return NGX_OK;
144 }
145 return NGX_ERROR; /* 0 or NGX_ERROR on the write side */
146 }
147 *off = *end = 0;
148
149 n = from->recv(from, buf, BRIX_RELAY_BUF);
150 if (n > 0) {
151 if (from == r->client) {
152 /* First client bytes: reject a peer not speaking root before
153 * it is forwarded (the tap only sees decoded kXR frames). */
154 brix_relay_guard_handshake(&r->guard, buf, (size_t) n);
155 }
156 brix_tap_stream_feed(dec, buf, (size_t) n); /* tap once */
157 if (brix_relay_guard_should_drop(&r->guard)) {
158 return NGX_ERROR; /* BOUNCE: never forward this chunk */
159 }
160 *end = (size_t) n;
161 *off = 0;
162 continue;
163 }
164 if (n == NGX_AGAIN) {
165 if (ngx_handle_read_event(from->read, 0) != NGX_OK) {
166 return NGX_ERROR;
167 }
168 ngx_add_timer(from->read, BRIX_RELAY_IDLE);
169 return NGX_OK;
170 }
171 return NGX_ERROR; /* EOF (0) or error: tear the relay down */
172 }
173}

231/* Promote a connected upstream to the running relay: install the four relay
232 * handlers and kick both directions once (the client->upstream pump reads the
233 * client's hello fresh — there is no pre-read prefix in transparent mode). */
234static void
235relay_begin(brix_relay_t *r)
236{
237 r->connected = 1;
238
239 r->client->read->handler = relay_client_read;
240 r->client->write->handler = relay_client_write;
241 r->upstream->read->handler = relay_upstream_read;
242 r->upstream->write->handler = relay_upstream_write;

245 &r->cu_dec) != NGX_OK)

257/* upstream writable while still connecting: connect completed (or timed out). */
258static void
259relay_connect_done(ngx_event_t *ev)
260{
261 ngx_connection_t *u = ev->data;
262 brix_relay_t *r = u->data;
263 int err;
264 socklen_t len = sizeof(err);
265
266 if (ev->timedout) {
267 ngx_log_error(NGX_LOG_ERR, r->client->log, NGX_ETIMEDOUT,
268 "xrootd relay: connect to %V timed out", r->peer.name);
269 relay_close(r);
270 return;
271 }
272 if (getsockopt(u->fd, SOL_SOCKET, SO_ERROR, (void *) &err, &len) == -1
273 || err != 0)
274 {
275 ngx_log_error(NGX_LOG_ERR, r->client->log, err,
276 "xrootd relay: connect to upstream failed");
277 relay_close(r);
278 return;
279 }
280 if (u->write->timer_set) {
281 ngx_del_timer(u->write);
282 }
283 relay_begin(r);
284}

288brix_relay_start(ngx_stream_session_t *s, ngx_connection_t *c, void *srv_conf)
289{
290 ngx_stream_brix_srv_conf_t *conf = srv_conf;
291 brix_ctx_t *ctx;
292 brix_relay_t *r;
293 ngx_connection_t *u;
294 ngx_int_t rc;
295
296 ctx = ngx_stream_get_module_ctx(s, ngx_stream_brix_module);
297
298 r = ngx_pcalloc(c->pool, sizeof(brix_relay_t));
299 if (r == NULL) {
300 return NGX_ERROR;
301 }
302 r->cu = ngx_pnalloc(c->pool, BRIX_RELAY_BUF);
303 r->uc = ngx_pnalloc(c->pool, BRIX_RELAY_BUF);
304 if (r->cu == NULL || r->uc == NULL) {
305 return NGX_ERROR;
306 }
307
308 r->s = s;
309 r->client = c;
310 r->metrics = ctx->metrics;
311
312 /* No pre-read prefix in transparent mode: the pump reads the client hello. */
313 r->cu_off = r->cu_end = 0;
314 r->uc_off = r->uc_end = 0;
315
316 /* Tap: a stable log copy (server error-log destination, no stale session
317 * appender), then register the audit sink + a decoder per direction. */
318 r->log = *c->log;
319 r->log.handler = NULL;
320 r->log.data = NULL;
321 r->log.action = NULL;
322 brix_tap_register_sink(&r->tap, relay_audit_sink, &r->log);
323 brix_tap_stream_init(&r->cu_dec, &r->tap, BRIX_TAP_C2U);
324 brix_tap_stream_init(&r->uc_dec, &r->tap, BRIX_TAP_U2C);
325
326 /* Bad-actor guard (brix_guard_stream): classify each tapped frame and
327 * let the pump drop the connection on a BOUNCE verdict. */
328 brix_relay_guard_init(&r->guard, conf->relay_guard_enable == 1,
329 &c->addr_text, &r->log);
330 if (r->guard.enable) {
331 brix_tap_register_sink(&r->tap, brix_relay_guard_sink, &r->guard);
332 }
333
334 /* The xrootd recv/send handlers and any deadline timer must not fire on this
335 * connection again — the relay owns it now. */
336 if (c->read->timer_set) {
337 ngx_del_timer(c->read);
338 }
339 if (c->write->timer_set) {
340 ngx_del_timer(c->write);
341 }
342 ctx->state = XRD_ST_PROXY;
343 ctx->relay = r;
344
345 r->peer.sockaddr = conf->relay_addr->sockaddr;
346 r->peer.socklen = conf->relay_addr->socklen;
347 r->peer.name = &conf->relay_name;
348 r->peer.get = relay_peer_get;
349 r->peer.free = relay_peer_free;
350 r->peer.log = c->log;
351 r->peer.log_error = NGX_ERROR_ERR;
352
353 rc = ngx_event_connect_peer(&r->peer);
354 if (rc == NGX_ERROR || rc == NGX_BUSY || rc == NGX_DECLINED) {
355 ngx_log_error(NGX_LOG_ERR, c->log, 0,
356 "xrootd relay: cannot connect to %V", &conf->relay_name);
357 return NGX_ERROR;
358 }
359
360 u = r->peer.connection;
361 r->upstream = u;
362 u->data = r;
363 u->pool = c->pool;
364 u->read->handler = relay_upstream_read;
365 u->write->handler = relay_connect_done;
366
367 if (rc == NGX_OK) {
368 /* Connected synchronously (loopback) — start relaying immediately. */
369 relay_begin(r);
370 } else {
371 /* NGX_AGAIN: connect in progress; arm a connect deadline. */
372 ngx_add_timer(u->write, BRIX_RELAY_IDLE);
373 }
374
375 return NGX_OK;
376}
