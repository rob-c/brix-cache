18 ngx_http_brix_shared_conf_t *conf, const char *root_default)
19{
20 ngx_conf_merge_value(conf->enable, prev->enable, 0);

22 * ngx_conf_merge_str_value computes sizeof(default)-1, which yields the
23 * pointer width (7 on 64-bit) rather than the actual string length — the
24 * empty-string default used by s3/cvmfs became {len:7,data:""}, so the
25 * pure-cache-node root.len==0 → "/" fallback never fired. Hand-roll. */

28 conf->root = prev->root;

350/*
351 * ngx_http_brix_shared_merge() — Merges shared preamble fields from parent to
352 * child using standard nginx merge macros. Called at the top of each protocol's
353 * merge_loc_conf function before protocol-specific merge logic runs.
354 *
355 * WHY: This is the SINGLE audit point for common.* config inheritance — every
356 * HTTP protocol (WebDAV, S3, cvmfs) calls it instead of hand-merging the same
357 * ~20 fields (which drifted per protocol and dropped the read-only enforcement
358 * in cvmfs). Defaults: enable=0, allow_write=0, compress=0, ktls=0,
359 * thread_pool_name="", storage_credential_dir=BRIX_CREDENTIAL_DIR_DEFAULT
360 * (tmpfs store, ensured 0700 by brix_shared_credential_dir_ensure above),
361 * tier grammar defaults as before.
362 *
363 * HOW: root_default parameterizes the one deliberate per-protocol difference
364 * (WebDAV exports default to "/", S3/cvmfs to ""). Ends by applying the hard
365 * read-only switch (see brix_shared_apply_read_only above) and merging pmark;
366 * returns NGX_CONF_OK or NGX_CONF_ERROR (pmark merge failure).
367 */
368static inline char *
369ngx_http_brix_shared_merge(ngx_conf_t *cf,
370 ngx_http_brix_shared_conf_t *prev,
371 ngx_http_brix_shared_conf_t *conf,
372 const char *root_default)
373{
374 char *rc;
375
376 rc = brix_shared_merge_core(cf, prev, conf, root_default);
377 if (rc != NGX_CONF_OK) {
378 return rc;
379 }
380 brix_shared_merge_backend(cf, prev, conf);
381 brix_shared_merge_tier(prev, conf);
382
383 /* Hard read-only: force allow_write off HERE so no protocol merge can
384 * forget the enforcement (it must win before token-scope checks). */
385 brix_shared_apply_read_only(conf, cf->log);
386
387 rc = brix_shared_merge_authx(cf, prev, conf);
388 if (rc != NGX_CONF_OK) {
389 return rc;
390 }
391 brix_shared_merge_net(prev, conf);
392
393 return brix_pmark_conf_merge(cf, &prev->pmark, &conf->pmark);
394}
