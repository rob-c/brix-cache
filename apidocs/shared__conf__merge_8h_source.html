16/* core runtime: enable/export root, write/security flags, logs, thread pool. */
17static inline char *
18brix_shared_merge_core(ngx_conf_t *cf, ngx_http_brix_shared_conf_t *prev,
19 ngx_http_brix_shared_conf_t *conf, const char *root_default)
20{
21 ngx_conf_merge_value(conf->enable, prev->enable, 0);
22 /* root_default is a runtime const char* parameter, not a string literal;
23 * ngx_conf_merge_str_value computes sizeof(default)-1, which yields the
24 * pointer width (7 on 64-bit) rather than the actual string length — the
25 * empty-string default used by s3/cvmfs became {len:7,data:""}, so the
26 * pure-cache-node root.len==0 → "/" fallback never fired. Hand-roll. */
27 if (conf->root.data == NULL) {
28 if (prev->root.data != NULL) {
29 conf->root = prev->root;
30 } else {
31 conf->root.data = (u_char *) root_default;
32 conf->root.len = ngx_strlen(root_default);
33 }
34 }
35 ngx_conf_merge_value(conf->allow_write, prev->allow_write, 0);
36 ngx_conf_merge_value(conf->durable_commit, prev->durable_commit, 1);
37 ngx_conf_merge_value(conf->verify_write, prev->verify_write, 0);
38 ngx_conf_merge_value(conf->require_pgwrite, prev->require_pgwrite, 0);
39 ngx_conf_merge_value(conf->data_substreams, prev->data_substreams, 1);
40 ngx_conf_merge_value(conf->read_only, prev->read_only, 0);
41 ngx_conf_merge_value(conf->read_only_public, prev->read_only_public, 0);
42 ngx_conf_merge_value(conf->compress, prev->compress, 0);
43 ngx_conf_merge_value(conf->strict_security, prev->strict_security, 0);
44 ngx_conf_merge_uint_value(conf->tls_require, prev->tls_require, 0);
45 ngx_conf_merge_str_value(conf->access_log, prev->access_log, "");
46 if (conf->access_log.len > 0
47 && ngx_strcmp(conf->access_log.data, (u_char *) "off") != 0)
48 {
49 conf->access_log_file = ngx_conf_open_file(cf->cycle,
50 &conf->access_log);
51 if (conf->access_log_file == NULL) {
52 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
53 "brix: cannot register HTTP access log \"%V\"",
54 &conf->access_log);
55 return NGX_CONF_ERROR;
56 }
57 } else {
58 conf->access_log_file = NULL;
59 }
60 ngx_conf_merge_value(conf->session_log, prev->session_log, 1);
61 ngx_conf_merge_value(conf->ktls, prev->ktls, 0); /* default OFF (phase-33 P5: opt-in, HW-offload-only) */
62 /* Trusted cache-store surface: default OFF everywhere, so the reserved-name
63 * 404 guard stays in force on every normal client location (default-deny). */
64 ngx_conf_merge_value(conf->cache_store_endpoint,
65 prev->cache_store_endpoint, 0);
66 ngx_conf_merge_value(conf->storage_staging, prev->storage_staging, 0);
67 ngx_conf_merge_str_value(conf->thread_pool_name, prev->thread_pool_name, "");
68 return NGX_CONF_OK;
69}

405 * thread_pool_name="", storage_credential_dir=BRIX_CREDENTIAL_DIR_DEFAULT

407 * tier grammar defaults as before.
408 *
409 * HOW: root_default parameterizes the one deliberate per-protocol difference
410 * (WebDAV exports default to "/", S3/cvmfs to ""). Ends by applying the hard
411 * read-only switch (see brix_shared_apply_read_only above) and merging pmark;
412 * returns NGX_CONF_OK or NGX_CONF_ERROR (pmark merge failure).
413 */

416 ngx_http_brix_shared_conf_t *prev,
417 ngx_http_brix_shared_conf_t *conf,
418 const char *root_default)
419{
420 char *rc;
421
422 rc = brix_shared_merge_core(cf, prev, conf, root_default);
423 if (rc != NGX_CONF_OK) {
424 return rc;
425 }
426 brix_shared_merge_backend(cf, prev, conf);
427 brix_shared_merge_tier(prev, conf);
428
429 /* Hard read-only: force allow_write off HERE so no protocol merge can
430 * forget the enforcement (it must win before token-scope checks). */

432

434 if (rc != NGX_CONF_OK) {
435 return rc;
436 }
437 brix_shared_merge_net(prev, conf);
438
439 return brix_pmark_conf_merge(cf, &prev->pmark, &conf->pmark);
440}
