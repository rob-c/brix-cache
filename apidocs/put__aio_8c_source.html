11/*
12 * s3_thread_pool — resolve (and cache) the async-I/O thread pool for this
13 * location.
14 *
15 * WHY: the S3 postconfiguration only resolves common.thread_pool on the
16 * *server-level* loc-conf, but `brix_s3 on` is normally inside a `location {}`
17 * block whose loc-conf never gets that pointer set — so the offload below would
18 * silently never engage. Mirror the WebDAV COPY/MOVE pattern
19 * (src/webdav/copy.c, move.c): resolve lazily at request time via ngx_cycle and
20 * cache the result into the loc-conf for subsequent requests.
21 */
22ngx_thread_pool_t *
23s3_thread_pool(ngx_http_s3_loc_conf_t *cf)
24{
25 static ngx_str_t default_name = ngx_string("default");
26 ngx_str_t *pname;
27 ngx_thread_pool_t *pool;
28
29 if (cf->common.thread_pool != NULL) {
30 return cf->common.thread_pool;
31 }
32 pname = cf->common.thread_pool_name.len > 0
33 ? &cf->common.thread_pool_name : &default_name;
34 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, pname);
35 if (pool != NULL) {
36 cf->common.thread_pool = pool; /* cache for subsequent requests */
37 }
38 return pool;
39}

42void
43s3_put_aio_thread(void *data, ngx_log_t *log)
44{
45 s3_put_aio_t *t = data;
46
47 (void) log;
48
49 t->io_errno = 0;
50
51 /*
52 * Phase 31 W2 / phase-46 W1a: stream every body buf straight through the
53 * unified write session — no full-body contiguous copy. The writer routes
54 * memory bufs via its write primitive and spooled bufs via write_fd (kernel
55 * copy_file_range from the nginx temp file), and a driver-backed remote-stage
56 * target with no kernel fd transparently through the same call. Both are
57 * blocking syscalls, which is exactly why they run here on the thread pool
58 * rather than the event loop.
59 */
60 if (brix_http_body_write_to_writer(t->r, t->writer) != NGX_OK) {
61 t->io_errno = errno;
62 t->nwritten = -1;
63 return;
64 }
65
66 t->nwritten = (ssize_t) t->len;
67}

70/* s3_put_aio_finish — event-loop completion body (verify/commit/stat/tag). Split
71 * out so s3_put_aio_done can run it inside the impersonation bracket: every op
72 * here is PATH-based and must act as the mapped user. */
73static void
74s3_put_aio_finish(s3_put_aio_t *t)
75{
76 ngx_http_request_t *r = t->r;
77 ngx_log_t *log = r->connection->log;
78
79 if (t->nwritten < 0 || (size_t) t->nwritten < t->len) {
80 brix_log_safe_path(log, NGX_LOG_ERR,
81 (ngx_uint_t) t->io_errno,
82 "s3: async write() failed for: \"%s\"",
83 t->final_path);
84 brix_vfs_writer_abort(t->writer);
85 s3_put_finalize_error(r);
86 return;
87 }
88
89 /* The write session's commit closes/promotes the temp itself — no manual close. */
90 if (s3_commit_put(r, log, t->root_canon, t->writer,
91 t->final_path) != NGX_OK)
92 {
93 if (s3_put_commit_conflict(r)) {
94 return;
95 }
96 brix_log_safe_path(log, NGX_LOG_ERR, ngx_errno,
97 "s3: async staged commit to \"%s\" failed",
98 t->final_path);
99 s3_put_finalize_error(r);
100 return;
101 }
102
103 /* S3 PutObject requires an ETag on the 200 response (the synchronous path
104 * sets it too — keep the offload path's response identical). */
105 {
106 brix_vfs_ctx_t fctx;
107 brix_vfs_stat_t fst;
108 char etag_buf[48];
109
110 brix_vfs_ctx_init(&fctx, r->pool, log, BRIX_PROTO_S3, t->root_canon,
111 NULL, BRIX_VFS_MUTATION_READ_ONLY, 0 /* is_tls */, NULL, t->final_path);
112 {
113 ngx_http_s3_loc_conf_t *acf =
114 ngx_http_get_module_loc_conf(r, ngx_http_brix_s3_module);
115 if (acf != NULL) {
116 /* Decorates a probe/stat-only ctx; Phase-2-forward — namespace
117 * scoping already routes the credential via the VFS ns gate. */
118 brix_vfs_ctx_bind_backend_cred(&fctx,
119 &acf->common.storage_credential_dir,
120 acf->common.storage_credential_fallback);
121 s3_vfs_bind_deleg(r, acf, &fctx);
122 }
123 }
124 if (brix_vfs_probe(&fctx, 1 /* no-follow */, &fst) == NGX_OK) {
125 struct stat final_sb;
126
127 ngx_memzero(&final_sb, sizeof(final_sb));
128 final_sb.st_mtime = fst.mtime;
129 final_sb.st_size = fst.size;
130 s3_etag(&final_sb, etag_buf, sizeof(etag_buf));
131 (void) s3_set_header(r, "ETag", etag_buf);
132 }
133 }
134
135 /* Full-object checksum verify (client-supplied) + echo; failures remove the
136 * object and send the matching 400. */
137 if (s3_put_checksum_failed(r, t->final_path, t->root_canon)) {
138 return;
139 }
140
141 /* x-amz-tagging (best-effort): store the request's tag set on the object. */
142 (void) s3_apply_put_tagging_header(r, t->final_path, t->root_canon);
143 /* x-amz-meta-* (best-effort): store the request's user metadata set. */
144 (void) s3_apply_put_user_metadata(r, t->final_path, t->root_canon);
145
146 brix_dashboard_http_add(r, (ngx_atomic_int_t) t->body_bytes);
147 BRIX_S3_METRIC_ADD(bytes_rx_total, t->body_bytes);
148 if (r->connection && r->connection->sockaddr
149 && r->connection->sockaddr->sa_family == AF_INET6) {
150 BRIX_S3_METRIC_ADD(bytes_rx_ipv6_total, t->body_bytes);
151 } else {
152 BRIX_S3_METRIC_ADD(bytes_rx_ipv4_total, t->body_bytes);
153 }
154 BRIX_S3_METRIC_INC(put_body_total[t->body_mode]);
155 s3_put_finalize_empty_ok(r);
156}

158/*
159 * s3_put_aio_done — thread-pool completion handler (event loop). The staged
160 * commit/verify/stat/tagging in s3_put_aio_finish are PATH-based and must run as
161 * the MAPPED user; this completion lost the per-request impersonation bracket
162 * s3_put_body_handler holds around the synchronous path, so without
163 * re-establishing it the atomic publish renames into the user's directory AS THE
164 * WORKER and fails EPERM. Mirrors the WebDAV async fix (put_body.c). The
165 * threaded write itself (s3_put_aio_thread) was fd-based — no impersonation.
166 */
167void
168s3_put_aio_done(ngx_event_t *ev)
169{
170 ngx_thread_task_t *task = ev->data;
171 s3_put_aio_t *t = task->ctx;
172 ngx_http_s3_req_ctx_t *rx =
173 ngx_http_get_module_ctx(t->r, ngx_http_brix_s3_module);
174
175 /* Balance the r->main->count++ at the task post that kept the request
176 * alive across the thread dispatch (webdav_put_aio_done precedent).
177 * Without it the finalize below leaves the request one reference high:
178 * the 200 is sent but the connection never re-enters keepalive, so the
179 * NEXT request on the same connection is never read - an sd_remote
180 * multipart commit (UploadPart then CompleteMPU back-to-back on one
181 * kept-alive transport connection) wedged for the full curl ceiling. */
182 t->r->main->count--;
183
184 brix_imp_request_begin(rx != NULL ? rx->identity : NULL);
185 s3_put_aio_finish(t);
186 brix_imp_request_end();
187}
