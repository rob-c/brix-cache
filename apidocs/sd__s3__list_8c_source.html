1/*
2 * sd_s3_list.c — S3 ListObjectsV2 (delimited, paged) for the shared S3 driver.
3 *
4 * WHAT: sd_s3_list_page() — one signed GET against the bucket root asking for a
5 * single directory level (delimiter '/'): <Contents> under the prefix are
6 * files, <CommonPrefixes> are sub-directories. Drives the sd_remote
7 * opendir/readdir namespace slots (finding #4).
8 * WHY: S3 has no readdir; a "directory listing" is a prefix+delimiter LIST. The
9 * logic lives here (once, ngx-free) rather than in the remote adapter so a
10 * future server-side lister reuses it.
11 * HOW: Ask sd_s3_list_scan.c for the signed page, then walk the body's
12 * <Contents>/<CommonPrefixes> blocks in document order, reporting each
13 * entry's BASENAME. The flat sibling (sd_s3_list_flat.c) shares every
14 * request-side primitive and differs only in the walk.
15 */

23/* Emit one entry from a raw <Key>/<Prefix> value [vs,ve) to cb, stripping the
24 * request prefix and (for dirs) the trailing '/'. Skips the directory-marker
25 * object and anything that does not sit directly under the prefix. Returns cb's
26 * stop signal (non-zero) or 0. */
27static int
28s3l_emit(const char *vs, const char *ve, const char *prefix, size_t plen,
29 int is_dir, sd_s3_list_cb cb, void *ud)
30{
31 char name[256];
32 char base[256];
33 int n;
34 size_t blen;
35
36 n = sd_s3l_xml_unescape(vs, ve, name, sizeof(name));
37 if (n < 0) {
38 return 0; /* unrepresentable name — skip */
39 }
40 if ((size_t) n < plen || memcmp(name, prefix, plen) != 0) {
41 return 0; /* not under the request prefix */
42 }
43 blen = (size_t) n - plen; /* basename (dirs keep trailing /) */
44 if (is_dir) {
45 if (blen == 0 || name[n - 1] != '/') {
46 return 0;
47 }
48 blen--; /* drop the trailing '/' */
49 }
50 if (blen == 0) {
51 return 0; /* the directory-marker object */
52 }
53 if (blen >= sizeof(base)) {
54 return 0; /* longer than a dirent can hold */
55 }
56 memcpy(base, name + plen, blen);
57 base[blen] = '\0';
58 if (memchr(base, '/', blen) != NULL) {
59 return 0; /* not a direct child (defensive) */
60 }
61 return cb(ud, base, is_dir);
62}

64/* Walk <Contents>/<CommonPrefixes> in document order; the top-level <Prefix>
65 * precedes both, so per-block extraction never mistakes it for an entry. */
66static void
67s3l_walk(const char *body, const char *end, const char *prefix, size_t plen,
68 sd_s3_list_cb cb, void *ud)
69{
70 const char *scan = body;
71 int stopped = 0;
72
73 while (!stopped && scan < end) {
74 const char *c = sd_s3l_find(scan, (size_t) (end - scan), "<Contents>");
75 const char *cp = sd_s3l_find(scan, (size_t) (end - scan),
76 "<CommonPrefixes>");
77 const char *ts;
78 const char *te;
79 const char *block;
80 const char *tag;
81 const char *open_tag;
82 const char *close_tag;
83 int is_dir;
84
85 if (c == NULL && cp == NULL) {
86 break;
87 }
88 is_dir = (c == NULL) || (cp != NULL && cp < c);
89 block = is_dir ? cp : c;
90 tag = is_dir ? "<CommonPrefixes>" : "<Contents>";
91 open_tag = is_dir ? "<Prefix>" : "<Key>";
92 close_tag = is_dir ? "</Prefix>" : "</Key>";
93
94 if (sd_s3l_first_text(block, end, open_tag, close_tag, &ts, &te)) {
95 stopped = s3l_emit(ts, te, prefix, plen, is_dir, cb, ud);
96 scan = te;
97 } else {
98 scan = block + strlen(tag);
99 }
100 }
101}

103int
104sd_s3_list_page(const sd_s3_open_params *p, const char *prefix,
105 const char *cont_in, sd_s3_list_cb cb, void *ud, int *truncated,
106 char *cont_out, size_t cont_cap, char *errbuf, size_t errcap)
107{
108 char qs[S3L_QS_CAP];
109 brix_s3_resp_t resp;
110 sd_s3_file *f = NULL;
111 const char *body;
112 size_t blen = 0;
113 size_t plen;
114 int rc;
115
116 if (p == NULL || prefix == NULL || cb == NULL || truncated == NULL
117 || cont_out == NULL || cont_cap == 0)
118 {
119 errno = EINVAL;
120 sd_s3_set_err(errbuf, errcap, "s3 list: bad parameters");
121 return -1;
122 }
123 *truncated = 0;
124 cont_out[0] = '\0';
125 plen = strlen(prefix);
126
127 if (sd_s3l_build_query(prefix, plen, cont_in, 1 /* delimited */, qs,
128 sizeof(qs), errbuf, errcap) != 0) {
129 return -1;
130 }
131 rc = sd_s3l_fetch(p, qs, &f, &resp, errbuf, errcap);
132 if (rc != 0) {
133 return rc;
134 }
135 if (f == NULL) {
136 /* sd_s3l_fetch's contract is rc == 0 ⇒ *f_out set. Checked rather than
137 * assumed: every failure arm there closes the handle, so a future arm
138 * that returns 0 without publishing one would land as a null deref on
139 * the very next line. */
140 errno = EIO;
141 sd_s3_set_err(errbuf, errcap, "s3 list: fetch returned no handle");
142 return -1;
143 }
144
145 body = f->transport->resp_body(&resp, &blen);
146 if (body == NULL) {
147 blen = 0;
148 body = "";
149 }
150 sd_s3l_page_meta(body, body + blen, truncated, cont_out, cont_cap);
151 s3l_walk(body, body + blen, prefix, plen, cb, ud);
152
153 f->transport->resp_free(&resp);
154 sd_s3_close(f);
155 return 0;
156}
