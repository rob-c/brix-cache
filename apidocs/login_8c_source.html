20/*
21 * WHAT: Validate the 8-byte fixed username field copied off the wire.
22 * WHY: A NUL in the middle silently truncates the field when treated as a C
23 * string, letting "a\x00evil" impersonate user "a"; the XRootD spec says
24 * the field is "null-padded ASCII", so reject NUL-embedded / non-printable
25 * bytes. Isolating the scan keeps the caller's control flow flat.
26 * HOW: Scan the up-to-8 leading non-NUL bytes; on the first out-of-range byte
27 * emit the exact kXR_ArgInvalid refusal and return NGX_DONE so the caller
28 * stops (brix_send_error itself returns NGX_OK, which must NOT continue).
29 * Returns NGX_OK when every byte is printable ASCII.
30 */
31static ngx_int_t
32brix_login_check_username(brix_ctx_t *ctx, ngx_connection_t *c,
33 const char user[9])
34{
35 int i;
36
37 for (i = 0; i < 8 && user[i] != '\0'; i++) {
38 if ((u_char) user[i] < 0x20 || (u_char) user[i] > 0x7e) {
39 brix_send_error(ctx, c, kXR_ArgInvalid,
40 "username contains invalid characters");
41 return NGX_DONE;
42 }
43 }
44
45 return NGX_OK;
46}

48/*
49 * WHAT: Run the pre-response login preconditions and parse the request into
50 * ctx->login (username, pid) plus a sanitized copy for logging.
51 * WHY: These checks (suspended, duplicate login, username validity) all short-
52 * circuit with an exact wire refusal before any response is built; keeping
53 * them together preserves the reference ordering and messages verbatim.
54 * HOW: On any refusal, emit the exact error and return NGX_DONE (brix_send_error
55 * returns NGX_OK, which must not fall through). On success populate
56 * ctx->login, mark the session logged_in, emit the debug trace, and return
57 * NGX_OK. `user`/`user_log` are caller-owned so the response stage can log.
58 */
59static ngx_int_t
60brix_login_precheck_and_parse(brix_ctx_t *ctx, ngx_connection_t *c,
61 ngx_stream_brix_srv_conf_t *conf, char user[9], char user_log[64])
62{
63 xrdw_login_req_t req;
64 ngx_int_t rc;
65
66 if (conf->cms.suspended) {
67 brix_send_error(ctx, c, kXR_Overloaded, "server suspended by manager");
68 return NGX_DONE;
69 }
70
71 /* The reference do_Login rejects a SECOND kXR_login on an already-logged-in
72 * connection ("if (Status) return Response.Send(kXR_InvalidRequest,
73 * \"duplicate login; already logged in\")", XrdXrootdXeq.cc:1095). Match the
74 * reference code and message verbatim. (Note: installed stock v5.9.5 happens
75 * to surface kXR_ArgMissing here, but the current reference source — and the
76 * correct semantics for a malformed-in-context request — is kXR_InvalidRequest;
77 * either way a live session cannot re-login.) */
78 if (ctx->login.logged_in) {
79 brix_send_error(ctx, c, kXR_InvalidRequest,
80 "duplicate login; already logged in");
81 return NGX_DONE;
82 }
83
84 xrdw_login_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
85
86 /* Username is an 8-byte fixed field on the wire, so copy and terminate it locally. */
87 ngx_memcpy(user, req.username, 8);
88 user[8] = '\0';
89
90 rc = brix_login_check_username(ctx, c, user);
91 if (rc != NGX_OK) {
92 return rc;
93 }
94
95 ngx_memcpy(ctx->login.user, user, sizeof(ctx->login.user));
96 ctx->login.pid = (uint32_t) req.pid;
97 /* §1.3: keep the advertised ability bitmasks — kXR_fullurl (bit 0)
98 * upgrades redirect targets to full root:// URLs (response/control.c);
99 * unknown bits are stored untouched and change nothing. */
100 ctx->login.ability = req.ability;
101 ctx->login.ability2 = req.ability2;
102 brix_sanitize_log_string(user, user_log, 64);
103
104 ngx_log_debug3(NGX_LOG_DEBUG_STREAM, c->log, 0,
105 "brix: login user=\"%s\" pid=%d auth=%s",
106 user_log, (int) req.pid,
107 (conf->auth == BRIX_AUTH_GSI) ? "gsi" :
108 (conf->auth == BRIX_AUTH_TOKEN) ? "token" :
109 (conf->auth == BRIX_AUTH_BOTH) ? "both" :
110 (conf->auth == BRIX_AUTH_SSS) ? "sss" :
111 (conf->auth == BRIX_AUTH_UNIX) ? "unix" :
112 (conf->auth == BRIX_AUTH_KRB5) ? "krb5" : "none");
113
114 /* Login marks the session as known; auth_done is deferred for GSI mode. */
115 ctx->login.logged_in = 1;
116
117 return NGX_OK;
118}

120/*
121 * WHAT: Emit the anonymous-mode login response (sessid only, single round-trip).
122 * WHY: BRIX_AUTH_NONE completes login immediately with no auth negotiation, so
123 * its response, session registration, and access log differ from the
124 * authenticated path and are kept separate for clarity.
125 * HOW: Mark auth_done, build the kXR_ok header + sessid body, record session
126 * start/identity, register the session, log access, count the metric, and
127 * queue the response (returning whatever brix_queue_response returns).
128 */
129static ngx_int_t
130brix_login_respond_anon(brix_ctx_t *ctx, ngx_connection_t *c, const char *user)
131{
132 u_char *buf;
133 size_t total;
134
135 /* Anonymous mode completes login in one round-trip with only the sessid. */
136 ctx->login.auth_done = 1;
137
138 total = XRD_RESPONSE_HDR_LEN + BRIX_SESSION_ID_LEN;
139 BRIX_PALLOC_OR_RETURN(buf, c->pool, total, NGX_ERROR);
140
141 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok,
142 BRIX_SESSION_ID_LEN,
143 (ServerResponseHdr *) buf);
144 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, ctx->login.sessid,
145 BRIX_SESSION_ID_LEN);
146
147 /* Session timing starts at successful login so later disconnect stats have an origin. */
148 ctx->totals.start = ngx_current_msec;
149 if (ctx->identity != NULL) {
150 ctx->identity->auth_method = BRIX_AUTHN_NONE;
151 }
152 brix_sess_auth_once(ctx->sess, BRIX_SESS_AM_ANON, "-", "-");
153 ctx->login.session_slot_hint =
154 brix_session_register(ctx->login.sessid, ctx->login.dn,
155 ctx->login.vo_list, 0);
156 brix_log_access(ctx, c, "LOGIN", "-", user, 1, 0, NULL, 0);
157 brix_count_login_ok(ctx);
158
159 return brix_queue_response(ctx, c, buf, total);
160}

162/* XrdSecProtocolztn client parameters, byte-frozen against the stock client.
163 *
164 * The reference server emits "<expiry>:<maxtsz>:" — a decimal minimum token
165 * lifetime the client must satisfy (0 = no requirement) and a decimal maximum
166 * accepted token size, each closed by a colon. XrdSecProtocolztn's constructor
167 * parses them with strtoll()/strtol() and demands the colon after each field
168 * plus maxtsz > 0; anything else aborts the login with "Secztn: Malformed
169 * client parameters", which is exactly what the earlier "v:10000" form did to
170 * every stock XrdCl. 4096 matches the reference server's default -maxsz. */
171#define BRIX_ZTN_PARMS "0:4096:"
172
173/*
174 * WHAT: Format ONE "&P=<proto>[,<parms>]" block for `proto` into `out`,
175 * returning the byte count the block needs (snprintf semantics: the
176 * would-be length even when it does not fit), or 0 when the server
177 * cannot advertise that protocol.
178 * WHY: The sec token is an ordered concatenation of per-protocol blocks, so
179 * each protocol's byte-frozen parameter string has to be addressable on
180 * its own rather than as one branch of a whole-token ladder. Returning 0
181 * for an unadvertisable protocol (krb5 with no service principal) is what
182 * keeps a partial configuration from emitting "&P=krb5,(null)".
183 * HOW: One snprintf per protocol; negative snprintf results are normalised to
184 * 0 so the caller's length arithmetic can never go backwards.
185 */
186static size_t
187brix_login_proto_parm(ngx_stream_brix_srv_conf_t *conf, ngx_uint_t proto,
188 char *out, size_t cap)
189{
190 /* Advertised GSI version drives the client's signed-DH decision:
191 * >=BRIX_GSI_VERS_DHSIGNED (10400) lets capable clients use the
192 * RSA-signed-DH variant. Default 10000 (unsigned, universally
193 * compatible) unless the brix_gsi_signed_dh policy opts in. */
194 unsigned gsi_ver = (conf->gsi_signed_dh != BRIX_GSI_SDH_OFF)
195 ? 10600u : 10000u;
196 int written;
197
198 switch (proto) {
199
200 case BRIX_AUTH_TOKEN:
201 /* Token: advertise ztn, no CA hash needed. */
202 written = snprintf(out, cap, "&P=ztn," BRIX_ZTN_PARMS);
203 break;
204
205 case BRIX_AUTH_GSI:
206 written = snprintf(out, cap, "&P=gsi,v:%u,c:ssl,ca:%s",
207 gsi_ver, conf->gsi_ca_hashes);
208 break;
209
210 case BRIX_AUTH_SSS:
211 /* XRootD SSS: bf32 ('0'), v2 server ('+'), client chooses keytab. */
212 written = snprintf(out, cap, "&P=sss,0.+%d:",
213 (int) conf->sss_lifetime);
214 break;
215
216 case BRIX_AUTH_UNIX:
217 written = snprintf(out, cap, "&P=unix");
218 break;
219
220 case BRIX_AUTH_KRB5:
221 /* A krb5 block without the service principal is unusable to the
222 * client, and conf->krb5.principal.data is NULL until configured. */
223 if (conf->krb5.principal.len == 0
224 || conf->krb5.principal.data == NULL)
225 {
226 return 0;
227 }
228 written = snprintf(out, cap, "&P=krb5,%s",
229 (const char *) conf->krb5.principal.data);
230 break;
231
232 case BRIX_AUTH_HOST:
233 /* Phase 52 WS-C: host auth asserts no credential — bare protocol id. */
234 written = snprintf(out, cap, "&P=host");
235 break;
236
237 case BRIX_AUTH_PWD:
238 /* Phase 52 WS-B: XrdSecpwd password protocol (v:10100, ssl crypto). */
239 written = snprintf(out, cap, "&P=pwd,v:10100,c:ssl");
240 break;
241
242 default:
243 return 0;
244 }
245
246 return (written < 0) ? 0 : (size_t) written;
247}

249/*
250 * WHAT: Format the full "&P=..." security parameter block for a resolved
251 * protocol set, returning the byte count INCLUDING the trailing NUL
252 * (clients treat the block as C-string data). A return greater than
253 * `cap` means the token did not fit and the caller must refuse.
254 * WHY: Stock XRootD advertises an arbitrary ORDERED protocol list — that
255 * order is the client's preference order — so the token is built by
256 * concatenating one block per protocol in the order protbind resolved,
257 * not by picking a single pre-baked string per auth mode.
258 * HOW: Append each protocol's block, tracking the total the token WOULD need
259 * separately from what actually fit, so a truncating token is detected
260 * instead of silently advertising a half-written protocol list.
261 */
262static size_t
263brix_login_build_parms(ngx_stream_brix_srv_conf_t *conf,
264 const brix_protbind_set_t *set, char *parms, size_t cap, int drop_ztn)
265{
266 size_t needed = 0;
267 ngx_uint_t index;
268
269 parms[0] = '\0';
270
271 for (index = 0; index < set->count; index++) {
272 size_t room = (needed < cap) ? cap - needed : 0;
273
274 /* ztn is a bearer token; on a cleartext connection (drop_ztn) it is
275 * withheld from the advertisement so a stock-compliant client never
276 * offers one where an on-path observer could replay it. */
277 if (drop_ztn && set->protos[index] == BRIX_AUTH_TOKEN) {
278 continue;
279 }
280
281 needed += brix_login_proto_parm(conf, set->protos[index],
282 parms + (cap - room), room);
283 }
284
285 return needed + 1;
286}

303/*
304 * WHAT: Emit the authenticated-mode login response (sessid + "&P=..." block).
305 * WHY: Authenticated modes send a text parameter block after the 16-byte
306 * session id so the client can select its security plugin; login succeeds
307 * here but auth negotiation continues in later round-trips.
308 * HOW: Re-fetch the live merged srv_conf, format the parameter block, guard its
309 * length (kXR_ServerError → NGX_DONE, since brix_send_error returns NGX_OK
310 * and must not fall through), build the kXR_ok header + sessid + parms,
311 * record session start, log access, count the metric, and queue.
312 */
313static ngx_int_t
314brix_login_respond_authenticated(brix_ctx_t *ctx, ngx_connection_t *c,
315 const char *user, const brix_protbind_set_t *set)
316{
317 ngx_stream_brix_srv_conf_t *conf;
318 /* One block per advertised protocol; the GSI block alone can carry an
319 * 80-byte CA-hash list, so size for the full BRIX_PROTBIND_MAX_PROTOS. */
320 char parms[1024];
321 size_t parms_len;
322 u_char *buf;
323 size_t total;
324 int drop_ztn;
325
326 /* Re-fetch the live merged srv_conf in case login inherited settings. */
327 conf = ngx_stream_get_module_srv_conf(ctx->session, ngx_stream_brix_module);
328
329 /* Stock parity: never advertise the ztn bearer protocol on a cleartext
330 * connection unless brix_ztn_cleartext explicitly opts back in. */
331 drop_ztn = (c->ssl == NULL || c->ssl->connection == NULL)
332 && !conf->ztn_cleartext;
333
334 parms_len = brix_login_build_parms(conf, set, parms, sizeof(parms),
335 drop_ztn);
336
337 /* Include the trailing NUL because clients treat the parameter block as C-string data. */
338 if (parms_len > sizeof(parms)) {
339 brix_send_error(ctx, c, kXR_ServerError,
340 "auth parameter block too long");
341 return NGX_DONE;
342 }
343
344 /* Token-only auth over cleartext with the ztn block withheld leaves the
345 * client nothing to authenticate with — refuse the login up front with
346 * the TLS-required error (the client's cue to reconnect with TLS),
347 * exactly as stock does, rather than a misleading config error. */
348 if (parms_len <= 1 && drop_ztn && brix_login_set_has_token(set)) {
349 brix_send_error(ctx, c, kXR_TLSRequired,
350 "token authentication requires a TLS-encrypted "
351 "connection");
352 return NGX_DONE;
353 }
354
355 /* An empty token would tell the client "authenticate, but with nothing" —
356 * the only way here is a protbind rule naming protocols this build cannot
357 * advertise, which is a configuration error, not a client error. */
358 if (parms_len <= 1) {
359 brix_send_error(ctx, c, kXR_ServerError,
360 "no usable authentication protocol configured");
361 return NGX_DONE;
362 }
363
364 total = XRD_RESPONSE_HDR_LEN + BRIX_SESSION_ID_LEN + parms_len;
365 BRIX_PALLOC_OR_RETURN(buf, c->pool, total, NGX_ERROR);
366
367 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok,
368 (uint32_t)(BRIX_SESSION_ID_LEN + parms_len),
369 (ServerResponseHdr *) buf);
370
371 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, ctx->login.sessid,
372 BRIX_SESSION_ID_LEN);
373 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN + BRIX_SESSION_ID_LEN,
374 parms, parms_len);
375
376 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
377 "brix: login->kXGS_init parms=\"%s\" ca_hashes=%s",
378 parms, conf->gsi_ca_hashes);
379
380 /* Successful login still marks the start of the session even though auth continues. */
381 ctx->totals.start = ngx_current_msec;
382 brix_log_access(ctx, c, "LOGIN", "-", user, 1, 0, NULL, 0);
383 brix_count_login_ok(ctx);
384
385 return brix_queue_response(ctx, c, buf, total);
386}

388/* Handle kXR_login — accept a client username, generate a session id (sessid),
389 * and begin auth negotiation (advertising the security protocols this peer is
390 * bound to). The advertised set comes from brix_protbind (per-host rules)
391 * refined over brix_auth; with no rules it is exactly what brix_auth alone
392 * allowed, so an unconfigured server's sec token is byte-identical. */
393ngx_int_t
394brix_handle_login(brix_ctx_t *ctx, ngx_connection_t *c,
395 ngx_stream_brix_srv_conf_t *conf)
396{
397 brix_protbind_set_t bound;
398 char user[9];
399 char user_log[64];
400 ngx_int_t rc;
401
402 rc = brix_login_precheck_and_parse(ctx, c, conf, user, user_log);
403 if (rc != NGX_OK) {
404 return (rc == NGX_DONE) ? NGX_OK : rc;
405 }
406
407 brix_protbind_resolve_ctx(ctx, c, conf->protbind, conf->auth, &bound);
408
409 if (!bound.require_auth) {
410 return brix_login_respond_anon(ctx, c, user);
411 }
412
413 rc = brix_login_respond_authenticated(ctx, c, user, &bound);
414 return (rc == NGX_DONE) ? NGX_OK : rc;
415}
