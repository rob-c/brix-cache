1/*
2 * gftp_feat.h — RFC 2389 FEAT capability probe for the outbound GridFTP driver.
3 *
4 * WHAT: one lazily-issued FEAT command per session, parsed into a bit mask of
5 * the extensions this driver is prepared to USE.
6 *
7 * WHY: the GridFTP extensions are opt-in on the wire and there is no safe way to
8 * discover them by trying. A command an origin has never heard of comes back
9 * 500 or 502 — which is indistinguishable from "the transfer failed" on some
10 * doors, and on others is answered with a plain-RETR-shaped reply that would
11 * hand the caller the WRONG BYTES (see the fallback note on gftp_feat_eret_get
12 * in gftp_data.c). Asking first is the only way to know.
13 *
14 * HOW: probed at most once per session, and only from a path that would
15 * actually use a capability — a stat, a listing or a whole-file read pays
16 * nothing for it, which matters because a session lives for exactly one
17 * storage-driver call and an unconditional FEAT would put a round trip on every
18 * metadata operation the driver performs.
19 */
20#ifndef BRIX_GFTP_FEAT_H
21#define BRIX_GFTP_FEAT_H
22
23#include "gftp_client.h"
24
25/* Extensions this driver uses. A bit is set only when the origin ADVERTISED
26 * it: an origin that supports one silently gets the RFC 959 path, which is
27 * correct, merely less efficient. Bits are added here as the callers that need
28 * them land. */
29#define GFTP_FEAT_ERET 0x1u /* GFD.020 5.3 bounded retrieve (W5.2) */
30#define GFTP_FEAT_SPAS 0x2u /* GFD.020 5.1 striped passive (W5.3) */
31
32/* The origin's advertised feature mask, probing on first use. Never fails:
33 * an origin that refuses or mangles FEAT reports NO extensions, which is the
34 * RFC 959 floor and exactly what this driver did before W5.2. */
35unsigned gftp_feat(gftp_session_t *session);

37#endif /* BRIX_GFTP_FEAT_H */
