1/*
2 * access_vfs_ctx.c - confined-VFS-ctx construction + delegated-credential
3 * binding for WebDAV storage ops.
4 *
5 * WHAT: The webdav_vfs_ctx_build constructor family (bare / _ns / _data) and
6 * webdav_vfs_bind_deleg, the binder that hands the request's captured
7 * forwardable credential (bearer JWT and/or user-supplied x509 proxy PEM) to
8 * the VFS under the export's resolved delegation mode.
9 *
10 * WHY: Every WebDAV method file used to hand-roll the same init +
11 * credential-binding sequence; one constructor family keeps the binding order
12 * (cred -> mint -> deleg) in a single place next to the deleg binder it
13 * composes. Split out of access.c (which keeps the access-phase gate itself)
14 * along that seam when access.c crossed the 600-line contract.
15 *
16 * HOW: Declarations live in webdav_auth.h beside the auth gate that captures
17 * the bytes these binders forward. See the per-function comments.
18 */

30/* ---- webdav_vfs_bind_deleg -------------------------------------------------
31 *
32 * WHAT: Bind the request's captured forwardable credential (bearer JWT and/or
33 * user-supplied full x509 proxy PEM) onto a cred-bound VFS ctx, using the
34 * export's resolved delegation mode. See webdav_auth.h for the contract.
35 *
36 * WHY: Called at every WebDAV brix_vfs_ctx_bind_backend_cred site so a delegated
37 * export authenticates the backend leg AS the inbound user rather than the
38 * shared service credential. The bytes were captured once at the auth gate
39 * (bearer in webdav_verify_bearer_token, proxy in the access-phase header
40 * capture) and stashed on the req ctx; here they are handed to the VFS.
41 *
42 * HOW: Reads conf->common.backend_delegation as the mode and the req ctx's
43 * bearer_token / deleg_proxy_pem as the bytes; brix_vfs_deleg_bind is a
44 * no-op for SELECT mode or when nothing was captured. A captured bearer
45 * that would be forwarded VERBATIM (any mode except EXCHANGE-with-endpoint,
46 * which re-audiences it) must first pass the backend audience gate
47 * (`brix_backend_token_audience_ok`, phase-70 §5.2 / P90-70.9) — on
48 * refusal the bearer is simply not bound (SELECT/service-cred policy then
49 * applies). EXCHANGE conf + the per-conf minted-token cache slot are
50 * stamped after the bind. */
51void
52webdav_vfs_bind_deleg(ngx_http_request_t *r,
53 ngx_http_brix_webdav_loc_conf_t *conf, brix_vfs_ctx_t *vctx)
54{
55 ngx_http_brix_webdav_req_ctx_t *rctx;
56 const ngx_str_t *bearer;
57
58 brix_http_vfs_bind_authz(r, &conf->common, conf->common.authdb_rules,
59 conf->common.vo_rules, vctx);
60 if (conf->common.backend_delegation == BRIX_CRED_SELECT) {
61 return;
62 }
63
64 rctx = ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
65 bearer = brix_proto_deleg_gate_bearer(
66 (rctx != NULL) ? &rctx->bearer_token : NULL,
67 &conf->common, r->connection->log);
68
69 (void) brix_vfs_deleg_bind(r->pool, vctx,
70 (enum brix_cred_mode) conf->common.backend_delegation,
71 bearer,
72 (rctx != NULL) ? &rctx->deleg_proxy_pem : NULL);
73
74 brix_proto_deleg_stamp_conf(vctx, &conf->common);
75
76 /* P90-70.4: stamp the export's trust store so the VFS deleg gate re-runs
77 * the RFC-3820 chain-trust check before materialising the proxy (no-op
78 * when nothing was bound or no CA store is configured). */
79 brix_vfs_deleg_set_ca_store(vctx, conf->ca_store, conf->common.verify_depth);
80}

83/*
84 * WHAT: Canonical confined-VFS-ctx constructors for WebDAV storage ops.
85 *
86 * WHY: Every method file used to hand-roll the same init + credential-binding
87 * sequence; one constructor family keeps the binding order (cred → mint →
88 * deleg) in a single place next to the deleg binder it composes.
89 *
90 * HOW: webdav_vfs_ctx_build initialises the bare confined ctx (identity from
91 * the req ctx, TLS from the connection). The _ns/_data variants add the
92 * export's per-user backend credential policy on top; only the data plane
93 * binds the opt-in mint CA — a namespace op that needs a credential the user
94 * doesn't have falls back per storage_credential_fallback (see prop_xattr.c).
95 * Callers that route through the export's storage driver assign vctx->sd
96 * themselves (brix_webdav_backend_instance).
97 */
98void
99webdav_vfs_ctx_build(ngx_http_request_t *r, const char *path,
100 brix_vfs_ctx_t *vctx)
101{
102 ngx_http_brix_webdav_loc_conf_t *conf =
103 ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
104 ngx_http_brix_webdav_req_ctx_t *wctx =
105 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
106
107 brix_vfs_ctx_init(vctx, r->pool, r->connection->log,
108 BRIX_PROTO_WEBDAV, conf->common.root_canon,
109 conf->common.cache_root_canon,
110 brix_vfs_policy_from_write_enable(conf->common.allow_write),
111 brix_http_request_is_tls(r),
112 (wctx != NULL) ? wctx->identity : NULL, path);
113
114 /* phase-107 C7: hand the VFS lock gate the same raw If/Lock-Token bytes
115 * the WebDAV edge matches against, so ownership answers agree. WebDAV is
116 * the only protocol with a way to present a lock token — every other
117 * plane's mutations stay "foreign" to a held lock. */
118 vctx->lock_token = webdav_lock_token_header(r);
119 brix_http_vfs_bind_authz(r, &conf->common, conf->common.authdb_rules,
120 conf->common.vo_rules, vctx);
121}

138 conf->common.storage_credential_mint_ttl);
139 }
140 webdav_vfs_bind_deleg(r, conf, vctx);
141}

165/* ---- Authorize one exact WebDAV resource operand -------------------------
166 *
167 * WHAT: Re-evaluates one logical/resolved resource pair through every active
168 * bearer, native authdb/VO, and XrdAcc policy. Returns NGX_OK on allow or
169 * NGX_HTTP_FORBIDDEN on denial, missing authority, or malformed input.
170 *
171 * WHY: COPY and MOVE are multi-resource operations. Their Destination header
172 * is not r->uri, so the ordinary access-phase decision cannot authorize it;
173 * the configurable VFS backstop is deliberately insufficient as an edge
174 * decision because OBSERVE and OFF modes do not refuse.
175 *
176 * HOW: Build the canonical request-identity VFS context on the confined path,
177 * force ENFORCE only on this request-local copy, and delegate the exact
178 * logical/resolved decision to the shared VFS authorization evaluator.
179 */
180ngx_int_t
181webdav_authorize_resource(ngx_http_request_t *r,
182 ngx_http_brix_webdav_loc_conf_t *conf,
183 const brix_vfs_authz_exact_t *exact)
184{
185 brix_vfs_ctx_t vctx;
186
187 if (r == NULL || conf == NULL || exact == NULL
188 || exact->resolved_path == NULL)
189 {
190 return NGX_HTTP_FORBIDDEN;
191 }
192
193 webdav_vfs_ctx_build(r, exact->resolved_path, &vctx);
194 vctx.authz.mode = BRIX_AUTHZ_BACKSTOP_ENFORCE;
195 if (brix_vfs_require_authorized_exact(&vctx, exact) != NGX_OK) {
196 return NGX_HTTP_FORBIDDEN;
197 }
198 return NGX_OK;
199}
