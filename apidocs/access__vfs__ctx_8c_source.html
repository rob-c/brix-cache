2 * access_vfs_ctx.c - confined-VFS-ctx construction + delegated-credential
3 * binding for WebDAV storage ops.
4 *
5 * WHAT: The webdav_vfs_ctx_build constructor family (bare / _ns / _data) and
6 * webdav_vfs_bind_deleg, the binder that hands the request's captured
7 * forwardable credential (bearer JWT and/or user-supplied x509 proxy PEM) to
8 * the VFS under the export's resolved delegation mode.
9 *
10 * WHY: Every WebDAV method file used to hand-roll the same init +

12 * (cred -> mint -> deleg) in a single place next to the deleg binder it

14 * along that seam when access.c crossed the 600-line contract.
15 *
16 * HOW: Declarations live in webdav_auth.h beside the auth gate that captures

29/* ---- webdav_vfs_bind_deleg -------------------------------------------------
30 *
31 * WHAT: Bind the request's captured forwardable credential (bearer JWT and/or
32 * user-supplied full x509 proxy PEM) onto a cred-bound VFS ctx, using the
33 * export's resolved delegation mode. See webdav_auth.h for the contract.
34 *
35 * WHY: Called at every WebDAV brix_vfs_ctx_bind_backend_cred site so a delegated
36 * export authenticates the backend leg AS the inbound user rather than the
37 * shared service credential. The bytes were captured once at the auth gate
38 * (bearer in webdav_verify_bearer_token, proxy in the access-phase header
39 * capture) and stashed on the req ctx; here they are handed to the VFS.
40 *
41 * HOW: Reads conf->common.backend_delegation as the mode and the req ctx's
42 * bearer_token / deleg_proxy_pem as the bytes; brix_vfs_deleg_bind is a
43 * no-op for SELECT mode or when nothing was captured. A captured bearer
44 * that would be forwarded VERBATIM (any mode except EXCHANGE-with-endpoint,
45 * which re-audiences it) must first pass the backend audience gate
46 * (`brix_backend_token_audience_ok`, phase-70 §5.2 / P90-70.9) — on
47 * refusal the bearer is simply not bound (SELECT/service-cred policy then
48 * applies). EXCHANGE conf + the per-conf minted-token cache slot are
49 * stamped after the bind. */
50void
51webdav_vfs_bind_deleg(ngx_http_request_t *r,
52 ngx_http_brix_webdav_loc_conf_t *conf, brix_vfs_ctx_t *vctx)
53{
54 ngx_http_brix_webdav_req_ctx_t *rctx;
55 const ngx_str_t *bearer;
56
57 brix_http_vfs_bind_authz(r, &conf->common, conf->common.authdb_rules,
58 conf->common.vo_rules, vctx);
59 if (conf->common.backend_delegation == BRIX_CRED_SELECT) {
60 return;
61 }
62
63 rctx = ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
64 bearer = brix_proto_deleg_gate_bearer(
65 (rctx != NULL) ? &rctx->bearer_token : NULL,
66 &conf->common, r->connection->log);
67
68 (void) brix_vfs_deleg_bind(r->pool, vctx,
69 (enum brix_cred_mode) conf->common.backend_delegation,
70 bearer,
71 (rctx != NULL) ? &rctx->deleg_proxy_pem : NULL);
72
73 brix_proto_deleg_stamp_conf(vctx, &conf->common);
74
75 /* P90-70.4: stamp the export's trust store so the VFS deleg gate re-runs
76 * the RFC-3820 chain-trust check before materialising the proxy (no-op
77 * when nothing was bound or no CA store is configured). */
78 brix_vfs_deleg_set_ca_store(vctx, conf->ca_store, conf->common.verify_depth);
79}

82/*
83 * WHAT: Canonical confined-VFS-ctx constructors for WebDAV storage ops.
84 *
85 * WHY: Every method file used to hand-roll the same init + credential-binding
86 * sequence; one constructor family keeps the binding order (cred → mint →
87 * deleg) in a single place next to the deleg binder it composes.
88 *
89 * HOW: webdav_vfs_ctx_build initialises the bare confined ctx (identity from
90 * the req ctx, TLS from the connection). The _ns/_data variants add the
91 * export's per-user backend credential policy on top; only the data plane
92 * binds the opt-in mint CA — a namespace op that needs a credential the user
93 * doesn't have falls back per storage_credential_fallback (see prop_xattr.c).
94 * Callers that route through the export's storage driver assign vctx->sd
95 * themselves (brix_webdav_backend_instance).
96 */
97void
98webdav_vfs_ctx_build(ngx_http_request_t *r, const char *path,
99 brix_vfs_ctx_t *vctx)
100{
101 ngx_http_brix_webdav_loc_conf_t *conf =
102 ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
103 ngx_http_brix_webdav_req_ctx_t *wctx =
104 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
105
106 brix_vfs_ctx_init(vctx, r->pool, r->connection->log,
107 BRIX_PROTO_WEBDAV, conf->common.root_canon,
108 conf->common.cache_root_canon,
109 brix_vfs_policy_from_write_enable(conf->common.allow_write),
110 brix_http_request_is_tls(r),
111 (wctx != NULL) ? wctx->identity : NULL, path);
112
113 /* phase-107 C7: hand the VFS lock gate the same raw If/Lock-Token bytes
114 * the WebDAV edge matches against, so ownership answers agree. WebDAV is
115 * the only protocol with a way to present a lock token — every other
116 * plane's mutations stay "foreign" to a held lock. */
117 vctx->lock_token = webdav_lock_token_header(r);
118 brix_http_vfs_bind_authz(r, &conf->common, conf->common.authdb_rules,
119 conf->common.vo_rules, vctx);
120}

137 conf->common.storage_credential_mint_ttl);
138 }
139 webdav_vfs_bind_deleg(r, conf, vctx);
140}
