1/*
2 * xrootdfs_identity.c — per-caller SSS identity for a multi-user mount.
3 *
4 * WHAT: xfs_ident_pool() and xfs_ident_mgr() hand back the metadata pool and
5 * the data manager belonging to the uid of the FUSE request currently
6 * being served, building them on first use. With --sss-identity off
7 * they return the mount-wide g_pool / g_mgr and nothing changes.
8 * WHY: one FUSE process serves every user on the box. The sss authenticator
9 * named the process owner (getpwuid(geteuid()) in sec_sss.c), so under
10 * `allow_other` every caller reached the server wearing the MOUNT
11 * OWNER's identity — user B reading user A's files with the server's
12 * blessing. A connection cannot change identity after kXR_login, so the
13 * identity must partition the connections rather than ride on them.
14 * HOW: a mutex-guarded fixed table of (uid -> name, opts, pool, mgr). Slots
15 * are never reused, so a pointer handed to a caller stays valid until
16 * unmount and no operation can have its connections freed underneath it;
17 * the price is a hard cap, and a mount that exceeds it REFUSES the extra
18 * identities (see xfs_ident_pool) rather than quietly serving them as the
19 * mount owner. Creation happens outside the lock behind a per-slot
20 * `creating` flag so one user's slow connect cannot stall every other
21 * user's metadata op. Same shape and lifetime as the g_pool/g_mgr pair
22 * it shards, and as the registry in client/lib/net/forksafe.c.
23 *
24 * The identity is only ever a PROPOSAL. The server's keytab decides whether
25 * the sss NAME TLV is honoured at all: a key with a fixed `user=` ignores it
26 * outright and only an `anybody`/`allusers` key accepts it
27 * (src/auth/sss/auth_request.c, sss_map_identity). A mount therefore cannot
28 * name its way into an identity the operator did not issue.
29 */

34/* Identities per mount. Slots are never recycled (see HOW), so this is also
35 * the number of distinct users one mount can serve. --max-identities raises it
36 * up to XFS_IDENT_HARD_MAX. */
37#define XFS_IDENT_DEFAULT_MAX 64
38#define XFS_IDENT_HARD_MAX 1024
39#define XFS_IDENT_NAME_MAX 64

63/*
64 * ident_login_name — resolve uid to its login name.
65 *
66 * A uid with no passwd entry gets its decimal form: the server still receives a
67 * DISTINCT, stable name per caller, which is what keeps two unmapped users from
68 * sharing one connection. It is not an escalation — an unknown name authorises
69 * as whatever the operator's keytab says, and typically as nobody.
70 */
71static void
72ident_login_name(uid_t uid, char *out, size_t n)
73{
74 struct passwd pw;
75 struct passwd *res = NULL;
76 char buf[1024];
77
78 if (getpwuid_r(uid, &pw, buf, sizeof(buf), &res) == 0
79 && res != NULL && res->pw_name != NULL && res->pw_name[0] != '\0')
80 {
81 snprintf(out, n, "%s", res->pw_name);
82 return;
83 }
84 snprintf(out, n, "%lu", (unsigned long) uid);
85}

88/* Find the slot for uid, or NULL. Caller holds the lock. */
89static xfs_ident *
90ident_find_locked(uid_t uid)
91{
92 int i;
93
94 for (i = 0; i < g_ident_n; i++) {
95 if (g_ident_tab[i].used && g_ident_tab[i].uid == uid) {
96 return &g_ident_tab[i];
97 }
98 }
99 return NULL;
100}

103/* Claim a fresh slot for uid, or NULL when the table is full. Holds lock. */
104static xfs_ident *
105ident_claim_locked(uid_t uid)
106{
107 xfs_ident *e;
108
109 if (g_ident_n >= g_ident_max) {
110 return NULL;
111 }
112 e = &g_ident_tab[g_ident_n++];
113 memset(e, 0, sizeof(*e));
114 e->used = 1;
115 e->creating = 1;
116 e->uid = uid;
117 return e;
118}

121/*
122 * ident_build — connect this identity's pool and manager. Runs OUTSIDE the
123 * table lock; the slot's `creating` flag keeps every other thread off it.
124 */
125static void
126ident_build(xfs_ident *e)
127{
128 brix_status st;
129
130 ident_login_name(e->uid, e->name, sizeof(e->name));
131
132 e->opts = g_opts;
133 e->opts.sss_user = e->name;
134
135 brix_status_clear(&st);
136 e->pool = brix_pool_create(&g_url, &e->opts, g_ident_conns, &st);
137 if (e->pool == NULL) {
138 fprintf(stderr, "xrootdfs: identity %s (uid %lu): meta pool: %s\n",
139 e->name, (unsigned long) e->uid, st.msg);
140 e->failed = 1;
141 return;
142 }
143
144 brix_status_clear(&st);
145 e->mgr = brix_mgr_create(&g_url, &e->opts, g_ident_streams, g_ident_streams,
146 g_max_stall, g_keepalive, g_max_retries, &st);
147 if (e->mgr == NULL) {
148 fprintf(stderr, "xrootdfs: identity %s (uid %lu): data manager: %s\n",
149 e->name, (unsigned long) e->uid, st.msg);
150 brix_pool_destroy(e->pool);
151 e->pool = NULL;
152 e->failed = 1;
153 }
154}

157/*
158 * ident_get — the slot for uid, fully built, or NULL.
159 *
160 * NULL means "this mount cannot serve that identity" — the table is full or the
161 * connect failed. The callers turn that into a refusal; none of them fall back
162 * to the mount-wide connections, because doing the work under the wrong
163 * identity is the exact failure this file exists to prevent.
164 */
165static xfs_ident *
166ident_get(uid_t uid)
167{
168 xfs_ident *e;
169
170 pthread_mutex_lock(&g_ident_lock);
171
172 for ( ;; ) {
173 e = ident_find_locked(uid);
174 if (e == NULL) {
175 e = ident_claim_locked(uid);
176 if (e == NULL) {
177 pthread_mutex_unlock(&g_ident_lock);
178 return NULL; /* table full — refuse, never fall back */
179 }
180 break; /* this thread builds it */
181 }
182 if (!e->creating) {
183 pthread_mutex_unlock(&g_ident_lock);
184 return e->failed ? NULL : e;
185 }
186 pthread_cond_wait(&g_ident_cond, &g_ident_lock);
187 }
188
189 pthread_mutex_unlock(&g_ident_lock);
190
191 ident_build(e);
192
193 pthread_mutex_lock(&g_ident_lock);
194 e->creating = 0;
195 pthread_cond_broadcast(&g_ident_cond);
196 pthread_mutex_unlock(&g_ident_lock);
197
198 return e->failed ? NULL : e;
199}

212/* Did uid fail for want of a slot (rather than a failed connect)? Checked only
213 * on the error path, so the extra lock round costs nothing in the common case. */
214static int
215ident_full(uid_t uid)
216{
217 int full;
218
219 pthread_mutex_lock(&g_ident_lock);
220 full = (ident_find_locked(uid) == NULL && g_ident_n >= g_ident_max);
221 pthread_mutex_unlock(&g_ident_lock);
222 return full;
223}

247/*
248 * xfs_ident_get — this caller's connections, or a negative errno.
249 *
250 * Returns 0 with the pool/mgr out-params set. The two failures are told apart
251 * because they mean different things in a log: -EMFILE is an operator
252 * capacity limit (raise --max-identities), -EACCES is this identity failing to
253 * connect or authenticate. Neither ever yields the mount-wide connections:
254 * serving a request under the wrong identity is the bug this file exists to
255 * prevent, so the request is refused instead. Either out-pointer may be NULL.
256 */
257int
258xfs_ident_get(brix_pool **pool, brix_mgr **mgr)
259{
260 xfs_ident *e;
261 uid_t uid;
262
263 if (!g_sss_ident) {
264 if (pool != NULL) { *pool = g_pool; }
265 if (mgr != NULL) { *mgr = g_mgr; }
266 return 0;
267 }
268
269 uid = ident_caller_uid();
270 e = ident_get(uid);
271 if (e == NULL) {
272 return ident_full(uid) ? -EMFILE : -EACCES;
273 }
274 if (pool != NULL) { *pool = e->pool; }
275 if (mgr != NULL) { *mgr = e->mgr; }
276 return 0;
277}
