9/*
10 * Phase 40 (a): build a per-transfer-UNIQUE temp path
11 * "<dst>.xrdcp-tmp.<pid>.<seq>". The pid alone is NOT unique under `-j`, whose
12 * batch workers are threads sharing one pid — two same-basename sources copied
13 * into one directory would otherwise collide on an identical temp name and
14 * interleave-corrupt it (then rename a garbage file into place, reported as
15 * success). A process-wide atomic sequence makes every concurrent transfer's
16 * temp distinct; pids already differ across separate processes. Returns 0, or -1
17 * if the composed path would not fit.
18 */
19int
20make_temp_path(const char *dst, char *out, size_t outsz)
21{
22 static atomic_ulong seq;
23 unsigned long s = atomic_fetch_add(&seq, 1ul);
24 if ((size_t) snprintf(out, outsz, "%s.xrdcp-tmp.%ld.%lu",
25 dst, (long) getpid(), s) >= outsz) {
26 return -1;
27 }
28 return 0;
29}

32/*
33 * open_download_temp — create a fresh private temp next to `dst` for the
34 * download+atomic-rename, and hand back its fd and name.
35 *
36 * WHY: the temp name is predictable (pid + counter), so an attacker with write
37 * access to the destination directory could pre-create it as a symlink and
38 * redirect our O_TRUNC onto a victim-owned file. O_EXCL refuses a
39 * pre-existing name and O_NOFOLLOW refuses a symlink, closing that race; we
40 * regenerate the name and retry on a stale collision so a leftover temp from
41 * a killed run doesn't wedge the transfer. Returns an fd (caller closes) and
42 * fills tmp[], or -1 with *st set.
43 */
44int
45open_download_temp(const char *dst, char *tmp, size_t tmpsz, brix_status *st)
46{
47 int attempt;
48
49 for (attempt = 0; attempt < 64; attempt++) {
50 int fd;
51 if (make_temp_path(dst, tmp, tmpsz) != 0) {
52 brix_status_set(st, XRDC_EUSAGE, 0, "destination path too long: %s", dst);
53 return -1;
54 }
55 fd = open(tmp, O_WRONLY | O_CREAT | O_EXCL | O_NOFOLLOW, 0644);
56 if (fd >= 0) {
57 return fd;
58 }
59 if (errno != EEXIST) {
60 brix_status_set(st, XRDC_EUSAGE, errno, "open %s: %s",
61 tmp, strerror(errno));
62 return -1;
63 }
64 }
65 brix_status_set(st, XRDC_EUSAGE, EEXIST,
66 "could not create a unique temp for %s", dst);
67 return -1;
68}

71/*
72 * Commit (or discard) a temp destination: on rc==0 rename `tmp`→`dest`
73 * atomically (downgrading rc to -1 with st set if the rename fails); on rc!=0
74 * drop the temp. Returns the final rc. Shared by every local-dest writer so the
75 * "atomic dest" guarantee lives in one place.
76 */
77int
78atomic_dest_finish(const char *tmp, const char *dest, int rc, brix_status *st)
79{
80 if (rc == 0) {
81 if (rename(tmp, dest) != 0) {
82 brix_status_set(st, XRDC_ESOCK, errno, "rename %s -> %s: %s",
83 tmp, dest, strerror(errno));
84 unlink(tmp);
85 return -1;
86 }
87 return 0;
88 }
89 unlink(tmp); /* drop the partial/cancelled/mismatched temp */
90 return rc;
91}

94/*
95 * WHAT: Reconnect a control connection to its home (redirector) endpoint,
96 * falling back to the currently-connected host/port when no home is set.
97 * WHY: Every resilient open/close retry re-establishes the session against the
98 * redirector (home_host/home_port) so a reopen re-runs the full locate,
99 * not a stale data-server address. Confining that host/port selection to
100 * one helper removes the duplicated `home ?: current` picking from both
101 * the download and upload retry loops.
102 * HOW: Prefer c->home_host / c->home_port when populated; otherwise use the
103 * live c->host / c->port. The reconnect result is intentionally ignored
104 * (the caller re-attempts the open and re-checks the deadline).
105 */
106void
107csctx_reopen_home(brix_conn *c, brix_status *st)
108{
109 const char *h = (c->home_host[0] != '\0') ? c->home_host : c->host;
110 int p = (c->home_port != 0) ? c->home_port : c->port;
111 (void) brix_reconnect(c, h, p, st);
112}

115/*
116 * WHAT: Decide whether a failed resilient attempt may be retried, sleeping a
117 * backoff step when it can.
118 * WHY: The download open, upload open, and upload close-commit loops share one
119 * give-up rule — stop on a non-retryable status, an operator cancel, or a
120 * blown stall deadline — and otherwise back off before the next attempt.
121 * One predicate keeps that rule identical across all three callers.
122 * HOW: Returns 1 (retry, after brix_backoff_sleep_fast) when the status is
123 * retryable, no cancel is pending, and the deadline is still in the future;
124 * returns 0 (give up) otherwise. `*attempt` is advanced on a retry so the
125 * backoff grows.
126 */
127int
128csctx_retry_gate(const brix_status *st, uint64_t deadline_ns, unsigned *attempt)
129{
130 if (!brix_status_retryable(st) || brix_copy_quit_requested()
131 || brix_mono_ns() >= deadline_ns) {
132 return 0;
133 }
134 brix_backoff_sleep_fast((*attempt)++);
135 return 1;
136}

139/*
140 * WHAT: Open the download source read handle, retrying reconnect+reopen within
141 * the stall deadline on a transport fault.
142 * WHY: The open is the last single-RTT step of setup and is just as sever-prone
143 * as connect/stat on a lossy link, so it rides the same resilient loop.
144 * Split out so download_stream_body stays a flat sequence.
145 * HOW: Issues brix_file_open_opaque when a compress opaque is set (ctx->opaque),
146 * else brix_file_open_read; on failure applies csctx_retry_gate and, if it
147 * permits, reconnects to home and retries. Returns 0 with *f open, or -1
148 * (st set) once the retry gate gives up.
149 */
150static int
151download_open_resilient(const copy_stream_ctx_t *ctx, brix_file *f,
152 brix_status *st)
153{
154 unsigned attempt = 0;
155 for (;;) {
156 int orc = (ctx->opaque != NULL)
157 ? brix_file_open_opaque(ctx->c, ctx->path,
158 ctx->opaque, 0, 0, 0, f, st)
159 : brix_file_open_read(ctx->c, ctx->path, f, st);
160 if (orc == 0) {
161 return 0;
162 }
163 if (!csctx_retry_gate(st, ctx->deadline_ns, &attempt)) {
164 return -1;
165 }
166 csctx_reopen_home(ctx->c, st); /* re-establish, then reopen */
167 }
168}

171/*
172 * WHAT: Open the source for read, stream the known-size body to the local sink,
173 * then close the remote handle — the whole "remote file is open" lifetime.
174 * WHY: Confining the open-read handle (and its secondary streams + scratch buf)
175 * to one helper lets the caller stay a flat early-return sequence: the file
176 * is always closed here, on every path, without a shared cleanup jump.
177 * HOW: open_read → streams_open(&ss) → pump(src, sink/sinkctx, si->size) →
178 * file_close. The caller supplies (sink, sinkctx): either a VFS file via
179 * pump_sink_local_vfs + pump_local_t, or the stdout fd via pump_sink_local.
180 * Returns 0 on a complete transfer, -1 (st set) otherwise. On open_read
181 * failure the streams are left untouched (ss.n stays 0, so the caller's
182 * streams_close is a no-op) — mirroring the original NULL-init. The
183 * connection is owned by the caller so it can run the post-transfer checksum
184 * before tearing down. NOTE: the invariant inputs (c/su/si/o/ss) ride in
185 * a download_body_ctx so the extern stays under the 5-parameter gate; the
186 * (sink, sinkctx) pump pair and st stay free (per-callsite / out-param).
187 */
188int
189download_stream_body(const download_body_ctx *j, pump_sink_fn sink,
190 void *sinkctx, brix_status *st)
191{
192 brix_conn *c = j->c;
193 const brix_url *su = j->su;
194 const brix_statinfo *si = j->si;
195 const brix_copy_opts *o = j->o;
196 brix_streamset *ss = j->ss;
197 brix_file f;
198 pump_remote_t src = {0};
199 int rc;
200 int stall = copy_stall_ms(o, 60000);
201 char opq[80];
202 copy_stream_ctx_t ctx = {0};
203
204 ctx.c = c;
205 ctx.path = su->path;
206 ctx.deadline_ns = brix_mono_ns() + (uint64_t) stall * 1000000ULL;
207
208 /* phase-42 W4: request inline read compression when --compress was given.
209 * Rides the open opaque; the server confirms via the open reply and
210 * brix_file_read transparently inflates. A server that doesn't support it
211 * just returns plaintext (f.read_codec stays 0), so this is always safe. */
212 if (o->compress != NULL && o->compress[0] != '\0') {
213 snprintf(opq, sizeof(opq), "xrootd.compress=%s", o->compress);
214 ctx.opaque = opq;
215 }
216 /* Open the remote handle, retried within max_stall on a transport fault
217 * (reconnecting between attempts). */
218 if (download_open_resilient(&ctx, &f, st) != 0) {
219 return -1;
220 }
221
222 /* M8: attach N-1 bound secondary streams to the (post-redirect) session. */
223 brix_streams_open(ss, c, o->streams, st);
224
225 /* remote (known si->size) → caller-supplied sink, with progress. Resilient:
226 * a sever mid-read reconnects + reopens at offset and adapts the request size,
227 * so a one-shot download rides out a flaky/lossy link. */
228 src.c = c;
229 src.f = &f;
230 src.pgrw = o->pgrw;
231 src.resilient = 1;
232 src.path = su->path;
233 src.opaque = ctx.opaque;
234 src.max_stall_ms = copy_stall_ms(o, 60000);
235 src.cur_chunk = XRDC_COPY_CHUNK;
236 /* Phase 94 parallel download: spread reads across the bound secondaries.
237 * Reads self-address by offset and any secondary miss falls back to the
238 * primary read, so this is safe against a server that won't serve bound
239 * reads. Inline read-compression stays single-stream (the server confirms
240 * the codec on the primary open; keep the decode path on that one conn). */
241 src.ss = (o->compress != NULL && o->compress[0] != '\0') ? NULL : ss;
242 src.rr_next = 0;
243 src.sec_reads = 0;
244 rc = transfer_pump(pump_src_remote, &src, sink, sinkctx,
245 si->size, o, si->size, st);
246
247 if (getenv("BRIX_STREAMS_DEBUG") != NULL) {
248 fprintf(stderr, "brix: download substreams=%d chunks-on-secondaries=%u\n",
249 ss->n, src.sec_reads);
250 }
251
252 {
253 brix_status throwaway;
254 brix_status_clear(&throwaway);
255 brix_file_close(c, &f, rc == 0 ? st : &throwaway);
256 }
257 return rc;
258}

260/*
261 * WHAT: Reconcile a completed download's checksum verdict into the transfer rc,
262 * emitting the "downloaded but NOT verified" note on an unverified query.
263 * WHY: Both the stdout and local-file download paths apply the identical
264 * MISMATCH→fail / UNVERIFIED→warn-and-clear rule after a good transfer;
265 * one helper keeps that policy in a single place and off copy_download's
266 * two branches.
267 * HOW: Runs cksum_verify(local_path may be NULL for stdout); a MISMATCH returns
268 * -1 (caller drops any committed file), an UNVERIFIED prints the note
269 * (unless silent) and clears st (a query hiccup is not a transfer failure),
270 * and an OK returns 0. `local_path` NULL ≡ stdout; the note names du->path.
271 */
272int
273download_reconcile_cksum(const download_job_t *job, const char *local_path,
274 brix_status *st)
275{
276 const brix_copy_opts *o = job->o;
277 int ck = cksum_verify(job->c, job->su->path, local_path, o->cksum,
278 o->silent, st);
279 if (ck == XRDC_CK_MISMATCH) {
280 return -1;
281 }
282 if (ck == XRDC_CK_UNVERIFIED) {
283 if (!o->silent) {
284 fprintf(stderr, "xrdcp: %s downloaded but checksum NOT verified: "
285 "%s\n", job->du->path, st->msg);
286 }
287 brix_status_clear(st);
288 }
289 return 0;
290}

293/*
294 * WHAT: Download the source body straight to stdout (no temp, no VFS, no commit),
295 * then verify its checksum.
296 * WHY: The stdio destination has no on-disk file to commit or unlink, so it is
297 * the simple half of copy_download; splitting it out keeps each branch a
298 * flat sequence.
299 * HOW: Pumps to STDOUT_FILENO via pump_sink_local; on success, when --cksum was
300 * given, reconciles the verdict with a NULL local path (cksum_verify skips
301 * gracefully, since stdout has no file). Connection is caller-owned.
302 */
303static int
304download_to_stdout(const download_job_t *job, brix_status *st)
305{
306 int stdoutfd = STDOUT_FILENO;
307 download_body_ctx dj = { job->c, job->su, job->si, job->o, job->ss };
308 int rc = download_stream_body(&dj, pump_sink_local, &stdoutfd, st);
309 if (rc == 0 && job->o->cksum != NULL) {
310 rc = download_reconcile_cksum(job, NULL, st);
311 }
312 return rc;
313}

316/*
317 * WHAT: Download the source body into a VFS-backed temp at du->path, then commit
318 * (fsync+rename) and verify the checksum, dropping the file on any failure.
319 * WHY: The local-file path adds the atomic temp+rename lifecycle and integrity
320 * drop-on-mismatch that the stdout path lacks; confining the VFS handle to
321 * this helper keeps its acquire/commit/abort/close linear (no shared jump).
322 * HOW: Opens du->path WRITE (+FORCE when -f) via brix_vfs_open, streams into it,
323 * and on success commits. A committed transfer with --cksum is reconciled;
324 * a genuine MISMATCH unlinks the committed-but-bad file and fails. On any
325 * failure before commit the temp is aborted (unlinked). Connection is
326 * caller-owned.
327 */
328static int
329download_to_local_file(const download_job_t *job, brix_status *st)
330{
331 const brix_copy_opts *o = job->o;
332 brix_vfs_file *vf = NULL;
333 brix_vfs_open_opts vopts = {0};
334 pump_local_t lc;
335 int committed = 0;
336 int rc;
337
338 vopts.io_uring = o->io_uring;
339 vopts.io_uring_direct = o->io_uring_direct;
340 vopts.expected_size = job->si->size;
341 vopts.cred = NULL;
342
343 if (brix_vfs_open(job->du->path,
344 XRDC_VFS_WRITE | (o->force ? XRDC_VFS_FORCE : 0),
345 &vopts, &vf, st) != 0) {
346 return -1;
347 }
348
349 lc.vf = vf;
350 {
351 download_body_ctx dj = { job->c, job->su, job->si, o, job->ss };
352 rc = download_stream_body(&dj, pump_sink_local_vfs, &lc, st);
353 }
354
355 /* Commit on success (fsync + rename temp→final); only then verify the
356 * checksum against the committed file. A genuine MISMATCH drops the
357 * committed file and returns error — it is an integrity failure, not a
358 * transient fault. A query hiccup (UNVERIFIED) keeps the good bytes. */
359 if (rc == 0) {
360 rc = brix_vfs_commit(vf, st);
361 if (rc == 0) {
362 committed = 1;
363 if (o->cksum != NULL) {
364 rc = download_reconcile_cksum(job, job->du->path, st);
365 if (rc != 0) {
366 unlink(job->du->path); /* drop committed-but-bad file */
367 }
368 }
369 }
370 }
371 if (rc != 0 && !committed) {
372 brix_vfs_abort(vf); /* discard the partial temp */
373 }
374 brix_vfs_close(vf);
375 return rc;
376}

380int
381copy_download(const brix_url *su, const brix_url *du, const brix_copy_opts *o,
382 const brix_opts *co, brix_status *st)
383{
384 brix_conn c;
385 brix_statinfo si;
386 brix_streamset ss;
387 download_job_t job = {0};
388 int to_stdout = (du->scheme == XRDC_SCHEME_STDIO);
389 int stall = copy_stall_ms(o, 60000);
390 int rc;
391
392 ss.n = 0; /* so the streams teardown is a no-op if we never bind */
393 if (resilient_setup(&c, su, co, &si, stall, st) != 0) {
394 return -1;
395 }
396 if (si.flags & kXR_isDir) {
397 brix_status_set(st, XRDC_EUSAGE, 0, "source is a directory (use -r, M5)");
398 brix_close(&c);
399 return -1;
400 }
401
402 job.c = &c;
403 job.su = su;
404 job.du = du;
405 job.si = &si;
406 job.o = o;
407 job.ss = &ss;
408
409 if (to_stdout) {
410 rc = download_to_stdout(&job, st);
411 brix_streams_close(&ss);
412 brix_close(&c);
413 return rc;
414 }
415
416 /* Local file path: existence-check preserving the original error message,
417 * then open via VFS (atomic temp+rename and optional io_uring inside the
418 * backend). commit() does fsync+rename; abort() unlinks the temp on any
419 * failure or checksum mismatch so the final destination is never partial. */
420 if (!o->force && access(du->path, F_OK) == 0) {
421 brix_status_set(st, XRDC_EUSAGE, 0,
422 "destination exists (use -f to overwrite): %s",
423 du->path);
424 brix_close(&c);
425 return -1;
426 }
427
428 /* Opt-in TRUE concurrent striped download (--parallel): one thread per bound
429 * connection, disjoint pwrite ranges. Returns 1 when it handled the transfer;
430 * 0 (not eligible) falls through to the serial resilient pump below. */
431 if (copy_download_parallel(&job, &rc, st)) {
432 brix_streams_close(&ss);
433 brix_close(&c);
434 return rc;
435 }
436
437 rc = download_to_local_file(&job, st);
438
439 brix_streams_close(&ss);
440 brix_close(&c);
441 return rc;
442}
