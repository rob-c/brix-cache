15static int
16s3_query_limits_ok(const brix_http_query_t *query)
17{
18 typedef struct {
19 const char *name;
20 size_t buffer_size;
21 size_t exact_len;
22 } s3_query_limit_t;
23 const s3_query_limit_t limits[] = {
24 { "uploadId", 25, 24 },
25 { "partNumber", 8, 0 },
26 { "list-type", 8, 0 },
27 { "encoding-type", 8, 0 },
28 { "fetch-owner", 8, 0 },
29 { "max-keys", 32, 0 },
30 { "max-uploads", 16, 0 },
31 { "max-parts", 16, 0 },
32 { "part-number-marker", 16, 0 },
33 { "key-marker", 256, 0 },
34 { "X-Amz-Algorithm", 64, 0 },
35 { "X-Amz-Credential", BRIX_S3_CREDENTIAL_BUF_SIZE, 0 },
36 { "X-Amz-Date", BRIX_S3_ISO8601_BUF_SIZE, 0 },
37 { "X-Amz-Expires", 32, 0 },
38 { "X-Amz-SignedHeaders", BRIX_S3_SIGNED_HEADERS_BUF_SIZE, 0 },
39 { "X-Amz-Signature", 128, 0 },
40 { "response-cache-control", BRIX_S3_CONDITIONAL_VAL_BUF, 0 },
41 { "response-content-disposition", BRIX_S3_CONDITIONAL_VAL_BUF, 0 },
42 { "response-content-encoding", BRIX_S3_CONDITIONAL_VAL_BUF, 0 },
43 { "response-content-language", BRIX_S3_CONDITIONAL_VAL_BUF, 0 },
44 { "response-content-type", 256, 0 },
45 { "response-expires", BRIX_S3_CONDITIONAL_VAL_BUF, 0 },
46 };
47 ngx_uint_t i;
48 size_t j;
49 size_t canonical_len = 0;
50 u_char encoded[BRIX_S3_QUERY_ENCODE_BUF_SIZE];
51
52 if (query->count > 64) {
53 return 0;
54 }
55
56 for (i = 0; i < query->count; i++) {
57 size_t buffer_size = BRIX_S3_HEADER_VALUE_BUF_SIZE;
58 size_t exact_len = 0;
59
60 for (j = 0; j < sizeof(limits) / sizeof(limits[0]); j++) {
61 if (query->params[i].name.len == ngx_strlen(limits[j].name)
62 && ngx_strncmp(query->params[i].name.data, limits[j].name,
63 query->params[i].name.len) == 0)
64 {
65 buffer_size = limits[j].buffer_size;
66 exact_len = limits[j].exact_len;
67 break;
68 }
69 }
70 if (query->params[i].value.len >= buffer_size
71 || (exact_len > 0 && query->params[i].value.len != exact_len))
72 {
73 return 0;
74 }
75 {
76 ssize_t name_len = brix_http_urlencode(query->params[i].name.data,
77 query->params[i].name.len, (char *) encoded, sizeof(encoded), "");
78 ssize_t value_len = brix_http_urlencode(query->params[i].value.data,
79 query->params[i].value.len, (char *) encoded, sizeof(encoded), "");
80
81 if (name_len < 0 || value_len < 0) {
82 return 0;
83 }
84 canonical_len += (i > 0) + (size_t) name_len + 1
85 + (size_t) value_len;
86 if (canonical_len >= BRIX_S3_CANONICAL_QUERY_BUF_SIZE) {
87 return 0;
88 }
89 }
90 }
91 return 1;
92}
