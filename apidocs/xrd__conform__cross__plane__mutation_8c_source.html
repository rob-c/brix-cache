51static int
52root_stat_size(xrd_conform_execution *execution, const char *target,
53 uint64_t expected, int absent)
54{
55 brix_url url;
56 brix_conn connection;
57 brix_statinfo info;
58 brix_status status;
59 char path[XRDC_PATH_MAX];
60 char reason[64];
61 int rc;
62 int matched;
63
64 brix_status_clear(&status);
65 if (brix_url_parse(target, &url, &status) != 0
66 || brix_conform_scratch_path(execution->run->config.scratch,
67 "brix-conform-xpr-mutation", path, sizeof(path)) != 0
68 || !xrd_conform_reserve(execution, 0, reason, sizeof(reason))
69 || brix_connect(&connection, &url, &execution->run->opts, &status) != 0)
70 return 0;
71 brix_status_clear(&status);
72 rc = brix_stat(&connection, path, &info, &status);
73 matched = absent ? rc != 0 && status.kxr == kXR_NotFound
74 : rc == 0 && info.size >= 0
75 && (uint64_t) info.size == expected;
76 brix_close(&connection);
77 return matched;

80static int
81s3_listed(xrd_conform_execution *execution, const brix_weburl *object)
82{
83 brix_weburl listing = *object;
84 brix_http_resp response;
85 const char *bucket_end = strchr(object->path + 1, '/');
86 const char *key;
87 char encoded[XRDC_PATH_MAX * 3];
88 char marker[XRDC_PATH_MAX + 16];
89 size_t bucket_length;
90 int rc;
91 int matched;
92
93 if (bucket_end == NULL) return 0;
94 key = bucket_end + 1;
95 bucket_length = (size_t) (bucket_end - object->path);
96 if (brix_http_urlencode((const unsigned char *) key, strlen(key), encoded,
97 sizeof(encoded), "") < 0
98 || (size_t) snprintf(listing.path, sizeof(listing.path),
99 "%.*s?list-type=2&max-keys=2&prefix=%s", (int) bucket_length,
100 object->path, encoded) >= sizeof(listing.path)
101 || (size_t) snprintf(marker, sizeof(marker), "<Key>%s</Key>", key)
102 >= sizeof(marker)) return 0;
103 rc = http_request(execution, &listing, "GET", NULL, NULL, 0, &response);
104 matched = rc == 0 && response.status == 200 && response.body != NULL
105 && strstr(response.body, "<IsTruncated>false</IsTruncated>") != NULL
106 && strstr(response.body, marker) != NULL;
107 if (rc == 0) brix_http_resp_free(&response);
108 return matched;
109}

162static mutation_observation
163observe_mutation(xrd_conform_execution *execution, const char *root,
164 const brix_weburl *http_object, const brix_weburl *s3_object,
165 int xpr_ready, const uint8_t *payload, size_t length, const char *token)
166{
167 mutation_observation observed = { 0, 0, 0, 0, 0, 0, 0 };
168
169 observed.put_ok = put_http_object(execution, http_object, token, payload,
170 length);
171 if (observed.put_ok) {
172 observed.root_ok = root_stat_size(execution, root, length, 0);
173 observed.http_bytes_ok = object_bytes(execution, http_object, payload,
174 length, 0);
175 if (xpr_ready) {
176 observed.list_ok = s3_listed(execution, s3_object);
177 observed.s3_bytes_ok = object_bytes(execution, s3_object, payload,
178 length, 0);
179 }
180 }
181 observed.common_cleanup_ok = delete_http_object(execution, http_object,
182 token) && root_stat_size(execution, root, 0, 1);
183 if (xpr_ready)
184 observed.xpr_cleanup_ok = observed.common_cleanup_ok
185 && object_bytes(execution, s3_object, payload, length, 1);
186 return observed;
187}

245void
246xrd_conform_probe_mutation_fixtures(xrd_conform_execution *execution)
247{
248 const brix_conform_result *xpr_row = xrd_conform_result_find(execution,
249 "XPR-009");
250 const brix_conform_result *htc_row = xrd_conform_result_find(execution,
251 "HTC-001");
252 const char *root = fixture_target(execution->run, "root");
253 const char *http = fixture_target(execution->run, "http");
254 const char *s3 = fixture_target(execution->run, "s3");
255 brix_weburl http_object;
256 brix_weburl s3_object;
257 uint8_t payload[XPR_MUTATION_SIZE];
258 char *token;
259 int xpr_active;
260 int htc_active;
261 int xpr_ready = 0;
262 mutation_observation observed;
263
264 xpr_active = result_active(xpr_row);
265 htc_active = result_active(htc_row);
266 if ((!xpr_active && !htc_active)
267 || execution->run->mutation_fixture_count == 0) return;
268 if (prepare_mutation_urls(execution, root, http, s3, &http_object,
269 &s3_object, &xpr_ready) != 0) return;
270 if (xpr_active && !xpr_ready && !htc_active) return;
271 brix_conform_fill(&execution->run->key, 0, payload, sizeof(payload));
272 token = brix_token_discover();
273 observed = observe_mutation(execution, root, &http_object, &s3_object,
274 xpr_ready, payload, sizeof(payload), token);
275 free(token);
276 if (htc_active) grade_http_mutation(execution, &observed);
277 if (!xpr_active || !xpr_ready) return;
278 grade_xpr_mutation(execution, &observed);
279}
