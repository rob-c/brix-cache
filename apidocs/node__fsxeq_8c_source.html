1/*
2 * node_fsxeq.c — 2.0 F17: run the operator program in place of a forwarded
3 * namespace op. See node_fsxeq.h for the contract.
4 *
5 * The three properties this file exists to hold, in the order they are checked:
6 *
7 * 1. CONFINEMENT. The built-in POSIX leg cannot be escaped: every path goes
8 * through openat2 RESOLVE_BENEATH under the export rootfd, so a hostile
9 * manager's "/../etc/shadow" fails EXDEV inside the kernel. An external
10 * program has no such floor — it is handed a path and trusted. So the
11 * path is gated LEXICALLY here, with the same rule recv_prepare.c applies
12 * to a forwarded stage path, BEFORE the program exists: absolute, bounded,
13 * and free of "..". This is the one security property the feature could
14 * quietly lose, and it is the reason the gate is in front of the fork
15 * rather than left to the operator's script.
16 *
17 * 2. THE WORKER SURVIVES A BAD PROGRAM. The run is a thread-pool task, so a
18 * program that blocks for its whole deadline costs one thread-pool slot
19 * and one forwarded op; the event loop keeps answering everything else.
20 * brix_subprocess_run supplies the rest: the double-forked agent (nginx's
21 * SIGCHLD handler reaps a worker's direct children, so a plain fork loses
22 * the exit status — F4's lesson), the program's own session, closefrom(3)
23 * of every inherited descriptor, and the SIGKILL of its whole process
24 * group when brix_cms_fsxeq_timeout expires.
25 *
26 * 3. THE REPLY IS STILL CMSD'S. Silent on success, kYR_error (kYR_EINVAL +
27 * text) on failure — byte-shape-identical to the built-in leg, so a stock
28 * manager cannot tell which leg answered. The manager link may have gone
29 * away while the program ran, so the completion handler re-checks it.
30 */

48/*
49 * One in-flight run. Allocated with the task from its own pool, which the
50 * completion handler destroys; the pool is the task's only owner, so a posted
51 * run leaks nothing whatever the program does.
52 *
53 * conn/conn_number snapshot the manager link the request arrived on. The ctx
54 * itself is per-worker and outlives every run (it is freed only at worker
55 * exit), but ctx->connection is re-dialled on every reconnect — answering a
56 * streamid from a dead link on a fresh one would be a reply to a request the
57 * new manager never made.
58 */
59typedef struct {
60 ngx_pool_t *pool;
61 ngx_brix_cms_ctx_t *ctx;
62 ngx_connection_t *conn;
63 ngx_atomic_uint_t conn_number;
64 uint32_t streamid;
65 unsigned timeout_ms;
66 const char *display; /* the configured command (NUL-term.) */
67 const char *opname;
68 char *argv[BRIX_CMS_FSXEQ_ARGV_MAX];
69 char numarg[32]; /* mode (octal) or trunc size */
70 char path[PATH_MAX];
71 char path2[PATH_MAX];
72 char out[FSXEQ_OUT_MAX];
73 int rc; /* brix_subprocess_run return */
74 int run_errno;
75 int exit_code;
76} fsxeq_task_t;

78/* Slot index for a planned action, or -1 for an action cms.fsxeq never
79 * covers (prepadd/prepdel are staging ops, not namespace ops). */
80static ngx_int_t
81fsxeq_slot(brix_cms_node_action_t action)
82{
83 switch (action) {
84 case XRDCMS_NACT_CHMOD: return BRIX_CMS_FSXEQ_CHMOD;
85 case XRDCMS_NACT_MKDIR: return BRIX_CMS_FSXEQ_MKDIR;
86 case XRDCMS_NACT_MKPATH: return BRIX_CMS_FSXEQ_MKPATH;
87 case XRDCMS_NACT_MV: return BRIX_CMS_FSXEQ_MV;
88 case XRDCMS_NACT_RM: return BRIX_CMS_FSXEQ_RM;
89 case XRDCMS_NACT_RMDIR: return BRIX_CMS_FSXEQ_RMDIR;
90 case XRDCMS_NACT_TRUNC: return BRIX_CMS_FSXEQ_TRUNC;
91 default: return -1;
92 }
93}

105/*
106 * The lexical confinement gate — see property 1 in the file header. Identical
107 * in rule to cms_prep_path_ok(): absolute, bounded, no ".." anywhere. The
108 * substring test also refuses a legitimate name containing "..", which is the
109 * conservative direction and keeps the two forwarded-op gates one rule.
110 */
111static int
112fsxeq_path_ok(const char *path)
113{
114 if (path == NULL || path[0] != '/' || ngx_strlen(path) >= PATH_MAX) {
115 return 0;
116 }
117 return ngx_strstr(path, "..") == NULL;
118}

120/* Both paths a plan can carry, gated. path2 is present only for mv. */
121static int
122fsxeq_plan_paths_ok(const brix_cms_node_plan_t *plan)
123{
124 if (!fsxeq_path_ok(plan->path)) {
125 return 0;

130/* Export-relative -> physical, the form stock hands its fsxeq program (the pfn,
131 * not the lfn). 0 on success, -1 when the join would not fit. */
132static int
133fsxeq_physical(const char *root_canon, const char *rel, char *buf, size_t cap)
134{
135 return brix_beneath_full_path(root_canon, rel, buf, cap) < (int) cap
136 ? 0 : -1;
137}

139/*
140 * Append the op's own arguments after the configured command line, the way
141 * XrdOucProg::Run() appends its arguments to the program's fixed prefix:
142 *
143 * chmod|mkdir|mkpath <mode octal> <path>
144 * trunc <size> <path>
145 * mv <path> <path2>
146 * rm|rmdir <path>
147 *
148 * Returns the total argv token count, or 0 when a path did not fit.
149 */
150static ngx_uint_t
151fsxeq_build_argv(fsxeq_task_t *t, const brix_cms_fsxeq_prog_t *prog,
152 const brix_cms_node_plan_t *plan, const char *root_canon)
153{
154 ngx_uint_t n = prog->argc;
155 ngx_uint_t i;
156
157 for (i = 0; i < prog->argc; i++) {
158 t->argv[i] = prog->argv[i];
159 }
160 if (fsxeq_physical(root_canon, plan->path, t->path, sizeof(t->path)) != 0) {
161 return 0;
162 }
163
164 switch (plan->action) {
165
166 case XRDCMS_NACT_MV:
167 if (fsxeq_physical(root_canon, plan->path2, t->path2,
168 sizeof(t->path2)) != 0)
169 {
170 return 0;
171 }
172 t->argv[n++] = t->path;
173 t->argv[n++] = t->path2;
174 break;
175
176 case XRDCMS_NACT_TRUNC:
177 snprintf(t->numarg, sizeof(t->numarg), "%lld",
178 (long long) plan->size);
179 t->argv[n++] = t->numarg;
180 t->argv[n++] = t->path;
181 break;
182
183 case XRDCMS_NACT_RM:
184 case XRDCMS_NACT_RMDIR:
185 t->argv[n++] = t->path;
186 break;
187
188 default: /* chmod, mkdir, mkpath — the mode-carrying ops */
189 snprintf(t->numarg, sizeof(t->numarg), "%04o",
190 (unsigned) (plan->mode & BRIX_CMS_MODE_BITS_MASK));
191 t->argv[n++] = t->numarg;
192 t->argv[n++] = t->path;
193 break;
194 }
195
196 t->argv[n] = NULL;
197 return n;
198}

200/* The thread-pool half: one bounded run, no nginx call anywhere. */
201static void
202fsxeq_thread(void *data, ngx_log_t *log)
203{
204 fsxeq_task_t *t = data;
205 brix_subprocess_req_t req;
206
207 (void) log;
208
209 t->out[0] = '\0';
210 req.argv = t->argv;
211 req.out = t->out;
212 req.outsz = sizeof(t->out);
213 req.timeout_ms = t->timeout_ms;
214
215 t->exit_code = -1;
216 t->rc = brix_subprocess_run(&req, NULL, &t->exit_code);
217 t->run_errno = errno;
218}

220/*
221 * Answer the manager, if the link the request arrived on is still the live one.
222 * A reconnect between post and completion retires the streamid with it.
223 */
224static void
225fsxeq_reply_error(fsxeq_task_t *t, const char *text)
226{
227 ngx_brix_cms_ctx_t *ctx = t->ctx;
228
229 if (ctx->connection == NULL || ctx->connection != t->conn
230 || ctx->connection->number != t->conn_number)
231 {
232 return;
233 }
234 (void) ngx_brix_cms_send_error(ctx, t->streamid, CMS_ERR_EINVAL, text);
235}

237/* The failure text, and the operator-facing log line that carries the program's
238 * own first words. Returns NULL when the run succeeded. */
239static const char *
240fsxeq_verdict(fsxeq_task_t *t, char *buf, size_t cap)
241{
242 if (t->rc != 0) {
243 return t->run_errno == ETIMEDOUT
244 ? "fsxeq program timed out" : "fsxeq program could not be run";
245 }
246 if (t->exit_code == 0) {
247 return NULL;
248 }
249 ngx_snprintf((u_char *) buf, cap - 1, "fsxeq program exited %d%Z",
250 t->exit_code);
251 return buf;
252}

254/* The event-loop half: log, reply, free. Nothing may touch the task after the
255 * pool goes. */
256static void
257fsxeq_done(ngx_event_t *ev)
258{
259 ngx_thread_task_t *task = ev->data;
260 fsxeq_task_t *t = task->ctx;
261 ngx_pool_t *pool = t->pool;
262 char buf[64];
263 const char *why;
264
265 (void) brix_rstrip(t->out);
266 why = fsxeq_verdict(t, buf, sizeof(buf));
267
268 if (why == NULL) {
269 ngx_log_debug3(NGX_LOG_DEBUG_EVENT, t->ctx->cycle->log, 0,
270 "brix: CMS node: fsxeq %s ok via \"%s\" (%s)",
271 t->opname, t->display, t->out);
272 } else {
273 ngx_log_error(NGX_LOG_NOTICE, t->ctx->cycle->log, 0,
274 "brix: CMS node: fsxeq %s path=%s via \"%s\" failed: %s%s%s",
275 t->opname, t->path, t->display, why,
276 t->out[0] ? "; said: " : "", t->out);
277 fsxeq_reply_error(t, why);
278 }
279
280 ngx_destroy_pool(pool);
281}

298/* Build and post the run. NGX_OK once the task owns the reply; NGX_DECLINED
299 * when nothing could be posted and the caller must answer the refusal. */
300static ngx_int_t
301fsxeq_post(ngx_brix_cms_ctx_t *ctx, uint32_t streamid, ngx_int_t slot,
302 const brix_cms_node_plan_t *plan)
303{
304 ngx_thread_pool_t *tp = brix_shared_thread_pool(&ctx->conf->common);
305 ngx_thread_task_t *task;
306 fsxeq_task_t *t;
307 ngx_pool_t *pool;
308
309 if (tp == NULL) {
310 return NGX_DECLINED;
311 }
312 pool = ngx_create_pool(BRIX_CMS_MAX_CONNECTIONS, ctx->cycle->log);
313 if (pool == NULL) {
314 return NGX_DECLINED;
315 }
316 task = ngx_thread_task_alloc(pool, sizeof(fsxeq_task_t));
317 if (task == NULL) {
318 ngx_destroy_pool(pool);
319 return NGX_DECLINED;
320 }
321 t = task->ctx;
322 fsxeq_task_init(t, pool, ctx, streamid, slot);
323
324 if (fsxeq_build_argv(t, ctx->conf->cms.fsxeq[slot], plan,
325 ctx->conf->common.root_canon) == 0)
326 {
327 ngx_destroy_pool(pool);
328 return NGX_DECLINED;
329 }
330
331 brix_task_bind(task, fsxeq_thread, fsxeq_done);
332 task->event.log = ctx->cycle->log;
333
334 if (ngx_thread_task_post(tp, task) != NGX_OK) {
335 ngx_destroy_pool(pool);
336 return NGX_DECLINED;
337 }
338 return NGX_OK;
339}

341ngx_int_t
342brix_cms_fsxeq_dispatch(ngx_brix_cms_ctx_t *ctx, u_char code,
343 uint32_t streamid, const brix_cms_node_plan_t *plan, int *taken)
344{
345 ngx_int_t slot = fsxeq_slot(plan->action);
346
347 *taken = 0;
348 if (slot < 0 || ctx->conf->cms.fsxeq[slot] == NULL) {
349 return NGX_OK;
350 }
351 *taken = 1;
352
353 if (!fsxeq_plan_paths_ok(plan)) {
354 ngx_log_error(NGX_LOG_NOTICE, ctx->cycle->log, 0,
355 "brix: CMS node: forwarded op code=%ui refused before fsxeq: "
356 "path denied", (ngx_uint_t) code);
357 return ngx_brix_cms_send_error(ctx, streamid, CMS_ERR_EINVAL,
358 "fsxeq path denied");
359 }
360
361 if (fsxeq_post(ctx, streamid, slot, plan) != NGX_OK) {
362 ngx_log_error(NGX_LOG_ERR, ctx->cycle->log, 0,
363 "brix: CMS node: fsxeq %s could not be posted (thread pool?)",
364 fsxeq_opname(slot));
365 return ngx_brix_cms_send_error(ctx, streamid, CMS_ERR_EINVAL,
366 "fsxeq program not runnable");
367 }
368 return NGX_OK;
369}
