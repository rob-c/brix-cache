99/* WHAT: Prove that accepted cache eviction causes a fresh origin fetch.
100 * WHY: Returning a stale cache block after successful eviction can preserve
101 * revoked or replaced content indefinitely.
102 * HOW: Populate a scratch object, evict it, replace equal-length origin bytes,
103 * and require the next cache read to equal the replacement exactly. */
104static void probe_evict(xrd_conform_execution *execution,
105 brix_conn *cache, brix_conn *origin)
106{
107 xrd_conform_scratch_object object;
108 uint8_t old_bytes[CAC_EVICT_SIZE];
109 uint8_t new_bytes[CAC_EVICT_SIZE];
110 uint8_t actual[CAC_EVICT_SIZE];
111 brix_status status;
112 int populated;
113
114 if (!xrd_conform_row_active(execution, "CAC-006")
115 && !xrd_conform_row_active(execution, "PRP-003")) return;
116 brix_status_clear(&status);
117 if (brix_conform_scratch_path(execution->run->config.scratch,
118 "brix-conform-cache-evict", object.path,
119 sizeof(object.path)) != 0) {
120 grade_evict(execution, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
121 "scratch-path-invalid", &status);
122 return;
123 }
124 if (write_version(execution, origin, &object, 0, old_bytes,
125 &status) != 0) {
126 grade_evict(execution, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
127 "cache-evict-origin-setup-failed", &status);
128 cleanup(execution, cache, origin, &object);
129 return;
130 }
131 populated = read_version(execution, cache, object.path, actual,
132 &status) == 0;
133 if (!populated || memcmp(actual, old_bytes, sizeof(actual)) != 0) {
134 grade_evict(execution,
135 populated ? BRIX_CONFORM_VERDICT_DIVERGENT
136 : BRIX_CONFORM_VERDICT_INCONCLUSIVE,
137 populated ? "cache-population-returned-wrong-bytes"
138 : "cache-evict-population-failed",
139 populated ? NULL : &status);
140 cleanup(execution, cache, origin, &object);
141 return;
142 }
143 if (prepare_path(execution, cache, object.path, kXR_stage, kXR_evict,
144 &status) != 0) {
145 grade_evict(execution, BRIX_CONFORM_VERDICT_DIVERGENT,
146 "cache-evict-refused", &status);
147 cleanup(execution, cache, origin, &object);
148 return;
149 }
150 if (write_version(execution, origin, &object, CAC_EVICT_SIZE,
151 new_bytes, &status) != 0) {
152 grade_evict(execution, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
153 "cache-evict-origin-replacement-failed", &status);
154 cleanup(execution, cache, origin, &object);
155 return;
156 }
157 if (read_version(execution, cache, object.path, actual, &status) != 0) {
158 grade_evict(execution, BRIX_CONFORM_VERDICT_DIVERGENT,
159 "cache-refetch-after-evict-failed", &status);
160 } else {
161 int fresh = memcmp(actual, new_bytes, sizeof(actual)) == 0;
162
163 grade_evict(execution,
164 fresh ? BRIX_CONFORM_VERDICT_CONFORMANT
165 : BRIX_CONFORM_VERDICT_DIVERGENT,
166 fresh ? "cache-evict-refetched-replacement-bytes"
167 : "cache-evict-served-stale-or-wrong-bytes",
168 NULL);
169 }
170 cleanup(execution, cache, origin, &object);
171}

173/* WHAT: Prove that kXR_fresh replaces a resident cache copy.
