65/* SUCCESS: every plane decodes to the expected scoped value. */
66static void
67test_parse_full(void)
68{
69 doctor_recon r;
70 memset(&r, 0, sizeof(r));
71 recon_init(&r, 0);
72 doctor_recon_parse_stats(STATS, &r);
73
74 /* link plane */
75 CHECK(r.conns_total == 123);
76 CHECK(r.bytes_in == 456789);
77 CHECK(r.bytes_out == 987654);
78 /* xrootd op plane (ops scoped; top-level err/rdr/dly) */
79 CHECK(r.ops_open == 50);
80 CHECK(r.ops_rd == 2000);
81 CHECK(r.ops_wr == 30);
82 CHECK(r.ops_err == 11);
83 CHECK(r.ops_rdr == 5);
84 CHECK(r.ops_dly == 2);
85 /* lgn sub-block: num here is 40, NOT the xrootd-level <num>3 or link <num>7 */
86 CHECK(r.lgn_num == 40);
87 CHECK(r.lgn_au == 37);
88 CHECK(r.lgn_af == 3);
89 /* ofs tpc */
90 CHECK(r.have_tpc == 1);
91 CHECK(r.tpc_grant == 8);
92 CHECK(r.tpc_deny == 2);
93 CHECK(r.tpc_err == 1);
94 /* oss capacity/inodes */
95 CHECK(r.oss_total == 1000000000);
96 CHECK(r.oss_free == 250000000);
97 CHECK(r.ino_total == 2048);
98 CHECK(r.ino_free == 1024);
99 /* http */
100 CHECK(r.have_http == 1);
101 CHECK(r.http_reqs == 777);
102 CHECK(r.http_in == 10);
103 CHECK(r.http_out == 20);
104 CHECK(r.http_tpc_pull == 3);
105 CHECK(r.http_tpc_push == 4);
106}

109/* SCOPING: doctor_recon_xml_i64 reads the tag only from the named block. */
110static void
111test_scoped_reads(void)
112{
113 /* <in>/<out> live in link, not xrootd. */
114 CHECK(doctor_recon_xml_i64(STATS, "link", "in") == 456789);
115 CHECK(doctor_recon_xml_i64(STATS, "link", "tot") == 123);
116 /* xrootd top-level <err> is 11 (ofs also has <err>1, must not leak). */
117 CHECK(doctor_recon_xml_i64(STATS, "xrootd", "err") == 11);
118 CHECK(doctor_recon_xml_i64(STATS, "ofs", "err") == 1);
119 /* a tag absent from the block => -1, even though it exists elsewhere. */
120 CHECK(doctor_recon_xml_i64(STATS, "link", "grnt") == -1);
121 /* an absent block => -1. */
122 CHECK(doctor_recon_xml_i64(STATS, "sched", "jobs") == -1);
123}

126/* ERROR / EDGE: empty, malformed, unterminated, and non-numeric inputs never
127 * crash and never fabricate a value — the "-1 = not reported" contract holds. */
128static void
129test_error_and_edge(void)
130{
131 doctor_recon r;
132
133 /* NULL / empty XML: parse is a no-op, sentinels intact. */
134 memset(&r, 0, sizeof(r));
135 recon_init(&r, 0);
136 doctor_recon_parse_stats(NULL, &r);
137 doctor_recon_parse_stats("", &r);
138 CHECK(r.conns_total == -1 && r.bytes_in == -1 && r.have_tpc == 0);
139
140 /* NULL args to the public reader. */
141 CHECK(doctor_recon_xml_i64(NULL, "link", "in") == -1);
142 CHECK(doctor_recon_xml_i64(STATS, NULL, "in") == -1);
143 CHECK(doctor_recon_xml_i64(STATS, "link", NULL) == -1);
144
145 /* block open tag but no </stats> close => treated as absent. */
146 CHECK(doctor_recon_xml_i64("<stats id=\"link\"><in>5</in>", "link", "in")
147 == -1);
148 /* tag opened but not closed within the block => still parses the digits up
149 * to end-of-buffer (strtoll stops at non-digit); a non-numeric body => -1. */
150 CHECK(doctor_recon_xml_i64("<stats id=\"link\"><in>x</in></stats>",
151 "link", "in") == -1);
152 /* negative values are honoured. */
153 CHECK(doctor_recon_xml_i64("<stats id=\"link\"><in>-9</in></stats>",
154 "link", "in") == -9);
155 /* "oss" must not match "oss.0" (closing-quote precision). */
156 CHECK(doctor_recon_xml_i64("<stats id=\"oss.0\"><tote>1</tote></stats>",
157 "oss", "tote") == -1);
158}

161/* CAPABILITY DECODE: bit set → names present; empty flags → empty string. */
162static void
163test_caps_decode(void)
164{
165 char buf[256];
166 int n;
167
168 /* no flags set => zero names, empty string (not garbage). */
169 buf[0] = 'x';
170 n = doctor_recon_caps_str(0, buf, sizeof(buf));
171 CHECK(n == 0 && buf[0] == '\0');
172
173 /* manager + TLS-available + gotoTLS + posc. */
174 n = doctor_recon_caps_str(kXR_isManager | kXR_haveTLS | kXR_gotoTLS
175 | kXR_supposc, buf, sizeof(buf));
176 CHECK(n == 4);
177 CHECK(strstr(buf, "manager") != NULL);
178 CHECK(strstr(buf, "tls-available") != NULL);
179 CHECK(strstr(buf, "tls-required") != NULL);
180 CHECK(strstr(buf, "posc") != NULL);
181 /* a bit NOT set must not appear. */
182 CHECK(strstr(buf, "proxy") == NULL);
183
184 /* server + proxy: comma-joined, order follows the table. */
185 n = doctor_recon_caps_str(kXR_isServer | kXR_attrProxy, buf, sizeof(buf));
186 CHECK(n == 2 && strcmp(buf, "server,proxy") == 0);
187}
