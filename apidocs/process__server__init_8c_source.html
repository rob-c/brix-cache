1/*
2 * process_server_init.c — the per-server worker-init ladder.
3 *
4 * WHAT: Owns brix_init_one_server() — every per-server init step for one
5 * enabled brix server block, in the frozen worker-init order (credential
6 * replay, log-fd capture, staging registry, export rootfd, checkpoint
7 * recovery, cache storage, XrdAcc, CMS, health-check, Pelican advertise,
8 * the two cache reaper timers, CRL/JWKS) — together with the file-local
9 * per-step helpers it delegates to.
10 * WHY: Split (phase-79 file-size cap) out of the former 925-line process.c.
11 * The per-server ladder is a self-contained concern: keeping it in one
12 * focused file leaves the init_process orchestrator (process.c) a short
13 * loop while preserving the EXACT initialization order the
14 * reload-semantics contract freezes. Only brix_init_one_server crosses
15 * the file boundary (declared in process_internal.h); every step helper
16 * stays file-local.
17 * HOW: brix_init_one_server runs the step helpers as a flat early-return
18 * sequence; a step returning NGX_ERROR aborts worker startup. The two
19 * reaper-timer arming helpers point their events at the callbacks defined
20 * in process_timers.c (brix_cache_reap_handler and, from
21 * reap_watermark.h, brix_cache_watermark_timer_handler). No behaviour
22 * change from the split.
23 */

41/* ---- Replay the remote-backend credential into this worker's VFS registry ----
42 *
43 * WHAT: Re-applies the server's configured storage credential to the
44 * process-global VFS backend registry entry for its export root. No return
45 * value — a missing credential/backend is a legitimate no-op.
46 *
47 * WHY: brix_vfs_backend_set_credential runs only at config parse, in the
48 * throwaway config-load process; the serving master/workers do not inherit
49 * that entry's credential, so their registry has the backend but an EMPTY
50 * credential and the origin login fails with "no credential set". The srv
51 * conf's fields ARE inherited reliably (nginx config), so replay them here,
52 * per worker, before any request resolves + builds the backend instance.
53 *
54 * HOW:
55 * 1. Skip unless both a storage credential name and an export root are set.
56 * 2. NUL-terminate the credential name and look it up in the credential
57 * block table; a miss is loud (error log) — a worker serving with a
58 * wiped credential must never be silent.
59 * 3. Map the credential through brix_credential_to_backend_cred() — the
60 * ONE shared mapper (P80.1). Its earlier hand-copied twin here dropped
61 * bearer + all three s3 fields, and because set_credential overwrites
62 * all 8 registry slots unconditionally, every worker spawn wiped the
63 * parse-time S3 keys to "" (phase-80 finding 1.1).
64 * 4. Install it on the export root's registry entry; set_credential also
65 * resets e->inst so the instance rebuilds with the credential on first
66 * use.
67 */
68static void
69brix_init_server_backend_credential(ngx_cycle_t *cycle,
70 ngx_stream_brix_srv_conf_t *xcf)
71{
72 char credz[256];
73 char bearer[4096];
74 const brix_credential_t *cred;
75 brix_vfs_backend_cred_t bcred;
76
77 if (xcf->common.storage_credential.len == 0
78 || xcf->common.root_canon[0] == '\0')
79 {
80 return;
81 }
82
83 ngx_cpystrn((u_char *) credz, xcf->common.storage_credential.data,
84 ngx_min(xcf->common.storage_credential.len + 1,
85 sizeof(credz)));
86 cred = brix_credential_lookup(credz);
87 if (cred == NULL) {
88 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
89 "brix: worker credential replay: no brix_credential \"%s\" "
90 "for export \"%s\" — backend keeps its (empty) worker-side "
91 "credential; upstream auth WILL fail", credz,
92 xcf->common.root_canon);
93 return;
94 }
95
96 if (brix_credential_to_backend_cred(cred, bearer, sizeof(bearer),
97 &bcred, cycle->log) != NGX_OK)
98 {
99 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
100 "brix: worker credential replay: cannot derive bearer for "
101 "brix_credential \"%s\" (export \"%s\") — credential NOT "
102 "installed; upstream auth WILL fail", credz,
103 xcf->common.root_canon);
104 return;
105 }
106 brix_vfs_backend_set_credential(xcf->common.root_canon, &bcred);
107}

109/* ---- Open this worker's fds onto the staging registry + resume sweep ----
110 *
111 * WHAT: Initialises the composable stage request registry from the server's
112 * (tape) control dir and registers the upload-resume TTL sweep for its stage
113 * dir. No return value — both are best-effort, idempotent no-ops when the
114 * respective dir is not configured.
115 *
116 * WHY: The migrated staging callers (prepare/tape_rest/open) record into the
117 * registry, which needs per-worker fds; abandoned upload-resume partials in
118 * the stage dir would otherwise leak disk. The legacy FRM
119 * queue/scheduler/purge worker-init is retired — the recall is driven by the
120 * sd_frm backend + the client poll model.
121 *
122 * HOW:
123 * 1. When FRM is enabled with a control dir, NUL-terminate the dir for the
124 * C API and init the stage registry (journal dir = the control dir).
125 * 2. When an upload stage dir is canonicalised, register the resume sweep
126 * (worker-0 only internally; the register itself is idempotent and arms
127 * a single timer for the first stage-dir server).
128 */
129static void
130brix_init_server_stage_registry(ngx_cycle_t *cycle,
131 ngx_stream_brix_srv_conf_t *xcf)
132{
133 if (xcf->frm.enable && xcf->frm.control_dir.len > 0) {
134 char _jd[NGX_MAX_PATH];
135 if (brix_str_cbuf(_jd, sizeof(_jd), &xcf->frm.control_dir) != NULL) {
136 (void) brix_stage_registry_init(_jd, cycle->log);
137 }
138 }
139
140 /* Phase 6 housekeeping: TTL-sweep abandoned upload-resume partials from
141 * the stage dir (worker-0 only; the register itself is idempotent and
142 * arms a single timer for the first stage-dir server). */
143 if (xcf->upload_stage_dir_canon[0] != '\0') {
144 brix_xfer_resume_sweep_register(cycle,
145 xcf->upload_stage_dir_canon);
146 }
147}

149/* ---- Startup advisor: an unstageable nearline export says so ONCE ----
150 *
151 * WHAT: One worker-0 [warn] (phase-107 C2 advisor note) when this export's
152 * composed chain declares a nearline tier that no recall slot can stage AND
153 * no brix_prepare_command fallback is configured: every kXR_prepare(kXR_stage)
154 * it will ever see answers kXR_Unsupported. Advisory — never fatal.
155 *
156 * WHY: "Can never stage" is a config-time fact; discovering it at the first
157 * client prepare hides it from the operator behind a client-side error. The
158 * capability walk itself lives in the VFS (brix_vfs_chain_nearline_
159 * unstageable), so this note can never disagree with brix_vfs_recall's own
160 * tier selection.
161 *
162 * HOW:
163 * 1. Skip when a prepare_command is configured (the fallback stages), when
164 * there is no local export root, or off worker 0 (one note, not N).
165 * 2. Resolve the export's chain head (NULL = default POSIX, never
166 * nearline) and ask the VFS probe; warn iff it says unstageable.
167 */
168static void
169brix_init_server_stage_advisor(ngx_cycle_t *cycle,
170 ngx_stream_brix_srv_conf_t *xcf)
171{
172 brix_sd_instance_t *chain;
173
174 if (xcf->prepare_command.len > 0
175 || xcf->common.root_canon[0] == '\0'
176 || ngx_worker != 0)
177 {
178 return;
179 }
180 chain = brix_vfs_backend_resolve(xcf->common.root_canon, cycle->log);
181 if (chain != NULL && brix_vfs_chain_nearline_unstageable(chain)) {
182 ngx_log_error(NGX_LOG_WARN, cycle->log, 0,
183 "brix: export \"%s\" is nearline but no tier of its driver "
184 "chain implements recall and no brix_prepare_command is "
185 "configured - kXR_prepare(kXR_stage) can never stage and will "
186 "answer kXR_Unsupported; configure brix_prepare_command or a "
187 "recall-capable nearline tier",
188 xcf->common.root_canon);
189 }
190}

192/* ---- Open the confined export-root fd for a data server ----
193 *
194 * WHAT: Opens the server's export root as an O_PATH directory fd for
195 * kernel-confined path operations. Returns NGX_OK on success or when no
196 * local export root is configured; NGX_ERROR (with an EMERG log) when the
197 * root cannot be opened.
198 *
199 * WHY: All confined path resolution (openat2 RESOLVE_IN_ROOT / O_NOFOLLOW
200 * walks) anchors on this fd; a data server that cannot open its export root
201 * must refuse to run rather than serve unconfined.
202 *
203 * HOW:
204 * 1. Skip servers with an empty root_canon (proxy/manager/supervisor).
205 * 2. open(root, O_PATH|O_DIRECTORY|O_CLOEXEC) into xcf->rootfd.
206 * 3. On failure log EMERG with errno and return NGX_ERROR.
207 */
208static ngx_int_t
209brix_init_server_rootfd(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *xcf)
210{
211 /* Only open rootfd for data servers with a local export root.
212 * Proxy/manager/supervisor servers leave root_canon empty. */
213 if (xcf->common.root_canon[0] == '\0') {
214 return NGX_OK;
215 }
216
217 xcf->rootfd = open(xcf->common.root_canon,
218 O_PATH | O_DIRECTORY | O_CLOEXEC);
219 if (xcf->rootfd < 0) {
220 ngx_log_error(NGX_LOG_EMERG, cycle->log, errno,
221 "brix: cannot open export root \"%s\" for "
222 "kernel-confined path operations",
223 xcf->common.root_canon);
224 return NGX_ERROR;
225 }
226
227 return NGX_OK;
228}

230/* ---- Arm the per-server stale-dirty cache reaper timer ----
231 *
232 * WHAT: Allocates and arms the hourly stale-dirty reaper timer for a server
233 * whose cache-state engine is active. Returns NGX_OK (including the
234 * not-configured no-op) or NGX_ERROR on allocation failure.
235 *
236 * WHY: The eviction guard protects dirty write-back staging files, so an
237 * abandoned flush would leak disk forever without a reaper that runs on a
238 * maintenance timer independent of occupancy.
239 *
240 * HOW:
241 * 1. Skip unless a reaper policy is armed (brix_cache_dirty_max_age for
242 * abandoned write-back, or brix_cache_cold_max_age for the cold read-fill
243 * purge — the same walk services both) and a state root resolves.
244 * 2. pcalloc the timer event from the cycle pool (NGX_ERROR on failure).
245 * 3. Point it at brix_cache_reap_handler with the srv conf as ev->data,
246 * mark cancelable, arm at the resolved first-tick delay.
247 */
248static ngx_int_t
249brix_init_server_cache_reap_timer(ngx_cycle_t *cycle,
250 ngx_stream_brix_srv_conf_t *xcf)
251{
252 if ((xcf->cache_dirty_max_age <= 0 && xcf->cache_cold_max_age <= 0)
253 || brix_cache_state_root(xcf) == NULL)
254 {
255 return NGX_OK;
256 }
257
258 xcf->cache_reap_timer = ngx_pcalloc(cycle->pool, sizeof(ngx_event_t));
259 if (xcf->cache_reap_timer == NULL) {
260 return NGX_ERROR;
261 }
262 xcf->cache_reap_timer->handler = brix_cache_reap_handler;
263 xcf->cache_reap_timer->data = xcf;
264 xcf->cache_reap_timer->log = cycle->log;
265 xcf->cache_reap_timer->cancelable = 1; /* don't delay graceful shutdown */
266 ngx_add_timer(xcf->cache_reap_timer,
267 brix_cache_reap_delay(BRIX_CACHE_REAP_FIRST_MS));
268
269 return NGX_OK;
270}

273 *
274 * WHAT: Allocates and arms the background CSI scrub timer for a server that set

276 * (including the not-configured no-op) or NGX_ERROR on allocation failure.
277 *
278 * WHY: CSI's hot read path only verifies the blocks a read fully spans, so a

280 * A periodic sweep surfaces at-rest rot proactively (metrics + error.log DIAG).
281 *

285 * 2. pcalloc the timer event from the cycle pool (NGX_ERROR on failure).
286 * 3. Point it at brix_csi_scrub_handler with the srv conf as ev->data, mark
287 * cancelable, arm at the fixed startup offset (steady state re-arms at the
288 * operator's interval).

295 return NGX_OK;
296 }
297

307
308 return NGX_OK;

311/* ---- Arm the per-server watermark-driven LRU reaper timer ----
312 *
313 * WHAT: Allocates and arms the proactive watermark LRU reaper timer for a
314 * server with a cache and a valid HIGH watermark. Returns NGX_OK (including
315 * the not-configured no-op) or NGX_ERROR on allocation failure.
316 *
317 * WHY: Waiting for occupancy-triggered eviction alone lets a busy cache
318 * overshoot its watermark; a periodic proactive reap keeps it bounded. A
319 * small per-worker jitter on the first tick keeps the workers from all
320 * firing together.
321 *
322 * HOW:
323 * 1. Skip unless a cache is configured, a state root resolves, and
324 * 0 < high_watermark < 1000000.
325 * 2. pcalloc the timer event from the cycle pool (NGX_ERROR on failure).
326 * 3. Point it at brix_cache_watermark_timer_handler, mark cancelable, arm
327 * at the resolved first-tick delay + per-worker jitter (bounded by the
328 * delay itself so a shortened test cadence stays short).
329 */
330static ngx_int_t
331brix_init_server_watermark_timer(ngx_cycle_t *cycle,
332 ngx_stream_brix_srv_conf_t *xcf)
333{
334 /* Arm the timer when a cache exists AND at least one reaper arm is live:
335 * a valid ppm FS watermark, OR the §4.7 owned-bytes cap (max_bytes > 0).
336 * Either alone is enough — brix_cache_watermark_purge runs whichever fires. */
337 if ((!xcf->cache && xcf->common.cache_store.len == 0)
338 || brix_cache_state_root(xcf) == NULL
339 || ((xcf->reaper.high_watermark <= 0
340 || xcf->reaper.high_watermark >= 1000000)
341 && xcf->reaper.max_bytes <= 0))
342 {
343 return NGX_OK;
344 }
345
346 xcf->reaper.timer =
347 ngx_pcalloc(cycle->pool, sizeof(ngx_event_t));
348 if (xcf->reaper.timer == NULL) {
349 return NGX_ERROR;
350 }
351 xcf->reaper.timer->handler =
352 brix_cache_watermark_timer_handler;
353 xcf->reaper.timer->data = xcf;
354 xcf->reaper.timer->log = cycle->log;
355 xcf->reaper.timer->cancelable = 1; /* don't delay graceful shutdown */
356 {
357 ngx_msec_t first = brix_cache_reap_delay(BRIX_CACHE_REAP_FIRST_MS);
358 ngx_msec_t span = (first < 1000) ? first : 1000;
359 ngx_msec_t jitter = (span > 0) ? (ngx_msec_t) (ngx_pid % span) : 0;
360
361 ngx_add_timer(xcf->reaper.timer, first + jitter);
362 }
363
364 return NGX_OK;
365}

367/* ---- Arm the per-server CRL reload timer + schedule JWKS refresh ----
368 *
369 * WHAT: Starts the CRL hot-reload timer for GSI servers that configured
370 * brix_crl_reload, and schedules the JWKS refresh in every case. Returns
371 * NGX_OK, or NGX_ERROR when the timer event cannot be allocated.
372 *
373 * WHY: Timers are per-worker because each nginx worker process has its own
374 * event loop and its own copy of the config pointers (but the X509_STORE*
375 * is shared within a worker). JWKS refresh is orthogonal to CRLs and must be
376 * scheduled whether or not a CRL timer is armed.
377 *
378 * HOW:
379 * 1. When the server is not GSI/BOTH, or has no CRL path/interval: just
380 * schedule the JWKS refresh and return NGX_OK.
381 * 2. Otherwise pcalloc the CRL timer event (NGX_ERROR on failure), point
382 * it at brix_crl_reload_handler, mark cancelable, arm at the configured
383 * interval and log a NOTICE.
384 * 3. Schedule the JWKS refresh for the GSI/BOTH server too.
385 */
386static ngx_int_t
387brix_init_server_crl_jwks(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *xcf)
388{
389 if ((xcf->auth != BRIX_AUTH_GSI && xcf->auth != BRIX_AUTH_BOTH
390 && !brix_protbind_any_names(xcf->protbind, BRIX_AUTH_GSI))
391 || xcf->crl.len == 0 || xcf->crl_reload == 0)
392 {
393 /* CRL timer not needed — but still check for JWKS refresh */
394 brix_token_jwks_schedule_refresh(cycle, xcf);
395 return NGX_OK;
396 }
397
398 /* Allocate and start the CRL reload timer */
399 xcf->crl_timer = ngx_pcalloc(cycle->pool, sizeof(ngx_event_t));
400 if (xcf->crl_timer == NULL) {
401 return NGX_ERROR;
402 }
403 xcf->crl_timer->handler = brix_crl_reload_handler;
404 xcf->crl_timer->data = xcf;
405 xcf->crl_timer->log = cycle->log;
406 xcf->crl_timer->cancelable = 1; /* don't delay graceful shutdown */
407
408 ngx_add_timer(xcf->crl_timer, (ngx_msec_t) xcf->crl_reload * 1000);
409
410 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
411 "brix: CRL reload timer started - interval=%ds "
412 "path=\"%s\"",
413 (int) xcf->crl_reload, xcf->crl.data);
414
415 /* Also check for JWKS refresh on GSI/BOTH servers */
416 brix_token_jwks_schedule_refresh(cycle, xcf);
417
418 return NGX_OK;
419}

421/* ---- Per-server worker initialization for one enabled brix server block ----
422 *
423 * WHAT: Runs every per-server init step for one enabled server conf, in the
424 * frozen worker-init order: credential replay, log-fd capture, staging
425 * registry, export rootfd, checkpoint recovery, cache storage, XrdAcc, CMS,
426 * health-check, Pelican advertise, cache reaper timers, CRL/JWKS. Returns
427 * NGX_OK, or NGX_ERROR to abort worker startup.
428 *
429 * WHY: Worker init is a flat sequence of independent subsystem
430 * initializations; keeping each server's ladder in one orchestrator keeps
431 * init_process itself a short loop while preserving the exact initialization
432 * order the reload-semantics contract freezes.
433 *
434 * HOW:
435 * 1. Replay the storage credential into the VFS backend registry.
436 * 2. Capture the nginx-managed access/audit log fds.
437 * 3. Init the staging registry + upload-resume sweep.
438 * 4. Open the confined export rootfd (fatal on failure).
439 * 5. Recover interrupted checkpoints, build cache SD storage instances,
440 * build XrdAcc tables — each fatal on failure.
441 * 6. Start the health-check timer + Pelican cache advertisement — internal
442 * no-ops when unconfigured. (The outbound CMS client is started earlier,
443 * per-block and worker-0-gated, in brix_cms_role_worker_init.)
444 * 7. Arm the stale-dirty and watermark cache reaper timers (fatal on
445 * allocation failure).
446 * 8. Arm the CRL reload timer / schedule JWKS refresh.
447 */
448ngx_int_t
449brix_init_one_server(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *xcf)
450{
451 brix_init_server_backend_credential(cycle, xcf);
452
453 /*
454 * Capture the nginx-managed log fds the master opened during
455 * ngx_init_cycle. We read file->fd here (not at config time, where it
456 * is still NGX_INVALID_FILE) so the hot access-/audit-log write paths
457 * keep using a plain int fd. nginx dup2()s a reopened log onto the same
458 * fd number on USR1, so this captured value stays valid for the worker's
459 * lifetime. A NULL handle means logging is disabled for this server.
460 */
461 xcf->access_log_fd = xcf->access_log_file != NULL
462 ? xcf->access_log_file->fd : NGX_INVALID_FILE;
463 xcf->proxy.audit_log_fd = xcf->proxy.audit_log_file != NULL
464 ? xcf->proxy.audit_log_file->fd : NGX_INVALID_FILE;
465
466 brix_init_server_stage_registry(cycle, xcf);
467
468 if (brix_init_server_rootfd(cycle, xcf) != NGX_OK) {
469 return NGX_ERROR;
470 }
471
472 if (brix_chkpoint_recover_root(cycle->log, xcf->common.root_canon)
473 != NGX_OK)
474 {
475 return NGX_ERROR;
476 }
477
478 /* The per-export storage backend (e.g. pblock) is registered at config
479 * time and built per worker lazily by the VFS backend registry on first
480 * use (brix_vfs_ctx_init → brix_vfs_backend_resolve); no per-server
481 * init_process creation is needed here. */
482
483 /* The cache performs all disk I/O through SD storage instances (POSIX
484 * driver on a per-worker rootfd by default, or a configured backend) —
485 * build them now. No-op unless a cache is configured. */
486 if (brix_cache_storage_init(xcf, cycle) != NGX_OK) {
487 return NGX_ERROR;
488 }
489
490 /* Build the per-worker XrdAcc tables + hot-reload timer (no-op unless
491 * this server uses `brix_authdb_engine xrdacc`). */
492 if (brix_acc_init_server(xcf, cycle) != NGX_OK) {
493 return NGX_ERROR;
494 }
495
496 /*
497 * The outbound CMS client is started in brix_cms_role_worker_init
498 * (process.c), which runs for every server block (including cms-only
499 * manager blocks with the data path disabled) and gates the client to
500 * worker 0 so a stock upstream cmsd admits a single connection per node.
501 * It must NOT be started here as well: this ladder runs on every worker,
502 * which would resurrect the per-worker connection collision.
503 */
504
505 /* Phase-107 C2 advisor: warn once when a nearline export can never
506 * stage (no recall-capable tier, no prepare_command). After the cache
507 * storage init so resolving the chain here builds it in its final,
508 * fully-registered shape. */
509 brix_init_server_stage_advisor(cycle, xcf);
510
511 /* Phase 22: start the active health-check timer (no-op if disabled). */
512 brix_hc_manager_start(cycle, xcf);
513
514 /* Pelican: start the cache advertisement timer (no-op unless
515 * brix_cache_advertise is on with a key + data-url configured). */
516 brix_cache_pelican_schedule_advertise(cycle, xcf);
517
518 /* Unified cache-state engine: arm the per-worker stale-dirty reaper when
519 * a state root resolves and a max age is set. Independent of occupancy. */
