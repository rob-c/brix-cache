10/*
11 * kXR_protocol - negotiate protocol version, auth protocols, and TLS support.
12 *
13 * This is the first XRootD request after the raw handshake. It does not log a
14 * user in; it only advertises server capabilities and, when configured, arms
15 * the native TLS upgrade for the following login/auth flow.
16 */
17
18/* WHAT: decoded client request + the server-side auth/TLS wants derived from it.
19 * WHY: the negotiation is a flag matrix over (client capability byte, configured
20 * auth mode, TLS availability); collecting the decoded flags in one struct
21 * lets each helper take a single explicit parameter instead of ten ints.
22 * HOW: filled once by protocol_parse_client_req(), then read-only everywhere. */
23typedef struct {
24 u_char client_flags; /* fifth byte of the kXR_protocol body */
25 int gsi; /* offer "gsi " SecurityProtocol entry */
26 int token; /* offer "ztn " SecurityProtocol entry */
27 int sss; /* offer "sss " SecurityProtocol entry */
28 int unx; /* offer "unix" SecurityProtocol entry */
29 int krb5; /* offer "krb5" SecurityProtocol entry */
30 int host; /* offer "host" SecurityProtocol entry */
31 int pwd; /* offer "pwd " SecurityProtocol entry */
32 int client_wants_tls; /* client set kXR_wantTLS (TLS is mandatory) */
33 int offer_tls; /* server can and will offer the TLS upgrade */
34} protocol_want_t;
35
36/* WHAT: decode the kXR_protocol request into a protocol_want_t.
37 * WHY: isolates the request-byte decode and auth-policy fan-out so the handler
38 * body only deals with negotiation and response assembly. The offered
39 * SecurityProtocol entries must name the SAME protocols the kXR_login sec
40 * token will advertise and kXR_auth will accept, so all three read the
41 * one resolved protbind set rather than re-deriving it from conf->auth.
42 * HOW: reads the client capability byte from the received body, resolves the
43 * per-host protocol binding for this peer, and computes the two TLS
44 * decisions from the client flags + listener TLS config. */
45static void
46protocol_parse_client_req(brix_ctx_t *ctx, ngx_connection_t *c,
47 ngx_stream_brix_srv_conf_t *conf, protocol_want_t *want)
48{
49 brix_protbind_set_t bound;
50
51 /* kXR_protocol packs client capability flags into the fifth byte of body[]. */
52 want->client_flags = ctx->recv.cur_body[4];
53
54 brix_protbind_resolve_ctx(ctx, c, conf->protbind, conf->auth, &bound);
55
56 want->gsi = brix_protbind_allows(&bound, BRIX_AUTH_GSI);
57 want->token = brix_protbind_allows(&bound, BRIX_AUTH_TOKEN);
58 want->sss = brix_protbind_allows(&bound, BRIX_AUTH_SSS);
59 want->unx = brix_protbind_allows(&bound, BRIX_AUTH_UNIX);
60 want->krb5 = brix_protbind_allows(&bound, BRIX_AUTH_KRB5);
61 want->host = brix_protbind_allows(&bound, BRIX_AUTH_HOST);
62 want->pwd = brix_protbind_allows(&bound, BRIX_AUTH_PWD);
63
64 /* kXR_wantTLS: client requires TLS; kXR_ableTLS: client is TLS-capable. */
65 want->client_wants_tls = (want->client_flags & kXR_wantTLS) ? 1 : 0;
66 want->offer_tls = (conf->tls && conf->tls_ctx != NULL
67 && (want->client_flags & (kXR_ableTLS | kXR_wantTLS)));
68}
69
70/* WHAT: the server-role capability bits (manager/supervisor/redirector/meta/
71 * proxy/cache) for the ServerProtocolBody flags word.
72 * WHY: splits the role half of the capability bit-ladder out of the
73 * negotiation so each half stays a small pure expression.
74 * HOW: pure mapping from the merged server conf; no side effects. */
75static uint32_t
76protocol_role_flags(const ngx_stream_brix_srv_conf_t *conf)
77{
78 return ((conf->manager_map || conf->manager_mode) ? kXR_isManager : 0)
79 | (conf->caps.supervisor
80 ? (kXR_isManager | kXR_attrSuper) : 0)
81 | ((conf->caps.virtual_redirector
82 || (conf->manager_map != NULL && conf->cms.addr == NULL))
83 ? (kXR_isManager | kXR_attrVirtRdr) : 0)
84 | (conf->caps.metadata_only ? kXR_attrMeta : 0)
85 | ((conf->proxy.enable > 0 || conf->proxy.upstreams != NULL)
86 ? kXR_attrProxy : 0)
87 | ((conf->cache_root.len > 0 || conf->cache_origin_host.len > 0)
88 ? kXR_attrCache : 0);
89}
90
91/* WHAT: the kXR_tls* requirement bits the brix_tls_require mask advertises.
92 * WHY: stock clients decide when to upgrade from the per-capability bits in
93 * the kXR_protocol reply (tlsSess/tlsData/tlsTPC/tlsLogin); advertising
94 * the enforced policy lets a conformant client upgrade pre-emptively
95 * instead of eating a kXR_TLSRequired refusal.
96 * HOW: pure BRIX_TLSREQ_* → kXR_tls* mapping; 0 mask → 0 (the flags word is
97 * byte-identical to the pre-feature advertisement). */
98static uint32_t
99protocol_tls_require_flags(const ngx_stream_brix_srv_conf_t *conf)
100{
101 ngx_uint_t mask = conf->common.tls_require;
102
103 return ((mask & BRIX_TLSREQ_LOGIN) ? kXR_tlsLogin : 0)
104 | ((mask & BRIX_TLSREQ_SESSION) ? kXR_tlsSess : 0)
105 | ((mask & BRIX_TLSREQ_DATA) ? kXR_tlsData : 0)
106 | ((mask & BRIX_TLSREQ_TPC) ? kXR_tlsTPC : 0);
107}
108
109/* WHAT: the full ServerProtocolBody flags word (host-order).
110 * WHY: the capability advertisement is the heart of kXR_protocol; keeping it
111 * one pure expression makes the negotiated bit-set reviewable at a glance.
112 * HOW: fixed server bits | role bits | per-conf feature bits | TLS bits. */
113static uint32_t
114protocol_negotiate_flags(const ngx_stream_brix_srv_conf_t *conf,
115 const protocol_want_t *want)
116{
117 return kXR_isServer
118 | kXR_suppgrw /* pgread/pgwrite always implemented */
119 | kXR_supposc /* POSC always implemented */
120 | protocol_role_flags(conf)
121 | (conf->caps.collapse_redir ? kXR_collapseRedir : 0)
122 | (conf->caps.recover_writes ? kXR_recoverWrts : 0)
123 | (want->offer_tls
124 ? (kXR_haveTLS | kXR_gotoTLS | kXR_tlsLogin
125 | protocol_tls_require_flags(conf))
126 : 0);
127}

129/* WHAT: write the 6-byte ServerResponseReqs_Protocol signing block at si.
130 * WHY: when the client sets kXR_secreqs the kXR_protocol reply carries this
131 * block *immediately* after the 8-byte ServerProtocolBody — the wire
132 * layout is ServerResponseBody_Protocol = { pval, flags, secreq }, so a
133 * conformant client reads theTag ('S') at that fixed offset and the
134 * seclvl right behind it. This implementation deliberately emits NO
135 * leading SecurityInfo header and NO binary SecurityProtocol entries:
136 * the list of offered auth protocols is advertised the standard way, as
137 * the "&P=gsi,...&P=ztn,..." string in the kXR_login response, never in
138 * the protocol reply. Prepending anything here shifts 'S' downfield and
139 * makes strict clients (go-hep, XrdRust) misread the block.
140 * HOW: emits theTag 'S', reserved/secver, secopt, the configured security
141 * level (seclvl=0 => no signing required), and secvsz=0 (no secvec). */
142static void
143protocol_write_sec_trailer(const ngx_stream_brix_srv_conf_t *conf, u_char *si)
144{
145 si[0] = 'S'; /* theTag */
146 si[1] = 0; /* rsvd */
147 si[2] = 0; /* secver (kXR_secver_0) */
148 si[3] = (conf->security_level >= 4) ? kXR_secOData : 0; /* secopt */
149 si[4] = (u_char) conf->security_level; /* seclvl */
150 si[5] = 0; /* secvsz=0: no secvec */
151}

154/* WHAT: short auth-mode label for the kXR_protocol debug log line.
155 * WHY: keeps the nine-way ternary chain out of the handler's log call.
156 * Debug-build only — its sole caller is ngx_log_debug4, which compiles
157 * out without NGX_DEBUG and would leave this unused under -Werror.
158 * HOW: pure priority chain over the offer flags — identical ordering to the
159 * historical inline expression so the logged string never changes. */
160static const char *
161protocol_auth_label(const protocol_want_t *want)
162{
163 return want->sss ? "sss" :
164 want->unx ? "unix" :
165 want->krb5 ? "krb5" :
166 want->host ? "host" :
167 want->pwd ? "pwd" :
168 want->gsi && want->token ? "both" :
169 want->gsi ? "gsi" :
170 want->token ? "token" : "none";
171}

174/* Handle kXR_protocol — negotiate server capabilities by building the fixed
175 * 8-byte ServerProtocolBody response (protocol version, flags, and the security
176 * requirement advertised to the client). */
177ngx_int_t
178brix_handle_protocol(brix_ctx_t *ctx, ngx_connection_t *c,
179 ngx_stream_brix_srv_conf_t *conf)
180{
181 ServerProtocolBody body;
182 u_char *buf;
183 size_t bodylen, total;
184 protocol_want_t want;
185
186 protocol_parse_client_req(ctx, c, conf, &want);
187
188 /* Reject if the client demands TLS but this listener has none configured. */
189 if (want.client_wants_tls && (!conf->tls || conf->tls_ctx == NULL)) {
190 return brix_send_error(ctx, c, kXR_TLSRequired,
191 "TLS required by client but not configured on this server");
192 }
193
194 /*
195 * Base kXR_protocol reply is the fixed 8-byte ServerProtocolBody.
196 * If the client advertised security negotiation support (kXR_secreqs),
197 * append the 6-byte ServerResponseReqs_Protocol signing block directly
198 * behind it — the wire layout is { pval, flags, secreq } with nothing in
199 * between.
200 */
201 bodylen = sizeof(body);
202 if (want.client_flags & kXR_secreqs) {
203 bodylen += 6; /* ServerResponseReqs_Protocol */
204 }
205
206 total = XRD_RESPONSE_HDR_LEN + bodylen;
207 BRIX_PALLOC_OR_RETURN(buf, c->pool, total, NGX_ERROR);
208
209 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok,
210 (uint32_t) bodylen,
211 (ServerResponseHdr *) buf);
212
213 body.pval = htonl(kXR_PROTOCOLVERSION);
214 body.flags = htonl(protocol_negotiate_flags(conf, &want));
215
216 /* Fixed 8-byte prefix every protocol reply starts with after the response header. */
217 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, &body, sizeof(body));
218
219 if (want.client_flags & kXR_secreqs) {
220 protocol_write_sec_trailer(conf,
221 buf + XRD_RESPONSE_HDR_LEN + sizeof(body));
222 }
223
224 ngx_log_debug4(NGX_LOG_DEBUG_STREAM, c->log, 0,
225 "brix: kXR_protocol ok (client_flags=0x%02xd "
226 "bodylen=%uz auth=%s tls=%d)",
227 (int) want.client_flags, bodylen,
228 protocol_auth_label(&want),
229 want.offer_tls);
230
231 if (want.offer_tls) {
232 ctx->tls_pending = 1;
233 }
234
235 return brix_queue_response(ctx, c, buf, total);
236}
