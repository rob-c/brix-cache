45/* Parse http(s)://host[:port][/...] — only scheme/host/port are used. */
46static int
47scan_parse_url(const char *url, scan_ep *ep)
48{
49 const char *p, *slash, *colon;
50 size_t hlen;
51
52 if (strncmp(url, "https://", 8) == 0) {
53 ep->tls = 1; p = url + 8; ep->port = 443;
54 } else if (strncmp(url, "http://", 7) == 0) {
55 ep->tls = 0; p = url + 7; ep->port = 80;
56 } else {
57 return -1;
58 }
59 slash = strchr(p, '/');
60 colon = strchr(p, ':');
61 if (colon != NULL && (slash == NULL || colon < slash)) {
62 hlen = (size_t) (colon - p);
63 ep->port = atoi(colon + 1);
64 } else {
65 hlen = slash ? (size_t) (slash - p) : strlen(p);
66 }
67 if (hlen == 0 || hlen >= sizeof(ep->host) || ep->port <= 0) {
68 return -1;
69 }
70 memcpy(ep->host, p, hlen);
71 ep->host[hlen] = '\0';
72 return 0;
73}

75/* POST /brix/login (password) → capture the session cookie. 0 / -1. */
76static int
77scan_login(const scan_ep *ep, const scan_args_t *sa,
78 char *cookie, size_t cksz, brix_status *st)
79{
80 brix_http_resp resp;
81 char body[256];
82 char sc[512];
83 int n, ok;
84
85 n = snprintf(body, sizeof(body), "password=%s", sa->password);
86 if (brix_http_req(ep->host, ep->port, ep->tls, "POST", "/brix/login",
87 "Content-Type: application/x-www-form-urlencoded\r\n",
88 body, (size_t) n, 15000, sa->insecure ? 0 : 1, NULL,
89 NULL, &resp, st) != 0)
90 {
91 return -1;
92 }
93 cookie[0] = '\0';
94 if (brix_http_header(&resp, "Set-Cookie", sc, sizeof(sc))) {
95 char *semi = strchr(sc, ';');
96 if (semi != NULL) {
97 *semi = '\0';
98 }
99 snprintf(cookie, cksz, "%s", sc);
100 }
101 ok = (resp.status == 200 || resp.status == 302) && cookie[0] != '\0';
102 brix_http_resp_free(&resp);
103 if (!ok) {
104 brix_status_set(st, XRDC_EAUTH, 0, "dashboard login failed (bad password?)");
105 return -1;
106 }
107 return 0;
108}

110/* Minimal JSON field extractor over one controlled NDJSON line: copies the value
111 * of "key" into out[outsz]. Strings are returned without surrounding quotes (no
112 * unescaping — values here are paths/hex/short tokens). 1 found, 0 absent. */
113static int
114scan_json_field(const char *line, const char *key, char *out, size_t outsz)
115{
116 char pat[48];
117 const char *p;
118 size_t o = 0;
119
120 snprintf(pat, sizeof(pat), "\"%s\":", key);
121 p = strstr(line, pat);
122 if (p == NULL) {
123 return 0;
124 }
125 p += strlen(pat);
126 if (*p == '"') {
127 p++;
128 while (*p != '\0' && *p != '"' && o + 1 < outsz) {
129 if (*p == '\\' && p[1] != '\0') {
130 p++; /* keep the escaped char verbatim */
131 }
132 out[o++] = *p++;
133 }
134 } else {
135 while (*p != '\0' && *p != ',' && *p != '}' && o + 1 < outsz) {
136 out[o++] = *p++;
137 }
138 }
139 out[o] = '\0';
140 return 1;
141}

143/*
144 * scan_render_summary — print the summary record; return its mismatch count.
145 * WHY: the summary line both renders AND yields the verify/compare exit
146 * signal, so it is the one record type with a return value.
147 * HOW: raw passthrough in --json/--summary mode, otherwise the frozen

150static long
151scan_render_summary(const char *buf, int as_json, int summary_only)
152{

291/* Render the NDJSON body: TSV (default), raw json, or summary-only. Returns the
292 * mismatch count seen in the summary (for the verify/compare exit code). */
293static long
294scan_render(const char *body, int as_json, int summary_only)
295{
296 const char *line = body;
297 long mismatch = 0;
298
299 while (line != NULL && *line != '\0') {
300 const char *nl = strchr(line, '\n');
301 size_t len = nl ? (size_t) (nl - line) : strlen(line);
302 char buf[4096];
303 char t[16];
304
305 if (len >= sizeof(buf)) {
306 len = sizeof(buf) - 1;
307 }
308 memcpy(buf, line, len);
309 buf[len] = '\0';
310
311 if (scan_json_field(buf, "t", t, sizeof(t))) {
312 if (strcmp(t, "summary") == 0) {
313 mismatch = scan_render_summary(buf, as_json, summary_only);
314 } else {
315 scan_render_row(buf, t, as_json, summary_only);
316 }
317 }
318 line = nl ? nl + 1 : NULL;
319 }
320 return mismatch;
321}

323/* URL-encode a query value (conservative: keep unreserved + '/'). */
324static void
325scan_qencode(const char *in, char *out, size_t outsz)
326{
327 static const char hex[] = "0123456789ABCDEF";
328 size_t o = 0;
329
330 for (; *in != '\0' && o + 4 < outsz; in++) {
331 unsigned char c = (unsigned char) *in;
332 if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')
333 || (c >= '0' && c <= '9') || c == '/' || c == '.' || c == '-'
334 || c == '_' || c == '~')
335 {
336 out[o++] = (char) c;
337 } else {
338 out[o++] = '%';
339 out[o++] = hex[c >> 4];
340 out[o++] = hex[c & 0xf];
341 }
342 }
343 out[o] = '\0';
344}

346/* Parse cmd_scan's flag ladder into `sa`. WHAT: --path/--algo/--password/
347 * --insecure/--json/--summary plus the single positional dashboard URL.
348 * WHY: cmd_scan's parse half is independent of the fetch/render halves; the
349 * decoded scan_args_t is what scan_login and the fetch consume.
350 * HOW: same first-match ladder as before; unknown dash-word or a second
351 * positional → -1 (caller emits usage). */
352static int
353scan_parse_scan_args(int argc, char **argv, scan_args_t *sa)
354{
355 int i;
356
357 for (i = 0; i < argc; i++) {
358 const char *a = argv[i];
359 if (strcmp(a, "--path") == 0 && i + 1 < argc) {
360 sa->path = argv[++i];
361 } else if (strcmp(a, "--algo") == 0 && i + 1 < argc) {
362 sa->alg = argv[++i];
363 } else if (strcmp(a, "--password") == 0 && i + 1 < argc) {
364 sa->password = argv[++i];
365 } else if (strcmp(a, "--insecure") == 0) {
366 sa->insecure = 1;
367 } else if (strcmp(a, "--json") == 0) {
368 sa->as_json = 1;
369 } else if (strcmp(a, "--summary") == 0) {
370 sa->summary_only = 1;
371 } else if (a[0] == '-') {
372 return -1;
373 } else if (sa->url == NULL) {
374 sa->url = a;
375 } else {
376 return -1;
377 }
378 }
379 return 0;
380}

382/* Run the authenticated GET against /brix/api/v1/scan. WHAT: builds the
383 * mode/path/alg query + Cookie header and fetches into *resp.
384 * WHY: the network half of cmd_scan, split from parse and render.
385 * HOW: query-encodes path/alg, formats the request, maps transport failure
386 * to the shell code and a non-200 to SX_ERROR with the same hints. */
387static int
388scan_fetch(const char *mode, const scan_ep *ep, const scan_args_t *sa,
389 const char *cookie, brix_http_resp *resp)
390{
391 char epath[2048], ealg[64], query[2240], hdr[640];
392 char fullpath[2304];
393 brix_status st;
394
395 brix_status_clear(&st);
396 scan_qencode(sa->path, epath, sizeof(epath));
397 scan_qencode(sa->alg, ealg, sizeof(ealg));
398 snprintf(query, sizeof(query), "mode=%s&path=%s&alg=%s", mode, epath, ealg);
399 snprintf(hdr, sizeof(hdr), "Cookie: %s\r\n", cookie);
400
401 snprintf(fullpath, sizeof(fullpath), "/brix/api/v1/scan?%s", query);
402 if (brix_http_req(ep->host, ep->port, ep->tls, "GET", fullpath, hdr,
403 NULL, 0, 120000, sa->insecure ? 0 : 1, NULL, NULL, resp, &st) != 0)
404 {
405 fprintf(stderr, "xrdstorascan: %s: %s\n", mode, st.msg);
406 return brix_shellcode(&st);
407 }
408 if (resp->status != 200) {
409 fprintf(stderr, "xrdstorascan: %s: server returned HTTP %d%s\n",
410 mode, resp->status,
411 resp->status == 404 ? " (scan disabled? — set brix_scan_root)"
412 : resp->status == 401 ? " (auth — check password)" : "");
413 brix_http_resp_free(resp);
414 return SX_ERROR;
415 }
416 return SX_OK;
417}

419int
420cmd_scan(const char *mode, int argc, char **argv, const char *prog)
421{
422 scan_args_t sa = { NULL, "/", "adler32", NULL, 0, 0, 0 };
423 scan_ep ep;
424 char cookie[512] = "";
425 brix_http_resp resp;
426 brix_status st;
427 long mismatch;
428 int rc;
429
430 sa.password = getenv("XRDSTORASCAN_PASSWORD");
431
432 /* --help as the first subcommand arg → print this mode's usage to stdout
433 * and exit cleanly (WS-2). */
434 if (argc >= 1 && strcmp(argv[0], "--help") == 0) {
435 printf("usage: %s %s <dashboard-url> [--path P] [--algo A]\n"
436 " [--password PW] [--insecure] [--json|--summary]\n"
437 " Server-side scan over the /brix/api/v1/scan admin endpoint.\n"
438 " auth via --password or $XRDSTORASCAN_PASSWORD\n",
439 prog, mode);
440 brix_usage_footer(stdout, prog);
441 return SX_OK;
442 }
443
444 if (scan_parse_scan_args(argc, argv, &sa) != 0 || sa.url == NULL) {
445 return usage(prog, SX_USAGE);
446 }
447 if (scan_parse_url(sa.url, &ep) != 0) {
448 fprintf(stderr, "xrdstorascan: %s needs an http(s):// dashboard URL\n", mode);
449 return SX_USAGE;
450 }
451 if (sa.password == NULL) {
452 fprintf(stderr, "xrdstorascan: %s needs --password or $XRDSTORASCAN_PASSWORD\n",
453 mode);
454 return SX_USAGE;
455 }
456
457 brix_status_clear(&st);
458 if (scan_login(&ep, &sa, cookie, sizeof(cookie), &st) != 0) {
459 fprintf(stderr, "xrdstorascan: %s\n", st.msg);
460 return brix_shellcode(&st);
461 }
462
463 rc = scan_fetch(mode, &ep, &sa, cookie, &resp);
464 if (rc != SX_OK) {
465 return rc;
466 }
467
468 mismatch = scan_render(resp.body ? resp.body : "", sa.as_json,
469 sa.summary_only);
470 brix_http_resp_free(&resp);
471
472 /* verify/compare: corruption found ⇒ non-zero for scripting */
473 if ((strcmp(mode, "verify") == 0 || strcmp(mode, "compare") == 0)
474 && mismatch > 0)
475 {
476 return SX_MISMATCH;
477 }
478 return SX_OK;
479}
