13/* pgw_fmt_detail()
14 * WHAT: Format the "<offset>+<len>" access-log/error detail string used by every
15 * pgwrite reply path.
16 * WHY: The parse, decode and completion paths must use identical detail bytes.
17 * HOW: 1. Format into the caller-owned buffer with a bounded snprintf. */
18void
19pgw_fmt_detail(char *buf, size_t bufsz, int64_t offset, size_t len)
20{
21 snprintf(buf, bufsz, "%lld+%zu", (long long) offset, len);

24/*
25 * pgwrite_handle_write_error — handle I/O error from pwrite()
26 * WHAT: Logs error access event, increments error metric, sends kXR_IOError response.
27 * WHY: Centralizes error handling for consistent logging and metrics.
28 * HOW: Formats detail string, logs via brix_log_access(), calls BRIX_OP_ERR(),
29 * sends error response via brix_send_error().
30 *
31 * Returns 1 (response sent).
32 */
33int
34pgwrite_handle_write_error(brix_ctx_t *ctx, ngx_connection_t *c, pgw_state_t *st,
35 const char *ioerr, ngx_int_t *rc)
36{
37 char write_detail[64];
38
39 pgw_fmt_detail(write_detail, sizeof(write_detail), st->offset, st->flat_sz);
40 brix_log_access(ctx, c, "WRITE", ctx->files[st->idx].path,
41 write_detail, 0, kXR_IOError, ioerr, 0);
42 BRIX_OP_ERR(ctx, BRIX_OP_WRITE);
43 *rc = brix_send_error(ctx, c, kXR_IOError, ioerr);
44 return 1;
45}

150/*
151 * pgwrite_send_reply — send CSE or status reply
152 * WHAT: Sends kXR_status reply — CSE retransmit list if bad pages, else status.
153 * WHY: Centralizes reply logic for clarity.
154 * HOW: Calls brix_send_pgwrite_cse() or brix_send_pgwrite_status().

158ngx_int_t
