1/*
2 * xrdceph_migrate_config.h — site-profile config file for the C++
3 * XrdCeph<->CephFS migration tools (shared by xrdceph_striper_migrate.cpp
4 * and xrdceph_cephfs_to_striper.cpp; format identical to the Python tools'
5 * pymigrate.common.load_tool_config).
6 *
7 * WHAT: A tiny header-only parser for the flat `key = value` profile passed
8 * via --config (or $XRDCEPH_MIGRATE_CONF): '#' comments (inline
9 * allowed), blank lines ignored, whitespace trimmed, empty value =
10 * unset. The key set is CLOSED — an unknown key is a hard error,
11 * because a typo'd pool name in a tool that can delete data must fail
12 * loudly, not silently fall back to a default.
13 *
14 * WHY: Operators define pools + connection identity once per site instead
15 * of retyping them per invocation; it also makes the client id and
16 * CephFS fs name configurable (both were hardcoded before).
17 *
18 * HOW: Header-only so the tools' single-file `g++ tool.cpp -l...` builds
19 * keep working. Precedence is applied by the caller:
20 * explicit CLI > config file > built-in default (see
21 * xrdceph_migrate_cfg_resolve).
22 *
23 * Besides the profile parser this also carries the CLI plumbing the two
24 * tools share: the argv walk (xrdceph_migrate_cli_walk), the common value
25 * options (xrdceph_migrate_cli_common), the positional-arity gate
26 * (xrdceph_migrate_cli_arity) and the resolution tail / driver
27 * (xrdceph_migrate_cfg_finish / xrdceph_migrate_cfg_resolve_with).
28 *
29 * Recognised keys:
30 * striper_pool meta_pool data_pool conf client fs_name dest_prefix strip
31 */

49inline std::string
50trim(const std::string &s)
51{
52 size_t b = s.find_first_not_of(" \t\r\n");
53 if (b == std::string::npos) { return ""; }
54 size_t e = s.find_last_not_of(" \t\r\n");
55 return s.substr(b, e - b + 1);
56}

70/* Parse `path` into *cfg. Returns true on success; on failure prints a
71 * one-line reason (with file:line) to stderr and returns false. */
72inline bool
73xrdceph_migrate_cfg_load(const std::string &path, xrdceph_migrate_cfg *cfg)
74{
75 using xrdceph_migrate_cfg_detail::trim;
76 using xrdceph_migrate_cfg_detail::known_key;
77
78 std::ifstream f(path);
79 if (!f) {
80 fprintf(stderr, "--config: cannot open %s\n", path.c_str());
81 return false;
82 }
83 std::string raw;
84 int lineno = 0;
85 while (std::getline(f, raw)) {
86 lineno++;
87 std::string line = raw;
88 size_t hash = line.find('#');
89 if (hash != std::string::npos) { line.erase(hash); }
90 line = trim(line);
91 if (line.empty()) { continue; }
92
93 size_t eq = line.find('=');
94 if (eq == std::string::npos) {
95 fprintf(stderr, "--config: %s:%d: expected 'key = value', "
96 "got \"%s\"\n", path.c_str(), lineno, raw.c_str());
97 return false;
98 }
99 std::string key = trim(line.substr(0, eq));
100 std::string val = trim(line.substr(eq + 1));
101 if (!known_key(key)) {
102 fprintf(stderr, "--config: %s:%d: unknown config key '%s' "
103 "(known: striper_pool, meta_pool, data_pool, conf, "
104 "client, fs_name, dest_prefix, strip)\n",
105 path.c_str(), lineno, key.c_str());
106 return false;
107 }
108 if (!val.empty()) { cfg->kv[key] = val; }
109 }
110 return true;
111}

113/* Precedence: explicit CLI (non-empty) > config file > built-in default. */
114inline std::string
115xrdceph_migrate_cfg_resolve(const std::string &cli_value,
116 const xrdceph_migrate_cfg &cfg,
117 const std::string &key,
118 const std::string &dflt = "")
119{
120 if (!cli_value.empty()) { return cli_value; }
121 auto it = cfg.kv.find(key);
122 if (it != cfg.kv.end()) { return it->second; }
123 return dflt;
124}

126/* The value options every migration tool takes: --strip --conf --config
127 * --threads. Returns 1 when `a` was consumed (advancing *i past the value). */
128inline int
129xrdceph_migrate_cli_common(const std::string &a, int *i, int argc, char **argv,
130 std::string *strip, std::string *conf,
131 std::string *config, int *threads)
132{
133 const std::pair<const char *, std::string *> sopts[] = {
134 { "--strip", strip }, { "--conf", conf }, { "--config", config },
135 };
136 for (auto &o : sopts) {
137 if (a == o.first && *i + 1 < argc) { *o.second = argv[++(*i)]; return 1; }
138 }
139 if (a == "--threads" && *i + 1 < argc) { *threads = atoi(argv[++(*i)]); return 1; }
140 return 0;
141}

143/* Shared argv walk: the tool's value options first (1 = consumed, -1 = hard
144 * error), then its boolean flags, then positionals; an unknown --option is
145 * refused. Returns 0 or the exit code. */
146inline int
147xrdceph_migrate_cli_walk(int argc, char **argv,
148 int (*value_opt)(const std::string &, int *, int, char **),
149 int (*flag_opt)(const std::string &),
150 std::vector<std::string> *pos)
151{
152 for (int i = 1; i < argc; i++) {
153 std::string a = argv[i];
154 int rc = value_opt(a, &i, argc, argv);
155 if (rc < 0) { return 2; }
156 if (rc > 0) { continue; }
157 if (flag_opt(a)) { continue; }
158 if (a.rfind("--", 0) == 0) {
159 fprintf(stderr, "unknown option %s\n", a.c_str());
160 return 2;
161 }
162 pos->push_back(a);
163 }
164 return 0;
165}

167/* Positional-arity gate: the tools take all three positionals or NONE (a
168 * partial mix with --config is ambiguous and refused). */
169inline int
170xrdceph_migrate_cli_arity(const std::vector<std::string> &pos, const char *prog,
171 const char *usage)
172{
173 if (pos.size() != 3 && pos.size() != 0) {
174 fprintf(stderr, "usage: %s %s [opts]\n"
175 " (give all three positionals, or none with --config)\n",
176 prog, usage);
177 return 2;
178 }
179 return 0;
180}

182/* Shared resolution tail: every required key must have resolved non-empty,
183 * then the ceph.conf fallback chain (--conf > profile `conf` > $CEPH_CONF >
184 * /etc/ceph/ceph.conf) and the worker-count clamp. Returns 0 or exit code. */
185inline int
186xrdceph_migrate_cfg_finish(const std::pair<const char *, std::string *> *req,
187 size_t nreq, const xrdceph_migrate_cfg &cfg,
188 std::string *conf, int *threads)
189{
190 for (size_t k = 0; k < nreq; k++) {
191 if (req[k].second->empty()) {
192 fprintf(stderr, "missing %s: pass positionals or set it in --config\n",
193 req[k].first);
194 return 2;
195 }
196 }
197 if (conf->empty()) { *conf = xrdceph_migrate_cfg_resolve("", cfg, "conf"); }
198 if (conf->empty()) { *conf = getenv("CEPH_CONF") ? getenv("CEPH_CONF")
199 : "/etc/ceph/ceph.conf"; }
200 if (*threads < 1) { *threads = 1; }
201 return 0;
202}

204/* Shared resolution driver: pick up $XRDCEPH_MIGRATE_CONF when --config was
205 * not given, load the profile, then run the tool's own resolve step on it. */
206inline int
207xrdceph_migrate_cfg_resolve_with(const std::vector<std::string> &pos,
208 const char *prog, std::string *config,
209 int (*resolve_fn)(const std::vector<std::string> &,
210 const char *,
211 const xrdceph_migrate_cfg &))
212{
213 if (config->empty() && getenv("XRDCEPH_MIGRATE_CONF") != NULL) {
214 *config = getenv("XRDCEPH_MIGRATE_CONF");
215 }
216 xrdceph_migrate_cfg cfg;
217 if (!config->empty() && !xrdceph_migrate_cfg_load(*config, &cfg)) { return 2; }
218 return resolve_fn(pos, prog, cfg);
219}
