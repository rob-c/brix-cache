1/*
2 * sss_id.c — per-connection SSS identity registry (the XrdSecsssID role).
3 *
4 * WHAT: Implements sss_id.h: bounded setters for one SSS entity, a file reader
5 * for a proxied credential, and a mutex-guarded lid → identity map.
6 * WHY: See sss_id.h. The security property this file exists to hold is that a
7 * lookup miss is an ERROR, never a substitution: a front end that cannot
8 * find the caller's identity must fail the request rather than mint one
9 * under whatever identity happens to be at hand.
10 * HOW: A dynamic array of fixed-size slots, doubled on growth and capped at
11 * XRDC_SSS_ID_SLOTS_MAX. Every mutator takes the registry mutex; find()
12 * copies the slot out under the same lock so a concurrent unregister
13 * cannot free the bytes a caller is still minting from. Slots are wiped
14 * with an explicit_bzero-equivalent memset through a volatile pointer on
15 * removal and teardown so a proxied credential does not outlive its use.
16 */

40/*
41 * WHAT: Overwrite `len` bytes at `p` with zeroes in a way the compiler may not
42 * elide.
43 * WHY: A slot holds a proxied credential — bearer material. A plain memset on
44 * memory that is about to be freed is a textbook dead-store elimination
45 * candidate, so the wipe has to go through a volatile pointer.
46 * HOW: Byte loop through a `volatile unsigned char *` (the portable idiom;
47 * explicit_bzero is not available everywhere the client builds).
48 */
49static void
50sss_id_wipe(void *p, size_t len)
51{
52 volatile unsigned char *v = (volatile unsigned char *) p;
53
54 while (len-- > 0) {
55 *v++ = 0;
56 }
57}

60/*
61 * WHAT: Copy `src` into dst[cap], failing when it does not fit.
62 * WHY: Truncating an identity field silently is the bug class this whole
63 * feature is meant to avoid: a truncated VO or role is a DIFFERENT
64 * authorization claim, not a shorter one. The server's parser fails the
65 * same way on an over-cap field, so the two ends agree.
66 * HOW: NULL/empty leaves dst empty and succeeds (the field is simply absent);
67 * otherwise strlen must be < cap. Returns 0 / -1.
68 */
69static int
70sss_id_set_field(char *dst, size_t cap, const char *src)
71{
72 size_t n;
73
74 if (src == NULL || src[0] == '\0') {
75 dst[0] = '\0';
76 return 0;
77 }
78 n = strlen(src);
79 if (n >= cap) {
80 return -1;
81 }
82 memcpy(dst, src, n + 1);
83 return 0;
84}

159int
160brix_sss_creds_read_file(const char *path, uint8_t *out, size_t out_max,
161 size_t *out_len)
162{
163 struct stat sb;
164 size_t got = 0;
165 int fd;
166
167 if (path == NULL || path[0] == '\0' || out == NULL || out_len == NULL) {
168 return -1;
169 }
170 *out_len = 0;
171
172 fd = open(path, O_RDONLY | O_NOFOLLOW | O_CLOEXEC);
173 if (fd < 0) {
174 return -1;
175 }
176 if (fstat(fd, &sb) != 0 || !S_ISREG(sb.st_mode)
177 || sb.st_size <= 0 || (size_t) sb.st_size > out_max)
178 {
179 close(fd);
180 return -1;
181 }
182
183 while (got < (size_t) sb.st_size) {
184 ssize_t r = read(fd, out + got, (size_t) sb.st_size - got);
185 if (r > 0) {
186 got += (size_t) r;
187 continue;
188 }
189 if (r < 0 && errno == EINTR) {
190 continue;
191 }
192 break;
193 }
194 close(fd);
195
196 if (got != (size_t) sb.st_size) {
197 sss_id_wipe(out, out_max);
198 return -1;
199 }
200 *out_len = got;
201 return 0;
202}

245static sss_id_slot *
246sss_id_slot_find(brix_sss_id_registry *reg, const char *lid)
247{
248 size_t i;
249
250 for (i = 0; i < reg->n; i++) {
251 if (strcmp(reg->slots[i].lid, lid) == 0) {
252 return &reg->slots[i];
253 }
254 }
255 return NULL;

259/* Grow the slot array by doubling (from 8), honouring the hard cap. 0 / -1. */
260static int
261sss_id_grow(brix_sss_id_registry *reg)
262{
263 size_t want = (reg->cap == 0) ? 8 : reg->cap * 2;
264 sss_id_slot *next;
265
266 if (reg->n < reg->cap) {
267 return 0;
268 }
269 if (reg->n >= XRDC_SSS_ID_SLOTS_MAX) {
270 return -1;
271 }
272 if (want > XRDC_SSS_ID_SLOTS_MAX) {
273 want = XRDC_SSS_ID_SLOTS_MAX;
274 }
275 next = calloc(want, sizeof(*next));
276 if (next == NULL) {
277 return -1;
278 }
279 if (reg->slots != NULL) {
280 memcpy(next, reg->slots, reg->n * sizeof(*next));
281 sss_id_wipe(reg->slots, reg->cap * sizeof(*next));
282 free(reg->slots);
283 }
284 reg->slots = next;
285 reg->cap = want;
286 return 0;
287}

290int
291brix_sss_id_register(brix_sss_id_registry *reg, const char *lid,
292 const brix_sss_ident *id)
293{
294 const char *key = sss_id_key(lid);
295 sss_id_slot *slot;
296 int rc = -1;
297
298 if (reg == NULL || id == NULL || strlen(key) >= XRDC_SSS_LID_MAX) {
299 return -1;
300 }
301
302 pthread_mutex_lock(&reg->mu);
303 slot = sss_id_slot_find(reg, key);
304 if (slot == NULL && sss_id_grow(reg) == 0) {
305 slot = &reg->slots[reg->n];
306 memcpy(slot->lid, key, strlen(key) + 1);
307 reg->n++;
308 }
309 if (slot != NULL) {
310 slot->id = *id;
311 rc = 0;
312 }
313 pthread_mutex_unlock(&reg->mu);
314 return rc;
315}

320{
321 const char *key = sss_id_key(lid);
322 sss_id_slot *slot;
323 int rc = -1;
324
325 if (reg == NULL) {
326 return -1;
327 }
328
329 pthread_mutex_lock(&reg->mu);
330 slot = sss_id_slot_find(reg, key);
331 if (slot != NULL) {
332 sss_id_wipe(slot, sizeof(*slot));
333 *slot = reg->slots[reg->n - 1];
334 sss_id_wipe(&reg->slots[reg->n - 1], sizeof(*slot));
335 reg->n--;
336 rc = 0;
337 }
338 pthread_mutex_unlock(&reg->mu);
339 return rc;
