28/* extended attributes (opt-in via --xattr) *//* FUSE uses "user.<x>" names; the server stores them under its own "user.U."
29 * prefix, so we send the bare "<x>" and the module re-prefixes. Only the user.*
30 * namespace is exposed. Returns "<x>" or NULL for a non-user.* name. */
31static const char *
32lg_xfs_xattr_to_fattr(const char *name)
33{
34 if (strncmp(name, "user.", 5) == 0 && name[5] != '\0') {
35 return name + 5;
36 }
37 return NULL;
38}

40int
41lg_xfs_getxattr(const char *path, const char *name, char *value, size_t size)
42{
43 brix_status st;
44 brix_conn *c;
45 int rc;
46 size_t vlen = 0;
47 const char *fname;
48
49 if (!lg_xattr) {
50 return -ENOTSUP;
51 }
52 brix_status_clear(&st);
53
54 /* Virtual read-only checksum xattr: user.XrdCks.<algo> → kXR_Qcksum hex. */
55 if (strncmp(name, XFS_CKS_XATTR_PFX, sizeof(XFS_CKS_XATTR_PFX) - 1) == 0) {
56 const char *algo = name + sizeof(XFS_CKS_XATTR_PFX) - 1;
57 char hex[160];
58
59 if (algo[0] == '\0') {
60 return -ENODATA;
61 }
62 c = brix_pool_checkout(lg_pool, &st);
63 if (c == NULL) {
64 return lg_xfs_err(&st);
65 }
66 rc = brix_query_cksum(c, path, algo, hex, sizeof(hex), &st);
67 brix_pool_checkin(lg_pool, c, rc == 0 ? 1 : lg_xfs_conn_healthy(&st));
68 if (rc != 0) {
69 return lg_xfs_err(&st);
70 }
71 vlen = strlen(hex);
72 if (size == 0) {
73 return (int) vlen;
74 }
75 if (vlen > size) {
76 return -ERANGE;
77 }
78 memcpy(value, hex, vlen);
79 return (int) vlen;
80 }
81
82 fname = lg_xfs_xattr_to_fattr(name);
83 if (fname == NULL) {
84 return -ENODATA;
85 }
86 c = brix_pool_checkout(lg_pool, &st);
87 if (c == NULL) {
88 return lg_xfs_err(&st);
89 }
90 rc = brix_fattr_get(c, path, fname, value, size, &vlen, &st);
91 brix_pool_checkin(lg_pool, c, rc == 0 ? 1 : lg_xfs_conn_healthy(&st));
92 if (rc != 0) {
93 return lg_xfs_err(&st);
94 }
95 if (size == 0) {
96 return (int) vlen;
97 }
98 if (vlen > size) {
99 return -ERANGE;
100 }
101 return (int) vlen;
102}

104/* Shared entry for the two xattr mutators: verify xattr support, reject the
105 * read-only checksum prefix, resolve NAME → backend fattr (absent → *err =
106 * absent_errno), then check a conn out of the pool. Returns the conn with
107 * *fattr set, or NULL with *err holding the FUSE error to return. */
108static brix_conn *
109lg_xfs_xattr_begin(const char *name, int absent_errno, const char **fattr,
110 brix_status *st, int *err)
111{
112 brix_conn *c;
113
114 if (!lg_xattr) {
115 *err = -ENOTSUP;
116 return NULL;
117 }
118 if (strncmp(name, XFS_CKS_XATTR_PFX, sizeof(XFS_CKS_XATTR_PFX) - 1) == 0) {
119 *err = -EACCES; /* checksum xattr is read-only */
120 return NULL;
121 }
122 *fattr = lg_xfs_xattr_to_fattr(name);
123 if (*fattr == NULL) {
124 *err = absent_errno;
125 return NULL;
126 }
127 brix_status_clear(st);
128 c = brix_pool_checkout(lg_pool, st);
129 if (c == NULL) {
130 *err = lg_xfs_err(st);
131 }
132 return c;
133}

139{
140 brix_pool_checkin(lg_pool, c, rc == 0 ? 1 : lg_xfs_conn_healthy(st));

230/* WHAT: handle a no-value dash-option (a bare flag), updating the driver's
231 * file-scope option globals in place.
232 * WHY: splitting the flag group from the value group (mirrors the async driver's
233 * aio_opt_novalue) keeps each option handler under the complexity gate; the
234 * option state already lives in file-scope globals, so this maps onto them
235 * directly (no new state introduced).
236 * HOW: returns 1 if `a` was a recognised flag, 0 otherwise. */
237static int
238lg_xfs_opt_novalue(const char *a)
239{
240 if (strcmp(a, "--tls") == 0) { lg_opts.want_tls = 1; return 1; }
241 if (strcmp(a, "--notlsok") == 0) { lg_opts.notlsok = 1; return 1; }
242 if (strcmp(a, "--noverifyhost") == 0) { lg_opts.verify_host = 0; return 1; }
243 if (strcmp(a, "--kernel-cache") == 0) { lg_kernel_cache = 1; return 1; }
244 if (strcmp(a, "--xattr") == 0) { lg_xattr = 1; return 1; }
245 return 0;
246}

248/* WHAT: handle a value-taking dash-option `a` with its already-extracted value
249 * word `v`, updating the driver's file-scope option globals in place.
250 * WHY: the value group split from the flag group (mirrors aio_opt_conn_value /
251 * aio_opt_cache_value) keeps this handler under the complexity gate.
252 * HOW: the caller only calls this when a next word exists, so `v` is always
253 * valid (a trailing value option therefore falls through to libfuse exactly
254 * as before). Returns 1 if `a` was a recognised value option, 0 otherwise. */
255static int
256lg_xfs_opt_value(const char *a, char *v)
257{
258 if (strcmp(a, "--auth") == 0) { lg_opts.auth_force = v; return 1; }
259 if (strcmp(a, "--max-conns") == 0) {
260 lg_max_conns = atoi(v);
261 if (lg_max_conns < 1) { lg_max_conns = 1; }
262 return 1;
263 }
264 if (strcmp(a, "--attr-timeout") == 0) { lg_attr_timeout = atof(v); return 1; }
265 if (strcmp(a, "--entry-timeout") == 0) { lg_entry_timeout = atof(v); return 1; }
266 if (strcmp(a, "--readahead") == 0) {
267 long n = atol(v);
268 lg_readahead = (n > 0) ? (size_t) n : 0;
269 return 1;
270 }
271 if (strcmp(a, "--writeback") == 0) {
272 long n = atol(v);
273 lg_writeback = (n > 0) ? (size_t) n : 0;
274 return 1;
275 }
276 if (strcmp(a, "--connect-timeout") == 0) { brix_tmo_set_connect_ms(atoi(v)); return 1; }
277 if (strcmp(a, "--io-timeout") == 0) { brix_tmo_set_io_ms(atoi(v)); return 1; }
278 return 0;
279}

281/* WHAT: split the command line into our options, the endpoint, and the libfuse
282 * passthrough vector.
283 * WHY: our known options are honored ANYWHERE on the line (before OR after the
284 * endpoint); unknown dash-args fall through to libfuse (so -f/-d/-s/-o still
285 * work); the first bare word is the endpoint, the next the mountpoint.
286 * HOW: each dash-arg is offered to the flag group then (when a next word exists)
287 * the value group, matching the original inline argc guards; -h/--help print
288 * usage and stop. Returns -1 to proceed with the mount, or a process exit
289 * code (0 for -h/--help encountered inline). */
290static int
291lg_xfs_parse_args(int argc, char **argv, char **fuse_argv, int *fuse_argc,
292 const char **endpoint)
293{
294 int i;
295
296 for (i = 1; i < argc; i++) {
297 char *a = argv[i];
298
299 if (a[0] == '-') {
300 if (lg_xfs_opt_novalue(a)) { continue; }
301 if (i + 1 < argc && lg_xfs_opt_value(a, argv[i + 1])) { i++; continue; }
302 if (strcmp(a, "-h") == 0 || strcmp(a, "--help") == 0) {
303 lg_usage();
304 return 0;
305 }
306 }
307 xfs_arg_passthrough(a, a[0] == '-', fuse_argv, fuse_argc, endpoint);
308 }
309 return -1;
310}

312/* WHAT: parse the endpoint, bring up the metadata connection pool, run the FUSE
313 * mount, and tear the pool down afterwards.
314 * WHY: isolates the mount bring-up + teardown ordering from the entry point so
315 * main() stays a short arg-parse → mount sequence.
316 * HOW: the pool connects one conn eagerly, so a bad endpoint/auth fails here
317 * (before fuse_main); the pool is destroyed after fuse_main returns.
318 * Returns the process exit code. */
319static int
320lg_xfs_root_mount(int fuse_argc, char **fuse_argv, const char *endpoint)
321{
322 brix_status st;
323 int rc;
324
325 brix_status_clear(&st);
326 if (brix_endpoint_parse(endpoint, &lg_url, &st) != 0) {
327 fprintf(stderr, "xrootdfs: %s\n", st.msg);
328 return 2;
329 }
330 /* The pool connects one conn eagerly, so a bad endpoint/auth fails here. */
331 lg_pool = brix_pool_create(&lg_url, &lg_opts, lg_max_conns, &st);
332 if (lg_pool == NULL) {
333 fprintf(stderr, "xrootdfs: connect %s:%d: %s\n",
334 lg_url.host, lg_url.port, st.msg);
335 return brix_shellcode(&st);
336 }
337 fprintf(stderr, "xrootdfs: mounted %s:%d (pool=%d, multi-threaded)\n",
338 lg_url.host, lg_url.port, lg_max_conns);
339
340 rc = fuse_main(fuse_argc, fuse_argv, &lg_xfs_ops, NULL);
341
342 brix_pool_destroy(lg_pool);
343 return rc;
344}

346/* Entry point for the synchronous fallback driver. Invoked by the unified
347 * xrootdfs front-end (apps/xrootdfs_main.c) when --legacy is given. */
348int
349xrootdfs_legacy_main(int argc, char **argv)
350{
351 const char *endpoint = NULL;
352 char *fuse_argv[64];
353 int fuse_argc = 0;
354 int rc;
355
356 if (argc < 3) {
357 lg_usage();
358 return 2;
359 }
360 memset(&lg_opts, 0, sizeof(lg_opts));
361 lg_opts.verify_host = 1;
362 brix_crypto_init();
363
364 fuse_argv[fuse_argc++] = argv[0];
365 /* This entry point only runs in --legacy mode (the binary is `xrootdfs`), so
366 * tag the kernel mount subtype explicitly: mounts then show as
367 * fuse.xrootdfs_legacy and `xrd mount` can still tell the two apart. */
368 fuse_argv[fuse_argc++] = (char *) "-osubtype=xrootdfs_legacy";
369
370 rc = lg_xfs_parse_args(argc, argv, fuse_argv, &fuse_argc, &endpoint);
371 if (rc >= 0) {
372 return rc;
373 }
374
375 if (endpoint == NULL || fuse_argc < 2) {
376 lg_usage();
377 return 2;
378 }
379 fuse_argv[fuse_argc] = NULL;
380
381 return lg_xfs_root_mount(fuse_argc, fuse_argv, endpoint);
382}
