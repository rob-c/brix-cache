1/*
2 * router.c — table-driven CMS opcode routing. See router.h.
3 *
4 * The tables mirror XrdCmsRouting.cc: the manager table merges the redirector
5 * routing (client-facing forwardable ops, initRDRrouting) with the server group
6 * (node->manager status frames, initRouter); the node table is the set of ops a
7 * data server executes when forwarded down from its manager (leaf — no FORWARD
8 * flag). Phase-61 W7 adds the two upward-leg valid-ops tables for explicit
9 * multi-tier roles: SUBMAN (manVOps — what a sub-manager accepts from its
10 * meta-manager) and SUPER (supVOps — what a supervisor accepts from its
11 * manager, namespace ops FORWARDed down to its own nodes).
12 */

54/*
55 * The routing matrix: one row per opcode, one column per role leg —
56 * manager initRDRrouting (forwardable client ops) + initRouter server
57 * group (node -> manager status/heartbeat frames)
58 * node initRouter leaf: ops a data server executes when forwarded down
59 * subman initMANrouting/manVOps (Phase-61 W7): a meta-manager may only ask
60 * for non-destructive service — stock cmsd "prohibit[s] a
61 * meta-manager from requesting potentially destructive actions"
62 * super initSUProuting/supVOps (Phase-61 W7): namespace mutations carry
63 * FORWARD — a supervisor pushes them DOWN to its own data nodes
64 * 0 = that role does not accept the opcode (lookup returns NULL: caller logs
65 * and drops, matching cmsd tolerance).
66 */

110static const brix_cms_route_t *
111route_scan(const brix_cms_route_t *tbl, size_t n, unsigned char code)
112{
113 size_t i;
114 for (i = 0; i < n; i++) {
115 if (tbl[i].code == code) {
116 return (tbl[i].flags != 0) ? &tbl[i] : NULL;
117 }
118 }
119 return NULL;
120}

122const brix_cms_route_t *
123brix_cms_route_lookup(brix_cms_role_t role, unsigned char code)
124{
125 switch (role) {
126 case XRDCMS_ROLE_NODE:
127 return route_scan(node_routes,
128 sizeof(node_routes) / sizeof(node_routes[0]), code);
129 case XRDCMS_ROLE_SUBMAN:
130 return route_scan(subman_routes,
131 sizeof(subman_routes) / sizeof(subman_routes[0]),
132 code);
133 case XRDCMS_ROLE_SUPER:
134 return route_scan(super_routes,
135 sizeof(super_routes) / sizeof(super_routes[0]),
136 code);
137 default:
138 return route_scan(manager_routes,
139 sizeof(manager_routes) / sizeof(manager_routes[0]),
140 code);
141 }
