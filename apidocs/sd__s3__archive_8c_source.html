1/*
2 * sd_s3_archive.c — the S3 archive (GLACIER / DEEP_ARCHIVE) primitives: read an
3 * object's archive state, and ask the store to restore it.
4 *
5 * WHAT: sd_s3_archive_state() — one signed HEAD returning the three response
6 * headers that between them say whether an object's bytes are readable
7 * right now; sd_s3_restore() — the RestoreObject POST that starts a
8 * temporary copy coming back to the online tier.
9 *
10 * WHY: These are the two halves the nearline SD slots (residency / recall) need
11 * from an object store, and they are the only S3 verbs in the whole client
12 * that concern data AVAILABILITY rather than data. Keeping them out of
13 * sd_s3_meta.c (which is about an object's user metadata) and out of
14 * sd_s3_write.c (which is about putting bytes) leaves each file with one
15 * subject — and, less tidily but more importantly, both of these are
16 * failure-tolerant advisory operations, which is a different contract from
17 * everything either neighbour does.
18 *
19 * HOW: Both are ordinary SigV4 requests. The state read shares the metadata
20 * path's HEAD leg (sd_s3_head_send) so three headers cost ONE round trip
21 * and the status verdict is not restated. The restore is a POST to
22 * `?restore` with an XML RestoreRequest body, signed through the extended
23 * signer for the same reason CreateMPU is: the canonical query string is
24 * part of the signature.
25 *
26 * NOTE ON `errno`: sd_s3_status_err maps the HTTP status onto errno for the
27 * caller, and every path here preserves it across the resp_free cleanup.
28 * The nearline slots decide policy from errno alone, so losing it to a
29 * cleanup call would turn "the object is not there" into "something went
30 * wrong".
31 */
32

39/* The restored-copy lifetime the driver asks for when the operator configured
40 * none. One day is the smallest value AWS accepts and the right default for a
41 * read-through cache: the cache tier owns the copy's usefulness from the moment
42 * the recall lands, so paying for archive-tier storage beyond that is the
43 * operator's deliberate choice, not ours. */
44#define SD_S3_RESTORE_DAYS_DEFAULT 1
45
46/* An x-amz-restore value is short and fixed-shape ("ongoing-request=\"false\",
47 * expiry-date=\"Fri, 01 Jan 2027 00:00:00 GMT\""); a storage class is one token.
48 * Anything longer is not something this build can classify, and the buffers
49 * below are the caller's, so this only bounds what we ask the transport for. */

51int
52sd_s3_archive_state(sd_s3_file *f, const sd_s3_archive_buf_t *out,
53 char *errbuf, size_t errcap)
54{
55 brix_s3_resp_t resp;
56 char auth[SD_S3_AUTH_HDRS_CAP];
57 int e;
58
59 if (f == NULL || out == NULL || out->storage_class == NULL
60 || out->restore == NULL || out->archive_status == NULL)
61 {
62 sd_s3_set_err(errbuf, errcap, "s3 archive-state: bad parameters");
63 return -1;
64 }
65 out->storage_class->buf[0] = '\0';
66 out->restore->buf[0] = '\0';
67 out->archive_status->buf[0] = '\0';
68
69 if (sd_s3_sign(f, "HEAD", "", auth, sizeof(auth)) != 0) {
70 sd_s3_set_err(errbuf, errcap, "s3 HEAD: SigV4 sign failed on %s",
71 f->key);
72 return -1;
73 }
74 if (sd_s3_head_send(f, auth, &resp, errbuf, errcap) != 0) {
75 return -1;
76 }
77 /* Each header is optional: a STANDARD object carries none of the three, and
78 * an absent header is an ANSWER here ("not archived"), never a failure. The
79 * buffers were emptied above, so a missed lookup leaves "" behind. */
80 (void) f->transport->resp_header(&resp, "x-amz-storage-class",
81 out->storage_class->buf,
82 out->storage_class->cap);
83 (void) f->transport->resp_header(&resp, "x-amz-restore",
84 out->restore->buf, out->restore->cap);
85 (void) f->transport->resp_header(&resp, "x-amz-archive-status",
86 out->archive_status->buf,
87 out->archive_status->cap);
88 e = errno;
89 f->transport->resp_free(&resp);
90 errno = e;
91 return 0;
92}

94int
95sd_s3_restore(const sd_s3_open_params *p, int days, char *errbuf, size_t errcap)
96{
97 sd_s3_file *f;
98 const sd_s3_sign_req_t req = { "POST", "restore=", NULL, 0 };
99 brix_s3_resp_t resp;
100 char wire[SD_S3_KEY_MAX + 16];
101 char auth[SD_S3_AUTH_HDRS_CAP];
102 char body[256];
103 int blen, pn, rc = 0, e;
104
105 if (p == NULL) {
106 sd_s3_set_err(errbuf, errcap, "s3 restore: bad parameters");
107 errno = EINVAL;
108 return -1;
109 }
110 if (days <= 0) {
111 days = SD_S3_RESTORE_DAYS_DEFAULT;
112 }
113 /* A read handle is how this client addresses an object; opening one signs
114 * nothing and issues no request, so this costs a struct, not a round trip. */
115 f = sd_s3_open_read(p, errbuf, errcap);
116 if (f == NULL) {
117 return -1;
118 }
119 pn = snprintf(wire, sizeof(wire), "%s?restore", f->key);
120 if (pn < 0 || (size_t) pn >= sizeof(wire)) {
121 sd_s3_set_err(errbuf, errcap, "s3 restore: key path too long");
122 sd_s3_close(f);
123 errno = ENAMETOOLONG;
124 return -1;
125 }
126 /* Tier Standard, not Expedited: Expedited is a paid capacity reservation
127 * that AWS can refuse outright with InsufficientCapacityError, and a recall
128 * this layer starts is a cache fill, not a user-visible latency promise. */
129 blen = snprintf(body, sizeof(body),
130 "<RestoreRequest><Days>%d</Days>"
131 "<GlacierJobParameters><Tier>Standard</Tier>"
132 "</GlacierJobParameters></RestoreRequest>", days);
133 if (sd_s3_sign_ext(f, &req, auth, sizeof(auth)) != 0) {
134 sd_s3_set_err(errbuf, errcap, "s3 RestoreObject: sign failed on %s",
135 f->key);
136 sd_s3_close(f);
137 return -1;
138 }
139 if (f->transport->request(f->tctx, f->host, f->port, f->tls, "POST", wire,
140 auth, body, (size_t) blen, f->timeout_ms, &resp,
141 errbuf, errcap) != 0)
142 {
143 sd_s3_close(f);
144 return -1;
145 }
146 /* 202 = restore started. 200 = a restored copy already exists (S3 answers
147 * OK and extends nothing). 409 = RestoreAlreadyInProgress, which is the
148 * SAME outcome as 202 from the caller's side — the object is on its way —
149 * and treating it as an error would make every concurrent reader of one
150 * archived object fail all but the first. */
151 if (resp.status != 200 && resp.status != 202 && resp.status != 409) {
152 rc = sd_s3_status_err(resp.status, "RestoreObject", f->key, errbuf,
153 errcap);
154 }
155 e = errno;
156 f->transport->resp_free(&resp);
157 sd_s3_close(f);
158 errno = e;
159 return rc;
160}
