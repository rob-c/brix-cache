35/* Two hex checksums are equal ignoring case and leading-zero padding (a client
36 * may send an un-padded adler32 while our compute zero-pads to the alg width). */
37static int
38webdav_hex_norm_equal(const char *a, const char *b)
39{
40 while (*a == '0') { a++; }
41 while (*b == '0') { b++; }
42 for (; *a != '\0' && *b != '\0'; a++, b++) {
43 if (ngx_tolower((u_char) *a) != ngx_tolower((u_char) *b)) {
44 return 0;
45 }
46 }
47 return *a == '\0' && *b == '\0';
48}

50/* Pick the first supported digest from Digest: (RFC 3230) then Content-MD5:. On
51 * BRIX_DIGEST_FOUND, *alg points at a static alg name and exp_hex holds the
52 * client-asserted value as lowercase hex. Unsupported Digest tokens are skipped
53 * (best-effort interop), so a Digest that names only algs we cannot compute reads
54 * as NONE — require_digest then decides whether that is acceptable.
55 *
56 * The grammar itself (token table, base64-vs-hex encoding per algorithm, the
57 * comma-separated scan) lives in core/compat/digest_header.c, shared with the
58 * outbound direction — the sd_http checksum-offload slot parses the SAME header
59 * off an origin's reply, and a client and an origin must not be understood by
60 * two different parsers. */
61static brix_digest_kind_t
62webdav_digest_select(ngx_http_request_t *r, const char **alg,
63 char *exp_hex, size_t exp_sz)
64{
65 ngx_table_elt_t *h;
66 brix_digest_kind_t kind;
67
68 h = brix_http_find_header(r, "Digest", sizeof("Digest") - 1);
69 if (h != NULL && h->value.len > 0) {
70 kind = brix_digest_header_scan(h->value.data, h->value.len, NULL,
71 alg, exp_hex, exp_sz);
72 if (kind != BRIX_DIGEST_NONE) {
73 return kind;
74 }
75 }
76
77 h = brix_http_find_header(r, "Content-MD5", sizeof("Content-MD5") - 1);
78 if (h != NULL && h->value.len > 0) {
79 if (brix_digest_value_hex(h->value.data, h->value.len, 1,
80 exp_hex, exp_sz) != NGX_OK)
81 {
82 return BRIX_DIGEST_BAD;
83 }
84 *alg = "md5";
85 return BRIX_DIGEST_FOUND;
86 }
87
88 return BRIX_DIGEST_NONE;
89}

91/*
92 * webdav_put_verify_ingest_digest — verify a client-asserted body digest over
93 * the fully-staged bytes, BEFORE the commit publishes them.
94 *
95 * A PUT may carry an end-to-end digest the client computed over what it sent
96 * (RFC-3230 `Digest:` — the WLCG/XrdHttp convention — or legacy `Content-MD5:`).
97 * The writer otherwise streams and commits whatever lands, so a byte flipped in
98 * flight past the TCP checksum is stored silently. When a usable digest is
99 * present we recompute it over the staged temp fd and refuse the commit (400) on
100 * mismatch — the client told us what it sent, so we can and must check it. With
101 * brix_webdav_require_digest on, a PUT that carries no usable digest is also
102 * refused, for deployments that decline writes they cannot verify.
103 *
104 * Returns NGX_OK to proceed to commit, or an HTTP status (>=400) to reject.
105 *
106 * Scope: the staged POSIX temp fd (brix_vfs_writer_fd). Coded bodies
107 * (Content-Encoding) are skipped — the stored plaintext would not match a digest
108 * over the encoded stream, matching the verify-on-write contract. A driver
109 * object target (S3) exposes no fd here and is verified on its own ingest path.
110 * The recompute is synchronous on the event thread (like §8.3 checksum-on-write);
111 * it only runs when a digest is asserted or required.
112 */
113ngx_int_t
114webdav_put_verify_ingest_digest(ngx_http_request_t *r,
115 ngx_http_brix_webdav_loc_conf_t *conf, brix_vfs_writer_t *writer,
116 const char *path)
117{
118 char exp_hex[129];
119 const char *alg = NULL;
120 brix_digest_kind_t kind;
121 ngx_fd_t fd;
122 ngx_table_elt_t *ce;
123 brix_integrity_info_t info;
124 brix_integrity_opts_t o;
125
126 /* A coded body is stored decoded; a digest over the encoded stream cannot be
127 * checked against it — skip (same limitation as verify-on-write). */
128 ce = brix_http_find_header(r, "Content-Encoding", sizeof("Content-Encoding") - 1);
129 if (ce != NULL && ce->value.len > 0) {
130 return NGX_OK;
131 }
132
133 kind = webdav_digest_select(r, &alg, exp_hex, sizeof(exp_hex));
134 if (kind == BRIX_DIGEST_BAD) {
135 return NGX_HTTP_BAD_REQUEST; /* known alg, bad value */
136 }
137 if (kind == BRIX_DIGEST_NONE) {
138 return conf->require_digest ? NGX_HTTP_BAD_REQUEST : NGX_OK;
139 }
140
141 fd = brix_vfs_writer_fd(writer);
142 if (fd == NGX_INVALID_FILE) {
143 /* Driver-backed object target — no local fd to hash here. */
144 return conf->require_digest ? NGX_HTTP_BAD_REQUEST : NGX_OK;
145 }
146
147 ngx_memzero(&o, sizeof(o));
148 o.require_regular_file = 1; /* hash the staged bytes fresh (no xattr cache) */
149 {
150 char procpath[32];
151 ngx_int_t irc;
152 int rfd;
153
154 /* The staged temp is opened write-only (staged_open_posix, O_WRONLY), and
155 * has no final path yet (pre-commit), so it cannot be pread directly nor
156 * reopened by name. Re-open the same open file description read-only via
157 * /proc/self/fd/<fd> — valid on Linux for an unlinked/O_TMPFILE staged
158 * temp — and hash the already-written bytes over the read handle. */
159 (void) ngx_snprintf((u_char *) procpath, sizeof(procpath),
160 "/proc/self/fd/%d%Z", (int) fd);
161 rfd = open(procpath, O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: DOMAIN_STAGE — read-back of staged PUT bytes for pre-commit ingest-digest verify (no final path yet) */
162 if (rfd < 0) {
163 /* Cannot obtain a readable view of what we must verify: fail closed. */
164 return NGX_HTTP_BAD_REQUEST;
165 }
166 irc = brix_integrity_get_fd(r->connection->log, rfd, NULL, path,
167 alg, &o, &info);
168 (void) close(rfd);
169 if (irc != NGX_OK) {
170 /* Cannot compute the digest we were asked to verify: fail closed. */
171 return NGX_HTTP_BAD_REQUEST;
172 }
173 }
174
175 if (!webdav_hex_norm_equal(exp_hex, info.hex)) {
176 ngx_log_error(NGX_LOG_ERR, r->connection->log, 0,
177 "brix_webdav: PUT ingest %s digest mismatch (asserted=%s got=%s)",
178 alg, exp_hex, info.hex);
179 brix_log_safe_path(r->connection->log, NGX_LOG_ERR, 0,
180 "brix_webdav: ingest digest mismatch rejects PUT for: \"%s\"", path);
181 return NGX_HTTP_BAD_REQUEST;
182 }
183 return NGX_OK;
184}
