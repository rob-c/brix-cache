18/* Decode the \ooo escapes mountinfo uses for spaces, tabs and backslashes. */
19static void
20mountinfo_unescape(const char *in, char *out, size_t cap)
21{
22 size_t o = 0;
23
24 while (*in != '\0' && o + 1 < cap) {
25 if (in[0] == '\\' && in[1] >= '0' && in[1] <= '7'
26 && in[2] >= '0' && in[2] <= '7' && in[3] >= '0' && in[3] <= '7')
27 {
28 out[o++] = (char) (((in[1] - '0') << 6) | ((in[2] - '0') << 3)
29 | (in[3] - '0'));
30 in += 4;
31 continue;
32 }
33 out[o++] = *in++;
34 }
35 out[o] = '\0';
36}

38/* One mountinfo line -> (mountpoint, fstype, source) -> cb. Malformed lines
39 * are skipped; the tokenizer is destructive on `line`. */
40static int
41mountinfo_emit_line(char *line, brix_plat_mount_cb cb, void *arg)
42{
43 char *fields[48];
44 int nf = 0, sep = -1, i;
45 char *tok, *save;
46 char mp[4096], src[4096];
47
48 for (tok = strtok_r(line, " \n", &save); tok != NULL && nf < 48;
49 tok = strtok_r(NULL, " \n", &save))
50 {
51 fields[nf++] = tok;
52 }
53 for (i = 0; i < nf; i++) {
54 if (strcmp(fields[i], "-") == 0) {
55 sep = i;
56 break;
57 }
58 }
59 if (sep < 5 || sep + 2 >= nf) {
60 return 0;
61 }
62 mountinfo_unescape(fields[4], mp, sizeof(mp));
63 mountinfo_unescape(fields[sep + 2], src, sizeof(src));
64 return cb(mp, fields[sep + 1], src, arg);
65}

67int
68brix_plat_mounts_walk(const char *table_path, brix_plat_mount_cb cb, void *arg)
69{
70 FILE *fp;
71 char *line = NULL;
72 size_t cap = 0;
73 int rc = 0;
74
75 fp = fopen(table_path != NULL ? table_path : "/proc/self/mountinfo", "r"); /* vfs-seam-allow: NOT_STORAGE - PAL mount-table read */
76 if (fp == NULL) {
77 return -1;
78 }
79 while (rc == 0 && getline(&line, &cap, fp) >= 0) {
80 rc = mountinfo_emit_line(line, cb, arg);
81 }
82 free(line);
83 fclose(fp);
84 return rc;
85}
