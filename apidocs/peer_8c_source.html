1/*
2 * peer.c — root:// binding for the protbind engine, and the shared
3 * per-connection peer-hostname cache every host-template policy reads.
4 *
5 * WHAT: Supplies the peer identity the pure engine needs for a stream session:
6 * the reverse-resolved peer hostname (resolved at most once per
7 * connection) and the peer IP already recorded at accept time, then calls
8 * brix_protbind_resolve().
9 *
10 * WHY: policy.c and match.c must stay free of brix_ctx_t so the HTTP frontend
11 * can reuse them, but SOMETHING has to own the one impure step — a
12 * blocking PTR lookup — and the cache that keeps it to one call per
13 * connection. Two subsystems match host templates against that name
14 * (protbind rules here, XrdAcc `h <host>`/`h .domain` records in
15 * ../authz/auth_gate.c), so the cache lives here once and both read it;
16 * otherwise a session with both features on would resolve twice.
17 *
18 * HOW: brix_protbind_peer_host_cached() fills the ctx->login.acc_host slot
19 * through brix_acc_resolve_peer() — the circuit-breaker-bounded
20 * getnameinfo path — and every later call returns the cached string.
21 * brix_protbind_peer_host() adds the protbind-specific short-circuit:
22 * when every template is a bare "*" no lookup is needed at all.
23 */
24

30#define BRIX_PROTBIND_HOST_MAX 256

32/* ---- Reverse-resolve the peer hostname, once per connection ----
33 *
34 * WHAT: Returns the peer's FQDN, or NULL when it has no PTR record, the DNS
35 * circuit breaker is open, or the name could not be cached. The result
36 * is borrowed from the connection pool and lives for the whole session.
37 *
38 * WHY: Reverse DNS blocks the event loop, so a session must pay for it at most
39 * once no matter how many host-template policies consult the name — the
40 * kXR_auth path alone re-evaluates protbind on every round. Caching a
41 * NULL result too (via the done flag) is deliberate: a peer with no PTR
42 * record must not trigger a fresh lookup per request.
43 *
44 * HOW: 1. On the first call, mark the cache resolved and attempt the lookup.
45 * 2. Copy any resulting name into the connection pool.
46 * 3. Return the cached pointer (NULL when resolution failed).
47 */
48const char *
49brix_protbind_peer_host_cached(struct brix_ctx_s *ctx, ngx_connection_t *c)
50{
51 char host_buf[BRIX_PROTBIND_HOST_MAX];
52 const char *resolved;
53
54 if (!ctx->login.acc_host_done) {
55 ctx->login.acc_host_done = 1;
56
57 resolved = brix_acc_resolve_peer(c->sockaddr, c->socklen,
58 host_buf, sizeof(host_buf));
59 if (resolved != NULL) {
60 size_t name_len = ngx_strlen(resolved);
61 char *owned = ngx_pnalloc(c->pool, name_len + 1);
62
63 if (owned != NULL) {
64 ngx_memcpy(owned, resolved, name_len + 1);
65 ctx->login.acc_host = owned;
66 }
67 }
68 }
69
70 return ctx->login.acc_host;
71}

73/* ---- Peer hostname for protbind template matching ----
74 *
75 * WHAT: Returns the cached peer hostname, or NULL when no configured template
76 * could consult it.
77 *
78 * WHY: The dominant configuration is a single `brix_protbind * <protos>`
79 * line, which is decidable from the wildcard alone. Asking that question
80 * before touching the resolver keeps the common case free of DNS entirely
81 * — and, just as importantly, keeps protbind from populating the shared
82 * cache for sessions that never needed it.
83 *
84 * HOW: Short-circuit on a wildcard-only ruleset, else read the shared cache.
85 */
86const char *
87brix_protbind_peer_host(struct brix_ctx_s *ctx, ngx_connection_t *c,
88 ngx_array_t *rules)
89{
90 if (!brix_protbind_needs_hostname(rules)) {
91 return NULL;
92 }
93
94 return brix_protbind_peer_host_cached(ctx, c);
95}

97/* ---- Resolve the effective protocol set for a root:// session ----
98 *
99 * WHAT: Fills *out with the ordered protocols this session may authenticate
100 * with, given the listener's brix_auth mode and its protbind rules.
101 *
102 * WHY: Both kXR_login (which advertises the set as the "&P=" sec token) and
103 * kXR_auth (which enforces membership before running a scheme) need the
104 * same verdict. Recomputing it from the connection rather than caching a
105 * set on brix_ctx_t keeps the per-connection struct unchanged and costs
106 * only a handful of string compares, the DNS lookup being cached.
107 *
108 * HOW: 1. Build the base set from the brix_auth mode.
109 * 2. Supply the (lazily resolved) hostname and the recorded peer IP.
110 * 3. Delegate the decision to the shared resolver.
111 */
112void
113brix_protbind_resolve_ctx(struct brix_ctx_s *ctx, ngx_connection_t *c,
114 ngx_array_t *rules, ngx_uint_t base_auth, brix_protbind_set_t *out)
115{
116 brix_protbind_set_t base;
117 const char *peer_ip;
118
119 brix_protbind_base_set(base_auth, &base);
120
121 peer_ip = (ctx->login.peer_ip[0] != '\0') ? ctx->login.peer_ip : NULL;
122
123 brix_protbind_resolve(rules, &base,
124 brix_protbind_peer_host(ctx, c, rules),
125 peer_ip, out);
126}
