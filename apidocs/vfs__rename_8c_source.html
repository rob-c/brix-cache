1/*
2 * vfs_rename.c — VFS rename / move.
3 *
4 * WHAT: Implements brix_vfs_rename(), which moves the resolved ctx (source)
5 * path to a caller-supplied, already-resolved destination
6 * brix_path_result_t.
7 *
8 * WHY: kXR_mv and WebDAV MOVE need a single write-gated, confined rename whose
9 * destination is verified to be inside the export root before the syscall,
10 * with one metric/access-log line — not an ad-hoc rename(2) per protocol.
11 *
12 * Also owns brix_vfs_exchange() (phase-107 C6): the atomic two-name swap
13 * — same gate, same confinement, same OP_RENAME meter, refused ENOTSUP
14 * where the backend has no primitive (never a two-rename emulation).
15 *
16 * HOW: Enforces brix_vfs_require_confined_mutation(MUTATE_RENAME) — EROFS on
17 * a read-only endpoint — then demands a non-NULL root_canon
18 * and a destination that is itself is_confined with a non-empty resolved
19 * path. It delegates the actual move to brix_ns_rename() (namespace
20 * layer), maps the returned status back to errno (sys_errno or EIO), and
21 * observes the operation as BRIX_METRIC_OP_RENAME on every path.
22 */
23#include "vfs_internal.h"
24#include "fs/backend/cache/sd_cache.h" /* brix_sd_cache_evict: leaf-dispatch bypasses the decorator's own evict */
25#include "vfs_backend_registry.h" /* brix_vfs_backend_durable (C3 barrier gate) */
26#include "core/compat/staged_file.h" /* brix_publish_dirsync (C3 POSIX barrier) */
27#include "fs/path/beneath.h" /* brix_beneath_strip_root (C6 EXDEV gate) */

29/* Thread-safe confined rename (no pool alloc, no metric) — relocates the
30 * namespace rename into the VFS layer for off-thread / pool-less callers (kXR_mv,
31 * WebDAV MOVE collection offload). Maps the namespace status to errno + an
32 * optional was_dir flag; see vfs.h. */
33ngx_int_t
34brix_vfs_rename_path(brix_sd_instance_t *sd, ngx_log_t *log,
35 const char *root_canon, const brix_n2n_cfg_t *n2n,
36 const char *src, const char *dst,
37 unsigned overwrite, int *was_dir_out)
38{
39 brix_ns_result_t res;
40
41 /* Non-POSIX backend: rename in the driver namespace (export-relative keys).
42 * was_dir is derived from a post-rename stat of the destination. */
43 if (sd != NULL && sd->driver != brix_sd_default_driver()
44 && sd->driver->rename != NULL)
45 {
46 char s[PATH_MAX];
47 char d[PATH_MAX];
48 ngx_int_t rc;
49
50 if (brix_path_export_to_pfn(root_canon, n2n, src, s, sizeof(s))
51 != NGX_OK
52 || brix_path_export_to_pfn(root_canon, n2n, dst, d, sizeof(d))
53 != NGX_OK)
54 {
55 return NGX_ERROR;
56 }
57 rc = sd->driver->rename(sd, s, d, overwrite ? 0 : 1);
58
59 if (was_dir_out != NULL) {
60 brix_sd_stat_t st;
61
62 *was_dir_out = (rc == NGX_OK && sd->driver->stat != NULL
63 && sd->driver->stat(sd, d, &st) == NGX_OK)
64 ? st.is_dir : 0;
65 }
66 return rc;
67 }
68
69 res = brix_ns_rename(log, root_canon, src, dst, overwrite ? 1 : 0);
70 if (was_dir_out != NULL) {
71 *was_dir_out = res.was_dir;
72 }
73 if (res.status == BRIX_NS_OK) {
74 return NGX_OK;
75 }
76 errno = res.sys_errno != 0 ? res.sys_errno
77 : brix_vfs_ns_status_errno(res.status);
78 return NGX_ERROR;
79}

81/*
82 * WHAT: Shared entry gate for the two-name driver mutations (rename/exchange):
83 * capability check, credential materialisation, and derivation of both
84 * export-relative keys.
85 * WHY: Both arms refuse identically (EPERM without DIRS_WRITE, the cred-gate
86 * errno when materialisation fails) and both book the refusal as
87 * OP_RENAME; one body keeps the refusal semantics from drifting apart.
88 * HOW: On refusal, sets errno, observes, returns NGX_ERROR — the caller just
89 * propagates. On NGX_OK the caller owns `store` and must wipe it after
90 * its driver call.
91 */
92static ngx_int_t
93brix_vfs_two_key_gate(brix_vfs_ctx_t *ctx, const brix_path_result_t *second,
94 const char *path, uint64_t start, const brix_sd_driver_t *drv,
95 brix_sd_ucred_t *store, brix_sd_cred_t *cred, int *use_cred,
96 char a_key[PATH_MAX], char b_key[PATH_MAX])
97{
98 int cred_err = 0;
99 int saved_errno;
100
101 ngx_memzero(store, sizeof(*store));
102 ngx_memzero(cred, sizeof(*cred));
103 if (!(drv->caps & BRIX_SD_CAP_DIRS_WRITE)) {
104 errno = EPERM;
105 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
106 NGX_ERROR, EPERM, start);
107 return NGX_ERROR;
108 }
109 if (brix_vfs_cred_gate_active(ctx)
110 && brix_vfs_ns_cred(ctx, store, cred, use_cred, &cred_err) != NGX_OK)
111 {
112 saved_errno = cred_err ? cred_err : EACCES;
113 errno = saved_errno;
114 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
115 NGX_ERROR, saved_errno, start);
116 return NGX_ERROR;
117 }
118 if (brix_path_resolved_to_pfn(ctx, path, a_key, PATH_MAX) != NGX_OK
119 || brix_path_resolved_to_pfn(ctx,
120 (const char *) second->resolved.data, b_key, PATH_MAX) != NGX_OK)
121 {
122 saved_errno = errno;
123 brix_sd_ucred_wipe(store);
124 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
125 NGX_ERROR, saved_errno, start);
126 errno = saved_errno;
127 return NGX_ERROR;
128 }
129 return NGX_OK;
130}

132/*
133 * WHAT: Rename two keys through a non-POSIX storage-driver namespace.
134 * WHY: Credential dispatch and cache eviction are distinct from POSIX rename.
135 * HOW: Gate capability/credentials, call the leaf driver, wipe, evict, observe.
136 */
137static ngx_int_t
138brix_vfs_rename_driver(brix_vfs_ctx_t *ctx, const brix_path_result_t *dst,
139 const char *path, uint64_t start, const brix_sd_driver_t *drv)
140{
141 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(ctx->sd);
142 brix_sd_ucred_t store;
143 brix_sd_cred_t cred;
144 ngx_int_t rc;
145 int use_cred = 0;
146 int saved_errno;
147 char src_key[PATH_MAX];
148 char dst_key[PATH_MAX];
149
150 if (brix_vfs_two_key_gate(ctx, dst, path, start, drv, &store, &cred,
151 &use_cred, src_key, dst_key) != NGX_OK)
152 {
153 return NGX_ERROR;
154 }
155 rc = drv->rename != NULL
156 ? brix_sd_rename_maybe_cred(leaf, src_key, dst_key, 0,
157 use_cred ? &cred : NULL)
158 : (errno = ENOTSUP, NGX_ERROR);
159 saved_errno = errno;
160 brix_sd_ucred_wipe(&store);
161
162 /* Phase-107 C3: durable-publish barrier on the DESTINATION name. Dispatched
163 * on the leaf (a decorator relays; NULL slot = the far end's rename is
164 * atomic-and-durable there). A failed barrier FAILS the rename — the name
165 * moved and cannot be moved back, but success would claim durability the
166 * store does not have. */
167 if (rc == NGX_OK && leaf != NULL && leaf->driver->sync_publish != NULL
168 && brix_vfs_backend_durable(ctx->root_canon)
169 && leaf->driver->sync_publish(leaf, dst_key) != NGX_OK)
170 {
171 saved_errno = errno ? errno : EIO;
172 ngx_log_error(NGX_LOG_CRIT, ctx->log, saved_errno,
173 "brix: rename: durable-publish barrier failed for "
174 "\"%s\"", dst_key);
175 errno = saved_errno;
176 rc = NGX_ERROR;
177 }
178 if (rc == NGX_OK) {
179 brix_metric_cache_evicted(brix_vfs_metrics_proto(ctx),
180 brix_sd_cache_evict(ctx->sd, src_key)
181 + brix_sd_cache_evict(ctx->sd, dst_key));
182 }
183 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
184 rc, saved_errno, start);
185 return rc;
186}

188/*
189 * WHAT: Shared public-entry gate for the two-name mutations (rename/exchange):
190 * the write-plane policy gate, then validation of the second endpoint.
191 * WHY: Both verbs are rename-class mutations with an identical refusal shape
192 * (policy errno, EINVAL for an unconfined/absent second name), booked as
193 * OP_RENAME; one body keeps the two entries from drifting apart.
194 * HOW: On refusal, sets errno, observes, returns NGX_ERROR — the caller just
195 * propagates.
196 */
197static ngx_int_t
198brix_vfs_two_name_entry(brix_vfs_ctx_t *ctx, const brix_path_result_t *second,
199 const char *path, uint64_t start)
200{
201 int saved_errno;
202
203 if (brix_vfs_confined_mutation_checked(ctx,
204 BRIX_VFS_MUTATE_RENAME) != NGX_OK)
205 {
206 saved_errno = errno;
207 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
208 NGX_ERROR, saved_errno, start);
209 return NGX_ERROR;
210 }
211
212 if (ctx->root_canon == NULL || second == NULL || !second->is_confined
213 || second->resolved.data == NULL)
214 {
215 errno = EINVAL;
216 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
217 NGX_ERROR, EINVAL, start);
218 return NGX_ERROR;
219 }
220 if (brix_vfs_require_authorized_target(ctx,
221 (const char *) second->resolved.data,
222 BRIX_VFS_MUTATE_RENAME) != NGX_OK)
223 {
224 saved_errno = errno;
225 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
226 NGX_ERROR, saved_errno, start);
227 return NGX_ERROR;
228 }
229
230 /* phase-107 C7: lock gate after the mutation gate (EROFS precedes EBUSY),
231 * on BOTH names — a rename removes the source and replaces the second
232 * endpoint, so a lock on either refuses. The second name's confinement
233 * was validated just above. */
234 if (brix_vfs_require_unlocked(ctx, BRIX_VFS_MUTATE_RENAME) != NGX_OK
235 || brix_vfs_require_unlocked_at(ctx,
236 (const char *) second->resolved.data,
237 BRIX_VFS_MUTATE_RENAME) != NGX_OK)
238 {
239 saved_errno = errno;
240 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
241 NGX_ERROR, saved_errno, start);
242 return NGX_ERROR;
243 }
244 return NGX_OK;
245}

247/* Move the resolved ctx source path to the confined destination `dst`.
248 * Write-gated; both endpoints must be confined. Metered as OP_RENAME.
249 * `overwrite_dirs` lets the rename replace an existing directory destination
250 * (its tree is removed first — WebDAV MOVE Overwrite:T); 0 keeps kXR_mv
251 * semantics where an existing dir dest fails with EEXIST. */
252ngx_int_t
253brix_vfs_rename(brix_vfs_ctx_t *ctx, const brix_path_result_t *dst,
254 unsigned overwrite_dirs)
255{
256 brix_ns_result_t res;
257 const char *path;
258 uint64_t start;
259 int saved_errno;
260 const brix_sd_driver_t *drv;
261
262 start = brix_vfs_now_ns();
263 path = brix_vfs_ctx_path(ctx);
264
265 if (brix_vfs_two_name_entry(ctx, dst, path, start) != NGX_OK) {
266 return NGX_ERROR;
267 }
268
269 /* Non-POSIX backend: rename within the driver namespace (both endpoints are
270 * keyed export-relative; the move carries content via the catalog).
271 * Dispatch on the leaf instance so brix_sd_rename_maybe_cred finds the
272 * leaf driver's rename_cred slot (decorators have only plain relays). */
273 drv = brix_vfs_ctx_driver(ctx);
274 if (drv != NULL)
275 return brix_vfs_rename_driver(ctx, dst, path, start, drv);
276
277 res = brix_ns_rename(ctx->log, ctx->root_canon,
278 brix_vfs_ctx_path(ctx),
279 (const char *) dst->resolved.data,
280 overwrite_dirs ? 1 : 0);
281 if (res.status == BRIX_NS_OK) {
282 /* Phase-107 C3: flush the destination's parent directory entry so the
283 * rename itself survives a crash (same barrier the staged commit
284 * carries; -1 anchor = dirsync opens its own confined root). Failure
285 * FAILS the rename — the name moved, but reporting success without
286 * durability is the exact bug the barrier removes. */
287 if (brix_vfs_backend_durable(ctx->root_canon)
288 && brix_publish_dirsync(ctx->log, -1, ctx->root_canon,
289 (const char *) dst->resolved.data) != NGX_OK)
290 {
291 saved_errno = errno ? errno : EIO;
292 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
293 NGX_ERROR, saved_errno, start);
294 errno = saved_errno;
295 return NGX_ERROR;
296 }
297 brix_vfs_neg_stat_forget(ctx->root_canon,
298 (const char *) dst->resolved.data);
299 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
300 NGX_OK, 0, start);
301 return NGX_OK;
302 }
303
304 /* A pre-checked refusal (e.g. destination is an existing directory →
305 * BRIX_NS_EXISTS) carries no syscall errno; map the status instead of
306 * collapsing it to EIO so mv/MOVE report the true category. */
307 errno = res.sys_errno != 0 ? res.sys_errno
308 : brix_vfs_ns_status_errno(res.status);
309 saved_errno = errno;
310 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
311 NGX_ERROR, saved_errno, start);
312 return NGX_ERROR;
313}

315/*
316 * WHAT: Exchange two keys through a non-POSIX storage-driver namespace.
317 * WHY: Same gate/cred/evict shape as rename, but BOTH names change content,
318 * so the C3 barrier and the cache eviction cover both keys.
319 * HOW: Gate capability/credentials, call the leaf driver (ENOTSUP when it has
320 * no primitive — never emulated), wipe, barrier both, evict both, observe.
321 */
322static ngx_int_t
323brix_vfs_exchange_driver(brix_vfs_ctx_t *ctx, const brix_path_result_t *other,
324 const char *path, uint64_t start, const brix_sd_driver_t *drv)
325{
326 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(ctx->sd);
327 brix_sd_ucred_t store;
328 brix_sd_cred_t cred;
329 ngx_int_t rc;
330 int use_cred = 0;
331 int saved_errno;
332 char a_key[PATH_MAX];
333 char b_key[PATH_MAX];
334
335 if (brix_vfs_two_key_gate(ctx, other, path, start, drv, &store, &cred,
336 &use_cred, a_key, b_key) != NGX_OK)
337 {
338 return NGX_ERROR;
339 }
340 rc = brix_sd_exchange_maybe_cred(leaf, a_key, b_key,
341 use_cred ? &cred : NULL);
342 saved_errno = errno;
343 brix_sd_ucred_wipe(&store);
344
345 /* Phase-107 C3: durable-publish barrier on BOTH names — each now maps to
346 * the other's content. Dispatched on the leaf (a decorator relays; NULL
347 * slot = the far end's swap is atomic-and-durable there). A failed barrier
348 * FAILS the exchange — the names swapped and cannot be un-swapped safely,
349 * but success would claim durability the store does not have. */
350 if (rc == NGX_OK && leaf != NULL && leaf->driver->sync_publish != NULL
351 && brix_vfs_backend_durable(ctx->root_canon)
352 && (leaf->driver->sync_publish(leaf, a_key) != NGX_OK
353 || leaf->driver->sync_publish(leaf, b_key) != NGX_OK))
354 {
355 saved_errno = errno ? errno : EIO;
356 ngx_log_error(NGX_LOG_CRIT, ctx->log, saved_errno,
357 "brix: exchange: durable-publish barrier failed for "
358 "\"%s\" <-> \"%s\"", a_key, b_key);
359 errno = saved_errno;
360 rc = NGX_ERROR;
361 }
362 if (rc == NGX_OK) {
363 brix_metric_cache_evicted(brix_vfs_metrics_proto(ctx),
364 brix_sd_cache_evict(ctx->sd, a_key)
365 + brix_sd_cache_evict(ctx->sd, b_key));
366 }
367 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
368 rc, saved_errno, start);
369 return rc;
370}

372/* Atomically exchange the resolved ctx path with the confined name `other`
373 * (phase-107 C6): no instant at which either name is missing. Write-gated as a
374 * rename-class mutation; both endpoints must be confined in the SAME export
375 * (EXDEV otherwise); metered as OP_RENAME. ENOTSUP where the backend has no
376 * primitive — NEVER emulated with two renames (§3.5): a caller that asked for
377 * an atomic swap would rather have a refusal than a window in which neither
378 * name resolves. */
379ngx_int_t
380brix_vfs_exchange(brix_vfs_ctx_t *ctx, const brix_path_result_t *other)
381{
382 brix_ns_result_t res;
383 const char *path;
384 uint64_t start;
385 int saved_errno;
386 const brix_sd_driver_t *drv;
387
388 start = brix_vfs_now_ns();
389 path = brix_vfs_ctx_path(ctx);
390
391 if (brix_vfs_two_name_entry(ctx, other, path, start) != NGX_OK) {
392 return NGX_ERROR;
393 }
394
395 /* Both names must live in THIS export: a cross-export swap is EXDEV, not a
396 * confinement escape. The driver arm needs the check here — its
397 * export-relative keying passes a foreign absolute path through UNCHANGED,
398 * which would hand the backend a key outside the export. (The POSIX arm
399 * re-derives the same answer inside brix_ns_exchange.) */
400 if (brix_beneath_strip_root(ctx->root_canon,
401 (const char *) other->resolved.data) == NULL)
402 {
403 errno = EXDEV;
404 saved_errno = errno;
405 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
406 NGX_ERROR, saved_errno, start);
407 return NGX_ERROR;
408 }
409
410 drv = brix_vfs_ctx_driver(ctx);
411 if (drv != NULL)
412 return brix_vfs_exchange_driver(ctx, other, path, start, drv);
413
414 res = brix_ns_exchange(ctx->log, ctx->root_canon, path,
415 (const char *) other->resolved.data);
416 if (res.status == BRIX_NS_OK) {
417 /* Phase-107 C3: both directory entries changed — flush both parents
418 * (same barrier rename carries for its destination). Failure FAILS the
419 * exchange: the names swapped, but reporting success without
420 * durability is the exact bug the barrier removes. */
421 if (brix_vfs_backend_durable(ctx->root_canon)
422 && (brix_publish_dirsync(ctx->log, -1, ctx->root_canon,
423 path) != NGX_OK
424 || brix_publish_dirsync(ctx->log, -1, ctx->root_canon,
425 (const char *)
426 other->resolved.data) != NGX_OK))
427 {
428 saved_errno = errno ? errno : EIO;
429 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
430 NGX_ERROR, saved_errno, start);
431 errno = saved_errno;
432 return NGX_ERROR;
433 }
434 brix_vfs_neg_stat_forget(ctx->root_canon, path);
435 brix_vfs_neg_stat_forget(ctx->root_canon,
436 (const char *) other->resolved.data);
437 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
438 NGX_OK, 0, start);
439 return NGX_OK;
440 }
441
442 errno = res.sys_errno != 0 ? res.sys_errno
443 : brix_vfs_ns_status_errno(res.status);
444 saved_errno = errno;
445 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_RENAME, NULL, 0,
446 NGX_ERROR, saved_errno, start);
447 return NGX_ERROR;
448}
