46static void
47conv_emit(conv_t *c, const char *fmt, ...)
48{
49 va_list ap;
50 int n;
51
52 if (c->over) {
53 return;
54 }
55 va_start(ap, fmt);
56 n = vsnprintf(c->doc + c->len, CONV_DOC_CAP - c->len, fmt, ap);
57 va_end(ap);
58 if (n < 0 || (size_t) n >= CONV_DOC_CAP - c->len) {
59 c->over = 1;
60 return;
61 }
62 c->len += (size_t) n;
63}

65static void
66conv_emit_raw(conv_t *c, const char *p, size_t n)
67{
68 if (c->over) {
69 return;
70 }
71 if (n >= CONV_DOC_CAP - c->len) {
72 c->over = 1;
73 return;
74 }
75 memcpy(c->doc + c->len, p, n);
76 c->len += n;
77 c->doc[c->len] = '\0';
78}

82/* An unlinked scratch file: the converted layer has to be pread-able for
83 * the registry push, and must not survive a crash. */
84static int
85conv_tmpfd(char *err, size_t errlen)
86{
87 const char *dir = getenv("TMPDIR");
88 char path[1100];
89 int fd;
90
91 snprintf(path, sizeof(path), "%s/brixoci-convert-XXXXXX",
92 dir != NULL && dir[0] != '\0' ? dir : "/tmp");
93 fd = mkstemp(path);
94 if (fd < 0) {
95 snprintf(err, errlen, "scratch file: %s", strerror(errno));
96 return BRIX_OCI_REG_ETRANSPORT;
97 }
98 unlink(path);
99 return fd;
100}

102/* The source layer as a readable fd positioned at byte 0. Layout blobs are
103 * verified against their path digest first; registry blobs are hashed on
104 * the way in by the fetch itself. */
105static int
106conv_src_fd(brixoci_end_t *s, const char *digest, char *err, size_t errlen)
107{
108 long long sz;
109 int fd, rc;
110
111 if (s->is_layout) {
112 rc = brix_oci_layout_blob_verify(&s->lay, digest, &sz, err, errlen);
113 return rc != BRIX_OCI_REG_OK
114 ? rc
115 : brix_oci_layout_blob_open(&s->lay, digest, err, errlen);
116 }
117 fd = conv_tmpfd(err, errlen);
118 if (fd < 0) {
119 return fd;
120 }
121 rc = brix_oci_reg_blob_fetch(&s->reg, s->name, digest, fd, err, errlen);
122 if (rc != BRIX_OCI_REG_OK || lseek(fd, 0, SEEK_SET) != 0) {
123 close(fd);
124 return rc != BRIX_OCI_REG_OK ? rc : BRIX_OCI_REG_ETRANSPORT;
125 }
126 return fd;
127}

129/* Land [fd, size) at the destination under `digest`. The caller vouches
130 * for the digest — it came from the writer that produced the bytes. */
131static int
132conv_put_fd(brixoci_end_t *d, const char *digest, int fd, long long size,
133 char *err, size_t errlen)
134{
135 char tmppath[1200];
136 int out_fd, rc;
137
138 if (lseek(fd, 0, SEEK_SET) != 0) {
139 snprintf(err, errlen, "converted layer: %s", strerror(errno));
140 return BRIX_OCI_REG_ETRANSPORT;
141 }
142 if (!d->is_layout) {
143 return brix_oci_reg_blob_push(&d->reg, d->name, digest, fd,
144 (size_t) size, err, errlen);
145 }
146 out_fd = brix_oci_layout_stage(&d->lay, tmppath, sizeof(tmppath), err,
147 errlen);
148 if (out_fd < 0) {
149 return out_fd;
150 }
151 rc = brixoci_fd_pump(fd, out_fd, err, errlen) == 0
152 ? BRIX_OCI_REG_OK
153 : BRIX_OCI_REG_ETRANSPORT;
154 close(out_fd);
155 if (rc == BRIX_OCI_REG_OK) {
156 rc = brix_oci_layout_commit(&d->lay, tmppath, digest, err, errlen);
157 }
158 if (rc != BRIX_OCI_REG_OK) {
159 unlink(tmppath);
160 }
161 return rc;
162}

164/* Land an in-memory document (the rewritten config) at the destination,
165 * reporting the digest it hashed to. */
166static int
167conv_put_mem(brixoci_end_t *d, const void *body, size_t len, char *dig,
168 size_t diglen, char *err, size_t errlen)
169{
170 brix_oci_digest_t h;
171 ssize_t w;
172 int fd, rc;
173
174 if (d->is_layout) {
175 return brix_oci_layout_blob_put_mem(&d->lay, body, len, dig, diglen,
176 err, errlen);
177 }
178 if (brix_oci_sha256(body, len, &h) != 0 ||
179 brix_oci_digest_format(&h, dig, diglen) < 0) {
180 snprintf(err, errlen, "sha256 of rewritten document failed");
181 return BRIX_OCI_REG_EPROTO;
182 }
183 fd = conv_tmpfd(err, errlen);
184 if (fd < 0) {
185 return fd;
186 }
187 w = write(fd, body, len);
188 rc = (w < 0 || (size_t) w != len)
189 ? BRIX_OCI_REG_ETRANSPORT
190 : brix_oci_reg_blob_push(&d->reg, d->name, dig, fd, len, err,
191 errlen);
192 if (rc == BRIX_OCI_REG_ETRANSPORT && w != (ssize_t) len) {
193 snprintf(err, errlen, "rewritten document: %s", strerror(errno));
194 }
195 close(fd);
196 return rc;
197}

199/* The source config blob, whole, into a malloc'd buffer. */
200static int
201conv_load(brixoci_end_t *s, const char *digest, char **out, size_t *outlen,
202 char *err, size_t errlen)
203{
204 char *buf;
205 off_t end;
206 ssize_t got;
207 int fd;
208
209 if (s->is_layout) {
210 return brix_oci_layout_blob_load(&s->lay, digest, CONV_BLOB_CAP, out,
211 outlen, err, errlen);
212 }
213 fd = conv_src_fd(s, digest, err, errlen);
214 if (fd < 0) {
215 return fd;
216 }
217 end = lseek(fd, 0, SEEK_END);
218 if (end < 0 || (size_t) end > CONV_BLOB_CAP) {
219 snprintf(err, errlen, "config blob %s is %lld bytes", digest,
220 (long long) end);
221 close(fd);
222 return BRIX_OCI_REG_EPROTO;
223 }
224 buf = malloc((size_t) end + 1);
225 got = buf == NULL ? -1 : pread(fd, buf, (size_t) end, 0); /* vfs-seam-allow: OCI layout blob or anonymous registry staging fd, never an export VFS object */
226 close(fd);
227 if (got < 0 || (size_t) got != (size_t) end) {
228 free(buf);
229 snprintf(err, errlen, "config blob %s: short read", digest);
230 return BRIX_OCI_REG_ETRANSPORT;
231 }
232 buf[end] = '\0';
233 *out = buf;
234 *outlen = (size_t) end;
235 return BRIX_OCI_REG_OK;
236}

248static int
249conv_one_layer(brixoci_end_t *s, brixoci_end_t *d, const char *digest,
250 conv_layer_t *out, char *err, size_t errlen)
251{
252 brix_stargz_stats_t st;
253 int in_fd, out_fd, rc;
254
255 in_fd = conv_src_fd(s, digest, err, errlen);
256 if (in_fd < 0) {
257 return in_fd;
258 }
259 out_fd = conv_tmpfd(err, errlen);
260 if (out_fd < 0) {
261 close(in_fd);
262 return out_fd;
263 }
264 rc = brix_stargz_convert(in_fd, out_fd, &st, err, errlen) == 0
265 ? BRIX_OCI_REG_OK
266 : BRIX_OCI_REG_EPROTO;
267 close(in_fd);
268 if (rc == BRIX_OCI_REG_OK) {
269 snprintf(out->digest, sizeof(out->digest), "%s", st.blob_digest);
270 snprintf(out->diffid, sizeof(out->diffid), "%s", st.diffid);
271 snprintf(out->toc, sizeof(out->toc), "%s", st.toc_digest);
272 out->size = st.blob_size;
273 rc = conv_put_fd(d, out->digest, out_fd, st.blob_size, err, errlen);
274 }
275 close(out_fd);
276 return rc;
277}

279/* One manifest layer element → one converted layer. Foreign layers are
280 * refused: their bytes live on a URL the registry does not hold, so there
281 * is nothing here to reframe. */
282static int
283conv_layer_el(brixoci_end_t *s, brixoci_end_t *d, const char *el, size_t en,
284 conv_layer_t *out, char *err, size_t errlen)
285{
286 char dig[BRIX_OCI_DIGEST_STRLEN], mt[128];
287
288 if (!brix_json_get_str(el, en, "digest", dig, sizeof(dig))) {
289 snprintf(err, errlen, "manifest layer has no digest");
290 return BRIX_OCI_REG_EPROTO;
291 }
292 if (!brix_json_get_str(el, en, "mediaType", mt, sizeof(mt))) {
293 snprintf(mt, sizeof(mt), "%s", OCI_MT_LAYER_GZ);
294 }
295 if (strstr(mt, "foreign") != NULL) {
296 snprintf(err, errlen, "layer %s is a foreign layer (%s) — its bytes "
297 "are not in this registry", dig, mt);
298 return BRIX_OCI_REG_EPROTO;
299 }
300 snprintf(out->mediatype, sizeof(out->mediatype), "%s", conv_layer_mt(mt));
301 return conv_one_layer(s, d, dig, out, err, errlen);
302}

304static int
305conv_layers(brixoci_end_t *s, brixoci_end_t *d, const brix_oci_desc_t *m,
306 conv_t *c, char *err, size_t errlen)
307{
308 const char *arr, *el;
309 size_t an, en, cur = 0;
310 int rc;
311
312 if (brix_json_get_raw(m->body, m->body_len, "layers", &arr, &an) != 1) {
313 snprintf(err, errlen, "source manifest has no layers to convert");
314 return BRIX_OCI_REG_EPROTO;
315 }
316 while (brix_json_arr_next(arr, an, &cur, &el, &en) == 1) {
317 if (c->n >= CONV_MAX_LAYERS) {
318 snprintf(err, errlen, "source manifest has more than %d layers",
319 CONV_MAX_LAYERS);
320 return BRIX_OCI_REG_EPROTO;
321 }
322 rc = conv_layer_el(s, d, el, en, &c->lay[c->n], err, errlen);
323 if (rc != BRIX_OCI_REG_OK) {
324 return rc;
325 }
326 c->n++;
327 }
328 if (c->n == 0) {
329 snprintf(err, errlen, "source manifest has no layers to convert");
330 return BRIX_OCI_REG_EPROTO;
331 }
332 return BRIX_OCI_REG_OK;
333}

353/* The rewritten config: the source document with its rootfs object
354 * replaced. rootfs carries exactly two spec fields, so rebuilding it and
355 * splicing it over the original's raw span leaves every other field —
356 * history, env, entrypoint — byte-identical. */
357static int
358conv_rewrite_config(conv_t *c, const char *body, size_t blen, char *err,
359 size_t errlen)
360{
361 const char *rootfs;
362 size_t rn, head;
363 char type[64];
364 int i, count;
365
366 if (brix_json_get_raw(body, blen, "rootfs", &rootfs, &rn) != 1) {
367 snprintf(err, errlen, "image config has no rootfs");
368 return BRIX_OCI_REG_EPROTO;
369 }
370 count = conv_count_diffids(rootfs, rn);
371 if (count != c->n) {
372 snprintf(err, errlen, "image config lists %d diff_ids for %d layers",
373 count, c->n);
374 return BRIX_OCI_REG_EPROTO;
375 }
376 if (!brix_json_get_str(rootfs, rn, "type", type, sizeof(type))) {
377 snprintf(type, sizeof(type), "layers");
378 }
379 head = (size_t) (rootfs - body);
380 c->len = 0;
381 c->over = 0;
382 c->doc[0] = '\0';
383 conv_emit_raw(c, body, head);
384 conv_emit(c, "{\"type\":\"%s\",\"diff_ids\":[", type);
385 for (i = 0; i < c->n; i++) {
386 conv_emit(c, "%s\"%s\"", i ? "," : "", c->lay[i].diffid);
387 }
388 conv_emit(c, "]}");
389 conv_emit_raw(c, rootfs + rn, blen - head - rn);
390 if (c->over) {
391 snprintf(err, errlen, "rewritten image config exceeds %u bytes",
392 (unsigned) CONV_DOC_CAP);
393 return BRIX_OCI_REG_EPROTO;
394 }
395 return BRIX_OCI_REG_OK;
396}

398static int
399conv_config(brixoci_end_t *s, brixoci_end_t *d, const brix_oci_desc_t *m,
400 conv_t *c, brix_oci_desc_t *cfg, char *err, size_t errlen)
401{
402 const char *el;
403 size_t en, blen;
404 char dig[BRIX_OCI_DIGEST_STRLEN], *body;
405 int rc;
406
407 if (brix_json_get_raw(m->body, m->body_len, "config", &el, &en) != 1 ||
408 !brix_json_get_str(el, en, "digest", dig, sizeof(dig))) {
409 snprintf(err, errlen, "source manifest has no config descriptor");
410 return BRIX_OCI_REG_EPROTO;
411 }
412 if (!brix_json_get_str(el, en, "mediaType", cfg->mediatype,
413 sizeof(cfg->mediatype))) {
414 snprintf(cfg->mediatype, sizeof(cfg->mediatype), "%s", OCI_MT_CONFIG);
415 }
416 rc = conv_load(s, dig, &body, &blen, err, errlen);
417 if (rc != BRIX_OCI_REG_OK) {
418 return rc;
419 }
420 rc = conv_rewrite_config(c, body, blen, err, errlen);
421 free(body);
422 if (rc != BRIX_OCI_REG_OK) {
423 return rc;
424 }
425 cfg->body_len = c->len;
426 return conv_put_mem(d, c->doc, c->len, cfg->digest, sizeof(cfg->digest),
427 err, errlen);
428}

440/* The rewritten manifest: new config descriptor, new layer descriptors,
441 * and per layer the annotation a stargz snapshotter reads the TOC digest
442 * from before it fetches anything else. */
443static int
444conv_manifest(conv_t *c, const brix_oci_desc_t *src, brix_oci_desc_t *out,
445 const brix_oci_desc_t *cfg, char *err, size_t errlen)
446{
447 int i;
448
449 snprintf(out->mediatype, sizeof(out->mediatype), "%s",
450 conv_manifest_mt(src->mediatype));
451 c->len = 0;
452 c->over = 0;
453 c->doc[0] = '\0';
454 conv_emit(c, "{\"schemaVersion\":2,\"mediaType\":\"%s\","
455 "\"config\":{\"mediaType\":\"%s\",\"digest\":\"%s\","
456 "\"size\":%lld},\"layers\":[",
457 out->mediatype, cfg->mediatype, cfg->digest,
458 (long long) cfg->body_len);
459 for (i = 0; i < c->n; i++) {
460 conv_emit(c, "%s{\"mediaType\":\"%s\",\"digest\":\"%s\","
461 "\"size\":%lld,\"annotations\":{\"%s\":\"%s\"}}",
462 i ? "," : "", c->lay[i].mediatype, c->lay[i].digest,
463 c->lay[i].size, BRIX_STARGZ_TOC_ANNOTATION, c->lay[i].toc);
464 }
465 conv_emit(c, "]}");
466 if (c->over) {
467 snprintf(err, errlen, "rewritten manifest exceeds %u bytes",
468 (unsigned) CONV_DOC_CAP);
469 return BRIX_OCI_REG_EPROTO;
470 }
471 out->body = c->doc;
472 out->body_len = c->len;
473 return BRIX_OCI_REG_OK;
474}

478static int
479conv_run(brixoci_end_t *src, brixoci_end_t *dst, conv_t *c,
480 const char *bindtag, brix_oci_desc_t *m, char *digest_out,
481 size_t dlen, char *err, size_t errlen)
482{
483 brix_oci_desc_t cfg, out;
484 int rc;
485
486 memset(&cfg, 0, sizeof(cfg));
487 memset(&out, 0, sizeof(out));
488 rc = conv_layers(src, dst, m, c, err, errlen);
489 if (rc == BRIX_OCI_REG_OK) {
490 rc = conv_config(src, dst, m, c, &cfg, err, errlen);
491 }
492 if (rc == BRIX_OCI_REG_OK) {
493 rc = conv_manifest(c, m, &out, &cfg, err, errlen);
494 }
495 if (rc == BRIX_OCI_REG_OK) {
496 /* out.body aliases the scratch document; the binding copies it. */
497 rc = brixoci_put_manifest(dst, bindtag, &out, digest_out, dlen, err,
498 errlen);
499 }
500 return rc;
501}

503int
504brixoci_convert_run(brixoci_end_t *src, brixoci_end_t *dst,
505 const brixoci_opts_t *o, char *digest_out, size_t dlen,
506 char *err, size_t errlen)
507{
508 brix_oci_desc_t m;
509 conv_t c;
510 const char *seltag, *bindtag;
511 int rc;
512
513 memset(&c, 0, sizeof(c));
514 c.lay = calloc(CONV_MAX_LAYERS, sizeof(*c.lay));
515 c.doc = malloc(CONV_DOC_CAP);
516 if (c.lay == NULL || c.doc == NULL) {
517 free(c.lay);
518 free(c.doc);
519 snprintf(err, errlen, "out of memory");
520 return BRIX_OCI_REG_ETRANSPORT;
521 }
522 brixoci_xfer_tags(src, dst, &seltag, &bindtag);
523 if (o->tag != NULL) {
524 /* Layout → layout has no reference to carry over, and an untagged
525 * entry is one podman/skopeo cannot name. --tag is how the caller
526 * says what the converted image is called at the destination. */
527 bindtag = o->tag;
528 }
529 rc = brixoci_src_manifest(src, seltag, o, &m, err, errlen);
530 if (rc == BRIX_OCI_REG_OK && brixoci_is_index(&m)) {
531 snprintf(err, errlen, "source manifest is an image index — convert "
532 "one image at a time with --platform");
533 rc = BRIX_OCI_REG_EPROTO;
534 } else if (rc == BRIX_OCI_REG_OK) {
535 rc = conv_run(src, dst, &c, bindtag, &m, digest_out, dlen, err,
536 errlen);
537 }
538 brix_oci_desc_free(&m);
539 free(c.lay);
540 free(c.doc);
541 return rc;
542}
