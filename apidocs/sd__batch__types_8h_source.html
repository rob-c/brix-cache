1/*
2 * sd_batch_types.h - the by-value slot contracts (phase-107 C4 + C6).
3 *
4 * WHAT: brix_sd_unlink_batch_t, the one value the unlink_many/_cred vtable
5 * slots take: a borrowed window of confined non-directory keys with a
6 * per-key result vector. And brix_sd_precond_t, the typed publish
7 * precondition staged_commit carries (C6): create-if-absent and
8 * compare-and-publish, expressed once instead of a boolean per driver.
9 * WHY: A DeleteObjects of 1,000 keys over a remote backend was 1,000 signed
10 * round trips whose whole purpose was to avoid exactly that. One batch
11 * value keeps the slot signatures stable and the contract in one place.
12 * HOW: The VFS chunker (vfs_unlink_many.c) fills a window and flushes; a tree
13 * delete batches only WITHIN one directory level (a prefix cannot go
14 * before its children - every driver but mirage has CAP_DIRS, so the
15 * per-level rule IS the rule, not a special case).
16 */
17#ifndef BRIX_SD_BATCH_TYPES_H
18#define BRIX_SD_BATCH_TYPES_H

27/* The batch ceiling: no call ever carries more keys than this. The cap bit
28 * (BRIX_SD_CAP_BULK_DELETE, "my unlink_many is a real batch, not a loop")
29 * decides where the WINDOW COMES FROM, not its size: with the bit, the rmtree
30 * walker actively accumulates up to this many keys per flush - buffering is
31 * worth it because each flush is one round trip. Without the bit the walker
32 * does not accumulate (a looping slot saves no round trips), but a slot a
33 * no-bit driver still implements (pblock: one transaction; ceph: one ioctx)
34 * is handed the full flat batch by brix_vfs_delete_many, where the client
35 * supplied the keys and accumulation cost nothing. */
36#define BRIX_SD_BULK_DELETE_WINDOW 1000

38/*
39 * The batch value.
40 *
41 * paths borrowed, n entries, each already confined by the caller (the slot
42 * never re-resolves), none a directory - directories are removed
43 * singly after their children, outside any window.
44 * n <= the driver's window; the VFS chunker guarantees it.
45 * errs caller-allocated, n entries. The slot writes 0 for success and a
46 * positive errno per key. ENOENT is written as ENOENT, never silently
47 * mapped - the CALLER decides idempotency, exactly as the S3
48 * DeleteObjects handler does.
49 * done number of leading entries the slot actually attempted; a transport
50 * failure at key k sets done = k and the slot returns NGX_ERROR with
51 * errno set, leaving errs[k..n) untouched (the chunker pre-fills them
52 * with ECANCELED so an untried key is never reported as deleted).
53 *
54 * Slot return: NGX_OK - every key attempted, per-key results in errs (which
55 * may still hold failures); NGX_ERROR - the batch itself failed. A NULL slot
56 * means the VFS runs the per-key loop; behaviour is identical, only slower.
57 */
58typedef struct {
59 const char *const *paths;
60 size_t n;
61 int *errs;
62 size_t done;
63} brix_sd_unlink_batch_t;

65/*
66 * The publish precondition (phase-107 C6) — the second argument of the
67 * staged_commit slot, replacing the old `int noreplace` boolean the wave-7 ABI
68 * change retired (a boolean has nowhere to put an entity tag).
69 *
70 * kind == 0 is NONE, so a zeroed struct — and a NULL pointer, which every
71 * caller with no condition passes — is today's unconditional replace: the same
72 * fail-safe-by-zero discipline the mutation policy uses, applied to a
73 * different question. A caller who forgets to fill the struct gets the OLD
74 * semantics, never an accidental refusal.
75 *
76 * Refusals (slot returns NGX_ERROR):
77 * EEXIST ABSENT and the target exists -> kXR_ItExists / 412
78 * ECANCELED MATCH_* and the compare failed -> 412 Precondition Failed
79 * ENOTSUP the driver cannot evaluate this kind at all; there is no
80 * honest emulation, so the VFS refuses rather than pretending
81 *
82 * etag is BORROWED (pointer + length, not copied; the caller keeps it alive
83 * across the call — it is bytes+len rather than ngx_str_t so this header stays
84 * ngx-free for the libxrdproto plane, like the batch value above). The tag
85 * grammar is the shared weak "mtime-size" form of core/http/etag.h unless the
86 * driver's own storage carries a native tag (S3: the real ETag header).
87 *
88 * atomic is an OUTPUT: the slot sets it when the decision was made AT the
89 * storage and could not have raced (BRIX_SD_CAP_PRECOND advertises the same
90 * property statically; the field reports it per call, because on `http` it is
91 * a runtime fact about the origin, not a compile-time fact about the driver).
92 * The protocol layer must not claim RFC 7232 semantics when it is 0. The
93 * parameter is therefore non-const by contract — the OUT bit is the reason.
94 * Set on REFUSALS too: a 412 the origin answered, an EEXIST that
95 * RENAME_NOREPLACE raised, or a verdict reached inside pblock's transaction
96 * is a storage-decided refusal, and the C6 advisory metric
97 * (brix_vfs_precond_advisory_total) keys on this bit after a failed commit —
98 * a refusal that leaves it 0 was a check-then-act compare.
99 */
100typedef enum {
101 BRIX_SD_PRECOND_NONE = 0, /* replace unconditionally */
102 BRIX_SD_PRECOND_ABSENT, /* create-if-absent (the old noreplace) */
103 BRIX_SD_PRECOND_MATCH_ETAG, /* replace iff the entity tag matches */
104 BRIX_SD_PRECOND_MATCH_META /* replace iff (size, mtime) matches */
105} brix_sd_precond_kind_t;

107typedef struct {
108 brix_sd_precond_kind_t kind;
109 const char *etag; /* MATCH_ETAG only; borrowed */
110 size_t etag_len;
111 off_t size; /* MATCH_META only */
112 time_t mtime; /* MATCH_META only */
113 unsigned atomic:1; /* OUT: storage decided, atomically */
114} brix_sd_precond_t;

116/* True when `pre` asks for create-if-absent — the one question the pre-C6
117 * boolean could express; drivers that only support ABSENT key on this. */
118#define brix_sd_precond_absent(pre) ((pre) != NULL && (pre)->kind == BRIX_SD_PRECOND_ABSENT)

120/* Evaluate a MATCH_* precondition against the target's (size, mtime) — the
121 * shared body for every stat-grammar driver (posix, and the VFS's non-driver
122 * compat arm) so the etag comparison never forks. MATCH_ETAG compares the
123 * shared "mtime-size" tag of core/http/etag.h (hex mtime, hex size, quoted),
124 * honouring RFC 7232 weak comparison: an optional leading W/ on the caller's
125 * tag is ignored. MATCH_META compares the fields directly. Returns 0 when
126 * the precondition holds, -1 with errno = ECANCELED when it does not, and -1
127 * with errno = ENOTSUP for a kind this evaluator does not know (a new enum
128 * member must be taught here explicitly, never silently passed). NONE and
129 * ABSENT are not questions about (size, mtime) and are the caller's job. */
130static inline int
131brix_sd_precond_eval_stat(const brix_sd_precond_t *pre, off_t size,
132 time_t mtime)
133{
134 if (pre->kind == BRIX_SD_PRECOND_MATCH_META) {
135 if (pre->size == size && pre->mtime == mtime) {
136 return 0;
137 }
138 errno = ECANCELED;
139 return -1;
140 }
141 if (pre->kind == BRIX_SD_PRECOND_MATCH_ETAG) {
142 char tag[48];
143 const char *want = pre->etag;
144 size_t want_len = pre->etag_len;
145 int n;
146
147 if (want != NULL && want_len > 2
148 && want[0] == 'W' && want[1] == '/')
149 {
150 want += 2; /* RFC 7232 §2.3.2 weak comparison */
151 want_len -= 2;
152 }
153 n = snprintf(tag, sizeof(tag), "\"%lx-%llx\"",
154 (unsigned long) mtime, (unsigned long long) size);
155 if (want != NULL && n > 0 && (size_t) n == want_len
156 && memcmp(tag, want, want_len) == 0)
157 {
158 return 0;
159 }
160 errno = ECANCELED;
161 return -1;
162 }
163 errno = ENOTSUP;
164 return -1;
165}
