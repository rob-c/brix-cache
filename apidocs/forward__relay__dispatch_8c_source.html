44/*
45 * Deferred request dispatch and main proxy entry point.
46 *
47 * WHAT:
48 * Handles lazy upstream connection, request queuing during bootstrap,
49 * and bound-secondary channel lazy-open for file handle translation.
50 *
51 * WHY:
52 * The transparent XRootD proxy must connect to the upstream server lazily
53 * (on first non-bootstrap request) rather than eagerly.
54 * Requests arriving before bootstrap completion are saved in proxy->saved_req
55 * and dispatched after bootstrap finishes via this function.
56 * Bound-secondary channels (kXR_bind) may carry read/readv requests for
57 * handles with no upstream mapping — a synthetic kXR_open is issued first.
58 *
59 * HOW:
60 * Main entry brix_proxy_dispatch():
61 * - Creates proxy ctx on first call
62 * - Connects to upstream lazily
63 * - Saves pending requests during bootstrap state
64 * - Forwards via brix_proxy_forward_request when idle
65 * Dispatch pending (after bootstrap):
66 * - Checks bound-secondary lazy-open case
67 * - Sets forwarding state and flushes request
68 */

98/* ---- Detect a bound-secondary read that needs a synthetic upstream open ----
99 *
100 * WHAT: Returns the local file handle carried by a queued read-family request
101 * (kXR_read / kXR_pgread / kXR_readv) that currently has no upstream
102 * mapping, or -1 when no lazy open is required. A non-negative result means
103 * the caller must issue brix_proxy_lazy_open before forwarding.
104 *
105 * WHY: A kXR_bind secondary channel can carry read requests for handles that
106 * were opened on the primary channel and never mapped upstream on this
107 * channel; those must trigger a synthetic kXR_open first. Isolating this
108 * wire-shape probe as a pure query keeps the dispatch decision explicit and
109 * keeps the orchestrator's cyclomatic complexity within cap.
110 *
111 * HOW: 1. Non-bound channels never lazy-open — return -1 immediately.
112 * 2. Extract the file-handle byte by opcode: read/pgread at byte 4,
113 * readv at the first byte past the request header, honoring the same
114 * minimum-length guards as the original inline check.
115 * 3. Return the handle only when it is in range and its fh_map slot is
116 * still BRIX_PROXY_FH_FREE (no upstream mapping); otherwise -1.
117 */
118static int
119brix_proxy_pending_lazy_fh(brix_proxy_ctx_t *proxy, brix_ctx_t *ctx,
120 const u_char *req, size_t len, uint16_t reqid)
121{
122 int check_fh = -1;
123
124 if (!ctx->is_bound) {
125 return -1;
126 }
127
128 if ((reqid == kXR_read || reqid == kXR_pgread) && len >= 5) {
129 check_fh = (int)(unsigned char) req[4];
130 } else if (reqid == kXR_readv && len > XRD_REQUEST_HDR_LEN + 4) {
131 check_fh = (int)(unsigned char) req[XRD_REQUEST_HDR_LEN];
132 }
133
134 if (check_fh >= 0 && check_fh < BRIX_MAX_FILES
135 && proxy->fh_map[check_fh].fh_state == BRIX_PROXY_FH_FREE)
136 {
137 return check_fh;
138 }
139 return -1;
140}

142/* ---- Install the queued request as the write buffer and flush it upstream ----
143 *
144 * WHAT: Points the proxy write buffer at the saved request, resets the response
145 * accumulator, and flushes to the upstream connection. Returns NGX_OK on a
146 * full or partial send (the write handler completes a partial one) and
147 * NGX_ERROR after aborting the proxy on send or read-arm failure.
148 *
149 * WHY: The forward tail (buffer install, flush, read arm) is a single-purpose
150 * side-effecting step; extracting it keeps brix_proxy_dispatch_pending a
151 * flat sequence and its complexity within the per-function cap without
152 * altering the send ordering.
153 *
154 * HOW: 1. Set wbuf to the request bytes and mark the proxy XRD_PX_FORWARDING.
155 * 2. Zero the response-parse cursors for the reply that will follow.
156 * 3. Flush; on error abort and return NGX_ERROR.
157 * 4. If the send is partial, return NGX_OK (write handler resumes it).
158 * 5. Otherwise arm the read event; on failure abort and return NGX_ERROR.
159 */
160static ngx_int_t
161brix_proxy_pending_flush(brix_proxy_ctx_t *proxy, u_char *req, size_t len)
162{
163 proxy->wbuf = req;
164 proxy->wbuf_len = len;
165 proxy->wbuf_pos = 0;
166 proxy->state = XRD_PX_FORWARDING;
167
168 proxy->rhdr_pos = 0;
169 proxy->resp_dlen = 0;
170 proxy->resp_body = NULL;
171 proxy->resp_body_pos = 0;
172
173 if (brix_proxy_flush(proxy) == NGX_ERROR) {
174 brix_proxy_abort(proxy, "proxy: send deferred request failed");
175 return NGX_ERROR;
176 }
177 if (proxy->wbuf_pos < proxy->wbuf_len) {
178 return NGX_OK; /* write handler will complete the send */
179 }
180 if (ngx_handle_read_event(proxy->conn->read, 0) != NGX_OK) {
181 brix_proxy_abort(proxy, "proxy: read arm failed after deferred send");
182 return NGX_ERROR;
183 }
184 return NGX_OK;
185}

187ngx_int_t
188brix_proxy_dispatch_pending(brix_proxy_ctx_t *proxy)
189{
190 brix_ctx_t *ctx = proxy->client_ctx;
191 ngx_connection_t *c = proxy->client_conn;
192 u_char *req = proxy->saved_req;
193 size_t len = proxy->saved_req_len;
194 int saved_lfh = proxy->saved_local_fh;
195 uint16_t reqid;
196 int lazy_fh;
197 ClientRequestHdr *hdr;
198
199 if (req == NULL) {
200 return NGX_OK;
201 }
202
203 hdr = (ClientRequestHdr *)(void *) req;
204 reqid = ntohs(hdr->requestid);
205
206 /* Bound secondary lazy-open check: kXR_read / kXR_pgread / kXR_readv */
207 lazy_fh = brix_proxy_pending_lazy_fh(proxy, ctx, req, len, reqid);
208 if (lazy_fh >= 0) {
209 proxy->saved_req = NULL; /* ownership to lazy_open */
210 return brix_proxy_lazy_open(proxy, ctx, c, lazy_fh, req, len);
211 }
212
213 /* Normal deferred dispatch (e.g. kXR_open queued during bootstrap) */
214 proxy->fwd_reqid = reqid;
215 proxy->fwd_streamid[0] = hdr->streamid[0];
216 proxy->fwd_streamid[1] = hdr->streamid[1];
217 proxy->fwd_local_fh = saved_lfh;
218 proxy->fwd_streaming = 0;
219 proxy->fwd_payload_len = len > XRD_REQUEST_HDR_LEN
220 ? len - XRD_REQUEST_HDR_LEN : 0;
221 proxy->saved_req = NULL; /* ownership transferred to wbuf */
222
223 /* Tap: this is the first post-bootstrap request (e.g. the kXR_open), queued
224 * directly here rather than through brix_proxy_forward_request — emit it. */
225 {
226 brix_tap_frame_t tf;
227 if (brix_tap_decode_request(req, len, &tf) > 0) {
228 brix_tap_emit(&proxy->tap, &tf, BRIX_TAP_C2U, NULL, 0);
229 }
230 }
231
232 return brix_proxy_pending_flush(proxy, req, len);
233}

280/* brix_proxy_session_may_reselect — may this session be moved to another data
281 * server right now? (phase-115 W2.1)
282 *
283 * WHAT: True when the pinned upstream is quiescent — bootstrapped to
284 * XRD_PX_IDLE with no file handle open.
285 * WHY: The forwarding gate in handshake/dispatch.c must let a *selection-
286 * pinned* session fall through to the manager for path-carrying opcodes,
287 * so a later selection can name a different node; but only while nothing
288 * is riding the current upstream. One upstream per session is the
289 * invariant: with a handle open (or a request in flight) the session
290 * stays where it is and the request rides the pinned node, even when
291 * that node does not hold the path.
292 * HOW: state == XRD_PX_IDLE and proxy_has_open_handles() == 0. The 255
293 * open-pending sentinel counts as "not open" on purpose: a pending open
294 * always leaves state != XRD_PX_IDLE, so the first clause covers it.
295 */
296int
297brix_proxy_session_may_reselect(const brix_proxy_ctx_t *proxy)
298{
299 return proxy != NULL
300 && proxy->state == XRD_PX_IDLE
301 && !proxy_has_open_handles(proxy);
302}

304/* proxy_repin_idle — the CMS selected a data server other than the one this
305 * session is pinned to (phase-115 W2.1).
306 *
307 * WHAT: If the pinned upstream is idle with no open handles, drop it, pin the
308 * new target and start a fresh connect; the caller's save-or-forward tail
309 * then parks the request until that bootstrap completes. Otherwise the
310 * request rides the pinned upstream unchanged.
311 * WHY: A proxy session carries one upstream connection and one fh_map; a
312 * session with files open on server A cannot also be on server B. The
313 * idle case is the common one (a client opening files one after another
314 * across the cluster) and costs one reconnect, exactly like the
315 * redirect follow-through.
316 * HOW: Returns NGX_OK to continue dispatch (re-pinned or kept), NGX_DECLINED
317 * after answering the client with *sent_rc (connect failure; the proxy
318 * ctx is gone, the next request starts over). */
319static ngx_int_t
320proxy_repin_idle(brix_ctx_t *ctx, ngx_connection_t *c,
321 ngx_stream_brix_srv_conf_t *conf, const char *host, uint16_t port,
322 ngx_int_t *sent_rc)
323{
324 brix_proxy_ctx_t *proxy = ctx->proxy;
325
326 if (!proxy_pinned_elsewhere(proxy, host, port)) {
327 return NGX_OK;
328 }
329 if (proxy->state != XRD_PX_IDLE || proxy_has_open_handles(proxy)) {
330 ngx_log_error(NGX_LOG_INFO, c->log, 0,
331 "brix: cms select: session pinned to %V:%ud keeps its "
332 "upstream (busy or files open); %s:%ud not adopted",
333 &proxy->pinned_host, (unsigned) proxy->pinned_port,
334 host, (unsigned) port);
335 return NGX_OK;
336 }
337 if (proxy_pin_target(proxy, host, port) != NGX_OK) {
338 *sent_rc = brix_send_error(ctx, c, kXR_ArgInvalid,
339 "proxy: selected host too long");
340 return NGX_DECLINED;
341 }
342 ngx_log_error(NGX_LOG_INFO, c->log, 0,
343 "brix: cms select: re-pinning idle session to %s:%ud",
344 host, (unsigned) port);
345 brix_proxy_reset_upstream(proxy);
346
347 if (brix_proxy_connect(proxy, c, conf) != NGX_OK) {
348 ctx->proxy = NULL;
349 ctx->proxy_fail_count++;
350 *sent_rc = brix_send_error(ctx, c, kXR_IOError,
351 "proxy: upstream connect failed");
352 return NGX_DECLINED;
353 }
354 return NGX_OK;
355}

357/* proxy_lazy_init — create ctx->proxy and start the upstream connect on the
358 * first forwarded opcode. Returns NGX_OK with ctx->proxy set (bootstrap may
359 * still be in flight), or NGX_DECLINED after answering the client itself with
360 * *sent_rc = the brix_send_error result the caller must return. */
361static ngx_int_t
362proxy_lazy_init(brix_ctx_t *ctx, ngx_connection_t *c,
363 ngx_stream_brix_srv_conf_t *conf, const char *host, uint16_t port,
364 ngx_int_t *sent_rc)
365{
366 brix_proxy_ctx_t *proxy;
367 int i;
368
369 /*
370 * Loop guard: a permanently-rejecting upstream (bad credential, all
371 * upstreams down) would otherwise make every re-dispatch allocate a
372 * fresh proxy ctx on c->pool and reconnect, spinning the worker at
373 * event-loop rate and growing the connection pool without bound. Once
374 * this connection has burned through its failure budget, stop retrying
375 * and return a hard error instead of spawning another proxy.
376 */
377 if (ctx->proxy_fail_count >= BRIX_PROXY_MAX_CONN_FAILS) {
378 ngx_log_error(NGX_LOG_ERR, c->log, 0,
379 "xrootd proxy: %ui consecutive upstream failures on "
380 "this connection — failing request, not retrying",
381 ctx->proxy_fail_count);
382 *sent_rc = brix_send_error(ctx, c, kXR_IOError,
383 "proxy: upstream unavailable");
384 return NGX_DECLINED;
385 }
386
387 proxy = ngx_pcalloc(c->pool, sizeof(brix_proxy_ctx_t));
388 if (proxy == NULL) {
389 *sent_rc = brix_send_error(ctx, c, kXR_IOError,
390 "proxy: out of memory");
391 return NGX_DECLINED;
392 }
393 for (i = 0; i < BRIX_MAX_FILES; i++) {
394 proxy->fh_map[i].fh_state = BRIX_PROXY_FH_FREE;
395 }
396 proxy->client_ctx = ctx;
397 proxy->client_conn = c;
398 proxy->conf = conf;
399 brix_proxy_tap_init(proxy, c);
400 proxy->fwd_local_fh = -1;
401 proxy->saved_local_fh = -1;
402 proxy->reconnect_left = (int) conf->proxy.reconnect_attempts;
403 proxy->splice_pipe[0] = -1;
404 proxy->splice_pipe[1] = -1;
405
406 if (host != NULL && proxy_pin_target(proxy, host, port) != NGX_OK) {
407 *sent_rc = brix_send_error(ctx, c, kXR_ArgInvalid,
408 "proxy: selected host too long");
409 return NGX_DECLINED;
410 }
411 ctx->proxy = proxy;
412
413 if (brix_proxy_connect(proxy, c, conf) != NGX_OK) {
414 ctx->proxy = NULL;
415 /* Count synchronous connect/selection failures too (e.g. all
416 * upstreams down) so they are bounded by the same per-connection
417 * budget as async bootstrap aborts. */
418 ctx->proxy_fail_count++;
419 *sent_rc = brix_send_error(ctx, c, kXR_IOError,
420 "proxy: upstream connect failed");
421 return NGX_DECLINED;
422 }
423 return NGX_OK;
424}

426/* main dispatch entry point — see proxy.h; host == NULL means the configured
427 * upstream, otherwise the session is pinned to host:port (CMS selection). */
428ngx_int_t
429brix_proxy_dispatch_to(brix_ctx_t *ctx, ngx_connection_t *c,
430 ngx_stream_brix_srv_conf_t *conf, const char *host, uint16_t port)
431{
432 brix_proxy_ctx_t *proxy;
433 ngx_int_t sent_rc;
434
435 /* lazy initialisation: connect to upstream on first dispatch; a later
436 * CMS selection naming another server re-pins the session when idle */
437 if (ctx->proxy == NULL) {
438 if (proxy_lazy_init(ctx, c, conf, host, port, &sent_rc) != NGX_OK) {
439 return sent_rc;
440 }
441 } else if (host != NULL
442 && proxy_repin_idle(ctx, c, conf, host, port, &sent_rc) != NGX_OK)
443 {
444 return sent_rc;
445 }
446
447 proxy = ctx->proxy;
448
449 /* if bootstrap is still in progress, save this request */
450 if (proxy->state != XRD_PX_IDLE) {
451 size_t total = XRD_REQUEST_HDR_LEN + ctx->recv.cur_dlen;
452 u_char *saved;
453
454 saved = ngx_alloc(total, c->log);
455 if (saved == NULL) {
456 return brix_send_error(ctx, c, kXR_IOError,
457 "proxy: out of memory saving request");
458 }
459 ngx_memcpy(saved, ctx->recv.hdr_buf, XRD_REQUEST_HDR_LEN);
460 if (ctx->recv.cur_dlen > 0 && ctx->recv.payload != NULL) {
461 ngx_memcpy(saved + XRD_REQUEST_HDR_LEN, ctx->recv.payload, ctx->recv.cur_dlen);
462 }
463
464 proxy->saved_req = saved;
465 proxy->saved_req_len = total;
466
467 /* Pre-allocate a local fh for a deferred open */
468 if (ctx->recv.cur_reqid == kXR_open) {
469 int local_fh = brix_proxy_alloc_local_fh(proxy);
470 if (local_fh < 0) {
471 ngx_free(saved);
472 proxy->saved_req = NULL;
473 return brix_send_error(ctx, c, kXR_IOError,
474 "proxy: no free file handles");
475 }
476 proxy->fh_map[local_fh].fh_state = BRIX_PROXY_FH_PENDING;
477 proxy->saved_local_fh = local_fh;
478 } else {
479 proxy->saved_local_fh = -1;
480 }
481
482 /* Suspend client read loop until bootstrap + deferred dispatch complete */
483 ctx->state = XRD_ST_PROXY;
484 return NGX_OK;
485 }
486
487 /* upstream is ready: forward the request now */
488 return brix_proxy_forward_request(proxy, ctx, c);
489}
