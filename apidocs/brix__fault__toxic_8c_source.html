1/*
2 * brix_fault_toxic.c — named, stackable, per-direction fault toxics (C1).
3 *
4 * WHAT: a table of named single-effect toxics (`toxic add <name> <type> <value>
5 * [dir]`), each a per-direction fault that composes on top of the core's
6 * base levers. `toxic remove <name>` drops one by name, the core `clear`
7 * verb empties the table, and `toxic list [json]` reports them.
8 *
9 * WHY: the base g_up/g_down levers are a single blended knob-set — you cannot
10 * add "100 ms latency" and later remove exactly that without remembering
11 * and undoing the delta. Named toxics make each discrete effect
12 * independently addressable (the Toxiproxy model), which is what a test
13 * harness wants when it layers and peels faults around a scenario; two of
14 * a kind STACK, where the flat single-field lever held exactly one.
15 *
16 * HOW: a fixed-size table under one mutex. A lock-free atomic direction mask
17 * (g_dir_mask) lets the per-buffer relay hot path skip composition entirely
18 * when nothing targets its direction; only when a toxic is present does
19 * fp_toxic_compose take the lock and fold the active toxics into the
20 * caller's snapshot. Composition rules are fixed and documented below so a
21 * stack of toxics has one well-defined meaning.
22 */

35/* dir: 1 = up (client→upstream), 2 = down (upstream→client), 3 = both. */
36struct fp_toxic {
37 char name[FP_TOXIC_NAME];
38 char type[FP_TOXIC_TYPE]; /* the effect keyword, echoed back verbatim */
39 int dir;
40 lever_t lv; /* exactly one field set; the rest are neutral */
41 int in_use;
42};

44static struct fp_toxic g_toxics[FP_TOXIC_MAX];
45static pthread_mutex_t g_toxic_lock = PTHREAD_MUTEX_INITIALIZER;

47/* Union of the dir masks of all live toxics (bit0=up, bit1=down). Read lock-free
48 * by the hot path; only ever written under g_toxic_lock. */
49static volatile int g_dir_mask = 0;
50
51/* Recompute g_dir_mask from the table. Caller holds g_toxic_lock. */
52static void
53recalc_mask(void)
54{
55 int m = 0;
56 for (int i = 0; i < FP_TOXIC_MAX; i++) {
57 if (g_toxics[i].in_use) {
58 m |= g_toxics[i].dir;
59 }
60 }
61 __atomic_store_n(&g_dir_mask, m, __ATOMIC_RELEASE);
62}

64/* Map a direction token to a mask (1/2/3); default to both (3) when absent or
65 * unrecognised — the direction is an optional trailing token. */
66static int
67dir_mask_of(const char *tok)
68{
69 if (strcmp(tok, "up") == 0) return 1;
70 if (strcmp(tok, "down") == 0) return 2;
71 return 3;
72}

81/* The settable lever fields, as a descriptor table rather than a strcmp ladder
82 * (coding-standards §8.6): a toxic type is a name, a slot, and how to read its
83 * operand. Adding a lever field is one row here and nothing else. */
84enum { FLD_INT, FLD_PPM, FLD_LONG };

87 const char *name;

91 { "latency", offsetof(lever_t, latency_ms), FLD_INT },
92 { "jitter", offsetof(lever_t, jitter_ms), FLD_INT },
93 { "chunk", offsetof(lever_t, chunk_bytes), FLD_INT },
94 { "rate", offsetof(lever_t, rate_kbps), FLD_INT },
95 { "drip_bytes", offsetof(lever_t, drip_bytes), FLD_INT },

108/* Set the single field named by `type` on `lv` to `value`. Probability types
109 * take a percent string (× 10000 → ppm), matching the core's lever grammar.
110 * Returns 0 on success, -1 if the type name is unknown. */
111static int
112set_field(lever_t *lv, const char *type, const char *value)
113{
114 for (size_t i = 0; i < sizeof(TOXIC_FIELDS) / sizeof(TOXIC_FIELDS[0]); i++) {
115 if (strcmp(type, TOXIC_FIELDS[i].name) != 0) {
116 continue;
117 }
118 void *slot = (char *) lv + TOXIC_FIELDS[i].off;
119 switch (TOXIC_FIELDS[i].kind) {
120 case FLD_LONG:
121 *(volatile long *) slot = atol(value);
122 break;
123 case FLD_PPM: /* operator writes a percentage; the lever holds ppm */
124 *(volatile int *) slot = (int) (strtod(value, NULL) * 10000.0 + 0.5);
125 break;
126 default:
127 *(volatile int *) slot = atoi(value);
128 break;
129 }
130 return 0;
131 }
132 return -1; /* unknown toxic type */
133}

153/* toxic add <name> <type> <value> [dir] */
154static int
155toxic_add(char *rest, char *reply, size_t rsz)
156{
157 char name[FP_TOXIC_NAME] = "", type[FP_TOXIC_TYPE] = "";
158 char value[32] = "", dir[8] = "";
159 int nf = sscanf(rest, "%31s %23s %31s %7s", name, type, value, dir);
160 if (nf < 3) {
161 reply_set(reply, rsz, "err: usage: toxic add <name> <type> <value> [up|down|both]\n");
162 return 1;
163 }
164
165 /* Validate the effect type before touching the table (so a bad type never
166 * displaces a duplicate-name or capacity verdict). */
167 lever_t lv;
168 memset(&lv, 0, sizeof lv);
169 if (set_field(&lv, type, value) != 0) {
170 reply_set(reply, rsz, "err: unknown toxic type\n");
171 return 1;
172 }
173 int dm = (nf >= 4) ? dir_mask_of(dir) : 3;
174
175 pthread_mutex_lock(&g_toxic_lock);
176 if (find_toxic(name) >= 0) {
177 pthread_mutex_unlock(&g_toxic_lock);
178 reply_set(reply, rsz, "err: exists\n");
179 return 1;
180 }
181 int slot = -1;
182 for (int i = 0; i < FP_TOXIC_MAX; i++) {
183 if (!g_toxics[i].in_use) { slot = i; break; }
184 }
185 if (slot < 0) {
186 pthread_mutex_unlock(&g_toxic_lock);
187 reply_set(reply, rsz, "err: too many toxics\n");
188 return 1;
189 }
190 snprintf(g_toxics[slot].name, FP_TOXIC_NAME, "%s", name);
191 snprintf(g_toxics[slot].type, FP_TOXIC_TYPE, "%s", type);
192 g_toxics[slot].dir = dm;
193 g_toxics[slot].lv = lv;
194 g_toxics[slot].in_use = 1;
195 recalc_mask();
196 pthread_mutex_unlock(&g_toxic_lock);
197
198 fp_reply(reply, rsz, "added %s\n", name);
199 return 1;
200}

202/* toxic remove <name> */
203static int
204toxic_remove(char *rest, char *reply, size_t rsz)
205{
206 char name[FP_TOXIC_NAME] = "";
207 if (sscanf(rest, "%31s", name) != 1) {
208 reply_set(reply, rsz, "err: usage: toxic remove <name>\n");
209 return 1;
210 }
211 pthread_mutex_lock(&g_toxic_lock);
212 int slot = find_toxic(name);
213 if (slot >= 0) {
214 g_toxics[slot].in_use = 0;
215 recalc_mask();
216 }
217 pthread_mutex_unlock(&g_toxic_lock);
218 if (slot < 0) {
219 reply_set(reply, rsz, "err: no such toxic\n");
220 } else fp_reply(reply, rsz, "removed %s\n", name);
221 return 1;
222}

224/* toxic list [json] — human head line is "toxics=N"; json is the machine form. */
225static int
226toxic_list(char *rest, char *reply, size_t rsz)
227{
228 if (!reply || !rsz) {
229 return 1;
230 }
231 int want_json = (strncmp(rest, "json", 4) == 0);
232
233 pthread_mutex_lock(&g_toxic_lock);
234 size_t o = 0;
235 if (want_json) {
236 o += (size_t) snprintf(reply + o, rsz - o, "{\"toxics\":[");
237 int first = 1;
238 for (int i = 0; i < FP_TOXIC_MAX && o < rsz; i++) {
239 if (!g_toxics[i].in_use) continue;
240 o += (size_t) snprintf(reply + o, rsz - o,
241 "%s{\"name\":\"%s\",\"type\":\"%s\",\"dir\":\"%s\"}",
242 first ? "" : ",",
243 g_toxics[i].name, g_toxics[i].type,
244 dir_name(g_toxics[i].dir));
245 first = 0;
246 }
247 if (o < rsz) o += (size_t) snprintf(reply + o, rsz - o, "]}\n");
248 } else {
249 int n = 0;
250 for (int i = 0; i < FP_TOXIC_MAX; i++) {
251 if (g_toxics[i].in_use) n++;
252 }
253 o += (size_t) snprintf(reply + o, rsz - o, "toxics=%d\n", n);
254 for (int i = 0; i < FP_TOXIC_MAX && o < rsz; i++) {
255 if (!g_toxics[i].in_use) continue;
256 o += (size_t) snprintf(reply + o, rsz - o, " %s %s %s\n",
257 g_toxics[i].name, g_toxics[i].type,
258 dir_name(g_toxics[i].dir));
259 }
260 }
261 pthread_mutex_unlock(&g_toxic_lock);
262 return 1;
263}

288/* Additive stack for delays/probabilities; tightest-wins for bottleneck fields. */
289static void
290compose_one(lever_t *s, const lever_t *t)
291{
292 s->latency_ms += t->latency_ms;
293 s->jitter_ms += t->jitter_ms;
294 s->drip_ms += t->drip_ms;
295 s->delayfirst_ms += t->delayfirst_ms;
296 if (t->reorder_ms > 0) {
297 s->reorder_ms += t->reorder_ms;
298 }
299
300#define ADD_PPM(f) do { long v = (long) s->f + t->f; s->f = v > 1000000 ? 1000000 : (int) v; } while (0)
301 ADD_PPM(lossy_ppm);
302 ADD_PPM(corrupt_ppm);
303 ADD_PPM(dup_ppm);
304 ADD_PPM(reorder_ppm);
305 ADD_PPM(drop_ppm);
306 ADD_PPM(repeat_ppm);
307#undef ADD_PPM
308
309 /* Bottleneck / fragmentation: the tightest non-zero value wins. */
310#define TIGHTEN(f) do { if (t->f > 0 && (s->f == 0 || t->f < s->f)) s->f = t->f; } while (0)
311 TIGHTEN(rate_kbps);
312 TIGHTEN(chunk_bytes);
313 TIGHTEN(drip_bytes);
314#undef TIGHTEN
315 if (t->truncate_at > 0 && (s->truncate_at == 0 || t->truncate_at < s->truncate_at)) {
316 s->truncate_at = t->truncate_at;
317 }
318}

320void
321fp_toxic_compose(int is_up, lever_t *snap)
322{
323 int want = is_up ? 1 : 2;
324 pthread_mutex_lock(&g_toxic_lock);
325 for (int i = 0; i < FP_TOXIC_MAX; i++) {
326 if (g_toxics[i].in_use && (g_toxics[i].dir & want)) {
327 compose_one(snap, &g_toxics[i].lv);
328 }
329 }
330 pthread_mutex_unlock(&g_toxic_lock);
331}
