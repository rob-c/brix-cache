34/* ---- Remove terminal line separators from a query value ----
35 *
36 * WHAT: Returns the length after stripping trailing CR/LF bytes in place.
37 * WHY: Servers differ only in whether the final Qconfig line has a newline.
38 * HOW: Walk backward over CR/LF and terminate at the first content boundary.
39 */
40static size_t
41trim_lines(char *value)
42{
43 size_t length = strlen(value);
44
45 while (length > 0
46 && (value[length - 1] == '\n' || value[length - 1] == '\r')) length--;
47 value[length] = '\0';
48 return length;
49}

83/* ---- Grade ordered multi-key Qconfig output ----
84 *
85 * WHAT: Exercises QRY-001 and QRY-014 with independently learned line values.
86 * WHY: Counting lines alone cannot prove request order or unknown-key continuity.
87 * HOW: Query each key alone, query the pair, and compare the exact composition.

95/* WHAT: Compare independently observed key/role values with their paired reply.
96 * WHY: Counting lines cannot prove key order or unknown-key continuity.
97 * HOW: Query both controls and the pair under the same budget, then compare
98 * the exact normalized composition, retaining error and successful evidence.
99 */
100static int
101query_config_role_pair(xrd_conform_execution *execution, brix_conn *connection,
102 const char *first_key, const char *pair, int require_first,
103 qconfig_pair_result *result)
104{
105 char first[256], second[256];
106 int written;
107
108 memset(result, 0, sizeof(*result));
109 if (query_config(execution, connection, first_key, first, sizeof(first),
110 &result->status) != 0
111 || query_config(execution, connection, "role", second, sizeof(second),
112 &result->status) != 0
113 || query_config(execution, connection, pair, result->combined,
114 sizeof(result->combined), &result->status) != 0) {
115 return 0;
116 }
117 if (require_first && first[0] == '\0') {
118 return 0;
119 }
120 written = snprintf(result->expected, sizeof(result->expected), "%s\n%s",
121 first, second);
122 return written >= 0 && (size_t) written < sizeof(result->expected)
123 && strcmp(result->combined, result->expected) == 0;
124}
