7/* WHAT: Match one XML element's local name at a '<' byte.
8 * WHY: WebDAV servers freely choose namespace prefixes and letter case.
9 * HOW: Skip slash/prefix, compare the local name and require a name boundary. */
10static int
11xml_local_name(const char *cursor, const char *name, int closing)
12{
13 const char *local;
14 const char *colon;
15 const char *end;
16 size_t length = strlen(name);
17
18 if (cursor == NULL || *cursor != '<') return 0;
19 cursor++;
20 if ((*cursor == '/') != closing) return 0;
21 if (closing) cursor++;
22 end = cursor;
23 while (*end != '\0' && *end != '>' && *end != ' ' && *end != '\t'
24 && *end != '\r' && *end != '\n') end++;
25 colon = memchr(cursor, ':', (size_t) (end - cursor));
26 local = colon == NULL ? cursor : colon + 1;
27 return (size_t) (end - local) == length
28 && strncasecmp(local, name, length) == 0;
29}

31/* WHAT: Count opening elements with one local name.
32 * WHY: DAV-001 requires exactly one response independent of prefix spelling.
33 * HOW: Walk bounded NUL-terminated decoded XML and inspect each '<'. */
34size_t
35xrd_conform_webdav_xml_count(const char *body, const char *name)
36{
37 const char *cursor = body;
38 size_t count = 0;
39
40 if (body == NULL) return 0;
41 while ((cursor = strchr(cursor, '<')) != NULL) {
42 if (xml_local_name(cursor, name, 0)) count++;
43 cursor++;
44 }
45 return count;
46}

48/* WHAT: Return the text between a named opening and closing element.
49 * WHY: Content-length comparison must parse digits only inside its property.
50 * HOW: Locate a prefix-agnostic opening tag, then the next matching close tag. */
51static int
52xml_element_text(const char *body, const char *name, char *out, size_t capacity)
53{
54 const char *cursor = body;
55 const char *text;
56 const char *close;
57 size_t length;
58
59 if (body == NULL || capacity == 0) return 0;
60 while ((cursor = strchr(cursor, '<')) != NULL
61 && !xml_local_name(cursor, name, 0)) cursor++;
62 if (cursor == NULL || (text = strchr(cursor, '>')) == NULL) return 0;
63 text++;
64 close = text;
65 while ((close = strchr(close, '<')) != NULL
66 && !xml_local_name(close, name, 1)) close++;
67 if (close == NULL) return 0;
68 length = (size_t) (close - text);
69 if (length == 0 || length >= capacity) return 0;
70 memcpy(out, text, length);
71 out[length] = '\0';
72 return 1;
73}

75/* WHAT: Validate every embedded HTTP status line in a multistatus body.
76 * WHY: DAV-009 must not accept malformed or missing per-resource outcomes.
77 * HOW: Require one status per response and a three-digit 2xx..5xx code. */
78static int
79xml_statuses_valid(const char *body)
80{
81 const char *cursor = body;
82 size_t statuses = 0;
83 size_t responses = xrd_conform_webdav_xml_count(body, "response");
84
85 if (body == NULL || responses == 0) return 0;
86 while ((cursor = strcasestr(cursor, "HTTP/1.1 ")) != NULL) {
87 const char *code = cursor + 9;
88

90 || code[1] < '0' || code[1] > '9'
91 || code[2] < '0' || code[2] > '9') return 0;
92 statuses++;
93 cursor = code + 3;
94 }

96}
