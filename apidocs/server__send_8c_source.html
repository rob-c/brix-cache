23/* ---- Acknowledge an admitted, authenticated CMS LOGIN ----
24 * WHAT: Return NGX_OK only after the complete manager LOGIN reply is sent.
25 * WHY: Reference nodes wait for this reply before processing state probes;
26 * registration without it creates false NotFound results and reconnects.
27 * HOW: 1. Identify this listening endpoint. 2. Encode a conservative manager
28 * reply with no optional features or storage capacity. 3. Send via frame IO.
29 */
30ngx_int_t
31brix_cms_srv_send_login_ack(brix_cms_srv_ctx_t *ctx)
32{
33 brix_cms_login_fields_t fields = {0};
34 u_char payload[BRIX_CMS_LOGIN_PAYLOAD_BUF];
35 u_char sid[BRIX_CMS_SID_BUF];
36 u_char *end;
37 size_t length;
38
39 if (ctx->c == NULL
40 || ngx_connection_local_sockaddr(ctx->c, NULL, 0) != NGX_OK) {
41 return NGX_ERROR;
42 }
43 end = ngx_snprintf(sid, sizeof(sid), "brix-manager@%s:%ui",
44 brix_host_identity(),
45 (ngx_uint_t) ngx_inet_get_port(ctx->c->local_sockaddr));
46 fields.sid.data = sid;
47 fields.sid.len = (size_t) (end - sid);
48 length = brix_cms_login_encode(payload, sizeof(payload), &fields);
49 if (length == 0) {
50 return NGX_ERROR;
51 }
52 return brix_cms_srv_send_frame(ctx, 0, CMS_RR_LOGIN, 0, payload, length);
53}

102/*
103 * brix_cms_srv_send_load — kYR_load reply to a node's kYR_usage query.
104 *
105 * WHAT: Sends a CMS_RR_LOAD frame echoing the request streamid, carrying the
106 * 6-byte load vector (cpu,net,xeq,mem,pag,dsk) and the aggregate free
107 * space in MB. WHY: Stock cmsd answers do_Usage with its current load
108 * report so a querying node/peer can weigh this manager; phase-89 W1
109 * brings the manager side to parity. HOW: Byte-exact with the node-side
110 * heartbeat in send.c — theLoad is a bare [2-byte len][6 bytes] Pup
111 * char-blob (NO scalar tag), dskFree a tagged int; total dlen = 13.
112 */
113ngx_int_t
114brix_cms_srv_send_load(brix_cms_srv_ctx_t *ctx, uint32_t streamid,
115 const u_char load6[6], uint32_t free_mb)
116{
117 u_char payload[16];
118 u_char *cursor;
119
120 cursor = payload;
121 ngx_brix_cms_put16(cursor, 6); /* blob length: 6 load bytes */
122 cursor += 2;
123 ngx_memcpy(cursor, load6, 6);
124 cursor += 6;
125 cursor = ngx_brix_cms_put_int(cursor, free_mb);
126
127 return brix_cms_srv_send_frame(ctx, streamid, CMS_RR_LOAD, 0,
128 payload, (size_t) (cursor - payload));
129}

159/*
160 * brix_cms_srv_send_try — §2.9/§2.13: redirect a node at login (kYR_try).
161 *
162 * WHAT: Sends a kYR_try frame on streamid 0 naming ONE alternate manager:
163 * payload = NUL-terminated host + 2-byte big-endian port — the same
164 * shape cms_frame_redirect parses on the receiving node.
165 * WHY: Two producers share it: the ManTree-style supervisor offload (a
166 * manager at max_direct redirecting a new server to a supervisor) and
167 * a blacklist entry's `redirect <host:port>` action. streamid 0 marks
168 * it as unsolicited — the node's redirect handler treats a 0-streamid
169 * try as "re-dial there", never as a locate answer.
170 * HOW: Pack host+NUL+port into a stack buffer; synchronous frame send.
171 */
172ngx_int_t
173brix_cms_srv_send_try(brix_cms_srv_ctx_t *ctx, const char *host,
174 uint16_t port)
175{
176 u_char payload[BRIX_CMS_STATE_SAFE_BUF + 3];
177 size_t host_len = ngx_strlen(host);
178
179 if (host_len == 0 || host_len >= sizeof(payload) - 3) {
180 return NGX_ERROR;
181 }
182 ngx_memcpy(payload, host, host_len);
183 payload[host_len] = '\0';
184 payload[host_len + 1] = (u_char) (port >> 8);
185 payload[host_len + 2] = (u_char) (port & 0xff);
186
187 return brix_cms_srv_send_frame(ctx, 0, CMS_RR_TRY, 0, payload,
188 host_len + 3);
189}
