48char *
49webdav_merge_base_conf(ngx_conf_t *cf, ngx_http_brix_webdav_loc_conf_t *prev,
50 ngx_http_brix_webdav_loc_conf_t *conf)
51{
52 /* Unified directives (brix_export, brix_cache_store, ...) live in the
53 * common module; pull the merged values for this location into our
54 * embedded preamble before protocol merge applies defaults. */
55 brix_http_common_adopt(cf, &conf->common);
56
57 /* Shared common.* preamble (incl. hard read-only enforcement + pmark);
58 * WebDAV exports default to "/" (pure cache nodes serve the whole ns). */
59 if (ngx_http_brix_shared_merge(cf, &prev->common, &conf->common, "/")
60 != NGX_CONF_OK)
61 {
62 return NGX_CONF_ERROR;
63 }
64 /* cache_root merge -> shared_merge (common.cache_root), phase-101 W8. */
65 /* tcp_congestion / trusted_ca(_dir) / verify_depth merges ->
66 * shared_merge (phase-105 W2/W3.5); vomsdir/voms_cert_dir (W4),
67 * authdb_rules (W5.2), vo_rules + crl/crl_mode/signing_policy (W4)
68 * likewise live there. */
69 ngx_conf_merge_uint_value(conf->auth, prev->auth,
70 WEBDAV_AUTH_OPTIONAL);
71 /* protbind is inherited whole (never merged element-wise) in
72 * ngx_http_brix_shared_merge (common.protbind) — phase-101 W4. The rule ORDER
73 * decides which template matches first, so a location that states any rule of
74 * its own owns the whole policy. */
75 /* XrdAcc merge moved into ngx_http_brix_shared_merge (common.acc), W2. */
76 ngx_conf_merge_value(conf->proxy_certs, prev->proxy_certs, 0);
77 /* client_ca_store merge -> shared_merge (common.client_ca_store), W2-105. */
78 ngx_conf_merge_value(conf->tape_rest, prev->tape_rest, 0);
79 /* Uploads staged + resumable by DEFAULT. Set brix_webdav_upload_resume
80 * off to opt out. */
81 /* upload_resume merge moved to shared_merge (common.upload_resume), W4. */
82 /* upload_stage_dir merge -> shared_merge (W4). */
83 ngx_http_brix_webdav_tpc_merge_loc_conf(conf, prev);
84 BRIX_MERGE_PTR(conf, prev, cors_origins);
85 BRIX_MERGE_PTR(conf, prev, header2cgi); /* §6.5 header->cgi map */
86 ngx_conf_merge_value(conf->cors_credentials, prev->cors_credentials, 0);
87 ngx_conf_merge_sec_value(conf->cors_max_age, prev->cors_max_age, 86400);
88 ngx_conf_merge_sec_value(conf->lock_timeout, prev->lock_timeout, 600);
89 ngx_conf_merge_value(conf->lock_startup_sweep, prev->lock_startup_sweep, 0);
90 ngx_conf_merge_value(conf->http_query_token, prev->http_query_token, 1);
91 ngx_conf_merge_value(conf->macaroon_max_validity,
92 prev->macaroon_max_validity, 86400);
93 ngx_conf_merge_str_value(conf->macaroon_location, prev->macaroon_location, "");
94 ngx_conf_merge_str_value(conf->checksum_on_write, prev->checksum_on_write, "");
95 ngx_conf_merge_uint_value(conf->checksum_xattr_format,
96 prev->checksum_xattr_format, BRIX_CKS_FMT_TEXT);
97 if (conf->checksum_xattr_format != BRIX_CKS_FMT_TEXT) {
98 /* §8.x: stock-interoperable binary XrdCksData write format (process-wide). */
99 brix_integrity_set_xattr_format(conf->checksum_xattr_format);
100 }
101 ngx_conf_merge_value(conf->dig_enable, prev->dig_enable, 0);
102 ngx_conf_merge_value(conf->require_digest, prev->require_digest, 0);
103
104 /* §6.6: HTML directory listing on GET (listingdeny/listingredir analog). */
105 ngx_conf_merge_value(conf->authz_endpoint, prev->authz_endpoint, 0);
106 ngx_conf_merge_str_value(conf->accel_redirect, prev->accel_redirect, "");
107 ngx_conf_merge_value(conf->html_listing, prev->html_listing, 0);
108 /* §6.11 maxdelay merge -> shared_merge (common.max_delay), phase-105 W3. */
109 ngx_conf_merge_str_value(conf->listing_redirect,
110 prev->listing_redirect, "");
111
112 /* §6.1: HTTP redirect-to-dataserver + signed-CGI handoff. */
113 ngx_conf_merge_value(conf->redirect_dataserver,
114 prev->redirect_dataserver, 0);
115 ngx_conf_merge_value(conf->redirect_port, prev->redirect_port, 0);
116 ngx_conf_merge_uint_value(conf->redirect_scheme, prev->redirect_scheme,
117 BRIX_WEBDAV_RDR_HTTP);
118 ngx_conf_merge_value(conf->redirect_window, prev->redirect_window, 120);
119 ngx_conf_merge_str_value(conf->http_secretkey, prev->http_secretkey, "");
120 ngx_conf_merge_ptr_value(conf->dig_exports, prev->dig_exports, NULL);
121 ngx_conf_merge_str_value(conf->dig_auth_file, prev->dig_auth_file, "");
122 /* delegation_endpoint merge -> shared_merge (common.delegation_endpoint),
123 * phase-105 W2. */
124
125 /* Phase 20 caches/limits merge -> shared_merge (common.token_cache_kv /
126 * common.rate_limit / common.rl_rules), phase-105 W1. */
127
128 /* Phase 21 introspection merges -> shared_merge (common.introspect_*),
129 * phase-105 W4.1. */
130 if (conf->revoke_kv == NULL) {
131 conf->revoke_kv = prev->revoke_kv;
132 }
133
134 ngx_conf_merge_ptr_value(conf->open_file_cache,
135 prev->open_file_cache, NULL);
136 ngx_conf_merge_uint_value(conf->open_file_cache_valid,
137 prev->open_file_cache_valid, 60);
138 ngx_conf_merge_uint_value(conf->open_file_cache_min_uses,
139 prev->open_file_cache_min_uses, 1);
140 ngx_conf_merge_value(conf->open_file_cache_errors,
141 prev->open_file_cache_errors, 0);
142 ngx_conf_merge_value(conf->open_file_cache_events,
143 prev->open_file_cache_events, 0);
144
145 /* pwd_file merge moved into ngx_http_brix_shared_merge (common.pwd_file), W4. */
146 /* token quartet merges -> shared_merge (W4); the clock-skew [0,300]
147 * clamp moved into shared_merge too (phase-105 W8) so s3/cvmfs enforce it. */
148 /* token_config merge -> shared_merge (common.token_config), W4. */
149 ngx_conf_merge_ptr_value(conf->token_registry, prev->token_registry, NULL);
150 /* macaroon_secret merge -> shared_merge (W4). */
151
152
153 return NGX_CONF_OK;
154}

208{
209 char cred_z[256];
210 char bearer[4096];

227 if (brix_credential_to_backend_cred(cred, bearer, sizeof(bearer),

231 }

300/*
301 * WHAT: refuse an auth optional/required export that configures no credential
302 * verifier at all. Returns NGX_CONF_OK or NGX_CONF_ERROR.
303 * WHY: without one, every client is rejected — the misconfiguration should
304 * fail `nginx -t`, not surface as 403s in production.
305 * HOW: accept any of an x509 CA (cadir/cafile), a token verifier (JWKS /
306 * issuer registry / macaroon secret), a Basic password db, or (§6.1) the
307 * shared redirect secret (brix_webdav_secretkey) that authenticates a
308 * signed redirect handoff.
309 */
310static char *
311webdav_require_verifier(ngx_conf_t *cf,
312 ngx_http_brix_webdav_loc_conf_t *conf)
313{
314 if ((conf->auth == WEBDAV_AUTH_OPTIONAL
315 || conf->auth == WEBDAV_AUTH_REQUIRED)
316 && conf->common.trusted_ca_dir.len == 0 && conf->common.trusted_ca.len == 0
317 && conf->common.token_jwks.len == 0 && conf->common.token_config.len == 0
318 && conf->common.token_macaroon_secret.len == 0 && conf->common.pwd_file.len == 0
319 && conf->http_secretkey.len == 0)
320 {
321 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
322 "brix_webdav: auth optional/required needs a credential verifier "
323 "— brix_trusted_ca / brix_trusted_ca_dir, brix_token_jwks / "
324 "brix_token_config / brix_macaroon_secret, or brix_pwd_file");
325 return NGX_CONF_ERROR;
326 }
327
328 return NGX_CONF_OK;
329}

331/*
332 * WHAT: stat-check the three x509 trust-material paths. Returns NGX_CONF_OK
333 * or NGX_CONF_ERROR.
334 * WHY: an unreadable CA directory yields an empty trust store, which fails
335 * every client certificate at request time instead of at config time.
336 * HOW: each path is checked for the exact kind (directory / regular file /
337 * either) and the access modes the verifier will need.
338 */
339static char *
340webdav_validate_ca_paths(ngx_conf_t *cf,
341 ngx_http_brix_webdav_loc_conf_t *conf)
342{
343 if (webdav_validate_path(cf, "brix_trusted_ca_dir", &conf->common.trusted_ca_dir,
344 WEBDAV_PATH_DIRECTORY, R_OK | X_OK) != NGX_OK
345 || webdav_validate_path(cf, "brix_trusted_ca", &conf->common.trusted_ca,
346 WEBDAV_PATH_REGULAR_FILE, R_OK) != NGX_OK
347 || webdav_validate_path(cf, "brix_crl", &conf->common.crl,
348 WEBDAV_PATH_FILE_OR_DIRECTORY, R_OK) != NGX_OK)
349 {
350 return NGX_CONF_ERROR;
351 }
352
353 return NGX_CONF_OK;

356static char *
357webdav_validate_auth_paths(ngx_conf_t *cf,
358 ngx_http_brix_webdav_loc_conf_t *conf)
359{
360 if (webdav_require_verifier(cf, conf) != NGX_CONF_OK) {
361 return NGX_CONF_ERROR;
362 }
363 /* E-1: a writable export that does not REQUIRE authentication lets an
364 * unauthenticated client create/overwrite/delete objects. Warn always;
365 * refuse under strict security. */
366 if (conf->common.allow_write && conf->auth != WEBDAV_AUTH_REQUIRED
367 && brix_shared_security_gate(cf, conf->common.strict_security,
368 "WebDAV export permits unauthenticated writes "
369 "(brix_allow_write on but brix_webdav_auth is not required)",
370 "brix_webdav_auth required") != NGX_OK)
371 {
372 return NGX_CONF_ERROR;
373 }
374 if (webdav_validate_ca_paths(cf, conf) != NGX_CONF_OK) {
375 return NGX_CONF_ERROR;
376 }
377 if (webdav_validate_cors_origins(cf, conf) != NGX_OK) {
378 return NGX_CONF_ERROR;
379 }
380 return NGX_CONF_OK;
381}

383static char *
384webdav_build_ca_store_once(ngx_conf_t *cf,
385 ngx_http_brix_webdav_loc_conf_t *conf)
386{
387 X509_STORE *store;
388 ngx_pool_cleanup_t *cln;
389 int crl_count = 0;
390
391 if (conf->auth != WEBDAV_AUTH_OPTIONAL
392 && conf->auth != WEBDAV_AUTH_REQUIRED)
393 {
394 return NGX_CONF_OK;
395 }
396 /* Token-only / Basic-only exports carry no x509 trust anchors: there is
397 * no store to build, and the cert tier declines at request time. */
398 if (conf->common.trusted_ca_dir.len == 0 && conf->common.trusted_ca.len == 0) {
399 return NGX_CONF_OK;
400 }
401 store = webdav_build_ca_store(cf->log, conf, &crl_count);
402 if (store == NULL) {
403 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
404 "brix_webdav: failed to build cached CA store");
405 return NGX_CONF_ERROR;
406 }
407 (void) webdav_check_pki_consistency(cf->log, conf);
408 cln = ngx_pool_cleanup_add(cf->pool, 0);
409 if (cln == NULL) {
410 X509_STORE_free(store);
411 return NGX_CONF_ERROR;
412 }
413 cln->handler = webdav_x509_store_cleanup;
414 cln->data = store;
415 conf->ca_store = store;
416 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
417 "brix_webdav: cached CA store built for root=\"%V\" crls=%d",
418 &conf->common.root, crl_count);
419 return NGX_CONF_OK;
420}

483webdav_validate_webdav_enabled(ngx_conf_t *cf, ngx_http_brix_webdav_loc_conf_t *prev,
484 ngx_http_brix_webdav_loc_conf_t *conf)
485{
486 (void) prev;
487
488 if (!conf->common.enable) {
489 return NGX_CONF_OK;
490 }
491 if (webdav_prepare_export_root(cf, conf) != NGX_CONF_OK
492 || webdav_finalize_authz_rules(cf, conf) != NGX_CONF_OK
493 || webdav_configure_storage_backend(cf, conf) != NGX_CONF_OK
494 || brix_http_open_rootfd(cf, &conf->common) != NGX_CONF_OK
495 || webdav_prepare_stage_and_cache(cf, conf) != NGX_CONF_OK
496 || brix_tier_register_stores(cf, &conf->common) != NGX_OK)
497 {
498 return NGX_CONF_ERROR;
499 }

501 if (webdav_validate_auth_paths(cf, conf) != NGX_CONF_OK
