1/*
2 * oci_errors.c — the §0.7.6 error envelope, the J.5 errno map, the guard line.
3 *
4 * WHAT: every way this plane says "no". One JSON envelope emitter, one
5 * errno → HTTP table for the read surface, one guard-core audit line,
6 * and the two response primitives (API-version header, complete-body
7 * send) the gate and the mirror handler share.
8 * WHY: a container client parses the envelope, not the status line: podman
9 * prints `code`/`message` straight to the operator, and `docker pull`
10 * distinguishes "repository does not exist" from "you are not allowed"
11 * purely by the code. Emitting the envelope in exactly one place is
12 * what keeps that contract from drifting per call site — and it is the
13 * same reason the errno map lives here rather than inline in the
14 * handler: the mapping is a POLICY (a full disk on a read is the
15 * ORIGIN's problem, hence 502, not the client's 500), and a policy
16 * spread over five call sites is a policy nobody can audit.
17 * HOW: two parallel tables indexed by brix_oci_err_t (no switch to fall out
18 * of sync with the enum), a bounded snprintf into a stack buffer, and
19 * the standard nginx send_header + one-buf output_filter tail. The
20 * envelope is emitted here, so callers return this function's rc, not
21 * the status — returning the status would have the core paint its own
22 * HTML error page over the JSON.
23 */

104ngx_int_t
105brix_oci_send_body(ngx_http_request_t *r, ngx_uint_t status,
106 const char *ctype, const u_char *body, size_t len)
107{
108 ngx_buf_t *b;
109 ngx_chain_t out;
110 ngx_int_t rc;
111
112 r->headers_out.status = status;
113 r->headers_out.content_length_n = (off_t) len;
114 r->headers_out.content_type.len = ngx_strlen(ctype);
115 r->headers_out.content_type.data = (u_char *) ctype;
116 r->headers_out.content_type_len = r->headers_out.content_type.len;
117
118 if (brix_oci_api_version_header(r) != NGX_OK) {
119 return NGX_HTTP_INTERNAL_SERVER_ERROR;
120 }
121
122 /* A bodiless answer (the 201s this surface returns carry their whole
123 * meaning in Location + Docker-Content-Digest) must go out as headers
124 * only: a zero-length in-memory buf is not a "special" buf, so the write
125 * filter logs "zero size buf in writer" and fails the request after the
126 * header has already left. */
127 if (len == 0) {
128 r->header_only = 1;
129 }
130
131 rc = ngx_http_send_header(r);
132 if (rc == NGX_ERROR || rc > NGX_OK || r->header_only) {
133 return rc;
134 }
135
136 b = ngx_calloc_buf(r->pool);
137 if (b == NULL) {
138 return NGX_HTTP_INTERNAL_SERVER_ERROR;
139 }
140 /* An in-memory body, never a file range: b->memory (invariant #2 — the
141 * cleartext file-backed/sendfile path is for cached objects only). */
142 b->pos = (u_char *) body;
143 b->last = (u_char *) body + len;
144 b->memory = 1;
145 b->last_buf = (r == r->main) ? 1 : 0;
146 b->last_in_chain = 1;
147
148 out.buf = b;
149 out.next = NULL;
150
151 return ngx_http_output_filter(r, &out);
152}

154ngx_int_t
155brix_oci_error(ngx_http_request_t *r, ngx_uint_t status,
156 brix_oci_err_t err, const char *detail)
157{
158 char esc[256];
159 char buf[640];
160 u_char *body;
161 int n;
162
163 /* `detail` is operator- or upstream-supplied and lands inside a JSON
164 * string: escape it, and drop it entirely rather than truncate mid-escape
165 * if it will not fit (a half-escaped tail is a broken envelope). */
166 if (detail != NULL
167 && brix_scan_json_escape(detail, ngx_strlen(detail), esc, sizeof(esc))
168 < 0)
169 {
170 detail = NULL;
171 }
172
173 n = snprintf(buf, sizeof(buf),
174 "{\"errors\":[{\"code\":\"%s\",\"message\":\"%s\"%s%s%s}]}\n",
175 brix_oci_err_code(err), oci_err_message(err),
176 (detail != NULL) ? ",\"detail\":\"" : "",
177 (detail != NULL) ? esc : "",
178 (detail != NULL) ? "\"" : "");
179 if (n < 0 || (size_t) n >= sizeof(buf)) {
180 return NGX_HTTP_INTERNAL_SERVER_ERROR;
181 }
182
183 body = ngx_pnalloc(r->pool, (size_t) n);
184 if (body == NULL) {
185 return NGX_HTTP_INTERNAL_SERVER_ERROR;
186 }
187 ngx_memcpy(body, buf, (size_t) n);
188
189 return brix_oci_send_body(r, status, "application/json", body,
190 (size_t) n);
191}

200ngx_int_t
201brix_oci_refuse(ngx_http_request_t *r, ngx_uint_t status, brix_oci_err_t err,
202 const char *detail)
203{
204 ngx_int_t rc = brix_oci_error(r, status, err, detail);
205
206 /* brix_oci_error() answers NGX_OK when the envelope went out — the same
207 * value every seam in this module uses for "carry on". A refusal that
208 * travels back through such a seam therefore has to arrive as something
209 * else, or the request proceeds past its own denial and answers twice.
210 * NGX_DONE is that value; each caller translates it at its own boundary
211 * (never returning it from an nginx content handler, which would leave
212 * the request suspended). */
213 return (rc == NGX_OK) ? NGX_DONE : rc;
214}

217ngx_uint_t
218brix_oci_errno_status(ngx_http_request_t *r, int err)
219{
220 if (err == ENOENT || err == ENOTDIR || err == ENAMETOOLONG) {
221 return NGX_HTTP_NOT_FOUND;
222 }
223
224 /* The confinement cascade refused the open. On a plane whose every path
225 * component came out of a validating classifier this cannot be a client
226 * spelling — it is a store whose permissions or symlinks have drifted,
227 * or a genuine escape attempt. Say so once, loudly, with the signal the
228 * fail2ban filter keys on. */
229 if (err == EACCES || err == EPERM || err == ELOOP || err == EXDEV) {
230 ngx_log_error(NGX_LOG_WARN, r->connection->log, err,
231 "oci: store open refused signal=oci_path_escape uri=\"%V\" "
232 "client=%V", &r->uri, &r->connection->addr_text);
233 return NGX_HTTP_FORBIDDEN;
234 }
235
236 /* Read surface: a bad or full store is the CACHE's failure, and from the
237 * client's seat the mirror is a gateway — 502, never a 500 that would
238 * have podman blame its own request. */
239 if (err == EIO || err == ENOSPC || err == EDQUOT) {
240 return NGX_HTTP_BAD_GATEWAY;
241 }
242
243 return brix_http_errno_to_status(err);
244}
