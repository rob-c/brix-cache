1/*
2 * oci_present.c — the two headers a registry client actually reads (App. B.1).
3 *
4 * WHAT: decide the `Content-Type` and `Docker-Content-Digest` for a cached
5 * object and attach them to the outgoing response.
6 * WHY: the bytes alone are not a valid registry answer. Podman and docker
7 * branch on the manifest media type — an OCI image index and a Docker
8 * manifest list are different objects with the same JSON shape, and the
9 * wrong `Content-Type` sends the client down the wrong unpack path — and
10 * both read `Docker-Content-Digest` to pin what they pulled, which is
11 * what makes a subsequent `podman image inspect` agree with the registry.
12 * A cache stores bytes; a mirror must also answer these two questions.
13 * HOW: both are pure functions of the object, so they are DERIVED and then
14 * memoized in the `.ocimeta` sidecar rather than trusted from anywhere:
15 * - a blob or digest-addressed manifest already names its digest in the
16 * key, and needs no I/O at all;
17 * - a TAG-addressed manifest is hashed once (bounded read through the
18 * VFS seam), its `mediaType` lifted from the JSON, and the pair
19 * written beside the body.
20 * A missing, unreadable or stale sidecar therefore costs one derivation,
21 * never a wrong answer — which is the whole reason the sidecar is a memo
22 * and not a source of truth.
23 */

40/* The spec caps a manifest at 4 MiB; anything larger is not a manifest we can
41 * describe, and hashing it on the event loop would be a stall besides. */
42#define OCI_MANIFEST_MAX (4 * 1024 * 1024)
43
44/* The default when a manifest carries no `mediaType` member — Docker schema-2
45 * manifests written before the field was mandatory. Serving them as the OCI
46 * type would make a docker client reject its own manifest. */
47#define OCI_CT_DOCKER_MANIFEST \
48 "application/vnd.docker.distribution.manifest.v2+json"
49#define OCI_CT_BLOB "application/octet-stream"

51/* Lift the digest out of the canonical key: for a blob it is the whole
52 * reference, for a manifest only when the reference is digest-shaped. */
53static int
54oci_digest_from_key(const ngx_http_brix_oci_ctx_t *ctx, char *out, size_t outsz)
55{
56 if (ctx->req.cls != BRIX_OCI_REQ_BLOB && !ctx->req.ref_is_digest) {
57 return -1;
58 }
59 return (brix_oci_digest_format(&ctx->req.digest, out, outsz) < 0) ? -1 : 0;
60}

62/* Hash a tag-addressed manifest and read its media type. sha256 is right here
63 * and not a shortcut: a tag is only ever bound by a push that arrived by tag,
64 * and such a push is filed under the algorithm we produce, so the digest this
65 * reports is the digest the store is keyed by. The read goes through
66 * brix_vfs_file_pread (invariant #12): the handle may be backed by a cache
67 * decorator over a remote store, and this is the one seam that knows how to
68 * read either. */
69static ngx_int_t
70oci_derive_manifest(ngx_http_request_t *r, brix_vfs_file_t *fh,
71 const brix_vfs_stat_t *vst, brix_oci_meta_t *meta)
72{
73 u_char *buf;
74 ssize_t n;
75 brix_oci_digest_t d;
76
77 if (vst->size <= 0 || vst->size > OCI_MANIFEST_MAX) {
78 return NGX_DECLINED;
79 }
80
81 buf = ngx_palloc(r->pool, (size_t) vst->size);
82 if (buf == NULL) {
83 return NGX_ERROR;
84 }
85 n = brix_vfs_file_pread(fh, buf, (size_t) vst->size, 0);
86 if (n != (ssize_t) vst->size) {
87 return NGX_DECLINED;
88 }
89
90 if (brix_oci_sha256(buf, (size_t) n, &d) != 0
91 || brix_oci_digest_format(&d, meta->digest, sizeof(meta->digest)) < 0)
92 {
93 return NGX_DECLINED;
94 }
95
96 if (!brix_json_get_str((const char *) buf, (size_t) n, "mediaType",
97 meta->content_type, sizeof(meta->content_type))
98 || meta->content_type[0] == '\0')
99 {
100 (void) ngx_cpystrn((u_char *) meta->content_type,
101 (u_char *) OCI_CT_DOCKER_MANIFEST,
102 sizeof(meta->content_type));
103 }
104 return NGX_OK;
105}

107/* A media type is echoed straight into a response header, so it is validated
108 * before it is trusted: the sidecar and the manifest JSON are both attacker-
109 * reachable in the sense that matters (an upstream we mirror chose them). */
110static int
111oci_ctype_sane(const char *ct)
112{
113 size_t i;
114
115 if (ct[0] == '\0') {
116 return 0;
117 }
118 for (i = 0; ct[i] != '\0'; i++) {
119 unsigned char c = (unsigned char) ct[i];
120
121 if (c <= 0x20 || c >= 0x7f || c == ',' || c == ';' || c == '"') {
122 return 0;
123 }
124 }
125 return 1;
126}

128/* The validator a registry client revalidates with. A manifest's digest IS
129 * its strong ETag (what a Docker Registry v2 emits), and it is the only
130 * validator that survives a refill of identical bytes: the serve pipeline's
131 * mtime+size ETag changes the moment the mirror re-fetches an unchanged tag
132 * past its TTL, turning every revalidation into a full body transfer. */
133int
134brix_oci_present_etag(const brix_oci_present_t *pres, char *buf, size_t buflen)
135{
136 int n;
137
138 if (pres == NULL || pres->meta.digest[0] == '\0') {
139 return 0;
140 }
141 n = snprintf(buf, buflen, "\"%s\"", pres->meta.digest);
142
143 return (n > 0 && (size_t) n < buflen);
144}

147/* Replace the validator the serve pipeline already registered rather than
148 * adding one: two ETag headers leave the client to pick, and a client that
149 * picks the mtime one is back to transferring bodies it holds. */
150static void
151oci_set_etag(ngx_http_request_t *r, const char *etag)
152{
153 ngx_str_t v;
154
155 v.len = ngx_strlen(etag);
156 v.data = ngx_pnalloc(r->pool, v.len);
157 if (v.data == NULL) {
158 return; /* the mtime validator stands */
159 }
160 ngx_memcpy(v.data, etag, v.len);
161
162 if (r->headers_out.etag != NULL) {
163 r->headers_out.etag->value = v;
164 return;
165 }
166 (void) brix_http_set_header_str(r, "ETag", &v, 0, &r->headers_out.etag);
167}

170ngx_int_t
171brix_oci_present_prepare(ngx_http_request_t *r,
172 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx,
173 brix_vfs_file_t *fh, const brix_vfs_stat_t *vst, const char *meta_base,
174 brix_oci_present_t *pres)
175{
176 ngx_log_t *log = r->connection->log;
177 ngx_int_t rc;
178
179 ngx_memzero(pres, sizeof(*pres));
180
181 /* A blob's digest IS its name: no sidecar, no read, no hash. Doing this
182 * before the sidecar load keeps the hot path (blobs are ~99% of an image
183 * pull by both count and bytes) free of any extra I/O. */
184 if (ctx->req.cls == BRIX_OCI_REQ_BLOB) {
185 if (oci_digest_from_key(ctx, pres->meta.digest,
186 sizeof(pres->meta.digest)) != 0)
187 {
188 return NGX_ERROR; /* classifier guarantees otherwise */
189 }
190 (void) ngx_cpystrn((u_char *) pres->meta.content_type,
191 (u_char *) OCI_CT_BLOB,
192 sizeof(pres->meta.content_type));
193 return NGX_OK;
194 }
195
196 rc = (meta_base != NULL)
197 ? brix_oci_meta_load(meta_base, vst->size, vst->mtime, &pres->meta,
198 log)
199 : NGX_DECLINED;
200 if (rc == NGX_OK && oci_ctype_sane(pres->meta.content_type)
201 && pres->meta.digest[0] != '\0')
202 {
203 /* A tag manifest served past its TTL is stale-if-error output from the
204 * cache tier (sd_cache_stale_serve_ok). The mirror cannot see that
205 * decision directly, but it wrote fetched_at itself, so the age of its
206 * own record is an exact and local answer. */
207 if (!ctx->req.ref_is_digest && lcf->manifest_ttl > 0
208 && pres->meta.fetched_at > 0
209 && ngx_time() - pres->meta.fetched_at > lcf->manifest_ttl)
210 {
211 pres->stale = 1;
212 ctx->stale = 1;
213 }
214 return NGX_OK;
215 }
216
217 /* No usable memo: derive from the bytes and (best-effort) leave one behind
218 * for the next hit. */
219 if (oci_derive_manifest(r, fh, vst, &pres->meta) != NGX_OK) {
220 /* Undescribable object: still serve it, but say nothing we cannot
221 * stand behind rather than guessing a digest. */
222 (void) ngx_cpystrn((u_char *) pres->meta.content_type,
223 (u_char *) OCI_CT_DOCKER_MANIFEST,
224 sizeof(pres->meta.content_type));
225 pres->meta.digest[0] = '\0';
226 return NGX_OK;
227 }
228
229 pres->meta.fetched_at = ngx_time();
230 /* D2.5: what "verified" may honestly claim. A digest-addressed object got
231 * here through the oci-digest verify — its key named the hash its bytes
232 * had to have, and a mismatch never became cache-visible. A TAG-addressed
233 * manifest has no such check available: the client supplied no digest, and
234 * the upstream's own Docker-Content-Digest is the upstream's word about
235 * the upstream's bytes. That is trust-on-first-fetch, and the record says
236 * so rather than laundering a derivation into a verification. */
237 pres->meta.verified = ctx->req.ref_is_digest ? 1 : 0;
238 if (meta_base != NULL) {
239 (void) brix_oci_meta_store(meta_base, vst->size, vst->mtime,
240 &pres->meta, log);
241 }
242
243 return NGX_OK;
244}

246void
247brix_oci_present_headers(ngx_http_request_t *r, ngx_fd_t fd, off_t file_size,
248 void *userdata)
249{
250 brix_oci_present_t *pres = userdata;
251
252 (void) fd;
253 (void) file_size;
254
255 if (pres == NULL) {
256 return;
257 }
258
259 /* Overrides whatever the shared serve pipeline guessed: the key has no
260 * extension, so its guess is always the location's default type. The
261 * buffer is the request-pool-allocated presentation record, so it outlives
262 * the header filter. */
263 if (oci_ctype_sane(pres->meta.content_type)) {
264 r->headers_out.content_type.data = (u_char *) pres->meta.content_type;
265 r->headers_out.content_type.len =
266 ngx_strlen(pres->meta.content_type);
267 r->headers_out.content_type_len = r->headers_out.content_type.len;
268 r->headers_out.content_type_lowcase = NULL;
269 }
270
271 if (pres->meta.digest[0] != '\0') {
272 char etag[BRIX_OCI_DIGEST_STRLEN + 3];
273
274 (void) brix_http_set_header(r, "Docker-Content-Digest",
275 pres->meta.digest, NULL);
276 if (brix_oci_present_etag(pres, etag, sizeof(etag))) {
277 oci_set_etag(r, etag);
278 }
279 }
280
281 (void) brix_oci_api_version_header(r);
282
283 /* RFC 9111 §5.5.1 — the response is knowingly past its freshness lifetime
284 * because the upstream could not be reached. A client that honours it can
285 * tell "the mirror is degraded" from "the tag genuinely still points here",
286 * which is exactly the distinction a CI pipeline needs. */
287 if (pres->stale) {
288 (void) brix_http_set_header(r, "Warning",
289 "110 - \"Response is Stale\"", NULL);
290 }
291}

295brix_oci_present_serve_opts(brix_http_serve_opts_t *opts,
296 brix_oci_present_t *pres)
297{
298 ngx_memzero(opts, sizeof(*opts));
299 opts->xfer_proto = BRIX_XFER_PROTO_OCI;
300 opts->op_name = "GET";
301 opts->identity = "anonymous";
302 opts->etag_flags = BRIX_ETAG_WEAK;
303 /* Never transcode: a client verifies the body against the digest we just
304 * advertised in Docker-Content-Digest, so any outbound codec — however
305 * reversible — turns every pull into a checksum failure. */
306 opts->compress = 0;
307 opts->pre_header_send = (pres != NULL) ? brix_oci_present_headers : NULL;
308 opts->pre_header_ud = pres;
309}
