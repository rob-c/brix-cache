1/*
2 * frame_hdr.h — XRootD response-frame field codecs (single source of truth).
3 *
4 * WHAT: unaligned-safe big-endian accessors + the small pure codecs both the
5 * module and the native client apply to server response frames:
6 * - xrd_resp_hdr_unpack: ServerResponseHeader (streamid/status/dlen)
7 * - xrd_wait_secs_parse: kXR_wait / kXR_waitresp retry-after seconds
8 * - xrd_error_body_decode: kXR_error [errnum][msg] (msg NOT NUL-guaranteed)
9 * WHY: these layouts were hand-decoded ~6× in the client (with unaligned
10 * `ntohl(*(uint32_t*)p)` casts — UB on non-aligned frame buffers — and two
11 * `%s` over-reads on the non-NUL error message) and parsed again on the
12 * server's proxy/upstream relay paths. One header keeps the wire facts in a
13 * single place and the access UB-free.
14 * HOW: header-only static inlines over the documented wire vocabulary — no ngx,
15 * no allocation, no OpenSSL. The memcpy+ntoh accessors are the exact idiom
16 * the safe server sites (compat/vendor_ext.c) already use.
17 *
18 * Clean-room: layouts from src/protocol (cross-checked vs XProtocol.hh).
19 */

29static inline uint16_t
30xrd_get_u16_be(const void *p)
31{
32 uint16_t v;
33 memcpy(&v, p, sizeof(v));
34 return ntohs(v);
35}

45static inline uint64_t
46xrd_get_u64_be(const void *p)
47{
48 uint32_t hi, lo;
49 memcpy(&hi, p, 4);
50 memcpy(&lo, (const uint8_t *) p + 4, 4);
51 return ((uint64_t) ntohl(hi) << 32) | (uint64_t) ntohl(lo);
52}

68static inline void
69xrd_put_u64_be(void *p, uint64_t v)
70{
71 uint32_t hi = htonl((uint32_t) (v >> 32));
72 uint32_t lo = htonl((uint32_t) (v & 0xffffffffu));
73 memcpy(p, &hi, 4);
74 memcpy((uint8_t *) p + 4, &lo, 4);
75}

77/* ---- client-request frame prologue ---- */
78/*
79 * Every ClientRequest arm opens with the same four bytes — `kXR_char
80 * streamid[2]` then a big-endian `kXR_unt16 requestid` — before the arm's own
81 * fields begin. xrd_creq_begin stamps exactly that prologue: zero the whole
82 * struct, tag the request slot in streamid[1], write the opcode.
83 *
84 * WHY: every outbound request site (the cache origin's auth/read/pgread/write
85 * legs, the TPC source and push legs) opened with the identical declare +
86 * memzero + streamid + requestid lines. That is a clone by shape — which is
87 * what tools/ci/check_duplication.py reports — but the reason to share it is
88 * that a site which forgets the zeroing ships uninitialised stack into a wire
89 * frame, and nothing on the wire says so. The per-op fields stay at the call
90 * site: they are the half that legitimately differs.
91 *
92 * The prologue is layout-identical across every arm, so a void* + size is
93 * enough and no arm-specific type has to be named here. streamid[0] stays 0:
94 * callers that need a two-byte tag write it themselves after this call.
95 */
96static inline void
97xrd_creq_begin(void *req, size_t reqsz, uint8_t stream_slot, uint16_t requestid)
98{
99 uint8_t *p = (uint8_t *) req;
100
101 memset(req, 0, reqsz);
102 p[1] = stream_slot;
103 xrd_put_u16_be(p + 2, requestid);
104}

106/*
107 * ServerResponseHeader = streamid[2] + status[2 BE] + dlen[4 BE] (8 bytes).
108 * streamid is an opaque 2-byte token echoed back; we read it big-endian so it
109 * compares equal to the value the sender wrote. Any of the out-params may be NULL.
110 */
111static inline void
112xrd_resp_hdr_unpack(const uint8_t h[8], uint16_t *streamid, uint16_t *status,
113 uint32_t *dlen)
114{
115 if (streamid != NULL) { *streamid = xrd_get_u16_be(h); }
116 if (status != NULL) { *status = xrd_get_u16_be(h + 2); }
117 if (dlen != NULL) { *dlen = xrd_get_u32_be(h + 4); }
118}

120/*
121 * kXR_wait / kXR_waitresp retry-after body = [int32 BE seconds]. Returns the
122 * advised seconds clamped to [1, cap]; uses `fallback` if the body is too short.
123 * The cap is caller-supplied on purpose — each role wants a different ceiling
124 * (a client honoring an origin's tape-recall vs a proxy absorbing for a client).
125 */
126static inline uint32_t
127xrd_wait_secs_parse(const uint8_t *body, uint32_t blen, uint32_t fallback,
128 uint32_t cap)
129{
130 uint32_t s = (blen >= 4 && body != NULL) ? xrd_get_u32_be(body) : fallback;
131 if (s < 1) { s = 1; }
132 if (s > cap) { s = cap; }
133 return s;
134}

136/*
137 * kXR_error body = [int32 BE errnum][message bytes]. The message is NOT
138 * NUL-terminated on the wire, so this returns a BOUNDED slice (*msg + *msglen) —
139 * callers MUST use the length (e.g. "%.*s"), never treat *msg as a C string.
140 * Returns 0 on success, -1 if the body is too short to hold errnum. Out-params
141 * other than the return may be NULL.
142 */
143static inline int
144xrd_error_body_decode(const uint8_t *body, uint32_t dlen, int *errnum,
145 const char **msg, size_t *msglen)
146{
147 if (body == NULL || dlen < 4) {
148 return -1;
149 }
150 if (errnum != NULL) { *errnum = (int) xrd_get_u32_be(body); }
151 if (msg != NULL) { *msg = (const char *) (body + 4); }
152 if (msglen != NULL) { *msglen = (size_t) (dlen - 4); }
153 return 0;
154}

156/*
157 * kXR_redirect body = [int32 BE port][host bytes, may end in NUL/CR/LF]. The
158 * host field may carry a "?<opaque>" tail: a redirector (notably EOS/cmsd)
159 * appends the open CAPABILITY (cap.sym/cap.msg) that the open MUST replay to the
160 * chosen data server, else the DS cannot authorize it and bounces the open back
161 * (an endless manager<->DS redirect loop). The host and opaque are split into
162 * their own bounded, NUL-terminated buffers so a long capability opaque (often
163 * >256 B) never truncates the connectable host. Leading '&'/'?' on the opaque
164 * are dropped (EOS sends "?&cap.sym="). `opaque` may be NULL (opaque_sz 0) when
165 * the caller does not replay capabilities. Returns 0 on success, -1 when the
166 * body is too short to hold the port or host/port out-params are missing. An
167 * EMPTY decoded host is left to the caller to judge. Shared by the native
168 * client's redirect follower and the destination's TPC multihop pull (F7).
169 */
170/* Host-field length: up to the first NUL, CR or LF, else the whole body. */
171static inline size_t
172xrd_redirect_field_len(const char *field, size_t flen)
173{
174 const char *end = memchr(field, '\0', flen);
175
176 if (end == NULL) { end = memchr(field, '\r', flen); }
177 if (end == NULL) { end = memchr(field, '\n', flen); }
178 return (end != NULL) ? (size_t) (end - field) : flen;
179}

181/* Copy at most dst_sz-1 bytes of src into dst, always NUL-terminating. */
182static inline void
183xrd_copy_clipped(char *dst, size_t dst_sz, const char *src, size_t len)
184{
185 if (len >= dst_sz) {
186 len = dst_sz - 1;
187 }
188 memcpy(dst, src, len);
189 dst[len] = '\0';
190}

192/* The capability opaque after '?', with the leading '&'/'?' padding some
193 * redirectors emit ("?&cap.sym=") dropped before it is clipped in. */
194static inline void
195xrd_redirect_opaque_copy(char *opaque, size_t opaque_sz, const char *o,
196 size_t olen)
197{
198 while (olen > 0 && (*o == '&' || *o == '?')) {
199 o++;
200 olen--;
201 }
202 xrd_copy_clipped(opaque, opaque_sz, o, olen);
203}

205static inline int
206xrd_redirect_body_decode(const uint8_t *body, uint32_t dlen, char *host,
207 size_t host_sz, int *port, char *opaque,
208 size_t opaque_sz)
209{
210 const char *field, *qmark;
211 size_t flen, hlen;
212
213 if (body == NULL || dlen < 5 || host == NULL || host_sz == 0
214 || port == NULL)
215 {
216 return -1;
217 }
218 *port = (int) xrd_get_u32_be(body);
219
220 field = (const char *) body + 4;
221 flen = xrd_redirect_field_len(field, (size_t) dlen - 4);
222 qmark = memchr(field, '?', flen);
223 hlen = (qmark != NULL) ? (size_t) (qmark - field) : flen;
224 xrd_copy_clipped(host, host_sz, field, hlen);
225
226 if (opaque != NULL && opaque_sz > 0) {
227 opaque[0] = '\0';
228 if (qmark != NULL) {
229 xrd_redirect_opaque_copy(opaque, opaque_sz, qmark + 1,
230 (size_t) (field + flen - (qmark + 1)));
231 }
232 }
233 return 0;
234}
