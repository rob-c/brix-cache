14/*
15 * brix_sss_identity_t — decoded identity fields from an SSS cleartext
16 * payload. Populated by brix_sss_parse_identity(). The v1 fields (name,
17 * grps, host, ip) are joined by the v2 entity breadth of
18 * release-2.0-readiness F9: vorg, role, endorsements and a proxied
19 * credential blob. Every cap is the shared wire constant from
20 * protocols/root/protocol/sss.h so the producers (proxy arm, native client)
21 * and this parser agree on what fits.
22 */
23typedef struct {
24 char name[BRIX_SSS_ENT_NAME_MAX];
25 char vorg[BRIX_SSS_ENT_VORG_MAX]; /* v2 VORG 0x02 */
26 char role[BRIX_SSS_ENT_ROLE_MAX]; /* v2 ROLE 0x03 */
27 char grps[BRIX_SSS_ENT_GRPS_MAX];
28 char endo[BRIX_SSS_ENT_ENDO_MAX]; /* v2 ENDO 0x05 endorsements */
29 char host[256];
30 char ip[128];
31 u_char creds[BRIX_SSS_ENT_CREDS_MAX]; /* v2 CRED 0x06 proxied credential, raw */
32 size_t creds_len; /* 0 = no CRED field */
33 int id_count;
34} brix_sss_identity_t;

59/*
60 * brix_sss_verify_blob — validate a self-contained SSS credential against a
61 * keytab, transport-independent (used by both the XRootD kXR_auth path and the
62 * CMS kYR_xauth handshake).
63 *
64 * Performs the full verification chain: outer "sss\0"+BF32 header, key-id
65 * lookup, Blowfish-CFB64 decrypt, CRC32 wrong-key detection, timestamp replay
66 * window (lifetime seconds), and identity TLV parse. Only the self-contained
67 * USEDATA credential form is accepted; the interactive SNDLID (server-supplies
68 * login-id) form returns NGX_DECLINED so the caller can reject it explicitly.
69 *
70 * Returns NGX_OK with *id_out filled and *key_out set to the matched key on
71 * success; NGX_DECLINED for an unsupported SNDLID credential; NGX_ERROR for any
72 * malformed/forged/expired credential (err filled). Does not allocate.
73 */
74ngx_int_t brix_sss_verify_blob(ngx_array_t *keys, time_t lifetime,
