38int
39brixoci_src_manifest(brixoci_end_t *s, const char *seltag,
40 const brixoci_opts_t *o, brix_oci_desc_t *m, char *err,
41 size_t errlen)
42{
43 char *body;
44 size_t blen;
45 int rc;
46
47 memset(m, 0, sizeof(*m));
48 if (!s->is_layout) {
49 return brix_oci_reg_resolve(&s->reg, &s->ref, o->platform, m, err,

51 }
52 rc = brix_oci_layout_index_get(&s->lay, seltag, m->digest,
53 sizeof(m->digest), m->mediatype,

57 sizeof(m->digest), m->mediatype,

59 }

61 return rc;
62 }
63 rc = brix_oci_layout_blob_load(&s->lay, m->digest,
64 OCI_COPY_MANIFEST_CAP, &body, &blen,

68 }

79/* Sequential fd → fd copy. 0 / -1 with errno-flavored err. */
80int
81brixoci_fd_pump(int in_fd, int out_fd, char *err, size_t errlen)
82{
83 char *buf = malloc(OCI_COPY_IOBUF);
84 ssize_t n;
85 int ok = buf != NULL;
86
87 while (ok && (n = read(in_fd, buf, OCI_COPY_IOBUF)) != 0) {
88 ssize_t off = 0;
89
90 if (n < 0) {
91 ok = errno == EINTR;
92 continue;
93 }
94 while (ok && off < n) {
95 ssize_t w = write(out_fd, buf + off, (size_t) (n - off));
96
97 if (w < 0) {
98 ok = errno == EINTR;
99 } else {
100 off += w;
101 }
102 }
103 }
104 if (!ok) {
105 snprintf(err, errlen, "blob copy: %s",
106 buf == NULL ? "out of memory" : strerror(errno));
107 }
108 free(buf);
109 return ok ? 0 : -1;
110}

112/* Land one blob in a layout: registry fetch (hash-on-stream) or local
113 * layout-to-layout copy, staged + committed. */
114static int
115copy_blob_to_layout(brixoci_end_t *s, brixoci_end_t *d, const char *digest,
116 char *err, size_t errlen)
117{
118 char tmppath[1200];
119 long long sz;
120 int out_fd, in_fd = -1, rc;
121
122 if (s->is_layout) {
123 rc = brix_oci_layout_blob_verify(&s->lay, digest, &sz, err, errlen);
124 if (rc != BRIX_OCI_REG_OK) {
125 return rc;
126 }
127 in_fd = brix_oci_layout_blob_open(&s->lay, digest, err, errlen);
128 if (in_fd < 0) {
129 return in_fd;
130 }
131 }
132 out_fd = brix_oci_layout_stage(&d->lay, tmppath, sizeof(tmppath), err,
133 errlen);
134 if (out_fd < 0) {
135 if (in_fd >= 0) {
136 close(in_fd);
137 }
138 return out_fd;
139 }
140 if (s->is_layout) {
141 rc = brixoci_fd_pump(in_fd, out_fd, err, errlen) == 0
142 ? BRIX_OCI_REG_OK
143 : BRIX_OCI_REG_ETRANSPORT;
144 close(in_fd);
145 } else {
146 rc = brix_oci_reg_blob_fetch(&s->reg, s->name, digest, out_fd, err,
147 errlen);
148 }
149 close(out_fd);
150 if (rc == BRIX_OCI_REG_OK) {
151 rc = brix_oci_layout_commit(&d->lay, tmppath, digest, err, errlen);
152 }
153 if (rc != BRIX_OCI_REG_OK) {
154 unlink(tmppath);
155 }
156 return rc;
157}

159/* Land one blob in a registry: layout push (verified first) or a
160 * registry-to-registry pump through an anonymous verified temp. */
161static int
162copy_blob_to_reg(brixoci_end_t *s, brixoci_end_t *d, const char *digest,
163 char *err, size_t errlen)
164{
165 long long sz;
166 struct stat st;
167 FILE *tf;
168 int fd, rc;
169
170 if (s->is_layout) {
171 rc = brix_oci_layout_blob_verify(&s->lay, digest, &sz, err, errlen);
172 if (rc != BRIX_OCI_REG_OK) {
173 return rc;
174 }
175 fd = brix_oci_layout_blob_open(&s->lay, digest, err, errlen);
176 if (fd < 0) {
177 return fd;
178 }
179 rc = brix_oci_reg_blob_push(&d->reg, d->name, digest, fd,
180 (size_t) sz, err, errlen);
181 close(fd);
182 return rc;
183 }
184 tf = tmpfile();
185 if (tf == NULL) {
186 snprintf(err, errlen, "tmpfile: %s", strerror(errno));
187 return BRIX_OCI_REG_ETRANSPORT;
188 }
189 fd = fileno(tf);
190 rc = brix_oci_reg_blob_fetch(&s->reg, s->name, digest, fd, err, errlen);
191 if (rc == BRIX_OCI_REG_OK) {
192 rc = fstat(fd, &st) == 0
193 ? brix_oci_reg_blob_push(&d->reg, d->name, digest, fd,
194 (size_t) st.st_size, err, errlen)
195 : BRIX_OCI_REG_ETRANSPORT;
196 }
197 fclose(tf);
198 return rc;
199}

201static int
202copy_one_blob(brixoci_end_t *s, brixoci_end_t *d, const char *digest,
203 char *err, size_t errlen)
204{
205 if (d->is_layout) {
206 char scratch[128];
207 long long sz;
208
209 /* Dedupe: an existing blob that VERIFIES is done; a corrupt one
210 * is treated as absent and re-landed (commit renames over it). */
211 if (brix_oci_layout_blob_verify(&d->lay, digest, &sz, scratch,
212 sizeof(scratch)) ==
213 BRIX_OCI_REG_OK) {
214 return BRIX_OCI_REG_OK;
215 }
216 return copy_blob_to_layout(s, d, digest, err, errlen);
217 }
218 /* Registry dedupe lives in the push itself (HEAD probe). */
219 return copy_blob_to_reg(s, d, digest, err, errlen);
220}

222/* Read the "digest" string out of the JSON object [el,en) and land that blob;
223 * `what` names the object in the error message ("config"/"layer"). Returns
224 * BRIX_OCI_REG_OK, BRIX_OCI_REG_EPROTO (no digest present), or copy_one_blob's
225 * transfer error. */
226static int
227copy_digest_blob(brixoci_end_t *s, brixoci_end_t *d, const char *el, size_t en,
228 const char *what, char *err, size_t errlen)
229{
230 char dig[BRIX_OCI_DIGEST_STRLEN];
231
232 if (!brix_json_get_str(el, en, "digest", dig, sizeof(dig))) {
233 snprintf(err, errlen, "manifest %s has no digest", what);
234 return BRIX_OCI_REG_EPROTO;
235 }
236 return copy_one_blob(s, d, dig, err, errlen);
237}

239/* Walk config.digest + layers[].digest and land each blob. */
240static int
241copy_blobs(brixoci_end_t *s, brixoci_end_t *d, const char *body,
242 size_t blen, char *err, size_t errlen)
243{
244 const char *arr, *el;
245 size_t an, en, cur = 0;
246 int rc;
247
248 if (brix_json_get_raw(body, blen, "config", &el, &en) == 1) {
249 rc = copy_digest_blob(s, d, el, en, "config", err, errlen);
250 if (rc != BRIX_OCI_REG_OK) {
251 return rc;
252 }
253 }
254 if (brix_json_get_raw(body, blen, "layers", &arr, &an) != 1) {
255 return BRIX_OCI_REG_OK; /* config-only artifact */
256 }
257 while (brix_json_arr_next(arr, an, &cur, &el, &en) == 1) {
258 rc = copy_digest_blob(s, d, el, en, "layer", err, errlen);
259 if (rc != BRIX_OCI_REG_OK) {
260 return rc;
261 }
262 }
263 return BRIX_OCI_REG_OK;
264}

266/* Bind the manifest at the destination; digest_out gets the canonical
267 * "<alg>:<hex>" that was bound/pushed. */
268int
269brixoci_put_manifest(brixoci_end_t *d, const char *bindtag,
270 const brix_oci_desc_t *m, char *digest_out, size_t dlen,
271 char *err, size_t errlen)
272{
273 char dig[BRIX_OCI_DIGEST_STRLEN];
274 int rc;
275
276 if (!d->is_layout) {
277 rc = brix_oci_reg_manifest_put(&d->reg, &d->ref, m->mediatype,
278 m->body, m->body_len, err, errlen);
279 snprintf(dig, sizeof(dig), "%s", m->digest);
280 } else {
281 rc = brix_oci_layout_blob_put_mem(&d->lay, m->body, m->body_len,
282 dig, sizeof(dig), err, errlen);
283 if (rc == BRIX_OCI_REG_OK && m->digest[0] != '\0' &&
284 strcmp(dig, m->digest) != 0) {
285 snprintf(err, errlen, "manifest bytes hash to %s, expected %s",
286 dig, m->digest);
287 rc = BRIX_OCI_REG_EVERIFY;
288 }
289 if (rc == BRIX_OCI_REG_OK) {
290 rc = brix_oci_layout_index_set(&d->lay, bindtag, dig,
291 m->mediatype, m->body_len, err,
292 errlen);
293 }
294 }
295 if (rc == BRIX_OCI_REG_OK && digest_out != NULL) {
296 snprintf(digest_out, dlen, "%s", dig);
297 }
298 return rc;
299}

316int
317brixoci_copy_run(brixoci_end_t *src, brixoci_end_t *dst,
318 const brixoci_opts_t *o, char *digest_out, size_t dlen,
319 char *err, size_t errlen)
320{
321 brix_oci_desc_t m;
322 const char *seltag = NULL, *bindtag = NULL;
323 int rc;
324
325 brixoci_xfer_tags(src, dst, &seltag, &bindtag);
326 rc = brixoci_src_manifest(src, seltag, o, &m, err, errlen);
327 if (rc != BRIX_OCI_REG_OK) {
328 return rc;
329 }
330 if (brixoci_is_index(&m)) {
331 brix_oci_desc_free(&m);
332 snprintf(err, errlen, "source manifest is an image index — pull "
333 "it with --platform to select one image");
334 return BRIX_OCI_REG_EPROTO;
335 }
336 rc = copy_blobs(src, dst, m.body, m.body_len, err, errlen);
337 if (rc == BRIX_OCI_REG_OK) {
338 rc = brixoci_put_manifest(dst, bindtag, &m, digest_out, dlen, err,
339 errlen);
340 }
341 brix_oci_desc_free(&m);
342 return rc;
