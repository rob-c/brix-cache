1/* File: redirect.c — native TPC multihop: follow a source's kXR_redirect
2 * WHAT: tpc_redirect_note() records the target a source named in a kXR_redirect
3 * reply to the pull's kXR_open (host, port, capability opaque);
4 * tpc_redirect_follow() decides whether the pull thread may dial it — hop
5 * budget, self-loop, and the same egress allowlist the first hop passed —
6 * and swaps it in as the next leg's source.
7 *
8 * WHY (release-2.0 F7): a real HEP source is often a manager/redirector (cmsd,
9 * EOS MGM, dCache door) that answers the open with "go to data server X".
10 * Before F7 the pull treated that reply as an open failure, so every native
11 * root:// TPC into a clustered source died at the first hop. The follow is
12 * bounded (brix_tpc_max_hops) and policy-checked on EVERY hop so a hostile
13 * or mis-configured source cannot steer the gateway at an internal host the
14 * operator never allowed (the request-forgery class the egress guard exists
15 * for) — the redirect target passes brix_tpc_source_guard here and the
16 * address-level SSRF check (I-DNS-3) again in tpc_connect().
17 *
18 * HOW: the shared frame_hdr.h decoder splits the ServerRedirectBody; the follow
19 * counts hops against conf->tpc_max_hops, refuses a target equal to the
20 * current source (a two-node ping-pong is still caught by the hop cap), runs
21 * the allowlist, then copies redir_host/redir_port into src_host/src_port.
22 * Runs on the pull thread: only the pull task and its immutable conf are
23 * touched (a refusal is counted on the event loop by done.c via
24 * t->redirect_refused). */

40int
41tpc_redirect_note(brix_tpc_pull_t *t, const u_char *body, uint32_t dlen)
42{
43 int port = 0;
44
45 if (xrd_redirect_body_decode(body, dlen, t->redir_host,
46 sizeof(t->redir_host), &port,
47 t->redir_opaque, sizeof(t->redir_opaque)) != 0
48 || t->redir_host[0] == '\0')
49 {
50 snprintf(t->err_msg, sizeof(t->err_msg),
51 "TPC source redirect body malformed");
52 t->xrd_error = kXR_ServerError;
53 return -1;
54 }
55 /* XRootD lets a redirector send port 0 ("the default"); a negative port
56 * selects the URL-list form this pull does not speak. */
57 if (port < 0 || port > BRIX_MAX_PORT) {
58 snprintf(t->err_msg, sizeof(t->err_msg),
59 "TPC source redirect to %s carries invalid port %d",
60 t->redir_host, port);
61 t->xrd_error = kXR_ServerError;
62 return -1;
63 }
64 t->redir_port = (uint16_t) port;
65 t->redirect_pending = 1;
66 return 0;
67}

70static int
71tpc_redirect_check_hops(brix_tpc_pull_t *t)
72{
73 ngx_uint_t cap = (t->conf != NULL) ? t->conf->tpc_max_hops : 0;
74
75 if (t->hops < cap) {
76 return 0;
77 }
78 snprintf(t->err_msg, sizeof(t->err_msg),
79 "TPC source %.128s:%u redirected to %.128s:%u but "
80 "brix_tpc_max_hops is %lu",
81 t->src_host, tpc_redirect_effective_port(t->src_port),
82 t->redir_host, tpc_redirect_effective_port(t->redir_port),
83 (unsigned long) cap);
84 t->xrd_error = kXR_NotAuthorized;
85 return -1;
86}

88/* WHAT: the egress-policy gate — the redirect target must pass the same source
89 * allowlist the client's original tpc.src passed on the event loop. */
90static int
91tpc_redirect_check_policy(brix_tpc_pull_t *t)
92{
93 char guard_err[256];
94
95 if (t->conf == NULL) {
96 return 0;
97 }
98 if (brix_tpc_source_guard_check(t->conf->common.tpc_source_guard,
99 t->conf->common.tpc_source_allow,
100 t->redir_host, guard_err,
101 sizeof(guard_err)) == 0)
102 {
103 return 0;
104 }
105 t->redirect_refused = 1;
106 snprintf(t->err_msg, sizeof(t->err_msg),
107 "TPC redirect to %.128s refused: %.200s", t->redir_host,
108 guard_err);
109 t->xrd_error = kXR_NotAuthorized;
110 return -1;
111}

113int
114tpc_redirect_follow(brix_tpc_pull_t *t, ngx_log_t *log)
115{
116 unsigned cur_port = tpc_redirect_effective_port(t->src_port);
117 unsigned new_port = tpc_redirect_effective_port(t->redir_port);
118
119 if (!t->redirect_pending) {
120 return -1;
121 }
122 t->redirect_pending = 0;
123
124 if (tpc_redirect_check_hops(t) != 0) {
125 return -1;
126 }
127 if (strcasecmp(t->redir_host, t->src_host) == 0 && new_port == cur_port) {
128 snprintf(t->err_msg, sizeof(t->err_msg),
129 "TPC source redirect loops back to itself (%s:%u)",
130 t->src_host, cur_port);
131 t->xrd_error = kXR_ServerError;
132 return -1;
133 }
134 if (tpc_redirect_check_policy(t) != 0) {
135 return -1;
136 }
137
138 t->hops++;
139 ngx_log_error(NGX_LOG_INFO, log, 0,
140 "brix: TPC hop %ui: %s:%ud -> %s:%ud for %s",
141 t->hops, t->src_host, cur_port, t->redir_host, new_port,
142 t->src_path);
143
144 ngx_cpystrn((u_char *) t->src_host, (u_char *) t->redir_host,
145 sizeof(t->src_host));
146 t->src_port = t->redir_port;
147 t->sessid_known = 0; /* a new host means a new session */
148 return 0;
149}
