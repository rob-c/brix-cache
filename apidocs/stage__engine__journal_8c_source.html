1/*
2 * stage_engine_journal.c — the durable journal + dead-letter substrate of the
3 * async-staging engine (split from stage_engine.c, phase-79).
4 *
5 * WHAT: Owns the per-worker journal directory and the crash-visibility records
6 * that back an async submit: minting a request id, persisting a QUEUED
7 * record, removing a completed one, bumping the retry counter, and moving a
8 * permanently-denied flush to the dead-letter directory. Also the engine's
9 * one-time init that sets the journal directory.
10 *
11 * WHY: stage_engine.c owned three concerns and exceeded the file-size cap. The
12 * durable-record substrate is a self-contained concern: the byte mover and
13 * the scheduler both reach it only through the seam in
14 * stage_engine_internal.h. The exact on-disk record I/O, the O_TRUNC crash-
15 * safe update, and the attempt/age dead-letter caps are unchanged by the
16 * split — this is a re-home, not a behavior change.
17 *
18 * HOW: Every record write opens O_WRONLY|O_CREAT|O_TRUNC|O_CLOEXEC, writes the
19 * full fixed-size brix_sreq_t, and fsyncs. stage_deny_terminal (public,
20 * shared by the scheduler and reconcile halves) bumps attempts, re-persists,
21 * and on cap dead-letters + emits a loud tombstone. The two dead-letter
22 * helpers stay file-local.
23 */

37/* ---- SP4: the durable async queue -----------------------------------------
38 * An async submit is DEFERRED rather than run inline: the request is appended to
39 * a per-worker in-memory pending list (holding the live src/dst instances, which
40 * are the memoised per-worker tier instances - they outlive the request) and,
41 * when a journal dir is configured, persisted as a small record for crash
42 * visibility/recovery. brix_stage_scheduler_tick() (a per-worker timer) drains
43 * the list, runs each mover, and drops the stage copy of a completed FLUSH. This
44 * generalises the FRM queue model to SD instances (section 11); the full physical
45 * extraction of src/frm/ is the remaining SP4/SP5 migration. */

145/* Persist (best-effort) a QUEUED request record so a crash leaves a recoverable
146 * row; removed on completion. Skipped when no journal dir is configured. */
147void
148stage_journal_write(const stage_pending_t *p)
149{
150 brix_sreq_t rec;
151 char path[1200];
152 int fd;
153
154 if (stage_engine_state.journal_dir[0] == '\0') {
155 return;
156 }
157 if ((size_t) snprintf(path, sizeof(path), "%s/%s.req",
158 stage_engine_state.journal_dir, p->reqid) >= sizeof(path))
159 {
160 return;
161 }
162 ngx_memzero(&rec, sizeof(rec));
163 snprintf(rec.reqid, sizeof(rec.reqid), "%s", p->reqid);
164 rec.kind = p->kind;
165 rec.state = BRIX_SREQ_QUEUED;
166 snprintf(rec.src_driver, sizeof(rec.src_driver), "%s",
167 (p->src->driver && p->src->driver->name) ? p->src->driver->name : "");
168 snprintf(rec.src_key, sizeof(rec.src_key), "%s", p->src_key);
169 snprintf(rec.dst_driver, sizeof(rec.dst_driver), "%s",
170 (p->dst->driver && p->dst->driver->name) ? p->dst->driver->name : "");
171 snprintf(rec.dst_key, sizeof(rec.dst_key), "%s", p->dst_key);
172 snprintf(rec.export_root, sizeof(rec.export_root), "%s", p->export_root);
173 rec.cred = p->cred; /* copy the owner identity into the durable record */
174 /* The cred's trailing bearer is IN-MEMORY ONLY: a live secret that would be
175 * expired by replay time. Scrub it from the stack record and persist only
176 * the identity prefix (BRIX_SREQ_IDENTITY_SIZE), so a raw token never lands
177 * on disk regardless of the write length below. */
178 ngx_memzero(rec.cred.bearer, sizeof(rec.cred.bearer));
179 rec.enqueued_at = (int64_t) time(NULL);
180
181 fd = open(path, O_WRONLY | O_CREAT | O_TRUNC | O_CLOEXEC, 0600);
182 if (fd < 0) {
183 return;
184 }
185 if (write(fd, &rec, BRIX_SREQ_IDENTITY_SIZE)
186 == (ssize_t) BRIX_SREQ_IDENTITY_SIZE)
187 {
188 (void) fsync(fd);
189 }
190 (void) close(fd);
191}

211/* Write the updated rec (with bumped attempts) back to the active journal slot.
212 *
213 * WHAT: Overwrites <journal_dir>/<reqid>.req atomically enough for a crash-safe
214 * update — O_TRUNC on the existing file is sufficient (the record is
215 * readable even if we crash mid-write; the next drive will re-bump).
216 *
217 * WHY: Persisting attempts across restarts is what prevents the unbounded retry
218 * loop: a restart would otherwise reset the in-memory state to zero while
219 * the disk record still carries the accumulated count.
220 *
221 * HOW: Open for write with O_TRUNC, write the full record, fsync. */
222static void
223stage_journal_update_rec(const char *journal_dir, const brix_sreq_t *rec)
224{
225 char path[1200];
226 int fd;
227
228 if (journal_dir == NULL || journal_dir[0] == '\0') {
229 return;
230 }
231 if ((size_t) snprintf(path, sizeof(path), "%s/%s.req",
232 journal_dir, rec->reqid) >= sizeof(path))
233 {
234 return;
235 }
236 fd = open(path, O_WRONLY | O_CREAT | O_TRUNC | O_CLOEXEC, 0600);
237 if (fd < 0) {
238 return;
239 }
240 /* Persist only the identity prefix — never the in-memory-only cred.bearer
241 * (a decoded rec already carries a zeroed bearer, but the shorter write keeps
242 * the on-disk record shape identical to stage_journal_write's). */
243 if (write(fd, rec, BRIX_SREQ_IDENTITY_SIZE)
244 == (ssize_t) BRIX_SREQ_IDENTITY_SIZE)
245 {
246 (void) fsync(fd);
247 }
248 (void) close(fd);
249}

251/* Stamp a decoded record FAILED (last_errno + finished_at) and re-persist it to
252 * the active journal slot. Shared tail of the two public FAILED helpers below;
253 * the record stays in the active journal so the scheduler / reconcile retry it. */
254static void
255stage_journal_persist_failed(const char *journal_dir, brix_sreq_t *rec,
256 int last_errno)
257{
258 char num[24], att[24];
259
260 rec->state = BRIX_SREQ_FAILED;
261 rec->last_errno = last_errno;
262 rec->finished_at = (int64_t) time(NULL);
263 stage_journal_update_rec(journal_dir, rec);
264 brix_stage_events_emit("engine", "failed", rec->reqid, rec->dst_key,
265 "errno",
266 brix_stage_events_num(num, sizeof(num), last_errno),
267 "attempts",
268 brix_stage_events_num(att, sizeof(att),
269 rec->attempts),
270 NULL);
271}

286void
287stage_journal_mark_failed(const char *journal_dir, const char *reqid,
288 int last_errno)
289{
290 char path[1200];
291 char rbuf[sizeof(brix_sreq_t)];
292 brix_sreq_t rec;
293 int fd;
294 ssize_t n;
295
296 if (journal_dir == NULL || journal_dir[0] == '\0') {
297 return;
298 }
299 if ((size_t) snprintf(path, sizeof(path), "%s/%s.req",
300 journal_dir, reqid) >= sizeof(path))
301 {
302 return;
303 }
304 fd = open(path, O_RDONLY | O_CLOEXEC);
305 if (fd < 0) {
306 return; /* record already removed / never written */
307 }
308 n = read(fd, rbuf, sizeof(rbuf));
309 (void) close(fd);
310 if (brix_sreq_decode(rbuf, (size_t) n, &rec) != NGX_OK) {
311 return; /* corrupt slot — leave it for reconcile */
312 }
313 /* 2.0 (2026-09-09): count THIS drive. A failed first drive is an attempt —
314 * "attempts a journal record may accumulate ... a transient failure" is what
315 * brix_frm_fail_retries caps (directives.md) — and it is the only number the
316 * `replayed` event can report about a record the sweep just rescued. Before,
317 * the inline/thread completion path persisted FAILED without bumping, so a
318 * record that failed once and was re-driven successfully published
319 * attempts=0, indistinguishable from a crash replay that never ran at all
320 * (which stays 0: it recorded no failure). */
321 stage_journal_bump_failed(journal_dir, &rec, last_errno);
322}

324void
325stage_journal_bump_failed(const char *journal_dir, brix_sreq_t *rec,

327{
328 if (rec == NULL) {
329 return;

335/* Move the active journal record to <journal_dir>/deadletter/<reqid>.req,
336 * creating the deadletter directory on demand (0700).
337 *
338 * WHAT: rename(2) from the active slot to the deadletter slot; creates the
339 * subdirectory on the first call. If rename fails (e.g. cross-device),
340 * falls back to copy + unlink.
341 *
342 * WHY: Moving the file out of the active journal directory ensures the
343 * scheduler and reconcile stop picking it up while preserving the bytes
344 * and the stage copy for operator recovery.
345 *
346 * HOW: Build the two paths, mkdir the deadletter dir (EEXIST is OK), rename. */
347static void
348stage_journal_move_to_deadletter(const char *journal_dir, const char *reqid,
349 ngx_log_t *log)
350{
351 char src_path[1200];
352 char dl_dir[1200];
353 char dst_path[1300];
354 int fd;
355 char buf[sizeof(brix_sreq_t)];
356 ssize_t n;
357
358 if (journal_dir == NULL || journal_dir[0] == '\0') {
359 return;
360 }
361 if ((size_t) snprintf(src_path, sizeof(src_path), "%s/%s.req",
362 journal_dir, reqid) >= sizeof(src_path))
363 {
364 return;
365 }
366 if ((size_t) snprintf(dl_dir, sizeof(dl_dir), "%s/deadletter",
367 journal_dir) >= sizeof(dl_dir))
368 {
369 return;
370 }
371 if ((size_t) snprintf(dst_path, sizeof(dst_path), "%s/%s.req",
372 dl_dir, reqid) >= sizeof(dst_path))
373 {
374 return;
375 }
376
377 if (mkdir(dl_dir, 0700) != 0 && errno != EEXIST) {
378 ngx_log_error(NGX_LOG_ERR, log, errno,
379 "xrootd stage: dead-letter mkdir \"%s\" failed", dl_dir);
380 return;
381 }
382
383 if (rename(src_path, dst_path) == 0) {
384 return;
385 }
386 /* rename failed (cross-device or other): copy + unlink as fallback */
387 fd = open(src_path, O_RDONLY | O_CLOEXEC);
388 if (fd < 0) {
389 return;
390 }
391 n = read(fd, buf, sizeof(buf));
392 (void) close(fd);
393 if (n > 0) {
394 int dst_fd = open(dst_path, O_WRONLY | O_CREAT | O_TRUNC | O_CLOEXEC,
395 0600);
396 if (dst_fd >= 0) {
397 ssize_t wn = write(dst_fd, buf, (size_t) n);
398 (void) wn; /* best-effort copy; -Werror=unused-result */
399 (void) fsync(dst_fd);
400 (void) close(dst_fd);
401 }
402 }
403 (void) unlink(src_path);
404}

406/* Dead-letter terminal for a permanently denied flush.
407 *
408 * WHAT: Increments rec->attempts, re-persists the updated record to the active
409 * journal slot, then checks whether the attempt cap
410 * (BRIX_STAGE_DENY_MAX_ATTEMPTS) or the age cap
411 * (BRIX_STAGE_DENY_MAX_AGE_SEC) has been reached. When either cap fires,
412 * moves the active record to <journal_dir>/deadletter/<reqid>.req, emits a
413 * loud NGX_LOG_ERR naming the reqid, principal, key and dst_key, and
414 * returns 1 (stop re-driving). Returns 0 when below both caps (keep
415 * retrying later).
416 *
417 * WHY: A BRIX_XFER_DENIED result means the per-user credential is permanently
418 * missing or expired; the deny mode opted out of service-cred fallback.
419 * Without a cap the scheduler tick and restart-reconcile re-drive the
420 * same record forever. The cap bounds the loop while preserving the
421 * stage copy for operator recovery: a dead-lettered write is NEVER flushed
422 * on the wrong identity — dead-letter means STOP, not "flush as service".
423 *
424 * HOW: Called by stage_complete (BRIX_XFER_DENIED) and stage_reconcile_one
425 * (errno==EACCES from brix_sd_stage_reflush). The rec pointer is the
426 * decoded brix_sreq_t from the on-disk record (already bumped by the
427 * caller's re-read, so we bump it here before persist). journal_dir is
428 * passed explicitly so the function is unit-testable without the module
429 * global. */
430int
431stage_deny_terminal(const char *journal_dir, const char *reqid,
432 brix_sreq_t *rec, ngx_log_t *log)
433{
434 int64_t age_sec;
435
436 /* Bump and persist: always update attempts so the count survives a restart.
437 * Even if we are about to dead-letter, the persisted count is in the
438 * deadletter copy (which survives after the move). */
439 rec->attempts++;
440 stage_journal_update_rec(journal_dir, rec);
441
442 /* Check both caps: attempt count OR age triggers dead-letter. */
443 age_sec = (int64_t) time(NULL) - rec->enqueued_at;
444
445 if (rec->attempts < stage_engine_state.max_attempts
446 && age_sec < (int64_t) BRIX_STAGE_DENY_MAX_AGE_SEC)
447 {
448 return 0; /* below both caps — keep record in active journal for retry */
449 }
450
451 /* Both-or-either cap reached: move to deadletter and emit a loud tombstone. */
452 ngx_log_error(NGX_LOG_ERR, log, 0,
453 "xrootd stage: flush DEAD-LETTERED (reqid=%s principal=\"%s\" "
454 "key=%s dst=\"%s\" attempts=%uD age=%lds) - "
455 "credential permanently missing/expired in deny mode; "
456 "stage copy retained in deadletter/ for operator recovery",
457 reqid,
458 rec->cred.principal[0] ? rec->cred.principal : "-",
459 rec->cred.key[0] ? rec->cred.key : "-",
460 rec->dst_key,
461 rec->attempts,
462 (long) age_sec);
463
464 stage_journal_move_to_deadletter(journal_dir, reqid, log);
465 stage_journal_note_deadletter(rec, "denied");
466 return 1;
467}

469int
470stage_retry_terminal(const char *journal_dir, brix_sreq_t *rec,
471 int last_errno, ngx_log_t *log)
472{
473 stage_journal_bump_failed(journal_dir, rec, last_errno);
474 if (rec->attempts < stage_engine_state.max_attempts) {
475 return 0; /* below brix_frm_fail_retries: keep FAILED */
476 }
477 ngx_log_error(NGX_LOG_ERR, log, 0,
478 "xrootd stage: flush DEAD-LETTERED (reqid=%s key=%s dst=\"%s\" "
479 "attempts=%uD errno=%d) - brix_frm_fail_retries reached while the "
480 "origin stays unreachable; stage copy retained in deadletter/ for "
481 "operator recovery",
482 rec->reqid, rec->src_key, rec->dst_key, rec->attempts, last_errno);
483 stage_journal_move_to_deadletter(journal_dir, rec->reqid, log);
484 stage_journal_note_deadletter(rec, "unreachable");
485 return 1;
486}
