6/*
7 * forward.h — krb5 GSSAPI credential forwarding to the backend/origin.
8 *
9 * WHAT: One-shot helper that, given a forwardable delegated GSS credential
10 * captured during inbound krb5 auth (a gss_cred_id_t obtained with
11 * GSS_C_DELEG_FLAG), initiates a fresh GSSAPI security context to the
12 * upstream/origin service principal so this node acts AS the inbound user.
13 *
14 * WHY: Phase-70 §5.7 — krb5 is only backend-usable by EXCHANGE (GSSAPI
15 * forwarding). When the client presents a forwardable ticket the node
16 * receives a delegated credential and can re-authenticate to the origin on
17 * the user's behalf, with no admin pre-provisioning.
18 *
19 * HOW: brix_krb5_deleg_to_origin() imports the origin service principal name,
20 * acquires nothing (it uses the supplied delegated cred as the initiator
21 * identity) and calls gss_init_sec_context() ONCE, returning the initial
22 * output token to send to the origin. This is deliberately a single
23 * init-context step: the multi-leg GSSAPI negotiation loop (feeding origin
24 * replies back through gss_init_sec_context until GSS_S_COMPLETE) belongs to
25 * the origin-auth caller, which owns the wire exchange and the context
26 * handle lifetime. Availability is reported by brix_krb5_forward_available()
27 * so callers can fall back to SELECT when forwarding is not compiled in or
28 * not supported.
29 */
30
31/*
32 * Return 1 if this build was compiled with krb5/GSSAPI support AND credential
33 * forwarding is available; 0 otherwise (caller falls back to SELECT).
34 */
35int brix_krb5_forward_available(void);

56/*
57 * Wire transceiver callback driving one negotiation leg (phase-70 §5.7 origin
58 * wire). brix_krb5_deleg_negotiate() calls it every time the local
59 * gss_init_sec_context() produces an output token that must reach the origin:
60 *
61 * wire_ctx — opaque caller state (the origin connection: kXR wire for
62 * origin_auth.c, an in-process acceptor for the live test).
63 * out_token — the token this leg produced (non-empty); send it to the origin.
64 * in_token — filled with the origin's reply token. Set len==0 (data may be
65 * NULL) when the origin replied with none. Its bytes must stay
66 * valid until the next wire() call or until negotiate() returns
67 * (the engine copies nothing — it feeds them straight back into
68 * gss_init_sec_context as the next input token).
69 * done — set to 1 iff the origin signalled the exchange is complete
70 * (kXR_ok, not kXR_authmore); else 0.
71 * log — for diagnostics.
72 *
73 * Returns NGX_OK to continue the negotiation, NGX_ERROR to abort it (the origin
74 * rejected the token or the transport failed — negotiate() then fails closed).
75 */
76typedef ngx_int_t (*brix_krb5_wire_fn)(void *wire_ctx,
77 const ngx_str_t *out_token, ngx_str_t *in_token, int *done, ngx_log_t *log);

79/*
80 * Drive the FULL multi-leg GSSAPI negotiation to the origin AS the delegated
81 * user (phase-70 §5.7). Where brix_krb5_deleg_to_origin() performs one
82 * gss_init_sec_context() step, this owns the whole loop: it initialises the
83 * security context, hands each output token to `wire` for delivery to the
84 * origin, feeds the origin's reply back into gss_init_sec_context(), and repeats
85 * until the local context reports GSS_S_COMPLETE. Mutual auth is requested, so
86 * the origin's AP-REP is verified before success — a spoofed origin cannot
87 * complete the exchange.
88 *
89 * pool — token bytes handed to `wire` are copied here.
90 * deleg_gss_cred — gss_cred_id_t (as void*) captured with GSS_C_DELEG_FLAG;
91 * the initiator identity. GSS_C_NO_CREDENTIAL (NULL) uses
92 * the process default cred.
93 * origin_service_princ — target service principal (GSS_KRB5_NT_PRINCIPAL_NAME).
94 * wire — transceiver invoked once per outbound token (above).
95 * wire_ctx — opaque state passed to `wire`.
96 * log — for gss/major-minor diagnostics.
97 *
98 * Returns NGX_OK once the initiator context is GSS_S_COMPLETE (the origin
99 * authenticated this node AS the delegated user), or NGX_ERROR — GSS failure,
100 * a wire/transport error, or the origin closing the exchange before completion.
101 * Never leaks the GSS context or target name on either path. Without krb5/GSSAPI
102 * support this reports unavailable and returns NGX_ERROR.
103 */
104ngx_int_t brix_krb5_deleg_negotiate(ngx_pool_t *pool, void *deleg_gss_cred,
105 const char *origin_service_princ, brix_krb5_wire_fn wire, void *wire_ctx,
106 ngx_log_t *log);
