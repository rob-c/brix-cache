4/*
5 * ssi_client.h — native (ngx-free) client for the SSI request/response plane.
6 *
7 * WHAT: open a session on "/.ssi/<service>", submit a request, collect the
8 * reply stream (zero or more alerts, then one terminal response), pull a
9 * streamed body, cancel, close.
10 *
11 * WHY: src/protocols/ssi/ is a complete server plane, but the only client that
12 * ever spoke to it was tests/ssi_client.cc driving the real C++
13 * libXrdSsi. The pure-C suite could not reach its own server's SSI
14 * surface without that stack — exactly the dependency the C suite exists
15 * to avoid — so the plane's wire behaviour was untested from this side.
16 *
17 * HOW: the calls mirror the server's responder hooks one-for-one
18 * (src/protocols/ssi/ssi_service.h): set_response -> EV_RESPONSE,
19 * alert -> EV_ALERT, a streaming response -> EV_PENDING + brix_ssi_read,
20 * error -> a filled brix_status. The control word is packed by the SHARED
21 * codec (src/protocols/ssi/ssi_rrinfo.c, in libxrdproto), so the bytes
22 * this client sends and the bytes the server decodes cannot drift.
23 *
24 * ------------------------------------------------------------------------
25 * THE TWO THINGS THAT ARE NOT OBVIOUS
26 *
27 * 1. ONE control word, TWO carriers. The 8-byte XrdSsiRRInfo rides the kXR
28 * OFFSET FIELD of a read/write, but rides the PAYLOAD of a kXR_query
29 * (brix_ssi_query takes `body`; brix_ssi_write takes `off8`). Swapping them
30 * does not look like a framing bug — the server answers "short SSI control"
31 * or silently decodes a different command — so the carrier is part of the
32 * contract, not an implementation detail.
33 *
34 * 2. TWO reply paths, and the client cannot choose which it gets. A SYNCHRONOUS
35 * service has its reply ready when the submit returns, and the client PULLS
36 * it with kXR_query(kXR_Qopaqug, RRInfo{Rwt}) — an ordinary kXR_ok, no attn
37 * envelope anywhere. A DEFERRING service acks the submit with kXR_waitresp
38 * and later PUSHES kXR_attn(asynresp) frames. brix_ssi_await() hides the
39 * difference; both paths end in the same classifier, because the discriminator
40 * is identical in both: the payload's leading XrdSsiRRInfoAttn tag byte, and
41 * nothing else. src/protocols/root/response/async.c builds the SAME envelope
42 * for an alert and for the terminal response (same actnum, same inner
43 * kXR_ok), so the transport genuinely cannot tell them apart.
44 *
45 * Together those are why await() returns ONE frame at a time and says which
46 * kind it was: the caller loops until it sees EV_RESPONSE.
47 * ------------------------------------------------------------------------
48 */
49

55/*
56 * An unbounded alert stream must not read as a hung server, so it carries its
57 * own budget and its own wording — distinct from the transport's "no async
58 * response after N frames" (XRDC_REDIR_MAX, frame.c), which counts frames of
59 * any kind and would blame the wire for a service that simply never finishes.
60 */
61#define BRIX_SSI_ALERT_MAX 64

63/* One SSI session: a kXR_open handle plus the request correlation state. */
64typedef struct {
65 brix_conn *conn; /* borrowed, not owned */
66 uint8_t fhandle[XRD_FHANDLE_LEN]; /* from the open reply */
67 uint32_t req_id; /* 24-bit; masked by the codec */
68 uint16_t sid; /* streamid the deferred reply rides */
69 int alerts_seen; /* against BRIX_SSI_ALERT_MAX */
70 int deferred; /* submit was acked with kXR_waitresp */
71} brix_ssi_sess;

73/* What one await() call surfaced, decided by the RRInfoAttn tag byte. */
74typedef enum {
75 BRIX_SSI_EV_ALERT = 0, /* '!' BRIX_SSI_ATTN_ALRT — progress alert */
76 BRIX_SSI_EV_RESPONSE, /* ':' BRIX_SSI_ATTN_FULL — terminal, inline */
77 BRIX_SSI_EV_PENDING /* '*' BRIX_SSI_ATTN_PEND — pull with _read() */
78} brix_ssi_ev;

80/*
81 * One delivered frame. The server lays a reply out as
82 * [XrdSsiRRInfoAttn pfx_len][metadata md_len][data]
83 * and NOTHING on the wire states the data length: it is only recoverable as
84 * total - pfx_len - md_len. `meta` and `data` therefore point INTO `raw` rather
85 * than being separate allocations — free `raw` once, via brix_ssi_reply_free,
86 * and never free the halves.
87 */
88typedef struct {
89 brix_ssi_ev ev;
90 uint8_t *raw; /* the whole payload; the only owned pointer */
91 const uint8_t *meta; /* NULL when meta_len == 0 */
92 uint32_t meta_len;
93 const uint8_t *data; /* NULL when data_len == 0 */
94 uint32_t data_len;
95} brix_ssi_reply;

100/*

108int brix_ssi_sess_open(brix_ssi_sess *s, brix_conn *c, const char *service,

111/*
112 * kXR_write the request bytes with an RXQ control word in the OFFSET field
113 * carrying req_id and the declared total size, so the server dispatches on the
114 * last byte rather than waiting for a read. A deferring service acks
115 * kXR_waitresp; that is recorded on the session and is NOT an error. 0 / -1.
116 */
117int brix_ssi_submit(brix_ssi_sess *s, const void *req, size_t len,
118 brix_status *st);
119
120/*
121 * Collect the NEXT frame of the reply stream and classify it. Returns 0 with
122 * *r filled (the caller frees it with brix_ssi_reply_free), or -1 with st set —
123 * including the alert-budget refusal, which is its own message and not the
124 * transport's. Call in a loop until r->ev is BRIX_SSI_EV_RESPONSE (or PENDING,
125 * which is also terminal for the metadata and hands the body to _read).
126 */
127int brix_ssi_await(brix_ssi_sess *s, brix_ssi_reply *r, brix_status *st);
128
129/*
130 * kXR_read with an RXQ control word in the offset field: pull the body of a
131 * response the server reported as PENDING rather than inline. Returns the byte
132 * count (0 at end of stream), or -1.
133 */
134int brix_ssi_read(brix_ssi_sess *s, void *buf, size_t len, brix_status *st);

136/*
137 * Cancel the in-flight request with a CAN control word. Sent on the QUERY
138 * carrier (ssi.c) rather than the write carrier, because a request already
139 * dispatched has no write left to ride on; the request-phase cancel in
140 * ssi_dispatch.c is reachable the same way once the write is in flight.
141 * 0 / -1.
142 */
143int brix_ssi_cancel(brix_ssi_sess *s, brix_status *st);

145/* kXR_close the handle. Best-effort; the session is unusable afterwards. */
