4/*
5 * cns.h — minimal Composite Cluster Name Space (§6).
6 *
7 * WHAT: data servers report namespace mutations (file create/closew, unlink, …)
8 * to the manager, which keeps a path→{size,mtime,server} inventory and can
9 * answer kXR_stat for any cluster file locally — without redirecting the
10 * client to the data server that holds it.
11 * WHY: XrdCnsd-style global namespace: a redirector can stat/size any file in the
12 * federation it manages. The existing CMS registry is prefix-based routing
13 * (which server exports which subtree); CNS adds per-file metadata.
14 * HOW: a new raw CMS frame (CMS_RR_CNS) carries one fixed-layout event over the
15 * existing data-server→manager CMS link. The data server emits on closew
16 * (brix_cns_emit); the manager applies into an in-memory inventory
17 * (brix_cns_apply) and looks it up from the stat handler (brix_cns_stat).
18 *
19 * Scope: the inventory is a fixed-slot POD table (cns_inventory.h). When a
20 * `brix_cns_zone` is registered (any collector block; wired in
21 * postconfiguration) it lives in an nginx SHM slab shared across every manager
22 * worker, so a mutation on worker A is visible to a stat on worker B; without a
23 * zone it falls back to a per-worker heap table (correct for the common
24 * single-worker redirector). Emit coverage is add/del/mkdir/rmdir.
25 */

47/*
48 * Fixed wire layout (raw, big-endian), followed by `name_len` path bytes:
49 * op[1] rsvd[3] size[8] mtime[8] name_len[2]
50 *
51 * BRIX_CNS_MV appends a second, length-prefixed path after the first:
52 * … name_len[2] oldpath[name_len] name2_len[2] newpath[name2_len]
53 * and carries the destination's directory-ness in the first reserved byte
54 * (rsvd[0], which every other op leaves zero). Extending the frame this way is
55 * safe in both directions: CMS_RR_CNS is a private frame only our own peers
56 * parse, a pre-MV manager's decoder ignores the trailing bytes and then rejects
57 * the unknown op in brix_cns_inv_apply (a no-op, not a corrupt entry), and a
58 * pre-MV data server simply never emits one.
59 */
60#define BRIX_CNS_HDR_LEN 22
