1/*
2 * oci_referrers.c — the referrers graph (D15.1).
3 *
4 * WHAT: write one descriptor per (subject, referrer) edge at push time, drop
5 * it at delete time, and assemble the edges of one subject into an
6 * image index on GET.
7 * WHY: a referrer is an ordinary manifest whose `subject` field points at
8 * another manifest — that is the whole of the artifact model, and the
9 * registry's job is to make the reverse direction answerable. Storing
10 * the DESCRIPTOR rather than a bare digest is what keeps the read path
11 * from re-opening and re-parsing every referring manifest to rebuild
12 * fields (mediaType, artifactType, size) that were already known to
13 * the push that created the edge.
14 * HOW: <repo>/referrers/<alg>/<subject>/<referrer> holds the descriptor
15 * JSON, and <repo>/manifests/<alg>/<referrer>.subject holds the
16 * subject hex, so DELETE cuts the edge with one read instead of a scan
17 * over every subject in the repository. Both names are 64-hex by
18 * grammar (§0.7.2), so neither can spell a path component of its own.
19 */

42/* <root>/repos/<name>/referrers/<alg>/<subject-hex>[/<referrer-hex>].
43 * `referrer` may be NULL to name the subject's directory itself. The
44 * algorithm component is the SUBJECT's: the directory answers "what refers to
45 * this subject", so it is the subject that names it. A referrer's own
46 * algorithm rides in the width of its filename. */
47static int
48oci_referrers_path(const brix_oci_store_t *st, const char *name,
49 size_t name_len, const brix_oci_digest_t *subject,
50 const brix_oci_digest_t *referrer, char *out, size_t outsz)
51{
52 char rest[2 * BRIX_OCI_HEXLEN_MAX + 32];
53 const char *alg = brix_oci_alg_name(subject->alg);
54 int n;
55
56 if (alg == NULL) {
57 return -1;
58 }
59 n = snprintf(rest, sizeof(rest), "referrers/%s/%s%s%s", alg, subject->hex,
60 (referrer != NULL) ? "/" : "",
61 (referrer != NULL) ? referrer->hex : "");
62 if (n < 0 || (size_t) n >= sizeof(rest)) {
63 return -1;
64 }
65 return brix_oci_store_repo_path(st, name, name_len, rest, out, outsz);
66}

69/* What a client sees in the index for this referrer. `artifactType` is the
70 * manifest's own when it declares one and its config media type otherwise —
71 * the fallback the image spec mandates, and the reason a cosign signature
72 * (which sets only the config type) is still filterable. */
73static json_t *
74oci_referrers_descriptor(json_t *doc, const char *dg, const char *ctype,
75 off_t size)
76{
77 const char *atype;
78 json_t *desc;
79 json_t *ann;
80
81 atype = json_string_value(json_object_get(doc, "artifactType"));
82 if (atype == NULL) {
83 atype = json_string_value(
84 json_object_get(json_object_get(doc, "config"), "mediaType"));
85 }
86
87 desc = json_pack("{s:s, s:s, s:I}", "mediaType", ctype, "digest", dg,
88 "size", (json_int_t) size);
89 if (desc == NULL) {
90 return NULL;
91 }
92 if (atype != NULL
93 && json_object_set_new(desc, "artifactType",
94 json_string(atype)) != 0)
95 {
96 json_decref(desc);
97 return NULL;
98 }
99
100 /* Annotations ride along because that is where the tooling puts the
101 * signature's own metadata — but they are attacker-sized, so they are
102 * included only when the whole descriptor still fits the per-edge cap,
103 * checked below by the caller after the dump. */
104 ann = json_object_get(doc, "annotations");
105 if (json_is_object(ann)
106 && json_object_set(desc, "annotations", ann) != 0)
107 {
108 json_decref(desc);
109 return NULL;
110 }
111 return desc;
112}

115/* Serialise `desc` into `out`, dropping the annotations rather than the whole
116 * edge when the result would exceed the per-edge cap. Length, or -1. */
117static int
118oci_referrers_dump(json_t *desc, char *out, size_t outsz)
119{
120 char *text;
121 size_t len;
122
123 text = json_dumps(desc, JSON_COMPACT);
124 if (text == NULL) {
125 return -1;
126 }
127 len = ngx_strlen(text);
128
129 if (len >= outsz) {
130 free(text);
131 if (json_object_del(desc, "annotations") != 0) {
132 return -1;
133 }
134 text = json_dumps(desc, JSON_COMPACT);
135 if (text == NULL) {
136 return -1;
137 }
138 len = ngx_strlen(text);
139 if (len >= outsz) {
140 free(text);
141 return -1;
142 }
143 }
144
145 ngx_memcpy(out, text, len + 1);
146 free(text);
147 return (int) len;
148}

153{
154 brix_oci_digest_t subject;
155 const char *sd;
156 json_t *subject_node;
157
158 subject_node = json_object_get(doc, "subject");
159 if (subject_node == NULL) {
160 return 0;
161 }
162 if (!json_is_object(subject_node)) {
163 *why = "subject must be a descriptor object";
164 return -1;
165 }
166 sd = json_string_value(json_object_get(subject_node, "digest"));
167 if (sd == NULL) {
168 *why = "subject carries no digest";
169 return -1;
170 }
171 if (brix_oci_digest_parse(sd, ngx_strlen(sd), &subject) != 0) {
172 *why = "subject digest is not a digest this registry stores";
173 return -1;
174 }
175 return 0;

179ngx_int_t
180brix_oci_referrers_index(const brix_oci_store_t *st,
181 const brix_oci_req_t *req, json_t *doc, const brix_oci_digest_t *d,
182 const char *ctype, off_t size, char *subj, size_t subjsz, ngx_log_t *log)
183{
184 brix_oci_digest_t subject;
185 char desc_text[BRIX_OCI_REFERRER_DESC_MAX];
186 char path[PATH_MAX];
187 char dg[BRIX_OCI_DIGEST_STRLEN];
188 const char *sd;
189 json_t *desc;
190 int len;
191
192 subj[0] = '\0';
193
194 sd = json_string_value(
195 json_object_get(json_object_get(doc, "subject"), "digest"));
196 if (sd == NULL) {
197 return NGX_OK; /* not a referrer: nothing to index */
198 }
199 /* Unreachable by way of the PUT path — brix_oci_referrers_subject_ok()
200 * has already refused a subject that does not parse — and checked anyway,
201 * because this hex is about to name a directory. */
202 if (brix_oci_digest_parse(sd, ngx_strlen(sd), &subject) != 0) {
203 ngx_log_error(NGX_LOG_ERR, log, 0,
204 "oci: manifest declares an unparsable subject digest");
205 return NGX_ERROR;
206 }
207
208 if (brix_oci_digest_format(d, dg, sizeof(dg)) < 0) {
209 return NGX_ERROR;
210 }
211
212 desc = oci_referrers_descriptor(doc, dg, ctype, size);
213 if (desc == NULL) {
214 return NGX_ERROR;
215 }
216 len = oci_referrers_dump(desc, desc_text, sizeof(desc_text));
217 json_decref(desc);
218 if (len < 0) {
219 return NGX_ERROR;
220 }
221
222 if (oci_referrers_path(st, req->name, req->name_len, &subject, d,
223 path, sizeof(path)) != 0
224 || brix_oci_store_publish_bytes(st, path, desc_text, (size_t) len,
225 BRIX_VFS_DOMAIN_REGISTRY, log) != NGX_OK)
226 {
227 return NGX_ERROR;
228 }
229
230 /* OCI-Subject reports the SUBJECT's digest — the thing the client asked
231 * us to attach to — not the referrer it just pushed. The same string is
232 * the back-pointer's payload: a self-describing digest, so the reader
233 * validates it through the ordinary grammar rather than trusting that
234 * whatever wrote the file meant sha256. */
235 if (brix_oci_digest_format(&subject, subj, subjsz) < 0) {
236 return NGX_ERROR;
237 }
238
239 /* The back-pointer is written second: an edge whose descriptor exists but
240 * whose back-pointer does not is merely undeletable-by-shortcut, while the
241 * reverse would be a DELETE that removes a descriptor never written. */
242 if (brix_oci_store_manifest_path(st, req->name, req->name_len, d,
243 ".subject", path, sizeof(path)) != 0
244 || brix_oci_store_publish_bytes(st, path, subj, ngx_strlen(subj),
245 BRIX_VFS_DOMAIN_REGISTRY, log) != NGX_OK)
246 {
247 return NGX_ERROR;
248 }
249 return NGX_OK;
250}

253void
254brix_oci_referrers_forget(const brix_oci_store_t *st, const char *name,
255 size_t name_len, const brix_oci_digest_t *d, ngx_log_t *log)
256{
257 brix_oci_digest_t subject;
258 char path[PATH_MAX];
259 char subject_str[BRIX_OCI_DIGEST_STRLEN];
260 ssize_t n;
261
262 if (brix_oci_store_manifest_path(st, name, name_len, d, ".subject",
263 path, sizeof(path)) != 0)
264 {
265 return;
266 }
267 n = brix_oci_store_get_text(path, subject_str, sizeof(subject_str));
268 if (n < 0) {
269 return; /* not a referrer: no edge to cut */
270 }
271 (void) brix_oci_store_remove(path, log);
272
273 /* The stored hex was written by the indexer above, but it is re-validated
274 * before it names a directory: a store an operator has edited by hand is
275 * still not allowed to build a path. */
276 if (brix_oci_digest_parse(subject_str, (size_t) n, &subject) != 0) {
277 return;
278 }
279 if (oci_referrers_path(st, name, name_len, &subject, d,
280 path, sizeof(path)) == 0)
281 {
282 (void) brix_oci_store_remove(path, log);
283 }
284}

287/* Does this stored descriptor pass the requested artifactType filter? A
288 * descriptor with no artifactType matches nothing but the unfiltered case —
289 * the spec's filter selects, it does not merely order. */
290static int
291oci_referrers_matches(const char *desc_text, const ngx_str_t *want)
292{
293 json_error_t jerr;
294 const char *atype;
295 json_t *desc;
296 int ok;
297
298 if (want->len == 0) {
299 return 1;
300 }
301 desc = json_loads(desc_text, 0, &jerr);
302 if (desc == NULL) {
303 return 0;
304 }
305 atype = json_string_value(json_object_get(desc, "artifactType"));
306 ok = (atype != NULL
307 && ngx_strlen(atype) == want->len
308 && ngx_strncmp(atype, want->data, want->len) == 0);
309 json_decref(desc);
310
311 return ok;
312}

315/* Walk the subject's directory, appending each descriptor that passes the
316 * filter into `body` as an array element. Entries written, and `*used` grown
317 * by what was appended. */
318static int
319oci_referrers_collect(const char *dir, const ngx_str_t *filter, u_char *body,
320 size_t *used)
321{
322 char desc_text[BRIX_OCI_REFERRER_DESC_MAX];
323 char path[PATH_MAX];
324 struct dirent *ent;
325 DIR *dh;
326 int count = 0;
327
328 dh = opendir(dir); /* vfs-seam-allow: DOMAIN_REGISTRY — registry's own referrers index, not a VFS export listing */
329 if (dh == NULL) {
330 return 0; /* unknown subject: an empty graph */
331 }
332
333 while (count < BRIX_OCI_REFERRERS_MAX
334 && (ent = readdir(dh)) != NULL) /* vfs-seam-allow: DOMAIN_REGISTRY — registry's own referrers index, not a VFS export listing */
335 {
336 size_t len;
337
338 if (ngx_strlen(ent->d_name) != BRIX_OCI_SHA256_HEXLEN
339 || snprintf(path, sizeof(path), "%s/%s", dir, ent->d_name)
340 >= (int) sizeof(path))
341 {
342 continue;
343 }
344 if (brix_oci_store_get_text(path, desc_text, sizeof(desc_text)) <= 0
345 || !oci_referrers_matches(desc_text, filter))
346 {
347 continue;
348 }
349
350 len = ngx_strlen(desc_text);
351 if (*used + len + 2 >= OCI_REFERRERS_BODY_MAX) {
352 break;
353 }
354 if (count > 0) {
355 body[(*used)++] = ',';
356 }
357 ngx_memcpy(body + *used, desc_text, len);
358 *used += len;
359 count++;
360 }
361 (void) closedir(dh);
362
363 return count;
364}

367ngx_int_t
368brix_oci_referrers_get(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
369 brix_oci_store_t *st)
370{
371 ngx_str_t filter = ngx_null_string;
372 char dir[PATH_MAX];
373 u_char *body;
374 size_t used;
375
376 if (ngx_http_arg(r, (u_char *) oci_referrers_filter,
377 sizeof(oci_referrers_filter) - 1, &filter) != NGX_OK)
378 {
379 ngx_str_null(&filter);
380 }
381
382 if (oci_referrers_path(st, ctx->req.name, ctx->req.name_len,
383 &ctx->req.digest, NULL, dir, sizeof(dir)) != 0)
384 {
385 return NGX_HTTP_INTERNAL_SERVER_ERROR;
386 }
387
388 body = ngx_palloc(r->pool, OCI_REFERRERS_BODY_MAX);
389 if (body == NULL) {
390 return NGX_HTTP_INTERNAL_SERVER_ERROR;
391 }
392 used = (size_t) snprintf((char *) body, OCI_REFERRERS_BODY_MAX,
393 "{\"schemaVersion\":2,\"mediaType\":\"%s\","
394 "\"manifests\":[", OCI_MT_INDEX);
395
396 (void) oci_referrers_collect(dir, &filter, body, &used);
397
398 body[used++] = ']';
399 body[used++] = '}';
400
401 /* Declaring the filter is mandatory when one was honoured: without this
402 * header a client cannot tell a filtered answer from a registry that
403 * ignored the parameter, and would treat "no signatures of this type" as
404 * "no signatures at all". */
405 if (filter.len > 0
406 && brix_http_set_header(r, "OCI-Filters-Applied",
407 oci_referrers_filter, NULL) != NGX_OK)
408 {
409 return NGX_HTTP_INTERNAL_SERVER_ERROR;
410 }
411
412 ctx->disp = BRIX_OCI_OUT_LOCAL;
413 return brix_oci_send_body(r, NGX_HTTP_OK, OCI_MT_INDEX, body, used);
414}
