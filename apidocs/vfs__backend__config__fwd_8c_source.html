1/*
2 * vfs_backend_config_fwd.c — the forward://<protocols> backend grammar (2.0 F5)
3 *
4 * WHAT `brix_storage_backend forward://root[,roots] permit=<host|.suffix> …`
5 * claims the export for the xroot_fwd driver: an export that names NO
6 * origin of its own, and instead relays each client-named
7 * `/root://host:port//path` key to that origin — the XCache forwarding
8 * proxy shape (upstream `pss.origin =`).
9 * WHY Every other backend line names one fixed origin, and the whole
10 * registry entry describes that one link (host/port/tls/credential). A
11 * forwarding export has none of that: the protocol set and the origin
12 * permit list ARE its configuration, so it needs its own parser that
13 * fills the entry with an empty origin and the two forwarding fields.
14 * HOW Runs before the fixed-origin scheme table in
15 * vfs_parse_xroot_or_driver_origin() and answers NGX_DECLINED unless the
16 * spec starts with "forward://". The permit list arrives later through
17 * the store-line params (`permit=`, vfs_backend_store_params.c), which
18 * refuses a forwarding export that ends up with none.
19 */
20

25/* "root", "roots" or "root,roots" (any order): the schemes a client may name.
26 * Anything else — an empty list, an unknown scheme, a dangling comma — is a
27 * grammar error, never a silently narrowed set. */
28static ngx_int_t
29vfs_fwd_parse_protocols(const u_char *list, size_t list_len, int *root,
30 int *roots)
31{
32 const u_char *end = list + list_len;
33 const u_char *tok = list;
34 const u_char *cut;
35 size_t len;
36
37 *root = 0;
38 *roots = 0;
39 if (list_len == 0) {
40 return NGX_ERROR;
41 }
42 while (tok <= end) {
43 cut = ngx_strlchr((u_char *) tok, (u_char *) end, ',');
44 if (cut == NULL) {
45 cut = end;
46 }
47 len = (size_t) (cut - tok);
48 if (len == sizeof("root") - 1 && ngx_strncmp(tok, "root", len) == 0) {
49 *root = 1;
50 } else if (len == sizeof("roots") - 1
51 && ngx_strncmp(tok, "roots", len) == 0)
52 {
53 *roots = 1;
54 } else {
55 return NGX_ERROR; /* empty token or unknown scheme */
56 }
57 if (cut == end) {
58 return NGX_OK;
59 }
60 tok = cut + 1; /* a trailing ',' yields an empty tok */
61 }
62 return NGX_ERROR;
63}

68{

70 brix_vfs_backend_entry_t *e;

72

74 return NGX_DECLINED;
75 }
76 if (vfs_fwd_parse_protocols(sb->data + pfxn, sb->len - pfxn,
77 &root, &roots) != NGX_OK)
78 {
79 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
80 "brix_storage_backend \"%V\": forward:// takes a protocol list of "
81 "root and/or roots (e.g. forward://root,roots)", sb);
82 return NGX_ERROR;
83 }
84
85 e = brix_vfs_backend_entry_claim(root_canon, "xroot_fwd");
86 if (e == NULL) {
87 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
88 "brix_storage_backend \"%V\": backend registry is full", sb);
89 return NGX_ERROR;
90 }
91 /* No origin of its own: the host is empty, the path root is "/", and the
92 * permit list stays empty until the store-line params stamp it (an export
93 * that never gets one is refused there). */
94 brix_vfs_backend_set_origin(e, "", 0, 0, "/", 0);
95 e->origin_family = family;
96 e->origin_fwd_allow_root = root ? 1 : 0;
97 e->origin_fwd_allow_roots = roots ? 1 : 0;
98 e->origin_fwd_permit[0] = '\0';
99 return NGX_OK;
