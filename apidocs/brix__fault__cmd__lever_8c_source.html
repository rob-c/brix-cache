29/* Strip a trailing up|down|both direction token from `args`; return which lever
30 * set(s) it names: 0 = both, 1 = up only, 2 = down only. */
31int
32dir_of(char *args)
33{
34 size_t L = strlen(args);
35 while (L > 0 && isspace((unsigned char) args[L - 1])) {
36 args[--L] = '\0';
37 }
38 char *sp = strrchr(args, ' ');
39 char *tok = sp ? sp + 1 : args;
40 int d = -1;
41 if (strcmp(tok, "up") == 0) d = 1;
42 else if (strcmp(tok, "down") == 0) d = 2;
43 else if (strcmp(tok, "both") == 0) d = 0;
44 if (d < 0) {
45 return 0; /* no direction token -> both, leave args untouched */
46 }
47 if (sp) {
48 *sp = '\0'; /* drop the token so numeric parse ignores it */
49 } else {
50 args[0] = '\0';
51 }
52 return d;
53}

80/* ---- Percent operand -> ppm ----
81 *
82 * WHAT: Converts a percentage operand to parts-per-million, returning 0 on
83 * success and -1 when the operand is negative.
84 *
85 * WHY: `toxicity` and `refuse` both take a percentage and both store ppm; one
86 * converter keeps the clamp and the rounding identical between them, so a
87 * fix to either cannot drift.
88 *
89 * HOW: 1. atof the operand and refuse a negative.
90 * 2. Clamp above 100% — over-saturating is an operator typo, not a fault.
91 * 3. Scale to ppm with round-to-nearest.
92 */
93static int
94fid_pct_to_ppm(const char *args, int *ppm)
95{
96 double pct = atof(args);
97
98 if (pct < 0.0) {
99 return -1;
100 }
101 if (pct > 100.0) {
102 pct = 100.0;
103 }
104
105 *ppm = (int) (pct * 10000.0 + 0.5);
106 return 0;
107}

129/* ---- Directional non-negative int levers ----
130 *
131 * WHAT: Handles the table above, returning 1 when `verb` was one of them (set
132 * or refused with an `err` reply) and 0 when it was not.
133 *
134 * WHY: `rate` here deliberately shadows the unchecked cmd_set_lever branch —
135 * cmd_set_dispatch wires this first so a negative KB/s is refused rather
136 * than latched.
137 *
138 * HOW: 1. Match the verb against the table.
139 * 2. Strip the direction token FIRST (dir_of rewrites `args`), then parse
140 * the remaining number, and refuse a negative without touching state.
141 * 3. Write the value through the row's offsetof on the named side(s).
142 */
143static int
144cmd_set_dir_int(const char *verb, char *args, char *reply, size_t rsz)
145{
146 size_t n;
147 int d, v;
148
149 for (n = 0; n < sizeof(fid_dir_ints) / sizeof(fid_dir_ints[0]); n++) {
150 if (strcmp(verb, fid_dir_ints[n].verb) != 0) {
151 continue;
152 }
153
154 d = dir_of(args);
155 v = atoi(args);
156 if (v < 0) {
157 fp_reply(reply, rsz, "err: %s < 0\n", verb);
158 return 1;
159 }
160
161 if (d != 2) {
162 *(volatile int *) ((volatile char *) &g_up
163 + fid_dir_ints[n].field) = v;
164 }
165 if (d != 1) {
166 *(volatile int *) ((volatile char *) &g_down
167 + fid_dir_ints[n].field) = v;
168 }
169 return 1;
170 }
171
172 return 0;
173}

176/* ---- set latency-dist <uniform|normal> <mean> [sigma] ----
177 *
178 * WHAT: Arms the jitter distribution on the named direction(s); always returns
179 * 1 (the verb is consumed either way, malformed operands producing an
180 * `err` reply and no state change).
181 *
182 * WHY: This is the only fidelity verb with a compound operand, so it owns its
183 * own parse rather than forcing the shared helpers to grow a shape field.
184 *
185 * HOW: 1. Strip the direction token, then sscanf shape + mean + optional sigma.
186 * 2. Refuse a missing/negative mean and an unknown shape name before any
187 * write, so a rejected command leaves the previous distribution armed.
188 * 3. Write mean, shape and sigma together — a partial arm would mean a
189 * normal distribution running with the previous sigma.
190 */
191static int
192cmd_set_latency_dist(char *args, char *reply, size_t rsz)
193{
194 int d = dir_of(args);
195 char shape[16] = "";
196 int mean = -1, sigma = 0, dist;
197 int nf = sscanf(args, "%15s %d %d", shape, &mean, &sigma);
198
199 if (nf < 2 || mean < 0) {
200 fp_reply(reply, rsz, "err: latency-dist <uniform|normal> <mean> [sigma]\n");
201 return 1;
202 }
203
204 if (strcmp(shape, "uniform") == 0) {
205 dist = 0;
206 } else if (strcmp(shape, "normal") == 0) {
207 dist = 1;
208 } else {
209 fp_reply(reply, rsz, "err: unknown distribution\n");
210 return 1;
211 }
212
213 FID_SET_DIR(d, jitter_ms, mean);
214 FID_SET_DIR(d, lat_dist, dist);
215 FID_SET_DIR(d, lat_sigma_ms, sigma);
216 return 1;
217}

220/* Phase-B fidelity verbs, each validated so a malformed operand is refused with
221 * an `err` reply and leaves the lever unchanged. Handles toxicity/slow-close/
222 * connect-delay/refuse/burst/latency-dist plus a validated `rate` (overriding
223 * the unchecked cmd_set_lever rate branch — wired first in cmd_set_dispatch).
224 * Returns 1 when `verb` was one of these, else 0. */
225int
226cmd_set_fidelity(const char *verb, char *args, char *reply, size_t rsz)
227{
228 int ppm;
229
230 if (strcmp(verb, "toxicity") == 0) {
231 int d = dir_of(args);
232 if (fid_pct_to_ppm(args, &ppm) != 0) {
233 fp_reply(reply, rsz, "err: toxicity < 0\n");
234 return 1;
235 }
236 if (d != 2) { g_toxicity_up_ppm = ppm; }
237 if (d != 1) { g_toxicity_down_ppm = ppm; }
238 return 1;
239 }
240
241 if (strcmp(verb, "refuse") == 0) {
242 if (fid_pct_to_ppm(args, &ppm) != 0) {
243 fp_reply(reply, rsz, "err: refuse < 0\n");
244 return 1;
245 }
246 g_refuse_ppm = ppm;
247 return 1;
248 }
249
250 if (strcmp(verb, "connect-delay") == 0 ||
251 strcmp(verb, "accept-delay") == 0) {
252 int ms = atoi(args);
253 if (ms < 0) {
254 fp_reply(reply, rsz, "err: connect-delay < 0\n");
255 return 1;
256 }
257 g_connect_delay_ms = ms;
258 return 1;
259 }
260
261 if (cmd_set_dir_int(verb, args, reply, rsz)) {
262 return 1;
263 }
264
265 if (strcmp(verb, "latency-dist") == 0) {
266 return cmd_set_latency_dist(args, reply, rsz);
267 }
268
269 return 0;

330/*
331 * WHAT: One `<verb> <n> [up|down|both]` scalar lever — its target field in
332 * lever_t and how the operand converts (plain int/long, or a percentage
333 * scaled to ppm).
334 * WHY: These verbs differ only in name, field and conversion, so express the
335 * family as data rather than an else-if ladder (coding-standards §8.6),
336 * matching fid_dir_ints above.

361/* Write one lever value through its offsetof on the direction(s) `d` names
362 * (0 both / 1 up / 2 down) — the field is volatile long for LV_LONG, else
363 * volatile int. */
364static void
365lever_store(int d, size_t field, long v, lever_val_t kind)
366{
367 volatile lever_t *sides[2] = { &g_up, &g_down };
368 int skip[2] = { d == 2, d == 1 };
369
370 for (int i = 0; i < 2; i++) {
371 if (skip[i]) {
372 continue;
373 }
374 volatile char *base = (volatile char *) sides[i];
375 if (kind == LV_LONG) {
376 *(volatile long *) (base + field) = v;
377 } else {
378 *(volatile int *) (base + field) = (int) v;
379 }
380 }
381}

383/* Directional traffic levers (latency/bandwidth/corruption). Each strips an
384 * optional up/down/both token via dir_of() then sets the field on the selected
385 * direction(s). `args` is mutated. Returns 1 if `verb` was a lever, else 0. */
386int
387cmd_set_lever(const char *verb, char *args)
388{
389 for (size_t n = 0; n < sizeof(lever_verbs) / sizeof(lever_verbs[0]); n++) {
390 if (strcmp(verb, lever_verbs[n].verb) != 0) {
391 continue;
392 }
393 int d = dir_of(args);
394 long v = lever_verbs[n].kind == LV_PCT_PPM
395 ? (long) (strtod(args, NULL) * 10000.0 + 0.5)
396 : atol(args);
397 lever_store(d, lever_verbs[n].field, v, lever_verbs[n].kind);
398 return 1;
399 }
400
401 /* Compound-operand levers — each owns its own parse. */
402 if (strcmp(verb, "drip") == 0) {
403 int d = dir_of(args); int b = 0, m = 0;
404 sscanf(args, "%d %d", &b, &m);
405 SET_DIR(d, drip_bytes, b); SET_DIR(d, drip_ms, m);
406 } else if (strcmp(verb, "reorder") == 0) {
407 int d = dir_of(args); double p = 0; int m = -1;
408 sscanf(args, "%lf %d", &p, &m);
409 SET_DIR(d, reorder_ppm, (int) (p * 10000.0 + 0.5));
410 if (m >= 0) { SET_DIR(d, reorder_ms, m); }
411 } else {
412 return 0;
413 }
414 return 1;
415}
