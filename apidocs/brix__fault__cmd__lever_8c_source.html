26/* Strip a trailing up|down|both direction token from `args`; return which lever
27 * set(s) it names: 0 = both, 1 = up only, 2 = down only. */
28int
29dir_of(char *args)
30{
31 size_t L = strlen(args);
32 while (L > 0 && isspace((unsigned char) args[L - 1])) {
33 args[--L] = '\0';
34 }
35 char *sp = strrchr(args, ' ');
36 char *tok = sp ? sp + 1 : args;
37 int d = -1;
38 if (strcmp(tok, "up") == 0) d = 1;
39 else if (strcmp(tok, "down") == 0) d = 2;
40 else if (strcmp(tok, "both") == 0) d = 0;
41 if (d < 0) {
42 return 0; /* no direction token -> both, leave args untouched */
43 }
44 if (sp) {
45 *sp = '\0'; /* drop the token so numeric parse ignores it */
46 } else {
47 args[0] = '\0';
48 }
49 return d;
50}

224/* Autonomous "chaos monkey": every g_chaos_ms, fire a random lever within safe
225 * bounds so a long soak sees an ever-shifting hostile network. Stops when its
226 * generation is superseded (a new `chaos` / `chaos off` / `clear`). */
227void *
228chaos_thread(void *arg)
229{
230 unsigned gen = (unsigned) (uintptr_t) arg;
231 unsigned seed = g_seed ^ (gen * 2654435761u) ^ 0x9e3779b9u;
232 while (__atomic_load_n(&g_chaos_gen, __ATOMIC_SEQ_CST) == gen) {
233 int ms = g_chaos_ms > 0 ? g_chaos_ms : 500;
234 usleep((useconds_t) ms * 1000);
235 if (__atomic_load_n(&g_chaos_gen, __ATOMIC_SEQ_CST) != gen) {
236 break;
237 }
238 char cmd[64];
239 switch (rand_r(&seed) % 8) {
240 case 0: snprintf(cmd, sizeof(cmd), "latency %u", rand_r(&seed) % 400); break;
241 case 1: snprintf(cmd, sizeof(cmd), "jitter %u", rand_r(&seed) % 200); break;
242 case 2: snprintf(cmd, sizeof(cmd), "corrupt %.3f",
243 (rand_r(&seed) % 50) / 100.0); break;
244 case 3: snprintf(cmd, sizeof(cmd), "drop-bytes %.3f",
245 (rand_r(&seed) % 30) / 100.0); break;
246 case 4: snprintf(cmd, sizeof(cmd), "chunk %u",
247 1u + rand_r(&seed) % 2048u); break;
248 case 5: snprintf(cmd, sizeof(cmd), "rate %u",
249 8u + rand_r(&seed) % 4096u); break;
250 case 6: snprintf(cmd, sizeof(cmd), "reorder %.2f 40",
251 (rand_r(&seed) % 2000) / 100.0); break;
252 default: snprintf(cmd, sizeof(cmd), "%s",
253 (rand_r(&seed) % 4 == 0) ? "reset" : "clear"); break;
254 }
255 apply_command(cmd, NULL, 0);
256 }
257 return NULL;

262ext_set_replace(int d, char *findtok, char *repltok, char *reply, size_t rsz)
263{
264 if (!findtok || strcmp(findtok, "off") == 0) {
265 pthread_mutex_lock(&g_ext_lock);

274 if (fl <= 0 || rl < 0) {

419ramp_thread(void *arg)
420{

434 apply_command(cmd, NULL, 0);
