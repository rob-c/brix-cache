1/*
2 * sd_s3_list_internal.h — private seam between the two S3 ListObjectsV2 listers
3 * and the scanner/request plumbing they share.
4 *
5 * The delimited lister (sd_s3_list.c, one directory level) and the flat one
6 * (sd_s3_list_flat.c, the backend-catalog verb) issue the SAME signed request
7 * differing only in whether `delimiter` is present, and read the same response
8 * schema. Everything common lives in sd_s3_list_scan.c and is declared here.
9 *
10 * Private to src/fs/backend/s3/ — nothing outside the S3 driver includes this;
11 * the public entry points are sd_s3_list_page / sd_s3_list_flat_page in sd_s3.h.
12 */
