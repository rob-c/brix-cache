44static int
45contains_bytes(const uint8_t *haystack, size_t haystack_length,
46 const char *needle)
47{
48 size_t needle_length = strlen(needle);
49 size_t index;
50
51 if (needle_length == 0 || needle_length > haystack_length) return 0;
52 for (index = 0; index <= haystack_length - needle_length; index++)
53 if (memcmp(haystack + index, needle, needle_length) == 0) return 1;
54 return 0;
55}

57static int
58raw_set(xrd_conform_execution *execution, brix_conn *connection,
59 const xrd_conform_scratch_object *object, const char *const *names,
60 size_t count, const uint8_t *value, size_t value_length,
61 brix_status *status)
62{

73 size_t payload_length = path_length;
74 size_t index;
75 int rc;

79 if (payload_length > UINT32_MAX || !xrd_conform_reserve_probe(execution, payload_length))
80 return -1;
81 payload = malloc(payload_length);
82 if (payload == NULL) return -1;
83 cursor = payload;

85 cursor += path_length;
86 for (index = 0; index < count; index++) {
87 size_t name_length = strlen(names[index]) + 1;
88
89 xrdw_put_u16(cursor, 0);
90 cursor += 2;
91 memcpy(cursor, names[index], name_length);
92 cursor += name_length;
93 }

95 xrdw_put_u32(cursor, (uint32_t) value_length);
96 cursor += 4;
97 memcpy(cursor, value, value_length);
98 cursor += value_length;
99 }

101 xrdw_fattr_req_pack(&fields, ((ClientRequestHdr *) &request)->body);

105 };
106 rc = brix_roundtrip(connection, &request, &request_payload, &response,
107 status);
108 }
109 free(payload);
110 if (rc == 0)
111 rc = wire_status == kXR_ok && body_length >= 2
112 && body[0] == 0 && body[1] == count ? 0 : -1;
