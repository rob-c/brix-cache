1/*
2 * sss_id_unittest.c — standalone unit driver for the SSS identity registry.
3 *
4 * WHAT: Exercises sss_id.c end to end: bounded setters, the proxied-credential
5 * file reader, register/replace/unregister/find, the hard slot cap, and
6 * the entity view handed to the credential builder.
7 * WHY: The registry's whole value is that a lookup miss NEVER substitutes
8 * another identity. That is a property no end-to-end test can show
9 * convincingly (a miss looks like an auth failure either way), so it is
10 * pinned here, at the only layer where the substitution would be visible.
11 * HOW: No harness: a `check()` macro counts failures and main() returns
12 * non-zero if any fired, so the Python wrapper
13 * (tests/test_release20_sss_unit.py) just asserts the exit status and the
14 * "all checks passed" line.
15 *
16 * gcc -Wall -Wextra -Werror -I src -I client/lib -o /tmp/sss_id_ut \
17 * client/lib/auth/sss/sss_id.c src/core/compat/sss_entity.c \
18 * src/core/compat/sss_bf.c src/core/compat/crc32_ieee.c \
19 * client/lib/auth/sss/sss_id_unittest.c -lcrypto -pthread
20 */

127static void
128test_creds_file(void)
129{
130 static const char TEMPLATE[] = "/tmp/brix_sss_id_ut_XXXXXX";
131 char path[sizeof(TEMPLATE)];
132 uint8_t buf[128];
133 size_t len = 0;
134 uint8_t payload[37];
135 int fd;
136
137 memset(payload, 0x5A, sizeof(payload));
138 memcpy(path, TEMPLATE, sizeof(path));
139 fd = mkstemp(path);
140 check("file: mkstemp", fd >= 0);
141 if (fd < 0) {
142 return;
143 }
144 check("file: write", write(fd, payload, sizeof(payload)) == (ssize_t) sizeof(payload));
145 close(fd);
146
147 check("file: read", brix_sss_creds_read_file(path, buf, sizeof(buf), &len) == 0);
148 check("file: length", len == sizeof(payload));
149 check("file: bytes", memcmp(buf, payload, sizeof(payload)) == 0);
150
151 check("file: too small a buffer fails closed",
152 brix_sss_creds_read_file(path, buf, 8, &len) == -1);
153 check("file: failed read reports no length", len == 0);
154
155 check("file: missing path fails",
156 brix_sss_creds_read_file("/nonexistent/brix/sss/creds", buf,
157 sizeof(buf), &len) == -1);
158 check("file: NULL path fails",
159 brix_sss_creds_read_file(NULL, buf, sizeof(buf), &len) == -1);
160 check("file: empty path fails",
161 brix_sss_creds_read_file("", buf, sizeof(buf), &len) == -1);
162 check("file: a directory is not a credential",
163 brix_sss_creds_read_file("/tmp", buf, sizeof(buf), &len) == -1);
164 unlink(path);
165
166 /* mkstemp rewrote the template in place; restore it before reusing it. */
167 memcpy(path, TEMPLATE, sizeof(path));
168 fd = mkstemp(path);
169 check("file: mkstemp (empty)", fd >= 0);
170 if (fd >= 0) {
171 close(fd);
172 check("file: an empty file is not a credential",
173 brix_sss_creds_read_file(path, buf, sizeof(buf), &len) == -1);
174 unlink(path);
175 }
176}

181static void
182test_registry_basics(void)
183{
184 brix_sss_id_registry *reg = brix_sss_id_create();
185 brix_sss_ident alice, bob, got;
186
187 check("reg: create", reg != NULL);
188 if (reg == NULL) {
189 return;
190 }
191 check("reg: starts empty", brix_sss_id_count(reg) == 0);
192
193 check("reg: alice", brix_sss_ident_set(&alice, "alice", "cms", "prod", NULL, "e1") == 0);
194 check("reg: bob", brix_sss_ident_set(&bob, "bob", "atlas", NULL, NULL, NULL) == 0);
195
196 check("reg: register alice", brix_sss_id_register(reg, "u1", &alice) == 0);
197 check("reg: register bob", brix_sss_id_register(reg, "u2", &bob) == 0);
198 check("reg: count", brix_sss_id_count(reg) == 2);
199
200 check("reg: find alice", brix_sss_id_find(reg, "u1", &got) == 0);
201 check("reg: alice name", strcmp(got.name, "alice") == 0);
202 check("reg: alice vorg", strcmp(got.vorg, "cms") == 0);
203 check("reg: alice endo", strcmp(got.endo, "e1") == 0);
204
205 check("reg: find bob", brix_sss_id_find(reg, "u2", &got) == 0);
206 check("reg: bob name", strcmp(got.name, "bob") == 0);
207
208 /* Replace in place: same lid, new identity, no new slot. */
209 check("reg: replace u1", brix_sss_id_register(reg, "u1", &bob) == 0);
210 check("reg: replace did not grow", brix_sss_id_count(reg) == 2);
211 check("reg: replaced find", brix_sss_id_find(reg, "u1", &got) == 0);
212 check("reg: replaced value", strcmp(got.name, "bob") == 0);
213
214 /* The default slot is a slot like any other. */
215 check("reg: default slot via NULL", brix_sss_id_register(reg, NULL, &alice) == 0);
216 check("reg: default slot via \"\"", brix_sss_id_find(reg, "", &got) == 0);
217 check("reg: default value", strcmp(got.name, "alice") == 0);
218
219 check("reg: unregister", brix_sss_id_unregister(reg, "u2") == 0);
220 check("reg: count after unregister", brix_sss_id_count(reg) == 2);
221 check("reg: unregister twice fails", brix_sss_id_unregister(reg, "u2") == -1);
222 check("reg: survivor still there", brix_sss_id_find(reg, "u1", &got) == 0);
223
224 brix_sss_id_destroy(reg);
225 brix_sss_id_destroy(NULL); /* NULL-safe */
226}

229/*
230 * The security property: an unregistered lid is a MISS, never a substitution.
231 * A front end that got a fallback identity here would present one user's
232 * request under another user's name.
233 */
234static void
235test_registry_miss_is_fail_closed(void)
236{
237 brix_sss_id_registry *reg = brix_sss_id_create();
238 brix_sss_ident alice, got;
239
240 check("miss: create", reg != NULL);
241 if (reg == NULL) {
242 return;
243 }
244 check("miss: alice", brix_sss_ident_set(&alice, "alice", "cms", NULL, NULL, NULL) == 0);
245 check("miss: register", brix_sss_id_register(reg, "u1", &alice) == 0);
246
247 memset(&got, 0xEE, sizeof(got));
248 check("miss: unknown lid is a miss", brix_sss_id_find(reg, "u9", &got) == -1);
249 check("miss: output zeroed on a miss", got.name[0] == '\0' && got.creds_len == 0);
250
251 memset(&got, 0xEE, sizeof(got));
252 check("miss: the sole slot is NOT a fallback for the default lid",
253 brix_sss_id_find(reg, "", &got) == -1);
254 check("miss: default-lid output zeroed", got.name[0] == '\0');
255
256 memset(&got, 0xEE, sizeof(got));
257 check("miss: a prefix does not match", brix_sss_id_find(reg, "u", &got) == -1);
258 memset(&got, 0xEE, sizeof(got));
259 check("miss: an extension does not match", brix_sss_id_find(reg, "u10", &got) == -1);
260
261 check("miss: NULL registry fails", brix_sss_id_find(NULL, "u1", &got) == -1);
262 check("miss: NULL out fails", brix_sss_id_find(reg, "u1", NULL) == -1);
263 check("miss: register on NULL registry fails",
264 brix_sss_id_register(NULL, "u1", &alice) == -1);
265 check("miss: register NULL ident fails", brix_sss_id_register(reg, "u1", NULL) == -1);
266 check("miss: unregister on NULL registry fails",
267 brix_sss_id_unregister(NULL, "u1") == -1);
268 check("miss: count of NULL is 0", brix_sss_id_count(NULL) == 0);
269
270 brix_sss_id_destroy(reg);
271}

274static void
275test_registry_bounds(void)
276{
277 brix_sss_id_registry *reg = brix_sss_id_create();
278 brix_sss_ident id;
279 char lid[XRDC_SSS_LID_MAX + 8];
280 int i, over = 0;
281
282 check("bounds: create", reg != NULL);
283 if (reg == NULL) {
284 return;
285 }
286 check("bounds: ident", brix_sss_ident_set(&id, "alice", NULL, NULL, NULL, NULL) == 0);
287
288 memset(lid, 'x', sizeof(lid) - 1);
289 lid[sizeof(lid) - 1] = '\0';
290 check("bounds: over-long lid refused", brix_sss_id_register(reg, lid, &id) == -1);
291 lid[XRDC_SSS_LID_MAX - 1] = '\0';
292 check("bounds: cap-1 lid accepted", brix_sss_id_register(reg, lid, &id) == 0);
293 check("bounds: unregister it", brix_sss_id_unregister(reg, lid) == 0);
294
295 for (i = 0; i < XRDC_SSS_ID_SLOTS_MAX + 4; i++) {
296 char key[XRDC_SSS_LID_MAX];
297
298 snprintf(key, sizeof(key), "u%d", i);
299 if (brix_sss_id_register(reg, key, &id) != 0) {
300 over++;
301 }
302 }
303 check("bounds: the slot cap holds",
304 brix_sss_id_count(reg) == XRDC_SSS_ID_SLOTS_MAX);
305 check("bounds: the overflow was refused, not silently dropped", over == 4);
306
307 brix_sss_id_destroy(reg);
308}
