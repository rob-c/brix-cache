1/*
2 * prop_xattr.c — xattr-based lock persistence for WebDAV.
3 *
4 * A WebDAV lock is encoded as a single xattr on the locked resource:
5 * token=<tok>|owner=<owner>|expires=<msec>|scope=<exclusive|shared>|depth=<infinity|0>
6 *
7 * XATTR_CREATE semantics make lock creation atomic across workers: if two
8 * workers race on the same unlocked path, exactly one setxattr(XATTR_CREATE)
9 * succeeds and the other gets EEXIST → NGX_DECLINED → 423 Locked.
10 */

21/*
22 * Phase 40: lock xattrs must be written/read/removed AS THE MAPPED USER under
23 * impersonation, else the worker (svc) cannot setxattr on the user-owned lock
24 * file (EACCES) and LOCK/UNLOCK break. These helpers take the request so they
25 * can resolve the export root and route through the VFS xattr surface, which
26 * delegates to brix_*xattr_confined_canon (the broker when map mode is active,
27 * the raw path-based syscall otherwise) while adding the OP_XATTR metric +
28 * access-log line — confinement and errno behaviour are unchanged.
29 */

31/*
32 * Build a transient VFS ctx for a confined xattr op on `path`. The xattr
33 * family is not allow_write-gated, so the allow_write flag threaded through
34 * does not affect set/remove behaviour. The _ns (namespace) credential build
35 * is deliberate: minting is reserved for data-plane GET/PUT/COPY sites, so a
36 * lock xattr op that needs a credential the user doesn't already have falls
37 * back per the configured storage_credential_fallback policy, same as every
38 * other namespace-only VFS ctx in this codebase (see mv.c's probe ctxs).
39 */
40static void
41webdav_lock_vfs_ctx_init(ngx_http_request_t *r, const char *path,
42 brix_vfs_ctx_t *vctx)
43{
44 ngx_http_brix_webdav_loc_conf_t *conf =
45 ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
46
47 webdav_vfs_ctx_build_ns(r, conf, path, vctx);
48 /* Phase-3 T1: route through the export's selected storage backend (NULL =
49 * default POSIX) so a remote-backed export's cred gate (brix_vfs_ns_cred,
50 * keyed on the leaf driver's stat_cred/setxattr_cred capability) actually
51 * engages for lock-state xattr ops — mirrors every other namespace ctx in
52 * this codebase (put.c, mv.c). Without this, vctx->sd stays NULL and the
53 * gate is structurally unreachable (brix_vfs_ctx_driver(ctx) == NULL), so
54 * a deny-mode no-cred user would silently touch the lock xattr via the
55 * bare local-fs path. On a LOCAL (POSIX, non-instance) export this is a
56 * no-op (brix_webdav_backend_instance returns NULL), so behaviour there is
57 * unchanged. */
58 vctx->sd = brix_webdav_backend_instance(conf, r->connection->log);
59}

70ngx_int_t
71webdav_lock_xattr_write(ngx_http_request_t *r, const char *path,
72 const webdav_lock_xattr_t *e, int flags)
73{
74 ngx_log_t *log = r->connection->log;
75 brix_vfs_ctx_t vctx;
76 char buf[WEBDAV_LOCK_XATTR_MAXLEN];
77
78 if (brix_lock_record_encode(e, buf, sizeof(buf)) != NGX_OK) {
79 ngx_log_error(NGX_LOG_ERR, log, 0,
80 "brix_webdav: lock xattr encode failed for \"%s\"", path);
81 return NGX_ERROR;
82 }
83
84 webdav_lock_vfs_ctx_init(r, path, &vctx);
85
86 if (brix_vfs_setxattr(&vctx, WEBDAV_LOCK_XATTR_KEY, buf, strlen(buf),
87 flags) != NGX_OK)
88 {
89 if (errno == EEXIST) {
90 return NGX_DECLINED; /* XATTR_CREATE race — another worker won */
91 }
92 ngx_log_error(NGX_LOG_ERR, log, errno,
93 "brix_webdav: setxattr lock on \"%s\" failed", path);
94 return NGX_ERROR;
95 }
96
97 return NGX_OK;
98}

100ngx_int_t
101webdav_lock_xattr_read(ngx_http_request_t *r, const char *path,
102 webdav_lock_xattr_t *e)
103{
104 ngx_log_t *log = r->connection->log;
105 brix_vfs_ctx_t vctx;
106 char buf[WEBDAV_LOCK_XATTR_MAXLEN];
107 ssize_t n;
108
109 webdav_lock_vfs_ctx_init(r, path, &vctx);
110
111 n = brix_vfs_getxattr(&vctx, WEBDAV_LOCK_XATTR_KEY, buf, sizeof(buf) - 1);
112 if (n < 0) {
113 /* No lock present, OR a backend that cannot store xattrs at all (object /
114 * remote root:// stores) — either way the resource carries no WebDAV lock,
115 * so a write may proceed. EACCES/EPERM here (added by the phase-2 backend
116 * credential bind above) means the per-user backend credential gate
117 * denied THIS lock-state probe — it does NOT mean "proceed unlocked and
118 * unchecked": the actual write/delete/move this check gates re-runs the
119 * SAME gate on its own data-plane VFS ctx and is independently refused
120 * with a clean 403 if the user has no credential, so declining here
121 * (treating the lock as unknown/absent) cannot bypass the deny — it can
122 * only, in the worst case, miss an existing lock for a caller who is
123 * about to be denied by the write path anyway. */
124 if (errno == ENODATA || errno == ENOATTR || errno == ENOENT
125 || errno == ENOTSUP || errno == EOPNOTSUPP || errno == ENOSYS
126 || errno == EACCES || errno == EPERM)
127 {
128 return NGX_DECLINED;
129 }
130 ngx_log_error(NGX_LOG_ERR, log, errno,
131 "brix_webdav: getxattr lock on \"%s\" failed", path);
132 return NGX_ERROR;
133 }
134
135 return brix_lock_record_decode(buf, (size_t) n, e);
136}

138ngx_int_t
139webdav_lock_xattr_delete(ngx_http_request_t *r, const char *path)
140{
141 ngx_log_t *log = r->connection->log;
142 brix_vfs_ctx_t vctx;
143
144 webdav_lock_vfs_ctx_init(r, path, &vctx);
145
146 if (brix_vfs_removexattr(&vctx, WEBDAV_LOCK_XATTR_KEY) != NGX_OK) {
147 if (errno == ENODATA || errno == ENOATTR || errno == ENOENT
148 || errno == ENOTSUP || errno == EOPNOTSUPP || errno == ENOSYS)
149 {
150 return NGX_OK; /* idempotent (incl. backends without xattr) */
151 }
152 ngx_log_error(NGX_LOG_WARN, log, errno,
153 "brix_webdav: removexattr lock on \"%s\" failed", path);
154 return NGX_ERROR;
155 }
156
157 return NGX_OK;
158}
