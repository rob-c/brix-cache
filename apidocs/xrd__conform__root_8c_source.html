9/* ---- Narrow an implicit ROOT profile from authenticated protocol claims ----
10 *
11 * WHAT: Reclassifies untouched ledger rows using the observed server role.
12 * WHY: Default scope must follow discovery rather than assume every family.
13 * HOW: Prefer cache/proxy attributes, then manager/data role bits; preserve
14 * explicit profiles and multi-plane unions, and never widen prior scope.
15 */
16static void
17apply_discovered_profile(xrd_conform_execution *execution, uint32_t flags)
18{
19 const char *profile = "full";
20 size_t index;
21
22 if (execution->run->profile_explicit
23 || execution->run->also_count != 0) return;
24 if ((flags & kXR_attrCache) != 0) profile = "cache";
25 else if ((flags & kXR_attrProxy) != 0) profile = "proxy";
26 else if ((flags & kXR_isManager) != 0
27 && (flags & kXR_isServer) == 0) profile = "manager";
28 else if ((flags & kXR_isServer) != 0
29 && (flags & kXR_isManager) == 0) profile = "data";
30 execution->run->config.profile = profile;
31 for (index = 0; index < execution->results.count; index++) {
32 brix_conform_result *result = &execution->results.results[index];
33
34 if (result->verdict != BRIX_CONFORM_VERDICT_INCONCLUSIVE) continue;
35 if (!brix_conform_clause_selected(result->clause,
36 &execution->run->config, result->reason,
37 sizeof(result->reason)))
38 result->verdict = BRIX_CONFORM_VERDICT_SKIPPED;
39 }
40}

42/* ---- Mark every selected row in a family with one verdict ----
43 *
44 * WHAT: Applies an explicit family-wide result to rows not filtered from scope.
45 * WHY: A missing advertised capability has one mandated skip reason per row.
46 * HOW: Walk results, match family, preserve already-skipped scope decisions.
47 */
48static void
49family_result(xrd_conform_execution *execution, const char *family,
50 brix_conform_verdict verdict, const char *reason)
51{
52 size_t index;
53
54 for (index = 0; index < execution->results.count; index++) {
55 brix_conform_result *result = &execution->results.results[index];
56
57 if (strcmp(result->clause->family, family) != 0
58 || result->verdict == BRIX_CONFORM_VERDICT_SKIPPED) continue;
59 result->verdict = verdict;
60 snprintf(result->reason, sizeof(result->reason), "%s", reason);
61 }
62}

64/* ---- Plant one bounded deterministic remote object ----
65 *
66 * WHAT: Creates/truncates path and writes PRF bytes through the typed API.
67 * WHY: A planted object gives read probes an independent content oracle.
68 * HOW: Generate and write 64 KiB chunks, under the prior scratch path gate.
69 */
70int
71xrd_conform_plant_object(xrd_conform_execution *execution,
72 brix_conn *connection,
73 const xrd_conform_scratch_object *object, uint64_t size,
74 brix_status *status)
75{
76 uint8_t buffer[CONFORM_READ_CHUNK];
77 brix_file file;
78 uint64_t offset = 0;
79 char reason[64];
80
81 if (brix_file_open_write(connection, object->path, 1, 0, &file, status) != 0)
82 return -1;
83 while (offset < size) {
84 size_t length = size - offset < sizeof(buffer)
85 ? (size_t) (size - offset) : sizeof(buffer);
86
87 if (!xrd_conform_reserve(execution, length, reason, sizeof(reason))) {
88 brix_file_close(connection, &file, status);
89 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
90 return -1;
91 }
92 brix_conform_fill(&execution->run->key, offset, buffer, length);
93 if (brix_file_write(connection, &file, (int64_t) offset, buffer,
94 length, status) != 0) {
95 brix_file_close(connection, &file, status);
96 return -1;
97 }
98 offset += length;
99 }
100 return brix_file_close(connection, &file, status);
101}

103/* ---- Read an entire object under configured ceilings ----
104 *
105 * WHAT: Returns an owned byte oracle for the remote object.
106 * WHY: Read-only degraded mode must learn content before grading fragments.
107 * HOW: Allocate the stat size, read sequential chunks and reject short data.
108 */
109uint8_t *
110xrd_conform_read_baseline(xrd_conform_execution *execution,
111 brix_conn *connection,
112 brix_file *file, uint64_t size, brix_status *status)
113{
114 uint8_t *bytes;
115 uint64_t offset = 0;
116 char reason[64];
117
118 if (size > SIZE_MAX || size > execution->run->config.byte_limit) {
119 brix_status_set(status, XRDC_EUSAGE, 0, "object exceeds byte ceiling");
120 return NULL;
121 }
122 bytes = malloc(size == 0 ? 1 : (size_t) size);
123 if (bytes == NULL) {
124 brix_status_set(status, XRDC_EIO, errno, "baseline allocation failed");
125 return NULL;
126 }
127 while (offset < size) {
128 size_t length = size - offset < CONFORM_READ_CHUNK
129 ? (size_t) (size - offset) : CONFORM_READ_CHUNK;
130 ssize_t got;
131
132 if (!xrd_conform_reserve(execution, length, reason, sizeof(reason))) {
133 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
134 free(bytes);
135 return NULL;
136 }
137 got = brix_file_read(connection, file, (int64_t) offset,
138 bytes + offset, length, status);
139 if (got != (ssize_t) length) {
140 if (got >= 0) brix_status_set(status, XRDC_EPROTO, 0,
141 "short baseline read at %llu", (unsigned long long) offset);
142 free(bytes);
143 return NULL;
144 }
145 offset += length;
146 }
147 return bytes;
148}

150/* ---- Grade one ordinary read against learned bytes ----
151 *
152 * WHAT: Issues a bounded read and records exact length/content conformance.
153 * WHY: RD boundary clauses require direct evidence at their named extent.
154 * HOW: Reserve resources, call typed read, compare return and optional bytes.
155 */
156void
157xrd_conform_grade_read(xrd_conform_execution *execution,
158 brix_conn *connection,
159 brix_file *file, const uint8_t *baseline, uint64_t size, const char *id,
160 int64_t offset, size_t requested)
161{
162 uint8_t *buffer;
163 uint64_t available;
164 size_t expected;
165 ssize_t got;
166 brix_status status;
167 char reason[64];
168 char expected_text[64];
169 char received_text[96];
170
171 if (offset < 0) available = 0;
172 else available = (uint64_t) offset < size ? size - (uint64_t) offset : 0;
173 expected = available < requested ? (size_t) available : requested;
174 buffer = malloc(requested == 0 ? 1 : requested);
175 if (buffer == NULL) {
176 xrd_conform_result_set(execution, id, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
177 "local-allocation-failed", NULL, NULL);
178 return;
179 }
180 if (!xrd_conform_reserve(execution, requested, reason, sizeof(reason))) {
181 xrd_conform_result_set(execution, id, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
182 reason, NULL, NULL);
183 free(buffer);
184 return;
185 }
186 brix_status_clear(&status);
187 got = brix_file_read(connection, file, offset, buffer, requested, &status);
188 snprintf(expected_text, sizeof(expected_text), "length=%zu byte-exact", expected);
189 snprintf(received_text, sizeof(received_text), "length=%lld kxr=%d",
190 (long long) got, status.kxr);
191 if (got == (ssize_t) expected
192 && (expected == 0 || memcmp(buffer, baseline + (size_t) offset,
193 expected) == 0)) {
194 xrd_conform_result_set(execution, id, BRIX_CONFORM_VERDICT_CONFORMANT,
195 "typed-read-byte-exact", expected_text, received_text);
196 } else {
197 xrd_conform_result_set(execution, id, BRIX_CONFORM_VERDICT_DIVERGENT,
198 got < 0 ? "typed-read-error" : "read-length-or-content-mismatch",
199 expected_text, received_text);
200 }
201 free(buffer);
202}

239/* ---- Grade selected RD assertions with a byte oracle ----
240 *
241 * WHAT: Runs the safe boundary/idempotence subset of the RD ledger.
242 * WHY: These assertions are fully observable through the typed read API.
243 * HOW: Exercise exact named offsets, then separately test negative offset.
244 */
245void
246xrd_conform_run_rd(xrd_conform_execution *execution, brix_conn *connection,
247 brix_file *file, const uint8_t *baseline, uint64_t size)
248{
249 uint64_t offset;
250
251 if (size <= SIZE_MAX) xrd_conform_grade_read(execution, connection, file, baseline,
252 size, "RD-001", 0, (size_t) size);
253 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-002", 0, 0);
254 if (size <= INT64_MAX) {
255 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-003",
256 (int64_t) size, 1);
257 if (size < INT64_MAX) xrd_conform_grade_read(execution, connection, file, baseline,
258 size, "RD-004", (int64_t) size + 1, 1);
259 }
260 if (size > 0) {
261 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-005",
262 (int64_t) size - 1, 2);
263 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-006", 0, 1);
264 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-007",
265 (int64_t) size - 1, 1);
266 }
267 for (offset = 4095; offset <= 4097; offset++) {
268 if (offset < size) xrd_conform_grade_read(execution, connection, file, baseline,
269 size, "RD-008", (int64_t) offset, 1);
270 }
271 if (size == 0) xrd_conform_grade_read(execution, connection, file, baseline, size,
272 "RD-010", 0, 1);
273 if (size > 0) {
274 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-015", 0,
275 size < 4096 ? (size_t) size : 4096);
276 xrd_conform_grade_read(execution, connection, file, baseline, size, "RD-015", 0,
277 size < 4096 ? (size_t) size : 4096);
278 }
279 grade_invalid_offset(execution, connection, file, "RD-030", -1);
280 grade_invalid_offset(execution, connection, file, "RD-032", INT64_MIN);
281 xrd_conform_probe_read_partitions(execution, connection, file, baseline,
282 size);
283 xrd_conform_probe_read_substreams(execution, connection, file, baseline,
284 size);
285}

366/* ---- Probe one open root object ----
367 *
368 * WHAT: Learns the baseline, runs RD/PGR and records cross-plane digest facts.
369 * WHY: Object/read ownership should be independent from connection cleanup.
370 * HOW: Stat, open, slurp under ceilings, grade, close on every open path.
371 */
372static int
373probe_root_object(xrd_conform_execution *execution, brix_conn *connection,
374 const char *path, brix_status *status)
375{
376 brix_statinfo statinfo;
377 brix_file file;
378 uint8_t *baseline;
379 brix_status close_status;
380 int connection_poisoned = 0;
381 int rc = -1;
382
383 brix_status_clear(status);
384 if (brix_stat(connection, path, &statinfo, status) != 0
385 || statinfo.size < 0 || (statinfo.flags & kXR_isDir) != 0) {
386 BRIX_CLIENT_LOG("ERROR",
387 "xrd conform: target object is not readable: %s\n",
388 status->msg);
389 return -1;
390 }
391 if (brix_file_open_read(connection, path, &file, status) != 0) {
392 BRIX_CLIENT_LOG("ERROR",
393 "xrd conform: target object open failed: %s\n",
394 status->msg);
395 return -1;
396 }
397 baseline = xrd_conform_read_baseline(execution, connection, &file,
398 (uint64_t) statinfo.size, status);
399 if (baseline == NULL) {
400 BRIX_CLIENT_LOG("ERROR",
401 "xrd conform: baseline read failed: %s\n", status->msg);
402 } else {
403 connection_poisoned = xrd_conform_probe_selected_object(execution,
404 connection, &file, path, baseline, (uint64_t) statinfo.size);
405 rc = 0;
406 }
407 free(baseline);
408 if (connection_poisoned) return rc;
409 brix_status_clear(&close_status);
410 if (brix_file_close(connection, &file, &close_status) != 0) rc = -1;
411 else probe_closed_handle(execution, connection, &file);
412 return rc;
413}

537/* ---- Run the root read-plane assessment ----
538 *
539 * WHAT: Returns zero after usable endpoint evidence, or run-incomplete code 3.
540 * WHY: Parse/connect/stat/open failures prevent any protocol conclusion.
541 * HOW: Optionally plant beneath scratch, learn a baseline, probe and clean up.
542 */
543int
544xrd_conform_run_root(xrd_conform_execution *execution)
545{
546 brix_url url;
547 brix_conn connection;
548 brix_status status;
549 xrd_conform_scratch_object scratch_object;
550 const char *path;
551 int planted = 0;
552 int rc = 3;
553
554 brix_status_clear(&status);
555 if (brix_endpoint_parse(execution->target, &url, &status) != 0
556 || (url.scheme != XRDC_SCHEME_ROOT && url.scheme != XRDC_SCHEME_ROOTS)) {
557 BRIX_CLIENT_LOG("ERROR",
558 "xrd conform: target is not a valid root:// URL: %s\n",
559 status.msg);
560 return 3;
561 }
562 if (brix_connect(&connection, &url, &execution->run->opts, &status) != 0) {
563 BRIX_CLIENT_LOG("ERROR",
564 "xrd conform: target connection failed: %s\n", status.msg);
565 return grade_root_connect_failure(execution, &status);
566 }
567 apply_discovered_profile(execution, connection.server_flags);
568 xrd_conform_probe_protocol_claims(execution, &url, &connection);
569 xrd_conform_probe_cms_manager(execution, &connection, url.path);
570 xrd_conform_probe_handshake_negatives(execution, &url);
571 xrd_conform_probe_framing(execution, &url);
572 xrd_conform_probe_framing_hostile(execution, &url);
573 xrd_conform_probe_login(execution, &url, &connection);
574 xrd_conform_probe_tls(execution, &url, &connection, 0);
575 xrd_conform_probe_flow(execution, &url);
576 if (xrd_conform_readonly_exact_selected(execution)) {
577 xrd_conform_probe_readonly_exact(execution, &connection);
578 rc = 0;
