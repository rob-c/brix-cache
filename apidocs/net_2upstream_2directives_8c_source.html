10/* ---- Split an upstream address into host + validated port ----
11 *
12 * WHAT: Parses a null-terminated upstream address copy into its host and port
13 * components, supporting two formats: an IPv6 literal "[addr]:port" (detected
14 * by a leading '[', split on ']' then the following ':') or a
15 * "hostname:port" / "IPv4:port" form (split on the LAST ':' via strrchr).
16 * Duplicates the host bytes into xcf->upstream_host via brix_pstrdupz(), parses
17 * the port digits with strtol(), and validates 1 <= port <= 65535 with no
18 * trailing non-digit characters. On success writes the parsed port into
19 * *port_out and returns NGX_OK. On any malformed address, missing port, host
20 * allocation failure, or out-of-range port it logs an emerg-level message
21 * (referencing raw, the original directive token) and returns NGX_ERROR
22 * without touching *port_out.
23 *
24 * WHY: Upstream directives commonly use "[::1]:1094" for IPv6 loopback, which
25 * needs bracket-based splitting rather than a naive colon split; the strrchr
26 * split on the last colon keeps the non-IPv6 path tolerant of colons in the
27 * host portion. Isolating the parse+validate logic keeps the directive setter
28 * flat and makes the accept/reject decision independently reviewable. Port
29 * range 1-65535 rejects zero, negative, and above-range values so nginx never
30 * starts with an upstream address that would fail every connection.
31 *
32 * HOW:
33 * 1. If the address begins with '[', locate the closing ']'; require it to be
34 * immediately followed by ':', else log "invalid address" and fail.
35 * 2. Duplicate the bytes between '[' and ']' as the host; parse the port from
36 * two characters past ']' (skipping "]:").
37 * 3. Otherwise locate the last ':'; if absent log "missing port" and fail.
38 * 4. Duplicate the bytes before that ':' as the host; parse the port from one
39 * character past it.
40 * 5. Reject a trailing non-digit (endp not at the terminator) or a port
41 * outside 1-65535 with an "invalid port" message.
42 * 6. Store the port in *port_out and return NGX_OK.
43 */
44static ngx_int_t
45brix_upstream_parse_host_port(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *xcf,
46 char *addr_copy, ngx_str_t *raw, long *port_out)
47{
48 char *endp;
49 long pnum;
50
51 if (addr_copy[0] == '[') {
52 /* IPv6 literal [addr]:port */
53 char *rb = strchr(addr_copy, ']');
54 if (rb == NULL || *(rb + 1) != ':') {
55 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
56 "brix_upstream: invalid address \"%V\"", raw);
57 return NGX_ERROR;
58 }
59 size_t hostlen = (size_t)(rb - addr_copy - 1);
60 if (brix_pstrdupz(cf->pool, &xcf->upstream_host,
61 (u_char *) addr_copy + 1, hostlen) != NGX_OK) {
62 return NGX_ERROR;
63 }
64 pnum = strtol(rb + 2, &endp, 10);
65 } else {
66 /* hostname:port or IPv4:port - split on last colon */
67 char *colon = strrchr(addr_copy, ':');
68 if (colon == NULL) {
69 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
70 "brix_upstream: missing port in \"%V\"", raw);
71 return NGX_ERROR;
72 }
73 size_t hostlen = (size_t)(colon - addr_copy);
74 if (brix_pstrdupz(cf->pool, &xcf->upstream_host,
75 (u_char *) addr_copy, hostlen) != NGX_OK) {
76 return NGX_ERROR;
77 }
78 pnum = strtol(colon + 1, &endp, 10);
79 }
80
81 if (*endp != '\0' || pnum <= 0 || pnum > BRIX_MAX_PORT) {
82 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
83 "brix_upstream: invalid port in \"%V\"", raw);
84 return NGX_ERROR;
85 }
86
87 *port_out = pnum;
88 return NGX_OK;
89}

91/*
92 * brix_upstream_preresolve — register the redirector as a runtime DNS target.
93 *
94 * WHAT: Hands "host:port" to the phase-116 target registry and publishes the
95 * registry-owned ngx_addr_t as xcf->upstream_addr. An IP literal is final
96 * at once; a hostname stays socklen == 0 until the worker resolves it.
97 * WHY: A redirector whose name does not resolve at `nginx -t` must not stop
98 * the server (I-DNS-1), and the answer must follow the visible resolv.conf
99 * for the process lifetime — including after the redirector moves.
100 * HOW: brix_dns_target_register(); start.c reads the target round-robin via
101 * brix_dns_target_next() and never touches libc DNS on the event loop.
102 */
103static ngx_int_t
104brix_upstream_preresolve(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *xcf)
105{
106 ngx_str_t host;
107
108 host.data = xcf->upstream_host.data;
109 host.len = ngx_strlen(xcf->upstream_host.data);
110 xcf->upstream_dns = brix_dns_target_register(cf, "brix_upstream", &host,
111 (in_port_t) xcf->upstream_port,
112 BRIX_AF_AUTO, SOCK_STREAM,
113 &xcf->common.dns);
114 if (xcf->upstream_dns == NULL) {
115 return NGX_ERROR;
116 }
117 xcf->upstream_addr = &xcf->upstream_dns->addr;
118 return NGX_OK;
119}

121/* ---- brix_upstream directive setter ----
122 *
123 * WHAT: nginx config-directive handler for the upstream redirector address. Copies
124 * the directive argument, splits and validates it into xcf->upstream_host and
125 * xcf->upstream_port, pre-resolves the host into xcf->upstream_addr, logs a
126 * notice confirming the configured target, and returns NGX_CONF_OK. Returns
127 * NGX_CONF_ERROR (with an emerg-level message already emitted) if the argument
128 * cannot be duplicated or fails host/port parsing.
129 *
130 * WHY: This is the single entry point nginx invokes when it encounters the
131 * directive. Configuration parsing runs once at startup with no concurrent
132 * access, so all allocations come from cf->pool and are freed with the config
133 * lifecycle. Delegating parse/validate and pre-resolution to helpers keeps this
134 * orchestrator a flat, reviewable sequence.
135 *
136 * HOW:
137 * 1. Duplicate the directive value (value[1]) into a null-terminated copy;
138 * fail on allocation error.
139 * 2. Parse and validate host + port via brix_upstream_parse_host_port(); fail
140 * if it rejects the address.
141 * 3. Store the validated port, then pre-resolve the host address.
142 * 4. Log the configured target at notice level and return NGX_CONF_OK.
143 */
144char *
145brix_conf_set_upstream(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
146{
147 ngx_stream_brix_srv_conf_t *xcf = conf;
148 ngx_str_t *value;
149 char *addr_copy;
150 long pnum = 0;
151
152 value = cf->args->elts;
153 (void) cmd;
154
155 addr_copy = brix_pstrdup_z(cf->pool, &value[1]);
156 if (addr_copy == NULL) {
157 return NGX_CONF_ERROR;
158 }
159
160 if (brix_upstream_parse_host_port(cf, xcf, addr_copy, &value[1], &pnum)
161 != NGX_OK)
162 {
163 return NGX_CONF_ERROR;
164 }
165 xcf->upstream_port = (uint16_t) pnum;
166
167 /* phase-116: register the redirector as a runtime DNS target; request
168 * handlers never call getaddrinfo() on the event-loop thread. */
169 if (brix_upstream_preresolve(cf, xcf) != NGX_OK) {
170 return NGX_CONF_ERROR;
171 }
172
173 BRIX_CONF_LOG(NGX_LOG_NOTICE, cf, 0,
174 "brix: upstream redirector: %s:%d",
175 (char *) xcf->upstream_host.data, (int) xcf->upstream_port);
176
177 return NGX_CONF_OK;
178}
