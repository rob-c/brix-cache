9/* ---- Split an upstream address into host + validated port ----
10 *
11 * WHAT: Parses a null-terminated upstream address copy into its host and port
12 * components, supporting two formats: an IPv6 literal "[addr]:port" (detected
13 * by a leading '[', split on ']' then the following ':') or a
14 * "hostname:port" / "IPv4:port" form (split on the LAST ':' via strrchr).
15 * Duplicates the host bytes into xcf->upstream_host via brix_pstrdupz(), parses
16 * the port digits with strtol(), and validates 1 <= port <= 65535 with no
17 * trailing non-digit characters. On success writes the parsed port into
18 * *port_out and returns NGX_OK. On any malformed address, missing port, host
19 * allocation failure, or out-of-range port it logs an emerg-level message
20 * (referencing raw, the original directive token) and returns NGX_ERROR
21 * without touching *port_out.
22 *
23 * WHY: Upstream directives commonly use "[::1]:1094" for IPv6 loopback, which
24 * needs bracket-based splitting rather than a naive colon split; the strrchr
25 * split on the last colon keeps the non-IPv6 path tolerant of colons in the
26 * host portion. Isolating the parse+validate logic keeps the directive setter
27 * flat and makes the accept/reject decision independently reviewable. Port
28 * range 1-65535 rejects zero, negative, and above-range values so nginx never
29 * starts with an upstream address that would fail every connection.
30 *
31 * HOW:
32 * 1. If the address begins with '[', locate the closing ']'; require it to be
33 * immediately followed by ':', else log "invalid address" and fail.
34 * 2. Duplicate the bytes between '[' and ']' as the host; parse the port from
35 * two characters past ']' (skipping "]:").
36 * 3. Otherwise locate the last ':'; if absent log "missing port" and fail.
37 * 4. Duplicate the bytes before that ':' as the host; parse the port from one
38 * character past it.
39 * 5. Reject a trailing non-digit (endp not at the terminator) or a port
40 * outside 1-65535 with an "invalid port" message.
41 * 6. Store the port in *port_out and return NGX_OK.
42 */
43static ngx_int_t
44brix_upstream_parse_host_port(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *xcf,
45 char *addr_copy, ngx_str_t *raw, long *port_out)
46{
47 char *endp;
48 long pnum;
49
50 if (addr_copy[0] == '[') {
51 /* IPv6 literal [addr]:port */
52 char *rb = strchr(addr_copy, ']');
53 if (rb == NULL || *(rb + 1) != ':') {
54 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
55 "brix_upstream: invalid address \"%V\"", raw);
56 return NGX_ERROR;
57 }
58 size_t hostlen = (size_t)(rb - addr_copy - 1);
59 if (brix_pstrdupz(cf->pool, &xcf->upstream_host,
60 (u_char *) addr_copy + 1, hostlen) != NGX_OK) {
61 return NGX_ERROR;
62 }
63 pnum = strtol(rb + 2, &endp, 10);
64 } else {
65 /* hostname:port or IPv4:port - split on last colon */
66 char *colon = strrchr(addr_copy, ':');
67 if (colon == NULL) {
68 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
69 "brix_upstream: missing port in \"%V\"", raw);
70 return NGX_ERROR;
71 }
72 size_t hostlen = (size_t)(colon - addr_copy);
73 if (brix_pstrdupz(cf->pool, &xcf->upstream_host,
74 (u_char *) addr_copy, hostlen) != NGX_OK) {
75 return NGX_ERROR;
76 }
77 pnum = strtol(colon + 1, &endp, 10);
78 }
79
80 if (*endp != '\0' || pnum <= 0 || pnum > BRIX_MAX_PORT) {
81 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
82 "brix_upstream: invalid port in \"%V\"", raw);
83 return NGX_ERROR;
84 }
85
86 *port_out = pnum;
87 return NGX_OK;
88}

90/*
91 * brix_upstream_preresolve — register the redirector as a runtime DNS target.
92 *
93 * WHAT: Hands "host:port" to the phase-116 target registry and publishes the
94 * registry-owned ngx_addr_t as xcf->upstream_addr. An IP literal is final
95 * at once; a hostname stays socklen == 0 until the worker resolves it.
96 * WHY: A redirector whose name does not resolve at `nginx -t` must not stop
97 * the server (I-DNS-1), and the answer must follow the visible resolv.conf
98 * for the process lifetime — including after the redirector moves.
99 * HOW: brix_dns_target_register(); start.c reads the target round-robin via
100 * brix_dns_target_next() and never touches libc DNS on the event loop.
101 */
102static ngx_int_t
103brix_upstream_preresolve(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *xcf)
104{
105 ngx_str_t host;
106
107 host.data = xcf->upstream_host.data;
108 host.len = ngx_strlen(xcf->upstream_host.data);
109 xcf->upstream_dns = brix_dns_target_register(cf, "brix_upstream", &host,
110 (in_port_t) xcf->upstream_port,
111 BRIX_AF_AUTO, SOCK_STREAM,
112 &xcf->common.dns);
113 if (xcf->upstream_dns == NULL) {
114 return NGX_ERROR;
115 }
116 xcf->upstream_addr = &xcf->upstream_dns->addr;
117 return NGX_OK;
118}

120/* ---- brix_upstream directive setter ----
121 *
122 * WHAT: nginx config-directive handler for the upstream redirector address. Copies
123 * the directive argument, splits and validates it into xcf->upstream_host and
124 * xcf->upstream_port, pre-resolves the host into xcf->upstream_addr, logs a
125 * notice confirming the configured target, and returns NGX_CONF_OK. Returns
126 * NGX_CONF_ERROR (with an emerg-level message already emitted) if the argument
127 * cannot be duplicated or fails host/port parsing.
128 *
129 * WHY: This is the single entry point nginx invokes when it encounters the
130 * directive. Configuration parsing runs once at startup with no concurrent
131 * access, so all allocations come from cf->pool and are freed with the config
132 * lifecycle. Delegating parse/validate and pre-resolution to helpers keeps this
133 * orchestrator a flat, reviewable sequence.
134 *
135 * HOW:
136 * 1. Duplicate the directive value (value[1]) into a null-terminated copy;
137 * fail on allocation error.
138 * 2. Parse and validate host + port via brix_upstream_parse_host_port(); fail
139 * if it rejects the address.
140 * 3. Store the validated port, then pre-resolve the host address.
141 * 4. Log the configured target at notice level and return NGX_CONF_OK.
142 */
143char *
144brix_conf_set_upstream(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
145{
146 ngx_stream_brix_srv_conf_t *xcf = conf;
147 ngx_str_t *value;
148 char *addr_copy;
149 long pnum = 0;
150
151 value = cf->args->elts;
152 (void) cmd;
153
154 addr_copy = brix_pstrdup_z(cf->pool, &value[1]);
155 if (addr_copy == NULL) {
156 return NGX_CONF_ERROR;
157 }
158
159 if (brix_upstream_parse_host_port(cf, xcf, addr_copy, &value[1], &pnum)
160 != NGX_OK)
161 {
162 return NGX_CONF_ERROR;
163 }
164 xcf->upstream_port = (uint16_t) pnum;
165
166 /* phase-116: register the redirector as a runtime DNS target; request
167 * handlers never call getaddrinfo() on the event-loop thread. */
168 if (brix_upstream_preresolve(cf, xcf) != NGX_OK) {
169 return NGX_CONF_ERROR;
170 }
171
172 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
173 "brix: upstream redirector: %s:%d",
174 (char *) xcf->upstream_host.data, (int) xcf->upstream_port);
175
176 return NGX_CONF_OK;
177}
