30static int
31dir_checksum_matches(xrd_conform_execution *execution,
32 brix_conn *connection, const brix_dirent *entries, size_t count,
33 const char *name, brix_status *status)
34{
35 xrd_conform_scratch_object object;
36 const brix_dirent *entry = find_entry(entries, count, name);
37 char leaf[64];
38 char expected[129];
39 char token[144];
40 int length;
41
42 length = snprintf(leaf, sizeof(leaf), "brix-conform-dir/%s", name);
43 if (length < 0 || (size_t) length >= sizeof(leaf)
44 || brix_conform_scratch_path(execution->run->config.scratch, leaf,
45 object.path, sizeof(object.path)) != 0

51 return length >= 0 && (size_t) length < sizeof(token)

201/* WHAT: Force a plain listing beyond one 64 KiB response frame.
202 * WHY: A successful small listing does not exercise continuation framing.
203 * HOW: Create 276 scratch-confined long-name directories, observe the client's
204 * kXR_oksofar count, validate the exact set, then remove every fixture. */
205void
206xrd_conform_probe_large_dir(xrd_conform_execution *execution,
207 brix_conn *connection)
208{
209 xrd_conform_scratch_object base;
210 brix_dirent *entries = NULL;
211 brix_status status;
212 size_t continuations = 0;
213 size_t created = 0;
214 size_t count = 0;
215 int exact;
216 int listed = 0;
217
218 if (!dir_row_active(execution, "DIR-007")) return;
219 if (execution->requests > execution->run->config.request_limit
220 || execution->run->config.request_limit - execution->requests
221 < LARGE_DIR_ENTRIES * 2U + 3U) {
222 xrd_conform_result_set(execution, "DIR-007",
223 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
224 "ceiling:request-limit-needs-555", NULL, NULL);
225 return;
226 }
227 if (brix_conform_scratch_path(execution->run->config.scratch,
228 "brix-conform-dir-large", base.path, sizeof(base.path)) != 0)
229 return;
230 brix_status_clear(&status);
231 if (!large_dir_reserve(execution, &status)
232 || brix_mkdir(connection, base.path, 0755, 1, &status) != 0) {
233 large_dir_result(execution, 0, "large-directory-create-failed",
234 &status);
235 return;
236 }
237 created = create_large_dir_children(execution, connection, &status);
238 if (created == LARGE_DIR_ENTRIES
239 && large_dir_reserve(execution, &status)
240 && brix_dirlist_observe(connection, base.path, 0, &entries, &count,
241 &continuations, &status) == 0) listed = 1;
242 exact = listed && large_listing_exact(entries, count);
243 free(entries);
244 cleanup_large_dir(execution, connection, &base, created, &status);
245 large_dir_result(execution,
246 listed && continuations > 0 && exact,
247 listed && continuations == 0 ? "large-listing-was-not-continued"
248 : "continued-listing-name-set-mismatch", &status);
249}
