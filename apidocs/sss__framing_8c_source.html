17/*
18 * WHAT: validate the fixed magic ("sss\0" + BF32 encoding marker), the
19 * key-name-size field, and the header-length/NUL-termination against the
20 * received datagram length.
21 * WHY: bundling the several untrusted-framing bounds checks into one pure test
22 * keeps the SSS parse helper flat and the deny condition auditable, and
23 * makes the bound machine-checkable under ASan+UBSan via a fuzz harness.
24 * HOW: pure — reads the payload and the datagram length, writes the computed
25 * header length via *hdr_len; returns 1 when every check passes, else 0.
26 */
27int
28brix_sss_header_framing_ok(const unsigned char *payload, size_t dlen,
29 size_t *hdr_len)
30{
31 unsigned char kn_size;
32
33 if (payload == NULL
34 || dlen < BRIX_SSS_HDR_LEN + BRIX_SSS_DATA_HDR_LEN + 4)
35 {
36 return 0;
37 }
38
39 if (payload[0] != 's' || payload[1] != 's' || payload[2] != 's'
40 || payload[3] != '\0' || payload[7] != BRIX_SSS_ENC_BF32)
41 {
42 return 0;
43 }
44
45 kn_size = payload[6];
46 if (kn_size != 0 && (kn_size > BRIX_SSS_NAME_MAX || (kn_size & 0x07))) {
47 return 0;
48 }
49
50 *hdr_len = BRIX_SSS_HDR_LEN + kn_size;
51 if (*hdr_len >= dlen || (kn_size && payload[*hdr_len - 1] != '\0')) {
52 return 0;
53 }
54
55 return 1;
56}
