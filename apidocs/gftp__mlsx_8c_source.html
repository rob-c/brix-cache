9static int
10ci_eq(const char *a, size_t alen, const char *b)
11{
12 size_t i;
13 for (i = 0; i < alen; i++) {
14 char c = a[i];
15 if (c >= 'A' && c <= 'Z') { c = (char) (c - 'A' + 'a'); }
16 if (b[i] == '\0' || c != b[i]) {
17 return 0;
18 }
19 }
20 return b[alen] == '\0';
21}

24/* Parse an unsigned decimal in [p,end); *ok=0 on non-digit/empty/overflow. */
25static unsigned long long
26parse_u64(const char *p, const char *end, int *ok)
27{
28 unsigned long long v = 0;
29 int n = 0;
30 *ok = 0;
31 for (; p < end; p++) {
32 if (*p < '0' || *p > '9') {
33 return 0;
34 }
35 if (v > (~0ULL - 9) / 10) { /* would overflow */
36 return 0;
37 }
38 v = v * 10 + (unsigned) (*p - '0');
39 n++;
40 }
41 *ok = (n > 0);
42 return v;
43}

46/* Days since the Unix epoch for a civil (y,m,d), m in 1..12. Hinnant's
47 * algorithm — timezone-free, so MLSx `modify=` (always UTC) needs no libc. */
48static long long
49days_from_civil(long long y, unsigned m, unsigned d)
50{
51 y -= (m <= 2);
52 {
53 long long era = (y >= 0 ? y : y - 399) / 400;
54 unsigned yoe = (unsigned) (y - era * 400);
55 unsigned doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
56 unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
57 return era * 146097 + (long long) doe - 719468;
58 }
59}

62/* Decode a modify= fact "YYYYMMDDHHMMSS" (optionally ".sss") to a UTC epoch.
63 * Returns 0 and sets *out on success, -1 on a malformed timestamp. */
64static int
65parse_modify(const char *p, size_t len, long long *out)
66{
67 int v[6]; /* year, month, day, hour, min, sec */
68 int i, j;
69 if (len < 14) {
70 return -1;
71 }
72 for (i = 0, j = 0; i < 6; i++) {
73 int width = (i == 0) ? 4 : 2;
74 int n = 0, k;
75 for (k = 0; k < width; k++, j++) {
76 if (p[j] < '0' || p[j] > '9') {
77 return -1;
78 }
79 n = n * 10 + (p[j] - '0');
80 }
81 v[i] = n;
82 }
83 if (v[1] < 1 || v[1] > 12 || v[2] < 1 || v[2] > 31
84 || v[3] > 23 || v[4] > 59 || v[5] > 60) /* 60: leap second */
85 {
86 return -1;
87 }
88 *out = days_from_civil(v[0], (unsigned) v[1], (unsigned) v[2]) * 86400LL
89 + (long long) v[3] * 3600 + (long long) v[4] * 60 + v[5];
90 return 0;
91}

94static void
95apply_fact(const char *key, size_t klen, const char *val, size_t vlen,
96 gftp_mlsx_ent_t *out)
97{
98 if (ci_eq(key, klen, "type")) {
99 out->is_dir = ci_eq(val, vlen, "dir")
100 || ci_eq(val, vlen, "cdir")
101 || ci_eq(val, vlen, "pdir");
102 return;
103 }
104 if (ci_eq(key, klen, "size")) {
105 int ok;
106 unsigned long long s = parse_u64(val, val + vlen, &ok);
107 if (ok) { out->size = s; out->has_size = 1; }
108 return;
109 }
110 if (ci_eq(key, klen, "modify")) {
111 long long t;
112 const char *dot = val;
113 size_t base = vlen;
114 size_t z;
115 for (z = 0; z < vlen; z++) { /* drop a fractional part */
116 if (dot[z] == '.') { base = z; break; }
117 }
118 if (parse_modify(val, base, &t) == 0) {
119 out->mtime = t; out->has_mtime = 1;
120 }
121 return;
122 }
123 /* Unknown facts (perm, unix.mode, unique, …) are ignored by design. */
124}

127/* A listed name is rejected outright when it carries a separator or a control
128 * byte: '/' would let a hostile origin walk the caller out of the listed
129 * directory, and NUL/CR/LF would truncate or split the rendered entry. */
130static int
131gftp_mlsx_name_ok(const char *name, size_t len)
132{
133 size_t i;
134
135 for (i = 0; i < len; i++) {
136 char c = name[i];
137
138 if (c == '\0' || c == '\r' || c == '\n' || c == '/') {
139 return 0;
140 }
141 }
142 return 1;
143}

146/* Walk the `;`-separated key=value facts in [0,sep) and fold each recognised
147 * one into *out. A stray token with no '=' is tolerated (skipped) — only a
148 * missing name is fatal, and that is the caller's check. */
149static void
150gftp_mlsx_apply_facts(const char *line, size_t sep, gftp_mlsx_ent_t *out)
151{
152 size_t i, fstart = 0;
153
154 for (i = 0; i <= sep; i++) {
155 size_t e;
156
157 if (i != sep && line[i] != ';') {
158 continue;
159 }
160 for (e = fstart; e < i; e++) {
161 if (line[e] == '=') { break; }
162 }
163 if (e < i && e > fstart) {
164 apply_fact(line + fstart, e - fstart,
165 line + e + 1, i - (e + 1), out);
166 }
167 fstart = i + 1;
168 }
169}

172int
173gftp_mlsx_parse(const char *line, size_t len, gftp_mlsx_ent_t *out)
174{
175 size_t sep, i;
176
177 out->name = NULL; out->name_len = 0; out->is_dir = 0;
178 out->has_size = 0; out->size = 0; out->has_mtime = 0; out->mtime = 0;
179
180 /* Facts and the pathname are separated by exactly one space; a fact value
181 * never contains a space, so the first space is the boundary. */
182 sep = len;
183 for (i = 0; i < len; i++) {
184 if (line[i] == ' ') { sep = i; break; }
185 }
186 if (sep == len || sep + 1 >= len) {
187 return -1; /* no name present */
188 }
189
190 out->name = line + sep + 1;
191 out->name_len = len - (sep + 1);
192 if (!gftp_mlsx_name_ok(out->name, out->name_len)) {
193 return -1; /* hostile / traversal name */
194 }
195
196 gftp_mlsx_apply_facts(line, sep, out);
197 return 0;
198}
