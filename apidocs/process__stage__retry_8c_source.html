1/*
2 * process_stage_retry.c — 2.0 F1 (ADR-3b): apply the process-wide stage
3 * engine settings the brix_frm_* directives publish, and arm the worker-0
4 * brix_frm_fail_backoff retry sweep.
5 *
6 * WHAT: brix_stage_engine_conf_apply() turns the merge-time snapshot
7 * (brix_frm_engine_conf) into the engine's runtime state: it creates the
8 * brix_frm_queue_path journal directory, installs brix_frm_copymax /
9 * brix_frm_fail_retries as the engine limits and returns the journal dir
10 * to init the engine with, and opens the brix_frm_stagemsg StageEvents
11 * feed (2.0 F2, stage_events.c). brix_init_stage_retry_timer() arms a worker-0
12 * timer that re-drives FAILED journal records every
13 * brix_frm_fail_backoff (a record that keeps failing is dead-lettered at
14 * brix_frm_fail_retries attempts).
15 *
16 * WHY: Until 2.0 those knobs were parsed and merged but never read; the
17 * engine was driven by $BRIX_STAGE_JOURNAL_DIR and compile-time
18 * constants, and a FAILED record was only re-driven by a restart. The
19 * directives now own the engine (ADR-3b in phase-89 §D.1); the env var
20 * stays as the fallback when no `brix_frm on` server publishes.
21 *
22 * HOW: 1. No published snapshot -> NULL (caller falls back to the env var);
23 * the retry timer stays unarmed so env-only journals keep their
24 * restart-only re-drive semantics.
25 * 2. mkdir(queue_path, 0700), EEXIST ok; ERR + NULL (in-memory engine)
26 * on any other failure — a broken tape config must not take the
27 * worker down.
28 * 3. Timer period = max(fail_backoff, BRIX_FRM_FAIL_BACKOFF_MIN_MS),
29 * first tick after brix_cache_reap_delay(); each tick sweeps records
30 * older than the period and re-arms unless exiting.
31 */

47const char *
48brix_stage_engine_conf_apply(ngx_cycle_t *cycle)
49{
50 const brix_frm_engine_conf_t *fe = brix_frm_engine_conf(cycle);
51
52 if (fe == NULL) {
53 return NULL;
54 }
55 brix_stage_engine_set_limits(fe->copymax, fe->fail_retries);
56 if (fe->stagemsg[0] != '\0'
57 && brix_stage_events_open(fe->stagemsg, cycle->log) == NGX_OK)
58 {
59 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
60 "brix: stage engine: StageEvents feed \"%s\"",
61 fe->stagemsg);
62 }
63 if (fe->queue_path[0] == '\0') {
64 return NULL; /* unreachable: brix_frm on demands it */
65 }
66 if (mkdir(fe->queue_path, BRIX_CRED_STAGE_DIR_MODE) != 0 && errno != EEXIST) {
67 ngx_log_error(NGX_LOG_ERR, cycle->log, errno,
68 "brix: brix_frm_queue_path \"%s\": mkdir failed; the stage "
69 "journal is in-memory only (no restart recovery)", fe->queue_path);
70 return NULL;
71 }
72 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
73 "brix: stage engine: journal=\"%s\" copymax=%ui fail_retries=%ui "
74 "fail_backoff=%M ms copy_timeout=%M ms",
75 fe->queue_path, fe->copymax, fe->fail_retries, fe->fail_backoff_ms,
76 fe->copy_timeout_ms);
77 return fe->queue_path;
78}

91ngx_int_t
92brix_init_stage_retry_timer(ngx_cycle_t *cycle)
93{
94 const brix_frm_engine_conf_t *fe = brix_frm_engine_conf(cycle);
95 brix_stage_retry_timer_t *t;
96
97 if (fe == NULL || ngx_worker != 0
98 || brix_stage_engine_journal_dir()[0] == '\0')
99 {
100 return NGX_OK;
101 }
102 t = ngx_pcalloc(cycle->pool, sizeof(*t));
103 if (t == NULL) {
104 return NGX_ERROR;
105 }
106 t->period_ms = fe->fail_backoff_ms;
107 if (t->period_ms < BRIX_FRM_FAIL_BACKOFF_MIN_MS) {
108 t->period_ms = BRIX_FRM_FAIL_BACKOFF_MIN_MS;
109 }
110 t->ev.handler = brix_stage_retry_tick;
111 t->ev.data = t;
112 t->ev.log = cycle->log;
113 t->ev.cancelable = 1; /* don't delay graceful shutdown */
114 ngx_add_timer(&t->ev, brix_cache_reap_delay(t->period_ms));
115
116 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
117 "brix: stage retry sweep armed (brix_frm_fail_backoff=%M ms, "
118 "brix_frm_fail_retries=%ui)", t->period_ms, fe->fail_retries);
119 return NGX_OK;
120}
