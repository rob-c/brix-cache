54/* 1 iff `dst_path` names the same object as `src_path`, so a COPY/MOVE onto it
55 * must be refused (RFC 4918 §9.8.5 / §9.9.4 -> 403). Call it only when the
56 * destination exists; both stats come from webdav_{copy,move}_probe.
57 *
58 * Path equality is the arm that always holds. (dev,ino) equality additionally
59 * catches a hardlink reached under a second name — but only where the backend
60 * HAS inodes, and a REMOTE namespace has none: gsiftp, http, s3 and xroot never
61 * fill brix_vfs_stat_t.ino/.dev, so every path on such an export stats as
62 * (0,0). An inode-only guard therefore read "source and destination are the
63 * same file" for EVERY destination that already existed, and answered 403 to
64 * every ordinary overwrite on every remote-backed export. The inode arm is
65 * consulted only when the backend supplied an identity to compare. */
66static ngx_inline int
67brix_webdav_same_object(const char *src_path, const char *dst_path,
68 const struct stat *src_sb, const struct stat *dst_sb)
69{
70 if (ngx_strcmp(src_path, dst_path) == 0) {
71 return 1;
72 }
73 if (src_sb->st_ino == 0 && src_sb->st_dev == 0) {
74 return 0; /* no backend identity to compare */
75 }
76 return src_sb->st_ino == dst_sb->st_ino && src_sb->st_dev == dst_sb->st_dev;
77}
