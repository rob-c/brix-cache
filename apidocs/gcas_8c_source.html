1/*
2 * gcas.c — phase-87 G13: cross-repo hardlink dedup of verified CAS objects.
3 *
4 * WHAT: after a cvmfs-cas-verified fill commits, bind the per-repo cache
5 * object and a canonical repo-agnostic name ("/.gcas/<2hex>/<hex><sfx>")
6 * to one inode; after an eviction, unlink the canonical once it is the
7 * last remaining name.
8 * WHY: a stratum serving N repos stores byte-identical CAS objects once per
9 * repo. The verify step proves key-hash == content-hash, so identity at
10 * the byte layer is safe to collapse — while keys, cinfo, authz and
11 * origin fetches stay strictly per-repo (no cross-repo leak surface).
12 * HOW: first appearance registers the canonical via link(2); later fills of
13 * the same hash adopt it via link-to-temp + rename(2) (atomic, open
14 * readers keep their inode). The filesystem's st_nlink is the combined
15 * refcount, so there is no bookkeeping to corrupt: losing the canonical
16 * early (e.g. LRU-reaped) only forfeits future dedup. Every path is
17 * best-effort — any failure leaves plain, correct per-repo copies.
18 */

51 }

83/* Adopt an EXISTING canonical: swap the per-repo name onto its inode via a
84 * temp link + rename (atomic, open readers keep their inode). Returns 0 when
85 * publish is finished (bound, skipped or failed terminally), -1 when the
86 * canonical was evicted underneath us and the caller should retry. */
87static int
88gcas_adopt(brix_cstore_t *cs, const char *key, const char *canon,
89 const char *obj, const struct stat *cst, const struct stat *ost)
90{
91 char tmp[PATH_MAX];
92
93 if (cst->st_ino == ost->st_ino) {
94 return 0; /* already bound */
95 }
96
97 if (cst->st_size != ost->st_size) {
98 /* Hash collision or a damaged canonical: never collapse
99 * mismatched bytes — keep the verified per-repo copy. */
100 ngx_log_error(NGX_LOG_WARN, cs->log, 0,
101 "gcas: canonical size mismatch for \"%s\" "
102 "(%O vs %O) — dedup skipped", key,
103 (off_t) cst->st_size, (off_t) ost->st_size);
104 return 0;
105 }
106
107 if (brix_cache_append_suffix(tmp, sizeof(tmp), obj, ".gclnk") != 0) {
108 return 0;
109 }
110 unlink(tmp); /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
111 if (link(canon, tmp) != 0) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
112 return -1; /* canonical evicted underneath us */
113 }
114 if (rename(tmp, obj) != 0) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
115 unlink(tmp); /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
116 return 0;
117 }
118
119 ngx_log_error(NGX_LOG_INFO, cs->log, 0,
120 "gcas: dedup \"%s\" onto canonical inode", key);
121 return 0;
122}

124void
125brix_gcas_publish(brix_cstore_t *cs, const char *key)
126{
127 int attempt;
128 char rel[PATH_MAX], canon[PATH_MAX], obj[PATH_MAX];
129 struct stat ost, cst;
130
131 if (cs == NULL || !cs->gcas
132 || brix_gcas_canonical_rel(key, rel, sizeof(rel)) != 0
133 || gcas_abs(cs, rel, canon, sizeof(canon)) != 0
134 || gcas_abs(cs, key, obj, sizeof(obj)) != 0)
135 {
136 return;
137 }
138
139 if (stat(obj, &ost) != 0) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
140 return; /* committed object already gone — nothing to bind */
141 }
142
143 /* Two attempts cover the register/adopt race: attempt 1 may lose a
144 * concurrent link(2) of the canonical (EEXIST) and then adopts it. */
145 for (attempt = 0; attempt < 2; attempt++) {
146
147 if (stat(canon, &cst) == 0) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
148 if (gcas_adopt(cs, key, canon, obj, &cst, &ost) == 0) {
149 return;
150 }
151 continue; /* canonical evicted underneath us — retry */
152 }
153
154 /* Canonical absent — register this verified copy as the canonical. */
155 if (brix_cache_ensure_parent(canon) != 0) {
156 return;
157 }
158 if (link(obj, canon) == 0) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
159 ngx_log_error(NGX_LOG_INFO, cs->log, 0,
160 "gcas: registered canonical for \"%s\"", key);
161 return;
162 }
163 /* EEXIST: lost the race — loop once and adopt the winner. */
164 }
165}

169{

172

175 || gcas_abs(cs, rel, canon, sizeof(canon)) != 0)
176 {
177 return; /* a /.gcas/... key itself classifies REJECT — no-op */
178 }
179
180 if (stat(canon, &cst) != 0 || cst.st_nlink > 1) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
181 return; /* other per-repo links still hold the inode */
182 }
183
184 if (unlink(canon) == 0) { /* vfs-seam-allow: cache-store gcas hardlink farm, svc-owned domain */
185 ngx_log_error(NGX_LOG_INFO, cs->log, 0,
