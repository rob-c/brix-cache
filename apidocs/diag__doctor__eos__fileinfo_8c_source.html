1/*
2 * diag_doctor_eos_fileinfo.c — unprivileged EOS FST discovery via `fileinfo`.
3 *
4 * EOS gates the farm-enumeration commands (`fs ls`, `node ls`) behind an admin
5 * role: an identity the MGM maps to `nobody` gets NotAuthorized on /proc/admin/.
6 * But `fileinfo` is a *user*-plane command, and on any readable file it prints
7 * the FSTs holding that file's replicas. So when doctor_eos_map's admin `fs ls`
8 * is gated, we fall back here: walk a bounded sample of files under the target
9 * root, read each one's `fileinfo` replica table, and union the FSTs they name.
10 *
11 * Coverage is partial by construction — only FSTs that happen to hold a sampled
12 * file surface — so callers label these nodes "via fileinfo replica sampling",
13 * never as a complete inventory. The MGM answers `fileinfo` with a human table
14 * (it ignores the `-m` monitoring flag on the /proc route), so doctor_eos_parse_
15 * fileinfo tokenises that table; every parser here is pure over a caller buffer
16 * and unit-tested off recorded eospublic output. Only the walk touches the wire.
17 */
18#include "diag_internal.h"

20#include "brix_ops.h"

30/* Extract the path component of an endpoint URL into `out` (absolute, leading
31 * single slash; any opaque `?...` and duplicate leading slashes dropped). The
32 * XRootD `root://host//eos` double slash and the plain `root://host/eos` both
33 * yield "/eos". "/" when the URL carries no path. Returns 0 always (out filled). */
34int
35doctor_eos_url_path(const char *url, char *out, size_t osz)
36{
37 const char *p, *q;
38 size_t j = 0;
39
40 if (out == NULL || osz == 0) { return -1; }
41 out[0] = '\0';
42 if (url == NULL) { return 0; }
43
44 p = strstr(url, "://");
45 p = (p != NULL) ? p + 3 : url; /* past the scheme, at the authority */
46 q = strchr(p, '/'); /* first slash ends the authority */
47 if (q == NULL) { snprintf(out, osz, "/"); return 0; }
48
49 while (*q == '/') { q++; } /* collapse the // (or /) separator */
50 out[j++] = '/';
51 for (; *q != '\0' && *q != '?' && j < osz - 1; q++) {
52 out[j++] = *q;
53 }
54 /* trim a trailing slash (but never the root "/") */
55 while (j > 1 && out[j - 1] == '/') { j--; }
56 out[j] = '\0';
57 return 0;
58}

60/* Copy `src` into `dst` (max dsz-1) dropping ANSI CSI escape sequences
61 * (ESC '[' ... final-byte in 0x40..0x7e) — the EOS table colours the `active`
62 * column, and a stripped copy tokenises cleanly. NUL-terminates `dst`. */
63static void
64eos_strip_ansi(const char *src, char *dst, size_t dsz)
65{
66 size_t j = 0;
67
68 while (*src != '\0' && j < dsz - 1) {
69 if (src[0] == 0x1b && src[1] == '[') {
70 src += 2;
71 while (*src != '\0' && (*src < 0x40 || *src > 0x7e)) { src++; }
72 if (*src != '\0') { src++; } /* consume the final byte */
73 continue;
74 }
75 dst[j++] = *src++;
76 }
77 dst[j] = '\0';
78}

80/* 1 iff `s` is non-empty and every byte is an ASCII digit. */
81static int
82eos_all_digits(const char *s)
83{
84 if (s == NULL || *s == '\0') { return 0; }
85 for (; *s != '\0'; s++) {
86 if (*s < '0' || *s > '9') { return 0; }
87 }
88 return 1;
89}

91/* 1 iff `s` contains at least one ASCII letter (a host column, not a number). */
92static int
93eos_has_alpha(const char *s)
94{
95 for (; s != NULL && *s != '\0'; s++) {
96 if ((*s >= 'a' && *s <= 'z') || (*s >= 'A' && *s <= 'Z')) { return 1; }
97 }
98 return 0;
99}

101/* Split `rec` (mutated in place, NUL-terminating each token) on runs of spaces
102 * and tabs, recording up to `max` token start pointers into `tok`. Returns the
103 * token count. */
104static int
105eos_tokenise(char *rec, char **tok, int max)
106{
107 int n = 0;
108
109 while (*rec != '\0' && n < max) {
110 while (*rec == ' ' || *rec == '\t') { rec++; }
111 if (*rec == '\0') { break; }
112 tok[n++] = rec;
113 while (*rec != '\0' && *rec != ' ' && *rec != '\t') { rec++; }
114 if (*rec != '\0') { *rec++ = '\0'; }
115 }
116 return n;
117}

119/* Copy the next '\n'-delimited line of [s,s+len) starting at *i into `out`,
120 * advancing *i past the newline. Returns 1 if a line was read, 0 at end. */
121static int
122eos_next_line(const char *s, int len, int *i, char *out, size_t osz)
123{
124 int j = 0;
125
126 if (*i >= len) { return 0; }
127 while (*i < len && s[*i] != '\n') {
128 if (j < (int) osz - 1) { out[j++] = s[*i]; }
129 (*i)++;
130 }
131 out[j] = '\0';
132 if (*i < len) { (*i)++; } /* consume the newline */
133 return 1;
134}

136/* 1 iff the tokenised line is an `fs-id host ...` replica row: at least through
137 * the `active` column, first two tokens all-digits, host token alphabetic —
138 * which rejects the box-drawing header/rule lines and the trailing summary. */
139static int
140eos_is_replica_row(char *const *tok, int nt)
141{
142 return nt >= 9 && eos_all_digits(tok[0]) && eos_all_digits(tok[1])
143 && eos_has_alpha(tok[2]);
144}

146/* Fill one replica record from a validated token row (geotag = tok[9], optional).
147 * Precision `%.*s` copies keep the FORTIFY truncation analyzer quiet on the
148 * pointer-typed tokens. */
149static void
150eos_fill_rep(char *const *tok, int nt, doctor_eos_rep *r)
151{
152 char *colon;
153
154 memset(r, 0, sizeof(*r));
155 snprintf(r->host, sizeof(r->host), "%.*s", (int) sizeof(r->host) - 1, tok[2]);
156 colon = strrchr(r->host, ':');
157 if (colon != NULL) { *colon = '\0'; r->port = atoi(colon + 1); }
158 if (r->port == 0) { r->port = 1095; } /* fileinfo omits the port */
159 snprintf(r->cfgstatus, sizeof(r->cfgstatus), "%.*s",
160 (int) sizeof(r->cfgstatus) - 1, tok[6]);
161 r->booted = (strcmp(tok[5], "booted") == 0);
162 r->active = (strcmp(tok[8], "online") == 0);
163 if (nt >= 10) {
164 snprintf(r->geotag, sizeof(r->geotag), "%.*s",
165 (int) sizeof(r->geotag) - 1, tok[9]);
166 }
167}

169/* Parse an EOS `fileinfo` reply's stdout span, extracting one replica record per
170 * table row into out[0..cap). Pure: no allocation, no wire. Returns the row
171 * count written. */
172int
173doctor_eos_parse_fileinfo(const char *sout, int len, doctor_eos_rep *out, int cap)
174{
175 int i = 0, k = 0, nt;
176 char raw[1024], rec[1024];
177 char *tok[16];
178
179 if (sout == NULL || out == NULL) { return 0; }
180
181 while (k < cap && eos_next_line(sout, len, &i, raw, sizeof(raw))) {
182 eos_strip_ansi(raw, rec, sizeof(rec));
183 nt = eos_tokenise(rec, tok, 16);
184 if (!eos_is_replica_row(tok, nt)) { continue; }
185 eos_fill_rep(tok, nt, &out[k++]);
186 }
187 return k;
188}

193static int
194eos_fst_present(const doctor_ep *arr, int start, int n, const char *host)

196 int i;
197

207/* Materialise one discovered replica FST as a doctor_ep at arr[*n], mirroring the
208 * `fs ls` path's field mapping (CMS-plane data-server typing, rw/ro from config-
209 * status) but tagged `sampled`. fileinfo carries no filesystem capacity, so the
210 * cfg space fields stay unset (the "NN% free" line is simply omitted). */
211static void
212eos_add_fst(doctor_ep *e, const doctor_eos_rep *r)
213{
214 memset(e, 0, sizeof(*e));
215 e->proto = DXP_ROOT;
216 e->port = r->port;
217 snprintf(e->host, sizeof(e->host), "%.*s", (int) sizeof(e->host) - 1,
218 r->host);
219
220 e->eos.kind = DOC_EOS_FST;
221 e->eos.sampled = 1;
222 snprintf(e->eos.geotag, sizeof(e->eos.geotag), "%.*s",
223 (int) sizeof(e->eos.geotag) - 1, r->geotag);
224 snprintf(e->eos.cfgstatus, sizeof(e->eos.cfgstatus), "%.*s",
225 (int) sizeof(e->eos.cfgstatus) - 1, r->cfgstatus);
226 e->eos.booted = r->booted;
227 e->eos.active = r->active;
228
229 e->cms.reported = 1;
230 e->cms.role = DOC_CMS_SERVER;
231 e->cms.write = (strstr(r->cfgstatus, "rw") != NULL);
232}

234/* Read one file's `fileinfo` replica table and append the FSTs it names that we
235 * have not already recorded into arr[*n..cap). Returns the count newly added. */
236static int
237eos_sample_file(brix_conn *c, const char *path, doctor_ep *arr, int cap,
238 int start, int *n, brix_status *st)
239{
240 char opaque[XRDC_PATH_MAX + 32];
241 char *body = NULL;
242 const char *sout;
243 int slen, nrep, i, added = 0;
244 doctor_eos_rep reps[16];
245
246 snprintf(opaque, sizeof(opaque), "mgm.cmd=fileinfo&mgm.path=%s", path);
247 if (doctor_eos_proc(c, "/proc/user/", opaque, &body, st) != 0) { return 0; }
248 if (doctor_eos_stdout(body, &sout, &slen) != 0) { free(body); return 0; }
249
250 nrep = doctor_eos_parse_fileinfo(sout, slen, reps,
251 (int) (sizeof(reps) / sizeof(reps[0])));
252 for (i = 0; i < nrep && *n < cap; i++) {
253 if (eos_fst_present(arr, start, *n, reps[i].host)) { continue; }
254 eos_add_fst(&arr[*n], &reps[i]);
255 (*n)++;
256 added++;
257 }
258 free(body);
259 return added;
260}

262/* Build "<dir>/<name>" into `full` (length-checked with memcpy, so a path too
263 * long to hold is skipped rather than silently truncated into a bogus one) and
264 * reject the "."/".." / empty entries. Returns 0 on success, -1 to skip. */
265static int
266eos_child_path(const char *dir, int dl, const char *name, char *full, size_t fsz)
267{
268 size_t nl, base;
269
270 if (name[0] == '\0' || strcmp(name, ".") == 0 || strcmp(name, "..") == 0) {
271 return -1;
272 }
273 nl = strlen(name);
274 base = (dl > 0 && dir[dl - 1] == '/') ? (size_t) dl : (size_t) dl + 1;
275 if (base + nl + 1 > fsz) { return -1; }
276 memcpy(full, dir, (size_t) dl);
277 if (base != (size_t) dl) { full[dl] = '/'; }
278 memcpy(full + base, name, nl + 1);
279 return 0;
280}

282/* Descend the caller-scoped walk state through one listed directory: push each
283 * child directory (bounded by the stack) and sample each child file (bounded by
284 * *files), unioning FSTs into arr. Stops early at the FST budget or cap. */
285static void
286eos_walk_dir(brix_conn *c, const char *dir, brix_dirent *ents, size_t ne,
287 char (*stack)[XRDC_PATH_MAX], int *sp, doctor_ep *arr, int cap,
288 int start, int *n, int *files, brix_status *st)
289{
290 int dl = (int) strlen(dir);
291 size_t i;
292
293 for (i = 0; i < ne && *n < cap; i++) {
294 char full[XRDC_PATH_MAX];
295 int isdir = ents[i].have_stat && (ents[i].st.flags & kXR_isDir);
296
297 if (eos_child_path(dir, dl, ents[i].name, full, sizeof(full)) != 0) {
298 continue;
299 }
300 if (isdir) {
301 if (*sp < EOS_WALK_STACK) {
302 snprintf(stack[(*sp)++], XRDC_PATH_MAX, "%.*s",
303 XRDC_PATH_MAX - 1, full);
304 }
305 } else if (*files < EOS_WALK_FILES) {
306 eos_sample_file(c, full, arr, cap, start, n, st);
307 (*files)++;
308 }
309 if ((*n - start) >= EOS_WALK_FSTS) { break; }
310 }
311}

313/* Bounded DFS from `root`: list directories, sample files, union the FSTs their
314 * replica tables name into arr[start..cap). Stops at the file/dir/FST budgets.
315 * Returns the number of distinct FSTs appended (*n advanced by the same count). */
316int
317doctor_eos_discover_fileinfo(brix_conn *c, const char *root, doctor_ep *arr,
318 int cap, int start, int *n, brix_status *st)
319{
320 char (*stack)[XRDC_PATH_MAX];
321 int sp = 0, files = 0, dirs = 0, before;
322
323 if (arr == NULL || n == NULL) { return 0; }
324 before = *n;
325 stack = malloc(sizeof(*stack) * EOS_WALK_STACK);
326 if (stack == NULL) { return 0; }
327
328 if (root == NULL || root[0] == '\0') { root = "/"; }
329 snprintf(stack[sp++], XRDC_PATH_MAX, "%.*s", XRDC_PATH_MAX - 1, root);
330
331 while (sp > 0 && files < EOS_WALK_FILES && dirs < EOS_WALK_DIRS
332 && (*n - start) < EOS_WALK_FSTS && *n < cap) {
333 char dir[XRDC_PATH_MAX];
334 brix_dirent *ents = NULL;
335 size_t ne = 0;
336
337 snprintf(dir, sizeof(dir), "%.*s", XRDC_PATH_MAX - 1, stack[--sp]);
338 brix_status_clear(st);
339 if (brix_dirlist(c, dir, 1, &ents, &ne, st) != 0) { continue; }
340 dirs++;
341 eos_walk_dir(c, dir, ents, ne, stack, &sp, arr, cap, start, n, &files,
342 st);
343 free(ents);
344 }
345 free(stack);
346 return *n - before;
347}
