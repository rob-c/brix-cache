4/*
5 * vfs_policy.h — the VFS mutation-policy contract (phase-105).
6 *
7 * WHAT: Declares the typed endpoint mutation policy
8 * (brix_vfs_mutation_policy_t), the bounded operation vocabulary
9 * (brix_vfs_mutation_op_t) used for diagnostics and metrics, the policy
10 * kernel every VFS mutation entry point consults
11 * (brix_vfs_require_mutation_policy / _mutation / _confined_mutation),
12 * the config-derivation helper, and the policy-bearing operation context
13 * (brix_vfs_export_op_ctx_t) that carries the same authority into the
14 * ctx-less raw export helpers.
15 *
16 * WHY: Before phase-105 "may this request mutate the export?" was a bare
17 * allow_write bit tested in a handful of places, so a mutator that forgot
18 * the test reached a storage driver, and a protocol edge that forgot its
19 * gate was the only thing standing between a read-only export and a
20 * write. Making the authority a typed value that zero-initialises to
21 * READ_ONLY, and routing every mutator through one kernel, turns "the
22 * endpoint is read-only" into a VFS invariant instead of an edge habit.
23 *
24 * HOW: The policy is a two-valued enum whose zero is READ_ONLY, so a zeroed or
25 * hand-built object fails closed. The kernel is pure: it validates its
26 * inputs, returns NGX_ERROR with errno = EROFS under a read-only policy,
27 * and performs no allocation, I/O, backend lookup, or credential
28 * selection — which is what lets it run BEFORE leaf resolution, cache
29 * invalidation, and capability probing at every call site.
30 *
31 * Requires: <ngx_core.h> and observability/metrics/unified.h (brix_proto_t)
32 * before inclusion; included from vfs.h, never directly.
33 */

48/* Whether this request's endpoint may modify exported storage. READ_ONLY is
49 * zero on purpose: a zeroed or partially-built context is read-only, so an
50 * omitted policy can never read as "writable". */
51typedef enum {
52 BRIX_VFS_MUTATION_READ_ONLY = 0,
53 BRIX_VFS_MUTATION_ALLOWED = 1
54} brix_vfs_mutation_policy_t;

56/* The bounded mutation vocabulary. Values are append-only and exist for
57 * diagnostics and low-cardinality metric labels (INVARIANT #8) — never for
58 * exempting a backend, a protocol, or a path. It carries no protocol verb, no
59 * backend name, and no user data. */

61 BRIX_VFS_MUTATE_OPEN = 0, /* write/create/truncate/append open */
62 BRIX_VFS_MUTATE_WRITE, /* byte write, writev, pgwrite, staged write */
63 BRIX_VFS_MUTATE_TRUNCATE, /* handle ftruncate and path-native truncate */
64 BRIX_VFS_MUTATE_SYNC, /* flush/fsync of a writable handle */
65 BRIX_VFS_MUTATE_MKDIR, /* directory creation, incl. parent chains */
66 BRIX_VFS_MUTATE_REMOVE, /* unlink and rmdir */
67 BRIX_VFS_MUTATE_RENAME, /* rename/move, same- and cross-backend */
68 BRIX_VFS_MUTATE_COPY, /* copy, recursive copy, server-side copy */
69 BRIX_VFS_MUTATE_SETATTR, /* chmod/chown/utimes/setattr */
70 BRIX_VFS_MUTATE_XATTR, /* set/remove xattr, dead props, tags */
71 BRIX_VFS_MUTATE_PUBLISH, /* staged commit, multipart complete, promote */
72 BRIX_VFS_MUTATE_STAGE, /* recall from nearline into the online buffer */
73 BRIX_VFS_MUTATE_EVICT, /* drop an online copy (cache or nearline) */
74 BRIX_VFS_MUTATE_LOCK, /* acquire/refresh/release a resource lock */
75 BRIX_VFS_MUTATE_DEDUP, /* CAS alias publish and alias reap */
76 BRIX_VFS_MUTATE_CREDENTIAL, /* credential materialization (phase-108 C11) */
77 BRIX_VFS_MUTATE_OP_COUNT /* never a real operation */
78} brix_vfs_mutation_op_t;

96/* ---- the policy kernel ----------------------------------------------------
97 * All three return NGX_OK when the operation may proceed, else NGX_ERROR with
98 * errno set: EINVAL for a missing context or an out-of-range operation, EINVAL
99 * for an unconfined path on the confined form, and EROFS when the endpoint
100 * policy is read-only. EROFS is emitted BEFORE any capability, credential, or
101 * backend answer, so a read-only endpoint never discloses which of the later
102 * gates would also have refused (Appendix I.5). */
103
104/* Pure form: decide from a policy value alone. Used by objects that carry the
105 * policy by value (handles, staged sessions, writers, queued jobs) and by the
106 * unit tests, which need the decision without a context. */
107ngx_int_t brix_vfs_require_mutation_policy(brix_vfs_mutation_policy_t policy,
108 brix_vfs_mutation_op_t op);

110/* Context form: validate `ctx` and decide from ctx->mutation_policy. Does not
111 * require a resolved path — for handle- and fd-based mutations whose target is
112 * already open. */

116/* Path form: confinement first (EINVAL), then policy (EROFS). Every path-based
117 * mutator uses this one. */
118ngx_int_t brix_vfs_require_confined_mutation(const brix_vfs_ctx_t *ctx,

121/* Carried form: decide from a policy an object copied at construction, and
122 * attribute the denial to `proto`. This is what an open handle, a staged
123 * session, a writer or a queued job uses — it must never re-read a context it
124 * may outlive, and it must still be counted the same way a context rejection
125 * is. EROFS on refusal, exactly one metric sample. */
126ngx_int_t brix_vfs_require_carried_mutation(brix_vfs_mutation_policy_t policy,
127 brix_proto_t proto, brix_vfs_mutation_op_t op);

129/* Per-export lock-enforcement mode (phase-107 C7), carried by the
130 * `brix_lock_enforcement` directive and distributed through the backend
131 * registry (brix_vfs_backend_lock_enforcement). STRICT is zero on purpose: an
132 * unregistered export, a zeroed entry, and every default-configured export all
133 * fail toward enforcement. ADVISORY refuses only at the WebDAV edge (whose
134 * RFC-4918 checks predate this gate); the VFS gate warns, books the refusal
135 * metric, and allows. OFF restores the pre-C7 behaviour: the gate performs no
136 * reads at all. */
137typedef enum {
138 BRIX_VFS_LOCK_STRICT = 0,
139 BRIX_VFS_LOCK_ADVISORY = 1,
140 BRIX_VFS_LOCK_OFF = 2
141} brix_vfs_lock_enforcement_t;

143/* Lock gate (phase-107 C7) — NOT part of the pure kernel: it reads lock
144 * records (xattrs), so it is impure and lives in vfs_lock_gate.c. Called AFTER
145 * brix_vfs_require_confined_mutation() at every site, so EROFS continues to
146 * precede every other refusal. NGX_OK when no live foreign lock covers ctx's
147 * target or the ctx carries a matching token; NGX_ERROR with errno = EBUSY
148 * under a live foreign lock, EINVAL for a missing/unconfined ctx. Enforcement
149 * mode comes from the export's merged config, never from an argument — a call
150 * site cannot opt itself out. */
151ngx_int_t brix_vfs_require_unlocked(brix_vfs_ctx_t *ctx,
152 brix_vfs_mutation_op_t op);

154/* Alternate-target form for the two-name mutations: ask the same question
155 * about `path` (a confined resolved path in ctx's export — a rename or copy
156 * DESTINATION whose confinement the call site has already validated) instead
157 * of ctx's own resolved target. Same contract and errno mapping. */

161/* Batch form for the delete window: gate every resolved path in ONE sweep,

169/* ---- policy-bearing raw/export operation context --------------------------
170 * The thread-safe raw helpers (vfs_ops.h) take a log + root_canon rather than a
171 * ctx, because they run off the event loop where a request context does not
172 * exist. This bundle carries the same authority onto them WITHOUT adding a
173 * boolean to every signature, so an off-thread export mutation is as gated as
174 * an on-thread one. Fields are borrowed; the bundle is a caller-owned value. */
175typedef struct {
176 ngx_log_t *log;
177 const char *root_canon;
178 const brix_n2n_cfg_t *n2n;
179 brix_vfs_mutation_policy_t mutation_policy;
180 brix_proto_t proto;
181} brix_vfs_export_op_ctx_t;
