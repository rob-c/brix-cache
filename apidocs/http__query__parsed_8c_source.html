15static int
16brix_http_query_hex(u_char c)
17{
18 return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f')
19 || (c >= 'A' && c <= 'F');
20}

22static int
23brix_http_query_raw_valid(const u_char *data, size_t len)
24{
25 size_t i;
26
27 for (i = 0; i < len; i++) {
28 if (data[i] == '\0') {
29 return 0;
30 }
31 if (data[i] != '%') {
32 continue;
33 }
34 if (i + 2 >= len || !brix_http_query_hex(data[i + 1])
35 || !brix_http_query_hex(data[i + 2])
36 || (data[i + 1] == '0' && data[i + 2] == '0'))
37 {
38 return 0;
39 }
40 i += 2;
41 }
42 return 1;
43}

45static int
46brix_http_query_count(ngx_str_t args, ngx_uint_t *count)
47{
48 size_t start;
49 size_t i;
50 ngx_uint_t total;
51
52 if (args.len == 0) {
53 *count = 0;
54 return 1;
55 }
56 if (args.data == NULL || args.len > BRIX_HTTP_QUERY_MAX_BYTES) {
57 return -1;
58 }
59
60 start = 0;
61 total = 1;
62 for (i = 0; i < args.len; i++) {
63 if (args.data[i] != '&') {
64 continue;
65 }
66 if (i == start || total >= BRIX_HTTP_QUERY_MAX_PARAMS) {
67 return -1;
68 }
69 start = i + 1;
70 total++;
71 }
72 if (start == args.len) {
73 return -1;
74 }
75 *count = total;
76 return 1;
77}

79static int
80brix_http_query_decode(ngx_pool_t *pool, const u_char *raw, size_t raw_len,
81 int require_name, ngx_str_t *decoded_out)
82{
83 char *decoded;
84 size_t decoded_size;
85
86 if ((require_name && (raw_len == 0 || raw_len > BRIX_HTTP_QUERY_MAX_NAME))
87 || !brix_http_query_raw_valid(raw, raw_len))
88 {
89 return -1;
90 }
91 decoded_size = raw_len > 0 ? raw_len + 1 : 2;
92 decoded = ngx_pnalloc(pool, decoded_size);
93 if (decoded == NULL
94 || brix_http_urldecode(raw, raw_len, decoded, decoded_size,
95 BRIX_URLDECODE_PLUS_TO_SPACE |
96 BRIX_URLDECODE_REJECT_NUL) != BRIX_URLDECODE_OK
97 || (require_name && decoded[0] == '\0'))
98 {
99 return -1;
100 }
101 decoded_out->data = (u_char *) decoded;
102 decoded_out->len = ngx_strlen(decoded);
103 return 1;
104}

106static int
107brix_http_query_store(ngx_pool_t *pool, brix_http_query_t *query)
108{
109 u_char *cursor;
110 u_char *end;
111 ngx_uint_t index;
112
113 cursor = query->raw.data;
114 end = cursor + query->raw.len;
115 index = 0;
116 while (cursor < end) {
117 u_char *segment_end = ngx_strlchr(cursor, end, '&');
118 u_char *equals;
119
120 if (segment_end == NULL) {
121 segment_end = end;
122 }
123 equals = ngx_strlchr(cursor, segment_end, '=');
124 query->params[index].raw_name.data = cursor;
125 query->params[index].raw_name.len =
126 (size_t) ((equals ? equals : segment_end) - cursor);
127 if (brix_http_query_decode(pool, cursor,
128 (size_t) ((equals ? equals : segment_end) - cursor), 1,
129 &query->params[index].name) < 0
130 || brix_http_query_decode(pool,
131 equals ? equals + 1 : segment_end,
132 equals ? (size_t) (segment_end - equals - 1) : 0, 0,
133 &query->params[index].value) < 0)
134 {
135 return -1;
136 }
137 query->params[index].has_value = (equals != NULL);
138 index++;
139 cursor = segment_end < end ? segment_end + 1 : end;
140 }
141 return 1;
142}

144int
145brix_http_query_parse(ngx_pool_t *pool, ngx_str_t args,
146 brix_http_query_t *query)
147{
148 ngx_uint_t count;
149
150 if (pool == NULL || query == NULL) {
151 return -1;
152 }
153 if (query->parsed) {
154 return 1;
155 }
156 if (brix_http_query_count(args, &count) < 0) {
157 return -1;
158 }
159 if (args.len > 0) {
160 query->raw.data = ngx_pnalloc(pool, args.len + 1);
161 if (query->raw.data == NULL) {
162 return -1;
163 }
164 ngx_memcpy(query->raw.data, args.data, args.len);
165 query->raw.data[args.len] = '\0';
166 query->raw.len = args.len;
167 }
168 if (count > 0) {
169 query->params = ngx_pcalloc(pool, count * sizeof(*query->params));
170 if (query->params == NULL) {
171 return -1;
172 }
173 }
174 query->count = count;
175 if (count > 0 && brix_http_query_store(pool, query) < 0) {
176 return -1;
177 }
178 query->parsed = 1;
179 return 1;
180}

182static int
183brix_http_query_name_eq(const ngx_str_t *name, const char *key,
184 size_t key_len, unsigned flags)
185{
186 if (name->len != key_len) {
187 return 0;
188 }
189 if (flags & BRIX_HTTP_QUERY_CASE_INSENSITIVE) {
190 return ngx_strncasecmp(name->data, (u_char *) key, key_len) == 0;
191 }
192 return ngx_strncmp(name->data, key, key_len) == 0;
193}

195/* ---- Find one unambiguous decoded query name ----
196 * WHAT: Return zero with a nullable match, or BRIX_HTTP_QUERY_DUPLICATE.
197 * WHY: Value lookup and presence routing must reject the same ambiguity.
198 * HOW: 1. Scan the validated immutable query. 2. Reject a second matching name.
199 * Callers validate query/key and retain their distinct empty-value rules.
200 */
201static int
202brix_http_query_find_singleton(const brix_http_query_t *query,
203 const char *key, unsigned flags, const brix_http_query_param_t **match)
204{
205 size_t key_len = ngx_strlen(key);
206 ngx_uint_t i;
207
208 *match = NULL;
209 for (i = 0; i < query->count; i++) {
210 if (!brix_http_query_name_eq(&query->params[i].name, key,
211 key_len, flags))
212 {
213 continue;
214 }
215 if (*match != NULL) {
216 return BRIX_HTTP_QUERY_DUPLICATE;
217 }
218 *match = &query->params[i];
219 }
220 return 0;
221}

223int
224brix_http_query_parsed_value(const brix_http_query_t *query,
225 const char *key, ngx_str_t *out, unsigned flags)
226{
227 const brix_http_query_param_t *match;
228
229 if (query == NULL || !query->parsed || key == NULL || out == NULL) {
230 return -1;
231 }
232 out->data = NULL;
233 out->len = 0;
234 if (brix_http_query_find_singleton(query, key, flags, &match)
235 == BRIX_HTTP_QUERY_DUPLICATE)
236 {
237 return BRIX_HTTP_QUERY_DUPLICATE;
238 }
239 if (match == NULL || !match->has_value
240 || (match->value.len == 0 && !(flags & BRIX_HTTP_QUERY_ALLOW_EMPTY)))
241 {
242 return 0;
243 }
244 *out = match->value;
245 return 1;
246}

248int
249brix_http_query_parsed_get(const brix_http_query_t *query,
250 const char *key, char *out, size_t outsz, unsigned flags)
251{
252 ngx_str_t value;
253 size_t copy;
254 int rc;
255
256 if (out == NULL || outsz == 0) {
257 return -1;
258 }
259 rc = brix_http_query_parsed_value(query, key, &value, flags);
260 if (rc != 1) {
261 return rc;
262 }
263 if (value.len >= outsz && !(flags & BRIX_HTTP_QUERY_TRUNCATE)) {
264 return -1;
265 }
266 copy = value.len < outsz - 1 ? value.len : outsz - 1;
267 ngx_memcpy(out, value.data, copy);
268 out[copy] = '\0';
269 return 1;
270}

272int
273brix_http_query_parsed_has(const brix_http_query_t *query,
274 const char *key, unsigned flags)
275{
276 const brix_http_query_param_t *match;
277
278 if (query == NULL || !query->parsed || key == NULL) {
279 return -1;
280 }
281 if (brix_http_query_find_singleton(query, key, flags, &match)
282 == BRIX_HTTP_QUERY_DUPLICATE)
283 {
284 return BRIX_HTTP_QUERY_DUPLICATE;
285 }
286 if (match == NULL) {
287 return 0;
288 }
289 if (!match->has_value) {
290 return 1;
291 }
292 if (match->value.len == 0 && (flags & BRIX_HTTP_QUERY_HAS_EMPTY_OK)) {
293 return 1;
294 }
295 return !!(flags & BRIX_HTTP_QUERY_HAS_VALUE_OK);
296}

298int
299brix_http_query_unique(const brix_http_query_t *query, unsigned flags)
300{
301 ngx_uint_t i;
302 ngx_uint_t j;
303
304 if (query == NULL || !query->parsed) {
305 return -1;
306 }
307 for (i = 0; i < query->count; i++) {
308 for (j = i + 1; j < query->count; j++) {
309 if (brix_http_query_name_eq(&query->params[i].name,
310 (const char *) query->params[j].name.data,
311 query->params[j].name.len, flags))
312 {
313 return BRIX_HTTP_QUERY_DUPLICATE;
314 }
315 }
316 }
317 return 1;
