1/*
2 * ftp_login.c — GridFTP client login: RFC 2228 AUTH GSSAPI/ADAT, or anonymous.
3 *
4 * WHAT: negotiate the session credential immediately after the greeting, then
5 * settle the data-channel protection parameters the server expects.
6 * WHY: which credential a transfer runs under is a security decision, so it is
7 * made in one place with one rule: a gsiftp:// URL means GSI, and a GSI
8 * endpoint is never downgraded to anonymous because the proxy is missing or
9 * the handshake failed — the copy fails instead, loudly.
10 * HOW: AUTH GSSAPI (334) opens the token loop; each 335 reply carries the peer's
11 * ADAT token, which the initiator (ftp_gsi.c) consumes and answers until the
12 * server returns 235. From then on the control channel is protected and
13 * ftp_ctl.c wraps every command transparently.
14 */
15#include "ftp_client.h"

18#include <string.h>
19#include <unistd.h>

21/* Extract the base64 argument of an "…ADAT=<token>…" reply text. */
22static char *
23adat_token(const char *text)
24{
25 const char *p = strstr(text, "ADAT=");
26 const char *e;
27 char *tok;
28 size_t n;
29
30 if (p == NULL) {
31 return NULL;
32 }
33 p += 5;
34 for (e = p; *e != '\0' && *e != ' ' && *e != '\r' && *e != '\n'; e++) {
35 /* token runs to the first whitespace */
36 }
37 n = (size_t) (e - p);
38 if (n == 0) {
39 return NULL;
40 }
41 tok = malloc(n + 1);
42 if (tok == NULL) {
43 return NULL;
44 }
45 memcpy(tok, p, n);
46 tok[n] = '\0';
47 return tok;
48}

51/* Send one ADAT token (base64 of `tok`) and read the server's reply. */
52static int
53adat_send(brix_ftp_sess *s, const uint8_t *tok, size_t len, brix_status *st)
54{
55 char *b64 = brix_ftp_b64_encode(tok, len);
56 int rc;
57
58 if (b64 == NULL) {
59 brix_status_set(st, XRDC_EAUTH, 0, "gsiftp: cannot encode ADAT token");
60 return -1;
61 }
62 rc = brix_ftp_cmd(s, st, "ADAT %s", b64);
63 free(b64);
64 return rc;
65}

68/* Decode the peer's ADAT token from the current reply (absent ⇒ empty token). */
69static int
70adat_recv(brix_ftp_sess *s, uint8_t **in, size_t *in_len, brix_status *st)
71{
72 char *b64;
73
74 *in = NULL;
75 *in_len = 0;
76 b64 = adat_token(s->text);
77 if (b64 == NULL) {
78 return 0;
79 }
80 *in = brix_ftp_b64_decode(b64, in_len);
81 free(b64);
82 if (*in == NULL) {
83 brix_status_set(st, XRDC_EPROTO, 0,
84 "gsiftp: server ADAT token is not base64");
85 return -1;
86 }
87 return 0;
88}

91/*
92 * Run the ADAT loop until the server reports completion (235). Bounded: a peer
93 * that keeps answering 335 without converging must not spin the client forever.
94 */
95static int
96gsi_token_loop(brix_ftp_sess *s, brix_status *st)
97{
98 uint8_t *out = NULL;
99 size_t out_len = 0;
100 int rounds;
101 int more;
102
103 more = brix_ftp_gss_step(s->gss, NULL, 0, &out, &out_len, st);
104 if (more < 0) {
105 return -1;
106 }
107 for (rounds = 0; rounds < 16; rounds++) {
108 int rc;
109
110 if (out_len == 0) {
111 free(out);
112 brix_status_set(st, XRDC_EAUTH, 0,
113 "gsiftp: GSI handshake produced no token");
114 return -1;
115 }
116 rc = adat_send(s, out, out_len, st);
117 free(out);
118 out = NULL;
119 out_len = 0;
120 if (rc != 0) {
121 return -1;
122 }
123 if (s->code == 235) {
124 s->secure = 1;
125 return 0;
126 }
127 if (s->code != 335) {
128 brix_status_set(st, XRDC_EAUTH, 0, "gsiftp: %d %s", s->code,
129 s->text);
130 return -1;
131 }
132 {
133 uint8_t *in = NULL;
134 size_t in_len = 0;
135
136 if (adat_recv(s, &in, &in_len, st) != 0) {
137 return -1;
138 }
139 more = brix_ftp_gss_step(s->gss, in, in_len, &out, &out_len, st);
140 free(in);
141 }
142 if (more < 0) {
143 return -1;
144 }
145 }
146 brix_status_set(st, XRDC_EAUTH, 0,
147 "gsiftp: GSI handshake did not converge");
148 return -1;
149}

152/*
153 * The RFC 959 access sequence: USER, then PASS only if the server asked for one
154 * (3xx). Anything but a final 2xx is a refusal, and `what` names which refusal it
155 * was — an anonymous login that was not accepted, or a GSI identity the server
156 * authenticated but would not authorize.
157 */
158static int
159ftp_user_pass(brix_ftp_sess *s, const char *user, const char *pass,
160 const char *what, brix_status *st)
161{
162 if (brix_ftp_cmd(s, st, "USER %s", user) != 0) {
163 return -1;
164 }
165 if (s->code >= 300 && s->code < 400) {
166 if (brix_ftp_cmd(s, st, "PASS %s", pass) != 0) {
167 return -1;
168 }
169 }
170 if (s->code < 200 || s->code > 299) {
171 brix_status_set(st, XRDC_EAUTH, 0, "gsiftp: %s: %d %s", what, s->code,
172 s->text);
173 return -1;
174 }
175 return 0;
176}

197static int
198gsi_login(brix_ftp_sess *s, const brix_opts *co, brix_status *st)
199{
200 /* The proxy path comes from $X509_USER_PROXY (xrdcp's --proxy sets it), so
201 * one discovery rule serves every scheme. */
202 const char *ca_dir = (co != NULL) ? co->ca_dir : NULL;
203 int insecure = (co != NULL) ? co->insecure_tls : 0;
204
205 if (brix_ftp_cmd(s, st, "AUTH GSSAPI") != 0) {
206 return -1;
207 }
208 if (s->code != 334) {
209 brix_status_set(st, XRDC_EAUTH, 0,
210 "gsiftp: server refused GSI authentication: %d %s",
211 s->code, s->text);
212 return -1;
213 }
214 s->gss = brix_ftp_gss_create(NULL, ca_dir, insecure, st);
215 if (s->gss == NULL) {
216 return -1;
217 }
218 if (gsi_token_loop(s, st) != 0) {
219 return -1;
220 }
221 return gsi_user(s, st);
222}
