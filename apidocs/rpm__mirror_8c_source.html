1/*
2 * rpm_mirror.c — the content handler for a brix_rpm_mirror location.
3 *
4 * WHAT: the entry point for every request on an RPM/dnf pull-through mirror:
5 * ctx + observer → gate → offloaded serve → coalesced fill → open +
6 * ranged response.
7 * WHY: a package mirror is a cache with a repository layout on top of it.
8 * Everything below the layout — request coalescing so a hundred
9 * simultaneous `dnf install`s of the same package cause ONE upstream
10 * GET, digest verify at the edge, ranged/conditional serving, bounded
11 * stale-if-error — is machinery this tree already runs for CVMFS and
12 * for the OCI mirror. The job of this file is to compose those pieces
13 * in the order the shared plane pins, and to add nothing of its own
14 * that could drift from them.
15 * HOW: the same wiring the OCI mirror uses, verbatim:
16 * brix_http_serve_offload_remote → brix_http_cache_fill_if_needed →
17 * brix_vfs_open → brix_http_serve_file_ranged. Never
18 * brix_cache_open_or_fill (it fills inline on the event loop), and
19 * never a raw open/read/sendfile in this directory (INVARIANT #12).
20 */

40/* The serve options this plane publishes. Weak ETags because the validator is
41 * derived from mtime+size: a refill of byte-identical metadata is a new file
42 * on disk, and a strong ETag would be claiming byte-identity we did not check
43 * (for the digest-named routes the cache tier verified the bytes, but the
44 * validator a client compares is still the local file's). */
45static void
46rpm_serve_opts(brix_http_serve_opts_t *opts)
47{
48 ngx_memzero(opts, sizeof(*opts));
49 opts->xfer_proto = BRIX_XFER_PROTO_RPM;
50 opts->op_name = "GET";
51 opts->identity = "anonymous";
52 opts->etag_flags = BRIX_ETAG_WEAK;
53}

56/* Re-entry trampoline: the completed fill re-runs the whole handler, which
57 * now takes the hit path. The disposition stays FILL — the bytes came from a
58 * fresh upstream pull, and an operator reading $brix_cache_status wants to
59 * know. */
60static ngx_int_t
61rpm_reenter(ngx_http_request_t *r, void *data)
62{
63 (void) data;
64 return ngx_http_brix_rpm_handler(r);

68/* Fill-failure interceptor. The tier has already exhausted its own
69 * stale-serve window by the time it calls this, so there is nothing left to
70 * salvage: report the upstream's own verdict where it is meaningful to a
71 * package client (a 404 is a missing package, which dnf reports as such) and
72 * 502 for everything else — the failure is ours-to-upstream, not the
73 * client's. An upstream 401/403 becomes 403: relaying the challenge would
74 * make dnf prompt for credentials THIS mirror cannot use. */
75static ngx_int_t
76rpm_fill_fail(ngx_http_request_t *r, void *data, ngx_int_t status)
77{
78 ngx_http_brix_rpm_ctx_t *ctx =
79 ngx_http_get_module_ctx(r, ngx_http_brix_rpm_module);
80
81 (void) data;
82
83 if (ctx != NULL) {
84 ctx->disp = BRIX_RPM_OUT_ERROR;
85 }
86 if (status == NGX_HTTP_NOT_FOUND) {
87 return NGX_HTTP_NOT_FOUND;
88 }
89 if (status == NGX_HTTP_UNAUTHORIZED || status == NGX_HTTP_FORBIDDEN) {
90 return NGX_HTTP_FORBIDDEN;
91 }
92 return NGX_HTTP_BAD_GATEWAY;
93}

96/* The two off-loop steps: a socket-wire serve, then a coalesced miss fill.
97 * NGX_DECLINED means the bytes are local now and the caller should open. */
98static ngx_int_t
99rpm_serve_or_fill(ngx_http_request_t *r, ngx_http_brix_rpm_loc_conf_t *lcf,
100 ngx_http_brix_rpm_ctx_t *ctx, brix_sd_instance_t *sd, const char *path)
101{
102 brix_http_serve_opts_t sopts;
103 ngx_int_t rc;
104
105 rpm_serve_opts(&sopts);
106
107 rc = brix_http_serve_offload_remote(r, sd, ctx->key, path, &sopts,
108 &lcf->common, NULL, NULL);
109 if (rc == NGX_DONE) {
110 return NGX_DONE;
111 }
112 if (rc == NGX_ERROR) {
113 ctx->disp = BRIX_RPM_OUT_ERROR;
114 return NGX_HTTP_INTERNAL_SERVER_ERROR;
115 }
116
117 rc = brix_http_cache_fill_if_needed(r, sd, ctx->key, &lcf->common, NULL,
118 rpm_reenter, NULL, rpm_fill_fail);
119 if (rc == NGX_DONE) {
120 ctx->disp = BRIX_RPM_OUT_FILL;
121 return NGX_DONE;
122 }
123 if (rc == NGX_ERROR) {
124 ctx->disp = BRIX_RPM_OUT_ERROR;
125 return NGX_HTTP_INTERNAL_SERVER_ERROR;
126 }
127
128 return NGX_DECLINED;
129}

132/* Open the now-local object and answer. A directory-shaped hit is a 404 here:
133 * the classifier already refused directory-shaped REQUESTS, so this is the
134 * cache store answering with something that is not the file we asked for. */
135static ngx_int_t
136rpm_open_respond(ngx_http_request_t *r, ngx_http_brix_rpm_loc_conf_t *lcf,
137 ngx_http_brix_rpm_ctx_t *ctx, brix_vfs_ctx_t *vctx, const char *path)
138{
139 brix_http_serve_opts_t opts;
140 brix_http_serve_result_t result;
141 brix_vfs_file_t *fh;
142 brix_vfs_stat_t vst;
143 int vfs_err = 0;
144
145 fh = brix_vfs_open(vctx, BRIX_VFS_O_READ, &vfs_err);
146 if (fh == NULL) {
147 ctx->disp = BRIX_RPM_OUT_ERROR;
148 return (vfs_err == NGX_ENOENT) ? NGX_HTTP_NOT_FOUND
149 : (vfs_err == NGX_EACCES) ? NGX_HTTP_FORBIDDEN
150 : NGX_HTTP_BAD_GATEWAY;
151 }
152 if (brix_vfs_file_stat(fh, &vst) != NGX_OK) {
153 brix_vfs_close(fh, r->connection->log);
154 ctx->disp = BRIX_RPM_OUT_ERROR;
155 return NGX_HTTP_INTERNAL_SERVER_ERROR;
156 }
157 if (vst.is_directory) {
158 brix_vfs_close(fh, r->connection->log);
159 ctx->disp = BRIX_RPM_OUT_ERROR;
160 return NGX_HTTP_NOT_FOUND;
161 }
162
163 /* A freshly pulled index names the two files this client asks for next,
164 * so warm them while this response is still being written (D15.10).
165 * Advisory in both directions: nothing below reads its result, and it
166 * refuses itself on every route but a repomd FILL. */
167 brix_rpm_prefetch_repomd(r, lcf, ctx, fh, vst.size, vctx->sd);
168
169 /* A resumed package download is the one range a repository client asks
170 * for, and dnf's own delta/resume logic depends on it being honoured. */
171 r->allow_ranges = 1;
172
173 rpm_serve_opts(&opts);
174 return brix_http_serve_file_ranged(r, fh, &vst, path, &opts, &result);
175}

178static ngx_int_t
179rpm_tier_get(ngx_http_request_t *r, ngx_http_brix_rpm_loc_conf_t *lcf,
180 ngx_http_brix_rpm_ctx_t *ctx)
181{
182 char path[PATH_MAX];
183 const char *root = lcf->common.root_canon;
184 brix_vfs_ctx_t vctx;
185 int is_tls = 0;
186 ngx_int_t rc;
187
188 rc = brix_http_mirror_key_path(root, ctx->key, ctx->key_len,
189 path, sizeof(path));
190 if (rc != NGX_OK) {
191 ctx->disp = BRIX_RPM_OUT_REFUSED;
192 return rc;
193 }
194
195 is_tls = brix_http_request_is_tls(r);
196 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log, BRIX_PROTO_RPM,
197 root, "", BRIX_VFS_MUTATION_READ_ONLY, is_tls, NULL, path);
198 brix_http_vfs_bind_no_rules(&lcf->common, &vctx);
199
200 vctx.sd = brix_vfs_backend_resolve(root, r->connection->log);
201 if (vctx.sd == NULL) {
202 ngx_log_error(NGX_LOG_ERR, r->connection->log, 0,
203 "rpm: no storage backend registered for \"%s\" — check "
204 "brix_rpm_mirror / brix_cache_store", root);
205 ctx->disp = BRIX_RPM_OUT_ERROR;
206 return NGX_HTTP_INTERNAL_SERVER_ERROR;
207 }
208
209 (void) brix_dashboard_http_start_identity(r, ctx->key, "anonymous", "",
210 BRIX_XFER_PROTO_RPM, BRIX_XFER_DIR_READ, "GET", -1);
211
212 rc = rpm_serve_or_fill(r, lcf, ctx, vctx.sd, path);
213 if (rc != NGX_DECLINED) {
214 return rc;
215 }
216
217 return rpm_open_respond(r, lcf, ctx, &vctx, path);
218}

223void
224brix_rpm_finalize_observe(void *data)
225{
226 ngx_http_request_t *r = data;
227 ngx_http_brix_rpm_ctx_t *ctx =
228 ngx_http_get_module_ctx(r, ngx_http_brix_rpm_module);
229
230 if (ctx == NULL || ctx->counted) {
231 return;
232 }
233 ctx->counted = 1;
234
235 /* One request, one row. Both indices are enums (INVARIANT #8): no
236 * repository path, package name or digest is ever a metric label. */
237 BRIX_RPM_METRIC_INC(requests_total[ctx->req.cls][ctx->disp]);
238 /* phase-110 W10: also feed the unified cache family (brix_cache_requests_total)
239 * so an RPM cache hit-rate is one query across planes. HIT/LOCAL = local
240 * store; FILL = origin fill (miss); REFUSED/ERROR are not cache outcomes. */
241 if (ctx->disp == BRIX_RPM_OUT_HIT || ctx->disp == BRIX_RPM_OUT_LOCAL) {
242 brix_metric_cache_result(BRIX_PROTO_RPM, 1, 0);
243 } else if (ctx->disp == BRIX_RPM_OUT_FILL) {
244 brix_metric_cache_result(BRIX_PROTO_RPM, 0, 0);
245 }
246}

251static ngx_http_brix_rpm_ctx_t *
252rpm_handler_ctx(ngx_http_request_t *r)
253{
254 ngx_http_brix_rpm_ctx_t *ctx =
255 ngx_http_get_module_ctx(r, ngx_http_brix_rpm_module);
256 ngx_pool_cleanup_t *cln;
257
258 if (ctx != NULL) {
259 return ctx; /* re-entry after a parked fill */
260 }
261 ctx = ngx_pcalloc(r->pool, sizeof(*ctx));
262 if (ctx == NULL) {
263 return NULL;
264 }
265 ngx_http_set_ctx(r, ctx, ngx_http_brix_rpm_module);
266
267 cln = ngx_pool_cleanup_add(r->pool, 0);
268 if (cln == NULL) {
269 return NULL;
270 }
271 cln->handler = brix_rpm_finalize_observe;
272 cln->data = r;
273
274 return ctx;
275}

278ngx_int_t
279ngx_http_brix_rpm_handler(ngx_http_request_t *r)
280{
281 ngx_http_brix_rpm_loc_conf_t *lcf =
282 ngx_http_get_module_loc_conf(r, ngx_http_brix_rpm_module);
283 ngx_http_brix_rpm_ctx_t *ctx;
284 ngx_int_t rc;
285
286 ctx = rpm_handler_ctx(r);
287 if (ctx == NULL) {
288 return NGX_HTTP_INTERNAL_SERVER_ERROR;
289 }
290
291 /* GET/HEAD only, so no request body is ever meaningful; the gate still
292 * polices the method itself, because discarding a body must not make an
293 * upload look like it was accepted. */
294 rc = ngx_http_discard_request_body(r);
295 if (rc != NGX_OK) {
296 return rc;
297 }
298
299 rc = brix_rpm_gate(r, ctx);
300 if (rc != NGX_DECLINED) {
301 return rc;
302 }
303
304 return rpm_tier_get(r, lcf, ctx);
