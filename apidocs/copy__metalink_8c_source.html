1/*
2 * copy_metalink.c — metalink virtual-redirector orchestration (phase-100).
3 *
4 * WHAT: Detects a metalink source in brix_copy, loads the document (local read
5 * or a bounded remote fetch through the normal copy engine), parses it
6 * via metalink.c, then runs the transfer against the ranked mirrors with
7 * ordered failover — handing the root-family mirror list to the
8 * extreme-copy engine when --sources asks for it.
9 * WHY: Upstream XrdCl treats .meta4/.metalink files as virtual redirectors;
10 * this file is that behavior for the BriX client, kept separate from the
11 * pure parser (metalink.c) and from the per-direction transfer bodies it
12 * reuses (copy_dispatch_one).
13 * HOW: copy_metalink_run = fetch -> parse -> per-mirror dispatch loop. Every
14 * inner dispatch runs with metalink_off forced on, so a mirror that
15 * itself names a .meta4 can never recurse.
16 */
17#include "copy_internal.h"
18#include "metalink.h"

41/* ---- Read a local metalink file into a fresh buffer, bounded ----
42 *
43 * WHAT: malloc-read `path` (at most XRDC_METALINK_MAX_BYTES) into buf/blen.
44 * 0, or -1 with st set (open/oversize/short-read). Caller frees *buf.
45 *
46 * WHY: The local branch of the fetch; the size cap runs BEFORE the read so a
47 * huge file never lands in memory.
48 *
49 * HOW: open O_RDONLY+O_NOFOLLOW-less (a user-named local path may be a
50 * symlink; that is normal CLI behavior) -> fstat gate -> full read loop.
51 */
52static int
53metalink_read_local(const char *path, char **buf, size_t *blen, brix_status *st)
54{
55 struct stat sb;
56 int fd = open(path, O_RDONLY);
57 char *data;
58 size_t got = 0;
59
60 if (fd < 0) {
61 brix_status_set(st, XRDC_EUSAGE, errno, "open %s: %s",
62 path, strerror(errno));
63 return -1;
64 }
65 if (fstat(fd, &sb) != 0 || !S_ISREG(sb.st_mode)
66 || sb.st_size <= 0 || sb.st_size > (off_t) XRDC_METALINK_MAX_BYTES) {
67 brix_status_set(st, XRDC_EPROTO, 0,
68 "metalink %s: not a regular file within the %u-byte cap",
69 path, XRDC_METALINK_MAX_BYTES);
70 close(fd);
71 return -1;
72 }
73 data = (char *) malloc((size_t) sb.st_size + 1);
74 if (data == NULL) {
75 brix_status_set(st, XRDC_EPROTO, 0, "out of memory");
76 close(fd);
77 return -1;
78 }
79 while (got < (size_t) sb.st_size) {
80 ssize_t n = read(fd, data + got, (size_t) sb.st_size - got);
81 if (n <= 0) {
82 brix_status_set(st, XRDC_EIO, errno, "read %s: %s",
83 path, n < 0 ? strerror(errno) : "short read");
84 free(data);
85 close(fd);
86 return -1;
87 }
88 got += (size_t) n;
89 }
90 close(fd);
91 data[got] = '\0';
92 *buf = data;
93 *blen = got;
94 return 0;
95}

98/* ---- Fetch a REMOTE metalink through the normal copy engine ----
99 *
100 * WHAT: Pull `src` (root://, http(s)://, davs://, …) into a private mkstemp
101 * temp, read it bounded, unlink the temp. 0 / -1 (st set).
102 *
103 * WHY: Reusing copy_dispatch_one means every transport, auth scheme and
104 * resilience knob the client has works for the metalink document itself —
105 * no second fetch stack to maintain.
106 *
107 * HOW: 1. mkstemp under $TMPDIR (0600, private). 2. Inner dispatch with
108 * metalink_off=1 (no recursion), silent, force (the temp exists), and
109 * every transfer-decoration (cksum/progress/xcp/zip/recursive) zeroed.
110 * 3. Bounded local read + unlink on every path.
111 */
112static int
113metalink_fetch_remote(const char *src, const brix_copy_opts *o,
114 const brix_opts *co, char **buf, size_t *blen,
115 brix_status *st)
116{
117 const char *tmpdir = getenv("TMPDIR");
118 char tmp[XRDC_PATH_MAX];
119 int fd, rc;
120 brix_copy_opts fetch;
121
122 if (tmpdir == NULL || tmpdir[0] == '\0') {
123 tmpdir = "/tmp";
124 }
125 if ((size_t) snprintf(tmp, sizeof(tmp), "%s/xrdcp-mtln-XXXXXX",
126 tmpdir) >= sizeof(tmp)) {
127 brix_status_set(st, XRDC_EUSAGE, 0, "TMPDIR path too long");
128 return -1;
129 }
130 fd = mkstemp(tmp);
131 if (fd < 0) {
132 brix_status_set(st, XRDC_EIO, errno, "mkstemp %s: %s",
133 tmp, strerror(errno));
134 return -1;
135 }
136 close(fd); /* the copy engine writes via its own temp+rename */
137
138 fetch = *o;
139 fetch.metalink_off = 1;
140 fetch.silent = 1;
141 fetch.force = 1;
142 fetch.cksum = NULL;
143 fetch.compress = NULL;
144 fetch.zip = 0;
145 fetch.zip_append = 0;
146 fetch.recursive = 0;
147 fetch.pgrw = 0;
148 fetch.parallel = 0;
149 fetch.sources = 0;
150 fetch.xcp_mirrors = NULL;
151 fetch.xcp_n_mirrors = 0;
152 fetch.progress = NULL;
153 fetch.progress_arg = NULL;
154 fetch.remove_source = 0;
155 fetch.dry_run = 0;
156
157 rc = copy_dispatch_one(src, tmp, &fetch, co, st);
158 if (rc == 0) {
159 rc = metalink_read_local(tmp, buf, blen, st);
160 }
161 unlink(tmp);
162 return rc;
163}

166/* ---- Load the metalink document bytes from wherever `src` lives ----
167 *
168 * WHAT: Local paths read directly; anything with a scheme the copy engine
169 * knows fetches through it. 0 / -1 (st set); caller frees *buf.
170 *
171 * WHY: One seam for the two load paths keeps copy_metalink_run flat.
172 *
173 * HOW: Web URLs and root-family URLs go remote; a bare path or file:// (the
174 * LOCAL scheme) reads directly; stdio ("-") is rejected (a stream has no
175 * re-openable mirrors to fail over to anyway).
176 */
177static int
178metalink_load(const char *src, const brix_copy_opts *o, const brix_opts *co,
179 char **buf, size_t *blen, brix_status *st)
180{
181 brix_url u;
182 brix_status parse_st;
183
184 if (brix_is_web_url(src)) {
185 return metalink_fetch_remote(src, o, co, buf, blen, st);
186 }
187 brix_status_clear(&parse_st);
188 if (brix_url_parse(src, &u, &parse_st) == 0) {
189 if (u.scheme == XRDC_SCHEME_ROOT || u.scheme == XRDC_SCHEME_ROOTS) {
190 return metalink_fetch_remote(src, o, co, buf, blen, st);
191 }
192 if (u.scheme == XRDC_SCHEME_LOCAL) {
193 return metalink_read_local(u.path, buf, blen, st);
194 }
195 }
196 brix_status_set(st, XRDC_EUSAGE, 0,
197 "metalink source %s: unsupported scheme", src);
198 return -1;
199}

202/* ---- May this failed mirror attempt fail over to the next mirror? ----
203 *
204 * WHAT: 1 to advance to the next mirror, 0 to stop with this verdict.
205 *
206 * WHY: Mirror failover exists for per-endpoint faults (dead host, missing
207 * replica, auth mismatch, corrupt copy). Verdicts that are identical for
208 * every mirror — usage errors (destination exists, bad flags), local
209 * disk failures, operator cancel — must surface immediately instead of
210 * being retried N times.
211 *
212 * HOW: Deny-list on the local verdict classes + the cooperative cancel flag;
213 * every transport/server/integrity class advances.
214 */
215static int
216metalink_failover_ok(const brix_status *st)
217{
218 if (brix_copy_quit_requested()) {
219 return 0;
220 }
221 if (st->kxr == XRDC_EUSAGE || st->kxr == XRDC_EIO) {
222 return 0;
223 }
224 return 1;
225}

228/* ---- Is this mirror a root-family URL the block engine can dial? ----
229 *
230 * WHAT: 1 for root:// roots:// xroot:// xroots:// spellings. Pure.
231 *
232 * WHY: The extreme-copy engine speaks the root:// wire only; web mirrors stay
233 * in the serial failover loop.
234 *
235 * HOW: Prefix table, case-insensitive (mirror documents are hand-written).
236 */
237static int
238metalink_url_is_root_family(const char *mirror_url)
239{
240 static const char *const roots[] = {
241 "root://", "roots://", "xroot://", "xroots://",
242 };
243 size_t i;
244
245 for (i = 0; i < sizeof(roots) / sizeof(roots[0]); i++) {
246 if (strncasecmp(mirror_url, roots[i], strlen(roots[i])) == 0) {
247 return 1;
248 }
249 }
250 return 0;
251}

254/* ---- Run the transfer against the ranked mirrors with ordered failover ----
255 *
256 * WHAT: The virtual-redirector core: fetch + parse the metalink named by
257 * `src`, then dispatch dst against urls[0], urls[1], … until one
258 * succeeds. 0 / -1 (st = last mirror's verdict, or the parse failure).
259 *
260 * WHY: This IS the metalink feature: mirror failover for a one-shot copy, the
261 * document's digest inherited as an integrity gate, and the mirror list
262 * handed to the extreme-copy engine when --sources wants it.
263 *
264 * HOW: 1. Load + parse (buffer freed immediately after parse). 2. Build the
265 * per-attempt opts once: metalink_off (no recursion), the document
266 * digest as a literal --cksum when the user gave none, and the
267 * root-family mirror subset wired to xcp_mirrors for --sources. 3.
268 * Ordered loop over mirrors: success returns; a non-failover verdict
269 * stops; otherwise note the failure (unless -s) and advance.
270 */
271int
272copy_metalink_run(const char *src, const char *dst, const brix_copy_opts *o,
273 const brix_opts *co, brix_status *st)
274{
275 brix_metalink ml;
276 char *buf = NULL;
277 size_t blen = 0;
278 brix_copy_opts attempt;
279 char cksum_spec[XRDC_METALINK_ALGO_MAX + XRDC_METALINK_HEX_MAX];
280 const char *root_mirrors[XRDC_METALINK_MAX_URLS];
281 size_t n_root = 0, i;
282 int rc = -1;
283
284 if (metalink_load(src, o, co, &buf, &blen, st) != 0) {
285 return -1;
286 }
287 rc = brix_metalink_parse(buf, blen, &ml, st);
288 free(buf);
289 if (rc != 0) {
290 return -1;
291 }
292
293 attempt = *o;
294 attempt.metalink_off = 1;
295
296 /* Inherit the document digest as a literal integrity gate when the user
297 * did not pick their own --cksum: a corrupt mirror then drops the download
298 * exactly like a failed --cksum today (committed-but-bad file unlinked). */
299 if (o->cksum == NULL && ml.hash_algo[0] != '\0') {
300 snprintf(cksum_spec, sizeof(cksum_spec), "%s:%s",
301 ml.hash_algo, ml.hash_hex);
302 attempt.cksum = cksum_spec;
303 }
304
305 /* The extreme-copy engine speaks the root:// wire only; hand it that
306 * subset of the ranked mirrors (best-first order preserved). */
307 for (i = 0; i < ml.n_urls; i++) {
308 if (metalink_url_is_root_family(ml.urls[i].rank_url)) {
309 root_mirrors[n_root++] = ml.urls[i].rank_url;
310 }
311 }
312 attempt.xcp_mirrors = (n_root > 0) ? root_mirrors : NULL;
313 attempt.xcp_n_mirrors = n_root;
314
315 for (i = 0; i < ml.n_urls; i++) {
316 rc = copy_dispatch_one(ml.urls[i].rank_url, dst, &attempt, co, st);
317 if (rc == 0) {
318 return 0;
319 }
320 if (!metalink_failover_ok(st)) {
321 return -1;
322 }
323 if (!o->silent && i + 1 < ml.n_urls) {
324 fprintf(stderr, "xrdcp: metalink mirror %s failed (%s); "
325 "trying next mirror\n",
326 ml.urls[i].rank_url, st->msg);
327 }
328 }
329 return rc;
330}
