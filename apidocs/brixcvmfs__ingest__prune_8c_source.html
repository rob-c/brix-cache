1/* brixcvmfs_ingest_prune.c — `brixcvmfs ingest prune` (phase-104 D8):
2 * retire unreferenced digest roots. Tool surface only (G14).
3 *
4 * brixcvmfs ingest prune --repo <repo_dir> [--prefix /images] [--keep N]
5 * [--keys-dir D] [--chunk-size N] [--dry-run] [--no-wait]
6 *
7 * Referenced = named by ANY memo file under <repo>/.brix-ingest/memo (the
8 * global/conservative rule: a root stays while any tag anywhere points at
9 * it). Unreferenced roots — the <repo>/.brix-ingest/roots<prefix> ledger
10 * minus the referenced set — are ordered newest-first by ledger mtime;
11 * --keep N spares the N newest; the rest are DELETEd in one publish under
12 * the transaction lock, then their ledger entries are unlinked. Ledger
13 * unlink comes after the publish, so a crash between the two leaves only a
14 * re-prunable ledger entry, never a dangling published root.
15 *
16 * `--layout layered` images (D15.6) add a second pass: once the image roots
17 * are gone, a layer root that no surviving image's `imglayers` record names
18 * is unreachable too, and is retired the same way in its own publish. The
19 * two passes are ordered, never merged — a layer is only provably orphaned
20 * after the images that composed it have actually left the tree.
21 *
22 * `bci_prune_old_mark`/`bci_root_forget` are the same rule reached from the
23 * other side: `ingest image --prune-old` retiring the root its tag just
24 * moved off. Deleting a digest root lives here, once.
25 */

63/* Normalize an optional prune prefix while retaining the root identity. */
64static int prune_prefix(ing_prune_opts_t *o, const char *value) {
65 size_t n = snprintf(o->prefix, sizeof(o->prefix), "%s", value);
66
67 if (n >= sizeof(o->prefix)) return -1;
68 while (n > 1 && o->prefix[n - 1] == '/') o->prefix[--n] = '\0';
69 return 0;

84/* Consume one prune option; *i advances when its option owns a value. */
85static int prune_option(int argc, char **argv, int *i, ing_prune_opts_t *o) {
86 const char *arg = argv[*i];
87 const char *value = *i + 1 < argc ? argv[*i + 1] : NULL;
88
89 if (strcmp(arg, "--dry-run") == 0) o->dry_run = 1;
90 else if (strcmp(arg, "--no-wait") == 0) o->no_wait = 1;
91 else if (strcmp(arg, "--prefix") == 0) {
92 if (value == NULL || prune_prefix(o, value) != 0) return ING_USAGE;
93 ++*i;
94 } else if (prune_value_option(arg, value, o) == 0) {
95 ++*i;
96 } else return ING_USAGE;
97 return ING_OK;
98}

110static int prune_cmp(const void *a, const void *b) {
111 const prune_root_t *x = a, *y = b;
112 return x->mtime < y->mtime ? 1 : x->mtime > y->mtime ? -1
113 : strcmp(x->dig.hex, y->dig.hex);
114}

116/* Roots ledger scan → unreferenced entries, newest-first. */
117static int prune_collect(const ing_prune_opts_t *o, const char *roots_dir,
118 prune_root_t **out, size_t *out_n) {
119 char memo_dir[ING_PATH_MAX];
120 snprintf(memo_dir, sizeof(memo_dir), "%s/.brix-ingest/memo", o->repo);
121 *out = NULL;
122 *out_n = 0;
123 DIR *d = opendir(roots_dir);
124 if (d == NULL) return ING_OK; /* nothing ever ingested here */
125 struct dirent *e;
126 size_t cap = 0;
127 while ((e = readdir(d)) != NULL) {
128 char sub[ING_PATH_MAX + 80], digest[BRIX_OCI_DIGEST_STRLEN];
129 brix_oci_digest_t dig;
130 struct stat st;
131 /* A ledger name is a bare hex whose width names its algorithm; a
132 * name that is not one is not this tool's to delete. */
133 if (brix_oci_digest_parse_hex(e->d_name, strlen(e->d_name), &dig) != 0
134 || snprintf(sub, sizeof(sub), "%s/%s", roots_dir, e->d_name)
135 >= (int) sizeof(sub)
136 || lstat(sub, &st) != 0 || !S_ISREG(st.st_mode)
137 || brix_oci_digest_format(&dig, digest, sizeof(digest)) < 0)
138 continue;
139 if (bci_memo_refs(memo_dir, digest, "") > 0) continue;
140 if (*out_n == cap) {
141 cap = cap != 0 ? cap * 2 : 8;
142 prune_root_t *nv = realloc(*out, cap * sizeof(*nv));
143 if (nv == NULL) {
144 closedir(d);
145 return bci_fail(ING_FAIL, "out of memory", NULL);
146 }
147 *out = nv;
148 }
149 prune_root_t *r = &(*out)[(*out_n)++];
150 r->dig = dig;
151 r->mtime = st.st_mtime;
152 r->victim = 0;
153 }
154 closedir(d);
155 qsort(*out, *out_n, sizeof(**out), prune_cmp);
156 return ING_OK;
157}

159static int prune_publish(const ing_prune_opts_t *o, prune_root_t *v,
160 size_t nv, size_t nvictims) {
161 cvmfs_changeset_t cs;
162 char err[1024], path[ING_PATH_MAX + 96];
163 long rev = 0;
164 memset(&cs, 0, sizeof(cs));
165 int rc = ING_OK;
166 for (size_t i = 0; rc == ING_OK && i < nv; i++) {
167 if (!v[i].victim) continue;
168 cvmfs_change_t *ch = bci_cs_append(&cs);
169 snprintf(path, sizeof(path), "%s/.images/%s/%s",
170 bci_pfx(o->prefix), brix_oci_alg_name(v[i].dig.alg),
171 v[i].dig.hex);
172 if (ch == NULL || (ch->path = strdup(path)) == NULL)
173 rc = bci_fail(ING_FAIL, "out of memory", NULL);
174 else
175 ch->op = CVMFS_CH_DELETE;
176 }
177 if (rc == ING_OK) rc = bci_lock_acquire(o->repo, o->no_wait);
178 if (rc == ING_OK) {
179 cvmfs_publish_opts_t po;
180 memset(&po, 0, sizeof(po));
181 po.repo_dir = o->repo;
182 po.keys_dir = o->keys_dir;
183 po.chunk_size = o->chunk_size;
184 rc = cvmfs_publish_run(&po, &cs, &rev, err, sizeof(err)) == 0
185 ? ING_OK : bci_fail(ING_FAIL, "publish failed", err);
186 bci_lock_release(o->repo);
187 }
188 cvmfs_changeset_free(&cs);
189 if (rc != ING_OK) return rc;
190 for (size_t i = 0; i < nv; i++) {
191 if (!v[i].victim) continue;
192 snprintf(path, sizeof(path), "%s/.brix-ingest/roots%s/%s",
193 o->repo, bci_pfx(o->prefix), v[i].dig.hex);
194 unlink(path);
195 snprintf(path, sizeof(path), "%s/.brix-ingest/imglayers%s/%s",
196 o->repo, bci_pfx(o->prefix), v[i].dig.hex);
197 unlink(path); /* layered: pass 2 reads what is left */
198 }
199 printf("pruned %zu root(s) under %s (revision %ld)\n",
200 nvictims, o->prefix, rev);
201 return ING_OK;
202}

206int
207bci_prune_old_mark(const char *repo, const char *prefix, const char *memo_path,
208 const char *old_digest, const char *new_digest,
209 cvmfs_changeset_t *cs)
210{
211 char memo_dir[ING_PATH_MAX], path[ING_PATH_MAX + 96];
212 brix_oci_digest_t old;
213 cvmfs_change_t *ch;
214
215 if (old_digest[0] == '\0' || strcmp(old_digest, new_digest) == 0) {
216 return 0; /* first publish, or the tag stayed */
217 }
218 snprintf(memo_dir, sizeof(memo_dir), "%s/.brix-ingest/memo", repo);
219 if (bci_memo_refs(memo_dir, old_digest, memo_path) > 0) {
220 return 0; /* another tag still points at it */
221 }
222 if (brix_oci_digest_parse(old_digest, strlen(old_digest), &old) != 0
223 || snprintf(path, sizeof(path), "%s/.images/%s/%s", bci_pfx(prefix),
224 brix_oci_alg_name(old.alg), old.hex)
225 >= (int) sizeof(path))
226 {
227 bci_fail(ING_FAIL, "unusable previous digest", old_digest);
228 return -1;
229 }
230 ch = bci_cs_append(cs);
231 if (ch == NULL || (ch->path = strdup(path)) == NULL) {
232 bci_fail(ING_FAIL, "out of memory", NULL);
233 return -1;
234 }
235 ch->op = CVMFS_CH_DELETE;
236 return 1;
237}

256static int prune_layers(const ing_prune_opts_t *o)
257{
258 cvmfs_changeset_t cs;
259 char err[1024], **hex = NULL;
260 long rev = 0;
261 int n, rc;
262
263 memset(&cs, 0, sizeof(cs));
264 n = bci_layer_orphans(o->repo, o->prefix, &cs, &hex);
265 if (n < 0) {
266 cvmfs_changeset_free(&cs);
267 return ING_FAIL;
268 }
269 if (n == 0) {
270 cvmfs_changeset_free(&cs);
271 return ING_OK; /* flat repositories land here */
272 }
273 if (o->dry_run) {
274 for (int i = 0; i < n; i++) {
275 printf("would prune layer %s\n", cs.v[i].path);
276 }
277 cvmfs_changeset_free(&cs);
278 bci_layer_release(hex, n);
279 return ING_OK;
280 }
281 rc = bci_lock_acquire(o->repo, o->no_wait);
282 if (rc == ING_OK) {
283 cvmfs_publish_opts_t po;
284 memset(&po, 0, sizeof(po));
285 po.repo_dir = o->repo;
286 po.keys_dir = o->keys_dir;
287 po.chunk_size = o->chunk_size;
288 rc = cvmfs_publish_run(&po, &cs, &rev, err, sizeof(err)) == 0
289 ? ING_OK : bci_fail(ING_FAIL, "layer publish failed", err);
290 bci_lock_release(o->repo);
291 }
292 cvmfs_changeset_free(&cs);
293 if (rc == ING_OK) {
294 bci_layer_forget(o->repo, o->prefix, hex, n);
295 printf("pruned %d layer root(s) under %s (revision %ld)\n",
296 n, o->prefix, rev);
297 } else {
298 bci_layer_release(hex, n); /* the roots are still published */
299 }
300 return rc;
301}

304int bci_prune_main(int argc, char **argv) {
305 ing_prune_opts_t o;
306 int rc = prune_parse(argc, argv, &o);
307 if (rc != ING_OK) return rc;
308 rc = bci_prefix_check(o.prefix);
309 if (rc != ING_OK) return rc;
310 char roots_dir[ING_PATH_MAX];
311 snprintf(roots_dir, sizeof(roots_dir), "%s/.brix-ingest/roots%s",
312 o.repo, bci_pfx(o.prefix));
313 prune_root_t *v = NULL;
314 size_t nv = 0, nvictims = 0;
315 rc = prune_collect(&o, roots_dir, &v, &nv);
316 for (size_t i = 0; rc == ING_OK && i < nv; i++) {
317 if (i < (size_t) o.keep) continue; /* newest-first: spare the N newest */
318 v[i].victim = 1;
319 nvictims++;
320 if (o.dry_run)
321 printf("would prune %s/.images/%s/%s\n", bci_pfx(o.prefix),
322 brix_oci_alg_name(v[i].dig.alg), v[i].dig.hex);
323 }
324 if (rc == ING_OK && nvictims == 0)
325 printf("nothing to prune under %s\n", o.prefix);
326 else if (rc == ING_OK && !o.dry_run)
327 rc = prune_publish(&o, v, nv, nvictims);
328 free(v);
329 /* Pass 2 runs even when pass 1 found nothing: a layer can also be
330 * orphaned by an image root removed some other way. */
331 if (rc == ING_OK)
332 rc = prune_layers(&o);
333 return rc;
334}
