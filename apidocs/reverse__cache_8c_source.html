1/*
2 * reverse_cache.c — per-worker reverse (PTR) answer cache (phase-116 W7).
3 *
4 * WHAT: A bounded cache keyed by the raw peer address (family + address
5 * bytes, port ignored) holding one FQDN, a negative marker, or an
6 * "in flight" marker with an expiry. The reverse consumers — XrdAcc
7 * `h` rules, protbind hostname templates, `host` auth, the TPC origin
8 * id — ask this cache on the event loop and never wait for it there.
9 * WHY: Reverse lookups used to happen inline on the event loop (a blocking
10 * PTR query per connection) — a slow resolver stalled every
11 * connection in the worker. Keying by address lets one background fill
12 * serve every connection from that peer for the record's TTL, and the
13 * in-flight marker keeps a burst of connections from starting a fill
14 * each (I-DNS-1).
15 * HOW: ngx_rbtree keyed by ngx_crc32_short over the address bytes with a
16 * memcmp on collision, plus an LRU queue — the forward cache's layout
17 * (cache.c). Per-process singleton guarded by a plain pthread mutex:
18 * the event loop probes and stores, the TPC pull thread stores through
19 * brix_dns_reverse_sync(). Nothing under the lock blocks or re-enters.
20 */
21#include "net/dns/dns.h"

23#include <pthread.h>

54static void
55dns_rcache_ready(void)
56{
57 if (dns_rcache.ready) {
58 return;
59 }
60 ngx_rbtree_init(&dns_rcache.tree, &dns_rcache.sentinel,
61 ngx_rbtree_insert_value);
62 ngx_queue_init(&dns_rcache.lru);
63 if (dns_rcache.max == 0) {
64 dns_rcache.max = BRIX_DNS_CACHE_MAX_DEFAULT;
65 }
66 dns_rcache.ready = 1;
67}

83/* The address bytes that identify a peer; 0 for an unsupported family. */
84static size_t
85dns_rcache_key(const struct sockaddr *sa, socklen_t len, u_char *addr,
86 uint32_t *key)
87{
88 size_t alen = 0;
89
90 if (sa == NULL) {
91 return 0;
92 }
93 if (sa->sa_family == AF_INET && len >= sizeof(struct sockaddr_in)) {
94 ngx_memcpy(addr, &((const struct sockaddr_in *) sa)->sin_addr, 4);
95 alen = 4;
96 }
97#if (NGX_HAVE_INET6)
98 if (sa->sa_family == AF_INET6 && len >= sizeof(struct sockaddr_in6)) {
99 ngx_memcpy(addr, &((const struct sockaddr_in6 *) sa)->sin6_addr, 16);
100 alen = 16;
101 }
102#endif
103 if (alen > 0) {
104 *key = ngx_crc32_short(addr, alen);
105 }
106 return alen;
107}

110static dns_rcache_entry_t *
111dns_rcache_find(uint32_t key, const u_char *addr, size_t alen,
112 sa_family_t family)
113{
114 ngx_rbtree_node_t *node = dns_rcache.tree.root;
115 ngx_rbtree_node_t *sentinel = dns_rcache.tree.sentinel;
116
117 while (node != sentinel) {
118 dns_rcache_entry_t *e;
119
120 if (key < node->key) {
121 node = node->left;
122 continue;
123 }
124 if (key > node->key) {
125 node = node->right;
126 continue;
127 }
128 e = (dns_rcache_entry_t *) node;
129 if (e->alen == alen && e->family == family
130 && ngx_memcmp(e->addr, addr, alen) == 0)
131 {
132 return e;
133 }
134 node = node->right; /* crc collision: keep walking */
135 }
136 return NULL;
137}

150static void
151dns_rcache_evict_lru(void)
152{
153 ngx_queue_t *q;
154
155 while (dns_rcache.entries >= dns_rcache.max
156 && !ngx_queue_empty(&dns_rcache.lru))
157 {
158 q = ngx_queue_last(&dns_rcache.lru);
159 dns_rcache_remove(ngx_queue_data(q, dns_rcache_entry_t, queue));
160 }
161}

164/* Insert a fresh entry for the key (any existing one already removed). */
165static dns_rcache_entry_t *
166dns_rcache_insert(uint32_t key, const u_char *addr, size_t alen,
167 sa_family_t family, time_t expires)
168{
169 dns_rcache_entry_t *e;
170
171 dns_rcache_evict_lru();
172 e = ngx_calloc(sizeof(dns_rcache_entry_t), ngx_cycle->log);
173 if (e == NULL) {
174 return NULL;
175 }
176 ngx_memcpy(e->addr, addr, alen);
177 e->alen = alen;
178 e->family = family;
179 e->expires = expires;
180 e->node.key = key;
181 ngx_rbtree_insert(&dns_rcache.tree, &e->node);
182 ngx_queue_insert_head(&dns_rcache.lru, &e->queue);
183 dns_rcache.entries++;
184 return e;
185}

188ngx_int_t
189brix_dns_rcache_lookup(const struct sockaddr *sa, socklen_t len, char *buf,
190 size_t buflen)
191{
192 u_char addr[16];
193 uint32_t key = 0;
194 size_t alen;
195 dns_rcache_entry_t *e;
196 ngx_int_t rc;
197
198 alen = dns_rcache_key(sa, len, addr, &key);
199 if (alen == 0) {
200 return NGX_DECLINED; /* no PTR for a non-IP peer */
201 }
202 pthread_mutex_lock(&dns_rcache_mu);
203 dns_rcache_ready();
204 e = dns_rcache_find(key, addr, alen, sa->sa_family);
205 if (e != NULL && e->expires <= ngx_time()) {
206 dns_rcache_remove(e);
207 e = NULL;
208 }
209 if (e == NULL) {
210 dns_rcache.misses++;
211 rc = NGX_AGAIN;
212 } else if (e->pending) {
213 rc = NGX_AGAIN;
214 } else if (e->negative) {
215 dns_rcache.negative_hits++;
216 rc = NGX_DECLINED;
217 } else {
218 dns_rcache.hits++;
219 ngx_queue_remove(&e->queue);
220 ngx_queue_insert_head(&dns_rcache.lru, &e->queue);
221 if (buf != NULL && buflen > 0) {
222 ngx_cpystrn((u_char *) buf, (u_char *) e->name, buflen);
223 }
224 rc = NGX_OK;
225 }
226 pthread_mutex_unlock(&dns_rcache_mu);
227 return rc;
228}

231unsigned
232brix_dns_rcache_mark_pending(const struct sockaddr *sa, socklen_t len,
233 time_t hold)
234{
235 u_char addr[16];
236 uint32_t key = 0;
237 size_t alen;
238 dns_rcache_entry_t *e;
239 unsigned marked = 0;
240
241 alen = dns_rcache_key(sa, len, addr, &key);
242 if (alen == 0) {
243 return 0;
244 }
245 pthread_mutex_lock(&dns_rcache_mu);
246 dns_rcache_ready();
247 e = dns_rcache_find(key, addr, alen, sa->sa_family);
248 if (e != NULL && e->expires > ngx_time()) {
249 pthread_mutex_unlock(&dns_rcache_mu);
250 return 0; /* known, or already in flight */
251 }
252 if (e != NULL) {
253 dns_rcache_remove(e);
254 }
255 e = dns_rcache_insert(key, addr, alen, sa->sa_family,
256 ngx_time() + (hold > 0 ? hold
257 : BRIX_DNS_PENDING_HOLD));
258 if (e != NULL) {
259 e->pending = 1;
260 marked = 1;
261 }
262 pthread_mutex_unlock(&dns_rcache_mu);
263 return marked;
264}

267void
268brix_dns_rcache_store(const struct sockaddr *sa, socklen_t len,
269 const char *name, time_t ttl)
270{
271 u_char addr[16];
272 uint32_t key = 0;
273 size_t alen;
274 dns_rcache_entry_t *e;
275
276 alen = dns_rcache_key(sa, len, addr, &key);
277 if (alen == 0 || ttl <= 0) {
278 return;
279 }
280 pthread_mutex_lock(&dns_rcache_mu);
281 dns_rcache_ready();
282 e = dns_rcache_find(key, addr, alen, sa->sa_family);
283 if (e != NULL) {
284 dns_rcache_remove(e);
285 }
286 e = dns_rcache_insert(key, addr, alen, sa->sa_family, ngx_time() + ttl);
287 if (e != NULL) {
288 if (name == NULL || name[0] == '\0') {
289 e->negative = 1;
290 } else {
291 ngx_cpystrn((u_char *) e->name, (u_char *) name, sizeof(e->name));
292 }
293 }
294 pthread_mutex_unlock(&dns_rcache_mu);
295}
