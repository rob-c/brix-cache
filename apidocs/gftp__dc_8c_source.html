14static int
15gftp_dc_port(gftp_session_t *session, unsigned *port)
16{
17 unsigned char ignored[4];
18
19 if (gftp_command(session, "EPSV") != 0) {
20 return -1;
21 }
22 if (session->code == 229) {
23 if (gftp_reply_parse_epsv(session->text, strlen(session->text), port)
24 == 0) {
25 return 0;
26 }
27 gftp_set_error(session, EPROTO, "GridFTP EPSV reply is malformed");
28 return -1;
29 }
30 if (gftp_command(session, "PASV") != 0 || session->code != 227) {
31 gftp_set_error(session, EPROTO, "GridFTP passive mode was refused");
32 return -1;
33 }
34 if (gftp_reply_parse_pasv(session->text, strlen(session->text), ignored,
35 port) != 0) {
36 gftp_set_error(session, EPROTO, "GridFTP PASV reply is malformed");
37 return -1;
38 }
39 return 0;
40}

42/* Dial one already-decided data target. `ip` is a NUMERIC address string —
43 * a literal parse, never a name lookup (phase-116) — and the caller is the one
44 * that decided it may be dialled: gftp_dc_open() pins it to the control
45 * channel's own peer, and gftp_spas.c refuses any stripe that is not that same
46 * peer before it ever reaches here. */
47static int
48gftp_dc_dial(gftp_session_t *session, const char *ip, unsigned port)
49{
50 brix_dns_addr_t addr;
51 ngx_str_t peer;
52 int fd;
53
54 if (port < 1024 || port > 65535 || ip[0] == '\0') {
55 gftp_set_error(session, EACCES,
56 "refusing unsafe GridFTP passive data target");
57 return -1;
58 }
59 peer.data = (u_char *) ip;
60 peer.len = strlen(ip);
61 if (brix_dns_parse_literal(&peer, &addr) != NGX_OK) {
62 gftp_set_error(session, EHOSTUNREACH,
63 "cannot resolve pinned GridFTP data peer");
64 return -1;
65 }
66 ngx_inet_set_port((struct sockaddr *) &addr.ss, (in_port_t) port);
67 fd = socket(addr.ss.ss_family, SOCK_STREAM, 0);
68 if (fd >= 0 && brix_connect_fd_deadline(fd, (struct sockaddr *) &addr.ss,
69 addr.len, session->timeout_ms) != 0) {
70 close(fd);
71 fd = -1;
72 }
73 if (fd < 0) {
74 gftp_set_error(session, ECONNREFUSED,
75 "cannot connect to pinned GridFTP data peer");
76 }
77 return fd;
78}

97int
98gftp_dc_secure(gftp_session_t *session, gftp_dc_t *dc)
99{
100 if (session->prot != GFTP_DPROT_P) {
101 return 0; /* a clear channel needs no handshake */
102 }
103 /* A protected channel that cannot complete its handshake is CLOSED, never
104 * demoted: the caller has no way to reach the clear path from here. */
105 if (gftp_dc_tls_start(session, dc) != 0) {
106 gftp_dc_close(dc);
107 return -1;
108 }
109 return 0;
110}
