1/* stream_common.c — ngx_stream_brix_common_module (phase-101 W3, variant A).

3 * The stream-plane analogue of ngx_http_brix_common_module: it owns the bare

182/*
183 * create_srv_conf — allocate the common module's srv conf and seed its embedded
184 * preamble with UNSET sentinels so parent->child inheritance can tell "not
185 * configured" from an explicit value (mirror of the HTTP common module).
186 */
187static void *
188brix_stream_common_create_srv_conf(ngx_conf_t *cf)
189{
190 ngx_stream_brix_common_conf_t *c;
191
192 c = ngx_pcalloc(cf->pool, sizeof(*c));
193 if (c == NULL) {
194 return NULL;
195 }
196 ngx_http_brix_shared_init(&c->common);
197 return c;
198}

200/*
201 * merge_srv_conf — inheritance-only fold (stream{} main -> server) that applies
202 * NO defaults; each stream protocol picks its own defaults after adopting.
203 * Because brix_stream_common_adopt() reads the server srv conf directly (values
204 * set at parse time), correctness does not depend on whether this fold has run
205 * before root/gridftp adopt.
206 */
207static char *
208brix_stream_common_merge_srv_conf(ngx_conf_t *cf, void *parent, void *child)
209{
210 ngx_stream_brix_common_conf_t *prev = parent;
211 ngx_stream_brix_common_conf_t *conf = child;
212
213 (void) cf;
214 brix_shared_adopt_unified(&conf->common, &prev->common);
215 return NGX_CONF_OK;
216}

233ngx_int_t
234brix_shared_clone_vo_rules(ngx_conf_t *cf, brix_shared_conf_t *conf)
235{
236 ngx_array_t *copy;
237 brix_vo_rule_t *src, *r;
238 ngx_uint_t i;
239
240 if (conf->vo_rules == NULL || conf->vo_rules->nelts == 0) {
241 return NGX_OK;
242 }
243
244 copy = ngx_array_create(cf->pool, conf->vo_rules->nelts,
245 sizeof(brix_vo_rule_t));
246 if (copy == NULL) {
247 return NGX_ERROR;
248 }
249
250 /* Shallow struct copy per rule: path/vo point into cf->pool (immutable), and
251 * .resolved is empty on the common owner (it never finalizes), so each plane
252 * finalizes this fresh array against its own root_canon. */
253 src = conf->vo_rules->elts;
254 for (i = 0; i < conf->vo_rules->nelts; i++) {
255 r = ngx_array_push(copy);
256 if (r == NULL) {
257 return NGX_ERROR;
258 }
259 *r = src[i];
260 }
261
262 conf->vo_rules = copy;
263 return NGX_OK;
264}
