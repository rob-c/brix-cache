1/* stream_common.c — ngx_stream_brix_common_module (phase-101 W3, variant A).
2 *
3 * The stream-plane analogue of ngx_http_brix_common_module: it owns the bare
4 * storage/x509 directive names once, so the root and gridftp stream modules can
5 * both adopt them instead of each registering its own (prefixed) copy. See
6 * stream_common.h for the design and docs/refactor/phase-101-config-surface-
7 * unification.md (W3) for the migration.
8 *
9 * Stage 1 (this commit) is the inert scaffold: the module registers NO
10 * directives, so behaviour is byte-identical to before — it only establishes
11 * the module + the adopt entry point. Stage 2 MOVEs the storage bare names

216/*
217 * create_srv_conf — allocate the common module's srv conf and seed its embedded
218 * preamble with UNSET sentinels so parent->child inheritance can tell "not
219 * configured" from an explicit value (mirror of the HTTP common module).
220 */
221static void *
222brix_stream_common_create_srv_conf(ngx_conf_t *cf)
223{
224 ngx_stream_brix_common_conf_t *c;
225
226 c = ngx_pcalloc(cf->pool, sizeof(*c));
227 if (c == NULL) {
228 return NULL;
229 }
230 ngx_http_brix_shared_init(&c->common);
231 return c;
232}

234/*
235 * merge_srv_conf — inheritance-only fold (stream{} main -> server) that applies
236 * NO defaults; each stream protocol picks its own defaults after adopting.
237 * Because brix_stream_common_adopt() reads the server srv conf directly (values
238 * set at parse time), correctness does not depend on whether this fold has run
239 * before root/gridftp adopt.
240 */
241static char *
242brix_stream_common_merge_srv_conf(ngx_conf_t *cf, void *parent, void *child)
243{
244 ngx_stream_brix_common_conf_t *prev = parent;
245 ngx_stream_brix_common_conf_t *conf = child;
246
247 (void) cf;
248 brix_shared_adopt_unified(&conf->common, &prev->common);
249 return NGX_CONF_OK;
250}

252void
253brix_stream_common_adopt(ngx_conf_t *cf, brix_shared_conf_t *dst)
254{
255 ngx_stream_brix_common_conf_t *scf, *mcf;
256 ngx_stream_conf_ctx_t *main_ctx;
257
258 /* During merge_srv_conf nginx sets cf->ctx to the current server's stream
259 * ctx (ngx_stream.c), so this returns THIS server's stream_common srv conf,
260 * whose directive fields were populated at parse time.
261 * brix_shared_adopt_unified only fills still-UNSET dst slots. */
262 scf = ngx_stream_conf_get_module_srv_conf(cf, ngx_stream_brix_common_module);
263 if (scf != NULL) {
264 brix_shared_adopt_unified(dst, &scf->common);
265 }
266
267 /* phase-116: stream{}-level values (brix_resolver & co. are main|srv).
268 * This module is registered AFTER the protocol modules, so at the moment a
269 * protocol adopts, the server's srv conf has not yet inherited from the
270 * main level — read the stream{} conf directly (the same slot
271 * ngx_stream_cycle_get_module_main_conf walks) and let it fill whatever
272 * the server left unset. */
273 main_ctx = (ngx_stream_conf_ctx_t *)
274 cf->cycle->conf_ctx[ngx_stream_module.index];
275 if (main_ctx != NULL) {
276 mcf = main_ctx->srv_conf[ngx_stream_brix_common_module.ctx_index];
277 if (mcf != NULL && mcf != scf) {
278 brix_shared_adopt_unified(dst, &mcf->common);
279 }
280 }

283ngx_int_t
284brix_shared_clone_vo_rules(ngx_conf_t *cf, brix_shared_conf_t *conf)
285{
286 ngx_array_t *copy;
287 brix_vo_rule_t *src, *r;
288 ngx_uint_t i;
289
290 if (conf->vo_rules == NULL || conf->vo_rules->nelts == 0) {
291 return NGX_OK;
292 }
293
294 copy = ngx_array_create(cf->pool, conf->vo_rules->nelts,
295 sizeof(brix_vo_rule_t));
296 if (copy == NULL) {
297 return NGX_ERROR;
298 }
299
300 /* Shallow struct copy per rule: path/vo point into cf->pool (immutable), and
301 * .resolved is empty on the common owner (it never finalizes), so each plane
302 * finalizes this fresh array against its own root_canon. */
303 src = conf->vo_rules->elts;
304 for (i = 0; i < conf->vo_rules->nelts; i++) {
305 r = ngx_array_push(copy);
306 if (r == NULL) {
307 return NGX_ERROR;
308 }
309 *r = src[i];
310 }
311
312 conf->vo_rules = copy;
313 return NGX_OK;
314}
