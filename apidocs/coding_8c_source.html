8/* WHAT: Read one bounded VFS object into request-owned coding buffers.
9 * WHY: Both codings require exact input, handle-bound metadata, and an output
10 * capacity no larger than input (a coded response must win over identity).
11 * HOW: Validate metadata, allocate bounded buffers, read until complete; refuse
12 * short/error reads so the caller retains its normal identity fallback.
13 */
14int
15brix_cvmfs_coding_load(ngx_http_request_t *r, brix_vfs_file_t *file,
16 brix_vfs_stat_t *metadata, size_t maximum, size_t *length,
17 unsigned char **input, unsigned char **output)
18{
19 size_t offset = 0, size;
20
21 if (brix_vfs_file_stat(file, metadata) != NGX_OK || metadata->is_directory
22 || metadata->size <= 0 || (size_t) metadata->size > maximum)
23 {
24 return -1;
25 }
26 size = (size_t) metadata->size;
27 *input = ngx_pnalloc(r->pool, size);
28 *output = ngx_pnalloc(r->pool, size);
29 if (*input == NULL || *output == NULL) {
30 return -1;
31 }
32 while (offset < size) {
33 ssize_t count = brix_vfs_file_pread(file, *input + offset,
34 size - offset, (off_t) offset);
35 if (count <= 0 || (size_t) count > size - offset) {
36 return -1;
37 }
38 offset += (size_t) count;
39 }
40 *length = size;
41 return 0;
42}
