1/*
2 * ftp_data.c — GridFTP client data channel: passive open, RETR, STOR, listings.
3 *
4 * WHAT: negotiate a passive data connection (EPSV, falling back to PASV), then
5 * run exactly one transfer over it in stream mode with the control channel
6 * supplying the preliminary (1xx) and completion (2xx) replies.
7 * WHY: stream mode over a passive connection is the one shape every GridFTP
8 * server supports, including the phase-82 gateway; extended block mode buys
9 * parallelism this client does not yet need and costs a second framing
10 * layer. Keeping open/pump/finish here means the transfer contract (never
11 * report success before the server's 226) lives in one place.
12 * HOW: the passive replies are decoded by the pure kernels gftp_reply_parse_epsv
13 * / gftp_reply_parse_pasv and screened by ftp_screen.c before any socket is
14 * dialled. Payload bytes move through caller-supplied sink/source adapters
15 * so the same pump serves a file, a listing buffer, or a progress-reporting
16 * copy. No goto: each step returns early and closes what it opened.
17 */

29/* Decode the passive reply currently in `s` into an address + port candidate. */
30static int
31ftp_passive_target(brix_ftp_sess *s, char *ip, size_t ipsz, unsigned *port,
32 brix_status *st)
33{
34 size_t tlen = strlen(s->text);
35
36 if (s->code == 229) {
37 if (gftp_reply_parse_epsv(s->text, tlen, port) != 0) {
38 brix_status_set(st, XRDC_EPROTO, 0, "gsiftp: malformed EPSV reply");
39 return -1;
40 }
41 snprintf(ip, ipsz, "%s", s->peer_ip); /* EPSV inherits the peer */
42 return 0;
43 }
44 {
45 unsigned char a[4];
46
47 if (gftp_reply_parse_pasv(s->text, tlen, a, port) != 0) {
48 brix_status_set(st, XRDC_EPROTO, 0, "gsiftp: malformed PASV reply");
49 return -1;
50 }
51 snprintf(ip, ipsz, "%u.%u.%u.%u", a[0], a[1], a[2], a[3]);
52 }
53 return 0;
54}

57int
58brix_ftp_data_open(brix_ftp_sess *s, int *dfd, brix_status *st)
59{
60 char ip[64];
61 unsigned port = 0;
62 const char *env = getenv("BRIX_GSIFTP_ALLOW_OFFPEER");
63 int allow_offpeer = (env != NULL && env[0] == '1');
64
65 *dfd = -1;
66 if (brix_ftp_cmd(s, st, "EPSV") != 0) {
67 return -1;
68 }
69 if (s->code != 229) {
70 if (brix_ftp_cmd(s, st, "PASV") != 0) {
71 return -1;
72 }
73 if (s->code != 227) {
74 brix_status_set(st, XRDC_EPROTO, 0,
75 "gsiftp: passive mode refused: %d %s", s->code,
76 s->text);
77 return -1;
78 }
79 }
80 if (ftp_passive_target(s, ip, sizeof(ip), &port, st) != 0) {
81 return -1;
82 }
83 if (!brix_ftp_data_addr_ok(s->peer_ip, ip, port, allow_offpeer)) {
84 /* The port is the hard gate — a passive data port below 1024 is a bounce
85 * attempt, never a real transfer. An off-peer *address* is merely
86 * ignored: we dial the control peer instead (curl's
87 * --ftp-skip-pasv-ip behaviour), which also keeps NAT'd servers working. */
88 if (!brix_ftp_data_addr_ok(s->peer_ip, s->peer_ip, port, 0)) {
89 brix_status_set(st, XRDC_EPROTO, 0,
90 "gsiftp: refusing data channel to port %u", port);
91 return -1;
92 }
93 snprintf(ip, sizeof(ip), "%s", s->peer_ip);
94 }
95
96 *dfd = brix_tcp_connect(ip, (int) port, s->timeout_ms, st);
97 if (*dfd < 0) {
98 return -1;
99 }
100 return 0;
101}

104/* Await the preliminary reply that opens a transfer (150/125). */
105static int
106ftp_await_preliminary(brix_ftp_sess *s, brix_status *st)
107{
108 if (s->code >= 100 && s->code < 200) {
109 return 0;
110 }
111 brix_status_set(st, (s->code == 550) ? XRDC_ENOENT : XRDC_EPROTO, 0,
112 "gsiftp: %d %s", s->code, s->text);
113 return -1;

142/*
143 * Open the data channel, issue one transfer verb, take the preliminary reply and
144 * hand back the connected socket plus a transfer buffer. On failure everything it
145 * opened is released and the caller returns straight away. 0 / -1 (st set).
146 */
147static int
148ftp_xfer_begin(brix_ftp_sess *s, const char *verb, const char *path, int *dfd,
149 uint8_t **buf, brix_status *st)
150{
151 *dfd = -1;
152 *buf = NULL;
153 if (brix_ftp_data_open(s, dfd, st) != 0) {
154 return -1;
155 }
156 if (brix_ftp_cmd(s, st, "%s %s", verb, path) != 0
157 || ftp_await_preliminary(s, st) != 0) {
158 close(*dfd);
159 return -1;
160 }
161 *buf = malloc(BRIX_FTP_DATA_CHUNK);
162 if (*buf == NULL) {
163 close(*dfd);
164 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
165 return -1;
166 }
167 return 0;
168}

176 return 0;

183/* One chunk of work in whichever direction the transfer runs: bytes moved, 0 at
184 * end of transfer, -1 on failure (st set). */
185typedef struct {
186 brix_ftp_sink_fn sink; /* RETR: socket → caller */
187 brix_ftp_src_fn src; /* STOR: caller → socket */
188 void *ctx;
189} ftp_pump;
190
191typedef ssize_t (*ftp_chunk_fn)(brix_ftp_sess *s, int dfd, uint8_t *buf,
192 const ftp_pump *p, brix_status *st);
193
194static ssize_t
195ftp_chunk_in(brix_ftp_sess *s, int dfd, uint8_t *buf, const ftp_pump *p,
196 brix_status *st)
197{
198 ssize_t n = brix_ftp_io_read(dfd, buf, BRIX_FTP_DATA_CHUNK, s->timeout_ms,
199 st);
200
201 if (n <= 0) {
202 return n;
203 }
204 if (p->sink(p->ctx, buf, (size_t) n, st) != 0) {
205 return -1;
206 }
207 return n;
208}

211static ssize_t
212ftp_chunk_out(brix_ftp_sess *s, int dfd, uint8_t *buf, const ftp_pump *p,
213 brix_status *st)
214{
215 ssize_t n = p->src(p->ctx, buf, BRIX_FTP_DATA_CHUNK, st);
216
217 if (n <= 0) {
218 return n;
219 }
220 if (brix_ftp_io_write_all(dfd, buf, (size_t) n, s->timeout_ms, st) != 0) {
221 return -1;
222 }
223 return n;
224}

227/*
228 * Run one whole transfer: open the channel, issue `verb`, pump chunks until the
229 * stream ends, then take the server's completion reply. `half_close` marks the
230 * outbound direction, where the FIN is the end-of-file marker and must reach the
231 * server before we wait for its 226.
232 */
233static int
234ftp_xfer(brix_ftp_sess *s, const char *verb, const char *path,
235 ftp_chunk_fn chunk, const ftp_pump *pump, const brix_copy_opts *o,
236 int64_t total, int half_close, brix_status *st)
237{
238 uint8_t *buf;
239 int64_t done = 0;
240 int dfd;
241 int rc = 0;
242
243 if (ftp_xfer_begin(s, verb, path, &dfd, &buf, st) != 0) {
244 return -1;
245 }
246 for (;;) {
247 ssize_t n;
248
249 if (ftp_cancelled(st)) {
250 rc = -1;
251 break;
252 }
253 n = chunk(s, dfd, buf, pump, st);
254 if (n < 0) {
255 rc = -1;
256 break;
257 }
258 if (n == 0) {
259 break;
260 }
261 done += n;
262 ftp_report(o, done, total);
263 }
264 free(buf);
265 if (half_close) {
266 (void) shutdown(dfd, SHUT_WR);
267 }
268 close(dfd);
269 if (rc != 0) {
270 return -1;
271 }
272 return ftp_await_completion(s, st);
273}

298int
299brix_ftp_data_slurp(brix_ftp_sess *s, const char *cmd, char **out,
300 size_t *out_len, brix_status *st)
301{
302 char *acc = NULL;
303 size_t len = 0;
304 uint8_t buf[8192];
305 int dfd = -1;
306 int rc = 0;
307
308 *out = NULL;
309 *out_len = 0;
310 if (brix_ftp_data_open(s, &dfd, st) != 0) {
311 return -1;
312 }
313 if (brix_ftp_cmd(s, st, "%s", cmd) != 0
314 || ftp_await_preliminary(s, st) != 0) {
315 close(dfd);
316 return -1;
317 }
318 for (;;) {
319 ssize_t n = brix_ftp_io_read(dfd, buf, sizeof(buf), s->timeout_ms, st);
320 char *bigger;
321
322 if (n < 0) {
323 rc = -1;
324 break;
325 }
326 if (n == 0) {
327 break;
328 }
329 bigger = realloc(acc, len + (size_t) n + 1);
330 if (bigger == NULL) {
331 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
332 rc = -1;
333 break;
334 }
335 acc = bigger;
336 memcpy(acc + len, buf, (size_t) n);
337 len += (size_t) n;
338 acc[len] = '\0';
339 }
340 close(dfd);
341 if (rc != 0) {
342 free(acc);
343 return -1;
344 }
345 if (ftp_await_completion(s, st) != 0) {
346 free(acc);
347 return -1;
348 }
349 if (acc == NULL) {
350 acc = calloc(1, 1);
351 if (acc == NULL) {
352 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
353 return -1;
354 }
355 }
356 *out = acc;
357 *out_len = len;
358 return 0;
359}
