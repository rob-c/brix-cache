10/* WHAT: Send 1025 descriptors on an isolated connection.
11 * WHY: Stock refusal drops the link, which must not poison later probes.
12 * HOW: Authenticate a second connection and send an all-zero descriptor block;
13 * count validation precedes handle validation and returns ArgTooLong. */
14static int
15probe_one_over(xrd_conform_execution *execution, brix_status *status)
16{
17 ClientWriteVRequest request;
18 brix_payload_ext payload;
19 brix_resp_out response;
20 brix_conn connection;
21 brix_opts opts = execution->run->opts;

24 uint8_t *body = NULL;
25 uint32_t body_length = 0;

27 uint16_t wire_status = 0;
28 int rc = -1;
29
30 opts.capture = NULL;
31 if (brix_endpoint_parse(execution->target, &url, status) != 0
32 || brix_connect(&connection, &url, &opts, status) != 0)
33 return -1;
34 descriptors = calloc(WVC_LIMIT + 1U, WVC_DESCRIPTOR_SIZE);
35 if (descriptors == NULL) {
36 brix_status_set(status, XRDC_EIO, errno,
37 "writev boundary allocation failed");
38 brix_close(&connection);
39 return -1;
40 }
41 memset(&request, 0, sizeof(request));
42 request.requestid = htons(kXR_writev);
43 xrdw_writev_req_t fields = { .options = 0 };
44 xrdw_writev_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
45 payload.data = descriptors;
46 payload.len = (WVC_LIMIT + 1U) * WVC_DESCRIPTOR_SIZE;
47 payload.dlen = payload.len;
48 response.status = &wire_status;
49 response.body = &body;
50 response.blen = &body_length;
51 brix_status_clear(status);
52 if (xrd_conform_reserve_status(execution, 0, status)
53 && brix_send_ext(&connection, &request, &payload, &stream_id,
54 status) == 0)
55 rc = brix_recv(&connection, stream_id, &response, status);
56 free(body);
57 free(descriptors);
58 brix_close(&connection);
59 return rc != 0 && status->kxr == kXR_ArgTooLong ? 0 : -1;
60}

92/* WHAT: Exercise the exact 1024/1025 kXR_writev segment boundary remotely.
93 * WHY: A local client cap alone cannot prove remote count enforcement.
94 * HOW: Write and read back 1024 distinct bytes, then isolate the dropping
95 * one-over request on a fresh connection and require kXR_ArgTooLong. */
96void
97xrd_conform_probe_writev_limits(xrd_conform_execution *execution,
98 brix_conn *connection, const xrd_conform_scratch_object *object)
99{
100 brix_writev_seg *segments = NULL;
101 uint8_t *expected = NULL;
102 uint8_t *actual = NULL;
103 brix_status status;
104 size_t index;
105 int at_limit = 0;
106 int over_limit;
107
108 segments = calloc(WVC_LIMIT, sizeof(*segments));
109 expected = malloc(WVC_LIMIT);
110 actual = malloc(WVC_LIMIT);
111 if (segments == NULL || expected == NULL || actual == NULL) {
112 xrd_conform_result_set(execution, "WVC-002",
113 BRIX_CONFORM_VERDICT_INCONCLUSIVE, "local-allocation-failed",
114 "1024 exact; 1025 kXR_ArgTooLong", NULL);
115 free(actual); free(expected); free(segments);
116 return;
117 }
118 brix_conform_fill(&execution->run->key, 0, expected, WVC_LIMIT);
119 for (index = 0; index < WVC_LIMIT; index++) {
120 segments[index].offset = (int64_t) index;
121 segments[index].len = 1;
122 segments[index].data = expected + index;
123 }
124 brix_status_clear(&status);
125 at_limit = probe_at_limit(execution, connection, object, segments,
126 expected, actual, &status);
127 over_limit = probe_one_over(execution, &status) == 0;
128 xrd_conform_result_set(execution, "WVC-002",
129 at_limit && over_limit ? BRIX_CONFORM_VERDICT_CONFORMANT
130 : BRIX_CONFORM_VERDICT_DIVERGENT,
131 at_limit && over_limit ? "writev-count-boundary-enforced"
132 : "writev-count-boundary-mismatch",
133 "1024 byte-exact; 1025 kXR_ArgTooLong",
134 at_limit && over_limit ? NULL : status.msg);
135 free(actual); free(expected); free(segments);
136}
