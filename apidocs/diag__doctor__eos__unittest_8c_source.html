29{
30 /* minimal quoting — enough for the constrained EOS token charset */
31 fputc('"', o);
32 for (; s && *s; s++) {
33 if (*s == '"' || *s == '\\') { fputc('\\', o); }
34 fputc(*s, o);
35 }
36 fputc('"', o);

86/* Capture a printf-emitting renderer by swapping the stdout stream (glibc). */
87static char *
88capture(void (*fn)(const doctor_ep *), const doctor_ep *e)
89{
90 char *buf = NULL;
91 size_t sz = 0;
92 FILE *ms = open_memstream(&buf, &sz);
93 FILE *save = stdout;
94 stdout = ms;
95 fn(e);
96 fflush(ms);
97 stdout = save;
98 fclose(ms);
99 return buf;
100}
