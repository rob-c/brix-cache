1/*
2 * resolve.c — the async resolve driver (phase-116 W2).
3 *
4 * WHAT: brix_dns_resolve(): IP-literal short-circuit, cache lookup, then
5 * either nginx's resolver (with resolv.conf search/ndots expansion and
6 * glibc-style attempts) or the thread-pool libc backend; answers are
7 * family-filtered, port-stamped, TTL-clamped and cached before the
8 * caller's handler runs. brix_dns_resolve_cancel() abandons a request
9 * whose owner is going away. Also the pool-free literal parser and
10 * numeric formatter every other layer shares.
11 * WHY: nginx's resolver knows nothing about search lists — `ngx_resolve_name`
12 * on "storage01" asks for the literal label and fails where `getent`
13 * succeeds. Driving the candidate list here gives every brix hostname
14 * the same semantics as the host resolver (I-DNS-3) while staying fully
15 * asynchronous (I-DNS-4). ngx_parse_addr() allocates from a pool, which
16 * a per-resolve caller cannot afford — hence the pool-free literal path.
17 * HOW: Candidates follow glibc: with >= ndots dots (or a trailing dot) the
18 * absolute name is tried first, then each search suffix; otherwise the
19 * search suffixes come first and the absolute name last. NXDOMAIN moves
20 * to the next candidate; TIMEDOUT/SERVFAIL retries the same candidate
21 * up to `attempts` times. Each hop is re-entered through a posted
22 * event so the next ngx_resolve_name() never runs inside the previous
23 * resolver callback. The in-flight nginx ctx is kept in req->rctx so a
24 * cancel can ngx_resolve_name_done() it; ngx_resolve_name() frees the
25 * ctx itself on NGX_ERROR and may run the handler inline on a resolver
26 * cache hit, so rctx is cleared inside the handler, never after the
27 * call. The handler is invoked exactly once per request.
28 */

96ngx_int_t
97brix_dns_parse_literal(const ngx_str_t *name, brix_dns_addr_t *out)
98{
99 u_char *p = name->data;
100 size_t len = name->len;
101 struct sockaddr_in *sin;
102 in_addr_t inaddr;
103#if (NGX_HAVE_INET6)
104 struct sockaddr_in6 *sin6;
105 struct in6_addr in6;
106#endif
107
108 if (len == 0) {
109 return NGX_DECLINED;
110 }
111 if (len > 2 && p[0] == '[' && p[len - 1] == ']') {
112 p++;
113 len -= 2;
114 }
115 ngx_memzero(out, sizeof(*out));
116 inaddr = ngx_inet_addr(p, len);
117 if (inaddr != INADDR_NONE) {
118 sin = (struct sockaddr_in *) &out->ss;
119 sin->sin_family = AF_INET;
120 sin->sin_addr.s_addr = inaddr;
121 out->len = sizeof(struct sockaddr_in);
122 return NGX_OK;
123 }
124#if (NGX_HAVE_INET6)
125 if (ngx_inet6_addr(p, len, in6.s6_addr) == NGX_OK) {
126 sin6 = (struct sockaddr_in6 *) &out->ss;
127 sin6->sin6_family = AF_INET6;
128 sin6->sin6_addr = in6;
129 out->len = sizeof(struct sockaddr_in6);
130 return NGX_OK;
131 }
132#endif
133 return NGX_DECLINED;
134}

157{

175static ngx_int_t
176dns_try_literal(brix_dns_req_t *req)
177{
178 brix_dns_addr_t a;
179
180 if (brix_dns_parse_literal(&req->name, &a) != NGX_OK) {
181 return NGX_DECLINED;
182 }
183 if ((req->af == BRIX_AF_INET && a.ss.ss_family != AF_INET)
184 || (req->af == BRIX_AF_INET6 && a.ss.ss_family != AF_INET6))
185 {
186 req->rc = NGX_ERROR;
187 req->error = "address family excluded by policy";
188 req->naddrs = 0;
189 } else {
190 req->addrs[0] = a;
191 req->naddrs = 1;
192 req->rc = NGX_OK;
193 }
194 req->literal = 1;
195 req->ttl = 0;
196 return NGX_OK;
197}

230static void
231dns_candidate_name(brix_dns_req_t *req, ngx_uint_t idx)
232{
233 const brix_resolv_conf_t *rc;
234 unsigned dots, absolute_first;
235 ngx_uint_t suffix;
236 size_t len = req->name.len;
237
238 if (len > 0 && req->name.data[len - 1] == '.') {
239 len--;
240 }
241 if (req->policy == NULL || !req->policy->search || idx >= req->ncand) {
242 ngx_memcpy(req->cbuf, req->name.data, len);
243 req->cname.data = req->cbuf;
244 req->cname.len = len;
245 return;
246 }
247 rc = &req->policy->rc;
248 dots = brix_resolv_conf_count_dots((const char *) req->name.data,
249 req->name.len);
250 absolute_first = (dots >= rc->ndots)
251 || (req->name.len > 0
252 && req->name.data[req->name.len - 1] == '.');
253
254 if ((absolute_first && idx == 0) || (!absolute_first && idx == rc->nsearch)) {
255 ngx_memcpy(req->cbuf, req->name.data, len);
256 req->cname.data = req->cbuf;
257 req->cname.len = len;
258 return;
259 }
260 suffix = absolute_first ? idx - 1 : idx;
261 req->cname.data = req->cbuf;
262 req->cname.len = ngx_sprintf(req->cbuf, "%*s.%s", len, req->name.data,
263 rc->search[suffix]) - req->cbuf;
264}

287static void
288dns_ngx_copy_answer(brix_dns_req_t *req, ngx_resolver_ctx_t *ctx)
289{
290 ngx_uint_t i, n = 0;
291 time_t ttl;
292
293 for (i = 0; i < ctx->naddrs && n < BRIX_DNS_MAX_ADDRS; i++) {
294 struct sockaddr *sa = ctx->addrs[i].sockaddr;
295
296 if ((req->af == BRIX_AF_INET && sa->sa_family != AF_INET)
297 || (req->af == BRIX_AF_INET6 && sa->sa_family != AF_INET6)
298 || ctx->addrs[i].socklen > sizeof(req->addrs[n].ss))
299 {
300 continue;
301 }
302 ngx_memcpy(&req->addrs[n].ss, sa, ctx->addrs[i].socklen);
303 req->addrs[n].len = ctx->addrs[i].socklen;
304 n++;
305 }
306 req->naddrs = n;
307 ttl = ctx->valid - ngx_time();
308 if (req->policy != NULL) {
309 if (ttl < req->policy->min_ttl) {
310 ttl = req->policy->min_ttl;
311 }
312 if (ttl > req->policy->max_ttl) {
313 ttl = req->policy->max_ttl;
314 }
315 } else if (ttl < BRIX_DNS_MIN_TTL_DEFAULT) {
316 ttl = BRIX_DNS_MIN_TTL_DEFAULT;
317 }
318 req->ttl = ttl;
319}

323static ngx_uint_t
324dns_ngx_result(ngx_int_t state, ngx_uint_t naddrs)
325{
326 if (state == NGX_OK) {
327 return naddrs > 0 ? BRIX_DNS_RESULT_OK : BRIX_DNS_RESULT_NXDOMAIN;
328 }
329 if (state == NGX_RESOLVE_NXDOMAIN) {
330 return BRIX_DNS_RESULT_NXDOMAIN;

333 return BRIX_DNS_RESULT_TIMEOUT;

339/* nginx resolver callback: decide, release the ctx, then re-enter via a
340 * posted event so the next ngx_resolve_name() runs outside this callback. */
341static void
342dns_ngx_handler(ngx_resolver_ctx_t *ctx)
343{
344 brix_dns_req_t *req = ctx->data;
345 ngx_int_t state = ctx->state;
346 ngx_uint_t attempts = req->policy ? req->policy->rc.attempts : 1;
347
348 req->rctx = NULL;
349 brix_dns_lookup_note(dns_ngx_result(state, ctx->naddrs));
350 if (state == NGX_OK && ctx->naddrs > 0) {
351 dns_ngx_copy_answer(req, ctx);
352 ngx_resolve_name_done(ctx);
353 if (req->naddrs == 0) {
354 req->rc = NGX_ERROR;
355 req->error = "no address in the requested family";
356 req->naddrs = 0;
357 brix_dns_req_finish(req);
358 return;
359 }
360 req->rc = NGX_OK;
361 req->error = NULL;
362 brix_dns_req_finish(req);
363 return;
364 }
365 ngx_resolve_name_done(ctx);
366
367 if ((state == NGX_RESOLVE_TIMEDOUT || state == NGX_RESOLVE_SERVFAIL)
368 && req->attempt + 1 < attempts)
369 {
370 req->attempt++; /* retry the same candidate */
371 } else if (req->cand + 1 < req->ncand) {
372 req->cand++; /* next search candidate */
373 req->attempt = 0;
374 } else {
375 dns_ngx_fail(req, state == NGX_OK ? NGX_RESOLVE_NXDOMAIN : state);
376 return;
377 }
378 req->ev.handler = dns_ngx_step;
379 req->ev.data = req;
380 req->ev.log = req->log;
381 ngx_post_event(&req->ev, &ngx_posted_events);
382}

385static void
386dns_ngx_step(ngx_event_t *ev)
387{
388 brix_dns_req_t *req = ev->data;
389 ngx_resolver_t *r = dns_usable_resolver(req);
390 ngx_resolver_ctx_t *ctx, temp;
391 ngx_msec_t timeout;
392
393 if (r == NULL) {
394 dns_ngx_fail(req, NGX_RESOLVE_TIMEDOUT);
395 return;
396 }
397 dns_candidate_name(req, req->cand);
398 temp.name = req->cname;
399 ctx = ngx_resolve_start(r, &temp);
400 if (ctx == NULL || ctx == NGX_NO_RESOLVER) {
401 dns_ngx_fail(req, NGX_RESOLVE_TIMEDOUT);
402 return;
403 }
404 timeout = (ngx_msec_t) (req->policy ? req->policy->rc.timeout : 5) * BRIX_CMS_SEC_TO_MS_MULTIPLIER;
405 if (timeout < BRIX_DNS_TIMEOUT_FLOOR_MS) {
406 timeout = BRIX_DNS_TIMEOUT_FLOOR_MS;
407 }
408 ctx->name = req->cname;
409 ctx->handler = dns_ngx_handler;
410 ctx->data = req;
411 ctx->timeout = timeout;
412 req->rctx = ctx;
413 if (ngx_resolve_name(ctx) != NGX_OK) {
414 /* nginx freed the ctx and did not run the handler */
415 req->rctx = NULL;
416 dns_ngx_fail(req, NGX_RESOLVE_TIMEDOUT);
417 }
418}

421ngx_int_t
422brix_dns_resolve(brix_dns_req_t *req)
423{
424 ngx_int_t rc;
425
426 req->rc = NGX_ERROR;
427 req->error = NULL;
428 req->naddrs = 0;
429 req->ttl = 0;
430 req->cached = 0;
431 req->literal = 0;
432 req->negative = 0;
433 req->cand = 0;
434 req->attempt = 0;
435 req->rctx = NULL;
436 req->task = NULL;
437 if (req->socktype == 0) {
438 req->socktype = SOCK_STREAM;
439 }
440 if (req->log == NULL) {
441 req->log = ngx_cycle->log;
442 }
443 if (req->name.len == 0 || req->name.len >= BRIX_RESOLV_DOMAIN_LEN) {
444 req->error = "empty or over-long hostname";
445 brix_dns_req_finish(req);
446 return NGX_OK;
447 }
448
449 if (dns_try_literal(req) == NGX_OK || brix_dns_cache_lookup(req) == NGX_OK) {
450 brix_dns_req_finish(req);
451 return NGX_OK;
452 }
453
454 if (dns_usable_resolver(req) != NULL) {
455 req->ncand = dns_candidate_count(req);
456 req->ev.handler = dns_ngx_step;
457 req->ev.data = req;
458 req->ev.log = req->log;
459 dns_ngx_step(&req->ev);
460 return NGX_OK;
461 }
462
463 rc = brix_dns_resolve_via_thread(req);
464 if (rc == NGX_OK) {
465 return NGX_OK;
466 }
467 if (rc == NGX_DECLINED) {
468 req->error = "no resolver and no thread pool available";
469 brix_dns_req_finish(req);
470 return NGX_OK;
471 }
472 return NGX_ERROR;
473}
