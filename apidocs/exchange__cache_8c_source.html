1/*
2 * exchange_cache.c — per-worker token-exchange result cache (see header).
3 *
4 * WHAT: Direct-mapped (subject-token, audience) → minted-token cache with a
5 * hard TTL clamp, for the phase-70 §5.4 EXCHANGE leg.
6 *
7 * WHY: Re-POSTing the RFC-8693 grant to the issuer on every VFS op is pure
8 * latency and issuer load; but a cache over credentials must (a) bind
9 * entries to the exact subject token so no principal can ever receive
10 * another's minted token, and (b) bound how long a revoked/rotated grant
11 * can keep being replayed. SHA-256 keying gives (a); the
12 * min(exp, +5 min) clamp gives (b).
13 *
14 * HOW: Mirrors worker_cache.c: one slot per hash bucket, colliding keys evict
15 * each other, no LRU bookkeeping, event-loop-only so lock-free. The
16 * minted token's `exp` is read by splitting the compact JWS and decoding
17 * its payload (no signature work — the issuer minted it over TLS a
18 * moment ago); an unparseable or already-expired mint is simply not
19 * cached. Tokens are secrets: nothing from a slot is ever logged.
20 */
21#include "exchange_cache.h"
22#include "b64url.h"
23#include "json.h"
24#include "core/compat/crypto.h" /* brix_sha256 */

26/* A minted token larger than this is served uncached rather than truncated. */
27#define BRIX_TX_CACHE_TOKEN_MAX 4096

29typedef struct {
30 u_char fp[32]; /* SHA-256(subject ‖ 0x0A ‖ aud); all-zero = empty */

62/* ---- tx_cache_slot_for -----------------------------------------------------
63 *
64 * WHAT: Hash (subject, aud) to the one slot that key can occupy, writing the
65 * 32-byte fingerprint to `fp`.
66 *
67 * WHY: The fingerprint covers the FULL subject token bytes — two different
68 * client tokens (even for the same principal) never share a slot hit —
69 * and a 0x0A separator keeps (s="a", aud="b") distinct from (s="ab").
70 * JWT text is base64url + dots, so 0x0A cannot occur inside either part.
71 *
72 * HOW: One incremental-free SHA-256 over a small stitched stack buffer would
73 * need subject-sized stack; instead hash the concatenation via two
74 * passes folded with the streaming-free helper: SHA-256(subject) is
75 * computed first, then SHA-256(that digest ‖ 0x0A ‖ aud). Collision
76 * resistance is inherited from SHA-256 composition. */
77static brix_tx_cache_slot_t *
78tx_cache_slot_for(brix_tx_cache_t *c, const ngx_str_t *subject,
79 const ngx_str_t *aud, u_char fp[32])
80{
81 u_char inner[32];
82 u_char buf[32 + 1 + 512];
83 size_t alen;
84 ngx_uint_t idx;
85
86 if (brix_sha256(subject->data, subject->len, inner) != 1) {
87 return NULL;
88 }
89
90 alen = (aud != NULL && aud->data != NULL) ? aud->len : 0;
91 if (alen > 512) {
92 return NULL; /* audiences are short URLs; refuse rather than clip */
93 }
94
95 ngx_memcpy(buf, inner, 32);
96 buf[32] = 0x0A;
97 if (alen > 0) {
98 ngx_memcpy(buf + 33, aud->data, alen);
99 }
100 if (brix_sha256(buf, 33 + alen, fp) != 1) {
101 return NULL;
102 }
103
104 idx = ((ngx_uint_t) fp[0] << 24 | (ngx_uint_t) fp[1] << 16
105 | (ngx_uint_t) fp[2] << 8 | (ngx_uint_t) fp[3]) % c->nslots;
106 return &c->slots[idx];
107}

109int
110brix_tx_cache_lookup(brix_tx_cache_t *cache, const ngx_str_t *subject,
111 const ngx_str_t *aud, time_t now, ngx_str_t *out)
112{
113 u_char fp[32];
114 brix_tx_cache_slot_t *slot;
115
116 if (cache == NULL || subject == NULL || subject->len == 0 || out == NULL) {
117 return 0;
118 }
119
120 slot = tx_cache_slot_for(cache, subject, aud, fp);
121 if (slot == NULL || slot->len == 0
122 || ngx_memcmp(slot->fp, fp, sizeof(fp)) != 0) {
123 return 0;
124 }
125 if (now >= slot->expire_at) {
126 slot->len = 0; /* dead entry — free the slot eagerly */
127 return 0;
128 }
129
130 out->data = slot->token;
131 out->len = slot->len;
132 return 1;
133}

135/* ---- tx_cache_minted_exp ---------------------------------------------------
136 *
137 * WHAT: Read the `exp` claim out of the freshly-minted compact JWS; 0 when the
138 * token cannot be parsed or carries no positive exp.
139 *
140 * WHY: The cache must never outlive the credential itself — `exp` is the
141 * issuer's own bound and the base the 5-minute clamp tightens. */
142static time_t
143tx_cache_minted_exp(const ngx_str_t *minted)
144{
145 xrdjwt_seg seg[3];
146 u_char payload[8192];
147 ssize_t plen;
148 int64_t exp = 0;
149
150 if (xrdjwt_split((const char *) minted->data, minted->len, seg) != 0) {
151 return 0;
152 }
153 plen = b64url_decode(seg[1].p, seg[1].n, payload, sizeof(payload));
154 if (plen <= 0) {
155 return 0;
156 }
157 if (json_get_int64((const char *) payload, (size_t) plen, "exp", &exp) != 0
158 || exp <= 0) {
159 return 0;
160 }
161 return (time_t) exp;
162}

164void
165brix_tx_cache_store(brix_tx_cache_t *cache, const ngx_str_t *subject,
166 const ngx_str_t *aud, const ngx_str_t *minted, time_t now)
167{
168 u_char fp[32];
169 time_t exp, cap;
170 brix_tx_cache_slot_t *slot;
171
172 if (cache == NULL || subject == NULL || subject->len == 0
173 || minted == NULL || minted->len == 0 || minted->data == NULL
174 || minted->len > BRIX_TX_CACHE_TOKEN_MAX) {
175 return;
176 }
177
178 exp = tx_cache_minted_exp(minted);
179 if (exp <= now) {
180 return; /* unparseable or already expired — never cache blind */
181 }
182
183 cap = now + BRIX_TX_CACHE_MAX_TTL_SECS;
184 slot = tx_cache_slot_for(cache, subject, aud, fp);
185 if (slot == NULL) {
186 return;
187 }
188
189 ngx_memcpy(slot->fp, fp, sizeof(fp));
190 ngx_memcpy(slot->token, minted->data, minted->len);
191 slot->token[minted->len] = '\0';
192 slot->len = minted->len;
193 slot->expire_at = (exp < cap) ? exp : cap;
194}
