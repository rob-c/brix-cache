25/* ---- File-local kXR_readv request working state ----
26 *
27 * WHAT: Bundles the per-request fields shared by brix_handle_readv's helper
28 * steps (config, wire view, segment count, per-element cap, and the two
29 * allocations once they exist). Not a wire structure — purely in-process.
30 *
31 * WHY: The validate / build-descriptors / async-post / sync-execute steps each

33 * counts within the gate and makes the data flow explicit.
34 *

36 * pointer to each step; steps that allocate (response_buffer, segment_descs)

129/* ---- Validate handles and compute the response upper bound ----
130 *
131 * WHAT: First pass over the wire segments: validates every file handle and
132 * accumulates req->max_response_bytes (capping each element at readv_seg_max).
133 * Returns 1 on success; returns 0 and stores the terminating status in
134 * *rc_out on a bad handle or an over-limit total.
135 *
136 * WHY: Allocation must happen only after the whole request is known sane; this
137 * isolates the sanity pass from the buffer setup and keeps the error triplet
138 * (metric + send_error) at a single named step.
139 *
140 * HOW:
141 * 1. For each segment decode the handle index and requested length.
142 * 2. Reject an invalid handle via brix_validate_read_handle (sets *rc_out).
143 * 3. Clamp the length to the per-element cap and add the framed size.
144 * 4. Reject the request if the running total exceeds BRIX_MAX_READV_TOTAL.
145 */
146static ngx_flag_t
147brix_readv_validate_and_size(brix_ctx_t *ctx, ngx_connection_t *c,
148 brix_readv_req_t *req, ngx_int_t *rc_out)
149{
150 size_t segment_index;
151
152 req->max_response_bytes = 0;
153 for (segment_index = 0; segment_index < req->segment_count;
154 segment_index++)
155 {
156 int handle_index;
157 uint32_t read_length;
158 ngx_int_t validate_rc;
159
160 handle_index =
161 (int) (unsigned char) req->wire_segments[segment_index].fhandle[0];
162 read_length =
163 (uint32_t) ntohl((uint32_t) req->wire_segments[segment_index].rlen);
164
165 if (!brix_validate_read_handle(ctx, c, handle_index, "READV",
166 BRIX_OP_READV, &validate_rc)) {
167 *rc_out = validate_rc;
168 return 0;
169 }
170
171 if ((size_t) read_length > req->readv_seg_max) {
172 read_length = (uint32_t) req->readv_seg_max;
173 }
174 req->max_response_bytes += BRIX_READV_SEGSIZE + read_length;
175
176 if (req->max_response_bytes > BRIX_MAX_READV_TOTAL) {
177 BRIX_OP_ERR(ctx, BRIX_OP_READV);
178 *rc_out = brix_send_error(ctx, c, kXR_ArgTooLong,
179 "readv total would exceed server limit");
180 return 0;
181 }
182 }
183
184 return brix_readv_validate_extents(ctx, c, req, rc_out);
185}

187/* ---- Build the response wire body and the segment descriptor array ----
188 *
189 * WHAT: Lays out the final kXR_readv response body in req->response_buffer and
190 * fills req->segment_descs so each descriptor's payload pointer names the
191 * exact place preadv() will land that segment's bytes. No return value; both
192 * buffers are pre-allocated by the orchestrator.
193 *
194 * WHY: The body layout and descriptor construction is a single pure pass with
195 * no I/O; isolating it keeps brix_handle_readv flat and the byte layout (the
196 * 4-byte fhandle, 4-byte length, 8-byte offset header per segment) reviewable
197 * in one place.
198 *
199 * HOW:
200 * 1. Walk a cursor over the response buffer, one framed segment at a time.
201 * 2. For each segment write the fhandle, clamped big-endian length, and raw
202 * offset into the leading header, then record fd/obj/offset/length and the
203 * header/payload pointers into the descriptor.
204 * 3. Advance the cursor past the header and clamped payload region.
205 */
206/* brix_readv_run_job — set up and run the READV VFS job for `req` into *job,
207 * routing any driver error text into err_msg[cap]. Returns job->io_errno (0 =
208 * success); the caller reads job->nio / job->out_size on success. Both the
209 * inline and the offloaded readv paths run their job through here. */
210static int
211brix_readv_run_job(brix_readv_req_t *req, brix_vfs_job_t *job,
212 char *err_msg, size_t cap)
213{
214 err_msg[0] = '\0';
215 ngx_memzero(job, sizeof(*job));
216 job->op = BRIX_VFS_IO_READV;
217 job->segs = req->segment_descs;
218 job->nsegs = req->segment_count;
219 job->err_msg = err_msg;
220 job->err_msg_cap = cap;
221 brix_vfs_io_execute(job);
222 return job->io_errno;
223}

226brix_readv_build_descriptors(brix_ctx_t *ctx, brix_readv_req_t *req)
227{
228 u_char *response_cursor = req->response_buffer;
229 size_t segment_index;
230
231 for (segment_index = 0; segment_index < req->segment_count;
232 segment_index++)
233 {
234 int handle_index;
235 uint32_t read_length;
236 uint32_t read_length_be;
237
238 handle_index =
239 (int) (unsigned char) req->wire_segments[segment_index].fhandle[0];
240 read_length = (uint32_t) ntohl(
241 (uint32_t) req->wire_segments[segment_index].rlen);
242
243 if ((size_t) read_length > req->readv_seg_max) {
244 read_length = (uint32_t) req->readv_seg_max;
245 }
246
247 ngx_memcpy(response_cursor,
248 req->wire_segments[segment_index].fhandle, 4);
249 read_length_be = htonl(read_length);
250 ngx_memcpy(response_cursor + 4, &read_length_be, 4);
251 ngx_memcpy(response_cursor + 8,
252 &req->wire_segments[segment_index].offset, 8);
253
254 req->segment_descs[segment_index].fd =
255 ctx->files[handle_index].fd;
256 req->segment_descs[segment_index].obj =
257 ctx->files[handle_index].sd_obj; /* Layer 3: driver or zeroed */
258 req->segment_descs[segment_index].handle_index = handle_index;
259 req->segment_descs[segment_index].offset = (off_t) (int64_t)
260 be64toh((uint64_t) req->wire_segments[segment_index].offset);
261 req->segment_descs[segment_index].read_length = read_length;
262 req->segment_descs[segment_index].header_read_length_ptr =
263 response_cursor + 4;
264 req->segment_descs[segment_index].payload_ptr =
265 response_cursor + BRIX_READV_SEGSIZE;
266
267 response_cursor += BRIX_READV_SEGSIZE + read_length;
268 }

271/* brix_readv_try_offload — §1.1 response offloading (do_Offload parity) for
272 * kXR_readv: the readv analog of brix_read_try_offload (read.c) — see there for
273 * the full safety rationale. When a readv carries a nonzero body[15] pathid whose
274 * bound secondary is on THIS worker AND quiescent, and the assembled response
275 * fits one streaming window, the whole response is built into a buffer owned by
276 * the SECONDARY (acquire+release both on its ctx → no cross-connection lifetime
277 * tangle) as one contiguous [8B kXR_ok header | assembled readv body] carrying
278 * the PRIMARY request's streamid, and queued on the secondary's out-ring.
279 * Returns 1 when routed to the secondary (*rc set), 0 to fall through to the
280 * normal primary-stream path (validate/size already ran; this only routes). */
281static ngx_flag_t
282brix_readv_try_offload(brix_ctx_t *ctx, ngx_connection_t *c,
283 brix_readv_req_t *req, ngx_int_t *rc)
284{
285 ngx_connection_t *sec_c;
286 brix_ctx_t *sec_ctx;
287 u_char *buf;
288 brix_vfs_job_t job;
289 char error_message[128];
290 size_t segment_index;
291
292 sec_ctx = brix_read_offload_secondary(ctx, c, req->pathid, &sec_c);
293 if (sec_ctx == NULL) {
294 return 0;
295 }
296
297 /* Single-frame only: a readv whose assembled response exceeds one streaming
298 * window belongs on the (budgeted) primary path. */
299 if (req->max_response_bytes > (size_t) BRIX_READ_WINDOW) {
300 return 0;
301 }
302
303 /* Frame buffer from the SECONDARY's pool: 8B header + the assembled body. */
304 buf = brix_acquire_read_buffer(sec_ctx, sec_c,
305 XRD_RESPONSE_HDR_LEN + req->max_response_bytes);
306 if (buf == NULL) {
307 return 0; /* secondary pool exhausted — fall back rather than fail */
308 }
309
310 req->segment_descs = brix_alloc_array(c->log, req->segment_count,
311 sizeof(brix_readv_seg_desc_t));
312 if (req->segment_descs == NULL) {
313 brix_release_read_buffer(sec_ctx, sec_c, buf);
314 return 0;
315 }
316
317 /* Lay the assembled readv body out AFTER the 8-byte response header. */
318 req->response_buffer = buf + XRD_RESPONSE_HDR_LEN;
319 brix_readv_build_descriptors(ctx, req);
320
321 if (brix_readv_run_job(req, &job, error_message, sizeof(error_message)) != 0) {
322 /* I/O failure: nothing has touched the secondary wire yet, so the error
323 * rides the PRIMARY control stream exactly like the normal path. */
324 ngx_free(req->segment_descs);
325 brix_release_read_buffer(sec_ctx, sec_c, buf);
326 BRIX_OP_ERR(ctx, BRIX_OP_READV);
327 *rc = brix_send_error(ctx, c, kXR_IOError,
328 error_message[0] ? error_message
329 : "readv I/O error");
330 return 1;
331 }
332
333 for (segment_index = 0; segment_index < req->segment_count;
334 segment_index++)
335 {
336 ctx->files[req->segment_descs[segment_index].handle_index].bytes_read +=
337 req->segment_descs[segment_index].read_length;
338 }
339 ngx_free(req->segment_descs);
340 ctx->totals.bytes += (size_t) job.nio;
341
342 if (req->rconf->access_log_fd != NGX_INVALID_FILE) {
343 char detail[64];
344 snprintf(detail, sizeof(detail), "%zu_segs", req->segment_count);
345 brix_log_access(ctx, c, "READV", "-", detail, 1, 0, NULL,
346 (size_t) job.nio);
347 }
348 BRIX_OP_OK(ctx, BRIX_OP_READV);
349
350 /* Stamp the kXR_ok header with the PRIMARY request's streamid, then queue the
351 * contiguous [hdr|body] frame on the SECONDARY's out-ring (it owns `buf`). */
352 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok, (uint32_t) job.out_size,
353 (ServerResponseHdr *) buf);
354 *rc = brix_queue_response_base(sec_ctx, sec_c, buf,
355 XRD_RESPONSE_HDR_LEN + job.out_size, buf);
356 if (*rc == NGX_OK) {
357 brix_metric_offload(BRIX_PROTO_ROOT); /* §1.1 observability */
358 }
359
360 /* Release on error OR full inline send (ring stayed empty); a parked-and-
361 * draining response keeps the buffer until its slot drains. Idempotent. */
362 if (*rc != NGX_OK || sec_ctx->out.count == 0) {
363 brix_release_read_buffer(sec_ctx, sec_c, buf);
364 }
365 return 1;
366}

368/* Handle kXR_readv — validate the request (payload a multiple of the readv
369 * element size, segment lengths within bounds), read all segments, and assemble
370 * the chained response. */
371ngx_int_t
372brix_handle_readv(brix_ctx_t *ctx, ngx_connection_t *c)
373{
374 /* phase-42 W4 invariant: readv is ALWAYS plaintext — it never consults
375 * read_codec. Inline read compression applies only to single-segment
376 * kXR_read; readv's interleaved per-segment framing (and its pgread-style
377 * integrity expectations) stay byte-for-byte intact. */
378 brix_readv_req_t req;
379 ngx_int_t rc;
380
381 ngx_memzero(&req, sizeof(req));
382
383 if (ctx->recv.payload == NULL || ctx->recv.cur_dlen == 0 ||
384 (ctx->recv.cur_dlen % BRIX_READV_SEGSIZE) != 0)
385 {
386 BRIX_OP_ERR(ctx, BRIX_OP_READV);
387 return brix_send_error(ctx, c, kXR_ArgInvalid,
388 "malformed readv request");
389 }
390
391 req.wire_segments = (readahead_list *) ctx->recv.payload;
392 req.segment_count = ctx->recv.cur_dlen / BRIX_READV_SEGSIZE;
393
394 /* Phase 27 W2/F1: explicit segment-count cap at the callsite (defense in
395 * depth over the recv-layer payload cap). */
396 if (req.segment_count == 0 || req.segment_count > BRIX_READV_MAXSEGS) {
397 BRIX_OP_ERR(ctx, BRIX_OP_READV);
398 return brix_send_error(ctx, c, kXR_ArgTooLong,
399 "readv segment count exceeds server limit");
400 }
401
402 /*
403 * §1.1: kXR_readv carries its response-offload pathid in the request HEADER
404 * body (byte 15), NOT the payload (which is entirely the read_list). Validate
405 * it like read/pgread — a nonzero pathid MUST name a live kXR_bind path of
406 * this session, else kXR_ArgInvalid "invalid path ID". readv previously
407 * ignored the pathid outright, an inconsistency vs read/pgread; the routing
408 * itself happens in brix_readv_try_offload once the request is sized.
409 */
410 req.pathid = (unsigned) ((ClientRequestHdr *) ctx->recv.hdr_buf)->body[15];
411 if (req.pathid != 0
412 && !brix_session_pathid_bound(ctx->is_bound ? ctx->bound_sessid
413 : ctx->login.sessid,
414 req.pathid))
415 {
416 BRIX_OP_ERR(ctx, BRIX_OP_READV);
417 return brix_send_error(ctx, c, kXR_ArgInvalid, "invalid path ID");
418 }
419
420 req.rconf = ngx_stream_get_module_srv_conf(
421 (ngx_stream_session_t *) c->data, ngx_stream_brix_module);
422
423 /* Per-element read cap = the configured readv segment size (official
424 * maxReadv_ior, advertised via Qconfig readv_ior_max). An element requesting
425 * more is served SHORT (capped): our native client reads each segment's
426 * actual returned length (see test_readv_variable_blocks). Well-behaved
427 * clients honour the advertised readv_ior_max and never request more; a
428 * client that ignores it and cannot handle a short element (e.g. XrdCl's
429 * VectorRead) must therefore size its elements within the cap. */
430 req.readv_seg_max = req.rconf->readv_segment_size;
431
432 /*
433 * First pass validates every file handle and computes the upper bound for
434 * the single response scratch buffer. Allocation happens only after the
435 * whole request is known to be sane.
436 */
437 if (!brix_readv_validate_and_size(ctx, c, &req, &rc)) {
438 return rc;
439 }
440
441 /*
442 * §1.1 response offloading: route an eligible readv response over the bound
443 * secondary channel (validate_and_size has run, so handles are good and
444 * max_response_bytes is known). Ineligible requests (pathid 0, cross-worker,
445 * busy secondary, large) fall through to the primary-stream path below.
446 */
447 if (brix_readv_try_offload(ctx, c, &req, &rc)) {
448 return rc;
449 }
450
451 brix_prefetch_readv_segments(ctx, c, req.wire_segments, req.segment_count,
452 req.readv_seg_max);
453
454 /* The primary response is one protocol body produced through a bounded
455 * scratch window. Segment headers remain byte-exact, which stock XrdCl
456 * requires when matching returned ranges to its destination buffers. */
457 return brix_readv_serve_windowed(ctx, c, req.rconf, req.wire_segments,
458 req.segment_count,
459 req.max_response_bytes);
460}
