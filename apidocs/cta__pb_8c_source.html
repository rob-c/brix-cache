61/* Copy a length-delimited protobuf string into a fixed buffer (truncate, NUL). */

63copy_str(char *dst, size_t dst_sz, const unsigned char *src, size_t n)
64{
65 if (n >= dst_sz) {
66 n = dst_sz - 1;
67 }
68 memcpy(dst, src, n);

72/* ---- table-driven request decoder ---------------------------------------
73 * Every message in the pinned table is the same flat shape — match a field
74 * number, else skip — so a single generic loop walks one row table; the
75 * per-message differences live entirely in pbf_rows[]. */
76
77typedef enum {
78 PBF_STR, /* LEN → copy_str into the cta_request_t buffer at off */
79 PBF_MSG, /* LEN → recurse into the rows of message `sub` */
80 PBF_U64, /* VARINT → raw value into the uint64_t at off */
81 PBF_EVENT, /* VARINT → map cta.eos.Workflow.EventType onto out->op */
82 PBF_QUERY, /* LEN → consume; mark the request as CTA_OP_QUERY */
83} pbf_kind_t;
84
85enum {
86 M_REQUEST, M_NOTIF, M_WF, M_CLIENT, M_META, M_SERVICE, M_REQUESTER,
87};

130/* Consume the value of matched row `i` from the reader into `out`. */
131static int
132decode_field(pb_reader *r, size_t i, cta_request_t *out)
133{
134 const unsigned char *d;
135 size_t n;
136 uint64_t ev;
137 pb_reader sub;
138
139 switch (pbf_rows[i].kind) {
140
141 case PBF_EVENT:
142 if (pb_read_varint(r, &ev) != 0) {
143 return -1;
144 }
145 out->op = ev == EV_CLOSEW ? CTA_OP_ARCHIVE
146 : ev == EV_PREPARE ? CTA_OP_RETRIEVE
147 : ev == EV_ABORT_PREPARE ? CTA_OP_CANCEL
148 : CTA_OP_UNKNOWN;
149 return 0;
150
151 case PBF_U64:
152 return pb_read_varint(r,
153 (uint64_t *) ((char *) out + pbf_rows[i].off));
154
155 default:
156 break;
157 }
158
159 if (pb_read_len_delim(r, &d, &n) != 0) {
160 return -1;
161 }
162
163 switch (pbf_rows[i].kind) {
164
165 case PBF_STR:
166 copy_str((char *) out + pbf_rows[i].off, pbf_rows[i].sz, d, n);
167 return 0;
168
169 case PBF_MSG:
170 sub.p = d; sub.end = d + n;
171 return decode_msg(&sub, pbf_rows[i].sub, out);
172
173 default:
174 /* PBF_QUERY — admin command (query/listing). The full AdminCmd parse
175 * is deferred; for routing it is enough to mark the op. */
176 out->op = CTA_OP_QUERY;
177 return 0;
178 }
179}

181/* Decode message `msg` (M_*): dispatch known fields, skip the rest. */
182static int
183decode_msg(pb_reader *r, unsigned char msg, cta_request_t *out)
184{
185 uint32_t f; int wt;
186
187 while (r->p < r->end) {
188 size_t i;
189
190 if (pb_read_tag(r, &f, &wt) != 0) {
191 return -1;
192 }
193 for (i = 0; i < PBF_NROWS; i++) {
194 if (pbf_rows[i].msg == msg && pbf_rows[i].field == f
195 && wt == pbf_wire_type(pbf_rows[i].kind))
196 {
197 break;
198 }
199 }
200 if (i == PBF_NROWS) {
201 if (pb_skip_field(r, wt) != 0) {
202 return -1;
203 }
204 } else if (decode_field(r, i, out) != 0) {
205 return -1;
206 }
207 }
208 return 0;
209}

211int
212cta_pb_decode_request(const unsigned char *buf, size_t len, cta_request_t *out)
213{
214 pb_reader r;
215
216 memset(out, 0, sizeof(*out));
217 out->op = CTA_OP_UNKNOWN;
218 r.p = buf; r.end = buf + len;
219 return decode_msg(&r, M_REQUEST, out);
220}

222int
223cta_pb_encode_response(cta_rsp_type_t type, const char *message_txt,
224 uint64_t archive_id,
225 unsigned char *out, size_t cap, size_t *out_len)
226{
227 pb_writer w = { out, 0, cap };
228
229 if (pb_write_varint_field(&w, F_RSP_TYPE, (uint64_t) type) != 0) {
230 return -1;
231 }
232 if (message_txt != NULL && message_txt[0] != '\0') {
233 if (pb_write_string(&w, F_RSP_MESSAGE_TXT, message_txt) != 0) {
234 return -1;
235 }
236 }
237 if (archive_id != 0) {
238 /* Response.archive_file_id is a string xattr; render the number. */
239 char idbuf[24];
240 int m = 0;
241 uint64_t v = archive_id;
242 char rev[24];
243 int k = 0;
244 while (v > 0) { rev[k++] = (char) ('0' + (v % 10)); v /= 10; }
245 while (k > 0) { idbuf[m++] = rev[--k]; }
246 idbuf[m] = '\0';
247 if (pb_write_string(&w, F_RSP_ARCHIVE_ID, idbuf) != 0) {
248 return -1;
249 }
250 }
251 *out_len = w.len;
252 return 0;
253}
