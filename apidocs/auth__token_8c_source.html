31/*
32 * webdav_check_token_scope — enforce WLCG/SciToken read or write scope for
33 * any WebDAV data-access method (GET, HEAD, PROPFIND, PUT, DELETE, MKCOL, …).
34 *
35 * If the request was authenticated via a bearer token (rctx->token_auth == 1),
36 * checks whether the token's scope covers the request URI path for the
37 * operation class (read vs write) derived from the HTTP method.
38 * Returns NGX_OK if the scope check passes or if auth was not token-based.
39 * Returns NGX_HTTP_FORBIDDEN if the token lacks scope for the URI.
40 *
41 * NOTE: scope is checked against the raw decoded URI path, not the filesystem
42 * path — the path-prefix invariant is enforced by the scope matching code in
43 * token/scopes.c (must be an exact prefix, not a partial directory name match).
44 */
45/*
46 *
47 * WHAT: Enforces WLCG/SciToken read or write scope authorization for WebDAV
48 * data-access methods. Derives the required op class (read vs write) from the
49 * HTTP method via webdav_token_op_class(), then checks whether the
50 * authenticated bearer token's scopes cover the request URI path. Returns
51 * NGX_OK if the scope check passes or if authentication was not token-based
52 * (e.g., GSI cert auth has no equivalent scope concept). Returns
53 * NGX_HTTP_FORBIDDEN when the token lacks sufficient permission for the URI.
54 *
55 * WHY: WLCG/SciToken grants fine-grained path-based access rights rather than
56 * binary allow/deny. A token might grant read-only access to /data/atlas but
57 * write access to /data/cms — this function prevents both cross-VO file reads
58 * and cross-VO file mutation by enforcing scope on every data-access method,
59 * not just writes. The raw URI path check (not filesystem path) is intentional
60 * because scope granularity must match the client-facing namespace, not the
61 * underlying storage layout.
62 *
63 * HOW: Retrieves request context and verifies token_auth flag is set; derives
64 * need_write from webdav_token_op_class(); copies r->uri into a
65 * null-terminated buffer for scope checking; calls either
66 * brix_token_check_write() or brix_token_check_read() (or the identity
67 * wrapper) to verify the URI path is covered by a matching scope prefix; logs
68 * warning and returns 403 if no matching scope found. */
69ngx_int_t
70webdav_check_token_scope(ngx_http_request_t *r, const char *method_name)
71{
72 ngx_http_brix_webdav_req_ctx_t *rctx;
73 char uri_path[WEBDAV_MAX_PATH];
74 size_t ulen;
75 int need_write;
76
77 rctx = ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
78 if (rctx == NULL || !rctx->token_auth) {
79 return NGX_OK;
80 }
81
82 ulen = r->uri.len < sizeof(uri_path) - 1
83 ? r->uri.len : sizeof(uri_path) - 1;
84 ngx_memcpy(uri_path, r->uri.data, ulen);
85 uri_path[ulen] = '\0';
86
87 need_write = (webdav_token_op_class(r) == BRIX_TOKEN_OP_WRITE);
88
89 if (rctx->identity != NULL) {
90 if (brix_identity_check_token_scope(rctx->identity, uri_path,
91 need_write) == NGX_OK)
92 {
93 return NGX_OK;
94 }
95 } else if (need_write
96 ? brix_token_check_write(rctx->token_scopes,
97 rctx->token_scope_count, uri_path)
98 : brix_token_check_read(rctx->token_scopes,
99 rctx->token_scope_count, uri_path))
100 {
101 return NGX_OK;
102 }
103
104 ngx_log_error(NGX_LOG_WARN, r->connection->log, 0,
105 "brix_webdav: token scope denies %s %s to \"%s\"",
106 method_name, need_write ? "write" : "read", uri_path);
107
108 return NGX_HTTP_FORBIDDEN;
109}

115/*
116 * webdav_bearer_from_query — extract a bearer token from ?authz= / ?access_token=.
117 *
118 * WHAT: query-string fallback used only when no Authorization header carries a
119 * Bearer token (davix/gfal2/xrdcp redirect + pre-signed-URL flows).
120 * WHY: XrdHttp accepts the token in the URL; matching it is required for WLCG
121 * client interop. The header path stays primary.
122 * HOW: copies the raw arg into r->pool, URL-decodes in place, strips an optional
123 * case-insensitive "Bearer " prefix, and enforces a length cap. On NGX_OK
124 * *out is a NUL-terminated pool slice. NGX_DECLINED when disabled/absent.
125 */
126static ngx_int_t
127webdav_bearer_from_query(ngx_http_request_t *r,
128 ngx_http_brix_webdav_loc_conf_t *conf, ngx_str_t *out)
129{
130 ngx_str_t raw;
131 size_t len;
132
133 if (!conf->http_query_token) {
134 return NGX_DECLINED;
135 }
136 if (brix_http_arg(r, "authz", 5, &raw) != NGX_OK
137 && brix_http_arg(r, "access_token", 12, &raw) != NGX_OK) {
138 return NGX_DECLINED;
139 }
140 len = brix_urldecode_inplace((char *) raw.data);
141 if (len >= 7 && ngx_strncasecmp(raw.data, (u_char *) "Bearer ", 7) == 0) {
142 raw.data += 7;
143 len -= 7;
144 while (len > 0 && raw.data[0] == ' ') { raw.data++; len--; }
145 }
146 if (len == 0 || len > WEBDAV_QUERY_TOKEN_MAX) {
147 return NGX_DECLINED;
148 }
149 out->data = raw.data;
150 out->len = len;
151 return NGX_OK;
152}

154/*
155 * wt_ensure_ctx — get-or-create the WebDAV request context with an identity.
156 *
157 * WHAT: Retrieves the per-request module context, allocating a fresh one (with
158 * an attached identity) on r->pool when absent, and back-fills an identity onto
159 * a pre-existing context that lacks one. Returns the context via *out.
160 * WHY: bearer-token auth runs after other auth phases may (or may not) have
161 * created the context; a NULL identity would crash the later claims-store step,
162 * so both the create and the repair paths must guarantee one. Factored out so
163 * the orchestrator's allocation-failure branches don't inflate its complexity.
164 * HOW: ngx_http_get_module_ctx(); on NULL allocate+attach identity+set_ctx; on a
165 * context with a NULL identity allocate one; map every OOM to
166 * NGX_HTTP_INTERNAL_SERVER_ERROR. Returns NGX_OK with *out populated otherwise.
167 */
168static ngx_int_t
169wt_ensure_ctx(ngx_http_request_t *r, ngx_http_brix_webdav_req_ctx_t **out)
170{
171 ngx_http_brix_webdav_req_ctx_t *ctx;
172
173 ctx = ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
174 if (ctx == NULL) {
175 ctx = ngx_pcalloc(r->pool, sizeof(*ctx));
176 if (ctx == NULL) {
177 return NGX_HTTP_INTERNAL_SERVER_ERROR;
178 }
179 ctx->identity = brix_identity_alloc(r->pool);
180 if (ctx->identity == NULL) {
181 return NGX_HTTP_INTERNAL_SERVER_ERROR;
182 }
183 ngx_http_set_ctx(r, ctx, ngx_http_brix_webdav_module);
184 } else if (ctx->identity == NULL) {
185 ctx->identity = brix_identity_alloc(r->pool);
186 if (ctx->identity == NULL) {
187 return NGX_HTTP_INTERNAL_SERVER_ERROR;
188 }
189 }
190
191 *out = ctx;
192 return NGX_OK;
193}

215/*
216 * webdav_add_nostore — attach Cache-Control: no-store to the response.
217 *
218 * WHAT: pushes a Cache-Control: no-store header onto headers_out.
219 * WHY: RFC 6750 §2.3 (SEC MUST) — responses to a query-transported token MUST
220 * carry no-store so the URL (with its embedded token) is never cached by an
221 * intermediary. Applies to the response regardless of the auth outcome.
222 * HOW: best-effort ngx_list_push; a push failure is non-fatal (the response
223 * still goes out, just without the advisory header).
224 */
225static void
226webdav_add_nostore(ngx_http_request_t *r)
227{
228 ngx_table_elt_t *h = ngx_list_push(&r->headers_out.headers);
229
230 if (h == NULL) {
231 return;
232 }
233 h->hash = 1;
234#if (nginx_version >= 1023000)
235 h->next = NULL;
236#endif
237 ngx_str_set(&h->key, "Cache-Control");
238 ngx_str_set(&h->value, "no-store");
239}

241/*
242 * wt_parse_header — obtain the presented bearer token and scrub it from logs.
243 *
244 * WHAT: Resolves the request's bearer token from the Authorization header,
245 * falling back to the ?authz=/?access_token= query parameter, enforces the
246 * RFC 6750 §2 single-transport MUST, and — once the token has been consumed for
247 * auth — length-preservingly redacts any URL-borne token from every loggable
248 * request field. Returns the token via *bearer.
249 * WHY: WLCG clients present the token either in the header (primary) or the URL
250 * (davix/gfal2 redirect + pre-signed flows); both must be accepted with the same
251 * precedence XrdHttp uses, but NOT both at once (§2 SEC MUST — a header+query
252 * collision is a confused-deputy vector → 400 invalid_request). A URL token
253 * must never reach access/error logs, so the scrub happens here, immediately
254 * after extraction, before any early return that could log the request line.
255 * HOW: no header → query fallback (NGX_DECLINED when absent); header present but
256 * not Bearer → query fallback; a malformed Bearer header → NGX_HTTP_UNAUTHORIZED;
257 * a header Bearer token together with a query token → NGX_HTTP_BAD_REQUEST; on a
258 * query-sourced success attach Cache-Control: no-store (§2.3); always redact.
259 */
260static ngx_int_t
261wt_parse_header(ngx_http_request_t *r,
262 ngx_http_brix_webdav_loc_conf_t *conf, ngx_str_t *bearer)
263{
264 ngx_str_t auth_hdr;
265 ngx_str_t qtok;
266 int from_query = 0;
267 int header_bearer = 0;
268 int rc;
269
270 if (r->headers_in.authorization == NULL) {
271 /* No Authorization header — try the ?authz= query fallback (§1). */
272 if (webdav_bearer_from_query(r, conf, bearer) != NGX_OK) {
273 return NGX_DECLINED;
274 }
275 from_query = 1;
276 } else {
277 auth_hdr = r->headers_in.authorization->value;
278 rc = brix_http_extract_bearer(&auth_hdr, bearer);
279 if (rc == NGX_DECLINED) {
280 /* Header present but not Bearer — still allow the query fallback. */
281 if (webdav_bearer_from_query(r, conf, bearer) != NGX_OK) {
282 return NGX_DECLINED;
283 }
284 from_query = 1;
285 } else if (rc != NGX_OK) {
286 return NGX_HTTP_UNAUTHORIZED;
287 } else {
288 header_bearer = 1;
289 }
290 }
291
292 /* RFC 6750 §2 (SEC MUST): a token MUST NOT be transmitted via more than one
293 * method. A Bearer Authorization header together with a ?authz=/?access_token=
294 * query token is a dual-transport request → 400 invalid_request. Redact the
295 * URL token before returning so the reject path never leaks it. */
296 if (header_bearer
297 && webdav_bearer_from_query(r, conf, &qtok) == NGX_OK)
298 {
299 wt_redact_query_token(r);
300 return NGX_HTTP_BAD_REQUEST;
301 }
302
303 /* §1: the token has now been consumed for auth — scrub any URL-borne token
304 * from every log source (length-preserving). */
305 wt_redact_query_token(r);
306
307 /* §2.3 (SEC MUST): a query-transported token requires Cache-Control:
308 * no-store on the response, set now since it applies regardless of the
309 * eventual auth outcome. */
310 if (from_query) {
311 webdav_add_nostore(r);
312 }
313
314 return NGX_OK;
315}

317static ngx_int_t
318wt_check_claims(ngx_http_request_t *r,
319 ngx_http_brix_webdav_req_ctx_t *ctx,
320 const char *token, size_t token_len,
321 brix_token_claims_t *claims)
322{
323 int i;
324
325 ctx->verified = 1;
326 ctx->token_auth = 1;
327 ctx->auth_source = "token";
328
329 /* Phase-70 §5.4: retain the raw JWT bytes for backend PASSTHROUGH. Copied
330 * onto r->pool (the wire `token`/`bearer.data` may point into a header value
331 * that later rewrites, e.g. query-token redaction) and never logged. */
332 ctx->bearer_token.data = ngx_pnalloc(r->pool, token_len);
333 if (ctx->bearer_token.data == NULL) {
334 return NGX_HTTP_INTERNAL_SERVER_ERROR;
335 }
336 ngx_memcpy(ctx->bearer_token.data, token, token_len);
337 ctx->bearer_token.len = token_len;
338
339 if (brix_identity_set_token_claims(ctx->identity, r->pool, claims)
340 != NGX_OK)
341 {
342 return NGX_HTTP_INTERNAL_SERVER_ERROR;
343 }
344 ngx_cpystrn((u_char *) ctx->dn, (u_char *) claims->sub, sizeof(ctx->dn));
345
346 ctx->token_scope_count = claims->scope_count;
347 for (i = 0; i < claims->scope_count && i < BRIX_MAX_TOKEN_SCOPES; i++) {
348 ctx->token_scopes[i] = claims->scopes[i];
349 }
350
351 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
352 "brix_webdav: token auth OK sub=\"%s\" scopes=%d",
353 claims->sub, claims->scope_count);
354
355 return NGX_OK;
356}

358/*
359 *
360 * WHAT: Validates WLCG/SciToken bearer tokens presented in HTTP Authorization headers using either JWKS-based JWT verification or macaroon secret-key validation. Extracts token claims (subject, scopes, expiration) and stores them in the request context for downstream operations. Supports grace-period key rotation — if a macaroon is rejected by the current secret but accepted by an old secret configured via conf->common.token_macaroon_secret_old, the token is still accepted with an informational log message indicating graceful migration during nginx -s reload.
361 *
362 * WHY: WebDAV clients authenticate using bearer tokens rather than GSI certificates or anonymous access. This function must handle both JWT (via JWKS key set) and macaroon formats since different WLCG sites use different token types. The grace-period fallback prevents immediate access disruption during secret key rotation — in-flight tokens should be accepted until they naturally expire, avoiding a "hard break" scenario where all active clients are suddenly denied after a config reload.
363 *
364 * HOW: Declines if no keys/secrets configured; parses macaroon secret for validation if present; creates or retrieves request context (declines if already token-authenticated to avoid redundant verification); extracts a Bearer token from Authorization with shared case-insensitive scheme parsing via wt_parse_header(); resolves validity via wt_check_issuer_keys() (caches → registry/JWKS → grace retry); caches freshly verified non-registry claims; on success stores claims (sub, scopes) in ctx via wt_check_claims() for downstream scope checks. */
365ngx_int_t
366webdav_verify_bearer_token(ngx_http_request_t *r,
367 ngx_http_brix_webdav_loc_conf_t *conf)
368{
369 ngx_http_brix_webdav_req_ctx_t *ctx = NULL;
370 brix_token_claims_t claims;
371 ngx_str_t bearer;
372 const char *token;
373 size_t token_len;
374 int rc;
375 int cache_hit = 0;
376 int via_registry = 0;
377 ngx_int_t crc;
378
379 u_char secret[64];
380 ssize_t slen = 0;
381
382 if (conf->jwks_key_count <= 0 && conf->common.token_macaroon_secret.len == 0
383 && conf->token_registry == NULL)
384 {
385 return NGX_DECLINED;
386 }
387
388 if (conf->common.token_macaroon_secret.len) {
389 slen = brix_macaroon_secret_parse(
390 (const char *) conf->common.token_macaroon_secret.data,
391 conf->common.token_macaroon_secret.len, secret, sizeof(secret));
392 }
393
394 crc = wt_ensure_ctx(r, &ctx);
395 if (crc != NGX_OK) {
396 return crc;
397 }
398
399 if (ctx->token_auth) {
400 return NGX_OK;
401 }
402
403 crc = wt_parse_header(r, conf, &bearer);
404 if (crc != NGX_OK) {
405 return crc;
406 }
407
408 token = (const char *) bearer.data;
409 token_len = bearer.len;
410
411 {
412 wt_validate_ctx_t v;
413
414 v.r = r;
415 v.conf = conf;
416 v.token = token;
417 v.token_len = token_len;
418 v.secret = secret;
419 v.slen = slen;
420 v.claims = &claims;
421
422 rc = wt_check_issuer_keys(&v, &cache_hit, &via_registry);
423 }
424
425 if (rc != 0) {
426 ngx_log_error(NGX_LOG_WARN, r->connection->log, 0,
427 "brix_webdav: bearer token validation failed");
428 return NGX_HTTP_UNAUTHORIZED;
429 }
430
431 /* Cache the freshly verified claims for subsequent presentations (L1 always,
432 * L2 when a SHM zone is configured). Registry-authorized tokens are never
433 * cached: the decision is path-dependent and the cache is token-keyed. */
434 if (!cache_hit && !via_registry) {
435 brix_token_l1_store(conf->token_l1, token, token_len, &claims);
436 if (conf->common.token_cache_kv != NULL) {
437 brix_token_cache_store(conf->common.token_cache_kv, token,
438 token_len, &claims);
439 }
440 }
441
442 return wt_check_claims(r, ctx, token, token_len, &claims);
443}
