2 * offload_registry.c — per-worker (sessid, pathid) -> secondary-connection map
3 * for pathid response offloading (audit §1.1/§7.3). See offload_registry.h.
4 *
5 * Pure C (no nginx dependency): the connection is an opaque `void *`, so the
6 * table is unit-testable with plain pointers. One nginx worker runs a single-
7 * threaded event loop, so no locking is needed.

23static int
24slot_matches(const brix_offload_slot_t *s, const unsigned char *sessid,
25 unsigned pathid)
26{
27 return s->conn != NULL
28 && s->pathid == pathid
29 && memcmp(s->sessid, sessid, BRIX_SESSION_ID_LEN) == 0;
30}

32int
33brix_offload_register(const unsigned char *sessid, unsigned pathid, void *conn)
34{
35 size_t i;
36 int free_idx = -1;
37
38 if (sessid == NULL || conn == NULL) {
39 return 0;
40 }
41
42 /* Re-point an existing (sessid, pathid) in place; remember the first free
43 * slot for the insert case in the same pass. */
44 for (i = 0; i < BRIX_OFFLOAD_MAX; i++) {
45 if (s_binds[i].conn == NULL) {
46 if (free_idx < 0) {
47 free_idx = (int) i;
48 }
49 continue;
50 }
51 if (slot_matches(&s_binds[i], sessid, pathid)) {
52 s_binds[i].conn = conn;
53 return 1;
54 }
55 }
56
57 if (free_idx < 0) {
58 return 0; /* table full — no offload slot */
59 }
60
61 memcpy(s_binds[free_idx].sessid, sessid, BRIX_SESSION_ID_LEN);
62 s_binds[free_idx].pathid = pathid;
63 s_binds[free_idx].conn = conn;
64 s_count++;
65 return 1;
66}

68void *
69brix_offload_lookup(const unsigned char *sessid, unsigned pathid)
70{
71 size_t i;
72
73 if (sessid == NULL || pathid == 0) {
74 return NULL; /* pathid 0 = the primary */
75 }
76 for (i = 0; i < BRIX_OFFLOAD_MAX; i++) {
77 if (slot_matches(&s_binds[i], sessid, pathid)) {
78 return s_binds[i].conn;
79 }
80 }
81 return NULL;
82}

84void
85brix_offload_unregister(void *conn)
86{
87 size_t i;
88
89 if (conn == NULL) {
90 return;
91 }
92 for (i = 0; i < BRIX_OFFLOAD_MAX; i++) {
93 if (s_binds[i].conn == conn) {
94 s_binds[i].conn = NULL;
95 s_binds[i].pathid = 0;
96 memset(s_binds[i].sessid, 0, BRIX_SESSION_ID_LEN);
97 if (s_count > 0) {
98 s_count--;
99 }
100 }
101 }
102}

111brix_offload_foreach(int (*cb)(void *ud, const unsigned char *sessid,

126 }
