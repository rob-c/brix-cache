1/*
2 * rpm_gate.c — method policing, path classification, cache-key derivation.
3 *
4 * WHAT: the first step of every request on a brix_rpm_mirror location.
5 * Refuses write-class methods (a pull-through mirror is GET/HEAD only),
6 * classifies the URI through the shared grammar kernel, and copies the
7 * URI into the request's canonical cache key.
8 * WHY: the classifier verdict is what every later decision reads — the
9 * freshness window (mutable vs immutable), the metric row, and whether
10 * the cache tier's rpm-repodata verify has a digest to check — so it
11 * must be computed exactly once, at the edge, on the only bytes the
12 * client controls. Refusing a write here is the same discipline the OCI
13 * mirror applies: a PUT that quietly 200s at a mirror is an operator
14 * believing a package was published that nothing holds.
15 * HOW: an early-return ladder. NGX_DECLINED means "a routable object" and is
16 * the only non-terminal verdict; everything else is an HTTP status the
17 * caller returns unchanged (the core paints the body).
18 */

25/* A write-class method aimed at a read-only mirror: the spec-shaped 405 +
26 * Allow for the client, and one audit line for the operator's jail. */
27static ngx_int_t
28rpm_refuse_write(ngx_http_request_t *r, ngx_http_brix_rpm_ctx_t *ctx)
29{
30 ctx->disp = BRIX_RPM_OUT_REFUSED;
31
32 brix_http_guard_audit(r, "rpm", GUARD_R_RPMWRITE, GUARD_OP_WRITE,
33 NGX_HTTP_NOT_ALLOWED);
34
35 if (brix_http_set_header(r, "Allow", "GET, HEAD", NULL) != NGX_OK) {
36 return NGX_HTTP_INTERNAL_SERVER_ERROR;
37 }
38 return NGX_HTTP_NOT_ALLOWED;
39}

42/* Classifier refusal → HTTP status. A path this grammar will not accept is a
43 * malformed request, not a missing file: answering 404 would tell a traversal
44 * probe that its shape was legal and only the target was absent. */
45static ngx_uint_t
46rpm_bad_status(brix_rpm_err_t err)
47{
48 return (err == BRIX_RPM_ERR_PATH_TOO_LONG)
49 ? NGX_HTTP_REQUEST_URI_TOO_LARGE : NGX_HTTP_BAD_REQUEST;
50}

53ngx_int_t
54brix_rpm_gate(ngx_http_request_t *r, ngx_http_brix_rpm_ctx_t *ctx)
55{
56 if (!(r->method & (NGX_HTTP_GET | NGX_HTTP_HEAD))) {
57 return rpm_refuse_write(r, ctx);
58 }
59
60 /* The URI is the key, so the cap is checked here rather than trusted:
61 * classify() enforces the same bound, but the copy below must be safe on
62 * its own terms — the two are one number (BRIX_RPM_KEY_MAX). */
63 if (r->uri.len == 0 || r->uri.len >= sizeof(ctx->key)) {
64 ctx->disp = BRIX_RPM_OUT_REFUSED;
65 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
66 }
67 ngx_memcpy(ctx->key, r->uri.data, r->uri.len);
68 ctx->key[r->uri.len] = '\0';
69 ctx->key_len = r->uri.len;
70
71 if (brix_rpm_classify(ctx->key, ctx->key_len, &ctx->req) != 0) {
72 ctx->classified = 1; /* BAD is a class, and it counts */
73 ctx->disp = BRIX_RPM_OUT_REFUSED;
74 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
75 "rpm: refusing \"%V\" - not a legal repository path", &r->uri);
76 return (ngx_int_t) rpm_bad_status(ctx->req.err);
77 }
78 ctx->classified = 1;
79
80 return NGX_DECLINED;
81}
