4/* ---- File: context.h — Per-connection session context (brix_ctx_t) ----
5 *
6 * PURPOSE:
7 * One brix_ctx_t per TCP connection, allocated from nginx pool.
8 * State machine runs on single worker thread — XRootD multiplexing handled
9 * via streamid matching on client side; server serializes responses.
10 *
11 * KEY DESIGN DECISIONS:
12 * 1. Reusable scratch buffers (malloc/realloc) prevent pool growth in
13 * long-lived xrdcp sessions. Per-request ngx_palloc would cause unbounded
14 * pool growth over connection lifetime.
15 * 2. AIO destruction guard (destroyed=1) prevents post-disconnect callback
16 * writes to freed memory — common bug in async I/O patterns.
17 * 3. Bind connections lazily reopen primary's canonical path in own worker —
18 * nginx workers cannot share post-fork fd integers safely.
19 * 4. Sigver lifecycle: kXGC_cert → signing_key=SHA-256(DH-secret)/active=1,
20 * sigver arrives → pending=1/envelope saved, next dispatch → HMAC verified.
21 * 5. TLS upgrade path intercepts next recv as ClientHello when kXR_haveTLS
22 * advertised — enables opportunistic TLS without separate port.
23 * 6. Lazy file table allocation (brix_files_ensure) — brix_file_t is ~10KB,
24 * so metadata-only sessions avoid paying ~170KB for unused table.
25 *
26 * STRUCT LAYOUT (grouped by concern):
27 * - Core: session pointer, state machine state
28 * - Input accumulation: recv sub-struct (hdr_buf[24], hdr_pos, cur_streamid/reqid/body/dlen)
29 * - Session auth: login sub-struct (sessid, logged_in, auth_done, login_user[9])
30 * - Identity: dn[512], primary_vo[128], vo_list[512], peer_ip[64], identity object
31 * - File table: files[BRIX_MAX_FILES] — index = XRootD handle, lazy alloc
32 * - Send paths: out sub-struct (wbuf flat OR wchain chain — mutually exclusive)
33 * - Scratch buffers: rd sub-struct (read_scratch, read_hdr_scratch, write_scratch)
34 * - AIO: rd sub-struct (read_aio_task reusable thread task)
35 * - Fast-path: out sub-struct (read_fast_* one-chunk read, zero allocation)
36 * - GSI: gsi sub-struct (gsi_dh_key freed after DH secret derived)
37 * - Token: token sub-struct (token_auth, token_scopes[BRIX_MAX_TOKEN_SCOPES])
38 * - Totals: totals sub-struct (session_bytes, session_bytes_tx/rx_ipv4/ipv6)
39 * - Sigver: sigver sub-struct (signing_key, signing_active, last_seqno)
40 * - Bind: is_bound, pathid, bound_sessid (secondary data channel)
41 * - Metrics: metrics pointer to shared-memory segment
42 * - TLS: tls_pending (awaiting ClientHello after kXR_haveTLS)
43 * - CMS: cms_wait_streamid (pending locate answer)
44 * - Proxy: proxy context, bearer_token[BRIX_BEARER_TOKEN_MAX], proxy_fail_count
45 * - PMark: pmark sub-struct (SciTags packet-marking flow)
46 * - Prepare: prepare sub-struct (kXR_prepare/kXR_stage polling)
47 * - Throttle: throttle sub-struct (per-user accounting)
48 * - Rate limit: rl sub-struct (Phase 25/33)
49 * - Deadlines: deadline sub-struct (Phase 39 network-fault resilience)
50 * - Misc: req_start, io_monitor, destroyed, disconnect_done, admin_paused
51 *
52 * THREAD SAFETY:
53 * All fields accessed from single worker thread. No locks needed.
54 *
55 * MEMORY MANAGEMENT:
56 * - Scratch buffers: malloc/realloc (not pool) to avoid per-request growth
57 * - File table: lazy alloc on first kXR_open (brix_files_ensure)
58 * - payload_buf: reused across requests, detached by async handlers
59 * - read_fast_*: pre-zeroed chain objects for common one-chunk response
60 * - AIO task: single reusable task per connection
61 * - Bearer token: stack-allocated [BRIX_BEARER_TOKEN_MAX] for proxy forward
62 *
63 * LIFECYCLE:
64 * Allocated: ngx_stream_brix_handler() on TCP accept
65 * Freed: on_disconnect() after connection close
66 * Trimmed: scratch buffers after request drain (BRIX_SCRATCH_TRIM_THRESHOLD)
67 * Guarded: destroyed=1 prevents post-disconnect callback writes
68 *
69 * STATE MACHINE:
70 * XRD_ST_HANDSHAKE → XRD_ST_REQ_HEADER → XRD_ST_REQ_PAYLOAD → XRD_ST_SENDING
71 * XRD_ST_AIO (blocking I/O) → XRD_ST_SENDING → XRD_ST_REQ_HEADER (next)
72 * Special states: XRD_ST_TLS_HANDSHAKE, XRD_ST_UPSTREAM, XRD_ST_WAITING_CMS,
73 * XRD_ST_WAITING_BAQ, XRD_ST_PROXY
74 */

76/*
77 * Per-connection context (brix_ctx_t).
78 *
79 * One instance per TCP connection, allocated from the nginx connection pool.
80 * The state machine runs on a single nginx worker thread; only one request
81 * is "in flight" at a time — XRootD multiplexing is handled via streamid
82 * matching on the client side, but the server serialises responses.
83 *
84 * Requires: state.h, file.h, tunables.h, protocol/protocol.h,
85 * token/token.h, metrics/metrics.h, and nginx/OpenSSL headers
86 * before inclusion.
87 */
88/*
89 * Output-queue slot (Phase 29 pipelining).
90 *
91 * Each slot owns the full send state for ONE in-flight response: the
92 * flat-buffer tail (small error/ok/status frames) OR the chain tail (read data,
93 * dirlist chunks), plus the reusable header/data/file chain structs that back a
94 * single-chunk read response. Bundling these per-slot (instead of one set of
95 * singletons on the connection) is what lets more than one response be in flight
96 * without the builders aliasing each other's chain memory.
97 *
98 * Only one of wbuf / wchain is active in a given slot at a time. wbuf points at
99 * the next unsent byte; wbuf_base is the owned allocation freed once the slot
100 * drains. wchain holds the remaining chain links; wchain_base (if set) backs a
101 * chain link and is freed after the chain fully drains. The read_fast_* structs
102 * are the pre-zeroed chain/buf/file objects reused for the common one-chunk read
103 * response so that path allocates nothing from the pool per request.
104 */
105typedef struct {
106 u_char *wbuf; /* pointer to the next byte to send */
107 size_t wbuf_len; /* total bytes in the buffer */
108 size_t wbuf_pos; /* bytes successfully sent so far */
109 u_char *wbuf_base; /* allocation base (freed when send completes) */
110
111 ngx_chain_t *wchain; /* remaining chain links to send */
112 off_t wchain_pending; /* unsent bytes in wchain, for cheap resume */
113 u_char *wchain_base; /* optional backing buffer, freed after drain */
114
115 ngx_chain_t read_fast_hdr_chain;
116 ngx_chain_t read_fast_body_chain;
117 ngx_buf_t read_fast_hdr_buf;
118 ngx_buf_t read_fast_body_buf;
119 ngx_file_t read_fast_file;
120
121 /*
122 * Per-slot response header storage (Phase 29 single-chunk; Phase 32 WS2
123 * widened to BRIX_SLOT_HDR_MAX so a MULTI-chunk sendfile read also keeps
124 * all its per-chunk ServerResponseHdrs here instead of the shared
125 * read_hdr_scratch). Owning the headers per-slot is what lets multiple
126 * (single- or multi-chunk) reads pipeline without their headers clobbering
127 * each other while a prior response is still draining.
128 */
129 u_char hdr_bytes[BRIX_SLOT_HDR_MAX];
130} brix_resp_slot_t;

132/*
133 * Phase 32 WS3 — concurrent-AIO read pipeline.
134 *
135 * A pool of per-in-flight-read buffers + thread tasks lets multiple
136 * memory-backed kXR_reads be outstanding at once (pipelined) instead of the
137 * single read_scratch / read_aio_task serial pair. Each entry is owned by one
138 * in-flight read from dispatch until the response it backs has fully drained
139 * from its out_ring slot, at which point brix_release_read_buffer() returns it
140 * to the pool. Buffers are raw ngx_alloc'd (Phase-31 discipline) and freed on
141 * disconnect. rd_inflight counts entries currently in use.
142 */
143typedef struct {
144 u_char *buf; /* raw-alloc'd data buffer (ngx_alloc/ngx_free) */
145 size_t size; /* allocated size of buf */
146 ngx_thread_task_t *task; /* per-entry thread task (kXR_read AIO) */
147 unsigned in_use:1; /* 1 while an in-flight read owns this entry */
148 unsigned hot:1; /* used since the last trim pass — trim skips
149 * (and clears) it, so a slot is only shrunk
150 * after a full request cycle without use.
151 * Freeing an 8 MiB slot on EVERY request made
152 * glibc mmap/munmap the block each time
153 * (equal-size churn never clears the dynamic
154 * mmap threshold): ~25% of streaming-pgread
155 * worker CPU went to page faults + zeroing. */
156} brix_read_slot_t;
