1/*
2 * vfs_backend_registry.h — per-export storage-backend resolution.
3 *
4 * WHAT: A tiny per-worker table mapping an export's canonical root to its
5 * selected non-POSIX storage backend (today: pblock). Config parsing
6 * registers the export's backend choice; brix_vfs_ctx_init() resolves a
7 * ctx's root_canon to a bound brix_sd_instance_t through this table, so
8 * EVERY VFS op (any protocol) routes to the driver without each handler
9 * threading the instance by hand.
10 *
11 * WHY: The backend instance must be created per worker (a SQLite connection
12 * cannot be shared across fork) and lazily (after fork), but the *choice*
13 * (root, backend name, block size) is known at config time. Splitting those
14 * — register the choice at config time, create the instance on first use
15 * per worker — lets one ctx_init lookup cover all ~50 ctx-build sites.
16 *
17 * HOW: brix_vfs_backend_config() records {root_canon, "pblock", block_size}
18 * at config time (deduped on reload). brix_vfs_backend_resolve() looks up
19 * root_canon; on the first hit in a given worker it builds the instance on
20 * the cycle pool and caches it. A root with no registered backend (or a
21 * no-sqlite build) resolves to NULL ⇒ the default POSIX path.
22 */

39/* brix_vfs_xroot_origin_t — the endpoint a remote root:// primary backend binds
40 * to, as parsed from the brix_storage_backend URL.
41 *
42 * WHAT: host/port plus the three declarations the URL SCHEME carries: `tls`
43 * (roots://), `family` (the export's address-family constraint) and
44 * `nearline` (root+tape:// — the origin fronts an MSS).
45 * WHY: the four fields are all int-or-string and trivially transposable
46 * positionally, and `nearline` is the second flag to join them; one named
47 * struct keeps the call site self-describing (the same reason
48 * brix_sd_xroot_origin_cfg_t exists one layer down).
49 * HOW: `host` need only outlive the call — it is copied onto the entry. */
50typedef struct {
51 const char *host; /* required: origin hostname (non-empty) */
52 int port; /* required: origin port (1..65535) */
53 int tls; /* non-zero: roots:// (TLS) */
54 int family; /* brix_af_policy_t address-family constraint */
55 int nearline; /* non-zero: root+tape:// — recall, do not block */
56} brix_vfs_xroot_origin_t;

89/* Record (at config time) that the export rooted at `root_canon` is backed by
90 * an S3 source (`host`:`port`, TLS iff `tls`, path-style `bucket`), served
91 * through the sd_remote driver over the shared libcurl S3 transport. The driver
92 * advertises CAP_RANGE_READ (+CAP_MEMFILE) — no CAP_RANDOM_WRITE — so in-place
93 * partial writes are rejected at the cap layer, while sequential uploads go
94 * through the whole-object staged path (.staged_* → single PUT or MPU) and
95 * deletes through .unlink (phase-71/phase-80). */
96/* The optional declarations an s3 origin URL carries in its query suffix. */
97typedef struct {

99 int nearline; /* "?nearline=1": the bucket is archive-backed
100 * (GLACIER/DEEP_ARCHIVE), so residency comes from the
101 * storage class and recall from RestoreObject. Arms
102 * BRIX_SD_CAP_NEARLINE — a declaration, like tape_api
103 * above, and subject to the same cache-tier contract. */

105 * readable. 0 leaves the driver's default. */
