43/* WHAT: Decide whether an error message is non-empty printable text.
44 * WHY: Wire errors must not hide binary/control data inside their dlen.
45 * HOW: Permit horizontal tabs and visible bytes, rejecting every other control. */
46static int
47message_readable(const char *message)
48{
49 const unsigned char *cursor = (const unsigned char *) message;
50
51 if (cursor == NULL || *cursor == '\0') return 0;
52 while (*cursor != '\0') {
53 if ((*cursor < 0x20 && *cursor != '\t') || *cursor == 0x7f) return 0;
54 cursor++;
55 }
56 return 1;
57}

59/* WHAT: Reject obvious host-path and credential disclosure in an error string.
60 * WHY: A correct status code is still unsafe when its explanation leaks local

62 * HOW: Match only high-confidence path roots and credential markers. */
63static int
64message_safe(const char *message)
65{
66 static const char *const forbidden[] = {

68 "Authorization:", "Bearer ", "BEGIN PRIVATE KEY",

79/* WHAT: Execute one labelled fault fixture and capture its first wire error.
80 * WHY: mapError conformance is external only when an actual server response is
81 * observed; compiling the mapping table is necessary but insufficient.
82 * HOW: Connect, read-open, then read one byte; accept failure at open or read,
83 * closing every successfully opened handle and connection. */
84int
85xrd_conform_observe_root_error(xrd_conform_execution *execution,
86 const char *target,
87 uint16_t *code, char message[128])
88{
89 brix_url url;
90 brix_conn connection;
91 brix_file file;
92 brix_status status;
93 brix_status close_status;
94 unsigned char byte;
95 char reason[64];
96 int opened = 0;
97 int failed = 0;
98
99 brix_status_clear(&status);
100 if (brix_endpoint_parse(target, &url, &status) != 0
101 || (url.scheme != XRDC_SCHEME_ROOT && url.scheme != XRDC_SCHEME_ROOTS)
102 || brix_connect(&connection, &url, &execution->run->opts, &status) != 0)
103 return -1;
104 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) {
105 brix_close(&connection);
106 return -1;
107 }
108 if (brix_file_open_read(&connection, url.path, &file, &status) != 0) {
109 failed = 1;
110 } else {
111 opened = 1;
112 if (!xrd_conform_reserve(execution, 1, reason, sizeof(reason))) {
113 brix_status_clear(&close_status);
114 brix_file_close(&connection, &file, &close_status);
115 brix_close(&connection);
116 return -1;
117 }
118 failed = brix_file_read(&connection, &file, 0, &byte, 1, &status) < 0;
119 }
120 *code = failed ? (uint16_t) status.kxr : 0;
121 snprintf(message, 128, "%.127s", failed
122 ? status.wire_message_len > 0 && status.wire_message_terminated
123 ? status.msg : ""
124 : "request-succeeded");
125 if (opened) {
126 brix_status_clear(&close_status);
127 brix_file_close(&connection, &file, &close_status);
128 }
129 brix_close(&connection);
130 return failed ? 0 : 1;
131}

133/* WHAT: Observe a fixture twice and derive its per-fixture safety facts.
134 * WHY: One request proves neither deterministic classification nor stable text.
135 * HOW: Run two fresh sessions and compare exact code/message pairs. */
136static error_fixture_observation
137observe_fixture(xrd_conform_execution *execution, const char *target)
138{
139 error_fixture_observation observation;
140 uint16_t repeat_code = 0;
141 char repeat_message[128] = "";
142 int first_rc;
143 int repeat_rc;
144
145 memset(&observation, 0, sizeof(observation));
146 observation.present = 1;
147 first_rc = xrd_conform_observe_root_error(execution, target,
148 &observation.code,
149 observation.message);
150 repeat_rc = xrd_conform_observe_root_error(execution, target,
151 &repeat_code, repeat_message);
152 observation.failed = first_rc == 0 && repeat_rc == 0;
153 observation.deterministic = observation.failed
154 && observation.code == repeat_code
155 && strcmp(observation.message, repeat_message) == 0;
156 observation.readable = observation.failed
157 && message_readable(observation.message);
158 observation.safe = observation.readable && message_safe(observation.message);
159 return observation;
160}

212 const error_fixture_observation observations[])
213{
