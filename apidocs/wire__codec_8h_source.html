1/*
2 * wire_codec.h — shared per-opcode XRootD wire-body codec (xrdwire).
3 *
4 * WHAT: Decoded structs + paired pack/unpack functions for the fixed-layout
5 * request bodies (and fixed response bodies) of each XRootD opcode. The
6 * client uses the *_pack (request) / *_unpack (response) halves; the server
7 * uses the mirror halves — one definition of every wire offset, shared via
8 * libxrdproto so the two cannot drift.
9 * WHY: The 16-byte ClientRequestHdr body and the fixed response bodies were
10 * hand-marshalled twice (client builds, server parses; and vice-versa).
11 * That implicit "offsets must match" contract is the largest remaining
12 * cross-component duplication and the classic wire-bug class. This codec
13 * generalises the proven pgio / vendor_ext / fattr_codec pattern to every
14 * opcode and makes the contract round-trip-testable with no process running.
15 * HOW: Pure C, ngx-free, big-endian fixed-offset copies over a caller buffer —
16 * the codec owns ONLY the fixed-offset fields. The shared framing header
17 * (streamid/requestid/dlen) stays in frame_hdr.h; the variable tail (path,
18 * payload, dirlist/stat strings) is appended/consumed by the caller. The
19 * codec never touches a socket, nginx, or the client's poll machinery.
20 *
21 * See docs/superpowers/specs/2026-06-27-xrdwire-codec-design.md.
22 */
23#ifndef BRIX_WIRE_CODEC_H
24#define BRIX_WIRE_CODEC_H

31/* Result codes (consumers map onto kXR_* / XRDC_* at the edge). */
32#define XRDW_OK 0
33#define XRDW_EINVAL (-1) /* NULL argument / bad field */
34#define XRDW_ETRUNC (-2) /* input shorter than the fixed region */

36/* The fixed request-body region of ClientRequestHdr (between requestid and dlen). */
37#define XRDW_BODY_LEN 16
38#define XRDW_FHANDLE_LEN 4
39
40/* Compile-time guards on the codec's foundational constants. Every pack/unpack
41 * memcpy/put/get below indexes into a XRDW_BODY_LEN-byte buffer at offsets up to
42 * 12 + XRDW_FHANDLE_LEN; if either constant drifts from the XRootD wire (the
43 * 16-byte ClientRequestHdr.body, 4-byte file handle) the offsets silently corrupt
44 * every request. These turn that drift into a build failure. The matching
45 * "codec == real ClientRequestHdr.body" tie lives at the module consumer
46 * (src/read/read.c), since this header is ngx-/XProtocol-free by design. */
47_Static_assert(XRDW_BODY_LEN == 16, "XRootD ClientRequestHdr body region is 16 bytes");
48_Static_assert(XRDW_FHANDLE_LEN == 4, "XRootD file handle is 4 bytes");
49_Static_assert(12 + XRDW_FHANDLE_LEN <= XRDW_BODY_LEN,
50 "fhandle at offset 12 must fit within the body region");

52/* ---- big-endian (network order) field helpers ------------------------- */
53
54static inline void xrdw_put_u16(uint8_t *p, uint16_t v)
55{
56 p[0] = (uint8_t) (v >> 8);
57 p[1] = (uint8_t) v;
58}
59
60static inline void xrdw_put_u32(uint8_t *p, uint32_t v)
61{
62 p[0] = (uint8_t) (v >> 24);
63 p[1] = (uint8_t) (v >> 16);
64 p[2] = (uint8_t) (v >> 8);
65 p[3] = (uint8_t) v;
66}
67
68static inline uint16_t xrdw_get_u16(const uint8_t *p)
69{
70 return (uint16_t) (((uint16_t) p[0] << 8) | p[1]);
71}
72
73static inline uint32_t xrdw_get_u32(const uint8_t *p)
74{
75 return ((uint32_t) p[0] << 24) | ((uint32_t) p[1] << 16)
76 | ((uint32_t) p[2] << 8) | (uint32_t) p[3];
77}
78
79static inline void xrdw_put_u64(uint8_t *p, uint64_t v)
80{
81 xrdw_put_u32(p, (uint32_t) (v >> 32));
82 xrdw_put_u32(p + 4, (uint32_t) v);
83}
84
85static inline uint64_t xrdw_get_u64(const uint8_t *p)
86{
87 return ((uint64_t) xrdw_get_u32(p) << 32) | xrdw_get_u32(p + 4);
88}

90/* ---- field-table pack/unpack engine ----------------------------------- */
91
92/*
93 * One row = one fixed field of a 16-byte body. The SAME table drives pack and
94 * unpack, so a wire offset cannot drift between the two directions (the bug
95 * class this codec exists to kill). Multi-byte integers are big-endian on the
96 * wire; signed struct fields round-trip through the matching-width unsigned
97 * type via memcpy.
98 */
99typedef enum {
100 XRDW_F_U8,
101 XRDW_F_U16,
102 XRDW_F_U32,
103 XRDW_F_U64,
104 XRDW_F_BYTES, /* len raw bytes, no byte order */
105} xrdw_ftype_t;
106
107typedef struct {
108 uint8_t wire_off; /* offset into the 16-byte body */
109 uint8_t type; /* xrdw_ftype_t */
110 uint8_t len; /* XRDW_F_BYTES only */
111 uint8_t struct_off; /* offsetof() in the decoded struct */
112} xrdw_field_t;
113
114#define XRDW_NFIELDS(tab) (sizeof(tab) / sizeof((tab)[0]))
115
116static inline void xrdw_field_put(uint8_t *body, const uint8_t *base,
117 const xrdw_field_t *f)
118{
119 uint16_t v16;
120 uint32_t v32;
121 uint64_t v64;
122
123 switch ((xrdw_ftype_t) f->type) {
124 case XRDW_F_U8:
125 body[f->wire_off] = base[f->struct_off];
126 break;
127 case XRDW_F_U16:
128 memcpy(&v16, base + f->struct_off, sizeof(v16));
129 xrdw_put_u16(body + f->wire_off, v16);
130 break;
131 case XRDW_F_U32:
132 memcpy(&v32, base + f->struct_off, sizeof(v32));
133 xrdw_put_u32(body + f->wire_off, v32);
134 break;
135 case XRDW_F_U64:
136 memcpy(&v64, base + f->struct_off, sizeof(v64));
137 xrdw_put_u64(body + f->wire_off, v64);
138 break;
139 case XRDW_F_BYTES:
140 memcpy(body + f->wire_off, base + f->struct_off, f->len);
141 break;
142 }
143}
144
145static inline void xrdw_field_get(const uint8_t *body, uint8_t *base,
146 const xrdw_field_t *f)
147{
148 uint16_t v16;
149 uint32_t v32;
150 uint64_t v64;
151
152 switch ((xrdw_ftype_t) f->type) {
153 case XRDW_F_U8:
154 base[f->struct_off] = body[f->wire_off];
155 break;
156 case XRDW_F_U16:
157 v16 = xrdw_get_u16(body + f->wire_off);
158 memcpy(base + f->struct_off, &v16, sizeof(v16));
159 break;
160 case XRDW_F_U32:
161 v32 = xrdw_get_u32(body + f->wire_off);
162 memcpy(base + f->struct_off, &v32, sizeof(v32));
163 break;
164 case XRDW_F_U64:
165 v64 = xrdw_get_u64(body + f->wire_off);
166 memcpy(base + f->struct_off, &v64, sizeof(v64));
167 break;
168 case XRDW_F_BYTES:
169 memcpy(base + f->struct_off, body + f->wire_off, f->len);
170 break;
171 }
172}

174/* Validate, zero the body, then lay each field down at its table offset. */
175static inline int xrdw_pack_fields(const void *r, uint8_t body[XRDW_BODY_LEN],
176 const xrdw_field_t *fields, size_t nfields)
177{
178 size_t i;
179
180 if (r == NULL || body == NULL) {
181 return XRDW_EINVAL;
182 }
183 memset(body, 0, XRDW_BODY_LEN);
184 for (i = 0; i < nfields; i++) {
185 xrdw_field_put(body, (const uint8_t *) r, &fields[i]);
186 }
187 return XRDW_BODY_LEN;
188}
189
190/* Validate, then lift each table field out of the body into the struct. */
191static inline int xrdw_unpack_fields(const uint8_t body[XRDW_BODY_LEN],
192 void *r, const xrdw_field_t *fields,
193 size_t nfields)
194{
195 size_t i;
196
197 if (body == NULL || r == NULL) {
198 return XRDW_EINVAL;
199 }
200 for (i = 0; i < nfields; i++) {
201 xrdw_field_get(body, (uint8_t *) r, &fields[i]);
202 }
203 return XRDW_OK;
204}
