1/*
2 * oci_authz.c — the registry surface's authorization gate (§D4.5).
3 *
4 * WHAT: decide, once per request, whether this client may act on the local
5 * registry and under what identity, and emit the spec's own refusal
6 * (401 + a `WWW-Authenticate` challenge naming our realm, or 403) when
7 * it may not.
8 * WHY: a push surface that answers 200 to an unauthenticated client is a
9 * supply-chain hole with a green checkmark on it: whatever it accepted
10 * is what every node on the site subsequently runs. So the gate is
11 * fail-closed in the strongest sense available here — a location that
12 * has not been told to accept writes cannot be talked into one, and an
13 * operator who genuinely wants an open registry has to say so in a
14 * directive whose name says what it does.
15 * The 401 shape matters as much as the refusal: `podman login` only
16 * works against a registry that answers an unauthenticated request with
17 * a challenge it can follow, so the header is part of the contract, not
18 * decoration.
19 * HOW: INVARIANT #3 order, without exception — `brix_allow_write` is
20 * consulted BEFORE any token is parsed, so a token with a generous
21 * scope can never open a location the operator kept read-only. Identity
22 * then comes from the plane the location already configured: a TLS
23 * client certificate if one is present, otherwise a bearer validated
24 * against `brix_oci_token_issuers`.
25 */

35/* The realm advertised in the challenge. A client follows it verbatim, so it
36 * names THIS location's own token endpoint rather than any upstream's — the
37 * registry surface never delegates its authentication to a mirror upstream. */
38#define OCI_REALM_HEADER_MAX 512
39
40/* Scopes are checked against a path, and the natural path for a repository is
41 * the store route it maps onto. "/v2/<name>" is stable, hierarchical and is
42 * exactly what an operator would write in a token's `storage.modify` scope. */
43#define OCI_SCOPE_PATH_MAX (BRIX_OCI_KEY_MAX)

56static const char *
57oci_bearer_of(ngx_http_request_t *r, size_t *len_out)
58{
59 ngx_table_elt_t *h = r->headers_in.authorization;
60
61 if (h == NULL || h->value.len <= sizeof("Bearer ") - 1) {
62 return NULL;
63 }
64 if (ngx_strncasecmp(h->value.data, (u_char *) "Bearer ",
65 sizeof("Bearer ") - 1) != 0)
66 {
67 return NULL;
68 }
69 *len_out = h->value.len - (sizeof("Bearer ") - 1);
70 return (const char *) h->value.data + sizeof("Bearer ") - 1;
71}

74/* 401 with the challenge a registry client knows how to follow. The realm is
75 * this location's own URL: a client that cannot resolve it will fall back to
76 * anonymous, which is the honest outcome for a site running no token plane. */
77static ngx_int_t
78oci_challenge(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
79 const char *detail)
80{
81 char hdr[OCI_REALM_HEADER_MAX];
82 int n;
83
84 ctx->disp = BRIX_OCI_OUT_REFUSED;
85
86 n = snprintf(hdr, sizeof(hdr),
87 "Bearer realm=\"%s://%.*s/v2/token\",service=\"%.*s\"",
88 (r->connection->ssl != NULL) ? "https" : "http",
89 (int) r->headers_in.server.len, r->headers_in.server.data,
90 (int) r->headers_in.server.len, r->headers_in.server.data);
91 if (n > 0 && (size_t) n < sizeof(hdr)
92 && brix_http_set_header(r, "WWW-Authenticate", hdr, NULL) != NGX_OK)
93 {
94 return NGX_HTTP_INTERNAL_SERVER_ERROR;
95 }
96
97 brix_oci_guard_emit(r, GUARD_R_AUTHFAIL, GUARD_OP_WRITE,
98 NGX_HTTP_UNAUTHORIZED);
99
100 return brix_oci_refuse(r, NGX_HTTP_UNAUTHORIZED,
101 BRIX_OCI_ERR_UNAUTHORIZED, detail);
102}

119/* Validate the presented bearer against every configured issuer, and require
120 * a write scope covering this repository. Returns NGX_OK with `principal`
121 * filled, NGX_DECLINED when no issuer accepted it, NGX_ABORT when a valid
122 * token simply lacks the scope (a 403, not a 401 — re-authenticating would
123 * hand back the same token). */
124static ngx_int_t
125oci_authz_bearer(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
126 const ngx_http_brix_oci_ctx_t *ctx, char *principal, size_t principal_len)
127{
128 brix_token_validate_args_t args;
129 brix_token_claims_t claims;
130 char path[OCI_SCOPE_PATH_MAX];
131 const char *tok;
132 size_t tok_len = 0;
133 int i;
134
135 tok = oci_bearer_of(r, &tok_len);
136 if (tok == NULL || lcf->issuers == NULL || lcf->issuers->count == 0) {
137 return NGX_DECLINED;
138 }
139
140 for (i = 0; i < lcf->issuers->count; i++) {
141 const brix_token_issuer_t *iss = &lcf->issuers->issuers[i];
142
143 if (!iss->enabled || iss->jwks_key_count == 0) {
144 continue;
145 }
146 ngx_memzero(&args, sizeof(args));
147 ngx_memzero(&claims, sizeof(claims));
148 args.log = r->connection->log;
149 args.token = tok;
150 args.token_len = tok_len;
151 args.keys = iss->jwks_keys;
152 args.key_count = iss->jwks_key_count;
153 args.expected_issuer = iss->issuer;
154 args.claims = &claims;
155
156 if (brix_token_validate(&args) != 0) {
157 continue;
158 }
159
160 /* A valid token from a trusted issuer. Whether it may WRITE is a
161 * separate question with a separate answer code — conflating the two
162 * is what sends clients into a re-login loop against a scope problem
163 * no login can fix. */
164 (void) snprintf(path, sizeof(path), "/v2/%.*s",
165 (int) ctx->req.name_len, ctx->req.name);
166
167 if (oci_is_write(r)
168 && !brix_token_check_write(claims.scopes, claims.scope_count,
169 path))
170 {
171 return NGX_ABORT;
172 }
173 (void) snprintf(principal, principal_len, "%s", claims.sub);
174 return NGX_OK;
175 }
176
177 return NGX_DECLINED;
178}

181ngx_int_t
182brix_oci_registry_authz(ngx_http_request_t *r,
183 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx,
184 char *principal, size_t principal_len)
185{
186 ngx_int_t rc;
187
188 principal[0] = '\0';
189
190 /* INVARIANT #3: the location's own write permission is consulted first
191 * and independently. No credential of any strength promotes a read-only
192 * location into a writable one. */
193 if (oci_is_write(r) && !lcf->common.allow_write) {
194 ctx->disp = BRIX_OCI_OUT_REFUSED;
195 brix_oci_guard_emit(r, GUARD_R_AUTHFAIL, GUARD_OP_WRITE,
196 NGX_HTTP_FORBIDDEN);
197 return brix_oci_refuse(r, NGX_HTTP_FORBIDDEN, BRIX_OCI_ERR_DENIED,
198 "this registry location is read-only "
199 "(brix_allow_write is off)");
200 }
201
202 rc = oci_authz_bearer(r, lcf, ctx, principal, principal_len);
203 if (rc == NGX_OK) {
204 return NGX_OK;
205 }
206 if (rc == NGX_ABORT) {
207 return oci_deny(r, ctx, "the presented token carries no write scope "
208 "for this repository");
209 }
210
211#if (NGX_HTTP_SSL)
212 /* A TLS client certificate the location already validated is an identity
213 * in its own right: the handshake did the proving, and requiring a bearer
214 * on top would break the x509-only deployments the tree already serves. */
215 if (r->connection->ssl != NULL && r->headers_in.authorization == NULL) {
216 ngx_str_t dn = ngx_null_string;
217
218 if (ngx_ssl_get_subject_dn(r->connection, r->pool, &dn) == NGX_OK
219 && dn.len > 0)
220 {
221 (void) snprintf(principal, principal_len, "%.*s",
222 (int) dn.len, dn.data);
223 return NGX_OK;
224 }
225 }
226#endif
227
228 if (lcf->registry_anon) {
229 /* The typed decision to run an open registry. Recorded as such in the
230 * identity so the access log distinguishes "nobody authenticated" from
231 * "somebody did, and it was anonymous". */
232 (void) snprintf(principal, principal_len, "anonymous");
233 return NGX_OK;
234 }
235
236 return oci_challenge(r, ctx,
237 "authentication required "
238 "(or set brix_oci_registry_allow_anonymous on)");
239}
