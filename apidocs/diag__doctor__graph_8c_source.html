1/*
2 * diag_doctor_graph.c — mesh topology diagram renderer (phase-93 extension).
3 *
4 * WHAT: draw the manager→data-server mesh that doctor_fanout discovered over the
5 * wire (a kXR_locate answer from the manager, i.e. the CMS subscription
6 * view) as an ASCII tree, a Graphviz DOT digraph, or a Mermaid graph.
7 * WHY: the fan-out already builds the topology as a doctor_ep[] (eps[0]=manager,
8 * eps[1..]=located data servers) with each node's scraped role/version/
9 * capacity/health; an operator asked to *see* the mesh, not just read a
10 * per-node table. DOT/Mermaid output is pipeable to `dot -Tpng` / a Mermaid
11 * renderer for a real picture.
12 * HOW: pure formatting over the already-scraped array — no new wire, no
13 * connection. The single edge class is manager→DS (the CMS star the locate
14 * answer expresses). PII-free beyond the cluster-member authorities that
15 * ARE the topology (no path, token, or credential). No goto; early-return.
16 */
17#include "diag_internal.h"

20typedef enum { MAP_ASCII, MAP_DOT, MAP_MERMAID } doctor_map_fmt;

22static doctor_map_fmt
23doctor_map_format(const char *s)
24{
25 if (s != NULL && strcmp(s, "dot") == 0) {
26 return MAP_DOT;
27 }
28 if (s != NULL && (strcmp(s, "mermaid") == 0 || strcmp(s, "mmd") == 0)) {
29 return MAP_MERMAID;
30 }
31 return MAP_ASCII; /* default + explicit "ascii"/"tree" */
32}

45/* 1 when the node acts as a manager/redirector: the CMS locate answer is
46 * authoritative, else a scraped "manager" role, else the connected-to root. */
47static int
48map_is_redirector(const doctor_ep *e, int is_root)
49{
50 if (e->cms.reported) {
51 return e->cms.role == DOC_CMS_MANAGER;
52 }
53 if (e->cfg.scraped && strstr(e->cfg.role, "manager") != NULL) {
54 return 1;
55 }
56 return is_root;
57}

60/* Resolve a node's role descriptor into `buf`. The CMS locate answer types the
61 * node even when we could not connect (redirector vs data server); otherwise the
62 * scraped Qconfig role, otherwise a generic default. Read/write access and a
63 * pending (queued/staging) marker are appended for CMS-reported data servers —
64 * this is the "what is it?" data the CMSD gives us for un-connectable hosts. */
65static void
66map_role(const doctor_ep *e, int is_root, char *buf, size_t bsz)
67{
68 const doctor_cmsloc *m = &e->cms;
69 const char *base;
70
71 if (e->eos.kind == DOC_EOS_FST) {
72 /* an EOS FST enumerated from the MGM — access from configstatus */
73 snprintf(buf, bsz, "EOS FST%s", m->write ? " rw" : " ro");
74 return;
75 }
76 if (m->reported) {
77 base = (m->role == DOC_CMS_MANAGER) ? "redirector" : "data server";
78 } else if (e->cfg.scraped && e->cfg.role[0]
79 && strcmp(e->cfg.role, "none") != 0) {
80 base = e->cfg.role;
81 } else {
82 base = is_root ? "manager" : "server";
83 }
84 snprintf(buf, bsz, "%s%s%s", base,
85 (m->reported && m->role == DOC_CMS_SERVER)
86 ? (m->write ? " rw" : " ro") : "",
87 (m->reported && m->pending) ? " pending" : "");
88}

91/* Health verdict (DOC_GREEN/YELLOW/RED) for colouring, EOS-FST-aware: an FST is
92 * enumerated from the MGM and never probed inbound, so its health comes from the
93 * MGM's booted/active flags rather than a connect result. */
94static int
95map_health(const doctor_ep *e)
96{
97 if (e->eos.kind == DOC_EOS_FST) {
98 if (e->eos.booted && e->eos.active) { return DOC_GREEN; }
99 if (e->eos.booted || e->eos.active) { return DOC_YELLOW; }
100 return DOC_RED;
101 }
102 return e->status;
103}

106/* Up-to-five short PII-free descriptor tokens for one node:
107 * role · host:port · {EOS banner | geotag | version} · "NN% free" · STATUS.
108 * Returns the count filled. */
109static int
110map_fields(const doctor_ep *e, int is_root, char f[5][80])
111{
112 const doctor_cfg *g = &e->cfg;
113 int pct, k = 0;
114
115 map_role(e, is_root, f[k++], 80);
116 snprintf(f[k++], 80, "%.72s:%d", e->host, e->port);
117 if (e->eos.kind == DOC_EOS_MGM) {
118 snprintf(f[k++], 80, "EOS %.20s v%.12s", e->eos.instance,
119 e->eos.version[0] ? e->eos.version : "?");
120 } else if (e->eos.kind == DOC_EOS_FST && e->eos.geotag[0]) {
121 snprintf(f[k++], 80, "geo=%.39s", e->eos.geotag);
122 } else if (g->scraped && g->version[0]) {
123 snprintf(f[k++], 80, "v%s", g->version);
124 }
125 pct = doctor_cfg_capacity_pct(g->space_total, g->space_free);
126 if (pct >= 0) {
127 snprintf(f[k++], 80, "%d%% free", pct);
128 }
129 if (e->eos.kind == DOC_EOS_FST) {
130 int h = map_health(e);
131 snprintf(f[k++], 80, "%s",
132 h == DOC_RED ? "RED" : h == DOC_YELLOW ? "YELLOW" : "GREEN");
133 } else {
134 snprintf(f[k++], 80, "%s",
135 e->skipped ? "SKIP(no IPv6)"
136 : e->connected ? doc_color(e->status) : "DOWN");
137 }
138 return k;
139}

142/* Join a node's fields with `sep` into one line. */
143static void
144map_join(const doctor_ep *e, int is_root, const char *sep, char *out, size_t osz)
145{
146 char f[5][80];
147 int i, k = map_fields(e, is_root, f);
148 size_t pos = 0;
149
150 out[0] = '\0';
151 for (i = 0; i < k; i++) {
152 pos += snprintf(out + pos, (pos < osz) ? osz - pos : 0,
153 "%s%s", i ? sep : "", f[i]);
154 if (pos >= osz) {
155 break;
156 }
157 }
158}

161static void
162map_render_ascii(const doctor_ep *eps, int n, FILE *out)
163{
164 char line[512];
165 int i;
166
167 fprintf(out, "Mesh topology — discovered via CMS locate (%d node%s):\n\n",
168 n, n == 1 ? "" : "s");
169 map_join(&eps[0], 1, " ", line, sizeof(line));
170 fprintf(out, " [%s]\n", line);
171 for (i = 1; i < n; i++) {
172 map_join(&eps[i], 0, " ", line, sizeof(line));
173 fprintf(out, " %s [%s]\n", (i == n - 1) ? "\\-" : "|-", line);
174 }
175 fprintf(out, "\n legend: root = the redirector we connected to; each branch "
176 "is a node it located over CMS — 'redirector' (subordinate "
177 "manager) or 'data server' (rw/ro = write/read access, 'pending' "
178 "= queued/staging). Role/access come from the locate answer, so "
179 "even an unreachable (SKIP) holder is still typed. 'EOS FST' "
180 "nodes come from the MGM's EOS command channel — the admin `fs "
181 "ls` inventory, or (when that is gated for this identity) the "
182 "distinct FSTs named by a sample of files' `fileinfo` replica "
183 "tables — with health from their booted/active flags, not an "
184 "inbound probe.\n");
185}

188/* Graphviz fill colour per health verdict. */
189static const char *
190map_fill(const doctor_ep *e)
191{
192 if (e->skipped) {
193 return "lightskyblue";
194 }
195 if (e->eos.kind == DOC_EOS_FST) {
196 int h = map_health(e); /* MGM-reported, never inbound-probed */
197 return h == DOC_RED ? "lightcoral"
198 : h == DOC_YELLOW ? "khaki" : "palegreen";
199 }
200 if (!e->connected) {
201 return "gray";
202 }
203 return e->status == DOC_RED ? "lightcoral"
204 : e->status == DOC_YELLOW ? "khaki" : "palegreen";
205}

208static void
209map_render_dot(const doctor_ep *eps, int n, FILE *out)
210{
211 char label[512];
212 int i;
213
214 fprintf(out, "digraph mesh {\n");
215 fprintf(out, " rankdir=TB;\n");
216 fprintf(out, " node [style=filled, fontname=\"monospace\"];\n");
217 for (i = 0; i < n; i++) {
218 map_join(&eps[i], i == 0, "\\n", label, sizeof(label));
219 fprintf(out, " n%d [label=\"%s\", shape=%s, fillcolor=%s];\n",
220 i, label, map_is_redirector(&eps[i], i == 0) ? "box3d" : "box",
221 map_fill(&eps[i]));
222 }
223 for (i = 1; i < n; i++) {
224 fprintf(out, " n0 -> n%d;\n", i);
225 }
226 fprintf(out, "}\n");
227}

230static void
231map_render_mermaid(const doctor_ep *eps, int n, FILE *out)
232{
233 char label[512];
234 int i;
235
236 fprintf(out, "graph TD\n");
237 for (i = 0; i < n; i++) {
238 map_join(&eps[i], i == 0, "<br/>", label, sizeof(label));
239 /* redirectors as a hexagon {{...}}, data servers as a rectangle [...] */
240 if (map_is_redirector(&eps[i], i == 0)) {
241 fprintf(out, " n%d{{\"%s\"}}\n", i, label);
242 } else {
243 fprintf(out, " n%d[\"%s\"]\n", i, label);
244 }
245 }
246 for (i = 1; i < n; i++) {
247 fprintf(out, " n0 --> n%d\n", i);
248 }
249 /* colour classes by health verdict */
250 fprintf(out, " classDef red fill:#f8b,stroke:#900;\n");
251 fprintf(out, " classDef yellow fill:#ff8,stroke:#990;\n");
252 fprintf(out, " classDef green fill:#8f8,stroke:#090;\n");
253 fprintf(out, " classDef down fill:#ccc,stroke:#666;\n");
254 fprintf(out, " classDef skip fill:#bde,stroke:#369,stroke-dasharray:4;\n");
255 for (i = 0; i < n; i++) {
256 int h = map_health(&eps[i]);
257 const char *cls = eps[i].skipped ? "skip"
258 : eps[i].eos.kind == DOC_EOS_FST
259 ? (h == DOC_RED ? "red"
260 : h == DOC_YELLOW ? "yellow" : "green")
261 : !eps[i].connected ? "down"
262 : eps[i].status == DOC_RED ? "red"
263 : eps[i].status == DOC_YELLOW ? "yellow" : "green";
264 fprintf(out, " class n%d %s;\n", i, cls);
265 }
266}

269/* Render the discovered mesh in the requested format. `format` is the
270 * --map-format value (NULL/"ascii"/"tree" → ASCII, "dot", "mermaid"/"mmd"). */
271void
272doctor_render_map(const doctor_ep *eps, int n, const char *format, FILE *out)
273{
274 if (eps == NULL || n < 1) {
275 return;
276 }
277 switch (doctor_map_format(format)) {
278 case MAP_DOT: map_render_dot(eps, n, out); break;
279 case MAP_MERMAID: map_render_mermaid(eps, n, out); break;
280 case MAP_ASCII:
281 default: map_render_ascii(eps, n, out); break;
282 }
283}
