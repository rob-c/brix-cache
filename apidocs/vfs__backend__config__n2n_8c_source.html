1/*
2 * vfs_backend_config_n2n.c — config-time validation and registration of the
3 * per-export logical-to-physical name mapping.
4 *
5 * WHAT: Parses brix_n2n_scheme and applies the optional pool/prefix values to
6 * the backend-registry entry for one canonical export root.
7 * WHY: Name mapping has its own validation contract and is independent of the
8 * backend-origin parser. Keeping it in a focused unit holds the config
9 * dispatcher below the source-size limit and keeps each validation step
10 * small enough to audit.
11 * HOW: Parse the scheme, resolve the existing backend entry, validate the
12 * backend/scheme combination and bounded strings, then replace the
13 * entry's mapping and invalidate its lazily built instance.
14 */
15#include "vfs_backend_config_internal.h"

17/* Parse one configured scheme name into the internal mapping vocabulary. */
18static ngx_int_t
19vfs_n2n_scheme_parse(const ngx_str_t *scheme, brix_n2n_scheme_t *out)
20{
21 if (scheme->len == sizeof("identity") - 1
22 && ngx_strncmp(scheme->data, "identity", scheme->len) == 0)
23 {
24 *out = BRIX_N2N_IDENTITY;
25 return NGX_OK;
26 }
27 if (scheme->len == sizeof("ral") - 1
28 && ngx_strncmp(scheme->data, "ral", scheme->len) == 0)
29 {
30 *out = BRIX_N2N_RAL;
31 return NGX_OK;
32 }
33 if (scheme->len == sizeof("cephfs_path") - 1
34 && ngx_strncmp(scheme->data, "cephfs_path", scheme->len) == 0)
35 {
36 *out = BRIX_N2N_CEPHFS_PATH;
37 return NGX_OK;
38 }
39 return NGX_DECLINED;
40}

42/* Refuse scheme/backend combinations that would address a different object. */
43static ngx_int_t
44vfs_n2n_validate_backend(ngx_conf_t *cf, const brix_vfs_backend_entry_t *entry,
45 brix_n2n_scheme_t scheme, const ngx_str_t *pool)
46{
47 if (scheme == BRIX_N2N_RAL && ngx_strcmp(entry->backend, "ceph") == 0) {
48 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "brix_n2n_scheme ral is invalid "
49 "for the ceph/RADOS backend: its pool is bound at the ioctx and "
50 "objects are named \"<prefix><lfn>\" — use cephfs_path");
51 return NGX_ERROR;
52 }
53 if (scheme == BRIX_N2N_RAL && (pool == NULL || pool->len == 0)) {
54 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
55 "brix_n2n_scheme ral requires brix_n2n_pool");
56 return NGX_ERROR;
57 }
58 return NGX_OK;
59}

61/* Validate that configured mapping components fit their registry fields. */
62static ngx_int_t
63vfs_n2n_validate_lengths(ngx_conf_t *cf,
64 const brix_vfs_backend_entry_t *entry, const ngx_str_t *pool,
65 const ngx_str_t *prefix)
66{
67 if (pool != NULL && pool->len >= sizeof(entry->n2n.pool)) {
68 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "brix_n2n_pool is too long "
69 "(%uz bytes; max %uz)", pool->len, sizeof(entry->n2n.pool) - 1);
70 return NGX_ERROR;
71 }
72 if (prefix != NULL && prefix->len >= sizeof(entry->n2n.prefix)) {
73 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "brix_n2n_prefix is too long "
74 "(%uz bytes; max %uz)", prefix->len,
75 sizeof(entry->n2n.prefix) - 1);
76 return NGX_ERROR;
77 }
78 return NGX_OK;
79}

81/* Store an already-validated mapping and invalidate the cached driver stack. */
82static void
83vfs_n2n_assign(brix_vfs_backend_entry_t *entry, brix_n2n_scheme_t scheme,
84 const ngx_str_t *pool, const ngx_str_t *prefix)
85{
86 entry->n2n.scheme = scheme;
87 if (pool != NULL && pool->len > 0) {
88 ngx_cpystrn((u_char *) entry->n2n.pool, pool->data,
89 ngx_min(pool->len + 1, sizeof(entry->n2n.pool)));
90 } else if (scheme != BRIX_N2N_RAL) {
91 entry->n2n.pool[0] = '\0';
92 }
93 if (prefix != NULL && prefix->len > 0) {
94 ngx_cpystrn((u_char *) entry->n2n.prefix, prefix->data,
95 ngx_min(prefix->len + 1, sizeof(entry->n2n.prefix)));
96 }
97 entry->inst = NULL;
98}

100/* Apply the explicit per-export N2N configuration; an unset scheme preserves
101 * the backend parser's derived default. Returns NGX_OK or a logged NGX_ERROR. */
102ngx_int_t
103brix_vfs_backend_config_n2n(ngx_conf_t *cf, const char *root_canon,
104 const ngx_str_t *scheme, const ngx_str_t *pool, const ngx_str_t *prefix)
105{
106 brix_vfs_backend_entry_t *entry;
107 brix_n2n_scheme_t parsed;
108
109 if (scheme == NULL || scheme->len == 0) {
110 return NGX_OK;
111 }
112 if (vfs_n2n_scheme_parse(scheme, &parsed) != NGX_OK) {
113 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "brix_n2n_scheme: unknown "
114 "scheme \"%V\" (want identity | ral | cephfs_path)", scheme);
115 return NGX_ERROR;
116 }
117 entry = brix_vfs_backend_entry_find(root_canon);
118 if (entry == NULL) {
119 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0, "brix_n2n_scheme: no storage "
120 "backend is configured for this export to translate names for");
121 return NGX_ERROR;
122 }
123 if (vfs_n2n_validate_backend(cf, entry, parsed, pool) != NGX_OK
124 || vfs_n2n_validate_lengths(cf, entry, pool, prefix) != NGX_OK)
125 {
126 return NGX_ERROR;
127 }
128 vfs_n2n_assign(entry, parsed, pool, prefix);
129 return NGX_OK;
130}
