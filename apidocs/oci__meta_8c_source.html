1/*
2 * oci_meta.c — the `.ocimeta` sidecar (Appendix B.1).
3 *
4 * WHAT: read and write the small flat-file record that rides beside every
5 * cached OCI object: the media type to echo, the content digest to
6 * advertise, the upstream ETag, when it was fetched, and whether the
7 * fill was digest-verified.
8 * WHY: a registry client does not accept an object on its bytes alone. Podman
9 * and docker both branch on the manifest's `Content-Type` (an OCI index
10 * and a Docker manifest list are different objects with the same shape)
11 * and both read `Docker-Content-Digest` to pin what they just pulled.
12 * Neither value is recoverable from a TAG-addressed cache entry's path,
13 * and re-deriving them per hit would mean hashing the manifest on the
14 * event loop for every pull. So they are memoized here, next to the
15 * bytes they describe, and the record is treated as a CACHE of a pure
16 * function of those bytes: absent or unparsable simply means "derive it
17 * again", never "serve something wrong".
18 * HOW: one `key=value` line per field, LF-separated, written to a `.tmp`
19 * sibling and renamed into place so a reader never sees a torn record.
20 * Deliberately not JSON: the record is machine-written and
21 * machine-read, an operator greps it, and a parser that cannot fail in
22 * interesting ways is worth more here than a schema.
23 *
24 * The raw file calls below are marked vfs-seam-allow (invariant #12): this is
25 * cache BOOKKEEPING beside the object, not object data. Routing it through
26 * brix_vfs_* would hand it to the cache decorator, which would treat a missing
27 * sidecar as a miss and try to FILL it from the upstream registry — an
28 * endpoint that has never heard of `.ocimeta`.
29 */

42/* Build "<body_path>.ocimeta" (or its ".tmp" staging sibling). */
43static int
44oci_meta_path(const char *body_path, const char *extra, char *out, size_t outsz)
45{
46 int n = snprintf(out, outsz, "%s%s%s", body_path, OCI_META_SUFFIX,
47 (extra != NULL) ? extra : "");
48
49 return (n < 0 || (size_t) n >= outsz) ? -1 : 0;
50}

52/* Copy a field value, stripping the trailing LF. Silently truncates: a field
53 * longer than its slot is a corrupt record, and the caller's "derive it
54 * again" path is the right answer for that. */
55static void
56oci_meta_field(char *dst, size_t dstsz, const char *val)
57{
58 size_t n = strlen(val);
59
60 while (n > 0 && (val[n - 1] == '\n' || val[n - 1] == '\r')) {
61 n--;
62 }
63 if (n >= dstsz) {
64 n = dstsz - 1;
65 }
66 memcpy(dst, val, n);
67 dst[n] = '\0';
68}

70static void
71oci_meta_apply(brix_oci_meta_t *out, const char *key, const char *val)
72{
73 if (strcmp(key, "content_type") == 0) {
74 oci_meta_field(out->content_type, sizeof(out->content_type), val);
75
76 } else if (strcmp(key, "digest") == 0) {
77 oci_meta_field(out->digest, sizeof(out->digest), val);
78
79 } else if (strcmp(key, "etag") == 0) {
80 oci_meta_field(out->etag, sizeof(out->etag), val);
81
82 } else if (strcmp(key, "fetched_at") == 0) {
83 out->fetched_at = (time_t) strtoll(val, NULL, 10);
84
85 } else if (strcmp(key, "size") == 0) {
86 out->size = (off_t) strtoll(val, NULL, 10);
87
88 } else if (strcmp(key, "mtime") == 0) {
89 out->mtime = (time_t) strtoll(val, NULL, 10);
90
91 } else if (strcmp(key, "verified") == 0) {
92 out->verified = (strtol(val, NULL, 10) != 0);
93 }
94}

96/* Does this record describe THESE bytes? A refill leaves the body's mtime
97 * later and (nearly always) its size different; either disagreement means the
98 * memo predates the object it sits beside. */
99static int
100oci_meta_describes(const brix_oci_meta_t *m, off_t size, time_t mtime)
101{
102 if (m->size != size) {
103 return 0;
104 }
105 return (m->mtime == 0 || mtime <= 0 || m->mtime == mtime);
106}

108ngx_int_t
109brix_oci_meta_load(const char *body_path, off_t size, time_t mtime,
110 brix_oci_meta_t *out, ngx_log_t *log)
111{
112 char path[PATH_MAX];
113 char line[OCI_META_LINE_MAX];
114 FILE *fp;
115
116 ngx_memzero(out, sizeof(*out));
117
118 if (oci_meta_path(body_path, NULL, path, sizeof(path)) != 0) {
119 return NGX_ERROR;
120 }
121
122 fp = fopen(path, "re");
123 if (fp == NULL) {
124 return NGX_DECLINED; /* absent: derive and re-store */
125 }
126
127 while (fgets(line, sizeof(line), fp) != NULL) {
128 char *eq = strchr(line, '=');
129
130 if (eq == NULL) {
131 continue; /* not a field: ignore */
132 }
133 *eq = '\0';
134 oci_meta_apply(out, line, eq + 1);
135 }
136
137 if (ferror(fp)) {
138 ngx_log_error(NGX_LOG_WARN, log, ngx_errno,
139 "oci: unreadable sidecar \"%s\"", path);
140 (void) fclose(fp);
141 return NGX_ERROR;
142 }
143 (void) fclose(fp);
144
145 if (!oci_meta_describes(out, size, mtime)) {
146 ngx_memzero(out, sizeof(*out));
147 return NGX_DECLINED; /* superseded: derive and re-store */
148 }
149
150 return NGX_OK;
151}

153ngx_int_t
154brix_oci_meta_store(const char *body_path, off_t size, time_t mtime,
155 brix_oci_meta_t *meta, ngx_log_t *log)
156{
157 char final[PATH_MAX];
158 char tmp[PATH_MAX];
159 FILE *fp;
160 int rc;
161
162 meta->size = size;
163 meta->mtime = mtime;
164
165 if (oci_meta_path(body_path, NULL, final, sizeof(final)) != 0
166 || oci_meta_path(body_path, ".tmp", tmp, sizeof(tmp)) != 0)
167 {
168 return NGX_ERROR;
169 }
170
171 fp = fopen(tmp, "we");
172 if (fp == NULL) {
173 /* A read-only or full store must never fail the pull: the sidecar is
174 * a memo, and its absence costs one derivation per hit, not an
175 * error. Log once at INFO so the condition is visible without
176 * turning a healthy cache into a noisy one. */
177 ngx_log_error(NGX_LOG_INFO, log, ngx_errno,
178 "oci: sidecar not written \"%s\"", tmp);
179 return NGX_DECLINED;
180 }
181
182 rc = fprintf(fp,
183 "content_type=%s\n"
184 "digest=%s\n"
185 "etag=%s\n"
186 "fetched_at=%lld\n"
187 "size=%lld\n"
188 "mtime=%lld\n"
189 "verified=%d\n",
190 meta->content_type, meta->digest, meta->etag,
191 (long long) meta->fetched_at, (long long) meta->size,
192 (long long) meta->mtime, meta->verified ? 1 : 0);
193
194 /* fflush before the rename: a rename of a file whose bytes are still in
195 * the stdio buffer publishes an empty record. fclose flushes too, but its
196 * failure would then be indistinguishable from a close failure. */
197 if (rc < 0 || fflush(fp) != 0) {
198 (void) fclose(fp);
199 (void) unlink(tmp); /* vfs-seam-allow: DOMAIN_REGISTRY — sidecar staging file, not object data */
200 return NGX_ERROR;
201 }
202 if (fclose(fp) != 0) {
203 (void) unlink(tmp); /* vfs-seam-allow: DOMAIN_REGISTRY — sidecar staging file, not object data */
204 return NGX_ERROR;
205 }
206
207 if (rename(tmp, final) != 0) { /* vfs-seam-allow: DOMAIN_REGISTRY — sidecar publish, not object data */
208 ngx_log_error(NGX_LOG_WARN, log, ngx_errno,
209 "oci: sidecar rename failed \"%s\"", final);
210 (void) unlink(tmp); /* vfs-seam-allow: DOMAIN_REGISTRY — sidecar staging file, not object data */
211 return NGX_ERROR;
212 }
213
214 return NGX_OK;
215}
