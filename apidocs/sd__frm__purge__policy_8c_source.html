1/*
2 * sd_frm_purge_policy.c — 2.0 F4: per-group purge rules + the external
3 * policy program of the tape-buffer purge engine.
4 *
5 * WHAT: The layer between sd_frm_purge.c's LRU walk and the
6 * brix_frm_purge_policy / brix_frm_purge_polprog directives: tags every
7 * candidate with the rule its brix_oss_space group falls under, gives
8 * each rule its own owned-bytes arm (hi/lo) and hold, and, for rules
9 * marked `polprog`, lets an operator program choose which of the
10 * group's eligible copies may go.
11 *
12 * WHY: frm_purged has `purge.policy {*|space} min max [hold] [polprog]` and
13 * `frm.purge.polprog`; sites use them to keep one VO's hot set while
14 * draining another's. Until this phase the engine had one export-wide
15 * LRU with a watermark pair and a cap (release-2.0 register F4).
16 *
17 * HOW: 1. init: rule_of(key) -> rule index per candidate; sum the owned
18 * bytes per rule; need = owned - lo once owned > hi.
19 * 2. consult: when any polprog rule is under pressure (its own arm or
20 * the export-wide one), write "<group> <touched> <size> <key>" for
21 * every candidate of a polprog rule to <online>/.brix-purge.candidates,
22 * run "<program> <candidates-file> <decision-file>" under the exec
23 * shared reparented runner's deadline, read the decision file back
24 * (one key per line) and mark the matching candidates approved.
25 * Any failure is fail-closed: the polprog rules release nothing this
26 * pass. A key the program names that is not one of ITS candidates is
27 * counted and ignored — the program chooses among eligible copies, it
28 * can never add one. Both files are unlinked afterwards.
29 * 3. The walk asks hold(), wants(), allows() and calls account() per
30 * release; log() writes one line per rule + one for the pass.
31 *
32 * Raw filesystem calls are legitimate here: this TU is under src/fs/backend/
33 * (invariant 12), working on the backend's private buffer directory.
34 */

72int
73frm_purge_policy_init(frm_purge_policy_t *pp,
74 const brix_sd_frm_purge_policy_t *pol, frm_purge_scan_t *s)
75{
76 size_t i;
77
78 memset(pp, 0, sizeof(*pp));
79 if (pol->rules == NULL || pol->nrules == 0) {
80 return 0;
81 }
82 pp->rules = calloc(pol->nrules, sizeof(*pp->rules));
83 if (pp->rules == NULL) {
84 return -1;
85 }
86 pp->n = pol->nrules;
87 for (i = 0; i < s->n; i++) {
88 frm_purge_cand_t *c = &s->v[i];
89
90 c->rule = policy_rule_index(pol, c->rel);
91 if (c->rule >= 0) {
92 pp->rules[c->rule].owned += c->size;
93 }
94 }
95 for (i = 0; i < pp->n; i++) {
96 const brix_sd_frm_purge_rule_t *r = &pol->rules[i];
97 uint64_t owned = pp->rules[i].owned;
98
99 if (owned > r->hi_bytes && owned > r->lo_bytes) {
100 pp->rules[i].need = owned - r->lo_bytes;
101 }
102 }
103 return 0;
104}

172static int
173policy_write_candidates(const brix_sd_frm_purge_policy_t *pol,
174 const frm_purge_scan_t *s, const char *path)
175{
176 FILE *fp;
177 int fd;
178 size_t i;
179
180 (void) unlink(path);
181 fd = open(path, O_WRONLY | O_CREAT | O_EXCL | O_CLOEXEC | O_NOFOLLOW, 0600);
182 if (fd < 0) {
183 return -1;
184 }
185 fp = fdopen(fd, "w");
186 if (fp == NULL) {
187 close(fd);
188 return -1;
189 }
190 for (i = 0; i < s->n; i++) {
191 const frm_purge_cand_t *c = &s->v[i];
192
193 if (!policy_rule_is_polprog(pol, c)) {
194 continue;
195 }
196 fprintf(fp, "%s %" PRId64 " %" PRIu64 " %s\n",
197 pol->rules[c->rule].name, (int64_t) c->touched, c->size,
198 c->rel);
199 }
200 return (fclose(fp) == 0) ? 0 : -1;
201}

203/* "<program> <candidates-file> <decision-file>", no shell, under the shared
204 * reparented runner's deadline. 0 = exited 0, else -1 with *why.
205 *
206 * brix_subprocess_run, not a direct child (2.0, 2026-09-09): nginx's SIGCHLD
207 * handler reaps a worker's children itself, so the policy program's status was
208 * stolen before this code could wait for it and EVERY pass ended "wait failed"
209 * — the groups released nothing, which is the safe direction but made the
210 * feature inert. The runner's agent is the program's only parent; it also
211 * enforces polprog_timeout_ms and SIGKILLs the whole process group on expiry. */
212static int
213policy_run_program(const brix_sd_frm_purge_policy_t *pol, const char *cands,
214 const char *decision, const char **why)
215{
216 char *argv[4];
217 brix_subprocess_req_t req;
218 int exit_code = -1;
219
220 argv[0] = (char *) pol->polprog;
221 argv[1] = (char *) cands;
222 argv[2] = (char *) decision;
223 argv[3] = NULL;
224
225 req.argv = argv;
226 req.out = NULL; /* the verdict is the decision file */
227 req.outsz = 0;
228 req.timeout_ms = (unsigned) pol->polprog_timeout_ms;
229
230 if (brix_subprocess_run(&req, NULL, &exit_code) != 0) {
231 *why = (errno == ETIMEDOUT) ? "deadline exceeded, killed"
232 : "cannot run";
233 return -1;
234 }
235 if (exit_code != 0) {
236 *why = "non-zero exit";
237 return -1;
238 }
239 return 0;
240}

251static frm_purge_cand_t **
252policy_index(const brix_sd_frm_purge_policy_t *pol, frm_purge_scan_t *s,
253 size_t *n_out)
254{
255 frm_purge_cand_t **idx = malloc((s->n ? s->n : 1) * sizeof(*idx));
256 size_t i, n = 0;
257
258 if (idx == NULL) {
259 return NULL;
260 }
261 for (i = 0; i < s->n; i++) {
262 if (policy_rule_is_polprog(pol, &s->v[i])) {
263 idx[n++] = &s->v[i];
264 }
265 }
266 qsort(idx, n, sizeof(*idx), policy_cmp_rel);
267 *n_out = n;
268 return idx;
269}

271static void
272policy_mark_line(frm_purge_policy_t *pp, frm_purge_cand_t **idx, size_t n,
273 char *line)
274{
275 frm_purge_cand_t key, *kp = &key, **hit;
276
277 line[strcspn(line, "\r\n")] = '\0';
278 if (line[0] == '\0') {
279 return;
280 }
281 key.rel = line;
282 hit = bsearch(&kp, idx, n, sizeof(*idx), policy_cmp_rel);
283 if (hit == NULL) {
284 pp->ignored++;
285 return;
286 }
287 (*hit)->approved = 1;
288 pp->approved++;
289}

293static int
294policy_read_decision(frm_purge_policy_t *pp,
295 const brix_sd_frm_purge_policy_t *pol, frm_purge_scan_t *s,
296 const char *path)
297{
298 frm_purge_cand_t **idx;
299 char line[FRM_POLPROG_LINE_MAX];
300 FILE *fp;
301 size_t n;
302 int fd;
303
304 fd = open(path, O_RDONLY | O_CLOEXEC | O_NOFOLLOW);
305 if (fd < 0) {
306 return -1;
307 }
308 fp = fdopen(fd, "r");
309 if (fp == NULL) {
310 close(fd);
311 return -1;
312 }
313 idx = policy_index(pol, s, &n);
314 if (idx == NULL) {
315 fclose(fp);
316 return -1;
317 }
318 while (fgets(line, sizeof(line), fp) != NULL) {
319 policy_mark_line(pp, idx, n, line);
320 }
321 free(idx);
322 fclose(fp);
323 return 0;
324}

326static int
327policy_exchange(frm_purge_policy_t *pp, const brix_sd_frm_purge_policy_t *pol,
328 frm_purge_scan_t *s, const char *cands, const char *decision,
329 const char **why)
330{
331 (void) unlink(decision);
332 if (policy_write_candidates(pol, s, cands) != 0) {
333 *why = "cannot write the candidate list";
334 return -1;
335 }
336 if (policy_run_program(pol, cands, decision, why) != 0) {
337 return -1;
338 }
339 if (policy_read_decision(pp, pol, s, decision) != 0) {
340 *why = "decision file unreadable";
341 return -1;
342 }
343 return 0;
344}

346void
347frm_purge_policy_consult(frm_purge_policy_t *pp,
348 const brix_sd_frm_purge_policy_t *pol, frm_purge_scan_t *s,
349 const char *online, uint64_t need, ngx_log_t *log)
350{
351 char cands[PATH_MAX], decision[PATH_MAX];
352 const char *why = "";
353 int rc;
354
355 if (!policy_needs_program(pp, pol, need)) {
356 pp->polprog = FRM_POLPROG_IDLE;
357 return;
358 }
359 if (policy_private_path(online, FRM_PURGE_CANDIDATES_NAME, cands,
360 sizeof(cands)) != 0
361 || policy_private_path(online, FRM_PURGE_DECISION_NAME, decision,
362 sizeof(decision)) != 0)
363 {
364 pp->polprog = FRM_POLPROG_FAILED;
365 ngx_log_error(NGX_LOG_ERR, log, errno,
366 "brix: tape purge \"%s\": policy program paths do not fit", online);
367 return;
368 }
369 rc = policy_exchange(pp, pol, s, cands, decision, &why);
370 (void) unlink(cands);
371 (void) unlink(decision);
372 if (rc != 0) {
373 pp->polprog = FRM_POLPROG_FAILED;
374 ngx_log_error(NGX_LOG_ERR, log, 0,
375 "brix: tape purge \"%s\": policy program \"%s\" failed (%s); its "
376 "groups release nothing this pass", online, pol->polprog, why);
377 return;
378 }
379 pp->polprog = FRM_POLPROG_OK;
380}

393frm_purge_policy_account(frm_purge_policy_t *pp, const frm_purge_cand_t *c)
394{
395 frm_purge_rule_state_t *r;
396
397 if (c->rule < 0 || (size_t) c->rule >= pp->n) {
398 return;
399 }
400 r = &pp->rules[c->rule];
401 r->released += c->size;
402 r->evicted++;
403 r->need = (r->need > c->size) ? r->need - c->size : 0;

406void
407frm_purge_policy_log(const frm_purge_policy_t *pp,
408 const brix_sd_frm_purge_policy_t *pol, const char *online,
409 const brix_sd_frm_purge_report_t *rep, ngx_log_t *log)
410{
411 static const char *const state[] = { "idle", "ok", "failed" };
412 ngx_uint_t level;
413 size_t i;
414
415 if (pp->n == 0) {
416 return;
417 }
418 for (i = 0; i < pp->n; i++) {
419 const brix_sd_frm_purge_rule_t *r = &pol->rules[i];
420 const frm_purge_rule_state_t *st = &pp->rules[i];
421 time_t hold = (r->hold_s > pol->min_age_s) ? r->hold_s : pol->min_age_s;
422
423 level = st->evicted ? NGX_LOG_NOTICE : NGX_LOG_INFO;
424 ngx_log_error(level, log, 0,
425 "brix: tape purge \"%s\" policy \"%s\": owned %uL -> %uL bytes "
426 "(hi=%uL lo=%uL hold=%T s%s), released %ui file(s), %uL bytes",
427 online, r->name, st->owned, st->owned - st->released, r->hi_bytes,
428 r->lo_bytes, hold, r->polprog ? ", polprog" : "", st->evicted,
429 st->released);
430 }
431 level = (pp->polprog == FRM_POLPROG_FAILED) ? NGX_LOG_ERR : NGX_LOG_INFO;
432 ngx_log_error(level, log, 0,
433 "brix: tape purge \"%s\" policy pass: rules=%uz held=%ui "
434 "unapproved=%ui polprog=%s approved=%ui ignored=%ui",
435 online, pp->n, rep->held, rep->unapproved, state[pp->polprog],
436 pp->approved, pp->ignored);
437}
