10/* WHAT: Open the exact local inode selected by the export's storage owner.
11 * WHY: Explicit POSIX instances must retain their root and N2N translation;
12 * the implicit local path has no instance to supply either.
13 * HOW: Use the confined VFS open for implicit storage, otherwise translate
14 * through the existing N2N seam and ask the POSIX backend for a private fd.
15 */
16static int
17lock_store_open(brix_vfs_ctx_t *ctx)
18{
19 char physical[PATH_MAX];
20
21 if (ctx->sd == NULL) {
22 return brix_vfs_open_fd(ctx->log, ctx->root_canon,
23 brix_vfs_ctx_path(ctx), O_RDONLY | O_NONBLOCK | O_NOFOLLOW, 0);
24 }
25 if (brix_path_resolved_to_pfn(ctx, brix_vfs_ctx_path(ctx), physical,
26 sizeof(physical)) != NGX_OK) {
27 return -1;
28 }

32/* WHAT: Observe one completed local transaction without losing its errno.
33 * WHY: Refusals and failed opens need the same accounting as successful writes.
34 * HOW: Preserve errno, emit the existing VFS xattr observation, then restore it.
35 */
36static ngx_int_t
37lock_store_observe(brix_vfs_ctx_t *ctx, uint64_t started, int result, size_t bytes)
38{
39 int saved_errno = result == 0 ? 0 : errno;
40 brix_vfs_observe_ctx_op(ctx, brix_vfs_ctx_path(ctx), BRIX_METRIC_OP_XATTR,
41 NULL, bytes, result == 0 ? NGX_OK : NGX_ERROR, saved_errno, started);
42 errno = saved_errno;
43 return result == 0 ? NGX_OK : NGX_ERROR;
44}

46/* WHAT: Encode one optional, bounded record or refuse malformed caller input.
47 * WHY: The public VFS boundary must not scan past an unterminated field.
48 * HOW: NULL means no record; check fixed-array bounds before shared encoding.
49 */
50static int
51lock_store_encode(const brix_lock_record_t *record, char *output, size_t capacity)
52{
53 if (record == NULL) {
54 output[0] = '\0';
55 return 0;
56 }
57 if (record->token[0] == '\0'
58 || memchr(record->token, 0, sizeof(record->token)) == NULL
59 || memchr(record->owner, 0, sizeof(record->owner)) == NULL)
60 {
61 errno = EINVAL;
62 return -1;
63 }
64 if (brix_lock_record_encode(record, output, capacity) != NGX_OK) {
65 errno = EOVERFLOW;
66 return -1;
67 }
68 return 0;
69}

71/* WHAT: Preserve the exact snapshot on the expected side of compare/exchange.
72 * WHY: Decoding may normalize a legacy expiry or ignore future fields; encoding
73 * that semantic record would never match the persisted bytes being replaced.
74 * HOW: Copy a bounded, NUL-free observed value; only constructed records encode.
75 * Replacement records always use lock_store_encode, never this snapshot. */
76static int
77lock_store_expected(const brix_lock_record_t *record, char *output, size_t capacity)
78{
79 if (record == NULL || record->observed_length == 0) {
80 return lock_store_encode(record, output, capacity);
81 }
82 if (record->observed_length >= sizeof(record->observed)
83 || record->observed_length >= capacity
84 || memchr(record->observed, '\0', record->observed_length) != NULL)
85 {
86 errno = EINVAL;
87 return -1;
88 }
89 memcpy(output, record->observed, record->observed_length);
90 output[record->observed_length] = '\0';
91 return 0;
92}

94/* WHAT: Open one confined inode and apply the already-encoded transaction.
95 * WHY: Descriptor lifetime and errno must remain independent of protocol code.
96 * HOW: Nonblocking no-follow open, backend CAS, close, restore the verdict.
97 */
98static int
99lock_store_change(brix_vfs_ctx_t *ctx, const char *previous, const char *next)
100{
101 int descriptor = lock_store_open(ctx);
102 int result, saved_errno;
103
104 if (descriptor < 0) {
105 return -1;
106 }
107 result = brix_posix_lock_store_update_fd(descriptor,
108 previous[0] == '\0' ? NULL : previous, strlen(previous),
109 next[0] == '\0' ? NULL : next, strlen(next));
110 saved_errno = result == 0 ? 0 : errno;
111 (void) close(descriptor);
112 errno = saved_errno;
113 return result;
114}

116/* WHAT: Execute the exact encoded transaction in the mapped owner's helper.
117 * WHY: A worker-held fd does not grant the owner's xattr write credentials.

137/* WHAT: Change a local lock record after the normal VFS mutation gates.
138 * WHY: Protocol code must not implement its own storage, authorization or
139 * inode selection. The native xattr PAL intentionally keeps rejecting flags
140 * which its kernel cannot implement for arbitrary, non-cooperating writers.
141 * HOW: Gate first; route mapped local operations to the broker, otherwise
142 * confined-open a fresh fd; execute the serialized backend transaction;
143 * observe and preserve errno.
144 */
145ngx_int_t
146brix_vfs_lock_store_update(brix_vfs_ctx_t *ctx,
147 const brix_lock_record_t *expected, const brix_lock_record_t *replacement)
148{
149 char previous[BRIX_LOCK_XATTR_MAXLEN], next[BRIX_LOCK_XATTR_MAXLEN];
150 uint64_t started = brix_vfs_now_ns();
151 int result;
152
153 if (brix_vfs_require_confined(ctx) != NGX_OK
154 || brix_vfs_gate_mutation(ctx, BRIX_VFS_MUTATE_XATTR) != NGX_OK
155 || brix_vfs_require_unlocked(ctx, BRIX_VFS_MUTATE_XATTR) != NGX_OK)
156 {
157 return lock_store_observe(ctx, started, -1, 0);
158 }
159 /* Match the existing xattr dispatch: POSIX has no backend-secret slots.
160 * A default credential-directory pointer alone does not make it remote.
161 * Broker identity and non-POSIX credential dispatch must stay authoritative. */
162 if (brix_vfs_ctx_driver(ctx) != NULL) {
163 errno = ENOTSUP;
164 return NGX_DECLINED;
165 }
166 if (expected == NULL && replacement == NULL) {
167 errno = EINVAL;
168 return lock_store_observe(ctx, started, -1, 0);
169 }
170 if (lock_store_expected(expected, previous, sizeof(previous)) != 0
171 || lock_store_encode(replacement, next, sizeof(next)) != 0) {
172 return lock_store_observe(ctx, started, -1, 0);
173 }
174 result = brix_imp_client_active()
175 ? lock_store_broker_change(ctx, previous, next)
176 : lock_store_change(ctx, previous, next);
177 return lock_store_observe(ctx, started, result, strlen(next));
178}
