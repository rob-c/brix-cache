1/*
2 * backend_async_queue.c — durable write-behind queue for backend namespace
3 * mutations (brix_backend_async). See backend_async_queue.h for the contract.
4 *
5 * WHAT: Per-worker coalescing + durable journalling of unlink/rmdir/rename/mkdir,
6 * with in-bulk drain against the backend and a protocol-supplied waker.
7 *
8 * WHY: Trade per-op latency for bulk backend efficiency without ever losing an
9 * acknowledged mutation across a crash. Split from the stage engine on
10 * purpose: a namespace mutation is not a byte transfer.
11 *
12 * HOW: A fixed-size pending array (per worker), each row carrying the durable
13 * record + the adapter's completion callback + coalescing thresholds. The
14 * size trigger fires inline in enqueue; the time trigger fires on the worker
15 * timer tick. Durability reuses the stage engine's journal DIRECTORY root
16 * (stage_journal_dir) but a private "backend/" subdir and a private record
17 * type + reconcile, so the two subsystems never scan each other's records.
18 */
19

36/* The worker's pending batch. Bounded: at capacity, enqueue drains first so a
37 * burst never grows unboundedly (and never silently drops). 4096 rows is far
38 * above any sane coalescing batch while staying a small fixed footprint. */
39#define BRIX_BAQ_MAX_PENDING 4096
40
41typedef struct {
42 brix_baq_rec_t rec; /* the durable record (also what we journal) */
43 brix_baq_done_pt done; /* adapter waker (NULL = replayed, no client) */
44 void *client; /* opaque conn/request token for `done` */
45 ngx_uint_t batch; /* this op's size trigger */
46 ngx_msec_t wait_ms; /* this op's time trigger */
47 ngx_msec_t enqueued_ms; /* monotonic enqueue stamp (time trigger) */
48 /* phase-105: the write posture and protocol of the endpoint that enqueued
49 * this op, captured by value at enqueue. The drain re-checks it rather than
50 * re-reading configuration the request no longer owns and a reload may
51 * since have flipped (Appendix D.8). */
52 brix_vfs_mutation_policy_t policy;
53 brix_proto_t proto;
54} brix_baq_pending_t;

93/* fsync a QUEUED record to <baq_journal_dir>/<reqid>.req. Best-effort: a failure
94 * costs durability for that one op, never correctness of the in-memory drain. */
95static void
96baq_journal_write(const brix_baq_rec_t *rec)
97{
98 char path[1300];
99 int fd;
100
101 if (baq_journal_dir[0] == '\0') {
102 return;
103 }
104 if ((size_t) snprintf(path, sizeof(path), "%s/%s.req",
105 baq_journal_dir, rec->reqid) >= sizeof(path))
106 {
107 return;
108 }
109 fd = open(path, O_WRONLY | O_CREAT | O_TRUNC | O_CLOEXEC, 0600);
110 if (fd < 0) {
111 return;
112 }
113 if (write(fd, rec, sizeof(*rec)) == (ssize_t) sizeof(*rec)) {
114 (void) fsync(fd);
115 }
116 (void) close(fd);
117}

136/*
137 * Drive ONE record against its backend. Returns the raw POSIX errno (0 == ok).
138 *
139 * WHAT: Dispatches on rec->op to the matching pool-less VFS path primitive, all
140 * of which confine to root_canon and are safe to call inline on the event
141 * loop (no pool alloc, no metric).
142 *
143 * WHY: Shared by the live drain (which delivers the raw errno to the client as
144 * the true result) and reconcile (which additionally squashes idempotent-
145 * benign errnos — see baq_apply_idempotent).
146 *
147 * HOW: RENAME needs the resolved backend instance (brix_vfs_rename_path takes an
148 * sd); the others take root_canon directly. Every primitive is the
149 * phase-105 policy-bearing export form, driven from `opctx` — the drain
150 * re-checks the posture the enqueuing endpoint captured (Appendix D.8's
151 * "worker/drain validates record and policy again") instead of re-reading
152 * a configuration this op no longer belongs to. A refusal surfaces as
153 * EROFS and reaches the parked client like any other backend errno.
154 */
155static int
156baq_apply(const brix_baq_rec_t *rec, const brix_vfs_export_op_ctx_t *opctx)
157{
158 switch ((brix_baq_op_t) rec->op) {
159
160 case BRIX_BAQ_UNLINK:
161 if (brix_vfs_export_unlink(opctx, rec->src_key) == 0) {
162 return 0;
163 }
164 /* kXR_rm parity: "unlink a file, rmdir a directory" non-recursively. The
165 * file-unlink primitive returns EISDIR on a directory target, so fall back
166 * to the empty-dir removal (which itself returns ENOTEMPTY on a non-empty
167 * dir) — matching brix_vfs_unlink's file-or-empty-dir behaviour exactly. */
168 if (errno == EISDIR
169 && brix_vfs_export_rmdir(opctx, rec->src_key) == 0)
170 {
171 return 0;
172 }
173 return errno ? errno : EIO;
174
175 case BRIX_BAQ_RMDIR:
176 return brix_vfs_export_rmdir(opctx, rec->src_key) == 0
177 ? 0 : (errno ? errno : EIO);
178
179 case BRIX_BAQ_MKDIR:
180 return brix_vfs_export_mkdir(opctx, rec->src_key,
181 (mode_t) rec->mode) == 0
182 ? 0 : (errno ? errno : EIO);
183
184 case BRIX_BAQ_RENAME: {
185 brix_sd_instance_t *sd = brix_vfs_backend_resolve(rec->root_canon,
186 opctx->log);
187
188 errno = 0;
189 if (brix_vfs_export_rename(opctx, sd, rec->src_key, rec->dst_key,
190 0 /* no overwrite */, NULL) == NGX_OK)
191 {
192 return 0;
193 }
194 return errno ? errno : EIO;
195 }
196
197 default:
198 return EINVAL; /* unknown op = corrupt record */
199 }
200}

212/* Reconcile-time idempotency: a mutation the client was told to wait for has
213 * already taken effect if the target is now in the desired end-state. Squash the
214 * benign "already done" errnos to success so the replay removes the record; any
215 * other errno is a real transient failure and the record is kept for a later try. */
216static int
217baq_apply_idempotent(const brix_baq_rec_t *rec,
218 const brix_vfs_export_op_ctx_t *opctx)
219{
220 int e = baq_apply(rec, opctx);
221
222 switch ((brix_baq_op_t) rec->op) {
223 case BRIX_BAQ_UNLINK:
224 case BRIX_BAQ_RMDIR:
225 return (e == ENOENT) ? 0 : e; /* already gone = done */
226 case BRIX_BAQ_MKDIR:
227 return (e == EEXIST) ? 0 : e; /* already present = done */
228 case BRIX_BAQ_RENAME:
229 return (e == ENOENT) ? 0 : e; /* source already moved = done */
230 default:
231 return e;
232 }
233}

237/* Drain the WHOLE batch: apply each op, deliver its real errno to its client, drop
238 * its durable record. Clears the pending list. Live path (has clients to wake). */
239static void
240baq_drain_all(void)
241{
242 ngx_log_t *log = (ngx_cycle != NULL) ? ngx_cycle->log : NULL;
243 ngx_uint_t i;
244
245 for (i = 0; i < baq_count; i++) {
246 brix_baq_pending_t *p = &baq_pending[i];
247 brix_vfs_export_op_ctx_t opctx;
248 int e;
249
250 baq_opctx(&opctx, &p->rec, log, p->policy, p->proto);
251 e = baq_apply(&p->rec, &opctx);
252
253 baq_journal_remove(p->rec.reqid);
254 if (p->done != NULL) {
255 p->done(p->client, e); /* wake the parked client (real result) */
256 }
257 }
258 baq_count = 0;
259}

261/*
262 * Deferred-drain event: the size trigger posts this instead of draining inline.
263 *
264 * WHAT: An nginx posted event whose handler runs baq_drain_all() after the current
265 * stack unwinds (processed from ngx_posted_events by the event loop).
266 *
267 * WHY: A batch of 1 (or any op that completes the batch) would otherwise drain
268 * INSIDE brix_baq_enqueue, firing the done() callback for the very client the
269 * caller is still in the middle of parking — the reply would race the caller's
270 * own state transition. Deferring guarantees every parked connection has fully
271 * reached its wait state before its waker fires.
272 *
273 * HOW: ngx_post_event is idempotent via ev->posted, so multiple enqueues in one
274 * loop iteration coalesce into a single drain.
275 */
276static void
277baq_drain_event_handler(ngx_event_t *ev)
278{
279 (void) ev;
280 baq_drain_all();
281}

283static void
284baq_post_drain(void)
285{
286 if (baq_drain_ev.handler == NULL) {
287 baq_drain_ev.handler = baq_drain_event_handler;
288 baq_drain_ev.data = NULL;
289 baq_drain_ev.log = (ngx_cycle != NULL) ? ngx_cycle->log : NULL;
290 }
291 ngx_post_event((&baq_drain_ev), (&ngx_posted_events));
292}

294/* Would the batch flush on the SIZE trigger? True once the queue depth reaches the
295 * smallest per-op batch threshold currently parked (a shared worker may hold ops
296 * from servers configured with different batch sizes). */
297static ngx_uint_t
298baq_size_triggered(void)
299{
300 ngx_uint_t i, min_batch = 0;
301
302 if (baq_count == 0) {
303 return 0;
304 }
305 for (i = 0; i < baq_count; i++) {
306 if (min_batch == 0 || baq_pending[i].batch < min_batch) {
307 min_batch = baq_pending[i].batch;
308 }
309 }
310 return (min_batch != 0 && baq_count >= min_batch);
311}
312
313/* Would the batch flush on the TIME trigger? True once any parked op has waited at
314 * least its own wait_ms (ngx_current_msec is the worker's monotonic clock). */
315static ngx_uint_t
316baq_time_triggered(void)
317{
318 ngx_uint_t i;
319
320 for (i = 0; i < baq_count; i++) {
321 if (baq_pending[i].wait_ms == 0) {
322 return 1; /* 0ms wait = flush at the next tick */
323 }
324 if (ngx_current_msec - baq_pending[i].enqueued_ms
325 >= baq_pending[i].wait_ms)
326 {
327 return 1;
328 }
329 }
330 return 0;
331}

333/* Map a queued verb onto the phase-105 mutation vocabulary, so a refusal is
334 * counted under the operation the client actually asked for. */
335static brix_vfs_mutation_op_t
336baq_mutation_op(brix_baq_op_t op)
337{
338 switch (op) {
339 case BRIX_BAQ_RMDIR: return BRIX_VFS_MUTATE_REMOVE;
340 case BRIX_BAQ_RENAME: return BRIX_VFS_MUTATE_RENAME;

349ngx_int_t
350brix_baq_enqueue(const brix_baq_req_t *req, brix_baq_done_pt done, void *client)
351{
352 brix_baq_pending_t *p;
353
354 if (req == NULL || req->root_canon == NULL || req->src_key == NULL
355 || done == NULL)
356 {
357 return NGX_ERROR;
358 }
359
360 /* phase-105: the gate runs HERE, before the journal record is written and
361 * before the client is parked. Deferring a mutation does not soften it — a
362 * read-only endpoint must not be able to leave an unlink sitting in a
363 * durable journal that a later replay would apply. The refusal is EROFS and
364 * is counted once; the caller falls back to running the op inline, where it
365 * meets the same refusal from the VFS and reports it to the client. */
366 if (brix_vfs_require_carried_mutation(req->policy, req->proto,
367 baq_mutation_op(req->op)) != NGX_OK)
368 {
369 return NGX_ERROR;
370 }
371
372 /* Reject any path that would not fit the fixed record fields rather than
373 * silently truncating it — a truncated path at drain would target the WRONG
374 * object. The caller falls back to running the op inline. */
375 if (strlen(req->root_canon) >= sizeof(baq_pending[0].rec.root_canon)
376 || strlen(req->src_key) >= sizeof(baq_pending[0].rec.src_key)
377 || (req->dst_key != NULL
378 && strlen(req->dst_key) >= sizeof(baq_pending[0].rec.dst_key)))
379 {
380 return NGX_ERROR;
381 }
382
383 /* At capacity: drain now to make room. If a full-queue drain still leaves us
384 * at capacity (should not happen — drain empties it), fail so the caller runs
385 * the op inline rather than overwrite a parked row. */
386 if (baq_count >= BRIX_BAQ_MAX_PENDING) {
387 baq_drain_all();
388 }
389 if (baq_count >= BRIX_BAQ_MAX_PENDING) {
390 return NGX_ERROR;
391 }
392
393 p = &baq_pending[baq_count];
394 ngx_memzero(&p->rec, sizeof(p->rec));
395 baq_reqid_mint(p->rec.reqid);
396 p->rec.op = (uint32_t) req->op;
397 snprintf(p->rec.root_canon, sizeof(p->rec.root_canon), "%s",
398 req->root_canon);
399 snprintf(p->rec.src_key, sizeof(p->rec.src_key), "%s", req->src_key);
400 if (req->dst_key != NULL) {
401 snprintf(p->rec.dst_key, sizeof(p->rec.dst_key), "%s", req->dst_key);
402 }
403 p->rec.mode = req->mode;
404 p->rec.enqueued_at = (int64_t) time(NULL);
405 p->done = done;
406 p->client = client;
407 p->batch = (req->batch == 0) ? 1 : req->batch;
408 p->wait_ms = req->wait_ms;
409 p->enqueued_ms = ngx_current_msec;
410 p->policy = req->policy; /* phase-105: captured, not re-read */
411 p->proto = req->proto;
412
413 baq_journal_write(&p->rec); /* durable BEFORE the client parks */
414 baq_count++;
415
416 /* Size trigger: post a deferred drain rather than draining inline, so this op's
417 * waker never fires before the caller has finished parking its client (a batch
418 * of 1 still behaves like a synchronous mutation with a durable record, just one
419 * event-loop hop later). */
420 if (baq_size_triggered()) {
421 baq_post_drain();
422 }
423 return NGX_OK;
424}

434void
435brix_baq_drop_client(void *client)
436{
437 ngx_uint_t i = 0;
438
439 /* Compact the array, dropping rows for `client`. Their durable records are
440 * intentionally LEFT on disk so an un-drained mutation is replayed at boot. */
441 while (i < baq_count) {
442 if (baq_pending[i].client == client) {
443 baq_pending[i] = baq_pending[baq_count - 1];
444 baq_count--; /* re-test the swapped-in row */
445 continue;
446 }
447 i++;
448 }
449}

453/* Re-apply ONE persisted record idempotently; remove it on success. Returns
454 * 1 replayed / 0 kept (transient failure) / -1 dropped (corrupt). */
455static int
456baq_reconcile_one(const char *path, ngx_log_t *log)
457{
458 brix_baq_rec_t rec;
459 char rbuf[sizeof(brix_baq_rec_t)];
460 int fd, e;
461 ssize_t n;
462
463 fd = open(path, O_RDONLY | O_CLOEXEC);
464 if (fd < 0) {
465 return 0;
466 }
467 n = read(fd, rbuf, sizeof(rbuf));
468 (void) close(fd);
469
470 if (n != (ssize_t) sizeof(rec)) {
471 (void) unlink(path); /* short/corrupt/oversized — drop */
472 return -1;
473 }
474 ngx_memcpy(&rec, rbuf, sizeof(rec));
475 if (rec.op > BRIX_BAQ_MKDIR) {
476 (void) unlink(path); /* unknown op — drop */
477 return -1;
478 }
479
480 {
481 brix_vfs_export_op_ctx_t opctx;
482
483 /* phase-105: crash reconcile runs at worker start, with no request and
484 * no endpoint configuration in scope, so the posture cannot be re-read.
485 * The record itself is the authority: brix_baq_enqueue is its only
486 * writer and refuses a read-only endpoint, so a record on disk is proof
487 * that a WRITABLE endpoint authorised exactly this mutation and that a
488 * client was told to wait for it. Re-deciding here would turn an
489 * already-accepted mutation into a silent loss across a reload — the
490 * one outcome this journal exists to prevent (Appendix D.8). The proto
491 * label never reaches a metric, because ALLOWED never refuses. */
492 baq_opctx(&opctx, &rec, log, BRIX_VFS_MUTATION_ALLOWED,
493 BRIX_PROTO_ROOT);
494 e = baq_apply_idempotent(&rec, &opctx);
495 }
496 if (e == 0) {
497 (void) unlink(path);
498 return 1;
499 }
500 ngx_log_error(NGX_LOG_WARN, log, e,
501 "xrootd backend-async: reconcile of \"%s\" (op %uD root \"%s\") kept "
502 "for retry", rec.src_key, rec.op, rec.root_canon);
503 return 0;
504}

506/*
507 * WHAT: Snapshot bounded asynchronous-journal request names before replay.
508 * WHY: Reconciliation unlinks entries, so directory iteration cannot drive it live.
509 * HOW: Copy only well-formed .req basenames into the caller's fixed table.
510 */
511static ngx_uint_t
512baq_snapshot_names(char names[][256], ngx_uint_t capacity)
513{
514 DIR *d;
515 struct dirent *de;
516 ngx_uint_t count = 0;
517
518 d = opendir(baq_journal_dir);
519 if (d == NULL)
520 return 0;
521 while ((de = readdir(d)) != NULL && count < capacity) {
522 size_t nlen = strlen(de->d_name);
523
524 if (nlen <= 4 || nlen >= 256 ||
525 strcmp(de->d_name + nlen - 4, ".req") != 0)
526 continue;
527 memcpy(names[count], de->d_name, nlen + 1);
528 count++;
529 }
530 closedir(d);
531 return count;
532}

534/*
535 * WHAT: Replay a snapshot of asynchronous journal requests and count outcomes.
536 * WHY: Startup reporting distinguishes successful, corrupt, and retryable records.
537 * HOW: Build bounded paths, reconcile each record, and increment its result bucket.
538 */
539static void
540baq_replay_names(char names[][256], ngx_uint_t count, ngx_log_t *log,
541 ngx_uint_t *replayed, ngx_uint_t *kept, ngx_uint_t *dropped)
542{
543 for (ngx_uint_t i = 0; i < count; i++) {
544 char path[1400];
545 int result;
546
547 if ((size_t) snprintf(path, sizeof(path), "%s/%s",
548 baq_journal_dir, names[i]) >= sizeof(path))
549 continue;
550 result = baq_reconcile_one(path, log);
551 if (result > 0)
552 (*replayed)++;
553 else if (result < 0)
554 (*dropped)++;
555 else
556 (*kept)++;
557 }
558}

560void
561brix_baq_reconcile(void)
562{
563 char names[1024][256];
564 ngx_uint_t count;
565 ngx_uint_t replayed = 0, kept = 0, dropped = 0;
566 ngx_log_t *log;
567
568 if (baq_journal_dir[0] == '\0' || ngx_cycle == NULL)
569 return;
570 log = ngx_cycle->log;
571 count = baq_snapshot_names(names, 1024);
572 baq_replay_names(names, count, log, &replayed, &kept, &dropped);
573
574 if (replayed > 0 || kept > 0 || dropped > 0) {
575 ngx_log_error(NGX_LOG_NOTICE, log, 0,
576 "xrootd backend-async: restart reconcile — %ui mutation(s) replayed, "
577 "%ui kept for retry, %ui dropped", replayed, kept, dropped);
578 }
