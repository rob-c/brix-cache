1/*
2 * sd_frm_arc.c — dataset archiver adapter: key grammar + vtable (phase-115 W3.1)
3 *
4 * WHAT: the decorator's brix_mss_adapter_t slots. Anything at or above the
5 * dataset depth passes straight to the wrapped adapter; members and the
6 * completion marker get the archive semantics described in sd_frm_arc.h.
7 *
8 * WHY: the frm driver (sd_frm.c) speaks only the adapter vtable, so wrapping
9 * the adapter gives every tape dialect the archiver without touching the
10 * driver's open/recall/staged-write paths.
11 *
12 * HOW: arc_classify() splits a key into (kind, dataset, member). Residency of a
13 * member answers from the index sidecar (OFFLINE, size known) so stat and
14 * dirlist never recall; recall_begin/recall_poll recall the ARCHIVE via
15 * the inner adapter and extract one member (sd_frm_arc_store.c);
16 * migrate(marker) defers the seal to the driver's backup queue and
17 * create_online refuses mutations of a sealed, sealing or reserved
18 * name -- the write side lives in sd_frm_arc_seal.c (2.0 F3).
19 */

34arc_has_reserved_suffix(const char *key)
35{

41/* Walk `key`'s components: returns the count, and *cut = first byte after the
42 * `depth`-th component (NULL when the key is shorter than that). */
43static unsigned
44arc_components(const char *key, unsigned depth, const char **cut)
45{
46 const char *p = key;
47 unsigned n = 0;
48
49 *cut = NULL;
50 while (*p == '/') {
51 p++;
52 }
53 while (*p != '\0') {
54 const char *e = strchr(p, '/');
55
56 n++;
57 if (n == depth) {
58 *cut = (e != NULL) ? e : p + strlen(p);
59 }
60 if (e == NULL) {
61 break;
62 }
63 for (p = e; *p == '/'; p++) { /* collapse "//" */ }
64 }
65 return n;
66}

68arc_kind_t
69arc_classify(const arc_ctx_t *c, const char *key, arc_key_t *out)
70{
71 const char *cut, *member;
72 unsigned n;
73
74 memset(out, 0, sizeof(*out));
75 out->key = key;
76 if (key == NULL) {
77 return ARC_KEY_PLAIN;
78 }
79 if (arc_has_reserved_suffix(key)) {
80 out->kind = ARC_KEY_RESERVED;
81 return out->kind;
82 }
83 n = arc_components(key, c->depth, &cut);
84 if (n <= c->depth || cut == NULL) {
85 return ARC_KEY_PLAIN;
86 }
87 if (snprintf(out->ds, sizeof(out->ds), "%s%.*s", (key[0] == '/') ? "" : "/",
88 (int) (cut - key), key) >= (int) sizeof(out->ds))
89 {
90 return ARC_KEY_PLAIN; /* absurdly long dataset name: per-file */
91 }
92 for (member = cut; *member == '/'; member++) { /* skip the separator */ }
93 out->member = member;
94 out->kind = (strcmp(member, BRIX_ARC_MARKER) == 0) ? ARC_KEY_MARKER : ARC_KEY_MEMBER;
95 return out->kind;
96}

138static int
139arc_residency(void *mss, const char *key, off_t *size_out, time_t *mtime_out)
140{
141 arc_ctx_t *c = mss;
142 arc_key_t k;
143 brix_zip_entry_t e;
144 time_t sealed = 0;
145 int r = c->inner->residency(c->ictx, key, size_out, mtime_out);
146
147 if (r != BRIX_RESIDENCY_ABSENT || arc_classify(c, key, &k) != ARC_KEY_MEMBER) {
148 return r;
149 }
150 r = arc_sidecar_lookup(c, k.ds, k.member, &e, &sealed);
151 if (r == 0) {
152 return BRIX_RESIDENCY_ABSENT; /* sealed dataset, no such member */
153 }
154 if (r < 0) {
155 if (errno != ENOENT || !arc_archive_exists(c, &k)) {
156 return BRIX_RESIDENCY_ABSENT;
157 }
158 e.size = 0; /* index lost: rebuilt on recall */
159 sealed = time(NULL);
160 }
161 if (size_out != NULL) {
162 *size_out = (off_t) e.size;
163 }
164 if (mtime_out != NULL) {
165 *mtime_out = sealed;
166 }
167 return BRIX_RESIDENCY_OFFLINE;
168}

184/* Make the dataset's archive online: 1 online (extract now), 0 recall in
185 * flight, -1 error (errno; ENOENT when the archive does not exist at all). */
186static int
187arc_archive_online(const arc_ctx_t *c, const arc_key_t *k, int begin)
188{
189 char akey[PATH_MAX];
190 int r;
191
192 if (arc_archive_key(k, akey, sizeof(akey)) != 0) {
193 errno = ENAMETOOLONG;
194 return -1;
195 }
196 r = c->inner->residency(c->ictx, akey, NULL, NULL);
197 if (r == BRIX_RESIDENCY_ONLINE) {
198 return 1;
199 }
200 if (r == BRIX_RESIDENCY_ABSENT) {
201 errno = ENOENT;
202 return -1;
203 }
204 if (begin && c->inner->recall_begin(c->ictx, akey) != 0) {
205 return -1;
206 }
207 return c->inner->recall_poll(c->ictx, akey);
208}

210static int
211arc_recall_begin(void *mss, const char *key)
212{
213 arc_ctx_t *c = mss;
214 arc_key_t k;
215 int r;
216
217 if (arc_classify(c, key, &k) != ARC_KEY_MEMBER
218 || c->inner->residency(c->ictx, key, NULL, NULL) != BRIX_RESIDENCY_ABSENT)
219 {
220 return c->inner->recall_begin(c->ictx, key); /* plain key or a per-file copy */
221 }
222 if (!arc_member_recallable(c, &k)) {
223 errno = ENOENT;
224 return -1;
225 }
226 r = arc_archive_online(c, &k, 1);
227 if (r <= 0) {
228 return r; /* 0: archive still staging */
229 }
230 return arc_extract(c, &k);
231}

233static int
234arc_recall_poll(void *mss, const char *key)
235{
236 arc_ctx_t *c = mss;
237 arc_key_t k;
238 int r;
239
240 if (arc_classify(c, key, &k) != ARC_KEY_MEMBER) {
241 return c->inner->recall_poll(c->ictx, key);
242 }
243 r = c->inner->residency(c->ictx, key, NULL, NULL);
244 if (r == BRIX_RESIDENCY_ONLINE) {
245 return 1;
246 }
247 if (r != BRIX_RESIDENCY_ABSENT) {
248 return c->inner->recall_poll(c->ictx, key); /* a legacy per-file copy */
249 }
250 r = arc_archive_online(c, &k, 0);
251 if (r != 1) {
252 return r;
253 }
254 return (arc_extract(c, &k) == 0) ? 1 : -1;
255}

268/* Durable-copy probe for the purge engine: a member is releasable only once
269 * its dataset is sealed AND the archive itself is on tape. */
270static int
271arc_on_tape(void *mss, const char *key)
272{
273 arc_ctx_t *c = mss;
274 arc_key_t k;
275 brix_zip_entry_t e;
276 char akey[PATH_MAX];
277 int inner_can = (c->inner->on_tape != NULL);
278
279 if (arc_classify(c, key, &k) != ARC_KEY_MEMBER) {
280 return inner_can ? c->inner->on_tape(c->ictx, key) : 1;
281 }
282 if (inner_can && c->inner->on_tape(c->ictx, key) == 1) {
283 return 1; /* legacy per-file copy */
284 }
285 if (arc_sidecar_lookup(c, k.ds, k.member, &e, NULL) != 1
286 || arc_archive_key(&k, akey, sizeof(akey)) != 0)
287 {
288 return 0; /* unsealed: the only copy */
289 }
290 return inner_can ? c->inner->on_tape(c->ictx, akey) : 1;
291}

319static int
320arc_list_seen(const arc_list_t *l, const char *name)
321{
322 unsigned i;
323
324 for (i = 0; i < l->n; i++) {
325 if (strcmp(l->seen[i], name) == 0) {
326 return 1;
327 }
328 }
329 return 0;
330}

332static int
333arc_list_inner_cb(void *ud, const char *name, int is_dir)
334{
335 arc_list_t *l = ud;
336 char *copy;
337
338 if (l->n == l->cap) {
339 unsigned ncap = l->cap ? l->cap * 2 : 32;
340 char **grown = realloc(l->seen, ncap * sizeof(*grown));
341
342 if (grown == NULL) {
343 return 1;
344 }
345 l->seen = grown;
346 l->cap = ncap;
347 }
348 /* Ownership transfers INTO the seen list; arc_list_free releases every
349 * entry. A failed strdup only costs this name its dedup slot, so the
350 * callback still runs — the walk is not the allocation's hostage.
351 *
352 * gcc 11's -fanalyzer cannot see a symbolic-index store through a
353 * parameter as an escape, so it reports the copy leaking at the return
354 * below. Suppress exactly that diagnostic, at exactly its emission point. */
355#pragma GCC diagnostic push
356#if defined(__GNUC__) && !defined(__clang__)
357#pragma GCC diagnostic ignored "-Wanalyzer-malloc-leak"
358#endif
359 copy = strdup(name);
360 if (copy != NULL) {
361 l->seen[l->n] = copy;
362 l->n++;
363 }
364 l->stopped = l->cb(l->ud, name, is_dir);
365 return l->stopped;
366#pragma GCC diagnostic pop
367}

369/* One sidecar member name → its first component, emitted once (the sidecar is
370 * sorted, so a directory's members are contiguous and `last_dir` dedupes). */
371static int
372arc_list_member_cb(void *ud, const char *name)
373{
374 arc_list_t *l = ud;
375 const char *slash = strchr(name, '/');
376 char first[BRIX_ZIP_NAME_MAX];
377 size_t n = (slash != NULL) ? (size_t) (slash - name) : strlen(name);
378
379 if (n == 0 || n >= sizeof(first)) {
380 return 0;
381 }
382 memcpy(first, name, n);
383 first[n] = '\0';
384 if ((slash != NULL && strcmp(first, l->last_dir) == 0) || arc_list_seen(l, first)) {
385 return 0;
386 }
387 if (slash != NULL) {
388 memcpy(l->last_dir, first, n + 1);
389 }
390 l->stopped = l->cb(l->ud, first, slash != NULL);
391 return l->stopped;
392}

405static int
406arc_list(void *mss, const char *key, int (*cb)(void *ud, const char *name, int is_dir), void *ud)
407{
408 arc_ctx_t *c = mss;
409 arc_list_t l;
410 int rc = 0;
411
412 if (!arc_is_dataset_dir(c, key)) {
413 if (c->inner->list == NULL) {
414 errno = ENOTSUP;
415 return -1;
416 }
417 return c->inner->list(c->ictx, key, cb, ud);
418 }
419 memset(&l, 0, sizeof(l));
420 l.cb = cb;
421 l.ud = ud;
422 if (c->inner->list != NULL) {
423 rc = c->inner->list(c->ictx, key, arc_list_inner_cb, &l);
424 }
425 if (rc != 0 && (errno != ENOENT || !arc_sealed(c, key))) {
426 arc_list_free(&l); /* neither a tape directory nor sealed */
427 return -1;
428 }
429 if (!l.stopped) {
430 rc = arc_sidecar_names(c, key, arc_list_member_cb, &l);
431 rc = (rc != 0 && errno == ENOENT) ? 0 : rc; /* unsealed: inner only */
432 }
433 arc_list_free(&l);
434 return rc;
435}

516/* One "key=value" of the query: only `arc=<1..8>` exists today. */
517static int
518arc_query_pair(const char *kv, size_t n, brix_sd_frm_opts_t *opts)
519{
520 char *end;
521 unsigned long v;
522
523 if (n <= 4 || strncmp(kv, "arc=", 4) != 0 || !isdigit((unsigned char) kv[4])) {
524 return -1;
525 }
526 v = strtoul(kv + 4, &end, 10);
527 if (end != kv + n || v < 1 || v > BRIX_ARC_MAX_DEPTH) {
528 return -1;
529 }
530 opts->arc_depth = (unsigned) v;
531 return 0;
532}

534int
535brix_sd_frm_parse_query(const char *q, brix_sd_frm_opts_t *opts)
536{
537 memset(opts, 0, sizeof(*opts));
538 while (q != NULL && *q != '\0') {
539 const char *amp = strchr(q, '&');
540 size_t n = (amp != NULL) ? (size_t) (amp - q) : strlen(q);
541
542 if (arc_query_pair(q, n, opts) != 0) {
543 errno = EINVAL;
544 return -1;
545 }
546 q = (amp != NULL) ? amp + 1 : q + n;
547 }
548 return 0;
