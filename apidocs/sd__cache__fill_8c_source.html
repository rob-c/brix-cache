2 * sd_cache_fill.c — the whole-file fill spine of the read-through cache.
3 *
4 * WHAT: Pulls a COMPLETE object from the wrapped source into the cache store and
5 * records its cinfo. Owns the four fill phases (acquire -> pump -> verify
6 * -> commit) and the sd_cache_fill orchestrator the interposed read-open
7 * and the async-offload entrypoint both drive.
8 *
9 * WHY: Split from sd_cache.c (phase-79) to keep every cache file under the
10 * ~500-line, one-concept-per-file cap. The fill spine is the single path
11 * that populates the cache; isolating it keeps the origin-read / stale-serve
12 * / verify / cinfo-publish logic reviewable apart from the vtable adapters
13 * (sd_cache.c), the slice/partial machinery (sd_cache_partial.c), and the
14 * namespace/staged forwarders (sd_cache_forward.c).
15 *
16 * HOW: cache_fill_acquire (source open + admission + staged fill-open) -> pump
17 * (origin read loop) -> verify (phase-68 cvmfs-cas) -> commit (publish +

19 * fill-key) through sd_cache_internal.h. The admission/policy +
20 * repo-metrics helpers it calls live in sd_cache_policy.c. Phase-85 F7:
21 * the same spine runs the cold-tier PROMOTE (cold store as the fill
22 * source, from_cold=1). The integrity-verify phase is split into
23 * sd_cache_fill_verify.c and the eviction-seam hot->cold copy
24 * (brix_sd_cache_demote) into sd_cache_fill_demote.c.

56/* Acquire everything a fill needs: source object, stat snapshot, staged store
57 * object, and the move buffer.
58 *
59 * WHAT: Opens the source for `key`, snapshots its stat, checks admission, and
60 * opens the staged cache-store object (which takes the fill lock — the
61 * O_EXCL + dead-owner-reclaim logic lives in the cstore layer, frozen).
62 *
63 * WHY: The acquire phase concentrates all the "can this fill happen at all?"
64 * decisions so the pump/commit phases run against a fully-provisioned
65 * state and can stay linear.
66 *
67 * HOW: brix_sd_open_maybe_cred so drivers with an open_cred slot use the
68 * per-user proxy. Returns NGX_OK (fs populated), NGX_DONE (stale copy
69 * served — caller reports success without filling), NGX_DECLINED
70 * (admission declined), or NGX_ERROR with errno set; every failure path
71 * releases whatever it had already acquired. */
72/* Open the fill source and snapshot its stat into fs. Returns NGX_OK,
73 * NGX_DONE (a stale copy may be served instead — caller reports success without
74 * filling), or NGX_ERROR with errno set. */
75static ngx_int_t
76cache_fill_open_src(sd_cache_inst_state *st, const char *key,
77 const brix_sd_cred_t *cred, sd_cache_fill_state_t *fs)
78{
79 brix_sd_instance_t *src = fs->src;
80 int err = 0;
81
82 fs->so = brix_sd_open_maybe_cred(src, key, BRIX_SD_O_READ, 0, cred, &err);
83 if (fs->so == NULL) {
84 /* A cold-tier or mesh-sibling miss is NOT an origin outage — never
85 * stale-serve it; the orchestrator falls back to the origin fill. */
86 if (!fs->from_cold && !fs->from_peer
87 && sd_cache_stale_serve_ok(st, key))
88 {
89 return NGX_DONE; /* bounded stale-if-error (phase-68) */
90 }
91 errno = err ? err : EIO;
92 return NGX_ERROR;
93 }
94 /* Every op on the OPEN OBJECT dispatches through so->driver, NOT src->driver:
95 * a decorator source (sd_stage) legitimately returns the tier-below's object
96 * from a read open ("read byte-I/O bypasses the decorator"), so the object's
97 * vtable can differ from the instance's — dispatching object ops through the
98 * instance vtable reinterprets a foreign object state (type confusion). */
99 if (fs->so->driver->pread == NULL) {
100 brix_sd_obj_release(fs->so);
101 errno = ENOSYS;
102 return NGX_ERROR;
103 }
104 /* open() does not guarantee a populated snap (the posix driver fstats lazily),
105 * so fstat the object for an accurate size/mtime/mode - the cinfo validity and
106 * the cached file's permission bits both depend on it. */
107 fs->snap = fs->so->snap;
108 if (fs->so->driver->fstat != NULL) {
109 (void) fs->so->driver->fstat(fs->so, &fs->snap);
110 }
111 return NGX_OK;
112}

115/* Admission for the just-opened source. NGX_OK to proceed (possibly as a
116 * passthrough fill), NGX_DECLINED when the object must not be cached at all.
117 *
118 * phase-92 store-then-evict passthrough: a caller that opted in (fs->allow_pt —
119 * the HTTP shared cache-fill plane) can still be served a remote object the
120 * admission policy declined, by filling it under a separate spool cap, serving
121 * the coalesced waiters through the normal cache-hit reenter, then evicting the
122 * key. The gate is deliberately narrow: only a size-known source within the
123 * passthrough cap qualifies, so a path-filtered object, an oversized object, or
124 * a size-unknown (no Content-Length) origin still declines to the 502 slow-path.
125 * fs->allow_pt is set in exactly ONE place — the shared HTTP cache-fill worker
126 * — so WebDAV, S3 AND cvmfs all inherit the opt-in by routing through it; only
127 * the root:// stream plane's own fill passes allow_pt = 0, leaving its decline
128 * unchanged. Pinned by tests/test_cache_passthrough_planes.py. */
129static ngx_int_t
130cache_fill_admit_src(sd_cache_inst_state *st, const char *key,
131 sd_cache_fill_state_t *fs)
132{
133 off_t pt_cap;
134
135 if (sd_cache_admit(&st->policy, key, fs->snap.size)) {
136 return NGX_OK;
137 }
138 pt_cap = sd_cache_passthrough_cap(&st->policy);
139 if (fs->allow_pt && st->policy.passthrough && fs->snap.size >= 0
140 && (pt_cap == 0 || fs->snap.size <= pt_cap))
141 {
142 fs->passthrough = 1;
143 return NGX_OK;
144 }
145 return NGX_DECLINED; /* too big / filtered - do not cache */
146}

149static ngx_int_t
150cache_fill_acquire(sd_cache_inst_state *st, const char *key,
151 const brix_sd_cred_t *cred, sd_cache_fill_state_t *fs)
152{
153 mode_t fmode;
154 ngx_int_t rc;
155
156 if (fs->src->driver->open == NULL) {
157 errno = ENOSYS;
158 return NGX_ERROR;
159 }
160 rc = cache_fill_open_src(st, key, cred, fs);
161 if (rc != NGX_OK) {
162 return rc;
163 }
164 /* SECURITY + correctness: the physical cache-store object is a svc-owned
165 * artifact that aggregates MANY users' bytes under one tree. Per-user
166 * authorization is enforced at the protocol gate (open_cache.c), and the
167 * CLIENT-FACING mode is carried in the cinfo record and served by
168 * sd_cache_stat() — decoupled from this physical mode. So force 0600:
169 * - owner rw is REQUIRED (XATTR meta_mode stores user.xrd.cinfo ON this
170 * object; Linux refuses user.* xattrs on a non-writable inode — a
171 * read-only 0444 source would otherwise block cinfo persistence, G3);
172 * - NO group/other bits, so a mapped low-priv uid cannot read another
173 * user's cached bytes by direct filesystem access (the source mode's
174 * 0644 previously leaked here). snap.mode still reaches clients via the
175 * cinfo record (ci.mode below, served by sd_cache_stat). */
176 fmode = S_IRUSR | S_IWUSR; /* 0600 — svc-owned cache artifact */
177
178 if (cache_fill_admit_src(st, key, fs) == NGX_DECLINED) {
179 brix_sd_obj_release(fs->so);
180 return NGX_DECLINED;
181 }
182
183 fs->staged = brix_cstore_fill_open(&st->cstore, key, fmode);
184 if (fs->staged == NULL) {
185 ngx_log_error(NGX_LOG_WARN, st->log, errno,
186 "sd_cache: fill_open on the cache store failed for \"%s\" - not cached",
187 key);
188 brix_sd_obj_release(fs->so);
189 return NGX_ERROR;
190 }
191 fs->buf = malloc(SD_CACHE_CHUNK);
192 if (fs->buf == NULL) {
193 brix_cstore_fill_abort(fs->staged);
194 brix_sd_obj_release(fs->so);
195 errno = ENOMEM;
196 return NGX_ERROR;
197 }
198 return NGX_OK;
199}

201/* Pump the source object's bytes into the staged store object.
202 *
203 * WHAT: The origin read loop — pread from the source, fill_write to the staged
204 * object, chunk by chunk, advancing fs->off.
205 *
206 * WHY: The pump is the fill's only long-running phase; isolating it keeps the
207 * read-error / stale-serve / errno-preservation logic in one place.
208 *
209 * HOW: On success frees the move buffer and closes the source (the staged
210 * object stays open for verify/commit) and returns NGX_OK. On a read or
211 * write failure it releases EVERYTHING (buffer, staged fill, source) and
212 * returns NGX_DONE if a bounded stale copy can be served, else NGX_ERROR
213 * with the READ's errno preserved. */
214static ngx_int_t
215cache_fill_pump(sd_cache_inst_state *st, const char *key,
216 sd_cache_fill_state_t *fs)
217{
218 for ( ;; ) {
219 ssize_t r = fs->so->driver->pread(fs->so, fs->buf, SD_CACHE_CHUNK,
220 fs->off);
221
222 if (r < 0) {
223 int read_err = errno; /* capture BEFORE any cleanup call */
224
225 if (read_err == EINTR) {
226 continue;
227 }
228 free(fs->buf);
229 brix_cstore_fill_abort(fs->staged);
230 brix_sd_obj_release(fs->so);
231 if (!fs->from_cold && !fs->from_peer
232 && sd_cache_stale_serve_ok(st, key))
233 {
234 return NGX_DONE; /* bounded stale-if-error (phase-68) */
235 }
236 /* Restore the READ's errno. sd_cache_stale_serve_ok() (and the
237 * cleanup calls) stat the absent cache entry and leave errno
238 * ENOENT — propagating that would make the fill layer classify a
239 * merely-broken origin connection (EIO/ETIMEDOUT) as the origin's
240 * definitive 404, turning a transient network fault into a
241 * cached, client-poisoning "not found" instead of a retry. */
242 errno = read_err;
243 return NGX_ERROR;
244 }
245 if (r == 0) {
246 /* Clean EOF. If the source declared its size, an EOF short of that
247 * size is a TRUNCATED transfer — the origin (or an in-path actor)
248 * cut the fill mid-stream. Committing fs->off bytes as a whole-file
249 * COMPLETE object would poison the cache with a permanently-short
250 * copy served as a valid hit to every subsequent client. Fail
251 * closed exactly like a mid-fill read error: release everything,
252 * serve a bounded stale copy if one exists, else EIO so the next
253 * read refetches. A size-unknown source (snap.size <= 0, e.g. an
254 * HTTP origin with no Content-Length) has no lower bound to check
255 * against, so a clean EOF is the only completion signal available. */
256 if (fs->snap.size > 0 && fs->off < fs->snap.size) {
257 ngx_log_error(NGX_LOG_ERR, st->log, 0,
258 "sd_cache: origin truncated fill for \"%s\" at %O of %O "
259 "bytes - not caching the short object", key,
260 (off_t) fs->off, (off_t) fs->snap.size);
261 free(fs->buf);
262 brix_cstore_fill_abort(fs->staged);
263 brix_sd_obj_release(fs->so);
264 if (!fs->from_cold && !fs->from_peer
265 && sd_cache_stale_serve_ok(st, key))
266 {
267 return NGX_DONE; /* bounded stale-if-error (phase-68) */
268 }
269 errno = EIO;
270 return NGX_ERROR;
271 }
272 break;
273 }
274 if (brix_cstore_fill_write(fs->staged, fs->buf, (size_t) r,
275 fs->off) < 0)
276 {
277 free(fs->buf);
278 brix_cstore_fill_abort(fs->staged);
279 brix_sd_obj_release(fs->so);
280 return NGX_ERROR;
281 }
282 fs->off += r;
283 if (fs->snap.size > 0 && fs->off >= fs->snap.size) {
284 break; /* size known: skip the EOF-probe round-trip */
285 }
286 }
287 free(fs->buf);
288 fs->buf = NULL;
289 /* Capture the origin's advertised content digest BEFORE releasing the source
290 * object — the query needs a live, open object. Only the xroot source offers
291 * an in-band digest (kXR_Qcksum); other backends (http/s3/posix) leave alg/hex
292 * empty and the verify phase decides best-effort/require on that. Skipped
293 * entirely unless a digest-verify policy is in force, so an OFF/CVMFS-CAS fill
294 * pays no round-trip. Mirrors the fetch.c commit-then-verify pattern. */
295 if ((st->policy.verify == BRIX_CACHE_VERIFY_BESTEFFORT
296 || st->policy.verify == BRIX_CACHE_VERIFY_REQUIRE)
297 && ngx_strcmp(brix_sd_backend_name(fs->src), "xroot") == 0)
298 {
299 brix_sd_xroot_query_checksum(fs->so, fs->origin_alg,
300 sizeof(fs->origin_alg), fs->origin_hex, sizeof(fs->origin_hex));
301 }
302 brix_sd_obj_release(fs->so);
303 fs->so = NULL;
304 return NGX_OK;
305}

307/* Publish the staged object and record its cinfo.
308 *
309 * WHAT: Accounts the upstream fill, commits (renames) the staged object into
310 * the store, and stores the whole-file COMPLETE cinfo built from the
311 * source stat snapshot (the fstat mode fix — origin perms in ci.mode).
312 *
313 * WHY: The commit phase is the fill's publication point; everything before it
314 * is invisible to readers, so all bookkeeping that describes a
315 * SUCCESSFUL fill belongs here.
316 *
317 * HOW: Whole-file COMPLETE cinfo (the 1 MiB granule keys validity; the present
318 * bitmap is all-set). A partial/slice fill is section 6.5 / SP2. A failed
319 * cinfo store is only a WARN — the object is cached but unrecorded, a
320 * safe miss (refill) next time. */
321static ngx_int_t
322cache_fill_commit(sd_cache_inst_state *st, const char *key,
323 sd_cache_fill_state_t *fs)
324{
325 brix_cache_cinfo_t ci;
326
327 if (!fs->from_cold && !fs->from_peer) {
328 /* a cold promote or a sibling fetch moved zero origin-WAN bytes */
329 sd_cache_note_origin_bytes(st, key, fs->off); /* WAN in, this attempt */
330 sd_cache_note_upstream(st, 1, fs->off, sd_cache_ms_since(&fs->t0));
331 }
332
333 if (brix_cstore_fill_commit(fs->staged) != NGX_OK) {
334 /* Failed commit keeps the handle valid — abort to release it. */
335 brix_cstore_fill_abort(fs->staged);
336 return NGX_ERROR;
337 }
338
339 ngx_memzero(&ci, sizeof(ci));
340 ci.magic = BRIX_CACHE_CINFO_MAGIC;
341 ci.version = BRIX_CACHE_CINFO_VERSION;
342 ci.block_size = BRIX_CACHE_DIRTY_BLOCK;
343 ci.size = (uint64_t) fs->off;
344 ci.mtime = (uint64_t) fs->snap.mtime;
345 ci.mode = (uint32_t) (fs->snap.mode & 0777); /* origin perms — served back
346 * so a read-only source is not
347 * masked by the owner-writable
348 * physical store object. */
349 ci.nblocks = brix_cache_cinfo_nblocks((uint64_t) fs->off, ci.block_size);
350 ci.flags = BRIX_CINFO_F_COMPLETE
351 | (fs->verified ? BRIX_CINFO_F_VERIFIED : 0);
352 ci.filled_at = (uint64_t) time(NULL);
353 if (fs->verified && fs->cks_alg[0] != '\0' && fs->cks_hex[0] != '\0') {
354 /* The verified digest — the producer side of xrdckverify --cache.
355 * Only a MATCHED digest is recorded; an unverified fill leaves the
356 * fields empty rather than asserting bytes nobody checked. */
357 ngx_cpystrn((u_char *) ci.cks_alg, (u_char *) fs->cks_alg,
358 sizeof(ci.cks_alg));
359 ci.cks_alg_len = (uint8_t) ngx_strlen(ci.cks_alg);
360 ngx_cpystrn((u_char *) ci.cks_hex, (u_char *) fs->cks_hex,
361 sizeof(ci.cks_hex));
362 ci.cks_len = (uint8_t) ngx_strlen(ci.cks_hex);
363 }
364 if (st->policy.cvmfs_manifest_ttl > 0 && sd_cache_is_manifest_key(key)) {
365 brix_cache_cinfo_set_expires(&ci,
366 (time_t) ci.filled_at + st->policy.cvmfs_manifest_ttl);
367 }
368
369 if (brix_cstore_cinfo_store(&st->cstore, key, &ci) != NGX_OK) {
370 /* The object is cached but unrecorded - a safe miss (refill) next time. */
371 ngx_log_error(NGX_LOG_WARN, st->log, errno,
372 "sd_cache: cinfo store failed for \"%s\" - will refill on next read",
373 key);
374 }
375
376 if (st->policy.global_cas && fs->verified
377 && st->policy.verify == BRIX_CACHE_VERIFY_CVMFS_CAS)
378 {
379 /* phase-87 G13: only cvmfs-cas binds the KEY's hash to the bytes
380 * (best-effort/require verify an origin-advertised digest instead) —
381 * with that proof, collapse byte-identical cross-repo copies. */
382 brix_gcas_publish(&st->cstore, key);
383 }
384 return NGX_OK;
385}

387/* One fill attempt from the tier `src` (the wrapped source, the cold store on
388 * a phase-85 F7 promote, or a mesh sibling on a phase-85 F8 peer fetch):
389 * acquire (source open + admission + staged fill-open) -> pump (read loop) ->
390 * verify (phase-68 cvmfs-cas) -> commit (publish + cinfo). NGX_DONE from
391 * acquire/pump means a bounded stale copy was served — reported as NGX_OK
392 * without caching (origin attempts only; a cold or peer attempt never
393 * stale-serves so its failure falls back to the origin). */
394static ngx_int_t
395sd_cache_fill_attempt(sd_cache_inst_state *st, const char *key,
396 const brix_sd_cred_t *cred, brix_sd_instance_t *src, int from_cold,
397 int from_peer, int allow_pt, int *out_pt)
398{
399 sd_cache_fill_state_t fs;
400 ngx_int_t rc;
401
402 ngx_memzero(&fs, sizeof(fs));
403 fs.src = src;
404 fs.from_cold = from_cold;
405 fs.from_peer = from_peer;
406 fs.allow_pt = allow_pt;
407 (void) clock_gettime(CLOCK_MONOTONIC, &fs.t0);
408
409 rc = cache_fill_acquire(st, key, cred, &fs);
410 if (rc == NGX_DONE) {
411 return NGX_OK; /* bounded stale-if-error (phase-68) */
412 }
413 if (rc != NGX_OK) {
414 return rc; /* NGX_DECLINED or NGX_ERROR */
415 }
416
417 rc = cache_fill_pump(st, key, &fs);
418 if (rc == NGX_DONE) {
419 return NGX_OK; /* bounded stale-if-error (phase-68) */
420 }
421 if (rc != NGX_OK) {
422 return NGX_ERROR;
423 }
424
425 if (cache_fill_verify(st, key, &fs) != NGX_OK) {
426 return NGX_ERROR;
427 }
428 rc = cache_fill_commit(st, key, &fs);
429 if (rc == NGX_OK && fs.passthrough && out_pt != NULL) {
430 *out_pt = 1; /* phase-92: serve-then-evict — caller drops key */
431 }
432 return rc;
433}

435/* Rendezvous (highest-random-weight) owner of `key` over the peer ring:
436 * FNV-1a 64 over "<label>\n<key>", owner = argmax across members. Every mesh
437 * node carries the IDENTICAL member list with identical labels, so all nodes
438 * agree on the owner with no coordination. The scheme (FNV-1a 64, offset
439 * 14695981039346656037, prime 1099511628211, '\n' separator, ties to the
440 * lower index) is mirrored bit-for-bit by tests/test_cvmfs_peer_mesh.py —
441 * change both together or not at all. */
442static uint64_t
443sd_cache_hrw_fnv1a64(const char *label, const char *key)
444{
445 uint64_t h = BRIX_FNV1A64_OFFSET_BASIS;
446 const char *p;
447
448 for (p = label; *p != '\0'; p++) {
449 h = (h ^ (uint64_t) (unsigned char) *p) * BRIX_FNV1A64_PRIME;
450 }
451 h = (h ^ (uint64_t) '\n') * BRIX_FNV1A64_PRIME;
452 for (p = key; *p != '\0'; p++) {
453 h = (h ^ (uint64_t) (unsigned char) *p) * BRIX_FNV1A64_PRIME;
454 }
455 return h;
456}

458static int
459sd_cache_peer_owner(const brix_sd_cache_peer_t *peers, int n,
460 const char *key)
461{
462 int i, owner = 0;
463 uint64_t best = 0, w;
464
465 for (i = 0; i < n; i++) {
466 w = sd_cache_hrw_fnv1a64(peers[i].label, key);
467 if (i == 0 || w > best) {
468 best = w;
469 owner = i;
470 }
471 }
472 return owner;
473}

475/* Fill `key` into the cache store and record its cinfo.
476 *
477 * WHAT: Pulls the object `key` into the cache store, building a whole-file
478 * cinfo on success. With a cold store tier attached (phase-85 F7) a
479 * verified PROMOTE from the cold copy is attempted first; with a sibling
480 * mesh attached (phase-85 F8) the key's rendezvous-owning peer is tried
481 * next; the origin (st->source) is the fallback and the only tier that
482 * stale-serves.
483 *
484 * WHY: The fill spine is the single path that populates the cache; threading
485 * the caller's per-user `cred` here ensures the source open uses the
486 * correct identity rather than the service credential. Routing the
487 * promote through the SAME spine means a cold object passes the identical
488 * cvmfs-cas / manifest-signature verify gate as an origin fill — a
489 * corrupt cold copy can never publish.
490 *
491 * HOW: Cold attempt (when st->cold): any failure but admission silently falls
492 * back to the origin; success drops the cold copy (move semantics).
493 * Peer attempt (when the ring owner of `key` is a non-self member with a
494 * built instance): one verified fetch from that sibling; any failure but
495 * admission silently falls back to the origin. Only the configured ring
496 * members are ever contacted (allowlist by construction — no request-
497 * derived authority reaches a peer fetch). Returns NGX_OK (object cached
498 * + cinfo stored, or stale-served), NGX_DECLINED (policy declined), or
499 * NGX_ERROR (failure). */
500ngx_int_t
501sd_cache_fill(sd_cache_inst_state *st, const char *key,
502 const brix_sd_cred_t *cred, int allow_pt, int *out_pt)
503{
504 if (out_pt != NULL) {
505 *out_pt = 0;
506 }
507 if (st->cold != NULL) {
508 ngx_int_t rc = sd_cache_fill_attempt(st, key, cred, st->cold, 1, 0,
509 allow_pt, out_pt);
510
511 if (rc == NGX_OK) {
512 /* Promoted: the hot tier owns the object now — drop the cold copy
513 * (best-effort; a leftover is simply re-evicted or re-demoted). */
514 if (st->cold->driver->unlink != NULL) {
515 (void) st->cold->driver->unlink(st->cold, key, 0);
516 }
517 ngx_log_error(NGX_LOG_INFO, st->log, 0,
518 "sd_cache: promoted \"%s\" from the cold tier", key);
519 return NGX_OK;
520 }
