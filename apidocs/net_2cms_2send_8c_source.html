26/* ngx_brix_cms_send_error — reply to a forwarded op that failed * WHAT: Sends a kYR_error reply frame: [4B big-endian ecode][text + NUL],
27 * echoing the request streamid. WHY: byte-exact with cmsd
28 * XrdCmsProtocol::Reply_Error — a data node answers a failed forwarded
29 * namespace op (Plane B) this way; success stays silent. HOW: pack the
30 * error code then the NUL-terminated text into a stack buffer and dispatch
31 * a CMS_RSP_ERROR frame. text is truncated to fit the frame cap. */

35{

38
39 tlen = (text != NULL) ? ngx_strlen(text) : 0;
40 if (tlen > sizeof(buf) - 5) { /* 4B ecode + text + NUL */
41 tlen = sizeof(buf) - 5;

52/*
53 * cms_login_mode — the CmsLoginData Mode word for this node's cluster role
54 * (Phase-61 W7, Pander parity). An explicit brix_cms_role picks the stock
55 * bits — server kYR_server, manager kYR_manager ALONE (a sub-manager
56 * subscribing to a meta-manager), supervisor kYR_manager|kYR_server; auto
57 * keeps the legacy derivation (server, plus manager when manager_mode).
58 */
59static uint32_t
60cms_login_mode(ngx_stream_brix_srv_conf_t *conf)
61{
62 switch (conf->cms.role) {
63 case BRIX_CMS_ROLE_SERVER:
64 return CMS_LOGIN_MODE;
65 case BRIX_CMS_ROLE_MANAGER:
66 return CMS_LOGIN_MODE_MANAGER;
67 case BRIX_CMS_ROLE_SUPERVISOR:
68 return CMS_LOGIN_MODE | CMS_LOGIN_MODE_MANAGER;
69 case BRIX_CMS_ROLE_PEER:
70 /* §2.17: a peer cluster's contact point — no kYR_server bit; the
71 * manager registers it as overflow capacity, never normal service. */
72 return CMS_LOGIN_MODE_PEER;
73 case BRIX_CMS_ROLE_PROXY:
74 /* §2.17: a proxy data server — serves data (kYR_server) through a
75 * proxy front (kYR_proxy). */
76 return CMS_LOGIN_MODE | CMS_LOGIN_MODE_PROXY;
77 default:
78 return CMS_LOGIN_MODE
79 | (conf->manager_mode ? CMS_LOGIN_MODE_MANAGER : 0);
80 }
81}

83/*
84 * ngx_brix_cms_send_login — initial CMS login frame with server capabilities.
85 *
86 * WHAT: Builds and sends the first CMS frame after establishing a TCP connection,
87 * reporting version, mode (client/manager), PID, filesystem space stats,
88 * listen port, exported paths, and reserved fields. WHY: The CMS manager
89 * needs this information to decide whether to route client requests here;
90 * without login the server is invisible in the cluster registry. HOW: Call
91 * stat_space() for total_gb/free_mb/util_pct (aggregate if manager mode),
92 * pack type/value payload fields in wire order via put_short/put_int, append
93 * exported paths from cms_export_paths(), send frame with CMS_RR_LOGIN code. */
94
95ngx_int_t
96ngx_brix_cms_send_login(ngx_brix_cms_ctx_t *ctx)
97{
98 u_char payload[1280];
99 u_char *payload_cursor;
100 u_char sid[256];
101 u_char pathbuf[640];
102 u_char envbuf[80];
103 u_char hostbuf[200];
104 u_char *pp;
105 ngx_str_t paths;
106 size_t sid_len;
107 size_t path_len;
108 size_t env_len;
109 size_t i;
110 size_t seg_start;
111 uint32_t total_gb;
112 uint32_t free_mb;
113 uint32_t util_pct;
114 u_char ptype;
115
116 /*
117 * Build the exported-path list in the real XrdCms wire format: newline-
118 * separated "<type> <path>" entries, type 'w' if writable else 'r'
119 * (e.g. "all.export / r/w" -> "w /"). brix_cms_paths may carry several
120 * colon-separated namespace prefixes ("/data:/atlas").
121 */
122 paths = ngx_brix_cms_export_paths(ctx->conf);
123 ptype = ctx->conf->common.allow_write ? (u_char) 'w' : (u_char) 'r';
124 pp = pathbuf;
125 seg_start = 0;
126 /*
127 * Loop runs to i == paths.len (one past the last byte) so the final
128 * segment is flushed by the same branch that handles a ':' delimiter — no
129 * separate post-loop emit needed. The "+ 4" slack in the bound check
130 * reserves room for the per-segment framing this iteration may append:
131 * one '\n' separator + the type char + a space (3) plus a margin byte.
132 */
133 for (i = 0; i <= paths.len; i++) {
134 if (i == paths.len || paths.data[i] == ':') {
135 size_t seglen = i - seg_start;
136 if (seglen > 0
137 && (size_t) (pp - pathbuf) + seglen + 4 < sizeof(pathbuf))
138 {
139 if (pp != pathbuf) {
140 *pp++ = '\n';
141 }
142 *pp++ = ptype;
143 *pp++ = ' ';
144 ngx_memcpy(pp, paths.data + seg_start, seglen);
145 pp += seglen;
146 }
147 seg_start = i + 1;
148 }
149 }
150 path_len = (size_t) (pp - pathbuf);
151
152 /*
153 * Server identity "<host>:<dport>" — opaque to the manager but must be
154 * stable and reasonably unique for this node.
155 */
156 if (gethostname((char *) hostbuf, sizeof(hostbuf)) != 0) {
157 ngx_memcpy(hostbuf, "nginx", sizeof("nginx"));
158 }
159 hostbuf[sizeof(hostbuf) - 1] = '\0';
160 sid_len = (size_t) (ngx_snprintf(sid, sizeof(sid), "%s:%d", hostbuf,
161 (int) ctx->conf->listen_port) - sid);
162
163 total_gb = 0;
164 free_mb = 0;
165 util_pct = 0;
166 (void) ngx_brix_cms_stat_space(ctx->conf, &total_gb, &free_mb,
167 &util_pct);
168
169 if (ctx->conf->manager_mode) {
170 uint32_t agg_free = 0, agg_util = 0;
171 brix_srv_aggregate_space(&agg_free, &agg_util);
172 if (agg_free > 0 || agg_util > 0) {
173 free_mb = agg_free;
174 util_pct = agg_util;
175 }
176 }
177
178 /*
179 * kYR_login payload — XrdOucPup wire order (CmsLoginData):
180 * Version sh, Mode int, HoldTime int, tSpace int, fSpace int, mSpace int,
181 * fsNum sh, fsUtil sh, dPort sh, sPort sh, [Fence],
182 * SID str, Paths str, ifList str, envCGI str.
183 * Scalars carry a 0x80/0xa0 type tag; strings are bare [len][data+NUL].
184 * The frame header datalen (set by send_frame) is the total payload length.
185 */
186 payload_cursor = payload;
187 payload_cursor = ngx_brix_cms_put_short(payload_cursor,
188 CMS_LOGIN_VERSION);
189 payload_cursor = ngx_brix_cms_put_int(payload_cursor,
190 cms_login_mode(ctx->conf));
191 payload_cursor = ngx_brix_cms_put_int(payload_cursor,
192 (uint32_t) getpid());
193 /* tSpace/fSpace: total disk (GB) and free (MB) reported to the manager. */
194 payload_cursor = ngx_brix_cms_put_int(payload_cursor, total_gb);
195 payload_cursor = ngx_brix_cms_put_int(payload_cursor, free_mb);
196 /*
197 * mSpace: minimum-free threshold (MB). Policy floor, not a live
198 * measurement — below this free space the manager should stop selecting
199 * us. Configurable via brix_cms_min_free (§2.4); default 100 MB, the
200 * historical NGX_BRIX_CMS_MIN_FREE_MB constant.
201 */
202 payload_cursor = ngx_brix_cms_put_int(payload_cursor,
203 (uint32_t) ctx->conf->cms.min_free_mb);
204 /* fsNum: number of exported filesystems — we present a single namespace. */
205 payload_cursor = ngx_brix_cms_put_short(payload_cursor, 1);
206 payload_cursor = ngx_brix_cms_put_short(payload_cursor,
207 (uint16_t) util_pct);
208 /*
209 * dPort: the data port clients are redirected to (our listen port).
210 * §2.12 (cms.altds): a configured alternate data server's port wins —
211 * the manager records the CONNECTION's peer address as the data host, so
212 * advertising the foreign port redirects clients to the co-located
213 * foreign data server instead of this node.
214 * sPort follows as 0 — we run no separate subscriber/admin port, so the
215 * field is present in wire order but advertised as unused.
216 */
217 payload_cursor = ngx_brix_cms_put_short(
218 payload_cursor,
219 (uint16_t) (ctx->conf->cms.altds_port > 0
220 ? ctx->conf->cms.altds_port
221 : ctx->conf->listen_port));
222 payload_cursor = ngx_brix_cms_put_short(payload_cursor, 0); /* sPort */
223
224 /*
225 * Trailing Pup strings in wire order: SID, Paths, then ifList and envCGI.
226 * ifList is always empty; envCGI carries "vnid=<id>" when brix_cms_vnid is
227 * configured (Phase-89 W9), else stays empty. Empty slots are still
228 * emitted as length-prefixed strings — the manager expects all four.
229 * The vnid value is truncated to 63 chars, the manager-side buffer bound.
230 */
231 env_len = 0;
232 if (ctx->conf->cms.vnid.len > 0) {
233 env_len = (size_t) (ngx_snprintf(envbuf, sizeof(envbuf), "vnid=%*s",
234 ngx_min(ctx->conf->cms.vnid.len, 63),
235 ctx->conf->cms.vnid.data) - envbuf);
236 }
237
238 payload_cursor = ngx_brix_cms_put_string(payload_cursor, sid, sid_len);
239 payload_cursor = ngx_brix_cms_put_string(payload_cursor, pathbuf,
240 path_len);
241 payload_cursor = ngx_brix_cms_put_string(payload_cursor, NULL, 0);
242 payload_cursor = ngx_brix_cms_put_string(payload_cursor,
243 env_len > 0 ? envbuf : NULL,
244 env_len);
245
246 return ngx_brix_cms_send_frame(ctx, 0, CMS_RR_LOGIN, 0, payload,
247 (size_t) (payload_cursor - payload));
248}

266/*
267 * ngx_brix_cms_send_load — periodic load heartbeat report.
268 *
269 * WHAT: Builds and sends a CMS_LOAD frame reporting current free disk space in MB,
270 * optionally aggregated across all servers when in manager mode. WHY: The CMS
271 * manager uses load reports to distribute client requests proportionally to
272 * available capacity; stale or missing reports cause the manager to stop routing
273 * here. HOW: Call stat_space() for free_mb (aggregate via brix_srv_aggregate_space
274 * if manager mode), pack payload as put_short(6) + 6 zero bytes + put_int(free_mb),
275 * send with CMS_RR_LOAD code. */
276
277ngx_int_t
278ngx_brix_cms_send_load(ngx_brix_cms_ctx_t *ctx)
279{
280 u_char payload[32];
281 u_char *payload_cursor;
282 uint32_t free_mb;
283 uint32_t util_pct;
284 uint8_t load5[5];
285
286 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, ctx->cycle->log, 0,
287 "brix: send_load: conn=%p", ctx->connection);
288
289 free_mb = 0;
290 util_pct = 0;
291 (void) ngx_brix_cms_stat_space(ctx->conf, NULL, &free_mb, &util_pct);
292
293 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, ctx->cycle->log, 0,
294 "brix: send_load: free_mb=%uD", free_mb);
295
296 if (ctx->conf->manager_mode) {
297 uint32_t agg_free = 0, agg_util = 0;
298 brix_srv_aggregate_space(&agg_free, &agg_util);
299 if (agg_free > 0) {
300 free_mb = agg_free;
301 }
302 }
303
304 /*
305 * kYR_load payload (CmsLoadRequest / lodArgs): theLoad is a Pup char-blob
306 * of 6 load bytes (cpu,net,xeq,mem,pag,dsk) — a bare [2-byte len][data] with
307 * NO scalar type tag — followed by dskFree as a tagged int. The first five
308 * bytes come from the Phase-89 W4 /proc meter (a meter fault degrades to
309 * zeros, never a failed heartbeat); dsk is the export-fs utilisation.
310 * §2.11: a FRESH external cms.perf-pgm feed overrides the /proc figures.
311 */
312 brix_cms_meter_sample(&ctx->meter, (uint64_t) ngx_current_msec, load5);
313 (void) brix_cms_perf_get(load5);
314
315 /*
316 * Hand-packed (put16 + manual cursor advance) rather than via put_string:
317 * put16 only writes the 2-byte length, so the cursor must be advanced by
318 * hand past both the length and the 6 data bytes before the tagged dskFree
319 * int is appended via put_int.
320 */
321 payload_cursor = payload;
322 ngx_brix_cms_put16(payload_cursor, 6); /* blob length: 6 load bytes */
323 payload_cursor += 2; /* skip the 2-byte length */
324 ngx_memcpy(payload_cursor, load5, 5); /* cpu,net,xeq,mem,pag */
325 payload_cursor[5] = (u_char) (util_pct > 100 ? 100 : util_pct); /* dsk */
326 payload_cursor += 6; /* advance past the 6 bytes */
327 payload_cursor = ngx_brix_cms_put_int(payload_cursor, free_mb);
328
329 return ngx_brix_cms_send_frame(ctx, 0, CMS_RR_LOAD, 0, payload,
330 (size_t) (payload_cursor - payload));
331}

386/*
387 * ngx_brix_cms_next_streamid — monotonic stream ID allocator.
388 *
389 * WHAT: Returns the next available stream ID for outgoing CMS frames on this
390 * manager link, advancing by streamid_stride and wrap-resetting to the
391 * link's seed near UINT32_MAX. WHY: Each CMS frame must carry a unique
392 * streamid so a reply correlates back to its pending entry — and the

403

405 ctx->next_streamid = ctx->streamid_seed;
406 }
407 ctx->next_streamid += stride;
