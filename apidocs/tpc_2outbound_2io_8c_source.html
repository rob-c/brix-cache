1/* File: tpc io — low-level socket helpers for native TPC data transfer
2 * WHAT: Three blocking I/O primitives used exclusively by the TPC thread-pool worker
3 * to read/write bytes between a remote XRootD origin and nginx's local file.
4 * All functions run inside NGX_THREADS detached workers — allocations use
5 * malloc/free (not ngx_palloc). No SSL wrapping; plain TCP sockets only.
6 *
7 * WHY: TPC transfers require the destination server to connect directly to the
8 * source XRootD endpoint and pull data over raw TCP. Unlike event-loop I/O
9 * which accumulates incrementally via callbacks, these helpers guarantee
10 * exact byte counts — partial sends/recv loop until complete. The recv_response
11 * function additionally parses XRootD ServerResponseHdr framing (status +
12 * payload length) and allocates the body for caller use.
13 *
14 * HOW: tpc_send_all iterates remaining bytes with send() loop, continues on EINTR,
15 * returns -1 on any other failure. tpc_recv_exact mirrors this pattern with
16 * recv(). tpc_recv_response reads fixed-size ServerResponseHdr first, then
17 * mallocs the payload body and calls tpc_recv_exact for the remainder.
18 * Caller must free the returned body pointer.
19 *
20 * phase-57 §F5: when the pull upgraded to TLS (t->tls != NULL, set by
21 * tpc_start_tls after a kXR_gotoTLS), send/recv route through SSL_write/
22 * SSL_read instead of the raw fd. The SSL sits on a blocking fd in the
23 * thread-pool worker, so a single SSL_write/SSL_read completes or errors;
24 * <=0 is treated as a hard failure exactly like the plain recv()/send() path.
25 *
26 * The transfer loop itself lives in io_xfer.c — one copy shared by both
27 * directions, in a TU free of nginx headers so it is unit-testable over a
28 * socketpair (io_xfer_unittest.c). */

40/* ---- Find the bound sub-stream that owns an fd ----------------------------
41 *
42 * WHAT: Returns the matching sub-stream slot, or NULL for the primary socket,
43 * an unknown fd, or a NULL task.
44 * WHY: TLS transport and verification state must resolve through one fd-to-slot
45 * rule so a secondary socket cannot inherit the primary's security state.
46 * HOW: Walk the bounded active sub-stream array and compare each owned fd. */
47static tpc_substream_t *
48tpc_session_substream(brix_tpc_pull_t *t, int fd)
49{
50 int i;
51
52 if (t == NULL) {
53 return NULL;
54 }
55 for (i = 0; i < t->nsub; i++) {
56 if (t->sub[i].fd == fd) {
57 return &t->sub[i];
58 }
59 }
60 return NULL;
61}

63/* ---- Resolve the TLS session for an fd -----------------------------------
64 *
65 * WHAT: Returns the SSL session owned by fd, or NULL for cleartext/no task.
66 * WHY: Every native-TPC I/O primitive must select the session belonging to the
67 * exact socket rather than assuming the primary connection.
68 * HOW: Resolve a sub-stream first; otherwise require an exact primary-fd match. */
69
70static SSL *
71tpc_session_ssl(brix_tpc_pull_t *t, int fd)
72{
73 tpc_substream_t *sub;
74
75 if (t == NULL) {
76 return NULL;
77 }
78 sub = tpc_session_substream(t, fd);
79 if (sub != NULL) {
80 return (SSL *) sub->tls;
81 }
82 return (fd == t->primary_fd) ? (SSL *) t->tls : NULL;
83}

85/* ---- Report authenticated TLS state for an fd ----------------------------
86 *
87 * WHAT: Returns 1 only when fd has a live SSL session whose chain and source
88 * name/IP checks completed successfully; otherwise returns 0.
89 * WHY: Bearer credentials need a final disclosure guard independent of the
90 * peer-controlled STARTTLS negotiation and configuration flags.
91 * HOW: Resolve the fd's sub-stream state, require both its SSL pointer and
92 * verified bit, or require an exact match to the primary session fd. */
93int
94tpc_io_has_verified_tls(brix_tpc_pull_t *t, int fd)
95{
96 tpc_substream_t *sub;
97
98 if (t == NULL) {
99 return 0;
100 }
101 sub = tpc_session_substream(t, fd);
102 if (sub != NULL) {
103 return (sub->tls != NULL && sub->tls_verified) ? 1 : 0;
104 }
105 if (fd != t->primary_fd) {
106 return 0;
107 }
108 return (t->tls != NULL && t->tls_verified) ? 1 : 0;
109}

149/*
150 *
151 * WHAT: Reads a full XRootD ServerResponseHdr (status code + payload length), then
152 * allocates and reads the payload body. Returns parsed status, malloc'd body,
153 * and payload length via output pointers. NULL body when dlen==0.
154 *
155 * WHY: TPC transfers require parsing wire protocol response frames before processing
156 * data — the header encodes whether the operation succeeded (kXR_ok) or failed
157 * with a specific kXR error code, plus how many bytes of payload follow. The
158 * malloc'd body lets callers handle variable-length responses without fixed buffers.
159 *
160 * HOW: Three-phase read → 1) tpc_recv_exact fetches sizeof(ServerResponseHdr) bytes,
161 * then ntohs/ntohl decode status and dlen from network byte order. 2) If dlen==0
162 * returns immediately with NULL body. 3) Validates dlen <= TPC_RESP_MAX_BODY,
163 * mallocs (dlen + 1) bytes for safety, calls tpc_recv_exact to read payload,
164 * null-terminates the buffer, sets output pointers, and returns 0 on success.
165 * Caller must free(*body) after use. Returns -1 on any I/O or allocation error. */
166int
167tpc_recv_response_sid(brix_tpc_pull_t *t, int fd, u_char streamid[2],
168 uint16_t *status, u_char **body, uint32_t *dlen)
169{
170 ServerResponseHdr hdr;
171 u_char *response_body;
172
173 if (tpc_recv_exact(t, fd, &hdr, sizeof(hdr)) != 0) {
174 return -1;
175 }
176
177 if (streamid != NULL) {
178 streamid[0] = hdr.streamid[0];
179 streamid[1] = hdr.streamid[1];
180 }
181 *status = ntohs(hdr.status);
182 *dlen = (uint32_t) ntohl(hdr.dlen);
183 *body = NULL;
184
185 if (*dlen == 0) {
186 return 0;
187 }
188
189 if (*dlen > TPC_RESP_MAX_BODY) {
190 return -1;
191 }
192
193 response_body = malloc((size_t) (*dlen) + 1);
194 if (response_body == NULL) {
195 return -1;
196 }
197
198 if (tpc_recv_exact(t, fd, response_body, *dlen) != 0) {
199 free(response_body);
200 return -1;
201 }
202
203 response_body[*dlen] = '\0';
204 *body = response_body;
205 return 0;
206}
