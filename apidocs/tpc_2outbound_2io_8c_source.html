1/* File: tpc io — low-level socket helpers for native TPC data transfer
2 * WHAT: Three blocking I/O primitives used exclusively by the TPC thread-pool worker
3 * to read/write bytes between a remote XRootD origin and nginx's local file.
4 * All functions run inside NGX_THREADS detached workers — allocations use
5 * malloc/free (not ngx_palloc). No SSL wrapping; plain TCP sockets only.
6 *
7 * WHY: TPC transfers require the destination server to connect directly to the
8 * source XRootD endpoint and pull data over raw TCP. Unlike event-loop I/O
9 * which accumulates incrementally via callbacks, these helpers guarantee
10 * exact byte counts — partial sends/recv loop until complete. The recv_response
11 * function additionally parses XRootD ServerResponseHdr framing (status +
12 * payload length) and allocates the body for caller use.
13 *
14 * HOW: tpc_send_all iterates remaining bytes with send() loop, continues on EINTR,
15 * returns -1 on any other failure. tpc_recv_exact mirrors this pattern with
16 * recv(). tpc_recv_response reads fixed-size ServerResponseHdr first, then
17 * mallocs the payload body and calls tpc_recv_exact for the remainder.
18 * Caller must free the returned body pointer.
19 *
20 * phase-57 §F5: when the pull upgraded to TLS (t->tls != NULL, set by
21 * tpc_start_tls after a kXR_gotoTLS), send/recv route through SSL_write/
22 * SSL_read instead of the raw fd. The SSL sits on a blocking fd in the
23 * thread-pool worker, so a single SSL_write/SSL_read completes or errors;
24 * <=0 is treated as a hard failure exactly like the plain recv()/send() path.
25 *
26 * The transfer loop itself lives in io_xfer.c — one copy shared by both
27 * directions, in a TU free of nginx headers so it is unit-testable over a
28 * socketpair (io_xfer_unittest.c). */

40/* WHAT: Resolve the TLS session that owns `fd` — a bound sub-stream's own
41 * session (F7) or the primary's — or NULL when that leg is cleartext.
42 * t may be NULL on the pre-session legs (connect/bootstrap probes). */
43
44static SSL *
45tpc_session_ssl(brix_tpc_pull_t *t, int fd)
46{
47 int i;
48
49 if (t == NULL) {
50 return NULL;
51 }
52 for (i = 0; i < t->nsub; i++) {
53 if (t->sub[i].fd == fd) {
54 return (SSL *) t->sub[i].tls;
55 }
56 }
57 return (SSL *) t->tls;
58}

98/*
99 *
100 * WHAT: Reads a full XRootD ServerResponseHdr (status code + payload length), then
101 * allocates and reads the payload body. Returns parsed status, malloc'd body,
102 * and payload length via output pointers. NULL body when dlen==0.
103 *
104 * WHY: TPC transfers require parsing wire protocol response frames before processing
105 * data — the header encodes whether the operation succeeded (kXR_ok) or failed
106 * with a specific kXR error code, plus how many bytes of payload follow. The
107 * malloc'd body lets callers handle variable-length responses without fixed buffers.
108 *
109 * HOW: Three-phase read → 1) tpc_recv_exact fetches sizeof(ServerResponseHdr) bytes,
110 * then ntohs/ntohl decode status and dlen from network byte order. 2) If dlen==0
111 * returns immediately with NULL body. 3) Validates dlen <= TPC_RESP_MAX_BODY,
112 * mallocs (dlen + 1) bytes for safety, calls tpc_recv_exact to read payload,
113 * null-terminates the buffer, sets output pointers, and returns 0 on success.
114 * Caller must free(*body) after use. Returns -1 on any I/O or allocation error. */
115int
116tpc_recv_response_sid(brix_tpc_pull_t *t, int fd, u_char streamid[2],
117 uint16_t *status, u_char **body, uint32_t *dlen)
118{
119 ServerResponseHdr hdr;
120 u_char *response_body;
121
122 if (tpc_recv_exact(t, fd, &hdr, sizeof(hdr)) != 0) {
123 return -1;
124 }
125
126 if (streamid != NULL) {
127 streamid[0] = hdr.streamid[0];
128 streamid[1] = hdr.streamid[1];
129 }
130 *status = ntohs(hdr.status);
131 *dlen = (uint32_t) ntohl(hdr.dlen);
132 *body = NULL;
133
134 if (*dlen == 0) {
135 return 0;
136 }
137
138 if (*dlen > TPC_RESP_MAX_BODY) {
139 return -1;
140 }
141
142 response_body = malloc((size_t) (*dlen) + 1);
143 if (response_body == NULL) {
144 return -1;
145 }
146
147 if (tpc_recv_exact(t, fd, response_body, *dlen) != 0) {
148 free(response_body);
149 return -1;
150 }
151
152 response_body[*dlen] = '\0';
153 *body = response_body;
154 return 0;
155}
