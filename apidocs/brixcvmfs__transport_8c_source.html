1/*
2 * brixcvmfs_transport.c — the CVMFS-brix libcurl fetch seam (Phase-38 split).
3 *
4 * WHAT: the injected transport the shared cvmfs fetch core calls to pull one
5 * object — DPI/loss-hardened range-resume GETs over a pooled libcurl
6 * handle, the G3 shared-dictionary transfer coding, and the G2
7 * .cvmfs-bundle POST — plus the pool lifecycle helpers.
8 * WHY: split from brixcvmfs.c to keep each TU within the file-size budget;

34/* Capacity of a mirror URL we are willing to build. The https form is one byte
35 * larger because s/http:/https:/ adds exactly one character, so any http URL
36 * that fits can always be rewritten — which is what keeps to_https() from ever
37 * having to truncate (and what silences -Wformat-truncation there). */

41typedef struct { unsigned char *buf; size_t cap, len; } curl_sink_t;

43static size_t curl_write_cb(char *ptr, size_t sz, size_t nm, void *ud) {
44 curl_sink_t *s = ud;
45 size_t n = sz * nm;
46 if (s->len + n > s->cap) return 0; /* overflow → curl aborts */
47 memcpy(s->buf + s->len, ptr, n);
48 s->len += n;
49 return n;
50}

52/* Extract scheme + host + port from a URL for env-proxy resolution. */
53static void url_host(const char *url, char *sch, size_t sl, char *host, size_t hl, int *port) {
54 sch[0] = host[0] = '\0'; *port = 0;
55 const char *p = url;
56 const char *sep = strstr(url, "://");
57 if (sep) { size_t n = (size_t)(sep - url); if (n < sl) { memcpy(sch, url, n); sch[n] = '\0'; } p = sep + 3; }
58 size_t n = 0;
59 while (p[n] && p[n] != ':' && p[n] != '/' && n < hl - 1) { host[n] = p[n]; n++; }
60 host[n] = '\0';
61 *port = (p[n] == ':') ? atoi(p + n + 1) : (strcmp(sch, "https") == 0 ? 443 : 80);
62}

64/* Persistent easy handles, pooled through the generic brix_cpool (the SAME
65 * slot/mutex/condvar engine the xrootdfs root:// and WebDAV-metadata paths use —
66 * uniformity across the two FUSE drivers, phase-86). libcurl's connection cache
67 * lives on the handle, so keeping handles across fetches is what keeps
68 * origin/proxy connections alive (keepalive reuse = fewer handshakes for a DPI
69 * middlebox to interfere with). checkout() hands a caller exclusive ownership of
70 * one handle for the duration of a transfer (libcurl easy handles are NOT
71 * concurrency-safe), so the foreground FUSE loop and the background prefetch
72 * worker each borrow their own handle without racing. Every per-request option
73 * is re-set on each call below, so no stale request state survives reuse; `-o
74 * fresh` still forces a new connection per request. */
75#define BRIX_CURL_POOL_SLOTS 4 /* foreground (-s serialized) + prefetch + slack */
76static brix_cpool *g_curl_pool;

96/* ---- trained shared dictionary, client side (phase-87 G3) --------------- *
97 * The proxy may hold a zstd dictionary trained on this repo's small objects
98 * (GET <repo>/.cvmfs-dict/current, X-Brix-Dict-Id = sha1 of the bytes). When
99 * -o dict / $BRIXCVMFS_DICT=1 is set, the dict is fetched ONCE per mount
100 * (memory-pinned; any fetch/verify failure disables the feature for the
101 * mount) and every CAS data GET then offers "X-Brix-Dict: <id>"; a response
102 * marked "Content-Encoding: zstd-dict" is decoded back to the STORED bytes
103 * inside the transport, so the fetch orchestrator's decompress+CAS-verify
104 * pipeline sees exactly what an identity response would carry. The id check
105 * is transport integrity only — TRUST stays with CAS verify: a hostile dict
106 * can only fail decode (zstd dictID mismatch) or produce bytes that fail
107 * the content hash; either way the client refetches identity. */
108
109typedef struct {
110 int zstd_dict; /* saw Content-Encoding: zstd-dict */
111 char id[CVMFS_DICT_ID_HEXLEN + 1]; /* saw X-Brix-Dict-Id (else "") */
112} brix_hdrwatch_t;

114/* Case-insensitive "Name: value" header line match; trims OWS + CRLF. */
115static int hdr_match(const char *p, size_t n, const char *name,
116 const char **val, size_t *vlen) {
117 size_t nl = strlen(name);
118 if (n < nl + 1 || strncasecmp(p, name, nl) != 0 || p[nl] != ':') return 0;
119 const char *v = p + nl + 1;
120 size_t rem = n - nl - 1;
121 while (rem && (*v == ' ' || *v == '\t')) { v++; rem--; }
122 while (rem && (v[rem-1] == '\r' || v[rem-1] == '\n'
123 || v[rem-1] == ' ' || v[rem-1] == '\t')) rem--;
124 *val = v; *vlen = rem;
125 return 1;
126}

128static size_t curl_hdr_cb(char *ptr, size_t sz, size_t nm, void *ud) {
129 brix_hdrwatch_t *hw = ud;
130 size_t n = sz * nm;
131 const char *v; size_t vl;
132 if (hdr_match(ptr, n, "content-encoding", &v, &vl)
133 && vl == sizeof("zstd-dict") - 1 && strncasecmp(v, "zstd-dict", vl) == 0)
134 hw->zstd_dict = 1;
135 else if (hdr_match(ptr, n, "x-brix-dict-id", &v, &vl)
136 && vl == CVMFS_DICT_ID_HEXLEN) {
137 memcpy(hw->id, v, vl);
138 hw->id[vl] = '\0';
139 }
140 return n;
141}

143/* One GET of `url`, RESUMING from `*got` bytes already in `out` (HTTP Range).
144 * Appends new bytes and updates *got. Returns the CURLcode. If a resume request
145 * comes back 200 (server ignored Range), the freshly re-sent from-0 bytes are
146 * slid to the front so the buffer stays a valid prefix.
147 * `dict_id` non-NULL adds "X-Brix-Dict: <id>"; `hw` non-NULL captures the
148 * response coding headers. Both are reset on the pooled handle every call so
149 * no request state leaks into the next borrower (same lesson as the G2 POST). */
150static CURLcode http_get_range(CURL **slot, const char *proxy, const char *url,
151 unsigned char *out, size_t outcap, size_t *got,
152 const char *dict_id, brix_hdrwatch_t *hw) {
153 if (*slot == NULL) *slot = curl_easy_init();
154 CURL *c = *slot;
155 if (c == NULL) return CURLE_FAILED_INIT;
156
157 size_t start = *got;
158 curl_sink_t sink = { out, outcap, start }; /* append after what we have */
159 struct curl_slist *req_hdrs = NULL;
160 if (dict_id != NULL) {
161 char line[64];
162 snprintf(line, sizeof(line), "X-Brix-Dict: %s", dict_id);
163 req_hdrs = curl_slist_append(NULL, line);
164 }
165 curl_easy_setopt(c, CURLOPT_URL, url);
166 curl_easy_setopt(c, CURLOPT_WRITEFUNCTION, curl_write_cb);
167 curl_easy_setopt(c, CURLOPT_WRITEDATA, &sink);
168 /* set-or-clear EVERY call: pooled-handle hygiene */
169 curl_easy_setopt(c, CURLOPT_HTTPHEADER, req_hdrs);
170 curl_easy_setopt(c, CURLOPT_HEADERFUNCTION, hw ? curl_hdr_cb : NULL);
171 curl_easy_setopt(c, CURLOPT_HEADERDATA, hw);
172 curl_easy_setopt(c, CURLOPT_CONNECTTIMEOUT, g_tcfg.connect_timeout_s);
173 curl_easy_setopt(c, CURLOPT_LOW_SPEED_LIMIT, g_tcfg.low_speed_bytes);
174 curl_easy_setopt(c, CURLOPT_LOW_SPEED_TIME, g_tcfg.low_speed_time_s);
175 curl_easy_setopt(c, CURLOPT_FOLLOWLOCATION, 1L);
176 curl_easy_setopt(c, CURLOPT_FAILONERROR, 1L); /* HTTP >=400 → error */
177 curl_easy_setopt(c, CURLOPT_TCP_KEEPALIVE, 1L);
178 /* Confine the transport to HTTP(S) for BOTH the first request and every
179 * redirect it follows. A poisoned mirror or a DPI middlebox can answer any
180 * request with a 3xx to file:///etc/passwd, an internal metadata IP
181 * (169.254.169.254), scp://, gopher://, … — FOLLOWLOCATION would chase it.
182 * The CAS layer hash-verifies content so a wrong body is caught, but that is
183 * no help if the redirect makes libcurl read a LOCAL file or poke an
184 * internal service in the first place: confine the scheme up front and cap
185 * the chain so a redirect loop can't wedge the mount either. The *_STR opts
186 * exist only from 7.85 (and the bitmask form is deprecated there), so gate
187 * on the header version — an enum option name is not #ifdef-able — keeping
188 * alma8-era libcurl portability. */
189#if CURL_AT_LEAST_VERSION(7, 85, 0)
190 curl_easy_setopt(c, CURLOPT_PROTOCOLS_STR, "http,https");
191 curl_easy_setopt(c, CURLOPT_REDIR_PROTOCOLS_STR, "http,https");
192#else
193 curl_easy_setopt(c, CURLOPT_PROTOCOLS, (long)(CURLPROTO_HTTP | CURLPROTO_HTTPS));
194 curl_easy_setopt(c, CURLOPT_REDIR_PROTOCOLS, (long)(CURLPROTO_HTTP | CURLPROTO_HTTPS));
195#endif
196 curl_easy_setopt(c, CURLOPT_MAXREDIRS, 4L);
197 /* TLS is defence-in-depth here (CVMFS content self-authenticates), but when
198 * a user asks for -o tls we must fail CLOSED against a MITM / intercepting
199 * DPI proxy rather than silently accept a forged cert. This restates
200 * libcurl's own default so a future edit can't quietly relax it. */
201 curl_easy_setopt(c, CURLOPT_SSL_VERIFYPEER, 1L);
202 curl_easy_setopt(c, CURLOPT_SSL_VERIFYHOST, 2L);
203 /* Reused handle: set these unconditionally so a prior request's resume
204 * offset / freshness flags can never leak into this one. */
205 curl_easy_setopt(c, CURLOPT_RESUME_FROM_LARGE, (curl_off_t) start);
206 curl_easy_setopt(c, CURLOPT_FRESH_CONNECT, g_tcfg.fresh_connect ? 1L : 0L);
207 curl_easy_setopt(c, CURLOPT_FORBID_REUSE, g_tcfg.fresh_connect ? 1L : 0L);
208 /* Proxy precedence: env proxy wins; else CVMFS-config proxy; else force direct. */
209 char sch[8], thost[256]; int tport;
210 url_host(url, sch, sizeof(sch), thost, sizeof(thost), &tport);
211 brix_proxy_t px;
212 if (brix_proxy_resolve(sch, thost, tport, &px)) {
213 curl_easy_setopt(c, CURLOPT_PROXY, px.url);
214 brix_proxy_report(&px, thost, tport);
215 } else if (proxy != NULL && strcmp(proxy, "DIRECT") != 0) {
216 curl_easy_setopt(c, CURLOPT_PROXY, proxy);
217 } else {
218 curl_easy_setopt(c, CURLOPT_PROXY, "");
219 }
220
221 CURLcode rc = curl_easy_perform(c);
222 if (req_hdrs != NULL) {
223 curl_easy_setopt(c, CURLOPT_HTTPHEADER, NULL); /* before the free */
224 curl_slist_free_all(req_hdrs);
225 }
226 long code = 0;
227 curl_easy_getinfo(c, CURLINFO_RESPONSE_CODE, &code);
228 if (start > 0 && code != 206 && sink.len > start) {
229 /* Range not honoured (200): the bytes appended after `start` are actually
230 * a fresh stream from offset 0 — slide them to the front. */
231 size_t fresh = sink.len - start;
232 memmove(out, out + start, fresh);
233 *got = fresh;
234 } else {
235 *got = sink.len;
236 }
237 return rc;
238}

240/* Rewrite an http:// url to https:// into `buf`; returns 1 if rewritten.
241 *
242 * A rewrite that would not fit reports "not rewritten" rather than emitting a
243 * truncated URL — a shortened URL is a *different* URL, so the TLS probe would
244 * request an object nobody asked for and the 4xx would be misread as "this
245 * mirror has no TLS". Sizing the destination BRIXCVMFS_URL_HTTPS_MAX makes that
246 * unreachable for our own caller; the check keeps the helper safe for any other. */
247static int to_https(const char *url, char *buf, size_t n) {
248 if (strncmp(url, "http://", 7) != 0) return 0;
249 int len = snprintf(buf, n, "https://%s", url + 7);
250 return (len >= 0 && (size_t) len < n) ? 1 : 0;
251}

253/* Mount-lifetime dict state (see the G3 block comment above curl_hdr_cb). */
254typedef struct {
255 int mode; /* 0=off · 1=armed (fetch on first data GET)
256 · 2=ready · -1=disabled for this mount */
257 pthread_mutex_t mu;
258 char id[CVMFS_DICT_ID_HEXLEN + 1];
259 unsigned char *bytes; /* malloc'd, memory-pinned until unmount */
260 size_t len;
261} brix_dict_state_t;

263static brix_dict_state_t g_dict = { 0, PTHREAD_MUTEX_INITIALIZER, "", NULL, 0 };
264
265/* One-shot dict pull + self-certification (sha1(body) == X-Brix-Dict-Id).
266 * Any failure means "this mount runs identity" — never retried, so a dead
267 * or dict-less proxy costs exactly one extra GET per mount. */
268static int brix_dict_fetch(CURL **slot, const char *proxy, const char *host) {
269 char url[1024];
270 snprintf(url, sizeof(url), "%s/.cvmfs-dict/current", host);
271 unsigned char *buf = malloc(CVMFS_DICT_MAX_BYTES);
272 if (buf == NULL) return -1;
273 size_t got = 0;
274 brix_hdrwatch_t hw; memset(&hw, 0, sizeof(hw));
275 char myid[CVMFS_DICT_ID_HEXLEN + 1];
276 if (http_get_range(slot, proxy, url, buf, CVMFS_DICT_MAX_BYTES, &got,
277 NULL, &hw) != CURLE_OK
278 || got == 0
279 || strlen(hw.id) != CVMFS_DICT_ID_HEXLEN
280 || cvmfs_dict_id(buf, got, myid) != 0
281 || memcmp(myid, hw.id, CVMFS_DICT_ID_HEXLEN) != 0)
282 {
283 free(buf);
284 return -1;
285 }
286 g_dict.bytes = buf;
287 g_dict.len = got;
288 memcpy(g_dict.id, myid, sizeof(myid));
289 return 0;
290}

292/* Armed → fetch once (serialized; losers of the race reuse the outcome).
293 * Returns 1 iff a verified dict is ready for use. */
294static int brix_dict_ensure(CURL **slot, const char *proxy, const char *host) {
295 int ready;
296 pthread_mutex_lock(&g_dict.mu);
297 if (g_dict.mode == 1) {
298 g_dict.mode = (brix_dict_fetch(slot, proxy, host) == 0) ? 2 : -1;
299 fprintf(stderr, "brixcvmfs: dict %s%s\n",
300 g_dict.mode == 2 ? "ready id=" : "disabled (no verified "
301 ".cvmfs-dict/current — serving identity)",
302 g_dict.mode == 2 ? g_dict.id : "");
303 }
304 ready = (g_dict.mode == 2);
305 pthread_mutex_unlock(&g_dict.mu);
306 return ready;
307}

309/* G3: offer the dict on CAS data GETs only (never manifests/catalog
310 * fetches — those exceed the size class and the server declines anyway).
311 * Lazily pulls .cvmfs-dict/current on the first qualifying fetch. */
312static int brix_dict_offer(CURL **slot, const char *proxy, const char *host,
313 const char *rel) {
314 return (g_dict.mode != 0 && g_dict.mode != -1

319/* dict-coded body → decode back to the STORED bytes in place. Failure
320 * (wrong/hostile dict, truncation, overflow) is never fatal: the caller
321 * drops the dict for THIS fetch and pulls identity. */
322static int brix_dict_decode(unsigned char *out, size_t outcap, size_t got,
323 size_t *outlen) {
324 unsigned char *dec = malloc(outcap);
325 size_t dlen = 0;
326 if (dec != NULL
327 && cvmfs_dict_decompress(g_dict.bytes, g_dict.len,
328 out, got, dec, outcap, &dlen) == 0)
329 {
330 memcpy(out, dec, dlen);
331 free(dec);
332 *outlen = dlen;
333 return 0;
334 }
335 free(dec);
336 return -1;
337}

339/* First attempt may probe https when prefer_tls is set; every later attempt
340 * (and non-TLS configs) uses the plain http mirror URL. */
341static const char *transport_url(int first, const char *httpurl,
342 char *httpsbuf, size_t cap, int *use_https) {
343 *use_https = (first && g_tcfg.prefer_tls
344 && to_https(httpurl, httpsbuf, cap));
345 return *use_https ? httpsbuf : httpurl;
346}

348/* Classify one attempt's outcome. 1 = done (*outlen set), -1 = hard failure
349 * (fail over to the next mirror), 2 = retry immediately (identity refetch
350 * after a dict drop, no backoff), 0 = retry via the stall budget. */
351static int transport_after(CURLcode rc, int use_https, const brix_hdrwatch_t *hw,
352 int *dict_on, unsigned char *out, size_t outcap,
353 size_t *got, size_t *last, size_t *outlen) {
354 if (rc == CURLE_OK && *dict_on && hw->zstd_dict) {
355 if (brix_dict_decode(out, outcap, *got, outlen) == 0) return 1;
356 *dict_on = 0; *got = 0; *last = 0;
357 return 2; /* identity refetch, no backoff */
358 }
359 if (rc == CURLE_OK) { *outlen = *got; return 1; }
360
361 /* A dict-coded response is not resumable (an identity 206 cannot
362 * continue a coded prefix): restart the object from 0. */
363 if (*dict_on && *got > 0) { *got = 0; *last = 0; }
364
365 /* hard 4xx over plain http = mirror lacks the object → fail over. A 4xx on
366 * the https probe just means no TLS — fall through to http. */
367 if (rc == CURLE_HTTP_RETURNED_ERROR && !use_https) return -1;
368
369 /* Range-blind server: libcurl aborts a resumed request answered 200 with
370 * CURLE_RANGE_ERROR before any body byte reaches the sink, so resume can
371 * never progress there — throw away the partial prefix and restart the
372 * whole object from 0 (the fetch layer's hash check keeps this safe). */
373 if (rc == CURLE_RANGE_ERROR) { *got = 0; *last = 0; }
374
375 return 0;
376}

378/* Progress bookkeeping between attempts. -1 = stall budget exhausted. */
379static int transport_stall(size_t got, size_t *last, int *stalls, int budget) {
380 if (got > *last) { *last = got; *stalls = 0; } /* progress → keep resuming */
381 else if (++*stalls > budget) return -1; /* no progress → give up route */
382
383 long ms = 200L << (*stalls < 6 ? *stalls : 6); /* backoff on stalls only */
384 struct timespec ts = { ms / 1000, (ms % 1000) * 1000000L };
385 nanosleep(&ts, NULL);
386 return 0;
387}

389/* DPI/loss-hardened transport: RANGE-RESUME on a severed transfer (so a large
390 * object survives per-chunk connection loss — the key to 10%+ loss), optional
391 * TLS-first, optional fresh-connection. As long as each attempt makes progress
392 * (more bytes received) we keep resuming up to a generous cap; only attempts that
393 * make NO progress count against the retry budget. Across-mirror failover +
394 * hash-verify are owned by the fetch layer. */
395int brixcvmfs_transport(const char *proxy, const char *host, const char *rel,
396 unsigned char *out, size_t outcap, size_t *outlen, void *ud) {
397 (void) ud; /* handles now come from g_curl_pool */
398 brix_status st; brix_status_clear(&st);
399 CURL **slot = brix_cpool_checkout(g_curl_pool, &st); /* CURL** == the classic slot */
400 if (slot == NULL) return -1;
401
402 char httpurl[BRIXCVMFS_URL_MAX];
403 int urllen = snprintf(httpurl, sizeof(httpurl), "%s/%s", host, rel);
404 if (urllen < 0 || (size_t) urllen >= sizeof(httpurl)) {
405 /* Truncated here would mean GETting a different object and hash-failing
406 * it against the one we asked for — fail the mirror instead. */
407 brix_cpool_checkin(g_curl_pool, slot, 1);
408 return -1;
409 }
410
411 int budget = g_tcfg.max_retries > 0 ? g_tcfg.max_retries : 6;
412 int hard_cap = 64; /* absolute ceiling on resume attempts */
413 int stalls = 0;
414 size_t got = 0, last = 0;
415 int ret = -1;
416 int dict_on = brix_dict_offer(slot, proxy, host, rel);
417
418 for (int i = 0; i < hard_cap; i++) {
419 char httpsbuf[BRIXCVMFS_URL_HTTPS_MAX];
420 int use_https = 0;
421 const char *url = transport_url(i == 0, httpurl, httpsbuf,
422 sizeof(httpsbuf), &use_https);
423
424 brix_hdrwatch_t hw; memset(&hw, 0, sizeof(hw));
425 CURLcode rc = http_get_range(slot, proxy, url, out, outcap, &got,
426 (dict_on && got == 0) ? g_dict.id : NULL,
427 dict_on ? &hw : NULL);
428 int act = transport_after(rc, use_https, &hw, &dict_on, out, outcap,
429 &got, &last, outlen);
430 if (act == 1) { ret = 0; break; }
431 if (act == -1) break;
432 if (act == 2) continue;
433 if (transport_stall(got, &last, &stalls, budget) != 0) break;
434 }
435 /* A libcurl easy handle stays healthy across transfers (it owns its own
436 * connection cache and re-establishes internally), so always check it back
437 * in reusable — never health-drop. */
438 brix_cpool_checkin(g_curl_pool, slot, 1);
439 return ret;
440}

442/* One bundle POST (phase-87 G2): want-list body up, framed reply down. Single
443 * attempt, no retry — the caller treats ANY failure as "no bundle" and the
444 * per-item fetch path does the real work. The pooled easy handle is shared
445 * with GET users, so POST mode is reverted before check-in. */
446int bundle_http_post(const char *proxy, const char *host,
447 const char *body, size_t body_len,
448 unsigned char *out, size_t outcap, size_t *outlen) {
449 brix_status st; brix_status_clear(&st);
450 CURL **slot = brix_cpool_checkout(g_curl_pool, &st);
451 if (slot == NULL) return -1;
452 if (*slot == NULL) *slot = curl_easy_init();
453 CURL *c = *slot;
454 int ret = -1;
455
456 if (c != NULL) {
457 char url[1024];
458 snprintf(url, sizeof(url), "%s/.cvmfs-bundle", host);
459 curl_sink_t sink = { out, outcap, 0 };
460 curl_easy_setopt(c, CURLOPT_URL, url);
461 curl_easy_setopt(c, CURLOPT_POST, 1L);
462 curl_easy_setopt(c, CURLOPT_POSTFIELDS, body);
463 curl_easy_setopt(c, CURLOPT_POSTFIELDSIZE_LARGE, (curl_off_t) body_len);
464 curl_easy_setopt(c, CURLOPT_WRITEFUNCTION, curl_write_cb);
465 curl_easy_setopt(c, CURLOPT_WRITEDATA, &sink);
466 curl_easy_setopt(c, CURLOPT_CONNECTTIMEOUT, g_tcfg.connect_timeout_s);
467 curl_easy_setopt(c, CURLOPT_LOW_SPEED_LIMIT, g_tcfg.low_speed_bytes);
