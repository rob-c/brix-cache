1/*
2 * vfs_dir.c — VFS directory enumeration (opendir/readdir/closedir).
3 *
4 * WHAT: Implements brix_vfs_opendir(), brix_vfs_readdir(), and
5 * brix_vfs_closedir() over the opaque brix_vfs_dir_t handle. readdir
6 * yields one entry per call as a pooled, NUL-terminated ngx_str_t, with an
7 * optional lstat of the child filled into an brix_vfs_stat_t.
8 *
9 * WHY: Directory listing (XRootD kXR_dirlist, WebDAV PROPFIND, S3 LIST) needs
10 * confinement, the "." / ".." filter, and a NGX_DONE end-of-stream signal
11 * to be handled once, the same way, for every protocol — rather than each
12 * front end driving opendir/readdir itself.
13 *
14 * HOW: opendir re-verifies confinement, pcalloc's the handle on ctx->pool,
15 * dups the resolved path, and opens the C-library DIR*; the open itself is
16 * observed as BRIX_METRIC_OP_DIRLIST. readdir loops skipping "."/".."
17 * (and entries are distinguished from a real error via errno-cleared
18 * readdir, returning NGX_DONE when the stream ends), copies the name into
19 * the pool, and optionally builds "<dir>/<name>" for an lstat. closedir
20 * calls closedir(3) and nulls the handle so it is idempotent.
21 */
22#include "vfs_internal.h"
23#include "core/compat/log_diag.h"
24#include "auth/impersonate/impersonate.h"

26/* vfs_opendir_fail — shared error tail for the opendir body.
27 *
28 * WHAT: Reports a failed opendir: copies `err` into *err_out (when the caller
29 * asked for it), meters the failure as OP_DIRLIST when `observe` is set,
30 * and returns NULL for the caller to relay.
31 * WHY: Every early-out of the opendir body ends the same way; folding the
32 * report into one helper keeps the body's control flow flat.
33 * HOW: Pure reporting — errno is deliberately NOT touched here so each call
34 * site preserves exactly the errno state the pre-refactor code left; the
35 * observed path is the ctx's resolved path (what every caller reports). */
36static brix_vfs_dir_t *
37vfs_opendir_fail(brix_vfs_ctx_t *ctx, int err, int observe, uint64_t start,
38 int *err_out)
39{
40 if (err_out != NULL) {
41 *err_out = err;
42 }
43 if (observe) {
44 brix_vfs_observe_ctx_op(ctx, brix_vfs_ctx_path(ctx),
45 BRIX_METRIC_OP_DIRLIST, NULL, 0,
46 NGX_ERROR, err, start);
47 }
48 return NULL;
49}

51/* vfs_dir_route — the SECURITY SEAM: export confinement + plane selection.
52 *
53 * WHAT: Verifies the resolved ctx path is confined to the export root, then
54 * selects the enumeration plane: a non-POSIX driver iterator (*drv_out
55 * set) or the confined POSIX/broker opendir (*drv_out NULL).
56 * WHY: Confinement MUST be re-verified before any directory is opened, and
57 * the driver-vs-broker decision must live in exactly one place so no
58 * future path can enumerate an unconfined directory.
59 * HOW: brix_vfs_require_confined() gates everything; on failure *err carries
60 * its errno for the caller's report (errno itself is left untouched). */
61static ngx_int_t
62vfs_dir_route(brix_vfs_ctx_t *ctx, const brix_sd_driver_t **drv_out, int *err)
63{
64 if (brix_vfs_require_confined(ctx) != NGX_OK) {
65 *err = errno;
66 return NGX_ERROR;
67 }
68 *drv_out = brix_vfs_ctx_driver(ctx);
69 return NGX_OK;
70}

72/* vfs_dir_open_driver — open a non-POSIX backend's directory iterator.
73 *
74 * WHAT: Resolves the export-relative logical path, loads the per-user backend
75 * credential when the export is credential-scoped, and opens the leaf
76 * driver's directory iterator into dh->sd_dir, filling the handle's
77 * driver-plane fields.
78 * WHY: Object/remote backends have no DIR*; enumeration goes through the
79 * driver's opendir/readdir verbs instead of the confined POSIX path.
80 * HOW: Returns NGX_OK, or NGX_ERROR with *err set to the errno to report
81 * (errno itself only set where the pre-refactor code set it: the
82 * credential-load failure). Dispatches on the leaf instance so
83 * brix_sd_opendir_maybe_cred finds the leaf driver's opendir_cred slot
84 * (decorators have only plain relays). */
85static ngx_int_t
86vfs_dir_open_driver(brix_vfs_ctx_t *ctx, brix_vfs_dir_t *dh,
87 const brix_sd_driver_t *drv, const char *path, int *err_out)
88{
89 const char *logical = brix_vfs_export_relative(ctx, path);
90 brix_sd_ucred_t store;
91 brix_sd_cred_t cred;
92 int use_cred = 0, cred_err = 0;
93 int err = 0;
94
95 /* Zero before the gate: it fills only the active credential kind; an
96 * unzeroed cred hands a garbage inactive pointer to the driver's cred
97 * slot (bearer PASSTHROUGH would leave x509_proxy dangling). */
98 ngx_memzero(&cred, sizeof(cred));
99
100 if (brix_vfs_cred_gate_active(ctx)) {
101 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
102 != NGX_OK)
103 {
104 *err_out = cred_err ? cred_err : EACCES;
105 errno = *err_out;
106 return NGX_ERROR;
107 }
108 }
109
110 if (drv->opendir == NULL) {
111 errno = ENOTSUP;
112 }
113 dh->sd_dir = (drv->opendir != NULL)
114 ? brix_sd_opendir_maybe_cred(brix_vfs_ns_leaf(ctx->sd),
115 logical, &err, use_cred ? &cred : NULL)
116 : NULL;
117 brix_sd_ucred_wipe(&store); /* secret consumed by opendir; erase (A-4/T4) */
118 if (dh->sd_dir == NULL) {
119 *err_out = (err != 0) ? err : errno;
120 return NGX_ERROR;
121 }
122
123 dh->sd = ctx->sd;
124 dh->drv = drv;
125 dh->sd_logical = brix_vfs_copy_path(ctx->pool, logical);
126 dh->pool = ctx->pool;
127 dh->log = ctx->log;
128 return NGX_OK;
129}

131/* vfs_dir_open_confined — the confined POSIX/broker opendir (do NOT weaken).
132 *
133 * WHAT: Opens dh->dir under export-root confinement and fills the handle's
134 * POSIX-plane fields.
135 * WHY: brix_opendir_confined_canon is the openat2 RESOLVE_IN_ROOT path from
136 * the symlink-escape fix — an outward symlink inside an export must stay
137 * invisible — and under impersonation it opens AS THE MAPPED USER
138 * (broker fdopendir) so a 0700 user-owned / 0770 group-restricted dir
139 * the unprivileged worker cannot itself open is enumerable by its
140 * legitimate owner/group-member; off impersonation it is a bare
141 * opendir().
142 * HOW: Returns NGX_OK, or NGX_ERROR with errno left exactly as the confined
143 * open set it. */
144static ngx_int_t
145vfs_dir_open_confined(brix_vfs_ctx_t *ctx, brix_vfs_dir_t *dh,
146 const char *path)
147{
148 dh->dir = brix_opendir_confined_canon(ctx->log, ctx->root_canon, path);
149 if (dh->dir == NULL) {
150 return NGX_ERROR;
151 }
152
153 dh->pool = ctx->pool;
154 dh->log = ctx->log;
155 dh->root_canon = ctx->root_canon;
156 return NGX_OK;
157}

159/* Shared opendir body. When `observe` is set the open is metered as OP_DIRLIST;
160 * the quiet variant (observe=0) skips the metric/access-log entirely — for bulk
161 * recursive walks (S3 ListObjects, WebDAV SEARCH) whose enclosing protocol op
162 * already accounts for the traversal and would otherwise emit one phantom
163 * OP_DIRLIST per visited subdirectory. */
164static brix_vfs_dir_t *
165brix_vfs_opendir_impl(brix_vfs_ctx_t *ctx, int *err_out, int observe)
166{
167 brix_vfs_dir_t *dh;
168 const char *path;
169 const brix_sd_driver_t *drv = NULL;
170 uint64_t start;
171 int err = 0;
172
173 start = brix_vfs_now_ns();
174
175 if (err_out != NULL) {
176 *err_out = 0;
177 }
178
179 if (vfs_dir_route(ctx, &drv, &err) != NGX_OK) {
180 return vfs_opendir_fail(ctx, err, observe, start, err_out);
181 }
182
183 path = brix_vfs_ctx_path(ctx);
184 dh = ngx_pcalloc(ctx->pool, sizeof(*dh));
185 if (dh == NULL) {
186 errno = ENOMEM;
187 return vfs_opendir_fail(ctx, ENOMEM, observe, start, err_out);
188 }
189
190 dh->path = brix_vfs_copy_path(ctx->pool, path);
191 if (dh->path == NULL) {
192 return vfs_opendir_fail(ctx, errno, observe, start, err_out);
193 }
194
195 if (drv != NULL) {
196 /* Non-POSIX backend: enumerate through the driver's iterator. */
197 if (vfs_dir_open_driver(ctx, dh, drv, path, &err) != NGX_OK) {
198 return vfs_opendir_fail(ctx, err, observe, start, err_out);
199 }
200 } else if (vfs_dir_open_confined(ctx, dh, path) != NGX_OK) {
201 return vfs_opendir_fail(ctx, errno, observe, start, err_out);
202 }
203
204 if (observe) {
205 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DIRLIST, NULL, 0,
206 NGX_OK, 0, start);
207 }
208 return dh;
209}

236/* vfs_readdir_fill_entry — copy an entry name into the pooled name_out.
237 *
238 * WHAT: Fills *name_out with a pooled, NUL-terminated copy of `name`.
239 * WHY: Both readdir variants, on both planes, end by handing the caller a
240 * pool-owned copy of the entry name; one helper kills the copy-paste.
241 * HOW: ngx_pnalloc(len+1) + memcpy + explicit NUL; NGX_ERROR with
242 * errno=ENOMEM on allocation failure. */
243static ngx_int_t
244vfs_readdir_fill_entry(ngx_pool_t *pool, const char *name,
245 ngx_str_t *name_out)
246{
247 name_out->len = ngx_strlen(name);
248 name_out->data = ngx_pnalloc(pool, name_out->len + 1);
249 if (name_out->data == NULL) {
250 errno = ENOMEM;
251 return NGX_ERROR;
252 }
253 ngx_memcpy(name_out->data, name, name_out->len);
254 name_out->data[name_out->len] = '\0';
255 return NGX_OK;
256}

258/* vfs_readdir_next_posix — next POSIX dirent, "." / ".." filtered out.
259 *
260 * WHAT: Yields the next non-dot entry of dh->dir into *de_out.
261 * WHY: Both readdir variants share the errno-cleared readdir(3) idiom that
262 * distinguishes end-of-stream from a real error, and the dot filter.
263 * HOW: errno=0 before each readdir(3): NULL with errno still 0 is NGX_DONE,
264 * NULL with errno set is NGX_ERROR; "." and ".." are skipped in-loop. */
265static ngx_int_t
266vfs_readdir_next_posix(brix_vfs_dir_t *dh, struct dirent **de_out)
267{
268 struct dirent *de;
269
270 for ( ;; ) {
271 errno = 0;
272 de = readdir(dh->dir);
273 if (de == NULL) {
274 return errno == 0 ? NGX_DONE : NGX_ERROR;
275 }
276
277 if (de->d_name[0] == '.'
278 && (de->d_name[1] == '\0'
279 || (de->d_name[1] == '.' && de->d_name[2] == '\0')))
280 {
281 continue;
282 }
283
284 *de_out = de;
285 return NGX_OK;
286 }
287}

289/* vfs_readdir_stat_child — confined lstat of one POSIX entry.
290 *
291 * WHAT: Joins "<dir>/<name>", lstats it under export confinement, and fills
292 * *stat_out on success.
293 * WHY: Per-entry stat must SKIP a bad entry rather than truncate the listing;
294 * returning NGX_ERROR here lets the scan loop drop the entry and go on.
295 * HOW: Off impersonation, a dirfd-relative fstatat of the bare entry name —
296 * O(1) per child, no join, no re-walk: dh->dir was opened through the
297 * RESOLVE_IN_ROOT-confined opendir, and readdir names contain no '/',
298 * so the child cannot escape the already-confined directory. Under
299 * impersonation, the snprintf join + brix_lstat_confined_canon AS THE
300 * MAPPED USER (broker-routed) is mandatory — mapped-user DAC — and is
301 * kept unchanged. lstat/AT_SYMLINK_NOFOLLOW keeps outward symlinks
302 * unfollowed on both paths. The caller skips a failed entry; only the
303 * benign unlink race (ENOENT) is silent — EIO/EACCES etc. are logged so
304 * a shrinking listing is diagnosable. */
305static ngx_int_t
306vfs_readdir_stat_child(brix_vfs_dir_t *dh, const char *name,
307 brix_vfs_stat_t *stat_out)
308{
309 char child[PATH_MAX];
310 struct stat st;
311 int n, rc;
312
313 if (!brix_imp_client_active()) {
314 int dfd = dirfd(dh->dir);
315
316 if (dfd < 0) {
317 /* Only reachable with a torn-down handle; fstatat would answer
318 * EBADF and the entry would vanish with a misleading errno. */
319 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
320 "xrootd[disk]: dirlist of \"%s\" has no directory "
321 "descriptor; entry \"%s\" omitted from the listing",
322 dh->path, name);
323 return NGX_ERROR;
324 }
325 rc = fstatat(dfd, name, &st, AT_SYMLINK_NOFOLLOW);
326 if (rc != 0) {
327 if (errno != ENOENT) {
328 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
329 "xrootd[disk]: dirlist stat of entry \"%s\" "
330 "under \"%s\" failed; entry omitted from the "
331 "listing", name, dh->path);
332 }
333 return NGX_ERROR;
334 }
335 brix_vfs_fill_stat(&st, stat_out);
336 return NGX_OK;
337 }
338
339 n = snprintf(child, sizeof(child), "%s/%s", dh->path, name);
340 if (n < 0 || (size_t) n >= sizeof(child)) {
341 errno = ENAMETOOLONG;
342 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
343 "xrootd[disk]: dirlist entry \"%s\" under \"%s\" joins "
344 "to an unrepresentable path; entry omitted",
345 name, dh->path);
346 return NGX_ERROR;
347 }
348 if (brix_lstat_confined_canon(dh->log, dh->root_canon, child,
349 &st, 1) != 0) {
350 if (errno != ENOENT) {
351 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
352 "xrootd[disk]: dirlist stat of \"%s\" failed; "
353 "entry omitted from the listing", child);
354 }
355 return NGX_ERROR;
356 }
357 brix_vfs_fill_stat(&st, stat_out);
358 return NGX_OK;
359}

361/* vfs_sd_stat_child — stat one driver-plane entry through the driver.
362 *
363 * WHAT: Joins the handle's logical directory with `name` and stats the child
364 * via dh->drv->stat into *sd_st.
365 * WHY: Both readdir variants build the same "<logical>/<name>" join (with
366 * the "/"-root special case) before the driver stat.
367 * HOW: Fixed PATH_MAX join via ngx_snprintf, then the driver's stat verb;
368 * the caller must have checked dh->drv->stat != NULL. */
369static ngx_int_t
370vfs_sd_stat_child(brix_vfs_dir_t *dh, const char *name, brix_sd_stat_t *sd_st)
371{
372 char child[PATH_MAX];
373
374 ngx_snprintf((u_char *) child, sizeof(child), "%s/%s%Z",
375 (dh->sd_logical[0] == '/' && dh->sd_logical[1] == '\0')
376 ? "" : dh->sd_logical, name);
377 return dh->drv->stat(dh->sd, child, sd_st);
378}

380/* vfs_readdir_sd — driver-plane body of brix_vfs_readdir.
381 *
382 * WHAT: Pulls the next entry from the driver iterator, stats the child
383 * through the same driver when the caller wants stats, and fills the
384 * pooled name.
385 * WHY: Keeps the driver iterator's skip-on-vanished-child scan out of the
386 * public entry point.
387 * HOW: Loops (instead of the former tail-recursion) so a child that vanished
388 * mid-scan is skipped; only NGX_DONE/NGX_ERROR from the iterator stop
389 * the caller's loop. */
390static ngx_int_t
391vfs_readdir_sd(brix_vfs_dir_t *dh, ngx_str_t *name_out,
392 brix_vfs_stat_t *stat_out)
393{
394 brix_sd_dirent_t de_sd;
395 brix_sd_stat_t sd_st;
396 ngx_int_t rc;
397
398 for ( ;; ) {
399 /* Zeroed so a driver that fills only the name yields d_type ==
400 * DT_UNKNOWN (= 0), never stack garbage. */
401 ngx_memzero(&de_sd, sizeof(de_sd));
402 rc = dh->drv->readdir(dh->sd_dir, &de_sd);
403 if (rc != NGX_OK) {
404 return rc; /* NGX_DONE / NGX_ERROR */
405 }
406 if (stat_out == NULL || dh->drv->stat == NULL) {
407 break;
408 }
409 if (vfs_sd_stat_child(dh, de_sd.name, &sd_st) == NGX_OK) {
410 brix_vfs_sd_stat_fill(&sd_st, stat_out);
411 break;
412 }
413 /* Child vanished mid-scan (ENOENT): skip it and keep enumerating.
414 * Any other driver stat failure is surfaced before the skip. */
415 if (errno != ENOENT) {
416 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
417 "xrootd[disk]: dirlist driver stat of entry \"%s\" "
418 "under \"%s\" failed; entry omitted from the "
419 "listing", de_sd.name, dh->sd_logical);
420 }
421 }
422
423 return vfs_readdir_fill_entry(dh->pool, de_sd.name, name_out);
424}

426/* Return the next entry: name as a pooled NUL-terminated ngx_str_t, plus an
427 * optional lstat of the child. Skips "." and ".."; returns NGX_DONE at
428 * end-of-stream and NGX_ERROR (errno set) on failure. */
429ngx_int_t
430brix_vfs_readdir(brix_vfs_dir_t *dh, ngx_str_t *name_out,
431 brix_vfs_stat_t *stat_out)
432{
433 struct dirent *de;
434 ngx_int_t rc;
435
436 if (dh == NULL || name_out == NULL
437 || (dh->dir == NULL && dh->sd_dir == NULL))
438 {
439 errno = EINVAL;
440 return NGX_ERROR;
441 }
442
443 /* Non-POSIX backend: enumerate through the driver iterator. */
444 if (dh->sd_dir != NULL) {
445 return vfs_readdir_sd(dh, name_out, stat_out);
446 }
447
448 /* Per-entry stat is folded into the scan so a single bad entry SKIPS
449 * rather than truncating the listing: a child that races an unlink
450 * (ENOENT), or whose joined path is unrepresentable, is dropped and the
451 * scan continues. (Only NGX_DONE/NGX_ERROR — true end/stream error —
452 * stop the caller's loop.) */
453 for ( ;; ) {
454 rc = vfs_readdir_next_posix(dh, &de);
455 if (rc != NGX_OK) {
456 return rc;
457 }
458 if (stat_out == NULL
459 || vfs_readdir_stat_child(dh, de->d_name, stat_out) == NGX_OK)
460 {
461 break;
462 }
463 }
464
465 return vfs_readdir_fill_entry(dh->pool, de->d_name, name_out);
466}

468/* Yield the next entry's name plus its KIND from the readdir d_type, with no
469 * per-entry stat — for callers that classify dir-vs-file on the fast path and
470 * only stat (via brix_vfs_probe) on a DT_UNKNOWN filesystem. Skips "."/"..";
471 * NGX_DONE at end-of-stream, NGX_ERROR (errno set) on failure. */
472/* vfs_posix_dtype_kind — map a readdir d_type onto the VFS entry kind.
473 *
474 * WHAT: DT_DIR/DT_REG/DT_UNKNOWN map onto their VFS kinds; anything else
475 * (symlink, fifo, socket, ...) is DT_OTHER.
476 * WHY: Callers classify dir-vs-file on the fast path and only stat on a
477 * DT_UNKNOWN filesystem; the mapping is a pure table.
478 * HOW: A single switch — no side effects. */
479static brix_vfs_dirent_kind_t
480vfs_posix_dtype_kind(unsigned char d_type)
481{
482 switch (d_type) {
483 case DT_DIR: return BRIX_VFS_DT_DIR;
484 case DT_REG: return BRIX_VFS_DT_REG;
485 case DT_UNKNOWN: return BRIX_VFS_DT_UNKNOWN;
486 default: return BRIX_VFS_DT_OTHER;
487 }
488}

490/* vfs_sd_entry_kind — classify one driver-plane entry as dir/file.
491 *
492 * WHAT: Returns the entry's kind, preferring the dirent's own d_type and
493 * falling back to a driver stat of the joined child.
494 * WHY: A backend that classifies during enumeration (POSIX readdir) makes
495 * the per-entry stat redundant; one that cannot leaves d_type ==
496 * DT_UNKNOWN (never guessed) and the stat verb decides. Kind is
497 * display/routing metadata only — NEVER an authorization input.
498 * HOW: Non-UNKNOWN d_type maps through the same pure table as the POSIX
499 * plane; on UNKNOWN, guard dh->drv->stat, join via vfs_sd_stat_child,
500 * and map is_dir → DT_DIR / DT_REG. */
501static brix_vfs_dirent_kind_t
502vfs_sd_entry_kind(brix_vfs_dir_t *dh, const brix_sd_dirent_t *de)
503{
504 brix_sd_stat_t sd_st;
505
506 if (de->d_type != DT_UNKNOWN) {
507 return vfs_posix_dtype_kind(de->d_type);
508 }
509
510 if (dh->drv->stat == NULL
511 || vfs_sd_stat_child(dh, de->name, &sd_st) != NGX_OK)
512 {
513 return BRIX_VFS_DT_UNKNOWN;
514 }
515 return sd_st.is_dir ? BRIX_VFS_DT_DIR : BRIX_VFS_DT_REG;
516}

518ngx_int_t
519brix_vfs_readdir_kind(brix_vfs_dir_t *dh, ngx_str_t *name_out,
520 brix_vfs_dirent_kind_t *kind_out)
521{
522 struct dirent *de;
523 ngx_int_t rc;
524
525 if (dh == NULL || name_out == NULL
526 || (dh->dir == NULL && dh->sd_dir == NULL))
527 {
528 errno = EINVAL;
529 return NGX_ERROR;
530 }
531
532 if (dh->sd_dir != NULL) {
533 brix_sd_dirent_t de_sd;
534
535 /* Zeroed: a name-only driver yields d_type == DT_UNKNOWN, not junk. */
536 ngx_memzero(&de_sd, sizeof(de_sd));
537 rc = dh->drv->readdir(dh->sd_dir, &de_sd);
538 if (rc != NGX_OK) {
539 return rc;
540 }
541 if (kind_out != NULL) {
542 *kind_out = vfs_sd_entry_kind(dh, &de_sd);
543 }
544 return vfs_readdir_fill_entry(dh->pool, de_sd.name, name_out);
545 }
546
547 rc = vfs_readdir_next_posix(dh, &de);
548 if (rc != NGX_OK) {
549 return rc;
550 }
551
552 if (kind_out != NULL) {
553 *kind_out = vfs_posix_dtype_kind(de->d_type);
554 }
555
556 return vfs_readdir_fill_entry(dh->pool, de->d_name, name_out);
557}

559/* Close the directory stream and null the handle (idempotent). Logs and returns
560 * NGX_ERROR if closedir(3) fails. */
561ngx_int_t
562brix_vfs_closedir(brix_vfs_dir_t *dh, ngx_log_t *log)
563{
564 if (dh == NULL || (dh->dir == NULL && dh->sd_dir == NULL)) {
565 return NGX_OK;
566 }
567
568 if (dh->sd_dir != NULL) {
569 ngx_int_t rc = (dh->drv->closedir != NULL)
570 ? dh->drv->closedir(dh->sd_dir) : NGX_OK;
571
572 dh->sd_dir = NULL;
573 return rc;
574 }
575
576 if (closedir(dh->dir) != 0) {
577 BRIX_DIAG_ERR(log != NULL ? log : dh->log, errno,
578 "xrootd[disk]: closedir failed for \"%s\"",
579 "the underlying directory stream returned an error on close — "
580 "usually an I/O error on the backing storage",
581 "check dmesg and the filesystem health for that path; the OS "
582 "reason is appended below",
583 dh->path != NULL ? dh->path : "-");
584 dh->dir = NULL;
585 return NGX_ERROR;
586 }
587
588 dh->dir = NULL;
589 return NGX_OK;
590}
