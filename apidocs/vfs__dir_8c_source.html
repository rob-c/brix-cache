1/*
2 * vfs_dir.c — VFS directory handle lifecycle (opendir/closedir).
3 *
4 * WHAT: Implements brix_vfs_opendir(), brix_vfs_opendir_quiet(),
5 * brix_vfs_dir_fd(), and brix_vfs_closedir() over the opaque
6 * brix_vfs_dir_t handle. The readdir family lives in vfs_dir_iter.c.
7 *
8 * WHY: Directory listing (XRootD kXR_dirlist, WebDAV PROPFIND, S3 LIST) needs
9 * confinement and driver-plane dispatch handled once, the same way, for
10 * every protocol — rather than each front end driving opendir itself.
11 *
12 * HOW: opendir re-verifies confinement, pcalloc's the handle on ctx->pool,
13 * dups the resolved path, and opens the C-library DIR* (or the driver's
14 * opendir slot); the open itself is observed as BRIX_METRIC_OP_DIRLIST.
15 * closedir calls closedir(3) (or the driver slot) and nulls the handle
16 * so it is idempotent.
17 */

19#include "core/compat/log_diag.h"

22/* vfs_opendir_fail — shared error tail for the opendir body.
23 *
24 * WHAT: Reports a failed opendir: copies `err` into *err_out (when the caller
25 * asked for it), meters the failure as OP_DIRLIST when `observe` is set,
26 * and returns NULL for the caller to relay.
27 * WHY: Every early-out of the opendir body ends the same way; folding the
28 * report into one helper keeps the body's control flow flat.
29 * HOW: Pure reporting — errno is deliberately NOT touched here so each call
30 * site preserves exactly the errno state the pre-refactor code left; the
31 * observed path is the ctx's resolved path (what every caller reports). */
32static brix_vfs_dir_t *
33vfs_opendir_fail(brix_vfs_ctx_t *ctx, int err, int observe, uint64_t start,
34 int *err_out)
35{
36 if (err_out != NULL) {
37 *err_out = err;
38 }
39 if (observe) {
40 brix_vfs_observe_ctx_op(ctx, brix_vfs_ctx_path(ctx),
41 BRIX_METRIC_OP_DIRLIST, NULL, 0,
42 NGX_ERROR, err, start);
43 }
44 return NULL;
45}

47/* vfs_dir_route — the SECURITY SEAM: export confinement + plane selection.
48 *
49 * WHAT: Verifies the resolved ctx path is confined to the export root, then
50 * selects the enumeration plane: a non-POSIX driver iterator (*drv_out
51 * set) or the confined POSIX/broker opendir (*drv_out NULL).
52 * WHY: Confinement MUST be re-verified before any directory is opened, and
53 * the driver-vs-broker decision must live in exactly one place so no
54 * future path can enumerate an unconfined directory.
55 * HOW: brix_vfs_require_confined() gates everything; on failure *err carries
56 * its errno for the caller's report (errno itself is left untouched). */
57static ngx_int_t
58vfs_dir_route(brix_vfs_ctx_t *ctx, const brix_sd_driver_t **drv_out, int *err)
59{
60 if (brix_vfs_require_confined(ctx) != NGX_OK) {
61 *err = errno;
62 return NGX_ERROR;
63 }
64 *drv_out = brix_vfs_ctx_driver(ctx);
65 return NGX_OK;
66}

68/* vfs_dir_open_driver — open a non-POSIX backend's directory iterator.
69 *
70 * WHAT: Resolves the export-relative logical path, loads the per-user backend
71 * credential when the export is credential-scoped, and opens the leaf
72 * driver's directory iterator into dh->sd_dir, filling the handle's
73 * driver-plane fields.
74 * WHY: Object/remote backends have no DIR*; enumeration goes through the
75 * driver's opendir/readdir verbs instead of the confined POSIX path.
76 * HOW: Returns NGX_OK, or NGX_ERROR with *err set to the errno to report
77 * (errno itself only set where the pre-refactor code set it: the
78 * credential-load failure). Dispatches on the leaf instance so
79 * brix_sd_opendir_maybe_cred finds the leaf driver's opendir_cred slot
80 * (decorators have only plain relays). */
81static ngx_int_t
82vfs_dir_open_driver(brix_vfs_ctx_t *ctx, brix_vfs_dir_t *dh,
83 const brix_sd_driver_t *drv, const char *path, int *err_out)
84{
85 char physical[PATH_MAX];
86 char canonical[PATH_MAX];
87 brix_sd_ucred_t store;
88 brix_sd_cred_t cred;
89 int use_cred = 0, cred_err = 0;
90 int err = 0;
91
92 /* Zero before the gate: it fills only the active credential kind; an
93 * unzeroed cred hands a garbage inactive pointer to the driver's cred
94 * slot (bearer PASSTHROUGH would leave x509_proxy dangling). */
95 ngx_memzero(&cred, sizeof(cred));
96
97 if (brix_path_resolved_to_pfn(ctx, path, physical, sizeof(physical))
98 != NGX_OK)
99 {
100 *err_out = errno;
101 return NGX_ERROR;
102 }
103 if (brix_path_pfn_to_lfn(ctx, physical, canonical, sizeof(canonical))
104 != NGX_OK)
105 {
106 *err_out = errno;
107 return NGX_ERROR;
108 }
109
110 if (brix_vfs_cred_gate_active(ctx)) {
111 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
112 != NGX_OK)
113 {
114 *err_out = cred_err ? cred_err : EACCES;
115 errno = *err_out;
116 return NGX_ERROR;
117 }
118 }
119
120 if (drv->opendir == NULL) {
121 errno = ENOTSUP;
122 }
123 dh->sd_dir = (drv->opendir != NULL)
124 ? brix_sd_opendir_maybe_cred(brix_vfs_ns_leaf(ctx->sd),
125 physical, &err, use_cred ? &cred : NULL)
126 : NULL;
127 brix_sd_ucred_wipe(&store); /* secret consumed by opendir; erase (A-4/T4) */
128 if (dh->sd_dir == NULL) {
129 *err_out = (err != 0) ? err : errno;
130 return NGX_ERROR;
131 }
132
133 dh->sd = ctx->sd;
134 dh->drv = drv;
135 dh->ctx = ctx;
136 dh->sd_logical = brix_vfs_copy_path(ctx->pool, canonical);
137 dh->sd_physical = brix_vfs_copy_path(ctx->pool, physical);
138 if (dh->sd_logical == NULL || dh->sd_physical == NULL) {
139 if (drv->closedir != NULL) {
140 (void) drv->closedir(dh->sd_dir);
141 }
142 dh->sd_dir = NULL;
143 *err_out = errno;
144 return NGX_ERROR;
145 }
146 dh->pool = ctx->pool;
147 dh->log = ctx->log;
148 return NGX_OK;
149}

151/* vfs_dir_open_confined — the confined POSIX/broker opendir (do NOT weaken).
152 *
153 * WHAT: Opens dh->dir under export-root confinement and fills the handle's
154 * POSIX-plane fields.
155 * WHY: brix_opendir_confined_canon is the openat2 RESOLVE_IN_ROOT path from
156 * the symlink-escape fix — an outward symlink inside an export must stay
157 * invisible — and under impersonation it opens AS THE MAPPED USER
158 * (broker fdopendir) so a 0700 user-owned / 0770 group-restricted dir
159 * the unprivileged worker cannot itself open is enumerable by its
160 * legitimate owner/group-member; off impersonation it is a bare
161 * opendir().
162 * HOW: Returns NGX_OK, or NGX_ERROR with errno left exactly as the confined
163 * open set it. */
164static ngx_int_t
165vfs_dir_open_confined(brix_vfs_ctx_t *ctx, brix_vfs_dir_t *dh,
166 const char *path)
167{
168 dh->dir = (ctx->rootfd >= 0)
169 ? brix_opendir_confined_canon_at(ctx->log, ctx->rootfd,
170 ctx->root_canon, path)
171 : brix_opendir_confined_canon(ctx->log, ctx->root_canon, path);
172 if (dh->dir == NULL) {
173 return NGX_ERROR;
174 }
175
176 dh->pool = ctx->pool;
177 dh->log = ctx->log;
178 dh->root_canon = ctx->root_canon;
179 return NGX_OK;
180}

182/* Shared opendir body. When `observe` is set the open is metered as OP_DIRLIST;
183 * the quiet variant (observe=0) skips the metric/access-log entirely — for bulk
184 * recursive walks (S3 ListObjects, WebDAV SEARCH) whose enclosing protocol op
185 * already accounts for the traversal and would otherwise emit one phantom
186 * OP_DIRLIST per visited subdirectory. */
187static brix_vfs_dir_t *
188brix_vfs_opendir_impl(brix_vfs_ctx_t *ctx, int *err_out, int observe)
189{
190 brix_vfs_dir_t *dh;
191 const char *path;
192 const brix_sd_driver_t *drv = NULL;
193 uint64_t start;
194 int err = 0;
195
196 start = brix_vfs_now_ns();
197
198 if (err_out != NULL) {
199 *err_out = 0;
200 }
201
202 if (brix_vfs_require_authorized_lookup(ctx) != NGX_OK) {
203 return vfs_opendir_fail(ctx, errno, observe, start, err_out);
204 }
205
206 if (vfs_dir_route(ctx, &drv, &err) != NGX_OK) {
207 return vfs_opendir_fail(ctx, err, observe, start, err_out);
208 }
209
210 path = brix_vfs_ctx_path(ctx);
211 dh = ngx_pcalloc(ctx->pool, sizeof(*dh));
212 if (dh == NULL) {
213 errno = ENOMEM;
214 return vfs_opendir_fail(ctx, ENOMEM, observe, start, err_out);
215 }
216
217 dh->path = brix_vfs_copy_path(ctx->pool, path);
218 if (dh->path == NULL) {
219 return vfs_opendir_fail(ctx, errno, observe, start, err_out);
220 }
221
222 if (drv != NULL) {
223 /* Non-POSIX backend: enumerate through the driver's iterator. */
224 if (vfs_dir_open_driver(ctx, dh, drv, path, &err) != NGX_OK) {
225 return vfs_opendir_fail(ctx, err, observe, start, err_out);
226 }
227 } else if (vfs_dir_open_confined(ctx, dh, path) != NGX_OK) {
228 return vfs_opendir_fail(ctx, errno, observe, start, err_out);
229 }
230
231 if (observe) {
232 brix_vfs_observe_ctx_op(ctx, path, BRIX_METRIC_OP_DIRLIST, NULL, 0,
233 NGX_OK, 0, start);
234 }
235 return dh;
236}

263/* Close the directory stream and null the handle (idempotent). Logs and returns
264 * NGX_ERROR if closedir(3) fails. */
265ngx_int_t
266brix_vfs_closedir(brix_vfs_dir_t *dh, ngx_log_t *log)
267{
268 if (dh == NULL || (dh->dir == NULL && dh->sd_dir == NULL)) {
269 return NGX_OK;
270 }
271
272 if (dh->sd_dir != NULL) {
273 ngx_int_t rc = (dh->drv->closedir != NULL)
274 ? dh->drv->closedir(dh->sd_dir) : NGX_OK;
275
276 dh->sd_dir = NULL;
277 return rc;
278 }
279
280 if (closedir(dh->dir) != 0) {
281 BRIX_DIAG_ERR(log != NULL ? log : dh->log, errno,
282 "xrootd[disk]: closedir failed for \"%s\"",
283 "the underlying directory stream returned an error on close — "
284 "usually an I/O error on the backing storage",
285 "check dmesg and the filesystem health for that path; the OS "
286 "reason is appended below",
287 dh->path != NULL ? dh->path : "-");
288 dh->dir = NULL;
289 return NGX_ERROR;
290 }
291
292 dh->dir = NULL;
293 return NGX_OK;
294}
