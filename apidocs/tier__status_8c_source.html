1/*
2 * tier_status.c - report a tier's readiness against its role contract
3 * (phase-64 section 2.2). Split out of tier_config.c, which owns the other
4 * half of tier.h's public surface (store-URL parsing): the two shared nothing
5 * but the header, and together they crossed the 600-line file cap.
6 *
7 * brix_tier_status checks the built driver against the per-role slot+cap
8 * contract: a missing slot/cap is a tracked "needs development" (NEEDS_DEV +
9 * the closing sub-project), never a hard failure (P1).
10 */

13/* The sub-project that closes a gap for (driver, role) - the §3 status matrix. */
14static const char *
15tier_sp_for(const char *driver, brix_tier_role_t role)
16{
17 (void) role;
18 if (ngx_strcmp(driver, "pblock") == 0 || ngx_strcmp(driver, "xroot") == 0) {
19 return "SP2";
20 }
21 if (ngx_strcmp(driver, "tape") == 0) {
22 return "SP5";
23 }
24 return "SP3"; /* writable http/s3 + rados stores */
25}
