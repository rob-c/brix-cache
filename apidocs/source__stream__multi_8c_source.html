2/* File: source_stream_multi.c — native TPC multi-stream read loop
3 * WHAT: tpc_stream_to_dst_multi() streams the source into the destination with
4 * one kXR_read window in flight per stream (the primary plus every bound
5 * sub-stream), demultiplexing replies by streamid across the sockets, and
6 * finishes through the same completion gate / fsync as the single-stream
7 * loop in source_stream.c.
8 *
9 * WHY (release-2.0 F7): parallel windows let a WAN pull use more than one
10 * socket's worth of bandwidth-delay product (stock xrootd `ofs.tpc streams`).
11 * The loop shares the frame classifier, the positional writer, the deadline
12 * and the finish gate with the single-stream loop so the two cannot drift.
13 *
14 * HOW: a round issues N reads at consecutive TPC_CHUNK_SIZE offsets, ALL on the
15 * primary socket, each tagged streamid {slot, 3} and carrying the 8-byte
16 * read_args whose pathid steers slot i's reply out of sub-stream i-1 (0 =
17 * the primary). Replies are drained with poll() over every socket (plus a
18 * TLS-buffered-bytes check, since poll cannot see data OpenSSL already read),
19 * each frame written at round_base + slot*CHUNK + bytes-so-far; a source that
20 * answers on the primary instead (cross-worker bind, oversize read) is
21 * handled identically because demux keys on the streamid, not the socket.
22 * tpc_stream_plan_round() judges the round: contiguous advance, EOF, or a
23 * hole (which fails the pull rather than commit a sparse file). */

56/* WHAT: issue slot's kXR_read for this round on the primary: header + the
57 * read_args payload naming the sub-stream that should carry the reply. */
58static int
59tpc_multi_send_slot(brix_tpc_pull_t *t, tpc_multi_round_t *r, int slot)
60{
61 u_char frame[sizeof(ClientReadRequest) + TPC_READ_ARGS_LEN];
62 ClientReadRequest *rdreq = (ClientReadRequest *) frame;
63 uint64_t offset = r->base + (uint64_t) slot * TPC_CHUNK_SIZE;
64 unsigned char pathid = (slot == 0) ? 0 : t->sub[slot - 1].pathid;
65
66 ngx_memzero(frame, sizeof(frame));
67 rdreq->streamid[0] = (u_char) slot;
68 rdreq->streamid[1] = TPC_STREAM_READ_TAG;
69 rdreq->requestid = htons(kXR_read);
70 ngx_memcpy(rdreq->fhandle, r->fhandle, XRD_FHANDLE_LEN);
71 rdreq->offset = (kXR_int64) brix_plat_htobe64(offset);
72 rdreq->rlen = htonl((kXR_int32) TPC_CHUNK_SIZE);
73 rdreq->dlen = htonl((kXR_int32) TPC_READ_ARGS_LEN);
74 tpc_stream_plan_read_args(frame + sizeof(ClientReadRequest), pathid);
75
76 if (tpc_send_all(t, r->fd, frame, sizeof(frame)) != 0) {
77 snprintf(t->err_msg, sizeof(t->err_msg),
78 "TPC kXR_read send failed at offset %llu (stream %d)",
79 (unsigned long long) offset, slot);
80 t->xrd_error = kXR_IOError;
81 return -1;
82 }
83 return 0;
84}

86/* WHAT: receive one frame from `fd` and account it to the slot its streamid
87 * names. Returns 0, or -1 with t->err_msg / t->xrd_error set. */
88static int
89tpc_multi_recv_frame(brix_tpc_pull_t *t, tpc_multi_round_t *r, int fd)
90{
91 u_char sid[2];
92 uint16_t status;
93 uint32_t dlen = 0;
94 u_char *body = NULL;
95 int slot;
96 uint64_t offset;
97
98 if (tpc_recv_response_sid(t, fd, sid, &status, &body, &dlen) != 0) {
99 snprintf(t->err_msg, sizeof(t->err_msg),
100 "TPC kXR_read recv failed at offset %llu",
101 (unsigned long long) r->base);
102 t->xrd_error = kXR_IOError;
103 return -1;
104 }
105
106 slot = (int) sid[0];
107 if (sid[1] != TPC_STREAM_READ_TAG || slot >= r->nslots || r->done[slot]) {
108 free(body);
109 snprintf(t->err_msg, sizeof(t->err_msg),
110 "TPC multi-stream reply for unexpected stream %u/%u",
111 (unsigned) sid[0], (unsigned) sid[1]);
112 t->xrd_error = kXR_ServerError;
113 return -1;
114 }
115
116 offset = r->base + (uint64_t) slot * TPC_CHUNK_SIZE;
117 if (tpc_stream_classify_frame(t, status, body, dlen, offset) != 0) {
118 free(body);
119 return -1;
120 }
121 if (dlen > 0 && body != NULL
122 && tpc_stream_write_frame(t, offset, &r->got[slot], body, dlen) != 0)
123 {
124 free(body);
125 return -1;
126 }
127 free(body);
128
129 if (status == kXR_ok) {
130 r->done[slot] = 1;
131 r->pending--;
132 }
133 return 0;
134}

136/* WHAT: pick the sockets that can be read without blocking: any with bytes
137 * already buffered inside its TLS session wins outright (poll cannot see
138 * those); otherwise poll the whole set. Returns the number ready, 0 on the
139 * idle timeout, -1 on a poll error. */
140static int
141tpc_multi_wait_ready(brix_tpc_pull_t *t, tpc_multi_round_t *r)
142{
143 int i, ready = 0;
144
145 for (i = 0; i < r->nslots; i++) {
146 r->pfd[i].revents = 0;
147 if (tpc_io_pending(t, r->pfd[i].fd) > 0) {
148 r->pfd[i].revents = POLLIN;
149 ready++;
150 }
151 }
152 if (ready > 0) {
153 return ready;
154 }
155 return poll(r->pfd, (nfds_t) r->nslots, BRIX_TPC_IO_TIMEOUT_SEC * 1000);
156}

158/* WHAT: drain every reply of the round, whichever socket it arrives on. */
159static int
160tpc_multi_drain_round(brix_tpc_pull_t *t, tpc_multi_round_t *r)
161{
162 while (r->pending > 0) {
163 int i, ready = tpc_multi_wait_ready(t, r);
164
165 if (ready <= 0) {
166 snprintf(t->err_msg, sizeof(t->err_msg),
167 "TPC multi-stream read %s at offset %llu",
168 ready == 0 ? "timed out" : "poll failed",
169 (unsigned long long) r->base);
170 t->xrd_error = kXR_IOError;
171 return -1;
172 }
173 for (i = 0; i < r->nslots; i++) {
174 if ((r->pfd[i].revents & (POLLIN | POLLERR | POLLHUP)) == 0) {
175 continue;
176 }
177 if (tpc_multi_recv_frame(t, r, r->pfd[i].fd) != 0) {
178 return -1;
179 }
180 }
181 }
182 return 0;
183}

185/* WHAT: send + drain + judge one round. Returns 1 to continue at *base
186 * advanced, 0 at EOF, -1 on failure (t->err_msg / t->xrd_error set). */
187static int
188tpc_multi_round(brix_tpc_pull_t *t, tpc_multi_round_t *r)
189{
190 size_t advance = 0;
191 int slot, eof = 0;
192
193 ngx_memzero(r->got, sizeof(r->got));
194 ngx_memzero(r->done, sizeof(r->done));
195 r->pending = r->nslots;
196
197 for (slot = 0; slot < r->nslots; slot++) {
198 if (tpc_multi_send_slot(t, r, slot) != 0) {
199 return -1;
200 }
201 }
202 if (tpc_multi_drain_round(t, r) != 0) {
203 return -1;
204 }
205 if (tpc_stream_plan_round(r->got, r->nslots, TPC_CHUNK_SIZE,
206 &advance, &eof) != 0)
207 {
208 snprintf(t->err_msg, sizeof(t->err_msg),
209 "TPC multi-stream round left a hole after offset %llu",
210 (unsigned long long) r->base);
211 t->xrd_error = kXR_IOError;
212 return -1;
213 }
214 r->base += advance;
215 return eof ? 0 : 1;
216}

218int
219tpc_stream_to_dst_multi(brix_tpc_pull_t *t, int fd, const u_char *fhandle)
220{
221 tpc_multi_round_t r;
222 time_t pull_start = time(NULL);
223 int slot, rc;
224
225 ngx_memzero(&r, sizeof(r));
226 r.fd = fd;
227 r.fhandle = fhandle;
228 r.nslots = 1 + t->nsub;
229 for (slot = 0; slot < r.nslots; slot++) {
230 r.pfd[slot].fd = tpc_multi_slot_fd(t, fd, slot);
231 r.pfd[slot].events = POLLIN;
232 }
233
234 do {
235 if (tpc_stream_check_deadline(t, pull_start, r.base) != 0) {
236 return -1;
237 }
238 /* Every socket is quiet between rounds — the only point a renewal
239 * kXR_auth can be carried on the primary (W8.2). */
240 if (tpc_cred_renew_if_due(t, fd) != 0) {
241 return -1;
242 }
243 rc = tpc_multi_round(t, &r);
244 } while (rc > 0);
245
246 if (rc < 0) {
247 return -1;
248 }
249 return tpc_stream_finish(t);
250}
