38int
39brix_platform_event_watch(int event_fd, int fd, uint32_t events)
40{
41 struct epoll_event ev;
42 int op = EPOLL_CTL_ADD;
43
44 /* Translate event mask to epoll events */
45 ev.events = 0;
46 if (events & BRIX_EVENT_READ) {
47 ev.events |= EPOLLIN;
48 }
49 if (events & BRIX_EVENT_WRITE) {
50 ev.events |= EPOLLOUT;
51 }
52 if (events & BRIX_EVENT_ERROR) {
53 ev.events |= EPOLLERR;
54 }
55
56 /* Filesystem events not supported by epoll - use inotify instead */
57 if (events & (BRIX_EVENT_DELETE | BRIX_EVENT_MODIFY | BRIX_EVENT_CREATE)) {
58 errno = EINVAL;
59 return -1;
60 }
61
62 ev.data.fd = fd;
63
64 if (epoll_ctl(event_fd, op, fd, &ev) < 0) {
65 return -1; /* errno set by epoll_ctl() */
66 }
67
68 return 0;
69}

73{

89 return -1; /* errno set by epoll_wait() */
