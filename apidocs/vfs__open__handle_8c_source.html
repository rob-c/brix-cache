1/*
2 * vfs_open_handle.c — VFS handle lifecycle, close, and the brix_vfs_file_*
3 * accessors.
4 *
5 * WHAT: Implements brix_vfs_close() and every read-only accessor over an open
6 * brix_vfs_file_t: fd / sd_obj / pread / sendfile-fd / can-sendfile /
7 * backend-name / path / size / mtime / from_cache / file_stat, plus the
8 * phase-71 memfd sendfile-proxy materialiser.
9 *
10 * WHY: Once brix_vfs_open() (vfs_open.c) hands back a confined, already-fstat'd
11 * handle, everything else is pure handle-state access and teardown. Keeping
12 * it separate from the open cascade keeps both files under the size cap and
13 * isolates the "given a handle" surface from the "build a handle" surface.
14 *
15 * HOW: Accessors read cached metadata captured at adopt time; brix_vfs_file_stat()
16 * answers from that cache on a read-only handle (stat_current) and otherwise
17 * re-stats through the backend's fstat slot. brix_vfs_file_sendfile_fd()
18 * delegates the zero-copy decision to the backend and, for a fd-less
19 * CAP_MEMFILE backend, materialises a handle-owned memfd once.
20 * brix_vfs_close() releases any memfd then the descriptor through the
21 * backend's close slot.
22 */

29ngx_int_t
30brix_vfs_close(brix_vfs_file_t *fh, ngx_log_t *log)
31{
32 if (fh == NULL) {
33 return NGX_OK;
34 }
35
36 /* phase-71 step 2: release a materialised memfd sendfile proxy, if any. This
37 * runs before the fd-based early-out below because a CAP_MEMFILE backend has
38 * obj.fd == NGX_INVALID_FILE yet may still own a memfd. */
39 if (fh->memfd != NGX_INVALID_FILE) {
40 (void) ngx_close_file(fh->memfd);
41 fh->memfd = NGX_INVALID_FILE;
42 }
43
44 /* A memory-served backend (xroot/pblock/ceph — CAP_MEMFILE, no kernel fd)
45 * still owns per-open driver state: an origin connection holding a live
46 * remote write handle, catalog state, a RADOS ioctx. Skipping the close
47 * slot for fd == NGX_INVALID_FILE leaked all of it — on a root:// backend
48 * the origin kept counting this session as an open writer, so the very
49 * next read-open of the object was denied by single-writer semantics. */
50 if (fh->obj.driver == NULL
51 || (fh->obj.fd == NGX_INVALID_FILE && fh->obj.state == NULL))
52 {
53 return NGX_OK;
54 }
55
56 /* Release the descriptor through the backend's close slot; the driver
57 * marks obj.fd invalid. The VFS keeps the error-log wrapper. */
58 if (fh->obj.driver->close(&fh->obj) != NGX_OK) {
59 BRIX_DIAG_ERR(log != NULL ? log : fh->log, ngx_errno,
60 "xrootd[disk]: close failed for \"%s\"",
61 "a deferred write error surfaced at close — typically the "
62 "filesystem filled up (ENOSPC) or the device returned an I/O error",
63 "check free space and dmesg for disk errors; the file may be "
64 "incomplete, so the client's write should be treated as failed",
65 fh->path != NULL ? fh->path : "-");
66 return NGX_ERROR;
67 }
68
69 return NGX_OK;
70}

118/* brix_vfs_memfile_materialize — phase-71 step 2 memfd sendfile proxy.
119 *
120 * WHAT: For a CAP_MEMFILE backend that exposes no kernel fd (obj.fd invalid,
121 * read_sendfile_fd declines), pread the whole object into an anonymous
122 * memfd ONCE and cache it on fh->memfd, so the VFS can hand every backend
123 * a uniform seekable fd for the sendfile / file-backed serve path.
124 * WHY: Removes the last backend-identity branch in the serve path: callers stop
125 * special-casing "no fd → build a memory buffer myself" and use one fd path.
126 * HOW: memfd_create, sized with ftruncate and filled through one mmap of
127 * itself, so the driver's worker-safe pread slot is asked for the whole
128 * remaining span at once rather than in 64 KiB steps (see
129 * brix_vfs_memfile_fill). The fd is owned by the handle and closed in
130 * brix_vfs_close. Returns the cached fd on repeat calls.
131 * NGX_INVALID_FILE on any failure (the caller then falls back to its
132 * legacy memory-backed path — no behaviour change). */
133
134/* Fill `map` (the whole object, already sized) from the driver.
135 *
136 * The span asked for is always ALL THAT IS LEFT, never a fixed chunk, and that
137 * is the point rather than a detail. The previous loop read 64 KiB at a time
138 * through a stack buffer, and on a driver whose pread opens a session per call
139 * — gsiftp does, one USER/PASS/TYPE/REST/RETR per read — materialising a
140 * 300 kB object cost FIVE logins and five transfers where one would do.
141 * Asking for the remainder lets a conforming driver answer in a single call
142 * and still lets a short-reading one finish; the memfd is mapped rather than
143 * written through a buffer because it already holds the whole object, so the
144 * mapping costs no memory the materialisation was not paying anyway. */
145static int
146brix_vfs_memfile_fill(brix_sd_obj_t *obj, u_char *map, off_t size)
147{
148 off_t off;
149
150 for (off = 0; off < size; /* advanced below */) {
151 ssize_t n = obj->driver->pread(obj, map + off,
152 (size_t) (size - off), off);
153
154 if (n <= 0) {
155 return NGX_ERROR;
156 }
157 off += n;
158 }
159 return NGX_OK;
160}

163/* The memfd sized and filled, or NGX_INVALID_FILE with the fd already closed. */
164static ngx_fd_t
165brix_vfs_memfile_build(brix_sd_obj_t *obj, off_t size)
166{
167 ngx_fd_t fd;
168 void *map;
169 int filled;
170
171 /* Use platform abstraction for anonymous fd (memfd on Linux, O_TMPFILE/mkstemp on macOS) */
172 fd = (ngx_fd_t) brix_plat_anon_fd("brix-vfs-memfile", NULL);
173 if (fd == NGX_INVALID_FILE) {
174 return NGX_INVALID_FILE;
175 }
176 if (size == 0) {
177 return fd; /* an empty object needs no transfer */
178 }
179 if (ftruncate(fd, size) != 0) {
180 (void) ngx_close_file(fd);
181 return NGX_INVALID_FILE;
182 }
183
184 map = mmap(NULL, (size_t) size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
185 if (map == MAP_FAILED) {
186 (void) ngx_close_file(fd);
187 return NGX_INVALID_FILE;
188 }
189 filled = brix_vfs_memfile_fill(obj, map, size);
190 (void) munmap(map, (size_t) size);
191
192 if (filled != NGX_OK || lseek(fd, 0, SEEK_SET) == (off_t) -1) {
193 (void) ngx_close_file(fd);
194 return NGX_INVALID_FILE;
195 }
196 return fd;
197}

200static ngx_fd_t
201brix_vfs_memfile_materialize(brix_vfs_file_t *fh)
202{
203 brix_sd_obj_t obj;
204 ngx_fd_t fd;
205
206 if (fh->memfd != NGX_INVALID_FILE) {
207 return fh->memfd; /* already materialised */
208 }
209
210 brix_vfs_handle_sd_obj(fh, &obj);
211 if (obj.driver == NULL || obj.driver->pread == NULL
212 || !(brix_sd_caps(obj.inst) & BRIX_SD_CAP_MEMFILE))
213 {
214 return NGX_INVALID_FILE;
215 }
216
217 fd = brix_vfs_memfile_build(&obj, fh->size);
218 if (fd == NGX_INVALID_FILE) {
219 return NGX_INVALID_FILE;
220 }
221
222 fh->memfd = fd;
223 return fd;
224}

226/* The fd only when the backend elects to back a zero-copy transfer of the whole
227 * object, else NGX_INVALID_FILE. The decision is delegated to the backend's
228 * read_sendfile_fd slot (want_zerocopy=1: the HTTP serve helper applies the
229 * TLS/cleartext choice itself). For a fd-less CAP_MEMFILE backend the VFS
230 * materialises a handle-owned memfd (phase-71 step 2) so the serve path is a
231 * uniform seekable fd for every backend. The contract gate for callers that
232 * build a sendfile / file-backed response. */
233ngx_fd_t
234brix_vfs_file_sendfile_fd(const brix_vfs_file_t *fh)
235{
236 return brix_vfs_file_sendfile_fd_window(fh, 0,
237 fh != NULL ? fh->size : 0);
238}

240/* brix_vfs_file_sendfile_fd_window — the sendfile fd for the window a caller is
241 * ABOUT TO SEND, rather than for the whole object.
242 *
243 * WHAT: identical to brix_vfs_file_sendfile_fd() for any backend that owns a
244 * real kernel fd; for a fd-less CAP_MEMFILE backend it materialises the
245 * memfd ONLY when [off, off+len) is the whole object.
246 * WHY: materialisation preads the ENTIRE object through the driver. On a
247 * remote backend (gsiftp, http, xroot, s3, ceph) that is a network
248 * transfer of the whole file, and it was being paid for every ranged
249 * read: a 256-byte Range on a 384 kB file fetched all 384 kB in six
250 * round trips, and the same request against a 10 GB object would have
251 * fetched 10 GB. Nothing was wrong with the response — the right bytes
252 * came back with the right Content-Range — so the cost was invisible from
253 * outside, and it silently cancelled every bounded-retrieve extension the
254 * drivers negotiate (GridFTP ERET P, HTTP Range, xroot kXR_read), which
255 * were all correctly asking the origin for a window and then being handed
256 * a whole-object request by the layer above.
257 * HOW: probe the backend first and unchanged — a driver that owns an fd is
258 * indifferent to the window, and narrowing what it is asked would change
259 * which offsets sd_block / sd_pblock accept. Only the materialisation
260 * fallback is gated: a strict sub-window declines and the caller falls
261 * back to its memory-backed path, which reads exactly the window it
262 * needs. A caller that will read NOTHING (a HEAD) passes len < 0, which
263 * can never equal a size and so never materialises.
264 *
265 * Declining costs nothing here: the memfd is a full COPY of an object the
266 * backend cannot sendfile, so the "zero-copy" path had already paid for
267 * the whole object once before sending a byte. */
268ngx_fd_t
269brix_vfs_file_sendfile_fd_window(const brix_vfs_file_t *fh, off_t off,
270 off_t len)
271{
272 ngx_fd_t fd;
273
274 if (fh == NULL) {
275 return NGX_INVALID_FILE;
276 }
277 fd = brix_vfs_handle_sendfile_fd(fh, 0, (size_t) fh->size, 1);
278 if (fd != NGX_INVALID_FILE) {
279 return fd;
280 }
281 if (off != 0 || len != fh->size) {
282 return NGX_INVALID_FILE;
283 }
284 /* fh is const by contract, but the memfd cache is an internal materialisation
285 * that does not change the observable file — safe to fill lazily here. */
286 return brix_vfs_memfile_materialize((brix_vfs_file_t *) fh);
287}

333ngx_int_t
334brix_vfs_file_stat(const brix_vfs_file_t *fh, brix_vfs_stat_t *stat_out)
335{
336 brix_sd_stat_t st;
337 brix_sd_obj_t obj;
338
339 if (fh == NULL || stat_out == NULL
340 || (fh->obj.fd == NGX_INVALID_FILE && fh->obj.driver == NULL))
341 {
342 /* A driver-backed handle (object/remote backend) has no kernel fd; it
343 * answers from cached metadata or the driver's fstat slot below. */
344 errno = EINVAL;
345 return NGX_ERROR;
346 }
347
348 /*
349 * phase-45 W2/R1: when the metadata cached at adopt time is authoritative,
350 * answer from it and skip a redundant fstat(2). This is the common read
351 * path (S3/WebDAV GET open the fd then immediately stat it). stat_current is
352 * set by adopt_fd only for read-only handles, whose file cannot change
353 * through them; a writable handle has it clear and always takes a live fstat.
354 */
355 if (fh->stat_current) {
356 ngx_memzero(stat_out, sizeof(*stat_out));
357 stat_out->size = fh->size;
358 stat_out->mtime = fh->mtime;
359 stat_out->ctime = fh->ctime;
360 stat_out->mode = (ngx_uint_t) fh->mode;
361 stat_out->ino = fh->ino;
362 stat_out->is_directory = S_ISDIR(fh->mode) ? 1 : 0;
363 stat_out->is_regular = S_ISREG(fh->mode) ? 1 : 0;
364 return NGX_OK;
365 }
366
367 /* Live re-stat through the backend's fstat slot (not a direct fstat(2)). */
368 brix_vfs_handle_sd_obj(fh, &obj);
369 if (obj.driver->fstat == NULL || obj.driver->fstat(&obj, &st) != NGX_OK) {
370 return NGX_ERROR;
371 }
372
373 brix_vfs_sd_stat_fill(&st, stat_out);
374 return NGX_OK;
375}
