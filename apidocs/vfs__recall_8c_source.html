1/*
2 * vfs_recall.c — the nearline lifecycle pair (phase-107 C2): prestage a
3 * nearline object into the online buffer, and evict an online copy.
4 *
5 * WHAT: brix_vfs_recall() initiates (or reports) a recall of the resolved ctx
6 * path from a nearline tier; brix_vfs_evict() drops the online copy the
7 * top of the driver chain holds. Both are typed export mutations
8 * (MUTATE_STAGE / MUTATE_EVICT) and pass the policy kernel FIRST, so a
9 * read-only endpoint answers EROFS before any capability, credential or
10 * registry question is asked (INVARIANT #12).
11 *
12 * WHY: Until this file, prestage ran only as the fork/exec prepare_command
13 * while every wire-native recall slot sat implemented and uncalled, and
14 * the kXR_prepare evict arm was a documented no-op. The two verbs also
15 * dispatch in OPPOSITE directions, which is why they share a file but
16 * not a body:
17 *
18 * - recall DESCENDS the decorator chain (brix_vfs_decorator_source, the
19 * residency seam's walk): the nearline authority is the buried leaf —
20 * asking a cache decorator to recall is asking the wrong tier.
21 * - evict dispatches on the TOP: "drop the online copy" means the copy
22 * the outermost tier holds (a cache store entry on a cache-fronted
23 * export); descending would evict the origin's copy instead.
24 *
25 * HOW: Gate, resolve the per-user credential when the cred gate is active
26 * (zeroed first — an unzeroed cred hands a garbage inactive pointer to
27 * the driver's cred slot), dispatch through the *_maybe_cred forwarders
28 * (which refuse in DENY mode rather than falling back to the service
29 * identity), wipe the credential store, book the C2 counters. The stage
30 * REGISTRY lifecycle (record-before-driver-call, delete-on-failure) is
31 * deliberately NOT here: it belongs to the protocol planes that own the
32 * reqid namespace (prepare.c, the Tape REST plane) — the VFS verb stays
33 * registry-agnostic so a caller with no reqid namespace (the cache
34 * tier's recall-at-fill) pays nothing for one.
35 */

38/* brix_vfs_recall — prestage the resolved ctx path from its nearline tier.
39 *
40 * WHAT: NGX_OK (already online — nothing to stage), NGX_AGAIN (recall queued /
41 * in flight; reqid_out (optional, ≤39 chars + NUL) carries the driver's
42 * parking handle, "" when it has none), or NGX_ERROR with errno set:
43 * EROFS (read-only endpoint, before anything else), ENOTSUP (no
44 * nearline tier in the chain — the caller falls back to
45 * prepare_command), EACCES (cred gate / DENY-mode refusal), or the
46 * driver's own errno (ENOENT for an unknown key, EBUSY, EIO).
47 * WHY: This is the VFS face of the recall slot the cache tier already drives
48 * at fill time; routing kXR_prepare(kXR_stage) here makes prestage one
49 * verb with one gate instead of a subprocess beside an uncalled slot.
50 * HOW: Same decorator walk as brix_vfs_residency — the FIRST tier advertising
51 * CAP_NEARLINE with a recall slot answers. Books brix_vfs_recall_total:
52 * online/queued/error by outcome; ENOTSUP books nothing (a capability
53 * probe, not a recall attempt); the registry-owning caller books
54 * `joined` itself when an existing record absorbs the request. */
55ngx_int_t
56brix_vfs_recall(brix_vfs_ctx_t *ctx, char reqid_out[40])
57{
58 brix_sd_instance_t *inst;
59 brix_sd_ucred_t store;
60 brix_sd_cred_t cred;
61 ngx_int_t rc;
62 int saved_errno;
63 int use_cred = 0, cred_err = 0;
64 char physical[PATH_MAX];
65
66 if (reqid_out != NULL) {
67 reqid_out[0] = '\0';
68 }
69 if (brix_vfs_gate_confined(ctx, BRIX_VFS_MUTATE_STAGE)
70 != NGX_OK)
71 {
72 return NGX_ERROR; /* EROFS/EINVAL; gate booked it */
73 }
74
75 for (inst = ctx->sd; inst != NULL;
76 inst = brix_vfs_decorator_source(inst))
77 {
78 if ((brix_sd_caps(inst) & BRIX_SD_CAP_NEARLINE) != 0
79 && (inst->driver->recall != NULL
80 || inst->driver->recall_cred != NULL))
81 {
82 break;
83 }
84 }
85 if (inst == NULL) {
86 errno = ENOTSUP; /* no nearline tier: use the
87 * prepare_command fallback */
88 return NGX_ERROR;
89 }
90
91 ngx_memzero(&cred, sizeof(cred)); /* never hand a garbage inactive
92 * pointer to a cred slot */
93 if (brix_vfs_cred_gate_active(ctx)) {
94 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
95 != NGX_OK)
96 {
97 errno = cred_err ? cred_err : EACCES;
98 brix_metric_vfs_recall(BRIX_VFS_RECALL_ERROR);
99 return NGX_ERROR;
100 }
101 }
102
103 if (brix_path_resolved_to_pfn(ctx, brix_vfs_ctx_path(ctx), physical,
104 sizeof(physical)) != NGX_OK)
105 {
106 brix_sd_ucred_wipe(&store);
107 brix_metric_vfs_recall(BRIX_VFS_RECALL_ERROR);
108 return NGX_ERROR;
109 }
110 rc = brix_sd_recall_maybe_cred(inst, physical,
111 reqid_out, use_cred ? &cred : NULL);
112 saved_errno = errno;
113 brix_sd_ucred_wipe(&store); /* secret consumed; erase (A-4/T4) */
114
115 brix_metric_vfs_recall(rc == NGX_OK ? BRIX_VFS_RECALL_ONLINE
116 : rc == NGX_AGAIN ? BRIX_VFS_RECALL_QUEUED
117 : BRIX_VFS_RECALL_ERROR);
118 errno = saved_errno;
119 return rc;
120}

122/* brix_vfs_evict — drop the online copy of the resolved ctx path at the TOP of
123 * the driver chain.
124 *
125 * WHAT: NGX_OK with *bytes_out (optional) = bytes actually reclaimed (0 when
126 * the copy was already absent — evict is idempotent — or the driver
127 * cannot know), or NGX_ERROR with errno set: EROFS first, ENOTSUP when
128 * the top driver has no evict slot (its online copy is the ONLY copy, or
129 * it is read-only to us), EACCES, or the driver's errno.
130 * WHY: The kXR_prepare evict arm and the Tape REST release verb need one
131 * gated seam; today the former is a logged no-op and the only real
132 * eviction path (the cache store's) is reachable only as a side effect
133 * of DELETE. OWNERSHIP of a reqid-bound evict (FRM-1) is the protocol
134 * caller's check, exactly like the registry lifecycle on recall.
135 * HOW: Top dispatch via brix_sd_evict_maybe_cred (never the descend walk —
136 * see the file banner), then book the reclaimed bytes under the
137 * dispatching driver's label. */
138ngx_int_t
139brix_vfs_evict(brix_vfs_ctx_t *ctx, uint64_t *bytes_out)
140{
141 brix_sd_ucred_t store;
142 brix_sd_cred_t cred;
143 uint64_t bytes = 0;
144 ngx_int_t rc;
145 int saved_errno;
146 int use_cred = 0, cred_err = 0;
147 char physical[PATH_MAX];
148
149 if (bytes_out != NULL) {
150 *bytes_out = 0;
151 }
152 if (brix_vfs_gate_confined(ctx, BRIX_VFS_MUTATE_EVICT)
153 != NGX_OK)
154 {
155 return NGX_ERROR; /* EROFS/EINVAL; gate booked it */
156 }
157 if (ctx->sd == NULL
158 || (ctx->sd->driver->evict == NULL
159 && ctx->sd->driver->evict_cred == NULL))
160 {
161 errno = ENOTSUP;
162 return NGX_ERROR;
163 }
164
165 ngx_memzero(&cred, sizeof(cred));
166 if (brix_vfs_cred_gate_active(ctx)) {
167 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
168 != NGX_OK)
169 {
170 errno = cred_err ? cred_err : EACCES;
171 return NGX_ERROR;
172 }
173 }
174
175 if (brix_path_resolved_to_pfn(ctx, brix_vfs_ctx_path(ctx), physical,
176 sizeof(physical)) != NGX_OK)
177 {
178 brix_sd_ucred_wipe(&store);
179 return NGX_ERROR;
180 }
181 rc = brix_sd_evict_maybe_cred(ctx->sd, physical,
182 &bytes, use_cred ? &cred : NULL);
183 saved_errno = errno;
184 brix_sd_ucred_wipe(&store);
185
186 if (rc == NGX_OK) {
187 if (bytes_out != NULL) {
188 *bytes_out = bytes;
189 }
190 brix_metric_vfs_evict(brix_sd_backend_name(ctx->sd), bytes);
191 }
192 errno = saved_errno;
193 return rc;
194}

196/* brix_vfs_chain_nearline_unstageable — startup advisor probe (phase-107 C2).
197 *
198 * WHAT: 1 iff `chain` (a composed driver chain head; NULL = default POSIX)
199 * declares CAP_NEARLINE on some tier while NO tier pairs that cap with a
200 * recall slot — i.e. brix_vfs_recall on this export can only ever answer
201 * ENOTSUP, and without a prepare_command the export can never stage.
202 * WHY: That configuration deserves one [warn] at worker startup, not a
203 * kXR_Unsupported at the first client prepare. The walk lives HERE so
204 * the advisor can never disagree with brix_vfs_recall's own tier
205 * selection: both iterate the same chain asking the same two questions.
206 * HOW: Same descent as brix_vfs_recall; no ctx (config-time — there is no
207 * request), no policy gate (a pure capability read mutates nothing). */
208int
209brix_vfs_chain_nearline_unstageable(brix_sd_instance_t *chain)
210{
211 brix_sd_instance_t *inst;
212 int nearline = 0;
213
214 for (inst = chain; inst != NULL;
215 inst = brix_vfs_decorator_source(inst))
216 {
217 if ((brix_sd_caps(inst) & BRIX_SD_CAP_NEARLINE) != 0) {
218 nearline = 1;
219 if (inst->driver->recall != NULL
220 || inst->driver->recall_cred != NULL)
221 {
222 return 0; /* recall-capable: stageable */
223 }
224 }
225 }
226 return nearline;
227}
