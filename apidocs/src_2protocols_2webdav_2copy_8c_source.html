26/*
27 * Map a VFS single-file-copy errno to the same HTTP status the prior
28 * brix_ns_local_copy → status mapping produced. brix_vfs_copy() returns
29 * NGX_ERROR with errno set (the namespace sys_errno), so reconstruct the
30 * namespace status the same way namespace_ops' errno_to_ns_status does and feed
31 * it to brix_http_map_ns_status — guaranteeing byte-for-byte parity with the
32 * old code (including DENIED→403, TOO_LONG→414, NO_SPACE→507, CONFLICT→409,
33 * IO_ERROR→500). The two COPY-specific overrides (EXISTS→412, NOT_FOUND→409)
34 * are applied by the caller before this helper is reached. The one departure
35 * from that parity is EBUSY: since phase-107 C7 it is the VFS lock gate's
36 * refusal errno, and the RFC 4918 §11.3 answer is 423 Locked (matching the
37 * shared errno table), never a generic 409.
38 */
39static ngx_int_t
40webdav_copy_errno_to_status(int err)
41{
42 brix_ns_status_t status;
43
44 switch (err) {
45 case 0: status = BRIX_NS_OK; break;
46 case ENOENT: status = BRIX_NS_NOT_FOUND; break;
47 case EACCES:
48 case EPERM:
49 case EXDEV:
50 case ELOOP: status = BRIX_NS_DENIED; break;
51 case EEXIST: status = BRIX_NS_EXISTS; break;
52 case ENOTEMPTY: status = BRIX_NS_NOT_EMPTY; break;
53 case ENAMETOOLONG: status = BRIX_NS_TOO_LONG; break;
54 case ENOSPC: status = BRIX_NS_NO_SPACE; break;
55#ifdef EDQUOT
56 case EDQUOT: status = BRIX_NS_NO_SPACE; break;
57#endif
58 case EBUSY: return NGX_HTTP_LOCKED;
59 case EINVAL: status = BRIX_NS_CONFLICT; break;
60 default: status = BRIX_NS_IO_ERROR; break;
61 }
62
63 return brix_http_map_ns_status(status);
64}

66/*
67 * webdav_copy_probe — confined stat of `path` (follow semantics, matching the
68 * prior brix_lstat_confined_canon nofollow=0) through the VFS probe, projected
69 * into the struct stat fields the COPY handler needs (ino/dev for the self-copy
70 * guard, mode for is-dir, mtime/size for the conditional checks). Non-metered
71 * (the COPY op accounts for itself). Returns NGX_OK / NGX_DECLINED (errno kept).
72 */
73static ngx_int_t
74webdav_copy_probe(ngx_http_request_t *r, const char *path, struct stat *sb)
75{
76 ngx_http_brix_webdav_loc_conf_t *conf =
77 ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
78 ngx_http_brix_webdav_req_ctx_t *rx =
79 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
80 brix_vfs_ctx_t vctx;
81 brix_vfs_stat_t vst;
82 int is_tls = brix_http_request_is_tls(r);
83
84 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log, BRIX_PROTO_WEBDAV,
85 conf->common.root_canon, conf->common.cache_root_canon,
86 brix_vfs_policy_from_write_enable(conf->common.allow_write),
87 is_tls, (rx != NULL) ? rx->identity : NULL, path);
88 /* Bind the export's per-user backend credential policy so this pre-copy
89 * probe (existence/self-copy/overwrite check) enforces deny for the
90 * REQUESTING USER on a remote-backed export, before any copy attempt. */
91 brix_vfs_ctx_bind_backend_cred(&vctx,
92 &conf->common.storage_credential_dir,
93 conf->common.storage_credential_fallback);
94 webdav_vfs_bind_deleg(r, conf, &vctx);
95
96 if (brix_vfs_probe(&vctx, 0 /* follow */, &vst) != NGX_OK) {
97 return NGX_DECLINED;
98 }
99
100 ngx_memzero(sb, sizeof(*sb));
101 sb->st_mode = (mode_t) vst.mode;
102 sb->st_size = vst.size;
103 sb->st_mtime = vst.mtime;
104 sb->st_ino = vst.ino;
105 sb->st_dev = vst.dev;
106 return NGX_OK;
107}

109/*
110 * WHAT: Parse the COPY request headers into req: decode the Destination URI into
111 * req->dest_decoded and set req->overwrite and req->depth_infinity.
112 *
113 * WHY: Header extraction/decoding is a self-contained first phase; isolating it
114 * keeps the handler a flat sequence of resolve → validate → execute.
115 *
116 * HOW: Requires a Destination header (400 if absent); Overwrite:F clears
117 * overwrite; Depth:"0" selects depth:0 (default is infinity). Extracts the path
118 * component of the Destination and URL-decodes it, propagating any helper error
119 * status. Returns NGX_OK on success.
120 */
121static ngx_int_t
122webdav_copy_parse_request(ngx_http_request_t *r, webdav_copy_req_t *req)
123{
124 ngx_table_elt_t *dest_hdr;
125 ngx_table_elt_t *depth_hdr;
126 const u_char *dest_path_start;
127 size_t dest_path_len;
128 ngx_int_t rc;
129
130 dest_hdr = webdav_tpc_find_header(r, "Destination",
131 sizeof("Destination") - 1);
132 if (dest_hdr == NULL) {
133 return NGX_HTTP_BAD_REQUEST;
134 }
135
136 req->overwrite = !brix_http_overwrite_forbidden(r);
137
138 depth_hdr = webdav_tpc_find_header(r, "Depth", sizeof("Depth") - 1);
139 if (depth_hdr != NULL
140 && depth_hdr->value.len == 1 && depth_hdr->value.data[0] == '0')
141 {
142 req->depth_infinity = 0;
143 }
144
145 dest_path_start = dest_hdr->value.data;
146 dest_path_len = dest_hdr->value.len;
147 rc = webdav_destination_extract_path(dest_path_start, dest_path_len,
148 &dest_path_start, &dest_path_len);
149 if (rc != NGX_OK) {
150 return rc;
151 }
152
153 return webdav_urldecode(dest_path_start, dest_path_len,
154 req->dest_decoded, sizeof(req->dest_decoded));
155}

157/*
158 * WHAT: Resolve and validate the source/destination pair: fill req->src_path,
159 * req->dst_path, their stat records, and req->dst_existed, then run the lock,
160 * self-copy, overwrite and conditional checks.
161 *
162 * WHY: This is the security/precondition gate between parsing and executing the
163 * copy — it enforces root confinement, INVARIANT 5's recursive child-lock check
164 * on the destination tree, the RFC 4918 §9.8.5 self-copy rejection, the
165 * Overwrite:F precondition, and If-Match/If-None-Match. Keeping it in one helper
166 * makes the deny paths auditable in isolation.
167 *
168 * HOW: Resolves the source under root_canon and probes it (ENOENT→404, else
169 * 500); resolves the destination and probes it (existence recorded, not
170 * required); runs webdav_check_locks_tree; rejects a same-(dev,ino) self-copy
171 * with 403; a pre-existing destination with Overwrite:F with 412; then the COPY
172 * conditionals. Returns NGX_OK when the copy may proceed, else the HTTP status.
173 */
174static ngx_int_t
175webdav_copy_resolve_pair(ngx_http_request_t *r,
176 ngx_http_brix_webdav_loc_conf_t *conf, webdav_copy_req_t *req)
177{
178 ngx_int_t rc;
179
180 rc = ngx_http_brix_webdav_resolve_path(r, conf->common.root_canon,
181 req->src_path, sizeof(req->src_path));
182 if (rc != NGX_OK) {
183 return rc;
184 }
185
186 if (webdav_copy_probe(r, req->src_path, &req->src_sb) != NGX_OK) {
187 return (errno == ENOENT) ? NGX_HTTP_NOT_FOUND
188 : NGX_HTTP_INTERNAL_SERVER_ERROR;
189 }
190
191 rc = webdav_resolve_destination_path(r->connection->log, "COPY",
192 conf->common.root_canon,
193 req->dest_decoded, req->dst_path,
194 sizeof(req->dst_path),
195 conf->common.cache_store_endpoint);
196 if (rc != NGX_OK) {
197 return rc;
198 }
199
200 req->dst_existed = (webdav_copy_probe(r, req->dst_path, &req->dst_sb)
201 == NGX_OK);
202
203 rc = webdav_check_locks_tree(r, req->dst_path);
204 if (rc != NGX_OK) {
205 return rc;
206 }
207
208 /* Reject copy-onto-self: same (dev, ino) means source and destination are
209 * the same file, which would corrupt/truncate it. RFC 4918 §9.8.5 -> 403. */
210 if (req->dst_existed
211 && req->src_sb.st_ino == req->dst_sb.st_ino
212 && req->src_sb.st_dev == req->dst_sb.st_dev)
213 {
214 return NGX_HTTP_FORBIDDEN;
215 }
216
217 if (req->dst_existed && !req->overwrite) {
218 return NGX_HTTP_PRECONDITION_FAILED;
219 }
220
221 return webdav_check_copy_conditionals(r, req->dst_path, req->dst_existed,
222 &req->dst_sb);
223}

225/*
226 * WHAT: Execute a single-file COPY for a resolved request through the metered
227 * VFS copy surface.
228 *
229 * WHY: Isolates the file branch of the handler, including the pre-delete of a
230 * destination directory (rename(2) cannot atomically replace a directory with a
231 * file) and the COPY-specific errno→HTTP mapping.
232 *
233 * HOW: Pre-deletes a pre-existing destination directory; builds the VFS context;
234 * runs brix_vfs_copy with staged-commit + xattr preservation. On failure it maps
235 * EEXIST→412 and ENOENT→409 (RFC 4918 overrides), everything else through the
236 * canonical mapper. On success it clones dead props and returns NGX_OK.
237 */
238static ngx_int_t
239webdav_copy_do_file(ngx_http_request_t *r,
240 ngx_http_brix_webdav_loc_conf_t *conf, const webdav_copy_req_t *req)
241{
242 brix_vfs_copy_opts_t copy_opts;
243 brix_vfs_ctx_t vctx;
244 brix_sd_precond_t precond;
245
246 if (req->dst_existed && S_ISDIR(req->dst_sb.st_mode)) {
247 (void) webdav_delete_path_recursive(r->connection->log,
248 conf->common.root_canon,
249 req->dst_path);
250 }
251
252 /* Data-plane build so a remote-backed export sees the REQUESTING USER's
253 * credential (cred + opt-in mint + delegation), matching the pre-copy
254 * probe's policy. The copy runs on the VFS surface directly, so no
255 * storage-driver instance is routed here. */
256 webdav_vfs_ctx_build_data(r, conf, req->src_path, &vctx);
257
258 ngx_memzero(&copy_opts, sizeof(copy_opts));
259 copy_opts.overwrite = req->overwrite ? 1 : 0;
260 copy_opts.preserve_xattrs = 1;
261 copy_opts.staged_commit = 1;
262
263 /* Carry If-Match / If-None-Match down to the copy's own commit (phase-107
264 * C6): webdav_check_copy_conditionals stays as the fast path that spares
265 * a doomed request the data copy, but the deciding evaluation now runs at
266 * the publish — a writer landing on dst mid-copy no longer wins silently.
267 * `precond` borrows header bytes from r's pool, alive across this call. */
268 brix_http_write_precond(r, &precond);
269 copy_opts.precond = &precond;
270
271 if (brix_vfs_copy(&vctx, req->dst_path, &copy_opts) != NGX_OK) {
272 /* COPY-specific RFC 4918 semantics that differ from the generic
273 * namespace→HTTP mapping: an existing dst with Overwrite:F is a
274 * precondition failure (412, not 409), and a missing destination
275 * parent is a Conflict (409, not 404). brix_vfs_copy reports the
276 * namespace failure via errno. */
277 int err = errno;
278 if (err == EEXIST || err == ECANCELED) {
279 /* EEXIST: Overwrite:F against an existing dst, or a carried
280 * If-None-Match:* (ABSENT) refused at the commit. ECANCELED: a
281 * carried If-Match (MATCH_*) refused at the commit (C6). Both are
282 * RFC 4918/7232 precondition failures. */
283 return NGX_HTTP_PRECONDITION_FAILED;
284 }
285 if (err == ENOENT) {
286 return NGX_HTTP_CONFLICT;
287 }
288 /* Everything else (DENIED→403, TOO_LONG→414, NO_SPACE→507,
289 * IO_ERROR→500) goes through the canonical mapper so a DAC denial
290 * — e.g. an impersonated cross-tenant copy into a 0700 dir — is a
291 * clean 403, not a blanket 500. */
292 return webdav_copy_errno_to_status(err);
293 }
294
295 webdav_dead_props_copy(r->connection->log, req->src_path, req->dst_path);
296 return NGX_OK;
297}

299/*
300 *
301 * WHAT: Implements RFC 4918 §9.8 WebDAV COPY operation for server-side file/directory duplication within the same export root. Orchestrates the complete copy lifecycle: parses Destination/Overwrite/Depth headers, resolves both source and destination paths under root confinement, validates locks on destination, handles conditional checks (If-Match/If-None-Match), performs atomic copy via an intermediate staged temp path, then renames to final destination. Returns 201 Created when target didn't exist or 204 No Content when replacing existing resource; cleanup on failure ensures no orphaned temp files remain.
302 *
303 * WHY: WebDAV clients need an HTTP endpoint that can duplicate resources within the same filesystem namespace — unlike xrdcp native TPC which transfers between different servers, this COPY is local-only (same root). The tmp_path atomic rename strategy prevents partial copies from being visible to other clients; if copy fails the temp file is immediately deleted. Lock validation ensures no concurrent modifications interfere with the operation. Depth=0 means single-item copy; Depth=infinity recursively copies subdirectories (delegated to webdav_copy_dir_recursive in fs/copy_engine.c).
304 *
305 * HOW: Extracts and decodes Destination URI from request headers (strips scheme prefix to get filesystem path); resolves source via resolve_path() with root_canon confinement; resolves destination via webdav_resolve_destination_path(); checks both paths exist via stat(); validates locks on destination (including recursive child locks if depth=infinity); compares inode/dev numbers to reject self-copy attempt; applies Overwrite header semantics (F=don't overwrite, default=T=allow overwrite); runs copy_conditionals() for If-Match/If-None-Match; creates a staged temp file for file copies or a temp directory for collection copies; copies data via webdav_copy_file() or webdav_copy_dir_recursive(); commits temp→dst via confined rename. */
306ngx_int_t
307webdav_handle_copy(ngx_http_request_t *r)
308{
309 ngx_http_brix_webdav_loc_conf_t *conf;
310 webdav_copy_req_t req;
311 ngx_int_t rc;
312
313 conf = ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
314
315 ngx_memzero(&req, sizeof(req));
316 req.depth_infinity = 1;
317
318 rc = webdav_copy_parse_request(r, &req);
319 if (rc != NGX_OK) {
320 return rc;
321 }
322
323 rc = webdav_copy_resolve_pair(r, conf, &req);
324 if (rc != NGX_OK) {
325 return rc;
326 }
327
328 if (S_ISDIR(req.src_sb.st_mode)) {
329 rc = webdav_copy_do_collection(r, conf, &req);
330 } else {
331 rc = webdav_copy_do_file(r, conf, &req);
332 }
333
334 /* NGX_DONE = collection copy offloaded to a thread (response sent later);
335 * any non-OK status is a failure to surface now. */
336 if (rc != NGX_OK) {
337 return rc;
338 }
339
340 /* A COPY publishes a NEW object at the destination, so the manager needs an
341 * ADD for it. The source is untouched and keeps its existing entry. (The
342 * offloaded collection copy reports from its own completion — it returned
343 * NGX_DONE above and never reaches here.) */
344 webdav_cns_note_written(r, req.dst_path);
345
346 return webdav_send_no_body(r, req.dst_existed ? NGX_HTTP_NO_CONTENT
347 : NGX_HTTP_CREATED);
348}
