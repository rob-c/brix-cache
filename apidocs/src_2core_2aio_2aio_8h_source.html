9/*
10 * AIO — async file I/O via the nginx thread pool, plus response builders.
11 *
12 * WHY AIO?
13 * nginx worker threads are single-threaded and must never block. Calling
14 * pread(2) directly would stall the event loop for the duration of the disk
15 * I/O, preventing any other connection from making progress. Instead we:
16 *
17 * 1. Allocate a task context struct (e.g. brix_read_aio_t) on the heap.
18 * 2. Post an ngx_thread_task_t to the nginx thread pool.
19 * 3. The thread-pool worker calls *_aio_thread() — pread/pwrite happens here.
20 * 4. When done, nginx posts the *_aio_done() callback back to the main thread.
21 * 5. The done callback queues the response and calls brix_aio_resume() to
22 * wake the connection.
23 *
24 * While AIO is in flight, ctx->state = XRD_ST_AIO and both read/write events
25 * are disarmed. ctx->destroyed is checked in every done callback so that a
26 * callback firing after client disconnect safely discards its work.
27 */
28

46/*
47 * brix_aio_metric_done — phase-56 D-2: file the op-latency histogram sample
48 * for a completed stream data-plane op (kXR_read/pgread/readv → READ,
49 * kXR_write/pgwrite/writev → WRITE).
50 *
51 * HISTOGRAM ONLY (brix_metric_op_latency): the stream plane's io_ops_total and
52 * io_bytes_{read,written} already reach the exporter via the legacy per-port
53 * fold in unified_export_io.c — which also covers the non-AIO paths (sendfile
54 * reads, inline fallbacks) — so booking them again here would double-count.
55 * The latency series is the one thing the AIO plane never emitted; its count
56 * is therefore the AIO-sampled subset of ops, not io_ops_total.
57 *
58 * start_ns is stamped at task-post time and consumed here in the done
59 * callback — both run on the event-loop thread, so the timestamp needs no
60 * synchronization; the worker never touches it. Errored completions file too
61 * (parity with the VFS inline observer, whose histogram includes error ops).
62 * Low-cardinality labels only (proto=stream, op) per INVARIANT #8.
63 */
64static ngx_inline void
65brix_aio_metric_done(uint64_t start_ns, brix_metric_op_t op)
66{
67 uint64_t now_ns = brix_phase_now_ns();
68
69 brix_metric_op_latency(BRIX_PROTO_ROOT, op,
70 (now_ns > start_ns) ? (ngx_msec_t) ((now_ns - start_ns) / 1000ull)
71 : 0);
72}
