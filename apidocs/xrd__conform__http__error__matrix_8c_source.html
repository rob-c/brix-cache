147/* WHAT: Grade a complete six-category ROOT-to-HTTP error-mapping row.
148 * WHY: ERR-043 and HTE-001 express the same cross-plane mapping contract.
149 * HOW: Require labelled fixtures for both planes of every category, then fold
150 * six exact, deterministic comparisons into the selected row. */
151static void
152grade_matrix(xrd_conform_execution *execution, const char *id)
153{
154 const http_error_pair *pairs;
155 const brix_conform_result *row;
156 size_t count;
157 size_t index;
158 size_t supplied = 0;
159 size_t matched = 0;
160 char received[96];
161
162 row = xrd_conform_result_find(execution, id);
163 if (row == NULL || row->verdict == BRIX_CONFORM_VERDICT_SKIPPED) return;
164 pairs = http_error_pairs(&count);
165 for (index = 0; index < count; index++) {
166 const char *root_target = fixture_target(execution->run,
167 pairs[index].name, 0);
168 const char *http_target = fixture_target(execution->run,
169 pairs[index].name, 1);
170
171 if (root_target == NULL && http_target == NULL) continue;
172 if (root_target == NULL || http_target == NULL) {
173 supplied++;
174 continue;
175 }
176 supplied++;
177 if (pair_matches(execution, &pairs[index], root_target, http_target))
178 matched++;
179 }
180 if (supplied == 0) return;
181 snprintf(received, sizeof(received), "pairs=%zu/%zu exact=%zu/%zu",
182 supplied, count, matched, count);
183 xrd_conform_result_set(execution, id,
184 supplied == count && matched == count
185 ? BRIX_CONFORM_VERDICT_CONFORMANT
186 : BRIX_CONFORM_VERDICT_DIVERGENT,
187 supplied == count && matched == count
188 ? "root-and-http-error-matrix-agrees"
189 : "root-and-http-error-matrix-incomplete-or-divergent",
190 "six labelled ROOT/HTTP errno pairs", received);
191}
