1/*
2 * oci_upstream_auth.c — the D1 upstream Bearer token dance.
3 *
4 * WHAT: mint (and cache) the bearer an upstream registry demands, and attach
5 * the supplier to the http storage instance so a cache fill can complete
6 * its own 401 retry without knowing any of this exists.
7 * WHY: every public registry answers an anonymous pull with a 401 and a
8 * challenge naming a token endpoint; the pull only proceeds after a
9 * second, differently-authenticated request to that endpoint. That is
10 * one extra round-trip per object — unacceptable per BLOB on a cold
11 * image pull, which is why the minted token is cached in SHM keyed by
12 * (upstream, scope) and shared across workers. It is also the single
13 * most dangerous instruction the plane accepts: a `realm=` is the
14 * upstream telling us which host to go hand a credential to. So the
15 * realm is checked against the upstream it claims to speak for before
16 * any credential leaves this process, and the Authorization header is
17 * dropped the moment a redirect crosses a host boundary.
18 * HOW: challenge → cache probe → (allowlisted) token GET → JSON extract →
19 * cache store, all on the fill worker thread over the same blocking
20 * libcurl transport the cache origin uses. The SHM zone is the shared
21 * brix_kv table (invariant #10: spin+yield, no I/O under the lock).
22 */

58/* Percent-encode `s` as a single query-parameter value. ngx_escape_uri does
59 * not NUL-terminate and does not bounds-check, so the two-pass form (measure,
60 * then write) is the only correct one: a scope is upstream-chosen text and a
61 * raw ':' or '&' surviving into the query would let it forge parameters. */
62static int
63oci_query_escape(const char *s, u_char *out, size_t outsz)
64{
65 size_t len = strlen(s);
66 uintptr_t esc;
67 u_char *end;
68
69 esc = ngx_escape_uri(NULL, (u_char *) s, len, NGX_ESCAPE_URI_COMPONENT);
70 if (len + 2 * esc >= outsz) {
71 return -1;
72 }
73 end = (u_char *) ngx_escape_uri(out, (u_char *) s, len,
74 NGX_ESCAPE_URI_COMPONENT);
75 *end = '\0';
76 return 0;
77}

79/* Point `leg` at the Location of a redirect. Absolute URLs are re-parsed in
80 * full (and drop the Basic credential when the host changes); a relative one
81 * keeps the current destination and replaces only the path. */
82static int
83oci_token_follow(oci_token_leg_t *leg, const char *loc)
84{
85 brix_oci_url_t u;
86 const char *query;
87 size_t path_len;
88
89 if (loc[0] == '/') {
90 return (snprintf(leg->path, sizeof(leg->path), "%s", loc)
91 < (int) sizeof(leg->path)) ? 0 : -1;
92 }
93
94 if (brix_oci_url_parse(loc, strlen(loc), &u) != 0) {
95 return -1;
96 }
97 if (strcasecmp(u.host, leg->host) != 0) {
98 /* The next leg is a different principal's endpoint, so our Basic
99 * credential does not travel with it. */
100 leg->send_basic = 0;
101 }
102 (void) snprintf(leg->host, sizeof(leg->host), "%s", u.host);
103 leg->port = u.port;
104 leg->tls = u.tls;
105
106 /* brix_oci_url_parse stops the path at '?'; carry the query across so a
107 * token endpoint may redirect while preserving service/scope. */
108 query = strchr(loc, '?');
109 path_len = (size_t) snprintf(leg->path, sizeof(leg->path), "%s%s",
110 u.path[0] ? u.path : "/",
111 (query != NULL) ? query : "");
112
113 return (path_len < sizeof(leg->path)) ? 0 : -1;
114}

116/* Build "Authorization: Basic <b64>\r\n" for the token endpoint, or "" when
117 * the mint is anonymous. The Basic credential authenticates a principal to
118 * the TOKEN service only — it is never sent to the registry data endpoints,
119 * and never survives a cross-host redirect. */
120static void
121oci_token_basic_header(const char *basic, char *out, size_t outsz)
122{
123 ngx_str_t src, dst;
124 u_char b64[684]; /* base64 of the 512-byte basic buffer */
125
126 out[0] = '\0';
127 if (basic == NULL || basic[0] == '\0') {
128 return;
129 }
130 src.data = (u_char *) basic;
131 src.len = strlen(basic);
132 dst.data = b64;
133 ngx_encode_base64(&dst, &src);
134
135 (void) snprintf(out, outsz, "Authorization: Basic %.*s\r\n",
136 (int) dst.len, dst.data);
137}

139/* One leg of the token request. Returns the HTTP status, or -1 on a transport
140 * failure. On a 3xx the Location is resolved into *leg for the next hop; a 3xx
141 * without a usable Location is a failure, not a hop we could repeat. */
142static int
143oci_token_leg(const brix_oci_upstream_t *up, oci_token_leg_t *leg,
144 char *body, size_t bodysz, size_t *body_len)
145{
146 const brix_s3_transport_t *tr = &brix_s3_origin_curl_transport;
147 brix_s3_resp_t resp;
148 char hdrs[1024];
149 char loc[1024];
150 const void *rb;
151 size_t blen = 0;
152 char errbuf[256];
153 int status;
154
155 hdrs[0] = '\0';
156 if (leg->send_basic) {
157 oci_token_basic_header(leg->basic, hdrs, sizeof(hdrs));
158 }
159
160 if (tr->request(NULL, leg->host, leg->port, leg->tls, "GET", leg->path,
161 hdrs[0] ? hdrs : NULL, NULL, 0, OCI_TOKEN_TIMEOUT_MS,
162 &resp, errbuf, sizeof(errbuf)) != 0)
163 {
164 ngx_log_error(NGX_LOG_ERR, up->log, 0,
165 "oci: token endpoint unreachable host=%s: %s", leg->host, errbuf);
166 return -1;
167 }
168 status = resp.status;
169
170 if (status >= 300 && status < 400) {
171 int rc = tr->resp_header(&resp, "location", loc, sizeof(loc));
172
173 tr->resp_free(&resp);
174 if (rc != 0 || oci_token_follow(leg, loc) != 0) {
175 ngx_log_error(NGX_LOG_ERR, up->log, 0,
176 "oci: token endpoint host=%s redirected without a usable "
177 "Location", leg->host);
178 return -1;
179 }
180 return status;
181 }
182
183 rb = tr->resp_body(&resp, &blen);
184 if (rb != NULL && blen > 0 && blen < bodysz) {
185 memcpy(body, rb, blen);
186 body[blen] = '\0';
187 *body_len = blen;
188 } else {
189 body[0] = '\0';
190 *body_len = 0;
191 }
192 tr->resp_free(&resp);
193
194 return status;
195}

197/* Turn a parsed WWW-Authenticate challenge into the one HTTP leg that mints
198 * the bearer: realm → host/port/tls/path, with the service and scope escaped
199 * into the query. 0 = `leg` is ready to run; -1 = the realm is unusable or
200 * off-boundary (already logged). */
201static int
202oci_token_leg_init(brix_oci_upstream_t *up, const brix_oci_challenge_t *ch,
203 const char *scope, const char *basic, oci_token_leg_t *leg)
204{
205 brix_oci_url_t realm;
206 u_char enc_scope[1024];
207 u_char enc_svc[768];
208
209 if (brix_oci_url_parse(ch->realm, strlen(ch->realm), &realm) != 0) {
210 ngx_log_error(NGX_LOG_ERR, up->log, 0,
211 "oci: challenge realm is not an absolute http(s) URL");
212 return -1;
213 }
214
215 /* THE gate: an upstream may only send us to its own token service, or to
216 * a host the operator named with brix_oci_upstream_auth_realm (§D15.11). */
217 if (!brix_oci_url_realm_allowed_ex(up->host, realm.host, &up->realms)) {
218 ngx_log_error(NGX_LOG_ERR, up->log, 0,
219 "oci: refusing token realm host=%s for upstream=%s "
220 "signal=oci_realm_refused", realm.host, up->host);
221 return -1;
222 }
223
224 /* A realm reached only because it was allowlisted is worth one line: the
225 * dance is cached per scope, so this is rare, and it is the audit record
226 * that the widened boundary is the one actually being used. */
227 if (up->realms.n > 0 && !brix_oci_url_realm_allowed(up->host, realm.host)) {
228 ngx_log_error(NGX_LOG_INFO, up->log, 0,
229 "oci: token realm host=%s is off-domain for upstream=%s and was "
230 "honoured by brix_oci_upstream_auth_realm", realm.host, up->host);
231 }
232
233 if (oci_query_escape(scope, enc_scope, sizeof(enc_scope)) != 0
234 || oci_query_escape(ch->service, enc_svc, sizeof(enc_svc)) != 0)
235 {
236 return -1;
237 }
238
239 memset(leg, 0, sizeof(*leg));
240 (void) snprintf(leg->host, sizeof(leg->host), "%s", realm.host);
241 leg->port = realm.port;
242 leg->tls = realm.tls;
243 leg->basic = basic;
244 leg->send_basic = 1;
245
246 return (snprintf(leg->path, sizeof(leg->path), "%s?service=%s&scope=%s",
247 realm.path[0] ? realm.path : "/", enc_svc, enc_scope)
248 >= (int) sizeof(leg->path)) ? -1 : 0;
249}

252/* Walk the token endpoint's redirects to the first non-3xx answer. The HTTP
253 * status, or -1 when a leg failed, a redirect tried to leave the upstream's
254 * trust boundary — a token minted off-boundary is a credential handed to a
255 * host the operator never named — or the chain never terminated within
256 * OCI_TOKEN_MAX_HOPS. */
257static int
258oci_token_run(brix_oci_upstream_t *up, oci_token_leg_t *leg,
259 char *body, size_t bodysz, size_t *body_len)
260{
261 int hop, status;
262
263 for (hop = 0; hop <= OCI_TOKEN_MAX_HOPS; hop++) {
264 status = oci_token_leg(up, leg, body, bodysz, body_len);
265 if (status < 0) {
266 return -1;
267 }
268 if (status < 300 || status >= 400) {
269 return status;
270 }
271 if (!brix_oci_url_realm_allowed_ex(up->host, leg->host, &up->realms)) {
272 ngx_log_error(NGX_LOG_ERR, up->log, 0,
273 "oci: refusing token redirect host=%s for upstream=%s "
274 "signal=oci_realm_refused", leg->host, up->host);
275 return -1;
276 }
277 }
278
279 /* Ran out of hops with a 3xx still on the wire. Returning the last status
280 * would hand the caller a redirect as though it were the token endpoint's
281 * answer; an unterminated chain is a failure, and the only status the
282 * contract above admits for one is -1. */
283 ngx_log_error(NGX_LOG_ERR, up->log, 0,
284 "oci: token redirect chain exceeded %d hops for upstream=%s "
285 "signal=oci_token_redirect_loop", OCI_TOKEN_MAX_HOPS, up->host);
286 return -1;
287}

290/* Lift the bearer and its lifetime out of the token endpoint's JSON. */
291static int
292oci_token_parse(brix_oci_upstream_t *up, const char *host, const char *body,
293 size_t body_len, char *tok, size_t toklen, long *expires_in)
294{
295 char expbuf[32];
296 long v;
297
298 /* Both spellings are in the wild: the Distribution spec says "token",
299 * OAuth2 says "access_token", and GitLab/Harbor emit both. */
300 if (!brix_json_get_str(body, body_len, "token", tok, toklen)
301 && !brix_json_get_str(body, body_len, "access_token", tok, toklen))
302 {
303 ngx_log_error(NGX_LOG_ERR, up->log, 0,
304 "oci: token response from %s carried no token", host);
305 return -1;
306 }
307 if (tok[0] == '\0') {
308 return -1;
309 }
310
311 *expires_in = OCI_TOKEN_DEFAULT_S;
312 if (brix_json_get_str(body, body_len, "expires_in", expbuf,
313 sizeof(expbuf)))
314 {
315 v = strtol(expbuf, NULL, 10);
316 if (v > 0) {
317 *expires_in = v;
318 }
319 }
320
321 return 0;
322}

325/* Mint one bearer: build the token leg from the challenge, walk it, and read
326 * the answer. Every failure path is a single counter here, so the three halves
327 * above can each fail plainly and stay readable. A 401/403 from the token
328 * endpoint itself sets *denied — the D16 gate needs "this credential was
329 * refused" kept apart from "the endpoint was unreachable", because only the
330 * former is a verdict about the principal. */
331static int
332oci_token_fetch(brix_oci_upstream_t *up, const brix_oci_challenge_t *ch,
333 const char *scope, const char *basic, char *tok, size_t toklen,
334 long *expires_in, int *denied)
335{
336 oci_token_leg_t leg;
337 char body[16384];
338 size_t body_len = 0;
339 int status;
340
341 if (oci_token_leg_init(up, ch, scope, basic, &leg) != 0) {
342 BRIX_OCI_METRIC_INC(token_fetch_total[BRIX_OCI_TOKEN_FAILED]);
343 return -1;
344 }
345
346 status = oci_token_run(up, &leg, body, sizeof(body), &body_len);
347 if (status != NGX_HTTP_OK || body_len == 0) {
348 if (status >= 0) {
349 ngx_log_error(NGX_LOG_ERR, up->log, 0,
350 "oci: token endpoint host=%s answered %d", leg.host, status);
351 }
352 if (denied != NULL
353 && (status == NGX_HTTP_UNAUTHORIZED
354 || status == NGX_HTTP_FORBIDDEN))
355 {
356 *denied = 1;
357 }
358 BRIX_OCI_METRIC_INC(token_fetch_total[BRIX_OCI_TOKEN_FAILED]);
359 return -1;
360 }
361
362 if (oci_token_parse(up, leg.host, body, body_len, tok, toklen,
363 expires_in) != 0)
364 {
365 BRIX_OCI_METRIC_INC(token_fetch_total[BRIX_OCI_TOKEN_FAILED]);
366 return -1;
367 }
368
369 BRIX_OCI_METRIC_INC(token_fetch_total[BRIX_OCI_TOKEN_FETCHED]);
370 return 0;
371}

374int
375brix_oci_token_get_cred(brix_oci_upstream_t *up, const char *path,
376 const char *challenge, const char *basic, const u_char *cred,
377 char *tok, size_t toklen, long *expires_in, int *denied)
378{
379 brix_oci_challenge_t ch;
380 char scope[1024];
381 long expires = 0;
382
383 if (denied != NULL) {
384 *denied = 0;
385 }
386 if (up == NULL || challenge == NULL || toklen == 0) {
387 return -1;
388 }
389 if (brix_oci_challenge_parse(challenge, strlen(challenge), &ch) != 0) {
390 /* Not a Bearer challenge (Basic, Negotiate, junk): nothing this
391 * function can mint, and guessing would send a credential to a
392 * scheme that never asked for one. */
393 return -1;
394 }
395
396 if (ch.scope[0] != '\0') {
397 (void) snprintf(scope, sizeof(scope), "%s", ch.scope);
398
399 } else if (brix_oci_pull_scope(path, scope, sizeof(scope)) != 0) {
400 return -1;
401 }
402
403 if (brix_oci_token_cache_get(up, scope, cred, tok, toklen) == 0) {
404 if (expires_in != NULL) {
405 *expires_in = -1; /* cached: lifetime unknown here */
406 }
407 return 0;
408 }
409
410 if (oci_token_fetch(up, &ch, scope, basic, tok, toklen, &expires,
411 denied) != 0)
412 {
413 return -1;
414 }
415
416 brix_oci_token_cache_put(up, scope, cred, tok, expires);
417 if (expires_in != NULL) {
418 *expires_in = expires;
419 }
420 return 0;
421}

466void
467brix_oci_bind_bearer(ngx_http_brix_oci_loc_conf_t *lcf,
468 brix_sd_instance_t *inst, ngx_log_t *log)
469{
470 if (lcf->bearer_bound || lcf->up == NULL || inst == NULL) {
471 return;
472 }
473
474 /* The instance handed in is the composed stack (cache decorators over the
475 * http source); the supplier belongs on the source that actually speaks
476 * to the registry. */
477 while (inst != NULL && ngx_strcmp(inst->driver->name, "http") != 0) {
478 inst = brix_sd_cache_source_instance(inst);
479 }
480 if (inst == NULL) {
481 return;
482 }
483
484 /* The fill runs on a worker thread long after this request is gone: the
485 * upstream descriptor is config-pool-owned (process lifetime) and the log
486 * is the cycle's, never the connection's. */
487 lcf->up->log = ngx_cycle->log;
488 sd_http_set_bearer_provider(inst, oci_bearer_provider, lcf->up);
489 lcf->bearer_bound = 1;
490
491 ngx_log_error(NGX_LOG_INFO, log, 0,
492 "oci: bearer supplier bound to upstream %s", lcf->up->base_url);
493}
