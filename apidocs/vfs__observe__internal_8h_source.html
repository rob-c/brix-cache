8/* Pick the protocol label for this ctx's metrics, defaulting to
9 * BRIX_PROTO_ROOT when ctx is NULL or its metrics_proto is out of range. */
10static ngx_inline brix_proto_t
11brix_vfs_metrics_proto(const brix_vfs_ctx_t *ctx)
12{
13 if (ctx == NULL || ctx->metrics_proto >= BRIX_PROTO_COUNT) {
14 return BRIX_PROTO_ROOT;
15 }

20/* phase-56 D-1: a real monotonic timestamp in NANOseconds for op-latency.
21 * Replaces the cached ngx_current_msec, which (a) only advances on event-loop
22 * ticks — so a synchronous metadata op that never yields reported 0 µs — and
23 * (b) is millisecond-resolution, quantizing the whole sub-ms band to 0/1000 µs.
24 * CLOCK_MONOTONIC is vDSO-backed (~20 ns/call, lost in the syscalls the op
25 * already makes) and gives honest sub-µs deltas. NOT CLOCK_MONOTONIC_COARSE —
26 * that is also ~1-4 ms granularity and would only fix (a), not the resolution. */
27static ngx_inline uint64_t
28brix_vfs_now_ns(void)
29{
30 struct timespec ts;
31
32 clock_gettime(CLOCK_MONOTONIC, &ts);
33 return (uint64_t) ts.tv_sec * 1000000000ull + (uint64_t) ts.tv_nsec;
34}

36/* Latency since start_ns in MICROseconds (start is an brix_vfs_now_ns()
37 * snapshot). Clamps to 0 if the monotonic clock appears to have gone backwards. */
38static ngx_inline ngx_msec_t
39brix_vfs_elapsed_usec(uint64_t start_ns)
40{
41 uint64_t now_ns = brix_vfs_now_ns();
42
43 if (now_ns < start_ns) {
44 return 0;
45 }
46
47 return (ngx_msec_t) ((now_ns - start_ns) / 1000ull);
48}

50/* Post-op observer: derive the error class from rc/sys_errno, compute latency
51 * from start_msec, then emit one metric (brix_metric_op_done) and one access
52 * log line (brix_access_log_emit) for op. bytes is the transferred count;
53 * result may be NULL. Borrows path (does not copy). Restores errno=sys_errno on
54 * return so the caller can propagate it unchanged.
55 *
56 * ctx == NULL means there is no request context — an internal maintenance op
57 * (e.g. the integrity code persisting checksum sidecars via the NULL-ctx
58 * f-xattr variants). Those are not client I/O: observing them would default
59 * the proto label to "stream" and misattribute s3/webdav-triggered sidecar
60 * touches, so they are deliberately not metered or access-logged. */
61static ngx_inline void
62brix_vfs_observe_ctx_op_ex(const brix_vfs_ctx_t *ctx, const char *path,
63 brix_metric_op_t op, const brix_vfs_io_result_t *result,
64 size_t bytes, ngx_int_t rc, int sys_errno, uint64_t start_ns,
65 unsigned meter_io)
66{
67 brix_err_class_t err;
68 ngx_msec_t latency_usec;
69
70 if (ctx == NULL) {
71 errno = sys_errno;
72 return;
73 }
74
75 err = rc == NGX_OK ? BRIX_ERR_NONE
76 : brix_metric_err_from_errno(sys_errno);
77 latency_usec = brix_vfs_elapsed_usec(start_ns);
78
79 /* phase-106 W1 / phase-110 W1-W4: fold this op into the request's or
80 * session's I/O monitor, which the uniform $brix_* variables read at log
81 * time (io_monitor.h). NULL monitor = an unmonitored path (metadata-only
82 * builders, internal maintenance) and is the common case; every helper
83 * below is a silent no-op on NULL.
84 * - op/path/outcome: candidate primary op under the weight rule, on
85 * success AND failure (a failed stat on a GET of a missing file is the
86 * outcome the operator wants: op=stat status=not_found).
87 * - backend time: successful ops only — a failed op's latency is error
88 * handling, not backend service time.
89 * - received bytes: a successful WRITE's count is what the client sent
90 * (the staged PUT commit observes its total once; a GET's cache fill
91 * writes through its own unmonitored ctx, so it never lands here).
92 * - SERVED bytes are deliberately NOT folded here: the client-facing
93 * serve is zero-copy (sendfile / output filter) and never reaches this
94 * observer, so the plane books result->bytes_sent at its serve site.
95 * Single-writer contract: see io_monitor.h. */
96 brix_io_monitor_record_op(ctx->io_monitor, op, path, err);
97 if (rc == NGX_OK) {
98 brix_io_monitor_add_latency(ctx->io_monitor, latency_usec);
99 if (op == BRIX_METRIC_OP_WRITE) {
100 brix_io_monitor_add_received(ctx->io_monitor, bytes);
101 }
102 }
103
104 /* meter_io == 0: the owning protocol books the unified io_ops/latency row
105 * for this operation itself (data-plane READ/WRITE via *_metrics_response,
106 * bytes via the per-protocol wire-ledger fold), so emitting it here too
107 * would double-count. Backend byte totals and the access-log line stay
108 * VFS-owned either way. */
109 if (meter_io) {
110 brix_metric_op_done(brix_vfs_metrics_proto(ctx), op, bytes,
111 latency_usec, err);
112 }
113
114 /* Per-backend storage byte totals (staged-commit writes, VFS-metered
115 * reads). ctx->sd == NULL is the default-POSIX instance. */
116 if (rc == NGX_OK && bytes > 0) {
117 brix_metric_backend_bytes(
118 ctx != NULL && ctx->sd != NULL ? brix_sd_backend_name(ctx->sd)
119 : "posix",
120 op, bytes);
121 }
122
123 brix_access_log_emit(ctx, path, op, result, bytes, err, latency_usec);
124
125 errno = sys_errno;
126}
