1/*
2 * cvmfs_module_build.c — cvmfs:// enabled-branch export/backend/cache build.
3 *
4 * WHAT: The two functions that run only when a location has brix_cvmfs on:
5 * - brix_cvmfs_reject_unsupported — EMERG at config load for storage
6 * grammar cvmfs cannot honour (staging, CAS slicing, explicit writes).
7 * - cvmfs_merge_cache — the enabled-branch build: reject unsupported grammar,
8 * force read-only, rewrite the posix backend, anchor the export root
9 * (defaulting to "/"), open the confinement rootfd, register the
10 * composable http backend, stamp the cache TTL/deadline knobs, compose the
11 * cache/stage tiers, then apply T19 origin selection (geo ranks / rtt
12 * probe registration) and the coords-without-geo warning.
13 *
14 * WHY: Split out of module.c (file-size gate). This is the one part of the merge
15 * that mutates persistent export state and registers backends — a distinct
16 * concern from the pure config-field merge (cvmfs_module_merge.c). Isolating
17 * it lets the merge orchestrator gate it behind a single enable check while
18 * the build steps stay in their exact original order. Every EMERG/rejection
19 * line and the build-step order are byte-frozen against the pre-split module.c.
20 *
21 * HOW: Verbatim move. brix_cvmfs_reject_unsupported stays static (its only
22 * caller, cvmfs_merge_cache, lives here); cvmfs_merge_cache is exported via
23 * cvmfs_module_internal.h (the merge orchestrator in cvmfs_module_merge.c
24 * invokes it) and imports cvmfs_geo_rank_config from cvmfs_module_georank.c
25 * through the same header.
26 */

46/*
47 * brix_cvmfs_reject_unsupported() — EMERG at config load for storage grammar
48 * that cvmfs cannot honour.
49 *
50 * WHAT: Checks three merged common-preamble fields for features that are
51 * structurally incompatible with a read-only content-addressed site cache:
52 * (1) staging — brix_stage on / brix_stage_store, (2) object slicing —
53 * brix_cache_slice_size, and (3) explicit write permission — brix_allow_write.
54 *
55 * WHY: Without an explicit rejection at config-load time these directives
56 * silently pass nginx -t, misleading operators into believing their config
57 * is active when it is not. cvmfs is structurally read-only; writes and
58 * staging have no meaning (CAS objects are immutable whole objects), and
59 * slicing never applies because the cache fill and CAS verification run on
60 * the whole object. A loud nginx -t rejection surfaces operator mistakes
61 * before traffic is served.
62 *
63 * HOW: Called at the top of the cvmfs-enabled branch in merge_loc_conf,
64 * after brix_http_common_adopt() and ngx_http_brix_shared_merge() have
65 * resolved unified directive values into conf->common (so brix_stage,
66 * brix_allow_write, and brix_cache_slice_size are visible). Returns
67 * NGX_CONF_ERROR on any violation; NGX_CONF_OK otherwise. The caller
68 * must check the return and propagate NGX_CONF_ERROR to nginx.
69 */
70static char *
71brix_cvmfs_reject_unsupported(ngx_conf_t *cf,
72 ngx_http_brix_cvmfs_loc_conf_t *conf)
73{
74 /* Staging (brix_stage on / brix_stage_store) implies mutability: a cvmfs
75 * cache is a read-through CAS store; write-back staging makes no sense. */
76 if (conf->common.stage_enable == 1 || conf->common.stage_store.len > 0) {
77 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
78 "brix_stage/brix_stage_store: cvmfs is a read-only protocol; "
79 "staging is not supported");
80 return NGX_CONF_ERROR;
81 }
82
83 /* CAS object slicing: cvmfs objects are immutable and must be filled and
84 * verified as whole units; per-slice CAS verification is undefined. */
85 if (conf->common.cache_slice_size > 0) {
86 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
87 "brix_cache_slice_size: cvmfs CAS objects are immutable whole "
88 "objects; slicing is not supported");
89 return NGX_CONF_ERROR;
90 }
91
92 /* Explicit write permission: the hard force (conf->common.allow_write = 0)
93 * below already prevents accidental writes, but an explicit on is a
94 * config mistake that deserves a loud rejection rather than silent no-op. */
95 if (conf->common.allow_write == 1) {
96 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
97 "brix_allow_write on: cvmfs is a read-only protocol; "
98 "write permission cannot be granted");
99 return NGX_CONF_ERROR;
100 }
101
102 return NGX_CONF_OK;
103}

105/*
106 * cvmfs_stratum0_apply() — phase-96 S13: resolve the brix_cvmfs_stratum0_root
107 * alias.
108 *
109 * WHAT: When the directive is set, refuse every cache-fill upstream shape in
110 * the same block (an http(s) brix_storage_backend, a brix_cache_store fill
111 * tier, a brix_cvmfs_upstream_allow proxy allowlist) and refuse an
112 * ambiguous double root (brix_export AND the alias), then anchor the
113 * export root at the Stratum-0 tree.
114 *
115 * WHY: A Stratum-0 has no upstream — it IS the origin. Letting fill grammar
116 * coexist would silently turn the master copy into a cache node; the alias
117 * exists precisely so the intent is visible and enforced at nginx -t.
118 *
119 * HOW: Runs before the posix-root rewrite so conf->common.root feeds the
120 * normal brix_prepare_export_root path unchanged. No-op when unset.
121 */
122static char *
123cvmfs_stratum0_apply(ngx_conf_t *cf, ngx_http_brix_cvmfs_loc_conf_t *conf)
124{
125 if (conf->cvmfs.stratum0_root.len == 0) {
126 return NGX_CONF_OK;
127 }
128
129 if (conf->common.root.len > 0) {
130 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
131 "brix_cvmfs_stratum0_root and brix_export both name an export "
132 "root - the alias replaces brix_export; configure exactly one");
133 return NGX_CONF_ERROR;
134 }
135
136 if (conf->common.storage_backend.len >= 4
137 && ngx_strncasecmp(conf->common.storage_backend.data,
138 (u_char *) "http", 4) == 0)
139 {
140 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
141 "brix_cvmfs_stratum0_root: a Stratum-0 has no upstream - remove "
142 "the http(s) brix_storage_backend from this block");
143 return NGX_CONF_ERROR;
144 }
145
146 if (conf->common.cache_store.len > 0) {
147 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
148 "brix_cvmfs_stratum0_root: a Stratum-0 serves its published "
149 "tree directly - remove brix_cache_store (cache-fill) from "
150 "this block");
151 return NGX_CONF_ERROR;
152 }
153
154 if (conf->cvmfs.upstream_allow != NULL) {
155 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
156 "brix_cvmfs_stratum0_root: a Stratum-0 has no upstream - remove "
157 "brix_cvmfs_upstream_allow (proxy mode) from this block");
158 return NGX_CONF_ERROR;
159 }
160
161 conf->common.root = conf->cvmfs.stratum0_root;
162 return NGX_CONF_OK;
163}

165/*
166 * cvmfs_merge_cache() — the cvmfs-enabled export/backend/cache build block.
167 *
168 * WHAT: When enable=1: reject unsupported storage grammar, force read-only,
169 * rewrite the posix backend, anchor the export root (defaulting to "/"),
170 * open the confinement rootfd, register the composable http backend, stamp
171 * the cache TTL/deadline knobs, compose the cache/stage tiers, then apply the
172 * T19 origin-selection (geo ranks / rtt probe registration) and the
173 * coords-without-geo warning.
174 *
175 * WHY: This is the one part of the merge that mutates persistent export state
176 * and registers backends; it only runs for cvmfs-enabled locations. Splitting
177 * it out lets the orchestrator gate it with a single early return while the
178 * build steps stay in their exact original order.
179 *
180 * HOW: Verbatim move of the `if (conf->cvmfs.enable)` block; returns
181 * NGX_CONF_ERROR on any step failure (identical EMERG/rejection lines), else
182 * NGX_CONF_OK. The caller only invokes it when enable is set.
183 */
184char *
185cvmfs_merge_cache(ngx_conf_t *cf, ngx_http_brix_cvmfs_loc_conf_t *conf)
186{
187 /* Reject storage grammar cvmfs cannot honour before doing anything
188 * else: staging, slicing, and explicit writes make no sense for a
189 * read-only content-addressed site cache. */
190 if (brix_cvmfs_reject_unsupported(cf, conf) != NGX_CONF_OK) {
191 return NGX_CONF_ERROR;
192 }
193
194 /* Phase-96 S13: Stratum-0 alias — refuse cache-fill upstream grammar,
195 * then anchor the export root at the published tree. */
196 if (cvmfs_stratum0_apply(cf, conf) != NGX_CONF_OK) {
197 return NGX_CONF_ERROR;
198 }
199
200 /* CVMFS is a read-only protocol: no directive can enable writes. */
201 conf->common.allow_write = 0;
202
203 /* "posix:<path>" backend names the local export tree (composable
204 * brix_root replacement) — same rewrite every protocol applies. */
205 brix_storage_backend_posix_root(&conf->common);
206
207 /* Pure cache node (the normal CVMFS shape): no local export tree, so
208 * the anchor falls back to "/" and the http(s) Stratum-1 origin URL
209 * routes every VFS op to sd_http. */
210 if (brix_http_merge_export_anchor(cf, &conf->common, "brix_cvmfs", 0)
211 != NGX_CONF_OK)
212 {
213 return NGX_CONF_ERROR;
214 }
215
216 /* Verify-mismatch evidence lands in the protocol's quarantine dir;
217 * MANIFEST-class fills get the protocol's TTL stamped (T12). */
218 conf->common.cache_quarantine_dir = conf->cvmfs.quarantine_dir;
219 conf->common.cache_manifest_ttl = conf->cvmfs.manifest_ttl;
220 conf->common.cache_offline_ttl = conf->cvmfs.offline_ttl;
221 /* phase-85 F1: the master-key PATH rides the common preamble; the tier
222 * registration loads the PEM once at config time into the cache policy. */
223 conf->common.cache_cvmfs_master_key = conf->cvmfs.master_key;
224 /* T20 never-drop deadlines for the shared fill machinery */
225 conf->common.cache_client_hold = conf->cvmfs.client_hold;
226 conf->common.cache_fill_max_life = conf->cvmfs.fill_max_life;
227
228 /* Phase-64: compose the cache/stage tiers over the backend. */
229 if (brix_tier_register_stores(cf, &conf->common) != NGX_OK) {
230 return NGX_CONF_ERROR;
231 }
232
233 /* T19 origin selection: geo ranks compute once at config time;
234 * rtt registers the per-worker probe; static keeps configured
235 * order (all ranks 0 — the pick is order-stable on ties). */
236 if (conf->cvmfs.origin_select == BRIX_CVMFS_SELECT_GEO) {
237 if (cvmfs_geo_rank_config(cf, conf) != NGX_CONF_OK) {
238 return NGX_CONF_ERROR;
239 }
240 } else if (conf->cvmfs.origin_select == BRIX_CVMFS_SELECT_RTT) {
241 brix_cvmfs_rtt_register(conf->common.root_canon,
242 conf->cvmfs.rtt_interval,
243 &conf->common.thread_pool_name);
244 }
245
246 return cvmfs_merge_services(cf, conf);
247}

249/* The phase-87 background-service registrations (scrub / learn / swarm):
250 * each rides the RTT-probe lifecycle — registered here at config time, the
251 * worker-0 timer / per-worker state arms at init_process. */
252static char *
253cvmfs_merge_services(ngx_conf_t *cf, ngx_http_brix_cvmfs_loc_conf_t *conf)
254{
255 /* Phase-87 G17: register the background CAS scrub for this export; the
256 * worker-0 timer arms at init_process (scrub.c, RTT-probe lifecycle). */
257 if (conf->cvmfs.scrub) {
258 brix_cvmfs_scrub_register(conf->common.root_canon,
259 conf->cvmfs.scrub_interval,
260 conf->cvmfs.scrub_rate,
261 &conf->common.thread_pool_name);
262 }
263
264 /* Phase-87 G11: register the predictive-prewarm learner for this export;
265 * the per-worker model + fill task build at init_process (learn.c). */
266 if (conf->cvmfs.learn) {
267 brix_cvmfs_learn_register(conf->common.root_canon,
268 &conf->common.thread_pool_name);
269 }
270
271 /* Phase-87 G12: the swarm extends the F8 static mesh — it needs the
272 * brix_cache_peers seed ring to know who this node IS. Fail loudly at
273 * config time instead of gossiping into the void. */
274 if (conf->cvmfs.swarm) {
275 if (conf->common.cache_peers == NULL) {
276 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
277 "brix_cvmfs_swarm requires brix_cache_peers (the seed "
278 "ring naming this node's own \"self=\" slot)");
279 return NGX_CONF_ERROR;
280 }
281 brix_cvmfs_swarm_register(conf->common.root_canon,
282 conf->cvmfs.swarm_interval,
283 &conf->common.thread_pool_name);
284 }
285
286 /* WARN (not NOTICE — config-parse NOTICE is dropped at cf->log ERR
287 * level) when coordinates were supplied but origin_select is not geo:
288 * the coordinates are silently ignored in all other modes and the
289 * operator may not realise their geo config has no effect. */
290 if (conf->cvmfs.origin_select != BRIX_CVMFS_SELECT_GEO
291 && conf->cvmfs.origin_coords != NULL)
292 {
293 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
294 "brix_cvmfs_origin_coords set but brix_cvmfs_origin_select "
295 "is not geo — coordinates are ignored");
296 }
297 return NGX_CONF_OK;
298}
