2 * cvmfs_module_build.c — cvmfs:// enabled-branch export/backend/cache build.
3 *
4 * WHAT: The two functions that run only when a location has brix_cvmfs on:
5 * - brix_cvmfs_reject_unsupported — EMERG at config load for storage
6 * grammar cvmfs cannot honour (staging, CAS slicing, explicit writes).
7 * - cvmfs_merge_cache — the enabled-branch build: reject unsupported grammar,
8 * force read-only, rewrite the posix backend, anchor the export root
9 * (defaulting to "/"), open the confinement rootfd, register the
10 * composable http backend, stamp the cache TTL/deadline knobs, compose the
11 * cache/stage tiers, then apply T19 origin selection (geo ranks / rtt
12 * probe registration) and the coords-without-geo warning.
13 *
14 * WHY: Split out of module.c (file-size gate). This is the one part of the merge
15 * that mutates persistent export state and registers backends — a distinct
16 * concern from the pure config-field merge (cvmfs_module_merge.c). Isolating
17 * it lets the merge orchestrator gate it behind a single enable check while
18 * the build steps stay in their exact original order. Every EMERG/rejection
19 * line and the build-step order are byte-frozen against the pre-split module.c.
20 *
21 * HOW: Verbatim move. brix_cvmfs_reject_unsupported stays static (its only
22 * caller, cvmfs_merge_cache, lives here); cvmfs_merge_cache is exported via
23 * cvmfs_module_internal.h (the merge orchestrator in cvmfs_module_merge.c
24 * invokes it) and imports cvmfs_geo_rank_config from cvmfs_module_georank.c
25 * through the same header.
26 */

48/*
49 * brix_cvmfs_reject_unsupported() — EMERG at config load for storage grammar
50 * that cvmfs cannot honour.
51 *
52 * WHAT: Checks three merged common-preamble fields for features that are
53 * structurally incompatible with a read-only content-addressed site cache:
54 * (1) staging — brix_stage on / brix_stage_store, (2) object slicing —
55 * brix_cache_slice_size, and (3) explicit write permission — brix_allow_write.
56 *
57 * WHY: Without an explicit rejection at config-load time these directives
58 * silently pass nginx -t, misleading operators into believing their config
59 * is active when it is not. cvmfs is structurally read-only; writes and
60 * staging have no meaning (CAS objects are immutable whole objects), and
61 * slicing never applies because the cache fill and CAS verification run on
62 * the whole object. A loud nginx -t rejection surfaces operator mistakes
63 * before traffic is served.
64 *
65 * HOW: Called at the top of the cvmfs-enabled branch in merge_loc_conf,
66 * after brix_http_common_adopt() and ngx_http_brix_shared_merge() have
67 * resolved unified directive values into conf->common (so brix_stage,
68 * brix_allow_write, and brix_cache_slice_size are visible). Returns
69 * NGX_CONF_ERROR on any violation; NGX_CONF_OK otherwise. The caller
70 * must check the return and propagate NGX_CONF_ERROR to nginx.
71 */
72static char *
73brix_cvmfs_reject_unsupported(ngx_conf_t *cf,
74 ngx_http_brix_cvmfs_loc_conf_t *conf)
75{
76 /* Staging (brix_stage on / brix_stage_store) implies mutability: a cvmfs
77 * cache is a read-through CAS store; write-back staging makes no sense. */
78 if (conf->common.stage_enable == 1 || conf->common.stage_store.len > 0) {
79 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
80 "brix_stage/brix_stage_store: cvmfs is a read-only protocol; "
81 "staging is not supported");
82 return NGX_CONF_ERROR;
83 }
84
85 /* CAS object slicing: cvmfs objects are immutable and must be filled and
86 * verified as whole units; per-slice CAS verification is undefined. */
87 if (conf->common.cache_slice_size > 0) {
88 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
89 "brix_cache_slice_size: cvmfs CAS objects are immutable whole "
90 "objects; slicing is not supported");
91 return NGX_CONF_ERROR;
92 }
93
94 /* Explicit write permission: the hard force (conf->common.allow_write = 0)
95 * below already prevents accidental writes, but an explicit on is a
96 * config mistake that deserves a loud rejection rather than silent no-op. */
97 if (conf->common.allow_write == 1) {
98 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
99 "brix_allow_write on: cvmfs is a read-only protocol; "
100 "write permission cannot be granted");
101 return NGX_CONF_ERROR;
102 }
103
104 return NGX_CONF_OK;
105}

107/*
108 * cvmfs_stratum0_apply() — phase-96 S13: resolve the brix_cvmfs_stratum0_root
109 * alias.
110 *
111 * WHAT: When the directive is set, refuse every cache-fill upstream shape in
112 * the same block (an http(s) brix_storage_backend, a brix_cache_store fill
113 * tier, a brix_cvmfs_upstream_allow proxy allowlist) and refuse an
114 * ambiguous double root (brix_export AND the alias), then anchor the
115 * export root at the Stratum-0 tree.
116 *
117 * WHY: A Stratum-0 has no upstream — it IS the origin. Letting fill grammar
118 * coexist would silently turn the master copy into a cache node; the alias
119 * exists precisely so the intent is visible and enforced at nginx -t.
120 *
121 * HOW: Runs before the posix-root rewrite so conf->common.root feeds the
122 * normal brix_prepare_export_root path unchanged. No-op when unset.
123 */
124static char *
125cvmfs_stratum0_apply(ngx_conf_t *cf, ngx_http_brix_cvmfs_loc_conf_t *conf)
126{
127 if (conf->cvmfs.stratum0_root.len == 0) {
128 return NGX_CONF_OK;
129 }
130
131 if (conf->common.root.len > 0) {
132 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
133 "brix_cvmfs_stratum0_root and brix_export both name an export "
134 "root - the alias replaces brix_export; configure exactly one");
135 return NGX_CONF_ERROR;
136 }
137
138 if (conf->common.storage_backend.len >= 4
139 && ngx_strncasecmp(conf->common.storage_backend.data,
140 (u_char *) "http", 4) == 0)
141 {
142 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
143 "brix_cvmfs_stratum0_root: a Stratum-0 has no upstream - remove "
144 "the http(s) brix_storage_backend from this block");
145 return NGX_CONF_ERROR;
146 }
147
148 if (conf->common.cache_store.len > 0) {
149 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
150 "brix_cvmfs_stratum0_root: a Stratum-0 serves its published "
151 "tree directly - remove brix_cache_store (cache-fill) from "
152 "this block");
153 return NGX_CONF_ERROR;
154 }
155
156 if (conf->cvmfs.upstream_allow != NULL) {
157 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
158 "brix_cvmfs_stratum0_root: a Stratum-0 has no upstream - remove "
159 "brix_cvmfs_upstream_allow (proxy mode) from this block");
160 return NGX_CONF_ERROR;
161 }
162
163 conf->common.root = conf->cvmfs.stratum0_root;
164 return NGX_CONF_OK;
165}

167/*
168 * cvmfs_merge_cache() — the cvmfs-enabled export/backend/cache build block.
169 *
170 * WHAT: When enable=1: reject unsupported storage grammar, force read-only,
171 * rewrite the posix backend, anchor the export root (defaulting to "/"),
172 * open the confinement rootfd, register the composable http backend, stamp
173 * the cache TTL/deadline knobs, compose the cache/stage tiers, then apply the
174 * T19 origin-selection (geo ranks / rtt probe registration) and the
175 * coords-without-geo warning.
176 *
177 * WHY: This is the one part of the merge that mutates persistent export state
178 * and registers backends; it only runs for cvmfs-enabled locations. Splitting
179 * it out lets the orchestrator gate it with a single early return while the
180 * build steps stay in their exact original order.
181 *
182 * HOW: Verbatim move of the `if (conf->cvmfs.enable)` block; returns
183 * NGX_CONF_ERROR on any step failure (identical EMERG/rejection lines), else
184 * NGX_CONF_OK. The caller only invokes it when enable is set.
185 */
186char *
187cvmfs_merge_cache(ngx_conf_t *cf, ngx_http_brix_cvmfs_loc_conf_t *conf)
188{
189 brix_export_root_opts_t root_opts;
190
191 /* Reject storage grammar cvmfs cannot honour before doing anything
192 * else: staging, slicing, and explicit writes make no sense for a
193 * read-only content-addressed site cache. */
194 if (brix_cvmfs_reject_unsupported(cf, conf) != NGX_CONF_OK) {
195 return NGX_CONF_ERROR;
196 }
197
198 /* Phase-96 S13: Stratum-0 alias — refuse cache-fill upstream grammar,
199 * then anchor the export root at the published tree. */
200 if (cvmfs_stratum0_apply(cf, conf) != NGX_CONF_OK) {
201 return NGX_CONF_ERROR;
202 }
203
204 /* CVMFS is a read-only protocol: no directive can enable writes. */
205 conf->common.allow_write = 0;
206
207 /* "posix:<path>" backend names the local export tree (composable
208 * brix_root replacement) — same rewrite every protocol applies. */
209 brix_storage_backend_posix_root(&conf->common);
210
211 /* Pure cache node (the normal CVMFS shape): no local export tree —
212 * anchor the namespace at "/" exactly like the stream plane's pure
213 * cache node; the location serves the "/" namespace, filling from
214 * the http backend into the cache store. */
215 if (conf->common.root.len == 0) {
216 ngx_str_set(&conf->common.root, "/");
217 }
218
219 root_opts.directive_name = "brix_cvmfs";
220 root_opts.allow_write = 0;
221 root_opts.required = 1;
222 root_opts.canon_size = sizeof(conf->common.root_canon);
223 if (brix_prepare_export_root(cf, &conf->common.root, &root_opts,
224 conf->common.root_canon) != NGX_CONF_OK)
225 {
226 return NGX_CONF_ERROR;
227 }
228
229 /* Persistent confinement rootfd (openat2 RESOLVE_BENEATH anchor). */
230 if (brix_http_open_rootfd(cf, &conf->common) != NGX_CONF_OK) {
231 return NGX_CONF_ERROR;
232 }
233
234 /* Register the composable storage backend (phase-63): the http(s)
235 * Stratum-1 origin URL routes every VFS op to sd_http. */
236 if (brix_vfs_backend_config_str(cf, conf->common.root_canon,
237 &conf->common.storage_backend, conf->common.pblock_block_size,
238 BRIX_AF_AUTO)
239 != NGX_OK)
240 {
241 return NGX_CONF_ERROR;
242 }
243
244 /* Verify-mismatch evidence lands in the protocol's quarantine dir;
245 * MANIFEST-class fills get the protocol's TTL stamped (T12). */
246 conf->common.cache_quarantine_dir = conf->cvmfs.quarantine_dir;
247 conf->common.cache_manifest_ttl = conf->cvmfs.manifest_ttl;
248 conf->common.cache_offline_ttl = conf->cvmfs.offline_ttl;
249 /* phase-85 F1: the master-key PATH rides the common preamble; the tier
250 * registration loads the PEM once at config time into the cache policy. */
251 conf->common.cache_cvmfs_master_key = conf->cvmfs.master_key;
252 /* T20 never-drop deadlines for the shared fill machinery */
253 conf->common.cache_client_hold = conf->cvmfs.client_hold;
254 conf->common.cache_fill_max_life = conf->cvmfs.fill_max_life;
255
256 /* Phase-64: compose the cache/stage tiers over the backend. */
257 if (brix_tier_register_stores(cf, &conf->common) != NGX_OK) {
258 return NGX_CONF_ERROR;
259 }
260
261 /* T19 origin selection: geo ranks compute once at config time;
262 * rtt registers the per-worker probe; static keeps configured
263 * order (all ranks 0 — the pick is order-stable on ties). */
264 if (conf->cvmfs.origin_select == BRIX_CVMFS_SELECT_GEO) {
265 if (cvmfs_geo_rank_config(cf, conf) != NGX_CONF_OK) {
266 return NGX_CONF_ERROR;
267 }
268 } else if (conf->cvmfs.origin_select == BRIX_CVMFS_SELECT_RTT) {
269 brix_cvmfs_rtt_register(conf->common.root_canon,
270 conf->cvmfs.rtt_interval,
271 &conf->common.thread_pool_name);
272 }
273
274 return cvmfs_merge_services(cf, conf);
275}

277/* The phase-87 background-service registrations (scrub / learn / swarm):
278 * each rides the RTT-probe lifecycle — registered here at config time, the
279 * worker-0 timer / per-worker state arms at init_process. */
280static char *
281cvmfs_merge_services(ngx_conf_t *cf, ngx_http_brix_cvmfs_loc_conf_t *conf)
282{
283 /* Phase-87 G17: register the background CAS scrub for this export; the
284 * worker-0 timer arms at init_process (scrub.c, RTT-probe lifecycle). */
285 if (conf->cvmfs.scrub) {
286 brix_cvmfs_scrub_register(conf->common.root_canon,
287 conf->cvmfs.scrub_interval,
288 conf->cvmfs.scrub_rate,
289 &conf->common.thread_pool_name);
290 }
291
292 /* Phase-87 G11: register the predictive-prewarm learner for this export;
293 * the per-worker model + fill task build at init_process (learn.c). */
294 if (conf->cvmfs.learn) {
295 brix_cvmfs_learn_register(conf->common.root_canon,
296 &conf->common.thread_pool_name);
297 }
298
299 /* Phase-87 G12: the swarm extends the F8 static mesh — it needs the
300 * brix_cache_peers seed ring to know who this node IS. Fail loudly at
301 * config time instead of gossiping into the void. */
302 if (conf->cvmfs.swarm) {
303 if (conf->common.cache_peers == NULL) {
304 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
305 "brix_cvmfs_swarm requires brix_cache_peers (the seed "
306 "ring naming this node's own \"self=\" slot)");
307 return NGX_CONF_ERROR;
308 }
309 brix_cvmfs_swarm_register(conf->common.root_canon,
310 conf->cvmfs.swarm_interval,
311 &conf->common.thread_pool_name);
312 }
313
314 /* WARN (not NOTICE — config-parse NOTICE is dropped at cf->log ERR
315 * level) when coordinates were supplied but origin_select is not geo:
316 * the coordinates are silently ignored in all other modes and the
317 * operator may not realise their geo config has no effect. */
318 if (conf->cvmfs.origin_select != BRIX_CVMFS_SELECT_GEO
319 && conf->cvmfs.origin_coords != NULL)
320 {
321 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
322 "brix_cvmfs_origin_coords set but brix_cvmfs_origin_select "
323 "is not geo — coordinates are ignored");
324 }
325 return NGX_CONF_OK;
326}
