114/* /proc/self/fd names every open descriptor: close the numeric ones >= lowfd
115 * (skipping the directory stream's own fd). 0 when the scan ran, -1 if /proc
116 * is unavailable so the caller falls back to a bounded loop. */
117static int
118close_from_procfs(int lowfd)
119{
120 DIR *dp = opendir("/proc/self/fd"); /* vfs-seam-allow: NOT_STORAGE - PAL fd hygiene before exec */
121 struct dirent *de;
122 int self;
123
124 if (dp == NULL) {
125 return -1;
126 }
127 self = dirfd(dp);
128 while ((de = readdir(dp)) != NULL) { /* vfs-seam-allow: NOT_STORAGE - PAL fd hygiene before exec */
129 int fd = (int) strtol(de->d_name, NULL, 10);
130
131 if (fd >= lowfd && fd != self && de->d_name[0] >= '0'
132 && de->d_name[0] <= '9')
133 {
134 close(fd);
135 }
136 }
137 closedir(dp);
138 return 0;
139}

141int
142brix_plat_close_from(int lowfd)
143{
144 long maxfd;
145 int fd;
146
147#if defined(SYS_close_range)
148 if (syscall(SYS_close_range, (unsigned int) lowfd, ~0U, 0) == 0) {
149 return 0;
150 }
151#endif
152 if (close_from_procfs(lowfd) == 0) {
153 return 0;
154 }
155 maxfd = sysconf(_SC_OPEN_MAX);
156 if (maxfd < 0) {
157 maxfd = 4096;
158 }
159 for (fd = lowfd; fd < maxfd; fd++) {
160 close(fd);
161 }
162 return 0;
163}

180int
181brix_plat_self_exe(char *buf, size_t cap)
182{
183 ssize_t n;
184
185 if (cap == 0) {
186 errno = ENAMETOOLONG;
187 return -1;
188 }
189 n = readlink("/proc/self/exe", buf, cap - 1); /* vfs-seam-allow: SEAM_CORRECT - PAL body reading procfs, not storage */
190 if (n < 0) {
191 return -1;
192 }
193 if ((size_t) n >= cap - 1) {
194 errno = ENAMETOOLONG; /* truncated: a partial path is no path */
195 return -1;
196 }
197 buf[n] = '\0';
198 return 0;
199}

201int
202brix_plat_boot_id(char *buf, size_t cap)
203{
204 char line[64]; /* a 36-char UUID and its newline, with room to spare */
205 int fd;
206 ssize_t n;
207 size_t len;
208
209 if (cap == 0) {
210 errno = ENAMETOOLONG;
211 return -1;
212 }
213 fd = open("/proc/sys/kernel/random/boot_id", O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: SEAM_CORRECT - PAL body reading procfs, not storage */
214 if (fd < 0) {
215 return -1;
216 }
217 /* Read the id whole, into a local, and only then check that it fits.
218 * Reading cap-1 bytes straight into `buf` handed the caller a PREFIX of
219 * the boot id and called it success; two boots sharing that prefix then
220 * compare equal, so brixcvmfs would read a stale transaction lock as live.
221 * brix_plat_self_exe above already refuses a truncated answer and Darwin's
222 * sysctl form fails ENOMEM on a short buffer — this is the same rule. */
223 n = read(fd, line, sizeof(line) - 1); /* vfs-seam-allow: SEAM_CORRECT - PAL body reading procfs, not storage */
224 close(fd);
225 if (n < 0) {
226 return -1;
227 }
228 line[n] = '\0';
229 line[strcspn(line, "\n")] = '\0';
230 len = strlen(line);
231 if (len == 0) {
232 errno = ENODATA;
233 return -1;
234 }
235 if (len >= cap) {
236 errno = ENAMETOOLONG; /* truncated: a partial boot id is no boot id */
237 return -1;
238 }
239 memcpy(buf, line, len + 1);
240 return 0;
241}

245/* pidfd_open(2) (Linux >= 5.3, same number on every architecture) gives a
246 * descriptor that becomes readable when the process exits, so one poll(2)
247 * enforces the deadline without SIGALRM and without a waitpid(WNOHANG) spin.
248 * A kernel without it answers "exited", degrading to an unbounded wait. */
249#ifndef __NR_pidfd_open
250#define __NR_pidfd_open 434
251#endif

253int
254brix_plat_wait_pid_timeout(pid_t pid, unsigned timeout_ms)
255{
256 struct pollfd pfd;
257 int pidfd = (int) syscall(__NR_pidfd_open, pid, 0U);
258 int r;
259
260 if (pidfd < 0) {
261 return 0;
262 }
263 pfd.fd = pidfd;
264 pfd.events = POLLIN;
265 while ((r = poll(&pfd, 1, (int) timeout_ms)) < 0 && errno == EINTR) {
266 /* the caller installs no handlers: restart the window */
267 }
268 (void) close(pidfd);
269 return r == 0;
270}
