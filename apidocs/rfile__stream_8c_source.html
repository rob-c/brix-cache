35brix_rfile_pump(brix_rfile *rf, int64_t offset, int64_t limit,
36 size_t chunk_size, brix_rfile_sink_fn sink, void *arg,
37 int64_t *moved, brix_status *st)
38{
39 uint8_t *buffer;
40 int64_t total = 0;
41 int rc = 0;
42
43 if (moved != NULL) {
44 *moved = 0;
45 }
46 if (!rfile_arguments_valid(rf, offset, limit, sink, st)) {
47 return -1;
48 }
49 if (chunk_size == 0) {
50 chunk_size = BRIX_RFILE_CHUNK_DEFAULT;
51 }
52 buffer = malloc(chunk_size);
53 if (buffer == NULL) {
54 brix_status_set(st, XRDC_EIO, ENOMEM, "resilient-file stream: out of memory");
55 return -1;
56 }
57 while (limit < 0 || total < limit) {
58 size_t want = chunk_size;
59 ssize_t got;
60 int sink_rc;
61
62 if (limit >= 0 && (int64_t) want > limit - total) {
63 want = (size_t) (limit - total);
64 }
65 if (want == 0) {
66 break;
67 }
68 got = brix_rfile_pread(rf, offset + total, buffer, want, st);
69 if (got < 0) {
70 rc = -1;
71 break;
72 }
73 if (got == 0) {
74 break;
75 }
76 sink_rc = sink(buffer, (size_t) got, offset + total, arg, st);
77 total += got;
78 if (sink_rc != 0) {
79 rc = sink_rc < 0 ? -1 : 0;
80 break;
81 }
82 }
83 free(buffer);
84 if (moved != NULL) {
85 *moved = total;
86 }
87 return rc;
88}

91rfile_fd_sink(const uint8_t *data, size_t len, int64_t offset, void *arg,
92 brix_status *st)
93{
94 rfile_fd_sink_t *sink = arg;
95 size_t written = 0;
96
97 (void) offset;
98 while (written < len) {
99 ssize_t n = write(sink->fd, data + written, len - written);
100 if (n < 0 && errno == EINTR) {
101 continue;
102 }
103 if (n <= 0) {
104 int cause = n < 0 ? errno : EIO;
105 brix_status_set(st, XRDC_EIO, cause,

107 return -1;
108 }
109 written += (size_t) n;
110 }

168int
169brix_rfile_slurp(brix_conn *c, const char *path, const char *opaque,
170 int64_t max_bytes, uint8_t **out, int64_t *len,
171 brix_status *st)
172{
173 brix_rfile file;
174 rfile_memory_sink_t sink;
175 size_t size;
176 int rc;
177
178 if (out == NULL || len == NULL) {
179 brix_status_set(st, XRDC_EUSAGE, EINVAL, "missing resilient-file slurp output");
180 return -1;
181 }
182 *out = NULL;
183 *len = 0;
184 if (rfile_slurp_size(c, path, max_bytes, &size, st) != 0) {
185 return -1;
186 }
187 sink.data = malloc(size == 0 ? 1 : size);
188 if (sink.data == NULL) {
189 brix_status_set(st, XRDC_EIO, ENOMEM, "resilient-file slurp: out of memory");
190 return -1;
191 }
192 sink.used = 0;
193 sink.capacity = size;
194 if (brix_rfile_open_read(c, path, opaque, 0, -1, &file, st) != 0) {
195 free(sink.data);
196 return -1;
197 }
198 rc = brix_rfile_pump(&file, 0, (int64_t) size, 0,
199 rfile_memory_sink, &sink, NULL, st);
200 {
201 brix_status ignored;
202 brix_status_clear(&ignored);
203 if (brix_rfile_close(&file, rc == 0 ? st : &ignored) != 0 && rc == 0) {
204 rc = -1;
205 }
206 }
207 if (rc != 0) {
208 free(sink.data);
209 return -1;
210 }
211 *out = sink.data;
212 *len = (int64_t) sink.used;
213 return 0;
214}
