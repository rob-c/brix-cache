34int
35brix_rfile_pump(brix_rfile *rf, int64_t offset, int64_t limit,
36 size_t chunk_size, brix_rfile_sink_fn sink, void *arg,
37 int64_t *moved, brix_status *st)
38{
39 uint8_t *buffer;
40 int64_t total = 0;
41 int rc = 0;
42
43 if (moved != NULL) {
44 *moved = 0;
45 }
46 if (!rfile_arguments_valid(rf, offset, limit, sink, st)) {
47 return -1;
48 }
49 if (chunk_size == 0) {
50 chunk_size = BRIX_RFILE_CHUNK_DEFAULT;
51 }
52 buffer = malloc(chunk_size);
53 if (buffer == NULL) {
54 brix_status_set(st, XRDC_EIO, ENOMEM, "resilient-file stream: out of memory");
55 return -1;
56 }
57 while (limit < 0 || total < limit) {
58 size_t want = chunk_size;
59 ssize_t got;
60 int sink_rc;
61
62 if (limit >= 0 && (int64_t) want > limit - total) {
63 want = (size_t) (limit - total);
64 }
65 if (want == 0) {
66 break;
67 }
68 got = brix_rfile_pread(rf, offset + total, buffer, want, st);
69 if (got < 0) {
70 rc = -1;
71 break;
72 }
73 if (got == 0) {
74 break;
75 }
76 sink_rc = sink(buffer, (size_t) got, offset + total, arg, st);
77 total += got;
78 if (sink_rc != 0) {
79 rc = sink_rc < 0 ? -1 : 0;
80 break;
81 }
82 }
83 free(buffer);
84 if (moved != NULL) {
85 *moved = total;
86 }
87 return rc;
88}

90static int
91rfile_fd_sink(const uint8_t *data, size_t len, int64_t offset, void *arg,
92 brix_status *st)
93{
94 rfile_fd_sink_t *sink = arg;
95 size_t written = 0;
96
97 (void) offset;
98 while (written < len) {
99 ssize_t n = write(sink->fd, data + written, len - written);
100 if (n < 0 && errno == EINTR) {
101 continue;
102 }
103 if (n <= 0) {
104 int cause = n < 0 ? errno : EIO;
105 brix_status_set(st, XRDC_EIO, cause,
106 "local write: %s", strerror(cause));
107 return -1;
108 }
109 written += (size_t) n;
110 }
111 return 0;
112}

114/* ---- Bytes of a bounded range still owed after `done` were delivered ----
115 *
116 * WHAT: Returns the limit for the remainder of a range: -1 stays -1 (EOF-driven,
117 * still unbounded), otherwise limit - done clamped at 0.
118 *
119 * WHY: The fast path may stop part-way through a bounded range and hand the
120 * rest to the serial pump; getting the residual limit wrong would either
121 * truncate the stream or read past the caller's window, so the arithmetic
122 * is named once instead of being inlined at the resume site.
123 *
124 * HOW: Branch on the unbounded sentinel, else subtract with a floor of 0.
125 */
126static int64_t
127rfile_limit_left(int64_t limit, int64_t done)
128{
129 if (limit < 0) {
130 return -1;
131 }
132 return (done >= limit) ? 0 : limit - done;
133}

135/* ---- Stream a range: pipelined first, serial for whatever is left ----
136 *
137 * WHAT: Streams [offset, offset+limit) (limit < 0 = to EOF) of `rf` to `sink`,
138 * filling *moved with the bytes delivered. 0 / -1 (st set).
139 *
140 * WHY: This is the bulk-transfer entry point, so it is where the pipelined
141 * reader earns its keep: the serial pump leaves exactly one kXR_read on
142 * the wire and idles a round trip per chunk. The fast path is a pure
143 * accelerator — anything it cannot finish is completed by the serial pump,
144 * which owns the reconnect/reopen retry policy — so no reliability
145 * behaviour depends on it succeeding.
146 *
147 * HOW: 1. Try brix_rfile_stream_fast; a 0 return means the range is done.
148 * 2. Otherwise clear the status (the fast path's failure is not the
149 * caller's verdict) and run the serial pump over what is left, then
150 * fold the fast path's bytes back into *moved.
151 */
152int
153brix_rfile_stream(brix_rfile *rf, int64_t offset, int64_t limit,
154 size_t chunk_size, brix_rfile_sink_fn sink, void *arg,
155 int64_t *moved, brix_status *st)
156{
157 int64_t fast_moved = 0;
158 int64_t left;
159 int rc;
160
161 rc = brix_rfile_stream_fast(rf, offset, limit, sink, arg, &fast_moved, st);
162 if (rc == 0) {
163 if (moved != NULL) {
164 *moved = fast_moved;
165 }
166 return 0;
167 }
168 left = rfile_limit_left(limit, fast_moved);
169 if (rc != BRIX_RFILE_FAST_OFF && left == 0) {
170 return -1; /* bounded range already covered: the failure IS the verdict */
171 }
172 brix_status_clear(st);
173 rc = brix_rfile_pump(rf, offset + fast_moved, left, chunk_size,
174 sink, arg, moved, st);
175 if (moved != NULL) {
176 *moved += fast_moved;
177 }
178 return rc;
179}

181/* ---- Drain a range straight to a file descriptor ----
182 *
183 * WHAT: brix_rfile_stream with the built-in write(2) sink. 0 / -1 (st set).
184 *
185 * WHY: xrdfs cat / tail's initial dump want exactly this and should not each
186 * carry their own write loop.
187 *
188 * HOW: Reject a bad descriptor, then delegate with rfile_fd_sink.
189 */
190int
191brix_rfile_drain_to_fd(brix_rfile *rf, int64_t offset, int64_t limit,
192 size_t chunk_size, int fd, int64_t *moved,
193 brix_status *st)
194{
195 rfile_fd_sink_t sink = {fd};
196
197 if (fd < 0) {
198 brix_status_set(st, XRDC_EUSAGE, EBADF, "invalid output descriptor");
199 return -1;
200 }
201 return brix_rfile_stream(rf, offset, limit, chunk_size, rfile_fd_sink,
202 &sink, moved, st);

206rfile_memory_sink(const uint8_t *data, size_t len, int64_t offset, void *arg,
207 brix_status *st)
208{
209 rfile_memory_sink_t *sink = arg;
210
211 (void) offset;
212 if (len > sink->capacity - sink->used) {
213 brix_status_set(st, XRDC_EPROTO, EOVERFLOW,
214 "remote file exceeded its declared size");
215 return -1;
216 }
217 memcpy(sink->data + sink->used, data, len);

219 return 0;

244int
245brix_rfile_slurp(brix_conn *c, const char *path, const char *opaque,
246 int64_t max_bytes, uint8_t **out, int64_t *len,
247 brix_status *st)
248{
249 brix_rfile file;
250 rfile_memory_sink_t sink;
251 size_t size;
252 int rc;
253
254 if (out == NULL || len == NULL) {
255 brix_status_set(st, XRDC_EUSAGE, EINVAL, "missing resilient-file slurp output");
256 return -1;
257 }
258 *out = NULL;
259 *len = 0;
260 if (rfile_slurp_size(c, path, max_bytes, &size, st) != 0) {
261 return -1;
262 }
263 sink.data = malloc(size == 0 ? 1 : size);
264 if (sink.data == NULL) {
265 brix_status_set(st, XRDC_EIO, ENOMEM, "resilient-file slurp: out of memory");
266 return -1;
267 }
268 sink.used = 0;
269 sink.capacity = size;
270 if (brix_rfile_open_read(c, path, opaque, 0, -1, &file, st) != 0) {
271 free(sink.data);
272 return -1;
273 }
274 rc = brix_rfile_pump(&file, 0, (int64_t) size, 0,
275 rfile_memory_sink, &sink, NULL, st);
276 {
277 brix_status ignored;
278 brix_status_clear(&ignored);
279 if (brix_rfile_close(&file, rc == 0 ? st : &ignored) != 0 && rc == 0) {
280 rc = -1;
281 }
282 }
283 if (rc != 0) {
284 free(sink.data);
285 return -1;
286 }
287 *out = sink.data;
288 *len = (int64_t) sink.used;
289 return 0;
290}
