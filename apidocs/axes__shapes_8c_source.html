6static int
7pairwise_selected(const size_t *at, const size_t *sizes, size_t axes)
8{
9 size_t left;
10 size_t right;
11
12 if (axes < 2) return 1;
13 for (left = 0; left < axes; left++) {
14 for (right = left + 1; right < axes; right++) {
15 size_t axis;
16 int matches = 1;
17
18 for (axis = 0; axis < axes; axis++) {
19 size_t expected;
20
21 if (axis == left || axis == right) continue;
22 expected = (at[left] + at[right] + axis) % sizes[axis];
23 if (at[axis] != expected) {
24 matches = 0;
25 break;
26 }
27 }
28 if (matches) return 1;
29 }
30 }
31 return 0;
32}

35brix_conform_product_selected(brix_conform_budget budget, const size_t *at,
36 const size_t *sizes, size_t axes)

38 size_t axis;
39
40 if (budget == BRIX_CONFORM_EXHAUSTIVE) return 1;

42 return pairwise_selected(at, sizes, axes);
