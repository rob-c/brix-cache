1/*
2 * brix_fault_proxy.c — brix-fault-proxy: a root-free TCP fault-injection proxy.
3 *
4 * WHAT: Relays TCP from a local listen port to an upstream host:port and injects
5 * network faults on demand — latency, jitter, partial writes, slow-drip,
6 * bandwidth caps, packet-loss (resets), payload corruption, duplication,
7 * deterministic mid-transfer truncation, connection black-holes, and full
8 * outages — set at startup on the command line or live over a control port
9 * while traffic flows. A "toxiproxy-lite" you can splice in front of any
10 * brix (or stock XRootD/HTTP) endpoint to prove a client survives a bad or
11 * actively hostile network, WITHOUT needing root: tc/netem needs
12 * CAP_NET_ADMIN; this is pure userspace sockets.
13 *
14 * DIRECTIONALITY: every byte-level lever is per-direction. `up` = client ->
15 * upstream (the request/upload path), `down` = upstream -> client (the
16 * response/download path). A control command with no direction token
17 * applies to BOTH; append ` up`, ` down`, or ` both` to target one.
18 *
19 * FAULT LEVERS (each off/0 by default; set at startup via the matching --flag or
20 * live by writing the command to the control port):
21 * latency <ms> [dir] delay every forwarded chunk (a high-RTT link)
22 * jitter <ms> [dir] delay each chunk by a uniform-random 0..<ms> ms —
23 * the application signature of IP reordering (see NOTE)
24 * chunk <bytes> [dir] split each write into <=N-byte segments (egress
25 * TCP_NODELAY) so the peer sees genuine partial reads
26 * drip <bytes> <ms> [dir] forward <bytes>, sleep <ms>, repeat (slow stream)
27 * rate <KB/s> [dir] pace the stream to a bandwidth ceiling (slow link)
28 * lossy <pct> [dir] with prob <pct>%/chunk, sever the stream (loss=reset)
29 * reorder <pct> [ms] [dir] with prob <pct>%/chunk, hold that chunk back <ms> ms
30 * corrupt <pct> [dir] flip a random bit in <pct>% of forwarded BYTES — a
31 * MITM tamper that a checksum-verifying peer must catch
32 * dup <pct> [dir] with prob <pct>%/chunk, deliver that chunk twice
33 * truncate-at <bytes> [dir] sever each connection once <bytes> have flowed in
34 * that direction (deterministic mid-transfer cut)
35 * fail-nth <n> fail exactly the Nth accepted connection, pass others
36 * heal-after <ms> auto-clear every lever after <ms> (a transient fault)
37 * one-shot self-heal (clear) the moment a sever/truncate fires
38 * drop graceful-FIN sever every live connection now
39 * reset abortive-RST sever every live connection now, and
40 * make subsequent lossy/truncate severs use RST too
41 * abortive <0|1> select RST (1) vs graceful FIN (0) for auto-severs
42 * half-close FIN the client->upstream path on live conns while the
43 * upstream->client path keeps flowing (tests half-open)
44 * hang accept new connections but never relay (a black hole)
45 * unhang resume relaying newly accepted connections
46 * block drop live conns AND refuse new ones (an outage window)
47 * unblock resume accepting
48 * clear reset every lever (levers=0, unblock, unhang)
49 * status report lever state and traffic/fault counters
50 *
51 * WHY: A client's resilience (reconnect + file-handle resumption, per-PDU read
52 * deadlines, low-speed aborts, checksum-verify-and-retry) needs the exact
53 * conditions of "bad wifi from a laptop abroad" — or an outright tampering
54 * MITM — injected deterministically and without privilege. The peer
55 * connects THROUGH this proxy; an operator (or a test) pulls the levers.
56 *
57 * HOW: Thread-per-connection relay (so per-chunk latency is a simple usleep); a
58 * global drop-epoch counter that, when bumped, makes every live relay thread
59 * sever and exit; a control thread parsing newline commands; an optional
60 * timeline thread replaying a scripted scenario. All randomness is per-thread
61 * rand_r() seeded from --seed for reproducibility.
62 *
63 * SECURITY: the control port is UNAUTHENTICATED — anyone who can reach it can pull
64 * the fault levers and reset live connections. Both the listen and control
65 * sockets therefore bind to loopback by DEFAULT; widening the bind address
66 * (--bind) exposes the control plane and is refused unless you also pass
67 * --insecure-bind, which prints a warning. Never expose the control port to
68 * an untrusted network.
69 *
70 * NOTE: packet LOSS is intentionally modelled as a connection sever, not byte
71 * drops: dropping bytes from an already-ACKed TCP stream corrupts it rather
72 * than emulating loss (which lives below TCP). Likewise true out-of-order
73 * PACKET delivery cannot be emulated by re-ordering bytes — TCP reassembles
74 * in order below us — so `jitter`/`reorder` inject the variable latency that
75 * IP reordering actually imposes on a TCP application. (True packet
76 * reordering needs `tc qdisc ... netem`, i.e. CAP_NET_ADMIN, which this
77 * avoids.) `corrupt`, by contrast, IS a real in-band byte mutation: it is
78 * what an active on-path attacker (or a flaky NIC past the TCP checksum)
79 * does, and it is precisely what an application-layer checksum exists to
80 * catch.
81 */

212/* Accept clients for `route` on `lfd`, spawning a detached relay thread per
213 * connection (subject to the outage/connection-cap levers). Returns when the
214 * route is stopped (fp_route_alive(route) == 0) or on a non-EINTR accept error.
215 * poll() with a short timeout makes the loop responsive to a `route del` without
216 * blocking indefinitely in accept(). */
217void
218fp_accept_loop(fp_route *route, int lfd)
219{
220 fp_event_set_route(fp_route_name(route)); /* tag refuse events on this thread */
221 while (fp_route_alive(route)) {
222 /* accept-pause: delay servicing the accept queue so pending SYNs / the
223 * accept backlog pile up (connect-timeout + backlog-overflow testing). */
224 if (g_accept_pause_ms > 0) {
225 usleep((useconds_t) g_accept_pause_ms * 1000);
226 }
227 struct pollfd lp = { lfd, POLLIN, 0 };
228 int pr = poll(&lp, 1, 200);
229 if (pr <= 0) {
230 continue; /* timeout (re-check alive) or EINTR */
231 }
232 int client = accept(lfd, NULL, NULL);
233 if (client < 0) {
234 if (errno == EINTR) {
235 continue;
236 }
237 break;
238 }
239 if (g_blocked) {
240 brix_fp_event(CBUMP(refused, 1), NULL, "refuse", "block", NULL, 0);
241 close(client); /* outage: refuse */
242 continue;
243 }
244 /* syn-drop: silently drop a fraction of accepted clients (no RST) so the
245 * connection looks like it never happened — connect-timeout under load. */
246 if (g_syn_drop_ppm > 0) {
247 static unsigned syn_seed = 0;
248 if (syn_seed == 0) { syn_seed = g_seed ? g_seed : 0x9e3779b9u; }
249 if ((unsigned) (rand_r(&syn_seed) % 1000000) < (unsigned) g_syn_drop_ppm) {
250 CBUMP(syn_dropped, 1);
251 close(client);
252 continue;
253 }
254 }
255 if (g_max_conns > 0
256 && __atomic_load_n(&C.active, __ATOMIC_RELAXED) >= (unsigned long) g_max_conns) {
257 brix_fp_event(CBUMP(refused, 1), NULL, "refuse", "max-conns", NULL, 0);
258 close(client); /* connection cap reached */
259 continue;
260 }
261 int one = 1;
262 setsockopt(client, IPPROTO_TCP, TCP_NODELAY, &one, sizeof(one));
263 relay_arg *ra = malloc(sizeof(*ra));
264 ra->client_fd = client;
265 ra->epoch = __atomic_load_n(&g_drop_epoch, __ATOMIC_SEQ_CST);
266 ra->conn_id = CBUMP(conns, 1);
267 ra->route = route;
268 CBUMP(active, 1);
269 fp_route_inc_conns(route);
270 pthread_t t;
271 if (pthread_create(&t, NULL, relay_thread, ra) != 0) {
272 close(client);
273 CDEC(active);
274 free(ra);
275 continue;
276 }
277 pthread_detach(t);
278 }
279}

294/* On SIGINT/SIGTERM, restore any host network state the privileged subsystem
295 * installed (qdisc / nft table / MTU) before exiting — otherwise a Ctrl-C would

297void

300 fp_priv_teardown();

304/* Arm the privileged subsystem if requested, wiring in the teardown hooks. On a
305 * hard failure (e.g. --privileged without root) prints the reason and returns
306 * FP_USAGE; returns FP_CONTINUE otherwise. */
307int
308fp_arm_privileged(const fp_config *cfg)
309{
310 if (!cfg->privileged) {
311 if (cfg->priv_iface != NULL) {
312 fprintf(stderr, "brix-fault-proxy: --priv-iface requires "
313 "--privileged\n");
314 return FP_USAGE;
315 }
316 return FP_CONTINUE;
317 }
318 int ports[FP_MAX_TARGETS];
319 for (int i = 0; i < g_ntargets; i++) {
320 ports[i] = g_targets[i].port;
321 }
322 const char *err = NULL;
323 if (fp_priv_enable(cfg->priv_iface, cfg->listen_port, ports, g_ntargets,
324 &err) != 0) {
325 fprintf(stderr, "brix-fault-proxy: --privileged: %s\n", err);
326 return FP_USAGE;
327 }
328 atexit(fp_priv_teardown);
329 signal(SIGINT, fp_on_signal);
330 signal(SIGTERM, fp_on_signal);
331 if (!cfg->quiet) {
332 fprintf(stderr, "brix-fault-proxy: privileged levers ARMED (iface=%s) — "
333 "host network state will be restored on exit\n",
334 cfg->priv_iface ? cfg->priv_iface : "(none)");
335 }
336 return FP_CONTINUE;
337}

352/* Parse `--udp "<listen> <host:port>"` and launch the datagram relay thread
353 * (detached). Best-effort: a malformed spec warns and is ignored (TCP runs). */
354static void
355fp_spawn_udp(const char *spec)
356{
357 if (spec == NULL) {
358 return;
359 }
360 fp_udp_cfg *uc = calloc(1, sizeof(*uc));
361 char uhost[256] = "";
362 if (uc && sscanf(spec, "%d %255[^:]:%d",
363 &uc->listen_port, uhost, &uc->port) == 3) {
364 snprintf(uc->host, sizeof(uc->host), "%s", uhost);
365 pthread_t ut;
366 if (pthread_create(&ut, NULL, fp_udp_thread, uc) == 0) {
367 pthread_detach(ut);
368 return;
369 }
370 free(uc);
371 return;
372 }
373 free(uc);
374 fprintf(stderr, "brix-fault-proxy: --udp needs \"<listen> <host:port>\"\n");
375}

378main(int argc, char **argv)
379{
380 fp_config cfg = { .bind_str = "127.0.0.1" };
381 int rc;
382
383 /* `brix-fault-proxy ctl <host:port> "<cmd>"` is a control-port CLIENT, not a
384 * proxy run — dispatch it before any listener/lever setup. */
385 if (argc >= 2 && strcmp(argv[1], "ctl") == 0) {
386 return fp_ctl_main(argc, argv);
387 }
388
389 reset_lever(&g_up);
390 reset_lever(&g_down);
391 fp_tls_cfg_init(&g_tls_up); /* sentinels (-1); zero would read as active */
392 fp_tls_cfg_init(&g_tls_down);
393 clock_gettime(CLOCK_MONOTONIC, &g_t0); /* replay-timestamp reference */
394
395 if ((rc = fp_parse_args(argc, argv, &cfg)) != FP_CONTINUE) {
396 return rc;
397 }
398 if (cfg.event_log != NULL && fp_event_open(cfg.event_log) != 0) {
399 fprintf(stderr, "brix-fault-proxy: cannot open --event-log '%s'\n",
400 cfg.event_log);
401 return FP_USAGE;
402 }
403 if ((rc = fp_setup_bind(cfg.bind_str, cfg.insecure, &g_bind_ss, &g_bind_len))
404 != FP_CONTINUE) {
405 return rc;
406 }
407
408 signal(SIGPIPE, SIG_IGN);
409
410 if ((rc = fp_arm_privileged(&cfg)) != FP_CONTINUE) {
411 return rc;
412 }
413
414 /* Route plane: the startup listener is the always-on "default" route; the
415 * core provides binding + the accept/relay engine, so dynamic routes reuse
416 * the one vetted bind address. */
417 fp_route_ops ops = { fp_core_bind_listen, fp_accept_loop };
418 fp_route_init(&ops);
419 fp_route *def = fp_route_register_default(cfg.listen_port);
420 for (int i = 0; i < g_ntargets; i++) {
421 fp_route_add_target(def, g_targets[i].host, g_targets[i].port);
422 }
423
424 int lfd = listen_sa(&g_bind_ss, g_bind_len, cfg.listen_port);
425 int ctlfd = listen_sa(&g_bind_ss, g_bind_len, cfg.control_port);
426 if (lfd < 0 || ctlfd < 0) {
427 fprintf(stderr, "brix-fault-proxy: bind failed (listen=%d control=%d)\n",
428 cfg.listen_port, cfg.control_port);
429 return FP_RUN;
430 }
431
432 pthread_t ct;
433 int *cfdp = malloc(sizeof(int));
434 *cfdp = ctlfd;
435 pthread_create(&ct, NULL, control_thread, cfdp);
436
437 fp_spawn_script(cfg.script_path);
438 fp_spawn_udp(cfg.udp_spec);
439
440 if (!cfg.quiet) {
441 fp_print_banner(&cfg);
442 }
443
444 fp_accept_loop(def, lfd); /* main thread owns the default route's accept */
445 return FP_OK;
446}
