18/* --delete (mirror: make the destination match the source) and
19 * --remove-source (move: delete each source once its transfer succeeds) are
20 * contradictory. Run together they destroy BOTH trees: on an upload the
21 * per-file source unlink runs before the mirror-delete pass, which then sees
22 * the now-missing local files and purges the freshly-uploaded remote copies.
23 * Reject the pair before any bytes (or unlinks) move. --delete also requires
24 * -r and --sync: without a recursive pass there is no listing to diff
25 * against; without --sync the extra-deletion semantics are ill-defined (we
26 * might delete a destination the caller wanted to keep). */
27static int
28xrdcp_validate_delete_matrix(const brix_copy_opts *opts, int sync_mode)
29{
30 if (opts->sync_delete && opts->remove_source) {
31 fprintf(stderr, "xrdcp: --delete and --remove-source are contradictory "
32 "(mirror vs move)\n");
33 return 50;
34 }
35 if (opts->sync_delete && !(opts->recursive && sync_mode)) {
36 fprintf(stderr, "xrdcp: --delete requires -r and --sync\n");
37 return 50;
38 }
39 return 0;
40}

42/* --continue writes the destination in place and trusts the existing
43 * partial — every mode that truncates, transforms, or re-frames the byte
44 * stream contradicts it. Reject the pairs before any bytes move. */
45static int
46xrdcp_validate_continue_matrix(const brix_copy_opts *opts)
47{
48 if (!opts->cont) {
49 return 0;
50 }
51 if (opts->force) {
52 fprintf(stderr, "xrdcp: --continue and --force are contradictory "
53 "(resume vs truncate)\n");
54 return 50;
55 }
56 if (opts->pgrw || (opts->compress != NULL && opts->compress[0])) {
57 fprintf(stderr, "xrdcp: --continue cannot combine with --pgrw or "
58 "--compress (byte-offset resume needs a plain "
59 "byte stream)\n");
60 return 50;
61 }
62 if (opts->zip || opts->zip_append) {
63 fprintf(stderr, "xrdcp: --continue cannot combine with --zip "
64 "modes\n");
65 return 50;
66 }
67 return 0;
68}

70/* ========================================================================
71 * WHAT: Validate argument combinations and derive implicit settings
72 * WHY: Many flag combinations are invalid (--delete requires -r+--sync,
73 * --delete conflicts with --remove-source, --verify implies --cksum, etc.)
74 * HOW: fold --sync into force/sync, then run the delete / continue matrices
75 * and the serial-only --xrate guard
76 * RETURNS: 0 on success, 50 on usage error
77 * ======================================================================== */
78static int
79xrdcp_validate_flag_matrix(brix_copy_opts *opts, int sync_mode, int verify)
80{
81 int rc;
82
83 /* --sync replaces destinations that differ, so the files it does copy must be
84 * allowed to overwrite (skipped ones are left untouched by the size check). */
85 if (sync_mode) {
86 opts->force = 1;
87 }
88 opts->sync = sync_mode; /* recursive walkers read o->sync (+ sync_cmp/algo) */
89
90 rc = xrdcp_validate_delete_matrix(opts, sync_mode);
91 if (rc != 0) {
92 return rc;
93 }
94
95 /* --verify: post-transfer checksum against the server. An explicit --cksum wins. */
96 if (verify && opts->cksum == NULL) {
97 opts->cksum = "adler32:source";
98 }
99
100 rc = xrdcp_validate_continue_matrix(opts);
101 if (rc != 0) {
102 return rc;
103 }
104
105 /* --xrate paces the serial pump; the striped/multi-source engines bypass
106 * it entirely, so the combination would silently run unpaced. */
107 if (opts->xrate_bps > 0 && (opts->parallel || opts->sources >= 2)) {
108 fprintf(stderr, "xrdcp: --xrate applies to the serial path and cannot "
109 "combine with --parallel/--sources\n");
110 return 50;
111 }
112
113 return 0;
114}

164/*
165 * WHAT: Fold the resilience posture (--max-stall / --no-retry / $XRDC_MAX_STALL_MS)
166 * from the shared brix_opts (o->conn) into the brix_copy_opts (o->copt).
167 * WHY: Those knobs are parsed/seeded into brix_opts by brix_opts_parse_arg and
168 * brix_opts_init, but the copy pump's give-up window is read from
169 * brix_copy_opts via copy_stall_ms(). Without this bridge the documented
170 * flag/env were silently no-ops for the transfer window — a hostile-network
171 * operator who set --max-stall to bound a slow-drip stall still got the 60 s
172 * default, so a tripped-deadline read would re-handshake for a full minute.
173 * HOW: no_retry (explicit fail-fast) dominates; otherwise a positive window is
174 * copied across. conn is the sole parse target, so this is the one place
175 * the posture is mirrored — copt is never written by a flag handler directly.
176 */
177static void
178finalize_resilience_posture(xrdcp_opts_t *o)
179{
180 if (o->conn->no_retry) {
181 o->copt->no_retry = 1;
182 } else if (o->conn->max_stall_ms > 0) {
183 o->copt->max_stall_ms = o->conn->max_stall_ms;
184 }
185}

187int
188xrdcp_validate_and_finalize_args(xrdcp_opts_t *o, xrdcp_lists_t *l, const char *prog)
189{
190 static char dstbuf[XRDC_PATH_MAX];
191 int rc;
192
193 finalize_resilience_posture(o);
194
195 rc = xrdcp_validate_flag_matrix(o->copt, o->sync_mode, o->verify);
196 if (rc != 0) {
197 return rc;
198 }
199
200 /* Need a destination (the last positional) and at least one source. */
201 if (l->pos.n < 1) {
202 usage(prog);
203 return 50;
204 }
205 brix_alias_resolve(l->pos.items[l->pos.n - 1], dstbuf, sizeof(dstbuf)); /* ~/.xrdrc */
206 o->dst = dstbuf;
207
208 rc = xrdcp_collect_sources(&l->pos, o->from, &l->srcs);
209 if (rc != 0) {
210 return rc;
211 }
212
213 /* --continue vs the journal family: two different resume systems; running
214 * both would double-account progress. One or the other. */
215 if (o->copt->cont && (o->resume || o->journal_path != NULL)) {
216 fprintf(stderr, "xrdcp: --continue and --journal/--resume are "
217 "different resume modes — pick one\n");
218 return 50;
219 }
220
221 /* --resume shorthand: derive journal path from the manifest path. Must come
222 * before nsrc==0 so the specific error fires even when there are no sources. */
223 rc = xrdcp_finalize_journal(o);
224 if (rc != 0) {
225 return rc;
226 }
227 if (l->srcs.n == 0) {
228 fprintf(stderr, "xrdcp: no source given\n");
229 usage(prog);
230 return 50;
231 }
232
233 return 0;
234}
