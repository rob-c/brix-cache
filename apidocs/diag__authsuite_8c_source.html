10/* base64url-encode (no padding) into out[outsz] — to assemble synthetic JWTs for
11 * negative tests. Returns 0, or -1 if the encoded form would not fit (the caller
12 * then skips the test rather than emitting a truncated token). */
13int
14dx_b64url_enc(const unsigned char *in, size_t n, char *out, size_t outsz)
15{
16 static const char T[] =
17 "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
18 size_t i, o = 0;
19 if ((n + 2) / 3 * 4 + 1 > outsz) { /* encoded length + NUL must fit */
20 return -1;
21 }
22 for (i = 0; i + 3 <= n; i += 3) {
23 uint32_t v = ((uint32_t) in[i] << 16) | ((uint32_t) in[i+1] << 8) | in[i+2];
24 out[o++] = T[(v >> 18) & 63]; out[o++] = T[(v >> 12) & 63];
25 out[o++] = T[(v >> 6) & 63]; out[o++] = T[v & 63];
26 }
27 if (n - i == 1) {
28 uint32_t v = (uint32_t) in[i] << 16;
29 out[o++] = T[(v >> 18) & 63]; out[o++] = T[(v >> 12) & 63];
30 } else if (n - i == 2) {
31 uint32_t v = ((uint32_t) in[i] << 16) | ((uint32_t) in[i+1] << 8);
32 out[o++] = T[(v >> 18) & 63]; out[o++] = T[(v >> 12) & 63];
33 out[o++] = T[(v >> 6) & 63];
34 }
35 out[o] = '\0';
36 return 0;
37}

40/* Build "<b64url(header)>.<b64url(payload)>.<sig>" into out. 0 / -1 on overflow. */
41int
42dx_make_jwt(const char *header, const char *payload, const char *sig,
43 char *out, size_t outsz)
44{
45 char h[192], p[320]; /* sized for the short fixed probe header/payload below */
46 if (dx_b64url_enc((const unsigned char *) header, strlen(header), h, sizeof(h)) != 0
47 || dx_b64url_enc((const unsigned char *) payload, strlen(payload), p, sizeof(p)) != 0) {
48 return -1;
49 }
50 if (strlen(h) + strlen(p) + strlen(sig) + 3 > outsz) {
51 return -1;
52 }
53 snprintf(out, outsz, "%s.%s.%s", h, p, sig);
54 return 0;
55}

110/*
111 * WHAT: the scheme table for the deep-dive URL parser — one row per recognised
112 * scheme prefix with its protocol, TLS flag, and default port.
113 * WHY: replaces a per-scheme strncmp ladder; adding a scheme is one row.
114 * HOW: dx_scheme_match scans top-to-bottom, first prefix match wins — longer
115 * prefixes ("roots://") MUST precede their proper prefixes ("root://").
116 */
117typedef struct {
118 const char *prefix; /* scheme prefix including "://" */
119 size_t len; /* strlen(prefix), for strncmp + cursor advance */
120 dx_proto proto; /* protocol battery to route to */
121 int tls; /* scheme implies TLS */
122 int defport; /* default port when the URL names none */
123} dx_scheme_t;

140/*
141 * WHAT: match + consume the URL's scheme prefix against DX_SCHEMES.
142 * WHY: isolates scheme recognition so dx_url_parse stays a small pipeline.
143 * HOW: on the first prefix match, fill proto/tls/port (the scheme default)
144 * and advance *p past the prefix. 0 on match, -1 for an unknown scheme.
145 */
146static int
147dx_scheme_match(const char **p, dx_proto *proto, int *tls, int *port)
148{
149 size_t k;
150
151 for (k = 0; k < sizeof(DX_SCHEMES) / sizeof(DX_SCHEMES[0]); k++) {
152 const dx_scheme_t *s = &DX_SCHEMES[k];
153 if (strncmp(*p, s->prefix, s->len) == 0) {
154 *proto = s->proto;
155 *tls = s->tls;
156 *port = s->defport;
157 *p += s->len;
158 return 0;
159 }
160 }
161 return -1;
162}

165/*
166 * WHAT: parse a bracketed [IPv6] host literal (with optional :port) at *p.
167 * WHY: IPv6 literals embed ':' so they need their own delimiter logic.
168 * HOW: copy the bytes between the brackets into host (must be non-empty and
169 * fit hsz), advance *p past ']', and read a port if ':' follows.
170 * 0 on success, -1 on a malformed or over-long literal.
171 */
172static int
173dx_host6_parse(const char **p, char *host, size_t hsz, int *port)
174{
175 const char *rb = strchr(*p, ']');
176 size_t n;
177
178 if (rb == NULL) {
179 return -1;
180 }
181 n = (size_t) (rb - (*p + 1));
182 if (n == 0 || n >= hsz) {
183 return -1;
184 }
185 memcpy(host, *p + 1, n);
186 host[n] = '\0';
187 *p = rb + 1;
188 if (**p == ':') {
189 *port = atoi(*p + 1);
190 }
191 return 0;
192}

195/*
196 * WHAT: parse a plain host name (with optional :port) at *p, up to '/' or NUL.
197 * WHY: companion to dx_host6_parse — the non-bracketed authority form.
198 * HOW: scan to the end of the authority remembering the last ':' (port
199 * separator), copy the host bytes (non-empty, must fit hsz), read the
200 * port if present, and leave *p at the authority end. 0 / -1.
201 */
202static int
203dx_hostname_parse(const char **p, char *host, size_t hsz, int *port)
204{
205 const char *colon = NULL, *e, *hend;
206 size_t n;
207
208 for (e = *p; *e != '\0' && *e != '/'; e++) {
209 if (*e == ':') {
210 colon = e;
211 }
212 }
213 hend = colon ? colon : e;
214 n = (size_t) (hend - *p);
215 if (n == 0 || n >= hsz) {
216 return -1;
217 }
218 memcpy(host, *p, n);
219 host[n] = '\0';
220 if (colon != NULL) {
221 *port = atoi(colon + 1);
222 }
223 *p = e;
224 return 0;
225}

228/*
229 * Parse a scheme://host[:port][/path] URL for the deep-dive router. Recognizes
230 * root[s]/xroot[s], http/https, dav/davs, s3/s3s, cms. Fills *u (proto, tls,
231 * host, port — a per-scheme default if absent — and path). Returns 0, or -1 if
232 * the scheme is unknown. IPv6 literals in [..] are accepted.
233 */
234int
235dx_url_parse(const char *url, dx_url_t *u)
236{
237 const char *p = url, *slash;
238 int rc;
239
240 u->tls = 0;
241 if (dx_scheme_match(&p, &u->proto, &u->tls, &u->port) != 0) {
242 return -1;
243 }
244 rc = (*p == '[') ? dx_host6_parse(&p, u->host, sizeof(u->host), &u->port)
245 : dx_hostname_parse(&p, u->host, sizeof(u->host), &u->port);
246 if (rc != 0) {
247 return -1;
248 }
249 if (u->port <= 0 || u->port > 65535) {
250 return -1;
251 }
252 slash = strchr(p, '/');
253 snprintf(u->path, sizeof(u->path), "%s", slash ? slash : "/");
254 return 0;
255}

258/* Classify an HTTP status into an "http" finding on e. */
259void
260dx_http_status(doctor_ep *e, const char *probe, int status)
261{
262 if (status >= 200 && status < 300) {
263 dx_record(e, &(dx_note){ probe, DX_OK, status, "request succeeded", "" });
264 } else if (status == 401 || status == 403) {
265 dx_record(e, &(dx_note){ probe, DX_WARN, status,
266 "access requires authentication/authorization (401/403)",
267 "provide a credential (Bearer token / cert) if this object should be reachable" });
268 } else if (status == 404 || status == 410) {
269 dx_record(e, &(dx_note){ probe, DX_WARN, status, "object not found (404/410)",
270 "verify the path/bucket/key exists on the server" });
271 } else if (status >= 300 && status < 400) {
272 dx_record(e, &(dx_note){ probe, DX_WARN, status, "server returned a redirect (3xx)",
273 "follow the Location target; check it is intended" });
274 } else if (status >= 500) {
275 dx_record(e, &(dx_note){ probe, DX_FAIL, status, "server error (5xx) on the request",
276 "check the server logs for this operation" });
277 } else if (status == 0) {
278 dx_record(e, &(dx_note){ probe, DX_FAIL, 0, "no HTTP status parsed (malformed/partial response)",
279 "the endpoint may not be an HTTP server on this port" });
280 } else {
281 dx_record(e, &(dx_note){ probe, DX_WARN, status, "unexpected HTTP status", "" });
282 }
283}

286/* Classify an HTTP-family transport failure (connect / TLS) on e. */
287void
288dx_http_fail(doctor_ep *e, int tls, const brix_status *st)
289{
290 const char *cause = "connection setup failed";
291 const char *remedy = "check the endpoint is up and the port is correct";
292 if (tls && st->kxr == XRDC_EAUTH) {
293 cause = "TLS verification failed (cert untrusted/expired/wrong host)";
294 remedy = "fix the server certificate chain, or pass --no-verify-tls for a self-signed test endpoint";
295 } else if (st->sys_errno == ECONNREFUSED) {
296 cause = "no listener on host:port (service down or wrong port)";
297 remedy = "start the gateway / verify the port and any firewall";
298 } else if (st->sys_errno == ETIMEDOUT || st->sys_errno == EHOSTUNREACH
299 || st->sys_errno == ENETUNREACH) {
300 cause = "host/network unreachable";
301 remedy = "check routing/firewall and that the host is up";
302 }
303 doc_issue(e, DOC_RED, "%s", cause);
304 dx_record(e, &(dx_note){ tls ? "tls" : "reachability", DX_FAIL, st->kxr, cause, remedy });
305}

338/* Scalar JSON scan (flat fields; no nesting awareness — sufficient for the SRR /
339 * Tape-REST documents). Extract the string value of "key":"value" into out. 1/0. */
340int
341js_str(const char *json, const char *key, char *out, size_t osz)
342{
343 char pat[64];
344 const char *p, *v, *e;
345 if (json == NULL) { return 0; }
346 snprintf(pat, sizeof(pat), "\"%s\"", key);
347 p = strstr(json, pat);
348 if (p == NULL) { return 0; }
349 p = strchr(p + strlen(pat), ':');
350 if (p == NULL) { return 0; }
351 p++;
352 while (*p == ' ' || *p == '\t') { p++; }
353 if (*p != '"') { return 0; }
354 v = p + 1;
355 /* Find the closing quote, honouring backslash escapes: a '"' preceded by an
356 * even number of backslashes is the delimiter; an odd number means it's \". */
357 e = v;
358 while (*e != '\0') {
359 if (*e == '"') {
360 const char *bs = e;
361 int nb = 0;
362 while (bs > v && bs[-1] == '\\') { nb++; bs--; }
363 if ((nb & 1) == 0) { break; }
364 }
365 e++;
366 }
367 if (*e != '"') { return 0; }
368 {
369 size_t n = (size_t) (e - v);
370 if (n >= osz) { n = osz - 1; }
371 memcpy(out, v, n);
372 out[n] = '\0';
373 }
374 return 1;
375}

378/* Sum every numeric "key": N occurrence in the document (e.g. all shares' sizes). */
379long long
380js_sum(const char *json, const char *key)
381{
382 char pat[64];
383 const char *p;
384 long long sum = 0;
385 if (json == NULL) { return 0; }
386 snprintf(pat, sizeof(pat), "\"%s\"", key);
387 p = json;
388 while ((p = strstr(p, pat)) != NULL) {
389 const char *c = strchr(p + strlen(pat), ':');
390 p += strlen(pat);
391 if (c != NULL) {
392 const char *n = c + 1;
393 long long v;
394 while (*n == ' ' || *n == '\t') { n++; }
395 v = strtoll(n, NULL, 10);
396 /* saturating add — a diagnostic total must never wrap to nonsense. */
397 if (v > 0 && sum > LLONG_MAX - v) { sum = LLONG_MAX; }
398 else if (v < 0 && sum < LLONG_MIN - v) { sum = LLONG_MIN; }
399 else { sum += v; }
400 }
401 }
402 return sum;
403}
