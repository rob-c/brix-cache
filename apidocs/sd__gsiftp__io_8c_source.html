11/* The sink is POSITIONED (`offset` is relative to the transfer's start) because
12 * MODE E delivers absolute-offset blocks that may arrive out of order. `cap`
13 * is the caller's buffer length: an origin that frames a block past the window
14 * it was asked for would otherwise write off the end of a read buffer, so the
15 * bound is re-checked here even though gftp_mode_e.c already refuses one. */
16typedef struct {
17 uint8_t *base;
18 size_t cap;
19} sd_gsiftp_sink;
20
21/* Surface the control-channel diagnosis the session already recorded.
22 *
23 * gftp_set_error() writes a specific, actionable sentence into session->error
24 * ("MODE E block at N is outside the requested window", "data channel timed
25 * out") and every caller discarded it, keeping only errno. A gsiftp GET that
26 * refused a block reached the operator as a bare read error with no reason
27 * anywhere in the log, which is exactly how long a wire-level bug can hide.
28 * errno is what the VFS needs; the sentence is what a human needs. */
29static void
30sd_gsiftp_log_session_error(const brix_sd_instance_t *inst,
31 const gftp_session_t *session, const char *what)
32{
33 if (inst == NULL || inst->log == NULL || session->error[0] == '\0') {
34 return;
35 }
36 ngx_log_error(NGX_LOG_ERR, inst->log, 0,
37 "gsiftp %s failed: %s", what, session->error);
38}

41static int
42sd_gsiftp_sink_copy(void *ctx, off_t offset, const uint8_t *data, size_t len)
43{
44 sd_gsiftp_sink *sink = ctx;
45
46 if (offset < 0 || (uint64_t) offset > sink->cap
47 || len > sink->cap - (uint64_t) offset)
48 {
49 errno = EPROTO;
50 return -1;
51 }
52 memcpy(sink->base + offset, data, len);
53 return 0;
54}

56static int
57sd_gsiftp_parse_size(const char *text, off_t *size)
58{
59 char *end;
60 unsigned long long value;
61
62 while (*text == ' ') {
63 text++;
64 }
65 errno = 0;
66 value = strtoull(text, &end, 10);
67 if (errno != 0 || end == text || value > (unsigned long long) INT64_MAX) {
68 errno = EPROTO;
69 return -1;
70 }
71 while (*end == ' ') {
72 end++;
73 }
74 if (*end != '\0') {
75 errno = EPROTO;
76 return -1;
77 }
78 *size = (off_t) value;
79 return 0;
80}

82static void
83sd_gsiftp_parse_mtime(const char *text, brix_sd_stat_t *out)
84{
85 char line[96];
86 gftp_mlsx_ent_t ent;
87 const char *start = text;
88 size_t len = 0;
89
90 while (*start == ' ') {
91 start++;
92 }
93 while (start[len] != '\0' && start[len] != ' ' && len < 32) {
94 len++;
95 }
96 if (len < 14 || snprintf(line, sizeof(line), "modify=%.*s; x",
97 (int) len, start) >= (int) sizeof(line)) {
98 return;
99 }
100 if (gftp_mlsx_parse(line, strlen(line), &ent) == 0 && ent.has_mtime) {
101 out->mtime = ent.mtime;
102 out->ctime = ent.mtime;
103 }
104}

107sd_gsiftp_stat_dir(gftp_session_t *session, const char *remote,

131sd_gsiftp_stat_file(gftp_session_t *session, const char *remote,
132 brix_sd_stat_t *out)
133{

147static ngx_int_t
148sd_gsiftp_stat_finish(gftp_session_t *session, const char *remote,
149 brix_sd_stat_t *out, int command_rc)
150{
151 int rc = command_rc;
152
153 if (rc == 0 && sd_gsiftp_stat_file(session, remote, out)) {
154 return NGX_OK;
155 }
156 if (rc == 0 && session->code == 550) {
157 rc = sd_gsiftp_stat_dir(session, remote, out);

160 rc = -1;

165ngx_int_t
166sd_gsiftp_stat_impl(brix_sd_instance_t *inst, const char *path,
167 brix_sd_stat_t *out, const brix_sd_cred_t *cred)
168{
169 sd_gsiftp_state *state = inst->state;
170 gftp_session_t session;
171 char remote[GSIFTP_PATH_CAP];
172 const char *proxy;
173 int err = 0;
174 int rc;
175
176 memset(out, 0, sizeof(*out));
177 if (sd_gsiftp_select_proxy(state, cred, &proxy, &err) != 0) {
178 return NGX_ERROR;
179 }
180 if (sd_gsiftp_path(state, path, remote) != 0
181 || sd_gsiftp_session(&session, state, proxy) != 0) {
182 return NGX_ERROR;
183 }
184 rc = gftp_command(&session, "SIZE %s", remote);
185 rc = sd_gsiftp_stat_finish(&session, remote, out, rc);
186 if (rc != NGX_OK) {
187 sd_gsiftp_log_session_error(inst, &session, "stat");
188 }
189 gftp_session_close(&session);
190 return rc;
191}

321ssize_t
322sd_gsiftp_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
323{
324 sd_gsiftp_obj_state *obj_state = obj->state;
325 sd_gsiftp_state *state = obj->inst->state;
326 sd_gsiftp_sink sink = { .base = buf, .cap = 0 };
327 gftp_session_t session;
328 size_t received;
329
330 if (off < 0) {
331 errno = EINVAL;
332 return -1;
333 }
334 if (off >= obj->snap.size || len == 0) {
335 return 0;
336 }
337 if (len > (size_t) (obj->snap.size - off)) {
338 len = (size_t) (obj->snap.size - off);
339 }
340 sink.cap = len; /* after the clamp: the sink bound is the REAL buffer use */
341 if (sd_gsiftp_session(&session, state,
342 obj_state->proxy[0] != '\0' ? obj_state->proxy : NULL) != 0) {
343 sd_gsiftp_log_session_error(obj->inst, &session, "session open");
344 return -1;
345 }
346 if (gftp_retrieve(&session, obj_state->path, off, len,
347 sd_gsiftp_sink_copy, &sink, &received) != 0) {
348 sd_gsiftp_log_session_error(obj->inst, &session, "retrieve");
349 gftp_session_close(&session);
350 return -1;
351 }
352 gftp_session_close(&session);
353 return (ssize_t) received;
354}

356ssize_t
357sd_gsiftp_preadv(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
358 off_t off)
359{
360 uint8_t *buffer;
361 size_t total = 0;
362 size_t copied = 0;
363 ssize_t got;
364 int i;
365
366 for (i = 0; i < iovcnt; i++) {
367 if (iov[i].iov_len > SIZE_MAX - total) {
368 errno = EOVERFLOW;
369 return -1;
370 }
371 total += iov[i].iov_len;
372 }
373 if (total == 0) {
374 return 0;
375 }
376 buffer = malloc(total);
377 if (buffer == NULL) {
378 errno = ENOMEM;
379 return -1;
380 }
381 got = sd_gsiftp_pread(obj, buffer, total, off);
382 if (got < 0) {
383 free(buffer);
384 return -1;
385 }
386 for (i = 0; i < iovcnt && copied < (size_t) got; i++) {
387 size_t take = iov[i].iov_len;
388
389 if (take > (size_t) got - copied) {
390 take = (size_t) got - copied;
391 }
392 memcpy(iov[i].iov_base, buffer + copied, take);
393 copied += take;
394 }
395 free(buffer);
396 return got;
397}
