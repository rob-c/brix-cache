25static int
26sd_gsiftp_parse_size(const char *text, off_t *size)
27{
28 char *end;
29 unsigned long long value;
30
31 while (*text == ' ') {
32 text++;
33 }
34 errno = 0;
35 value = strtoull(text, &end, 10);
36 if (errno != 0 || end == text || value > (unsigned long long) INT64_MAX) {
37 errno = EPROTO;
38 return -1;
39 }
40 while (*end == ' ') {
41 end++;
42 }
43 if (*end != '\0') {
44 errno = EPROTO;
45 return -1;
46 }
47 *size = (off_t) value;
48 return 0;
49}

51static void
52sd_gsiftp_parse_mtime(const char *text, brix_sd_stat_t *out)
53{
54 char line[96];
55 gftp_mlsx_ent_t ent;
56 const char *start = text;
57 size_t len = 0;
58
59 while (*start == ' ') {
60 start++;
61 }
62 while (start[len] != '\0' && start[len] != ' ' && len < 32) {
63 len++;
64 }
65 if (len < 14 || snprintf(line, sizeof(line), "modify=%.*s; x",
66 (int) len, start) >= (int) sizeof(line)) {
67 return;
68 }
69 if (gftp_mlsx_parse(line, strlen(line), &ent) == 0 && ent.has_mtime) {
70 out->mtime = ent.mtime;
71 out->ctime = ent.mtime;
72 }
73}

99static int
100sd_gsiftp_stat_file(gftp_session_t *session, const char *remote,
101 brix_sd_stat_t *out)
102{
103 if (session->code != 213

105 return 0;
106 }
107 out->mode = S_IFREG | 0644;
108 out->is_reg = 1;
109 if (gftp_command(session, "MDTM %s", remote) == 0

111 sd_gsiftp_parse_mtime(session->text, out);

116static ngx_int_t
117sd_gsiftp_stat_finish(gftp_session_t *session, const char *remote,
118 brix_sd_stat_t *out, int command_rc)
119{
120 int rc = command_rc;
121
122 if (rc == 0 && sd_gsiftp_stat_file(session, remote, out)) {
123 return NGX_OK;
124 }
125 if (rc == 0 && session->code == 550) {
126 rc = sd_gsiftp_stat_dir(session, remote, out);
127 } else if (rc == 0) {
128 errno = EPROTO;
129 rc = -1;
130 }
131 return rc == 0 ? NGX_OK : NGX_ERROR;

287ssize_t
288sd_gsiftp_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
289{
290 sd_gsiftp_obj_state *obj_state = obj->state;
291 sd_gsiftp_state *state = obj->inst->state;
292 sd_gsiftp_sink sink = { .cursor = buf };
293 gftp_session_t session;
294 size_t received;
295
296 if (off < 0) {
297 errno = EINVAL;
298 return -1;
299 }
300 if (off >= obj->snap.size || len == 0) {
301 return 0;
302 }
303 if (len > (size_t) (obj->snap.size - off)) {
304 len = (size_t) (obj->snap.size - off);
305 }
306 if (sd_gsiftp_session(&session, state,
307 obj_state->proxy[0] != '\0' ? obj_state->proxy : NULL) != 0) {
308 return -1;
309 }
310 if (gftp_retrieve(&session, obj_state->path, off, len,
311 sd_gsiftp_sink_copy, &sink, &received) != 0) {
312 gftp_session_close(&session);
313 return -1;
314 }
315 gftp_session_close(&session);
316 return (ssize_t) received;
317}

319ssize_t
320sd_gsiftp_preadv(brix_sd_obj_t *obj, const struct iovec *iov, int iovcnt,
321 off_t off)
322{
323 uint8_t *buffer;
324 size_t total = 0;
325 size_t copied = 0;
326 ssize_t got;
327 int i;
328
329 for (i = 0; i < iovcnt; i++) {
330 if (iov[i].iov_len > SIZE_MAX - total) {
331 errno = EOVERFLOW;
332 return -1;
333 }
334 total += iov[i].iov_len;
335 }
336 if (total == 0) {
337 return 0;
338 }
339 buffer = malloc(total);
340 if (buffer == NULL) {
341 errno = ENOMEM;
342 return -1;
343 }
344 got = sd_gsiftp_pread(obj, buffer, total, off);
345 if (got < 0) {
346 free(buffer);
347 return -1;
348 }
349 for (i = 0; i < iovcnt && copied < (size_t) got; i++) {
350 size_t take = iov[i].iov_len;
351
352 if (take > (size_t) got - copied) {
353 take = (size_t) got - copied;
354 }
355 memcpy(iov[i].iov_base, buffer + copied, take);
356 copied += take;
357 }
358 free(buffer);
359 return got;
360}
