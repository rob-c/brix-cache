1/*
2 * sts_sign.c — SigV4 request construction for the S3 STS exchange (phase-79).
3 *
4 * WHAT: build the canonical (sorted, percent-encoded) STS query string and
5 * SigV4-sign it with the node's backend *service* credentials over the
6 * "sts" service, producing the exact query AWS/MinIO recomputes and checks.
7 * WHY: split out of sts.c (formerly 815 lines) as the pure request-building
8 * concern — no I/O, no allocation — so each STS file owns one job. The
9 * byte output is unchanged from the pre-split code: same canonical layout,
10 * same signing-key derivation, same signature.
11 * HOW: sts_build_action_qs emits Action=AssumeRole (+RoleArn/RoleSessionName)
12 * or Action=GetSessionToken in SigV4 canonical byte order; sts_sign_query
13 * forms the canonical request + string-to-sign, derives the signing key
14 * with brix_sigv4_signing_key(), and appends "&X-Amz-Signature=<hex>".
15 *
16 * The reused building blocks:
17 * - SigV4 signing key: brix_sigv4_signing_key() (core/compat/sigv4.h)
18 * - HMAC / SHA-256: brix_hmac_sha256/brix_sha256 (core/compat/crypto.h)
19 */
20#include "sts_internal.h"
21
22#include "core/compat/sigv4.h"
23#include "core/compat/crypto.h"

26/*
27 * Lowercase-hex-encode `n` bytes of `in` into `out` (needs 2*n+1 bytes, incl.
28 * NUL). Used for the payload hash and the final signature, both of which SigV4
29 * carries as hex.
30 */
31static void
32sts_hex(const uint8_t *in, size_t n, char *out)
33{
34 static const char hx[] = "0123456789abcdef";
35 size_t i;
36
37 for (i = 0; i < n; i++) {
38 out[2 * i] = hx[(in[i] >> 4) & 0xf];
39 out[2 * i + 1] = hx[in[i] & 0xf];
40 }
41 out[2 * n] = '\0';
42}

45/*
46 * sts_sign_query — build and SigV4-sign the STS request query string.
47 *
48 * WHAT: given the sorted action query (`action_qs`, already canonical and
49 * percent-encoded, WITHOUT the trailing X-Amz-Signature) and the target
50 * host, compute the SigV4 signature and return the full signed query.
51 * WHY: the STS endpoint authenticates the node by SigV4 over its service
52 * credentials; this is the exact string AWS/MinIO recomputes and checks.
53 * HOW: canonical request = "GET\n/\n<qs>\nhost:<h>\n\nhost\nUNSIGNED?..." with an
54 * empty-body SHA-256; string-to-sign wraps it with the credential scope;
55 * signing key from brix_sigv4_signing_key over "sts"; signature =
56 * HMAC(key, string_to_sign). Result "<qs>&X-Amz-Signature=<hex>" is
57 * written NUL-terminated into `out` (size `outsz`). NGX_OK / NGX_ERROR.
58 *
59 * The timestamps, host and config are read from `req` (amzdate is
60 * "YYYYMMDDTHHMMSSZ", datestamp its "YYYYMMDD" prefix).
61 */
62ngx_int_t
63sts_sign_query(const sts_req_t *req, const char *action_qs,
64 char *out, size_t outsz)
65{
66 const brix_s3_sts_conf_t *cf = req->cf;
67 const char *host = req->host;
68 const char *amzdate = req->amzdate;
69 const char *datestamp = req->datestamp;
70 uint8_t empty_hash[32];
71 char empty_hex[65];
72 char canonical[4096];
73 uint8_t canon_hash[32];
74 char canon_hex[65];
75 char scope[128];
76 char to_sign[512];
77 uint8_t key[32];
78 uint8_t sig[32];
79 char sig_hex[65];
80 int n;
81
82 /* Empty request body — STS AssumeRole carries all params in the query. */
83 if (!brix_sha256((const uint8_t *) "", 0, empty_hash)) {
84 return NGX_ERROR;
85 }
86 sts_hex(empty_hash, 32, empty_hex);
87
88 n = ngx_snprintf((u_char *) canonical, sizeof(canonical),
89 "GET\n/\n%s\nhost:%s\nx-amz-date:%s\n\nhost;x-amz-date\n%s",
90 action_qs, host, amzdate, empty_hex) - (u_char *) canonical;
91 if (n <= 0 || (size_t) n >= sizeof(canonical)) {
92 return NGX_ERROR;
93 }
94 if (!brix_sha256((const uint8_t *) canonical, (size_t) n, canon_hash)) {
95 return NGX_ERROR;
96 }
97 sts_hex(canon_hash, 32, canon_hex);
98
99 ngx_snprintf((u_char *) scope, sizeof(scope), "%s/%V/sts/aws4_request%Z",
100 datestamp, &cf->region);
101
102 n = ngx_snprintf((u_char *) to_sign, sizeof(to_sign),
103 "AWS4-HMAC-SHA256\n%s\n%s\n%s",
104 amzdate, scope, canon_hex) - (u_char *) to_sign;
105 if (n <= 0 || (size_t) n >= sizeof(to_sign)) {
106 return NGX_ERROR;
107 }
108
109 if (!brix_sigv4_signing_key(cf->svc_sk.data, cf->svc_sk.len,
110 datestamp, (const char *) cf->region.data, "sts", key)) {
111 return NGX_ERROR;
112 }
113 if (!brix_hmac_sha256(key, 32, (const uint8_t *) to_sign, (size_t) n, sig)) {
114 return NGX_ERROR;
115 }
116 sts_hex(sig, 32, sig_hex);
117
118 n = ngx_snprintf((u_char *) out, outsz, "%s&X-Amz-Signature=%s%Z",
119 action_qs, sig_hex) - (u_char *) out;
120 if (n <= 0 || (size_t) n >= outsz) {
121 return NGX_ERROR;
122 }
123 return NGX_OK;
124}

127/*
128 * sts_build_action_qs — build the canonical (sorted, encoded) STS query string,
129 * WITHOUT the trailing X-Amz-Signature.
130 *
131 * Parameters appear in SigV4 canonical (byte-sorted-by-name) order. When a role
132 * ARN is configured we emit Action=AssumeRole (+ RoleArn/RoleSessionName);
133 * otherwise Action=GetSessionToken. `req->credential` is the pre-encoded
134 * "AKID%2FDATE%2Fregion%2Fsts%2Faws4_request" X-Amz-Credential value.
135 */
136ngx_int_t
137sts_build_action_qs(const sts_req_t *req, char *out, size_t outsz)
138{
139 const brix_s3_sts_conf_t *cf = req->cf;
140 const char *rsn = req->rsn;
141 const char *credential = req->credential;
142 const char *amzdate = req->amzdate;
143 int ttl = req->ttl;
144 int n;
145
146 if (cf->role_arn.len > 0) {
147 n = ngx_snprintf((u_char *) out, outsz,
148 "Action=AssumeRole"
149 "&DurationSeconds=%d"
150 "&RoleArn=%V"
151 "&RoleSessionName=%s"
152 "&Version=2011-06-15"
153 "&X-Amz-Algorithm=AWS4-HMAC-SHA256"
154 "&X-Amz-Credential=%s"
155 "&X-Amz-Date=%s"
156 "&X-Amz-SignedHeaders=host%%3Bx-amz-date%Z",
157 ttl, &cf->role_arn, rsn, credential, amzdate)
158 - (u_char *) out;
159 } else {
160 n = ngx_snprintf((u_char *) out, outsz,
161 "Action=GetSessionToken"
162 "&DurationSeconds=%d"
163 "&Version=2011-06-15"
164 "&X-Amz-Algorithm=AWS4-HMAC-SHA256"
165 "&X-Amz-Credential=%s"
166 "&X-Amz-Date=%s"
167 "&X-Amz-SignedHeaders=host%%3Bx-amz-date%Z",
168 ttl, credential, amzdate)
169 - (u_char *) out;
170 }
171
172 if (n <= 0 || (size_t) n >= outsz) {
173 return NGX_ERROR;
174 }
175 return NGX_OK;
176}

183/*
184 * sts_form_encode — percent-encode `in` into `out` for an application/x-www-
185 * form-urlencoded value, leaving only the RFC-3986 unreserved set verbatim.
186 * WHY: an AssumeRole RoleArn/RoleSessionName may carry ':' '/' '+' '=' which are
187 * form delimiters or decode to space; the SigV4 signature is over the raw
188 * body bytes, so a mis-parsed value would still verify but AssumeRole would
189 * then reject it. Returns the written length, or -1 on overflow.
190 */
191static int
192sts_form_encode(const u_char *in, size_t inlen, char *out, size_t outsz)
193{
194 static const char hx[] = "0123456789ABCDEF";
195 size_t i, o = 0;
196
197 for (i = 0; i < inlen; i++) {
198 u_char c = in[i];
199 int unreserved = (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')
200 || (c >= '0' && c <= '9')
201 || c == '-' || c == '.' || c == '_' || c == '~';
202 if (unreserved) {
203 if (o + 1 >= outsz) { return -1; }
204 out[o++] = (char) c;
205 } else {
206 if (o + 3 >= outsz) { return -1; }
207 out[o++] = '%';
208 out[o++] = hx[(c >> 4) & 0xf];
209 out[o++] = hx[c & 0xf];
210 }
211 }
212 if (o >= outsz) { return -1; }
213 out[o] = '\0';
214 return (int) o;
215}

218/*
219 * sts_build_post — build the MinIO-dialect AssumeRole POST (see header). The
220 * signed headers are content-type;host;x-amz-content-sha256;x-amz-date and the
221 * credential scope uses literal '/' (header-auth), unlike the %2F query form.
222 */
223/* ngx_snprintf renders without a NUL guarantee and returns the end pointer;
224 * treat an empty render or one that filled the buffer as failure. Returns the
225 * byte count, or -1 — one guard for every length-critical SigV4 rendering. */
226static int
227sts_rendered(const char *buf, const u_char *end, size_t cap)
228{
229 ptrdiff_t n = end - (const u_char *) buf;
230
231 return (n <= 0 || (size_t) n >= cap) ? -1 : (int) n;
232}

235/* Render the AssumeRole POST body (form-encoded) and its payload hash into pd.
236 * Returns the body length, or -1. RoleArn is omitted when the operator left it
237 * unset — MinIO's dialect derives the role from the service key (phase-70). */
238static int
239sts_build_body(const sts_req_t *req, sts_post_t *pd)
240{
241 const brix_s3_sts_conf_t *cf = req->cf;
242 char rsn_enc[200];
243 char arn_enc[512];
244 uint8_t body_hash[32];
245 u_char *end;
246 int n;
247
248 if (sts_form_encode((const u_char *) req->rsn, ngx_strlen(req->rsn),
249 rsn_enc, sizeof(rsn_enc)) < 0) {
250 return -1;
251 }
252
253 if (cf->role_arn.len > 0) {
254 if (sts_form_encode(cf->role_arn.data, cf->role_arn.len,
255 arn_enc, sizeof(arn_enc)) < 0) {
256 return -1;
257 }
258 end = ngx_snprintf((u_char *) pd->body, sizeof(pd->body),
259 "Action=AssumeRole&DurationSeconds=%d&RoleArn=%s"
260 "&RoleSessionName=%s&Version=2011-06-15",
261 req->ttl, arn_enc, rsn_enc);
262 } else {
263 end = ngx_snprintf((u_char *) pd->body, sizeof(pd->body),
264 "Action=AssumeRole&DurationSeconds=%d"
265 "&RoleSessionName=%s&Version=2011-06-15",
266 req->ttl, rsn_enc);
267 }
268 n = sts_rendered(pd->body, end, sizeof(pd->body));
269 if (n < 0) {
270 return -1;
271 }
272 if (!brix_sha256((const uint8_t *) pd->body, (size_t) n, body_hash)) {
273 return -1;
274 }
275 sts_hex(body_hash, 32, pd->content_sha256);
276 return n;
277}

280/* Canonical request → its hex hash. POST, empty query, four sorted signed
281 * headers, body hash as the payload hash. Header names are already canonical
282 * (lowercase). Returns NGX_OK / NGX_ERROR; canon_hex must hold 65 bytes. */
283static ngx_int_t
284sts_canonical_hex(const sts_req_t *req, const sts_post_t *pd, char *canon_hex)
285{
286 char canonical[4096];
287 uint8_t canon_hash[32];
288 int n;
289
290 n = sts_rendered(canonical,
291 ngx_snprintf((u_char *) canonical, sizeof(canonical),
292 "POST\n/\n\n"
293 "content-type:%s\nhost:%s\nx-amz-content-sha256:%s\n"
294 "x-amz-date:%s\n\n"
295 "content-type;host;x-amz-content-sha256;x-amz-date\n%s",
296 BRIX_STS_POST_CTYPE, req->host, pd->content_sha256,
297 req->amzdate, pd->content_sha256),
298 sizeof(canonical));
299 if (n < 0) {
300 return NGX_ERROR;
301 }
302 if (!brix_sha256((const uint8_t *) canonical, (size_t) n, canon_hash)) {
303 return NGX_ERROR;
304 }
305 sts_hex(canon_hash, 32, canon_hex);
306 return NGX_OK;
307}

311static ngx_int_t
312sts_signature_hex(const sts_req_t *req, const char *scope, const char *canon_hex,

343{
344 const brix_s3_sts_conf_t *cf = req->cf;
345 char scope[128];

354 return NGX_ERROR;
355 }
356
357 ngx_snprintf((u_char *) scope, sizeof(scope), "%s/%V/sts/aws4_request%Z",
358 req->datestamp, &cf->region);
359
360 if (sts_signature_hex(req, scope, canon_hex, sig_hex) != NGX_OK) {
361 return NGX_ERROR;
362 }
363
364 if (sts_rendered(pd->authorization,
365 ngx_snprintf((u_char *) pd->authorization,
366 sizeof(pd->authorization),
367 "AWS4-HMAC-SHA256 Credential=%V/%s, "
368 "SignedHeaders=content-type;host;x-amz-content-sha256;"
369 "x-amz-date, Signature=%s", &cf->svc_ak, scope, sig_hex),
370 sizeof(pd->authorization)) < 0)
371 {
372 return NGX_ERROR;
373 }
374 return NGX_OK;
