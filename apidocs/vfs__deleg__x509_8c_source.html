1/* vfs_deleg_x509.c — RFC-3820 proxy-chain trust gate for delegated creds.
2 *
3 * WHAT: The X.509 half of the delegation live-cred materialiser: parse the
4 * captured proxy PEM into a certificate chain, re-verify it against the
5 * CA store bound on the bag (phase-70 §5.1 / P90-70.4), and stage the
6 * PEM into a request-scoped temp file for the backend driver.
7 *
8 * WHY: Split out of vfs_deleg.c, which crossed the 600-line cap
9 * (coding-standards §1). The chain-trust trio is a self-contained
10 * OpenSSL-facing unit; the parent TU keeps the strategy dispatch and the
11 * SSS/krb5/STS arms, which need no X509 machinery.
12 *
13 * HOW: Seam declarations live in vfs_deleg_internal.h — the parent lends
14 * brix_vfs_deleg_pem_is_valid()/_temp_cleanup(), this TU lends
15 * brix_vfs_deleg_proxy() back. brix_vfs_deleg_deny() is the shared

32/* ---- brix_vfs_deleg_chain_is_trusted ---------------------------------------
33 *
34 * WHAT: In-gate RFC-3820 chain-trust re-verify of the PASSTHROUGH proxy PEM
35 * against the CA store bound on the bag (phase-70 §5.1, P90-70.4).
36 *
37 * WHY: The capture sites enforce transport + leaf-DN identity, but chain trust
38 * must also hold at the one seam where captured bytes become a backend
39 * credential — with a store bound this gate fails closed even if a future
40 * capture site forgets its own check.
41 *
42 * HOW: No store bound → pass (the setter was never wired on this protocol; the
43 * capture-side gate applies alone). Otherwise parse the PEM into a
44 * STACK_OF(X509) via brix_vfs_deleg_chain_parse, take cert 0 as the leaf,
45 * and run brix_gsi_verify_chain with client_purpose=0 — the same call
46 * shape as webdav's delegation_chain_trusted (RFC-3820 proxies accepted;
47 * the helper logs the specific failure). Certs are freed either way. */
48
49/* Parse every CERTIFICATE block out of `pem` into a new non-empty
50 * STACK_OF(X509) (caller sk_X509_pop_free's it), or NULL. A forwarded grid
51 * proxy is "proxy cert, PRIVATE KEY, issuing chain" — a bare
52 * PEM_read_bio_X509 loop (delegation_parse_chain's shape) would stop at the
53 * key block and lose the chain, so generic PEM blocks are read and only the
54 * certificates kept; the key bytes are never copied out of the bag. */
55static STACK_OF(X509) *
56brix_vfs_deleg_chain_parse(const u_char *pem, size_t len)
57{
58 BIO *bio;
59 STACK_OF(X509) *chain;
60 char *name = NULL;
61 char *header = NULL;
62 unsigned char *der = NULL;
63 long der_len = 0;
64
65 bio = BIO_new_mem_buf(pem, (int) len);
66 if (bio == NULL) {
67 return NULL;
68 }
69
70 chain = sk_X509_new_null();
71 if (chain == NULL) {
72 BIO_free(bio);
73 return NULL;
74 }
75
76 while (PEM_read_bio(bio, &name, &header, &der, &der_len) == 1) {
77 if (ngx_strcmp(name, PEM_STRING_X509) == 0) {
78 const unsigned char *p = der;
79 X509 *cert = d2i_X509(NULL, &p, der_len);
80
81 if (cert != NULL && sk_X509_push(chain, cert) <= 0) {
82 X509_free(cert); /* partial chain → verify fails closed */
83 }
84 }
85 OPENSSL_free(name);
86 OPENSSL_free(header);
87 OPENSSL_free(der);
88 name = header = NULL;
89 der = NULL;
90 }
91 ERR_clear_error(); /* the terminating PEM_read failure is expected */
92 BIO_free(bio);
93
94 if (sk_X509_num(chain) == 0) {
95 sk_X509_pop_free(chain, X509_free);
96 return NULL;
97 }
98
99 return chain;
100}

102static int
103brix_vfs_deleg_chain_is_trusted(brix_vfs_ctx_t *ctx)
104{
105 brix_deleg_live_t *live = ctx->deleg_live;
106 X509 *leaf;
107 STACK_OF(X509) *chain;
108 brix_gsi_verify_result_t res;
109 ngx_int_t rc;
110
111 if (live->ca_store == NULL) {
112 return 1;
113 }
114
115 chain = brix_vfs_deleg_chain_parse(live->proxy_pem.data,
116 live->proxy_pem.len);
117 if (chain == NULL) {
118 return 0;
119 }
120
121 leaf = sk_X509_value(chain, 0);
122
123 rc = brix_gsi_verify_chain(ctx->log, (X509_STORE *) live->ca_store, leaf,
124 chain, live->ca_verify_depth, &res,
125 0 /* GSI: accept RFC-3820 proxies */);
126 sk_X509_pop_free(chain, X509_free);
127
128 if (rc != NGX_OK) {
129 ngx_log_error(NGX_LOG_ERR, ctx->log, 0,
130 "brix: PASSTHROUGH proxy chain failed CA re-verify at the "
131 "delegation gate - denying");
132 return 0;
133 }
134
135 return 1;
136}

138/* ---- brix_vfs_deleg_proxy --------------------------------------------------
139 *
140 * WHAT: Materialise a PASSTHROUGH full x509 proxy: validate the PEM, write it to
141 * a 0600 temp, register the unlink+zero cleanup, and point cred->x509_proxy
142 * at the temp path.
143 *
144 * WHY: brix_cache_origin_auth_gsi() authenticates from a proxy FILE path, so
145 * any strategy that materialises a proxy PEM at a 0600 path reuses the
146 * origin leg unchanged (§5.1). The private key must never be logged and
147 * must be unlinked on pool teardown.
148 *
149 * HOW: brix_vfs_deleg_pem_is_valid() rejects non-PEM bytes → deny;
150 * brix_vfs_deleg_chain_is_trusted() re-runs the RFC-3820 chain-trust gate
151 * when the capture site bound a CA store → deny on failure. Then
152 * brix_proxy_gsi_write_pem_temp() creates the owner-only temp; the path is
153 * copied onto the pool and a cleanup registered to unlink+zero it.
154 *
155 * RFC-3820 chain-trust (phase-70 §5.1): the full gate is (1) chain parses AND is
156 * unexpired; (2) leaf DN EQUALS the front-door authenticated DN (no
157 * privilege swap); (3) chain is RFC-3820-valid AND trusted by the export's
158 * CA store via brix_gsi_verify_chain(..., client_purpose=0); (4) TLS-only
159 * transport. (2) and (4) are enforced at CAPTURE (deleg_capture.c matches
160 * the leaf DN against the authenticated identity over TLS;
161 * gsi_promote_fullproxy DN-matches the root:// push); (1)+(3) are enforced
162 * HERE whenever the capture site stamped the export's CA store via
163 * brix_vfs_deleg_set_ca_store (P90-70.4 — webdav binds conf->ca_store,
164 * root:// binds conf->gsi_store). With no store bound the seam enforces
165 * PEM well-formedness and relies on the capture-side gate. */
166ngx_int_t
167brix_vfs_deleg_proxy(brix_vfs_ctx_t *ctx, brix_sd_cred_t *cred,
168 int *use_cred, int *err_out)
169{
170 brix_deleg_live_t *live = ctx->deleg_live;
171 char tmp[NGX_MAX_PATH];
172 char *path;
173 size_t path_len;
174 brix_deleg_temp_t *payload;
175 ngx_pool_cleanup_t *cln;
176 ngx_str_t princ = ngx_string("");
177
178 if (!brix_vfs_deleg_pem_is_valid(live->proxy_pem.data,
179 live->proxy_pem.len)) {
180 return brix_vfs_deleg_deny(ctx, use_cred, err_out,
181 BRIX_CRED_FAIL_PEM);
182 }
183
184 if (!brix_vfs_deleg_chain_is_trusted(ctx)) {
185 return brix_vfs_deleg_deny(ctx, use_cred, err_out,
186 BRIX_CRED_FAIL_CHAIN);
187 }
188
189 if (brix_proxy_gsi_write_pem_temp(live->proxy_pem.data,
190 live->proxy_pem.len, tmp, sizeof(tmp)) != 0) {
191 if (ctx->identity != NULL) {
192 princ = ctx->identity->subject;
193 }
194 ngx_log_error(NGX_LOG_ERR, ctx->log, ngx_errno,
195 "brix: failed to materialise PASSTHROUGH proxy temp for "
196 "principal=\"%V\"", &princ);
197 return brix_vfs_deleg_deny(ctx, use_cred, err_out,
198 BRIX_CRED_FAIL_MATERIALISE);
199 }
200
201 path_len = ngx_strlen(tmp);
202 path = ngx_pnalloc(ctx->pool, path_len + 1);
203 if (path == NULL) {
204 (void) unlink(tmp); /* vfs-seam-allow: config-domain PASSTHROUGH proxy credential temp (not export storage) */
205 brix_metric_cred_fail(brix_vfs_metrics_proto(ctx),
206 BRIX_CRED_FAIL_MATERIALISE);
207 errno = ENOMEM;
208 if (err_out != NULL) {
209 *err_out = ENOMEM;
210 }
211 *use_cred = 0;
212 return NGX_ERROR;
213 }
214 ngx_memcpy(path, tmp, path_len);
215 path[path_len] = '\0';
216
217 cln = ngx_pool_cleanup_add(ctx->pool, sizeof(*payload));
218 if (cln == NULL) {
219 (void) unlink(path); /* vfs-seam-allow: config-domain PASSTHROUGH proxy credential temp (not export storage) */
220 ngx_memzero(path, path_len);
221 brix_metric_cred_fail(brix_vfs_metrics_proto(ctx),
222 BRIX_CRED_FAIL_MATERIALISE);
223 errno = ENOMEM;
224 if (err_out != NULL) {
225 *err_out = ENOMEM;
226 }
227 *use_cred = 0;
228 return NGX_ERROR;
229 }
230 payload = cln->data;
231 payload->path = path;
232 cln->handler = brix_vfs_deleg_temp_cleanup;
233
234 cred->x509_proxy = path;
235 cred->mode = BRIX_CRED_PASSTHROUGH;
236 *use_cred = 1;
237
238 brix_metric_cred_deleg(brix_vfs_metrics_proto(ctx),
239 (ngx_uint_t) brix_vfs_backend_mode(ctx), BRIX_CRED_OUTCOME_USER);
240 return NGX_OK;
241}
