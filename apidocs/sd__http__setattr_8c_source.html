1/*
2 * sd_http_setattr.c — advisory POSIX metadata over WebDAV dead properties: the
3 * http driver's setattr pair.
4 *
5 * WHAT: `setattr` / `setattr_cred` — kXR_chmod and kXR_setattr against an HTTP
6 * origin, persisted as the ONE reserved dead property every object
7 * backend shares (BRIX_META_ADVISORY_XATTR), read-modify-written.
8 *
9 * WHY: RFC 4918 gives a resource a display name, a length and two timestamps;
10 * it gives it no mode, no owner and no settable mtime. Without this slot
11 * the VFS read the missing driver entry as "nothing to do" and told the
12 * client a chmod had succeeded that never left this host — the same silent
13 * lie the ceph and s3 drivers used to tell, and it is worse over WebDAV
14 * than anywhere else because the http driver already carries a full
15 * xattr plane, so the metadata had a home and simply was not being put in
16 * it. The advisory blob is what stat() overlays, so a mode written here is
17 * the mode the next stat reports.
18 *
19 * HOW: Composed entirely from slots this driver already exports — the sentinel
20 * translation (brix_meta_advisory_from_setattr), one named-prop PROPFIND
21 * (sd_http_getxattr_common) and one PROPPATCH (sd_http_setxattr_cred) —
22 * so there is no second wire spelling of the property and no second
23 * credential path. A request carrying nothing representable returns
24 * success BEFORE the round trip: an atime-only setattr costs no requests.
25 * Existence needs no probe of its own: PROPPATCH on an absent resource is
26 * 404, which the property writer already maps to ENOENT, so this file
27 * never has to decide whether a resource exists.
28 */

39/* The encoded blob is short and bounded by the codec's own grammar (a version
40 * token plus five small integers); this is the same ceiling the ceph and s3
41 * carriers use, so a blob written through one backend cannot straddle two
42 * limits when the object is later served through another. */
43#define SD_HTTP_ADVISORY_MAX 512

46/*
47 * WHAT: The one setattr body; `cred` NULL runs as the instance's service
48 * identity, non-NULL as the requesting user.
49 * WHY: The plain and _cred slots differ only in that pointer, and the
50 * read-modify-write in between must be identical for both — a second copy
51 * is a second chance to overlay a delta onto the wrong base blob.
52 * HOW: Size-enquiry-free single read into a fixed buffer: ENODATA (no property
53 * yet, or a 404 propstat) starts from an empty blob, and any other read
54 * failure aborts rather than silently discarding metadata that IS there.
55 * ERANGE is aborted too, for exactly that reason — a blob too big for this
56 * buffer is one this build cannot safely rewrite.
57 */
58static ngx_int_t
59sd_http_setattr_common(brix_sd_instance_t *inst, const char *path,
60 const brix_sd_setattr_t *attr, const brix_sd_cred_t *cred)
61{
62 brix_meta_advisory_t delta;
63 char blob[SD_HTTP_ADVISORY_MAX];
64 ssize_t n;
65 int len;
66
67 if (inst == NULL || inst->state == NULL || path == NULL || attr == NULL) {
68 errno = EINVAL;
69 return NGX_ERROR;
70 }
71 if (!brix_meta_advisory_from_setattr(attr, &delta)) {
72 return NGX_OK; /* nothing representable — no round trip */
73 }
74
75 n = sd_http_getxattr_common(inst, path, BRIX_META_ADVISORY_XATTR, blob,
76 sizeof(blob) - 1, cred);
77 if (n < 0) {
78 if (errno != ENODATA) {
79 return NGX_ERROR; /* EACCES/ENOTSUP/ERANGE/EIO — do not guess */
80 }
81 n = 0;
82 }
83 blob[n] = '\0';
84
85 len = brix_meta_advisory_patch(blob, sizeof(blob), &delta);
86 if (len < 0) {
87 errno = EIO;
88 return NGX_ERROR;
89 }
90 return sd_http_setxattr_cred(inst, path, BRIX_META_ADVISORY_XATTR, blob,
91 (size_t) len, 0, cred);
92}
