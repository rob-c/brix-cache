9/*
10 * WHAT: This file implements the SSS authentication identity layer. Includes copy_packed_string() for safe bounded string extraction from wire-format TLV fields, brix_sss_parse_identity() for parsing decrypted identity blocks (username, groups, host/IP), brix_sss_auth_failed() for sending auth failure responses, and brix_sss_send_authmore() for generating encrypted login-ID challenges with Blowfish-CFB + CRC32 integrity.
11 *
12 * WHY: SSS authentication requires a challenge-response protocol where the server encrypts an identity block (containing random nonce + generation timestamp) using Blowfish-CFB, appends CRC32 for integrity verification, and sends it to the client. The client decrypts, verifies CRC, extracts its username/groups, then re-encrypts with its shared secret key. This file handles both sides of that exchange.
13 *
14 * HOW: Four functions → copy_packed_string() (bounded string extraction from TLV value bytes) for safe parsing; brix_sss_parse_identity() (TLV stream parser) reading type/length/value triples and extracting name/grps/host fields into identity struct; brix_sss_auth_failed() (failure response builder); brix_sss_send_authmore() (challenge generation: RAND_bytes → gen_time encode → TLV LGID body → CRC32 append → Blowfish-CFB encrypt → wire response). */

16/*
17 * WHAT: Safe bounded string copy from TLV value bytes into fixed-size C-string buffer. Handles packed (NUL-terminated) strings by stripping trailing NUL before length check, then clamps to dst_len - 1 and appends local NUL terminator. Used by brix_sss_parse_identity() to extract username, groups, and host/IP fields from decrypted identity TLV blocks without risking buffer overflow.
18 */
19brix_sss_copy_packed_string(char *dst, size_t dst_len,
20 const u_char *src, size_t len)
21{
22 if (len > 0 && src[len - 1] == '\0') {
23 len--;
24 }
25
26 if (len >= dst_len) {
27 len = dst_len - 1;
28 }
29
30 ngx_memcpy(dst, src, len);
31 dst[len] = '\0';
32}

34/*
35 * brix_sss_parse_identity — parse the decrypted SSS identity TLV block.
36 *
37 * After decryption and CRC32 verification, the cleartext body (past the
38 * BRIX_SSS_DATA_HDR_LEN fixed header) contains a compact TLV stream:
39 * [1-byte field_type][2-byte big-endian field_len][field_len bytes of value]
40 *
41 * Known field types:
42 * BRIX_SSS_TYPE_NAME (0x01) — username
43 * BRIX_SSS_TYPE_VORG (0x02) — virtual organisation (v2 entity, F9)
44 * BRIX_SSS_TYPE_ROLE (0x03) — VO role (v2 entity, F9)
45 * BRIX_SSS_TYPE_GRPS (0x04) — comma-separated groups
46 * BRIX_SSS_TYPE_ENDO (0x05) — endorsements (v2 entity, F9)
47 * BRIX_SSS_TYPE_CRED (0x06) — proxied credential, raw (v2 entity, F9)
48 * BRIX_SSS_TYPE_RAND (0x07) — random nonce (not counted for id_count)
49 * BRIX_SSS_TYPE_LGID (0x10) — local group ID
50 * BRIX_SSS_TYPE_HOST (0x20) — source hostname or [IP] literal
51 *
52 * This parser is security-sensitive: every length check must be explicit and
53 * complete before advancing cursor. A wrong check can turn a malformed
54 * credential into an out-of-bounds read. Entity strings are never
55 * truncated into a different identity: a string at or over its cap, or a
56 * CRED blob over BRIX_SSS_ENT_CREDS_MAX, fails the whole parse.
57 *
58 * Returns: NGX_OK if at least one non-random field was found, NGX_ERROR if
59 * the block is empty, truncated, or carries an over-cap field.
60 */
61/* One packed-string entity field: wire tag → its slot in brix_sss_identity_t.
62 * The table IS the field list; add a v2 string field here, nowhere else. */
63typedef struct {
64 uint8_t type;
65 size_t off;
66 size_t cap;
67} sss_string_field_t;
68
69static const sss_string_field_t sss_string_fields[] = {
70 { BRIX_SSS_TYPE_NAME, offsetof(brix_sss_identity_t, name), BRIX_SSS_ENT_NAME_MAX },
71 { BRIX_SSS_TYPE_VORG, offsetof(brix_sss_identity_t, vorg), BRIX_SSS_ENT_VORG_MAX },
72 { BRIX_SSS_TYPE_ROLE, offsetof(brix_sss_identity_t, role), BRIX_SSS_ENT_ROLE_MAX },
73 { BRIX_SSS_TYPE_GRPS, offsetof(brix_sss_identity_t, grps), BRIX_SSS_ENT_GRPS_MAX },
74 { BRIX_SSS_TYPE_ENDO, offsetof(brix_sss_identity_t, endo), BRIX_SSS_ENT_ENDO_MAX },
75};

77/*
78 * sss_store_field — file one bounds-checked TLV into the identity.
79 *
80 * WHAT: table lookup for the packed-string fields, the HOST/[IP] split, and
81 * the raw CRED copy; unknown tags are ignored.
82 * WHY: the v2 fields must land by cap, not by truncation — a NAME or DN cut
83 * at 255 bytes is a different principal, a cut credential is garbage that
84 * still looks like one. Refusing here makes the whole parse fail closed.
85 * HOW: returns NGX_ERROR for a string whose character count (NUL excluded)
86 * reaches its cap or a CRED over BRIX_SSS_ENT_CREDS_MAX; NGX_OK otherwise.
87 */
88static ngx_int_t
89sss_store_field(brix_sss_identity_t *id, uint8_t type,
90 const u_char *value, size_t len)
91{
92 size_t i, chars;
93
94 chars = (len > 0 && value[len - 1] == '\0') ? len - 1 : len;
95
96 for (i = 0; i < sizeof(sss_string_fields) / sizeof(sss_string_fields[0]); i++) {
97 if (sss_string_fields[i].type != type) {
98 continue;
99 }
100 if (chars >= sss_string_fields[i].cap) {
101 return NGX_ERROR;
102 }
103 brix_sss_copy_packed_string((char *) id + sss_string_fields[i].off,
104 sss_string_fields[i].cap, value, len);
105 return NGX_OK;
106 }
107
108 if (type == BRIX_SSS_TYPE_HOST) {
109 if (len > 0 && value[0] == '[') {
110 brix_sss_copy_packed_string(id->ip, sizeof(id->ip), value, len);
111 } else {
112 brix_sss_copy_packed_string(id->host, sizeof(id->host), value, len);
113 }
114 return NGX_OK;
115 }
116
117 if (type == BRIX_SSS_TYPE_CRED) {
118 if (len > sizeof(id->creds)) {
119 return NGX_ERROR;
120 }
121 ngx_memcpy(id->creds, value, len);
122 id->creds_len = len;
123 }
124 return NGX_OK;
125}

127ngx_int_t
128brix_sss_parse_identity(const u_char *data, size_t len,
129 brix_sss_identity_t *id)
130{
131 const u_char *cursor;
132 const u_char *end;
133
134 ngx_memzero(id, sizeof(*id));
135 cursor = data;
136 end = data + len;
137
138 /*
139 * The decrypted SSS identity block is a compact TLV stream:
140 * [1-byte type][2-byte big-endian length][value bytes]
141 *
142 * Keep cursor movement explicit; this parser is security-sensitive and a
143 * mistaken length check can turn malformed credentials into memory reads.
144 */
145 while (cursor < end) {
146 const u_char *field_value;
147 uint8_t field_type;
148 size_t field_len;
149
150 field_type = *cursor++;
151 if ((size_t) (end - cursor) < 2) {
152 return NGX_ERROR;
153 }
154
155 field_len = ((size_t) cursor[0] << 8) | (size_t) cursor[1];
156 cursor += 2;
157 if ((size_t) (end - cursor) < field_len) {
158 return NGX_ERROR;
159 }
160
161 field_value = cursor;
162
163 if (field_type != BRIX_SSS_TYPE_RAND) {
164 id->id_count++;
165 }
166
167 if (sss_store_field(id, field_type, field_value, field_len) != NGX_OK) {
168 return NGX_ERROR;
169 }
170
171 cursor += field_len;
172 }
173
174 return id->id_count > 0 ? NGX_OK : NGX_ERROR;
175}

185brix_sss_send_authmore(brix_ctx_t *ctx, ngx_connection_t *c,
186 const brix_sss_key_t *key, const u_char *hdr, size_t hdr_len)
187{
188 u_char clear[BRIX_SSS_DATA_HDR_LEN + 1 + 2 + sizeof("nobody")];
189 u_char plain[sizeof(clear) + 4];
190 u_char cipher[sizeof(clear) + 4];
191 u_char *buf;
192 u_char *clear_cursor;
193 size_t clear_len, cipher_len, total, out_len;
194 uint32_t crc, gen_time;
195
196 ngx_memzero(clear, sizeof(clear));
197 RAND_bytes(clear, 32);
198 gen_time = (uint32_t) (ngx_time() - BRIX_SSS_BASE_TIME);
199 brix_sss_write_be32(clear + 32, gen_time);
200 clear[39] = BRIX_SSS_OPT_USEDATA;
201
202 /*
203 * Auth-more asks the client to use an explicit local identity. The
204 * identity body is the same TLV format parsed above; sizeof("nobody")
205 * intentionally includes the terminating NUL expected by older clients.
206 */
207 clear_cursor = clear + BRIX_SSS_DATA_HDR_LEN;
208 *clear_cursor++ = BRIX_SSS_TYPE_LGID;
209 *clear_cursor++ = 0;
210 *clear_cursor++ = (u_char) sizeof("nobody");
211 ngx_memcpy(clear_cursor, "nobody", sizeof("nobody"));
212 clear_cursor += sizeof("nobody");
213 clear_len = (size_t) (clear_cursor - clear);
214
215 ngx_memcpy(plain, clear, clear_len);
216 crc = htonl(brix_sss_crc32(plain, clear_len));
217 ngx_memcpy(plain + clear_len, &crc, sizeof(crc));
218
219 if (brix_sss_bf32_crypt(1, key->key, key->key_len,
220 plain, clear_len + sizeof(crc),
221 cipher, sizeof(cipher), &out_len)
222 != NGX_OK)
223 {
224 return NGX_ERROR;
225 }
226 cipher_len = out_len;
227
228 total = XRD_RESPONSE_HDR_LEN + hdr_len + cipher_len;
229 BRIX_PALLOC_OR_RETURN(buf, c->pool, total, NGX_ERROR);
230
231 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_authmore,
232 (uint32_t) (hdr_len + cipher_len),
233 (ServerResponseHdr *) buf);
234 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, hdr, hdr_len);
235 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN + hdr_len, cipher, cipher_len);
236
237 return brix_queue_response(ctx, c, buf, total);
238}
