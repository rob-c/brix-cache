27/* WHAT: Decode one fixed-width decimal field without permissive libc syntax.
28 * WHY: Expiry text is authorization input. HOW: Validate and accumulate digits. */
29static ngx_int_t
30macaroon_parse_decimal(const char *text, size_t length, int *out)
31{
32 size_t index;
33 int value = 0;
34
35 if (length == 0) {
36 return NGX_ERROR;
37 }
38 for (index = 0; index < length; index++) {
39 int digit;
40
41 if (text[index] < '0' || text[index] > '9') {
42 return NGX_ERROR;
43 }
44 digit = text[index] - '0';
45 if (value > (INT_MAX - digit) / 10) {
46 return NGX_ERROR;
47 }
48 value = value * 10 + digit;
49 }
50 *out = value;
51 return NGX_OK;
52}

54/* WHAT: Verify the literal positions in YYYY-MM-DDTHH:MM:SSZ.
55 * WHY: Numeric parsing alone could accept alternate separators.
56 * HOW: Compare every non-digit format position using a local shape template. */
57static ngx_int_t
58macaroon_timestamp_shape_valid(const char *text, size_t length)
59{
60 const char shape[] = "0000-00-00T00:00:00Z";
61 size_t index;
62
63 if (length != sizeof(shape) - 1) {
64 return NGX_ERROR;
65 }
66 for (index = 0; index < sizeof(shape) - 1; index++) {
67 if (shape[index] != '0' && text[index] != shape[index]) {
68 return NGX_ERROR;
69 }
70 }
71 return NGX_OK;
72}

74/* WHAT: Decode all six timestamp fields into one UTC broken-down time.
75 * WHY: Table-driven fixed widths keep strict parsing auditable and bounded.
76 * HOW: Parse local offset/width tables, then map values into struct tm. */
77static ngx_int_t
78macaroon_parse_timestamp_fields(const char *text, struct tm *parsed)
79{
80 const size_t offsets[] = {0, 5, 8, 11, 14, 17};
81 const size_t widths[] = {4, 2, 2, 2, 2, 2};
82 int values[6];
83 size_t index;
84
85 for (index = 0; index < 6; index++) {
86 if (macaroon_parse_decimal(text + offsets[index], widths[index],
87 &values[index]) != NGX_OK) {
88 return NGX_ERROR;
89 }
90 }
91 memset(parsed, 0, sizeof(*parsed));
92 parsed->tm_year = values[0] - BRIX_UNIX_EPOCH_YEAR;
93 parsed->tm_mon = values[1] - 1;
94 parsed->tm_mday = values[2];
95 parsed->tm_hour = values[3];
96 parsed->tm_min = values[4];
97 parsed->tm_sec = values[5];
98 parsed->tm_isdst = -1;
99 return NGX_OK;
100}

102/* WHAT: Check that UTC conversion preserved every caller-supplied field.
103 * WHY: timegm normalizes impossible dates such as February 31.
104 * HOW: Compare the six security-relevant calendar fields after round-trip. */
105static ngx_int_t
106macaroon_timestamp_matches(const struct tm *expected,
107 const struct tm *actual)
108{
109 if (actual->tm_year != expected->tm_year
110 || actual->tm_mon != expected->tm_mon
111 || actual->tm_mday != expected->tm_mday
112 || actual->tm_hour != expected->tm_hour
113 || actual->tm_min != expected->tm_min
114 || actual->tm_sec != expected->tm_sec) {
115 return NGX_ERROR;
116 }
117 return NGX_OK;

120/* WHAT: Convert exactly YYYY-MM-DDTHH:MM:SSZ to UTC or return -1.
121 * WHY: Malformed expiry restrictions must not become unbounded credentials.
122 * HOW: Validate shape/fields and reject any normalization on a UTC round-trip. */
123static time_t
124parse_iso8601(const char *text, size_t length)
125{
126 struct tm parsed;
127 struct tm expected;
128 struct tm checked;
129 time_t expiry;
130
131 if (macaroon_timestamp_shape_valid(text, length) != NGX_OK
132 || macaroon_parse_timestamp_fields(text, &parsed) != NGX_OK) {
133 return (time_t)-1;
134 }
135 expected = parsed;
136 expiry = timegm(&parsed);
137 if (expiry == (time_t)-1 || gmtime_r(&expiry, &checked) == NULL
138 || macaroon_timestamp_matches(&expected, &checked) != NGX_OK) {
139 return (time_t)-1;
140 }
141 return expiry;
142}

154/* WHAT: Serialize all structured permission:path entries to scope_raw.
155 * WHY: Partial output would make authorization views disagree.
156 * HOW: Append with checked bounds; revoke and error on overflow. */
157static ngx_int_t
158macaroon_rebuild_scope_raw(brix_token_claims_t *claims)
159{
160 static const char *perm_names[] = {
161 "storage.read", "storage.write",
162 "storage.create", "storage.modify"
163 };
164 size_t off = 0;
165 int si;
166
167 claims->scope_raw[0] = '\0';
168 for (si = 0; si < claims->scope_count; si++) {
169 brix_token_scope_t *sc = &claims->scopes[si];
170 unsigned int bits[4] = {
171 sc->read, sc->write, sc->create, sc->modify
172 };
173 int bi;
174 for (bi = 0; bi < 4; bi++) {
175 size_t plen, pathlen, needed;
176 if (!bits[bi]) {
177 continue;
178 }
179 plen = strlen(perm_names[bi]);
180 pathlen = strlen(sc->path);
181 needed = plen + pathlen + 2 + (off > 0 ? 1 : 0);
182 if (off >= sizeof(claims->scope_raw)
183 || needed > sizeof(claims->scope_raw) - off) {
184 macaroon_revoke_claim_scopes(claims);
185 return NGX_ERROR;
186 }
187 if (off > 0) {
188 claims->scope_raw[off++] = ' ';
189 }
190 memcpy(claims->scope_raw + off, perm_names[bi], plen);
191 off += plen;
192 claims->scope_raw[off++] = ':';
193 memcpy(claims->scope_raw + off, sc->path, pathlen);
194 off += pathlen;
195 claims->scope_raw[off] = '\0';
196 }
197 }
198 return NGX_OK;
199}

201/* WHAT: Narrow, retain, or revoke one scope by path. WHY: Caveats are AND
202 * restrictions. HOW: Use the shared boundary matcher and choose the deeper
203 * containing path, clearing permissions for disjoint paths. */
204static int
205macaroon_apply_path_to_scope(ngx_log_t *log, brix_token_scope_t *scope,
206 const char *caveat_path, size_t caveat_path_len)
207{
208 if (brix_token_scope_path_matches(scope->path, caveat_path)) {
209 if (strcmp(scope->path, caveat_path) == 0) {
210 return 0;
211 }
212 BRIX_LOG(NGX_LOG_INFO, log, 0,
213 "brix_macaroon: path: caveat \"%s\" narrows "
214 "scope path \"%s\" → \"%s\"", caveat_path, scope->path,
215 caveat_path);
216 memcpy(scope->path, caveat_path, caveat_path_len + 1);
217 return 1;
218 }
219
220 if (brix_token_scope_path_matches(caveat_path, scope->path)) {
221 return 0;
222 }
223
224 if (scope->read || scope->write || scope->create || scope->modify) {
225 BRIX_LOG(NGX_LOG_INFO, log, 0,
226 "brix_macaroon: path: caveat \"%s\" revokes "
227 "scope path \"%s\" (disjoint)", caveat_path, scope->path);
228 }
229 scope->read = scope->write = scope->create = scope->modify = 0;
230 return 1;
231}

233/* WHAT: Narrow claims by every path. WHY: Repeated paths are cumulative AND
234 * restrictions. HOW: Cross caveats with scopes and rebuild changed output. */
235ngx_int_t
236macaroon_apply_path_caveats(ngx_log_t *log,
237 brix_token_claims_t *claims,
238 char path_caveats[][BRIX_SCOPE_PATH_MAX], int n_path_caveats)
239{
240 int ci, si;
241 int narrowed = 0;
242
243 for (ci = 0; ci < n_path_caveats; ci++) {
244 const char *cp = path_caveats[ci];
245 size_t cplen = strlen(cp);
246
247 for (si = 0; si < claims->scope_count; si++) {
248 narrowed |= macaroon_apply_path_to_scope(log, &claims->scopes[si],
249 cp, cplen);
250 }
251 }
252
253 if (narrowed) {
254 return macaroon_rebuild_scope_raw(claims);
255 }
256 return NGX_OK;
257}

259/* WHAT: Compute one permission-and-path overlap. WHY: Discharges may only
260 * attenuate parent authority. HOW: AND permissions, choose the deeper
261 * containing path, and reject disjoint paths. */
262static int
263macaroon_scope_intersection(const brix_token_scope_t *parent,
264 const brix_token_scope_t *restriction, brix_token_scope_t *result)
265{
266 const char *path;
267 size_t path_length;
268
269 ngx_memzero(result, sizeof(*result));
270 result->read = parent->read && restriction->read;
271 result->write = parent->write && restriction->write;
272 result->create = parent->create && restriction->create;
273 result->modify = parent->modify && restriction->modify;
274 if (!result->read && !result->write && !result->create
275 && !result->modify) {
276 return 0;
277 }
278
279 if (brix_token_scope_path_matches(parent->path, restriction->path)) {
280 path = restriction->path;
281 } else if (brix_token_scope_path_matches(restriction->path,
282 parent->path)) {
283 path = parent->path;
284 } else {
285 return 0;
286 }
287 path_length = strlen(path);
288 if (path_length == 0 || path_length >= sizeof(result->path)) {
289 return 0;
290 }
291 ngx_memcpy(result->path, path, path_length + 1);
292 return 1;
293}

295/* WHAT: Merge or append one bounded intersection. WHY: Several pairs may
296 * yield the same path. HOW: OR same-path permissions or append with a cap. */
297static ngx_int_t
298macaroon_merge_intersection(brix_token_scope_t *scopes, int *scope_count,
299 const brix_token_scope_t *candidate)
300{
301 int scope_index;
302
303 for (scope_index = 0; scope_index < *scope_count; scope_index++) {
304 if (strcmp(scopes[scope_index].path, candidate->path) != 0) {
305 continue;
306 }
307 scopes[scope_index].read |= candidate->read;
308 scopes[scope_index].write |= candidate->write;
309 scopes[scope_index].create |= candidate->create;
310 scopes[scope_index].modify |= candidate->modify;
311 return NGX_OK;
312 }
313 if (*scope_count >= BRIX_MAX_TOKEN_SCOPES) {
314 return NGX_ERROR;
315 }
316 scopes[*scope_count] = *candidate;
317 (*scope_count)++;
318 return NGX_OK;
319}

321/* WHAT: Replace root scopes with the exact discharge intersection.
322 * WHY: Permissions and paths must attenuate together.
323 * HOW: Cross, merge bounded overlaps, then atomically rebuild text. */
324ngx_int_t
325macaroon_intersect_claim_scopes(ngx_log_t *log,
326 brix_token_claims_t *claims, const brix_token_claims_t *restriction)
327{
328 brix_token_scope_t narrowed[BRIX_MAX_TOKEN_SCOPES];
329 int narrowed_count = 0;
330 int parent_index, restriction_index;
331
332 ngx_memzero(narrowed, sizeof(narrowed));
333 for (parent_index = 0; parent_index < claims->scope_count; parent_index++) {
334 for (restriction_index = 0;
335 restriction_index < restriction->scope_count;
336 restriction_index++) {
337 brix_token_scope_t candidate;
338
339 if (!macaroon_scope_intersection(&claims->scopes[parent_index],
340 &restriction->scopes[restriction_index], &candidate)) {
341 continue;
342 }
343 if (macaroon_merge_intersection(narrowed, &narrowed_count,
344 &candidate) != NGX_OK) {
345 BRIX_LOG(NGX_LOG_WARN, log, 0,
346 "brix_macaroon: discharge scope intersection exceeds limit");
347 macaroon_revoke_claim_scopes(claims);
348 return NGX_ERROR;
349 }
350 }
351 }
352
353 ngx_memzero(claims->scopes, sizeof(claims->scopes));
354 if (narrowed_count > 0) {
355 ngx_memcpy(claims->scopes, narrowed,
356 (size_t)narrowed_count * sizeof(narrowed[0]));
357 }
358 claims->scope_count = narrowed_count;
359 return macaroon_rebuild_scope_raw(claims);
360}

368/* ---- Map one exact WLCG activity to effective permission bits ----
369 *
370 * WHAT: Return the permission bit for one known activity, or zero if unknown.
371 * WHY: Unknown activity text must not be ignored inside a signed caveat.
372 * HOW: Compare exact lengths and bytes; group equivalent read/write activities.
373 */
374static unsigned int
375macaroon_permission_for_activity(const char *activity, size_t activity_len)
376{
377 if ((activity_len == 8 && memcmp(activity, "DOWNLOAD", 8) == 0)
378 || (activity_len == 4 && memcmp(activity, "LIST", 4) == 0)
379 || (activity_len == 5 && memcmp(activity, "STAGE", 5) == 0)) {
380 return MACAROON_ACTIVITY_READ;
381 }
382 if ((activity_len == 6 && memcmp(activity, "UPLOAD", 6) == 0)
383 || (activity_len == 6 && memcmp(activity, "DELETE", 6) == 0)) {
384 return MACAROON_ACTIVITY_WRITE;
385 }
386 if (activity_len == 6 && memcmp(activity, "MANAGE", 6) == 0) {
387 return MACAROON_ACTIVITY_MODIFY;
388 }
389 return 0;
390}

392/* ---- Append one effective activity scope to the bounded parser buffer ----
393 *
394 * WHAT: Append scope:/ and return NGX_OK, or NGX_ERROR without truncation.
395 * WHY: Silently dropping a caveat-derived scope can change intersection
396 * semantics between signed input and enforced authority.
397 * HOW: Compute required bytes by subtraction, add a separator when needed,
398 * then copy the fixed scope and root path.
399 */
400static ngx_int_t
401macaroon_append_activity_scope(brix_macaroon_parse_state_t *state,
402 const char *scope)
403{
404 size_t scope_len = strlen(scope);
405 size_t needed = scope_len + 2 + (state->scope_off > 0 ? 1 : 0);
406
407 if (state->scope_off >= sizeof(state->scope_buf)
408 || needed >= sizeof(state->scope_buf) - state->scope_off) {
409 return NGX_ERROR;
410 }
411 if (state->scope_off > 0) {
412 state->scope_buf[state->scope_off++] = ' ';
413 }
414 memcpy(state->scope_buf + state->scope_off, scope, scope_len);
415 state->scope_off += scope_len;
416 state->scope_buf[state->scope_off++] = ':';
417 state->scope_buf[state->scope_off++] = '/';
418 state->scope_buf[state->scope_off] = '\0';
419 return NGX_OK;
420}

423 *

425 * WHY: Each activity caveat is an AND restriction while comma-separated

451/* ---- Parse and intersect one activity: caveat ----
452 *
453 * WHAT: Parse comma-separated known activities and intersect their union with
454 * prior activity caveats; return NGX_ERROR for malformed/unknown input.
455 * WHY: Repeated caveats attenuate authority. Unioning them allowed a holder
456 * to append activity:UPLOAD to a read-only token and gain write access.
457 * HOW: Build one caveat mask, rejecting empty/unknown elements, AND it after
458 * the first caveat, then rebuild the effective scope buffer.
459 */
460static ngx_int_t
461macaroon_parse_activity_caveat(brix_macaroon_parse_state_t *state,
462 const u_char *caveat, size_t caveat_len)
463{
464 const char *activity = (const char *)(caveat + 9);
465 size_t remaining = caveat_len - 9;
466 unsigned int caveat_mask = 0;
467
468 if (remaining == 0) {
469 return NGX_ERROR;
470 }
471 while (remaining > 0) {
472 const char *comma = memchr(activity, ',', remaining);
473 size_t activity_len = comma != NULL
474 ? (size_t)(comma - activity) : remaining;
475 unsigned int permission;
476
477 if (activity_len == 0) {
478 return NGX_ERROR;
479 }
480 permission = macaroon_permission_for_activity(activity, activity_len);
481 if (permission == 0) {
482 return NGX_ERROR;
483 }
484 caveat_mask |= permission;
485 if (comma == NULL) {
486 remaining = 0;
487 } else {
488 activity += activity_len + 1;
489 remaining -= activity_len + 1;
490 if (remaining == 0) {
491 return NGX_ERROR;
492 }
493 }
494 }
495
496 state->activity_mask = state->activity_seen
497 ? state->activity_mask & caveat_mask : caveat_mask;
498 state->activity_seen = 1;
499 return macaroon_rebuild_activity_scopes(state);
500}

550/* ---- Classify one first-party caveat with fail-closed semantics ----
551 *
552 * WHAT: Apply one supported caveat and return NGX_OK; reject unsupported or
553 * malformed signed caveats with NGX_ERROR.
554 * WHY: Silently ignoring a condition makes the server authorize more than the
555 * macaroon expresses.
556 * HOW: Dispatch exact activity:/before:/path: prefixes and reject everything
557 * else without interpreting it as a third-party caveat.

559ngx_int_t
560macaroon_parse_first_party_caveat(brix_macaroon_parse_state_t *state,
561 const u_char *caveat, size_t caveat_len)
562{

564 return macaroon_parse_activity_caveat(state, caveat, caveat_len);
