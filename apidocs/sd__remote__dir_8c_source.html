29/* ---- directory listing (S3 ListObjectsV2, delimited + paged) --------------
30 *
31 * WHAT: opendir/readdir/closedir over the object catalog as a POSIX-shaped
32 * single directory level — <Contents> under the prefix are files,
33 * <CommonPrefixes> are sub-directories (finding #4).
34 * WHY: S3 has no readdir; a WebDAV PROPFIND / xrdfs ls / recursive walk over an
35 * s3:// export previously hit the NULL opendir slot and reported ENOTSUP.
36 * HOW: opendir derives the S3 key prefix from the export-relative path (no I/O)
37 * and readdir pages ListObjectsV2 lazily via the shared sd_s3_list_page,
38 * buffering one page of decoded basenames at a time; closedir frees the
39 * malloc-owned handle (this driver runs off the event loop with no pool).
40 * Object stores expose no per-object owner/mode, so d_type is DT_DIR for a
41 * CommonPrefixes entry and DT_REG otherwise — the VFS stats on anything it
42 * cannot classify, so a coarse d_type is a cheap hint, never authority. */
43typedef struct {
44 char name[256];
45 unsigned char d_type;
46} sd_remote_dirent;
47
48/* Upper bounds for the copied per-user SigV4 credential below. ak/sk/region
49 * match the ucred store's own limits; the STS session token has no small bound
50 * in the API, so 4 KiB with a REFUSAL (never a truncation) on overflow — a
51 * clipped session token would surface as an inscrutable SignatureDoesNotMatch
52 * pages into the listing rather than as an error at opendir. */
53#define SD_REMOTE_DIR_AK_MAX 128
54#define SD_REMOTE_DIR_SK_MAX 256
55#define SD_REMOTE_DIR_REGION_MAX 64
56#define SD_REMOTE_DIR_SESSION_MAX 4096
57
58typedef struct {
59 brix_sd_instance_t *inst;
60 char prefix[768]; /* S3 key prefix, "" or "dir/" */
61 char cont[2048]; /* NextContinuationToken for the next page */
62 int started; /* fetched at least one page */
63 int truncated; /* more pages remain */
64 sd_remote_dirent *ents; /* current page, grown on demand */
65 size_t n;
66 size_t cap;
67 size_t cursor;
68 /* Per-user SigV4 credential, COPIED at opendir. The listing is paged lazily
69 * from readdir, so the signing material has to outlive the opendir call that
70 * borrowed *cred — the same lifetime trap sd_http_obj_state's bearer copy
71 * documents. `have_cred` distinguishes "this handle signs as a user" from
72 * "no credential" so an empty string is never mistaken for one. */
73 int have_cred;
74 char ak [SD_REMOTE_DIR_AK_MAX];
75 char sk [SD_REMOTE_DIR_SK_MAX];
76 char region [SD_REMOTE_DIR_REGION_MAX];
77 char session[SD_REMOTE_DIR_SESSION_MAX];
78} sd_remote_dir_state;

80/* sd_s3_list_page callback: append one decoded entry to the page buffer. A
81 * realloc failure stops the page (returns 1) and surfaces as a short page. */
82static int
83sd_remote_dir_add(void *ud, const char *name, int is_dir)
84{
85 sd_remote_dir_state *ds = ud;
86
87 if (ds->n == ds->cap) {
88 size_t nc = (ds->cap != 0) ? ds->cap * 2 : 64;
89 void *nb = realloc(ds->ents, nc * sizeof(*ds->ents));
90
91 if (nb == NULL) {
92 return 1;
93 }
94 ds->ents = nb;
95 ds->cap = nc;
96 }
97 snprintf(ds->ents[ds->n].name, sizeof(ds->ents[ds->n].name), "%s", name);
98 ds->ents[ds->n].d_type = (unsigned char) (is_dir ? DT_DIR : DT_REG);
99 ds->n++;
100 return 0;
101}

103/* Fetch the next ListObjectsV2 page into the (reset) buffer. 0 / -1 (errno). */
104static int
105sd_remote_dir_fetch(sd_remote_dir_state *ds)
106{
107 const brix_sd_remote_cfg_t *cfg = ds->inst->state;
108 sd_s3_open_params p;
109 char root[300];
110 char cont_out[2048];
111 char errbuf[256];
112 int truncated = 0;
113
114 snprintf(root, sizeof(root), "/%s/", cfg->bucket); /* bucket-root canon URI */
115 sd_remote_s3_params(cfg, root, &p);
116 if (ds->have_cred) {
117 /* Every page of this listing signs as the identity that opened it, not
118 * as the export — including the continuation pages fetched long after
119 * opendir returned. */
120 sd_remote_params_cred(&p, ds->ak, ds->sk,
121 (ds->region[0] != '\0') ? ds->region : NULL,
122 (ds->session[0] != '\0') ? ds->session : NULL);
123 }
124
125 ds->n = 0;
126 ds->cursor = 0;
127 errno = 0;
128 if (sd_s3_list_page(&p, ds->prefix, ds->started ? ds->cont : "",
129 sd_remote_dir_add, ds, &truncated, cont_out, sizeof(cont_out),
130 errbuf, sizeof(errbuf)) != 0)
131 {
132 if (errno == 0) { errno = EIO; }
133 return -1;
134 }
135 ds->truncated = truncated;
136 snprintf(ds->cont, sizeof(ds->cont), "%s", cont_out);
137 ds->started = 1;
138 return 0;
139}

141/* Copy one credential string into a fixed handle buffer. Returns 0, or -1 when
142 * the value does not fit: a SILENTLY truncated key or session token signs a
143 * request that the store rejects with an opaque SignatureDoesNotMatch, so the
144 * only safe answer is to refuse the open. NULL/empty leaves the slot empty. */
145static int
146sd_remote_dir_copy_str(char *dst, size_t cap, const char *src)
147{
148 size_t n;
149
150 if (src == NULL || *src == '\0') {
151 dst[0] = '\0';
152 return 0;
153 }
154 n = strlen(src);
155 if (n >= cap) {
156 return -1;
157 }
158 memcpy(dst, src, n + 1);
159 return 0;
160}

162/* Latch the requesting user's SigV4 material onto the handle. Returns 0 (signs
163 * as the user), 1 (no credential to apply — sign as the export) or -1 with
164 * *err_out set. */
165static int
166sd_remote_dir_take_cred(sd_remote_dir_state *ds, const brix_sd_cred_t *cred,
167 int *err_out)
168{
169 int gate = sd_remote_cred_gate(cred);
170
171 if (gate < 0) {
172 if (err_out != NULL) { *err_out = EACCES; }
173 return -1;
174 }
175 if (gate == 0) {
176 return 1;
177 }
178 if (sd_remote_dir_copy_str(ds->ak, sizeof(ds->ak), cred->s3_ak) != 0
179 || sd_remote_dir_copy_str(ds->sk, sizeof(ds->sk), cred->s3_sk) != 0
180 || sd_remote_dir_copy_str(ds->region, sizeof(ds->region),
181 cred->s3_region) != 0
182 || sd_remote_dir_copy_str(ds->session, sizeof(ds->session),
183 cred->s3_session) != 0)
184 {
185 if (err_out != NULL) { *err_out = E2BIG; }
186 return -1;
187 }
188 ds->have_cred = 1;
189 return 0;
190}

192/* Wipe the copied secret before the handle's memory goes back to the allocator:
193 * a freed page holding a live secret key is one heap re-use away from another
194 * request's buffer. */
195static void
196sd_remote_dir_wipe_cred(sd_remote_dir_state *ds)
197{
198 OPENSSL_cleanse(ds->sk, sizeof(ds->sk));
199 OPENSSL_cleanse(ds->session, sizeof(ds->session));
200}

202/* Shared body of opendir/opendir_cred: derive the key prefix (no I/O) and, when
203 * a usable credential came in, latch it for the lazy pages readdir will fetch. */
204static brix_sd_dir_t *
205sd_remote_opendir_impl(brix_sd_instance_t *inst, const char *path,
206 const brix_sd_cred_t *cred, int *err_out)
207{
208 sd_remote_dir_state *ds;
209 brix_sd_dir_t *dir;
210 const char *rel = (path != NULL) ? path : "/";
211 size_t n;
212
213 ds = calloc(1, sizeof(*ds));
214 dir = calloc(1, sizeof(*dir));
215 if (ds == NULL || dir == NULL) {
216 free(ds);
217 free(dir);
218 if (err_out != NULL) { *err_out = ENOMEM; }
219 return NULL;
220 }
221 ds->inst = inst;
222
223 if (cred != NULL && sd_remote_dir_take_cred(ds, cred, err_out) < 0) {
224 sd_remote_dir_wipe_cred(ds);
225 free(ds);
226 free(dir);
227 return NULL;
228 }
229
230 /* export-relative path -> S3 key prefix: drop the leading '/', ensure a
231 * trailing '/' so LIST returns children of THIS level (root -> ""). */
232 while (*rel == '/') { rel++; }
233 n = strlen(rel);
234 if (n + 1 >= sizeof(ds->prefix)) {
235 sd_remote_dir_wipe_cred(ds);
236 free(ds);
237 free(dir);
238 if (err_out != NULL) { *err_out = ENAMETOOLONG; }
239 return NULL;
240 }
241 memcpy(ds->prefix, rel, n);
242 if (n > 0 && ds->prefix[n - 1] != '/') { ds->prefix[n++] = '/'; }
243 ds->prefix[n] = '\0';
244
245 dir->inst = inst;
246 dir->state = ds;
247 return dir;
248}

256/* Cred-scoped opendir: the whole listing runs as the requesting user.
257 *
258 * WHY: this driver already signed every namespace op (stat/mkdir/rename/unlink)
259 * and, since the metadata-read fix, every xattr read under the caller's own
260 * SigV4 keys — but opendir had no *_cred sibling, so brix_sd_opendir's
261 * forwarder fell through to the plain slot and LISTED THE BUCKET AS THE
262 * EXPORT. A user whose keys are scoped to one prefix saw every sibling
263 * prefix in the bucket, and the entries came back looking perfectly normal.
264 * HOW: sd_remote_cred_gate classifies the credential exactly as the other slots
265 * do (usable keypair / unusable-under-deny / no S3 material), and the
266 * material is COPIED, because ListObjectsV2 is paged lazily from readdir
267 * long after *cred stops being ours to hold. */
268brix_sd_dir_t *
269sd_remote_opendir_cred(brix_sd_instance_t *inst, const char *path,
270 int *err_out, const brix_sd_cred_t *cred)
271{
272 return sd_remote_opendir_impl(inst, path, cred, err_out);
273}

275ngx_int_t
276sd_remote_readdir(brix_sd_dir_t *d, brix_sd_dirent_t *out)
277{
278 sd_remote_dir_state *ds = d->state;
279
280 for ( ;; ) {
281 if (ds->cursor < ds->n) {
282 snprintf(out->name, sizeof(out->name), "%s",
283 ds->ents[ds->cursor].name);
284 out->d_type = ds->ents[ds->cursor].d_type;
285 ds->cursor++;
286 return NGX_OK;
287 }
288 if (ds->started && !ds->truncated) {
289 return NGX_DONE;
290 }
291 if (sd_remote_dir_fetch(ds) != 0) {
292 return NGX_ERROR;
293 }
294 }
