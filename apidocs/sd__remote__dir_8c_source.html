27/* ---- directory listing (S3 ListObjectsV2, delimited + paged) --------------
28 *
29 * WHAT: opendir/readdir/closedir over the object catalog as a POSIX-shaped
30 * single directory level — <Contents> under the prefix are files,
31 * <CommonPrefixes> are sub-directories (finding #4).
32 * WHY: S3 has no readdir; a WebDAV PROPFIND / xrdfs ls / recursive walk over an
33 * s3:// export previously hit the NULL opendir slot and reported ENOTSUP.
34 * HOW: opendir derives the S3 key prefix from the export-relative path (no I/O)
35 * and readdir pages ListObjectsV2 lazily via the shared sd_s3_list_page,
36 * buffering one page of decoded basenames at a time; closedir frees the
37 * malloc-owned handle (this driver runs off the event loop with no pool).
38 * Object stores expose no per-object owner/mode, so d_type is DT_DIR for a
39 * CommonPrefixes entry and DT_REG otherwise — the VFS stats on anything it
40 * cannot classify, so a coarse d_type is a cheap hint, never authority. */
41typedef struct {
42 char name[256];
43 unsigned char d_type;
44} sd_remote_dirent;

46typedef struct {

58/* sd_s3_list_page callback: append one decoded entry to the page buffer. A
59 * realloc failure stops the page (returns 1) and surfaces as a short page. */
60static int
61sd_remote_dir_add(void *ud, const char *name, int is_dir)
62{
63 sd_remote_dir_state *ds = ud;
64
65 if (ds->n == ds->cap) {
66 size_t nc = (ds->cap != 0) ? ds->cap * 2 : 64;
67 void *nb = realloc(ds->ents, nc * sizeof(*ds->ents));
68
69 if (nb == NULL) {
70 return 1;
71 }
72 ds->ents = nb;
73 ds->cap = nc;
74 }
75 snprintf(ds->ents[ds->n].name, sizeof(ds->ents[ds->n].name), "%s", name);
76 ds->ents[ds->n].d_type = (unsigned char) (is_dir ? DT_DIR : DT_REG);
77 ds->n++;
78 return 0;
79}

81/* Fetch the next ListObjectsV2 page into the (reset) buffer. 0 / -1 (errno). */
82static int
83sd_remote_dir_fetch(sd_remote_dir_state *ds)
84{
85 const brix_sd_remote_cfg_t *cfg = ds->inst->state;
86 sd_s3_open_params p;
87 char root[300];
88 char cont_out[2048];
89 char errbuf[256];
90 int truncated = 0;
91
92 snprintf(root, sizeof(root), "/%s/", cfg->bucket); /* bucket-root canon URI */
93 sd_remote_s3_params(cfg, root, &p);
94
95 ds->n = 0;
96 ds->cursor = 0;
97 errno = 0;
98 if (sd_s3_list_page(&p, ds->prefix, ds->started ? ds->cont : "",
99 sd_remote_dir_add, ds, &truncated, cont_out, sizeof(cont_out),
100 errbuf, sizeof(errbuf)) != 0)
101 {
102 if (errno == 0) { errno = EIO; }
103 return -1;
104 }
105 ds->truncated = truncated;
106 snprintf(ds->cont, sizeof(ds->cont), "%s", cont_out);
107 ds->started = 1;
108 return 0;
109}

111brix_sd_dir_t *
112sd_remote_opendir(brix_sd_instance_t *inst, const char *path, int *err_out)
113{
114 sd_remote_dir_state *ds;
115 brix_sd_dir_t *dir;
116 const char *rel = (path != NULL) ? path : "/";
117 size_t n;
118
119 ds = calloc(1, sizeof(*ds));
120 dir = calloc(1, sizeof(*dir));
121 if (ds == NULL || dir == NULL) {
122 free(ds);
123 free(dir);
124 if (err_out != NULL) { *err_out = ENOMEM; }
125 return NULL;
126 }
127 ds->inst = inst;
128
129 /* export-relative path -> S3 key prefix: drop the leading '/', ensure a
130 * trailing '/' so LIST returns children of THIS level (root -> ""). */
131 while (*rel == '/') { rel++; }
132 n = strlen(rel);
133 if (n + 1 >= sizeof(ds->prefix)) {
134 free(ds);
135 free(dir);
136 if (err_out != NULL) { *err_out = ENAMETOOLONG; }
137 return NULL;
138 }
139 memcpy(ds->prefix, rel, n);
140 if (n > 0 && ds->prefix[n - 1] != '/') { ds->prefix[n++] = '/'; }
141 ds->prefix[n] = '\0';
142
143 dir->inst = inst;
144 dir->state = ds;
145 return dir;
146}

148ngx_int_t
149sd_remote_readdir(brix_sd_dir_t *d, brix_sd_dirent_t *out)
150{
151 sd_remote_dir_state *ds = d->state;
152
153 for ( ;; ) {
154 if (ds->cursor < ds->n) {
155 snprintf(out->name, sizeof(out->name), "%s",
156 ds->ents[ds->cursor].name);
157 out->d_type = ds->ents[ds->cursor].d_type;
158 ds->cursor++;
159 return NGX_OK;
160 }
161 if (ds->started && !ds->truncated) {
162 return NGX_DONE;
163 }
164 if (sd_remote_dir_fetch(ds) != 0) {
165 return NGX_ERROR;
166 }
167 }

172{
173 sd_remote_dir_state *ds;
174
175 if (d == NULL || d->state == NULL) {
176 return NGX_OK;
177 }
178 ds = d->state;
179 free(ds->ents);
180 free(ds);
181 d->state = NULL;
182 free(d); /* malloc-owned shell (no pool off the event loop) */
183 return NGX_OK;
184}
