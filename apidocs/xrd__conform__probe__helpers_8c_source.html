9/* WHAT: Read exactly one raw frame into caller-bounded body storage.
10 * WHY: Flow and framing probes inspect raw replies without allocating from an
11 * untrusted wire length, and must refuse a length beyond their own distinct cap.
12 * HOW: Initialize outputs; read/unpack through shared framing helpers; enforce
13 * the caller's cap before reading the body. No status or stream-ID filtering.
14 */
15int
16xrd_conform_read_bounded_frame(brix_conn *connection, uint16_t *sid,
17 uint16_t *wire_status, uint32_t *length, void *body, size_t capacity,
18 brix_status *status)
19{
20 uint8_t header[XRD_RESPONSE_HDR_LEN];
21
22 *sid = 0;
23 *wire_status = 0;
24 *length = 0;
25 memset(body, 0, capacity);
26 if (brix_read_full(&connection->io, header, sizeof(header), status) != 0) {
27 return -1;
28 }
29 xrd_resp_hdr_unpack(header, sid, wire_status, length);
30 if (*length > capacity) {
31 brix_status_set(status, XRDC_EPROTO, 0,
32 "conformance response body too large: %u", *length);
33 return -1;
34 }
35 if (*length != 0 && brix_read_full(&connection->io, body, *length, status) != 0) {
36 return -1;
37 }
38 return 0;
39}

41/* WHAT: Consume one bounded whitespace-delimited locate token.
42 * WHY: Holder selection and access-bit checks must use identical boundaries.
43 * HOW: Skip separators, retain the token extent, advance the caller's cursor.
44 */
45int
46xrd_conform_next_locate_token(const char **cursor, const char **start,
47 size_t *length)
48{
49 const char *next = *cursor;
50 if (next == NULL) {
51 return 0;
52 }
53 while (*next == ' ' || *next == '\t' || *next == '\n') {
54 next++;
55 }
56 *start = next;
57 while (*next != '\0' && *next != ' ' && *next != '\t' && *next != '\n') {
58 next++;
59 }
60 *length = (size_t) (next - *start);
61 *cursor = next;
62 return *length != 0;
63}
