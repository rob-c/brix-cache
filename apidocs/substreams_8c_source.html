1/* File: substreams.c — native TPC multi-stream: bind extra data channels
2 * WHAT: tpc_substreams_open() dials up to (streams_requested - 1) additional
3 * sockets to the source, runs the transport bootstrap (handshake +
4 * kXR_protocol, TLS when the source demands it) on each, and joins them to
5 * the primary's session with kXR_bind; tpc_substreams_close() releases them.
6 *
7 * WHY (release-2.0 F7): the pre-F7 pull moved one 1 MiB window at a time over one
8 * socket, so a native TPC could not use the per-stream bandwidth a WAN path
9 * leaves on the table (stock xrootd's `ofs.tpc streams`). Sub-streams are
10 * best-effort: a source without kXR_bind (brix_data_substreams off, an old
11 * server) or one that refuses the join simply leaves the pull single-stream —
12 * a failed extra channel never fails the transfer, and the count is bounded
13 * by brix_tpc_streams on the destination regardless of the client's hint.
14 *
15 * HOW: XRootD wire — the secondary connection performs the ordinary handshake
16 * and kXR_protocol, then sends kXR_bind carrying the 16-byte session id the
17 * primary's kXR_login returned; the server answers kXR_ok with a 1-byte body,
18 * the pathid (1-253) that kXR_read read_args name to steer a reply out of
19 * that channel. Each sub-stream is registered in t->sub[] BEFORE its
20 * bootstrap so the fd-keyed TLS lookups in io.c/tls.c see it. */

32/* WHAT: send kXR_bind on the sub-stream and take the pathid from the reply. */
33static int
34tpc_substream_bind(brix_tpc_pull_t *t, tpc_substream_t *sub)
35{
36 ClientBindRequest br;
37 uint16_t status;
38 uint32_t dlen = 0;
39 u_char *body = NULL;
40
41 ngx_memzero(&br, sizeof(br));
42 ngx_memcpy(br.streamid, tpc_bind_streamid, sizeof(br.streamid));
43 br.requestid = htons(kXR_bind);
44 ngx_memcpy(br.sessid, t->sessid, BRIX_SESSION_ID_LEN);
45
46 if (tpc_send_all(t, sub->fd, &br, sizeof(br)) != 0) {
47 snprintf(t->err_msg, sizeof(t->err_msg), "TPC kXR_bind send failed");
48 return -1;
49 }
50 if (tpc_recv_response(t, sub->fd, &status, &body, &dlen) != 0) {
51 snprintf(t->err_msg, sizeof(t->err_msg), "TPC kXR_bind recv failed");
52 return -1;
53 }
54 if (status != kXR_ok || body == NULL || dlen < 1 || body[0] == 0) {
55 int rerr = 0;
56 const char *rmsg = "";
57 size_t rmsglen = 0;
58
59 if (status == kXR_error) {
60 (void) xrd_error_body_decode(body, dlen, &rerr, &rmsg, &rmsglen);
61 }
62 snprintf(t->err_msg, sizeof(t->err_msg),
63 "TPC kXR_bind rejected (status=%u code=%d) %.*s",
64 (unsigned) status, rerr, (int) rmsglen, rmsg);
65 free(body);
66 return -1;
67 }
68 sub->pathid = body[0];
69 free(body);
70 return 0;
71}

85/* WHAT: connect + transport-bootstrap + bind one more sub-stream into the next
86 * free slot. Returns 0 with t->nsub advanced, -1 (slot released, t->err_msg
87 * set) when the source would not give us another channel. */
88static int
89tpc_substream_open_one(brix_tpc_pull_t *t)
90{
91 tpc_substream_t *sub = &t->sub[t->nsub];
92 int fd = tpc_connect(t);
93
94 if (fd < 0) {
95 return -1;
96 }
97 ngx_memzero(sub, sizeof(*sub));
98 sub->fd = fd;
99 t->nsub++; /* visible to the fd-keyed TLS lookups from here */
100
101 if (tpc_bootstrap_transport(t, fd) != 0
102 || tpc_substream_bind(t, sub) != 0)
103 {
104 tpc_substream_release(t, sub); /* while the slot is still visible */
105 t->nsub--;
106 return -1;
107 }
108 return 0;
109}

111int
112tpc_substreams_open(brix_tpc_pull_t *t, ngx_log_t *log)
113{
114 int want = t->streams_requested - 1;
115 int saved_err = t->xrd_error;
116 char reason[sizeof(t->err_msg)];
117
118 if (want <= 0) {
119 return 0;
120 }
121 if (want > TPC_SUBSTREAMS_MAX) {
122 want = TPC_SUBSTREAMS_MAX;
123 }
124 if (!t->sessid_known) {
125 ngx_log_error(NGX_LOG_INFO, log, 0,
126 "brix: TPC multi-stream: source %s returned no session "
127 "id; pulling single-stream", t->src_host);
128 return 0;
129 }
130
131 reason[0] = '\0';
132 while (t->nsub < want) {
133 if (tpc_substream_open_one(t) != 0) {
134 ngx_cpystrn((u_char *) reason, (u_char *) t->err_msg,
135 sizeof(reason));
136 break;
137 }
138 }
139
140 ngx_log_error(NGX_LOG_INFO, log, 0,
141 "brix: TPC multi-stream: %d of %d requested sub-streams "
142 "bound to %s%s%s", t->nsub, want, t->src_host,
143 reason[0] != '\0' ? " - " : "", reason);
144
145 /* A refused extra channel is not a transfer failure: leave the pull's
146 * error state exactly as the caller had it. */
147 t->err_msg[0] = '\0';
148 t->xrd_error = saved_err;
149 return 0;
150}
