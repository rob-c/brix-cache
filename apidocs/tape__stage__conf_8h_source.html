1/*
2 * tape_stage_conf.h — per-server tape/stage directive config.
3 *
4 * WHAT: The `brix_frm_*` directive config struct + its init/merge/watermark
5 * helpers. Embedded as `frm` in ngx_stream_brix_srv_conf_t (name kept for
6 * back-compat) and read by the staging callers (prepare/tape_rest/open) for
7 * enable/stage_ttl/async_recall/stage_wait/control_dir.
8 *
9 * WHY: The FRM-dissolution (§13b, phase-64 P6) moved the staging IMPLEMENTATION to
10 * the composable stage request registry (src/fs/xfer/) + the sd_frm backend, so
11 * src/frm/ was deleted. This header is the config surface that survives: the
12 * directive names + parse/merge, lifted out of the deleted src/frm/frm.h so the
13 * many `brix_frm_*` directives keep working unchanged. The legacy
14 * FRM-implementation knobs (stagecmd/copycmd/residency_cmd/queue_path) are
15 * accepted for config compatibility but no longer drive a bespoke engine; the
16 * purge watermark/cap/interval trio drives the phase-115 W3.2 purge engine.
17 */

25/* 2.0 F4: one `brix_frm_purge_policy {*|<group>} <hi> <lo> [hold <time>]
26 * [polprog]` line. Thresholds are bytes (percent = 0) or ppm of the group's
27 * brix_oss_space quota (percent = 1); the engine scales them at arm time. */

64/* 2.0 F1 (ADR-3b, 2026-09-08): the stage engine, its FAILED-record retry sweep
65 * and the exec MSS adapter are process-wide, so the seven values below are
66 * published ONCE per configuration by the `brix_frm on` server blocks — every
67 * such block must agree on each value it sets (the merge refuses a
68 * disagreement with [emerg]). Unset values fall back to the defaults below,
69 * which reproduce the pre-2.0 built-in behaviour. */
70#define BRIX_FRM_COPYMAX_DEFAULT 8 /* was STAGE_MAX_INFLIGHT */
71#define BRIX_FRM_FAIL_RETRIES_DEFAULT 5 /* was the deny attempt cap */
72#define BRIX_FRM_FAIL_BACKOFF_DEFAULT_MS 60000
73#define BRIX_FRM_FAIL_BACKOFF_MIN_MS 1000 /* sweep period floor */
