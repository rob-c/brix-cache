1/*
2 * stream_variables.c — the nginx-variable surface for the root:// stream plane
3 * (phase 106 W2; renamed to the uniform $brix_* vocabulary by phase 110 W2,
4 * and the phase-106 $brix_session_* aliases removed by phase 112).
5 *
6 * WHAT: Registers the session-scoped variables an operator needs to write a
7 * `stream {}` access_log for root:// traffic, and implements their
8 * get_handlers.
9 *
10 * WHY: Before this file the stream plane registered NO variables at all
11 * (ngx_stream_add_variable appeared nowhere in the tree), so the
12 * flagship protocol was invisible to nginx's own access_log. Operators
13 * could see aggregate Prometheus counters or unstructured error-log
14 * lines, and nothing else: there was no way to answer "who moved what,
15 * over which session" with the logging tool they already use.
16 *
17 * HOW: SESSION scope, deliberately. A stream session is long-lived and
18 * carries many XRootD ops, so a variable evaluated at log time describes
19 * the SESSION, not an op — "the path" or "the status" would be
20 * ill-defined, and inventing a per-op stream log is a separate problem
21 * nginx's access_log cannot express. Every variable here is therefore a
22 * session total or a session-stable identity, which is exactly what a
23 * single access_log line at session close should carry.
24 *
25 * Handler rules, per phase 106's variable-handler trap:
26 * 1. ctx may be NULL — a connection can be closed before the module
27 * attaches its context (a port scan, a TLS failure, a client that
28 * hangs up mid-handshake). Every handler answers "-" instead of
29 * dereferencing.
30 * 2. Values are either static strings or copied into the CONNECTION
31 * pool, which outlives the log phase. Never the request pool: there
32 * isn't one on the stream plane.
33 * 3. Identity is session-stable once login completes, so these are
34 * cacheable; the byte totals change as the session runs and are
35 * marked NOCACHEABLE.
36 *
37 * SECURITY: these expose the SUBJECT of an identity (DN, VO, login
38 * user), never the credential that proved it. ctx->bearer_token and the
39 * GSI/sigver key material are deliberately absent and must stay absent —
40 * a variable is loggable, and a logged bearer token is a credential
41 * leak.
42 */

60 *

62 * the cloned-logic shape the duplication guard rejects, and nginx already
63 * provides the mechanism to avoid it: the `data` member of ngx_stream_variable_t
64 * is passed to the handler, so one handler can serve every variable of a given
65 * shape with `data` selecting the field.

92/*
93 * brix_stream_session_auth — how this session authenticated.
94 *
95 * Deliberately NOT login.logged_in / login.auth_done. Those are LIVE
96 * authorization state, and kXR_endsess clears both on purpose
97 * (session/lifecycle.c:96-97) so a client cannot keep using the connection
98 * after ending its session. A well-behaved client therefore ends its session
99 * before closing, and by the time nginx writes the access-log line both flags
100 * read 0 — every real transfer would be logged as "brix never ran". A log
101 * variable needs the HISTORICAL fact, not the current permission, so this
102 * reads evidence that is never revoked: the presented login name, and is_bound
103 * for the kXR_bind parallel DATA channel, which inherits the primary session's
104 * auth and carries no login of its own (and is where the bytes actually move).
105 */
106static const char *
107brix_stream_session_auth(const brix_ctx_t *ctx)
108{
109 ngx_uint_t bit;
110
111 /* Derive WHICH method from historical session evidence (the comment above),
112 * then render it through the SHARED name function so the stream
113 * $brix_auth_method speaks the exact vocabulary the HTTP variable, the JSON
114 * "auth_method" key and the brix_auth_total{method} label speak (phase-110
115 * rule 2 / R12). */
116 if (ctx->token.auth) {
117 bit = BRIX_AUTHN_TOKEN;
118 } else if (ctx->login.dn[0] != '\0') {
119 bit = BRIX_AUTHN_GSI;
120 } else if (ctx->login.user[0] != '\0' || ctx->is_bound) {
121 bit = BRIX_AUTHN_NONE; /* → "none" */
122 } else {
123 return NULL; /* caller reports the sentinel */
124 }
125 return brix_metric_auth_method_name(bit);
126}

191/*
192 * Identity on the stream plane (phase-110 W2): the SAME meaning as the HTTP
193 * variables. The canonical source is ctx->identity (the phase-2 identity object
194 * every plane populates at authn); the legacy login.* fields are the fallback
195 * for a session that authenticated through a path that filled only them, so a
196 * DN or VO carried only on the login record is still reported.
197 */
198static const char *
199brix_stream_identity_field(ngx_stream_session_t *sess, const brix_ctx_t *ctx,
200 brix_stream_var_e which)
201{
202 const brix_identity_t *id = ctx->identity;
203
204 switch (which) {
205 case BRIX_SV_DN:
206 return brix_stream_id_or(id != NULL ? brix_identity_dn_cstr(id) : NULL,
207 ctx->login.dn);
208 case BRIX_SV_VO:
209 /* The FULL VO list; the login.primary_vo fallback covers a session
210 * whose VO reached only the login record. */
211 return brix_stream_id_or(id != NULL ? brix_identity_vo_csv_cstr(id)
212 : NULL, ctx->login.primary_vo);
213 case BRIX_SV_FQAN:
214 /* The OPERATIVE FQAN — the FIRST field of the VO list, matching the
215 * HTTP handler's "first verified entry", NOT the whole list. */
216 return brix_stream_first_vo(sess, id, ctx);
217 case BRIX_SV_SUB:
218 return brix_stream_subject(id, ctx);
219 case BRIX_SV_ISSUER:
220 return brix_stream_issuer(id); /* root:// rarely carries one → "-" */
221 case BRIX_SV_USER:
222 return brix_stream_mapped_user(id);
223 default:
224 return NULL;
225 }
226}

229/* The operative FQAN for $brix_fqan: the first comma-separated field of the
230 * identity's VO list (matching the HTTP handler's first-vo_list-entry rule),
231 * else the phase-106 login.primary_vo, else NULL (sentinel). A multi-VO CSV is
232 * truncated at the first comma into a connection-pool copy so the value is a
233 * single NUL-terminated FQAN, never the whole list. */
234static const char *
235brix_stream_first_vo(ngx_stream_session_t *s, const brix_identity_t *id,
236 const brix_ctx_t *ctx)
237{
238 const char *csv = id != NULL ? brix_identity_vo_csv_cstr(id) : NULL;
239 const char *comma;
240 char *first;
241 size_t n;
242
243 if (csv == NULL || csv[0] == '\0') {
244 return ctx->login.primary_vo[0] != '\0' ? ctx->login.primary_vo : NULL;
245 }
246 comma = strchr(csv, ',');
247 if (comma == NULL) {
248 return csv; /* single field, already terminated */
249 }
250 n = (size_t) (comma - csv);
251 first = ngx_pnalloc(s->connection->pool, n + 1);
252 if (first == NULL) {
253 return csv; /* degrade to the full list */
254 }
255 ngx_memcpy(first, csv, n);
256 first[n] = '\0';
257 return first;
258}

261/* Session-stable transport/config string fields. */
262static const char *
263brix_stream_conf_field(ngx_stream_session_t *s, const brix_ctx_t *ctx,
264 brix_stream_var_e which)
265{
266 ngx_http_brix_shared_conf_t *c;
267
268 switch (which) {
269 case BRIX_SV_PROTOCOL:
270 return (ctx->protocol_label[0] != '\0') ? ctx->protocol_label : "root";
271 case BRIX_SV_AUTH:
272 return brix_stream_session_auth(ctx);
273 case BRIX_SV_TLS:
274 return (s->connection != NULL && s->connection->ssl != NULL)
275 ? "on" : "off";
276 case BRIX_SV_TIER:
277 c = brix_stream_common_conf(s);
278 /* NULL instance = default POSIX backend (vfs.h), as on the HTTP plane. */
279 return c == NULL ? NULL
280 : (c->storage_instance != NULL
281 ? brix_sd_backend_name(c->storage_instance) : "posix");
282 case BRIX_SV_ORIGIN:
283 c = brix_stream_common_conf(s);
284 /* A root:// origin is a local export path with no user:pass@ userinfo,
285 * so no stripping is needed; it is NUL-terminated storage. */
286 return (c != NULL && c->storage_backend.len > 0)
287 ? (const char *) c->storage_backend.data : NULL;
288 default:
289 return NULL;
290 }
291}

294/* Primary-op string facts the session's I/O monitor recorded. */
295static const char *
296brix_stream_monitor_field(const brix_ctx_t *ctx, brix_stream_var_e which)
297{
298 const brix_io_monitor_t *m = &ctx->io_monitor;
299
300 switch (which) {
301 case BRIX_SV_OP:
302 return m->have_op ? brix_metric_op_name(m->op) : NULL;
303 case BRIX_SV_STATUS:
304 return (m->have_op || m->err != BRIX_ERR_NONE)
305 ? brix_metric_err_name(m->err) : NULL;
306 case BRIX_SV_CACHE:
307 return m->cache != BRIX_CACHE_STATUS_NONE
308 ? brix_metric_cache_status_name(m->cache) : NULL;
309 case BRIX_SV_PATH:
310 return m->path_len > 0 ? m->path : NULL;
311 case BRIX_SV_CHECKSUM:
312 return m->have_checksum ? m->checksum : NULL;
313 default:
314 return NULL;
315 }
316}

350/* $brix_protocol, the identity set, $brix_{auth,tls,tier,origin,op,status,
351 * cache,path,checksum} — every string-valued stream variable. */
352static ngx_int_t
353brix_stream_var_str(ngx_stream_session_t *s, ngx_stream_variable_value_t *v,
354 uintptr_t data)
355{
356 brix_stream_var_e which = (brix_stream_var_e) data;
357 brix_ctx_t *ctx = brix_stream_var_ctx(s);
358 const char *val;
359
360 if (ctx == NULL) {
361 /* $brix_tls needs no ctx: the transport is a property of the
362 * connection, and reporting "-" for a TLS connection brix never got to
363 * serve would be wrong. */
364 if (which == BRIX_SV_TLS) {
365 return brix_stream_var_cstr(s, v,
366 (s->connection != NULL && s->connection->ssl != NULL)
367 ? "on" : "off", 0);
368 }
369 return brix_stream_var_none_value(v, 0);
370 }
371
372 val = brix_stream_is_identity(which)
373 ? brix_stream_identity_field(s, ctx, which)
374 : brix_stream_str_field(s, ctx, which);
375 /* The monitor-derived fields (op/status/cache/path/checksum) change as the
376 * session proceeds, so they are NOCACHEABLE; the session-stable ones are
377 * cacheable. The flag on the variable-table row governs nginx's caching;
378 * here we just publish the value. */
379 return brix_stream_var_cstr(s, v, val, 0);
380}

414/* Render `usec` microseconds as seconds.mmm (the $request_time / $session_time
415 * shape) for $brix_backend_time and $brix_duration. */
416static ngx_int_t
417brix_stream_var_seconds(ngx_stream_session_t *s, ngx_stream_variable_value_t *v,
418 ngx_msec_int_t ms)
419{
420 u_char *p;
421
422 p = ngx_pnalloc(s->connection->pool, NGX_TIME_T_LEN + 4);
423 if (p == NULL) {
424 return brix_stream_var_none_value(v, 1);
425 }
426 ms = ngx_max(ms, 0);
427 v->len = (unsigned) (ngx_sprintf(p, "%T.%03M", (time_t) ms / 1000,
428 ms % 1000) - p);
429 v->valid = 1;
430 v->no_cacheable = 1;
431 v->not_found = 0;
432 v->data = p;
433 return NGX_OK;
434}

453/* $brix_duration — wall time of the session, byte-identical to nginx's
454 * $session_time (rule 6: the one transport fact brix twins, because nginx
455 * spells it $request_time on HTTP and $session_time on stream). */
456static ngx_int_t
457brix_stream_var_duration(ngx_stream_session_t *s,
458 ngx_stream_variable_value_t *v, uintptr_t data)
459{
460 ngx_time_t *tp = ngx_timeofday();
461
462 (void) data;
463 return brix_stream_var_seconds(s, v,
464 (ngx_msec_int_t) ((tp->sec - s->start_sec) * 1000
465 + (tp->msec - s->start_msec)));
466}
