23/* WHAT: Login without invoking a client authentication module.
24 * WHY: The negative must send intentionally invalid material even when the host
25 * has no real credential for the advertised security protocol.
26 * HOW: Handshake only, then issue the standard anonymous login and retain its

28static int
29auth_fixture_login(xrd_conform_execution *execution, const brix_url *url,
30 brix_conn *connection, char security[256], brix_status *status)
31{
32 ClientLoginRequest login;
33 brix_opts options = execution->run->opts;
34 const uint8_t zero_sid[2] = { 0, 0 };
35 uint16_t wire_status = kXR_error;
36 uint8_t *body = NULL;
37 uint32_t body_length = 0;
38 size_t security_length;
39 int rc;
40
41 options.capture = NULL;
42 if (brix_connect_no_login(connection, url, &options, status) != 0) return -1;
43 xrd_pack_login_request(&login, zero_sid, 1, "conform",
44 (uint8_t) (kXR_ver005 | kXR_asyncap));
45 rc = auth_exchange(execution, connection, (ClientRequestHdr *) &login,
46 NULL, &wire_status, &body, &body_length, status);
47 if (rc != 0 || wire_status != kXR_ok
48 || body_length <= BRIX_SESSION_ID_LEN) {
49 free(body);
50 return -1;
51 }
52 security_length = body_length - BRIX_SESSION_ID_LEN;
53 if (security_length >= 256) security_length = 255;
54 memcpy(security, body + BRIX_SESSION_ID_LEN, security_length);
55 security[security_length] = '\0';
56 free(body);
57 return 0;
58}

72/* WHAT: Require a labelled probe for every protocol any fixture advertises.
73 * WHY: Omitting a difficult mechanism must not turn "every enabled protocol"
74 * into a vacuous pass over an operator-selected subset.
75 * HOW: Parse each anchored &P=name entry with the shared grammar boundaries and
76 * require its exact name to exist in the complete fixture vector. */
77int
78xrd_conform_auth_fixtures_cover(const xrd_conform_run *run,
79 const char *security)
80{
81 const char *entry = security;
82 size_t count = 0;
83
84 while ((entry = strstr(entry, "&P=")) != NULL) {
85 const char *name = entry + 3;
86 const char *end = name;
87
88 while (*end != '\0' && *end != ',' && *end != '&') end++;
89 if (end == name || !auth_fixture_named(run, name,
90 (size_t) (end - name))) return 0;
91 count++;
92 entry = end;
93 }
94 return count != 0;
95}

97/* WHAT: Send malformed credentials for one explicitly advertised protocol.
98 * WHY: ERR-047 tests the server's wire classification, not local credential
99 * discovery or the client's ability to complete a successful login.

102static int
103auth_failure_matches(xrd_conform_execution *execution,

113 char security[256];
114 uint16_t wire_status = kXR_error;

138 xrdw_auth_req_pack(&body, ((ClientRequestHdr *) &auth)->body);
139 rc = auth_exchange(execution, &connection, (ClientRequestHdr *) &auth,
140 &payload, &wire_status, &reply, &reply_length, &status);
141 if (rc != 0)

143 || status.kxr == kXR_NotAuthorized;
144 free(reply);

147}

149/* WHAT: Grade invalid credentials for every explicitly enabled protocol.
150 * WHY: A generic FSError hides an authentication failure and defeats clients'
151 * security-aware retry and diagnostic behavior.
152 * HOW: Require every labelled fixture to advertise its label and return one of
153 * the two permitted authentication-class wire codes. */
154void
155xrd_conform_probe_auth_fixtures(xrd_conform_execution *execution)
156{
157 const brix_conform_result *row;
158 size_t index;
159 size_t matched = 0;
160 char received[80];
161
162 row = xrd_conform_result_find(execution, "ERR-047");
163 if (row == NULL || row->verdict == BRIX_CONFORM_VERDICT_SKIPPED
164 || execution->run->auth_fixture_count == 0) return;
165 for (index = 0; index < execution->run->auth_fixture_count; index++)
166 if (auth_failure_matches(execution,
167 &execution->run->auth_fixtures[index])) matched++;
168 snprintf(received, sizeof(received), "protocols=%zu exact=%zu",
169 execution->run->auth_fixture_count, matched);
170 xrd_conform_result_set(execution, "ERR-047",
171 matched == execution->run->auth_fixture_count
172 ? BRIX_CONFORM_VERDICT_CONFORMANT
173 : BRIX_CONFORM_VERDICT_DIVERGENT,
174 matched == execution->run->auth_fixture_count
175 ? "all-supplied-auth-protocols-use-auth-class-errors"
176 : "auth-protocol-returned-non-auth-class-error",
177 "each enabled protocol returns kXR_AuthFailed or kXR_NotAuthorized",
178 received);
179}
