12static int
13message_is_human_readable(const char *message)
14{
15 const unsigned char *cursor = (const unsigned char *) message;
16
17 if (cursor == NULL || *cursor == '\0') return 0;
18 while (*cursor != '\0') {

56 && message_is_human_readable(status.msg),

59 brix_status_clear(&repeated);

62 xrd_conform_basic_grade(execution, "ERR-041",

68static int
69strict_unsigned(const char *value)
70{
71 const unsigned char *cursor = (const unsigned char *) value;
72
73 if (cursor == NULL || *cursor == '\0') return 0;
74 while (*cursor != '\0' && *cursor != '\n') {
75 if (*cursor < '0' || *cursor > '9') return 0;
76 cursor++;
77 }
78 return *cursor == '\0' || cursor[1] == '\0';
79}

114static void
115probe_query(xrd_conform_execution *execution, brix_conn *connection,
116 const char *path)
117{
118 char reply[512];

120 char reason[64];
121 brix_status status;
122 unsigned long long total;
123 unsigned long long free_bytes;
124 int rc;
125
126 if (!xrd_conform_basic_reserve(execution, reason)) return;
127 brix_status_clear(&status);
128 rc = brix_query(connection, kXR_Qconfig, "tpc", reply, sizeof(reply),
129 &status);
130 xrd_conform_basic_grade(execution, "QRY-002",
131 rc == 0 && strict_unsigned(reply), "qconfig-tpc-is-numeric", &status);
132 if (!xrd_conform_basic_reserve(execution, reason)) return;
133 brix_status_clear(&status);
134 rc = brix_query_cksum(connection, path, "sha256", digest, sizeof(digest),
135 &status);
136 if (rc == 0) {
137 int checksum_ok = strcmp(digest, execution->root_digest) == 0;
138
139 xrd_conform_observe_root_checksum(execution, digest, checksum_ok);
140 xrd_conform_basic_grade(execution, "QRY-004", checksum_ok,
141 "server-checksum-matches-adopted-bytes", &status);
142 } else
143 xrd_conform_result_set(execution, "QRY-004",
144 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
145 status.kxr == kXR_Unsupported ? "sha256-not-advertised"
146 : "checksum-query-failed",
147 NULL, status.msg);
148 probe_crc_variants(execution, connection, path);
149 if (!xrd_conform_basic_reserve(execution, reason)) return;
150 brix_status_clear(&status);
151 rc = brix_query_cksum(connection, path, "brix-unknown", digest,
152 sizeof(digest), &status);
153 xrd_conform_basic_grade(execution, "QRY-006", rc != 0
154 && (status.kxr == kXR_Unsupported || status.kxr == kXR_ArgInvalid),
155 "unsupported-checksum-refused", &status);
156 if (!xrd_conform_basic_reserve(execution, reason)) return;
157 brix_status_clear(&status);
158 rc = brix_query(connection, kXR_Qspace, path, reply, sizeof(reply),
159 &status);
160 if (rc != 0) {
161 xrd_conform_result_set(execution, "QRY-008",
162 BRIX_CONFORM_VERDICT_INCONCLUSIVE, "qspace-query-failed", NULL,
163 status.msg);
164 return;
165 }
166 brix_parse_qspace(reply, &total, &free_bytes);
167 xrd_conform_basic_grade(execution, "QRY-008",
168 total > 0 && free_bytes <= total,
169 "qspace-free-does-not-exceed-total", &status);
170}
