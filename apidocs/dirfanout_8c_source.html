1/*
2 * fs/dirfanout.c — cluster-wide directory enumeration (W7.2b).
3 *
4 * WHAT: brix_dirlist_all() — the union of one directory's entries across every
5 * data server a CMS manager names for that path, deduplicated by name.
6 *
7 * WHY: A manager answers kXR_dirlist by REDIRECTING to a single registered
8 * data server (src/protocols/root/dirlist/handler.c), so a plain dirlist
9 * against a manager enumerates ONE node's view of the directory, not the
10 * cluster's. Every file whose only replica lives elsewhere is invisible.
11 * Stock XRootD solves this client-side in XrdFfsPosix_readdirall; this is
12 * the same contract for the brix client, so a FUSE mount over a manager
13 * lists what the cluster holds rather than what one node happens to hold.
14 *
15 * HOW: kXR_locate the directory on the manager connection → keep the 'S'/'s'
16 * data-server tokens (manager 'M'/'m' entries are skipped: dialing a
17 * manager would just re-locate) → dial each one with the manager
18 * connection's own opts → brix_dirlist → concatenate → sort by name and
19 * drop adjacent duplicates (a replicated file is named by several nodes).
20 *
21 * Failure policy, deliberately asymmetric:
22 * * a node that answers "not found" contributes nothing and is fine —
23 * a directory need not exist on every server;
24 * * ANY other per-node failure (connect, auth, protocol) fails the
25 * whole call. A silently short listing is worse than an error: the
26 * caller cannot tell a partial answer from a complete one, and
27 * "file missing" is exactly the bug this function exists to fix.
28 */

48/* The node table's host field is the SAME width as the URL host it is copied
49 * into, so that copy is a fixed-size memcpy rather than a bounded string copy

67/* ---- Is this locate token a data server we should ask? ----
68 *
69 * WHAT: 1 for a "S<r|w><host>:<port>" / "s..." token with a non-empty body.
70 *
71 * WHY: 'M'/'m' are managers — asking one re-enters the same redirect we are
72 * working around. Anything else is a token shape we do not model.
73 *
74 * HOW: Leading byte class plus a length check for the two-byte prefix.
75 */
76static int
77dirfan_is_server(const char *token)
78{
79 return (token[0] == 'S' || token[0] == 's')
80 && token[1] != '\0' && token[2] != '\0';
81}

84/* ---- Have we already recorded this endpoint? ----
85 *
86 * WHAT: 1 when host:port is already in the node table.
87 *
88 * WHY: A locate can name the same server twice (read and write entries for
89 * the same holder). Listing it twice would double every name and make
90 * the dedup pass do work that never needed doing.
91 *
92 * HOW: Linear scan — the table is at most DIRFAN_MAX_NODES.
93 */
94static int
95dirfan_seen(const dirfan_node *nodes, size_t n, const brix_url *u)
96{
97 size_t i;
98
99 for (i = 0; i < n; i++) {
100 if (nodes[i].port == u->port
101 && strcmp(nodes[i].host, u->host) == 0) {
102 return 1;
103 }
104 }
105 return 0;
106}

109/* ---- Data servers named by a locate reply ----
110 *
111 * WHAT: Fill `nodes` with the distinct data-server endpoints in `reply`.
112 * Returns the count.
113 *
114 * WHY: The union is taken over exactly these; everything downstream is a
115 * loop over this table.
116 *
117 * HOW: Whitespace-split, keep server tokens, and reuse brix_url_parse for the
118 * host:port split so bracketed IPv6 literals are handled by the one
119 * parser that already knows their shape — never a second hand-rolled
120 * splitter (a token like "Sr[::1]:1094" has colons inside the host).
121 */
122static size_t
123dirfan_nodes_from_locate(char *reply, dirfan_node *nodes, size_t max)
124{
125 char *cursor = reply, *token, *save = NULL;
126 size_t n = 0;
127
128 while (n < max && (token = strtok_r(cursor, " \t\r\n", &save)) != NULL) {
129 char url[512];
130 brix_url u;
131 brix_status ust;
132
133 cursor = NULL;
134 if (!dirfan_is_server(token)) {
135 continue;
136 }
137 if ((size_t) snprintf(url, sizeof(url), "root://%s//", token + 2)
138 >= sizeof(url)) {
139 continue;
140 }
141 brix_status_clear(&ust);
142 if (brix_url_parse(url, &u, &ust) != 0 || u.host[0] == '\0') {
143 continue;
144 }
145 if (dirfan_seen(nodes, n, &u)) {
146 continue;
147 }
148 snprintf(nodes[n].host, sizeof(nodes[n].host), "%s", u.host);
149 nodes[n].port = u.port;
150 n++;
151 }
152 return n;
153}

156/* ---- Append one node's listing to the union ----
157 *
158 * WHAT: Grow the accumulator and copy `add` entries in. 0 / -1 (st set).
159 *
160 * WHY: The union's size is not known until the last node has answered, so the
161 * array doubles rather than being sized from a guess.
162 *
163 * HOW: realloc to the next power-of-two-ish capacity, then memcpy.
164 */
165static int
166dirfan_append(dirfan_acc *acc, const brix_dirent *add, size_t n_add,
167 brix_status *st)
168{
169 size_t want = acc->n + n_add;
170 brix_dirent *grown;
171
172 if (n_add == 0) {
173 return 0;
174 }
175 if (want > acc->cap) {
176 size_t cap = (acc->cap == 0) ? 64 : acc->cap;
177
178 while (cap < want) {
179 cap *= 2;
180 }
181 grown = realloc(acc->ents, cap * sizeof(*grown));
182 if (grown == NULL) {
183 brix_status_set(st, XRDC_EIO, errno,
184 "dirlist-all: out of memory for %zu entries", cap);
185 return -1;
186 }
187 acc->ents = grown;
188 acc->cap = cap;
189 }
190 memcpy(acc->ents + acc->n, add, n_add * sizeof(*add));
191 acc->n = want;
192 return 0;
193}

196/* ---- Did this node simply not have the directory? ----
197 *
198 * WHAT: 1 for a not-found-class failure.
199 *
200 * WHY: A data server that holds no part of a directory is a normal member of
201 * a cluster, not an error — its absence must not fail the enumeration.
202 *
203 * HOW: Same predicate the CLI hint layer uses (cli_hint.c): the wire code or
204 * a locally-mapped ENOENT.
205 */
206static int
207dirfan_not_found(const brix_status *st)
208{
209 return st->kxr == kXR_NotFound || st->sys_errno == ENOENT;
210}

213/* ---- List one data server and fold it into the union ----
214 *
215 * WHAT: Dial `nd`, dirlist acc->path, append. 0 on success or a clean miss,
216 * -1 on any other failure (st carries it).
217 *
218 * WHY: Each holder is asked directly, so the manager's single-node redirect
219 * is bypassed entirely.
220 *
221 * HOW: Compose a URL for the endpoint, connect with the manager connection's
222 * stored opts (same credentials, same TLS posture), dirlist, close. The
223 * connection is closed on every path — a listing must not leak a socket
224 * per node per readdir.
225 */
226static int
227dirfan_one(const brix_conn *mgr, const dirfan_node *nd, dirfan_acc *acc,
228 brix_status *st)
229{
230 brix_conn c;
231 brix_url u;
232 brix_dirent *ents = NULL;
233 size_t n = 0;
234 int rc;
235
236 memset(&u, 0, sizeof(u));
237 u.scheme = mgr->tls_strict ? XRDC_SCHEME_ROOTS : XRDC_SCHEME_ROOT;
238 memcpy(u.host, nd->host, sizeof(u.host)); /* equal widths, see the assert */
239 u.port = nd->port;
240 snprintf(u.path, sizeof(u.path), "%s", acc->path);
241
242 if (brix_connect(&c, &u, &mgr->opts, st) != 0) {
243 return -1;
244 }
245 rc = brix_dirlist(&c, acc->path, acc->want_stat, &ents, &n, st);
246 brix_close(&c);
247 if (rc != 0) {
248 if (dirfan_not_found(st)) {
249 brix_status_clear(st);
250 return 0;
251 }
252 return -1;
253 }
254 rc = dirfan_append(acc, ents, n, st);
255 free(ents);
256 return rc;
257}

269/* ---- Collapse duplicate names in place ----
270 *
271 * WHAT: Sort the union and drop adjacent equal names. Returns the kept count.
272 *
273 * WHY: A replicated file is named by every node holding it; a caller that saw
274 * it twice would render it twice. The FIRST occurrence is kept, so a
275 * stat-bearing entry is never replaced by a later stat-less one from a
276 * node that answered a want_stat=1 dirlist with only names.
277 *
278 * HOW: qsort by name (stable enough for this purpose — duplicates are equal
279 * records to the caller) then a single compaction sweep.
280 */
281static size_t
282dirfan_dedup(brix_dirent *ents, size_t n)
283{
284 size_t i, keep = 0;
285
286 if (n < 2) {
287 return n;
288 }
289 qsort(ents, n, sizeof(*ents), dirfan_cmp);
290 for (i = 1; i < n; i++) {
291 if (strcmp(ents[keep].name, ents[i].name) == 0) {
292 continue;
293 }
294 keep++;
295 if (keep != i) {
296 ents[keep] = ents[i];
297 }
298 }
299 return keep + 1;
300}

303int
304brix_dirlist_all(brix_conn *c, const char *path, int want_stat,
305 brix_dirent **ents, size_t *count, brix_status *st)
306{
307 char reply[DIRFAN_REPLY_MAX];
308 dirfan_node nodes[DIRFAN_MAX_NODES];
309 dirfan_acc acc;
310 size_t n_nodes, i;
311
312 *ents = NULL;
313 *count = 0;
314
315 /* No locate support, or a locate that names nobody: this endpoint is not a
316 * manager fronting other holders, so its own listing IS the answer. */
317 if (brix_locate(c, path, reply, sizeof(reply), st) != 0) {
318 brix_status_clear(st);
319 return brix_dirlist(c, path, want_stat, ents, count, st);
320 }
321 n_nodes = dirfan_nodes_from_locate(reply, nodes, DIRFAN_MAX_NODES);
322 if (n_nodes == 0) {
323 return brix_dirlist(c, path, want_stat, ents, count, st);
324 }
325
326 memset(&acc, 0, sizeof(acc));
327 acc.path = path;
328 acc.want_stat = want_stat;
329
330 for (i = 0; i < n_nodes; i++) {
331 if (dirfan_one(c, &nodes[i], &acc, st) != 0) {
332 free(acc.ents);
333 return -1;
334 }
335 }
336
337 *ents = acc.ents;
338 *count = dirfan_dedup(acc.ents, acc.n);
339 return 0;
340}
