26/* Stream `size` lower bytes of `rel` into fd via the injected reader. A short
27 * read before `size` is a lower-layer lie → -EIO. */
28static int ov_copyup_stream(int fd, const char *rel, uint64_t size,
29 brix_ov_read_fn read_lower, void *ud) {
30 unsigned char *buf = malloc(OV_COPYUP_CHUNK);
31 if (buf == NULL) return -ENOMEM;
32
33 uint64_t off = 0;
34 int rc = 0;
35 while (rc == 0 && off < size) {
36 size_t want = size - off > OV_COPYUP_CHUNK ? OV_COPYUP_CHUNK
37 : (size_t) (size - off);
38 size_t got = 0;
39 rc = read_lower(ud, rel, off, want, buf, &got);
40 if (rc == 0 && got == 0) rc = -EIO; /* premature EOF */
41 for (size_t done = 0; rc == 0 && done < got; ) {
42 ssize_t w = pwrite(fd, buf + done, got - done, (off_t) (off + done)); /* vfs-seam-allow: copy-up write to upper writable overlay layer, not export data */
43 if (w < 0) rc = -errno;
44 else done += (size_t) w;
45 }
46 off += got;
47 }
48 free(buf);
49 return rc;
50}

52int brix_overlay_copyup(const brix_overlay *ov, const char *rel,
53 const struct stat *lower_st,
54 brix_ov_read_fn read_lower, void *ud) {
55 char leaf[OV_NAME_MAX], tmp[OV_NAME_MAX + sizeof(BRIX_OV_TMP_PREFIX)];
56 int parent = ov_walk_parent_mk(ov, rel, leaf, sizeof(leaf), 1, NULL, NULL);
57 if (parent < 0) return parent;
58
59 snprintf(tmp, sizeof(tmp), BRIX_OV_TMP_PREFIX "%s", leaf);
60 int fd = openat(parent, tmp, O_WRONLY | O_CREAT | O_TRUNC | O_NOFOLLOW | O_CLOEXEC,
61 lower_st->st_mode & 07777);
62 if (fd < 0) { int e = errno; close(parent); return -e; }
63
64 /* the openat mode is umask-filtered — restate the exact lower bits */
65 int rc = fchmod(fd, lower_st->st_mode & 07777) != 0 ? -errno : 0;
66 if (rc == 0)
67 rc = ov_copyup_stream(fd, rel, (uint64_t) lower_st->st_size, read_lower, ud);
68 if (rc == 0) {
69 struct timespec tv[2] = { { lower_st->st_mtime, 0 }, { lower_st->st_mtime, 0 } };
70 if (futimens(fd, tv) != 0) rc = -errno;
71 }
72 close(fd);
73
74 if (rc == 0 && renameat(parent, tmp, parent, leaf) != 0) rc = -errno;
75 if (rc != 0) unlinkat(parent, tmp, 0); /* leave no torn trace */
76 close(parent);
77
78 return rc == 0 ? brix_overlay_whiteout_clear(ov, rel) : rc;
79}

83static int ov_nameset_add(brix_ov_nameset *s, char flag, const char *name) {
84 size_t need = strlen(name) + 2; /* flag + name + NUL */
85 if (s->used + need > s->cap) {
86 size_t ncap = s->cap ? s->cap * 2 : 512;
87 while (ncap < s->used + need) ncap *= 2;
88 char *nb = realloc(s->buf, ncap);
89 if (nb == NULL) return -ENOMEM;
90 s->buf = nb;
91 s->cap = ncap;
92 }
93 s->buf[s->used] = flag;
94 memcpy(s->buf + s->used + 1, name, need - 1);
95 s->used += need;
96 s->count++;
97 return 0;
98}

100int brix_overlay_read_upper(const brix_overlay *ov, const char *rel,
101 brix_ov_nameset *set, int *opaque) {
102 memset(set, 0, sizeof(*set));
103 *opaque = 0;
104
105 int dir = ov_open_dir(ov, rel);
106 if (dir == -ENOENT || dir == -ENOTDIR) return 0; /* nothing in upper */
107 if (dir < 0) return dir;
108
109 DIR *d = fdopendir(dir); /* owns dir from here */
110 if (d == NULL) { int e = errno; close(dir); return -e; }
111
112 int rc = 0;
113 struct dirent *e;
114 while (rc == 0 && (e = readdir(d)) != NULL) {
115 const char *n = e->d_name;
116 if (strcmp(n, ".") == 0 || strcmp(n, "..") == 0) continue;
117 if (strcmp(n, BRIX_OV_OPQ_NAME) == 0) { *opaque = 1; continue; }
118 if (strncmp(n, BRIX_OV_TMP_PREFIX, sizeof(BRIX_OV_TMP_PREFIX) - 1) == 0) continue;
119 if (strncmp(n, BRIX_OV_WH_PREFIX, sizeof(BRIX_OV_WH_PREFIX) - 1) == 0)
120 rc = ov_nameset_add(set, 'w', n + sizeof(BRIX_OV_WH_PREFIX) - 1);
121 else
122 rc = ov_nameset_add(set, 'u', n);
123 }
124 closedir(d);
125 if (rc != 0) brix_ov_nameset_free(set);
126 return rc;
127}

159/* Wrong-mountpoint guard: <mountdir>/.brixwrites/upper must be a directory.
160 * On success *upper_out (asprintf'd, caller frees) holds its path. */
161static int ov_cli_guard(const char *mountdir, char **upper_out) {
162 if (asprintf(upper_out, "%s/" BRIX_OV_DIRNAME "/" BRIX_OV_UPPER_DIRNAME,
163 mountdir) < 0)
164 return 1;
165 struct stat st;
166 if (lstat(*upper_out, &st) == 0 && S_ISDIR(st.st_mode)) return 0;
167 fprintf(stderr, "brixMount: no " BRIX_OV_DIRNAME
168 " overlay under %s (not a cvmfs-rw mountpoint?)\n", mountdir);
169 free(*upper_out);
170 *upper_out = NULL;
171 return 2;
172}

211/* ---- Emit one upper-tree entry during --overlay-list ----
212 *
213 * WHAT: Classifies dirent `name` in the upper directory `dirp` (union path
214 * `rel`): "." "..", the opaque marker and tmp files are skipped; a
215 * whiteout prints a "deleted" line; a subdirectory prints "dir" and
216 * recurses; any other entry prints its change kind. Returns 0, or 1 on
217 * an allocation failure (which stops the enclosing scan).
218 *
219 * WHY: Splitting the per-entry decision ladder out of ov_cli_list_dir keeps
220 * each function small and single-purpose while preserving the exact
221 * skip/whiteout/recurse ordering the listing relies on.
222 *
223 * HOW: 1. Return early for reserved/skipped names.
224 * 2. Print and return for a whiteout marker.
225 * 3. Build the child union path and absolute path; on OOM free and fail.
226 * 4. Recurse into a subdirectory, else print the file's change kind.
227 */
228static int ov_cli_list_entry(const char *mountdir, const char *upper_root,
229 const char *rel, const char *dirp,
230 const char *name, FILE *out) {
231 if (strcmp(name, ".") == 0 || strcmp(name, "..") == 0) return 0;
232 if (strcmp(name, BRIX_OV_OPQ_NAME) == 0) return 0;
233 if (strncmp(name, BRIX_OV_TMP_PREFIX, sizeof(BRIX_OV_TMP_PREFIX) - 1) == 0) return 0;
234
235 if (strncmp(name, BRIX_OV_WH_PREFIX, sizeof(BRIX_OV_WH_PREFIX) - 1) == 0) {
236 fprintf(out, "deleted %s%s%s\n", rel, rel[0] ? "/" : "",
237 name + sizeof(BRIX_OV_WH_PREFIX) - 1);
238 return 0;
239 }
240
241 char *crel = NULL, *full = NULL;
242 if (asprintf(&crel, "%s%s%s", rel, rel[0] ? "/" : "", name) < 0
243 || asprintf(&full, "%s/%s", dirp, name) < 0) {
244 free(crel);
245 return 1;
246 }
247
248 int rc = 0;
249 struct stat st;
250 if (lstat(full, &st) == 0 && S_ISDIR(st.st_mode)) {
251 fprintf(out, "dir %s\n", crel);
252 rc = ov_cli_list_dir(mountdir, upper_root, crel, out);
253 } else {
254 char kind[64];
255 ov_cli_kind(mountdir, crel, kind, sizeof(kind));
256 fprintf(out, "%s %s\n", kind, crel);
257 }
258 free(crel);
259 free(full);
260 return rc;
261}

272/* Recursively delete everything inside dirfd (never following symlinks). */
273static int ov_cli_reset_contents(int dirfd) {
274 int lfd = openat(dirfd, ".", O_RDONLY | O_DIRECTORY | O_CLOEXEC);
275 if (lfd < 0) return 1;
276 DIR *d = fdopendir(lfd); /* owns lfd */
277 if (d == NULL) { close(lfd); return 1; }
278
279 int rc = 0;
280 struct dirent *e;
281 while (rc == 0 && (e = readdir(d)) != NULL) {
282 const char *n = e->d_name;
283 if (strcmp(n, ".") == 0 || strcmp(n, "..") == 0) continue;
284 struct stat st;
285 if (fstatat(dirfd, n, &st, AT_SYMLINK_NOFOLLOW) != 0) { rc = 1; break; }
286 if (S_ISDIR(st.st_mode)) {
287 int sub = openat(dirfd, n, O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC);
288 if (sub < 0) { rc = 1; break; }
289 rc = ov_cli_reset_contents(sub);
290 close(sub);
291 if (rc == 0 && unlinkat(dirfd, n, AT_REMOVEDIR) != 0) rc = 1;
292 } else if (unlinkat(dirfd, n, 0) != 0) {
293 rc = 1;
294 }
295 }
296 closedir(d);
297 return rc;
298}
