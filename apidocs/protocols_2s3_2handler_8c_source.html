1/*
2 * handler.c — S3 content handler: URI parsing, SigV4 auth gate, and method dispatch.
3 *
4 * WHAT: The nginx HTTP content handler for all S3 REST API operations. Every S3 request
5 * enters here — it parses the path-style URI (/<bucket>/<key>), verifies the AWS
6 * SigV4 signature, resolves the key to a filesystem path, and dispatches to the
7 * appropriate method handler (GetObject, PutObject, DeleteObject, ListObjectsV2, etc.).
8 *
9 * WHY: S3 clients use path-style URIs that must be parsed into bucket + key before any
10 * filesystem operations. The SigV4 auth gate rejects unsigned requests early without
11 * parsing overhead. Write operations check cf->common.allow_write before body read to reject
12 * writes on read-only endpoints without consuming client bandwidth.
13 *
14 * HOW:
15 * 1. Check cf->common.enable → NGX_DECLINED if disabled
16 * 2. Determine method_slot (list vs object) + track bytes_rx metric per IP version
17 * 3. Verify SigV4 signature — fail fast with XML error on invalid
18 * 4. Parse URI into bucket+key via s3_parse_uri()
19 * 5. Dispatch list requests (ListObjectsV2, ListMultipartUploads, ListParts)
20 * 6. Check special empty-key flags (uploads → InitiateMPU, delete → DeleteObjects)
21 * 7. Reject bare GET /<bucket>/? (empty key without flag) as InvalidURI
22 * 8. Resolve key to fs_path via s3_resolve_key() — AccessDenied on escape
23 * 9. Dispatch by HTTP method: GET/HEAD/PUT/DELETE/POST → specific handler
24 * 10. Unknown methods → 405 Method Not Allowed
25 *
26 * Pool allocation: path_copy uses ngx_pnalloc(r->pool, PATH_MAX) to survive until the
27 * async PUT/MPU-complete callback fires after body reading completes.
28 *
29 * Phase-79 file-size split: the URI parser and the post-auth dispatch tree moved to
30 * the sibling handler_dispatch.c (URI parse, token-scope gate, bucket/empty-key/list
31 * routing) and handler_object_route.c (object-key method routing). This file retains
32 * the entry handler, the SigV4/XrdAcc auth gate, request classification, OPTIONS/CORS,
33 * and the delegation/pmark hooks. Cross-file symbols are declared in
34 * s3_handler_internal.h.
35 */

56/* Map an S3 request method to the XrdAcc operation it requires. */
57brix_acc_op_t
58s3_method_aop(ngx_http_request_t *r)
59{
60 switch (r->method) {
61 case NGX_HTTP_GET: return BRIX_AOP_READ; /* GetObject / ListObjects */
62 case NGX_HTTP_HEAD: return BRIX_AOP_STAT;
63 case NGX_HTTP_PUT: return BRIX_AOP_CREATE;
64 case NGX_HTTP_POST: return BRIX_AOP_CREATE; /* multipart upload */
65 case NGX_HTTP_DELETE: return BRIX_AOP_DELETE;
66 default: return BRIX_AOP_STAT;
67 }
68}

70/*
71 * s3_acc_check — XrdAcc tier for S3 (when `brix_acc_format xrdacc`).
72 * Returns NGX_OK (allow / not selected) or NGX_HTTP_FORBIDDEN (deny).
73 */
74static ngx_int_t
75s3_acc_check(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
76 brix_identity_t *id)
77{
78 const char *name = "", *vorg = "", *role = "", *grp = "";
79 char host[64], path[1024];
80 size_t n;
81 ngx_int_t rc;
82
83 if (cf->common.acc.format != BRIX_AUTHDB_FORMAT_XRDACC) {
84 return NGX_OK;
85 }
86 if (id != NULL) {
87 name = brix_identity_dn_cstr(id); /* S3 access key (or subject) */
88 vorg = brix_identity_acc_vorg_cstr(id);
89 role = brix_identity_acc_role_cstr(id);
90 grp = brix_identity_acc_group_cstr(id);
91 }
92 n = ngx_min(r->connection->addr_text.len, sizeof(host) - 1);
93 ngx_memcpy(host, r->connection->addr_text.data, n);
94 host[n] = '\0';
95
96 /* Opt-in reverse DNS for `h <host>`/`h .domain` rules (per request). */
97 if (cf->common.acc.resolve_hosts) {
98 char hbuf[256];
99 const char *h = brix_acc_resolve_peer(r->connection->sockaddr,
100 r->connection->socklen,
101 hbuf, sizeof(hbuf));
102 if (h != NULL) {
103 n = ngx_min(ngx_strlen(h), sizeof(host) - 1);
104 ngx_memcpy(host, h, n);
105 host[n] = '\0';
106 }
107 }
108
109 n = ngx_min(r->uri.len, sizeof(path) - 1);
110 ngx_memcpy(path, r->uri.data, n);
111 path[n] = '\0';
112
113 rc = brix_acc_http_authorize(r->pool, r->connection->log,
114 &cf->common.acc, name, host, vorg, role, grp,
115 s3_method_aop(r), path);
116 return (rc == NGX_ERROR) ? NGX_HTTP_FORBIDDEN : NGX_OK;
117}

119/*
120 *
121 * WHAT: Detects whether the HTTP request is a ListObjectsV2 operation by searching for the
122 * "list-type=2" query parameter in the request args. Returns 1 if found, 0 otherwise.
123 * Only checks GET requests with non-empty query arguments — other methods cannot be lists.
124 *
125 * WHY: ListObjectsV2 has fundamentally different filesystem behavior from GetObject (it returns XML
126 * listing of objects rather than serving a single file). This detection allows the main handler
127 * to dispatch to separate metrics slots and distinct handler functions for list vs object operations. */
128/*
129 * HOW: Checks that r->method is GET and reads the exact list-type query
130 * parameter with the shared query parser. Returns 1 only when list-type=2.
131 */
132/*
133 * s3_rate_limit — phase-105 W1: the [brix_rate_limit] token-bucket gate,
134 * byte-parallel to webdav's access_rate_limit. Runs BEFORE the auth burden
135 * (SigV4/bearer evaluation) and keys by client IP — the HTTP-plane keying
136 * (rate_limit.h: "key=ip … used before identity is known"); key=dn keying is
137 * stream-plane semantics. kv == NULL (directive unset) = no limit.
138 * Before this gate existed, brix_rate_limit in an S3 location parsed cleanly
139 * and enforced NOTHING (the 101-W1 silent-no-op class, on a DoS knob).
140 */
141static ngx_int_t
142s3_rate_limit(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf)
143{
144 ngx_str_t *ip;
145
146 if (cf->common.rate_limit.kv == NULL || r->connection == NULL) {
147 return NGX_OK;
148 }
149
150 ip = &r->connection->addr_text;
151
152 if (brix_rate_limit_check(&cf->common.rate_limit,
153 (const char *) ip->data, ip->len)
154 != NGX_OK)
155 {
156 return NGX_HTTP_TOO_MANY_REQUESTS;
157 }
158
159 return NGX_OK;
160}

176static ngx_flag_t
177s3_is_post_object_form(ngx_http_request_t *r)
178{
179 ngx_table_elt_t *ct;
180
181 if (r->method != NGX_HTTP_POST || r->args.len != 0) {
182 return 0;
183 }
184
185 ct = brix_http_find_header(r, "Content-Type",
186 sizeof("Content-Type") - 1);
187 if (ct == NULL || ct->value.len < sizeof("multipart/form-data") - 1) {
188 return 0;
189 }
190
191 return ngx_strncasecmp(ct->value.data, (u_char *) "multipart/form-data",
192 sizeof("multipart/form-data") - 1) == 0;
193}

208static ngx_int_t
209s3_add_preflight_headers(ngx_http_request_t *r, const ngx_str_t *allow)
210{
211 ngx_table_elt_t *origin;
212 ngx_table_elt_t *req_headers;
213
214 origin = brix_http_find_header(r, "Origin", sizeof("Origin") - 1);
215 if (origin == NULL) {
216 return NGX_OK;
217 }
218
219 if (brix_http_set_header(r, "Access-Control-Allow-Origin", "*", NULL)
220 != NGX_OK
221 || brix_http_set_header(r, "Vary", "Origin", NULL) != NGX_OK
222 || brix_http_set_header_str(r, "Access-Control-Allow-Methods",
223 allow, 0, NULL) != NGX_OK)
224 {
225 return NGX_ERROR;
226 }
227
228 req_headers = brix_http_find_header(
229 r, "Access-Control-Request-Headers",
230 sizeof("Access-Control-Request-Headers") - 1);
231 if (req_headers != NULL
232 && !brix_http_str_has_ctl(req_headers->value.data,
233 req_headers->value.len))
234 {
235 if (brix_http_set_header_str(r, "Access-Control-Allow-Headers",
236 &req_headers->value, 0, NULL) != NGX_OK)
237 {
238 return NGX_ERROR;
239 }
240 } else if (brix_http_set_header(r, "Access-Control-Allow-Headers",
241 "Authorization, Content-Type, "
242 "Content-Length, Content-MD5, "
243 "x-amz-content-sha256, x-amz-date, "
244 "x-amz-security-token, x-amz-copy-source, "
245 "x-amz-checksum-crc64nvme, "
246 "x-amz-checksum-algorithm, "
247 "x-amz-sdk-checksum-algorithm, "
248 "Range",
249 NULL) != NGX_OK)
250 {
251 return NGX_ERROR;
252 }
253
254 return brix_http_set_header(r, "Access-Control-Max-Age", "86400", NULL);
255}

257static ngx_int_t
258s3_handle_options(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf)
259{
260 ngx_str_t allow;
261
262 if (brix_http_operation_allow_header(r->pool,
263 brix_s3_operations, brix_s3_operations_count,
264 s3_allow_flags(cf), &allow) != NGX_OK)
265 {
266 return NGX_HTTP_INTERNAL_SERVER_ERROR;
267 }
268
269 if (brix_http_set_header_str(r, "Allow", &allow, 0, NULL) != NGX_OK) {
270 return NGX_HTTP_INTERNAL_SERVER_ERROR;
271 }
272
273 if (s3_add_preflight_headers(r, &allow) != NGX_OK) {
274 return NGX_HTTP_INTERNAL_SERVER_ERROR;
275 }
276
277 r->headers_out.status = NGX_HTTP_OK;
278 r->headers_out.content_length_n = 0;
279
280 ngx_http_send_header(r);
281 return ngx_http_send_special(r, NGX_HTTP_LAST);
282}

319 *
320 * WHAT: Begin SciTags packet marking for a plain S3 GET or PUT when
321 * brix_pmark + brix_pmark_http_plain are both on; ended via a request-pool
322 * cleanup.
323 * WHY: Extracted from ngx_http_s3_handler; S3 has no TPC, so only plain GET/PUT
324 * are marked. Keeping it here shrinks the entry handler's decision count.
325 * HOW: No-op unless enabled and method is GET/PUT. Copies uri/args into fixed
326 * buffers (matching original semantics) and calls brix_pmark_http_mark.

328static void
329s3_pmark_begin_if_enabled(ngx_http_request_t *r, ngx_http_s3_req_ctx_t *s3ctx,
330 ngx_http_s3_loc_conf_t *cf)
331{
332 u_char pth[2048], cgi[512];
333
334 if (!(cf->common.pmark.enable && cf->common.pmark.http_plain
335 && (r->method == NGX_HTTP_GET || r->method == NGX_HTTP_PUT)))
336 {
337 return;
338 }
339
340 ngx_cpystrn(pth, r->uri.data, ngx_min(r->uri.len + 1, sizeof(pth)));
341 if (r->args.len) {
342 ngx_cpystrn(cgi, r->args.data, ngx_min(r->args.len + 1, sizeof(cgi)));
343 } else {
344 cgi[0] = '\0';
345 }
346 brix_pmark_http_mark(&cf->common.pmark, r->pool, r->connection,
347 (r->method == NGX_HTTP_PUT),
348 brix_identity_vo_csv_cstr(s3ctx->identity),
349 brix_identity_dn_cstr(s3ctx->identity),
350 (const char *) pth, (const char *) cgi);
351}

353/*
354 * s3_tls_gate - the per-capability TLS gate (brix_tls_require) for one request.
355 *
356 * WHAT: Builds the capability mask this request exercises and refuses it with
357 * 403 when brix_tls_require demands TLS for any exercised capability on a
358 * cleartext connection.
359 * WHY: Extracted from ngx_http_s3_handler to keep the entry handler's branch
360 * density under the CCN gate; shares the generic VFS mask/check with the
361 * stream and WebDAV planes. Every request exercises `session`; GET/PUT/POST
362 * move object bytes → `data`; a SigV4-signed request presents a credential →
363 * `login` (S3 has no TPC). Refused cleartext requests get a plain 403
364 * before any SigV4 or parse work.
365 * HOW: No-op (NGX_OK) when brix_tls_require is unset or the mask clears.
366 */
367static ngx_int_t
368s3_tls_gate(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf)
369{
370 ngx_uint_t caps;
371
372 if (cf->common.tls_require == 0) {
373 return NGX_OK;
374 }
375
376 caps = BRIX_TLSREQ_SESSION;
377 if (r->method == NGX_HTTP_GET || r->method == NGX_HTTP_PUT
378 || r->method == NGX_HTTP_POST)
379 {
380 caps |= BRIX_TLSREQ_DATA;
381 }
382 if (r->headers_in.authorization != NULL) {
383 caps |= BRIX_TLSREQ_LOGIN;
384 }
385 caps = brix_tls_gate_refused(cf->common.tls_require, caps,
386 r->connection->ssl != NULL);
387 if (caps != 0) {
388 ngx_log_error(NGX_LOG_WARN, r->connection->log, 0,
389 "brix: cleartext S3 request refused by brix_tls_require %s "
390 "(method=%V)", brix_tls_cap_name(caps), &r->method_name);
391 return NGX_HTTP_FORBIDDEN;
392 }
393 return NGX_OK;
394}

400/*
401 * ngx_http_s3_handler - nginx HTTP content handler for the S3 API.
402 *
403 * Implements the AWS S3 path-style API subset needed by XrdClS3 and other
404 * HEP data transfer tools. Dispatch order:
405 *
406 * 1. Parse the URI into bucket + key using s3_parse_uri().
407 * 2. Verify AWS SigV4 signature (s3_verify_sigv4).
408 * 3. ListObjectsV2 (GET /<bucket>/?list-type=2) -> s3_handle_list().
409 * 4. ListMultipartUploads (GET /<bucket>/?uploads) -> s3_handle_list_multipart_uploads().
410 * 5. ListParts (GET /<bucket>/<key>?uploadId=<id>) -> s3_handle_list_parts().
411 * 6. GetObject (GET /<bucket>/<key>) -> s3_handle_get().
412 * 7. HeadObject (HEAD /<bucket>/<key>) -> s3_handle_head().
413 * 8. PutObject / UploadPart (PUT /<bucket>/<key>) -> s3_put_body_handler().
414 * 9. DeleteObject / Abort (DELETE /<bucket>/<key>) -> s3_handle_delete().
415 * 10. CompleteMultipartUpload (POST /<bucket>/<key>?uploadId=<id>) -> s3_multipart_complete_body_handler().
416 * 11. InitiateMultipartUpload (POST /<bucket>/<key>?uploads) -> s3_handle_multipart_initiate().
417 *
418 * The fs_path (filesystem path within conf->common.root_canon) is resolved by
419 * s3_resolve_key() and stored in r->pool for the async PUT callback. The URI
420 * parse and post-auth method dispatch live in handler_dispatch.c and
421 * handler_object_route.c; this handler runs the auth gate and hands off to
422 * s3_dispatch_after_auth().
423 *
424 * Pool allocation: path_copy uses ngx_pnalloc(r->pool, PATH_MAX) so it
425 * survives until the PUT body callback fires after body reading completes.
426 */
427/*
428 * WHY: The dispatch order is critical — list requests are checked before object access to avoid treating empty-key GETs as invalid URIs. SigV4 verification happens early so unsigned requests fail fast without parsing overhead. Write operations (PUT/DELETE/POST) check cf->common.allow_write before body read to reject writes on read-only endpoints without consuming client bandwidth.
429 *
430 * HOW: Gets location config, returns NGX_DECLINED if disabled. Determines method_slot (list vs object). Tracks per-IP-version bytes_rx metric. Verifies SigV4 signature — fails with appropriate XML error if invalid. Parses URI into bucket+key via s3_parse_uri() — rejects mismatched bucket (-1) or malformed URI (0). Checks list-type=2, uploads flag, delete flag before empty-key rejection. Resolves key to fs_path via s3_resolve_key(). Dispatches by HTTP method: GET→list_parts/get; HEAD→head; PUT→upload_part_copy/copy_object/put_body; DELETE→multipart_abort/delete; POST→multipart_initiate/complete. Unknown methods → 405.
431 */
432ngx_int_t
433ngx_http_s3_handler(ngx_http_request_t *r)
434{
435 ngx_http_s3_loc_conf_t *cf;
436 ngx_int_t rc;
437 int is_list_request;
438 ngx_flag_t is_post_object_form;
439 ngx_uint_t method_slot;
440 ngx_http_s3_req_ctx_t *s3ctx;
441
442 cf = ngx_http_get_module_loc_conf(r, ngx_http_brix_s3_module);
443 if (!cf->common.enable) {
444 return NGX_DECLINED;
445 }
446
447 rc = s3_tls_gate(r, cf);
448 if (rc != NGX_OK) {
449 return rc;
450 }
451
452 /* AGPL-3.0 sec.13: offer remote users the source (X-Source header). */
453 brix_http_source_offer(r);
454
455 BRIX_PCALLOC_OR_RETURN(s3ctx, r->pool, sizeof(*s3ctx), NGX_HTTP_INTERNAL_SERVER_ERROR);
456 s3ctx->identity = brix_identity_alloc(r->pool);
457 if (s3ctx->identity == NULL) {
458 return NGX_HTTP_INTERNAL_SERVER_ERROR;
459 }
460 ngx_http_set_ctx(r, s3ctx, ngx_http_brix_s3_module);
461
462 /*
463 * ListObjectsV2 is an HTTP GET on the wire, but it has very different
464 * filesystem behavior from GetObject, so it gets a separate metrics slot.
465 */
466 is_list_request = s3_is_list_request(r);
467 is_post_object_form = s3_is_post_object_form(r);
468 method_slot = is_list_request ? BRIX_S3_METHOD_LIST
469 : s3_metrics_method_slot(r);
470 s3_metrics_request_method(method_slot);
471
472 /* phase-105 W1: per-client-IP rate limit, before the auth burden
473 * (mirrors webdav's access-phase ordering — reject cheap first). */
474 rc = s3_rate_limit(r, cf);
475 if (rc != NGX_OK) {
476 return s3_metrics_return_method(r, method_slot, rc);
477 }
478
479 if (r->method == NGX_HTTP_OPTIONS) {
480 return s3_metrics_return_method(r, method_slot,
481 s3_handle_options(r, cf));
482 }
483
484 if (!is_post_object_form) {
485 rc = s3_verify_sigv4(r, cf, s3ctx->identity);
486 if (rc != NGX_OK) {
487 return s3_metrics_return_method(r, method_slot, rc);
488 }
489 }
490
491 /* XrdAcc engine (when brix_acc_format xrdacc) */ rc = s3_acc_check(r, cf, s3ctx->identity);
492 if (rc != NGX_OK) {
493 return s3_metrics_return_method(r, method_slot, rc);
494 }
495
496 /*
497 * An auth gate that rejected the request by emitting its OWN XML error body
498 * (e.g. s3_verify_sigv4's "Missing/Malformed Authorization") returns NGX_OK
499 * but leaves r->header_sent set. Continuing to dispatch would let GetObject
500 * call ngx_http_send_header a second time ("header already sent" alert) and
501 * overwrite the sent 4xx with headers_out.status = 200 — hiding the auth
502 * failure from the LOG phase (metrics AND the bad-actor guard). Stop here so
503 * the already-sent status stands.
504 */
505 if (r->header_sent) {
506 return s3_metrics_return_method(r, method_slot, NGX_OK);
507 }
508
509 /* Phase-70 §5.1: capture an optional user-supplied x509 delegation proxy
510 * (X-Brix-Delegate-Proxy). No-op unless this export delegates to a backend;
511 * a rejected header (403) short-circuits dispatch. */
512 rc = s3_capture_delegate_proxy(r, s3ctx, cf);
513 if (rc != NGX_OK) {
514 return s3_metrics_return_method(r, method_slot, rc);
515 }
516
517 s3_sess_begin_request(r, method_slot);
518 s3_sess_attempt_request(r, method_slot);
519
520 /*
521 * SciTags packet marking (phase-34). S3 has no TPC, so only plain
522 * GET/PUT are marked, and only when brix_pmark_http_plain is on.
523 * Begun here post-SigV4; ended via a request-pool cleanup.
524 */
525 s3_pmark_begin_if_enabled(r, s3ctx, cf);
526
527 /*
528 * Phase 40: bracket the whole post-auth dispatch with the impersonation
529 * principal (now that auth has populated s3ctx->identity). This covers the
530 * SYNCHRONOUS ops (GET/HEAD/DELETE/list/multipart-initiate) — which open /
531 * stat / unlink as the mapped user, so e.g. a GetObject cannot read a file
532 * the mapped user has no UNIX permission for (a cross-tenant leak when run as
533 * the worker). The async body handlers (PUT/POST/multipart-complete/
534 * delete-objects) return NGX_DONE and re-establish the principal in their own
535 * callbacks, so clearing it here on return is correct. No-op unless map mode.
536 */
537 brix_imp_request_begin(s3ctx->identity);
538 rc = s3_dispatch_after_auth(r, cf, s3ctx, method_slot,
539 is_list_request, is_post_object_form);
540 brix_imp_request_end();
541 return rc;
542}
