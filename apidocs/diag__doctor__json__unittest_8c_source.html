2 * diag_doctor_json_unittest.c — standalone unit test for the `xrddiag doctor
3 * --json` document assembler (diag_doctor_json.c).
4 *
5 * cc -std=c11 -Wall -Wextra -Werror -Ilib -I../src -I../shared \
6 * -DXRDPROTO_NO_NGX apps/diag/diag_doctor_json_unittest.c -o /tmp/ut \
7 * && /tmp/ut (run from client/)
8 *
9 * Exit 0 = all checks pass. Pure C — no server, no connection, no libbrix: the
10 * TU under test is #included and every extern it calls is satisfied by a stub
11 * here. The sub-object emitters (config/latency/recon/eos) are each owned by
12 * their own TU and each writes its OWN leading comma, so the property this
13 * suite pins is the assembler's: every sub-object must be written INSIDE the
14 * endpoint object, i.e. before the `}` that closes it. A regression that emits
15 * one after the brace yields a document that is still plausible-looking text
16 * but is not parseable JSON — exactly the `eos` defect this suite was written
17 * for. The C side asserts byte ordering; the Python driver
18 * (tests/test_doctor_json_unit.py) feeds the printed document to a real JSON
19 * parser, which is the assertion that actually matters.

53/* Minimal but correct JSON string emitter (the real one is fjson.c). */
54void
55fjson_str(FILE *out, const char *s)
56{
57 fputc('"', out);
58 for (; s != NULL && *s != '\0'; s++) {
59 if (*s == '"' || *s == '\\') {
60 fprintf(out, "\\%c", *s);
61 } else if ((unsigned char) *s < 0x20) {
62 fprintf(out, "\\u%04x", (unsigned char) *s);
63 } else {
64 fputc(*s, out);
65 }
66 }
67 fputc('"', out);
68}

116/* Byte offset of `needle` in `hay`, or -1. */
117static long
118at(const char *hay, const char *needle)
119{
120 const char *p = strstr(hay, needle);
121
122 return p == NULL ? -1 : (long) (p - hay);
123}
124
125/* Structural nesting depth at the first occurrence of `needle`, counting only
126 * brackets outside string literals; -1 when the needle is absent. This is what
127 * makes the placement assertion decisive: an object member sits one level below
128 * the object that owns it, so a sub-object flushed out past the endpoint's
129 * closing brace lands at the DEPTH OF THE ARRAY, not of its sibling keys —
130 * a difference no substring search can see. */
131static int
132key_depth(const char *doc, const char *needle)
133{
134 const char *hit = strstr(doc, needle);
135 const char *p;
136 int depth = 0, instr = 0;
137
138 if (hit == NULL) {
139 return -1;
140 }
141 for (p = doc; p < hit; p++) {
142 if (instr) {
143 instr = (*p == '\\') ? (p++, 1) : (*p != '"');
144 continue;
145 }
146 if (*p == '"') {
147 instr = 1;
148 } else if (*p == '{' || *p == '[') {
149 depth++;
150 } else if (*p == '}' || *p == ']') {
151 depth--;
152 }
153 }
154 return depth;
155}

199/* Render the two-endpoint document into a heap buffer the caller frees. */
200static char *
201render(void)
202{
203 doctor_ep eps[2];
204 char *buf = NULL;
205 size_t len = 0;
206 FILE *ms = open_memstream(&buf, &len);
207
208 if (ms == NULL) {
209 return NULL;
210 }
211 build_eps(eps);
212 doctor_emit_json(eps, 2, ms);
213 fclose(ms);
214 return buf;

217int
218main(void)
219{
220 char *doc = render();
221 long eos0, eos1, brace0, cms0;
222 int depth;
223
224 if (doc == NULL) {
225 printf("FAIL: open_memstream\n");
226 return 1;
227 }
228
229 /* Every sub-object is present and nothing is emitted for the absent ones. */
230 CHECK(at(doc, "\"eos\":{\"kind\":\"mgm\"") >= 0);
231 CHECK(at(doc, "\"eos\":{\"kind\":\"fst\"") >= 0);
232 CHECK(at(doc, "\"latency\":{\"samples\":5}") >= 0);
233 CHECK(at(doc, "\"recon\"") < 0); /* unprobed => nothing at all */
234 CHECK(at(doc, "\"cross_endpoint_analysis\":{\"hops\":1}") >= 0);
235
236 /* The defect this suite exists for: `eos` must land INSIDE the endpoint
237 * object — i.e. at the same nesting depth as the endpoint's own keys, and
238 * before the `},{` that starts the next endpoint. */
239 eos0 = at(doc, "\"eos\":{\"kind\":\"mgm\"");
240 brace0 = at(doc, "},{\"protocol\"");
241 cms0 = at(doc, "\"cms\":{\"reported\":true,\"role\":\"manager\"");
242 CHECK(brace0 > 0);
243 CHECK(eos0 > 0 && eos0 < brace0);
244 CHECK(cms0 > 0 && cms0 < eos0); /* documented emit order */
245 depth = key_depth(doc, "\"protocol\""); /* an endpoint's own key */
246 CHECK(depth > 0);
247 CHECK(key_depth(doc, "\"eos\":{\"kind\":\"mgm\"") == depth);
248 CHECK(key_depth(doc, "\"eos\":{\"kind\":\"fst\"") == depth);
249 CHECK(key_depth(doc, "\"latency\":") == depth);
250 CHECK(key_depth(doc, "\"cms\":") == depth);
251 CHECK(key_depth(doc, "\"config\":") == depth);
252
253 /* The last endpoint's eos must precede the array terminator. */
254 eos1 = at(doc, "\"eos\":{\"kind\":\"fst\"");
255 CHECK(eos1 > 0 && eos1 < at(doc, "],\"cross_endpoint_analysis\""));
256
257 /* Strings carrying JSON metacharacters are escaped, not passed through. */
258 CHECK(at(doc, "quoted \\\"issue\\\" with a backslash \\\\") >= 0);
259
260 /* Hand the whole document to the Python driver's real JSON parser. */
261 printf("JSON %s", doc);
262 if (doc[strlen(doc) - 1] != '\n') {
263 printf("\n");
264 }
265 free(doc);
266
267 if (g_fail == 0) {
268 printf("all doctor-json assembler checks passed\n");
269 }
270 return g_fail == 0 ? 0 : 1;
271}
