1/*
2 * crypto.c — shared OpenSSL HMAC-SHA256 and SHA-256 helpers.
3 *
4 * Provides two cryptographic primitives used across XRootD/WebDAV/S3:
5 * HMAC-SHA256 for GSI kXR_sigver request signing, and plain SHA-256 for
6 * digest computation.
7 *
8 * EVP_MAC and EVP_MD objects are fetched once at worker init
9 * (brix_crypto_init) and freed at worker exit (brix_crypto_cleanup).
10 * Each per-request call creates a lightweight CTX from the singleton,
11 * avoiding the registry search and allocation cost of EVP_MAC_fetch per call.
12 */

46{
47 EVP_MAC_CTX *ctx;
48 OSSL_PARAM params[2];
49 size_t outlen = 32;
50 int ok = 0;
51
52 if (s_hmac_mac == NULL) { return 0; }
53
54 params[0] = OSSL_PARAM_construct_utf8_string("digest", "SHA256", 0);
55 params[1] = OSSL_PARAM_construct_end();
56
57 ctx = EVP_MAC_CTX_new(s_hmac_mac);
58 if (ctx
59 && EVP_MAC_init(ctx, key, keylen, params) == 1
60 && EVP_MAC_update(ctx, data, datalen) == 1
61 && EVP_MAC_final(ctx, out, &outlen, 32) == 1)
62 {
63 ok = 1;
64 }
65
66 EVP_MAC_CTX_free(ctx);
67 return ok;

134/* ---- brix_secret_page_guard (Phase-28 F3 / P90-28.1) -----------------------
135 *
136 * WHAT: Excludes the pages holding [p, p+len) from core dumps
137 * (madvise MADV_DONTDUMP) and pins them off swap (mlock). Best-effort and
138 * never load-bearing: 0 = fully applied (or len==0), -1 = at least one knob
139 * failed (caller logs and continues).
140 *
141 * WHY: F1 wipes secrets at end-of-life, but a LONG-LIVED secret (keytab key
142 * array, admin bearer token, macaroon root-secret hex) sits in worker memory
143 * for the process lifetime — one core dump or swap-out would persist it to
144 * disk. Guarding the pages closes both channels without touching how the
145 * secret is used.
146 *
147 * HOW: Round p down / p+len up to the page boundary (madvise and mlock are
148 * page-granular and EINVAL on unaligned starts), then apply both knobs
149 * independently so one failing (e.g. RLIMIT_MEMLOCK exhausted) does not skip
150 * the other. MADV_DONTDUMP is Linux-specific — compiled out elsewhere and
151 * that knob then counts as success (nothing to apply). */
152int
153brix_secret_page_guard(const void *p, size_t len)
154{
155 size_t page, off;
156 uintptr_t start;
157 int rc = 0;
158
159 if (p == NULL || len == 0) {
160 return 0;
161 }
162
163 page = (size_t) sysconf(_SC_PAGESIZE);
164 start = (uintptr_t) p & ~(page - 1);
165 off = (size_t) ((uintptr_t) p - start);
166
167#ifdef MADV_DONTDUMP
168 if (madvise((void *) start, len + off, MADV_DONTDUMP) != 0) {
169 rc = -1;
170 }
171#endif
172 if (mlock((void *) start, len + off) != 0) {
173 rc = -1;
174 }
175 return rc;
176}
