47/* Consume one value byte or nested JSON token. 1 means a scalar completed. */
48static int
49value_step(const char *s, size_t n, size_t *i, int *depth)
50{
51 char c = s[*i];
52
53 if (c == '"') {
54 return str_skip(s, n, i);
55 }
56 if (c == '{' || c == '[') {

65 return -1;

67 (*i)++;
68 return 0;

78/* Advance *i past exactly one JSON value (leading ws consumed here). The
79 * container walk is iterative — depth is a counter, not a stack — so the
80 * depth cap bounds work without bounding recursion. 0 / -1. */
81static int
82value_skip(const char *s, size_t n, size_t *i)
83{
84 int depth = 0;
85
86 ws_skip(s, n, i);
87 if (*i >= n) {
88 return -1;
89 }
90 do {
91 int scalar = value_step(s, n, i, &depth);
92
93 if (scalar < 0) {
94 return -1;
95 }
96 if (scalar || depth == 0) {
97 return 0;
98 }
99 } while (*i < n);
100 return -1;
101}

103/* Read one member after an object opening delimiter. 1 field / 0 end / -1. */
104static int
105object_field(const char *json, size_t len, size_t *i, const char **k,
106 size_t *klen, const char **value, size_t *vlen)
107{
108 size_t start;
109
110 ws_skip(json, len, i);
111 if (*i < len && json[*i] == '}') {
112 return 0;
113 }
114 if (*i >= len || json[*i] != '"') {
115 return -1;
116 }
117 start = ++*i;
118 if (str_skip(json, len, i) != 0) {
119 return -1;
120 }
121 *k = json + start;
122 *klen = *i - start - 1;
123 ws_skip(json, len, i);
124 if (*i >= len || json[*i] != ':') {
125 return -1;
126 }
127 (*i)++;
128 ws_skip(json, len, i);
129 start = *i;
130 if (value_skip(json, len, i) != 0) {
131 return -1;
132 }
133 *value = json + start;
134 *vlen = *i - start;
135 return 1;
