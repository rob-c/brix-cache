29/* cms_addr_is_loopback — classify the resolved manager address as on-host
30 * (127.0.0.0/8 or ::1). A loopback manager is reached with no network latency, so
31 * it gets the most aggressive cold-start settle profile. */
32static ngx_uint_t
33cms_addr_is_loopback(struct sockaddr *sa)
34{
35 if (sa == NULL) {
36 return 0;
37 }
38
39 if (sa->sa_family == AF_INET) {
40 struct sockaddr_in *sin = (struct sockaddr_in *) sa;
41 return (ntohl(sin->sin_addr.s_addr) & 0xFF000000u) == 0x7F000000u;
42 }
43
44#if (NGX_HAVE_INET6)
45 if (sa->sa_family == AF_INET6) {
46 struct sockaddr_in6 *sin6 = (struct sockaddr_in6 *) sa;
47 return IN6_IS_ADDR_LOOPBACK(&sin6->sin6_addr) ? 1 : 0;
48 }
49#endif
50
51 return 0;
52}

54/* cms_ctx_create — allocate and arm ONE manager link's heartbeat context:
55 * seed the reconnect backoff from cms_interval (capped at
56 * NGX_BRIX_CMS_BACKOFF_INITIAL), resolve the cold-start settle profile from
57 * THIS manager's locality, seed the disjoint streamid lane (seed = manager
58 * index, stride = manager count), and schedule the first connect. */
59static ngx_brix_cms_ctx_t *
60cms_ctx_create(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *conf,
61 brix_cms_manager_ent_t *ent, ngx_uint_t index, ngx_uint_t nmgrs)
62{
63 ngx_brix_cms_ctx_t *ctx;
64
65 ctx = ngx_pcalloc(cycle->pool, sizeof(ngx_brix_cms_ctx_t));
66 if (ctx == NULL) {
67 return NULL;
68 }
69
70 ctx->cycle = cycle;
71 ctx->conf = conf;
72 ctx->mgr_addr = ent->addr;
73 ctx->mgr_name = ent->raw;
74 ctx->backoff = ngx_min((ngx_msec_t) conf->cms.interval * 1000,
75 (ngx_msec_t) NGX_BRIX_CMS_BACKOFF_INITIAL);
76 ctx->in_need = NGX_BRIX_CMS_HDR_LEN;
77 ctx->start_ns = brix_phase_now_ns();
78 ctx->streamid_seed = (uint32_t) index;
79 ctx->streamid_stride = (uint32_t) nmgrs;
80 ctx->next_streamid = ctx->streamid_seed;
81
82 /*
83 * Resolve the cold-start settle profile once. A loopback manager (same host)
84 * is reached with no network latency, so it gets the most aggressive defaults;
85 * an explicit directive overrides the locality default. The fast-retry interval
86 * is floored so a misconfigured "0" can never become a busy connect-storm.
87 */
88 ctx->is_loopback = cms_addr_is_loopback(ent->addr->sockaddr);
89
90 ctx->fast_retry = (conf->cms.connect_retry != NGX_CONF_UNSET_MSEC)
91 ? conf->cms.connect_retry
92 : (ctx->is_loopback ? NGX_BRIX_CMS_FASTRETRY_LOOPBACK
93 : NGX_BRIX_CMS_FASTRETRY_REMOTE);
94 if (ctx->fast_retry < NGX_BRIX_CMS_FASTRETRY_FLOOR) {
95 ctx->fast_retry = NGX_BRIX_CMS_FASTRETRY_FLOOR;
96 }
97 ctx->fast_window = ctx->is_loopback ? NGX_BRIX_CMS_FASTWIN_LOOPBACK
98 : NGX_BRIX_CMS_FASTWIN_REMOTE;
99
100 ctx->timer.handler = ngx_brix_cms_timer;
101 ctx->timer.data = ctx;
102 ctx->timer.log = cycle->log;
103 /*
104 * The heartbeat re-arms itself every cms_interval (connect.c:194,298). Mark
105 * it cancelable so a draining worker (ngx_exiting after `nginx -s reload`)
106 * exits as soon as its connections close, instead of being held alive to the
107 * shutdown timeout by the next pending heartbeat. The cluster keepalive is
108 * best-effort; dropping a final tick at teardown is harmless (the reconnect
109 * path is already ngx_exiting-guarded). Matches the io_uring panic timer.
110 */
111 ctx->timer.cancelable = 1;
112
113 {
114 /* First-connect delay: directive override, else the locality default
115 * (0 for loopback — connect immediately; the fast-retry handles a miss). */
116 ngx_msec_t init_delay = (conf->cms.initial_delay != NGX_CONF_UNSET_MSEC)
117 ? conf->cms.initial_delay
118 : (ctx->is_loopback ? NGX_BRIX_CMS_INITDELAY_LOOPBACK
119 : NGX_BRIX_CMS_INITDELAY_REMOTE);
120 ngx_brix_cms_schedule(ctx, init_delay);
121 }
122
123 return ctx;
124}

126/* ngx_brix_cms_start — worker-init entry point (from config/process.c): start
127 * one independent heartbeat link PER configured manager (concurrent logins into
128 * every redundant manager, stock ManList parity). Each worker that runs the
129 * client keeps its own connections. */
130
131void
132ngx_brix_cms_start(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *conf)
133{
134 ngx_uint_t i, nmgrs;
135 brix_cms_manager_ent_t *ents;
136
137 if (conf->cms.managers == NULL || conf->cms.nctxs != 0) {
138 return;
139 }
140
141 nmgrs = conf->cms.managers->nelts;
142 ents = conf->cms.managers->elts;
143
144 conf->cms.ctxs = ngx_pcalloc(cycle->pool,
145 nmgrs * sizeof(ngx_brix_cms_ctx_t *));
146 if (conf->cms.ctxs == NULL) {
147 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
148 "brix: CMS heartbeat allocation failed");
149 return;
150 }
151
152 for (i = 0; i < nmgrs; i++) {
153 conf->cms.ctxs[i] = cms_ctx_create(cycle, conf, &ents[i], i, nmgrs);
154 if (conf->cms.ctxs[i] == NULL) {
155 ngx_log_error(NGX_LOG_ERR, cycle->log, 0,
156 "brix: CMS heartbeat allocation failed");
157 conf->cms.ctxs = NULL;
158 conf->cms.nctxs = 0;
159 return;
160 }
161 }
162
163 conf->cms.nctxs = nmgrs;
164 conf->cms.rr = 0;
165
166 /* §2.11: external machine-load feed (no-op unless brix_cms_perf_pgm). */
167 brix_cms_perf_start(cycle, conf);
168
169 /* §2.12: foreign-data-server liveness monitor (no-op unless
170 * brix_cms_altds ... monitor). */
171 brix_cms_altds_start(cycle, conf);
172
173 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
174 "brix: CMS heartbeat starting for manager %V (%ui manager(s))",
175 &conf->cms.manager, nmgrs);
176}

178/* ngx_brix_cms_pick_ctx — rotate locate/query traffic across the logged-in
179 * manager links; see the declaration in cms_internal.h for the contract. */
180
181ngx_brix_cms_ctx_t *
182ngx_brix_cms_pick_ctx(ngx_stream_brix_srv_conf_t *conf)
183{
184 ngx_uint_t i, start;
185 ngx_brix_cms_ctx_t *ctx;
186
187 if (conf->cms.nctxs == 0) {
188 return NULL;
189 }
190
191 start = conf->cms.rr++;
192
193 for (i = 0; i < conf->cms.nctxs; i++) {
194 ctx = conf->cms.ctxs[(start + i) % conf->cms.nctxs];
195 if (ctx->connection != NULL && ctx->logged_in) {
196 return ctx;
197 }
198 }
199
200 return conf->cms.ctxs[start % conf->cms.nctxs];
201}

203/*
204 * CMS auto-role derivation.
205 *
206 * The role a node plays in the cmsd cluster is derived — not hand-configured —
207 * from two facts already present on the merged server conf:
208 *
209 * manager_capable = manager_mode (set by `brix_manager_mode on`, or
210 * auto-derived from `brix_cms_server on`; see
211 * server_module.c: a block that accepts downstream nodes
212 * IS a manager).
213 * has_upstream = a `brix_cms_manager host:port` resolved into cms.addr,
214 * i.e. this node registers UP to a parent manager.
215 *
216 * manager_capable & !has_upstream -> MANAGER (top of the tree)
217 * manager_capable & has_upstream -> SUB-MANAGER (accepts downstream AND
218 * registers up + aggregates)
219 * !manager_capable & has_upstream -> CLIENT (leaf data server / relay)
220 * !manager_capable & !has_upstream -> NONE (not a cmsd node)
221 */
222/*
223 * The DERIVED node role, computed here from (manager_capable, has_upstream).
224 * Deliberately its own vocabulary — distinct from the operator-declared config
225 * role (BRIX_CMS_ROLE_* in conf_structs.h) and the wire-routing role
226 * (XRDCMS_ROLE_* / brix_cms_role_t in router.h). Keep this prefix separate so
227 * the three role namespaces never collide in a shared translation unit.
228 */
229typedef enum {
230 BRIX_CMS_NODEROLE_NONE = 0,
231 BRIX_CMS_NODEROLE_CLIENT,
232 BRIX_CMS_NODEROLE_MANAGER,
233 BRIX_CMS_NODEROLE_SUBMANAGER
234} brix_cms_noderole_t;
235
236static brix_cms_noderole_t
237brix_cms_noderole_derive(ngx_flag_t manager_capable, ngx_flag_t has_upstream)
238{
239 if (manager_capable) {
240 return has_upstream ? BRIX_CMS_NODEROLE_SUBMANAGER
241 : BRIX_CMS_NODEROLE_MANAGER;
242 }
243 return has_upstream ? BRIX_CMS_NODEROLE_CLIENT : BRIX_CMS_NODEROLE_NONE;
244}
245
246static const char *
247brix_cms_noderole_name(brix_cms_noderole_t role)
248{
249 switch (role) {
250 case BRIX_CMS_NODEROLE_MANAGER: return "manager";
251 case BRIX_CMS_NODEROLE_SUBMANAGER: return "sub-manager";
252 case BRIX_CMS_NODEROLE_CLIENT: return "client";
253 default: return "none";
254 }
255}

257/*
258 * brix_cms_role_worker_init — per-worker CMS bring-up + role proof for ONE
259 * server block. Called for every stream server (process.c), including a
260 * dedicated cms-only manager/sub-manager block whose main-module data path is
261 * disabled (common.enable off) — that block still owns a cmsd role and, when it
262 * has an upstream, an outbound client to start.
263 *
264 * Two jobs:
265 * 1. PROOF-IN-LOGS: emit exactly one NOTICE naming this node's derived role
266 * (manager / sub-manager / client) and how it was derived. Worker 0 only,
267 * because every worker shares this config and N identical lines help no one.
268 * 2. SINGLE UPSTREAM CONNECTION: start the outbound CMS client on WORKER 0
269 * ONLY. A stock cmsd admits one connection per node identity (SID =
270 * host:listen_port, identical across workers), so N per-worker connections
271 * collide as "already logged in" and get 30s-blacklisted. Gating the
272 * client to a single worker makes brix register cleanly with a STOCK
273 * upstream cmsd and removes the self-collision entirely. (Tradeoff: the
274 * §7.1 pending-locate bridge, which needs the client on the worker holding
275 * the suspended session, is unavailable in this single-connection mode —
276 * aggregation, which is what a sub-manager needs, works over one link.)
277 */
278void
279brix_cms_role_worker_init(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *xcf)
280{
281 brix_cms_noderole_t role;
282 ngx_flag_t has_upstream;
283 ngx_flag_t manager_capable;
284
285 has_upstream = (xcf->cms.addr != NULL);
286 manager_capable = (xcf->manager_mode == 1);
287 role = brix_cms_noderole_derive(manager_capable, has_upstream);
288
289 if (role == BRIX_CMS_NODEROLE_NONE) {
290 return;
291 }
292
293 if (ngx_worker == 0) {
294 if (has_upstream) {
295 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
296 "brix: cmsd role: this node is a %s "
297 "(listen :%d, upstream_manager=%V (+%ui more), "
298 "accepts_downstream=%s, aggregates_up=%s)",
299 brix_cms_noderole_name(role), (int) xcf->listen_port,
300 &xcf->cms.manager,
301 xcf->cms.managers->nelts - 1,
302 manager_capable ? "yes" : "no",
303 role == BRIX_CMS_NODEROLE_SUBMANAGER ? "yes" : "no");
304 } else {
305 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
306 "brix: cmsd role: this node is a %s "
307 "(listen :%d, upstream_manager=none, "
308 "accepts_downstream=%s, aggregates_up=no)",
309 brix_cms_noderole_name(role), (int) xcf->listen_port,
310 manager_capable ? "yes" : "no");
311 }
312 }
313
314 /* Single outbound connection: worker 0 only (see the note above). */
315 if (has_upstream && ngx_worker == 0) {
316 ngx_brix_cms_start(cycle, xcf);
317 }
318}
