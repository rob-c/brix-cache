1/*
2 * vfs_internal.h — implementation-private definitions shared by the vfs_*.c units.
3 *
4 * WHAT: Defines the real handle structs hidden behind vfs.h's opaque typedefs
5 * (brix_vfs_file_s, brix_vfs_dir_s), the inline confinement/write
6 * guards (brix_vfs_require_confined; mutation authority lives in
7 * vfs_policy.h), the
8 * ctx-path accessor (brix_vfs_ctx_path), the metrics/access-log observer
9 * helpers (brix_vfs_observe_ctx_op / brix_vfs_observe_file_op and the
10 * elapsed-usec/proto helpers they use), and the cross-unit prototypes
11 * (fill_stat, copy_path, adopt_fd, pread_full, pwrite_full).
12 *
13 * WHY: Every vfs_*.c file needs the same guard-then-syscall-then-observe
14 * pattern and the same handle layout. Centralising it here keeps the
15 * per-op files thin and guarantees that confinement re-verification and
16 * metric/log emission happen identically for every operation.
17 *
18 * HOW: The guards reject any ctx whose resolved path is empty or not confined,
19 * setting errno; mutation authority is decided separately by the
20 * vfs_policy.c kernel, which returns EROFS for a read-only endpoint. The
21 * observer helpers translate an rc/errno into an brix_err_class_t,
22 * compute latency from a start ngx_current_msec, then call
23 * brix_metric_op_done + brix_access_log_emit and restore errno so the
24 * caller can return it untouched. Only this header is included by the
25 * vfs_*.c units; protocol handlers include vfs.h instead.
26 */

54/* Per-request delegation live-cred bag (phase-70 §4). Carries the raw
55 * forwardable credential BYTES the front door captured — distinct from the
56 * dir-based select bound by brix_vfs_ctx_bind_backend_cred. The front door
57 * fills exactly the fields for the strategy it captured:
58 * PASSTHROUGH bearer — bearer holds the raw JWT text;
59 * PASSTHROUGH x509 — have_proxy_pem=1 + proxy_pem holds the full proxy PEM
60 * (cert chain + private key) the user voluntarily supplied.
61 * `mode` is the resolved brix_cred_mode. All bytes are owned by the request pool
62 * and must never be logged. The vfs.h forward declaration names this struct so a
63 * pointer can hang on brix_vfs_ctx_t without exposing the layout to handlers. */
64struct brix_deleg_live_s {
65 int have_proxy_pem; /* 1 = proxy_pem holds a full x509 proxy */
66 ngx_str_t proxy_pem; /* full proxy PEM (chain + key); never log */
67 ngx_str_t bearer; /* raw JWT text; never log */
68 enum brix_cred_mode mode; /* resolved delegation strategy */
69 /* Phase-70 §5.4 EXCHANGE conf (borrowed from conf; NUL-terminated). When
70 * `mode` is BRIX_CRED_EXCHANGE and `tx.endpoint` is set, the cred gate trades
71 * the live bearer for a backend-audienced token via brix_token_exchange();
72 * when tx.endpoint is unset EXCHANGE degrades to verbatim bearer passthrough.
73 * `tx_audience` is the first backend_token_aud entry (target audience). These
74 * are populated by brix_vfs_deleg_set_exchange() at capture time — when that
75 * call is absent tx.endpoint stays empty and the verbatim fallback applies. */
76 brix_token_exchange_conf_t tx;
77 ngx_str_t tx_audience;
78 /* Borrowed pointer at the conf's per-worker minted-token cache slot
79 * (phase-70 §5.4 / P90-70.9): the EXCHANGE leg lazily creates a
80 * brix_tx_cache_t there (ngx_cycle->pool) and consults it before POSTing
81 * the RFC-8693 grant. NULL ⇒ no caching (every op re-exchanges). */
82 void **tx_cache_slot;
83 /* Phase-70 §5.1 in-gate chain re-verify (P90-70.4). Borrowed X509_STORE*
84 * (conf-owned; typed void* so this header stays OpenSSL-free) + max proxy
85 * chain depth (0 = OpenSSL default), stamped by
86 * brix_vfs_deleg_set_ca_store() at capture time. NULL ⇒ the PASSTHROUGH
87 * materialiser skips the re-verify and relies on the capture-side gate. */
88 void *ca_store;
89 ngx_uint_t ca_verify_depth;
90 /* Phase-70 §5.6 SSS identity injection (P90-70.3). Borrowed conf bytes
91 * (NUL-terminated — conf tokens are). Non-empty ⇒ when the bag carries NO
92 * forwardable bytes (no pem, no bearer) the gate materialises an SSS
93 * credential asserting the caller's principal, signed with this keytab.
94 * Proven credential bytes always win over injection. Stamped by
95 * brix_vfs_deleg_set_sss() — which, unlike the other setters, allocates
96 * the bag itself when none is bound (injection needs no captured bytes). */
97 ngx_str_t sss_keytab;
98 /* Phase-70 §5.5 S3 STS EXCHANGE. Borrowed brix_s3_sts_conf_t* (typed void*
99 * so the field costs no extra coupling beyond the sts.h already included).
100 * Non-NULL => when the bag carries NO forwardable bytes AND the leaf backend
101 * accepts BRIX_SD_CRED_S3, the gate exchanges the node's S3 service cred for
102 * temporary creds scoped to the caller via brix_vfs_deleg_sts_cred(). Like
103 * SSS injection this needs no captured bytes, so brix_vfs_deleg_set_sts()
104 * allocates the bag when none is bound. The conf (5 borrowed conf-owned
105 * ngx_str_t + ttl) is built on the request pool at the capture site. */
106 const void *sts;
107 /* Phase-70 §5.7 krb5 GSSAPI EXCHANGE. Unlike SSS/STS this DOES have captured
108 * bytes: the front door captures the user's forwarded TGT and serialises it
109 * to a 0600 FILE ccache (brix_krb5_cred_to_ccache) — a live gss_cred_id_t is
110 * request-scoped and cannot ride the async brix_cache_fill_t, so the path is
111 * the async-safe carry, mirroring the gsi proxy-PEM→0600-path trick.
112 * `krb5_ccache` is that path and `krb5_origin_princ` the derived origin
113 * service principal (host/<fqdn>@REALM); both are borrowed NUL-terminated
114 * strings on the request pool. Non-empty krb5_ccache ⇒ when the leaf accepts
115 * BRIX_SD_CRED_GSS_KRB5 the gate carries them onto the cred and the origin
116 * leg re-imports via brix_krb5_cred_from_ccache. Stamped by
117 * brix_vfs_deleg_set_krb5(). A full x509 proxy still wins above. */
118 ngx_str_t krb5_ccache;
119 ngx_str_t krb5_origin_princ;
120};

275/* Build a transient storage-driver object view from a ctx + fd: the bound
276 * instance (or NULL for the default backend), that backend's driver, and the
277 * fd. Used to ask the backend to perform/decide a per-fd operation without the
278 * VFS hard-coding any concrete driver. "No explicit backend" resolves to
279 * brix_sd_default_driver() rather than naming POSIX. */
280static ngx_inline void
281brix_vfs_ctx_sd_obj(const brix_vfs_ctx_t *ctx, ngx_fd_t fd,

302/* Ask the handle's backend for a sendfile-able fd over [off, off+len), passing
303 * the VFS's storage-neutral zero-copy verdict; returns the fd, or
304 * NGX_INVALID_FILE when the backend declines (or has no read_sendfile_fd slot).
305 * This is the single place the VFS consults the backend's sendfile decision. */
306static ngx_inline ngx_fd_t
307brix_vfs_handle_sendfile_fd(const brix_vfs_file_t *fh, off_t off,
308 size_t len, unsigned want_zerocopy)
309{
310 brix_sd_obj_t obj;
311
312 brix_vfs_handle_sd_obj(fh, &obj);
313 if (obj.driver == NULL || obj.driver->read_sendfile_fd == NULL) {
314 return NGX_INVALID_FILE;
315 }
316 return obj.driver->read_sendfile_fd(&obj, off, len, want_zerocopy);
317}

332/* Deep-copy `ctx` and the buffers it POINTS at — the resolved confined path,
333 * root_canon and the lock-token presentation, all of which can live in a caller
334 * stack frame or request header buffer — onto `pool`,
335 * returning a self-contained ctx that outlives the caller. Every other ctx member
336 * points at config- or connection-scoped memory that already outlives any write
337 * session, so a shallow struct copy carries them correctly. Returns NULL
338 * (errno = ENOMEM) on allocation failure. Shared by the staged-upload handle and
339 * the unified writer — both persist across the open→write→commit request
340 * boundary, so a ctx pointing at request stack buffers would be a use-after-free
341 * at commit. */
342static ngx_inline brix_vfs_ctx_t *
343brix_vfs_ctx_pool_clone(const brix_vfs_ctx_t *ctx, ngx_pool_t *pool)
344{
345 brix_vfs_ctx_t *copy;
346 const char *rpath;
347
348 if (ctx == NULL) {
349 errno = EINVAL;
350 return NULL;
351 }
352 rpath = brix_vfs_ctx_path(ctx);
353
354 copy = ngx_palloc(pool, sizeof(*copy));
355 if (copy == NULL) {
356 errno = ENOMEM;
357 return NULL;
358 }
359 *copy = *ctx;
360
361 if (rpath != NULL) {
362 size_t n = ngx_strlen(rpath);
363 u_char *d = ngx_pnalloc(pool, n + 1);
364 if (d == NULL) {
365 errno = ENOMEM;
366 return NULL;
367 }
368 ngx_memcpy(d, rpath, n + 1);
369 copy->resolved.resolved.data = d;
370 copy->resolved.resolved.len = n;
371 }
372 if (ctx->root_canon != NULL) {
373 size_t n = ngx_strlen(ctx->root_canon);
374 u_char *d = ngx_pnalloc(pool, n + 1);
375 if (d == NULL) {
376 errno = ENOMEM;
377 return NULL;
378 }
379 ngx_memcpy(d, ctx->root_canon, n + 1);
380 copy->root_canon = (const char *) d;
381 }
382 /* phase-107 C7: the lock-token presentation borrows the request's header
383 * buffer, which dies while this clone lives on — carried BY VALUE like
384 * the mutation policy; dropping it would 423 the lock's own holder. */
385 if (ctx->lock_token != NULL) {
386 size_t n = ngx_strlen(ctx->lock_token);
387 u_char *d = ngx_pnalloc(pool, n + 1);
388 if (d == NULL) {
389 errno = ENOMEM;
390 return NULL;
391 }
392 ngx_memcpy(d, ctx->lock_token, n + 1);
393 copy->lock_token = (const char *) d;
394 }
395 return copy;
396}

398/* Read guard: assert the ctx has a non-empty, kernel-confined resolved path.
399 * Returns NGX_OK if confined, else NGX_ERROR with errno=EINVAL. Every wire op
400 * must pass this before touching the filesystem. */
401static ngx_inline ngx_int_t
402brix_vfs_require_confined(const brix_vfs_ctx_t *ctx)
403{
404 const char *path = brix_vfs_ctx_path(ctx);
405
406 if (ctx == NULL || path == NULL || path[0] == '\0'
407 || !ctx->resolved.is_confined)
408 {
409 errno = EINVAL;
410 return NGX_ERROR;
411 }
412
413 return NGX_OK;
414}

449/* Translate a namespace status into a faithful POSIX errno. The namespace layer
450 * sets res.sys_errno for syscall failures but leaves it 0 for the conditions it
451 * derives itself (notably BRIX_NS_NOT_EMPTY from its own emptiness probe), so
452 * callers that collapse a failed brix_ns_* result to errno must use this for
453 * the sys_errno==0 case rather than a blanket EIO — otherwise a non-empty rmdir
454 * surfaces as EIO/500 instead of ENOTEMPTY/409. */
455static ngx_inline int
456brix_vfs_ns_status_errno(brix_ns_status_t status)
457{
458 switch (status) {
459 case BRIX_NS_OK: return 0;
460 case BRIX_NS_NOT_FOUND: return ENOENT;
461 case BRIX_NS_DENIED: return EACCES;
462 case BRIX_NS_EXISTS: return EEXIST;
463 case BRIX_NS_CONFLICT: return ENOTDIR;
464 case BRIX_NS_NOT_EMPTY: return ENOTEMPTY;
465 case BRIX_NS_TOO_LONG: return ENAMETOOLONG;
466 case BRIX_NS_NO_SPACE: return ENOSPC;
467 case BRIX_NS_IO_ERROR: return EIO;
468 }
469
470 return EIO;
471}
