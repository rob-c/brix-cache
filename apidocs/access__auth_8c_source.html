1/*
2 * access_auth.c - WebDAV access-phase authentication gate.
3 *
4 * WHAT: The credential-source tier of the WebDAV access phase: GSI proxy
5 * cert, bearer token, and Basic password sources, the RFC 7617 Basic
6 * challenge for browser clients, and the policy gate (access_authenticate)
7 * that runs them in order and applies auth=required/optional/none.
8 *
9 * WHY: Split out of access.c so the authentication surface is grouped by
10 * concern and individually reviewable while access.c keeps the access-phase
11 * orchestration and the XrdAcc/authdb authorization tiers.
12 *
13 * HOW: access_authenticate is the sole cross-split entry point (declared in
14 * access_internal.h); the per-source wrappers and the Basic challenge stay
15 * file-local statics. Each outcome increments exactly the same metric slot
16 * as before the decomposition.
17 */

29/*
30 * access_try_cert — GSI proxy-certificate credential source.
31 *
32 * WHAT: Attempts client-certificate authentication and records the cert-OK
33 * metrics on success.
34 *
35 * WHY: The cert tier is tried FIRST (before bearer token), matching the
36 * root:// auth ordering; metrics must attribute the session to GSI.
37 *
38 * HOW: Thin wrapper over webdav_verify_proxy_cert; returns its verdict.
39 */
40static ngx_int_t
41access_try_cert(ngx_http_request_t *r,
42 ngx_http_brix_webdav_loc_conf_t *conf)
43{
44 ngx_int_t rc = webdav_verify_proxy_cert(r, conf);
45
46 if (rc == NGX_OK) {
47 BRIX_WEBDAV_METRIC_INC(
48 auth_total[BRIX_WEBDAV_AUTH_RESULT_CERT_OK]);
49 brix_metric_auth(BRIX_PROTO_WEBDAV, BRIX_AUTHN_GSI, 1);
50 }
51 return rc;
52}

104/*
105 * access_basic_challenge — RFC 7617 Basic challenge for browser clients.
106 *
107 * WHAT: Attaches `WWW-Authenticate: Basic realm="brix"` to the response and
108 * returns NGX_HTTP_UNAUTHORIZED, or NGX_HTTP_INTERNAL_SERVER_ERROR on
109 * allocation failure.
110 *
111 * WHY: A browser only shows its login prompt (and re-prompts after a wrong
112 * password) on 401 + a challenge header; a bare 403 is a dead end. Emitted
113 * only when Basic is actually enabled on the export, so cert/token-only
114 * exports keep their historical 403 and never invite a password prompt that
115 * cannot succeed.
116 *
117 * HOW: The same headers_out wiring nginx's own auth_basic module uses — push
118 * the header and point headers_out.www_authenticate at it so the special-
119 * response path emits it.
120 */
121static ngx_int_t
122access_basic_challenge(ngx_http_request_t *r)
123{
124 ngx_table_elt_t *h = ngx_list_push(&r->headers_out.headers);
125
126 if (h == NULL) {
127 return NGX_HTTP_INTERNAL_SERVER_ERROR;
128 }
129 h->hash = 1;
130#if (nginx_version >= 1023000)
131 h->next = NULL;
132#endif
133 ngx_str_set(&h->key, "WWW-Authenticate");
134 ngx_str_set(&h->value, "Basic realm=\"brix\"");
135 r->headers_out.www_authenticate = h;
136 return NGX_HTTP_UNAUTHORIZED;
137}

139/*
140 * webdav_bearer_enabled — does this export accept bearer tokens?
141 *
142 * WHAT: true when JWKS keys, a macaroon secret, or a token registry is
143 * configured on the location.
144 * WHY: RFC 6750 challenge semantics (401 + WWW-Authenticate: Bearer) apply only
145 * on a bearer-protected resource; a cert-only export keeps its historical 403.
146 * HOW: mirrors the DECLINED guard in webdav_verify_bearer_token.
147 */
148static int
149webdav_bearer_enabled(ngx_http_brix_webdav_loc_conf_t *conf)
150{
151 return conf->jwks_key_count > 0
152 || conf->token_macaroon_secret.len > 0
153 || conf->token_registry != NULL;
154}

156/*
157 * access_bearer_challenge — RFC 6750 §3 Bearer challenge / error response.
158 *
159 * WHAT: attaches `WWW-Authenticate: Bearer realm="brix"` (optionally with an
160 * `error="..."` attribute) and returns the given HTTP status.
161 * WHY: RFC 6750 §3 requires an unauthenticated/invalid-token request on a
162 * bearer-protected resource to return 401 + WWW-Authenticate: Bearer (not 403,
163 * which means insufficient_scope); a dual-transport request returns
164 * 400 invalid_request. The header lets a client discover the scheme and, for
165 * invalid_token, know it should refresh rather than escalate.
166 * HOW: same headers_out wiring nginx's auth_basic uses — push the header and
167 * point headers_out.www_authenticate at it so the special-response path emits it.
168 */
169static ngx_int_t
170access_bearer_challenge(ngx_http_request_t *r, ngx_int_t status,
171 const char *error)
172{
173 ngx_table_elt_t *h = ngx_list_push(&r->headers_out.headers);
174
175 if (h == NULL) {
176 return NGX_HTTP_INTERNAL_SERVER_ERROR;
177 }
178 h->hash = 1;
179#if (nginx_version >= 1023000)
180 h->next = NULL;
181#endif
182 ngx_str_set(&h->key, "WWW-Authenticate");
183 if (error != NULL) {
184 h->value.data = ngx_pnalloc(r->pool,
185 sizeof("Bearer realm=\"brix\", error=\"\"") - 1 + ngx_strlen(error));
186 if (h->value.data == NULL) {
187 return NGX_HTTP_INTERNAL_SERVER_ERROR;
188 }
189 h->value.len = ngx_sprintf(h->value.data,
190 "Bearer realm=\"brix\", error=\"%s\"", error) - h->value.data;
191 } else {
192 ngx_str_set(&h->value, "Bearer realm=\"brix\"");
193 }
194 r->headers_out.www_authenticate = h;
195 return status;
196}

198/*
199 * access_try_proto — run the HTTP credential source bound to one protbind id.
200 *
201 * WHAT: Maps a BRIX_AUTH_* protocol id onto its HTTP transport and runs it;
202 * returns NGX_DECLINED for an id with no HTTP transport at all.
203 *
204 * WHY: protbind is protocol-agnostic, so a rule may legitimately name a scheme
205 * that only exists on the root:// wire (sss, unix, krb5, host). Skipping it
206 * here — rather than rejecting it at parse time — lets a site write ONE
207 * `<host> only gsi ztn sss` policy and have each frontend honour the part it
208 * can actually speak, which is exactly what XRootD's shared sec.protbind does.
209 *
210 * HOW: A switch over the three ids with an HTTP transport; the wrappers above
211 * keep their metric attribution unchanged.
212 */
213static ngx_int_t
214access_try_proto(ngx_uint_t proto, ngx_http_request_t *r,
215 ngx_http_brix_webdav_loc_conf_t *conf)
216{
217 switch (proto) {
218 case BRIX_AUTH_GSI: return access_try_cert(r, conf);
219 case BRIX_AUTH_TOKEN: return access_try_token(r, conf);
220 case BRIX_AUTH_PWD: return access_try_basic(r, conf);
221 default: return NGX_DECLINED; /* no HTTP transport */
222 }
223}

225/*
226 * access_protbind_set — resolve this request's ordered credential-source set.
227 *
228 * WHAT: Fills *out with the protocols this peer may authenticate with, in the
229 * order they are to be tried.
230 *
231 * WHY: Without any brix_webdav_protbind rule the answer must be byte-identical
232 * to the historical gate — cert, then token, then Basic, enabled iff auth is
233 * not `none` — so an existing config cannot change behaviour.
234 *
235 * HOW: Build that historical order as the base set, hand the resolver the peer
236 * IP (always available) plus the reverse-resolved hostname (only when some
237 * template actually needs one — the wildcard-only case must never pay for DNS),
238 * and let the shared engine apply the first matching rule.
239 */
240static void
241access_protbind_set(ngx_http_request_t *r,
242 ngx_http_brix_webdav_loc_conf_t *conf,
243 brix_protbind_set_t *out)
244{
245 brix_protbind_set_t base;
246 char peer_ip[NGX_INET6_ADDRSTRLEN + 1];
247 char host_buf[256];
248 const char *peer_host = NULL;
249 size_t n;
250
251 brix_protbind_http_base(conf->auth != WEBDAV_AUTH_NONE, &base);
252
253 n = ngx_min(r->connection->addr_text.len, sizeof(peer_ip) - 1);
254 ngx_memcpy(peer_ip, r->connection->addr_text.data, n);
255 peer_ip[n] = '\0';
256
257 if (brix_protbind_needs_hostname(conf->protbind)) {
258 peer_host = brix_acc_resolve_peer(r->connection->sockaddr,
259 r->connection->socklen,
260 host_buf, sizeof(host_buf));
261 }
262
263 brix_protbind_resolve(conf->protbind, &base, peer_host, peer_ip, out);
264}

266/*
267 * access_authenticate — the authentication gate.
268 *
269 * WHAT: Runs the credential sources bound to this peer, in order, and applies
270 * the location's auth policy to the outcome. With no brix_webdav_protbind
271 * rule that order is the historical one: GSI proxy cert, bearer token, then
272 * Basic password.
273 *
274 * WHY: auth=required rejects unauthenticated requests — with a 401 Basic
275 * challenge when a pwd db is configured (so browsers prompt for credentials),
276 * else the historical 403; auth=optional lets them proceed as anonymous;
277 * auth=none (or a `<host> none` binding) skips verification entirely. Each
278 * outcome increments exactly the same metric slot as before the decomposition.
279 *
280 * HOW: Returns NGX_OK to continue (authenticated or anonymous) or the
281 * metrics-counted rejection. A challenge is only offered for a scheme this
282 * peer is actually bound to — challenging Basic on a host whose binding
283 * excludes pwd would invite a password that can never be accepted.
284 */
285ngx_int_t
286access_authenticate(ngx_http_request_t *r,
287 ngx_http_brix_webdav_loc_conf_t *conf)
288{
289 brix_protbind_set_t bound;
290 ngx_int_t auth_rc = NGX_DECLINED;
291 ngx_int_t token_rc = NGX_DECLINED;
292 ngx_uint_t index;
293
294 access_protbind_set(r, conf, &bound);
295
296 if (!bound.require_auth) {
297 BRIX_WEBDAV_METRIC_INC(
298 auth_total[BRIX_WEBDAV_AUTH_RESULT_NONE]);
299 brix_metric_auth(BRIX_PROTO_WEBDAV, BRIX_AUTHN_NONE, 1);
300 return NGX_OK;
301 }
302
303 for (index = 0; index < bound.count; index++) {
304 auth_rc = access_try_proto(bound.protos[index], r, conf);
305 if (auth_rc == NGX_OK) {
306 break;
307 }
308 if (bound.protos[index] == BRIX_AUTH_TOKEN) {
309 token_rc = auth_rc;
310 /* RFC 6750 §2 (SEC MUST): a dual-transport bearer token (header +
311 * query) is a hard 400 invalid_request — it must NOT fall through
312 * to a weaker source, nor to anonymous. */
313 if (token_rc == NGX_HTTP_BAD_REQUEST) {
314 break;
315 }
316 }
317 }
318
319 if (token_rc == NGX_HTTP_BAD_REQUEST) {
320 BRIX_WEBDAV_METRIC_INC(
321 auth_total[BRIX_WEBDAV_AUTH_RESULT_REJECTED]);
322 brix_metric_auth(BRIX_PROTO_WEBDAV, BRIX_AUTHN_NONE, 0);
323 return webdav_metrics_return(r,
324 access_bearer_challenge(r, NGX_HTTP_BAD_REQUEST, "invalid_request"));
325 }
326
327 if (auth_rc != NGX_OK && conf->auth == WEBDAV_AUTH_REQUIRED) {
328 BRIX_WEBDAV_METRIC_INC(
329 auth_total[BRIX_WEBDAV_AUTH_RESULT_REJECTED]);
330 brix_metric_auth(BRIX_PROTO_WEBDAV, BRIX_AUTHN_NONE, 0);
331 ngx_log_error(NGX_LOG_WARN, r->connection->log, 0,
332 "brix_webdav: unauthenticated request rejected"
333 " (auth=required)");
334 if (conf->pwd_file.len > 0
335 && brix_protbind_allows(&bound, BRIX_AUTH_PWD))
336 {
337 return webdav_metrics_return(r, access_basic_challenge(r));
338 }
339 /* RFC 6750 §3 (MUST): on a bearer-protected export, no/invalid credential
340 * → 401 + WWW-Authenticate: Bearer (403 = insufficient_scope, emitted by
341 * the authz tier for a valid-but-unscoped token). Attribute
342 * error="invalid_token" only when a bearer was actually presented but
343 * failed validation (token_rc == 401); a missing credential gets the bare
344 * challenge. Cert-only exports keep the historical 403. */
345 if (webdav_bearer_enabled(conf)
346 && brix_protbind_allows(&bound, BRIX_AUTH_TOKEN))
347 {
348 const char *err =
349 (token_rc == NGX_HTTP_UNAUTHORIZED) ? "invalid_token" : NULL;
350 return webdav_metrics_return(r,
351 access_bearer_challenge(r, NGX_HTTP_UNAUTHORIZED, err));
352 }
353 return webdav_metrics_return(r, NGX_HTTP_FORBIDDEN);
354 }
355
356 if (auth_rc != NGX_OK) {
357 BRIX_WEBDAV_METRIC_INC(
358 auth_total[BRIX_WEBDAV_AUTH_RESULT_ANONYMOUS]);
359 brix_metric_auth(BRIX_PROTO_WEBDAV, BRIX_AUTHN_NONE, 1);
360 }
361
362 return NGX_OK;
363}
