6/* ---- Measure the complete login payload before writing any bytes ----
7 * WHAT: Return its size, or 0 for invalid spans, wire lengths or capacity.
8 * WHY: Both login directions share the exact scalar/string order and bounds.
9 * HOW: 1. Reserve 40 scalar and eight empty-string bytes. 2. Bound each tail.
10 */
11static size_t
12cms_login_payload_size(size_t capacity, const brix_cms_login_fields_t *fields)
13{
14 const ngx_str_t *strings[] = {
15 &fields->sid, &fields->paths, &fields->interfaces, &fields->environment
16 };
17 size_t required = 48;
18
19 if (capacity < required) {
20 return 0;
21 }
22 for (size_t index = 0; index < sizeof(strings) / sizeof(strings[0]); index++) {
23 size_t length = strings[index]->len;
24 if (length == 0) {
25 continue;
26 }
27 if (strings[index]->data == NULL || length > UINT16_MAX - 1
28 || length + 1 > capacity - required) {
29 return 0;
30 }
31 required += length + 1;
32 }
33 return required;
34}

36/* ---- Encode a complete, bounded CmsLoginData payload ----
37 * WHAT: Return the encoded size, or 0 without modifying output on error.
38 * WHY: A manager must acknowledge LOGIN, using the same wire schema as a node.
39 * HOW: 1. Validate all spans. 2. Use the shared Pup primitives in schema order.
40 */
41size_t
42brix_cms_login_encode(u_char *output, size_t capacity,
43 const brix_cms_login_fields_t *fields)
44{

46 size_t required;
47
48 if (output == NULL || fields == NULL) {

61 cursor = ngx_brix_cms_put_short(cursor, fields->fs_num);
62 cursor = ngx_brix_cms_put_short(cursor, fields->fs_util);
63 cursor = ngx_brix_cms_put_short(cursor, fields->data_port);

67 cursor = ngx_brix_cms_put_string(cursor, fields->interfaces.data, fields->interfaces.len);

70}

72/*
73 * wire.c — CMS frame header and XrdOucPup payload codec
74 *
75 * WHAT: Low-level get/put primitives for the XRootD CMS wire format. Two layers
76 * live here: (1) raw big-endian scalar I/O for the fixed CMS frame header
77 * (get16/get32, put16/put32), and (2) the tagged/packed payload codec
78 * (put_short, put_int, put_string) that serialises the variable-length
79 * fields of a kYR_login payload.
80 *
81 * WHY: The real XrdCms manager (cmsd) speaks a specific binary dialect. The
82 * header is plain big-endian; the payload after the Fence is encoded by
83 * XrdOucPup, which tags scalars with a type byte but writes strings with NO
84 * tag — the parser tells them apart by the PT_short (0x80) bit of the first
85 * byte. Getting the byte layout wrong silently breaks interop with cmsd.
86 *
87 * HOW: Scalars are tagged: put_short emits CMS_PT_SHORT (0x80) then a 2-byte BE
88 * value; put_int emits CMS_PT_INT (0xa0) then a 4-byte BE value. Strings are
89 * packed as [2B BE length][raw bytes][trailing NUL], where the length field
90 * counts the NUL (strlen+1) to match XrdOucPup::Pack; an empty/absent string
91 * is a bare 2-byte zero length with no data and no NUL. Every put_* returns
92 * the cursor advanced past the bytes it wrote so callers can chain writes.
93 */
94

130/* ngx_brix_cms_put_short — write a CMS type-tagged short: the CMS_PT_SHORT (0x80)
131 * tag then a big-endian uint16_t (version/port-style payload values); returns the
132 * advanced cursor. */
133u_char *
134ngx_brix_cms_put_short(u_char *p, uint16_t value)
135{
136 *p++ = CMS_PT_SHORT;
137 ngx_brix_cms_put16(p, value);
138 /* p already stepped over the tag byte above, so +2 covers only the value;
139 * net 3 bytes (1 tag + 2 BE) consumed from the caller's original cursor. */
140 return p + 2;
141}

144 * then a big-endian uint32_t (space metrics, streamid keys); returns the advanced

156/* ngx_brix_cms_put_string — encode a string in XrdOucPup::Pack format (kYR_login
157 * payload after the Fence): a 2-byte big-endian length (counting a trailing NUL, so
158 * strlen+1) then the bytes + NUL; a NULL/empty string is a bare 2-byte zero length.
159 * No type tag byte — the Parser tells a string from a scalar by the absent PT_short
160 * (0x80) bit. Returns the advanced cursor. */
161u_char *
162ngx_brix_cms_put_string(u_char *p, const u_char *data, size_t len)
163{
164 if (data == NULL || len == 0) {
165 ngx_brix_cms_put16(p, 0);
166 return p + 2;
167 }
168
169 ngx_brix_cms_put16(p, (uint16_t) (len + 1));
170 p += 2;
171 ngx_memcpy(p, data, len);
172 p += len;
173 *p++ = '\0';
174 return p;
175}
