1/*
2 * vfs_authz.h — the VFS authorization backstop (phase-108 C12), internal API.
3 *
4 * WHAT: Declares position 1.5 of the §3.4 mutation ordering — the check that
5 * runs after the phase-105 mutation-policy kernel (position 1, EROFS) and
6 * before the lock check (position 2, EBUSY): does THIS identity have the
7 * privilege the operation needs on the resolved path? It is a BACKSTOP,
8 * not a relocation — the protocol edge still refuses first, composes the
9 * wire-correct denial, and owns the SHM verdict cache. The VFS kernel is
10 * the checkpoint that cannot be forgotten when a new handler reaches
11 * storage without asking (§C12.4).
12 *
13 * WHY: Authorization is decided today at 28 protocol-edge sites in several
14 * vocabularies, held together by comments ("deliberately auth-free so
15 * this gate cannot be bypassed"). A comment is a convention the next
16 * author must be told about; a backstop is what replaces a convention.
17 * The backstop re-derives the verdict from the IDENTITY — never from the
18 * edge's cache — so a disagreement means a real bypass, not a stale memo.
19 *
20 * HOW: Reuses the exact three-tier evaluator the edge runs (auth_gate.c), in
21 * its identity-only forms: native authdb via brix_check_authdb_identity,
22 * VO ACL via brix_check_vo_acl_identity, token scope via
23 * brix_identity_check_token_scope; the xrdacc arm memoizes its
24 * brix_acc_entity_t on the identity (identity.h acc_entity) so position
25 * 1.5 stays allocation-free. Fail-closed: an unbound ctx, an identity it
26 * cannot interpret, or an operation with no privilege mapping is a
27 * refusal — a counter+WARN under OBSERVE, EACCES under ENFORCE.
28 *
29 * The two GATE wrappers below are what the VFS mutators actually call: they
30 * fuse position 1 and position 1.5 into one line so the backstop is inherited,
31 * not remembered. The raw brix_vfs_require_confined_mutation / _require_mutation
32 * kernel forms are banned at mutator call sites by
33 * tools/ci/check_authz_backstop.py; only these wrappers (and the phase-105
34 * kernel itself, and the carried-policy delayed-work forms) may reach them.
35 */

42/* The backstop's per-evaluation outcome, mirrored as a bounded metric label by
43 * BRIX_AUTHZ_BACKSTOP_RESULT_COUNT (observability/metrics/unified.h); vfs_authz.c
44 * carries the compile-time equality check. AGREE = the backstop reached the same
45 * verdict the edge did (the normal case); EDGE_MISSING = the backstop would have
46 * refused but the edge already allowed the request (the bypass class this item
47 * exists to catch); NO_RULES = a bound export with no rules (allow-all today);
48 * UNBOUND = a ctx that reached a mutation without binding a rule set. The last
49 * two are honest facts, not errors — but UNBOUND fails closed under ENFORCE. */
50typedef enum {
51 BRIX_AUTHZ_BACKSTOP_AGREE = 0,
52 BRIX_AUTHZ_BACKSTOP_EDGE_MISSING,
53 BRIX_AUTHZ_BACKSTOP_NO_RULES,
54 BRIX_AUTHZ_BACKSTOP_UNBOUND,
55 BRIX_AUTHZ_BACKSTOP_RESULT_N /* never a real result — the array bound */
56} brix_authz_backstop_result_t;

58/* Map a bounded mutation operation onto the BRIX_AUTH_* privilege the edge
59 * already passes by hand for it (§C12.3). Returns the primary needed-privilege
60 * mask; *also_delete is set to 1 for RENAME/COPY, which the tree already treats
61 * as UPDATE on the destination AND DELETE on the source (mv.c precedent). An
62 * operation with no export-storage mapping (DEDUP, CREDENTIAL — governed by the
63 * typed domain assert, not a rule) returns 0, which the kernel reads as
64 * "undecidable" ⇒ fail-closed. Pure. */
65uint32_t brix_vfs_authz_level_for_op(brix_vfs_mutation_op_t op,
66 int *also_delete);

68/* Position 1.5 for a mutation: re-derive the authorization verdict for `op` on
69 * ctx->resolved from ctx->identity + ctx->authz. NGX_OK when the backstop
70 * agrees (or is OFF, or the export is allow-all); NGX_ERROR with errno EACCES
71 * under ENFORCE when it would refuse. Under OBSERVE a refusal is recorded
72 * (counter + WARN) and NGX_OK is returned — the edge stays authoritative for
73 * one release. Never consults the SHM verdict cache. */
74ngx_int_t brix_vfs_require_authorized(const brix_vfs_ctx_t *ctx,
75 brix_vfs_mutation_op_t op);

88/* GATE wrapper — confined path mutators. Runs position 1 (confinement + the
89 * phase-105 mutation-policy kernel, EROFS/EINVAL, NULL-safe) then position 1.5

91 * The EROFS non-disclosure property survives because position 1 answers first:
92 * a read-only endpoint returns EROFS and the backstop never speaks. */
93ngx_int_t brix_vfs_gate_confined(const brix_vfs_ctx_t *ctx,
94 brix_vfs_mutation_op_t op);

96/* GATE wrapper — handle/ctx mutators that gate on the ctx policy without a
