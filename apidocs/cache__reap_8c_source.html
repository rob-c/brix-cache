53/*
54 * WHAT: Immutable, per-walk context threaded through the recursive reaper.
55 * WHY: Collapses reap_dir's fixed argument set (cutoff/device/log/metrics/
56 * store/data-root) into a single pointer so the recursive call and every
57 * extracted helper carry one param instead of seven — keeping each
58 * function within the ≤5-param budget without introducing globals.
59 * HOW: Populated once by brix_cache_reap_dirty from the resolved config and
60 * root stat, then passed by const pointer through the whole walk. `dir`
61 * stays a separate argument because it changes per recursion level.
62 */
63typedef struct {
64 time_t cutoff; /* aged-out boundary (now - max_age) */
65 time_t cold_cutoff;/* clean read-fill boundary; 0 = off */
66 dev_t dev; /* root device — never cross it */
67 ngx_log_t *log; /* reap-event log sink */
68 ngx_brix_srv_metrics_t *slot; /* SHM metrics slot (may be NULL) */
69 brix_cstore_t *cstore; /* data-removal adapter (may be NULL)*/
70 const char *data_root; /* cstore key prefix (may be NULL) */
71} reap_ctx_t;

73/*
74 * WHAT: Decide whether a directory entry names a candidate regular file, and
75 * when so return its full path in `child` (a PATH_MAX buffer).
76 * WHY: Isolates the same-device, dot-skip, sidecar-skip, path-build and type
77 * gating from the classification/removal logic so each stays single-job.
78 * HOW: Skips "."/".."/sidecars, builds `dir/name` (skips on truncation),
79 * lstats it, and rejects anything off the root device. Directories are
80 * signalled to the caller (which recurses); non-regular files are
81 * skipped. Returns 1 when `child` describes a regular candidate,
82 * 0 to skip this entry, and sets *is_dir when a same-device subdir.
83 */
84static int
85reap_entry_candidate(const char *dir, const struct dirent *de,
86 const reap_ctx_t *rc, char *child, int *is_dir, time_t *last_touch)
87{
88 struct stat st;
89
90 *is_dir = 0;
91 *last_touch = 0;
92
93 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0) {
94 return 0;
95 }
96 if (brix_xmeta_is_sidecar_name(de->d_name)) {
97 return 0;
98 }
99 if (snprintf(child, PATH_MAX, "%s/%s", dir, de->d_name) >= PATH_MAX) {
100 return 0;
101 }
102 if (lstat(child, &st) != 0 || st.st_dev != rc->dev) {
103 return 0;
104 }
105 if (S_ISDIR(st.st_mode)) {
106 *is_dir = 1;
107 return 0;
108 }
109 /* Cold-age signal for the clean read-fill purge: the LATER of atime and
110 * mtime. atime alone is not trustworthy — relatime coarsens it and noatime
111 * freezes it entirely, which would make every file look ancient and purge a
112 * hot cache. Taking the later of the two degrades safely: on a noatime mount
113 * the age is measured from the fill instead of the last read, so the purge
114 * is conservative (it can only ever be too slow, never too eager). */
115 *last_touch = (st.st_atime > st.st_mtime) ? st.st_atime : st.st_mtime;
116 return S_ISREG(st.st_mode) ? 1 : 0;
117}

119/*
120 * WHAT: Classify WHY a tracked regular file is reapable (or that it is not).
121 * WHY: Concentrates the aged-dirty vs finished-write-back decision — the
122 * behaviour-critical eviction policy — in one place off the walk loop.
123 * HOW: Reads the cinfo state; keeps untracked files. Dirty data reaps only
124 * once aged past cutoff (incomplete when a flush was started, abandoned
125 * otherwise). Clean-but-flushed staging copies reap once the last flush
126 * ages out (COMPLETED). A clean read-fill is purged only when the
127 * cold-age policy is armed and it has gone untouched past cold_cutoff
128 * (COLD); otherwise it is left for occupancy-driven eviction. Returns 1
129 * with *reason set when reapable, else 0.
130 */
131static int
132reap_classify(const char *child, const reap_ctx_t *rc, time_t last_touch,
133 brix_cache_cinfo_state_t *cs, brix_cache_reap_reason_t *reason)
134{
135 if (brix_cache_cinfo_state(child, cs) != NGX_OK) {
136 return 0; /* no record → keep (untracked file) */
137 }
138
139 if (cs->is_dirty) {
140 /* Un-flushed data: reap only once it has aged past the max-age. */
141 if (cs->dirty_since == 0 || (time_t) cs->dirty_since > rc->cutoff) {
142 return 0; /* dirty but not yet aged → keep */
143 }
144 *reason = (cs->flush_gen > 0) ? BRIX_CACHE_REAP_INCOMPLETE
145 : BRIX_CACHE_REAP_ABANDONED;
146 return 1;
147 }
148 if (cs->flush_gen > 0) {
149 /* Clean AND written back at least once: a finished write-back staging
150 * copy whose bytes are safely on the origin. Reclaim it once the last
151 * flush has aged out. */
152 if (cs->last_flush == 0 || (time_t) cs->last_flush > rc->cutoff) {
153 return 0; /* completed but not yet aged → keep */
154 }
155 *reason = BRIX_CACHE_REAP_COMPLETED;
156 return 1;
157 }
158 /* Clean read-through fill (audit §4.2, upstream pfc.purgecoldfiles): purge
159 * once it has gone untouched past the cold horizon, INDEPENDENT of
160 * occupancy — a cache that never fills its watermark otherwise keeps cold
161 * objects forever. Losing them is free: they are re-fetchable from the
162 * origin. Disarmed (cold_cutoff == 0) this is the previous behaviour. */
163 if (rc->cold_cutoff != 0 && last_touch != 0
164 && last_touch <= rc->cold_cutoff) {
165 *reason = BRIX_CACHE_REAP_COLD;
166 return 1;
167 }
168 return 0; /* clean read-fill → keep (evictable) */
169}

171/*
172 * WHAT: Remove one classified-reapable file (data + sidecars), bump the
173 * metric, and log the reap event.
174 * WHY: Keeps the destructive edge (unlink/evict + counter + log) in a single
175 * helper so the walk loop reads as pure control flow.
176 * HOW: Removes the DATA through the cstore adapter (phase-64 P3/G5: the policy
177 * layer never unlinks via the store driver itself); cstore_evict also
178 * drops the object's cinfo + L1 entry. The .meta sidecar is the legacy
179 * stats plane the cstore does not own, dropped by reap_unlink_sidecars.
180 * The adapter is keyed by the path beneath the DATA root, which only
181 * exists for a legacy `brix_cache_root` cache — a store-configured cache
182 * (`brix_cache_store`) leaves it empty, and the store's own root is
183 * already where the walk is rooted, so those reap by direct unlink.
184 *
185 * The removal is then VERIFIED: the adapter is best-effort by contract
186 * (it returns OK even on a failed unlink), so an unremoved data file
187 * used to be reported as reaped and — with its .cinfo sidecar gone —
188 * looked untracked on every later pass and was never revisited: a leak
189 * that logged as a success. A survivor is unlinked directly, and a file
190 * that STILL survives is logged as an error and not counted.
191 */
192static void
193reap_remove(const char *child, const reap_ctx_t *rc,
194 const brix_cache_cinfo_state_t *cs, brix_cache_reap_reason_t reason)
195{
196 struct stat leftover;
197
198 if (rc->cstore != NULL && rc->data_root != NULL && rc->data_root[0] != '\0'
199 && ngx_strncmp(child, rc->data_root, ngx_strlen(rc->data_root)) == 0)
200 {
201 (void) brix_cstore_evict(rc->cstore, child + ngx_strlen(rc->data_root));
202 } else {
203 (void) unlink(child);
204 }
205 /* Verify, then fall back — never report a reap that did not happen. */
206 if (lstat(child, &leftover) == 0) {
207 (void) unlink(child);
208 }
209 if (lstat(child, &leftover) == 0) {
210 ngx_log_error(NGX_LOG_ERR, rc->log, 0,
211 "brix: cache reaper could not remove \"%s\" (left in place)",
212 child);
213 return;
214 }
215 reap_unlink_sidecars(child);
216
217 if (rc->slot != NULL) {
218 (void) ngx_atomic_fetch_add(&rc->slot->cache_dirty_reaped[reason], 1);
219 }
220 if (reason == BRIX_CACHE_REAP_COLD) {
221 ngx_log_error(NGX_LOG_NOTICE, rc->log, 0,
222 "brix: cache purged cold file (untouched past "
223 "brix_cache_cold_max_age, re-fetchable): \"%s\"", child);
224 } else if (reason == BRIX_CACHE_REAP_COMPLETED) {
225 ngx_log_error(NGX_LOG_NOTICE, rc->log, 0,
226 "brix: cache reaped completed write-back file "
227 "(flushed, reclaimed): \"%s\"", child);
228 } else {
229 ngx_log_error(NGX_LOG_WARN, rc->log, 0,
230 "brix: cache reaped stale-dirty file (reason=%s, %uL "
231 "un-flushed bytes discarded): \"%s\"",
232 reason == BRIX_CACHE_REAP_INCOMPLETE ? "incomplete"
233 : "abandoned",
234 (unsigned long) (cs->dirty_hi - cs->dirty_lo), child);
235 }
236}

238/*
239 * WHAT: Recursively scan `dir` (same device as root), reaping aged write-back
240 * files: aged-dirty data (abandoned/incomplete → discarded) and finished
241 * write-back staging copies (completed → reclaimed).
242 * WHY: The dirty-reaper's tree walk; bounds abandoned staging the eviction
243 * guard would otherwise protect forever.
244 * HOW: Per entry: reap_entry_candidate gates to a same-device regular file
245 * (recursing into same-device subdirs), reap_classify decides if/why it
246 * is reapable, reap_remove does the destructive work. Each reaped file
247 * bumps rc->slot->cache_dirty_reaped[reason] (slot may be NULL). Returns
248 * the count reaped in this subtree.
249 */
250static ngx_uint_t
251reap_dir(const char *dir, const reap_ctx_t *rc)
252{
253 DIR *dp;
254 struct dirent *de;
255 ngx_uint_t n = 0;
256 char child[PATH_MAX];
257
258 dp = opendir(dir);
259 if (dp == NULL) {
260 return 0;
261 }
262
263 while ((de = readdir(dp)) != NULL) {
264 brix_cache_cinfo_state_t cs;
265 brix_cache_reap_reason_t reason = BRIX_CACHE_REAP_ABANDONED;
266 int is_dir = 0;
267 time_t last_touch = 0;
268
269 if (!reap_entry_candidate(dir, de, rc, child, &is_dir, &last_touch)) {
270 if (is_dir) {
271 n += reap_dir(child, rc);
272 }
273 continue;
274 }
275 if (!reap_classify(child, rc, last_touch, &cs, &reason)) {
276 continue;
277 }
278 reap_remove(child, rc, &cs, reason);
279 n++;
280 }
281
282 closedir(dp);
283 return n;
284}

286ngx_uint_t
287brix_cache_reap_dirty(const ngx_stream_brix_srv_conf_t *conf, ngx_log_t *log)
288{
289 const char *root;
290 struct stat rs;
291 reap_ctx_t rc;
292
293 /* Either policy alone is reason enough to walk: the dirty horizon bounds
294 * abandoned write-back, the cold horizon purges untouched read-fills. */
295 if (conf == NULL
296 || (conf->cache_dirty_max_age == 0 && conf->cache_cold_max_age == 0)) {
297 return 0;
298 }
299 root = brix_cache_state_root(conf);
300 if (root == NULL || stat(root, &rs) != 0) {
301 return 0;
302 }
303
304 /* Data removal routes through the cstore adapter (P3/G5); data_root is
305 * cache_root (== the state root for the default co-located cache). The state
306 * tree is still walked raw above — it is the POSIX state plane (.cinfo state
307 * records), not the store's data objects, so it is not a store-driver touch. */
308 ngx_memzero(&rc, sizeof(rc));
309 rc.cutoff = time(NULL) - conf->cache_dirty_max_age;
310 /* 0 keeps the cold purge disarmed (reap_classify tests for it explicitly),
311 * so an unset horizon can never be read as "everything is cold". */
312 rc.cold_cutoff = (conf->cache_cold_max_age > 0)
313 ? time(NULL) - conf->cache_cold_max_age : 0;
314 rc.dev = rs.st_dev;
315 rc.log = log;
316 rc.slot = reap_metrics_slot(conf);
317 rc.cstore = brix_cache_storage_cstore(conf);
318 rc.data_root = (const char *) conf->cache_root.data;
319
320 return reap_dir(root, &rc);
321}
