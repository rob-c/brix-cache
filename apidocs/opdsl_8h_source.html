1/*
2 * opdsl.h — declarative operation pipeline for the BriX client (§7.16, the
3 * XrdClOperations analog).
4 *
5 * WHAT: compose a sequence of client operations as data — each a step that runs
6 * against a brix_conn — and execute them as a pipeline that SHORT-CIRCUITS on
7 * the first failure into an optional "otherwise" handler (the Operation |
8 * Operation ... >> Handle / Then-Else model of XrdCl).
9 * WHY: callers currently hand-roll open→read→close ladders with ad-hoc error
10 * cleanup at every site. A declarative chain makes the success path linear and
11 * the failure path single-sourced, exactly as the upstream C++ DSL does, in
12 * plain C.
13 * HOW: brix_opd_new() builds an empty pipeline; brix_opd_step() appends a step
14 * (a function pointer + opaque arg); brix_opd_otherwise() sets the failure
15 * handler; brix_opd_run() executes steps in order until one returns non-zero,
16 * then runs the otherwise-handler (if any) and returns -1. All-success → 0.
17 * Pure control-flow: the steps themselves bind to the brix_file_open_read,
18 * brix_stat, brix_file_read (etc.) calls, so the engine has zero wire
19 * knowledge and is unit-testable with stubs.
20 */
21#ifndef BRIX_OPS_OPDSL_H
22#define BRIX_OPS_OPDSL_H

26/* A single pipeline step: run `arg` against `c`, reporting through `st`.
27 * Return 0 to continue the pipeline, non-zero to fail it. */
28typedef int (*brix_opd_step_fn)(brix_conn *c, void *arg, brix_status *st);

30/* Opaque pipeline handle. */
31typedef struct brix_opd brix_opd_t;

33/* Create an empty pipeline, or NULL on OOM. */

36/* Append a step (label is borrowed, for diagnostics). Returns the same pipeline
37 * for chaining, or NULL if the pipeline/args are invalid (the pipeline is left
38 * usable). */
39brix_opd_t *brix_opd_step(brix_opd_t *p, const char *label,

42/* Set the failure handler run after the first failing step (the Else leg).
43 * Optional; at most one. */
44void brix_opd_otherwise(brix_opd_t *p, brix_opd_step_fn fn, void *arg);
45
46/* Execute the pipeline against `c`. Runs steps in order; on the first step that
47 * returns non-zero, runs the otherwise-handler (if set) and returns -1 with the
48 * failing step's status in *st. All steps succeeding returns 0. The index of the
49 * failing step (or the count on success) is stored through *ran when non-NULL. */
50int brix_opd_run(brix_opd_t *p, brix_conn *c, brix_status *st, size_t *ran);
51
52/* Number of steps appended so far. */
53size_t brix_opd_len(const brix_opd_t *p);
54
55/* Free the pipeline (does not touch the step args — the caller owns those). */
56void brix_opd_free(brix_opd_t *p);

58#endif /* BRIX_OPS_OPDSL_H */
