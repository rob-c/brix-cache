13/* Per-connection reassembly state: MODE E blocks are framed independently on
14 * each data connection, so a partially-read header or payload belongs to ONE
15 * connection and never to the transfer as a whole. */
16typedef struct {
17 uint8_t hdr[FTP_EB_HDR];
18 size_t hdr_got;
19 uint64_t remain; /* payload bytes still owed for the current block */
20 uint64_t pos; /* absolute offset of the next payload byte */
21 int done; /* EOD seen, or the peer closed */
22} gftp_eb_conn_t;

24typedef struct {
25 gftp_mode_e_req_t *req;
26 gftp_eb_conn_t *cs;
27 ftp_eb_range_t *ranges;
28 size_t nranges;
29 uint64_t eod_seen;
30 long eof_total; /* -1 until an EOF block arrives */
31 int stopped; /* window edge reached; stop, do not fail */
32} gftp_eb_recv_t;

35/* Claim [lo, hi) for this transfer. Refuses a block outside the requested
36 * window and any block overlapping bytes already committed — see the security
37 * note in gftp_mode_e.h. Returns 0 / -1. */
38static int
39gftp_eb_claim(gftp_session_t *session, gftp_eb_recv_t *r, uint64_t off,
40 uint64_t count)
41{
42 uint64_t base = (uint64_t) r->req->start;
43
44 if (off < base || off - base > r->req->limit
45 || count > r->req->limit - (off - base))
46 {
47 gftp_set_error(session, EPROTO,
48 "GridFTP MODE E block at %llu is outside the requested window",
49 (unsigned long long) off);
50 return -1;
51 }
52 if (ftp_eb_range_overlaps(r->ranges, r->nranges, (off_t) off,
53 (off_t) (off + count)))
54 {
55 gftp_set_error(session, EPROTO,
56 "GridFTP MODE E block at %llu overlaps committed bytes",
57 (unsigned long long) off);
58 return -1;
59 }
60 if (r->nranges == GFTP_EB_MAX_RANGES) {
61 gftp_set_error(session, EPROTO,
62 "GridFTP MODE E transfer is too fragmented (%d ranges)",
63 GFTP_EB_MAX_RANGES);
64 return -1;
65 }
66 r->ranges[r->nranges].lo = (off_t) off;
67 r->ranges[r->nranges].hi = (off_t) (off + count);
68 r->nranges++;
69 return 0;
70}

73/* Is this block pure surplus - bytes the caller never asked for, from an origin
74 * that was never told where to stop?
75 *
76 * Only on an UNBOUNDED transfer (REST+RETR names no length), only once every
77 * byte of the window is committed, and only for a block that BEGINS at or past
78 * the end. All three are needed. Dropping the third would let an origin skip
79 * bytes and call the hole surplus; dropping the second would stop the receiver
80 * the moment the window happened to fill, which on any object smaller than the
81 * window is at EOF - and the `overlap` fault's replayed block arrives after
82 * exactly that point, so the refusal it exists to prove would never fire. */
83static int
84gftp_eb_surplus(const gftp_eb_recv_t *r, u_char desc, uint64_t off)
85{
86 uint64_t end = (uint64_t) r->req->start + r->req->limit;
87
88 return !r->req->bounded
89 && (desc & FTP_EB_EOF) == 0
90 && r->req->received >= r->req->limit
91 && off >= end;
92}

95/* How much of a block starting inside the window may be committed.
96 *
97 * Only ever shrinks, and only on an unbounded transfer: a 64 KiB chunk answering
98 * a 256-byte Range request straddles the end, and the bytes inside the window
99 * are still exactly the ones that were asked for. Committing the head of a
100 * straddling block leaves the connection mid-payload, so the caller stops
101 * immediately afterwards - the next 17 bytes on that socket are data, and
102 * reading them as a header would invent a block out of the file's contents. */
103static uint64_t
104gftp_eb_window_clamp(const gftp_eb_recv_t *r, uint64_t off, uint64_t count)
105{
106 uint64_t end = (uint64_t) r->req->start + r->req->limit;
107
108 if (r->req->bounded || off >= end || count <= end - off) {
109 return count;
110 }
111 return end - off;
112}

115/* A complete 17-byte header arrived on connection `i`: set up its block. */
116static int
117gftp_eb_header(gftp_session_t *session, gftp_eb_recv_t *r, unsigned i)
118{
119 gftp_eb_conn_t *c = &r->cs[i];
120 uint64_t count;
121 uint64_t offset;
122 u_char desc;
123
124 ftp_eb_unpack(c->hdr, &desc, &count, &offset);
125 c->hdr_got = 0;
126
127 if (gftp_eb_surplus(r, desc, offset)) {
128 r->stopped = 1;
129 return 0;
130 }
131 if (desc & FTP_EB_EOF) {
132 r->eof_total = (long) offset; /* no payload; EOD total is in OFFSET */
133 }
134 if ((desc & FTP_EB_EOF) == 0 && count > 0) {
135 uint64_t take = gftp_eb_window_clamp(r, offset, count);
136
137 if (gftp_eb_claim(session, r, offset, take) != 0) {
138 return -1;
139 }
140 r->stopped = (take < count) ? 1 : r->stopped;
141 c->remain = take;
142 c->pos = offset;
143 }
144 if (desc & FTP_EB_EOD) {
145 r->eod_seen++;
146 /* Any payload on an EOD block is drained first; the connection retires
147 * once `remain` reaches zero, which is why `done` is set here and the
148 * drain loop still runs. */
149 c->done = 1;
150 }
151 return 0;
152}

155/* Move up to one read's worth of payload from connection `i` to the sink. */
156static int
157gftp_eb_payload(gftp_session_t *session, gftp_eb_recv_t *r, unsigned i)
158{
159 uint8_t buf[GFTP_EB_CHUNK];
160 gftp_eb_conn_t *c = &r->cs[i];
161 size_t want = sizeof(buf);
162 ssize_t n;
163
164 if (c->remain < (uint64_t) want) {
165 want = (size_t) c->remain;
166 }
167 n = gftp_dc_read(session, &r->req->conns[i], buf, want);
168 if (n < 0) {
169 return -1;
170 }
171 if (n == 0) {
172 gftp_set_error(session, EPROTO,
173 "GridFTP MODE E connection closed mid-block");
174 return -1;
175 }
176 if (r->req->sink(r->req->ctx, (off_t) (c->pos - (uint64_t) r->req->start),
177 buf, (size_t) n) != 0)
178 {
179 return -1;
180 }
181 c->pos += (uint64_t) n;
182 c->remain -= (uint64_t) n;
183 r->req->received += (uint64_t) n;
184 return 0;
185}

188/* One readable event on connection `i`: header bytes, then payload bytes. */
189static int
190gftp_eb_step(gftp_session_t *session, gftp_eb_recv_t *r, unsigned i)
191{
192 gftp_eb_conn_t *c = &r->cs[i];
193 ssize_t n;
194
195 if (c->remain > 0) {
196 return gftp_eb_payload(session, r, i);
197 }
198 n = gftp_dc_read(session, &r->req->conns[i], c->hdr + c->hdr_got,
199 FTP_EB_HDR - c->hdr_got);
200 if (n < 0) {
201 return -1;
202 }
203 if (n == 0) {
204 if (c->hdr_got != 0) {
205 gftp_set_error(session, EPROTO,
206 "GridFTP MODE E connection closed mid-header");
207 return -1;
208 }
209 c->done = 1; /* clean close with no block pending */
210 return 0;
211 }
212 c->hdr_got += (size_t) n;
213 if (c->hdr_got < FTP_EB_HDR) {
214 return 0;
215 }
216 return gftp_eb_header(session, r, i);
217}

220/* True once every connection has retired and the EOF block's promised EOD count
221 * has been met. An EOF that promises more EODs than ever arrive leaves this
222 * false — a truncated transfer is never reported as complete. It is
223 * gftp_eb_poll() that then ends the transfer, as a failure: see the no-live-
224 * connection branch there. */
225static int
226gftp_eb_complete(const gftp_eb_recv_t *r)
227{
228 unsigned i;
229
230 for (i = 0; i < r->req->nconns; i++) {
231 if (!r->cs[i].done || r->cs[i].remain > 0) {
232 return 0;
233 }
234 }
235 return r->eof_total < 0 || r->eod_seen >= (uint64_t) r->eof_total;
236}

239/* Poll every live connection once; step each readable one. */
240static int
241gftp_eb_poll(gftp_session_t *session, gftp_eb_recv_t *r)
242{
243 struct pollfd fds[GFTP_STREAMS_MAX];
244 unsigned map[GFTP_STREAMS_MAX];
245 nfds_t n = 0;
246 unsigned i;
247 int rc;
248
249 for (i = 0; i < r->req->nconns; i++) {
250 if (r->cs[i].done && r->cs[i].remain == 0) {
251 continue;
252 }
253 fds[n].fd = r->req->conns[i].fd;
254 fds[n].events = POLLIN;
255 fds[n].revents = 0;
256 map[n] = i;
257 n++;
258 }
259 if (n == 0) {
260 /* Every connection has retired, yet the caller only polls while the
261 * transfer is INCOMPLETE — so the EOF block promised more EOD blocks
262 * than ever arrived. Nothing can still read: returning 0 here would
263 * spin `while (!gftp_eb_complete())` forever, burning a whole worker
264 * thread on an origin that merely has to lie about its EOD count once.
265 * A truncated transfer has to end, and it has to end as a failure. */
266 gftp_set_error(session, EPROTO,
267 "GridFTP MODE E transfer ended after %llu of %ld promised EOD blocks",
268 (unsigned long long) r->eod_seen, r->eof_total);
269 return -1;
270 }
271 do {
272 rc = poll(fds, n, session->timeout_ms);
273 } while (rc < 0 && errno == EINTR);
274 if (rc <= 0) {
275 gftp_set_error(session, (rc == 0) ? ETIMEDOUT : EIO,
276 "GridFTP MODE E data channel %s", (rc == 0) ? "timed out" : "failed");
277 return -1;
278 }
279 for (i = 0; i < (unsigned) n; i++) {
280 if (fds[i].revents != 0
281 && gftp_eb_step(session, r, map[i]) != 0)
282 {
283 return -1;
284 }
285 }
286 return 0;
287}

290int
291gftp_mode_e_receive(gftp_session_t *session, gftp_mode_e_req_t *req)
292{
293 gftp_eb_recv_t r;
294 gftp_eb_conn_t cs[GFTP_STREAMS_MAX];
295 int rc = 0;
296
297 req->received = 0;
298 if (req->nconns == 0 || req->nconns > GFTP_STREAMS_MAX) {
299 gftp_set_error(session, EINVAL, "GridFTP MODE E stream count is invalid");
300 return -1;
301 }
302 memset(&r, 0, sizeof(r));
303 memset(cs, 0, sizeof(cs));
304 r.req = req;
305 r.cs = cs;
306 r.eof_total = -1;
307 r.ranges = calloc(GFTP_EB_MAX_RANGES, sizeof(*r.ranges));
308 if (r.ranges == NULL) {
309 gftp_set_error(session, ENOMEM, "cannot allocate GridFTP MODE E ranges");
310 return -1;
311 }
312 while (rc == 0 && !r.stopped && !gftp_eb_complete(&r)) {
313 rc = gftp_eb_poll(session, &r);
314 }
315 /* A straddling block is drained to the window edge before the stop takes
316 * effect, so the loop runs while `stopped` is set and `remain` is non-zero
317 * on the connection carrying it. */
318 while (rc == 0 && r.stopped && r.cs[0].remain > 0) {
319 rc = gftp_eb_poll(session, &r);
320 }
321 req->stopped_early = r.stopped;
322 free(r.ranges);
323 return rc;
324}

327int
328gftp_mode_e_send(gftp_session_t *session, gftp_dc_t *dc, gftp_source_fn source,
329 void *ctx)
330{
331 uint8_t buf[GFTP_EB_CHUNK];
332 uint8_t hdr[FTP_EB_HDR];
333 uint64_t off = 0;
334
335 for (;;) {
336 ssize_t n = source(ctx, buf, sizeof(buf));
337
338 if (n < 0) {
339 return -1;
340 }
341 if (n == 0) {
342 break;
343 }
344 ftp_eb_pack(hdr, 0 /* data */, (uint64_t) n, off);
345 if (gftp_dc_write_all(session, dc, hdr, sizeof(hdr)) != 0
346 || gftp_dc_write_all(session, dc, buf, (size_t) n) != 0)
347 {
348 return -1;
349 }
350 off += (uint64_t) n;
351 }
352 /* One connection, so one EOD, and the EOF block promises exactly that —
353 * the same trailer the inbound door emits (ev/ftp_ev_mode_e.c). */
354 ftp_eb_pack(hdr, (uint8_t) (FTP_EB_EOF | FTP_EB_EOD), 0, 1);
355 return gftp_dc_write_all(session, dc, hdr, sizeof(hdr));
356}
