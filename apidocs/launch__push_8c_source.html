5 *
6 * WHAT: the two event-thread entry points of the push dialect.
7 * brix_tpc_prepare_push runs at kXR_open time: the operator opt-in
8 * (brix_tpc_push), the destination host/path validation, the source-egress
9 * allowlist and the cached SSRF verdict on the host this server would DIAL,
10 * then it PARKS the intent on ctx->tpc_push_pending and declines so the
11 * ordinary read-open resolves, authorizes and opens the local file.
12 * brix_tpc_push_apply_pending stamps that park onto the finished handle.
13 * brix_tpc_start_push runs at the second kXR_sync: it registers the transfer,

99/* WHAT: the security ladder, evaluated against the DESTINATION this server
100 * would dial. NGX_OK to proceed; NGX_OK is never returned after a refusal —
101 * *answered is set instead and holds the finished rc.
102 * WHY: keeps brix_tpc_prepare_push flat and the gates independently reviewable,
103 * exactly as tpc_prepare_check_preconditions does for the pull.
104 * HOW: opt-in off → kXR_Unsupported; no thread pool → kXR_ServerError; empty
105 * host/path/key → kXR_ArgInvalid; egress allowlist → kXR_NotAuthorized + guard
106 * signal + counter; cached SSRF verdict < 0 → kXR_NotAuthorized. A verdict of
107 * "unknown" proceeds: the connect-time recheck is the authoritative one. */
108static ngx_int_t
109tpc_push_check_preconditions(brix_ctx_t *ctx, ngx_connection_t *c,
110 ngx_stream_brix_srv_conf_t *conf, const brix_tpc_params_t *tpc,
111 ngx_int_t *answered)
112{
113 char policy_err[512];
114 char egress_err[512];
115 uint16_t dport;
116
117 if (!conf->tpc_push) {
118 *answered = tpc_push_refuse(ctx, c, tpc->dst_host, kXR_Unsupported,
119 "native TPC push is disabled (brix_tpc_push off)");
120 return NGX_ERROR;
121 }
122
123 if (conf->common.thread_pool == NULL) {
124 *answered = tpc_push_refuse(ctx, c, tpc->dst_host, kXR_ServerError,
125 "TPC push requires brix_thread_pool to be configured");

127 }
128

133 "invalid or incomplete TPC push destination");

135 }

138 * asked to dial must appear on brix_tpc_source_allow when the guard is on.
139 * Same list as the pull's: the directive names the peers this server may
140 * originate to, and a push originates to its destination. */
141 if (brix_tpc_source_guard_check(conf->common.tpc_source_guard,
142 conf->common.tpc_source_allow, tpc->dst_host,
143 egress_err, sizeof(egress_err))
144 != 0)
145 {
146 BRIX_SRV_METRIC_INC(ctx, tpc_egress_refused_total);
147 tpc_push_egress_signal(ctx, c, tpc->dst_host);

153 /*
154 * SSRF range gate. The verdict comes from an IP literal or the per-worker
155 * DNS cache; a definitively prohibited address refuses the open here. An
156 * unknown name is NOT parked on an async resolve as the pull's open is:
157 * the pull must decide before it CREATES the destination file, whereas a
158 * push has created nothing and dialled nothing at open time. The pull
159 * thread re-checks every candidate address it actually connects to
160 * (connect.c, I-DNS-3), so an offending address is still refused — one
161 * frame later, with no local side effect to undo.
162 */
163 dport = tpc->dst_port ? tpc->dst_port : TPC_DEFAULT_PORT;
164 if (brix_tpc_check_src_policy(conf, tpc->dst_host, dport, policy_err,
165 sizeof(policy_err)) < 0)
166 {
167 *answered = tpc_push_refuse(ctx, c, tpc->dst_host, kXR_NotAuthorized,
168 policy_err);
169 return NGX_ERROR;
170 }
171
172 return NGX_OK;
173}

249/* WHAT: register this push in the shared TPC transfer registry and return its
250 * id (0 on failure / registry full).
251 * WHY: a push is a transfer like any other — the dashboard, the metrics and the
252 * abandoned-transfer reaper must see it. The URL is built the same way as the
253 * pull's, but the REMOTE side is the destination, so src_url names the local
254 * file and dst_path the remote one.
255 * HOW: bracket an IPv6 literal destination, format "root://host:port/path",
256 * fill an brix_tpc_transfer_t (PROTO_STREAM/DIR_PUSH/PENDING), registry add. */
257static uint64_t
258tpc_register_push_transfer(ngx_connection_t *c, brix_file_t *file)
259{
260 brix_tpc_transfer_t transfer;
261 ngx_str_t src_url;
262 ngx_str_t dst_url;
263 u_char dst_buf[PATH_MAX + 320];
264 u_char *last;
265 char hostport[288];
266
267 if (file == NULL || file->path == NULL) {
268 return 0;
269 }
270
271 brix_format_host_port(file->tpc_src_host,
272 file->tpc_src_port ? file->tpc_src_port : TPC_DEFAULT_PORT,
273 hostport, sizeof(hostport));
274 last = ngx_snprintf(dst_buf, sizeof(dst_buf), "root://%s%s", hostport,
275 file->tpc_src_path);
276
277 src_url.data = (u_char *) file->path;
278 src_url.len = ngx_strlen(file->path);
279 dst_url.data = dst_buf;
280 dst_url.len = (size_t) (last - dst_buf);
281
282 ngx_memzero(&transfer, sizeof(transfer));
283 transfer.protocol = BRIX_TPC_PROTO_STREAM;
284 transfer.direction = BRIX_TPC_DIR_PUSH;
285 transfer.src_url = src_url;
286 transfer.dst_path = dst_url;
287 transfer.state = BRIX_TPC_STATE_PENDING;
288
289 return brix_tpc_registry_add(&transfer, c->log, 0);
290}
