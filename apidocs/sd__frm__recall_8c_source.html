1/*
2 * sd_frm_recall.c — the frm driver's recall family: the bounded synchronous
3 * recall every open and cache fill goes through, and the recall / recall_cred
4 * vtable verbs kXR_prepare and the cache tier drive.
5 *
6 * WHAT: frm_ensure_online() asks the bound MSS adapter for residency, starts a
7 * recall for an OFFLINE object and polls once; sd_frm_recall{,_cred}()
8 * wrap it for the driver vtable and book the one unified kind=tape
9 * ledger line for a genuine recall.
10 *
11 * WHY: Moved out of sd_frm.c when 2.0 F2 pushed it past the 600-line cap
12 * (coding-standards §1), the same split the staged-write family took.
13 * F2 adds the StageEvents feed lines (brix_frm_stagemsg): recall-begin /
14 * recall-online / recall-failed, keyed by the MSS key — never a
15 * credential.
16 *
17 * HOW: Same contract as before the move: 0 = online, -1 with errno ENOENT
18 * (absent), EAGAIN (an async MSS is still staging — HTTP 202 / kXR_wait
19 * park, §9.2; a retry re-polls and the eventual residency hit emits no
20 * line of its own) or EIO. `*recalled` says whether THIS call started
21 * a tape->cache recall so the ledger books real recalls only.
22 */

45int
46frm_ensure_online(sd_frm_state *st, const char *key, int *recalled)
47{
48 off_t sz = 0;
49 time_t mt = 0;
50 int res = st->mss->residency(st->mss_ctx, key, &sz, &mt);
51 int p;
52
53 if (recalled != NULL) { *recalled = 0; }
54 if (res == BRIX_RESIDENCY_ONLINE) {
55 return 0;
56 }
57 if (res == BRIX_RESIDENCY_ABSENT) {
58 errno = ENOENT;
59 return -1;
60 }
61 if (recalled != NULL) { *recalled = 1; }
62 brix_stage_events_emit("frm", "recall-begin", NULL, key, NULL);
63 errno = 0;
64 if (st->mss->recall_begin(st->mss_ctx, key) != 0) {
65 /* W3.1: a member missing from its sealed archive is NotFound, else EIO. */
66 return frm_recall_fail(key, (errno == ENOENT) ? ENOENT : EIO);
67 }
68 /* One poll: a synchronous adapter is online now; an async MSS is still
69 * staging → EAGAIN (HTTP 202 / root kXR_wait park, §9.2); a retry re-polls. */
70 p = st->mss->recall_poll(st->mss_ctx, key);
71 if (p == 1) {
72 brix_stage_events_emit("frm", "recall-online", NULL, key, NULL);
73 return 0;
74 }
75 if (p < 0) {
76 return frm_recall_fail(key, EIO);
77 }
78 errno = EAGAIN;
79 return -1;
80}

82static ngx_int_t
83sd_frm_recall_common(brix_sd_instance_t *inst, const char *key,
84 char reqid_out[40], const char *principal)
85{
86 sd_frm_state *st = SD_FRM_ST(inst);
87 ngx_log_t *log = (ngx_cycle != NULL) ? ngx_cycle->log : NULL;
88 int recalled = 0;
89
90 if (reqid_out != NULL) {
91 reqid_out[0] = '\0'; /* synchronous recall: no parking handle */
92 }
93 if (frm_ensure_online(st, key, &recalled) == 0) {
94 /* The cache-fill path (sd_cache) drives every nearline miss through this
95 * verb, so a genuine tape->cache recall books its one unified ledger
96 * line here (kind=tape, dir=in) — the sync counterpart to the async
97 * stage_engine RECALL emit (finding #12). Byte count = the now-online
98 * object size. An already-online object (recalled==0) is a plain fill. */
99 if (recalled) {
100 off_t sz = 0;
101 time_t mt = 0;
102
103 (void) st->mss->residency(st->mss_ctx, key, &sz, &mt);
104 brix_xfer_finish(BRIX_XFER_TAPE, "in", key, principal,
105 (size_t) (sz > 0 ? sz : 0), BRIX_XFER_OK, 0, log);
106 }
107 return NGX_OK; /* online now - the cache tier does a normal fill */
108 }
109 {
110 int e = errno;
111
112 /* Terminal recall failure books a kind=tape/error line; EAGAIN (async
113 * still in flight) is non-terminal and must not be recorded. */
114 if (recalled && e != EAGAIN) {
115 brix_xfer_finish(BRIX_XFER_TAPE, "in", key, principal, 0,
116 BRIX_XFER_SRC_ERR, e, log);
117 }
118 return (e == EAGAIN) ? NGX_AGAIN : NGX_ERROR;
119 }
120}
