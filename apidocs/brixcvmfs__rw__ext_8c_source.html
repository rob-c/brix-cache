30static int rw_mkdir(const char *path, mode_t mode) {
31 const char *pr = pt_rel(path);
32 if (pr != NULL) return mkdirat(g_writes_fd, pr, mode) != 0 ? -errno : 0;
33 if (rw_reserved(path)) return -EPERM;
34 if (merged_exists(path)) return -EEXIST;
35
36 int rc = rw_ensure_parents(path);
37 if (rc != 0) return rc;
38 rc = brix_overlay_mkdir(&g_ov, ov_rel(path), mode);
39 if (rc != 0) return rc;
40 brix_overlay_whiteout_clear(&g_ov, ov_rel(path));
41
42 cvmfs_dirent_t e; /* re-created deleted lower dir */
43 if (lower_resolve(path, &e) == 1 && lower_is_dir(&e))
44 brix_overlay_set_opaque(&g_ov, ov_rel(path));
45 return 0;
46}

48static int rw_unlink(const char *path) {
49 const char *pr = pt_rel(path);
50 if (pr != NULL) return unlinkat(g_writes_fd, pr, 0) != 0 ? -errno : 0;
51 if (rw_reserved(path)) return -ENOENT; /* invisible vocabulary */
52
53 ov_loc_t l;
54 int rc = ov_locate(path, &l);
55 if (rc != 0) return rc;
56 if (l.s == BRIX_OV_MASKED) return -ENOENT;
57
58 if (l.s == BRIX_OV_UPPER) {
59 if (S_ISDIR(l.st.st_mode)) return -EISDIR;
60 rc = brix_overlay_unlink_upper(&g_ov, ov_rel(path));
61 if (rc != 0) return rc;
62 return l.lower == 1 ? brix_overlay_whiteout_set(&g_ov, ov_rel(path)) : 0;
63 }
64 if (l.lower == 0) return -ENOENT;
65 if (lower_is_dir(&l.e)) return -EISDIR;
66 return brix_overlay_whiteout_set(&g_ov, ov_rel(path));
67}

81/*
82 * WHAT: rmdir shape check — is `path` a directory in the merged view?
83 * WHY: rmdir must refuse masked/absent paths and non-directories in either
84 * layer, in exactly this order, before the emptiness scan pays for a
85 * catalog listing.
86 * HOW: classify upper first (its errors win), then resolve lower; fills `l`
87 * and `*lower_dir` for the removal step on success.
88 */
89static int rw_rmdir_shape(const char *path, ov_loc_t *l, int *lower_dir) {
90 int rc = rw_classify(path, &l->st, &l->s);
91 if (rc != 0) return rc;
92 if (l->s == BRIX_OV_MASKED) return -ENOENT;
93 if (l->s == BRIX_OV_UPPER && !S_ISDIR(l->st.st_mode)) return -ENOTDIR;
94
95 l->lower = lower_resolve(path, &l->e);
96 if (l->lower < 0) return -EIO;
97 *lower_dir = l->lower == 1 && lower_is_dir(&l->e);
98 if (l->s == BRIX_OV_NONE && !*lower_dir)
99 return l->lower == 1 ? -ENOTDIR : -ENOENT;
100 return 0;
101}

103/*
104 * WHAT: merged-empty check — does the union view of dir `path` hold anything?
105 * WHY: any real upper entry, or any un-whiteouted lower entry (unless the
106 * dir is opaque), blocks the removal — same rule as kernel overlayfs.
107 * HOW: scan the upper nameset for 'u' entries first; only when that comes up
108 * empty walk the lower listing with the whiteout set as a filter.
109 */
110static int rw_dir_merged_nonempty(const char *path, int lower_dir, int *nonempty) {
111 brix_ov_nameset set;
112 int opaque = 0;
113 int rc = brix_overlay_read_upper(&g_ov, ov_rel(path), &set, &opaque);
114 if (rc != 0) return rc;
115 *nonempty = 0;
116 for (size_t i = 0; !*nonempty; i++) {
117 char flag = 0;
118 const char *name = brix_ov_nameset_at(&set, i, &flag);
119 if (name == NULL) break;
120 if (flag == 'u') *nonempty = 1;
121 }
122 if (!*nonempty && lower_dir && !opaque) {
123 rw_lowcount_t c = { &set, 0 };
124 cvmfs_catalog_readdir(brixcvmfs_client()->root_catalog,
125 brixcvmfs_cat_path(path), rw_lowcount_cb, &c);
126 *nonempty = c.nonempty;
127 }
128 brix_ov_nameset_free(&set);
129 return 0;
130}

132static int rw_rmdir(const char *path) {
133 const char *pr = pt_rel(path);
134 if (pr != NULL) return unlinkat(g_writes_fd, pr, AT_REMOVEDIR) != 0 ? -errno : 0;
135 if (rw_reserved(path)) return -ENOENT;
136
137 ov_loc_t l;
138 int lower_dir = 0;
139 int rc = rw_rmdir_shape(path, &l, &lower_dir);
140 if (rc != 0) return rc;
141
142 int nonempty = 0;
143 rc = rw_dir_merged_nonempty(path, lower_dir, &nonempty);
144 if (rc != 0) return rc;
145 if (nonempty) return -ENOTEMPTY;
146
147 if (l.s == BRIX_OV_UPPER) {
148 rc = brix_overlay_rmdir_upper(&g_ov, ov_rel(path));
149 if (rc != 0) return rc;
150 }
151 return lower_dir ? brix_overlay_whiteout_set(&g_ov, ov_rel(path)) : 0;
152}

154/*
155 * WHAT: rename admission checks shared by every union rename.
156 * WHY: flag/vocabulary/NOREPLACE gating is identical for files and dirs and
157 * must run before either side is resolved.
158 * HOW: pure predicate chain, 0 or -err.
159 */
160static int rw_rename_preflight(const char *from, const char *to, unsigned int flags) {
161 if (flags & ~(unsigned int) RENAME_NOREPLACE) return -EINVAL;
162 if (rw_reserved(from) || rw_reserved(to)) return -EPERM;
163 if ((flags & RENAME_NOREPLACE) && merged_exists(to)) return -EEXIST;
164 return 0;
165}

167/* target sanity: renaming anything onto a visible directory is refused */
168static int rw_rename_target_ok(const char *to) {
169 ov_loc_t t;
170 int rc = ov_locate_visible(to, &t);
171 if (rc != 0) return rc;
172 if (t.s == BRIX_OV_UPPER && S_ISDIR(t.st.st_mode)) return -EISDIR;
173 if (t.lower == 1 && lower_is_dir(&t.e)) return -EISDIR;
174 return 0;
175}

177/*
178 * WHAT: the shared rename tail — move the upper entry `from` → `to`.
179 * WHY: files and pure-upper dirs finish identically: mirror the target's
180 * parent chain, move in upper, unmask the target, and (when the source
181 * still exists in lower) whiteout the old name so it disappears.
182 * HOW: `mask_from` says whether lower still shows the source.
183 */
184static int rw_rename_commit(const char *from, const char *to, int mask_from) {
185 int rc = rw_ensure_parents(to);
186 if (rc != 0) return rc;
187 rc = brix_overlay_rename_upper(&g_ov, ov_rel(from), ov_rel(to));
188 if (rc != 0) return rc;
189 brix_overlay_whiteout_clear(&g_ov, ov_rel(to));
190 return mask_from ? brix_overlay_whiteout_set(&g_ov, ov_rel(from)) : 0;
191}

193/*
194 * WHAT: directory rename — same-layer (pure-upper) fast path only.
195 * WHY: only a pure-upper dir moves atomically; anything still merged with
196 * lower is -EXDEV (mv falls back to copy+delete, exactly like kernel
197 * overlayfs without redirect_dir).
198 * HOW: an opaque upper dir over a lower dir counts as pure-upper.
199 */
200static int rw_rename_dir(const char *from, const char *to,
201 int from_upper_dir, int from_lower_dir) {
202 if (!from_upper_dir) return -EXDEV;
203 brix_ov_nameset set;
204 int opaque = 0;
205 int rc = brix_overlay_read_upper(&g_ov, ov_rel(from), &set, &opaque);
206 if (rc != 0) return rc;
207 brix_ov_nameset_free(&set);
208 if (from_lower_dir && !opaque) return -EXDEV;
209 return rw_rename_commit(from, to, from_lower_dir);
210}

212/*
213 * WHAT: file rename — cross-layer copy-up path.
214 * WHY: a lower-only file must be materialised in upper before it can move;
215 * lower symlinks never move (no copy-up semantics for them).
216 * HOW: copy-up when lower-only, then the shared commit tail.
217 */
218static int rw_rename_file(const char *from, const char *to, const ov_loc_t *f) {
219 if (f->s == BRIX_OV_NONE) {
220 if (f->lower == 0) return -ENOENT;
221 if (!lower_is_file(&f->e)) return -EPERM; /* lower symlinks don't move */
222 int rc = rw_copyup(from, &f->e);
223 if (rc != 0) return rc;
224 }
225 return rw_rename_commit(from, to, f->lower == 1);
226}

228static int rw_rename(const char *from, const char *to, unsigned int flags) {
229 const char *prf = pt_rel(from), *prt = pt_rel(to);
230 if ((prf != NULL) != (prt != NULL)) return -EXDEV; /* no crossing the seam */
231 if (prf != NULL)
232 return renameat(g_writes_fd, prf, g_writes_fd, prt) != 0 ? -errno : 0;
233
234 int rc = rw_rename_preflight(from, to, flags);
235 if (rc != 0) return rc;
236
237 ov_loc_t f;
238 rc = ov_locate(from, &f);
239 if (rc != 0) return rc;
240 if (f.s == BRIX_OV_MASKED) return -ENOENT;
241
242 rc = rw_rename_target_ok(to);
243 if (rc != 0) return rc;
244
245 int from_lower_dir = f.lower == 1 && lower_is_dir(&f.e);
246 int from_upper_dir = f.s == BRIX_OV_UPPER && S_ISDIR(f.st.st_mode);
247 if (from_upper_dir || from_lower_dir)
248 return rw_rename_dir(from, to, from_upper_dir, from_lower_dir);
249 return rw_rename_file(from, to, &f);
250}

264static int rw_readlink(const char *path, char *buf, size_t size) {
265 const char *pr = pt_rel(path);
266 if (pr != NULL) {
267 ssize_t n = readlinkat(g_writes_fd, pt_at(pr), buf, size - 1);
268 if (n < 0) return -errno;
269 buf[n] = '\0';
270 return 0;
271 }
272 brix_ov_state s;
273 struct stat st;
274 int rc = rw_classify(path, &st, &s);
275 if (rc != 0) return rc;
276 if (s == BRIX_OV_UPPER) return brix_overlay_readlink(&g_ov, ov_rel(path), buf, size);
277 if (s == BRIX_OV_MASKED) return -ENOENT;
278 return brixcvmfs_op_readlink(path, buf, size);

285typedef int (*rw_meta_fn)(const char *rel, void *arg);

287static int rw_meta_apply(const char *path, rw_meta_fn apply, void *arg) {
288 if (rw_reserved(path)) return -EPERM;
289
290 struct stat st;
291 brix_ov_state s;
292 int rc = rw_classify(path, &st, &s);
293 if (rc != 0) return rc;
294 if (s == BRIX_OV_UPPER) return apply(ov_rel(path), arg);
295 if (s == BRIX_OV_MASKED) return -ENOENT;
296
297 cvmfs_dirent_t e;
298 int lower = lower_resolve(path, &e);
299 if (lower < 0) return -EIO;
300 if (lower == 0) return -ENOENT;
301
302 if (lower_is_dir(&e)) {
303 rc = rw_ensure_parents(path);
304 if (rc == 0) rc = brix_overlay_mkdirs(&g_ov, ov_rel(path), rw_lower_dir_mode, NULL);
305 } else if (lower_is_file(&e)) {
306 rc = rw_copyup(path, &e);
307 } else {
308 return -EOPNOTSUPP; /* lower symlink metadata */
309 }
310 return rc != 0 ? rc : apply(ov_rel(path), arg);
311}

366static int rw_getxattr(const char *path, const char *name, char *value, size_t size) {
367 if (pt_rel(path) != NULL) return -ENODATA;
368 if (strcmp(name, "user.overlay") == 0) {
369 const char *v = rw_overlay_xattr(path);
370 if (v == NULL) return -ENODATA;
371 size_t n = strlen(v);
372 if (size == 0) return (int) n; /* size probe */
373 if (n > size) return -ERANGE;
374 memcpy(value, v, n);
375 return (int) n;
376 }
377 brix_ov_state s;
378 struct stat st;
379 if (rw_classify(path, &st, &s) != 0) return -ENODATA;
380 if (s == BRIX_OV_UPPER) return -ENODATA; /* cvmfs attrs are lower-only */
381 if (s == BRIX_OV_MASKED) return -ENODATA;
382 return brixcvmfs_op_getxattr(path, name, value, size);
383}

386 static const char mine[] = "user.overlay";
387 int n = brixcvmfs_op_listxattr(path, NULL, 0);
388 if (n < 0) n = 0;

424int brixcvmfs_setup_rw(const char *mnt, const char *writes_override) {
425 char dir[600];
426 if (writes_override != NULL && writes_override[0] != '\0') {
427 snprintf(dir, sizeof(dir), "%s", writes_override);
428 } else {
429 mkdir(mnt, 0755); /* ensure the mountpoint exists */
430 snprintf(dir, sizeof(dir), "%s/" BRIX_OV_DIRNAME, mnt);
431 }
432 if (mkdir(dir, 0755) != 0 && errno != EEXIST) {
433 fprintf(stderr, "brixcvmfs: cannot create overlay dir %s\n", dir);
434 return -1;
435 }
436 g_writes_fd = open(dir, O_RDONLY | O_DIRECTORY | O_CLOEXEC);
437 if (g_writes_fd < 0 || brix_overlay_init(&g_ov, g_writes_fd) != 0) {
438 fprintf(stderr, "brixcvmfs: overlay at %s unusable\n", dir);
439 if (g_writes_fd >= 0) { close(g_writes_fd); g_writes_fd = -1; }
440 return -1;
441 }
442 return 0;
443}
