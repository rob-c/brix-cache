12/*
13 * kXR_bind — attach a secondary TCP data channel to an existing session.
14 *
15 * Secondary connections are used by xrdcp for parallel data transfer. The
16 * client establishes a primary connection (handshake + kXR_login + auth),
17 * then opens additional TCP connections that skip login and send kXR_bind
18 * with the primary session's sessid.
19 *
20 * The server assigns a pathid (1–253) to the secondary. The client then
21 * tags kXR_read/kXR_write payloads with this pathid to indicate which data
22 * channel the request arrived on (allowing the server to load-balance or
23 * multiplex across channels).
24 *
25 * Bound connections are intentionally narrower than full sessions: they may
26 * read primary-published handles, but they may not open, close, write, stat,
27 * or otherwise manipulate files independently. The primary remains the
28 * authority that decides which handles exist; secondaries are data channels.
29 */
30
31/* §1.4: the attach core shared by the local path and the migration target.
32 * Runs everything AFTER the sessid is known: registry lookup, identity
33 * inheritance, pathid assignment, offload registration and the kXR_ok reply
34 * (streamid read from ctx->recv.cur_streamid — the adopting worker stamps it
35 * from the migrate message before calling). */
36ngx_int_t
37brix_bind_attach(brix_ctx_t *ctx, ngx_connection_t *c,
38 const u_char sessid[BRIX_SESSION_ID_LEN])
39{
40 unsigned claimed_pathid;
41 u_char pathid;
42 u_char *buf;
43 size_t total;
44 ngx_uint_t token_auth = 0;
45 ngx_uint_t request_signing = 0;
46
47 /*
48 * The primary connection inserts its session registry entry at the point
49 * where the session is allowed to perform file I/O: anonymous login for
50 * auth=none, or successful kXR_auth for authenticated deployments.
51 */
52 if (!brix_session_lookup(sessid,
53 ctx->login.dn, sizeof(ctx->login.dn),
54 ctx->login.vo_list, sizeof(ctx->login.vo_list),
55 &token_auth, &request_signing))
56 {
57 brix_log_access(ctx, c, "BIND", "-", "-",
58 0, kXR_NotAuthorized,
59 "sessid not found in session registry", 0);
60 return brix_send_error(ctx, c, kXR_NotAuthorized,
61 "bind sessid not recognised");
62 }
63 ctx->token.auth = (int) token_auth;
64 ctx->sigver.inherited_required = request_signing ? 1 : 0;
65 if (ctx->identity != NULL) {
66 if (token_auth) {
67 if (brix_identity_set_subject(ctx->identity, c->pool, ctx->login.dn,
68 BRIX_AUTHN_TOKEN) != NGX_OK
69 || brix_identity_set_cstr(c->pool, &ctx->identity->dn,
70 ctx->login.dn) != NGX_OK)
71 {
72 return NGX_ERROR;
73 }
74 } else if (brix_identity_set_dn(ctx->identity, c->pool, ctx->login.dn,
75 BRIX_AUTHN_GSI) != NGX_OK)
76 {
77 return NGX_ERROR;
78 }
79 if (brix_identity_set_vos_csv(ctx->identity, c->pool,
80 ctx->login.vo_list) != NGX_OK)
81 {
82 return NGX_ERROR;
83 }
84 }
85
86 /* Claim and publish an unused path ID under the registry mutex. This is
87 * also the enforcement point for the advertised per-session bind limit. */
88 if (brix_session_pathid_claim(sessid, BRIX_BIND_MAX,
89 &claimed_pathid) != 1)
90 {
91 brix_log_access(ctx, c, "BIND", "-", "-", 0, kXR_ArgTooLong,
92 "session data-substream limit reached", 0);
93 return brix_send_error(ctx, c, kXR_ArgTooLong,
94 "maximum bound data streams reached");
95 }
96 pathid = (u_char) claimed_pathid;
97
98 total = XRD_RESPONSE_HDR_LEN + 1;
99 buf = ngx_palloc(c->pool, total);
100 if (buf == NULL) {
101 brix_session_pathid_unbind(sessid, claimed_pathid);
102 return NGX_ERROR;
103 }
104
105 /* Bind the session: record the primary's sessid and inherit auth state. */
106 ngx_memcpy(ctx->bound_sessid, sessid, BRIX_SESSION_ID_LEN);
107 ctx->is_bound = 1;
108 ctx->pathid = (int) pathid;
109 ctx->login.logged_in = 1; /* secondary skips kXR_login */
110 ctx->login.auth_done = 1; /* identity inherited from registry lookup above */
111
112 /* §1.2: publish the pathid on the session's registry entry so ANY worker
113 * can validate pathid-tagged requests against the session's live binds
114 * (stock refuses an unbound pathid with kXR_ArgInvalid). Cleared again in
115 * the disconnect path when this secondary goes away. */
116 /* §1.1 response offloading: record this secondary's per-worker connection so
117 * a later read/readv handler can route a pathid-tagged response out its
118 * socket. Best-effort — a full table just means this channel keeps using the
119 * control stream (never an error). Cleared in the disconnect path. */
120 (void) brix_offload_register(ctx->bound_sessid, (unsigned) pathid, c);
121
122 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, c->log, 0,
123 "brix: kXR_bind: pathid=%d sessid=%02xd%02xd...",
124 (int) pathid,
125 (unsigned) sessid[0],
126 (unsigned) sessid[1]);
127
128 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok, 1,
129 (ServerResponseHdr *) buf);
130 buf[XRD_RESPONSE_HDR_LEN] = pathid;
131
132 brix_log_access(ctx, c, "BIND", "-", "-", 1, 0, NULL, 0);
133
134 return brix_queue_response(ctx, c, buf, total);
135}

137/* Handle kXR_bind — attach a secondary TCP data channel to an existing primary
138 * session: look the session up in the registry and assign a pathid (1-253; 0 is
139 * the primary). Bound connections are capability-restricted (e.g. read primary-
140 * published handles via pathid-tagged kXR_read).
141 *
142 * §1.4: when the session's primary lives on ANOTHER worker (reuseport scatters
143 * the client's connections across workers), first try to migrate this
144 * secondary's fd to the owning worker over the pre-fork channel pair — response
145 * offloading (§1.1-§1.3) requires primary and secondary on the same event
146 * loop. On success the owner adopts the socket and sends the bind reply; the
147 * local connection is abandoned by returning NGX_ERROR, whose recv-loop BREAK
148 * teardown closes only this worker's fd reference without writing a byte (the
149 * finalize status is a log-side artifact — nothing is on the wire). */
150ngx_int_t
151brix_handle_bind(brix_ctx_t *ctx, ngx_connection_t *c,
152 ngx_stream_brix_srv_conf_t *conf)
153{
154 xrdw_sessid_req_t req;
155
156 /*
157 * brix_data_substreams off: refuse the bind so the client streams every
158 * request — and its data — inline on the primary connection (pathid 0). A
159 * data path is an optimisation, not a requirement (go-hep session.go), so a
160 * refused bind is a clean fallback; this is the supported posture when
161 * fronting a client that would otherwise stream WRITE payloads on a secondary
162 * connection, which BriX does not yet service as a cross-connection data-path.
163 */
164 if (!conf->common.data_substreams) {
165 brix_log_access(ctx, c, "BIND", "-", "-", 0, kXR_Unsupported,
166 "data substreams disabled (brix_data_substreams off)", 0);
167 return brix_send_error(ctx, c, kXR_Unsupported,
168 "data substreams disabled on this server");
169 }
170
171 xrdw_sessid_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
172
173 if (brix_bind_migrate_try(ctx, c, req.sessid) == NGX_OK) {
174 return NGX_ERROR; /* fd handed to the owning worker; quiet local close */
175 }
176
177 return brix_bind_attach(ctx, c, req.sessid);
178}
