1/*
2 * protbind.h — per-host authentication-protocol binding policy (XRootD
3 * `sec.protbind`), expressed protocol-agnostically.
4 *
5 * WHAT: Declares the rule/result types and the pure engine that turns an
6 * ordered list of `<host-template> [none | [only] <proto>...]` rules
7 * plus a fallback protocol set into the ORDERED set of authentication
8 * protocols a particular peer may use.
9 *
10 * WHY: BriX historically bound exactly one auth scheme per listener
11 * (`brix_auth`), with `both` as the single hard-coded composition of
12 * ztn+gsi. Stock XRootD instead binds an arbitrary ORDERED protocol
13 * list per host template, which is what sites use to say "grid hosts
14 * must use gsi, the local batch farm may use sss or unix, everyone
15 * else gets tokens". Expressing that as one generic engine — with no
16 * dependency on the root:// session, the HTTP request, or any storage
17 * backend — lets every VFS frontend share ONE policy implementation and
18 * one config grammar instead of growing a private copy each.
19 *
20 * HOW: Callers build a *base* set (what the listener would allow with no
21 * protbind rules at all — `brix_protbind_base_set` does that for the
22 * stream `brix_auth` modes, `brix_protbind_http_base` for the HTTP
23 * credential sources), then call `brix_protbind_resolve` with the peer's
24 * hostname and/or IP. The first matching rule wins; `none` yields an
25 * empty set with require_auth == 0, `only` yields exactly the listed
26 * protocols, and the default form yields the listed protocols followed
27 * by the base protocols not already named. `brix_protbind_allows` then
28 * gates an offered credential and the ordered `protos[]` drives whatever
29 * the frontend advertises (the root:// "&P=" sec token, the order HTTP
30 * credential sources are tried in).
31 *
32 * Requires: nginx core headers (ngx_config.h / ngx_core.h) and
33 * core/types/tunables.h (BRIX_AUTH_*) before inclusion.
34 */
35

39/* Upper bound on protocols in one rule / one resolved set. Seven schemes
40 * exist (gsi, ztn, sss, unix, krb5, host, pwd); 8 leaves one slot of slack
41 * and keeps both structs small enough to live on the stack. */
42#define BRIX_PROTBIND_MAX_PROTOS 8
43
44/* Rule modes — the three shapes of the directive's tail. */
45#define BRIX_PROTBIND_ALL 0 /* listed protocols first, then the base set */
46#define BRIX_PROTBIND_ONLY 1 /* listed protocols and nothing else */
47#define BRIX_PROTBIND_NONE 2 /* matching hosts authenticate with nothing */
48
49/*
50 * brix_protbind_rule_t — one parsed `brix_protbind` directive.
51 *
52 * host_tpl is the raw template (`*`, `*.cern.ch`, `lxplus*`, `pps*.gridpp.uk`,
53 * or a literal name/address); protos[] holds `count` BRIX_AUTH_* ids in the
54 * order they were written, which IS the advertisement order.
55 */
56typedef struct {
57 ngx_str_t host_tpl;
58 ngx_uint_t mode; /* BRIX_PROTBIND_* */
59 ngx_uint_t count; /* protocols listed */
60 ngx_uint_t protos[BRIX_PROTBIND_MAX_PROTOS];/* BRIX_AUTH_* ids */
61} brix_protbind_rule_t;
62
63/*
64 * brix_protbind_set_t — the resolved, ordered protocol set for one peer.
65 *
66 * require_auth is 0 only for the `none` outcome (and for a base set built from
67 * an auth-less listener): the caller must then complete the session
68 * anonymously rather than advertising an empty credential list.
69 */
70typedef struct {
71 ngx_uint_t count;
72 ngx_uint_t protos[BRIX_PROTBIND_MAX_PROTOS];
73 ngx_flag_t require_auth;
74} brix_protbind_set_t;

82/* Does `host` match one template? A single `*` acts as a wildcard span, so
83 * "*", "pref*", "*suffix" and "pref*suffix" all work (XrdOucNList semantics);

85 * matches even a NULL/empty host, so `*` rules never need a DNS lookup. */

88/* Would evaluating `rules` ever need a resolved peer HOSTNAME (i.e. is any
