2 * vfs_backend_config_s3.c — the s3:// and root(s)://-or-local-driver origin
3 * parsers plus their per-driver registry-entry builders (phase-79 file-size
4 * split of vfs_backend_config.c).
5 *
6 * WHAT: Parses "s3://host[:port]/bucket" into an S3 source backend and
7 * "root://host:port" / "roots://host:port" into a remote root:// primary

9 * brix_vfs_backend_config). Includes the s3 and xroot entry builders.
10 * WHY: The s3 and root(s):// network-origin schemes form the last cohesive

24/* Register an S3 source backend for an export (phase-64): the export's bytes
25 * live in a remote S3 bucket, served over the shared libcurl S3 transport
26 * (signed Range GET; writes are staged whole-object uploads — single PUT or
27 * MPU — via the driver's .staged_* slots, plus .unlink/DELETE). bucket is the
28 * path-style bucket name; port defaults to 80/443 by tls when the URL omits
29 * it. The driver caps are CAP_RANGE_READ|CAP_MEMFILE — no CAP_RANDOM_WRITE, so
30 * in-place partial writes are rejected at the cap layer while sequential
31 * uploads go through the staged path. */
32void
33brix_vfs_backend_config_s3(const char *root_canon, const char *host,
34 int port, int tls, const char *bucket,
35 const brix_vfs_s3_origin_opts_t *opts)
36{
37 brix_vfs_backend_entry_t *e;
38
39 if (root_canon == NULL || root_canon[0] == '\0' || host == NULL
40 || host[0] == '\0' || bucket == NULL || bucket[0] == '\0'
41 || port <= 0 || port > 65535)
42 {
43 return;
44 }
45 e = brix_vfs_backend_entry_claim(root_canon, "s3");
46 if (e == NULL) {
47 return;
48 }
49 /* origin_path carries the bucket */
50 brix_vfs_backend_set_origin(e, host, port, tls, bucket,
51 (opts != NULL) ? opts->put_checksum : 0);
52 /* Both are stamped unconditionally so a reload that drops "?nearline=1"
53 * leaves the export a plain bucket rather than keeping the previous cycle's
54 * declaration — the cap commits the export to a cache tier, so a stale one
55 * would be a config-time failure nobody asked for. */
56 e->origin_nearline = (opts != NULL && opts->nearline) ? 1 : 0;
57 e->origin_restore_days = (opts != NULL) ? opts->restore_days : 0;
58}

94/* Split an s3:// authority "host[:port]" into host length and numeric port,
95 * defaulting to 7480 (radosgw) when no ":port" is present. Returns NGX_OK, or
96 * NGX_ERROR after an [emerg] for a bad port. */
97static ngx_int_t
98vfs_parse_s3_port(ngx_conf_t *cf, const u_char *authority,
99 size_t authority_len, vfs_origin_parse_t *out)
100{
101 const u_char *colon = NULL;
102 size_t i;
103
104 for (i = authority_len; i > 0; i--) {
105 if (authority[i - 1] == ':') {
106 colon = authority + i - 1;
107 break;
108 }
109 }
110
111 if (colon == NULL) {
112 out->host_len = authority_len;
113 out->port = 7480; /* radosgw S3 default */
114 return NGX_OK;
115 }
116
117 {
118 ngx_int_t pn;
119 pn = ngx_atoi((u_char *) colon + 1,
120 (size_t) (authority + authority_len - (colon + 1)));
121 if (pn == NGX_ERROR || pn <= 0 || pn > 65535) {
122 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
123 "brix_storage_backend: invalid s3 origin port");
124 return NGX_ERROR;
125 }
126 out->host_len = (size_t) (colon - authority);
127 out->port = (int) pn;
128 }
129
130 return NGX_OK;
131}

133/* Parse "s3://host[:port]/bucket" into host/port/base(bucket). Returns NGX_OK,
134 * or NGX_ERROR after an [emerg] for a malformed URL. */
135static ngx_int_t
136vfs_parse_s3_url(ngx_conf_t *cf, const ngx_str_t *sb, vfs_origin_parse_t *out)
137{
138 u_char *authority = sb->data + sizeof("s3://") - 1;
139 size_t rem = sb->len - (sizeof("s3://") - 1);
140 size_t i, slash = rem, authority_len, path_len, host_len;
141 const char *path;
142
143 for (i = 0; i < rem; i++) {
144 if (authority[i] == '/') {
145 slash = i;
146 break;
147 }
148 }
149 authority_len = slash;
150 path = (slash < rem) ? (const char *) (authority + slash) : "";
151 path_len = (slash < rem) ? rem - slash : 0;
152
153 if (vfs_parse_s3_port(cf, authority, authority_len, out) != NGX_OK) {
154 return NGX_ERROR;
155 }
156 host_len = out->host_len;
157 while (path_len > 0 && path[0] == '/') {
158 path++;
159 path_len--;
160 }
161 /* Trim an optional "?opt=val" query suffix (e.g. ?put_checksum=1, #12) so the
162 * bucket name stays clean; the caller scans the raw URL for the option. */
163 {
164 size_t q;
165 for (q = 0; q < path_len; q++) {
166 if (path[q] == '?') { path_len = q; break; }
167 }
168 }
169 if (host_len == 0 || host_len >= out->host_cap || path_len == 0
170 || path_len >= out->base_cap)
171 {
172 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
173 "brix_storage_backend: s3:// needs \"//host[:port]/bucket\"");
174 return NGX_ERROR;
175 }
176
177 ngx_memcpy(out->host, authority, host_len);
178 out->host[host_len] = '\0';
179 ngx_memcpy(out->base, path, path_len);
180 out->base[path_len] = '\0';
181 return NGX_OK;
182}

184/* "s3://host[:port]/bucket" → a read-only S3 source backend. Returns NGX_OK on a
185 * handled segment, NGX_ERROR after an [emerg] for a malformed one, or
186 * NGX_DECLINED when the scheme is not ours. */
187ngx_int_t
188vfs_parse_s3_origin(ngx_conf_t *cf, const char *root_canon, const ngx_str_t *sb)
189{
190 char host[256], bucket[256];
191 vfs_origin_parse_t parsed;
192
193 /* "s3://host[:port]/bucket" → a read-only S3 source backend (path-style): the
194 * first path segment is the bucket. Default port 7480 (radosgw), tls off — the
195 * same defaults the generic tier parser uses for an s3 store. */
196 if (sb->len < sizeof("s3://") - 1
197 || ngx_strncmp(sb->data, "s3://", sizeof("s3://") - 1) != 0)
198 {
199 return NGX_DECLINED;
200 }
201
202 ngx_memzero(&parsed, sizeof(parsed));
203 parsed.host = host;
204 parsed.host_cap = sizeof(host);
205 parsed.base = bucket; /* origin_path carries the bucket */
206 parsed.base_cap = sizeof(bucket);
207
208 if (vfs_parse_s3_url(cf, sb, &parsed) != NGX_OK) {
209 return NGX_ERROR;
210 }
211
212 /* #12: opt-in origin-enforced body integrity. "s3://host/bucket?put_checksum=1"
213 * makes every PUT/UploadPart carry a signed x-amz-checksum-crc32 so the origin
214 * validates the bytes and rejects a wire-corrupted upload with 400 BadDigest —
215 * the outbound-leg analogue of the ingest Content-MD5 gate (#7). Off by default
216 * (UNSIGNED-PAYLOAD): unknown-header-rejecting origins stay working untouched. */
217 {
218 brix_vfs_s3_origin_opts_t opts;
219
220 ngx_memzero(&opts, sizeof(opts));
221 opts.put_checksum = (ngx_strstr(sb->data, "put_checksum=1") != NULL);
222 /* "?nearline=1" declares the bucket archive-backed: residency then comes
223 * from x-amz-storage-class / x-amz-restore and recall from
224 * RestoreObject (sd_remote_nearline.c). It is a DECLARATION and never
225 * inferred from a storage class seen at runtime, because arming the cap
226 * commits the export to carrying a cache tier as the recall target
227 * (§9.4) — inferring it would turn a working bucket into a config-time
228 * failure the first time someone tiered one object to GLACIER.
229 * "?restore_days=N" tunes how long the restored copy stays readable. */
230 opts.nearline = (ngx_strstr(sb->data, "nearline=1") != NULL);
231 opts.restore_days = brix_vfs_origin_opt_int(sb->data, "restore_days=",
232 0);
233 brix_vfs_backend_config_s3(root_canon, parsed.host, parsed.port, 0,
234 parsed.base, &opts);
235 }
236 return NGX_OK;
237
238}

240/* Configure a bare local-driver backend (the non-root:// fallback). Misconfig
241 * guard: a URL-ish value no scheme claimed used to be SILENTLY ignored (a
242 * `pblock:<dir>` typo ran the default POSIX backend). The legacy single-colon
243 * posix:/pblock: spellings stay accepted (warned); any other colon form is an
244 * [emerg]. Returns NGX_OK, or NGX_ERROR on an unrecognized scheme. */
245static ngx_int_t
246vfs_config_local_backend(ngx_conf_t *cf, const char *root_canon,
247 const ngx_str_t *sb, size_t block_size)
248{
249 if (memchr(sb->data, ':', sb->len) != NULL) {
250 if ((sb->len > sizeof("posix:") - 1
251 && ngx_strncmp(sb->data, "posix:", sizeof("posix:") - 1) == 0)
252 || (sb->len > sizeof("pblock:") - 1
253 && ngx_strncmp(sb->data, "pblock:", sizeof("pblock:") - 1) == 0))
254 {
255 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
256 "brix_storage_backend \"%V\": the single-colon form is "
257 "legacy — use \"%s\" bare (default root) or "
258 "\"pblock://<dir>[?opts]\"", sb,
259 sb->data[1] == 'o' ? "posix" : "pblock");
260 } else {
261 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
262 "brix_storage_backend \"%V\": unrecognized backend scheme "
263 "(known: posix, pblock, pblock://, mirage:, block:, "
264 "root://, roots://, root+tape://, roots+tape://, "
265 "tape://, frm://, ftp://, gsiftp://, http(s)://, s3://, "
266 "ceph:, rados:, cephfsro:)", sb);
267 return NGX_ERROR;
268 }
269 }
270 brix_vfs_backend_config(root_canon, sb, block_size);
271 return NGX_OK;
272}

274/* The remote root:// spellings of a primary backend URL, longest first. "roots"
275 * selects TLS; the "+tape" pair additionally declares that the origin fronts an
276 * MSS, which arms the driver's nearline pair (residency from kXR_stat's
277 * kXR_offline, recall via kXR_prepare/kXR_stage) instead of letting a first read
278 * block a worker for the length of a tape mount. None of the four is a prefix of
279 * another today, but the scan is ordered so it never starts to matter. */
280static const struct {
281 const char *prefix;
282 size_t len;
283 int tls;
284 int nearline;
285} vfs_xroot_scheme_table[] = {
286 { "roots+tape://", sizeof("roots+tape://") - 1, 1, 1 },
287 { "root+tape://", sizeof("root+tape://") - 1, 0, 1 },
288 { "roots://", sizeof("roots://") - 1, 1, 0 },
289 { "root://", sizeof("root://") - 1, 0, 0 },
290};

292/* "root[s][+tape]://host:port" → a remote root:// primary backend; any other
293 * value is a local driver name (pblock/posix) handled by
294 * brix_vfs_backend_config. Returns NGX_OK, or NGX_ERROR after an [emerg] for a
295 * malformed remote origin. */
296ngx_int_t
297vfs_parse_xroot_or_driver_origin(ngx_conf_t *cf, const char *root_canon,
298 const ngx_str_t *sb, size_t block_size, int family)
299{
300 u_char *addr = NULL;
301 size_t addrn = 0;
302 int is_roots = 0;
303 int is_nearline = 0;
304 size_t s;
305
306 for (s = 0; s < sizeof(vfs_xroot_scheme_table)
307 / sizeof(vfs_xroot_scheme_table[0]); s++)
308 {
309 size_t n = vfs_xroot_scheme_table[s].len;
310
311 if (sb->len > n
312 && ngx_strncmp(sb->data, vfs_xroot_scheme_table[s].prefix, n) == 0)
313 {
314 addr = sb->data + n;
315 addrn = sb->len - n;
316 is_roots = vfs_xroot_scheme_table[s].tls;
317 is_nearline = vfs_xroot_scheme_table[s].nearline;
318 break;
319 }
320 }
321
322 if (addr == NULL) {
323 return vfs_config_local_backend(cf, root_canon, sb, block_size);
324 }
325
326 {
327 u_char *colon = NULL;
328 size_t i, hostn;
329 ngx_int_t portnum;
330 char host[256];
331
332 /* Split host:port on the LAST colon (a bracketed [v6]:port keeps it). */
333 for (i = addrn; i > 0; i--) {
334 if (addr[i - 1] == ':') { colon = addr + i - 1; break; }
335 }
336 if (colon == NULL) {
337 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
338 "brix_storage_backend: remote origin needs host:port");
339 return NGX_ERROR;
340 }
341 hostn = (size_t) (colon - addr);
342 portnum = ngx_atoi(colon + 1, (size_t) (addr + addrn - (colon + 1)));
343 if (hostn == 0 || hostn >= sizeof(host) || portnum == NGX_ERROR
344 || portnum <= 0 || portnum > 65535)
345 {
346 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
347 "brix_storage_backend: invalid remote origin host:port");
348 return NGX_ERROR;
349 }
350 ngx_memcpy(host, addr, hostn);
351 host[hostn] = '\0';
352 {
353 brix_vfs_xroot_origin_t o = {
354 .host = host,
355 .port = (int) portnum,
356 .tls = is_roots,
357 .family = family,
358 .nearline = is_nearline,
359 };
360
361 brix_vfs_backend_config_xroot(root_canon, &o);
362 }
363 }
364
365 return NGX_OK;
366}
