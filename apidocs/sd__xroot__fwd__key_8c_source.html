25/* The scheme is the run of letters/digits ending at ':'; anything else before
26 * the ':' — or no ':' at all inside the first component — means the client
27 * named no origin. */
28static int
29fwd_parse_scheme(const char **pp, int *tls_out)
30{
31 const char *p = *pp;
32 size_t n = 0;
33
34 while (p[n] != '\0' && p[n] != ':' && p[n] != '/') {
35 n++;
36 }
37 if (p[n] != ':' || n == 0) {
38 return ENOENT;
39 }
40 if (n == 4 && strncmp(p, "root", 4) == 0) {
41 *tls_out = 0;
42 } else if (n == 5 && strncmp(p, "roots", 5) == 0) {
43 *tls_out = 1;
44 } else {
45 return ENOTSUP;
46 }
47 *pp = p + n + 1;
48 return 0;
49}

51static int
52fwd_authority_char_ok(char c)
53{
54 if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
55 || (c >= '0' && c <= '9'))
56 {
57 return 1;
58 }
59 return c == '.' || c == '-' || c == ':' || c == '[' || c == ']';
60}

62/* Copy the authority (up to the next '/' or NUL) into `auth`, then split it
63 * with the shared host:port core. Rejects an empty authority, any character
64 * outside the hostname / IPv6-literal alphabet, and an oversized one. */
65static int
66fwd_parse_authority(const char **pp, brix_sd_xroot_fwd_target_t *t)
67{
68 const char *p = *pp;
69 char auth[FWD_AUTH_MAX];
70 size_t n = 0;
71
72 while (p[n] != '\0' && p[n] != '/') {
73 if (!fwd_authority_char_ok(p[n])) {
74 return EINVAL;
75 }
76 n++;
77 }
78 if (n == 0) {
79 return EINVAL;
80 }
81 if (n >= sizeof(auth)) {
82 return ENAMETOOLONG;
83 }
84 memcpy(auth, p, n);
85 auth[n] = '\0';
86 if (brix_split_host_port(auth, t->host, sizeof(t->host), &t->port,
87 BRIX_SD_XROOT_FWD_PORT_DEF) != 0)
88 {
89 return (n >= BRIX_SD_XROOT_FWD_HOST_MAX) ? ENAMETOOLONG : EINVAL;
90 }
91 *pp = p + n;
92 return 0;
93}

95int
96brix_sd_xroot_fwd_parse_key(const char *key, brix_sd_xroot_fwd_target_t *t)
97{
98 const char *p;
99 size_t n;
100 int rc;
101
102 if (key == NULL || t == NULL) {
103 return EINVAL;
104 }
105 memset(t, 0, sizeof(*t));
106 p = fwd_skip_slashes(key);
107 rc = fwd_parse_scheme(&p, &t->tls);
108 if (rc != 0) {
109 return rc;
110 }
111 p = fwd_skip_slashes(p); /* "root:" "//" or "root:" "/" */
112 rc = fwd_parse_authority(&p, t);
113 if (rc != 0) {
114 return rc;
115 }
116 p = fwd_skip_slashes(p); /* "host//file" and "host/file" */
117 n = strlen(p);
118 if (n + 2 > sizeof(t->path)) {
119 return ENAMETOOLONG;
120 }
121 t->path[0] = '/';
122 memcpy(t->path + 1, p, n + 1);
123 return 0;
124}
