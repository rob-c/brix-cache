1/*
2 * identity_matrix_conf.c — directive setters for the TPC identity matrix
3 * (2.0 F18). One setter per directive, shared by EVERY plane: the shared conf
4 * preamble (ngx_http_brix_shared_conf_t) is member 0 of both
5 * ngx_stream_brix_srv_conf_t and ngx_http_brix_common_conf_t, so the cast from
6 * a plane's conf to the preamble is valid — the same arrangement
7 * brix_conf_set_tpc_verify_checksum uses in core/config/policy.c.
8 *
9 * Every rejection here is an [emerg] at `nginx -t`. That is the point: a
10 * confinement control that silently ignores a line it could not parse is a
11 * control that quietly permits everything the operator meant to forbid.
12 */
13#include "tpc/common/identity_matrix.h"

16/* Lazily create one of the three rule arrays. NULL means "no rules" all the
17 * way down (merge NULL-inherits, brix_tpc_matrix_check reads NULL as vacuous),
18 * so the array only exists once a directive has actually been written. */
19static ngx_array_t *
20tpc_matrix_array(ngx_conf_t *cf, ngx_array_t **slot, size_t size)
21{
22 if (*slot == NULL) {
23 *slot = ngx_array_create(cf->pool, 4, size);
24 }
25 return *slot;
26}

29/* Assign one `<key> <pattern>` pair into `rule`, or return the offending key.
30 * A repeated key inside one rule is refused rather than last-wins: `allow vo a
31 * vo b` reads as a disjunction but would silently become `vo b` alone. */
32static const char *
33tpc_allow_assign(brix_tpc_allow_rule_t *rule, ngx_str_t *key, ngx_str_t *val)
34{
35 const char **slot = NULL;
36
37 if (key->len == 2 && ngx_strncmp(key->data, "dn", 2) == 0) {
38 slot = &rule->dn;
39 } else if (key->len == 5 && ngx_strncmp(key->data, "group", 5) == 0) {
40 slot = &rule->group;
41 } else if (key->len == 4 && ngx_strncmp(key->data, "host", 4) == 0) {
42 slot = &rule->host;
43 } else if (key->len == 2 && ngx_strncmp(key->data, "vo", 2) == 0) {
44 slot = &rule->vo;
45 }
46
47 if (slot == NULL) {
48 return "unknown selector";
49 }
50 if (*slot != NULL) {
51 return "duplicate selector";
52 }
53 if (val->len == 0) {
54 return "empty pattern";
55 }
56 *slot = (const char *) val->data;
57 return NULL;
58}

61/*
62 * brix_tpc_allow_identity dn|group|host|vo <pattern> [<key> <pattern>]...
63 *
64 * One rule per directive; the selectors written on the line are ANDed, and
65 * separate directives OR — stock xrootd's `ofs.tpc allow [dn ..] [group ..]
66 * [host ..] [vo ..]`. Configuring ANY rule makes the allow stage fail-closed:
67 * a subject matching no rule is denied, never defaulted.
68 */
69char *
70brix_tpc_conf_allow_identity(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
71{
72 ngx_http_brix_shared_conf_t *sc = conf;
73 ngx_str_t *value = cf->args->elts;
74 brix_tpc_allow_rule_t *rule;
75 ngx_uint_t i;
76 const char *why;
77
78 (void) cmd;
79
80 if ((cf->args->nelts - 1) % 2 != 0) {
81 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
82 "brix_tpc_allow_identity: expects <selector> <pattern> pairs "
83 "(dn, group, host, vo)");
84 return NGX_CONF_ERROR;
85 }
86 if (tpc_matrix_array(cf, &sc->tpc_allow_identity, sizeof(*rule)) == NULL) {
87 return NGX_CONF_ERROR;
88 }
89
90 rule = ngx_array_push(sc->tpc_allow_identity);
91 if (rule == NULL) {
92 return NGX_CONF_ERROR;
93 }
94 ngx_memzero(rule, sizeof(*rule));
95
96 for (i = 1; i < cf->args->nelts; i += 2) {
97 why = tpc_allow_assign(rule, &value[i], &value[i + 1]);
98 if (why != NULL) {
99 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
100 "brix_tpc_allow_identity: %s \"%V\" (known: dn, group, host, "
101 "vo)", why, &value[i]);
102 return NGX_CONF_ERROR;
103 }
104 }
105 return NGX_CONF_OK;
106}

109/* The party a `require` rule names. -1 = unknown. */
110static int
111tpc_require_party(const ngx_str_t *tok)
112{
113 if (tok->len == 3 && ngx_strncmp(tok->data, "all", 3) == 0) {
114 return BRIX_TPC_PARTY_ALL;
115 }
116 if (tok->len == 6 && ngx_strncmp(tok->data, "client", 6) == 0) {
117 return BRIX_TPC_PARTY_CLIENT;
118 }
119 if (tok->len == 4 && ngx_strncmp(tok->data, "dest", 4) == 0) {
120 return BRIX_TPC_PARTY_DEST;
121 }
122 return -1;
123}

126/*
127 * Canonical auth label for an operator token, or NULL when unknown. The
128 * canonical spellings are exactly brix_identity_auth_label()'s, so a rule can
129 * never name a method the identity layer would report differently. XRootD's
130 * own protocol names are accepted as aliases (`ztn` is its name for the WLCG
131 * bearer token, `unix`/`krb5`/`gsi`/`sss`/`pwd`/`host` are already identical)
132 * so an ofs.tpc stanza transplants without translation. "none" is refused: a
133 * rule demanding that a party be UNauthenticated is never what an operator
134 * writing a confinement control meant.
135 */
136static const char *
137tpc_require_auth(const ngx_str_t *tok)
138{
139 static const struct { const char *in; const char *out; } map[] = {
140 { "gsi", "GSI" }, { "token", "TOKEN" }, { "ztn", "TOKEN" },
141 { "sss", "SSS" }, { "s3key", "S3KEY" }, { "krb5", "KRB5" },
142 { "unix", "UNIX" }, { "host", "HOST" }, { "pwd", "PWD" },
143 };
144 ngx_uint_t i;
145
146 for (i = 0; i < sizeof(map) / sizeof(map[0]); i++) {
147 size_t n = ngx_strlen(map[i].in);
148 if (tok->len == n
149 && ngx_strncasecmp(tok->data, (u_char *) map[i].in, n) == 0)
150 {
151 return map[i].out;
152 }
153 }
154 return NULL;
155}

158/*
159 * brix_tpc_require all|client|dest <auth>
160 *
161 * Repeatable. Rules naming the same party OR (any one satisfies it); a party
162 * no rule names is unconstrained even when the other party is constrained.
163 * `client` binds the leg the initiating client opened, `dest` the leg the peer
164 * SERVER opened on its behalf (the one carrying tpc.org) — so `require dest`
165 * can never be satisfied by the client's own credential.
166 */
167char *
168brix_tpc_conf_require(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
169{
170 ngx_http_brix_shared_conf_t *sc = conf;
171 ngx_str_t *value = cf->args->elts;
172 brix_tpc_require_rule_t *rule;
173 int party;
174 const char *auth;
175
176 (void) cmd;
177
178 party = tpc_require_party(&value[1]);
179 if (party < 0) {
180 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
181 "brix_tpc_require: unknown party \"%V\" (known: all, client, dest)",
182 &value[1]);
183 return NGX_CONF_ERROR;
184 }
185
186 auth = tpc_require_auth(&value[2]);
187 if (auth == NULL) {
188 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
189 "brix_tpc_require: unknown authentication method \"%V\" (known: "
190 "gsi, token (ztn), sss, s3key, krb5, unix, host, pwd)", &value[2]);
191 return NGX_CONF_ERROR;
192 }
193
194 if (tpc_matrix_array(cf, &sc->tpc_require, sizeof(*rule)) == NULL) {
195 return NGX_CONF_ERROR;
196 }
197 rule = ngx_array_push(sc->tpc_require);
198 if (rule == NULL) {
199 return NGX_CONF_ERROR;
200 }
201 rule->party = party;
202 rule->auth = auth;
203 return NGX_CONF_OK;
204}

207/*
208 * brix_tpc_restrict <path> [<path>...]
209 *
210 * Repeatable, and every argument on a line is appended — the stock
211 * ngx_conf_set_str_array_slot keeps only the first, which for a confinement
212 * list would silently drop everything after it (the footgun already fixed for
213 * brix_tpc_source_allow). Paths must be absolute; the total is capped so the
214 * per-request pointer vector in brix_tpc_matrix_check stays a stack array.
215 */
216char *
217brix_tpc_conf_restrict(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
218{
219 ngx_http_brix_shared_conf_t *sc = conf;
220 ngx_str_t *value = cf->args->elts;
221 ngx_str_t *slot;
222 ngx_uint_t i;
223
224 (void) cmd;
225
226 if (tpc_matrix_array(cf, &sc->tpc_restrict, sizeof(ngx_str_t)) == NULL) {
227 return NGX_CONF_ERROR;
228 }
229
230 for (i = 1; i < cf->args->nelts; i++) {
231 if (value[i].len == 0 || value[i].data[0] != '/') {
232 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
233 "brix_tpc_restrict: \"%V\" must be an absolute path",
234 &value[i]);
235 return NGX_CONF_ERROR;
236 }
237 if (sc->tpc_restrict->nelts >= BRIX_TPC_RESTRICT_MAX) {
238 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
239 "brix_tpc_restrict: at most %d prefixes",
240 BRIX_TPC_RESTRICT_MAX);
241 return NGX_CONF_ERROR;
242 }
243 slot = ngx_array_push(sc->tpc_restrict);
244 if (slot == NULL) {
245 return NGX_CONF_ERROR;
246 }
247 *slot = value[i];
248 }
249 return NGX_CONF_OK;
250}
