1/*
2 * identity_matrix_conf.c — directive setters for the TPC identity matrix
3 * (2.0 F18). One setter per directive, shared by EVERY plane: the shared conf
4 * preamble (ngx_http_brix_shared_conf_t) is member 0 of both
5 * ngx_stream_brix_srv_conf_t and ngx_http_brix_common_conf_t, so the cast from
6 * a plane's conf to the preamble is valid — the same arrangement
7 * brix_conf_set_tpc_verify_checksum uses in core/config/policy.c.
8 *
9 * Every rejection here is an [emerg] at `nginx -t`. That is the point: a
10 * confinement control that silently ignores a line it could not parse is a
11 * control that quietly permits everything the operator meant to forbid.
12 */

17/* Lazily create one of the three rule arrays. NULL means "no rules" all the
18 * way down (merge NULL-inherits, brix_tpc_matrix_check reads NULL as vacuous),
19 * so the array only exists once a directive has actually been written. */
20static ngx_array_t *
21tpc_matrix_array(ngx_conf_t *cf, ngx_array_t **slot, size_t size)
22{
23 if (*slot == NULL) {
24 *slot = ngx_array_create(cf->pool, 4, size);
25 }
26 return *slot;
27}

30/* Assign one `<key> <pattern>` pair into `rule`, or return the offending key.
31 * A repeated key inside one rule is refused rather than last-wins: `allow vo a
32 * vo b` reads as a disjunction but would silently become `vo b` alone. */
33static const char *
34tpc_allow_assign(brix_tpc_allow_rule_t *rule, ngx_str_t *key, ngx_str_t *val)
35{
36 const char **slot = NULL;
37
38 if (key->len == 2 && ngx_strncmp(key->data, "dn", 2) == 0) {
39 slot = &rule->dn;
40 } else if (key->len == 5 && ngx_strncmp(key->data, "group", 5) == 0) {
41 slot = &rule->group;
42 } else if (key->len == 4 && ngx_strncmp(key->data, "host", 4) == 0) {
43 slot = &rule->host;
44 } else if (key->len == 2 && ngx_strncmp(key->data, "vo", 2) == 0) {
45 slot = &rule->vo;
46 }
47
48 if (slot == NULL) {
49 return "unknown selector";
50 }
51 if (*slot != NULL) {
52 return "duplicate selector";
53 }
54 if (val->len == 0) {
55 return "empty pattern";
56 }
57 *slot = (const char *) val->data;
58 return NULL;
59}

62/*
63 * brix_tpc_allow_identity dn|group|host|vo <pattern> [<key> <pattern>]...
64 *
65 * One rule per directive; the selectors written on the line are ANDed, and
66 * separate directives OR — stock xrootd's `ofs.tpc allow [dn ..] [group ..]
67 * [host ..] [vo ..]`. Configuring ANY rule makes the allow stage fail-closed:
68 * a subject matching no rule is denied, never defaulted.
69 */
70char *
71brix_tpc_conf_allow_identity(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
72{
73 ngx_http_brix_shared_conf_t *sc = conf;
74 ngx_str_t *value = cf->args->elts;
75 brix_tpc_allow_rule_t *rule;
76 ngx_uint_t i;
77 const char *why;
78
79 (void) cmd;
80
81 if ((cf->args->nelts - 1) % 2 != 0) {
82 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
83 "brix_tpc_allow_identity: expects <selector> <pattern> pairs "
84 "(dn, group, host, vo)");
85 return NGX_CONF_ERROR;
86 }
87 if (tpc_matrix_array(cf, &sc->tpc_allow_identity, sizeof(*rule)) == NULL) {
88 return NGX_CONF_ERROR;
89 }
90
91 rule = ngx_array_push(sc->tpc_allow_identity);
92 if (rule == NULL) {
93 return NGX_CONF_ERROR;
94 }
95 ngx_memzero(rule, sizeof(*rule));
96
97 for (i = 1; i < cf->args->nelts; i += 2) {
98 why = tpc_allow_assign(rule, &value[i], &value[i + 1]);
99 if (why != NULL) {
100 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
101 "brix_tpc_allow_identity: %s \"%V\" (known: dn, group, host, "
102 "vo)", why, &value[i]);
103 return NGX_CONF_ERROR;
104 }
105 }
106 return NGX_CONF_OK;
107}

110/* The party a `require` rule names. -1 = unknown. */
111static int
112tpc_require_party(const ngx_str_t *tok)
113{
114 if (tok->len == 3 && ngx_strncmp(tok->data, "all", 3) == 0) {
115 return BRIX_TPC_PARTY_ALL;
116 }
117 if (tok->len == 6 && ngx_strncmp(tok->data, "client", 6) == 0) {
118 return BRIX_TPC_PARTY_CLIENT;
119 }
120 if (tok->len == 4 && ngx_strncmp(tok->data, "dest", 4) == 0) {
121 return BRIX_TPC_PARTY_DEST;
122 }
123 return -1;
124}

127/*
128 * Canonical auth label for an operator token, or NULL when unknown. The
129 * canonical spellings are exactly brix_identity_auth_label()'s, so a rule can
130 * never name a method the identity layer would report differently. XRootD's
131 * own protocol names are accepted as aliases (`ztn` is its name for the WLCG
132 * bearer token, `unix`/`krb5`/`gsi`/`sss`/`pwd`/`host` are already identical)
133 * so an ofs.tpc stanza transplants without translation. "none" is refused: a
134 * rule demanding that a party be UNauthenticated is never what an operator
135 * writing a confinement control meant.
136 */
137static const char *
138tpc_require_auth(const ngx_str_t *tok)
139{
140 static const struct { const char *in; const char *out; } map[] = {
141 { "gsi", "GSI" }, { "token", "TOKEN" }, { "ztn", "TOKEN" },
142 { "sss", "SSS" }, { "s3key", "S3KEY" }, { "krb5", "KRB5" },
143 { "unix", "UNIX" }, { "host", "HOST" }, { "pwd", "PWD" },
144 };
145 ngx_uint_t i;
146
147 for (i = 0; i < sizeof(map) / sizeof(map[0]); i++) {
148 size_t n = ngx_strlen(map[i].in);
149 if (tok->len == n
150 && ngx_strncasecmp(tok->data, (u_char *) map[i].in, n) == 0)
151 {
152 return map[i].out;
153 }
154 }
155 return NULL;
156}

159/*
160 * brix_tpc_require all|client|dest <auth>
161 *
162 * Repeatable. Rules naming the same party OR (any one satisfies it); a party
163 * no rule names is unconstrained even when the other party is constrained.
164 * `client` binds the leg the initiating client opened, `dest` the leg the peer
165 * SERVER opened on its behalf (the one carrying tpc.org) — so `require dest`
166 * can never be satisfied by the client's own credential.
167 */
168char *
169brix_tpc_conf_require(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
170{
171 ngx_http_brix_shared_conf_t *sc = conf;
172 ngx_str_t *value = cf->args->elts;
173 brix_tpc_require_rule_t *rule;
174 int party;
175 const char *auth;
176
177 (void) cmd;
178
179 party = tpc_require_party(&value[1]);
180 if (party < 0) {
181 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
182 "brix_tpc_require: unknown party \"%V\" (known: all, client, dest)",
183 &value[1]);
184 return NGX_CONF_ERROR;
185 }
186
187 auth = tpc_require_auth(&value[2]);
188 if (auth == NULL) {
189 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
190 "brix_tpc_require: unknown authentication method \"%V\" (known: "
191 "gsi, token (ztn), sss, s3key, krb5, unix, host, pwd)", &value[2]);
192 return NGX_CONF_ERROR;
193 }
194
195 if (tpc_matrix_array(cf, &sc->tpc_require, sizeof(*rule)) == NULL) {
196 return NGX_CONF_ERROR;
197 }
198 rule = ngx_array_push(sc->tpc_require);
199 if (rule == NULL) {
200 return NGX_CONF_ERROR;
201 }
202 rule->party = party;
203 rule->auth = auth;
204 return NGX_CONF_OK;
205}

208/*
209 * brix_tpc_restrict <path> [<path>...]
210 *
211 * Repeatable, and every argument on a line is appended — the stock
212 * ngx_conf_set_str_array_slot keeps only the first, which for a confinement
213 * list would silently drop everything after it (the footgun already fixed for
214 * brix_tpc_source_allow). Paths must be absolute; the total is capped so the
215 * per-request pointer vector in brix_tpc_matrix_check stays a stack array.
216 */
217char *
218brix_tpc_conf_restrict(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
219{
220 ngx_http_brix_shared_conf_t *sc = conf;
221 ngx_str_t *value = cf->args->elts;
222 ngx_str_t *slot;
223 ngx_uint_t i;
224
225 (void) cmd;
226
227 if (tpc_matrix_array(cf, &sc->tpc_restrict, sizeof(ngx_str_t)) == NULL) {
228 return NGX_CONF_ERROR;
229 }
230
231 for (i = 1; i < cf->args->nelts; i++) {
232 if (value[i].len == 0 || value[i].data[0] != '/') {
233 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
234 "brix_tpc_restrict: \"%V\" must be an absolute path",
235 &value[i]);
236 return NGX_CONF_ERROR;
237 }
238 if (sc->tpc_restrict->nelts >= BRIX_TPC_RESTRICT_MAX) {
239 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
240 "brix_tpc_restrict: at most %d prefixes",
241 BRIX_TPC_RESTRICT_MAX);
242 return NGX_CONF_ERROR;
243 }
244 slot = ngx_array_push(sc->tpc_restrict);
245 if (slot == NULL) {
246 return NGX_CONF_ERROR;
247 }
248 *slot = value[i];
249 }
250 return NGX_CONF_OK;
251}
