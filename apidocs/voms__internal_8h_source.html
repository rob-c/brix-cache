9/*
10 * ABI-compatible struct definitions for voms-2.1.3 / EL9.
11 *
12 * We dlopen() libvoms at runtime (to avoid a hard dependency) and call its
13 * VOMS_Retrieve() / VOMS_Destroy() functions through function pointers.
14 * These struct layouts match the libvoms ABI; only fields we dereference are
15 * given meaningful types. The rest are opaque placeholders so the byte offsets
16 * stay correct across minor library versions.
17 *
18 * DO NOT change field order or add/remove fields — that would break the dlopen
19 * ABI contract.
20 */
21
22/*
23 * One VOMS attribute item: a group/role/capability triple from one AC.
24 * group: VOMS group name, e.g. "/cms/Production"
25 * role: role name, e.g. "pilot" (may be NULL for group-only membership)
26 * cap: capability string (rarely used; usually NULL)
27 */
28struct voms_data_item {
29 char *group;
30 char *role;
31 char *cap;
32};

34/*
35 * One VOMS attribute certificate (AC) entry: metadata about one VO's AC
36 * embedded in the X.509 proxy.
37 * voname: the VO name, e.g. "cms"
38 * fqan: NULL-terminated array of fully-qualified attribute names, e.g.
39 * ["/cms/Production/Role=pilot/Capability=NULL", NULL]
40 * std: NULL-terminated array of voms_data_item pointers (structured fqan)
41 */
42struct voms_entry {
43 int siglen; /* length of signature bytes */
44 char *signature; /* AC signature (opaque) */
45 char *user; /* subject DN of the proxy holder */
46 char *userca; /* issuer DN of the holder's CA */
47 char *server; /* VOMS server DN */
48 char *serverca; /* VOMS server CA DN */
49 char *voname; /* VO name (primary key for attribute lookups) */
50 char *uri; /* VOMS server URI */
51 char *date1; /* AC validity start (ISO 8601) */
52 char *date2; /* AC validity end */
53 int type; /* AC type (opaque) */
54 struct voms_data_item **std; /* structured attribute list, NULL-terminated */
55 char *custom; /* custom attribute data (opaque) */
56 int datalen; /* byte count of custom data */
57 int version; /* AC version number */
58 char **fqan; /* FQANs as strings, NULL-terminated */
59 char *serial; /* AC serial number */
60 void *ac; /* raw OpenSSL AC pointer (opaque) */
61 void *holder; /* AC holder certificate pointer (opaque) */
62};

64/*
65 * Top-level VOMS data container returned by VOMS_Init().
66 * data: NULL-terminated array of voms_entry pointers (one per VO AC found)
67 * volen: number of entries in data[]
68 * real: internal VOMS library state (opaque; do not access)
69 */
70struct voms_data {

76 int volen; /* number of VO AC entries */

79};
