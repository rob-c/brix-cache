119/* ---- resilient.c — network resilience for the synchronous tools ----
120 *
121 * Brings xrootdfs-style recovery (reconnect + full re-auth + handle reopen +
122 * offset resume + bounded backoff) to one-shot CLI flows, lifted from the proven
123 * xrdcp pump (copy.c) and the async mfile layer (aio_mgr.c). Two seams:
124 * - brix_with_resilience(): wrap any stateless op (stat/ls/query/...) so it is
125 * re-issued after a sever, gated by an idempotency class.
126 * - brix_rfile: a synchronous file handle that reopens + resumes mid-transfer.
127 * Both are no-ops (single attempt) when the window is 0, so --no-retry restores
128 * the exact legacy fail-fast path. Raw ops (and copy.c) are untouched. */
129
130/* Idempotency class for brix_with_resilience — governs re-issue after a sever. */
131typedef enum {
132 XRDC_OP_READONLY, /* stat/ls/locate/query/statvfs: retry freely */
133 XRDC_OP_IDEMPOTENT, /* chmod: re-apply is harmless — retry freely */
134 XRDC_OP_MUTATION_NORMALIZE, /* mkdir/rm/rmdir/mv/prepare: re-issue ONCE, then
135 * treat benign_errno (EEXIST/ENOENT) as success */
136 XRDC_OP_UNSAFE /* never auto-retry */
137} brix_op_class;
