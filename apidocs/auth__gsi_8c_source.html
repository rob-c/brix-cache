1/* File: auth_gsi.c — outbound GSI (X.509 proxy) authentication for the transparent upstream
2 * WHAT: the two client-side rounds of an XrdSecgsi handshake the upstream connector
3 * performs when the upstream's kXR_login advert lists `gsi` and
4 * brix_upstream_x509_proxy is configured (phase 115 W2.4).
5 * round 1 brix_upstream_send_gsi_certreq — parse the advert parms, mint the
6 * signed-DH kXGC_certreq via the shared gsi_core wrapper, send it as
7 * kXR_auth "gsi" and mark up->gsi_round = 1.
8 * round 2 brix_upstream_gsi_respond — the reply (kXR_authmore + kXGS_cert) is in
9 * up->resp_body: verify the server's kXRS_x509 leaf against
10 * conf->gsi_store, load the proxy PEM + key, build the kXGC_cert answer
11 * with brix_gsi_build_cert_response, send it, mark up->gsi_round = 2.
12 * The final kXR_ok/kXR_error is judged by bootstrap.c (brix_upstream_bs_auth).
13 *
14 * WHY: before this file the connector only knew ztn, and only when the upstream
15 * answered kXR_login with kXR_authmore (a stub convention). A real brix or stock
16 * XRootD server answers kXR_ok + session id + "&P=gsi,...&P=ztn,..." and then
17 * refuses the unauthenticated relayed request — so a GSI-only redirector could
18 * not sit behind a transparent proxy at all. Sharing the kernel (gsi_core), the
19 * credential loaders (auth/gsi/cred_load) and the peer-leaf verifier
20 * (auth/crypto/gsi_verify) with the cache origin and the TPC destination keeps
21 * the three server-side GSI clients byte-identical on the wire.
22 *
23 * HOW: every step is a small early-return function; heap results (certreq, PEM,
24 * key, payload) are freed on every exit; failures log the specific reason at ERR
25 * on the upstream connection log and return NGX_ERROR so the caller aborts with
26 * a client-visible kXR_ServerError. Verification policy: no gsi_store (no
27 * brix_trusted_ca) → warn once per handshake and proceed (the operator opted out,
28 * as on the cache origin); missing bucket or verdict != 1 → refuse.
29 */

60/* upstream_gsi_verify_server — MITM checkpoint before agreeing a session secret.
61 * Returns NGX_OK when the leaf verifies (or no trust store is configured),
62 * NGX_ERROR (reason logged) when the server offered no certificate or it fails. */
63static ngx_int_t
64upstream_gsi_verify_server(brix_upstream_t *up,
65 ngx_stream_brix_srv_conf_t *conf)
66{
67 const uint8_t *pem = NULL;
68 size_t pem_len = 0;
69
70 if (conf->gsi_store == NULL) {
71 ngx_log_error(NGX_LOG_WARN, up->conn->log, 0,
72 "brix: upstream gsi: no brix_trusted_ca configured; "
73 "upstream server certificate not verified");
74 return NGX_OK;
75 }
76 if (brix_gsi_find_bucket(up->resp_body, up->resp_dlen, (uint32_t) kXRS_x509,
77 &pem, &pem_len) != 0
78 || pem_len == 0)
79 {
80 ngx_log_error(NGX_LOG_ERR, up->conn->log, 0,
81 "brix: upstream gsi: server presented no certificate "
82 "to verify");
83 return NGX_ERROR;
84 }
85 /* Unparseable (-1) and rejected (0) both fail closed: the upstream is a
86 * credentialed peer, so "could not evaluate" is not "verified". */
87 if (brix_gsi_verify_peer_leaf(conf->gsi_store, pem, pem_len) != 1) {
88 ngx_log_error(NGX_LOG_ERR, up->conn->log, 0,
89 "brix: upstream gsi: server certificate verification "
90 "failed");
91 return NGX_ERROR;
92 }
93 return NGX_OK;
94}

96/* upstream_gsi_load_cred — proxy chain from brix_upstream_x509_proxy, key from
97 * brix_upstream_x509_key when set, else from the proxy PEM itself. On success
98 * the caller owns *pem (free) and *key (brix_evp_pkey_free); on failure both are
99 * NULL and the reason is logged. */
100static ngx_int_t
101upstream_gsi_load_cred(brix_upstream_t *up, ngx_stream_brix_srv_conf_t *conf,
102 uint8_t **pem, size_t *pem_len, EVP_PKEY **key)
103{
104 const char *proxy_path = (const char *) conf->upstream_x509_proxy.data;
105 const char *key_path = (conf->upstream_x509_key.len > 0)
106 ? (const char *) conf->upstream_x509_key.data
107 : proxy_path;
108
109 *pem = brix_gsi_cred_load_pem(proxy_path, pem_len);
110 *key = brix_gsi_cred_load_key(key_path);
111 if (*pem != NULL && *key != NULL) {
112 return NGX_OK;
113 }
114 ngx_log_error(NGX_LOG_ERR, up->conn->log, 0,
115 "brix: upstream gsi: cannot load credential "
116 "brix_upstream_x509_proxy \"%V\"%s",
117 &conf->upstream_x509_proxy,
118 (*pem != NULL) ? " (private key)" : "");
119 free(*pem);
120 brix_evp_pkey_free(*key);
121 *pem = NULL;
122 *key = NULL;
123 return NGX_ERROR;
124}

126ngx_int_t
127brix_upstream_gsi_respond(brix_upstream_t *up, ngx_stream_brix_srv_conf_t *conf)
128{
129 uint8_t *pem = NULL;
130 size_t pem_len = 0;
131 EVP_PKEY *key = NULL;
132 uint8_t *payload = NULL;
133 uint32_t plen = 0;
134 char err[128];
135 ngx_int_t rc;
136
137 if (upstream_gsi_verify_server(up, conf) != NGX_OK
138 || upstream_gsi_load_cred(up, conf, &pem, &pem_len, &key) != NGX_OK)
139 {
140 return NGX_ERROR;
141 }
142
143 err[0] = '\0';
144 if (brix_gsi_build_cert_response(up->resp_body, up->resp_dlen, pem, pem_len,
145 key, &payload, &plen, err, sizeof(err))
146 != 0)
147 {
148 ngx_log_error(NGX_LOG_ERR, up->conn->log, 0,
149 "brix: upstream gsi: cert response build failed: %s", err);
150 rc = NGX_ERROR;
151 } else {
152 up->gsi_round = 2;
153 rc = brix_upstream_send_auth_frame(up, "gsi", payload, plen);
154 }
155
156 free(payload);
157 free(pem);
158 brix_evp_pkey_free(key);
159 return rc;
160}
