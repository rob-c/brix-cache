1/*
2 * sd_http_mutate.c — namespace mutations for the HTTP-origin storage driver:
3 * DELETE / MKCOL / MOVE against the WebDAV origin.
4 *
5 * WHAT: The namespace-facing vtable slots — unlink/unlink_cred (DELETE behind
6 * the POSIX delete gate), mkdir/mkdir_cred (MKCOL), rename/rename_cred
7 * (MOVE with Overwrite semantics) — plus the shared request path
8 * (sd_http_ns_send), the WebDAV-status→errno map, and the delete gate
9 * (type + emptiness probes) that keeps a recursive WebDAV DELETE from
10 * ever being issued where POSIX would refuse ENOTEMPTY.
11 *
12 * WHY: Split out of sd_http_write.c (600-line ratchet): the staged-upload
13 * path (buffer + one PUT) and the namespace mutations are two concepts;
14 * this file owns the latter. Writes never fail over — a mutation on a
15 * non-primary origin would split-brain the store — so every request here
16 * targets endpoint 0 via sd_http_write_path, exactly like the commit PUT.
17 *
18 * HOW: A `cred` presents the requesting user's bearer (Authorization header)
19 * or x509 proxy (mutual-TLS client cert) to the origin; cred==NULL falls
20 * back to the instance static header / anonymous. The same cred_gate the
21 * read/write legs use refuses a proxy-only cred the transport cannot
22 * present in deny mode.
23 */

30/* sd_http_ns_send — issue a body-less namespace request (DELETE/MKCOL/MOVE) on
31 * endpoint 0. When `cert_pem` is set AND the transport can present a client cert
32 * (request_cred), the request goes over a mutual-TLS leg carrying the per-user
33 * x509 proxy; otherwise the plain request (whose `hdrs` already carries any
34 * Authorization/Destination lines). Shared by the plain and credential-scoped
35 * namespace slots so the transport selection lives in exactly one place, mirroring
36 * sd_http_staged_commit's PUT selection. Returns the transport rc (0 = wire OK). */
37static int
38sd_http_ns_send(sd_http_inst_state *is, const char *method, const char *path,
39 const char *hdrs, const char *cert_pem, brix_s3_resp_t *resp,
40 char *errbuf, size_t errcap)
41{
42 if (cert_pem != NULL && cert_pem[0] != '\0'
43 && is->transport->request_cred != NULL)
44 {
45 return is->transport->request_cred(is->tctx, is->eps[0].host,
46 is->eps[0].port, is->eps[0].tls, method, path, hdrs,
47 NULL, 0, is->timeout_ms, cert_pem, resp, errbuf, errcap);
48 }
49 return is->transport->request(is->tctx, is->eps[0].host, is->eps[0].port,
50 is->eps[0].tls, method, path, hdrs, NULL, 0, is->timeout_ms,
51 resp, errbuf, errcap);
52}

54/* sd_http_status_to_errno — map a WebDAV mutation status to a POSIX errno for the
55 * delete/mkdir/rename slots. 401/403 → EACCES, 404/409 → ENOENT (target or its parent
56 * absent), 405 → EEXIST (method not allowed on an existing collection), 412 →
57 * EEXIST (Overwrite:F precondition — dst already present), anything else → EIO.
58 * The caller decides which codes count as success before calling this. */
59static int
60sd_http_status_to_errno(long status)
61{
62 switch (status) {
63 case 401:
64 case 403: return EACCES;
65 case 404:
66 case 409: return ENOENT;
67 case 405:
68 case 412: return EEXIST;
69 default: return EIO;
70 }
71}

73/* sd_http_coll_empty — 1 iff the collection at `path` has no children, 0 if it
74 * has at least one, -1 (errno set) if the question could not be answered.
75 *
76 * A WebDAV DELETE of a collection is RECURSIVE (RFC 4918 §9.6), while the VFS
77 * only ever calls this slot non-recursively (a recursive delete is walked by
78 * brix_vfs_driver_rmtree). Without an emptiness gate the two disagree in the
79 * most expensive possible direction: `xrdfs rmdir` of a populated collection —
80 * an operation POSIX refuses with ENOTEMPTY — would erase the entire subtree on
81 * any origin that implements DELETE to spec. Asking first makes the answer the
82 * origin's dialect-independent one. */
83static int
84sd_http_coll_empty(brix_sd_instance_t *inst, const char *path)
85{
86 brix_sd_dir_t *d;
87 brix_sd_dirent_t ent;
88 int err = EIO, rc;
89
90 d = sd_http_opendir(inst, path, &err);
91 if (d == NULL) {
92 errno = err ? err : EIO;
93 return -1;
94 }
95 rc = sd_http_readdir(d, &ent);
96 sd_http_closedir(d);
97 return (rc == NGX_DONE) ? 1 : 0;
98}

100/* sd_http_delete_gate — decide whether `path` may be deleted by this slot, given
101 * the caller's `is_dir` (VFS require_empty_dir) request. 0 = proceed, -1 with
102 * errno set = refuse without issuing anything.
103 *
104 * The rules are the POSIX ones the default backend applies, expressed over
105 * WebDAV's single DELETE method:
106 * absent → ENOENT (a delete that removed nothing is not success)
107 * rmdir of a non-dir → ENOTDIR
108 * any non-empty coll → ENOTEMPTY (rm and rmdir alike; see sd_http_coll_empty)
109 * empty collection → allowed for BOTH spellings, matching brix_ns_delete,
110 * which removes an empty directory on a plain rm
111 * no WebDAV at the origin (PROPFIND 405/501) → a file delete proceeds (such an
112 * origin has no collections); an rmdir is ENOTSUP
113 * rather than a guess. */
114static int
115sd_http_delete_gate(brix_sd_instance_t *inst, sd_http_inst_state *is,
116 const char *path, int is_dir, const char *auth_hdr, const char *cert_pem)
117{
118 int is_coll = 0, perr = 0, empty;
119
120 if (sd_http_probe_type(is, path, auth_hdr, cert_pem,
121 &is_coll, &perr) != 0) {
122 if (perr != ENOTSUP) { errno = perr; return -1; }
123 if (is_dir) { errno = ENOTSUP; return -1; }
124 return 0;
125 }
126 if (is_dir && !is_coll) {
127 errno = ENOTDIR;
128 return -1;
129 }
130 if (!is_coll) {
131 return 0;
132 }
133 empty = sd_http_coll_empty(inst, path);
134 if (empty < 0) {
135 return -1; /* errno set by the probe */
136 }
137 if (!empty) {
138 errno = ENOTEMPTY;
139 return -1;
140 }
141 return 0;
142}

144/* sd_http_unlink_common — shared DELETE path for the plain and credential-scoped
145 * unlink slots. The type/emptiness rules live in sd_http_delete_gate; nothing
146 * reaches the origin until they pass, so a mis-typed or over-broad delete can
147 * never be issued at all. Three behaviours this replaced were each a way of
148 * destroying or mis-reporting data: `is_dir` was discarded (an rmdir of a
149 * regular FILE deleted it), a 404 was reported as success (a delete of a
150 * missing object looked like a delete of a real one), and a non-empty
151 * collection was handed to DELETE (a recursive wipe where POSIX refuses
152 * ENOTEMPTY — this origin answers 409, which the shared status map read as
153 * ENOENT and the root layer then treated as idempotent rmdir SUCCESS).
154 *
155 * A `cred` presents the requesting user's bearer (Authorization header) or x509
156 * proxy (mutual-TLS client cert) to the origin — the gate probes and the DELETE
157 * is issued under the same identity; cred==NULL falls back to the instance
158 * static header / anonymous, exactly as before. The same cred_gate the
159 * read/write legs use refuses a proxy-only cred the transport cannot present in
160 * deny mode. */
161static ngx_int_t
162sd_http_unlink_common(brix_sd_instance_t *inst, const char *path, int is_dir,
163 const brix_sd_cred_t *cred)
164{
165 sd_http_inst_state *is = inst->state;
166 brix_s3_resp_t resp;
167 char errbuf[256], full[SD_HTTP_PATH_MAX];
168 char open_auth[SD_HTTP_AUTH_MAX];
169 const char *open_cert, *auth_hdr;
170
171 if (sd_http_cred_gate(is, cred) != 0) {
172 return NGX_ERROR; /* errno = EACCES (set by gate) */
173 }
174 open_cert = sd_http_resolve_open_cred(is, cred, open_auth, sizeof(open_auth));
175 auth_hdr = open_auth[0] ? open_auth
176 : (is->auth_hdr[0] ? is->auth_hdr : NULL);
177
178 if (sd_http_delete_gate(inst, is, path, is_dir, auth_hdr, open_cert) != 0) {
179 return NGX_ERROR; /* errno set by the gate */
180 }
181
182 sd_http_write_path(is, path, full, sizeof(full));
183
184 if (sd_http_ns_send(is, "DELETE", full, auth_hdr, open_cert, &resp,
185 errbuf, sizeof(errbuf)) != 0)
186 {
187 errno = EIO;
188 return NGX_ERROR;
189 }
190 if (resp.status != 204 && resp.status != 200) {
191 int status = (int) resp.status;
192
193 is->transport->resp_free(&resp);
194 /* 404 here means the entry the gate saw was removed underneath us — a
195 * concurrent delete, which is still "it is not there and I did not
196 * remove it": ENOENT, same as unlink(2) losing that race. 409 on a
197 * DELETE is the collection-not-empty conflict (the gate's race window:
198 * a child appeared after the emptiness probe), NOT the missing-parent
199 * ENOENT the shared status map assumes for MKCOL/MOVE. */
200 errno = (status == 404) ? ENOENT
201 : (status == 409) ? ENOTEMPTY
202 : sd_http_status_to_errno(status);
203 return NGX_ERROR;
204 }
205 is->transport->resp_free(&resp);
206 return NGX_OK;
207}

223/* sd_http_mkdir — create a collection at `path` via WebDAV MKCOL (RFC 4918 §9.3).
224 * `mode` is ignored: a WebDAV collection has no POSIX mode, and the VFS treats a
225 * best-effort chmod as a no-op success (sd.h setattr contract). Writes never fail
226 * over (a mutation on a non-primary origin would split-brain the store), so this
227 * targets endpoint 0 exactly like unlink/commit. 201 Created is success; 405 means
228 * the collection already exists (→ EEXIST); 409 means the parent is missing. */
229/* sd_http_mkdir_common — shared MKCOL path for the plain and credential-scoped
230 * mkdir slots (see sd_http_mkdir for the WebDAV semantics). A `cred` presents the
231 * requesting user's bearer / x509 proxy to the origin; cred==NULL falls back to the
232 * instance static header, exactly as before. */
233static ngx_int_t
234sd_http_mkdir_common(brix_sd_instance_t *inst, const char *path, mode_t mode,
235 const brix_sd_cred_t *cred)
236{
237 sd_http_inst_state *is = inst->state;
238 brix_s3_resp_t resp;
239 char errbuf[256], full[SD_HTTP_PATH_MAX];
240 char open_auth[SD_HTTP_AUTH_MAX];
241 const char *open_cert, *auth_hdr;
242
243 (void) mode;
244 if (sd_http_cred_gate(is, cred) != 0) {
245 return NGX_ERROR; /* errno = EACCES (set by gate) */
246 }
247 open_cert = sd_http_resolve_open_cred(is, cred, open_auth, sizeof(open_auth));
248 auth_hdr = open_auth[0] ? open_auth
249 : (is->auth_hdr[0] ? is->auth_hdr : NULL);
250
251 sd_http_write_path(is, path, full, sizeof(full));
252 if (sd_http_ns_send(is, "MKCOL", full, auth_hdr, open_cert, &resp,
253 errbuf, sizeof(errbuf)) != 0)
254 {
255 errno = EIO;
256 return NGX_ERROR;
257 }
258 if (resp.status != 201 && resp.status != 200) {
259 errno = sd_http_status_to_errno(resp.status);
260 is->transport->resp_free(&resp);
261 return NGX_ERROR;
262 }
263 is->transport->resp_free(&resp);

281/* sd_http_rename — rename/move `src` to `dst` via WebDAV MOVE (RFC 4918 §9.9).
282 * The Destination header must be a full absolute URI on this origin, so it is
283 * composed from endpoint 0's scheme/host/port and the write-path of `dst`.
284 * `noreplace` sends Overwrite: F so an existing destination fails 412 (→ EEXIST)
285 * rather than being clobbered; otherwise Overwrite: T replaces it. 201 (created)
286 * and 204 (replaced) are success. Any per-instance auth header is preserved
287 * alongside the MOVE-specific headers. Endpoint 0 only (writes never fail over). */
288/* sd_http_rename_common — shared MOVE path for the plain and credential-scoped
289 * rename slots (see sd_http_rename for the WebDAV semantics). A `cred` presents the
290 * requesting user's bearer (folded into the MOVE header block, so the origin
291 * authorizes BOTH the source and the Destination leg as the user) or x509 proxy
292 * (mutual-TLS client cert); cred==NULL falls back to the instance static header.
293 * The header block is sized for a full Destination URI PLUS a forwarded bearer
294 * (SD_HTTP_AUTH_MAX) so a large JWT never truncates to a spurious ENAMETOOLONG. */
295static ngx_int_t
296sd_http_rename_common(brix_sd_instance_t *inst, const char *src, const char *dst,
297 int noreplace, const brix_sd_cred_t *cred)
298{
299 sd_http_inst_state *is = inst->state;
300 brix_s3_resp_t resp;
301 char errbuf[256], srcfull[SD_HTTP_PATH_MAX];
302 char dstfull[SD_HTTP_PATH_MAX];
303 char hdrs[SD_HTTP_PATH_MAX + SD_HTTP_AUTH_MAX + 128];
304 char open_auth[SD_HTTP_AUTH_MAX];
305 const char *open_cert, *eff_auth;
306 int n;
307
308 if (sd_http_cred_gate(is, cred) != 0) {
309 return NGX_ERROR; /* errno = EACCES (set by gate) */
310 }
311 open_cert = sd_http_resolve_open_cred(is, cred, open_auth, sizeof(open_auth));
312 /* The per-user bearer header (open_auth) wins over the instance static; "" =
313 * anonymous/static. Folded into the MOVE header block below. */
314 eff_auth = open_auth[0] ? open_auth
315 : (is->auth_hdr[0] ? is->auth_hdr : "");
316
317 sd_http_write_path(is, src, srcfull, sizeof(srcfull));
318 sd_http_write_path(is, dst, dstfull, sizeof(dstfull));
319
320 /* Destination is an absolute URI on this origin; append Overwrite and the
321 * resolved auth header. A default HTTP/HTTPS port is emitted explicitly — it is
322 * always valid in an authority and keeps the composition branch-free. */
323 n = snprintf(hdrs, sizeof(hdrs),
324 "Destination: %s://%s:%d%s\r\nOverwrite: %c\r\n%s",
325 is->eps[0].tls ? "https" : "http",
326 is->eps[0].host, is->eps[0].port, dstfull,
327 noreplace ? 'F' : 'T', eff_auth);
328 if (n <= 0 || (size_t) n >= sizeof(hdrs)) {
329 errno = ENAMETOOLONG;
330 return NGX_ERROR;
331 }
332
333 if (sd_http_ns_send(is, "MOVE", srcfull, hdrs, open_cert, &resp,
334 errbuf, sizeof(errbuf)) != 0)
335 {
336 errno = EIO;
337 return NGX_ERROR;
338 }
339 if (resp.status != 201 && resp.status != 204) {
340 errno = sd_http_status_to_errno(resp.status);
341 is->transport->resp_free(&resp);
342 return NGX_ERROR;
343 }
344 is->transport->resp_free(&resp);
345 return NGX_OK;
346}
