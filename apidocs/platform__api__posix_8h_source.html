1/*
2 * src/platform/platform_api_posix.h - the POSIX surface the PAL guarantees on every host
3 *
4 * WHAT: The Linux/glibc spellings the tree is written against (byte-order
5 * names, Linux-shaped xattr calls, O_PATH, SOCK_CLOEXEC, RESOLVE_*,
6 * PR_*, CAP_*, ...) plus brix_plat_* entry points for the operations
7 * whose Darwin form is not a one-line alias (openat2, renameat2, wake
8 * descriptors, peer credentials, birth time, block-device size, ...).
9 * Reached ONLY through platform/platform_api.h.
10 * WHY: Phase 119 left ~80 `#if defined(__APPLE__)` sites, each a private port
11 * of one call that no PAL fixture exercised. tools/ci/check_platform_leak.py
12 * now rejects any OS-detection macro or OS-private header outside the
13 * PAL owners (src/platform/, client/lib/platform/, shared/cvmfs/platform/),
14 * so every such branch lives here or in a host adapter beneath it.
15 * HOW: This header is host-free: the names are guaranteed by the host's
16 * host_posix.h (linux/: the real glibc headers; darwin/: adapters and
17 * constants, e.g. Linux-shaped xattr calls with ENOATTR -> ENODATA;
18 * windows/: ssize_t and the openat2 ABI), pulled by one computed
19 * #include below. Where the semantics differ (not just the spelling)
20 * the tree calls a brix_plat_* function whose body lives under
21 * src/platform/<host>/; the Linux body is the plain syscall.
22 */
23

48/**
49 * openat2(2): open `rel` beneath rootfd under `resolve` (RESOLVE_* flags).
50 * Linux issues the syscall (ENOSYS on a pre-5.6 kernel); Darwin walks the
51 * path component by component on an fd stack and gives the same verdicts
52 * (EXDEV / ELOOP). O_CLOEXEC is always added; `mode` applies with O_CREAT.
53 * @return fd, or -1 with errno
54 */

163/**
164 * A new O_RDONLY|O_CLOEXEC descriptor on the same file as `fd` (which may be
165 * write-only and, on Linux, unlinked), positioned at offset 0.
166 * Linux re-opens /proc/self/fd/<fd>; Darwin has no such alias for a write-only
167 * descriptor (/dev/fd refuses a mode the original lacks), so it re-opens the
168 * F_GETPATH name and checks the inode still matches, falling back to dup(2)
169 * when the original is itself readable. Fails EACCES when neither works.
170 * @return the descriptor, or -1 with errno
171 */
172int brix_plat_fd_reopen_readonly(int fd);

174/** posix_spawn file action closing every descriptor >= lowfd in the child. */
175/* Linux queues one closefrom file action; Darwin instead sets
176 * POSIX_SPAWN_CLOEXEC_DEFAULT on `attr` (every descriptor not dup2()'d by a
177 * file action is closed in the child), because a queued close of one of the
178 * process's guarded descriptors makes posix_spawn fail with EBADF there.
179 * `attr` must be initialised and passed to posix_spawn(). */
180int brix_plat_spawn_closefrom(posix_spawn_file_actions_t *fa,

226/**
