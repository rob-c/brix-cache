18/* WHAT: Return whether an error body contains high-risk disclosure markers.
19 * WHY: Filesystem paths and credentials must never be reflected to a client.
20 * HOW: Scan the bounded decoded body and the discovered bearer value. */
21static int
22error_body_safe(const brix_http_resp *response, const char *token)
23{
24 static const char *const markers[] = {
25 "/etc/", "/var/", "/home/", "/Users/", "BEGIN PRIVATE KEY",
26 "Authorization:", "Bearer ", "AWS_SECRET_ACCESS_KEY"
27 };
28 size_t index;
29
30 if (response->body == NULL) return 1;
31 for (index = 0; index < sizeof(markers) / sizeof(markers[0]); index++)
32 if (strcasestr(response->body, markers[index]) != NULL) return 0;
33 return token == NULL || token[0] == '\0'
34 || strstr(response->body, token) == NULL;
35}

147/* WHAT: Compare absolute-form GET bytes to the origin-form baseline.
148 * WHY: Proxies and direct clients may legally use either request-target form.

150static void
151probe_absolute_form(xrd_conform_execution *execution, const brix_weburl *url,

217/* WHAT: Send request lines exactly at and one byte above the configured cap.
218 * WHY: HTE-005 cannot infer a deployment-specific parser limit honestly.
219 * HOW: Frame bounded raw GETs and require an explicit 414 for both lengths. */
220static void
221probe_request_line_limit(xrd_conform_execution *execution,
222 const brix_weburl *url, const char *token)
223{
224 unsigned limit = execution->run->http_request_line_limit;
225 unsigned delta;
226 int pass = 1;
227
228 if (!xrd_conform_row_active(execution, "HTE-005")) return;
229 if (limit == 0) {
230 xrd_conform_result_set(execution, "HTE-005",
231 BRIX_CONFORM_VERDICT_SKIPPED,
232 "fixture-required:request-line-limit-boundary", NULL, NULL);
233 return;
234 }
235 for (delta = 0; delta < 2; delta++) {
236 size_t target_length = (size_t) limit + delta - 15U;
237 char *target = malloc(target_length + 1U);
238 xrd_conform_raw_http request;
239 brix_http_resp response;
240 brix_status status;
241 int rc;
242
243 if (target == NULL) {
244 pass = 0;
245 continue;
246 }
247 target[0] = '/';
248 memset(target + 1, 'a', target_length - 1U);
249 target[target_length] = '\0';
250 memset(&request, 0, sizeof(request));
251 request.method = "GET";
252 request.target = target;
253 request.expected_bytes = 4096;
254 memset(&response, 0, sizeof(response));
255 brix_status_clear(&status);
256 rc = xrd_conform_http_raw_request(execution, url, token, &request,
257 &response, &status);
258 pass = pass && rc == 0 && response.status == 414;
259 if (rc == 0) brix_http_resp_free(&response);
260 free(target);
261 }
262 xrd_conform_grade_received(execution, "HTE-005", pass, "request-line-boundary-rejected",
263 "request-line-boundary-not-414", "HTTP 414 at limit and limit+1");
264}
