277static int
278consume_key_once(brix_conn *source, const xrd_conform_scratch_object *object,
279 const char *opaque, uint8_t expected, brix_status *status)
280{
281 brix_file consumption;
282 brix_status close_status;
283 uint8_t observed = 0;
284 ssize_t got;
285

294static void
295probe_key_semantics(xrd_conform_execution *execution, brix_conn *source,
296 const brix_url *source_url, const xrd_conform_scratch_object *object,
297 const uint8_t payload[TPC_PAYLOAD_SIZE])
298{
299 char key[64];
300 char register_opaque[512];
301 char consume_opaque[512];
302 brix_file registration;
303 brix_file replay;
304 brix_status status;
305 int registered;
306 int consumed = 0;
307 int replay_refused = 0;
308
309 if (!xrd_conform_row_active(execution, "TPC-002")
310 && !xrd_conform_row_active(execution, "TPC-007")) return;
311 snprintf(key, sizeof(key), "phase118-%.32s", execution->run->key_hex);
312 snprintf(register_opaque, sizeof(register_opaque),
313 "tpc.key=%s&tpc.dst=%s&tpc.stage=copy"
314 "&tpc.phase118_unknown=must-be-ignored",
315 key, source_url->host);
316 snprintf(consume_opaque, sizeof(consume_opaque),
317 "tpc.key=%s&tpc.org=phase118&tpc.stage=copy"
318 "&tpc.phase118_unknown=must-be-ignored", key);
319 registered = open_read_opaque(source, object->path, register_opaque,
320 &registration, &status) == 0;
321 if (registered) {
322 brix_status close_status;
323
324 brix_status_clear(&close_status);
325 brix_file_close(source, &registration, &close_status);
326 consumed = consume_key_once(source, object, consume_opaque,
327 payload[0], &status);
328 }
329 if (consumed) {
330 brix_status close_status;
331
332 if (open_read_opaque(source, object->path, consume_opaque, &replay,
333 &status) == 0) {
334 brix_status_clear(&close_status);
335 brix_file_close(source, &replay, &close_status);
336 } else {
337 replay_refused = status.kxr == kXR_NotAuthorized;
338 }
339 }
340 grade_key_replay(execution, registered, consumed, replay_refused, &status);
341 grade_key_unknown_field(execution, registered, consumed, &status);

369static int
370setup_tpc_pair(xrd_conform_execution *execution, tpc_probe_state *state)
371{
372 memset(state, 0, sizeof(*state));
373 state->opts = execution->run->opts;
374 state->opts.capture = NULL;
375 state->copy_elapsed = (time_t) execution->run->config.wall_seconds;
376 brix_status_clear(&state->status);
377 if (parse_pair(execution, &state->source_url, &state->destination_url,
378 &state->status) != 0) return -1;
379 if (brix_conform_scratch_path(execution->run->config.scratch,
380 "brix-conform-tpc-source", state->source_object.path,
381 sizeof(state->source_object.path)) != 0
382 || brix_conform_scratch_path(execution->run->config.scratch,
383 "brix-conform-tpc-destination", state->destination_object.path,
384 sizeof(state->destination_object.path)) != 0
385 || xrd_conform_tpc_root_url(&state->source_url,
386 state->source_object.path, state->source_text,
387 sizeof(state->source_text)) != 0
388 || xrd_conform_tpc_root_url(&state->destination_url,
389 state->destination_object.path, state->destination_text,
390 sizeof(state->destination_text)) != 0
391 || reserve_probe(execution, &state->status) != 0) return -2;
392 return 0;

424static void
425perform_tpc_copy(xrd_conform_execution *execution, tpc_probe_state *state)
426{
427 brix_copy_opts copy = { 0 };
428 int copy_needed;
429 time_t started;
430
431 state->claims_ok = xrd_conform_tpc_query(&state->source, &state->source_claim,
432 &state->status) == 0
433 && xrd_conform_tpc_query(&state->destination, &state->destination_claim,
434 &state->status) == 0;
435 state->planted = write_source(execution, &state->source,
436 &state->source_object, state->payload, &state->status) == 0;
437 if (!state->planted) {
438 grade_failure(execution, "tpc-source-plant-failed", &state->status);
439 return;
440 }
441 if (execution->run->tpc_fixture_kind == NULL)
442 probe_key_semantics(execution, &state->source, &state->source_url,
443 &state->source_object, state->payload);
444 copy_needed = xrd_conform_row_active(execution, "TPC-001")
445 || xrd_conform_row_active(execution, "TPC-008")
446 || xrd_conform_row_active(execution, "CLM-009")
447 || execution->run->tpc_fixture_kind != NULL;
448 state->destination_ready = prepare_destination(&state->destination,
449 &state->destination_object, &state->status);
450 if (!state->destination_ready) {
451 grade_destination_setup(execution, &state->status);
452 return;
453 }
454 if (!copy_needed) return;
455 copy.tpc_mode = XRDC_TPC_ONLY;
456 copy.force = 1;
457 copy.posc = 1;
458 copy.silent = 1;
459 started = time(NULL);
460 state->copy_ok = brix_copy(state->source_text, state->destination_text,
461 &copy, &state->opts, &state->status) == 0;
462 if (started != (time_t) -1) {
463 time_t finished = time(NULL);
464
465 if (finished != (time_t) -1) state->copy_elapsed = finished - started;
466 }
467 if (state->copy_ok)
468 state->exact = xrd_conform_tpc_destination_exact(&state->destination,
469 state->destination_object.path, state->payload,
470 sizeof(state->payload), &state->status);
471 else
472 state->absent = xrd_conform_tpc_destination_absent(&state->destination,
473 state->destination_object.path);
474 if (execution->run->tpc_fixture_kind != NULL
475 && strcmp(execution->run->tpc_fixture_kind, "egress-blocked") == 0)
476 query_source_attempts(state);
477}

488static void
489grade_tpc_claim(xrd_conform_execution *execution,
490 const tpc_probe_state *state)
491{
492 int advertised;
493 int agrees;
494
495 if (!state->destination_ready
496 || execution->run->tpc_fixture_kind != NULL) return;
497 advertised = state->claims_ok && state->source_claim != 0
498 && state->destination_claim != 0;
499 agrees = state->planted && state->claims_ok

501 record_tpc_claim_state(execution, state, agrees);
502 if (!xrd_conform_row_active(execution, "TPC-008")) return;
503 xrd_conform_result_set(execution, "TPC-008",

505 : agrees ? BRIX_CONFORM_VERDICT_CONFORMANT
506 : BRIX_CONFORM_VERDICT_DIVERGENT,
507 !state->planted ? "tpc-source-plant-failed"
508 : !state->claims_ok ? "qconfig-tpc-not-numeric"
509 : agrees ? "qconfig-tpc-agrees-with-observed-transfer"
510 : "qconfig-tpc-claim-behavior-mismatch",
511 "both numeric TPC claims agree with byte-exact behavior", NULL);

537/* WHAT: Execute a native TPC-only transfer between two explicit ROOT endpoints.
538 * WHY: A capability claim is meaningful only when reconciled with exact bytes.
539 * HOW: Plant PRF bytes under scratch, query both claims, hard-fail TPC copy,
540 * reread the destination independently, grade behavior/claim agreement, clean. */
541void
542xrd_conform_probe_tpc_pair(xrd_conform_execution *execution)
543{
544 tpc_probe_state state;
545 int setup;
546
547 if (!xrd_conform_family_active(execution, "TPC")
548 && !xrd_conform_row_active(execution, "CLM-009")) return;
549 if (execution->run->tpc_fixture_kind != NULL
550 && strcmp(execution->run->tpc_fixture_kind,
551 "scoped-delegation") == 0) {
552 xrd_conform_probe_tpc_scoped(execution);
553 return;
554 }
555 grade_required_tpc_fixtures(execution);
556 setup = setup_tpc_pair(execution, &state);
557 if (setup == -1) {
558 grade_failure(execution, "fixture-required:paired-tpc-endpoints",
559 &state.status);
560 if (xrd_conform_row_active(execution, "TPC-008"))
561 xrd_conform_result_set(execution, "TPC-008",
562 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
563 "fixture-required:paired-tpc-endpoints", NULL,
564 state.status.msg);
565 return;
566 }
567 if (setup != 0) {
568 grade_failure(execution, "tpc-probe-setup-failed", &state.status);
569 return;
570 }
571 if (connect_tpc_pair(execution, &state) != 0) return;
572 perform_tpc_copy(execution, &state);
573 grade_tpc_pair(execution, &state);
574 cleanup_pair(&state.source, &state.destination, &state.source_object,
575 &state.destination_object);
576 brix_close(&state.destination);
577 brix_close(&state.source);
578}
