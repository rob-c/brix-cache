29/* Open the catalog and prepare `sql`, binding `bind1`/`bind2` as text to ?1/?2
30 * when non-NULL. On success the caller steps *q and MUST finalize it + close *h;
31 * on failure *q is NULL and *h is closed here. Returns 1 when *q is ready. */
32static int
33pbut_open_prepare(const char *root, const char *sql, const char *bind1,
34 const char *bind2, sqlite3 **h, sqlite3_stmt **q)
35{
36 char db[PATH_MAX];
37
38 *h = NULL;
39 *q = NULL;
40 snprintf(db, sizeof(db), "%s/catalog.db", root);
41 CHECK(sqlite3_open(db, h) == SQLITE_OK, "catalog db open");
42 if (sqlite3_prepare_v2(*h, sql, -1, q, NULL) != SQLITE_OK) {
43 sqlite3_close(*h);
44 *h = NULL;
45 *q = NULL;
46 return 0;
47 }
48 if (bind1 != NULL) {
49 sqlite3_bind_text(*q, 1, bind1, -1, SQLITE_STATIC);
50 }
51 if (bind2 != NULL) {
52 sqlite3_bind_text(*q, 2, bind2, -1, SQLITE_STATIC);
53 }
54 return 1;
55}

57void
58pbut_query_text(const char *root, const char *sql, const char *bind1,
59 char *out, size_t cap)
60{
61 sqlite3 *h;
62 sqlite3_stmt *q;
63
64 out[0] = '\0';
65 if (!pbut_open_prepare(root, sql, bind1, NULL, &h, &q)) {
66 return;
67 }
68 if (sqlite3_step(q) == SQLITE_ROW) {
69 const unsigned char *b = sqlite3_column_text(q, 0);
70
71 snprintf(out, cap, "%s", b ? (const char *) b : "");
72 }
73 sqlite3_finalize(q);
74 sqlite3_close(h);
75}

209/* test_mkdir_trailing_slash — a GridFTP `-cd` MKD arrives as "/dir/" (trailing
210 * slash). The driver must canonicalise it so the catalog keys the directory
211 * without the slash: parent_of("/ts/") would otherwise split into the missing
212 * parent "/ts" + empty leaf → a spurious ENOENT. Covers success (the create
213 * takes and the child is reachable — no orphan), the security/idempotency edge
214 * (a second slash-variant create maps to the same key ⇒ EEXIST, not a dup), and
215 * the error edge (a genuinely-missing parent is still rejected ENOENT — the
216 * strip must not manufacture phantom ancestors). */
217void
218test_mkdir_trailing_slash(brix_sd_instance_t *inst)
219{
220 brix_sd_stat_t st;
221
222 /* success: "/ts/" creates, and the canonical key "/ts" is a directory. */
223 CHECK(D->mkdir(inst, "/ts/", 0755) == NGX_OK, "mkdir trailing slash");
224 CHECK(D->stat(inst, "/ts", &st) == NGX_OK && st.is_dir,
225 "canonical key /ts is a directory");
226 /* the child resolves through the canonical parent — the row is not orphaned. */
227 CHECK(write_file(inst, "/ts/child", "z", 1) == 0, "child under /ts");
228
229 /* idempotency/security: the slash variant maps to the same key ⇒ EEXIST. */
230 errno = 0;
231 CHECK(D->mkdir(inst, "/ts/", 0755) == NGX_ERROR && errno == EEXIST,
232 "second /ts/ is EEXIST (canonical dedup), errno=%d", errno);
233
234 /* error: a real missing parent still ENOENTs — the strip is slash-only. */
235 errno = 0;
236 CHECK(D->mkdir(inst, "/nope/deep/", 0755) == NGX_ERROR && errno == ENOENT,
237 "missing parent still ENOENT, errno=%d", errno);
238}

383typedef struct {
384 brix_sd_instance_t *inst;
385 int id;
386 int ops;
387 int ok;
388} thread_arg;
389
390/* Each thread writes + reads back its own private files through the SHARED
391 * instance (shared SQLite connection + blob dir), proving the per-worker handle
392 * is safe across a thread pool and that blob I/O runs in parallel. */
393static void *
394thread_body(void *p)
395{
396 thread_arg *a = p;
397 int i;
398
399 for (i = 0; i < a->ops; i++) {
400 char path[64];
401 char buf[32];
402 char want[32];
403 int wn = snprintf(want, sizeof(want), "t%d-op%d", a->id, i);
404
405 snprintf(path, sizeof(path), "/conc/t%d-%d", a->id, i);
406 if (write_file(a->inst, path, want, (size_t) wn) != 0) {
407 return NULL;
408 }
409 if (read_file(a->inst, path, buf, sizeof(buf)) != wn
410 || memcmp(buf, want, (size_t) wn) != 0)
411 {
412 return NULL;
413 }
414 }
415 a->ok = 1;
416 return NULL;
417}

442/* Each forked child opens its OWN instance on the same root and writes a batch;
443 * the parent then verifies every child's files landed. Exercises cross-process
444 * WAL locking + the busy-timeout retry path. */
445void
446test_processes(const char *root, brix_sd_instance_t *inst)
447{
448 enum { NPROC = 4, NOPS = 25 };
449 pid_t pids[NPROC];
450 int i, total = 0, rc;
451 brix_sd_dir_t *dir;
452 brix_sd_dirent_t de;
453 int err = 0;
454
455 CHECK(D->mkdir(inst, "/mp", 0755) == NGX_OK, "mkdir mp");
456
457 for (i = 0; i < NPROC; i++) {
458 pid_t pid = fork();
459
460 if (pid == 0) {
461 brix_sd_instance_t cinst = {0};
462 brix_sd_pblock_conf_t conf = { root, 5000, 0, NULL, 0 };
463 int j, bad = 0;
464
465 cinst.driver = D;
466 if (D->init(&cinst, &conf) != NGX_OK) {
467 _exit(2);
468 }
469 for (j = 0; j < NOPS; j++) {
470 char path[64];
471
472 snprintf(path, sizeof(path), "/mp/p%d-%d", (int) getpid(), j);
473 if (write_file(&cinst, path, "x", 1) != 0) {
474 bad = 1;
475 break;
476 }
477 }
478 D->cleanup(&cinst);
479 _exit(bad ? 1 : 0);
480 }
481 CHECK(pid > 0, "fork %d", i);
482 pids[i] = pid;
483 }
484
485 for (i = 0; i < NPROC; i++) {
486 int status = 0;
487
488 waitpid(pids[i], &status, 0);
489 CHECK(WIFEXITED(status) && WEXITSTATUS(status) == 0,
490 "child %d exit status %d", i, status);
491 }
492
493 dir = D->opendir(inst, "/mp", &err);
494 CHECK(dir != NULL, "opendir mp");
495 if (dir != NULL) {
496 while ((rc = D->readdir(dir, &de)) == NGX_OK) {
497 total++;
498 }
499 D->closedir(dir);
500 }
501 CHECK(total == NPROC * NOPS, "expected %d files, found %d", NPROC * NOPS,
502 total);
503}
