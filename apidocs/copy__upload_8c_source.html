8/*
9 * WHAT: Open the upload destination write handle, retrying reconnect+reopen
10 * within the stall deadline on a transport fault.
11 * WHY: A restart can hit during connect/open, before the write loop's resilient
12 * sink is reached; retrying the INITIAL open is safe because nothing is
13 * written yet, so a fresh create/truncate retry is idempotent (matches
14 * download_open_resilient and brix_rfile_open_write). Split out so
15 * upload_stream_body stays a flat sequence.
16 * HOW: Issues brix_file_open_opaque (write=1, force, posc) when a compress
17 * opaque is set (ctx->opaque), else brix_file_open_write; on failure applies
18 * csctx_retry_gate and, if it permits, reconnects to home and retries.
19 * Returns 0 with *f open, or -1 (st set) once the retry gate gives up.
20 * `force`/`posc` are the destination open flags.
21 */
22static int
23upload_open_resilient(const copy_stream_ctx_t *ctx, int force, int posc,
24 brix_file *f, brix_status *st)
25{
26 unsigned attempt = 0;
27 for (;;) {
28 int orc = (ctx->opaque != NULL)
29 ? brix_file_open_opaque(ctx->c, ctx->path,
30 ctx->opaque, 1, force, posc,
31 f, st)
32 : brix_file_open_write(ctx->c, ctx->path, force, posc, f, st);
33 if (orc == 0) {
34 return 0;
35 }
36 if (!csctx_retry_gate(st, ctx->deadline_ns, &attempt)) {
37 return -1;
38 }
39 csctx_reopen_home(ctx->c, st);
40 }
41}

44/*
45 * WHAT: Resiliently perform the upload's final close, which is the COMMIT
46 * (renames the staged partial onto the destination).
47 * WHY: A restart landing on the final close would otherwise leave a
48 * fully-written-but-uncommitted partial. Retrying the close within the
49 * stall window — reconnecting + reopening IN PLACE (no truncate) between
50 * attempts — commits it, since the bytes are all present on the server.
51 * Split out so upload_stream_body stays flat.
52 * HOW: Loops brix_file_close (sink->c / sink->f are the connection + handle);
53 * on failure, gives up on cancel/deadline, else backs off and reopens the
54 * partial in place (pump_sink_reopen) to re-commit. If reopen-in-place
55 * fails a PRIOR close may already have committed with its ack lost to the
56 * sever — for a known size (total >= 0) confirm the commit by the
57 * destination's (sink->path) size and treat as success; otherwise fall back
58 * to retryability. Returns 0 (committed) or -1 (st set).
59 */
60static int
61upload_close_commit_resilient(pump_remote_t *sink, int64_t total,
62 brix_status *st)
63{
64 uint64_t deadline = brix_mono_ns()
65 + (uint64_t) sink->max_stall_ms * 1000000ULL;
66 unsigned attempt = 0;
67 for (;;) {
68 if (brix_file_close(sink->c, sink->f, st) == 0) {
69 return 0;
70 }
71 if (brix_copy_quit_requested() || brix_mono_ns() >= deadline) {
72 return -1;
73 }
74 brix_backoff_sleep_fast(attempt++);
75 if (pump_sink_reopen(sink, st) == 0) {
76 continue; /* reopened the partial in place — loop re-commits */
77 }
78 /*
79 * Reopen-in-place failed. A PRIOR close may have already committed
80 * (renamed the staged partial onto the destination) with its ack lost
81 * to the sever — so the partial is gone and reopen-update NotFounds.
82 * Confirm the commit by the destination's size and treat as success.
83 * (total < 0 = stdin: no known size, so fall back to retryability.)
84 */
85 if (total >= 0) {
86 brix_statinfo si;
87 if (brix_stat(sink->c, sink->path, &si, st) == 0
88 && si.size == total) {
89 brix_status_clear(st);
90 return 0;
91 }
92 }
93 if (!brix_status_retryable(st) || brix_mono_ns() >= deadline) {
94 return -1;
95 }
96 }
97}

100/*
101 * WHAT: Verify a persisted upload's checksum, reconciling the verdict into rc.
102 * WHY: After the commit the file is on the server, so its checksum can be
103 * compared against the local source digest; confining that policy here
104 * keeps upload_stream_body's tail a single call.
105 * HOW: For stdin (XRDC_SCHEME_STDIO) there is no on-disk file, so passes NULL
106 * as the local path (cksum_verify skips gracefully). MISMATCH returns -1;
107 * UNVERIFIED prints the note (unless silent) and clears st (a could-not-
108 * verify is not a transfer failure); OK returns 0.
109 */
110static int
111upload_reconcile_cksum(brix_conn *c, const brix_url *su, const brix_url *du,
112 const brix_copy_opts *o, brix_status *st)
113{
114 const char *ck_local = (su->scheme == XRDC_SCHEME_STDIO) ? NULL : su->path;
115 int ck = cksum_verify(c, du->path, ck_local, o->cksum, o->silent, st);
116 if (ck == XRDC_CK_MISMATCH) {
117 return -1;
118 }
119 if (ck == XRDC_CK_UNVERIFIED) {
120 if (!o->silent) {
121 fprintf(stderr, "xrdcp: uploaded but checksum NOT verified: %s\n",
122 st->msg);
123 }
124 brix_status_clear(st); /* could-not-verify is not a transfer failure */
125 }
126 return 0;
127}

130/*
131 * WHAT: Connect the destination, open it for write, stream bytes from the
132 * caller-supplied (src, srcctx) into it, then tear the whole remote side
133 * down (file close on success, checksum, bound streams, connection) — the
134 * entire "destination session is up" lifetime.
135 * WHY: Confining the connection / write handle / secondary streams to one helper
136 * keeps copy_upload() a flat early-return sequence whose only lingering
137 * resource is the caller-owned VFS handle. Both pre-open failure paths
138 * (connect, open_write) return early without entering the finish teardown.
139 * HOW: connect → open_write → streams_open → transfer_pump(src→remote) → finish.
140 * `total` is the known source size for progress and resilient-close checks
141 * (-1 for stdin / unknown). src is either pump_src_local (stdin) or
142 * pump_src_local_vfs (local file via brix_vfs). su->path is used as the
143 * local checksum source path (NULL ≡ stdin → cksum_verify skips gracefully).
144 * NOTE: the 8-parameter signature is a frozen extern (called from
145 * copy_block.c) — decomposed body-only, per phase-75 G4.
146 */
147int
148upload_stream_body(const upload_body_ctx *j, pump_src_fn src, void *srcctx,
149 brix_status *st)
150{
151 const brix_url *su = j->su;
152 const brix_url *du = j->du;
153 const brix_copy_opts *o = j->o;
154 const brix_opts *co = j->co;
155 int64_t total = j->total;
156 brix_conn c;
157 brix_file f;
158 brix_streamset ss;
159 pump_remote_t sink = {0};
160 int rc;
161 int stall = copy_stall_ms(o, 60000);
162 char opq[80];
163 copy_stream_ctx_t ctx = {0};
164
165 ss.n = 0;
166 if (brix_connect_resilient(&c, du, co, st) != 0) {
167 return -1;
168 }
169
170 /* phase-42 W5: request inline write compression when --compress was given —
171 * the server decompresses each payload on ingest (brix_file_write compresses
172 * transparently once the handle's write_codec is learned). A server that
173 * doesn't support it returns plaintext (write_codec stays 0), so this is safe.
174 * Streams are disabled under write compression (the secondaries would carry
175 * raw payloads the server can't frame). */
176 ctx.c = &c;
177 ctx.path = du->path;
178 ctx.deadline_ns = brix_mono_ns() + (uint64_t) stall * 1000000ULL;
179 if (o->compress != NULL && o->compress[0] != '\0') {
180 snprintf(opq, sizeof(opq), "xrootd.compress=%s", o->compress);
181 ctx.opaque = opq;
182 }
183 /* Resilient INITIAL open (retry+reconnect within the stall window). */
184 if (upload_open_resilient(&ctx,
185 o->force | (o->coerce ? XRDC_OPEN_COERCE : 0),
186 o->posc, &f, st) != 0) {
187 brix_close(&c);
188 return -1;
189 }
190
191 /* M8: attach N-1 bound secondary streams to the (post-redirect) session.
192 * Skip them when write compression is active (see above). */
193 if (f.write_codec == 0) {
194 brix_streams_open(&ss, &c, o->streams, st);
195 }
196
197 /* local src → remote (EOF-driven), with progress (total = file size or -1).
198 * The sink is resilient: a transport sever mid-upload reconnects, reopens the
199 * destination IN PLACE (no truncate) and re-issues from the same offset, so an
200 * upload survives an nginx restart and resumes from where it left off. This
201 * needs the bytes below the offset to still be on the server: true for a
202 * direct-to-final write (default, posc off) and for a server with
203 * brix_upload_resume on (deterministic preserved partial). Re-issuing the
204 * same buffer at the same offset is idempotent. */
205 sink.c = &c;
206 sink.f = &f;
207 sink.pgrw = o->pgrw;
208 sink.resilient = 1;
209 sink.path = du->path;
210 sink.posc = o->posc;
211 sink.max_stall_ms = copy_stall_ms(o, 60000);
212 /* Phase 94: hand the bound-stream set to the sink so it spreads write chunks
213 * across the substreams (parallel upload). Only for plain (non-pgwrite,
214 * non-compressed) writes — pgwrite carries per-page CRCs and compression frames
215 * are primary-only; both keep the single-stream path. */
216 sink.ss = (o->pgrw || f.write_codec != 0) ? NULL : &ss;
217 sink.rr_next = 0;
218 sink.sec_writes = 0;
219 rc = transfer_pump(src, srcctx, pump_sink_remote, &sink, -1, o, total, st);
220
221 /* Phase 94 diagnostic: how many chunks the bound secondaries actually carried
222 * (0 = single-stream or full fallback to the primary). Opt-in so normal output
223 * is unchanged. */
224 if (getenv("BRIX_STREAMS_DEBUG") != NULL) {
225 fprintf(stderr, "brix: upload substreams=%d chunks-on-secondaries=%u\n",
226 ss.n, sink.sec_writes);
227 }
228
229 /* Only close the remote file cleanly on success: with POSC, abandoning the
230 * handle (connection teardown without close) makes the server discard the
231 * partial upload, which is exactly the atomicity we want on error. The close
232 * is the COMMIT, so it is retried resiliently within the stall window. */
233 if (rc == 0) {
234 rc = upload_close_commit_resilient(&sink, total, st);
235 }
236 /* The file is persisted after close — verify its checksum now (connection
237 * still open), comparing our local source digest against the server's. */
238 if (rc == 0 && o->cksum != NULL) {
239 rc = upload_reconcile_cksum(&c, su, du, o, st);
240 }
241 brix_streams_close(&ss);
242 brix_close(&c);
243 return rc;
244}

247int
248copy_upload(const brix_url *su, const brix_url *du, const brix_copy_opts *o,
249 const brix_opts *co, brix_status *st)
250{
251 if (su->scheme == XRDC_SCHEME_STDIO) {
252 /* stdio path: pump from raw STDIN_FILENO; no VFS open */
253 int stdinfd = STDIN_FILENO;
254 upload_body_ctx uj = { su, du, o, co, -1 /* size unknown */ };
255 return upload_stream_body(&uj, pump_src_local, &stdinfd, st);
256 }
257
258 /* Local file path: open via VFS (io_uring selection inside the backend) */
259 {
260 brix_vfs_file *vf = NULL;
261 brix_vfs_open_opts vopts;
262 brix_vfs_stat vst;
263 brix_status tmp_st;
264 pump_local_t lc;
265 int64_t total = -1;
266 int rc;
267
268 vopts.io_uring = o->io_uring;
269 vopts.io_uring_direct = o->io_uring_direct;
270 vopts.expected_size = -1; /* read-only open; hint unused */
271 vopts.cred = NULL;
272
273 if (brix_vfs_open(su->path, XRDC_VFS_READ, &vopts, &vf, st) != 0) {
274 return -1;
275 }
276 brix_status_clear(&tmp_st);
277 if (brix_vfs_fstat(vf, &vst, &tmp_st) == 0) {
278 total = vst.size;
279 }
280
281 lc.vf = vf;
282 {
283 upload_body_ctx uj = { su, du, o, co, total };
284 rc = upload_stream_body(&uj, pump_src_local_vfs, &lc, st);
285 }
286 brix_vfs_close(vf);
287 return rc;
288 }

302int
303copy_web_upload(const brix_url *su, const brix_weburl *du, const brix_copy_opts *o,
304 const brix_opts *co, brix_status *st)
305{
306 char hdrs[8192];
307 brix_vfs_file *vf = NULL;
308 brix_vfs_open_opts vopts;
309 brix_vfs_stat vst;
310 int status = 0, rc;
311
312 if (su->scheme == XRDC_SCHEME_STDIO) {
313 brix_status_set(st, XRDC_EUSAGE, 0,
314 "web upload needs a regular local file (Content-Length); "
315 "stdin not supported");
316 return -1;
317 }
318
319 /* Open the local source through the VFS (byte I/O dispatches to the shared
320 * SD driver), then fstat it for the Content-Length the PUT must promise. */
321 vopts.io_uring = o ? o->io_uring : 0;
322 vopts.io_uring_direct = o ? o->io_uring_direct : 0;
323 vopts.expected_size = -1; /* read-only open; hint unused */
324 vopts.cred = NULL;
325 if (brix_vfs_open(su->path, XRDC_VFS_READ, &vopts, &vf, st) != 0) {
326 return -1;
327 }
328 if (brix_vfs_fstat(vf, &vst, st) != 0) {
329 brix_vfs_close(vf);
330 return -1;
331 }
332 if (vst.is_dir) {
333 /* st_size is only a reliable Content-Length for a non-directory file. */
334 brix_vfs_close(vf);
335 brix_status_set(st, XRDC_EUSAGE, 0,
336 "web upload source must be a regular file: %s", su->path);
337 return -1;
338 }
339 {
340 web_auth_ctx a = { du, "PUT", o, co, st };
341 if (web_auth_headers(&a, hdrs, sizeof(hdrs)) != 0) {
342 brix_vfs_close(vf);
343 return -1;
344 }
345 }
346 {
347 /* Resilient by default: Content-Range PUT chunks that reconnect + resume
348 * from the server's durable offset, so the upload survives an nginx
349 * restart (server brix_webdav_upload_resume). A plain server commits on
350 * the first whole-range chunk, so a single-shot upload still works. */
351 int stall = copy_stall_ms(o, XRDC_DEFAULT_MAX_STALL_MS);
352 char proxybuf[512];
353 const char *pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
354 rc = brix_http_upload_resumable(du->host, du->port, du->tls, du->path,
355 hdrs[0] ? hdrs : NULL, web_upload_src_vfs, vf,
356 (long long) vst.size,
357 co ? co->verify_host : 1, co ? co->ca_dir : NULL, pcert,
358 XRDC_WEB_TIMEOUT_MS, stall, &status, st);
359 }
360 brix_vfs_close(vf);
361 if (rc == 0 && o && !o->silent) {
362 fprintf(stderr, "xrdcp: uploaded %lld bytes (HTTP %d)\n",
363 (long long) vst.size, status);
364 }
365 return rc;
366}
