1/*
2 * sd_cache_prefetch.c — background block prefetch for slice partial objects
3 * (parity audit §4.1: XrdPfc's speculative successor-block fills).
4 *
5 * WHAT: The cache decorator's read_advise vtable slot. A WILLNEED hint on a
6 * slice partial object posts a detached thread-pool job that fills the
7 * hinted range's ABSENT blocks from the source, so a sequential reader
8 * finds its successor blocks already cached instead of paying an origin
9 * round-trip per block. Bounded by policy.prefetch_jobs (in-flight cap,
10 * 0 = off) and policy.prefetch_window (max bytes speculation may run
11 * ahead of the hint cursor — a rolling runway, see the frontier note
12 * in sd_cache_read_advise).
13 *
14 * WHY: Before this, the only prefetch was the local POSIX_FADV_WILLNEED hint
15 * — driver-backed (partial-cache) handles got nothing, and every cold
16 * block cost a synchronous origin read on the serving path. The generic
17 * seam is the existing read_advise slot: the sequential-window engines
18 * (root:// read prefetch, the HTTP memory-backed serve loop) hint
19 * through the handle's driver object, and this decorator turns the hint
20 * into background fills. Any future driver can do the same.
21 *
22 * HOW: EVENT-LOOP ONLY (documented on the slot): the hint snapshots a
23 * self-contained heap job — key, captured per-user credential, block
24 * range — so the job shares NOTHING with the event-loop handle and
25 * survives its close. The thread half opens its OWN partial object
26 * (sd_cache_partial_open re-adopts the cinfo bitmap, so blocks the
27 * foreground filled meanwhile are skipped) and fills the still-absent
28 * blocks via sd_cache_fill_block — the same "pure driver pread/pwrite"
29 * doctrine brix_sd_cache_fill_key already runs on worker threads. The
30 * done callback (event loop) drops the in-flight count and publishes
31 * the process-wide counters. Concurrent foreground fills of the same
32 * block are idempotent (same bytes, same offsets) — the pre-existing
33 * two-workers-one-object scenario, not a new concurrency class.
34 */

46/* Self-contained job: everything the thread needs, copied at post time. inst
47 * and st are per-worker registry-owned (worker-lifetime — they outlive every
48 * request); all request-scoped state is embedded by value. */
49typedef struct {
50 ngx_pool_t *pool; /* owns the task + this ctx */
51 brix_sd_instance_t *inst; /* the cache decorator instance */

86/* Worker thread: open an own partial object for the key and fill the absent
87 * blocks in [blk_first, blk_last]. The fresh open re-adopts the cinfo bitmap,
88 * so blocks the foreground (or another job) filled meanwhile are skipped. */
89static void
90sd_cache_prefetch_thread(void *data, ngx_log_t *log)
91{
92 sd_cache_prefetch_job_t *j = data;
93 brix_sd_cred_t rc;
94 const brix_sd_cred_t *credp = NULL;
95 brix_sd_obj_t *o;
96 sd_cache_partial_t *p;
97 uint64_t blk;
98 int e = 0;
99
100 (void) log;
101
102 if (j->cred_proxy[0] != '\0') {
103 ngx_memzero(&rc, sizeof(rc));
104 rc.x509_proxy = j->cred_proxy;
105 rc.key = j->cred_key;
106 rc.principal = j->cred_principal;
107 credp = &rc;
108 }
109
110 o = sd_cache_partial_open(j->inst, j->st, j->key, credp, &e);
111 if (o == NULL) {
112 j->failed = 1;
113 return;
114 }
115 p = o->state;
116
117 for (blk = j->blk_first; blk <= j->blk_last; blk++) {
118 if (p->bitmap != NULL && blk < p->nblocks
119 && brix_cache_cinfo_block_present(p->bitmap, blk))
120 {
121 continue;
122 }
123 if (sd_cache_fill_block(p, blk) != 0) {
124 j->failed = 1;
125 break;
126 }
127 j->filled++;
128 }
129
130 brix_sd_obj_release(o); /* close + free the heap shell */
131}

133/* Event loop: drop the in-flight count, publish counters, free the job. */
134static void
135sd_cache_prefetch_done(ngx_event_t *ev)
136{
137 ngx_thread_task_t *task = ev->data;
138 sd_cache_prefetch_job_t *j = task->ctx;
139 ngx_brix_metrics_t *shm = prefetch_shm();
140
141 if (j->st->prefetch_active > 0) {
142 j->st->prefetch_active--;
143 }
144 if (shm != NULL) {
145 BRIX_ATOMIC_ADD(&shm->unified.cache_prefetch_blocks_total, j->filled);
146 if (j->failed) {
147 BRIX_ATOMIC_INC(&shm->unified.cache_prefetch_failures_total);
148 }
149 }
150 ngx_log_debug3(NGX_LOG_DEBUG_STREAM, j->st->log, 0,
151 "brix: cache prefetch done: %ui blocks, failed=%d, "
152 "in-flight=%ui", (ngx_uint_t) j->filled, (int) j->failed,
153 j->st->prefetch_active);
154 ngx_destroy_pool(j->pool); /* frees the task + ctx */
155}

157/* Skip leading blocks already present so a hint over cached bytes costs
158 * nothing; returns the first absent block at or after b0 (or b1 + 1). (Benign
159 * race: a foreground fill from an AIO worker may flip a bit under this read —
160 * worst case the job re-checks and skips it.) */
161static uint64_t
162sd_cache_prefetch_skip_present(sd_cache_partial_t *p, uint64_t b0, uint64_t b1)
163{
164 while (b0 <= b1 && p->bitmap != NULL && b0 < p->nblocks
165 && brix_cache_cinfo_block_present(p->bitmap, b0))
166 {
167 b0++;
168 }
169 return b0;
170}

172/* Resolve a WILLNEED hint to the block range [*b0_out, *b1_out] worth posting,
173 * or return 0 when nothing needs fetching. The operator window caps how far
174 * speculation may run past the CURRENT hint cursor (blocks whose first byte
175 * lies before off + window), and the per-handle frontier ratchet skips blocks
176 * an earlier hint on this handle already queued. Together they turn the
177 * engines' repeated [cursor, cursor + engine-window) hints into a continuous
178 * runway that stays at most `window` bytes ahead of the reader — no
179 * burst-then-starve, no compounding runaway, no double-posting. */
180static int
181sd_cache_prefetch_range(sd_cache_partial_t *p, sd_cache_inst_state *st,
182 off_t off, size_t len, uint64_t *b0_out, uint64_t *b1_out)
183{
184 off_t end;
185 uint64_t b0, b1;
186
187 if (p->block_size == 0 || off < 0 || len == 0 || off >= p->size) {
188 return 0;
189 }
190 end = ((off_t) len > p->size - off) ? p->size : off + (off_t) len;
191
192 b0 = (uint64_t) off / p->block_size;
193 b1 = (uint64_t) (end - 1) / p->block_size;
194
195 if (st->policy.prefetch_window > 0) {
196 uint64_t limit = ((uint64_t) off + st->policy.prefetch_window
197 + p->block_size - 1) / p->block_size;
198
199 if (limit == 0 || b0 >= limit) {
200 return 0;
201 }
202 if (b1 >= limit) {
203 b1 = limit - 1;
204 }
205 }
206 if (b0 < p->prefetch_next_blk) {
207 b0 = p->prefetch_next_blk;
208 }
209 b0 = sd_cache_prefetch_skip_present(p, b0, b1);
210 if (b0 > b1) {
211 return 0; /* already cached, queued, or past window */
212 }
213
214 *b0_out = b0;
215 *b1_out = b1;
216 return 1;
217}

219/* Snapshot a self-contained job for blocks [b0, b1] and post it to the
220 * "default" thread pool; on success advance the handle's frontier ratchet and
221 * the in-flight/jobs counters. Every failure path is a silent no-op — a hint
222 * is advisory. */
223static void
224sd_cache_prefetch_post(brix_sd_obj_t *obj, sd_cache_partial_t *p,
225 sd_cache_inst_state *st, uint64_t b0, uint64_t b1)
226{
227 sd_cache_prefetch_job_t *j;
228 ngx_thread_pool_t *tp;
229 ngx_pool_t *pool;
230 ngx_thread_task_t *task;
231 ngx_brix_metrics_t *shm;
232
233 tp = prefetch_thread_pool();
234 if (tp == NULL) {
235 return;
236 }
237 pool = ngx_create_pool(4096, st->log);
238 if (pool == NULL) {
239 return;
240 }
241 task = ngx_thread_task_alloc(pool, sizeof(sd_cache_prefetch_job_t));
242 if (task == NULL) {
243 ngx_destroy_pool(pool);
244 return;
245 }
246 j = task->ctx;
247 j->pool = pool;
248 j->inst = obj->inst;
249 j->st = st;
250 j->blk_first = b0;
251 j->blk_last = b1;
252 ngx_cpystrn((u_char *) j->key, (u_char *) p->key, sizeof(j->key));
253 ngx_cpystrn((u_char *) j->cred_proxy, (u_char *) p->cred_proxy,
254 sizeof(j->cred_proxy));
255 ngx_cpystrn((u_char *) j->cred_key, (u_char *) p->cred_key,
256 sizeof(j->cred_key));
257 ngx_cpystrn((u_char *) j->cred_principal, (u_char *) p->cred_principal,
258 sizeof(j->cred_principal));
259
260 brix_task_bind(task, sd_cache_prefetch_thread, sd_cache_prefetch_done);
261 task->event.log = st->log;
262 if (ngx_thread_task_post(tp, task) != NGX_OK) {
263 ngx_destroy_pool(pool);
264 return;
265 }
266 st->prefetch_active++;
267 p->prefetch_next_blk = b1 + 1; /* advance this handle's ratchet */
268
269 shm = prefetch_shm();
270 if (shm != NULL) {
271 BRIX_ATOMIC_INC(&shm->unified.cache_prefetch_jobs_total);
272 }
273 ngx_log_debug3(NGX_LOG_DEBUG_STREAM, st->log, 0,
274 "brix: cache prefetch posted: blocks %uL..%uL, in-flight=%ui",
275 b0, b1, st->prefetch_active);
276}

278ngx_int_t
279sd_cache_read_advise(brix_sd_obj_t *obj, off_t off, size_t len, int advice)
280{
281 sd_cache_partial_t *p;
282 sd_cache_inst_state *st;
283 uint64_t b0, b1;
284
285 /* Only a WILLNEED hint triggers speculative work: SEQUENTIAL is a no-op
286 * and RANDOM must never amplify origin traffic (XrdPfc disable-on-random
287 * parity — the engines already suppress hints on random access). */
288 if (obj == NULL || obj->state == NULL || obj->inst == NULL
289 || advice != BRIX_SD_ADV_WILLNEED)
290 {
291 return NGX_OK;
292 }
293 p = obj->state;
294 st = SD_CACHE_ST(obj->inst);
295
296 if (st->policy.prefetch_jobs == 0 /* feature off */
297 || st->prefetch_active >= st->policy.prefetch_jobs) /* in-flight cap */
298 {
299 return NGX_OK;
300 }
301 if (sd_cache_prefetch_range(p, st, off, len, &b0, &b1)) {
302 sd_cache_prefetch_post(obj, p, st, b0, b1);
303 }
304 return NGX_OK;
305}
