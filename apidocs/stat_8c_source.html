23/*
24 * brix_make_vfs_body — kXR_stat(kXR_vfs) / `xrdfs statvfs` response body.
25 *
26 * The reference format (XrdCl StatInfoVFS::ParseServerResponse) is SIX
27 * space-separated integers, each a bare number:
28 * "<nodesRW> <freeRW_MB> <utilRW%> <nodesStaging> <freeStaging_MB> <utilStaging%>"
29 * (We previously emitted the 4-field "id size flags mtime" stat line here, which
30 * the stock client rejects as "Invalid response".) Free space comes from
31 * statvfs(2) on the export root; staging is reported as 0 (no tape staging tier).
32 */
33static void
34brix_make_vfs_body(ngx_stream_brix_srv_conf_t *conf, char *out, size_t outsz)
35{
36 struct statvfs vfs;
37 const char *root = conf->common.root_canon[0]
38 ? conf->common.root_canon : "/";
39 long long free_mb = 0, total_mb = 0;
40 int util = 0;
41 int nrw = conf->common.allow_write ? 1 : 0;
42
43 /* Phase-83 F5: a backend implementing the optional `space` slot reports its
44 * own quota-aware logical space (pblock: quota total minus rollup usage);
45 * only exports without a space model fall back to raw statvfs(2). */
46 {
47 brix_sd_instance_t *inst = brix_vfs_backend_resolve(root, ngx_cycle->log);
48
49 if (inst != NULL && inst->driver->space != NULL) {
50 brix_sd_space_t sp;
51
52 if (inst->driver->space(inst, &sp) == NGX_OK) {
53 free_mb = (long long) (sp.free_bytes / 1048576u);
54 total_mb = (long long) (sp.total_bytes / 1048576u);
55 if (total_mb > 0) {
56 util = (int) (100.0 * (double) (total_mb - free_mb)
57 / (double) total_mb);
58 }
59 snprintf(out, outsz, "%d %lld %d 0 0 0", nrw, free_mb, util);
60 return;
61 }
62 }
63 }
64
65 if (statvfs(root, &vfs) == 0) {
66 unsigned long bs = vfs.f_frsize ? vfs.f_frsize : vfs.f_bsize;
67 free_mb = (long long) ((double) vfs.f_bavail * (double) bs / 1048576.0);
68 total_mb = (long long) ((double) vfs.f_blocks * (double) bs / 1048576.0);
69 if (total_mb > 0) {
70 util = (int) (100.0 * (double) (total_mb - free_mb)
71 / (double) total_mb);
72 }
73 }
74
75 snprintf(out, outsz, "%d %lld %d 0 0 0", nrw, free_mb, util);
76}

172/*
173 * stat_symlink_follow_fallback — confined follow of a host-absolute symlink.
174 *
175 * Follow fallback for an in-export symlink with a host-ABSOLUTE target:
176 * RESOLVE_IN_ROOT chroots the absolute target and lands on ENOENT, where
177 * stock follows it on the real fs. Match stock, but CONFINE via realpath —
178 * accept only when the canonical target is within the export root (an
179 * escaping link is rejected; brix_realpath_existing keeps the common
180 * genuine-miss leg at one failing open instead of glibc realpath's
181 * per-component readlink walk). Read-only, so the resolve/stat TOCTOU window
182 * is benign. Returns 0 with *tgt->st filled from the confirmed in-root
183 * target, or -1 (errno describes the last failing probe) to keep the
184 * original miss.
185 */
186static int
187stat_symlink_follow_fallback(brix_ctx_t *ctx, ngx_connection_t *c,
188 ngx_stream_brix_srv_conf_t *conf, stat_target_t *tgt)
189{
190 char real[PATH_MAX];
191 size_t rl = ngx_strlen(conf->common.root_canon);
192 brix_vfs_ctx_t rvctx;
193 brix_vfs_stat_t rvst;
194
195 if (brix_realpath_existing(tgt->full_path, real) == NULL
196 || ngx_strncmp(real, conf->common.root_canon, rl) != 0
197 || (real[rl] != '/' && real[rl] != '\0'))
198 {
199 return -1;
200 }
201
202 /* The canonical target is confirmed within the export root;
203 * read its metadata through the VFS (the symlink chain is
204 * already resolved, so a non-follow probe is exact). */
205 stat_vfs_ctx_prepare(ctx, c, conf, real, &rvctx);
206 if (brix_vfs_probe(&rvctx, 0 /* follow */, &rvst) != NGX_OK) {
207 return -1;
208 }
209
210 brix_vfs_to_struct_stat(&rvst, tgt->st);
211 return 0;
212}

215 * stat_vfs_query — path-mode metadata through the VFS seam.
216 *
217 * kXR_statNoFollow (vendor): lstat the final component so a symlink reports
218 * as itself (kXR_other + target-length size) for FUSE getattr; default
219 * follows symlinks exactly as before. Both go through the VFS seam
220 * (impersonation-aware, RESOLVE_IN_ROOT-confined); the result is projected
221 * back into struct stat for the response code. An ENOENT on a followable

223 * link still stats like stock. Returns 0 with *tgt->st filled, or -1 with
224 * errno describing the failure for the caller's error mapping.

237 : brix_vfs_statf(&vctx, &vst)) == NGX_OK) ? 0 : -1;

280/*
281 * stat_query_path — path-mode stat: resolve, authorize, query metadata.
282 *
283 * dlen > 0 means the payload names a path to resolve and stat(2). Extract
284 * and sanity-check the client path (".." rejected outright, internal
285 * sidecars reported absent), give redirector routing first refusal, then
286 * confine the path beneath the export root, run the auth gate, and read the
287 * metadata through the VFS seam. On success the cache-residency and
288 * nearline flag bits land in tgt->extra_flags. Returns 1 to continue with
289 * the common response tail; 0 when a response was already arranged (*rc
290 * carries the handler's return value).
291 */
292static int
293stat_query_path(brix_ctx_t *ctx, ngx_connection_t *c,
294 ngx_stream_brix_srv_conf_t *conf, stat_target_t *tgt, ngx_int_t *rc)
295{
296 if (!brix_extract_path(c->log, ctx->recv.payload, ctx->recv.cur_dlen,
297 tgt->reqpath_buf, tgt->reqpath_size, 1)) {
298 BRIX_BAIL_ERR(ctx, c, BRIX_OP_STAT, "STAT", "-", "-",
299 kXR_ArgInvalid, "invalid path payload", rc);
300 }
301 tgt->reqpath = tgt->reqpath_buf;
302
303 /* Reject any ".." component outright (the reference does not normalize
304 * ".."). This op resolves through the kernel RESOLVE_BENEATH, which
305 * would silently collapse an in-tree "..", so the guard is explicit. */
306 if (brix_reject_dotdot_path(ctx, c, BRIX_OP_STAT, "STAT", tgt->reqpath)) {
307 *rc = ctx->write_rc;
308 return 0;
309 }
310
311 /* Internal artifacts (sidecars, upload temps) are invisible → report as
312 * absent, never leaking their size/mtime/existence. A trusted cache-store
313 * surface (brix_cache_store_endpoint on) is the sole exception: the cache
314 * node stats its own "<key>.cinfo" sidecar before reading it back.
315 *
316 * The error TEXT is strerror(ENOENT), not a literal of its own: the genuine
317 * miss below answers strerror(errno), so any distinct wording here would
318 * make the refusal readable as "this name is reserved" — the same oracle
319 * the 404-not-403 choice in core/compat/path.c exists to close. */
320 if (!conf->common.cache_store_endpoint
321 && brix_is_internal_name(tgt->reqpath))
322 {
323 BRIX_BAIL_ERR(ctx, c, BRIX_OP_STAT, "STAT", tgt->reqpath, "-",
324 kXR_NotFound, strerror(ENOENT), rc);
325 }
326
327 if (!stat_manager_route(ctx, c, conf, tgt->reqpath, rc)) {
328 return 0;
329 }
330
331 brix_beneath_full_path(conf->common.root_canon, tgt->reqpath,
332 tgt->full_path, tgt->full_path_size);
333
334 if (brix_auth_gate(ctx, c, BRIX_OP_STAT, "STAT",
335 tgt->reqpath, tgt->full_path, conf,
336 BRIX_AUTH_LOOKUP, 0) != NGX_OK) {
337 *rc = ctx->write_rc;
338 return 0;
339 }
340
341 if (stat_vfs_query(ctx, c, conf, tgt) != 0) {
342 /* E-4: a missing path from a principal already over its miss budget is
343 * answered with kXR_wait instead of NotFound, throttling a stat-harvest
344 * loop while a legitimate client (which rarely misses) sails through. */
345 if (errno == ENOENT && conf->negcache.threshold > 0) {
346 unsigned w = brix_negcache_note_miss(ctx, conf->negcache.threshold,
347 conf->negcache.window_ms,
348 conf->negcache.backoff_s);
349 if (w > 0) {
350 ngx_log_error(NGX_LOG_INFO, c->log, 0,
351 "xrootd negcache: kXR_wait %ud for STAT miss %s",
352 w, tgt->reqpath);
353 BRIX_OP_ERR(ctx, BRIX_OP_STAT);
354 *rc = brix_send_wait(ctx, c, w);
355 return 0;
356 }
357 }
358 BRIX_BAIL_ERR(ctx, c, BRIX_OP_STAT, "STAT", tgt->reqpath,
359 "-", brix_kxr_from_errno(errno),
360 strerror(errno), rc);
361 }
362
363 tgt->extra_flags = brix_cache_path_flag(conf, tgt->reqpath);
364 tgt->extra_flags |= stat_residency_flags(ctx, c, conf, tgt->full_path);
365
366 return 1;
367}

369/*
370 * stat_query_handle — handle-based stat: fhandle[0] is our slot index.
371 *
372 * The cached path is only for logging; the real metadata comes from fstat().
373 * Two handle shapes need a size correction: a ZIP member's fd is the ARCHIVE
374 * (take the archive metadata, override size with the member's UNCOMPRESSED
375 * length) and a driver-backed handle's bare fd is only block 0 (ask the
376 * storage driver's worker-safe fstat slot for the catalog-backed metadata).
377 * Returns 1 to continue with the common response tail; 0 when an error
378 * response was sent (*rc carries the handler's return value).
379 */
380static int
381stat_query_handle(brix_ctx_t *ctx, ngx_connection_t *c,
382 stat_target_t *tgt, ngx_int_t *rc)
383{
384 int idx = (int)(unsigned char) tgt->req->fhandle[0];
385 struct stat *st = tgt->st;
386
387 if (!brix_validate_file_handle(ctx, c, idx, "STAT",
388 BRIX_OP_STAT, rc)) {
389 return 0;
390 }
391
392 tgt->full_path[0] = '\0';
393 ngx_cpystrn((u_char *) tgt->full_path,
394 (u_char *) (ctx->files[idx].path != NULL
395 ? ctx->files[idx].path : "-"),
396 tgt->full_path_size);
397
398 if (ctx->files[idx].zip_mode) {
399 /*
400 * ZIP member (phase-57 W2): the handle's fd is the ARCHIVE, so a
401 * plain fstat would report the archive's size. Take the archive's
402 * metadata (mode/mtime/ino) but override the size with the member's
403 * UNCOMPRESSED length (cached_size) so the client sees the logical
404 * file size.
405 */
406 if (fstat(ctx->files[idx].fd, st) != 0) {
407 BRIX_BAIL_ERR(ctx, c, BRIX_OP_STAT, "STAT", tgt->full_path, "-",
408 kXR_IOError, strerror(errno), rc);
409 }
410 st->st_size = (off_t) ctx->files[idx].cached_size;
411 } else if (ctx->files[idx].sd_obj.driver != NULL
412 && ctx->files[idx].sd_obj.driver
413 != brix_sd_default_driver()) {
414 /*
415 * Driver-backed handle (e.g. pblock): the bare fd is only block 0,
416 * so a plain fstat would report the block size, not the logical
417 * object size. Ask the storage driver for the object's
418 * (catalog-backed) metadata via its worker-safe fstat slot.
419 */
420 brix_sd_stat_t sdst;
421
422 if (ctx->files[idx].sd_obj.driver->fstat == NULL
423 || ctx->files[idx].sd_obj.driver->fstat(
424 &ctx->files[idx].sd_obj, &sdst) != NGX_OK)
425 {
426 BRIX_BAIL_ERR(ctx, c, BRIX_OP_STAT, "STAT", tgt->full_path,
427 "-", kXR_IOError, strerror(errno), rc);
428 }
429 ngx_memzero(st, sizeof(*st));
430 st->st_size = sdst.size;
431 st->st_mtime = sdst.mtime;
432 st->st_ctime = sdst.ctime;
433 st->st_ino = sdst.ino;
434 st->st_nlink = 1;
435 st->st_mode = sdst.mode ? (mode_t) sdst.mode
436 : (sdst.is_dir ? (S_IFDIR | 0755) : (S_IFREG | 0644));
437 } else if (fstat(ctx->files[idx].fd, st) != 0) {
438 BRIX_BAIL_ERR(ctx, c, BRIX_OP_STAT, "STAT", tgt->full_path, "-",
439 kXR_IOError, strerror(errno), rc);
440 }
441
442 tgt->extra_flags = ctx->files[idx].from_cache ? kXR_cachersp : 0;
443
444 return 1;
445}
