2 * source_stream.c when that file crossed the 600-line cap (coding-standards §1).
3 * WHAT: tpc_stat_source() kXR_stats the origin by path to capture the pull's

27/*
28 * tpc_stat_parse_size — WHAT: read the size (2nd whitespace token) out of an
29 * XRootD kXR_stat reply body "id size flags mtime". WHY: the size is the pull's
30 * authoritative completion target; the surrounding tokens are irrelevant here.
31 * HOW: skip token 0 (id), then strtoull the leading digits of token 1. Returns 0
32 * with *size set, or -1 if there is no numeric second token.
33 */
34static int
35tpc_stat_parse_size(const char *body, uint64_t *size)
36{
37 const char *p = body;
38 char *end;
39 unsigned long long v;
40
41 while (*p == ' ') {
42 p++;
43 }
44 while (*p != '\0' && *p != ' ') { /* skip token 0 (id) */
45 p++;
46 }
47 while (*p == ' ') {
48 p++;
49 }
50 if (*p < '0' || *p > '9') {
51 return -1;
52 }
53 errno = 0;
54 v = strtoull(p, &end, 10);
55 if (end == p || errno != 0) {
56 return -1;
57 }
58 *size = (uint64_t) v;
59 return 0;
60}

62/*
63 * tpc_stat_source — kXR_stat the remote source by path to capture its
64 * authoritative size (the pull's real completion signal). See source_internal.h.
65 * A distinct streamid tag (4) separates the reply from open(2)/read(3)/close(2).
66 * A source that errors the stat, or returns an unparseable body, leaves
67 * src_size_known=0 — only a socket/framing failure aborts the pull here.
68 */
69int
70tpc_stat_source(brix_tpc_pull_t *t, int fd)
71{
72 u_char req[sizeof(ClientStatRequest) + PATH_MAX];
73 ClientStatRequest streq;
74 size_t pathlen = strlen(t->src_path);
75 size_t total = sizeof(ClientStatRequest) + pathlen;
76 uint16_t status;
77 uint32_t dlen;
78 u_char *body = NULL;
79
80 t->src_size = 0;
81 t->src_size_known = 0;
82
83 if (total > sizeof(req)) {
84 snprintf(t->err_msg, sizeof(t->err_msg),
85 "TPC src path too long for stat");
86 t->xrd_error = kXR_ArgTooLong;
87 return -1;
88 }
89
90 ngx_memzero(&streq, sizeof(streq));
91 streq.streamid[1] = 4; /* distinct tag: stat replies */
92 streq.requestid = htons(kXR_stat);
93 streq.dlen = htonl((kXR_int32) pathlen);
94 ngx_memcpy(req, &streq, sizeof(streq));
95 ngx_memcpy(req + sizeof(streq), t->src_path, pathlen);
96
97 if (tpc_send_all(t, fd, req, total) != 0) {
98 snprintf(t->err_msg, sizeof(t->err_msg), "TPC kXR_stat send failed");
99 t->xrd_error = kXR_IOError;
100 return -1;
101 }
102 if (tpc_recv_response(t, fd, &status, &body, &dlen) != 0) {
103 snprintf(t->err_msg, sizeof(t->err_msg), "TPC kXR_stat recv failed");
104 t->xrd_error = kXR_IOError;
105 return -1;
106 }
107
108 if (status == kXR_ok && body != NULL
109 && tpc_stat_parse_size((const char *) body, &t->src_size) == 0)
110 {
111 t->src_size_known = 1;
112 }
113 free(body);
114 return 0;
115}

131/*
132 * WHAT: Query the remote source for its whole-file checksum response.
133 * WHY: Separate XRootD request framing from digest parsing and comparison.
134 * HOW: Send kXR_query(kXR_Qcksum) on stream tag 5, receive one response, and
135 * return its allocated successful body to the caller.
136 */
137static int
138tpc_query_source_checksum(brix_tpc_pull_t *t, int fd, u_char **body_out)
139{
140 u_char req[sizeof(ClientQueryRequest) + PATH_MAX];
141 ClientQueryRequest qreq;
142 size_t pathlen = strlen(t->src_path);
143 size_t total = sizeof(ClientQueryRequest) + pathlen;
144 uint16_t status;
145 uint32_t dlen;
146 u_char *body = NULL;
147
148 if (total > sizeof(req)) {
149 snprintf(t->err_msg, sizeof(t->err_msg),
150 "TPC src path too long for checksum query");
151 t->xrd_error = kXR_ArgTooLong;
152 return -1;
153 }
154 ngx_memzero(&qreq, sizeof(qreq));
155 qreq.streamid[1] = 5;
156 qreq.requestid = htons(kXR_query);
157 qreq.infotype = htons(kXR_Qcksum);
158 qreq.dlen = htonl((kXR_int32) pathlen);
159 ngx_memcpy(req, &qreq, sizeof(qreq));
160 ngx_memcpy(req + sizeof(qreq), t->src_path, pathlen);
161
162 if (tpc_send_all(t, fd, req, total) != 0) {
163 snprintf(t->err_msg, sizeof(t->err_msg),
164 "TPC checksum query send failed");
165 t->xrd_error = kXR_IOError;
166 return -1;
167 }
168 if (tpc_recv_response(t, fd, &status, &body, &dlen) != 0) {
169 snprintf(t->err_msg, sizeof(t->err_msg),
170 "TPC checksum query recv failed");
171 t->xrd_error = kXR_IOError;
172 return -1;
173 }
174 if (status != kXR_ok || body == NULL) {
175 free(body);
176 return tpc_checksum_error(t,
177 "TPC checksum verify: source supplied no checksum");
178 }
179 *body_out = body;
180 return 0;
181}

184/*
185 * WHAT: Parse the source's "algorithm digest" checksum response.
186 * WHY: Bound both untrusted tokens before passing them to crypto/logging code.
187 * HOW: Split at the first space, trim leading separator space, terminate at
188 * whitespace, and copy only tokens fitting the caller's buffers.
189 */
190static int
191tpc_parse_source_checksum(brix_tpc_pull_t *t, const u_char *body,
192 char *alg, size_t alg_cap, char *hex, size_t hex_cap)
193{
194 const char *text = (const char *) body;
195 const char *value;
196 size_t alglen;
197 size_t vlen;
198
199 value = strchr(text, ' ');
200 if (value == NULL) {
201 return tpc_checksum_error(t,
202 "TPC checksum verify: malformed source checksum reply");
203 }
204 alglen = (size_t) (value - text);
205 if (alglen == 0 || alglen >= alg_cap) {
206 return tpc_checksum_error(t,
207 "TPC checksum verify: bad source checksum type");
208 }
209 ngx_memcpy(alg, text, alglen);
210 alg[alglen] = '\0';
211 do {
212 value++;
213 } while (*value == ' ');
214 vlen = strcspn(value, " \n\r");
215 if (vlen == 0 || vlen >= hex_cap) {
216 return tpc_checksum_error(t,
217 "TPC checksum verify: bad source checksum value");
218 }
219 ngx_memcpy(hex, value, vlen);
220 hex[vlen] = '\0';
221 return 0;
222}

225/*
226 * tpc_verify_dst_hex_writer — sum the destination through the writer session
227 * that still owns its bytes.
228 *
229 * WHY: when the export hands the pull a brix_vfs_writer_t the data lives in the
230 * session's staged temp (or in-place handle) and is NOT yet at dst_path, so
231 * re-opening the final path would sum the wrong object — or nothing at all.
232 * HOW: prefer the session's kernel fd (staged temps are opened read-write); a
233 * driver-backed object exposes none, so fall back to reading the object block by
234 * block through its driver.
235 */
236static ngx_int_t
237tpc_verify_dst_hex_writer(brix_tpc_pull_t *t, const char *alg, ngx_log_t *log,
238 char *hex, size_t hexsz)
239{
240 brix_checksum_alg_t parsed;
241 char normalized[32];
242 ngx_fd_t fd = brix_vfs_writer_fd(t->dst_writer);
243 ngx_int_t rc;
244
245 if (fd != NGX_INVALID_FILE) {
246 return brix_checksum_hex_name_fd(alg, (int) fd, t->dst_path, log, hex,
247 hexsz, normalized, sizeof(normalized));
248 }
249
250 rc = brix_checksum_parse(alg, strlen(alg), &parsed, normalized,
251 sizeof(normalized));
252 if (rc != NGX_OK) {
253 return rc;
254 }
255 return brix_checksum_hex_obj(parsed, &t->dst_obj, t->dst_path, log, hex,
256 hexsz);
257}

260/*
261 * tpc_verify_dst_hex — hex-encode `alg` over the bytes this pull just wrote.
262 *
263 * WHY: the handle the pull WROTE through is not a handle it can READ. A kXR_open
264 * that asks only to write maps to O_WRONLY (open_flags.h), so summing t->dst_fd
265 * makes read(2) fail EBADF and every checksum-verified pull failed closed on a
266 * clean copy — the defect carried unfixed out of the 2026-08-19 session
267 * (history-testing-and-incidents §13 coda; the writer theory recorded there is
268 * the OTHER arm, below).
269 * HOW: with no writer session the bytes are already fsynced at dst_path, so take
270 * a fresh confined O_RDONLY fd on it — beneath the same root_canon done.c
271 * unlinks through, so the read is confined exactly as the write was — sum it and
272 * close. With a writer session the bytes are still in the session; delegate.
273 */
274static ngx_int_t
275tpc_verify_dst_hex(brix_tpc_pull_t *t, const char *alg, ngx_log_t *log,
276 char *hex, size_t hexsz)
277{
278 char normalized[32];
279 ngx_int_t rc;
280 int fd;
281
282 if (t->dst_writer != NULL) {
283 return tpc_verify_dst_hex_writer(t, alg, log, hex, hexsz);
284 }
285
286 fd = brix_vfs_open_fd(log, t->conf->common.root_canon, t->dst_path,
287 O_RDONLY | O_NOFOLLOW, 0);
288 if (fd < 0) {
289 return NGX_ERROR;
290 }
291
292 rc = brix_checksum_hex_name_fd(alg, fd, t->dst_path, log, hex, hexsz,
293 normalized, sizeof(normalized));
294 close(fd);
295 return rc;
296}

299/*
300 * tpc_verify_source_checksum — opt-in post-copy integrity for the TPC pull.
301 * kXR_query(kXR_Qcksum) the source (distinct streamid tag 5), parse the
302 * "<alg> <hex>" reply, recompute the same algorithm over the written destination
303 * with brix_checksum_hex_name_fd, and fail closed (kXR_ChkSumErr) on any of:
304 * source cannot supply a checksum, malformed reply, an algorithm brix cannot
305 * compute, a destination read failure, or a digest mismatch. See
306 * source_internal.h.
307 */
308int
309tpc_verify_source_checksum(brix_tpc_pull_t *t, int fd)
310{
311 u_char *body = NULL;
312 char alg[32];
313 char src_hex[2 * EVP_MAX_MD_SIZE + 1];
314 char local_hex[2 * EVP_MAX_MD_SIZE + 1];
315 ngx_log_t *log = (t->c != NULL) ? t->c->log : ngx_cycle->log;
316
317 if (tpc_query_source_checksum(t, fd, &body) != 0) {
318 return -1;
319 }
320 if (tpc_parse_source_checksum(t, body, alg, sizeof(alg), src_hex,
321 sizeof(src_hex)) != 0)
322 {
323 free(body);
324 return -1;
325 }
326 free(body);
327
328 /* Recompute the SAME algorithm over the destination and compare. */
329 if (tpc_verify_dst_hex(t, alg, log, local_hex, sizeof(local_hex)) != NGX_OK)
330 {
331 snprintf(t->err_msg, sizeof(t->err_msg),
332 "TPC checksum verify: cannot compute %s on destination", alg);
333 t->xrd_error = kXR_ChkSumErr;
334 return -1;
335 }
336
337 if (strcasecmp(local_hex, src_hex) != 0) {
338 snprintf(t->err_msg, sizeof(t->err_msg),
339 "TPC checksum mismatch: source %s=%s destination=%s",
340 alg, src_hex, local_hex);
341 t->xrd_error = kXR_ChkSumErr;
342 return -1;
343 }
344
345 return 0;
346}
