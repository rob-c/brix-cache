2 *
3 * WHAT: Decodes base64url-encoded input (RFC 4648 URL-safe variant) into raw binary.
4 *
5 * - Uses '-' instead of '+' and '_' instead of '/' (URL-safe alphabet)
6 * - Validates padded length <= BRIX_B64_DECODE_MAX bytes (overflow prevention)
7 * - Converts '-' -> '+' and '_' -> '/' via char-by-char replacement in stack buffer
8 * - Pads remainder with '=' for OpenSSL decoder alignment
9 * - Calculates decoded_max = padded_len/4*3 - pad_count; rejects if > out_max
10 * - Decodes via OpenSSL EVP_ENCODE_CTX API into private buffer (not caller's)
11 * - EVP_DecodeInit() initializes context
12 * - EVP_DecodeUpdate() processes main block (returns out_len)
13 * - EVP_DecodeFinal() handles padding (returns tmp_len)
14 * - Total = out_len + tmp_len, re-checked against out_max, memcpy'd out
15 * - Returns -1 on any validation or decoding failure
16 * - Stack-only: BRIX_B64_DECODE_MAX input buffer + BRIX_B64_DECODE_MAX*3/4 decode buffer
17 *
18 * WHY: Required for JWT token payloads and opaque continuation tokens in URLs.
19 *
20 * - '-'/'_' substitution avoids percent-encoding in URLs/HTTP headers
21 * - OpenSSL EVP API provides verified crypto decoding (no reimplementation)
22 * - BRIX_B64_DECODE_MAX cap prevents DoS via oversized inputs
23 * - Thread-safe: pure function, no shared state, stack-only allocation
24 */

33 * WHAT: locate the two '.' separators of "header.payload.signature" and return

49 if (dot1 == NULL) {

56 seg[0].p = tok; seg[0].n = (size_t) (dot1 - tok);

62ssize_t b64url_decode(const char *in, size_t in_len, uint8_t *out, size_t out_max) {
63 size_t padded_len = in_len + (4 - in_len % 4) % 4;
64 if (padded_len > BRIX_B64_DECODE_MAX) return -1;
65 char tmp[BRIX_B64_DECODE_MAX]; /* Max base64 decode buffer */
66 /* OpenSSL writes three bytes for every four base64 characters and subtracts
67 * the padding from the *reported* count only (EVP_DecodeBlock semantics), so
68 * a padded token has up to two bytes written past the length it decodes to.
69 * The capacity check below deliberately admits an out_max sized to that
70 * decoded length, so decoding straight into `out` overruns a caller who
71 * sized exactly — pre-auth, on every bearer token. Decode into our own
72 * buffer instead and hand back only the bytes that are really there; the
73 * client's ftp_gsi_cred.c meets the same hazard by allocating the full 3/4
74 * width. Do NOT collapse this back on the grounds that it does not
75 * reproduce: OpenSSL 3.5 no longer writes those bytes, 3.0.x does, and the
76 * contract cannot depend on which one is linked. */
77 uint8_t raw[BRIX_B64_DECODE_MAX / 4 * 3]; /* Decoded output (3/4 of input) */
78 size_t i;
79 for (i = 0; i < in_len; i++) {
80 if (in[i] == '-') tmp[i] = '+';
81 else if (in[i] == '_') tmp[i] = '/';
82 else tmp[i] = in[i];
83 }
84 for (; i < padded_len; i++) tmp[i] = '=';
85 size_t pad = 0;
86 while (pad < padded_len && tmp[padded_len - pad - 1] == '=') pad++;
87 size_t decoded_max = padded_len / 4 * 3 - pad;
88 if (decoded_max > out_max) return -1;
89 EVP_ENCODE_CTX *ctx = EVP_ENCODE_CTX_new();
90 if (!ctx) return -1;
91 EVP_DecodeInit(ctx);
92 int out_len = 0, tmp_len = 0;
93 if (EVP_DecodeUpdate(ctx, raw, &out_len, (unsigned char*)tmp, (int)padded_len) < 0) {
94 EVP_ENCODE_CTX_free(ctx);
95 return -1;
96 }
97 if (EVP_DecodeFinal(ctx, raw + out_len, &tmp_len) < 0) {
98 EVP_ENCODE_CTX_free(ctx);
99 return -1;
100 }
101 EVP_ENCODE_CTX_free(ctx);
102 /* Widen each operand BEFORE the add so the sum is computed in ssize_t,
103 * not int (bugprone-misplaced-widening-cast). */
104 ssize_t decoded = (ssize_t) out_len + (ssize_t) tmp_len;
105 if (decoded < 0 || (size_t) decoded > out_max) return -1;
106 memcpy(out, raw, (size_t) decoded);
107 return decoded;
108}
109/* HOW: Padded length calculation and validation.
110 *
111 * - padded_len = in_len + (4 - in_len % 4) % 4; rejects if > BRIX_B64_DECODE_MAX
112 * - Stack tmp[BRIX_B64_DECODE_MAX] for converted input
113 * - Loop i=0→in_len: '-'→'+', '_'→'/', else copy unchanged
114 * - Pad i→padded_len with '=' characters
115 * - Count pad by scanning backwards from padded_len-1 for '='
116 * - decoded_max = padded_len/4*3 - pad; rejects if > out_max
117 * - EVP_ENCODE_CTX_new(); returns -1 on NULL
118 * - EVP_DecodeInit(ctx)
119 * - EVP_DecodeUpdate(ctx,raw,&out_len,tmp,(int)padded_len); -1 on error
120 * - EVP_DecodeFinal(ctx,raw+out_len,&tmp_len); -1 on error
121 * - EVP_ENCODE_CTX_free(ctx)
122 * - Sum out_len + tmp_len in ssize_t; rejects if > out_max
123 * - memcpy(decoded bytes) from raw to out; returns decoded length

126/*
127 *
128 * WHAT: Encodes source bytes into base64url string (RFC 4648 URL-safe variant).
129 *
130 * - Uses '-' instead of '+' and '_' instead of '/' (URL-safe alphabet)
131 * - Custom 64-char lookup table: A-Z, a-z, 0-9, '-', '_'
132 * - Processes 3-byte chunks -> 4 output chars per iteration
133 * - Loop: i+2 < slen && di+4 < dsz-1 (checks source + dest capacity)
134 * - Partial final group: i < slen but i+2 >= slen
135 * - 1 remaining byte -> 2 output chars
136 * - 2 remaining bytes -> 3 output chars
137 * - Null-terminates: dst[di] = '\0'
138 * - Returns length implicitly via di counter (caller checks dsz capacity)
139 *
140 * WHY: URL-safe strings for JWT tokens, continuation tokens, HTTP transport.
141 *
142 * - No percent-encoding needed for '+' and '/' in URLs/headers
143 * - Minimal impl avoids OpenSSL for non-crypto encoding
144 * - Stack-only (static lookup table) - no heap pressure
145 * - Thread-safe: pure function, static table, no shared state
146 */

148/*
149 * Minimal base64url encode — key string → opaque continuation token.
150 * */
151void
152b64url_encode(const char *src, size_t slen, char *dst, size_t dsz)
153{
154 static const char tbl[] =
155 "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
156 size_t di = 0;
157 size_t i;
158
159 for (i = 0; i + 2 < slen && di + 4 < dsz - 1; i += 3) {
160 uint32_t v = ((uint32_t)(unsigned char)src[i] << 16)
161 | ((uint32_t)(unsigned char)src[i + 1] << 8)
162 | ((uint32_t)(unsigned char)src[i + 2]);
163 dst[di++] = tbl[(v >> 18) & 0x3f];
164 dst[di++] = tbl[(v >> 12) & 0x3f];
165 dst[di++] = tbl[(v >> 6) & 0x3f];
166 dst[di++] = tbl[ v & 0x3f];
167 }
168 if (i < slen && di + 2 < dsz - 1) {
169 uint32_t v = (uint32_t)(unsigned char)src[i] << 16;
170 if (i + 1 < slen) {
171 v |= (uint32_t)(unsigned char)src[i + 1] << 8;
172 }
173 dst[di++] = tbl[(v >> 18) & 0x3f];
174 dst[di++] = tbl[(v >> 12) & 0x3f];
175 if (i + 1 < slen) {
176 dst[di++] = tbl[(v >> 6) & 0x3f];
177 }
178 }
179 dst[di] = '\0';
180}
181/* HOW: Encoding algorithm.
182 *
183 * - Static 64-char lookup table: A-Z,a-z,0-9,-,_ (base64url alphabet)
184 * - Initializes di=0, i=0
185 * - Main loop (i+2<slen && di+4<dsz-1):
186 * - Reads 3 source bytes into uint32_t v via left-shifts
187 * - Extracts 6-bit groups: (v>>18)&0x3f, (v>>12)&0x3f, (v>>6)&0x3f, v&0x3f
188 * - Maps each to tbl[] char at dst[di++]
189 * - Partial final group (i<slen && di+2<dsz-1):
190 * - Reads 1-2 remaining bytes into v via shift
191 * - Extracts 6-bit groups -> 2-3 output chars based on available source
192 * - Null-terminates: dst[di] = '\0'
193 * - Returns implicitly via di counter (caller checks dsz before calling)
194 */
