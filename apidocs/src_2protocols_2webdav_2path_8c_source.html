13/* ---- Normalize one decoded logical WebDAV path --------------------------
14 *
15 * WHAT: Collapses repeated separators and removes trailing separators except
16 * for the root path, in place. NULL is a no-op.
17 *
18 * WHY: Authorization and storage must use one spelling for paths that the
19 * filesystem treats as identical, including independently decoded
20 * Destination headers.
21 *
22 * HOW: Compact the string with read/write cursors, then trim trailing '/'
23 * bytes while retaining a single root separator.
24 */
25void
26webdav_normalize_decoded_path(char *path)
27{
28 char *src;
29 char *dst;
30 size_t length;
31
32 if (path == NULL) {
33 return;
34 }
35
36 src = path;
37 dst = path;
38 while (*src != '\0') {
39 if (*src == '/' && dst > path && dst[-1] == '/') {
40 src++;
41 continue;
42 }
43 *dst++ = *src++;
44 }
45 *dst = '\0';
46
47 length = (size_t) (dst - path);
48 while (length > 1 && path[length - 1] == '/') {
49 path[--length] = '\0';
50 }
51}

53/* ---- Decode the request URI into the canonical logical path ------------
54 *
55 * WHAT: Produces the normalized logical WebDAV path used by authorization and
56 * VFS resolution. Returns NGX_OK or the decoder's HTTP error status.
57 *
58 * WHY: nginx has already normalized the wire URI once; comparing r->uri in an
59 * authorization gate and decoding it again for storage lets encoded bytes
60 * name different policy and storage paths.
61 *
62 * HOW: Apply the module's NUL-rejecting decoder, then delegate separator
63 * normalization to the shared in-place helper.
64 */
65ngx_int_t
66webdav_decode_request_path(ngx_http_request_t *r, char *out, size_t outsz)
67{
68 ngx_int_t rc;
69
70 rc = webdav_urldecode(r->uri.data, r->uri.len, out, outsz);
71 if (rc != NGX_OK) {
72 return rc;
73 }
74 webdav_normalize_decoded_path(out);
75 return NGX_OK;
76}

78/* Wrapper: maps brix_http_resolve_path() return codes to NGX_HTTP_*,
79 * using 409 Conflict when the destination parent cannot be resolved

81static ngx_int_t

94webdav_resolve_destination_path(ngx_log_t *log, const char *op_label,
95 const char *root_canon, const char *decoded_path,

97{
98 char stripped[PATH_MAX];
99 size_t dlen;
100 int rc;
101
102 (void) log;
103 (void) op_label;
104
105 dlen = strlen(decoded_path);
106 while (dlen > 1 && decoded_path[dlen - 1] == '/')
107 dlen--;
108
109 if (dlen >= sizeof(stripped))
110 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
111 memcpy(stripped, decoded_path, dlen);
112 stripped[dlen] = '\0';
113
114 rc = brix_http_resolve_path_ex(root_canon, stripped, out, outsz,
115 allow_internal ? 1u : 0u);
116 if (rc == 0)
117 return NGX_OK;
118 /* 404 from shared resolver → parent doesn't exist → 409 Conflict */
119 if (rc == 404)
120 return NGX_HTTP_CONFLICT;
121 return map_resolve_rc(rc);
122}
