1/*
2 * digest_header.c — RFC-3230 `Digest:` header grammar (see digest_header.h).
3 *
4 * The token table is the single source of truth for three facts per algorithm:
5 * the wire token, the canonical brix name, and whether the value is base64 or
6 * hex. It grew out of the WebDAV PUT ingest-digest parser, which now shares it
7 * with the sd_http checksum-offload slot rather than keeping a private copy.
8 */
9
10#include "digest_header.h"
11
12#include "hex.h" /* brix_hex_encode — lowercase, NUL-terminated */

14/* Decode scratch for a base64 value. Sized by ngx_base64_decoded_length(), NOT
15 * by the digest width: that bound is ((len+3)/4)*3, which for a padded sha-512
16 * (88 chars) is 66 — three bytes over the 64 the digest actually occupies. A
17 * buffer sized at 64 makes every sha-512 fail the pre-decode bound check and
18 * read as an unusable value. The true width is still enforced downstream, by
19 * the hex output not fitting BRIX_DIGEST_HEX_MAX. */
20#define BRIX_DIGEST_RAW_MAX 72
21
22/* `b64` = the value is base64 (md5/sha per RFC 3230 + RFC 1864); otherwise it
23 * is lowercase hex (the WLCG/dCache convention for the CRC family). `hexw` is
24 * the algorithm's fixed hex width, used to re-pad a value an origin trimmed.
25 *
26 * Both spellings of each SHA name are listed because origins disagree: RFC 3230
27 * registers the hyphenated form, several WLCG endpoints send the bare one. The
28 * hyphenated spelling comes first so the reverse lookup that builds a
29 * `Want-Digest:` request asks in the registered form. */
30static const struct {
31 const char *tok;
32 size_t toklen;
33 const char *alg;
34 int b64;
35 size_t hexw;
36} digest_tokens[] = {
37 { "md5", 3, "md5", 1, 32 },
38 { "sha-256", 7, "sha256", 1, 64 },
39 { "sha256", 6, "sha256", 1, 64 },
40 { "sha-512", 7, "sha512", 1, 128 },
41 { "sha512", 6, "sha512", 1, 128 },
42 { "sha-1", 5, "sha1", 1, 40 },
43 { "sha1", 4, "sha1", 1, 40 },
44 { "adler32", 7, "adler32", 0, 8 },
45 { "crc32c", 6, "crc32c", 0, 8 },
46 { "crc32", 5, "crc32", 0, 8 },
47};

51const char *
52brix_digest_wire_token(const char *canon_alg)
53{
54 size_t i;
55
56 if (canon_alg == NULL) {
57 return NULL;
58 }
59 for (i = 0; i < DIGEST_NTOKENS; i++) {
60 if (ngx_strcmp(canon_alg, digest_tokens[i].alg) == 0) {
61 return digest_tokens[i].tok;
62 }
63 }
64 return NULL;

67/*
68 * WHAT: Classify one ASCII hexadecimal digit from a checksum header.
69 * WHY: Value decoding keeps output sizing separate from byte grammar.
70 * HOW: Accept decimal digits and either case of the six hex letters.
71 */
72static int
73digest_is_hex(u_char value)
74{
75 return (value >= '0' && value <= '9') ||
76 (value >= 'a' && value <= 'f') ||
77 (value >= 'A' && value <= 'F');
78}

80/*
81 * WHAT: Copy a hex-encoded digest value into `out`, lowercased.
82 * WHY: Split from the base64 leg so neither branch carries the other's nesting.
83 * HOW: Validate the WHOLE value before writing a byte of it — a malformed
84 * value must never be reported as a digest, since the caller treats what
85 * it gets as truth, and a rejected value must leave `out` untouched
86 * rather than seeded with a shorter prefix that still looks like hex.
87 */
88static ngx_int_t
89digest_hex_copy(const u_char *val, size_t vlen, char *out, size_t outsz)
90{
91 size_t i;
92
93 if (vlen + 1 > outsz) {
94 return NGX_ERROR;
95 }
96 for (i = 0; i < vlen; i++) {
97 if (!digest_is_hex(val[i])) {
98 return NGX_ERROR;
99 }
100 }
101 for (i = 0; i < vlen; i++) {
102 out[i] = (char) ngx_tolower(val[i]);
103 }
104 out[vlen] = '\0';
105 return NGX_OK;
106}

108/*
109 * WHAT: Decode a base64 digest value and hex-encode it into `out`.
110 * WHY: RFC 3230 carries md5/sha values in base64; every consumer here compares
111 * hex. HOW: bound the decode against a stack buffer first (no pool on a
112 * driver thread), then ngx_decode_base64 + brix_hex_encode.
113 */
114static ngx_int_t
115digest_b64_hex(const u_char *val, size_t vlen, char *out, size_t outsz)
116{
117 u_char raw[BRIX_DIGEST_RAW_MAX];
118 ngx_str_t src, dst;
119
120 if (ngx_base64_decoded_length(vlen) > sizeof(raw)) {
121 return NGX_ERROR;
122 }
123 src.data = (u_char *) val;
124 src.len = vlen;
125 dst.data = raw;
126 if (ngx_decode_base64(&dst, &src) != NGX_OK) {
127 return NGX_ERROR;
128 }
129 if (dst.len == 0 || dst.len * 2 + 1 > outsz) {
130 return NGX_ERROR;
131 }
132 brix_hex_encode(raw, dst.len, out);
133 return NGX_OK;
134}

136ngx_int_t
137brix_digest_value_hex(const u_char *val, size_t vlen, int is_b64, char *out,
138 size_t outsz)
139{
140 if (vlen == 0 || outsz == 0) {
141 return NGX_ERROR;
142 }
143 return is_b64 ? digest_b64_hex(val, vlen, out, outsz)
144 : digest_hex_copy(val, vlen, out, outsz);

147void
148brix_digest_hex_pad(const char *canon_alg, char *hex, size_t hex_sz)
149{
150 size_t i, have, want = 0;
151
152 for (i = 0; i < DIGEST_NTOKENS; i++) {
153 if (ngx_strcmp(canon_alg, digest_tokens[i].alg) == 0) {
154 want = digest_tokens[i].hexw;
155 break;
156 }
157 }
158 have = ngx_strlen(hex);
159 if (want == 0 || have >= want || want + 1 > hex_sz) {
160 return;
161 }
162 ngx_memmove(hex + (want - have), hex, have + 1);
163 ngx_memset(hex, '0', want - have);
164}

166/* Where one matched pair's result is delivered. Bundled so the per-pair helper
167 * stays readable: `want` NULL means "the first supported algorithm wins". */
168typedef struct {

172 size_t hex_sz;
173} digest_sink_t;

175/* Strip leading and trailing linear whitespace from a [*s, *s+*len) slice. */
176static void
177digest_tok_trim(const u_char **s, size_t *len)
178{
179 const u_char *p = *s;
180 size_t n = *len;
181
182 while (n > 0 && (*p == ' ' || *p == '\t')) { p++; n--; }
183 while (n > 0 && (p[n - 1] == ' ' || p[n - 1] == '\t')) { n--; }
184 *s = p;
185 *len = n;
186}

188/*
189 * WHAT: Decide one trimmed `token=value` pair. Returns 1 when the pair settles
190 * the scan (*kind then FOUND or BAD), 0 to keep looking.
191 * WHY: An origin may list several digests; only the one the caller asked for
192 * (or, with no request, the first we understand) may end the scan.
193 * HOW: Table match on the token → skip when a specific algorithm was asked for
194 * and this is not it → normalise the value per the table's encoding flag.
195 */
196static int
197digest_pair(const u_char *tok, size_t tlen, const u_char *v, size_t vlen,
198 const digest_sink_t *sink, brix_digest_kind_t *kind)
199{
200 size_t i;
201
202 for (i = 0; i < DIGEST_NTOKENS; i++) {
203 if (tlen != digest_tokens[i].toklen
204 || ngx_strncasecmp((u_char *) tok,
205 (u_char *) digest_tokens[i].tok, tlen) != 0)
206 {
207 continue;
208 }
209 if (sink->want != NULL
210 && ngx_strcmp(sink->want, digest_tokens[i].alg) != 0)
211 {
212 return 0; /* a digest, but not the one asked for */
213 }
214 if (brix_digest_value_hex(v, vlen, digest_tokens[i].b64,
215 sink->hex_out, sink->hex_sz) != NGX_OK)
216 {
217 *kind = BRIX_DIGEST_BAD;
218 } else {
219 if (sink->alg_out != NULL) { *sink->alg_out = digest_tokens[i].alg; }
220 *kind = BRIX_DIGEST_FOUND;
221 }
222 return 1;
223 }
224 return 0;
225}

227brix_digest_kind_t
228brix_digest_header_scan(const u_char *val, size_t len, const char *want_canon,
229 const char **alg_out, char *hex_out, size_t hex_sz)
230{
231 digest_sink_t sink = { want_canon, alg_out, hex_out, hex_sz };
232 brix_digest_kind_t kind = BRIX_DIGEST_NONE;
233 const u_char *p = val, *end = val + len;
234
235 if (val == NULL || hex_out == NULL || hex_sz == 0) {
236 return BRIX_DIGEST_NONE;
237 }
238 while (p < end) {
239 const u_char *comma = ngx_strlchr((u_char *) p, (u_char *) end, ',');
240 const u_char *iend = comma ? comma : end;
241 const u_char *eq = ngx_strlchr((u_char *) p, (u_char *) iend, '=');
242
243 if (eq != NULL) {
244 const u_char *tok = p, *v = eq + 1;
245 size_t tlen = (size_t) (eq - p);
246 size_t vlen = (size_t) (iend - (eq + 1));
247
248 digest_tok_trim(&tok, &tlen);
249 digest_tok_trim(&v, &vlen);
250 if (digest_pair(tok, tlen, v, vlen, &sink, &kind)) {
251 return kind;
252 }
253 }
254 p = comma ? comma + 1 : end;
255 }
256 return BRIX_DIGEST_NONE;
257}
