1/*
2 * aio_io_dispatch.c - what a received frame does to its request
3 *
4 * WHAT: The frame half of the async connection: routing one fully-received
5 * frame to its in-flight request (kXR_attn unwrapping, oksofar
6 * accumulation, waitresp deferral, terminal completion), plus the
7 * bookkeeping for a body received straight into the caller's buffer.
8 *
9 * WHY: Split out of aio_io.c, which had grown past the size limit carrying
10 * two unrelated concerns. The other half is byte movement — draining the
11 * write queue, filling rbuf, the TLS/plain/direct read steps — and it
12 * only ever touches a request through the four entry points declared for
13 * it in aio_internal.h. The seam is where the bytes stop being bytes.
14 *
15 * HOW: Behaviour-identical to the code it came from; the only change is that
16 * aconn_direct_{target,begin,frame_done} are no longer static, because
17 * the parser and the direct read step that call them stayed behind.
18 */
19#include "aio_internal.h"

22/* kXR_waitresp: the reply arrives later as an unsolicited kXR_attn(asynresp).
23 * Park the request across the server's advertised delay — extend the deadline
24 * (never shorten it) past that delay plus margin so the timeout sweep doesn't
25 * fail the request while the server is legitimately working, and mark it
26 * deferred so its eventual completion is not taken as an RTT sample (it would
27 * measure the deferral, not the link). Mirrors the sync path's read-window
28 * extension (frame.c recv_after_waitresp); clamps match frame.c (570 s + 30 s). */
29static void
30areq_note_deferral(brix_areq *r, const uint8_t *body, uint32_t dlen)
31{
32 uint64_t secs = xrd_wait_secs_parse(body, dlen, 0, 570);
33
34 r->deferred = 1;
35 if (r->deadline_ns != 0) {
36 uint64_t want = brix_mono_ns() + secs * 1000000000ULL + 30000000000ULL;
37 if (want > r->deadline_ns) {
38 r->deadline_ns = want;
39 }
40 }
41}

44/* Dispatch one response (direct, or unwrapped from an asynresp envelope) to its
45 * in-flight request by streamid. Handles response statuses ONLY — kXR_attn never
46 * reaches here from the wire (aconn_dispatch_frame routes it first), so a nested
47 * attn inside an asynresp lands in the unexpected-status arm and fails the
48 * request cleanly instead of recursing. */
49static void
50aconn_dispatch_response(brix_aconn *ac, uint16_t sid, uint16_t stat,
51 const uint8_t *body, uint32_t dlen)
52{
53 brix_areq *r = reqmap_get(&ac->inflight, sid);
54 if (r == NULL) {
55 return; /* late frame for a completed/cancelled request — ignore */
56 }
57
58 switch (stat) {
59 case kXR_oksofar:
60 (void) areq_accumulate(r, body, dlen);
61 return; /* more frames follow; keep the entry */
62
63 case kXR_waitresp:
64 areq_note_deferral(r, body, dlen);
65 return; /* the deferred reply arrives as kXR_attn(asynresp); keep the
66 * entry (a repeat waitresp simply re-arms the wait) */
67
68 case kXR_ok:
69 if (!r->deferred) {
70 aconn_note_rtt(ac, r); /* RTT sample feeds the adaptive deadline/RTO */
71 }
72 /* fallthrough */
73 case kXR_redirect: /* delivered to the upper layer (M2/M3 acts) */
74 case kXR_wait:
75 case kXR_authmore:
76 (void) areq_accumulate(r, body, dlen);
77 reqmap_del(&ac->inflight, sid);
78 areq_complete(r, 0, stat, NULL);
79 return;
80
81 case kXR_error: {
82 int errnum = 0;
83 const char *emsg = "";
84 size_t emlen = 0;
85 brix_status st;
86 /* msg is NOT NUL-terminated on the wire — decode to a bounded slice and
87 * print with %.*s (the old %s on body+4 was a heap over-read). */
88 xrd_error_body_decode(body, dlen, &errnum, &emsg, &emlen);
89 brix_status_set(&st, errnum, 0, "%.*s (%s)", (int) emlen,
90 emsg ? emsg : "", brix_kxr_name(errnum));
91 reqmap_del(&ac->inflight, sid);
92 areq_complete(r, -1, (uint16_t) errnum, &st);
93 return;
94 }
95
96 default: {
97 brix_status st;
98 brix_status_set(&st, XRDC_EPROTO, 0, "unexpected response status %u", stat);
99 reqmap_del(&ac->inflight, sid);
100 areq_complete(r, -1, XRDC_EPROTO, &st);
101 return;
102 }
103 }
104}

107/* Handle an unsolicited kXR_attn push. Body = [actnum u32 BE][payload]. Only
108 * kXR_asynresp carries a reply — envelope [actnum 4][reserved 4][inner
109 * ServerResponseHdr 8][data] — which is unwrapped and dispatched by its INNER
110 * streamid (the outer one is not a request key: asyncms uses {0,0}, and a
111 * deferring server need not mirror the deferred sid). Everything else (asyncms
112 * text, obsolete actions, truncated envelopes) is informational at most: drop
113 * it — it must never complete or fail an in-flight request. */
114static void
115aconn_handle_attn(brix_aconn *ac, const uint8_t *body, uint32_t dlen)
116{
117 uint16_t esid, estat;
118 uint32_t edlen;
119
120 if (dlen < 4 || xrd_get_u32_be(body) != (uint32_t) kXR_asynresp) {
121 return; /* not a deferred reply (e.g. asyncms notice) — ignore */
122 }
123 if (dlen < 16) {
124 return; /* truncated asynresp envelope: no inner header to trust */
125 }
126 xrd_resp_hdr_unpack(body + 8, &esid, &estat, &edlen);
127 if (edlen > dlen - 16) {
128 edlen = dlen - 16; /* never read past the outer frame */
129 }
130 aconn_dispatch_response(ac, esid, estat, body + 16, edlen);
131}

134/* Dispatch one fully-received frame: unsolicited kXR_attn pushes are routed
135 * BEFORE any in-flight lookup (their outer streamid is not a reliable request
136 * key); everything else is a direct response matched by streamid. */
137void
138aconn_dispatch_frame(brix_aconn *ac, uint16_t sid, uint16_t stat,
139 const uint8_t *body, uint32_t dlen)
140{
141 ac->last_activity_ns = brix_mono_ns(); /* we heard from the server */
142
143 if (stat == kXR_attn) {
144 aconn_handle_attn(ac, body, dlen);
145 return;
146 }
147 aconn_dispatch_response(ac, sid, stat, body, dlen);
148}

151/* ---- Complete a frame whose body was received straight into a caller buffer ----
152 *
153 * WHAT: Finishes the frame identified by ac->rx_sid / ac->rx_stat once all of
154 * its body bytes have landed in the request's dst, then clears the
155 * direct-receive state. A kXR_oksofar frame just leaves the request
156 * in-flight for its next frame; a kXR_ok completes it.
157 *
158 * WHY: The body bytes were consumed outside aconn_dispatch_response (that is
159 * the whole point — they were never buffered), so the terminal half of
160 * that dispatch has to be reachable on its own. Keeping it here, next to
161 * the dispatch it completes, keeps the two halves of the split frame
162 * readable as one thing.
163 *
164 * HOW: 1. Clear rx_direct/rx_need first so any error path below sees a clean
165 * connection state.
166 * 2. kXR_oksofar → return (more frames follow for the same request).
167 * 3. kXR_ok → note the RTT sample, drop the in-flight entry and complete.
168 */
169void
170aconn_direct_frame_done(brix_aconn *ac)
171{
172 brix_areq *r = ac->rx_direct;
173 uint16_t sid = ac->rx_sid;
174 uint16_t stat = ac->rx_stat;
175
176 ac->rx_direct = NULL;
177 ac->rx_need = 0;
178 if (stat == kXR_oksofar) {
179 return;
180 }
181 if (!r->deferred) {
182 aconn_note_rtt(ac, r);
183 }
184 reqmap_del(&ac->inflight, sid);
185 areq_complete(r, 0, stat, NULL);
186}

189/* ---- Can this frame's body be received straight into the caller's buffer? ----
190 *
191 * WHAT: Returns the in-flight request when the frame (sid, stat) carries read
192 * data for a request that supplied a landing buffer big enough for dlen,
193 * and NULL otherwise.
194 *
195 * WHY: Direct receive is only safe for the data-bearing statuses of a request
196 * that asked for it. Error/redirect/wait/attn bodies are small control
197 * payloads that the ordinary buffered dispatch decodes.
198 *
199 * HOW: Look the streamid up; require r->dst and a non-empty data frame.
200 */
201brix_areq *
202aconn_direct_target(brix_aconn *ac, uint16_t sid, uint16_t stat, uint32_t dlen)
203{
204 brix_areq *r;
205
206 if (stat != kXR_ok && stat != kXR_oksofar) {
207 return NULL;
208 }
209 r = reqmap_get(&ac->inflight, sid);
210 if (r == NULL || r->dst == NULL || dlen == 0) {
211 return NULL;
212 }
213 return r;
214}

217/* ---- Start receiving one frame's body straight into the caller's buffer ----
218 *
219 * WHAT: Consumes the frame header from rbuf, copies whatever of its body is
220 * already buffered into the request's dst, and either finishes the frame
221 * (everything was buffered) or arms the direct-receive state for the rest.
222 * Returns 0 when the parse loop may continue with the next frame, 1 when
223 * it must stop (the remainder comes off the socket), and -1 when the frame
224 * would overrun the caller's buffer (the connection is failed here).
225 *
226 * WHY: A bulk read reply is far larger than anything that can arrive in one
227 * read(2), so the tail of it should land in the caller's buffer instead of
228 * being staged in rbuf and copied again. Handling the already-buffered
229 * head here is what makes the switch seamless at any frame boundary. An
230 * over-long body is refused BEFORE a byte is written, because from here on
231 * the bytes go straight into memory the caller owns.
232 *
233 * HOW: 1. Refuse dlen beyond the buffer's remaining capacity.
234 * 2. Advance past the 8-byte header.
235 * 3. Copy min(available, dlen) into dst via areq_accumulate.
236 * 4. All of it? finish the frame and continue parsing. Otherwise record
237 * rx_direct/rx_need/rx_sid/rx_stat and stop the loop (rbuf is now
238 * empty, so nothing is stranded behind the direct read).
239 */
240int
241aconn_direct_begin(brix_aconn *ac, brix_areq *r, uint16_t sid, uint16_t stat,
242 uint32_t dlen)
243{
244 size_t avail;
245 uint32_t take;
246
247 if (dlen > r->dst_cap - r->acc_len) {
248 brix_status st;
249 brix_status_set(&st, XRDC_EPROTO, 0,
250 "reply body %u exceeds the caller buffer", dlen);
251 aconn_on_transport_error(ac, &st);
252 return -1;
253 }
254 ac->rbuf.start += XRD_RESPONSE_HDR_LEN;
255 avail = ac->rbuf.len - ac->rbuf.start;
256 take = (avail < (size_t) dlen) ? (uint32_t) avail : dlen;
257 (void) areq_accumulate(r, ac->rbuf.buf + ac->rbuf.start, take);
258 ac->rbuf.start += take;
259
260 ac->rx_direct = r;
261 ac->rx_need = dlen - take;
262 ac->rx_sid = sid;
263 ac->rx_stat = stat;
264 if (ac->rx_need == 0) {
265 aconn_direct_frame_done(ac);
266 return 0;
267 }
268 return 1;
269}
