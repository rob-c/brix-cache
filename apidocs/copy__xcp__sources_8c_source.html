17/* ---- Effective block size ----
18 *
19 * WHAT: BRIX_XCP_BLOCK bytes when set (clamped to [64 KiB, 64 MiB]), else the
20 * shared 8 MiB copy chunk.
21 *
22 * WHY: Tests and tuning need block granularity without recompiling; the clamp
23 * keeps a hostile/typo value from degenerating the block table.
24 *
25 * HOW: getenv + strtoull + clamp.
26 */
27size_t
28xcp_block_size(void)
29{
30 const char *env = getenv("BRIX_XCP_BLOCK");
31 unsigned long long v;
32
33 if (env == NULL || env[0] == '\0') {
34 return XRDC_COPY_CHUNK;
35 }
36 v = strtoull(env, NULL, 10);
37 if (v < XRDC_XCP_BLOCK_MIN) { v = XRDC_XCP_BLOCK_MIN; }
38 if (v > XRDC_XCP_BLOCK_MAX) { v = XRDC_XCP_BLOCK_MAX; }
39 return (size_t) v;
40}

43/* ---- Compose the canonical URL for a parsed root-family endpoint ----
44 *
45 * WHAT: "<scheme>://host:port/<abs-path>" (the double-slash spelling, since
46 * the path keeps its leading '/'). 0 / -1 when it does not fit.
47 *
48 * WHY: The locate fallback and the single-source duplication both need a
49 * dialable URL string for endpoints we only hold in parsed form.
50 *
51 * HOW: Scheme from u->scheme (TLS keeps roots://); IPv6 hosts re-bracket.
52 */
53static int
54xcp_compose_url(const brix_url *u, char *out, size_t outsz)
55{
56 const char *scheme = (u->scheme == XRDC_SCHEME_ROOTS) ? "roots" : "root";
57 int v6 = (strchr(u->host, ':') != NULL);
58
59 return ((size_t) snprintf(out, outsz, "%s://%s%s%s:%d/%s",
60 scheme, v6 ? "[" : "", u->host, v6 ? "]" : "",
61 u->port, u->path) >= outsz) ? -1 : 0;
62}

65/* ---- Replica URLs from the metalink mirror list ----
66 *
67 * WHAT: Copy up to `want` root-family mirror URLs (already ranked best-first
68 * by the parser) into the worker slots. Returns the count.
69 *
70 * WHY: A metalink with N mirrors IS the replica list — no discovery needed.
71 *
72 * HOW: The resolver already filtered to root-family; bound-copy each.
73 */
74static size_t
75xcp_sources_from_mirrors(const download_job_t *job, xcp_worker_t *w,
76 size_t want)
77{
78 const brix_copy_opts *o = job->o;
79 size_t n = 0, i;
80
81 for (i = 0; i < o->xcp_n_mirrors && n < want; i++) {
82 const char *m = o->xcp_mirrors[i];
83 if (m == NULL || strlen(m) >= sizeof(w[n].url)) {
84 continue;
85 }
86 snprintf(w[n].url, sizeof(w[n].url), "%s", m);
87 n++;
88 }

93/* ---- Replica URLs from a kXR_locate on the control connection ----
94 *
95 * WHAT: Ask the (already connected) source endpoint where the file lives and
96 * turn each data-server token into a replica URL. Returns the count.
97 *
98 * WHY: Without a metalink, a clustered source (CMS manager) can still name
99 * multiple holders — upstream XCp does exactly this discovery.
100 *
101 * HOW: brix_locate → whitespace-split tokens shaped "S<r|w><host>:<port>"
102 * ('S'/'s' server entries; manager 'M'/'m' entries are skipped — dialing
103 * a manager would just re-locate). Bracketed IPv6 hosts pass through
104 * verbatim inside the composed URL.
105 */
106static size_t
107xcp_sources_from_locate(const download_job_t *job, xcp_worker_t *w,
108 size_t want)
109{
110 char reply[4096], *cursor, *token, *save = NULL;
111 const char *scheme =
112 (job->su->scheme == XRDC_SCHEME_ROOTS) ? "roots" : "root";
113 brix_status lst;
114 size_t n = 0;
115
116 brix_status_clear(&lst);
117 if (brix_locate(job->c, job->su->path, reply, sizeof(reply), &lst) != 0) {
118 return 0; /* no locate support / error: caller falls back */
119 }
120 cursor = reply;
121 while (n < want && (token = strtok_r(cursor, " \t\r\n", &save)) != NULL) {
122 cursor = NULL;
123 if ((token[0] != 'S' && token[0] != 's') || token[1] == '\0'
124 || token[2] == '\0') {
125 continue;
126 }
127 if ((size_t) snprintf(w[n].url, sizeof(w[n].url), "%s://%s/%s",
128 scheme, token + 2, job->su->path)
129 >= sizeof(w[n].url)) {
130 continue;
131 }
132 n++;
133 }
134 return n;
135}

138/* ---- Build the replica list for this transfer ----
139 *
140 * WHAT: Fill worker URL slots from, in preference order: metalink mirrors,
141 * locate discovery, then the single source duplicated up to `want`.
142 * Returns the worker count (0 only when even the source URL cannot be
143 * composed).
144 *
145 * WHY: One place owns the "where can these bytes come from" policy and its
146 * documented divergence: with only one known replica the source URL is
147 * duplicated, because parallel TCP streams to one host still help on
148 * high-latency links and it keeps --sources honest on single servers.
149 *
150 * HOW: Providers in order until >= 2 replicas exist; a single replica is
151 * duplicated to the asked width; >= 2 distinct replicas are used as-is.
152 */
153size_t
154xcp_build_sources(const download_job_t *job, xcp_worker_t *w, size_t want)
155{
156 size_t n = xcp_sources_from_mirrors(job, w, want);
157
158 if (n < 2) {
159 n += xcp_sources_from_locate(job, w + n, want - n);
160 }
161 if (n == 0) {
162 if (xcp_compose_url(job->su, w[0].url, sizeof(w[0].url)) != 0) {
163 return 0;
164 }
165 n = 1;
166 }
167 if (n == 1) { /* one known replica: duplicate it up to the asked width */
168 while (n < want) {
169 snprintf(w[n].url, sizeof(w[n].url), "%s", w[0].url);
170 n++;
171 }
172 }
173 return n;
174}
