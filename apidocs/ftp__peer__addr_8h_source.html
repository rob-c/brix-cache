1/*
2 * ftp_peer_addr.h — canonical GridFTP transport-peer address comparison.
3 *
4 * WHAT: normalizes IPv4, IPv6, and IPv4-mapped IPv6 socket addresses and
5 * compares their host identity while deliberately ignoring transport ports.
6 *
7 * WHY: passive GridFTP data sockets inherit the authenticated control
8 * session's filesystem authority. A clear data socket therefore has to come
9 * from the same transport peer; accepting the first connector lets a racing
10 * host steal a download or inject an upload. One predicate shared by MODE S
11 * and MODE E prevents representation differences from splitting that boundary.
12 *
13 * HOW: each supported sockaddr is copied into a fixed canonical form. Mapped
14 * IPv6 addresses become their four-byte IPv4 address; native IPv6 retains its
15 * scope identifier so equal link-local bytes on different interfaces do not
16 * alias. Unknown families, short structures, and NULL inputs fail closed.
17 */
18

34/* ---- Canonicalize one transport-peer address ----
35 *
36 * WHAT: Writes a port-independent canonical address to `out` and returns 1;
37 * returns 0 for malformed, unsupported, or absent input.
38 *
39 * WHY: IPv4 and IPv4-mapped IPv6 are two wire representations of the same
40 * host and must make the same security decision, while truncated sockaddrs
41 * must never be interpreted from uninitialized bytes.
42 *
43 * HOW: 1. Validate pointers and the family-specific structure length.
44 * 2. Copy IPv4 bytes directly or fold a mapped IPv6 address to IPv4.
45 * 3. Preserve all native IPv6 bytes and its scope identifier.
46 */
47static inline int
48brix_ftp_peer_addr_canonical(const struct sockaddr *address,
49 socklen_t address_len, brix_ftp_peer_addr_t *out)
50{
51 const struct sockaddr_in *address4;
52 const struct sockaddr_in6 *address6;
53
54 if (address == NULL || out == NULL
55 || address_len < (socklen_t) sizeof(struct sockaddr))
56 {
57 return 0;
58 }
59
60 memset(out, 0, sizeof(*out));
61 if (address->sa_family == AF_INET) {
62 if (address_len < (socklen_t) sizeof(*address4)) {
63 return 0;
64 }
65 address4 = (const struct sockaddr_in *) address;
66 out->family = AF_INET;
67 memcpy(out->bytes, &address4->sin_addr, sizeof(address4->sin_addr));
68 return 1;
69 }
70
71 if (address->sa_family != AF_INET6
72 || address_len < (socklen_t) sizeof(*address6))
73 {
74 return 0;
75 }
76
77 address6 = (const struct sockaddr_in6 *) address;
78 if (IN6_IS_ADDR_V4MAPPED(&address6->sin6_addr)) {
79 out->family = AF_INET;
80 memcpy(out->bytes, &address6->sin6_addr.s6_addr[12],
81 sizeof(struct in_addr));
82 return 1;
83 }
84
85 out->family = AF_INET6;
86 memcpy(out->bytes, &address6->sin6_addr, sizeof(address6->sin6_addr));
87 out->scope_id = address6->sin6_scope_id;
88 return 1;
89}

92/* ---- Compare two transport peers without their ephemeral ports ----
93 *
94 * WHAT: Returns 1 only when both inputs are valid supported socket addresses
95 * naming the same canonical host; otherwise returns 0.
96 *
97 * WHY: a passive data connection uses a new port by design, so port equality
98 * would reject every legitimate transfer, while raw family/byte comparison
99 * would let mapped-address spelling create inconsistent admission decisions.
100 *
101 * HOW: 1. Canonicalize both inputs with the fail-closed helper above.
102 * 2. Require the same canonical family and IPv6 scope.
103 * 3. Compare exactly four IPv4 bytes or sixteen IPv6 bytes.
104 */
105static inline int
106brix_ftp_peer_addr_equal(const struct sockaddr *left, socklen_t left_len,
107 const struct sockaddr *right, socklen_t right_len)
108{
109 brix_ftp_peer_addr_t left_address;
110 brix_ftp_peer_addr_t right_address;
111 size_t byte_count;
112
113 if (!brix_ftp_peer_addr_canonical(left, left_len, &left_address)
114 || !brix_ftp_peer_addr_canonical(right, right_len, &right_address))
115 {
116 return 0;
117 }
118 if (left_address.family != right_address.family
119 || left_address.scope_id != right_address.scope_id)
120 {
121 return 0;
122 }
123
124 byte_count = left_address.family == AF_INET
125 ? sizeof(struct in_addr) : sizeof(struct in6_addr);
126 return memcmp(left_address.bytes, right_address.bytes, byte_count) == 0;
127}
