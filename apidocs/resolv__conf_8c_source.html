1/*
2 * resolv_conf.c — resolv.conf(5) parser (phase-116 W1).
3 *
4 * WHAT: The implementation behind resolv_conf.h: defaults, a line tokenizer,
5 * keyword handlers for nameserver/search/domain/options, the env
6 * overrides, and a bounded file loader.
7 * WHY: Must behave like the host resolver (glibc) so a hostname that works
8 * for `getent hosts` on the box works the same way inside brix: last
9 * search/domain line wins, options accumulate, unknown tokens are
10 * ignored, and hard limits are silently enforced.
11 * HOW: No allocation — every value lands in the fixed arrays of
12 * brix_resolv_conf_t; the loader reads at most 64 KiB (resolv.conf is a
13 * few lines; a bigger file is not one). Pure C so the unittest builds it
14 * without nginx.

62/* Copy a bounded token into a NUL-terminated buffer; 0 when it does not fit. */
63static int
64resolv_copy_token(char *dst, size_t dstsz, const char *tok, size_t len)
65{
66 if (len == 0 || len >= dstsz) {
67 return 0;
68 }
69 memcpy(dst, tok, len);
70 dst[len] = '\0';
71 return 1;

75/* The digits after "host:": 1..65535 and nothing else. */
76static int
77resolv_port_ok(const char *p, size_t len)
78{
79 unsigned v = 0;
80 size_t i;
81
82 if (len == 0 || len > 5) {
83 return 0;
84 }
85 for (i = 0; i < len; i++) {
86 if (p[i] < '0' || p[i] > '9') {
87 return 0;
88 }
89 v = v * 10 + (unsigned) (p[i] - '0');
90 }
91 return v >= 1 && v <= BRIX_DNS_PORT_MAX;
92}

95/* inet_pton() on a bounded host token (zone already stripped). */
96static int
97resolv_is_literal(const char *host, size_t len, int family)
98{
99 char buf[BRIX_RESOLV_NS_LEN];
100 unsigned char addr[16];
101
102 if (len == 0 || len >= sizeof(buf)) {
103 return 0;
104 }
105 memcpy(buf, host, len);
106 buf[len] = '\0';
107 return inet_pton(family, buf, addr) == 1;

111/* "[v6]" or "[v6]:port" (zone inside the brackets dropped) -> stored as
112 * "v6" or "[v6]:port". */
113static int
114resolv_bracketed_ns(const char *tok, size_t len, char *dst, size_t dstsz)
115{
116 const char *close = memchr(tok, ']', len);
117 const char *pct;
118 size_t inner, rest;
119 int n;
120
121 if (close == NULL) {
122 return 0;
123 }
124 inner = (size_t) (close - tok - 1);
125 pct = memchr(tok + 1, '%', inner);
126 if (pct != NULL) {
127 inner = (size_t) (pct - (tok + 1));
128 }
129 if (!resolv_is_literal(tok + 1, inner, AF_INET6)) {
130 return 0;
131 }
132 rest = len - (size_t) (close + 1 - tok);
133 if (rest == 0) {
134 return resolv_copy_token(dst, dstsz, tok + 1, inner);
135 }
136 if (close[1] != ':' || !resolv_port_ok(close + 2, rest - 1)) {
137 return 0;
138 }
139 n = snprintf(dst, dstsz, "[%.*s]%.*s", (int) inner, tok + 1,
140 (int) rest, close + 1);
141 return n > 0 && (size_t) n < dstsz;
142}

145/* "v4", "v4:port" or a bare "v6" (zone dropped) -> stored as written. */
146static int
147resolv_plain_ns(const char *tok, size_t len, char *dst, size_t dstsz)
148{
149 const char *pct = memchr(tok, '%', len);
150 const char *colon, *second;
151 size_t after;
152
153 if (pct != NULL) {
154 len = (size_t) (pct - tok);
155 }
156 colon = memchr(tok, ':', len);
157 if (colon == NULL) {
158 return resolv_is_literal(tok, len, AF_INET)
159 && resolv_copy_token(dst, dstsz, tok, len);
160 }
161 after = len - (size_t) (colon + 1 - tok);
162 second = memchr(colon + 1, ':', after);
163 if (second == NULL) {
164 return resolv_is_literal(tok, (size_t) (colon - tok), AF_INET)
165 && resolv_port_ok(colon + 1, after)
166 && resolv_copy_token(dst, dstsz, tok, len);
167 }
168 return resolv_is_literal(tok, len, AF_INET6)
169 && resolv_copy_token(dst, dstsz, tok, len);
170}

173/* WHAT: nameserver <addr>[:port] — the first nameserver line replaces the
174 * 127.0.0.1 default; only address literals are kept (v4, v4:port, v6,
175 * [v6], [v6]:port), an IPv6 zone ("%eth0") is dropped because nginx's
176 * resolver cannot bind a scope id, and ":port" is the brix extension
177 * that lets an unprivileged stub serve the tests.
178 * WHY: glibc ignores a nameserver line it cannot parse as an address. If
179 * the token were passed through, nginx's `resolver` would look the name
180 * up at configuration time — the blocking startup DNS dependency
181 * phase 116 exists to remove. A file whose nameserver lines are all
182 * unparseable therefore ends with nnameservers == 0, which the builder
183 * reports and turns into the libc thread-pool fallback. */
184static void
185resolv_kw_nameserver(brix_resolv_conf_t *rc, const char *tok, size_t len,
186 unsigned *seen_ns)
187{
188 char *dst;
189 int kept;
190
191 if (!*seen_ns) {
192 rc->nnameservers = 0;
193 *seen_ns = 1;
194 }
195 if (rc->nnameservers >= BRIX_RESOLV_MAX_NS) {
196 return;
197 }
198 dst = rc->nameservers[rc->nnameservers];
199 kept = (len > 0 && tok[0] == '[')
200 ? resolv_bracketed_ns(tok, len, dst, BRIX_RESOLV_NS_LEN)
201 : resolv_plain_ns(tok, len, dst, BRIX_RESOLV_NS_LEN);
202 if (kept) {
203 rc->nnameservers++;
204 }
205}

208/* search/domain: the LAST such line wins (glibc), so reset then append. */
209static void
210resolv_kw_search(brix_resolv_conf_t *rc, const char *rest, size_t len)
211{
212 size_t i = 0;
213
214 rc->nsearch = 0;
215 while (i < len && rc->nsearch < BRIX_RESOLV_MAX_SEARCH) {
216 size_t start, tlen;
217
218 while (i < len && (rest[i] == ' ' || rest[i] == '\t')) {
219 i++;
220 }
221 start = i;
222 while (i < len && rest[i] != ' ' && rest[i] != '\t') {
223 i++;
224 }
225 tlen = i - start;
226 if (tlen > 0 && rest[start] != '.'
227 && resolv_copy_token(rc->search[rc->nsearch],
228 BRIX_RESOLV_DOMAIN_LEN, rest + start, tlen))
229 {
230 /* a trailing dot on a search domain is dropped (glibc) */
231 size_t n = strlen(rc->search[rc->nsearch]);
232 if (rc->search[rc->nsearch][n - 1] == '.') {
233 rc->search[rc->nsearch][n - 1] = '\0';
234 }
235 rc->nsearch++;
236 }
237 }
238}

252/* One "options" token: ndots:N, timeout:N, attempts:N, rotate; anything
253 * else (edns0, single-request, ...) is ignored. */
254static void
255resolv_option(brix_resolv_conf_t *rc, const char *tok, size_t len)
256{
257 const char *colon = memchr(tok, ':', len);
258 size_t nlen = colon ? (size_t) (colon - tok) : len;
259 unsigned val = colon ? (unsigned) strtoul(colon + 1, NULL, 10) : 0;
260
261 if (colon == NULL) {
262 if (nlen == 6 && memcmp(tok, "rotate", 6) == 0) {
263 rc->rotate = 1;
264 }
265 return;
266 }
267 if (nlen == 5 && memcmp(tok, "ndots", 5) == 0) {
268 rc->ndots = val > RESOLV_NDOTS_MAX ? RESOLV_NDOTS_MAX : val;
269 } else if (nlen == 7 && memcmp(tok, "timeout", 7) == 0) {
270 rc->timeout = resolv_clamp(val, RESOLV_TIMEOUT_MAX);
271 } else if (nlen == 8 && memcmp(tok, "attempts", 8) == 0) {
272 rc->attempts = resolv_clamp(val, RESOLV_ATTEMPTS_MAX);
273 }
274}

299/* Keyword tokens are matched by (length, text); a table keeps resolv_line a
300 * flat walk instead of a chain of memcmp branches. */
301typedef enum { RESOLV_KW_NONE, RESOLV_KW_NAMESERVER, RESOLV_KW_SEARCH,
302 RESOLV_KW_OPTIONS } resolv_kw_t;
303
304static resolv_kw_t
305resolv_keyword(const char *kw, size_t len)
306{
307 static const struct { const char *name; size_t len; resolv_kw_t kw; } tab[] = {
308 { "nameserver", 10, RESOLV_KW_NAMESERVER },
309 { "search", 6, RESOLV_KW_SEARCH },
310 { "domain", 6, RESOLV_KW_SEARCH },
311 { "options", 7, RESOLV_KW_OPTIONS },
312 };
313 size_t i;
314
315 for (i = 0; i < sizeof(tab) / sizeof(tab[0]); i++) {
316 if (tab[i].len == len && memcmp(kw, tab[i].name, len) == 0) {
317 return tab[i].kw;
318 }
319 }
320 return RESOLV_KW_NONE;
321}

331/* Split " keyword rest " into its keyword and its trimmed remainder.
332 * Returns 0 when the line carries no keyword or no argument. */
333static int
334resolv_split_line(const char *line, size_t len, const char **kw, size_t *kw_len,
335 const char **rest, size_t *rest_len)
336{
337 size_t i = 0, start;
338
339 while (i < len && resolv_is_blank(line[i])) {
340 i++;
341 }
342 start = i;
343 while (i < len && !resolv_is_blank(line[i])) {
344 i++;
345 }
346 *kw = line + start;
347 *kw_len = i - start;
348 while (i < len && resolv_is_blank(line[i])) {
349 i++;
350 }
351 *rest = line + i;
352 *rest_len = len - i;
353 while (*rest_len > 0
354 && (resolv_is_blank((*rest)[*rest_len - 1])
355 || (*rest)[*rest_len - 1] == '\r'))
356 {
357 (*rest_len)--;
358 }
359 return *kw_len != 0 && *rest_len != 0;
360}

363/* One line of resolv.conf: "keyword arg..." — comments and blank lines were
364 * dropped by the caller. sortlist / lookup / unknown keywords are ignored,
365 * as glibc does. */
366static void
367resolv_line(brix_resolv_conf_t *rc, const char *line, size_t len,
368 unsigned *seen_ns)
369{
370 const char *kw, *rest;
371 size_t kw_len, rest_len, tlen = 0;
372
373 if (!resolv_split_line(line, len, &kw, &kw_len, &rest, &rest_len)) {
374 return;
375 }
376 switch (resolv_keyword(kw, kw_len)) {
377 case RESOLV_KW_NAMESERVER:
378 while (tlen < rest_len && !resolv_is_blank(rest[tlen])) {
379 tlen++;
380 }
381 resolv_kw_nameserver(rc, rest, tlen, seen_ns);
382 break;
383 case RESOLV_KW_SEARCH:
384 resolv_kw_search(rc, rest, rest_len);
385 break;
386 case RESOLV_KW_OPTIONS:
387 resolv_kw_options(rc, rest, rest_len);
388 break;
389 case RESOLV_KW_NONE:
390 break;
391 }
392}

395void
396brix_resolv_conf_parse_text(brix_resolv_conf_t *rc, const char *text,
397 size_t len)
398{
399 size_t i = 0;
400 unsigned seen_ns = 0;
401
402 while (i < len) {
403 size_t start = i, llen;
404 const char *hash;
405
406 while (i < len && text[i] != '\n') {
407 i++;
408 }
409 llen = i - start;
410 if (i < len) {
411 i++; /* skip the newline */
412 }
413 /* comments start with '#' or ';' anywhere on the line */
414 hash = memchr(text + start, '#', llen);
415 if (hash != NULL) {
416 llen = (size_t) (hash - (text + start));
417 }
418 hash = memchr(text + start, ';', llen);
419 if (hash != NULL) {
420 llen = (size_t) (hash - (text + start));
421 }
422 resolv_line(rc, text + start, llen, &seen_ns);
423 }
424}

441int
442brix_resolv_conf_load(brix_resolv_conf_t *rc, const char *path)
443{
444 FILE *fp;
445 char *buf;
446 size_t n;
447
448 brix_resolv_conf_defaults(rc);
449 if (path == NULL || path[0] == '\0') {
450 path = BRIX_RESOLV_DEFAULT_PATH;
451 }
452
453 fp = fopen(path, "r");
454 if (fp == NULL) {
455 brix_resolv_conf_apply_env(rc, getenv("LOCALDOMAIN"),
456 getenv("RES_OPTIONS"));
457 return -1;
458 }
459 buf = malloc(RESOLV_READ_MAX);
460 if (buf == NULL) {
461 fclose(fp);
462 brix_resolv_conf_apply_env(rc, getenv("LOCALDOMAIN"),
463 getenv("RES_OPTIONS"));
464 return -1;
465 }
466 n = fread(buf, 1, RESOLV_READ_MAX, fp);
467 if (n == 0 && ferror(fp)) {
468 /* The path opened but holds no readable content — a directory (fopen
469 * succeeds, the first read is EISDIR) or an I/O error. That is the
470 * operator's own path being unusable, not an empty resolv.conf, so
471 * report it like an unopenable file instead of silently keeping the
472 * 127.0.0.1 default. A genuinely empty file reads 0 bytes with no
473 * error and still defaults, as glibc does.
474 */
475 fclose(fp);
476 free(buf);
477 brix_resolv_conf_apply_env(rc, getenv("LOCALDOMAIN"),
478 getenv("RES_OPTIONS"));
479 return -1;
480 }
481 fclose(fp);
482
483 rc->defaulted = 0;
484 brix_resolv_conf_parse_text(rc, buf, n);
485 free(buf);
486 brix_resolv_conf_apply_env(rc, getenv("LOCALDOMAIN"),
487 getenv("RES_OPTIONS"));
488 return 0;
489}
