10static char *
11gftp_adat_argument(const char *text)
12{
13 const char *start = strstr(text, "ADAT=");
14 const char *end;
15 char *copy;
16 size_t len;
17
18 if (start == NULL) {
19 return NULL;
20 }
21 start += sizeof("ADAT=") - 1;
22 for (end = start; *end != '\0' && *end != ' ' && *end != '\r'
23 && *end != '\n'; end++) {
24 /* scan token */
25 }
26 len = (size_t) (end - start);
27 if (len == 0) {
28 return NULL;
29 }
30 copy = malloc(len + 1);
31 if (copy != NULL) {
32 memcpy(copy, start, len);
33 copy[len] = '\0';
34 }
35 return copy;
36}

58static int
59gftp_adat_send(gftp_session_t *session, const uint8_t *token, size_t token_len)
60{
61 char *encoded = gftp_base64_encode(token, token_len);
62 char *line;
63 size_t encoded_len;
64 int rc;
65
66 if (encoded == NULL) {
67 gftp_set_error(session, ENOMEM, "cannot encode GridFTP ADAT token");
68 return -1;
69 }
70 encoded_len = strlen(encoded);
71 if (encoded_len + sizeof("ADAT \r\n") > GFTP_ADAT_LINE_CAP) {
72 free(encoded);
73 gftp_set_error(session, EOVERFLOW, "GridFTP ADAT token exceeds limit");
74 return -1;
75 }
76 line = malloc(encoded_len + sizeof("ADAT \r\n"));
77 if (line == NULL) {
78 free(encoded);
79 gftp_set_error(session, ENOMEM, "cannot allocate GridFTP ADAT command");
80 return -1;
81 }
82 memcpy(line, "ADAT ", sizeof("ADAT ") - 1);
83 memcpy(line + sizeof("ADAT ") - 1, encoded, encoded_len);
84 memcpy(line + sizeof("ADAT ") - 1 + encoded_len, "\r\n",
85 sizeof("\r\n"));
86 free(encoded);
87 rc = gftp_socket_write_all(session, session->fd, line, encoded_len + 7);
88 free(line);
89 if (rc == 0) {
90 rc = gftp_read_reply(session);
91 }
92 return rc;
93}

95static int
96gftp_gsi_loop(gftp_session_t *session)
97{
98 uint8_t *output = NULL;
99 size_t output_len = 0;
100 int round;
101 int more;
102
103 more = gftp_gsi_step(session->gsi, NULL, 0, &output, &output_len, session);
104 if (more < 0) {
105 return -1;
106 }
107 for (round = 0; round < 16; round++) {
108 uint8_t *input = NULL;
109 size_t input_len = 0;
110
111 if (output_len == 0 || gftp_adat_send(session, output, output_len) != 0) {
112 free(output);
113 return -1;
114 }
115 free(output);
116 output = NULL;
117 output_len = 0;
118 if (session->code == 235) {
119 session->secure = 1;
120 return 0;
121 }
122 if (session->code != 335
123 || gftp_adat_input(session, &input, &input_len) != 0) {
124 free(input);
125 gftp_set_error(session, EACCES, "GridFTP GSI authentication refused");
126 return -1;
127 }
128 more = gftp_gsi_step(session->gsi, input, input_len,
129 &output, &output_len, session);
130 free(input);
131 if (more < 0) {
132 free(output);
133 return -1;
134 }
135 }
136 free(output);
137 gftp_set_error(session, ELOOP, "GridFTP GSI handshake did not converge");
138 return -1;
139}

141static int
142gftp_user_pass(gftp_session_t *session, const char *user, const char *password)
143{
144 if (gftp_command(session, "USER %s", user) != 0) {
145 return -1;
146 }
147 if (session->code >= 300 && session->code < 400
148 && gftp_command(session, "PASS %s", password) != 0) {
149 return -1;
150 }
151 if (session->code < 200 || session->code > 299) {
152 gftp_set_error(session, EACCES, "GridFTP identity is not authorized");
153 return -1;
154 }

159gftp_gsi_login(gftp_session_t *session, const gftp_session_cfg_t *cfg)
160{
161 if (cfg->proxy_path == NULL || cfg->proxy_path[0] == '\0') {

163 return -1;

176int
177gftp_authenticate(gftp_session_t *session, const gftp_session_cfg_t *cfg)
178{
179 if (cfg->require_gsi) {
180 if (gftp_gsi_login(session, cfg) != 0) {
181 return -1;
182 }
183 /* The data leg remains clear but is bound to the authenticated control
184 * peer. A PROT-P engine can replace these policy commands later without
185 * changing the storage-driver API. */
186 (void) gftp_command(session, "PBSZ 0");
187 (void) gftp_command(session, "PROT C");
188 (void) gftp_command(session, "DCAU N");
189 } else if (gftp_user_pass(session, "anonymous", "brix@") != 0) {
190 return -1;
191 }
