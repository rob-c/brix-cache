7/* File: src/core/aio/uring_probe.c — io_uring detection, gating, and the §32
8 * startup fail-fast validator (phase-79 split from uring.c).
9 * WHAT: Owns the "should this build/host/worker use io_uring at all" decisions:
10 * the memoized runtime opcode probe, the per-block config scan folded into
11 * one per-worker verdict, and the config-time hard-fail for `brix_io_uring
12 * on`. The ring singleton and its bring-up live in uring.c; the CQE

58/*
59 * brix_uring_runtime_available — authoritative, memoized opcode probe.
60 *
61 * Stands up a throwaway 8-entry ring, asks io_uring_get_probe whether every
62 * required server data opcode (READ/WRITE/READV/WRITEV/FSYNC) is supported,
63 * tears it down, and caches the verdict for the lifetime of the process. Never
64 * parses a kernel version: containers/seccomp routinely block io_uring_setup
65 * even on new kernels, so only an actual setup+probe is trustworthy.
66 *
67 * Returns 1 iff a ring can be created and all required opcodes are supported.
68 */
69ngx_int_t
70brix_uring_runtime_available(void)
71{
72 static const int required[] = {
73 IORING_OP_READ, IORING_OP_WRITE, IORING_OP_READV,
74 IORING_OP_WRITEV, IORING_OP_FSYNC
75 };
76 static ngx_int_t cached = -1; /* -1 = not yet probed */
77 struct io_uring ring;
78 struct io_uring_probe *probe;
79 ngx_int_t ok = 1;
80 ngx_uint_t i;
81
82 if (cached != -1) {
83 return cached;
84 }
85
86 if (io_uring_queue_init(8, &ring, 0) < 0) {
87 cached = 0; /* seccomp / no syscall / old kernel */
88 return 0;
89 }
90
91 probe = io_uring_get_probe_ring(&ring);
92 if (probe == NULL) {
93 io_uring_queue_exit(&ring);
94 cached = 0;
95 return 0;
96 }
97
98 for (i = 0; i < sizeof(required) / sizeof(required[0]); i++) {
99 if (!io_uring_opcode_supported(probe, required[i])) {
100 ok = 0;
101 break;
102 }
103 }
104
105 io_uring_free_probe(probe);
106 io_uring_queue_exit(&ring);
107
108 cached = ok;
109 return cached;
110}

112/*
113 * uring_scan_conf — fold every enabled server block's io_uring settings.
114 *
115 * WHAT: Scans the stream server blocks and merges their io_uring demand into
116 * *scan: whether any block wants the ring, whether any requires it
117 * (`on`), the max queue depth, the restriction preference, and the
118 * first configured panic-file path.
119 * WHY: The ring is a per-worker singleton, so per-block settings must be
120 * folded into one verdict before bring-up.
121 * HOW: Linear scan skipping disabled/off blocks; max for depth, AND for
122 * restrictions, first-wins for the panic file.
123 *
124 * Returns 1 if at least one enabled block wants io_uring, 0 otherwise.
125 */
126static ngx_int_t
127uring_scan_conf(ngx_cycle_t *cycle, brix_uring_scan_t *scan)
128{
129 ngx_stream_core_main_conf_t *cmcf;
130 ngx_stream_core_srv_conf_t **cscfp;
131 ngx_stream_brix_srv_conf_t *xcf;
132 ngx_uint_t i;
133 ngx_int_t want = 0;
134
135 cmcf = ngx_stream_cycle_get_module_main_conf(cycle, ngx_stream_core_module);
136 if (cmcf == NULL) {
137 return 0;
138 }
139
140 cscfp = cmcf->servers.elts;
141 for (i = 0; i < cmcf->servers.nelts; i++) {
142 xcf = ngx_stream_conf_get_module_srv_conf(cscfp[i],
143 ngx_stream_brix_module);
144 if (!xcf->common.enable || xcf->io_uring == BRIX_IO_URING_OFF) {
145 continue;
146 }
147 want = 1;
148 if (xcf->io_uring == BRIX_IO_URING_ON) {
149 scan->mode_on = 1;
150 }
151 if (xcf->io_uring_queue_depth > scan->depth) {
152 scan->depth = xcf->io_uring_queue_depth;
153 }
154 if (!xcf->io_uring_restrict) {
155 scan->want_restrict = 0;
156 }
157 if (scan->panic_file.len == 0 && xcf->io_uring_panic_file.len > 0) {
158 scan->panic_file = xcf->io_uring_panic_file;
159 }
160 }
161
162 return want;
163}

165/*
166 * uring_probe_features — decide whether this worker should bring a ring up.
167 *
168 * WHAT: Folds the configuration (uring_scan_conf) and the kernel capability
169 * probe into one go/no-go verdict, clamping the queue depth.
170 * WHY: Both "nobody wants io_uring" and "auto blocks on a host without it"
171 * are silent thread-pool outcomes that must be decided BEFORE any ring
172 * resource exists — keeping the post-creation fallback decision point
173 * (brix_uring_init_fail) single.
174 * HOW: Early-returns NGX_DECLINED for either skip reason (logging the
175 * `auto`-unavailable NOTICE), otherwise clamps depth into [8,4096] and
176 * returns NGX_OK to proceed with bring-up. Non-static: also called from
177 * brix_uring_init_worker() in uring.c.
178 */
179ngx_int_t
180uring_probe_features(ngx_cycle_t *cycle, brix_uring_scan_t *scan)
181{
182 scan->mode_on = 0;
183 scan->want_restrict = 1;
184 scan->depth = BRIX_IO_URING_QUEUE_DEPTH;
185 ngx_str_null(&scan->panic_file);
186
187 if (!uring_scan_conf(cycle, scan)) {
188 return NGX_DECLINED; /* every enabled block has io_uring off */
189 }
190
191 /* AUTO blocks need the per-process probe (seccomp-accurate); ON blocks
192 * already passed it at config time but re-checking is cheap + memoized. */
193 if (!scan->mode_on && !brix_uring_runtime_available()) {
194 BRIX_LOG(NGX_LOG_NOTICE, cycle->log, 0,
195 "brix: io_uring unavailable on this host; using the thread pool");
196 return NGX_DECLINED;
197 }
198
199 if (scan->depth < 8) { scan->depth = 8; }
200 if (scan->depth > BRIX_URING_MAX_DEPTH) { scan->depth = BRIX_URING_MAX_DEPTH; }
201
202 return NGX_OK;
203}

215/*
216 * brix_uring_validate_conf — §32 startup fail-fast (ADR-16).
217 *
218 * If any enabled server block requests `brix_io_uring on`, io_uring MUST be
219 * provided or startup fails (NGX_ERROR -> nginx -t exits non-zero, master
220 * refuses to start). Two independent gates:
221 * (1) compile-time: a stub build can never satisfy `on` — caught with no
222 * kernel needed, so CI / `nginx -t` flags it anywhere;
223 * (2) runtime: a liburing build whose probe fails (old kernel / seccomp).
224 * `off` and `auto` always pass (auto degrades silently per worker).
225 *
226 * Returns NGX_OK to allow startup, NGX_ERROR to abort it.
227 */
228ngx_int_t
229brix_uring_validate_conf(ngx_conf_t *cf)
230{
231 if (!brix_uring_any_block_on(cf)) {
232 return NGX_OK;
233 }
234
235#if !(BRIX_HAVE_LIBURING)
236 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
237 "\"brix_io_uring on\" requires a build with liburing, but this binary "
238 "was compiled WITHOUT it. Rebuild with BRIX_ENABLE_IO_URING=1 and "
239 "liburing-devel installed, or set \"brix_io_uring auto\" to allow "
240 "silent fallback to the thread pool.");
241 return NGX_ERROR;
242#else
243 if (!brix_uring_runtime_available()) {
244 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
245 "\"brix_io_uring on\" requested but io_uring is unavailable on "
246 "this host (io_uring_setup/opcode probe failed). This is typically a "
247 "seccomp policy (Docker/containerd default profiles block io_uring) "
248 "or a kernel older than %d.%d. Set \"brix_io_uring auto\" to fall "
249 "back to the thread pool, or enable io_uring at the host/container "
250 "level.",
251 BRIX_IO_URING_MIN_KERNEL_MAJOR, BRIX_IO_URING_MIN_KERNEL_MINOR);
252 return NGX_ERROR;
253 }
254 return NGX_OK;
255#endif
256}
