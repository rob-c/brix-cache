48/* Compute the kXR_open options word for a (re)open with the given force tri-state.
49 *
50 * WHAT: Returns the wire options bitmask — kXR_open_read for a read handle, or
51 * kXR_open_updt|kXR_mkpath plus (force==1) kXR_delete / (force==0) kXR_new /
52 * (force==2) nothing, plus kXR_posc when POSC is set — for a write handle.
53 * WHY: The force tri-state (truncate vs create-excl vs update-in-place) is exactly
54 * what makes reopen NON-destructive: reopen passes force==2 so recovery never
55 * re-truncates or re-creates. Isolating it keeps mfile_do_open flat.
56 * HOW: 1) read handle → kXR_open_read, done.
57 * 2) write handle → base updt|mkpath; OR in delete/new per force (force==2
58 * adds neither = update in place); OR in posc when requested. */
59static uint16_t
60mfile_open_options(const brix_mfile *mf, int force)
61{
62 uint16_t options;
63
64 if (!mf->writable) {
65 return kXR_open_read;
66 }
67 options = kXR_open_updt | kXR_mkpath;
68 if (force == 1) {
69 options |= kXR_delete; /* truncate/overwrite */
70 } else if (force == 0) {
71 options |= kXR_new; /* create-excl */
72 } /* force == 2: update in place */
73 if (mf->posc) {
74 options |= kXR_posc;
75 }
76 return options;
77}

79/* Learn the inline-compression codec from a kXR_open reply body.
80 *
81 * WHAT: Resets mf->read_codec/write_codec to 0, then (write handle → write_codec,
82 * read handle → read_codec) adopts the server's codec id when the reply
83 * confirms it. Returns 0 on success (including "no codec"), or -1 with *st
84 * set when the server negotiated a codec this build cannot decode.
85 * WHY: phase-42 W4/W5. Per the dual-check contract (codec_core.h) cptype is a
86 * legacy field a non-cooperating server may reuse; trusting it alone would
87 * inflate plaintext and corrupt data, so it is adopted ONLY alongside the
88 * cpsize magic. Failing an un-decodable codec avoids asymmetric-build
89 * corruption (copying compressed bytes through as plaintext). Re-learned on
90 * every (re)open, so a fault mid-transfer transparently re-negotiates.
91 * HOW: 1) clear both codecs (stock servers leave cpsize 0 → stays plaintext).
92 * 2) ServerOpenBody is fhandle[4] cpsize[4] cptype[4]; a body under 12 bytes
93 * carries no codec fields → return 0.
94 * 3) parse big-endian cpsize and cptype[0]; unless cpsize == the magic and
95 * the id is in [1, BRIX_CODEC_MAX) → return 0 (plaintext).
96 * 4) if this build cannot decode the id → status + return -1.
97 * 5) store it in the write- or read-side codec per mf->writable, return 0. */
98static int
99mfile_learn_codec(brix_mfile *mf, const uint8_t *body, uint32_t blen,
100 brix_status *st)
101{
102 mf->read_codec = 0;
103 mf->write_codec = 0;
104 if (blen < 12) {
105 return 0;
106 }
107 uint32_t cpsize = ((uint32_t) body[4] << 24) | ((uint32_t) body[5] << 16)
108 | ((uint32_t) body[6] << 8) | (uint32_t) body[7];
109 uint8_t cid = body[8]; /* cptype[0] */
110 if (cpsize != BRIX_INLINE_CMP_MAGIC || cid < 1 || cid >= BRIX_CODEC_MAX) {
111 return 0;
112 }
113 /* Server confirmed codec `cid`; if this build cannot handle it, fail the open
114 * rather than silently copying compressed bytes as plaintext. */
115 if (!brix_codec_available(cid)) {
116 brix_status_set(st, XRDC_EUNSUPPORTED, 0,
117 "server negotiated inline-compression codec %u that this "
118 "client build cannot decode", (unsigned) cid);
119 return -1;
120 }
121 if (mf->writable) {
122 mf->write_codec = cid;
123 } else {
124 mf->read_codec = cid;
125 }
126 return 0;
127}

129/* Build + send a kXR_open with the given force tri-state (see brix_file_open_opaque).
130 * On success copies the fresh handle into mf->fhandle. Caller holds no lock. */
131static int
132mfile_do_open(brix_mfile *mf, int force, brix_status *st)
133{
134 ClientOpenRequest req;
135 uint16_t options = mfile_open_options(mf, force);
136 uint16_t kxr = 0;
137 uint8_t *body = NULL;
138 uint32_t blen = 0;
139 char *payload;
140 size_t need;
141 int plen;
142
143 need = strlen(mf->path) + 1 + (mf->have_opaque ? strlen(mf->opaque) + 1 : 0);
144 payload = (char *) malloc(need);
145 if (payload == NULL) {
146 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (open payload)");
147 return -1;
148 }
149 plen = mf->have_opaque
150 ? snprintf(payload, need, "%s?%s", mf->path, mf->opaque)
151 : snprintf(payload, need, "%s", mf->path);
152
153 memset(&req, 0, sizeof(req));
154 req.requestid = htons(kXR_open);
155 {
156 xrdw_open_req_t b = { .mode = (uint16_t) (mf->writable ? 0644 : 0),
157 .options = options, .optiont = 0 };
158 xrdw_open_req_pack(&b, ((ClientRequestHdr *) &req)->body);
159 }
160
161 brix_aio_opts o = { mf->max_stall_ms, mf->max_retries, 1 /*open is idempotent*/ };
162 int rc = brix_aio_call_ex(mf->ac, &req, payload, (uint32_t) plen, &o,
163 &kxr, &body, &blen, st);
164 free(payload);
165 if (rc != 0) {
166 return -1;
167 }
168 if (blen < XRD_FHANDLE_LEN) {
169 brix_status_set(st, XRDC_EPROTO, 0, "open reply too short (%u bytes)", blen);
170 free(body);
171 return -1;
172 }
173 memcpy(mf->fhandle, body, XRD_FHANDLE_LEN);
174
175 /* phase-42 W4/W5: learn the inline-compression codec from the open reply.
176 * Re-learned on every (re)open, so a fault mid-transfer re-negotiates. */
177 rc = mfile_learn_codec(mf, body, blen, st);
178 free(body);
179 return rc;
180}

182brix_mfile *
183brix_mfile_open(brix_aconn *ac, const char *path, int writable, int force,
184 int posc, const char *opaque, int max_stall_ms, int max_retries,
185 brix_status *st)
186{
187 brix_mfile *mf = (brix_mfile *) calloc(1, sizeof(*mf));
188 if (mf == NULL) {
189 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (mfile)");
190 return NULL;
191 }
192 mf->ac = ac;
193 snprintf(mf->path, sizeof(mf->path), "%s", path);
194 if (opaque != NULL && opaque[0] != '\0') {
195 snprintf(mf->opaque, sizeof(mf->opaque), "%s", opaque);
196 mf->have_opaque = 1;
197 }
198 mf->writable = writable;
199 mf->posc = posc;
200 mf->max_stall_ms = max_stall_ms;
201 mf->max_retries = max_retries;
202 pthread_mutex_init(&mf->lock, NULL);
203
204 if (mfile_do_open(mf, force, st) != 0) {
205 pthread_mutex_destroy(&mf->lock);
206 free(mf);
207 return NULL;
208 }
209 mf->have_handle = 1;
210 mf->gen = 1;
211 return mf;
212}

214/* A failure that a reopen + retry might fix: a transient transport fault, or a
215 * stale handle after the session was re-established. */
216static int
217mfile_should_reopen(const brix_status *st)
218{
219 return brix_status_retryable(st) || st->kxr == kXR_FileNotOpen;
220}

222/* Reopen the file NON-destructively if no one else already did (generation match).
223 * Returns 0 if the file is open (by us or a racing caller), -1 on failure. */
224static int
225mfile_reopen(brix_mfile *mf, uint64_t expected_gen, brix_status *st)
226{
227 pthread_mutex_lock(&mf->lock);
228 if (mf->gen != expected_gen && mf->have_handle) {
229 pthread_mutex_unlock(&mf->lock);
230 return 0; /* someone else reopened while we waited */
231 }
232 mf->have_handle = 0;
233 int rc = mfile_do_open(mf, mf->writable ? 2 /*update, no truncate*/ : -1, st);
234 if (rc == 0) {
235 mf->have_handle = 1;
236 mf->gen++;
237 }
238 pthread_mutex_unlock(&mf->lock);
239 return rc;
240}

242/* Snapshot the current handle + generation (+ optional read/write codecs) under
243 * the lock. rcodec/wcodec may be NULL for callers that don't need them. */
244static int
245mfile_snapshot(brix_mfile *mf, uint8_t fh[XRD_FHANDLE_LEN], uint64_t *gen,
246 uint8_t *rcodec, uint8_t *wcodec)
247{
248 pthread_mutex_lock(&mf->lock);
249 int ok = mf->have_handle;
250 if (ok) {
251 memcpy(fh, mf->fhandle, XRD_FHANDLE_LEN);
252 }
253 if (rcodec != NULL) {
254 *rcodec = mf->read_codec;
255 }
256 if (wcodec != NULL) {
257 *wcodec = mf->write_codec;
258 }
259 *gen = mf->gen;
260 pthread_mutex_unlock(&mf->lock);
261 return ok;
262}

264ssize_t
265brix_mfile_pread(brix_mfile *mf, int64_t off, void *buf, size_t len,
266 brix_status *st)
267{
268 /* Retry transport faults until the max_stall patience window elapses, with
269 * exponential backoff between attempts — NOT a fixed small count. Under
270 * sustained loss (a flapping firewall) a fixed count is exhausted in a tight
271 * spin long before the patience window, surfacing a spurious EIO; bounding by
272 * the deadline rides the loss out as long as progress is possible. */
273 uint64_t deadline = now_ns() + (uint64_t) mf->max_stall_ms * 1000000ULL;
274 unsigned attempt = 0;
275
276 for (;;) {
277 uint8_t fh[XRD_FHANDLE_LEN];
278 uint64_t gen;
279 uint8_t codec = 0;
280 if (!mfile_snapshot(mf, fh, &gen, &codec, NULL)) {
281 if (mfile_reopen(mf, gen, st) != 0) {
282 if (now_ns() >= deadline) {
283 return -1;
284 }
285 brix_backoff_sleep_fast(attempt++);
286 }
287 continue;
288 }
289
290 ClientReadRequest req;
291 memset(&req, 0, sizeof(req));
292 req.requestid = htons(kXR_read);
293 {
294 xrdw_read_req_t b = { .offset = off, .rlen = (int32_t) len };
295 memcpy(b.fhandle, fh, XRD_FHANDLE_LEN);
296 xrdw_read_req_pack(&b, ((ClientRequestHdr *) &req)->body);
297 }
298
299 uint16_t kxr = 0;
300 uint8_t *body = NULL;
301 uint32_t blen = 0;
302 brix_aio_opts o = { mf->max_stall_ms, 0, 0 /*we own reopen/retry*/ };
303 if (brix_aio_call_ex(mf->ac, &req, NULL, 0, &o, &kxr, &body, &blen, st) == 0) {
304 /* phase-42 W4: a compressed handle returns one self-contained codec
305 * frame per request (aio_call_ex has already accumulated all
306 * oksofar frames into `body`); inflate it into the caller's buffer.
307 * A corrupt frame is not a transport fault — do not reopen/retry. */
308 if (codec != 0) {
309 ssize_t pl = brix_inflate_frame(codec, body, blen, buf, len, st);
310 free(body);
311 return pl;
312 }
313 size_t n = (blen < len) ? blen : len;
314 if (n > 0) {
315 memcpy(buf, body, n);
316 }
317 free(body);
318 return (ssize_t) n;
319 }
320 if (!mfile_should_reopen(st) || now_ns() >= deadline) {
321 return -1;
322 }
323 brix_backoff_sleep_fast(attempt++); /* don't tight-spin under loss */
324 (void) mfile_reopen(mf, gen, st); /* refresh handle, then retry */
325 }
326}

328int
329brix_mfile_pwrite(brix_mfile *mf, int64_t off, const void *buf, size_t len,
330 brix_status *st)
331{
332 /* Deadline-bounded retry with backoff (see brix_mfile_pread). */
333 uint64_t deadline = now_ns() + (uint64_t) mf->max_stall_ms * 1000000ULL;
334 unsigned attempt = 0;
335
336 for (;;) {
337 uint8_t fh[XRD_FHANDLE_LEN];
338 uint64_t gen;
339 uint8_t wcodec = 0;
340 if (!mfile_snapshot(mf, fh, &gen, NULL, &wcodec)) {
341 if (mfile_reopen(mf, gen, st) != 0) {
342 if (now_ns() >= deadline) {
343 return -1;
344 }
345 brix_backoff_sleep_fast(attempt++);
346 }
347 continue;
348 }
349
350 /* phase-42 W5: a compression-negotiated write handle compresses each
351 * payload as a self-contained frame; the server decompresses on ingest.
352 * The request offset stays the PLAINTEXT offset. */
353 const void *payload = buf;
354 size_t plen = len;
355 uint8_t *frame = NULL;
356 if (wcodec != 0 && len > 0) {
357 size_t flen = 0;
358 frame = brix_deflate_frame(wcodec, buf, len, &flen, st);
359 if (frame == NULL) {
360 return -1; /* compress failure is not a transport fault */
361 }
362 payload = frame;
363 plen = flen;
364 }
365
366 ClientWriteRequest req;
367 memset(&req, 0, sizeof(req));
368 req.requestid = htons(kXR_write);
369 {
370 xrdw_write_req_t b = { .offset = off, .pathid = 0 };
371 memcpy(b.fhandle, fh, XRD_FHANDLE_LEN);
372 xrdw_write_req_pack(&b, ((ClientRequestHdr *) &req)->body);
373 }
374
375 uint16_t kxr = 0;
376 brix_aio_opts o = { mf->max_stall_ms, 0, 0 };
377 int rc = brix_aio_call_ex(mf->ac, &req, payload, (uint32_t) plen, &o,
378 &kxr, NULL, NULL, st);
379 free(frame);
380 if (rc == 0) {
381 return 0;
382 }
383 if (!mfile_should_reopen(st) || now_ns() >= deadline) {
384 return -1;
385 }
386 brix_backoff_sleep_fast(attempt++);
387 (void) mfile_reopen(mf, gen, st);
388 }
389}

391int
392brix_mfile_sync(brix_mfile *mf, brix_status *st)
393{
394 uint8_t fh[XRD_FHANDLE_LEN] = { 0 };
395 uint64_t gen;
396 if (!mfile_snapshot(mf, fh, &gen, NULL, NULL)) {
397 if (mfile_reopen(mf, gen, st) != 0) {
398 return -1;
399 }
400 (void) mfile_snapshot(mf, fh, &gen, NULL, NULL);
401 }
402 ClientSyncRequest req;
403 memset(&req, 0, sizeof(req));
404 req.requestid = htons(kXR_sync);
405 {
406 xrdw_sync_req_t b;
407 memcpy(b.fhandle, fh, XRD_FHANDLE_LEN);
408 xrdw_sync_req_pack(&b, ((ClientRequestHdr *) &req)->body);
409 }
410 uint16_t kxr = 0;
411 /* A sync after a reopen is meaningless (fresh handle), but re-issuing is
412 * harmless; do not loop — a failed sync is reported, not retried forever. */
413 return brix_aio_call(mf->ac, &req, NULL, 0, &kxr, NULL, NULL,
414 mf->max_stall_ms, st);
415}
