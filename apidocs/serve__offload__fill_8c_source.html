1/*
2 * serve_offload_fill.c — fetch the window an off-loop serve will send, not the
3 * object it lives in.
4 *
5 * WHAT: given an open remote object and the request's raw Range header, copy
6 * into a scratch fd exactly the bytes the response will carry.
7 *
8 * WHY: the off-loop serve (http_serve_offload.c) materialises a remote object
9 * into a scratch temp file and then serves the response out of that file. It
10 * materialised the WHOLE object, always — so a 256-byte Range on a 384 kB
11 * object fetched all 384 kB across the network, and the same request against a
12 * 10 GB object would have fetched 10 GB. Nothing looked wrong from outside:
13 * the right bytes came back with the right Content-Range.
14 *
15 * That cost was invisible for as long as the offload was reached only by
16 * `xroot`. Phase-115 §P6 made every blocking-wire driver declare itself, which
17 * correctly routed `gsiftp` here too — and gsiftp is the driver whose bounded
18 * retrieve (GridFTP `ERET P <off> <len>`) is VISIBLE in the origin's command
19 * log. The origin was then asked for `ERET P 0 384000` to serve a 256-byte
20 * window, which is the same defect phase-115 §P4 had already fixed one layer up
21 * in brix_vfs_file_sendfile_fd_window(): that layer narrowed the read and this
22 * one widened it straight back, silently cancelling every bounded-retrieve
23 * extension the drivers negotiate.
24 *
25 * HOW: the window is derived with the SAME parser the serve itself will use
26 * (brix_http_parse_range, via brix_http_serve_file_ranged) against the SAME
27 * size, so the window fetched and the window sent cannot disagree — they are
28 * one computation run twice. It is then written into the scratch fd at its
29 * TRUE offset over a hole of the full object size, so nothing downstream has to
30 * know a narrowing happened: the serve re-derives the window and reads exactly
31 * these bytes. The hole costs no disk and is never read for this request.
32 *
33 * A header that is absent, unparseable, unsatisfiable, or too large for the
34 * caller's buffer yields the whole object. This may only ever narrow what is
35 * FETCHED, never narrow what can be SENT.
36 */

49/* The window [off, off+len) of a `size`-byte object that a response carrying
50 * this Range header will send. */
51static void
52serve_fill_window(const u_char *hdr, size_t hdr_len, off_t size,
53 xvfs_window_t *win)
54{
55 brix_http_range_t rng;
56
57 win->off = 0;
58 win->len = size;
59 win->end = 0;
60
61 if (hdr == NULL || hdr_len == 0) {
62 return;
63 }
64 brix_http_parse_range(hdr, hdr_len, size, &rng);
65 if (!rng.present || !rng.satisfiable) {
66 /* Unsatisfiable is deliberately NOT narrowed to nothing: the serve
67 * answers 416 without reading the scratch, but a parser that ever
68 * disagreed about satisfiability would then be serving a hole. */
69 return;
70 }
71 win->off = rng.start;
72 win->len = rng.end - rng.start + 1;
73}

85/* Size the scratch to the whole object, so a narrowed window lands at its true
86 * offset. Skipped for a whole-object copy, which fills every byte itself and
87 * must keep reporting the size it actually managed to read. */
88static int
89serve_fill_hole(brix_sd_obj_t *dst, const xvfs_window_t *win, off_t size)
90{
91 if (win->len >= size) {
92 return 0;
93 }
94 return (xvfs_ftruncate(dst, size) == 0) ? 0 : (errno ? errno : EIO);
95}

97int
98brix_serve_offload_fill(brix_sd_obj_t *obj, int tmp_fd,
99 const u_char *range_hdr, size_t range_len,
100 brix_serve_offload_fill_t *out)
101{
102 brix_sd_obj_t dst; /* worker-owned scratch, driver-routed */
103 brix_sd_stat_t snap;
104 xvfs_window_t win;
105 u_char *buf;
106 int err;
107
108 serve_fill_snap(obj, &snap);
109 out->mtime = snap.mtime;
110 out->size = snap.size;
111
112 buf = malloc(BRIX_SERVE_OFFLOAD_FILL_CHUNK);
113 if (buf == NULL) {
114 return ENOMEM;
115 }
116 brix_sd_posix_wrap(&dst, tmp_fd);
117 serve_fill_window(range_hdr, range_len, snap.size, &win);
118
119 err = serve_fill_hole(&dst, &win, snap.size);
120 if (err == 0
121 && xvfs_drain_window(obj, &dst, buf, BRIX_SERVE_OFFLOAD_FILL_CHUNK,
122 &win) != 0)
123 {
124 err = errno ? errno : EIO;
125 }
126 free(buf);
127
128 /* A whole-object copy reports what it READ, not what the origin claimed:
129 * an origin whose size is a lie must not make the response promise bytes
130 * that are not in the scratch. A narrowed copy reports the object size,
131 * because that is what its Content-Range denominator has to be — and its
132 * window is all-or-nothing, so there is nothing to disagree about. */
133 if (err == 0 && win.len >= snap.size) {
134 out->size = win.end;
135 }
136 return err;
137}
