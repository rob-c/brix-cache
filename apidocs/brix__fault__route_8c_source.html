1/*
2 * brix_fault_route.c — named routes: independent listener→upstream mappings (C2).
3 *
4 * WHAT: a table of named routes, each a (listener port → own target pool) with
5 * its own connection/byte counters. `route add <name> <port> <host:port
6 * [,host:port...]>` starts a new listener; `route del <name>` stops it;
7 * `route list [json]` reports every route and its counters. The default
8 * route is the startup listener, always present, never deletable.
9 *
10 * WHY: one process can then stand in for a handful of endpoints during a test,
11 * and traffic is attributable per endpoint — neither of which the single
12 * global listener can do. The global fault levers and toxics still apply
13 * uniformly, so a scenario is configured once and every route feels it.
14 *
15 * HOW: a fixed-size table under one mutex. A route slot, once bound to a name,
16 * keeps that identity for the process lifetime: `del` stops the listener
17 * (closing it and joining the accept thread) but leaves the struct — and
18 * thus its counters — valid, so in-flight relay threads that still hold the
19 * route pointer can finish writing counters with no use-after-free, and a
20 * later `add` of the same name simply restarts the same slot. Binding and
21 * the accept/relay engine live in the core, reached only through the
22 * registered fp_route_ops, so a dynamic route reuses the one vetted
23 * loopback-gated bind address (only the port differs) and cannot widen the
24 * unauthenticated control plane's exposure.
25 */

128/* Parse a comma-separated host:port list into `route` (replacing its pool).
129 * Returns 0 on success, -1 on any malformed entry or overflow. */
130static int
131parse_targets(fp_route *route, const char *spec)
132{
133 char tmp[512];
134 snprintf(tmp, sizeof tmp, "%s", spec);
135 route->ntargets = 0;
136 for (char *tok = strtok(tmp, ","); tok != NULL; tok = strtok(NULL, ",")) {
137 const char *colon = strrchr(tok, ':');
138 if (colon == NULL || colon == tok || colon[1] == '\0') {
139 return -1;
140 }
141 size_t hlen = (size_t) (colon - tok);
142 if (hlen >= sizeof route->targets[0].host) {
143 return -1;
144 }
145 int p = atoi(colon + 1);
146 if (p <= 0 || p > 65535) {
147 return -1;
148 }
149 char host[256];
150 memcpy(host, tok, hlen);
151 host[hlen] = '\0';
152 if (fp_route_add_target(route, host, p) != 0) {
153 return -1;
154 }
155 }
156 return route->ntargets > 0 ? 0 : -1;
157}

159/* Accept-thread trampoline: run the core accept loop, then mark not listening. */
160struct accept_arg { fp_route *route; int lfd; };
161
162static void *
163route_accept_trampoline(void *arg)
164{
165 struct accept_arg *a = arg;
166 fp_route *route = a->route;
167 int lfd = a->lfd;
168 free(a);
169 g_ops.accept_loop(route, lfd); /* returns once route->alive == 0 */
170 close(lfd);
171 __atomic_store_n(&route->listening, 0, __ATOMIC_RELEASE);
172 return NULL;
173}

175/* Bind a listener and spawn its accept thread for `route`. Caller holds the
176 * lock. Returns 0 on success, -1 on bind/thread failure. */
177static int
178start_listener(fp_route *route, char *reply, size_t rsz)
179{
180 int lfd = g_ops.bind_listen(route->listen_port);
181 if (lfd < 0) {
182 /* The bind address is the one vetted loopback template — only the port
183 * varies — so a bind failure is a port collision in all realistic cases. */
184 fp_reply(reply, rsz, "err: port in use: cannot bind %d\n",
185 route->listen_port);
186 return -1;
187 }
188 struct accept_arg *a = malloc(sizeof *a);
189 if (a == NULL) {
190 close(lfd);
191 fp_reply(reply, rsz, "err: out of memory\n");
192 return -1;
193 }
194 route->alive = 1;
195 route->listen_fd = lfd;
196 a->route = route;
197 a->lfd = lfd;
198 if (pthread_create(&route->accept_tid, NULL, route_accept_trampoline, a) != 0) {
199 close(lfd);
200 free(a);
201 route->listen_fd = -1;
202 fp_reply(reply, rsz, "err: cannot start accept thread\n");
203 return -1;
204 }
205 __atomic_store_n(&route->listening, 1, __ATOMIC_RELEASE);
206 return 0;
207}

209/* route add <name> <port> <host:port[,host:port...]> */
210static int
211route_add(char *rest, char *reply, size_t rsz)
212{
213 char name[FP_ROUTE_NAME] = "", targets[400] = "";
214 int port = 0;
215 if (sscanf(rest, "%31s %d %399s", name, &port, targets) != 3) {
216 fp_reply(reply, rsz, "err: usage: route add <name> <port> <host:port,...>\n");
217 return 1;
218 }
219 if (port <= 0 || port > 65535) {
220 fp_reply(reply, rsz, "err: bad port\n");
221 return 1;
222 }
223 if (strcmp(name, "default") == 0) {
224 fp_reply(reply, rsz, "err: 'default' is reserved\n");
225 return 1;
226 }
227
228 pthread_mutex_lock(&g_route_lock);
229 fp_route *r = find_by_name(name);
230 int reused = r != NULL;
231 if (r == NULL) {
232 r = alloc_slot();
233 if (r == NULL) {
234 pthread_mutex_unlock(&g_route_lock);
235 fp_reply(reply, rsz, "err: route table full\n");
236 return 1;
237 }
238 } else if (__atomic_load_n(&r->listening, __ATOMIC_ACQUIRE)) {
239 pthread_mutex_unlock(&g_route_lock);
240 fp_reply(reply, rsz, "err: route '%s' already exists\n", name);
241 return 1;
242 }
243
244 /* Populate/refresh identity + pool (counters persist across a restart). */
245 if (!reused) {
246 snprintf(r->name, sizeof r->name, "%s", name);
247 r->conns = r->up_bytes = r->down_bytes = 0;
248 r->rr = 0;
249 }
250 r->listen_port = port;
251 r->is_default = 0;
252 r->listen_fd = -1;
253 if (parse_targets(r, targets) != 0) {
254 if (!reused) {
255 memset(r, 0, sizeof *r); /* release a freshly-taken slot */
256 }
257 pthread_mutex_unlock(&g_route_lock);
258 fp_reply(reply, rsz, "err: bad target list\n");
259 return 1;
260 }
261 r->in_use = 1;
262
263 if (start_listener(r, reply, rsz) != 0) {
264 if (!reused) {
265 memset(r, 0, sizeof *r);
266 }
267 pthread_mutex_unlock(&g_route_lock);
268 return 1; /* start_listener wrote the reply */
269 }
270 pthread_mutex_unlock(&g_route_lock);
271 fp_reply(reply, rsz, "ok: route %s listening on port %d\n", name, port);
272 return 1;
273}

275/* route del <name> — stop the listener; keep the (now idle) slot + counters. */
276static int
277route_del(char *rest, char *reply, size_t rsz)
278{
279 char name[FP_ROUTE_NAME] = "";
280 if (sscanf(rest, "%31s", name) != 1) {
281 fp_reply(reply, rsz, "err: usage: route del <name>\n");
282 return 1;
283 }
284 if (strcmp(name, "default") == 0) {
285 fp_reply(reply, rsz, "err: cannot delete default route\n");
286 return 1;
287 }
288
289 pthread_mutex_lock(&g_route_lock);
290 fp_route *r = find_by_name(name);
291 if (r == NULL || !__atomic_load_n(&r->listening, __ATOMIC_ACQUIRE)) {
292 pthread_mutex_unlock(&g_route_lock);
293 fp_reply(reply, rsz, "err: no such route\n");
294 return 1;
295 }
296 pthread_t tid = r->accept_tid;
297 __atomic_store_n(&r->alive, 0, __ATOMIC_RELEASE); /* accept loop will return */
298 pthread_mutex_unlock(&g_route_lock);
299
300 pthread_join(tid, NULL); /* wait for the accept loop + fd close */
301 fp_reply(reply, rsz, "ok: route %s removed\n", name);
302 return 1;
303}

314static void
315route_emit_json(char *reply, size_t rsz)
316{
317 size_t o = (size_t) snprintf(reply, rsz, "{\"routes\":[");
318 int first = 1;
319
320 for (int i = 0; i < FP_ROUTE_MAX && o < rsz; i++) {
321 fp_route *r = &g_routes[i];
322 if (!route_reportable(r)) {
323 continue;
324 }
325 o += (size_t) snprintf(reply + o, rsz - o,
326 "%s{\"name\":\"%s\",\"port\":%d,\"targets\":%d,"
327 "\"conns\":%lu,\"up_bytes\":%lu,\"down_bytes\":%lu}",
328 first ? "" : ",", r->name, r->listen_port, r->ntargets,
329 r->conns, r->up_bytes, r->down_bytes);
330 first = 0;
331 }
332 if (o < rsz) {
333 snprintf(reply + o, rsz - o, "]}\n");
334 }
335}
