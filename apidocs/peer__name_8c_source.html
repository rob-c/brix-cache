1/*
2 * peer_name.c — accept-time wait for the peer's reverse-DNS name (phase-116).
3 *
4 * WHAT: When this listener's policy consults the peer's hostname (XrdAcc `h`
5 * rules via brix_acc_resolve_hosts, a protbind host template, or
6 * `brix_auth host`), look the PTR up ONCE at accept, off the event loop,
7 * and only then start the protocol pump. Listeners that never consult
8 * the name pay three flag tests.
9 * WHY: The consumers (auth gate, protbind, host auth) run in the middle of
10 * kXR_login/kXR_auth handling, where they can only probe a cache — the
11 * blocking getnameinfo() they used to call stalled the worker (the
12 * Phase 51 breaker merely bounded it). Waiting at accept keeps every
13 * later brix_acc_resolve_peer() probe a hit. The wait is bounded by the
14 * resolver's own timeout*attempts; a peer that connects and says nothing
15 * is bounded the same way it always was once the pump arms its timers.
16 * HOW: brix_conn_peer_name_wait(): a cached probe first (hit → pump now);
17 * otherwise a brix_dns_rev_req_t on c->pool with a pool cleanup that
18 * cancels an in-flight lookup, idle event handlers so a stray read event
19 * cannot re-enter the stream content phase while nothing is armed, and a
20 * completion handler that calls the pump. The pump is passed in so
21 * handler.c keeps ownership of its static conn_pump().
22 */

38static ngx_flag_t
39conn_peer_name_needed(const ngx_stream_brix_srv_conf_t *sconf)
40{
41 return sconf->common.acc.resolve_hosts > 0
42 || sconf->auth == BRIX_AUTH_HOST
43 || brix_protbind_needs_hostname(sconf->protbind)
44 /* 2.0 F18: `brix_tpc_allow_identity host <pattern>` is a fourth
45 * host-template consumer. Without the wait its rule would be
46 * evaluated against a name that had not landed yet and deny every
47 * TPC — a confinement control must fail closed, but not by accident
48 * of timing. */
49 || brix_tpc_matrix_needs_hostname(sconf->common.tpc_allow_identity);
50}

85ngx_int_t
86brix_conn_peer_name_wait(ngx_stream_session_t *s, ngx_connection_t *c,
87 brix_conn_pump_pt pump)
88{
89 ngx_stream_brix_srv_conf_t *sconf;
90 conn_peer_name_wait_t *w;
91 ngx_pool_cleanup_t *cln;
92 char name[BRIX_DNS_REVERSE_NAME_LEN];
93
94 sconf = ngx_stream_get_module_srv_conf(s, ngx_stream_brix_module);
95 if (!conn_peer_name_needed(sconf) || c->sockaddr == NULL
96 || c->socklen > sizeof(struct sockaddr_storage)
97 || brix_dns_reverse_cached(c->sockaddr, c->socklen, name,
98 sizeof(name)) != NGX_AGAIN)
99 {
100 return NGX_OK;
101 }
102
103 w = ngx_pcalloc(c->pool, sizeof(*w));
104 cln = (w != NULL) ? ngx_pool_cleanup_add(c->pool, 0) : NULL;
105 if (cln == NULL) {
106 return NGX_OK; /* no memory: decide numerically */
107 }
108 cln->handler = conn_peer_name_cleanup;
109 cln->data = w;
110
111 ngx_memcpy(&w->req.ss, c->sockaddr, c->socklen);
112 w->req.len = c->socklen;
113 w->req.policy = sconf->common.dns.policy;
114 w->req.log = c->log;
115 w->req.handler = conn_peer_name_done;
116 w->req.data = w;
117 w->c = c;
118 w->pump = pump;
119
120 c->read->handler = conn_peer_name_idle;
121 c->write->handler = conn_peer_name_idle;
122
123 w->inline_call = 1;
124 if (brix_dns_reverse(&w->req) != NGX_OK) {
125 w->done = 1;
126 return NGX_OK; /* not started: decide numerically */
127 }
128 w->inline_call = 0;
129
130 return w->done ? NGX_OK : NGX_AGAIN;
131}
