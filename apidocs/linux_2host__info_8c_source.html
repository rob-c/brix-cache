40#if BRIX_ARCH_ARM64

44 /* x86-64 user mappings occupy the lower canonical 47-bit range. */

71static void
72sysload_net(brix_plat_sysload_t *sample)
73{
74 struct ifaddrs *all;
75 struct ifaddrs *at;
76
77 if (getifaddrs(&all) != 0) return;
78 for (at = all; at != NULL; at = at->ifa_next) {
79 struct rtnl_link_stats *stats;
80
81 if (at->ifa_addr == NULL || at->ifa_data == NULL
82 || at->ifa_addr->sa_family != AF_PACKET
83 || (at->ifa_flags & IFF_LOOPBACK) != 0) continue;
84 stats = at->ifa_data;
85 sample->net_bytes += (uint64_t) stats->rx_bytes + stats->tx_bytes;
86 sample->have |= BRIX_SYSLOAD_HAVE_NET;
87 }
88 freeifaddrs(all);
89}

91static void
92sysload_paging(brix_plat_sysload_t *sample)
93{
94 char key[64];
95 unsigned long long value;
96 FILE *input = fopen("/proc/vmstat", "r");
97
98 if (input == NULL) return;
99 while (fscanf(input, "%63s %llu", key, &value) == 2) {
100 if (strcmp(key, "pgmajfault") == 0) {
101 sample->paging_events = (uint64_t) value;
102 sample->have |= BRIX_SYSLOAD_HAVE_PAGING;
103 break;
104 }
105 }
106 (void) fclose(input);
107}

109int
110brix_plat_sysload_sample(brix_plat_sysload_t *sample)
111{
112 if (sample == NULL) {
113 errno = EINVAL;
114 return -1;
115 }
116 memset(sample, 0, sizeof(*sample));
117 if (getloadavg(&sample->load1, 1) == 1) {
118 sample->have |= BRIX_SYSLOAD_HAVE_LOAD;
119 }
120 sample->total_memory = brix_plat_total_memory();
121 sample->available_memory = brix_plat_available_memory();
122 if (sample->total_memory != 0) sample->have |= BRIX_SYSLOAD_HAVE_MEMORY;
123 sysload_net(sample);
124 sysload_paging(sample);
125 return sample->have != 0 ? 0 : -1;
126}
