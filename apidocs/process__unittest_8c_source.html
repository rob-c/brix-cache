279TEST(execvpe_echo)
280{
281 /* Test that we can execute a simple command */
282 char *argv[] = {"echo", "hello", "world", NULL};
283 char *envp[] = {"TEST=value", NULL};
284
285 /* Fork a child to test execvpe */
286 int pid = _fork();
287 if (pid == 0) {
288 /* Child - exec */
289 brix_plat_execvpe("echo", argv, envp);
290 /* Should not return */
291 _exit(255);
292 } else if (pid > 0) {
293 /* Parent - wait */
294 int status;
295 _waitpid(pid, &status, 0);
296 ASSERT(status == 0, "echo exited successfully");
297 } else {
298 /* Fork failed - skip test */
299 printf("SKIP (fork failed)\n");
300 tests_failed--; /* Compensate for RUN_TEST increment */
301 return;
302 }
303}
