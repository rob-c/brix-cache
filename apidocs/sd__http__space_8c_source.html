1/*
2 * sd_http_space.c — the `space` vtable slot for the http origin driver.
3 *
4 * WHAT: Answers "how big is this backend, and how much of it is left?" from the
5 * ORIGIN's own RFC-4331 quota properties, over one Depth:0 PROPFIND on the
6 * export root.
7 *
8 * WHY: Without this slot kXR_statvfs / kXR_Qspace / kXR_QFSinfo / SRR fall back
9 * to a statvfs(2) of the gateway's own export directory — for an http
10 * origin that directory holds nothing at all, so a client sizing a
11 * transfer against it is reading the gateway's spool, not the storage it
12 * is about to write to. This is the http sibling of sd_xroot_space, which
13 * closed exactly the same wrong answer for root:// origins.
14 *
15 * HOW: RFC 4331 defines a pair of live properties, `DAV:quota-available-bytes`
16 * (what THIS principal may still write) and `DAV:quota-used-bytes`. They
17 * are live properties outside RFC 4918, so an empty ("allprop") PROPFIND
18 * is not required to return them — the request therefore carries an
19 * explicit named-prop body, which is why sd_http_req_t grew an optional
20 * entity. The 207 is read with the same bounded, namespace-agnostic tag
21 * scanner the directory listing uses; nothing here parses XML generally.
22 *
23 * There is no `total` property: RFC 4331 §3 makes used+available the only
24 * defined derivation, and it is the one every WebDAV client makes. A
25 * missing or unparsable half means the origin does not report quota, which
26 * is NGX_ERROR here so the caller falls back to the local statvfs — a
27 * fabricated total would be worse than no answer.
28 */
29

37/* The named-prop request. Kept to the two properties actually consulted: an
38 * allprop would be shorter to write but is not obliged to carry either of them,
39 * and asking for more than is read invites an origin to do more work than the
40 * answer needs. */
41static const char sd_http_quota_body[] =
42 "<?xml version=\"1.0\" encoding=\"utf-8\"?>"
43 "<D:propfind xmlns:D=\"DAV:\"><D:prop>"
44 "<D:quota-available-bytes/><D:quota-used-bytes/>"
45 "</D:prop></D:propfind>";

47/* A quota multistatus is a few hundred bytes. Anything past this is not the
48 * reply this reader understands, and copying it to NUL-terminate it (the tag
49 * scanner walks a C string) would be unbounded work on a driver thread. */
50#define SD_HTTP_QUOTA_XML_MAX (64u * 1024u)
51
52/* XML S production (RFC 4918 bodies are XML 1.0): the only bytes allowed to sit
53 * between a start tag and its character data, or after it. */
54static int
55sd_http_quota_ws(char c)
56{
57 return c == ' ' || c == '\t' || c == '\r' || c == '\n';
58}

60static const char *
61sd_http_quota_skip_ws(const char *p, const char *end)
62{
63 while (p < end && sd_http_quota_ws(*p)) {
64 p++;
65 }

69/*
70 * WHAT: Accumulate one unsigned decimal run at *pp, advancing it past the run.
71 * WHY: strtoull would take a sign, whitespace and a base prefix, and reports
72 * overflow only through errno — none of which is wanted for a value this
73 * server republishes as a capacity.
74 * HOW: Digits only, with a pre-multiply ceiling so an over-wide value is
75 * refused rather than wrapped; an empty run is a refusal, not a zero.
76 */
77static int
78sd_http_quota_u64(const char **pp, const char *end, uint64_t *out)
79{
80 const char *p = *pp;
81 uint64_t v = 0;
82 int digits = 0;
83
84 while (p < end && *p >= '0' && *p <= '9') {
85 if (v > (UINT64_MAX - (uint64_t) (*p - '0')) / 10) {
86 return -1; /* not a size we can represent */
87 }
88 v = v * 10 + (uint64_t) (*p - '0');
89 digits++;
90 p++;
91 }
92 *pp = p;
93 *out = v;
94 return (digits > 0) ? 0 : -1;
95}

97/*
98 * WHAT: Read the unsigned decimal text of element `local` into *out.
99 * WHY: Both quota properties are "a single non-negative integer" (RFC 4331 §3);
100 * an origin that does not support one still lists it, as an EMPTY element
101 * inside a 404 propstat, so "the tag is present" is not "there is a value"
102 * and must not be read as a zero.
103 * HOW: Locate the start tag with the shared scanner, refuse the self-closing
104 * form (`<…/>`), then accumulate digits with an overflow ceiling and
105 * require the run to end on markup or whitespace — so `12x` is rejected
106 * whole rather than read as 12.
107 */
108static int
109sd_http_quota_prop(const char *xml, const char *end, const char *local,
110 uint64_t *out)
111{
112 const char *p = sd_http_xml_open(xml, end, local);
113 const char *gt;
114 uint64_t v = 0;
115
116 if (p == NULL) {
117 return -1;
118 }
119 /* p is the '<' itself, so the '>' that ends the start tag is always past it
120 * and gt[-1] is always readable. */
121 gt = memchr(p, '>', (size_t) (end - p));
122 if (gt == NULL || gt[-1] == '/') {
123 return -1; /* empty element: no value at all */
124 }
125 p = sd_http_quota_skip_ws(gt + 1, end);
126 if (sd_http_quota_u64(&p, end, &v) != 0) {
127 return -1;
128 }
129 if (p < end && *p != '<' && !sd_http_quota_ws(*p)) {
130 return -1; /* trailing junk: not a bare integer */
131 }
132 *out = v;
133 return 0;
134}

136/*
137 * WHAT: Turn one 207 quota body into the space triple.
138 * WHY: Split from the request leg so the wire handling and the arithmetic are
139 * separately readable, and so the NUL-terminated copy the tag scanner
140 * needs has one owner.
141 * HOW: Bounded malloc + copy, both properties or nothing, total = used +
142 * available with an overflow refusal.
143 */
144static int
145sd_http_quota_parse(const void *body, size_t blen, brix_sd_space_t *out)
146{
147 char *xml;
148 uint64_t avail = 0, used = 0;
149 int rc;
150
151 if (body == NULL || blen == 0 || blen > SD_HTTP_QUOTA_XML_MAX) {
152 return -1;
153 }
154 xml = malloc(blen + 1);
155 if (xml == NULL) {
156 return -1;
157 }
158 memcpy(xml, body, blen);
159 xml[blen] = '\0';
160
161 rc = sd_http_quota_prop(xml, xml + blen, "quota-available-bytes", &avail);
162 if (rc == 0) {
163 rc = sd_http_quota_prop(xml, xml + blen, "quota-used-bytes", &used);
164 }
165 free(xml);
166 if (rc != 0) {
167 return -1;
168 }
169 if (avail > UINT64_MAX - used) {
170 return -1;
171 }
172 out->free_bytes = avail;
173 out->used_bytes = used;
174 out->total_bytes = avail + used;
175 return 0;
176}
