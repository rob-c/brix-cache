1/*
2 * http_conditionals.c - shared HTTP conditional request checks (RFC 7232).
3 *
4 * WHAT: Evaluates If-None-Match, If-Match, If-Modified-Since, and Overwrite headers
5 * against resource metadata. Returns 304 Not Modified, 412 Precondition Failed,
6 * or NGX_OK (conditions pass) for each conditional type.
7 *
8 * WHY: RFC 7232 §3 requires servers to validate conditional requests before returning
9 * content. WebDAV PUT/MOVE/COPY and S3 GET/PUT all use these checks. Shared
10 * implementation ensures consistent behaviour across both modules.
11 *
12 * HOW: ETag path generates resource ETag from mtime/size, then compares against header
13 * list entries (case-insensitive, weak-equivalence support). Time path parses
14 * HTTP date via ngx_parse_http_time(), compares against st_mtime. Overwrite
15 * checks header value equals "F".
16*/

26/*
27 * brix_http_strip_weak - strip W/ prefix from ETag string.
28 *
29 * WHAT: If the data starts with "W/", advances data pointer by 2 and reduces len
30 * by 2. Otherwise leaves data and len unchanged.
31 *
32 * WHY: RFC 7232 §2.3 defines weak equivalence: W/"etag1" ≡ "etag1" for If-None-Match
33 * comparison. This helper normalises both resource ETag and header entries before
34 * string comparison when flags include BRIX_HTTP_COND_WEAK_EQUIV.
35 *
36 * HOW: len >= 2 && data[0]=='W' && data[1]=='/' → data+=2, len-=2.
37 */
38
39static void
40brix_http_strip_weak(const char **data, size_t *len)
41{
42 if (*len >= 2 && (*data)[0] == 'W' && (*data)[1] == '/') {
43 *data += 2;
44 *len -= 2;
45 }
46}

48/*
49 * brix_http_next_etag_token - pull the next comma-separated entry from an ETag list.
50 *
51 * WHAT: Advances *cursor past leading separators, then captures one entry up to the
52 * next comma (trailing spaces/tabs trimmed). Returns 1 with tok_start/tok_len set
53 * when an entry was found, 0 (cursor at end) when the list is exhausted.
54 *
55 * WHY: If-Match / If-None-Match carry a comma-separated list of entity-tags
56 * (RFC 7232 §2.3, §3.1). Isolating the tokeniser keeps the comparison loop in
57 * brix_http_etag_list_contains flat and independently testable.
58 *
59 * HOW: 1. Skip leading ' ', '\t', ','. 2. If at end, publish cursor and return 0.
60 * 3. Mark start; scan to next ',' or end. 4. Trim trailing OWS. 5. Publish
61 * cursor, token start and length; return 1.
62 */
63
64static ngx_flag_t
65brix_http_next_etag_token(const char **cursor, const char *end,
66 const char **tok_start, size_t *tok_len)
67{
68 const char *p = *cursor;
69 const char *start;
70 const char *stop;
71
72 while (p < end && (*p == ' ' || *p == '\t' || *p == ',')) {
73 p++;
74 }
75 if (p >= end) {
76 *cursor = p;
77 return 0;
78 }
79
80 start = p;
81 while (p < end && *p != ',') {
82 p++;
83 }
84 stop = p;
85
86 while (stop > start && (stop[-1] == ' ' || stop[-1] == '\t')) {
87 stop--;
88 }
89
90 *cursor = p;
91 *tok_start = start;
92 *tok_len = (size_t) (stop - start);
93 return 1;
94}

96ngx_int_t
97brix_http_etag_list_contains(const ngx_str_t *header, const char *etag,
98 unsigned flags)
99{
100 const char *cursor;
101 const char *end;
102 const char *etag_data;
103 size_t etag_len;
104 const char *tok_start;
105 size_t tok_len;
106
107 if (header == NULL || header->data == NULL || etag == NULL) {
108 return NGX_DECLINED;
109 }
110
111 cursor = (const char *) header->data;
112 end = cursor + header->len;
113 etag_data = etag;
114 etag_len = strlen(etag);
115
116 if (flags & BRIX_HTTP_COND_WEAK_EQUIV) {
117 brix_http_strip_weak(&etag_data, &etag_len);
118 }
119
120 while (brix_http_next_etag_token(&cursor, end, &tok_start, &tok_len)) {
121 if (flags & BRIX_HTTP_COND_WEAK_EQUIV) {
122 brix_http_strip_weak(&tok_start, &tok_len);
123 }
124
125 if (tok_len == etag_len
126 && ngx_strncmp(tok_start, etag_data, tok_len) == 0)
127 {
128 return NGX_OK;
129 }
130 }
131
132 return NGX_DECLINED;
133}

135/*
136 * cond_header_present - is a conditional header usable?
137 *
138 * WHAT: True when the header exists with a non-empty value.
139 * WHY: An empty conditional header is invalid per RFC 9110; treating it as
140 * absent (rather than as a never-matching list) keeps a malformed client
141 * from turning every request into a 412.
142 * HOW: NULL / len check.
143 */
144static ngx_flag_t
145cond_header_present(const ngx_table_elt_t *h)
146{
147 return h != NULL && h->value.len > 0;
148}

150/*
151 * cond_is_wildcard - is the header value the `*` any-representation token?
152 *
153 * WHAT: True for a value that is exactly "*".
154 * WHY: RFC 9110 §13.1.1/§13.1.2: `*` selects any existing representation and
155 * must be recognised before list matching.
156 * HOW: Length-1 compare (nginx has already trimmed surrounding whitespace).
157 */
158static ngx_flag_t
159cond_is_wildcard(const ngx_str_t *v)
160{
161 return v->len == 1 && v->data[0] == '*';
162}

164/*
165 * brix_http_validator_selects - does an entity-tag validator select this resource?
166 *
167 * WHAT: Returns true when the header value (a `*` wildcard or a comma-separated
168 * entity-tag list) selects the current representation. A missing resource is
169 * never selected.
170 *
171 * WHY: RFC 7232 §3.1/§3.2 define the same "does this validator match the selected
172 * representation" test for both If-Match and If-None-Match — the two differ only
173 * in what they do with the answer. Sharing the predicate keeps that answer
174 * byte-identical across every conditional callsite.
175 *
176 * HOW: 1. No representation ⇒ not selected. 2. `*` ⇒ selected (a representation
177 * exists). 3. Otherwise the list must contain the resource's entity-tag.
178 */
179
180static ngx_flag_t
181brix_http_validator_selects(const ngx_str_t *value, ngx_flag_t resource_exists,
182 const char *etag, unsigned condition_flags)
183{
184 if (!resource_exists) {
185 return 0;
186 }
187 if (cond_is_wildcard(value)) {
188 return 1;
189 }
190 return brix_http_etag_list_contains(value, etag, condition_flags) == NGX_OK;
191}

193/*
194 * brix_http_eval_if_unmodified_since - apply an If-Unmodified-Since precondition.
195 *
196 * WHAT: Returns NGX_HTTP_PRECONDITION_FAILED (412) when the resource was modified
197 * after the header date, otherwise NGX_DECLINED (precondition passes / proceed).
198 *
199 * WHY: RFC 7232 §3.4: a write must not proceed if the representation changed since the
200 * client last saw it. An unparseable date is ignored (treated as passing) per the
201 * same section.
202 *
203 * HOW: Parse the HTTP date; if valid and mtime is strictly newer, fail with 412.
204 */
205
206static ngx_int_t
207brix_http_eval_if_unmodified_since(const ngx_table_elt_t *if_unmod, time_t mtime)
208{
209 time_t t = ngx_parse_http_time(if_unmod->value.data, if_unmod->value.len);
210
211 if (t != (time_t) NGX_ERROR && mtime > t) {
212 return NGX_HTTP_PRECONDITION_FAILED;
213 }
214 return NGX_DECLINED;
215}

218 * brix_http_eval_if_modified_since - apply an If-Modified-Since precondition.
219 *
220 * WHAT: Returns NGX_HTTP_NOT_MODIFIED (304) when the resource has not been modified
221 * since the header date, otherwise NGX_DECLINED (precondition passes / proceed).

235 return NGX_HTTP_NOT_MODIFIED;
236 }
237 return NGX_DECLINED;

240ngx_int_t
241brix_http_check_etag_preconditions(ngx_http_request_t *r,
242 ngx_flag_t resource_exists, const struct stat *st, unsigned etag_flags,
243 unsigned condition_flags)
244{
245 ngx_table_elt_t *if_match;
246 ngx_table_elt_t *if_none_match;
247 char etag_buf[64];
248 const char *etag;
249
250 if_match = r->headers_in.if_match;
251 if_none_match = r->headers_in.if_none_match;
252
253 if (if_match == NULL && if_none_match == NULL) {
254 return NGX_OK;
255 }
256
257 etag = NULL;
258 if (resource_exists && st != NULL) {
259 brix_http_etag_str(etag_buf, sizeof(etag_buf),
260 st->st_mtime, st->st_size, etag_flags);
261 etag = etag_buf;
262 }
263
264 /* If-Match: fail unless the validator selects the existing representation. */
265 if (if_match != NULL
266 && !brix_http_validator_selects(&if_match->value, resource_exists, etag,
267 condition_flags))
268 {
269 brix_http_precond_refused_edge(r);
270 return NGX_HTTP_PRECONDITION_FAILED;
271 }
272
273 /* If-None-Match: fail when the validator DOES select it (write semantics). */
274 if (if_none_match != NULL
275 && brix_http_validator_selects(&if_none_match->value, resource_exists,
276 etag, condition_flags))
277 {
278 brix_http_precond_refused_edge(r);
279 return NGX_HTTP_PRECONDITION_FAILED;
280 }
281
282 return NGX_OK;
283}

285/*
286 * brix_http_eval_match_group - first RFC-7232 precedence step (If-Match, else
287 * If-Unmodified-Since).
288 *
289 * WHAT: Returns NGX_HTTP_PRECONDITION_FAILED (412) when the step's precondition fails,
290 * otherwise NGX_DECLINED (proceed to the If-None-Match step).
291 *
292 * WHY: RFC 7232 §6 fixes the order: If-Match takes precedence and, when it is absent,
293 * If-Unmodified-Since is evaluated in its place (the two are mutually exclusive at
294 * this step). Encapsulating the pair preserves that "else" relationship exactly.
295 *
296 * HOW: 1. If-Match present ⇒ 412 unless the validator selects the representation;
297 * otherwise proceed. 2. Else, in time mode, an If-Unmodified-Since header is
298 * applied. 3. Neither present ⇒ proceed.
299 */
300
301static ngx_int_t
302brix_http_eval_match_group(ngx_table_elt_t *if_match, ngx_table_elt_t *if_unmod,
303 ngx_flag_t resource_exists, const char *etag, unsigned condition_flags,
304 unsigned time_mode, time_t mtime)
305{
306 if (cond_header_present(if_match)) {
307 if (!brix_http_validator_selects(&if_match->value, resource_exists, etag,
308 condition_flags))
309 {
310 return NGX_HTTP_PRECONDITION_FAILED;
311 }
312 return NGX_DECLINED;
313 }
314
315 if (time_mode && cond_header_present(if_unmod)) {
316 return brix_http_eval_if_unmodified_since(if_unmod, mtime);
317 }
318
319 return NGX_DECLINED;
320}

322/*
323 * brix_http_eval_none_match_group - second RFC-7232 precedence step (If-None-Match,
324 * else If-Modified-Since).
325 *
326 * WHAT: Returns NGX_HTTP_NOT_MODIFIED (304) or NGX_HTTP_PRECONDITION_FAILED (412) when
327 * the step's precondition triggers, otherwise NGX_DECLINED (conditions pass).
328 *
329 * WHY: RFC 7232 §6 evaluates If-None-Match after the If-Match step and, when it is
330 * absent, evaluates If-Modified-Since in its place (reads only). A selecting
331 * If-None-Match means the client already holds the representation: 304 on reads,
332 * 412 on writes.
333 *
334 * HOW: 1. If-None-Match present ⇒ if the validator selects it, 304 for reads / 412 for
335 * writes; otherwise proceed. 2. Else, for reads in time mode, apply
336 * If-Modified-Since. 3. Neither present ⇒ proceed.
337 */
338
339static ngx_int_t
340brix_http_eval_none_match_group(ngx_table_elt_t *if_none,
341 ngx_table_elt_t *if_mod, ngx_flag_t resource_exists, const char *etag,
342 unsigned condition_flags, unsigned read_mode, unsigned time_mode,
343 time_t mtime)
344{
345 if (cond_header_present(if_none)) {
346 if (brix_http_validator_selects(&if_none->value, resource_exists, etag,
347 condition_flags))
348 {
349 return read_mode ? NGX_HTTP_NOT_MODIFIED
350 : NGX_HTTP_PRECONDITION_FAILED;
351 }
352 return NGX_DECLINED;
353 }
354
355 if (read_mode && time_mode && cond_header_present(if_mod)) {
356 return brix_http_eval_if_modified_since(if_mod, mtime);
357 }
358
359 return NGX_DECLINED;
360}

362ngx_int_t
363brix_http_eval_preconditions(ngx_http_request_t *r,
364 ngx_flag_t resource_exists, time_t mtime, off_t size,
365 unsigned etag_flags, unsigned condition_flags)
366{
367 ngx_table_elt_t *if_match = r->headers_in.if_match;
368 ngx_table_elt_t *if_none = r->headers_in.if_none_match;
369 ngx_table_elt_t *if_unmod = r->headers_in.if_unmodified_since;
370 ngx_table_elt_t *if_mod = r->headers_in.if_modified_since;
371 unsigned read_mode = condition_flags & BRIX_HTTP_COND_READ;
372 unsigned time_mode = condition_flags & BRIX_HTTP_COND_TIME;
373 char etag_buf[64];
374 const char *etag = NULL;
375 ngx_int_t decision;
376
377 if (resource_exists) {
378 brix_http_etag_str(etag_buf, sizeof(etag_buf), mtime, size,
379 etag_flags);
380 etag = etag_buf;
381 }
382
383 /* Step 1: If-Match, or (absent) If-Unmodified-Since — may 412. */
384 decision = brix_http_eval_match_group(if_match, if_unmod, resource_exists,
385 etag, condition_flags, time_mode,
386 mtime);
387 if (decision != NGX_DECLINED) {
388 return decision;
389 }
390
391 /* Step 2: If-None-Match, or (absent) If-Modified-Since — may 304/412. */
392 decision = brix_http_eval_none_match_group(if_none, if_mod, resource_exists,
393 etag, condition_flags, read_mode,
394 time_mode, mtime);
395 if (decision != NGX_DECLINED) {
396 return decision;
397 }
398
399 return NGX_OK;
400}

430/*
431 * brix_http_write_precond - lift a write's conditional headers into the typed
432 * storage precondition (phase-107 C6).
433 *
434 * WHAT: Fills `pre` from If-Match / If-None-Match so the PUBLISH (staged
435 * commit) re-decides the precondition at the storage instead of trusting
436 * an edge check that raced. Zeroes the struct first, so every form this
437 * lift cannot express carries NONE — the edge evaluation stays the
438 * deciding authority for those, exactly today's behaviour.
439 *
440 * WHY: The edge check (brix_http_check_etag_preconditions and the S3
441 * evaluator) runs before the body is even staged; a writer landing
442 * between that check and the rename used to win silently. Carrying the
443 * condition down lets a capable backend decide atomically (and report it
444 * via pre->atomic); the edge check is kept as a fast path that spares
445 * the client the body upload on an already-doomed request.
446 *
447 * HOW: RFC 9110 §13.2.2 precedence — If-Match first. Only forms with an exact
448 * storage meaning are lifted:
449 * If-Match: "<one-tag>" -> MATCH_ETAG (value borrowed from r's pool,
450 * alive across the commit; W/ handled by the
451 * evaluator per §2.3.2 weak comparison)
452 * If-None-Match: * -> ABSENT (create-if-absent)
453 * A list (ETag grammar forbids ',' inside a tag, so a comma IS a list),
454 * If-Match: *, or an empty value stays NONE. When If-Match is present it
455 * wins the lift and If-None-Match remains edge-checked, mirroring the
456 * evaluator's precedence.
457 */
458void
459brix_http_write_precond(ngx_http_request_t *r, brix_sd_precond_t *pre)
460{
461 ngx_table_elt_t *if_match = r->headers_in.if_match;
462 ngx_table_elt_t *if_none_match = r->headers_in.if_none_match;
463
464 ngx_memzero(pre, sizeof(*pre)); /* NONE: fail-safe-by-zero */
465
466 if (cond_header_present(if_match)) {
467 if (!(if_match->value.len == 1 && if_match->value.data[0] == '*')
468 && ngx_strlchr(if_match->value.data,
469 if_match->value.data + if_match->value.len,
470 ',') == NULL)
471 {
472 pre->kind = BRIX_SD_PRECOND_MATCH_ETAG;
473 pre->etag = (const char *) if_match->value.data;
474 pre->etag_len = if_match->value.len;
475 }
476 return;
477 }
478
479 if (cond_header_present(if_none_match)
480 && if_none_match->value.len == 1 && if_none_match->value.data[0] == '*')
481 {
482 pre->kind = BRIX_SD_PRECOND_ABSENT;
483 }
484}

486/*
487 * brix_http_precond_refused_edge - book an EDGE-answered publish-precondition
488 * refusal into the C6 metric pair (phase-107).
489 *
490 * WHAT: Bumps brix_vfs_precond_failed_total{kind} and the advisory counter for
491 * a 412 the pre-body fast path answered, classifying the kind from the
492 * request's conditional headers by RFC 9110 §13.2.2 precedence.
493 *
494 * WHY: The commit-path observer (brix_vfs_precond_refused_observe) only sees
495 * refusals that reach the storage; the edge fast path spares a doomed
496 * request its body upload and answers 412 before any commit exists, so
497 * without this the COMMON refusal was invisible to the one metric that
498 * counts refusals. An edge check is a stat-then-compare with the publish
499 * still to come — non-atomic by construction — so every edge refusal
500 * books advisory, and under the compat label: the deciding compare is
501 * the gateway's own stat grammar, not any storage driver's.
502 *
503 * HOW: If-Match present -> etag (it is evaluated first; when both headers are
504 * present and If-Match passed, an If-None-Match refusal still books as
505 * etag — the same precedence rule the typed lift above applies). Else
506 * If-None-Match "*" -> absent, a named tag -> etag. The date-based
507 * conditionals never reach a write-plane 412 (write mode strips
508 * COND_READ|COND_TIME), so the tag classes are exhaustive here.
509 */
510void
511brix_http_precond_refused_edge(ngx_http_request_t *r)
512{
513 ngx_table_elt_t *if_match = r->headers_in.if_match;
514 ngx_table_elt_t *if_none = r->headers_in.if_none_match;
515 ngx_uint_t kind = (ngx_uint_t) BRIX_SD_PRECOND_MATCH_ETAG;
516
517 if (if_match == NULL && if_none != NULL
518 && if_none->value.len == 1 && if_none->value.data[0] == '*')
519 {
520 kind = (ngx_uint_t) BRIX_SD_PRECOND_ABSENT;
521 }
522 brix_metric_vfs_precond_failed(kind);
523 brix_metric_vfs_precond_advisory(NULL /* -> "posix": the gateway decided */);
524}
