4/*
5 * sd_frm.h - the nearline (tape/MSS) backend FS driver (phase-64 SP5, P4/P6).
6 *
7 * WHAT: A capability-typed SD driver whose objects are NEARLINE: an object is
8 * either ONLINE (present in the MSS disk buffer, directly readable) or
9 * OFFLINE (on tape, minutes-to-hours away). It advertises CAP_NEARLINE and
10 * implements the `recall` slot; the composing registry therefore REQUIRES a
11 * cache tier in front of it (G8, the recall target). A read of an offline
12 * object is a RECALL into the online buffer - exactly the cache miss-fill,
13 * but sourced from tape.
14 *
15 * WHY: P6 makes tape "just another backend tier": instead of a parallel FRM
16 * subsystem that drivers call, tape residency becomes an SD driver behind
17 * the same seam (stat/open/pread/recall/staged_*). A node composes
18 * cache(frm-backend) and serves tape transparently - no special FRM code
19 * path above the seam. (This SP5 increment lands the driver + an in-tree
20 * stub MSS adapter and a SYNCHRONOUS recall; the async park/wake of a slow
21 * recall via the stage_engine waiter, and the dissolution of the legacy
22 * src/frm/ subsystem, are the remaining P6 migration.)
23 *
24 * HOW: Tape access is abstracted behind a pluggable MSS adapter
25 * (brix_mss_adapter_t): residency (online/offline/absent), recall
26 * begin/poll (tape -> online buffer), and migrate (online -> tape). The
27 * `tape://` store-URL selects the adapter; "stub" is the built-in
28 * local-directory simulation used for tests. recall ensures the object is
29 * online (driving the adapter), then sd_cache fills from the online buffer
30 * through this driver's open/pread; staged_* writes the online buffer then
31 * migrates to tape. See docs/refactor/phase-64-fully-tiered-composable-
32 * storage.md (section 9, 13b, 26, Appendix I).
33 */

43/*
44 * The pluggable MSS adapter - how the frm driver talks to the tape system. One
45 * vtable; the tape:// store-URL selects it (stub | exec/hpss/cta stagecmd |
46 * lib/libhpss/libcta dlopen). All calls are blocking and run on the cache-fill /
47 * scheduler path (off the event loop).
48 */
49typedef struct {
50 const char *name; /* "stub" | "exec" | ... */
51 /* Is `key` online, nearline, or offline/absent? Fills size/mtime when known. */
52 int (*residency)(void *mss, const char *key, off_t *size_out,
53 time_t *mtime_out);
54 /* Begin a recall of `key` into the MSS online buffer; 0 started / -1 error. */
55 int (*recall_begin)(void *mss, const char *key);
56 /* Poll a recall: 1 online (ready), 0 in-flight, -1 error. */
57 int (*recall_poll)(void *mss, const char *key);
58 /* Migrate `key` from the online buffer to tape (after a staged write).
59 * 0 shipped / -1 error (errno) / BRIX_MSS_MIGRATE_DEFERRED = the object
60 * is a dataset completion marker under the archiver (2.0 F3): nothing
61 * moved yet, the driver must drive seal(key) -- through the durable
62 * stage engine when it has one -- to compose and ship the archive. */
63 int (*migrate)(void *mss, const char *key);
64 /* 2.0 F3: seal the dataset whose completion marker is `key` (compose
65 * every online member into the archive object, ship it, write the
66 * index sidecar, migrate the marker). 0 ok -- also when the dataset
67 * is already sealed, so a journal replay is idempotent -- / -1 error:
68 * EINVAL not a marker, ENOENT the marker's online copy is gone, else
69 * the tape tier's errno (transient, retried by the engine). Only the
70 * archiver decorator implements it; a NULL slot => ENOTSUP. */
71 int (*seal)(void *mss, const char *key);
72 /* Drop the online copy of `key` (free buffer space). */
73 int (*purge)(void *mss, const char *key);
74 /* Phase-115 W3.2: does the MSS hold a DURABLE copy of `key`, regardless of
75 * the online buffer? 1 yes / 0 no / -1 cannot tell (errno set). The purge
76 * engine releases an online copy only on 1; a NULL slot means the adapter
77 * cannot answer and the engine falls back to the staged-write invariant
78 * (ONLINE ⇒ migrated) documented on sd_frm_evict. */
79 int (*on_tape)(void *mss, const char *key);
80 /* Atomic two-name exchange of the ONLINE-BUFFER copies (phase-107 C6):
81 * renameat2(RENAME_EXCHANGE) on <base>/.online/<a|b>. Both names must be
82 * online (ENOENT otherwise — recall first); the driver migrates both keys
83 * afterwards so tape truth catches up. A NULL slot means this MSS cannot
84 * swap — the driver reports ENOTSUP, never a two-rename emulation. */
85 int (*exchange)(void *mss, const char *a, const char *b);
86 /* §3.7 MSS namespace enumeration (the stock `rsscmd dread` analog): call
87 * cb(ud, name, is_dir) for each entry of MSS directory `key` until done
88 * (a nonzero cb return stops early). 0 ok / -1 error (errno set). A NULL
89 * slot means this MSS cannot enumerate — the driver reports ENOTSUP,
90 * exactly the pre-slot behaviour. */
91 int (*list)(void *mss, const char *key,
92 int (*cb)(void *ud, const char *name, int is_dir), void *ud);
93 /* §3.7 MSS-side directory creation (the stock `rsscmd rcreate` analog):
94 * create directory `key` (and its parents) on the MSS. 0 ok / -1 error
95 * (errno set). NULL slot ⇒ the MSS namespace cannot be extended — the
96 * driver reports ENOTSUP, the pre-slot behaviour. */
97 int (*mkpath)(void *mss, const char *key, mode_t mode);
98 /* Open the online-buffer file of an ONLINE `key` for reading; fd or -1. */
99 int (*open_online)(void *mss, const char *key);
100 /* Create+open the online-buffer file of `key` for a staged write; fd or -1. */
101 int (*create_online)(void *mss, const char *key, mode_t mode);
102 /* Phase-107 C3: make the LOCAL directory entry of `key`'s just-published
103 * artifact durable (fsync its parent directory). 0 ok / -1 error (errno
104 * set). NULL slot ⇒ the adapter keeps nothing local worth flushing. */
105 int (*sync_publish)(void *mss, const char *key);
106 void (*destroy)(void *mss);
107} brix_mss_adapter_t;

164/*
165 * Phase-115 W3.2 — the tape-buffer purge engine (sd_frm_purge.c).
166 *
167 * The online buffer behind a tape:// tier fills with recalled and freshly
168 * migrated copies; nothing above the seam knows which of them are safe to
169 * drop. brix_sd_frm_purge() releases LRU online copies until BOTH arms are
170 * satisfied: the filesystem occupancy of the buffer's mount is back under
171 * `lo_ppm` (arm 1, armed when the tick observed occupancy > `hi_ppm`) and
172 * the bytes the buffer OWNS are back under `max_bytes` (arm 2, armed when
173 * `max_bytes` > 0). A copy is released only when the adapter's on_tape probe
174 * confirms a durable copy (or the adapter cannot answer and the staged-write
175 * invariant ONLINE => migrated holds), it is older than `min_age_s`, and the
176 * `is_pinned` callback does not claim it (a live stage request). Symlinks are
177 * never followed, nothing outside the online root is ever touched.
178 *
179 * 2.0 F4 — per-group rules (brix_frm_purge_policy) and the policy program
180 * (brix_frm_purge_polprog, sd_frm_purge_policy.c): `rule_of` maps a key to
181 * one of `rules`; each rule is an owned-bytes arm of its own (hi/lo) with
182 * an optional longer hold, and a `polprog` rule releases only the copies
183 * the program approved this pass. The export-wide arms above still reach
184 * into every group.
185 */
186typedef struct {
187 const char *name; /* the brix_oss_space group, or "*" */
188 uint64_t hi_bytes; /* the group's own arm: trigger */
189 uint64_t lo_bytes; /* ... and target */
190 time_t hold_s; /* keep copies touched more recently */
191 int polprog; /* releases need the program's approval */
192} brix_sd_frm_purge_rule_t;
193
194typedef struct {
195 ngx_uint_t hi_ppm; /* arm 1 trigger; 0 or >= 1000000 disables it */
196 ngx_uint_t lo_ppm; /* arm 1 target */
197 off_t max_bytes; /* arm 2 (owned-bytes cap); <= 0 disables it */
198 time_t min_age_s; /* never release a copy touched more recently */
199 uint64_t fs_total; /* statvfs of the online root's mount (arm 1) */
200 uint64_t fs_used;
201 int (*is_pinned)(void *ud, const char *key); /* may be NULL */
202 void *ud;
203 /* 2.0 F4: NULL/0 = no per-group rules; rule_of answers an index into
204 * rules or -1; polprog NULL = no program; the deadline bounds one run. */
205 const brix_sd_frm_purge_rule_t *rules;
206 size_t nrules;
207 int (*rule_of)(void *ud, const char *key);
208 const char *polprog;
209 ngx_msec_t polprog_timeout_ms;
210} brix_sd_frm_purge_policy_t;
