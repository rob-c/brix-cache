31/* Build the exact covered open request before deriving its signature. */
32static void
33build_open(ClientOpenRequest *request, uint16_t stream_id, const char *path)
34{
35 xrdw_open_req_t fields = {
36 .mode = 0, .options = kXR_open_read, .optiont = 0
37 };
38 uint32_t length = (uint32_t) strlen(path);
39 uint32_t encoded = htonl(length);
40
41 memset(request, 0, sizeof(*request));
42 request->streamid[0] = (uint8_t) (stream_id >> 8);
43 request->streamid[1] = (uint8_t) stream_id;
44 request->requestid = htons(kXR_open);
45 xrdw_open_req_pack(&fields, ((ClientRequestHdr *) request)->body);
46 memcpy((uint8_t *) request + 20, &encoded, sizeof(encoded));
47}

49static uint8_t *
50make_signature(const brix_conn *connection, uint64_t sequence,
51 const ClientOpenRequest *request, const char *path, int nodata,
52 size_t *signature_length)
53{
54 brix_gsi_cipher_t cipher;
55
56 if (!brix_gsi_cipher_lookup(connection->gsi_deleg_cipher, &cipher)
57 || connection->gsi_deleg_keylen < (size_t) cipher.key_len) return NULL;
58 return brix_gsi_sigver_sign(&cipher, connection->gsi_deleg_key,
59 connection->gsi_deleg_use_iv, sequence, (const uint8_t *) request,
60 (const uint8_t *) path, strlen(path), nodata, signature_length);
61}

64write_prefix(xrd_conform_execution *execution, brix_conn *connection,
65 uint16_t stream_id, uint16_t expectrid, uint8_t version, uint8_t flags,
66 uint64_t sequence, uint8_t crypto, const uint8_t *signature,
67 size_t signature_length, brix_status *status)
68{
69 ClientSigverRequest request;
70 xrdw_sigver_req_t fields = {
71 .expectrid = expectrid,
72 .version = version,
73 .flags = flags,
74 .seqno = sequence,
75 .crypto = crypto,
76 };
77
78 if (signature_length > UINT32_MAX
79 || !xrd_conform_reserve_probe(execution, sizeof(request) + signature_length)) return -1;
80 memset(&request, 0, sizeof(request));
81 request.streamid[0] = (uint8_t) (stream_id >> 8);
82 request.streamid[1] = (uint8_t) stream_id;
83 request.requestid = htons(kXR_sigver);

85 request.dlen = (int32_t) htonl((uint32_t) signature_length);
86 brix_status_clear(status);
87 if (brix_write_full(&connection->io, &request, sizeof(request), status) != 0)
88 return -1;
89 return brix_write_full(&connection->io, signature, signature_length,

108static sigver_observation
109signed_open(xrd_conform_execution *execution, brix_conn *connection,
110 const char *path, uint16_t expectrid, uint8_t version, uint8_t flags,
111 uint64_t sequence, uint8_t crypto, int hash_without_payload)
112{
113 ClientOpenRequest request;
114 brix_status status;
115 brix_payload payload = { path, (uint32_t) strlen(path) };
116 uint16_t stream_id = connection->next_sid++;
117 uint8_t *signature;
118 size_t signature_length = 0;
119 sigver_observation failed = { .rc = -1 };
120
121 build_open(&request, stream_id, path);
122 signature = make_signature(connection, sequence, &request, path,
123 hash_without_payload, &signature_length);
124 if (signature == NULL) return failed;
125 if (write_prefix(execution, connection, stream_id, expectrid, version,
126 flags, sequence, crypto, signature, signature_length, &status) != 0
127 || !xrd_conform_reserve_probe(execution, sizeof(request) + payload.len)
128 || brix_write_full(&connection->io, &request, sizeof(request),
129 &status) != 0
130 || brix_write_full(&connection->io, path, payload.len, &status) != 0) {
131 free(signature);
132 failed.kxr = status.kxr;
133 return failed;
134 }
135 free(signature);
136 return receive_one(connection, stream_id, &status);
137}

139/* WHAT: Cover an open with the plain SHA-256 form authorized by secOFrce.
140 * WHY: Merely observing the option is not evidence that the advertised form is
141 * both accepted and verified; a server that ignores the bytes is unsafe.
142 * HOW: Hash the exact sequence/header/payload span, optionally corrupt the
143 * digest, then send it as the sigver body without session encryption. */
144static sigver_observation
145unencrypted_open(xrd_conform_execution *execution, brix_conn *connection,
146 const char *path, int corrupt)
147{
148 ClientOpenRequest request;
149 brix_status status;
150 brix_payload payload = { path, (uint32_t) strlen(path) };
151 uint16_t stream_id = connection->next_sid++;
152 uint8_t hash[32];
153 sigver_observation failed = { .rc = -1 };
154
155 build_open(&request, stream_id, path);
156 if (!brix_gsi_sigver_hash(1, (const uint8_t *) &request,
157 (const uint8_t *) path, strlen(path), 0, hash))
158 return failed;
159 if (corrupt) hash[0] ^= 0x80;
160 if (write_prefix(execution, connection, stream_id, kXR_open, 0, 0, 1,
161 kXR_SHA256_sig, hash, sizeof(hash), &status) != 0
162 || !xrd_conform_reserve_probe(execution, sizeof(request) + payload.len)
163 || brix_write_full(&connection->io, &request, sizeof(request),
164 &status) != 0
165 || brix_write_full(&connection->io, path, payload.len, &status) != 0)
166 {
167 failed.kxr = status.kxr;
168 return failed;
169 }
170 return receive_one(connection, stream_id, &status);
171}

219static void
220probe_one(xrd_conform_execution *execution, const brix_url *url,
221 const char *id)
222{
223 brix_conn connection;
224 brix_status status;
225 sigver_observation first = { .rc = -1 };
226 sigver_observation observed = { .rc = -1 };
227
228 if (connect_signing(execution, url, &connection, &status) != 0) return;
229 if (strcmp(id, "SIG-003") == 0) {
230 observed = signed_open(execution, &connection, url->path, kXR_stat,
231 0, 0, 1, kXR_SHA256_sig, 0);
232 grade(execution, id, is_refusal(observed),
233 "wrong-expectrid-refused", observed);
234 } else if (strcmp(id, "SIG-004") == 0) {
235 first = signed_open(execution, &connection, url->path, kXR_open,
236 0, 0, 10, kXR_SHA256_sig, 0);
237 observed = invalid_prefix(execution, &connection, url->path, 0,
238 kXR_SHA256_sig, 10);
239 grade(execution, id, first.rc == 0 && is_refusal(observed),
240 "replayed-sequence-refused", observed);
241 } else if (strcmp(id, "SIG-005") == 0) {
242 observed = signed_open(execution, &connection, url->path, kXR_open,
243 0, 0, UINT64_C(1) << 40, kXR_SHA256_sig, 0);
244 grade(execution, id, observed.rc == 0,
245 "forward-sequence-gap-accepted", observed);
246 } else if (strcmp(id, "SIG-006") == 0) {
247 if (connection.sec_level >= 4) {
248 xrd_conform_result_set(execution, id,
249 BRIX_CONFORM_VERDICT_SKIPPED,
250 "claim-excludes-nodata:pedantic-payload-signing", NULL, NULL);
251 } else {
252 observed = signed_open(execution, &connection, url->path, kXR_open,
253 0, kXR_nodata_sig, 1, kXR_SHA256_sig, 1);
254 grade(execution, id, observed.rc == 0,
255 "nodata-signature-ignored-payload-bytes", observed);
256 }
257 } else if (strcmp(id, "SIG-007") == 0) {
258 observed = invalid_prefix(execution, &connection, url->path, 0,
259 0x02, 1);
260 grade(execution, id, is_refusal(observed),
261 "unknown-signature-hash-refused", observed);
262 } else if (strcmp(id, "SIG-008") == 0) {
263 observed = invalid_prefix(execution, &connection, url->path, 1,
264 kXR_SHA256_sig, 1);
265 grade(execution, id, is_refusal(observed),
266 "unknown-signature-version-refused", observed);
267 }
268 brix_close(&connection);
269}
