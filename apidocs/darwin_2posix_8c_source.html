25int
26brix_plat_umount(const char *path, int lazy)
27{
28 /* no lazy detach on Darwin: MNT_FORCE is the closest for the EBUSY retry */
29 return unmount(path, lazy ? MNT_FORCE : 0);
30}

57/* macFUSE unmounts when the daemon exits and rejects the Linux libfuse
58 * spelling that asks for exactly that ("fuse: unknown option(s): `-o
59 * auto_unmount'"), so the element is dropped rather than passed through. */
60int
61brix_plat_fuse_opt_supported(const char *opt)
62{
63 if (opt == NULL || opt[0] == '\0') {
64 return 0;
65 }
66 return strcmp(opt, "auto_unmount") != 0;
67}

69const char *
70brix_plat_fuse_host_opts(void)
71{
72 /* No "._name" AppleDouble sidecars: they are created and written through
73 * the filesystem and then looked up by name before a staged write lands,
74 * which fails every write with ENXIO on a filesystem that commits on
75 * close. Extended attributes still go through the driver's own xattr ops. */
76 return "noappledouble";
77}
