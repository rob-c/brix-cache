29/* Parse "key=<type>[:<prefix>]" into rule->key_type / rule->key_match.
30 * sizeof("key=")-1 == 4 strips the literal prefix; the value tail is then split
31 * on the first ':' into <type> and an optional <prefix> (used only by VOLUME). */
32ngx_int_t
33rl_parse_key(ngx_conf_t *cf, ngx_str_t *v, brix_rl_rule_t *rule)
34{
35 ngx_str_t val = { v->len - (sizeof("key=") - 1),
36 v->data + sizeof("key=") - 1 };
37 u_char *colon;
38 ngx_uint_t i;
39 ngx_str_t type = val, prefix = { 0, NULL };
40
41 /* If a ':' is present, narrow `type` to the part before it and point
42 * `prefix` at the (non-NUL-terminated) remainder after it. */
43 colon = ngx_strlchr(val.data, val.data + val.len, ':');
44 if (colon != NULL) {
45 type.len = colon - val.data;
46 prefix.data = colon + 1;
47 prefix.len = val.data + val.len - (colon + 1);
48 }
49
50 for (i = 0; i < sizeof(rl_key_names) / sizeof(rl_key_names[0]); i++) {
51 if (type.len == ngx_strlen(rl_key_names[i].name)
52 && ngx_strncmp(type.data, rl_key_names[i].name, type.len) == 0)
53 {
54 rule->key_type = rl_key_names[i].type;
55 return NGX_OK;
56 }
57 }
58 if (type.len == 6 && ngx_strncmp(type.data, "volume", 6) == 0) {
59 rule->key_type = BRIX_RL_KEY_VOLUME;
60 rule->key_match = prefix; /* points into cf->args memory (persists) */
61 return NGX_OK;
62 }
63 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
64 "xrootd rate limit: unknown key \"%V\" (expected"
65 " vo|issuer|ip|dn|subject|volume[:<prefix>])", &type);
66 return NGX_ERROR;
67}

69/* Parse "<N>r/s" → requests/s (returns the integer N, or NGX_ERROR). */
70ngx_int_t
71rl_parse_req_rate(ngx_str_t *v)
72{
73 if (v->len < 4 || ngx_strncmp(v->data + v->len - 3, "r/s", 3) != 0) {
74 return NGX_ERROR;

79/* Shared "<N>[k|m|g]" → bytes parser (phase-101 W7 dedup). nginx's own
80 * ngx_parse_size handles ONLY k/m — this adds the g (gibibyte) suffix that the
81 * bandwidth-rate grammar already accepted, with an equivalent overflow guard, so
82 * every ratelimit size (rate, burst, zone) accepts k/m/g uniformly through ONE
83 * routine instead of two divergent parsers. Returns NGX_ERROR on bad input. */
84static ssize_t
85rl_parse_size_bytes(ngx_str_t *v)
86{
87 if (v->len >= 2) {
88 u_char u = v->data[v->len - 1];
89
90 if (u == 'g' || u == 'G') {
91 ngx_str_t num = *v;
92 ngx_int_t n;
93
94 num.len--; /* drop the 'g' */
95 n = ngx_atoi(num.data, num.len);
96 if (n == NGX_ERROR
97 || n > (ngx_int_t) (NGX_MAX_SIZE_T_VALUE >> 30))
98 {
99 return NGX_ERROR; /* bad digits / overflow */
100 }
101 return (ssize_t) ((size_t) n << 30);
102 }
103 }
104 return ngx_parse_size(v); /* k/m + plain digits, with nginx's overflow guard */
105}

107/* Parse "<N>[k|m|g]/s" → bytes/s. Returns NGX_ERROR on bad input.
108 * Strip the trailing "/s", then parse the "<N>[k|m|g]" remainder as bytes. */
109ssize_t
110rl_parse_bw_rate(ngx_str_t *v)
111{
112 ngx_str_t num = *v;
113
114 if (v->len < 3 || ngx_strncmp(v->data + v->len - 2, "/s", 2) != 0) {
115 return NGX_ERROR;
116 }
117 num.len = v->len - 2; /* drop "/s"; remainder is bytes/s */
118 return rl_parse_size_bytes(&num);
119}

121/* Parse "<N>[k|m|g]" → bytes (burst / zone size). */
122ssize_t
123rl_parse_size(ngx_str_t *v)
124{
125 return rl_parse_size_bytes(v); /* k/m/g — see rl_parse_size_bytes */
126}

129/* zone directive: brix_rate_limit_zone zone=NAME:SIZE */
130char *
131brix_rl_zone_directive(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
132{
133 ngx_str_t *value = cf->args->elts;
134 ngx_str_t spec, name, sizestr;
135 u_char *colon;
136 ssize_t size;
137 ngx_uint_t i;
138
139 (void) cmd; (void) conf;
140
141 name.len = 0; name.data = NULL;
142 sizestr.len = 0; sizestr.data = NULL;
143
144 for (i = 1; i < cf->args->nelts; i++) {
145 if (value[i].len > 5
146 && ngx_strncmp(value[i].data, "zone=", 5) == 0)
147 {
148 spec.data = value[i].data + 5;
149 spec.len = value[i].len - 5;
150 colon = ngx_strlchr(spec.data, spec.data + spec.len, ':');
151 if (colon == NULL) {
152 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
153 "brix_rate_limit_zone: expected zone=NAME:SIZE");
154 return NGX_CONF_ERROR;
155 }
156 name.data = spec.data;
157 name.len = colon - spec.data;
158 sizestr.data = colon + 1;
159 sizestr.len = spec.data + spec.len - (colon + 1);
160 }
161 }
162
163 if (name.len == 0 || sizestr.len == 0) {
164 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
165 "brix_rate_limit_zone: missing zone=NAME:SIZE");
166 return NGX_CONF_ERROR;
167 }
168 size = rl_parse_size(&sizestr);
169 if (size == NGX_ERROR || size <= 0) {
170 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
171 "brix_rate_limit_zone: bad size \"%V\"", &sizestr);
172 return NGX_CONF_ERROR;
173 }
174
175 if (brix_rl_zone_add(cf, &name, (size_t) size, NULL) != NGX_OK) {
176 return NGX_CONF_ERROR;
177 }
178 return NGX_CONF_OK;
179}
