1/*
2 * vfs_open_adopt.c — VFS ctx priming, handle adoption, and shared path helpers.
3 *
4 * WHAT: Hosts the handle-construction half of the VFS open unit:
5 * brix_vfs_ctx_init() (prime a per-request ctx), brix_vfs_fill_stat()
6 * (struct stat -> brix_vfs_stat_t), brix_vfs_copy_path() (pool-dup a C
7 * string), brix_vfs_adopt_fd()/brix_vfs_adopt_obj() (wrap an already-open
8 * fd / a backend object in a brix_vfs_file_t), and
9 * brix_vfs_export_relative()/_root() (export-root-relative path form).
10 *
11 * WHY: These are the pieces the open orchestrator (vfs_open.c) and the cache
12 * layer both build handles with. Splitting them out of vfs_open.c keeps
13 * both files under the size cap while leaving the open cascade in one place.
14 * Every symbol here is already public (declared in vfs_internal.h / vfs.h)
15 * because external units (the cache layer, vfs_staged.c) also call them —
16 * so nothing new crosses a boundary; this is purely a file-size split.
17 *
18 * HOW: brix_vfs_adopt_fd()/_obj() capture open-time metadata through the
19 * backend's fstat slot (never a direct fstat(2)) into a freshly pcalloc'd
20 * handle, dup the path, and record from_cache/is_tls/stat_current.
21 * brix_vfs_export_relative_root() strips an absolute export root from a
22 * confined path so an inst-keyed storage driver keys its namespace on the
23 * logical path. See vfs_open.c for the open cascade that consumes these.
24 */
25#include "vfs_internal.h"
26#include "vfs_backend_registry.h"

28/* Populate a per-request brix_vfs_ctx_t with the fields the HTTP front ends
29 * (WebDAV, S3) set identically: a transient (rootfd = -1) confined open of an
30 * already-resolved canonical path. Zeroes the ctx first, then fills pool/log,
31 * the metrics proto, the export + cache roots (deriving cache_enabled), the
32 * endpoint mutation policy, the TLS flag, the identity, and the resolved path
33 * (is_confined). Callers may still adjust individual fields afterwards (e.g.
34 * cache_writethrough config). Kept HTTP-agnostic on purpose so the header stays
35 * stream-includable — callers pass pool/log/is_tls extracted from their own
36 * request object.
37 *
38 * The mutation policy is stored BEFORE the backend is resolved, and any value
39 * outside the enum is normalised to READ_ONLY (phase-105 §C.2): the initializer
40 * returns void, so coercion is the only way to guarantee that a caller passing
41 * a stray integer — or a flag that never survived a config merge — gets a
42 * closed endpoint rather than an open one. */
43void
44brix_vfs_ctx_init(brix_vfs_ctx_t *vctx, ngx_pool_t *pool, ngx_log_t *log,
45 brix_proto_t proto, const char *root_canon, const char *cache_root_canon,
46 brix_vfs_mutation_policy_t mutation_policy, int is_tls,
47 brix_identity_t *identity, const char *resolved_path)
48{
49 if (vctx == NULL) {
50 return;
51 }
52
53 ngx_memzero(vctx, sizeof(*vctx));
54 vctx->rootfd = -1;
55 vctx->pool = pool;
56 vctx->log = log;
57 vctx->metrics_proto = proto;
58 vctx->root_canon = root_canon;
59 vctx->mutation_policy = (mutation_policy == BRIX_VFS_MUTATION_ALLOWED)
60 ? BRIX_VFS_MUTATION_ALLOWED : BRIX_VFS_MUTATION_READ_ONLY;
61 /* Resolve the export's selected storage backend (NULL ⇒ default POSIX) so
62 * every VFS op on this ctx routes through the driver without each handler
63 * threading the instance. Per-worker, lazily created on first use. */
64 vctx->sd = brix_vfs_backend_resolve(root_canon, log);
65 /* phase-108 A.4: bind the export's resolved name translation from the same
66 * registry entry (borrowed, worker-lifetime). NULL ⇒ IDENTITY — the generic
67 * brix_path_lfn_to_pfn / _pfn_to_lfn stage reads this. */
68 vctx->n2n = brix_vfs_backend_n2n(root_canon);
69 vctx->cache_root_canon = cache_root_canon;
70 vctx->cache_enabled =
71 (cache_root_canon != NULL && cache_root_canon[0] != '\0') ? 1 : 0;
72 vctx->is_tls = is_tls ? 1 : 0;
73 vctx->identity = identity;
74 if (resolved_path != NULL) {
75 vctx->resolved.resolved.data = (u_char *) resolved_path;
76 vctx->resolved.resolved.len = ngx_strlen(resolved_path);
77 vctx->resolved.is_confined = 1;
78 }
79}

98/* Copy a struct stat into the protocol-neutral brix_vfs_stat_t (zeroes the
99 * output first; sets is_directory/is_regular from the mode). No-op on NULL. */
100void
101brix_vfs_fill_stat(const struct stat *st, brix_vfs_stat_t *out)
102{
103 if (st == NULL || out == NULL) {
104 return;
105 }
106
107 ngx_memzero(out, sizeof(*out));
108 out->size = st->st_size;
109 out->mtime = st->st_mtime;
110 out->ctime = st->st_ctime;
111 out->atime = st->st_atime;
112 out->mode = (ngx_uint_t) st->st_mode;
113 out->ino = st->st_ino;
114 out->dev = st->st_dev;
115 out->uid = st->st_uid;
116 out->gid = st->st_gid;
117 out->blocks = st->st_blocks;
118 out->is_directory = S_ISDIR(st->st_mode) ? 1 : 0;
119 out->is_regular = S_ISREG(st->st_mode) ? 1 : 0;
120}

122char *
123brix_vfs_copy_path(ngx_pool_t *pool, const char *path)
124{
125 size_t len;
126 char *copy;
127
128 if (pool == NULL || path == NULL) {
129 errno = EINVAL;
130 return NULL;
131 }
132
133 len = strlen(path);
134 copy = ngx_pnalloc(pool, len + 1);
135 if (copy == NULL) {
136 errno = ENOMEM;
137 return NULL;
138 }
139
140 ngx_memcpy(copy, path, len);
141 copy[len] = '\0';
142 return copy;
143}

145/* Wrap an already-open fd in a freshly pcalloc'd brix_vfs_file_t: fstat the
146 * fd into the cached metadata, dup the path, and record attrs.from_cache/is_tls.
147 * attrs.writable gates the stat_current fast path (set only for read-only
148 * handles). Used by both brix_vfs_open() and the cache layer's open path. */
149ngx_int_t
150brix_vfs_adopt_fd(brix_vfs_ctx_t *ctx, const char *path, ngx_fd_t fd,
151 brix_vfs_adopt_attrs_t attrs, brix_vfs_file_t **out)
152{
153 brix_sd_stat_t st;
154 brix_sd_obj_t obj;
155 brix_vfs_file_t *fh;
156
157 if (out == NULL) {
158 errno = EINVAL;
159 return NGX_ERROR;
160 }
161 *out = NULL;
162
163 if (ctx == NULL || path == NULL || fd == NGX_INVALID_FILE) {
164 errno = EINVAL;
165 return NGX_ERROR;
166 }
167
168 /* Capture the open-time metadata through the backend's fstat slot rather
169 * than a direct fstat(2): the VFS records the cached size/mtime/etc. but
170 * leaves the syscall to the storage driver. */
171 brix_vfs_ctx_sd_obj(ctx, fd, &obj);
172 if (obj.driver->fstat == NULL || obj.driver->fstat(&obj, &st) != NGX_OK) {
173 return NGX_ERROR;
174 }
175
176 fh = ngx_pcalloc(ctx->pool, sizeof(*fh));
177 if (fh == NULL) {
178 errno = ENOMEM;
179 return NGX_ERROR;
180 }
181
182 fh->path = brix_vfs_copy_path(ctx->pool, path);
183 if (fh->path == NULL) {
184 return NGX_ERROR;
185 }
186
187 fh->obj = obj; /* backend object built above (driver + inst + fd) */
188 fh->pool = ctx->pool;
189 fh->log = ctx->log;
190 fh->ctx = ctx;
191 fh->mutation_policy = ctx->mutation_policy;
192 fh->metrics_proto = ctx->metrics_proto;
193 fh->authz = ctx->authz;
194 fh->identity = ctx->identity;
195 fh->io_monitor = ctx->io_monitor;
196 fh->root_canon = ctx->root_canon;
197 fh->size = st.size;
198 fh->mtime = st.mtime;
199 fh->ctime = st.ctime;
200 fh->ino = st.ino;
201 fh->mode = st.mode;
202 fh->from_cache = attrs.from_cache ? 1 : 0;
203 fh->is_tls = ctx->is_tls;
204 /* phase-110 W4: the open IS the request's data op as far as the monitor
205 * can see — the client-facing serve is zero-copy and never reaches the
206 * per-op observer — so record it as read/write here. Weight 2 displaces the
207 * stat that preceded it (a GET logs op=read, not op=stat); a later composite
208 * op (copy/tpc) displaces this in turn. */
209 brix_io_monitor_record_op(ctx->io_monitor,
210 attrs.writable ? BRIX_METRIC_OP_WRITE : BRIX_METRIC_OP_READ,
211 path, BRIX_ERR_NONE);
212 /* Trust the open-time metadata for stat() only on a read-only handle: the
213 * file cannot change through it, so the fstat above stays authoritative for
214 * the handle's lifetime. A writable handle leaves this 0 so brix_vfs_file_stat()
215 * always issues a live fstat (its bytes/mtime/size move as it is written). */
216 fh->stat_current = attrs.writable ? 0 : 1;
217 fh->memfd = NGX_INVALID_FILE;
218
219 *out = fh;
220 return NGX_OK;
221}

223/* brix_vfs_adopt_obj — build a handle from an object the backend's open slot
224 * already produced, PRESERVING its per-open state (e.g. an object backend's
225 * block map / metadata), rather than rebuilding from a bare fd (which would drop
226 * that state). Captures open-time metadata via the object's own fstat slot, then
227 * moves the object into the handle by value. A heap-allocated shell (driver set
228 * obj->heap_shell, e.g. pblock's malloc'd object) is freed once copied; the
229 * per-open `state` lives on and is released by driver->close at handle close.
230 * Used for the non-POSIX storage drivers; POSIX keeps the adopt_fd path. Also
231 * the cache hit-serve path (src/cache/open.c) adopts a cache-storage driver
232 * object — hence public (declared in vfs.h). */

237/* brix_vfs_export_relative — the export-root-relative ("logical") form of a
238 * confined path, which is what an inst-keyed storage-driver op expects (per the
239 * SD seam contract). Some callers (WebDAV/S3) resolve to an absolute path under
240 * root_canon; strip that prefix so a non-POSIX backend keys its namespace on the
241 * logical path. Returns `path` unchanged when it is not under root_canon.
242 * Declared in vfs_internal.h; shared with vfs_staged.c. */
243/* Path-based form: strip `root_canon` (an absolute export root) from `path`. */
244const char *
245brix_vfs_export_relative_root(const char *path, const char *root_canon)
246{
247 size_t rlen;
248
249 if (root_canon == NULL || path == NULL) {
250 return path;
251 }
252 rlen = ngx_strlen(root_canon);
253 if (rlen == 0 || ngx_strncmp(path, root_canon, rlen) != 0) {
254 return path;
255 }
256 if (path[rlen] == '/') {
257 return path + rlen; /* "/sub/file" */
258 }
259 if (path[rlen] == '\0') {
260 return "/"; /* the export root itself */
261 }
262 return path; /* a sibling that merely shares the prefix */
263}

267{
268 return brix_vfs_export_relative_root(path, ctx->root_canon);

271ngx_int_t
272brix_vfs_adopt_obj(brix_vfs_ctx_t *ctx, const char *path,
273 brix_sd_obj_t *o, unsigned writable, brix_vfs_file_t **out)
274{
275 brix_sd_stat_t st;
276 brix_vfs_file_t *fh;
277
278 *out = NULL;
279 if (ctx == NULL || path == NULL || o == NULL) {
280 errno = EINVAL;
281 return NGX_ERROR;
282 }
283 if (o->driver->fstat == NULL || o->driver->fstat(o, &st) != NGX_OK) {
284 return NGX_ERROR;
285 }
286
287 fh = ngx_pcalloc(ctx->pool, sizeof(*fh));
288 if (fh == NULL) {
289 errno = ENOMEM;
290 return NGX_ERROR;
291 }
292 fh->path = brix_vfs_copy_path(ctx->pool, path);
293 if (fh->path == NULL) {
294 return NGX_ERROR;
295 }
296
297 fh->obj = *o; /* driver + instance + fd + per-open state */
298 if (o->heap_shell) {
299 free(o); /* shell copied into fh->obj; release it */
300 }
301 fh->obj.heap_shell = 0; /* fh->obj is embedded, not a heap shell */
302
303 fh->pool = ctx->pool;
304 fh->log = ctx->log;
305 fh->ctx = ctx;
306 fh->mutation_policy = ctx->mutation_policy;
307 fh->metrics_proto = ctx->metrics_proto;
308 fh->authz = ctx->authz;
309 fh->identity = ctx->identity;
310 fh->io_monitor = ctx->io_monitor;
311 fh->root_canon = ctx->root_canon;
312 fh->size = st.size;
313 fh->mtime = st.mtime;
314 fh->ctime = st.ctime;
315 fh->ino = st.ino;
316 fh->mode = st.mode;
317 fh->from_cache = 0;
318 fh->is_tls = ctx->is_tls;
319 /* Object-backed opens are data operations too. Cache hits and native
320 * object backends enter through adopt_obj rather than adopt_fd; recording
321 * the open here keeps the request/session monitor independent of which
322 * storage driver supplied the handle. */
323 brix_io_monitor_record_op(ctx->io_monitor,
324 writable ? BRIX_METRIC_OP_WRITE : BRIX_METRIC_OP_READ,
325 path, BRIX_ERR_NONE);
326 fh->stat_current = writable ? 0 : 1;
327 fh->memfd = NGX_INVALID_FILE;
328
329 *out = fh;
330 return NGX_OK;
331}
