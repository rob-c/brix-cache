18/* WHAT: Hash a namespace key. WHY: Share one deterministic candidate/index hash.
19 * HOW: 1. Fold every byte. 2. Return the full-width value. */
20static uint64_t
21xfs_path_hash(const char *path)
22{
23 uint64_t hash = UINT64_C(14695981039346656037);
24 for (; *path; path++) {
25 hash = (hash ^ (unsigned char) *path) * UINT64_C(1099511628211);
26 }
27 return hash;
28}

136/* WHAT: Reserve a node and alias, returning negative errno on exhaustion.
137 * WHY: A successful remote create/link must never lose local bookkeeping.
138 * HOW: 1. Check hard limits. 2. Allocate all memory. 3. Charge the reservation. */
139int
140xfs_nodes_prepare(xfs_nodes *table, const char *path, xfs_node_ticket *ticket)
141{
142 size_t bytes = strlen(path) + 1;
143 memset(ticket, 0, sizeof(*ticket));
144 if (bytes > XRDC_PATH_MAX) {
145 return -ENAMETOOLONG;
146 }
147 while (table->next_ino != UINT64_MAX && xfs_nodes_find(table, table->next_ino)) {
148 table->next_ino++;
149 }
150 if (table->nodes >= XFS_NODE_LIMIT || table->aliases >= XFS_ALIAS_LIMIT
151 || bytes > XFS_ALIAS_BYTES - table->alias_bytes
152 || table->next_ino == UINT64_MAX) {
153 return -ENOSPC;
154 }
155 ticket->node = calloc(1, sizeof(*ticket->node));
156 ticket->alias = calloc(1, sizeof(*ticket->alias));
157 if (ticket->alias) {
158 ticket->alias->path = strdup(path);
159 }
160 if (!ticket->node || !ticket->alias || !ticket->alias->path) {
161 if (ticket->alias) {
162 free(ticket->alias->path);
163 }
164 free(ticket->alias);
165 free(ticket->node);
166 memset(ticket, 0, sizeof(*ticket));
167 return -ENOMEM;
168 }
169 ticket->node->ino = table->next_ino++;
170 ticket->node->generation = ticket->node->ino;
171 table->nodes++;
172 table->aliases++;
173 table->alias_bytes += bytes;
174 return 0;
175}

209/* WHAT: Suggest a stable directory inode, or zero when a collision is unknown.
210 * WHY: readdir's advisory inode should agree with a later lookup/stat. Unknown
211 * WebDAV identities derive from exact paths but are NEVER merged by hash.
212 * HOW: 1. Prefer known server/path owners. 2. Select server id or path hash.
213 * 3. Refuse occupied candidate ids rather than aliasing unrelated objects. */
214uint64_t
215xfs_nodes_ino_hint(xfs_nodes *table, const char *path, uint64_t backend)
216{
217 xfs_node *node = xfs_backend_find(table, backend);
218 if (node) {
219 return node->ino;
220 }
221 xfs_alias *alias = xfs_nodes_alias(table, path);
222 if (alias && (!backend || !alias->node->backend || alias->node->backend == backend)) {
223 return alias->node->ino;
224 }
225 uint64_t candidate = backend > 1 ? backend : xfs_path_hash(path) | (UINT64_C(1) << 63);
226 return xfs_nodes_find(table, candidate) ? 0 : candidate;
227}

241/* WHAT: Publish a prepared name without allocating. WHY: Preserve hard-link ids.
242 * HOW: 1. Prefer the linked node or matching identity. 2. Reconcile the alias.
243 * 3. Publish a fresh node only when necessary. 4. Return unused capacity. */
244xfs_node *
245xfs_nodes_bind(xfs_nodes *table, xfs_node_ticket *ticket, uint64_t backend,
246 xfs_node *preferred)
247{
248 xfs_alias *existing = xfs_nodes_alias(table, ticket->alias->path);
249 xfs_node *node = preferred ? preferred : xfs_backend_find(table, backend);
250 if (!node && existing && (!backend || !existing->node->backend
251 || existing->node->backend == backend)) {
252 node = existing->node;
253 }
254 if (!node) {
255 node = ticket->node;
256 ticket->node = NULL;
257 uint64_t candidate = xfs_nodes_ino_hint(table, ticket->alias->path, backend);
258 if (table->nodes == 1 && strcmp(ticket->alias->path, "/") == 0) {
259 candidate = 1; /* the permanent root is the only reserved node id */
260 }
261 if (candidate) {
262 node->ino = candidate;
263 }
264 size_t bucket = node->ino % XFS_NODE_BUCKETS;
265 node->id_next = table->ids[bucket];
266 table->ids[bucket] = node;
267 }
268 xfs_backend_attach(table, node, backend);
269 if (existing && existing->node != node) {
270 xfs_alias_remove(table, existing);
271 existing = NULL;
272 }
273 if (!existing) {
274 xfs_alias *alias = ticket->alias;
275 ticket->alias = NULL;
276 alias->node = node;
277 alias->node_next = node->aliases;
278 node->aliases = alias;
279 xfs_alias_hash(table, alias);
280 }
281 xfs_nodes_cancel(table, ticket);
282 return node;
283}

285/* WHAT: Reclaim a node with no kernel/handle owners. WHY: Bound mount memory.
286 * HOW: 1. Retain root and referenced nodes. 2. Drop indexes and aliases. 3. Free. */
287static void
288xfs_node_collect(xfs_nodes *table, xfs_node *node)
289{
290 if (node->ino == 1 || node->lookups || node->opens) {
291 return;
292 }
293 xfs_node **cursor = &table->ids[node->ino % XFS_NODE_BUCKETS];
294 while (*cursor != node) {
295 cursor = &(*cursor)->id_next;
296 }
297 *cursor = node->id_next;
298 if (node->backend) {
299 cursor = &table->backends[node->backend % XFS_NODE_BUCKETS];
300 while (*cursor != node) {
301 cursor = &(*cursor)->backend_next;
302 }
303 *cursor = node->backend_next;
304 }
305 while (node->aliases) {
306 xfs_alias_remove(table, node->aliases);
307 }
308 table->nodes--;
309 free(node);
310}

377/* WHAT: Prepare one renamed path. WHY: Never allocate after successful rename.
378 * HOW: 1. Check path/budget bounds. 2. Allocate replacement and record. */
379static int
380xfs_rename_add(xfs_nodes *table, xfs_alias *alias, const char *from,
381 const char *to, xfs_rename_change **changes)
382{
383 size_t length = strlen(to) + strlen(alias->path + strlen(from)) + 1;
384 if (length > XRDC_PATH_MAX) {
385 return -ENAMETOOLONG;
386 }
387 if (length > XFS_ALIAS_BYTES - table->alias_bytes) {
388 return -ENOSPC;
389 }
390 xfs_rename_change *change = calloc(1, sizeof(*change));
391 if (!change) {
392 return -ENOMEM;
393 }
394 change->replacement = malloc(length);
395 if (!change->replacement) {
396 free(change);
397 return -ENOMEM;
398 }
399 snprintf(change->replacement, length, "%s%s", to, alias->path + strlen(from));
400 table->alias_bytes += length;
401 change->alias = alias;
402 change->next = *changes;
403 *changes = change;
404 return 0;
405}

407/* WHAT: Reserve all subtree rename keys. WHY: Open descendants survive rename.
408 * HOW: 1. Scan names with a boundary check. 2. Reserve each new path. 3. Roll back
409 * all reservations on error; the namespace has not yet changed. */
410int
411xfs_nodes_rename_prepare(xfs_nodes *table, const char *from, const char *to,
412 xfs_rename_change **changes)
413{
414 *changes = NULL;
415 for (size_t bucket = 0; bucket < XFS_NODE_BUCKETS; bucket++) {
416 for (xfs_alias *alias = table->paths[bucket]; alias; alias = alias->hash_next) {
417 if (!xfs_path_inside(alias->path, from)) {
418 continue;
419 }
420 int rc = xfs_rename_add(table, alias, from, to, changes);
421 if (rc != 0) {
422 xfs_nodes_rename_finish(table, *changes, to, 0);
423 *changes = NULL;
424 return rc;
425 }
426 }
427 }
428 return 0;
429}

431/* WHAT: Commit or cancel reserved rename keys. WHY: Mirror only server success.
432 * HOW: 1. Remove an overwritten destination. 2. Swap prepared keys on commit.
433 * 3. Return all old or cancelled string capacity. */
434void
435xfs_nodes_rename_finish(xfs_nodes *table, xfs_rename_change *changes,
436 const char *to, int commit)
437{
438 if (commit) {
439 xfs_nodes_unlink(table, to);
440 }
441 while (changes) {

443 if (commit) {

448 changes->replacement = old;

450 }

452 free(changes->replacement);
