1/*
2 * defsfile.c — load the SciTags experiment/activity registry (JSON).
3 *
4 * WHAT: Parse the scitags "defsfile" — the JSON registry that maps experiment
5 * and activity NAMES to their numeric ids — into an in-memory table the
6 * mapping layer (mapping.c) resolves config rules against. Mirrors XRootD's
7 * XRootD's packet-marking config loader.
8 *
9 * WHY: SciTags config is written in terms of names (experiment "atlas",
10 * activity "write"), but the wire/firefly carries numeric ids; the registry is
11 * the authoritative name→id source (published at scitags.org / by the WLCG).
12 *
13 * HOW: jansson (already linked, see config:70-77 and src/dashboard/api.c) parses
14 * the document {"experiments":[{"expName","expId","activities":[{"activityName",
15 * "activityId"}]}]}. Everything is copied into pool-allocated, NUL-terminated
16 * ngx_str_t so the jansson tree can be freed immediately. A missing file is
17 * NGX_DECLINED (not an error — only a problem if mappings reference names).
18 */

27/* Copy a jansson string into a pool-allocated, NUL-terminated ngx_str_t. */
28static ngx_int_t
29pmark_dup_json_str(ngx_pool_t *pool, json_t *js, ngx_str_t *out)
30{
31 const char *s;
32 size_t len;
33
34 if (!json_is_string(js)) {
35 return NGX_ERROR;
36 }
37 s = json_string_value(js);
38 len = ngx_strlen(s);
39
40 out->data = ngx_pnalloc(pool, len + 1);
41 if (out->data == NULL) {
42 return NGX_ERROR;
43 }
44 ngx_memcpy(out->data, s, len);
45 out->data[len] = '\0';
46 out->len = len;
47 return NGX_OK;
48}

51ngx_int_t
52brix_pmark_defsfile_load(const char *path, ngx_pool_t *pool,
53 ngx_array_t **out, ngx_log_t *log)
54{
55 json_t *root, *exps, *e;
56 json_error_t jerr;
57 ngx_array_t *defs;
58 size_t i;
59
60 *out = NULL;
61
62 if (path == NULL || *path == '\0') {
63 return NGX_DECLINED;
64 }
65 if (access(path, R_OK) != 0) {
66 return NGX_DECLINED; /* absent/unreadable → "no registry" */
67 }
68
69 root = json_load_file(path, 0, &jerr);
70 if (root == NULL) {
71 BRIX_LOG(NGX_LOG_ERR, log, 0,
72 "pmark: cannot parse defsfile \"%s\": %s (line %d)",
73 path, jerr.text, jerr.line);
74 return NGX_ERROR;
75 }
76
77 exps = json_object_get(root, "experiments");
78 if (!json_is_array(exps)) {
79 json_decref(root);
80 BRIX_LOG(NGX_LOG_ERR, log, 0,
81 "pmark: defsfile \"%s\" has no \"experiments\" array", path);
82 return NGX_ERROR;
83 }
84
85 defs = ngx_array_create(pool, json_array_size(exps),
86 sizeof(brix_pmark_exp_def_t));
87 if (defs == NULL) {
88 json_decref(root);
89 return NGX_ERROR;
90 }
91
92 json_array_foreach(exps, i, e) {
93 brix_pmark_exp_def_t *def;
94 json_t *acts, *a;
95 size_t j;
96
97 if (!json_is_object(e)) {
98 continue;
99 }
100
101 def = ngx_array_push(defs);
102 if (def == NULL) {
103 json_decref(root);
104 return NGX_ERROR;
105 }
106 ngx_memzero(def, sizeof(*def));
107
108 if (pmark_dup_json_str(pool, json_object_get(e, "expName"),
109 &def->exp_name) != NGX_OK)
110 {
111 def->exp_name.len = 0;
112 }
113 def->exp_id = (ngx_uint_t) json_integer_value(json_object_get(e, "expId"));
114
115 acts = json_object_get(e, "activities");
116 if (!json_is_array(acts)) {
117 continue;
118 }
119 def->activities = ngx_array_create(pool, json_array_size(acts),
120 sizeof(brix_pmark_named_t));
121 if (def->activities == NULL) {
122 json_decref(root);
123 return NGX_ERROR;
124 }
125
126 json_array_foreach(acts, j, a) {
127 brix_pmark_named_t *act;
128
129 if (!json_is_object(a)) {
130 continue;
131 }
132 act = ngx_array_push(def->activities);
133 if (act == NULL) {
134 json_decref(root);
135 return NGX_ERROR;
136 }
137 ngx_memzero(act, sizeof(*act));
138 if (pmark_dup_json_str(pool, json_object_get(a, "activityName"),
139 &act->name) != NGX_OK)
140 {
141 act->name.len = 0;
142 }
143 act->id = (ngx_uint_t)
144 json_integer_value(json_object_get(a, "activityId"));
145 }
146 }
147
148 json_decref(root);
149 *out = defs;
150 return NGX_OK;
151}

154ngx_uint_t
155brix_pmark_defs_exp_id(ngx_array_t *defs, ngx_str_t *name)
156{
157 brix_pmark_exp_def_t *d;
158 ngx_uint_t i;
159
160 if (defs == NULL || name == NULL) {
161 return 0;
162 }
163 d = defs->elts;
164 for (i = 0; i < defs->nelts; i++) {
165 if (d[i].exp_name.len == name->len
166 && ngx_strncmp(d[i].exp_name.data, name->data, name->len) == 0)
167 {
168 return d[i].exp_id;
169 }
170 }
171 return 0;
172}

188 }
189 {
190 brix_pmark_named_t *a = d[i].activities->elts;
191 for (j = 0; j < d[i].activities->nelts; j++) {
192 if (a[j].name.len == name->len
193 && ngx_strncmp(a[j].name.data, name->data, name->len) == 0)
194 {
195 return a[j].id;
196 }
197 }
198 }
199 }
200 return 0;
