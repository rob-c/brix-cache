1/*
2 * subprocess.h — capture a child process's stdout (shared).
3 *
4 * WHAT: fork/exec a command, drain its stdout into a caller buffer, reap it.
5 * WHY: the oidc-agent / RFC-8693 token fetch is done by fork/exec'ing
6 * `oidc-token` or `curl` and reading stdout — the SAME pipe→fork→dup2→
7 * drain→waitpid skeleton appears in the native client (credrefresh.c) and
8 * the module's TPC token paths (tpc/tpc_token.c, webdav/tpc_cred.c). Both
9 * run the fork SYNCHRONOUSLY (the module does it off the event loop), so —
10 * unusually for resilience code — the mechanism itself is genuinely shared.
11 * HOW: runs the command under a double-forked, reparented agent (the
12 * xfer_spawn.c shape) that waits for it and relays the raw wait status
13 * over a socketpair, so a host SIGCHLD handler (nginx's, on the worker's
14 * MAIN thread — a sigprocmask block is per-thread and does not reach it)
15 * can never reap the command first and leave a stolen status reading as
16 * "exit 0" (history-testing-and-incidents §24.15). The helper is
17 * status-agnostic — it returns the child's exit code + the raw bytes; each
18 * caller maps to its own error type and parses (JSON vs plain).
19 * ngx-free; libc only. (libxrdproto) Unit test: subprocess_unittest.c.
20 */

26/*
27 * One command run under the reparented agent, in its general form.
28 *
29 * out == NULL runs the command with the caller's own stdout (no capture pipe);
30 * timeout_ms > 0 arms a deadline the AGENT enforces — it SIGKILLs the command's
31 * whole process group (the command is a session leader) and the run reports
32 * -1 with errno == ETIMEDOUT. brix_subprocess_capture() is this struct with a
33 * buffer and no deadline.
34 */
35typedef struct {
36 char *const *argv; /* NULL-terminated; argv[0] resolved via PATH */
37 char *out; /* stdout capture buffer, or NULL to inherit */
38 size_t outsz; /* its size incl. the NUL (0 iff out == NULL) */
39 unsigned timeout_ms; /* 0 = wait as long as the command takes */
40} brix_subprocess_req_t;
41
42/*
43 * Run argv[] (argv[0] resolved via PATH), capturing up to outsz-1 bytes of its
44 * stdout into out (always NUL-terminated). *out_len = bytes captured (excl NUL);
45 * *exit_code = the child's exit status (WEXITSTATUS). Both out-params may be NULL.
46 * Returns 0 if the child ran to a normal exit (check *exit_code for its result),
47 * -1 on pipe/fork failure, if the child was killed by a signal, or if its exit
48 * status could not be obtained — never a fabricated 0.
49 */
50int brix_subprocess_capture(char *const argv[], char *out, size_t outsz,
51 size_t *out_len, int *exit_code);
52
53/*
54 * The general form of the above: optional capture, optional deadline. Returns 0
55 * if the command ran to a normal exit (check *exit_code), -1 otherwise — with
56 * errno == ETIMEDOUT when the deadline expired and the command's process group
57 * was killed, and errno == EINVAL for a malformed request. Both out-params may
58 * be NULL. Never a fabricated 0.
59 */
60int brix_subprocess_run(const brix_subprocess_req_t *req, size_t *out_len,
61 int *exit_code);
