2 * subprocess_unittest.c — standalone unit test for brix_subprocess_capture().
3 *
4 * gcc -Wall -Wextra -Werror -pthread -I src -o /tmp/subprocess_ut \
5 * src/core/compat/subprocess.c src/core/compat/subprocess_unittest.c \
6 * && /tmp/subprocess_ut
7 *
8 * Exit 0 = all checks pass. Verifies stdout capture + exit-code propagation
9 * (0/N), exec failure (127), kill-by-signal (-1), NUL-termination + truncation

66/* nginx's shape: ngx_signal_handler → ngx_process_get_status → waitpid(-1, WNOHANG). */
67static void
68reaper_handler(int signo)
69{
70 int st;
71
72 (void) signo;
73 while (waitpid(-1, &st, WNOHANG) > 0) {
74 reaper_stolen++;
75 }
76}

78/* The worker's main thread: SIGCHLD unblocked, and also polling — the worst case. */
79static void *
80reaper_thread(void *arg)
81{
82 sigset_t unblock;
83 struct timespec nap = { 0, 50000 };
84 int st;
85
86 (void) arg;
87 sigemptyset(&unblock);
88 sigaddset(&unblock, SIGCHLD);
89 pthread_sigmask(SIG_UNBLOCK, &unblock, NULL);
90 while (!reaper_stop) {
91 while (waitpid(-1, &st, WNOHANG) > 0) {
92 reaper_stolen++;
93 }
94 nanosleep(&nap, NULL);
95 }
96 return NULL;
97}

155/* ---- Verify the runner removes parent descriptors before exec ----
156 *
157 * WHAT: Opens a descriptor in the parent and confirms the command cannot see
158 * it through /proc; records a unit failure on setup or contract failure.
159 *
160 * WHY: A failed close_range fallback would expose worker sockets, logs, and
161 * request pipes to an operator command even when ordinary exit tests pass.
162 *
163 * HOW:
164 * 1. Open /dev/null in the parent at a known descriptor number.
165 * 2. Ask a shell run by the helper whether that descriptor survived exec.
166 * 3. Require a successful shell with a false existence test, then close it.
167 */
168static void
169check_descriptor_hygiene(void)
170{
171 char command[96];
172 char out[16];
173 int fd, ec = -1;
174
175 fd = open("/dev/null", O_RDONLY);
176 if (fd < 0) {
177 check("descriptor hygiene: open", -1, 0);
178 return;
179 }
180 snprintf(command, sizeof(command), "test ! -e /proc/$$/fd/%d", fd);
181 check("descriptor hygiene: runner rc",
182 run_sh(command, out, sizeof(out), NULL, &ec), 0);
183 check("descriptor hygiene: inherited fd closed", ec, 0);
184 close(fd);
185}

187static void
188check_hostile_reaper(void)
189{
190 struct sigaction sa;
191 pthread_t th;
192 sigset_t block;
193 char out[16];
194 int ec, wrong = 0, i;
195
196 memset(&sa, 0, sizeof(sa));
197 sa.sa_handler = reaper_handler;
198 sigemptyset(&sa.sa_mask);
199 sigaction(SIGCHLD, &sa, NULL);
200
201 /* This thread plays the nginx thread-pool worker: it never handles SIGCHLD.
202 * The reaper thread plays the worker's main thread, which does. */
203 sigemptyset(&block);
204 sigaddset(&block, SIGCHLD);
205 pthread_sigmask(SIG_BLOCK, &block, NULL);
206 pthread_create(&th, NULL, reaper_thread, NULL);
207
208 for (i = 0; i < REAPER_ROUNDS; i++) {
209 ec = -2;
210 if (run_sh("exit 7", out, sizeof(out), NULL, &ec) != 0 || ec != 7) {
211 wrong++;
212 }
213 }
214 reaper_stop = 1;
215 pthread_join(th, NULL);
216 signal(SIGCHLD, SIG_DFL);
217 pthread_sigmask(SIG_UNBLOCK, &block, NULL);
218
219 if (wrong != 0) {
220 printf("FAIL hostile reaper: %d of %d runs reported the wrong status "
221 "(reaper stole %d children)\n", wrong, REAPER_ROUNDS, (int) reaper_stolen);
222 fails++;
223 } else {
224 printf("ok hostile reaper (%d/%d exit 7; reaper collected %d)\n",
225 REAPER_ROUNDS, REAPER_ROUNDS, (int) reaper_stolen);
226 }
227
228 /* Reparent invariant: nothing is left for THIS process to reap. */
229 errno = 0;
230 {
231 pid_t w = waitpid(-1, NULL, WNOHANG);
232 if (!(w == -1 && errno == ECHILD)) {
233 printf("FAIL reparent invariant: waitpid=%d errno=%d "
234 "(a child was left for us to reap)\n", (int) w, errno);
235 fails++;
236 } else {
237 printf("ok reparent invariant (no reapable child)\n");
238 }
239 }
240}
