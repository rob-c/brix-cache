1/*
2 * staged_file.h — Atomic temp-file write lifecycle.
3 *
4 * Provides brix_staged_file_t (a struct tracking fd, tmp_path, active flag) and three
5 * operations: open → commit/abort. Used by S3 PUT, WebDAV PUT for crash-safe writes:
6 * 1) Open a unique temp file with O_EXCL inside root_canon
7 * 2) Write data to the fd
8 * 3) Commit (rename temp → final path) or abort (close + unlink)
9 */

20/*
21 * brix_staged_file_t — Tracks the lifecycle of a staged (temp) file.
22 *
23 * Fields:
24 * fd — open file descriptor for the temp file (NGX_INVALID_FILE when closed)
25 * active — 1 while the temp file is open and data can be written; 0 after commit/abort
26 * tmp_path — filesystem path of the temp file (used by commit to rename, abort to unlink)
27 */
28typedef struct {
29 ngx_fd_t fd;
30 ngx_flag_t active;
31 char tmp_path[PATH_MAX];
32 mode_t final_mode; /* intended mode to publish at commit; the temp is
33 * kept 0600 during the write so a peer mapped uid
34 * cannot read an in-progress upload. */
35} brix_staged_file_t;

37/*
38 * brix_staged_open_req_t — the pure (side-effect-free) inputs that describe WHERE
39 * and HOW a staged temp is opened, bundled so the open entry points stay within
40 * the argument gate.
41 *
42 * WHAT: the confinement root + destination path + creation mode shared by both
43 * staged-open variants, plus the variant-specific selectors (open_flags /
44 * attempts for the random O_EXCL temp; principal / stage_dir for the
45 * deterministic identity-keyed resume partial).
46 * WHY: threading these as discrete parameters pushed brix_staged_open (7 args)
47 * and brix_staged_open_resume (8 args) past the 5-argument budget. They are
48 * all read-only request description — no hidden global state — so gathering
49 * them into one value bundle keeps the data flow explicit while the
50 * side-effecting args (log, the output staged/cur_size) stay direct params.
51 * HOW: callers fill the fields relevant to their variant (unused fields are
52 * ignored by that variant) and pass a pointer; the function reads it and
53 * never mutates it.
54 *
55 * Fields:
56 * root_canon — canonical root the temp must reside within
57 * final_path — destination path (derives the temp name)
58 * mode — file creation mode to publish at commit
59 * open_flags — [open] base open flags (O_EXCL added internally)
60 * attempts — [open] max EEXIST retries (0 → 16)
61 * principal — [resume] identity keying the deterministic partial name
62 * stage_dir — [resume] fast-device stage dir, or NULL to stage under root
63 */
64typedef struct {
65 const char *root_canon;
66 const char *final_path;
67 mode_t mode;
68 int open_flags;
69 ngx_uint_t attempts;
70 const char *principal;
71 const char *stage_dir;
72} brix_staged_open_req_t;

86/* brix_commit_staged() — atomically publish a staged file onto its final path,
87 * across filesystems (rename fast-path; copy+rename on EXDEV for a fast-cache
88 * stage dir on a different device than the storage). fd = open staged fd (for
89 * the durability fsync) or NGX_INVALID_FILE. final_mode = the mode to publish
90 * on the committed object (the temp is written 0600); pass 0 to leave the temp's
91 * current mode untouched (callers whose temp already carries the intended mode).
92 * flags: BRIX_COMMIT_RELAXED skips the pre-publish data fsync (phase-51 C1) for
93 * a same-filesystem rename commit — the operator opted out of per-close
94 * durability via `brix_durable_commit off` (stock-XRootD close semantics). The
95 * cross-device and cross-backend movers keep their own fsyncs: there the copy
96 * itself is the publish and a torn copy WOULD be visible.
97 * See staged_file.c. */
98#define BRIX_COMMIT_RELAXED 0x1u

138/*
139 * brix_publish_dirsync — the phase-107 C3 durable-publish barrier.
140 *
141 * Make the DIRECTORY ENTRY of the just-published object durable: open the
142 * PARENT directory of `final_path` (absolute under root_canon, or already
143 * root-relative) through the confined-fd machinery — O_RDONLY|O_DIRECTORY,
144 * never O_PATH (fsync on an O_PATH fd is EBADF; that inert cast-away barrier
145 * is the defect this helper replaces) and never by re-resolving the full path
146 * by name — fsync it, close it. `rootfd` is the confinement anchor (an O_PATH
147 * root fd is fine as an openat anchor); pass -1 to have the helper open and
148 * close a transient anchor on root_canon. Returns NGX_OK, or NGX_ERROR with
149 * errno (EIO fsync failed, ENOENT the parent vanished). Callers gate on
150 * brix_vfs_backend_durable(root_canon) and treat failure as failing the
151 * publish (the name is already visible — report EIO, log crit).
152 */
153ngx_int_t brix_publish_dirsync(ngx_log_t *log, int rootfd,
154 const char *root_canon, const char *final_path);
