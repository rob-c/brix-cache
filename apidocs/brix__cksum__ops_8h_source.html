10/* Parse one line of the sha256sum-style tree-audit manifest: "<hex> <rel>\n".
11 * Two spaces separate the hex digest from the relative path; the trailing newline
12 * (or CRLF) is stripped. Returns 0 on success with hex[hexsz] and rel[relsz]
13 * filled (NUL-terminated); returns -1 when the line is malformed (no double-space

48/* Which recorded-checksum sources to consult. */
49typedef enum {
50 XRDC_CKV_AUTO = 0, /* cache sidecars (.cinfo/.meta) AND storage (xattr/.cks) */
51 XRDC_CKV_CACHE, /* proxy cache only: <file>.cinfo / <file>.meta cks fields */

55/* Outcome of a verification. */
56typedef enum {
57 XRDC_CKV_OK = 0, /* a recorded checksum was found and matches */
58 XRDC_CKV_MISMATCH, /* recorded != recomputed (corruption) */
59 XRDC_CKV_NO_RECORD, /* no recorded checksum found for this file/algo */
60 XRDC_CKV_UNSUPPORTED, /* recorded with an algorithm this engine cannot compute */
61 XRDC_CKV_ERROR /* I/O / access error */
62} brix_ckv_result;

78/* ---- cli_cksum.c (shared checksum-tool front-end) ---- */
79/* Process-exit conventions shared by the front-end tools (phase-49):
80 * USAGE — bad arguments / URL parse / local open (was the bare `return 50`)
81 * IO — runtime I/O failure
82 * AUTH — authentication/authorization failure

84 * code; these are for the cases that never produced a status. */
85#define XRDC_EXIT_USAGE 50
86#define XRDC_EXIT_IO 51
87#define XRDC_EXIT_AUTH 53

89/* The whole body of xrdcrc32c / xrdcrc64 / xrdadler32: checksum a LOCAL file or a
