1/*
2 * rfile_fast.c — pipelined (many-reads-in-flight) fast path for brix_rfile.
3 *
4 * WHAT: brix_rfile_stream_fast() streams a byte range of an already-open
5 * resilient file to a brix_rfile_sink_fn while keeping several kXR_read
6 * requests in flight on ONE connection, delivering the chunks to the sink
7 * in strict file order. It returns 0 (range complete / EOF / sink asked
8 * to stop), BRIX_RFILE_FAST_OFF (nothing was attempted — the caller must
9 * run its ordinary serial loop), or -1 with *moved holding the bytes that
10 * WERE delivered, so the caller resumes serially from there.
11 * WHY: the serial pump (brix_rfile_pump / transfer_pump) issues exactly one
12 * kXR_read and then blocks on the reply, so the link idles for a whole
13 * round trip per chunk and the sink write never overlaps the next read.
14 * On a fast link that is the single largest cost in the client. Keeping
15 * N reads outstanding hides both the RTT and the server's per-request
16 * work, which is what every other fast XRootD client does.
17 * HOW: the existing async transport core (core/aio) already owns an epoll loop
18 * that pipelines many requests over one socket and demultiplexes replies
19 * by streamid — this file only drives it. The connection is handed to a
20 * loop with brix_aconn_attach, a ring of `depth` slots each carries one
21 * outstanding kXR_read, the loop thread parks each reply in its slot and
22 * signals, and the calling thread consumes slots strictly in ring order.
23 * On teardown the connection is detached and its fd put back in blocking
24 * mode so the ordinary synchronous helpers own it again.
25 *
26 * Fallback: a connection with GSI request signing active cannot go async (see
27 * core/aio/aio.h), and paged/inline-compressed reads have their own framing —
28 * all three return BRIX_RFILE_FAST_OFF so the serial path handles them.
29 *
30 * This file is the session around the pipeline: eligibility, the env tunables,
31 * attach/detach and the socket's blocking mode. The ring and the fill → wait →
32 * deliver loop over it are in rfile_fast_ring.c, behind rfile_fast.h.
33 */

51/* Per-request read size and how many of them ride the wire at once. 4 MiB × 4
52 * keeps ~16 MiB outstanding: enough to cover a LAN round trip plus the server's
53 * page-cache read at multi-GiB/s, while staying small enough that a chunk is
54 * handed to the sink long before the whole file is buffered. */

62/* ---- Read an unsigned tunable from the environment ----
63 *
64 * WHAT: Returns the value of environment variable `name` clamped to
65 * [lo, hi], or `dflt` when it is unset, empty or unparseable.
66 *
67 * WHY: The chunk size and pipeline depth are the two knobs whose best value
68 * depends on the link (RTT × bandwidth) and on the server's read-ahead.
69 * Exposing them as environment overrides lets a deployment tune without a
70 * rebuild, while the clamp keeps a typo from asking for a 4 GiB request.
71 *
72 * HOW: 1. getenv; return dflt when absent or empty.
73 * 2. strtoul; return dflt when nothing was consumed.
74 * 3. Clamp into [lo, hi] and return.
75 */
76static unsigned
77fast_env_u(const char *name, unsigned dflt, unsigned lo, unsigned hi)
78{
79 const char *text = getenv(name);
80 char *end = NULL;
81 unsigned long value;
82
83 if (text == NULL || text[0] == '\0') {
84 return dflt;
85 }
86 value = strtoul(text, &end, 10);
87 if (end == text) {
88 return dflt;
89 }
90 if (value < (unsigned long) lo) {
91 return lo;
92 }
93 if (value > (unsigned long) hi) {
94 return hi;
95 }
96 return (unsigned) value;
97}

100/* ---- One line of pipeline diagnostics, when asked for ----
101 *
102 * WHAT: Writes "brix pipeline: <what> ..." to stderr when $XRDC_PIPELINE_TRACE
103 * is set, and nothing otherwise.
104 *
105 * WHY: Whether a given transfer took the pipelined path — and with what depth
106 * and chunk, or for what reason it declined — is invisible from the
107 * outside: both paths produce identical bytes. A single opt-in line turns
108 * "is the fast path even running?" from a rebuild into one run, which is
109 * exactly the question every throughput investigation starts with.
110 *
111 * HOW: getenv guard, then fprintf; `detail` is a number whose meaning depends
112 * on `what` (bytes, depth, or the status code that declined it).
113 */
114static void
115fast_trace(const char *what, long long a, long long b)
116{
117 if (getenv("XRDC_PIPELINE_TRACE") != NULL) {
118 fprintf(stderr, "brix pipeline: %s %lld %lld\n", what, a, b);
119 }
120}

125/* ---- Poison a session that is no longer frame-aligned ----
126 *
127 * WHAT: Shuts the socket down in both directions, ignoring failures.
128 *
129 * WHY: When the pipelined reader has to abandon requests whose replies are
130 * still on the wire, the next synchronous read would take those bytes for
131 * the answer to its own request — silent cross-talk, and with a hostile or
132 * merely confused peer a frame boundary the caller never validated. There
133 * is no way to re-synchronise a stream whose remaining length is unknown,
134 * so the only safe move is to make the session unusable: every later
135 * operation then fails with an ordinary transport error, which is exactly
136 * what the serial pump's reconnect+reopen+resume loop already handles.
137 *
138 * HOW: shutdown(2) both directions; a negative fd is a no-op. The fd itself is
139 * left open — the connection is the caller's to close or reconnect.
140 */
141static void
142fast_poison_session(int fd)
143{
144 if (fd >= 0) {
145 (void) shutdown(fd, SHUT_RDWR);
146 }
147}

150/* ---- Put a socket back into blocking mode ----
151 *
152 * WHAT: Clears O_NONBLOCK on `fd`, ignoring failures.
153 *
154 * WHY: brix_aconn_attach switches the connection's fd to non-blocking and owns
155 * it for as long as it is attached; the synchronous helpers that run
156 * afterwards (close, checksum query, the serial resume pump) assume a
157 * blocking socket. Restoring the flag at exactly one place — the fast
158 * path's teardown — keeps that contract local to the code that broke it.
159 *
160 * HOW: F_GETFL then F_SETFL without O_NONBLOCK; a negative fd is a no-op.
161 */
162static void
163fast_restore_blocking(int fd)
164{
165 int flags;
166
167 if (fd < 0) {
168 return;
169 }
170 flags = fcntl(fd, F_GETFL, 0);
171 if (flags >= 0) {
172 (void) fcntl(fd, F_SETFL, flags & ~O_NONBLOCK);
173 }
174}

177/* ---- Is this file eligible for the pipelined path? ----
178 *
179 * WHAT: Returns 1 when `rf` is a plain (non-paged, non-compressed) read handle
180 * on a live connection, 0 otherwise.
181 *
182 * WHY: kXR_pgread replies use kXR_status framing with per-page CRC32c and
183 * inline-compressed reads arrive as codec frames; both are decoded by
184 * their own synchronous readers, not by the generic reply path this file
185 * drives. Rather than duplicate that decoding, those handles keep the
186 * serial pump — correctness first, speed only where it is free.
187 *
188 * HOW: Require a connection, pgrw == 0 and a zero read_codec.
189 */
190static int
191fast_eligible(const brix_rfile *rf)
192{
193 return rf != NULL && rf->c != NULL && rf->pgrw == 0
194 && rf->f.read_codec == 0;
195}

198/* ---- Stream a range with several reads in flight ----
199 *
200 * WHAT: Public entry point. Streams [offset, offset+limit) (or to EOF when
201 * limit < 0) of the open resilient file `rf` to `sink`, keeping several
202 * kXR_reads outstanding. Returns 0 on a complete range / EOF / sink stop,
203 * BRIX_RFILE_FAST_OFF when the fast path was not attempted at all, or -1
204 * with *st set. *moved (never NULL-checked away) always reports the bytes
205 * actually delivered, so a caller can resume serially after a failure.
206 *
207 * WHY: The serial pump leaves one request on the wire at a time; this is the
208 * whole point of the file (see the header block). Returning the partial
209 * progress instead of retrying internally is deliberate: the serial pump
210 * already owns reconnect + reopen + adaptive shrink, so the fast path
211 * stays a pure accelerator and there is exactly one retry policy.
212 *
213 * HOW: 1. Reject ineligible handles and zero-length ranges up front.
214 * 2. Create a loop and attach the connection (signing-active connections
215 * are refused here — that is the documented async limitation).
216 * 3. Size the ring, run the pipeline, then close the aconn (which fails
217 * anything still outstanding) and destroy the loop.
218 * 4. Restore blocking mode, sweep the ring, report moved + rc.
219 */
220int
221brix_rfile_stream_fast(brix_rfile *rf, int64_t offset, int64_t limit,
222 brix_rfile_sink_fn sink, void *arg, int64_t *moved,
223 brix_status *st)
224{
225 brix_loop *loop;
226 brix_aconn *ac;
227 fast_ring ring;
228 fast_state state;
229 fast_job job;
230 int rc;
231 int fd;
232 int aligned;
233
234 if (moved != NULL) {
235 *moved = 0;
236 }
237 if (!fast_eligible(rf) || sink == NULL || offset < 0 || limit == 0
238 || fast_env_u("XRDC_PIPELINE", 1, 0, 1) == 0) {
239 fast_trace("off (ineligible) pgrw/codec", rf ? rf->pgrw : -1,
240 rf ? rf->f.read_codec : -1);
241 return BRIX_RFILE_FAST_OFF;
242 }
243
244 memset(&job, 0, sizeof(job));
245 job.rf = rf;
246 job.sink = sink;
247 job.arg = arg;
248 job.offset = offset;
249 job.limit = limit;
250 job.chunk = fast_env_u("XRDC_PIPELINE_CHUNK", FAST_CHUNK_DEFAULT,
251 FAST_CHUNK_MIN, FAST_CHUNK_MAX);
252 job.depth = fast_env_u("XRDC_PIPELINE_DEPTH", FAST_DEPTH_DEFAULT,
253 1, FAST_DEPTH_MAX);
254 job.deadline_ms = rf->max_stall_ms;
255
256 loop = brix_loop_create(st);
257 if (loop == NULL) {
258 fast_trace("off (no loop)", 0, 0);
259 brix_status_clear(st);
260 return BRIX_RFILE_FAST_OFF;
261 }
262 fd = rf->c->io.fd;
263 ac = brix_aconn_attach(loop, rf->c, st);
264 if (ac == NULL) {
265 fast_trace("off (attach refused)", 0, 0);
266 brix_loop_destroy(loop);
267 brix_status_clear(st);
268 return BRIX_RFILE_FAST_OFF;
269 }
270 if (fast_ring_init(&ring, job.depth, job.chunk, st) != 0) {
271 brix_aconn_close(ac);
272 brix_loop_destroy(loop);
273 fast_restore_blocking(fd);
274 return -1;
275 }
276
277 fast_trace("on depth/chunk", (long long) job.depth, (long long) job.chunk);
278 memset(&state, 0, sizeof(state));
279 rc = fast_run(ac, &job, &ring, &state, st);
280 /* Only a clean finish can leave the socket frame-aligned; on any failure the
281 * still-outstanding replies are abandoned in the pipe (see fast_drain). */
282 aligned = (rc == 0) ? fast_drain(&job, &ring, &state) : 0;
283
284 brix_aconn_close(ac); /* synchronous: fails every still-open request */
285 brix_loop_destroy(loop);
286 fast_restore_blocking(fd);
287 fast_ring_free(&ring);
288
289 if (!aligned) {
290 fast_poison_session(fd); /* replies still in flight: never read them */
291 }
292 fast_trace("done rc/moved", (long long) rc, (long long) state.moved);
293 if (moved != NULL) {
294 *moved = state.moved;
295 }
296 return rc;
297}
