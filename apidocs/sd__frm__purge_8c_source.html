1/*
2 * sd_frm_purge.c — phase-115 W3.2: the tape-buffer purge engine.
3 *
4 * WHAT: One LRU pass over a tape:// tier's online buffer (<base>/.online)
5 * that releases migrated, unpinned, cold copies until the filesystem
6 * watermark and/or the owned-bytes cap are satisfied. Pure libc + the
7 * instance's MSS adapter; the nginx timer that paces it lives in
8 * core/config/process_frm_purge.c.
9 *
10 * WHY: The WLCG Tape REST / kXR_prepare surface recalls into the buffer and
11 * staged writes land there before migrating, but nothing ever released
12 * a copy on its own — an operator either ran a cron over the directory
13 * or watched the buffer fill (parity audit §3.4). The cache tier's LRU
14 * reaper cannot do it: the buffer is the BACKEND's, invisible above the
15 * seam, and a cache-style unlink of an un-migrated copy loses data.
16 *
17 * HOW: 1. Take a non-blocking flock on <online>/.brix-purge.lock so two
18 * workers (or a worker and an operator's manual run) never race.
19 * 2. nftw(FTW_PHYS|FTW_MOUNT) the online root collecting regular files
20 * (path, size, last-touch = max(atime, mtime)); symlinks are counted
21 * and never followed, the lock file is skipped.
22 * 3. Compute the byte target: arm 1 = used - lo*total/1e6 when the tick
23 * saw occupancy > hi; arm 2 = owned - max_bytes; need = max of the
24 * armed arms (<= 0 => nothing to do).
25 * 4. Sort by last-touch ascending and walk: skip young / pinned /
26 * on_tape != 1, else adapter->purge(key). Stop once `need` is met.
27 * 5. Book brix_frm_purge_total per release + brix_vfs_evict_bytes_total
28 * {driver="frm"} once; NOTICE one summary line.
29 * 2.0 F4: between 3 and 4 sd_frm_purge_policy.c tags every candidate
30 * with its brix_frm_purge_policy rule (a per-group owned-bytes arm
31 * with its own hold) and may run the operator's policy program
32 * once; the walk asks it per copy and books held/unapproved.
33 *
34 * Raw filesystem calls here are legitimate: this TU is UNDER src/fs/backend/
35 * (invariant 12), operating on the backend's own private buffer directory.
36 */

84static int
85frm_purge_push(frm_purge_scan_t *s, const char *rel, const struct stat *sb)
86{
87 frm_purge_cand_t *c;
88
89 if (s->n >= FRM_PURGE_MAX_CANDIDATES) {
90 return 0; /* cap: next tick continues */
91 }
92 if (s->n == s->cap) {
93 size_t ncap = s->cap ? s->cap * 2 : 256;
94 frm_purge_cand_t *nv = realloc(s->v, ncap * sizeof(*nv));
95
96 if (nv == NULL) {
97 s->oom = 1;
98 return -1;
99 }
100 s->v = nv;
101 s->cap = ncap;
102 }
103 c = &s->v[s->n];
104 c->rel = strdup(rel);
105 if (c->rel == NULL) {
106 s->oom = 1;
107 return -1;
108 }
109 c->size = (uint64_t) sb->st_size;
110 c->touched = (sb->st_atime > sb->st_mtime) ? sb->st_atime : sb->st_mtime;
111 c->rule = -1;
112 c->approved = 0;
113 s->n++;
114 return 0;
115}

117static int
118frm_purge_visit(const char *path, const struct stat *sb, int flag,
119 struct FTW *ftw)
120{
121 frm_purge_scan_t *s = frm_purge_active_scan;
122 const char *rel;
123
124 (void) ftw;
125 if (s == NULL) {
126 return -1;
127 }
128 if (flag == FTW_SL || flag == FTW_SLN) {
129 s->symlinks++; /* never followed (FTW_PHYS) */
130 return 0;
131 }
132 if (flag != FTW_F || !S_ISREG(sb->st_mode)) {
133 return 0;
134 }
135 rel = path + s->root_len; /* "/<key>" */
136 if (strncmp(rel, FRM_PURGE_PRIVATE_PREFIX,
137 sizeof(FRM_PURGE_PRIVATE_PREFIX) - 1) == 0)
138 {
139 return 0; /* the engine's own files */
140 }
141 s->owned += (uint64_t) sb->st_size;
142 return frm_purge_push(s, rel, sb);
143}

145static int
146frm_purge_cmp_lru(const void *a, const void *b)
147{
148 const frm_purge_cand_t *x = a, *y = b;
149
150 if (x->touched != y->touched) {
151 return (x->touched < y->touched) ? -1 : 1;
152 }
153 return strcmp(x->rel, y->rel);
154}

169/* Bytes the pass must release: the larger demand of the armed arms, 0 when
170 * neither arm is armed or both are already satisfied. */
171static uint64_t
172frm_purge_need(const brix_sd_frm_purge_policy_t *pol, uint64_t owned)
173{
174 uint64_t need = 0;
175
176 if (pol->hi_ppm > 0 && pol->hi_ppm < 1000000 && pol->fs_total > 0) {
177 uint64_t occ_ppm = (uint64_t) ((pol->fs_used * 1000000.0) / pol->fs_total);
178 uint64_t lo_bytes = (uint64_t) (pol->fs_total * (pol->lo_ppm / 1000000.0));
179
180 if (occ_ppm > pol->hi_ppm && pol->fs_used > lo_bytes) {
181 need = pol->fs_used - lo_bytes;
182 }
183 }
184 if (pol->max_bytes > 0 && owned > (uint64_t) pol->max_bytes) {
185 uint64_t cap_need = owned - (uint64_t) pol->max_bytes;
186
187 if (cap_need > need) {
188 need = cap_need;
189 }
190 }
191 return need;
192}

194/* Classify + release one candidate; returns the bytes freed (0 when skipped). */
195static uint64_t
196frm_purge_one(sd_frm_state *st, const brix_sd_frm_purge_policy_t *pol,
197 const frm_purge_cand_t *c, time_t now, brix_sd_frm_purge_report_t *rep,
198 const frm_purge_policy_t *pp)
199{
200 int on_tape;
201
202 if (now - c->touched < pol->min_age_s) {
203 rep->young++;
204 return 0;
205 }
206 if (now - c->touched < frm_purge_policy_hold(pol, c)) {
207 rep->held++; /* 2.0 F4 rule hold */
208 return 0;
209 }
210 if (pol->is_pinned != NULL && pol->is_pinned(pol->ud, c->rel)) {
211 rep->pinned++;
212 return 0;
213 }
214 /* A NULL probe means the adapter cannot tell; ONLINE copies then exist only
215 * via a completed staged commit (sd_frm_staged.c migrates before publish)
216 * or a recall, so the invariant ONLINE => on tape holds and we proceed. */
217 on_tape = (st->mss->on_tape != NULL) ? st->mss->on_tape(st->mss_ctx, c->rel)
218 : 1;
219 if (on_tape != 1) {
220 rep->unmigrated++;
221 return 0;
222 }
223 if (!frm_purge_policy_allows(pp, pol, c)) {
224 rep->unapproved++; /* 2.0 F4 polprog rule */
225 return 0;
226 }
227 if (st->mss->purge == NULL || st->mss->purge(st->mss_ctx, c->rel) != 0) {
228 rep->failed++;
229 return 0;
230 }
231 rep->evicted++;
232 BRIX_FRM_METRIC_INC(purge_total);
233 return c->size;
234}

236/* LRU walk: a copy is considered while the export-wide `need` is unmet or
237 * its own rule still wants bytes (2.0 F4); the walk ends once neither holds. */
238static void
239frm_purge_release(sd_frm_state *st, const brix_sd_frm_purge_policy_t *pol,
240 frm_purge_scan_t *s, uint64_t need, brix_sd_frm_purge_report_t *rep,
241 frm_purge_policy_t *pp)
242{
243 time_t now = time(NULL);
244 size_t i;
245
246 qsort(s->v, s->n, sizeof(s->v[0]), frm_purge_cmp_lru);
247 for (i = 0; i < s->n; i++) {
248 const frm_purge_cand_t *c = &s->v[i];
249 uint64_t freed;
250
251 if (rep->bytes >= need && !frm_purge_policy_wants(pp, c)) {
252 if (!frm_purge_policy_pending(pp)) {
253 break;
254 }
255 continue;
256 }
257 freed = frm_purge_one(st, pol, c, now, rep, pp);
258 if (freed > 0) {
259 rep->bytes += freed;
260 frm_purge_policy_account(pp, c);
261 }
262 }
263}

265static int
266frm_purge_lock(const char *online, char *lockpath, size_t cap, ngx_log_t *log)
267{
268 int fd;
269
270 if (snprintf(lockpath, cap, "%s/" FRM_PURGE_LOCK_NAME, online) >= (int) cap) {
271 errno = ENAMETOOLONG;
272 return -1;
273 }
274 fd = open(lockpath, O_RDWR | O_CREAT | O_CLOEXEC | O_NOFOLLOW, 0600);
275 if (fd < 0) {
276 ngx_log_error(NGX_LOG_ERR, log, errno,
277 "brix: tape purge cannot open lock \"%s\"", lockpath);
278 return -1;
279 }
280 if (flock(fd, LOCK_EX | LOCK_NB) != 0) {
281 int saved = errno;
282
283 close(fd);
284 errno = saved;
285 return -2;
286 }
287 return fd;
288}

290static void
291frm_purge_log(ngx_log_t *log, const char *online,
292 const brix_sd_frm_purge_policy_t *pol, const brix_sd_frm_purge_report_t *rep)
293{
294 ngx_uint_t occ_before = 0, occ_after = 0;
295 ngx_uint_t level = rep->evicted ? NGX_LOG_NOTICE : NGX_LOG_INFO;
296
297 if (pol->fs_total > 0) {
298 uint64_t used_after = (pol->fs_used > rep->bytes) ? pol->fs_used - rep->bytes : 0;
299
300 occ_before = (ngx_uint_t) ((pol->fs_used * 1000000.0) / pol->fs_total);
301 occ_after = (ngx_uint_t) ((used_after * 1000000.0) / pol->fs_total);
302 }
303 ngx_log_error(level, log, 0,
304 "brix: tape purge \"%s\": released %ui file(s), %uL bytes "
305 "(occupancy %ui -> %ui ppm, owned %uL -> %uL bytes; skipped young=%ui "
306 "pinned=%ui unmigrated=%ui symlink=%ui failed=%ui)",
307 online, rep->evicted, rep->bytes, occ_before, occ_after,
308 rep->owned_before, rep->owned_after, rep->young, rep->pinned,
309 rep->unmigrated, rep->symlinks, rep->failed);
310}

312ngx_int_t
313brix_sd_frm_purge(brix_sd_instance_t *inst,
314 const brix_sd_frm_purge_policy_t *pol, brix_sd_frm_purge_report_t *rep,
315 ngx_log_t *log)
316{
317 char online[PATH_MAX], lockpath[PATH_MAX];
318 frm_purge_scan_t scan;
319 frm_purge_policy_t pp;
320 uint64_t need;
321 int lockfd, rc;
322
323 ngx_memzero(rep, sizeof(*rep));
324 if (brix_sd_frm_online_root(inst, online, sizeof(online)) != NGX_OK) {
325 return NGX_ERROR;
326 }
327 if (access(online, F_OK) != 0 && errno == ENOENT) {
328 return NGX_OK; /* nothing recalled or staged yet: no buffer */
329 }
330 lockfd = frm_purge_lock(online, lockpath, sizeof(lockpath), log);
331 if (lockfd == -2) {
332 return NGX_DECLINED;
333 }
334 if (lockfd < 0) {
335 return NGX_ERROR;
336 }
337
338 ngx_memzero(&scan, sizeof(scan));
339 scan.root_len = strlen(online);
340 frm_purge_active_scan = &scan;
341 rc = nftw(online, frm_purge_visit, FRM_PURGE_NFTW_FDS, FTW_PHYS | FTW_MOUNT);
342 frm_purge_active_scan = NULL;
343 if (rc != 0 || scan.oom) {
344 ngx_log_error(NGX_LOG_ERR, log, scan.oom ? ENOMEM : errno,
345 "brix: tape purge walk of \"%s\" failed", online);
346 frm_purge_scan_free(&scan);
347 close(lockfd);
348 return NGX_ERROR;
349 }
350
351 rep->symlinks = scan.symlinks;
352 rep->owned_before = scan.owned;
353 if (frm_purge_policy_init(&pp, pol, &scan) != 0) {
354 ngx_log_error(NGX_LOG_ERR, log, ENOMEM,
355 "brix: tape purge of \"%s\": cannot size the policy table", online);
356 frm_purge_scan_free(&scan);
357 close(lockfd);
358 return NGX_ERROR;
359 }
360 need = frm_purge_need(pol, scan.owned);
361 if (need > 0 || frm_purge_policy_pending(&pp)) {
362 frm_purge_policy_consult(&pp, pol, &scan, online, need, log);
363 frm_purge_release(SD_FRM_ST(inst), pol, &scan, need, rep, &pp);
364 }
365 rep->owned_after = scan.owned - rep->bytes;
366 if (rep->bytes > 0) {
367 brix_metric_vfs_evict("frm", rep->bytes);
368 }
369 frm_purge_log(log, online, pol, rep);
370 frm_purge_policy_log(&pp, pol, online, rep, log);
371
372 frm_purge_policy_free(&pp);
373 frm_purge_scan_free(&scan);
374 close(lockfd); /* releases the flock */
375 return NGX_OK;
376}
