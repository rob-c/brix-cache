9/* Issue a WILLNEED read-ahead hint for [offset, offset+length) on fd through
10 * the Storage Driver seam (phase-56 B-2). Best-effort; a no-op on bad args
11 * or a driver without the read_advise slot. */
12void
13brix_prefetch_fd_range(ngx_log_t *log, int fd, off_t offset, size_t length)
14{
15 brix_sd_obj_t obj;
16
17 if (fd < 0 || offset < 0 || length < BRIX_READ_PREFETCH_MIN) {
18 return;
19 }
20
21 brix_sd_posix_wrap(&obj, fd);
22 if (obj.driver->read_advise != NULL
23 && obj.driver->read_advise(&obj, offset, length, BRIX_SD_ADV_WILLNEED)
24 != NGX_OK)
25 {
26 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, log, errno,
27 "brix: WILLNEED read-ahead hint ignored: %s",
28 strerror(errno));
29 }
30}

54/* ---- Validate a read-file prefetch request and compute its clamped end ----
55 *
56 * WHAT: Applies the argument guards for brix_prefetch_read_file() and, when the
57 * request is prefetchable, writes the clamped end offset [offset, end) to
58 * *end_out and returns 1. Returns 0 (with *end_out untouched) when the handle
59 * is invalid, the read is too small, the offset+length would overflow off_t, or
60 * the range collapses to nothing after clamping to file_size.
61 *
62 * WHY: Isolating the guard/clamp arithmetic keeps the offset/length math (an
63 * overflow-safe hot-path calculation) in one small, reviewable place and lets
64 * the orchestrator read as a flat sequence of steps without a nested guard
65 * ladder.
66 *
67 * HOW:
68 * 1. Reject a null handle, closed fd, negative offset, or sub-threshold read.
69 * 2. Reject when offset + length would overflow NGX_MAX_OFF_T_VALUE.
70 * 3. Compute end = offset + length, clamping down to file_size when known.
71 * 4. Reject a range that clamped away to <= offset; otherwise publish end.
72 */
73static ngx_flag_t
74brix_prefetch_read_end(brix_file_t *file, off_t offset, size_t length,
75 off_t file_size, off_t *end_out)
76{
77 off_t end;
78
79 /* A handle is prefetchable through its raw fd (POSIX fadvise) OR through
80 * its driver object (read_advise — e.g. the partial-cache background
81 * prefetch); reject only when it has neither. */
82 if (file == NULL || (file->fd < 0 && file->sd_obj.driver == NULL)
83 || offset < 0 || length < BRIX_READ_PREFETCH_MIN)
84 {
85 return 0;
86 }
87
88 if ((off_t) length > NGX_MAX_OFF_T_VALUE - offset) {
89 return 0;
90 }
91
92 end = offset + (off_t) length;
93 if (file_size > 0 && end > file_size) {
94 end = file_size;
95 }
96 if (end <= offset) {
97 return 0;
98 }
99
100 *end_out = end;
101 return 1;
102}

104/* ---- Compute the windowed WILLNEED hint range for a sequential read ----
105 *
106 * WHAT: For a read that continues the handle's sequential stream, writes the
107 * read-ahead hint range [*hint_start, *hint_end) that should be issued ahead of
108 * end. When the handle's existing read_ahead_end already extends past the
109 * low-water mark (end + LOW_WATER), no new hint is needed and both outputs are
110 * set equal (an empty range the caller skips).
111 *
112 * WHY: The sequential windowing — low-water suppression, start clamped to the
113 * already-prefetched frontier, and end extended by the read-ahead window and
114 * clamped to file_size — is the subtle part of the prefetch policy. Extracting
115 * it as a pure computation (state read from file, results returned by pointer,
116 * no I/O) keeps the arithmetic exact and independently reviewable.
117 *
118 * HOW:
119 * 1. Compute the low-water mark end + LOW_WATER (saturating at off_t max).
120 * 2. If read_ahead_end is already past low-water, emit an empty range (skip).
121 * 3. Start the hint at the further of read_ahead_end (the prefetched frontier)
122 * and offset.
123 * 4. End the hint at end + WINDOW (saturating), clamped down to file_size.
124 */
125static void
126brix_prefetch_seq_window(brix_file_t *file, off_t offset, off_t end,
127 off_t file_size, off_t *hint_start, off_t *hint_end)
128{
129 off_t low_water_end;
130
131 if (end > NGX_MAX_OFF_T_VALUE - BRIX_READ_PREFETCH_LOW_WATER) {
132 low_water_end = NGX_MAX_OFF_T_VALUE;
133 } else {
134 low_water_end = end + BRIX_READ_PREFETCH_LOW_WATER;
135 }
136
137 if (file->read_ahead_end > low_water_end) {
138 /* Already prefetched past the low-water mark; issue no new hint. */
139 *hint_start = 0;
140 *hint_end = 0;
141 return;
142 }
143
144 *hint_start = (file->read_ahead_end > offset)
145 ? file->read_ahead_end : offset;
146
147 if (end > NGX_MAX_OFF_T_VALUE - BRIX_READ_PREFETCH_WINDOW) {
148 *hint_end = NGX_MAX_OFF_T_VALUE;
149 } else {
150 *hint_end = end + BRIX_READ_PREFETCH_WINDOW;
151 }
152 if (file_size > 0 && *hint_end > file_size) {
153 *hint_end = file_size;
154 }
155}

157/* Sequential-aware prefetch for a read handle: detect sequential access from the
158 * handle's read_last_end / read-ahead window and issue a windowed WILLNEED hint
159 * ahead of the current read. No-op for random access. */
160void
161brix_prefetch_read_file(ngx_log_t *log, brix_file_t *file, off_t offset,
162 size_t length, off_t file_size)
163{
164 off_t end = 0, hint_start = 0, hint_end = 0;
165 ngx_flag_t sequential;
166
167 if (!brix_prefetch_read_end(file, offset, length, file_size, &end)) {
168 return;
169 }
170
171 sequential = (file->read_last_end < 0 || offset == file->read_last_end);
172 if (!sequential) {
173 /* Random access resets the read-ahead frontier and hints exactly the
174 * requested range — but only on a raw fd (a page-cache warm-up). A
175 * driver-backed handle gets NO hint on random access: the read itself
176 * fills exactly what it needs, and a speculative background fill of
177 * the same blocks would double the origin traffic (XrdPfc
178 * disable-on-random parity). */
179 file->read_ahead_end = 0;
180 if (file->sd_obj.driver == NULL) {
181 hint_start = offset;
182 hint_end = end;
183 }
184 } else if (file->sd_obj.driver != NULL) {
185 /* Driver-backed sequential stream: hint the full engine window off
186 * the CURRENT read cursor on every read. The engine-side
187 * read_ahead_end frontier is wrong here — it assumes the whole hint
188 * was honored, but the driver clamps to its own prefetch window,
189 * which would starve a long stream after the first burst. The
190 * driver keeps a per-handle rolling frontier instead, so repeated
191 * hints over already-queued or cached blocks cost one bitmap peek. */
192 hint_start = end;
193 hint_end = (end > NGX_MAX_OFF_T_VALUE - BRIX_READ_PREFETCH_WINDOW)
194 ? NGX_MAX_OFF_T_VALUE : end + BRIX_READ_PREFETCH_WINDOW;
195 if (file_size > 0 && hint_end > file_size) {
196 hint_end = file_size;
197 }
198 } else {
199 brix_prefetch_seq_window(file, offset, end, file_size,
200 &hint_start, &hint_end);
201 }
202
203 if (hint_end > hint_start) {
204 if (file->sd_obj.driver != NULL) {
205 brix_prefetch_obj_range(log, &file->sd_obj, hint_start,
206 (size_t) (hint_end - hint_start));
207 } else {
208 brix_prefetch_fd_range(log, file->fd, hint_start,
209 (size_t) (hint_end - hint_start));
210 }
211 file->read_ahead_end = hint_end;
212 }
213
214 file->read_last_end = end;
215}

228/* Decode + bounds-check one readv segment against the open-handle table.
229 * Returns 1 with fd_out, start_out and end_out filled for a hintable range,
230 * 0 for a segment to skip (unopened handle, zero/overflowing range). */
231static int
232prefetch_decode_segment(brix_ctx_t *ctx, const readahead_list *seg,
233 size_t readv_seg_max, int *fd_out, off_t *start_out, off_t *end_out)
234{
235 int handle_index;
236 int64_t request_offset;
237 uint32_t request_length;
238 off_t segment_start;
239 off_t segment_end;
240
241 handle_index = (int) (unsigned char) seg->fhandle[0];
242 if (handle_index < 0 || handle_index >= BRIX_MAX_FILES
243 || ctx->files[handle_index].fd < 0)
244 {
245 return 0;
246 }
247
248 request_length = (uint32_t) ntohl((uint32_t) seg->rlen);
249 if ((size_t) request_length > readv_seg_max) {
250 request_length = (uint32_t) readv_seg_max;
251 }
252 if (request_length == 0) {
253 return 0;
254 }
255
256 request_offset = (int64_t) be64toh((uint64_t) seg->offset);
257 if (request_offset < 0
258 || (off_t) request_length > NGX_MAX_OFF_T_VALUE - request_offset)
259 {
260 return 0;
261 }
262
263 segment_start = (off_t) request_offset;
264 segment_end = segment_start + (off_t) request_length;
265 if (segment_end <= segment_start) {
266 return 0;
267 }
268
269 *fd_out = ctx->files[handle_index].fd;
270 *start_out = segment_start;
271 *end_out = segment_end;
272 return 1;
273}

275/* Issue read-ahead hints for the segments of a kXR_readv request. */
276void
277brix_prefetch_readv_segments(brix_ctx_t *ctx, ngx_connection_t *c,
278 readahead_list *segments, size_t segment_count, size_t readv_seg_max)
279{
280 int merged_fd = -1;
281 off_t merged_start = 0;
282 off_t merged_end = 0;
283 size_t i;
284
285 if (ctx->files == NULL) {
286 return; /* no handle open yet — nothing to hint */
287 }
288
289 for (i = 0; i < segment_count; i++) {
290 int fd;
291 off_t segment_start;
292 off_t segment_end;
293
294 if (!prefetch_decode_segment(ctx, &segments[i], readv_seg_max,
295 &fd, &segment_start, &segment_end))
296 {
297 continue;
298 }
299
300 /*
301 * readv often arrives as nearby ranges against the same file. Merge
302 * those ranges before issuing the WILLNEED hint so one vector request
303 * does not turn into many tiny kernel hints.
304 */
305 if (merged_fd == fd
306 && segment_start >= merged_start
307 && segment_start <= merged_end + (off_t) BRIX_READV_PREFETCH_GAP
308 && segment_end - merged_start <= (off_t) BRIX_READV_PREFETCH_MAX)
309 {
310 if (segment_end > merged_end) {
311 merged_end = segment_end;
312 }
313 continue;
314 }
315
316 brix_prefetch_flush(c->log, merged_fd, merged_start, merged_end);
317 merged_fd = fd;
318 merged_start = segment_start;
319 merged_end = segment_end;
320 }
321
322 brix_prefetch_flush(c->log, merged_fd, merged_start, merged_end);
323}
