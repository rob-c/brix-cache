38/*
39 * WHAT: The complete parsed-option + connection state main threads through the
40 * parse → credential → dispatch pipeline.
41 * WHY: These flags and pointers were passed loose (up to 29 parameters), which
42 * is unreviewable and trips the parameter gate. Bundling them keeps every
43 * pipeline helper at a reviewable arity with explicit, single-owner data
44 * flow; `copt`/`conn` point at main's stack objects, the rest are the
45 * parse outputs that live beyond `brix_copy_opts`.
46 * HOW: main zero-inits one instance, points `copt`/`conn` at its locals, and
47 * hands the address to each pipeline stage. Parse writes the scalars;
48 * later stages read them. Byte-frozen: same values, same order.
49 */
50typedef struct {
51 brix_copy_opts *copt; /* the parsed brix_copy_opts (main's local) */
52 brix_opts *conn; /* the connection opts (main's local) */
53 struct brix_cred_store *cred_store; /* built by the credential stage; INERT */
54 const char *from; /* --from manifest path (NULL = none) */
55 const char *journal_path; /* --journal <path> or derived from --resume */
56 const char *oidc_account; /* --oidc-account (or $OIDC_ACCOUNT) */
57 const char *proxy; /* --proxy X.509 proxy path override */
58 const char *dst; /* resolved destination (last positional) */
59 int resume; /* --resume: derive journal from --from */
60 int retries; /* --retry <n> budget */
61 int jobs; /* -j/--jobs concurrency */
62 int force_progress;/* --progress */
63 int no_progress; /* -N/--no-progress */
64 int verify; /* --verify */
65 int auto_refresh; /* --auto-refresh */
66 int sync_mode; /* --sync / --sync-check */
67} xrdcp_opts_t;

69/*
70 * WHAT: The set of strdup-owned string vectors main owns for one invocation.
71 * WHY: The positional/source/expanded/exclude/include lists were passed as
72 * five (array,count,cap) triples — 15 loose parameters — through the same
73 * helpers. One struct keeps the pipeline signatures under the gate and
74 * makes ownership (main frees them all) explicit.
75 * HOW: main zero-inits it; parse fills `pos` (+ srcs via the manifest), the
76 * credential stage fills `exp`, and main frees every vector on exit.
77 */
78typedef struct {
79 xrdcp_strlist pos; /* positional args (sources + dst) */
80 xrdcp_strlist srcs; /* sources after manifest merge, pre-glob */
81 xrdcp_strlist exp; /* sources after glob expansion */
82 xrdcp_strlist excl; /* --exclude patterns */
83 xrdcp_strlist incl; /* --include patterns */
84} xrdcp_lists_t;
