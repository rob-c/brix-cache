8/* Outcome of the two-phase name-list read (fattr_list_read_names). When
9 * `done` is set the caller must return `rc` verbatim (a terminal send already
10 * happened, whether OK-empty or an error); otherwise (raw, actual) hold the
11 * NUL-separated name list to build the response from. */
12typedef struct {
13 int done; /* terminal response already sent → return rc */
14 ngx_int_t rc; /* value to return when done */
15 char *raw; /* name-list buffer (valid when !done) */
16 ssize_t actual; /* bytes in raw (valid when !done) */
17} fattr_list_read_t;

19/*
20 *
21 * WHAT: Two-phase read of the full NUL-separated xattr name list: probe the size,
22 * allocate, then read the names into a buffer.
23 *
24 * WHY: Factors the size-probe + allocation + read (and their several terminal
25 * responses: no-xattr-support, empty list, alloc failure, read error) out of
26 * fattr_list, so the handler body is just "read then build".
27 *
28 * HOW: Returns a fattr_list_read_t. When `done` is set the caller returns `rc`
29 * (a terminal send already happened — OK-empty or error); otherwise (raw,
30 * actual) hold the name list. A +4096 slack absorbs names added between the
31 * two phases (TOCTOU). Behaviour and error codes are identical to the inline
32 * original. */
33static fattr_list_read_t
34fattr_list_read_names(brix_ctx_t *ctx, ngx_connection_t *c,
35 brix_vfs_ctx_t *vctx, const char *path, int fd)
36{
37 fattr_list_read_t out = { 0 };
38 char *raw;
39 size_t raw_cap;
40 ssize_t list_sz;
41 ssize_t actual;
42
43 /* Phase 1: probe (size, NULL) to learn how big the name list is. */
44 list_sz = path ? brix_vfs_listxattr(vctx, NULL, 0)
45 : brix_vfs_flistxattr(vctx, fd, NULL, 0);
46 if (list_sz < 0) {
47 out.done = 1;
48 /* Filesystem/backend with no xattr support → empty list, not an error.
49 * Three errnos mean exactly that here, depending on where the "no
50 * xattrs" verdict came from: the local POSIX path and the VFS's own
51 * NULL-driver checks report ENOTSUP/EOPNOTSUPP (vfs_xattr.c), while the
52 * storage seam reports a driver without the slot as ENOSYS
53 * (brix_sd_listxattr_maybe_cred, sd_cred_forward.h) — which is what a
54 * composed tier over an xattr-less leaf (e.g. brix_stage over an http
55 * origin) produces. Treating ENOSYS as a hard error made fattr list
56 * answer kXR_FSError for a perfectly healthy backend that simply has no
57 * extended attributes. */
58 if (errno == ENOTSUP || errno == EOPNOTSUPP || errno == ENOSYS) {
59 BRIX_OP_OK(ctx, BRIX_OP_FATTR);
60 out.rc = brix_send_ok(ctx, c, NULL, 0);
61 return out;
62 }
63 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
64 out.rc = brix_send_error(ctx, c, kXR_FSError, "listxattr failed");
65 return out;
66 }
67 if (list_sz == 0) {
68 out.done = 1;
69 BRIX_OP_OK(ctx, BRIX_OP_FATTR);
70 out.rc = brix_send_ok(ctx, c, NULL, 0);
71 return out;
72 }
73
74 /* +BRIX_ROOT_FATTR_PATH_SLACK absorbs names added between the two reads. */
75 if (brix_size_add((size_t) list_sz, BRIX_ROOT_FATTR_PATH_SLACK,
76 &raw_cap) != NGX_OK)
77 {
78 out.done = 1;
79 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
80 out.rc = brix_send_error(ctx, c, kXR_ArgTooLong,
81 "fattr list too large");
82 return out;
83 }
84 raw = ngx_palloc(c->pool, raw_cap);
85 if (raw == NULL) {
86 out.done = 1;
87 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
88 out.rc = brix_send_error(ctx, c, kXR_NoMemory, "out of memory");
89 return out;
90 }
91
92 /* Phase 2: read the actual NUL-separated name list into raw. */
93 actual = path ? brix_vfs_listxattr(vctx, raw, raw_cap)
94 : brix_vfs_flistxattr(vctx, fd, raw, raw_cap);
95 if (actual < 0) {
96 out.done = 1;
97 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
98 out.rc = brix_send_error(ctx, c, kXR_FSError, "listxattr failed");
99 return out;
100 }
101 if ((size_t) actual > raw_cap) {
102 out.done = 1;
103 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
104 out.rc = brix_send_error(ctx, c, kXR_FSError,
105 "invalid listxattr length");
106 return out;
107 }
108 if (actual == 0) {
109 out.done = 1;
110 BRIX_OP_OK(ctx, BRIX_OP_FATTR);
111 out.rc = brix_send_ok(ctx, c, NULL, 0);
112 return out;
113 }
114
115 out.raw = raw;
116 out.actual = actual;
117 return out;
118}

120/*
121 *
122 * WHAT: Reads one direct attribute value into a caller-bounded response slot.
123 *
124 * WHY: Factors the kXR_fa_aData value-append out of the build loop. Note getxattr
125 * uses the full internal key while the emitted name is the stripped form.
126 *
127 * HOW: Probe first, enforce both the protocol maximum and destination capacity,
128 * then perform the bounded read. A disappearing/unreadable value retains
129 * the historical zero-length representation; a growth race fails closed. */
130static ngx_int_t
131fattr_list_read_value(brix_vfs_ctx_t *vctx, const char *path, int fd,
132 const char *full_key, u_char *dst, size_t dst_cap, size_t *value_len)
133{
134 ssize_t expected;
135 ssize_t actual;
136
137 *value_len = 0;
138 expected = path ? brix_vfs_getxattr(vctx, full_key, NULL, 0)
139 : brix_vfs_fgetxattr(vctx, fd, full_key, NULL, 0);
140 if (expected < 0) return NGX_OK;
141 if ((size_t) expected > kXR_faMaxVlen || (size_t) expected > dst_cap) {
142 errno = ERANGE;
143 return NGX_ERROR;
144 }
145 if (expected == 0) return NGX_OK;
146
147 actual = path ? brix_vfs_getxattr(vctx, full_key, dst, (size_t) expected)
148 : brix_vfs_fgetxattr(vctx, fd, full_key, dst,
149 (size_t) expected);
150 if (actual < 0) return errno == ERANGE ? NGX_ERROR : NGX_OK;
151 if (actual > expected || (size_t) actual > kXR_faMaxVlen) {
152 errno = ERANGE;
153 return NGX_ERROR;
154 }
155 *value_len = (size_t) actual;
156 return NGX_OK;
157}

228/*
229 *
230 * WHAT: Builds the fattrList response body into `resp` (capacity resp_cap) by
231 * striding the raw NUL-separated name list, keeping managed "user.U.*" keys,
232 * and (when aData) appending each attribute's value. Returns the byte length.
233 *
234 * WHY: Factors the response-assembly loop out of fattr_list so the handler is a
235 * thin read→build→send sequence.
236 *
237 * HOW: Each kept name has its full 7-byte internal prefix stripped so the client
238 * sees the bare name it set (matching the reference XeqFALsd). Entries that
239 * would exceed resp_cap reject the complete response; a partial successful
240 * list would hide attributes and make retry behavior ambiguous. */
241static ssize_t
242fattr_list_build(ngx_pool_t *pool, brix_vfs_ctx_t *vctx, const char *path,
243 int fd, int aData, const char *raw, ssize_t actual, u_char *resp,
244 size_t resp_cap)
245{
246 /* wp = write cursor into resp; lp/lend stride the raw name list. */
247 const char *lp = raw;
248 const char *lend = raw + actual;
249 size_t used = 0;
250
251 while (lp < lend) {
252 const char *nul;
253 size_t full_nlen;
254 brix_fattr_stored_t stored;
255 int decoded;
256
257 nul = memchr(lp, '\0', (size_t) (lend - lp));
258 if (nul == NULL) {
259 errno = EIO;
260 return -1;
261 }
262 full_nlen = (size_t) (nul - lp);
263 decoded = fattr_storage_decode_list(pool, vctx, path, fd, lp,
264 &stored);
265 if (decoded < 0) return -1;
266 if (decoded == BRIX_FATTR_LIST_EMIT
267 && fattr_list_append_entry(vctx, path, fd, aData, lp, &stored,
268 resp, resp_cap, &used) != NGX_OK)
269 return -1;
270 /* Advance past this name and its NUL terminator. */
271 lp += full_nlen + 1;
272 }
273
274 return (ssize_t) used;
275}

277/*
278 *
279 * WHAT: Handles kXR_fattrList by calling listxattr(path, NULL, 0) for path-based
280 * operations or flistxattr(fd, NULL, 0) for open-file-handle operations to
281 * query the size of all attribute names stored on the filesystem, then reads
282 * the full NUL-separated name string into a buffer. Filters entries to only
283 * those prefixed with BRIX_FATTR_XKEY_PFX (the user namespace prefix used by
284 * nginx-xrootd), stripping the 5-byte prefix from returned names so clients see
285 * bare attribute keys. When kXR_fa_aData flag is set, also reads each filtered
286 * attribute's value via getxattr/fgetxattr and appends it to the response.
287 *
288 * WHY: XRootD fattrList returns all user-space extended attributes on a file;
289 * nginx-xrootd uses only the user namespace prefixed with "user." so filtering
290 * ensures only managed attributes are visible. The two-phase list (query size
291 * then read) prevents over-allocation. ENOTSUP/EOPNOTSUPP handled gracefully —
292 * filesystems without xattr support return empty response rather than error.
293 * Thread safety: operates only on provided ctx, c, pool and local stack variables.
294 *
295 * HOW: Recurse dispatch → two-phase read (fattr_list_read_names) → build
296 * (fattr_list_build) → send. Each stage is a small helper; this body is the
297 * straight-line composition. */
298ngx_int_t fattr_list(brix_ctx_t *ctx, ngx_connection_t *c,
299 brix_vfs_ctx_t *vctx, const char *path, int fd,
300 int options) {
301 int aData = (options & kXR_fa_aData);
302 ngx_int_t recurse_rc = NGX_OK;
303 fattr_list_read_t rd;
304 size_t resp_cap;
305 ssize_t resp_len;
306 u_char *resp;
307
308 /* Directory + recurse flag → take the subtree-walk path instead. */
309 if (fattr_list_try_recurse(ctx, c, vctx, path, options, &recurse_rc)) {
310 return recurse_rc;
311 }
312
313 rd = fattr_list_read_names(ctx, c, vctx, path, fd);
314 if (rd.done) {
315 return rd.rc;
316 }
317
318 /* Keep response memory bounded. The builder accounts exact decoded values
319 * and rejects the whole request before an entry would cross this cap. */
320 resp_cap = BRIX_ROOT_FATTR_RESP_CAP;
321 resp = ngx_palloc(c->pool, resp_cap);
322 if (resp == NULL) {
323 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
324 return brix_send_error(ctx, c, kXR_NoMemory, "out of memory");
325 }
326
327 resp_len = fattr_list_build(c->pool, vctx, path, fd, aData, rd.raw,
328 rd.actual, resp, resp_cap);
329 if (resp_len < 0) {
330 int build_errno = errno;
331
332 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
333 if (build_errno == ERANGE || build_errno == E2BIG) {
334 return brix_send_error(ctx, c, kXR_ArgTooLong,
335 "fattr list response too large");
336 }
337 return brix_send_error(ctx, c, kXR_FSError, "fattr list failed");
338 }
339
340 BRIX_OP_OK(ctx, BRIX_OP_FATTR);
341 if (resp_len == 0) {
342 return brix_send_ok(ctx, c, NULL, 0);
343 }
344 return brix_send_ok(ctx, c, resp, (uint32_t) resp_len);
345}
