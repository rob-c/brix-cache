1/*
2 * sd_pblock.c — the pblock ("pseudo-block") Storage Driver: a full-capability,
3 * block-based drop-in for POSIX.
4 *
5 * WHAT: Implements brix_sd_pblock_driver, a complete backend that stores each
6 * object's bytes striped across fixed-size POSIX "block" files and the
7 * entire logical namespace + metadata in a SQLite catalog
8 * (sd_pblock_catalog.c). It advertises the same capabilities as the POSIX
9 * driver and implements every vtable slot. This file owns the instance and
10 * object lifecycle (init/cleanup, open/close and their helpers) and the
11 * driver descriptor; the byte I/O (sd_pblock_io.c), namespace/directory/
12 * xattr ops (sd_pblock_namespace.c) and staged atomic publish
13 * (sd_pblock_staged.c) live in siblings reached through
14 * sd_pblock_internal.h.
15 *
16 * WHY: Striping bulk content into fixed-size blocks (default 64 MiB, set per
17 * file at creation and configurable per export) is the defining property
18 * of a block backend, and splitting that content from the namespace/
19 * metadata keeps the hot byte path free of any database work. SQLite is
20 * touched only at metadata boundaries. Split from a single 1111-line file
21 * (phase-79) to hold every pblock file under the ~500-line, one-concept cap.
22 *
23 * HOW: An object's bytes live at <data>/<aa>/<bb>/<blob_id>/<block_index>;
24 * block 0 is opened persistently as obj->fd (so small files keep a real
25 * fd for zero-copy sendfile), and higher blocks are opened transiently per
26 * I/O. Reads/writes map [off,off+len) across blocks (holes read as zeros);
27 * writes update the cached size/mtime in memory and flush to the catalog
28 * on fsync/close. The whole file is ngx-free (libc + sqlite, malloc-owned
29 * state) so it is identical in the module and the standalone unit test.
30 * Compiled only when the build found libsqlite3 (BRIX_HAVE_SQLITE).
31 */

78/* Adapter from the catalog's flat row callback to the driver's enumerate cb. */

85static int
86pblock_enum_thunk(void *vctx, const char *path, int is_dir, int64_t size,
87 int64_t mtime)
88{
89 pblock_enum_ctx_t *ec = vctx;
90 brix_sd_catalog_ent_t ent;
91
92 if (is_dir) {
93 return 0; /* enumerate reports stored objects, not directories */
94 }
95 memset(&ent, 0, sizeof(ent));
96 ent.key = path; /* the catalog IS the namespace: key == logical path */
97 ent.path = path;
98 ent.have_stat = ec->want_stat;
99 ent.size = (off_t) size;
100 ent.mtime = (time_t) mtime;
101 return ec->cb(ec->ctx, &ent);
102}

128/* driver->space — quota-aware capacity numbers for kXR_statvfs/SRR. Answers
129 * only when a byte quota is armed (then total = the quota, used = the usage
130 * rollup); otherwise NGX_ERROR/ENOTSUP so callers fall back to statvfs(2) on
131 * the backing filesystem — byte-for-byte the pre-F5 behaviour. */
132static ngx_int_t
133sd_pblock_space(brix_sd_instance_t *inst, brix_sd_space_t *out)
134{
135 pblock_state_t *st = inst->state;
136 int64_t bytes, inodes;
137
138 if (st == NULL || !st->quota || st->quota_bytes <= 0) {
139 errno = ENOTSUP;
140 return NGX_ERROR;
141 }
142 if (pblock_quota_usage(st, "total", 0, &bytes, &inodes) != 0) {
143 return NGX_ERROR;
144 }
145 out->total_bytes = (uint64_t) st->quota_bytes;
146 out->used_bytes = bytes > 0 ? (uint64_t) bytes : 0;
147 out->free_bytes = bytes < st->quota_bytes
148 ? (uint64_t) (st->quota_bytes - bytes) : 0;
149 return NGX_OK;
150}

191 return NGX_OK;

201/* driver->evict (phase-107 C2) — release the simulated online copy: demote the
202 * residency row to NEARLINE (the simulated tape copy survives; a later recall
203 * restages it — the exact inverse of sd_pblock_recall). ENOTSUP when the
204 * export is not armed nearline: without a tape model the block store's copy is
205 * the ONLY copy and evicting it would be a delete wearing a different verb
206 * (the sd.h contract). Reports the object's catalog size as the bytes
207 * released — the logical online footprint the simulation just gave up (no
208 * physical block moves; this is bookkeeping, exactly like the recall's sleep).
209 * The *_cred twin gates W_OK against catalog ownership and delegates here
210 * (sd_pblock_cred.c) — pblock is its own multi-user authority, unlike frm's
211 * service-only MSS adapter. */
212ngx_int_t
213sd_pblock_evict(brix_sd_instance_t *inst, const char *path,
214 uint64_t *bytes_out)
215{
216 pblock_state_t *st = inst->state;
217 pblock_meta meta;
218 int rc;
219
220 if (bytes_out != NULL) {
221 *bytes_out = 0;
222 }
223 if (st == NULL || !st->nearline) {
224 errno = ENOTSUP;
225 return NGX_ERROR;
226 }
227 rc = pblock_catalog_lookup(st->cat, path, &meta);
228 if (rc != 0) {
229 errno = rc == 1 ? ENOENT : errno;
230 return NGX_ERROR;
231 }
232 if (meta.is_dir) {
233 errno = EISDIR;
234 return NGX_ERROR;
235 }
236 if (pblock_nearline_evict(st, path) != 0) {
237 return NGX_ERROR; /* errno set (ENOENT lost / EIO) */
238 }
239 if (bytes_out != NULL && meta.size > 0) {
240 *bytes_out = (uint64_t) meta.size;
241 }
242 return NGX_OK;
243}

247/* sd_pblock_dedup_publish — driver->dedup_publish for pblock.
248 *
249 * WHAT: Collapse the (content-verified) object at `path` onto a byte-identical
250 * existing blob via the F10 refs machinery. The canonical alias `canon`

252 * no extra namespace entry is needed and eviction is refcount-driven

258 * hardlink farm has no meaning here, but refcounted-blob dedup gives the
259 * same one-copy-per-content guarantee with byte-verified folds.

262 * committed blob by its recorded content hash and folds when a candidate
263 * byte-verifies.
264 */

273 errno = ENOTSUP;
274 return NGX_ERROR;
275 }
276 return pblock_refs_dedup_existing(st, path) < 0 ? NGX_ERROR : NGX_OK;

281/* Atomic two-name exchange (phase-107 C6): one catalog transaction swaps both
282 * subtrees (pblock_catalog_exchange), so a concurrent reader never sees either
283 * name absent. Lives here rather than sd_pblock_namespace.c because that file
284 * sits at the 600-line cap. Side tables (nearline residency, anomaly history,
285 * lease rows) swap post-commit through the catalog's reserved temp name —
286 * exact-path, best-effort, the same discipline pblock_rename_observe applies
287 * after pblock_catalog_rename. Lease checks come first, like rename. */
288ngx_int_t
289sd_pblock_exchange(brix_sd_instance_t *inst, const char *a, const char *b)
290{
291 static const char tmp[] = "/\001brix-xchg\001";
292 pblock_state_t *st = inst->state;
293 ngx_int_t rc;
294
295 if (st->locks
296 && (pblock_locks_ns_check(st, a, 0) != 0
297 || pblock_locks_ns_check(st, b, 0) != 0))
298 {
299 return NGX_ERROR;
300 }
301
302 rc = pblock_catalog_exchange(st->cat, a, b) == 0 ? NGX_OK : NGX_ERROR;
303 if (rc == NGX_OK) {
304 if (st->nearline) {
305 pblock_nearline_rename(st, a, tmp);
306 pblock_nearline_rename(st, b, a);
307 pblock_nearline_rename(st, tmp, b);
308 }
309 if (st->lab != NULL) {
310 pblock_anomaly_rename(st, a, tmp);
311 pblock_anomaly_rename(st, b, a);
312 pblock_anomaly_rename(st, tmp, b);
313 }
314 if (st->locks) {
315 pblock_locks_rename(st, a, tmp);
316 pblock_locks_rename(st, b, a);
317 pblock_locks_rename(st, tmp, b);
318 }
319 }
320 if (st->audit) {
321 pblock_audit_log(st->cat, "exchange", a, b, 0, 0,
322 rc == NGX_OK ? 0 : -1, rc == NGX_OK ? 0 : errno);
323 }
324 return rc;
325}
