21/* blfile_parse_ipv4 — dotted-quad text (len bytes, not NUL-terminated) to a
22 * host-byte-order u32. Returns 0 on success, -1 if not an IPv4 literal. */
23static int
24blfile_parse_ipv4(const char *s, size_t len, uint32_t *out)
25{
26 char buf[INET_ADDRSTRLEN];
27 struct in_addr a;
28
29 if (len == 0 || len >= sizeof(buf)) {
30 return -1;
31 }
32 memcpy(buf, s, len);
33 buf[len] = '\0';
34 if (inet_pton(AF_INET, buf, &a) != 1) {
35 return -1;
36 }
37 *out = ntohl(a.s_addr);
38 return 0;
39}

41/* blfile_parse_uint — bounded decimal parse of [s, s+len). Returns the value
42 * or -1 on empty/non-digit/overflow-past-max input. */
43static long
44blfile_parse_uint(const char *s, size_t len, long max)
45{
46 long v = 0;
47 size_t i;
48
49 if (len == 0) {
50 return -1;
51 }
52 for (i = 0; i < len; i++) {
53 if (s[i] < '0' || s[i] > '9') {
54 return -1;
55 }
56 v = v * 10 + (s[i] - '0');
57 if (v > max) {
58 return -1;
59 }
60 }
61 return v;
62}

64/* blfile_parse_cidr — IPv4 CIDR entry text ("a.b.c.d/n", slash points at the
65 * '/'). Returns 0 with out filled as a CIDR entry, -1 on malformed input. */
66static int
67blfile_parse_cidr(const char *line, size_t len, const char *slash,
68 brix_cms_blfile_entry_t *out)
69{
70 uint32_t net;
71 long v;
72
73 if (blfile_parse_ipv4(line, (size_t) (slash - line), &net) != 0) {
74 return -1;
75 }
76 v = blfile_parse_uint(slash + 1, len - (size_t) (slash - line) - 1, 32);
77 if (v < 0) {
78 return -1;
79 }
80 out->is_cidr = 1;
81 out->mask = (v == 0) ? 0 : (uint32_t) (0xffffffffu << (32 - v));
82 out->net = net & out->mask;
83 return 0;
84}

86/* blfile_find_port_colon — locate the port separator in a host[:port] line.
87 * Bracketed IPv6 ([::1]:1094) keeps its colons inside the brackets; only a
88 * colon AFTER the closing bracket (or in a bracket-less line with exactly one
89 * colon) is a port separator. A bare IPv6 literal (multiple colons, no
90 * brackets) is taken whole as the host text. Returns 0 with *colon set (NULL
91 * when there is no port), -1 on a malformed bracketed line. */
92static int
93blfile_find_port_colon(const char *line, size_t len, const char **colon)
94{
95 *colon = NULL;
96 if (line[0] == '[') {
97 const char *rb = memchr(line, ']', len);
98
99 if (rb == NULL) {
100 return -1;
101 }
102 if ((size_t) (rb - line) + 1 < len) {
103 if (rb[1] != ':') {
104 return -1;
105 }
106 *colon = rb + 1;
107 }
108 } else {
109 const char *first = memchr(line, ':', len);
110
111 if (first != NULL
112 && memchr(first + 1, ':', len - (size_t) (first - line) - 1) == NULL)
113 {
114 *colon = first;
115 }
116 }
117 return 0;
118}

120/* blfile_parse_hostspec — the first token of a line: exact host, host:port,
121 * IPv4 CIDR, or (§2.13) a `*` wildcard pattern (at most one star, whole-host
122 * — a pattern takes no :port suffix, matching stock hostpat rules). */
123static int
124blfile_parse_hostspec(const char *line, size_t len,
125 brix_cms_blfile_entry_t *out)
126{
127 const char *slash;
128 const char *colon;
129 const char *star;
130 long v;
131
132 slash = memchr(line, '/', len);
133 if (slash != NULL) {
134 /* IPv4 CIDR: a.b.c.d/n */
135 return blfile_parse_cidr(line, len, slash, out);
136 }
137
138 star = memchr(line, '*', len);
139 if (star != NULL) {
140 /* §2.13: wildcard pattern — exactly one star, no port suffix. */
141 if (memchr(star + 1, '*', len - (size_t) (star - line) - 1) != NULL
142 || len == 0 || len >= sizeof(out->host))
143 {
144 return -1;
145 }
146 memcpy(out->host, line, len);
147 out->host[len] = '\0';
148 out->is_pattern = 1;
149 return 0;
150 }
151
152 if (blfile_find_port_colon(line, len, &colon) != 0) {
153 return -1;
154 }
155
156 if (colon != NULL) {
157 v = blfile_parse_uint(colon + 1, len - (size_t) (colon - line) - 1,
158 65535);
159 if (v < 1) {
160 return -1;
161 }
162 out->port = (uint16_t) v;
163 len = (size_t) (colon - line);
164 }
165
166 if (len == 0 || len >= sizeof(out->host)) {
167 return -1;
168 }
169 memcpy(out->host, line, len);
170 out->host[len] = '\0';
171 return 0;
172}

174/* blfile_parse_redirect — §2.13: the `redirect <host:port>` action tail.
175 * The target must carry an explicit port (kYR_try names a concrete manager
176 * endpoint — guessing one would bounce nodes into the void). */
177static int
178blfile_parse_redirect(const char *tok, size_t len,
179 brix_cms_blfile_entry_t *out)
180{
181 const char *colon;
182 long v;
183
184 if (blfile_find_port_colon(tok, len, &colon) != 0 || colon == NULL) {
185 return -1;
186 }
187 v = blfile_parse_uint(colon + 1, len - (size_t) (colon - tok) - 1, 65535);
188 if (v < 1) {
189 return -1;
190 }
191 len = (size_t) (colon - tok);
192 if (len == 0 || len >= sizeof(out->redirect_host)) {
193 return -1;
194 }
195 memcpy(out->redirect_host, tok, len);
196 out->redirect_host[len] = '\0';
197 out->redirect_port = (uint16_t) v;
198 out->has_redirect = 1;
199 return 0;
200}

202/*
203 * WHAT: advance `p` past any spaces/tabs, stopping at `end`; returns the new
204 * cursor.
205 * WHY: the action tail is whitespace-separated in two places and both must
206 * treat tab and space identically — one skipper keeps them from drifting.
207 * HOW: bounded walk; never reads at or past `end`.
208 */
209static const char *
210blfile_skip_ws(const char *p, const char *end)
211{
212 while (p < end && (*p == ' ' || *p == '\t')) {
213 p++;
214 }
215
216 return p;
217}

220/*
221 * WHAT: parse the ` redirect <host:port>` action tail in [tail,tail_end) into
222 * *out. Returns 0 on success, -1 on anything malformed.
223 * WHY: §2.13 allows exactly one action keyword with exactly one argument.
224 * Interior whitespace in the argument is rejected rather than truncated
225 * at, so a line that means two things can never be read as one — a
226 * silently-truncated redirect target would bounce nodes at the wrong
227 * manager.
228 * HOW: 1. skip leading whitespace and require the literal `redirect` followed
229 * by whitespace (not a prefix of a longer keyword).
230 * 2. skip to the argument and reject an empty or whitespace-bearing one.
231 * 3. hand the argument to blfile_parse_redirect for host/port validation.
232 */
233static int
234blfile_parse_action_tail(const char *tail, const char *tail_end,
235 brix_cms_blfile_entry_t *out)
236{
237 size_t kw_len = sizeof("redirect") - 1;
238 size_t arg_len;
239
240 tail = blfile_skip_ws(tail, tail_end);
241
242 if ((size_t) (tail_end - tail) <= kw_len + 1
243 || memcmp(tail, "redirect", kw_len) != 0
244 || (tail[kw_len] != ' ' && tail[kw_len] != '\t'))
245 {
246 return -1;
247 }
248
249 tail = blfile_skip_ws(tail + kw_len, tail_end);
250 arg_len = (size_t) (tail_end - tail);
251
252 if (arg_len == 0
253 || memchr(tail, ' ', arg_len) != NULL
254 || memchr(tail, '\t', arg_len) != NULL)
255 {
256 return -1;
257 }
258
259 return blfile_parse_redirect(tail, arg_len, out);
260}

263int
264brix_cms_blfile_parse_line(const char *line, size_t len,
265 brix_cms_blfile_entry_t *out)
266{
267 const char *ws;
268 size_t head_len;
269
270 if (line == NULL || out == NULL || len == 0) {
271 return -1;
272 }
273 memset(out, 0, sizeof(*out));
274
275 /* §2.13: split an optional ` redirect <host:port>` action tail off the
276 * hostspec head. Any other interior whitespace stays malformed. */
277 ws = memchr(line, ' ', len);
278 if (ws == NULL) {
279 ws = memchr(line, '\t', len);
280 }
281 head_len = (ws != NULL) ? (size_t) (ws - line) : len;
282
283 if (ws != NULL && blfile_parse_action_tail(ws, line + len, out) != 0) {
284 return -1;
285 }
286
287 return blfile_parse_hostspec(line, head_len, out);
288}

290int
291brix_cms_blfile_entry_matches(const brix_cms_blfile_entry_t *e,
292 const char *host, uint16_t port)
293{
294 if (e == NULL || host == NULL) {
295 return 0;
296 }
297
298 if (e->is_cidr) {
299 uint32_t addr;
300
301 if (blfile_parse_ipv4(host, strlen(host), &addr) != 0) {
302 return 0; /* CIDR entries only cover IPv4 hosts */
303 }
304 return (addr & e->mask) == e->net;
305 }
306
307 if (e->is_pattern) {
308 /* §2.13: one-`*` wildcard span, XrdOucNList rules — same matcher the
309 * protbind host templates use, so the two grammars can never drift. */
310 ngx_str_t tpl;
311
312 tpl.data = (u_char *) e->host;
313 tpl.len = strlen(e->host);
314 return brix_protbind_host_match(&tpl, host) ? 1 : 0;
315 }
316
317 if (strcmp(e->host, host) != 0) {
318 return 0;
319 }
320 return e->port == 0 || e->port == port;
321}

343/* blfile_scan_line — find the '\n'-terminated line at *pos in buf[0..got),
344 * trim surrounding whitespace, and advance *pos past the line. Returns the
345 * trimmed bounds in *start / *end (start == end for a blank line). */
346static void
347blfile_scan_line(const char *buf, size_t got, size_t *pos,
348 size_t *start, size_t *end)
349{
350 size_t eol = *pos;
351
352 while (eol < got && buf[eol] != '\n') {
353 eol++;
354 }
355
356 /* Trim surrounding whitespace. */
357 *start = *pos;
358 *end = eol;
359 while (*start < *end && (buf[*start] == ' ' || buf[*start] == '\t'
360 || buf[*start] == '\r')) {
361 (*start)++;
362 }
363 while (*end > *start && (buf[*end - 1] == ' ' || buf[*end - 1] == '\t'
364 || buf[*end - 1] == '\r')) {
365 (*end)--;
366 }
367 *pos = eol + 1;
368}

370/* blfile_reload — parse the file at path into bl->entries, skipping comments,
371 * blank lines, and (with a warning) malformed lines. Never fails the poll:
372 * an unreadable file simply keeps the previous entry set. */
373static void
374blfile_reload(brix_cms_blfile_t *bl, const char *path, ngx_log_t *log)
375{
376 /* vfs-seam-allow: operator blacklist file is host config, not managed storage */
377 FILE *f = fopen(path, "r");
378 char buf[BRIX_CMS_BLFILE_MAX_BYTES];
379 size_t got, pos;
380
381 if (f == NULL) {
382 ngx_log_error(NGX_LOG_WARN, log, ngx_errno,
383 "brix: cms blacklist file \"%s\" unreadable; "
384 "keeping previous %ui entries", path, bl->nentries);
385 return;
386 }
387 got = fread(buf, 1, sizeof(buf) - 1, f);
388 fclose(f);
389 buf[got] = '\0';
390
391 bl->nentries = 0;
392
393 for (pos = 0; pos < got; /* advanced inside */) {
394 size_t start, end;
395
396 /* Scan and trim the next line, then drop blanks and '#' comments. */
397 blfile_scan_line(buf, got, &pos, &start, &end);
398 if (start == end || buf[start] == '#') {
399 continue;
400 }
401
402 if (bl->nentries >= BRIX_CMS_BLFILE_MAX_ENTRIES) {
403 ngx_log_error(NGX_LOG_WARN, log, 0,
404 "brix: cms blacklist file \"%s\": more than %d "
405 "entries; rest ignored", path,
406 BRIX_CMS_BLFILE_MAX_ENTRIES);
407 break;
408 }
409
410 if (brix_cms_blfile_parse_line(buf + start, end - start,
411 &bl->entries[bl->nentries]) != 0)
412 {
413 ngx_log_error(NGX_LOG_WARN, log, 0,
414 "brix: cms blacklist file \"%s\": skipping "
415 "malformed line \"%*s\"", path,
416 (int) (end - start), buf + start);
417 continue;
418 }
419 bl->nentries++;
420 }
421
422 ngx_log_error(NGX_LOG_NOTICE, log, 0,
423 "brix: cms blacklist file \"%s\" loaded: %ui entries",
424 path, bl->nentries);
425}

427/*
428 * blfile_refresh — the shared rate-limited stat/reload half of both polls.
429 *
430 * WHAT: Applies the poll rate limit, re-reads the file on an mtime change,
431 * and reports (1) whether enforcement should run this call.
432 * WHY: Blacklist and whitelist modes differ only in the enforcement rule;
433 * one refresh keeps the stat cadence and reload behaviour identical.
434 * HOW: Exactly the original poll prologue, factored.
435 */
436static int
437blfile_refresh(brix_cms_blfile_t *bl, const ngx_str_t *path,
438 ngx_uint_t force, ngx_log_t *log)
439{
440 char pathbuf[1024];
441 struct stat st;
442
443 if (bl == NULL || path == NULL || path->len == 0
444 || path->len >= sizeof(pathbuf))
445 {
446 return 0;
447 }
448 if (!force && bl->next_poll != 0 && ngx_current_msec < bl->next_poll) {
449 return 0;
450 }
451 bl->next_poll = ngx_current_msec + BRIX_CMS_BLFILE_POLL_MS;
452
453 memcpy(pathbuf, path->data, path->len);
454 pathbuf[path->len] = '\0';
455
456 if (stat(pathbuf, &st) != 0) { /* vfs-seam-allow: DOMAIN_CONFIG — operator config file, not managed storage */
457 ngx_log_error(NGX_LOG_WARN, log, ngx_errno,
458 "brix: cms blacklist file \"%s\" stat failed; "
459 "keeping previous %ui entries", pathbuf, bl->nentries);
460 } else if (st.st_mtime != bl->mtime) {
461 blfile_reload(bl, pathbuf, log);
462 bl->mtime = st.st_mtime;
463 }
464
465 return bl->nentries > 0;
466}

468/* blfile_enforce — walk a registry snapshot and blacklist every server whose
469 * membership test (`matching` for blacklist mode, non-matching for whitelist
470 * mode) says it must not serve. The snapshot is a by-value copy, so no
471 * registry lock is held while we walk (brix_srv_blacklist locks per call).
472 * Heap, not stack — 128 snapshot entries are ~80 KB. */
473static void
474blfile_enforce(const brix_cms_blfile_t *bl, ngx_msec_t blacklist_ms,
475 int want_match, ngx_log_t *log)
476{
477 brix_srv_snapshot_entry_t *snap;
478 ngx_uint_t n, s;
479
480 snap = ngx_alloc(sizeof(*snap) * BRIX_SRV_REGISTRY_SLOTS, log);
481 if (snap == NULL) {
482 return;
483 }
484 n = brix_srv_snapshot(snap, BRIX_SRV_REGISTRY_SLOTS, ngx_current_msec);
485 for (s = 0; s < n; s++) {
486 int matched =
487 brix_cms_blfile_find(bl, snap[s].host, snap[s].port) != NULL;
488
489 if (matched == want_match) {
490 brix_srv_blacklist(snap[s].host, snap[s].port, blacklist_ms);
491 }
492 }
493 ngx_free(snap);
494}
