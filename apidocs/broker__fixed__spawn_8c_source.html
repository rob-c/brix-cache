14/* WHAT: Verify every realized uid/gid slot and supplementary group.
15 * WHY: Neither a saved root ID nor an inherited group may lend authority.
16 * HOW: Read all slots/groups back and require an exact order-independent set. */
17static int
18fixed_verify_ids(const brix_idmap_creds_t *creds)
19{
20 uid_t real_uid, effective_uid, saved_uid;
21 gid_t real_gid, effective_gid, saved_gid;
22 brix_idmap_creds_t realized = *creds;
23
24 if (brix_plat_getresuid(&real_uid, &effective_uid, &saved_uid) != 0
25 || brix_plat_getresgid(&real_gid, &effective_gid, &saved_gid) != 0
26 || brix_plat_prctl(PR_GET_NO_NEW_PRIVS, 0) != 1)
27 {
28 return -1;
29 }
30 if (real_uid != creds->uid || effective_uid != creds->uid
31 || saved_uid != creds->uid || real_gid != creds->gid
32 || effective_gid != creds->gid || saved_gid != creds->gid)
33 {
34 errno = EPERM;
35 return -1;
36 }
37 realized.ngroups = getgroups(BRIX_IDMAP_MAXGROUPS, realized.groups);
38 if (!brix_imp_fixed_creds_equal(creds, &realized)) {
39 errno = EPERM;
40 return -1;
41 }
42 return 0;
43}

45/* WHAT: Irreversibly become one non-root account, without Linux capabilities.
46 * WHY: Request-serving helpers and their supervisor must never retain root.
47 * HOW: Set no-new-privileges, groups, all gid/uid slots, then verify the result. */
48int
49brix_imp_fixed_drop(const brix_idmap_creds_t *creds)
50{
51 struct rlimit no_core = {0, 0};
52
53 if (creds == NULL || creds->uid == 0 || creds->gid == 0
54 || creds->ngroups < 0 || creds->ngroups > BRIX_IDMAP_MAXGROUPS)
55 {
56 errno = EPERM;
57 return -1;
58 }
59 if (setrlimit(RLIMIT_CORE, &no_core) != 0
60 || brix_plat_prctl(PR_SET_NO_NEW_PRIVS, 1) != 0
61 || syscall(SYS_setgroups, (size_t) creds->ngroups, creds->groups) != 0
62 || brix_plat_setresgid(creds->gid, creds->gid, creds->gid) != 0
63 || brix_plat_setresuid(creds->uid, creds->uid, creds->uid) != 0)
64 {
65 return -1;
66 }
67 return fixed_verify_ids(creds);
68}

70/* WHAT: Set bounded native helper transport buffers and waiting time.
71 * WHY: One frame/payload must fit, and a dead helper cannot stall every user.
72 * HOW: Enlarge both socket buffers and install symmetric 30-second deadlines. */
73static int
74fixed_channel_options(int channel, int supervisor)
75{
76 int capacity = 2 * (IMP_XATTR_MAX + sizeof(imp_req_t));
77 struct timeval deadline = {30, 0};
78
79 if (setsockopt(channel, SOL_SOCKET, SO_SNDBUF, &capacity, sizeof(capacity)) != 0
80 || setsockopt(channel, SOL_SOCKET, SO_RCVBUF, &capacity, sizeof(capacity)) != 0
81 || setsockopt(channel, SOL_SOCKET, SO_SNDTIMEO, &deadline, sizeof(deadline)) != 0)
82 {
83 return -1;
84 }
85 if (supervisor) {
86 return setsockopt(channel, SOL_SOCKET, SO_RCVTIMEO, &deadline, sizeof(deadline));
87 }
88 return 0; /* a healthy helper may remain idle indefinitely */
89}

91/* WHAT: Enter the fixed account and sandbox before acknowledging readiness.
92 * WHY: The supervisor must never dispatch to an unconfined or root helper.
93 * HOW: Seal unrelated descriptors, irreversibly drop, confine, then serve. */
94static void
95fixed_child(int channel, int rootfd, const brix_idmap_creds_t *creds,
96 ngx_log_t *log)
97{
98 int logfd = log && log->file ? log->file->fd : -1;
99 imp_rep_t ready;
100
101 brix_imp_broker_seal_fds(channel, rootfd, log);
102 if (brix_imp_fixed_drop(creds) != 0
103 || brix_plat_broker_confine(channel, rootfd, logfd) != 0)
104 {
105 _exit(EXIT_FAILURE);
106 }
107 ngx_memzero(&ready, sizeof(ready));
108 if (imp_send_reply(channel, &ready, -1, NULL, 0) != 0) {
109 _exit(EXIT_FAILURE);
110 }
111 brix_imp_fixed_serve(channel, rootfd, creds, log);
112 _exit(0);
113}

115/* WHAT: Spawn and attest one process-descriptor-owned, fixed-identity helper.
116 * WHY: Partial setup must close only its owned child/channel, without pid races.
117 * HOW: Create a private record socket, fork, and require post-confinement ACK. */
118int
119brix_imp_fixed_spawn(int rootfd, const brix_idmap_creds_t *creds,
120 int *channel, int *process_fd, ngx_log_t *log)
121{
122 int pair[2], descriptor = -1, received = -1;
123 pid_t child;
124 imp_rep_t ready;
125
126 if (socketpair(AF_UNIX, SOCK_SEQPACKET | SOCK_CLOEXEC, 0, pair) != 0) {
127 return -1;
128 }
129 if (fixed_channel_options(pair[0], 1) != 0
130 || fixed_channel_options(pair[1], 0) != 0)
131 {
132 close(pair[0]);
133 close(pair[1]);
134 return -1;
135 }
136 child = brix_plat_broker_fork(&descriptor);
137 if (child == 0) {
138 fixed_child(pair[1], rootfd, creds, log);
139 }
140 close(pair[1]);
141 if (child < 0) {
142 close(pair[0]);
143 return -1;
144 }
145 int status = imp_recv_reply(pair[0], &ready, &received, NULL, 0);
146 if (received >= 0) { close(received); }
147 if (status != 0 || ready.status != 0 || ready.rep_flags != 0 || received >= 0) {
148 close(pair[0]);
149 close(descriptor);
150 errno = EPERM;
151 return -1;
152 }
153 *channel = pair[0];
154 *process_fd = descriptor;
155 return 0;
156}
