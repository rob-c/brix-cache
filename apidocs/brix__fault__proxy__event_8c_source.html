1/*
2 * brix_fault_proxy_event.c — optional JSONL fault-event log (D2).
3 *
4 * WHAT: when `--event-log FILE` (or the live `event-log <path>` verb) is set,
5 * append one JSON object per discrete fault event — sever (with a reason),
6 * truncate cut, batched per-read corruption, duplicate delivery, and
7 * connection refusal — so a run leaves a machine-readable audit trail.
8 *
9 * WHY: the human `status` line and Prometheus `metrics` are point-in-time
10 * aggregates; they cannot tell you *which* connection was cut *when* or
11 * *why*. A JSONL stream gives per-event provenance for offline analysis
12 * without turning the proxy into a payload capture (an explicit non-goal:
13 * NO relayed bytes are ever written to the log).
14 *
15 * HOW: a single append-only fd guarded by one mutex (events are discrete and
16 * off the per-byte hot path, so a global lock is cheap and correct). When
17 * no log is configured the fd is -1 and brix_fp_event() returns on the
18 * first branch — zero cost. Each line is built with bounded snprintf and
19 * written under the lock so concurrent relay threads never interleave.
20 */

31static pthread_mutex_t g_event_mu = PTHREAD_MUTEX_INITIALIZER;

33/* Per-thread "route" tag. A route accept thread names its route once; every
34 * other thread leaves it empty and events read "default". */
35static __thread char t_event_route[64] = "";
36
37/* Set (or clear, with NULL/"") the route tag for events emitted on this thread. */
38void
39fp_event_set_route(const char *name)
40{
41 if (name == NULL || name[0] == '\0') {
42 t_event_route[0] = '\0';
43 return;
44 }
45 size_t i = 0;
46 while (name[i] != '\0' && i + 1 < sizeof t_event_route) {
47 t_event_route[i] = name[i];
48 i++;
49 }
50 t_event_route[i] = '\0';
51}

53/* Open (or replace) the event-log sink. Returns 0 on success, -1 if the path
54 * cannot be opened for append — the caller fails closed (startup exit / err:). */
55int
56fp_event_open(const char *path)
57{
58 int fd = open(path, O_WRONLY | O_CREAT | O_APPEND, 0644);
59 if (fd < 0) {
60 return -1;
61 }
62 pthread_mutex_lock(&g_event_mu);
63 if (g_event_fd >= 0) {
64 close(g_event_fd);
65 }
66 g_event_fd = fd;
67 pthread_mutex_unlock(&g_event_mu);
68 return 0;
69}

78/* Append one formatted field to the event line, advancing `*n`. Returns 0 on
79 * success and -1 once the line is full — an event that would not fit whole is
80 * dropped rather than written truncated (a half-object is not valid JSONL). */
81static int __attribute__((format(printf, 4, 5)))
82ev_append(char *line, size_t cap, int *n, const char *fmt, ...)
83{
84 if (*n < 0 || (size_t) *n >= cap) {
85 return -1;
86 }
87 va_list ap;
88 va_start(ap, fmt);
89 int m = vsnprintf(line + *n, cap - (size_t) *n, fmt, ap);
90 va_end(ap);
91 if (m < 0 || (size_t) (*n + m) >= cap) {
92 return -1;
93 }
94 *n += m;
95 return 0;
96}

98/* Append one JSONL event. `dir`/`reason`/`numkey` are optional (NULL omits the
99 * field); `numval` is emitted only when `numkey` is non-NULL. No payload bytes
100 * are ever included — only structural metadata. */
101void
102brix_fp_event(unsigned long conn, const char *dir, const char *event,
103 const char *reason, const char *numkey, long numval)
104{
105 if (g_event_fd < 0) {
106 return; /* zero-cost when --event-log is unset */
107 }
108
109 struct timeval tv;
110 gettimeofday(&tv, NULL);
111 double t = (double) tv.tv_sec + (double) tv.tv_usec / 1e6;
112
113 const char *route = t_event_route[0] != '\0' ? t_event_route : "default";
114 char line[320];
115 int n = 0;
116
117 if (ev_append(line, sizeof line, &n, "{\"t\":%.2f,\"route\":\"%s\",\"conn\":%lu",
118 t, route, conn) != 0) {
119 return;
120 }
121 if (dir != NULL &&
122 ev_append(line, sizeof line, &n, ",\"dir\":\"%s\"", dir) != 0) {
123 return;
124 }
125 if (ev_append(line, sizeof line, &n, ",\"event\":\"%s\"", event) != 0) {
126 return;
127 }
128 if (reason != NULL &&
129 ev_append(line, sizeof line, &n, ",\"reason\":\"%s\"", reason) != 0) {
130 return;
131 }
132 if (numkey != NULL &&
133 ev_append(line, sizeof line, &n, ",\"%s\":%ld", numkey, numval) != 0) {
134 return;
135 }
136 if (n >= (int) sizeof line - 2) {
137 return; /* no room for the closing "}\n" */
138 }
139 line[n++] = '}';
140 line[n++] = '\n';
141
142 pthread_mutex_lock(&g_event_mu);
143 ssize_t wr = write(g_event_fd, line, (size_t) n);
144 (void) wr; /* best-effort: a full log must not disrupt the relay */
145 pthread_mutex_unlock(&g_event_mu);
146}
