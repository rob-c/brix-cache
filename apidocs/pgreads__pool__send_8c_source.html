8 * kXR_pgread.
9 *
10 * WHAT: The half of the pgread offload that runs ON THE POOL THREAD and owns
11 * the secondary's cleartext socket for the duration of one frame: the
12 * budgeted POLLOUT send loop (brix_pool_send_range), the single-frame

25/* §1.2 pool-send pacing: one POLLOUT wait quantum and the wait budget after
26 * which the worker hands the unsent tail back to the event loop (a slow
27 * reader must not park a pool thread indefinitely — xrootd blocks its writer
28 * threads here; we bound the wait and fall back to the out-ring). */
29#define BRIX_POOL_SEND_WAIT_MS 25
30#define BRIX_POOL_SEND_WAIT_MAX 12
31
32/*
33 * brix_pool_send_range — §1.2 budgeted send of the wire image bytes
34 * [base + t->pool_sent, base + end) on the SECONDARY's cleartext socket.
35 *
36 * The watermark t->pool_sent advances as bytes land; the wait budget counts
37 * CONSECUTIVE zero-progress polls (reset on any progress) so it measures a
38 * genuinely stalled reader (~BRIX_POOL_SEND_WAIT_MAX × _MS), never frame
39 * size. Returns NGX_OK (watermark reached end), NGX_AGAIN (budget
40 * exhausted — the caller hands the tail to the ring), NGX_ERROR (hard
41 * socket error, t->pool_send_errno set).
42 */
43static ngx_int_t
44brix_pool_send_range(brix_pgread_aio_t *t, const u_char *base, size_t end)
45{
46 ssize_t n;
47 ngx_uint_t waits;
48 struct pollfd pfd;
49
50 waits = 0;
51
52 while (t->pool_sent < end) {
53 n = send(t->sec_c->fd, base + t->pool_sent, end - t->pool_sent,
54 MSG_NOSIGNAL);
55
56 if (n > 0) {
57 t->pool_sent += (size_t) n;
58 waits = 0;
59 continue;
60 }
61
62 if (n == -1 && errno == EINTR) {
63 continue;
64 }
65
66 if (n == -1 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
67 if (waits >= BRIX_POOL_SEND_WAIT_MAX) {
68 return NGX_AGAIN; /* stalled reader: hand the tail over */
69 }
70 waits++;
71 pfd.fd = t->sec_c->fd;
72 pfd.events = POLLOUT;
73 pfd.revents = 0;
74 (void) poll(&pfd, 1, BRIX_POOL_SEND_WAIT_MS);
75 continue;
76 }
77
78 t->pool_send_errno = (n == -1 && errno != 0) ? errno : EPIPE;
79 return NGX_ERROR;
80 }
81
82 return NGX_OK;
83}

85/*
86 * brix_pgread_pool_send — §1.2: send the finished single [status|pages] frame
87 * on the SECONDARY's cleartext socket from THIS pool thread (the data is
88 * cache-hot from the read+CRC pass; the event loop never touches the bytes).
89 *
90 * Concurrency contract: the per-connection send token (CAS) plus send_busy —
91 * a busy ring means the head frame may be mid-send, and interleaving into it
92 * would corrupt the stream (whole frames may legally reorder, frame BYTES may
93 * not). Declining is always safe: the done epilogue queues the frame on the
94 * ring exactly as before pool-send existed.
95 *
96 * Reports through t->pool_sent / pool_sent_all / pool_token_held /
97 * pool_send_errno; on a short send within the wait budget the token is KEPT
98 * (pool_token_held) so the done epilogue can park the remainder at the ring
99 * HEAD before releasing it.
100 */
101void
102brix_pgread_pool_send(brix_pgread_aio_t *t)
103{
104 brix_ctx_t *sec = t->sec_ctx;
105 u_char *p;
106 ngx_int_t rc;
107
108 if (t->nread < 0) {
109 return; /* I/O error: the done handler answers on the PRIMARY
110 * control stream; the secondary wire stays untouched */
111 }
112
113 if (!ngx_atomic_cmp_set(&sec->out.send_token, 0, 1)) {
114 return; /* the event loop owns the socket — ring path serves it */
115 }
116
117 if (sec->out.send_busy) {
118 /* Parked frames exist and the head one may be mid-frame: sending now
119 * could interleave into it. Decline; the ring keeps wire order. */
120 sec->out.send_token = 0;
121 return;
122 }
123
124 p = t->scratch - sizeof(ServerStatusResponse_pgRead);
125 brix_build_pgread_status_sid(t->streamid, t->offset,
126 (uint32_t) t->out_size, kXR_FinalResult,
127 (ServerStatusResponse_pgRead *) p);
128 t->pool_image_len = sizeof(ServerStatusResponse_pgRead) + t->out_size;
129 t->pool_frames = 1;
130
131 rc = brix_pool_send_range(t, p, t->pool_image_len);
132
133 if (rc == NGX_OK) {
134 t->pool_sent_all = 1;
135 sec->out.send_token = 0;
136 return;
137 }
138
139 if (rc == NGX_ERROR) {
140 sec->out.send_token = 0; /* dead socket: nothing more rides it */
141 return;
142 }
143
144 /* Short send inside the wait budget: keep the token — the done epilogue
145 * parks the remainder at the ring head, THEN releases it (a worker that
146 * wins the token next must already see send_busy = 1). */
147 t->pool_token_held = 1;
148}

150/* §1.3 chunk sizing: cuts land on the ABSOLUTE 4 KiB page grid (the
151 * client's page accounting is grid-absolute), so a chunk runs from cur to
152 * the next grid line at most BRIX_PGREAD_STREAM_CHUNK away, capped by the
153 * bytes remaining. */
154static size_t
155brix_pgread_chunk_len(off_t cur, size_t left)
156{
157 off_t grid_end;
158 size_t clen;
159
160 grid_end = (off_t) (((uint64_t) cur + (uint64_t) BRIX_PGREAD_STREAM_CHUNK)
161 & ~((uint64_t) kXR_pgPageSZ - 1));
162 clen = (size_t) (grid_end - cur);
163 return (clen > left) ? left : clen;
164}

166/* Read error inside the §1.3 chunk loop. Wire untouched → classic error
167 * shape (token released, nread = -1, not chunked; the done handler answers
168 * on the PRIMARY control stream). After a committed partial → latch
169 * chunk_error so the done epilogue terminates the train with kXR_error on
170 * the SECONDARY under the request sid. Returns 1 when the caller must
171 * return (classic shape), 0 when it must break out of the loop. */
172static ngx_flag_t
173brix_pgread_stream_read_error(brix_pgread_aio_t *t, int err)
174{
175 t->io_errno = err;
176 if (t->pool_sent == 0) {
177 t->sec_ctx->out.send_token = 0;
178 t->nread = -1;
179 t->pool_chunked = 0;
180 return 1;
181 }
182 t->chunk_error = 1;
183 return 0;
184}

186/*
187 * brix_pgread_pool_stream — §1.3 chunked pgread streaming: read, CRC-encode
188 * and SEND the reply as a train of kXR_PartialResult status frames (final
189 * chunk kXR_FinalResult), all from THIS pool thread under one send-token
190 * hold.
191 *
192 * WHY: with one request in flight per substream, per-request wall latency IS
193 * the throughput; a monolithic frame makes the client wait the full
194 * read+CRC of the request before its first byte. Chunking overlaps the
195 * client's receive+verify of chunk k with this thread's read+CRC of chunk
196 * k+1 — exactly how the reference server streams ~1.4 MiB partials.
197 *
198 * HOW: chunk frames are laid out back-to-back in the slot buffer
199 * ([hdr|enc][hdr|enc]…), so the already-parked/park-front machinery applies
200 * verbatim to the image tail: t->pool_sent is a single watermark into the
201 * image and a stall parks [pool_sent, pool_image_len) at the ring head.
202 * Chunk cuts land on the absolute 4 KiB page grid (the client's page
203 * accounting is grid-absolute); a stall stops sending but keeps BUILDING the
204 * remaining chunks so the parked tail is always a complete frame train.
205 * A read error before any byte hit the wire reports the classic error shape
206 * (error on the PRIMARY); after a committed partial it sets t->chunk_error
207 * and the done epilogue queues kXR_error on the SECONDARY under the request
208 * sid, behind any parked tail.
209 *
210 * Returns 1 when this thread took ownership of the reply (the caller must
211 * not run the classic read path), 0 when the token was unavailable — the
212 * classic single-frame path serves the request unchanged.
213 */

275 }

278 }

282 break; /* EOF: that frame already went out as kXR_FinalResult */
283 }
284 w += sizeof(ServerStatusResponse_pgRead) + job.out_size;
285 cur += (off_t) clen;
286 left -= clen;
287 }
288
289 t->pool_image_len = image;
290

292 sec->out.send_token = 0; /* dead socket: nothing more rides it */
293 return 1;
294 }
295
296 if (t->pool_sent == image) {
297 if (!t->chunk_error) {
298 t->pool_sent_all = 1;
299 }
300 sec->out.send_token = 0;
301 return 1;
302 }
303
304 /* Unsent image tail (stall, possibly plus a later chunk_error): keep the
305 * token — the done epilogue parks the tail at the ring head, THEN
306 * releases it (§1.2). */
307 t->pool_token_held = 1;
308 return 1;
