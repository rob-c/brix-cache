1/*
2 * brix_fault_toxic.h — named, stackable, per-direction fault toxics (C1).
3 *
4 * The core owns the two live base levers (g_up/g_down); this module holds a
5 * separately addressable table of *named* single-effect toxics that compose ON
6 * TOP of that base into the relay's per-buffer snapshot. Naming lets an operator
7 * or test add and remove one discrete effect at a time — `toxic add slow
8 * latency 100 up`, `toxic remove slow` — without disturbing the base levers or
9 * the other toxics, exactly the Toxiproxy model. Grammar:
10 * toxic add <name> <type> <value> [up|down|both] (dir optional, default both)
11 * toxic remove <name>
12 * toxic list [json]
13 * The core `clear` verb empties the whole table (via fp_toxic_reset).
14 */

22/* Handle a `toxic <sub> ...` control command (add / remove / list). `args`
23 * is the text after the "toxic" verb (mutated). Always writes a reply; returns
24 * 1 (the verb is always consumed here once the core has matched "toxic"). */
25int fp_toxic_cmd(char *args, char *reply, size_t rsz);

27/* Fast-path gate: 1 if any toxic could affect this direction (is_up != 0 ⇒ the
28 * client→upstream leg). Lock-free so the zero-toxic relay path pays nothing. */
29int fp_toxic_active(int is_up);
30
31/* Compose every active toxic for this direction onto `*snap` (a caller-owned
32 * copy of the base levers — never the live levers). Delays stack additively,
33 * probabilities add (capped), and bottleneck fields take the tightest value. */
34void fp_toxic_compose(int is_up, lever_t *snap);

36/* Drop every toxic (hooked into the core's clear_all). */
37void fp_toxic_reset(void);
