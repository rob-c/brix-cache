1/* brixcvmfs_ingest.c — `brixcvmfs ingest` front-end + the `dir` verb
2 * (phase-104 D9). Unprivileged, tool-surface only (G14): no FUSE, no root.
3 *
4 * brixcvmfs ingest dir <src_dir> --repo <repo_dir> --prefix /sw/foo/1.2
5 * [--delete] [--follow-symlinks=no] [--dry-run] [--keys-dir D]
6 * [--chunk-size N] [--no-wait]
7 *
8 * The folder → Tier-0 path: <src_dir> is scanned exactly as a transaction
9 * upper tree (cvmfs_changeset_scan — containment, hardlink groups, xattrs),
10 * the changeset is re-rooted under --prefix (cvmfs_changeset_reprefix), and
11 * the phase-96 publish engine applies it. Without --delete the ingest is
12 * add/overwrite-only (the safe default); --delete marks the prefix root
13 * opaque so published content absent from src is removed (mirror-exact).
14 *
15 * A src tree is NOT an overlay upper, so names spelling the reserved
16 * ".brix.wh./.brix.opq" grammar are refused rather than interpreted, and
17 * symlinks are stored verbatim — never followed (the scan's O_NOFOLLOW
18 * discipline; --follow-symlinks=no is the only accepted spelling).
19 *
20 * Serialization: the phase-96 transaction lock (.brixtxn/lock) is taken for
21 * the publish leg — `repo transaction` and `ingest` can never interleave.
22 */

44int bci_mkdir_p(const char *path, unsigned mode) {
45 char buf[ING_PATH_MAX];
46 if (snprintf(buf, sizeof(buf), "%s", path) >= (int) sizeof(buf))
47 return -1;
48 for (char *p = buf + 1; *p != '\0'; p++) {
49 if (*p != '/') continue;
50 *p = '\0';
51 if (mkdir(buf, mode) != 0 && errno != EEXIST) return -1;
52 *p = '/';
53 }
54 return mkdir(buf, mode) == 0 || errno == EEXIST ? 0 : -1;
55}

67int bci_write_atomic(const char *path, const char *data) {
68 char dir[ING_PATH_MAX], tmp[ING_PATH_MAX + 8];
69 if (snprintf(dir, sizeof(dir), "%s", path) >= (int) sizeof(dir))
70 return -1;
71 char *slash = strrchr(dir, '/');
72 if (slash != NULL) {
73 *slash = '\0';
74 if (bci_mkdir_p(dir, 0755) != 0) return -1;
75 }
76 snprintf(tmp, sizeof(tmp), "%s.tmp", path);
77 FILE *f = fopen(tmp, "w");
78 if (f == NULL) return -1;
79 int ok = fputs(data, f) >= 0;
80 ok = fclose(f) == 0 && ok;
81 if (!ok || rename(tmp, path) != 0) {
82 unlink(tmp);
83 return -1;
84 }
85 return 0;
86}

95int bci_lock_acquire(const char *repo_dir, int no_wait) {
96 char txn[ING_PATH_MAX], lock[ING_PATH_MAX + 8];
97 snprintf(txn, sizeof(txn), "%s/.brixtxn", repo_dir);
98 snprintf(lock, sizeof(lock), "%s/lock", txn);
99 int waited = 0;
100 for (;;) {
101 if (mkdir(txn, 0755) != 0 && errno != EEXIST)
102 return bci_fail(ING_FAIL, "cannot create transaction dir", txn);
103 if (brixcvmfs_tx_lock_take(lock) == 0)
104 return ING_OK;
105 /* ENOENT: the holder retired .brixtxn between mkdir and open — retry */
106 if (errno != EEXIST && errno != ENOENT)
107 return bci_fail(ING_FAIL, "cannot take transaction lock", lock);
108 if (errno == EEXIST) {
109 char upper[ING_PATH_MAX + 16];
110 struct stat st;
111 snprintf(upper, sizeof(upper), "%s/upper", txn);
112 /* A crashed ingest leaves a lock with nothing behind it; a repo
113 * transaction always has its durable upper — never broken. */
114 if (lstat(upper, &st) != 0 && brixcvmfs_tx_lock_stale(lock)) {
115 fprintf(stderr,
116 "brixcvmfs ingest: breaking stale lock (holder gone)\n");
117 unlink(lock);
118 continue;
119 }
120 char who[32];
121 snprintf(who, sizeof(who), "pid %d", brixcvmfs_tx_lock_pid(lock));
122 if (no_wait)
123 return bci_fail(ING_BUSY, "transaction lock is held", who);
124 if (!waited) {
125 fprintf(stderr,
126 "brixcvmfs ingest: waiting for transaction lock (%s)\n",
127 who);
128 waited = 1;
129 }
130 struct timespec ts = { 0, 200 * 1000 * 1000 };
131 nanosleep(&ts, NULL);
132 }
133 }
134}

153int bci_memo_refs(const char *memo_dir, const char *digest, const char *skip) {
154 DIR *d = opendir(memo_dir);
155 if (d == NULL) return 0;
156 int n = 0;
157 struct dirent *e;
158 while ((e = readdir(d)) != NULL) {
159 if (strcmp(e->d_name, ".") == 0 || strcmp(e->d_name, "..") == 0)
160 continue;
161 char sub[ING_PATH_MAX];
162 struct stat st;
163 if (snprintf(sub, sizeof(sub), "%s/%s", memo_dir, e->d_name)
164 >= (int) sizeof(sub)
165 || lstat(sub, &st) != 0)
166 continue;
167 if (S_ISDIR(st.st_mode)) {
168 n += bci_memo_refs(sub, digest, skip);
169 continue;
170 }
171 if (strcmp(sub, skip) == 0) continue;
172 char line[ING_PATH_MAX + 128];
173 if (bci_read_line(sub, line, sizeof(line)) != 0) continue;
174 const char *dig = bci_memo_digest(line);
175 if (dig != NULL && strcmp(dig, digest) == 0) n++;
176 }
177 closedir(d);
178 return n;
179}

227/* Parse one dir-ingest option while the caller owns argv iteration. */
228static int ing_dir_option(int argc, char **argv, int *i, ing_dir_opts_t *o,
229 const char **prefix) {
230 const char *arg = argv[*i];
231
232 if (strcmp(arg, "--repo") == 0 && *i + 1 < argc)
233 o->repo = argv[++*i];
234 else if (strcmp(arg, "--prefix") == 0 && *i + 1 < argc)
235 *prefix = argv[++*i];
236 else if (strcmp(arg, "--keys-dir") == 0 && *i + 1 < argc)
237 o->keys_dir = argv[++*i];
238 else if (strcmp(arg, "--chunk-size") == 0 && *i + 1 < argc)
239 o->chunk_size = atol(argv[++*i]);
240 else if (strcmp(arg, "--delete") == 0)
241 o->del = 1;
242 else if (strcmp(arg, "--dry-run") == 0)
243 o->dry_run = 1;
244 else if (strcmp(arg, "--no-wait") == 0)
245 o->no_wait = 1;
246 else if (strcmp(arg, "--follow-symlinks=no") == 0)
247 return ING_OK;
248 else if (strncmp(arg, "--follow-symlinks=", 18) == 0)
249 return bci_fail(ING_USAGE, "symlinks are never followed", arg);
250 else
251 return ing_dir_usage();
252 return ING_OK;
253}

255static int ing_dir_parse(int argc, char **argv, ing_dir_opts_t *o) {
256 const char *prefix = NULL;
257 memset(o, 0, sizeof(*o));
258 if (argc < 2 || argv[1][0] == '-') return ing_dir_usage();
259 o->src = argv[1];
260 for (int i = 2; i < argc; i++) {
261 int rc = ing_dir_option(argc, argv, &i, o, &prefix);
262 if (rc != ING_OK) return rc;
263 }
264 if (o->repo == NULL || prefix == NULL) return ing_dir_usage();
265 size_t n = snprintf(o->prefix, sizeof(o->prefix), "%s", prefix);
266 if (n >= sizeof(o->prefix))
267 return bci_fail(ING_USAGE, "bad --prefix", "too long");
268 while (n > 1 && o->prefix[n - 1] == '/') o->prefix[--n] = '\0';
269 return ING_OK;
270}

272/* A src tree is not an overlay upper: reserved grammar is refused, never
273 * interpreted (a whiteout scans as DELETE, a .brix.opq as the opaque flag).
274 * The upper ROOT has no ADD_DIR op to carry the opaque flag, so a top-level
275 * .brix.opq must be probed directly — the scan drops it silently. */
276static int ing_dir_grammar_check(const char *src, const cvmfs_changeset_t *cs) {
277 char opq[ING_PATH_MAX + 16];
278 struct stat st;
279 snprintf(opq, sizeof(opq), "%s/.brix.opq", src);
280 if (lstat(opq, &st) == 0)
281 return bci_fail(ING_FAIL,
282 "src tree carries reserved overlay grammar (.brix.opq)", src);
283 for (size_t i = 0; i < cs->n; i++) {
284 if (cs->v[i].op == CVMFS_CH_DELETE)
285 return bci_fail(ING_FAIL,
286 "src tree carries reserved overlay grammar (.brix.wh.*)",
287 cs->v[i].path);
288 if (cs->v[i].op == CVMFS_CH_ADD_DIR && cs->v[i].opaque)
289 return bci_fail(ING_FAIL,
290 "src tree carries reserved overlay grammar (.brix.opq)",
291 cs->v[i].path);
292 }
293 return ING_OK;
294}

299 for (size_t i = 0; i < cs->n; i++)

301 && strcmp(cs->v[i].path, prefix) == 0)

317static int ing_dir_publish(const ing_dir_opts_t *o, const cvmfs_changeset_t *cs) {
318 cvmfs_publish_opts_t po;
319 char err[1024];
320 long rev = 0;
321 memset(&po, 0, sizeof(po));
322 po.repo_dir = o->repo;
323 po.keys_dir = o->keys_dir;
324 po.chunk_size = o->chunk_size;
325 int rc = bci_lock_acquire(o->repo, o->no_wait);
326 if (rc != ING_OK) return rc;
327 rc = cvmfs_publish_run(&po, cs, &rev, err, sizeof(err)) == 0
328 ? ING_OK : bci_fail(ING_FAIL, "publish failed", err);
329 bci_lock_release(o->repo);
330 if (rc == ING_OK)
331 printf("published revision %ld (%zu changes under %s)\n",
332 rev, cs->n, o->prefix);
333 return rc;
334}

336static int ing_dir_main(int argc, char **argv) {
337 ing_dir_opts_t o;
338 cvmfs_changeset_t cs;
339 char err[1024];
340 int rc = ing_dir_parse(argc, argv, &o);
341 if (rc != ING_OK) return rc;
342 rc = bci_prefix_check(o.prefix);
343 if (rc != ING_OK) return rc;
344 if (cvmfs_changeset_scan(o.src, &cs, err, sizeof(err)) != 0)
345 return bci_fail(ING_FAIL, "src scan failed", err);
346 rc = ing_dir_grammar_check(o.src, &cs);
347 if (rc == ING_OK
348 && cvmfs_changeset_reprefix(&cs, o.prefix, err, sizeof(err)) != 0)
349 rc = bci_fail(ING_FAIL, "prefix remap failed", err);
350 if (rc == ING_OK) {
351 if (o.del) ing_dir_mark_mirror(&cs, o.prefix);
352 rc = o.dry_run ? ing_dir_dry_run(&cs, &o) : ing_dir_publish(&o, &cs);
353 }
354 cvmfs_changeset_free(&cs);
355 return rc;
356}
