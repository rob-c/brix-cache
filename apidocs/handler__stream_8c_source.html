20/*
21 * brix_dirlist_entry_fmt_t — formatted per-entry output produced by
22 * brix_dirlist_entry_meta and consumed by brix_dirlist_append_entry.
23 *
24 * Carries the rendered stat line, the optional checksum token, whether stat
25 * data is valid for this entry, and the total wire bytes the entry needs so
26 * the streaming loop can decide when to flush a kXR_oksofar chunk.
27 */
28typedef struct {
29 char statbuf[256]; /* rendered stat line */
30 char cksum_token[EVP_MAX_MD_SIZE * 2 + 64]; /* "algo:hex" token */
31 ngx_flag_t have_stat; /* statbuf is valid */
32 size_t need; /* wire bytes required */
33} brix_dirlist_entry_fmt_t;

35/*
36 * brix_dirlist_name_is_unsafe — detect control characters in a directory
37 * entry name that would corrupt the newline-delimited kXR_dirlist wire format.
38 *
39 * The XRootD dirlist response separates entries with '\n'. A filename
40 * containing '\n' (or any other byte < 0x20) would be misinterpreted as a
41 * record separator by the client. Such filenames are skipped silently.
42 *
43 * Returns 1 (unsafe, skip this entry), 0 (safe to include).
44 */
45static ngx_flag_t
46brix_dirlist_name_is_unsafe(const char *name)
47{
48 const u_char *p;
49
50 for (p = (const u_char *) name; *p != '\0'; p++) {
51 if (*p < 0x20 || *p == 0x7f) {
52 return 1;
53 }
54 }
55
56 return 0;
57}

59/*
60 * brix_dirlist_entry_skip — decide whether a directory entry must be
61 * excluded from the listing.
62 *
63 * Skips "." and "..", this gateway's internal control artifacts (e.g. the
64 * checkpoint recovery lock — a stock XRootD export shows no such files;
65 * mirror that), internal metadata/staging artifacts (cache sidecars, stage
66 * markers, in-flight upload temps — never enumerable to a client), and names
67 * with control bytes that would corrupt the wire format (logged at WARN).
68 *
69 * Returns 1 (skip this entry), 0 (include it).
70 */
71static ngx_flag_t
72brix_dirlist_entry_skip(ngx_log_t *log, const char *name)
73{
74 char safe_name[256];
75
76 if (name[0] == '.' && (name[1] == '\0' ||
77 (name[1] == '.' && name[2] == '\0')))
78 {
79 return 1;
80 }
81
82 if (ngx_strncmp(name, ".nginx-xrootd",
83 sizeof(".nginx-xrootd") - 1) == 0)
84 {
85 return 1;
86 }
87
88 if (brix_is_internal_name(name)) {
89 return 1;
90 }
91
92 if (brix_dirlist_name_is_unsafe(name)) {
93 brix_sanitize_log_string(name, safe_name, sizeof(safe_name));
94 ngx_log_error(NGX_LOG_WARN, log, 0,
95 "brix: dirlist skipping entry with control bytes \"%s\"",
96 safe_name);
97 return 1;
98 }
99
100 return 0;
101}

103/* ---- Omit a not-online entry under the kXR_online option ----
104 *
105 * WHAT: Returns 1 when the kXR_online dirlist option is set and the entry's
106 * file data is not currently online (nearline/offline residency), 0 to
107 * include the entry in the listing.
108 *
109 * WHY: kXR_online asks for "entries whose data is currently online" only
110 * (protocol/flags.h). The residency authority is the same
111 * brix_vfs_residency() decorator walk the stat/statx handlers advertise
112 * kXR_offline through, so the filter can never disagree with what a
113 * stat of the same entry reports.
114 *
115 * HOW: 1. Pass directories through — only file data can be offline.
116 * 2. Build the entry's absolute path; on overflow include the entry
117 * (the same fail-open choice the checksum token makes with
118 * "algo:none" — a too-long name must not vanish from the listing).
119 * 3. Probe residency with a transient read-only VFS ctx and omit only
120 * on a definite NEARLINE/OFFLINE answer (the stat predicate: a
121 * probe error or an export with no nearline tier reports online).
122 */
123static int
124brix_dirlist_entry_not_online(brix_ctx_t *ctx, ngx_connection_t *c,
125 brix_dirlist_walk_t *walk, const char *name,
126 const brix_vfs_stat_t *vst)
127{
128 brix_vfs_ctx_t residency_ctx;
129 brix_sd_residency_t residency;
130 char entry_path[PATH_MAX];
131 int written;
132
133 if (!walk->want_online || vst->is_directory) {
134 return 0;
135 }
136
137 written = snprintf(entry_path, sizeof(entry_path), "%s/%s",
138 walk->full_path, name);
139 if (written < 0 || (size_t) written >= sizeof(entry_path)) {
140 return 0;
141 }
142
143 brix_vfs_ctx_init(&residency_ctx, c->pool, c->log, BRIX_PROTO_ROOT,
144 walk->conf->common.root_canon, NULL, BRIX_VFS_MUTATION_READ_ONLY,
145 0 /* is_tls */, ctx->identity, entry_path);
146 brix_root_vfs_bind_session(ctx, walk->conf, &residency_ctx);
147
148 return brix_vfs_residency(&residency_ctx, &residency, NULL) == NGX_OK
149 && (residency == BRIX_SD_RES_NEARLINE
150 || residency == BRIX_SD_RES_OFFLINE);
151}

153/*
154 * brix_dirlist_entry_meta — render the optional stat line and checksum
155 * token for one entry, accumulating the wire bytes it needs into fmt->need.
156 *
157 * readdir already did the no-follow lstat (and skipped any entry whose stat
158 * failed), so vst is valid whenever want_stat is set. The dcksm variant
159 * uses the dcksm stat-body format; the checksum token falls back to
160 * "algo:none" when the entry path would overflow PATH_MAX.
161 *
162 * fmt->need must be pre-seeded with the bare name length + '\n'.
163 */
164static void
165brix_dirlist_entry_meta(ngx_log_t *log, brix_dirlist_walk_t *walk,
166 const char *name, const brix_vfs_stat_t *vst,
167 brix_dirlist_entry_fmt_t *fmt)
168{
169 struct stat entry_st;
170 char entry_path[PATH_MAX];
171
172 fmt->cksum_token[0] = '\0';
173 fmt->have_stat = 0;
174
175 if (walk->want_stat) {
176 ngx_memzero(&entry_st, sizeof(entry_st));
177 entry_st.st_mode = (mode_t) vst->mode;
178 entry_st.st_size = vst->size;
179 entry_st.st_mtime = vst->mtime;
180 entry_st.st_ctime = vst->ctime;
181 entry_st.st_ino = vst->ino;
182 entry_st.st_dev = vst->dev;
183 entry_st.st_uid = vst->uid;
184 entry_st.st_gid = vst->gid;
185 entry_st.st_blocks = vst->blocks;
186
187 fmt->have_stat = 1;
188 if (walk->want_cksum) {
189 brix_dirlist_make_dcksm_stat_body(&entry_st, fmt->statbuf,
190 sizeof(fmt->statbuf));
191 } else {
192 brix_make_stat_body(&entry_st, 0, 0, fmt->statbuf,
193 sizeof(fmt->statbuf));
194 }
195 fmt->need += strlen(fmt->statbuf) + 1;
196 }
197
198 if (walk->want_cksum && fmt->have_stat) {
199 int n;
200
201 n = snprintf(entry_path, sizeof(entry_path), "%s/%s",
202 walk->full_path, name);
203 if (n < 0 || (size_t) n >= sizeof(entry_path)) {
204 snprintf(fmt->cksum_token, sizeof(fmt->cksum_token),
205 "%s:none", walk->cksum_algo);
206 } else {
207 brix_dirlist_cksum_req_t creq;
208
209 ngx_memzero(&creq, sizeof(creq));
210 creq.log = log;
211 creq.dfd = brix_vfs_dir_fd(walk->dh);
212 creq.name = name;
213 creq.path = entry_path;
214 creq.st = &entry_st;
215 creq.algo = walk->cksum_algo;
216 creq.policy = walk->mutation_policy;
217
218 brix_dirlist_checksum_token(&creq, fmt->cksum_token,
219 sizeof(fmt->cksum_token));
220 }
221 fmt->need += strlen(fmt->cksum_token) + sizeof(" [ ]") - 1;
222 }
223}

225/*
226 * brix_dirlist_flush_chunk — frame the accumulated chunk as a kXR_oksofar
227 * continuation and queue it, resetting the accumulator for the next chunk.
228 *
229 * Returns the brix_queue_response result (NGX_OK on success).
230 */
231static ngx_int_t
232brix_dirlist_flush_chunk(brix_ctx_t *ctx, ngx_connection_t *c,
233 brix_dirlist_walk_t *walk)
234{
235 ngx_int_t rc;
236
237 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_oksofar,
238 (uint32_t) walk->chunk_pos,
239 (ServerResponseHdr *) walk->chunk);
240
241 rc = brix_queue_response(ctx, c, walk->chunk,
242 XRD_RESPONSE_HDR_LEN + walk->chunk_pos);
243 if (rc != NGX_OK) {
244 return rc;
245 }
246
247 walk->chunk_pos = 0;
248 return NGX_OK;
249}

251/*
252 * brix_dirlist_append_entry — copy one formatted entry into the chunk
253 * accumulator: name '\n' [statline [" [ token ]"] '\n'].
254 *
255 * The caller has already guaranteed fmt->need bytes of headroom by flushing
256 * a full chunk beforehand, so the copies cannot overflow chunk_cap.
257 */
258static void
259brix_dirlist_append_entry(brix_dirlist_walk_t *walk, const char *name,
260 size_t nlen, const brix_dirlist_entry_fmt_t *fmt)
261{
262 u_char *data = walk->chunk + XRD_RESPONSE_HDR_LEN;
263
264 ngx_memcpy(data + walk->chunk_pos, name, nlen);
265 walk->chunk_pos += nlen;
266 data[walk->chunk_pos++] = '\n';
267
268 if (walk->want_stat && fmt->have_stat) {
269 size_t slen;
270
271 slen = strlen(fmt->statbuf);
272 ngx_memcpy(data + walk->chunk_pos, fmt->statbuf, slen);
273 walk->chunk_pos += slen;
274
275 if (walk->want_cksum) {
276 int n;
277
278 n = snprintf((char *) data + walk->chunk_pos,
279 walk->chunk_cap - walk->chunk_pos,
280 " [ %s ]", fmt->cksum_token);
281 if (n > 0) {
282 walk->chunk_pos += (size_t) n;
283 }
284 }
285
286 data[walk->chunk_pos++] = '\n';
287 }
288}

290/*
291 * brix_dirlist_send_final — frame the last accumulated chunk as kXR_ok to
292 * signal end-of-listing, log the access, and bump the success metric.
293 *
294 * The trailing '\n' of the final entry is replaced by a NUL terminator (an
295 * empty listing sends a zero-length body).
296 *
297 * Returns the brix_queue_response result.
298 */
299static ngx_int_t
300brix_dirlist_send_final(brix_ctx_t *ctx, ngx_connection_t *c,
301 brix_dirlist_walk_t *walk)
302{
303 u_char *data = walk->chunk + XRD_RESPONSE_HDR_LEN;
304 size_t final_len;
305
306 if (walk->chunk_pos == 0) {
307 final_len = 0;
308 } else {
309 data[walk->chunk_pos - 1] = '\0';
310 final_len = walk->chunk_pos;
311 }
312
313 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok,
314 (uint32_t) final_len,
315 (ServerResponseHdr *) walk->chunk);
316
317 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
318 "brix: kXR_dirlist final chunk %uz bytes",
319 walk->chunk_pos);
320
321 brix_log_access(ctx, c, "DIRLIST", walk->reqpath,
322 walk->want_cksum ? "dcksm"
323 : (walk->want_stat ? "stat" : "-"),
324 1, 0, NULL, 0);
325 BRIX_OP_OK(ctx, BRIX_OP_DIRLIST);
326
327 return brix_queue_response(ctx, c, walk->chunk,
328 XRD_RESPONSE_HDR_LEN + final_len);
329}

331/*
332 * brix_dirlist_stream_entries — the chunked streaming loop: enumerate the
333 * open directory, format each visible entry, flush full 64KB chunks as
334 * kXR_oksofar frames, and finish with the kXR_ok final frame.
335 *
336 * Owns walk->dh from here on: the handle is closed on every exit path
337 * (queue failure mid-stream or normal end-of-directory).
338 *
339 * Returns the final queue result (NGX_OK on success).
340 */
341ngx_int_t
342brix_dirlist_stream_entries(brix_ctx_t *ctx, ngx_connection_t *c,
343 brix_dirlist_walk_t *walk)
344{
345 ngx_str_t entry_name;
346 brix_vfs_stat_t entry_vst;
347 ngx_int_t rc;
348
349 /* kXR_online needs per-entry stat data (to pass directories through), so
350 * it turns on the readdir lstat like dstat does — the response format
351 * still emits stat lines only under want_stat. */
352 /* Borrowed names (zero-copy): each entry is fully consumed — skip
353 * checks, meta, chunk append — before the next readdir invalidates it. */
354 while (brix_vfs_readdir_borrow(walk->dh, &entry_name,
355 (walk->want_stat || walk->want_online)
356 ? &entry_vst : NULL) == NGX_OK) {
357 const char *name = (char *) entry_name.data;
358 size_t nlen = entry_name.len;
359 brix_dirlist_entry_fmt_t fmt;
360
361 if (brix_dirlist_entry_skip(c->log, name)) {
362 continue;
363 }
364
365 if (brix_dirlist_entry_not_online(ctx, c, walk, name, &entry_vst)) {
366 continue;
367 }
368
369 fmt.need = nlen + 1;
370 brix_dirlist_entry_meta(c->log, walk, name, &entry_vst, &fmt);
371
372 if (walk->chunk_pos + fmt.need > walk->chunk_cap) {
373 rc = brix_dirlist_flush_chunk(ctx, c, walk);
374 if (rc != NGX_OK) {
375 brix_vfs_closedir(walk->dh, c->log);
376 return rc;
377 }
378 }
379
380 brix_dirlist_append_entry(walk, name, nlen, &fmt);
381 }
382
383 brix_vfs_closedir(walk->dh, c->log);
384
385 return brix_dirlist_send_final(ctx, c, walk);
386}
