1/*
2 * auth/krb5/kxr_wire.c — kXR krb5 auth wire codec (§5.7). See kxr_wire.h.
3 *
4 * The frame vocabulary is byte-frozen and rendered by hand here (no XProtocol
5 * structs) so the unit links into both nginx and the standalone krb5 harness:
6 *
7 * ClientAuthRequest (24 bytes, sent per leg):
8 * [0..1] streamid = {0,1} (the connector stream)
9 * [2..3] requestid = kXR_auth, big-endian
10 * [4..15] reserved = 0
11 * [16..19] credtype = "krb5"
12 * [20..23] dlen = token length, big-endian
13 * then <dlen> credential (GSS token) bytes.
14 *
15 * ServerResponseHeader (8 bytes, read per leg):
16 * [0..1] streamid (echoed, ignored here)
17 * [2..3] status big-endian (kXR_ok / kXR_authmore / kXR_error / ...)
18 * [4..7] dlen big-endian — the reply token (or error body) length.
19 */

62ngx_int_t
63brix_krb5_kxr_classify(uint16_t status, u_char *body, uint32_t dlen,
64 ngx_str_t *in_token, int *done)
65{
66 in_token->data = NULL;
67 in_token->len = 0;
68 *done = 0;
69
70 if (status == kXR_authmore) {
71 in_token->data = body; /* borrowed; feed back to gss_init_sec_context */
72 in_token->len = dlen;
73 *done = 0;
74 return NGX_OK;
75 }
76 if (status == kXR_ok) {
77 in_token->data = body; /* may carry a final AP-REP; may be empty */
78 in_token->len = dlen;
79 *done = 1;
80 return NGX_OK;
81 }
82 return NGX_ERROR; /* kXR_error or any unexpected status */
83}

86ngx_int_t
87brix_krb5_kxr_wire(void *wire_ctx, const ngx_str_t *out_token,
88 ngx_str_t *in_token, int *done, ngx_log_t *log)
89{
90 brix_krb5_kxr_wire_t *w = wire_ctx;
91 u_char hdr[24];
92 u_char rhdr[8];
93 uint16_t status;
94 uint32_t dlen;
95 u_char *body = NULL;
96
97 in_token->data = NULL;
98 in_token->len = 0;
99 *done = 0;
100
101 /* Release the previous leg's reply — the engine has consumed it by now. */
102 if (w->reply != NULL) {
103 free(w->reply);
104 w->reply = NULL;
105 }
106
107 /* ClientAuthRequest header + credential payload. */
108 memset(hdr, 0, sizeof hdr);
109 hdr[1] = 1; /* connector stream */
110 put_u16_be(hdr + 2, (uint16_t) kXR_auth);
111 memcpy(hdr + 16, "krb5", 4);
112 put_u32_be(hdr + 20, (uint32_t) out_token->len);
113
114 if (w->send(w->io, hdr, sizeof hdr) != NGX_OK
115 || (out_token->len > 0
116 && w->send(w->io, out_token->data, out_token->len) != NGX_OK))
117 {
118 ngx_log_error(NGX_LOG_ERR, log, 0,
119 "brix: krb5 origin leg - auth request write failed");
120 return NGX_ERROR;
121 }
122
123 /* ServerResponseHeader. */
124 if (w->recv(w->io, rhdr, sizeof rhdr) != NGX_OK) {
125 ngx_log_error(NGX_LOG_ERR, log, 0,
126 "brix: krb5 origin leg - reply header read failed");
127 return NGX_ERROR;
128 }
129 status = get_u16_be(rhdr + 2);
130 dlen = get_u32_be(rhdr + 4);
131
132 if (dlen > w->max_body) {
133 ngx_log_error(NGX_LOG_ERR, log, 0,
134 "brix: krb5 origin leg - reply body %uD exceeds cap %uD",
135 dlen, w->max_body);
136 return NGX_ERROR;
137 }
138 if (dlen > 0) {
139 body = malloc(dlen);
140 if (body == NULL) {
141 return NGX_ERROR;
142 }
143 if (w->recv(w->io, body, dlen) != NGX_OK) {
144 free(body);
145 ngx_log_error(NGX_LOG_ERR, log, 0,
146 "brix: krb5 origin leg - reply body read failed");
147 return NGX_ERROR;
148 }
149 w->reply = body; /* borrowed by the engine until the next call/end */
150 }
151
152 if (brix_krb5_kxr_classify(status, body, dlen, in_token, done) != NGX_OK) {
153 ngx_log_error(NGX_LOG_ERR, log, 0,
154 "brix: krb5 origin leg - origin rejected token (status %ui)",
155 (ngx_uint_t) status);
156 return NGX_ERROR;
157 }
158 return NGX_OK;
159}
