1/*
2 * auth/krb5/kxr_wire.c — kXR krb5 auth wire codec (§5.7). See kxr_wire.h.
3 *
4 * The frame vocabulary is byte-frozen and rendered by hand here (no XProtocol
5 * structs) so the unit links into both nginx and the standalone krb5 harness:
6 *
7 * ClientAuthRequest (24 bytes, sent per leg):
8 * [0..1] streamid = {0,1} (the connector stream)
9 * [2..3] requestid = kXR_auth, big-endian
10 * [4..15] reserved = 0
11 * [16..19] credtype = "krb5"
12 * [20..23] dlen = token length, big-endian
13 * then <dlen> credential (GSS token) bytes.
14 *
15 * ServerResponseHeader (8 bytes, read per leg):
16 * [0..1] streamid (echoed, ignored here)
17 * [2..3] status big-endian (kXR_ok / kXR_authmore / kXR_error / ...)
18 * [4..7] dlen big-endian — the reply token (or error body) length.
19 */

63ngx_int_t
64brix_krb5_kxr_classify(uint16_t status, u_char *body, uint32_t dlen,
65 ngx_str_t *in_token, int *done)
66{
67 in_token->data = NULL;
68 in_token->len = 0;
69 *done = 0;
70
71 if (status == kXR_authmore) {
72 in_token->data = body; /* borrowed; feed back to gss_init_sec_context */
73 in_token->len = dlen;
74 *done = 0;
75 return NGX_OK;
76 }
77 if (status == kXR_ok) {
78 in_token->data = body; /* may carry a final AP-REP; may be empty */
79 in_token->len = dlen;
80 *done = 1;
81 return NGX_OK;
82 }
83 return NGX_ERROR; /* kXR_error or any unexpected status */
84}

87ngx_int_t
88brix_krb5_kxr_wire(void *wire_ctx, const ngx_str_t *out_token,
89 ngx_str_t *in_token, int *done, ngx_log_t *log)
90{
91 brix_krb5_kxr_wire_t *w = wire_ctx;
92 u_char hdr[24];
93 u_char rhdr[8];
94 uint16_t status;
95 uint32_t dlen;
96 u_char *body = NULL;
97
98 in_token->data = NULL;
99 in_token->len = 0;
100 *done = 0;
101
102 /* Release the previous leg's reply — the engine has consumed it by now. */
103 if (w->reply != NULL) {
104 free(w->reply);
105 w->reply = NULL;
106 }
107
108 /* ClientAuthRequest header + credential payload. */
109 memset(hdr, 0, sizeof hdr);
110 hdr[1] = 1; /* connector stream */
111 put_u16_be(hdr + 2, (uint16_t) kXR_auth);
112 memcpy(hdr + 16, "krb5", 4);
113 put_u32_be(hdr + 20, (uint32_t) out_token->len);
114
115 if (w->send(w->io, hdr, sizeof hdr) != NGX_OK
116 || (out_token->len > 0
117 && w->send(w->io, out_token->data, out_token->len) != NGX_OK))
118 {
119 BRIX_LOG(NGX_LOG_ERR, log, 0,
120 "brix: krb5 origin leg - auth request write failed");
121 return NGX_ERROR;
122 }
123
124 /* ServerResponseHeader. */
125 if (w->recv(w->io, rhdr, sizeof rhdr) != NGX_OK) {
126 BRIX_LOG(NGX_LOG_ERR, log, 0,
127 "brix: krb5 origin leg - reply header read failed");
128 return NGX_ERROR;
129 }
130 status = get_u16_be(rhdr + 2);
131 dlen = get_u32_be(rhdr + 4);
132
133 if (dlen > w->max_body) {
134 BRIX_LOG(NGX_LOG_ERR, log, 0,
135 "brix: krb5 origin leg - reply body %uD exceeds cap %uD",
136 dlen, w->max_body);
137 return NGX_ERROR;
138 }
139 if (dlen > 0) {
140 body = malloc(dlen);
141 if (body == NULL) {
142 return NGX_ERROR;
143 }
144 if (w->recv(w->io, body, dlen) != NGX_OK) {
145 free(body);
146 BRIX_LOG(NGX_LOG_ERR, log, 0,
147 "brix: krb5 origin leg - reply body read failed");
148 return NGX_ERROR;
149 }
150 w->reply = body; /* borrowed by the engine until the next call/end */
151 }
152
153 if (brix_krb5_kxr_classify(status, body, dlen, in_token, done) != NGX_OK) {
154 BRIX_LOG(NGX_LOG_ERR, log, 0,
155 "brix: krb5 origin leg - origin rejected token (status %ui)",
156 (ngx_uint_t) status);
157 return NGX_ERROR;
158 }
159 return NGX_OK;
160}
