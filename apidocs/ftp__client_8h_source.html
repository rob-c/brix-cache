1/*
2 * ftp_client.h — private split contract for the client-role GridFTP engine.
3 * Not a public API: include only from client/lib/protocols/ftp/ and the xfer
4 * sibling that routes gsiftp:// copies (client/lib/xfer/copy_gsiftp.c).
5 *
6 * WHAT: one blocking gsiftp://(ftp://) control session — dial, RFC 2228 GSI
7 * login, RFC 959/3659 command exchange, and a stream-mode data channel —
8 * expressed over the pure parser kernels in src/fs/backend/gsiftp/.
9 * WHY: phase-82 gave the tree a GridFTP *gateway* (it only ever emits replies);
10 * xrdcp needs the initiator half to move data to and from the grid's
11 * gsiftp endpoints without libglobus. Keeping the session state in one
12 * small struct lets the control, security, and data concerns live in
13 * separate translation units that share no globals.
14 * HOW: brix_ftp_connect() dials and captures the greeting; brix_ftp_login()
15 * negotiates AUTH GSSAPI (or anonymous); brix_ftp_cmd() sends one command
16 * and parses exactly one reply (transparently GSS-wrapping the command and
17 * unwrapping the 63x reply once the security layer is up); the data-channel
18 * helpers open a passive connection and pump one transfer.
19 */

29/* Both buffers must hold one whole reply line, and the largest of those is an
30 * `ADAT=<base64>` token carrying a TLS flight (server certificate chain) — kilo-
31 * bytes, not the hundreds of bytes a plain FTP reply needs. The session is
32 * therefore heap-allocated by its owner, never a stack local. */
33#define BRIX_FTP_CTL_BUF 32768 /* control receive buffer */
34#define BRIX_FTP_TEXT_MAX 32768 /* final-line reply text kept per command */
35#define BRIX_FTP_CMD_MAX 4096 /* longest command line we will emit */
36#define BRIX_FTP_DATA_CHUNK (1024u * 1024u) /* data-channel transfer buffer */
37
38struct brix_ftp_gss; /* opaque GSI initiator engine (ftp_gsi.c) */
39
40typedef struct {
41 int fd; /* control socket, -1 when closed */
42 int timeout_ms;/* per-poll idle budget for control + data */
43 size_t rlen; /* bytes pending in rbuf */
44 char rbuf[BRIX_FTP_CTL_BUF];
45 struct brix_ftp_gss *gss; /* GSI engine, NULL until AUTH GSSAPI runs */
46 int secure; /* 1 ⇒ commands/replies are GSS-wrapped */
47 int code; /* last reply's 3-digit code */
48 char text[BRIX_FTP_TEXT_MAX]; /* last final-line text */
49 char peer_ip[64]; /* control peer, numeric */
50} brix_ftp_sess;
