1/*
2 * resolv_conf.h — resolv.conf(5) parser (phase-116 W1).
3 *
4 * WHAT: Parses the resolver configuration file the process can see
5 * (nameserver / search / domain / options ndots,timeout,attempts,rotate)
6 * plus the LOCALDOMAIN and RES_OPTIONS environment overrides, into a
7 * fixed-size, allocation-free struct with glibc's defaults pre-applied.
8 * WHY: nginx open-source never reads resolv.conf — every `resolver` directive
9 * is hand-written. BriX fills nginx's resolver slots from this struct
10 * (resolver_build.c) and drives search/ndots semantics (resolve.c), so
11 * one file is the single truth for "what the host resolver would do".
12 * HOW: Pure C, no nginx headers — the same code is compiled standalone by
13 * resolv_conf_unittest.c. Parsing is line-oriented; unknown keywords and
14 * options are ignored (glibc behaviour); limits match glibc (3
15 * nameservers, 6 search domains, ndots<=15, timeout<=30, attempts<=5).
16 * A missing or unreadable file yields the glibc defaults (127.0.0.1,
17 * ndots:1, timeout:5, attempts:2) and reports `defaulted = 1`.
18 */

27#define BRIX_RESOLV_DOMAIN_LEN 256 /* RFC 1035 name limit + NUL */
28#define BRIX_RESOLV_DEFAULT_PATH "/etc/resolv.conf"

30typedef struct {
31 char nameservers[BRIX_RESOLV_MAX_NS][BRIX_RESOLV_NS_LEN];
32 unsigned nnameservers;
33 char search[BRIX_RESOLV_MAX_SEARCH][BRIX_RESOLV_DOMAIN_LEN];
34 unsigned nsearch;
35 unsigned ndots; /* options ndots:N (default 1, max 15) */
36 unsigned timeout; /* options timeout:N (seconds, default 5, max 30) */
37 unsigned attempts; /* options attempts:N (default 2, max 5) */
38 unsigned rotate:1; /* options rotate */
39 unsigned defaulted:1; /* file unreadable: glibc defaults in force */
40 unsigned env_search:1; /* LOCALDOMAIN replaced the file's search list */
41} brix_resolv_conf_t;
