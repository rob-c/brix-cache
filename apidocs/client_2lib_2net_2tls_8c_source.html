1/*
2 * tls.c — in-protocol TLS for the native client (roots:// and GSI+TLS).
3 *
4 * WHAT: Upgrade the live socket to a TLS session (after the kXR_protocol reply,
5 * before kXR_login) and transfer bytes transparently through it. Server
6 * certificate verification against a CA hash dir ($X509_CERT_DIR) is on by
7 * default; a name-check escape (--noverifyhost) relaxes only the hostname.
8 * WHY: XRootD encrypts the control/data stream in-protocol; the same SSL session
9 * then carries login, GSI auth, and all file ops.
10 * HOW: The fd goes non-blocking at upgrade time; SSL_connect and the read/write
11 * loops drive OpenSSL's WANT_READ/WANT_WRITE through the same poll(2) +
12 * timeout discipline the cleartext path uses. sock.c branches to these when
13 * io->ssl != NULL, so no other file changes for TLS. Byte transfer itself
14 * lives in tls_io.c, split out when this file crossed the 600-line cap; the
15 * two helpers both halves need are declared in tls_internal.h.
16 */

45/* poll one fd for `events`; >0 ready, 0 timeout, <0 error (EINTR retried). */
46int
47brix_tls_wait_io(int fd, short events, int timeout_ms)
48{
49 struct pollfd pfd;
50 int pr;
51
52 pfd.fd = fd;
53 pfd.events = events;
54 do {
55 pr = poll(&pfd, 1, timeout_ms);
56 } while (pr < 0 && errno == EINTR && !brix_copy_quit_requested());
57 return pr;
58}

61/* Drive a non-blocking SSL_connect to completion (poll on WANT_*). 0 / -1. */
62static int
63tls_do_connect(SSL *ssl, brix_io *io, int verify_peer, brix_status *st)
64{
65 for (;;) {
66 int ret, err;
67
68 ERR_clear_error();
69 ret = SSL_connect(ssl);
70 if (ret == 1) {
71 break;
72 }
73 err = SSL_get_error(ssl, ret);
74 if (err == SSL_ERROR_WANT_READ || err == SSL_ERROR_WANT_WRITE) {
75 short ev = (err == SSL_ERROR_WANT_READ) ? POLLIN : POLLOUT;
76 if (brix_tls_wait_io(io->fd, ev, io->timeout_ms) <= 0) {
77 brix_status_set(st, XRDC_ESOCK, 0, "TLS handshake timed out");
78 return -1;
79 }
80 continue;
81 }
82 {
83 long vr = SSL_get_verify_result(ssl);
84 if (vr != X509_V_OK) {
85 brix_status_set(st, XRDC_EAUTH, 0, "TLS verify failed: %s",
86 X509_verify_cert_error_string(vr));
87 } else {
88 brix_tls_err(st, XRDC_ESOCK, "SSL_connect");
89 }
90 }
91 return -1;
92 }
93
94 if (verify_peer) {
95 long vr = SSL_get_verify_result(ssl);
96 if (vr != X509_V_OK) {
97 brix_status_set(st, XRDC_EAUTH, 0, "TLS verify failed: %s",
98 X509_verify_cert_error_string(vr));
99 return -1;
100 }
101 }
102 return 0;
103}

105int
106brix_tls_upgrade(brix_conn *c, int verify_peer, int verify_host,
107 const char *ca_dir, brix_status *st)
108{
109 SSL_CTX *ctx;
110 SSL *ssl;
111 int fl;
112
113 ctx = SSL_CTX_new(TLS_client_method());
114 if (ctx == NULL) {
115 brix_tls_err(st, XRDC_ESOCK, "SSL_CTX_new");
116 return -1;
117 }
118 SSL_CTX_set_min_proto_version(ctx, TLS1_2_VERSION);
119
120 if (verify_peer) {
121 SSL_CTX_set_verify(ctx, SSL_VERIFY_PEER, NULL);
122 if (ca_dir != NULL && ca_dir[0] != '\0') {
123 if (SSL_CTX_load_verify_locations(ctx, NULL, ca_dir) != 1) {
124 brix_tls_err(st, XRDC_EAUTH, "load CA dir");
125 SSL_CTX_free(ctx);
126 return -1;
127 }
128 } else {
129 SSL_CTX_set_default_verify_paths(ctx);
130 }
131 /* Permit proxy certs in the chain (GSI deployments). */
132 X509_VERIFY_PARAM_set_flags(SSL_CTX_get0_param(ctx),
133 X509_V_FLAG_ALLOW_PROXY_CERTS);
134 } else {
135 SSL_CTX_set_verify(ctx, SSL_VERIFY_NONE, NULL);
136 }
137
138 ssl = SSL_new(ctx);
139 if (ssl == NULL) {
140 brix_tls_err(st, XRDC_ESOCK, "SSL_new");
141 SSL_CTX_free(ctx);
142 return -1;
143 }
144
145 if (verify_peer && verify_host && c->host[0] != '\0') {
146 SSL_set1_host(ssl, c->host);
147 }
148
149 /* Non-blocking for poll-driven handshake/IO; SSL handles record framing. */
150 fl = fcntl(c->io.fd, F_GETFL, 0);
151 if (fl >= 0) {
152 (void) fcntl(c->io.fd, F_SETFL, fl | O_NONBLOCK);
153 }
154 SSL_set_fd(ssl, c->io.fd);
155 SSL_set_connect_state(ssl);
156
157 if (tls_do_connect(ssl, &c->io, verify_peer, st) != 0) {
158 SSL_free(ssl);
159 SSL_CTX_free(ctx);
160 return -1;
161 }
162
163 c->io.ssl = ssl; /* struct ssl_st* == SSL* */
164 c->ssl_ctx = ctx;
165 return 0;
166}

169/*
170 * Standalone TLS client handshake on an already-connected socket (io->fd), for the
171 * general HTTP(S) client — independent of the root:// in-protocol upgrade. Mirrors
172 * brix_tls_upgrade but takes a bare brix_io + an explicit SNI/verify host. On
173 * success io->ssl holds the live SSL and *out_ctx the SSL_CTX (caller frees both via
174 * brix_tls_client_free). 0 / -1.
175 */
176/* Prepare the client SSL_CTX: load a client certificate for mutual TLS (davs GSI
177 * delegation) when given — the proxy PEM holds cert + private key + EEC chain in
178 * one file, so one path loads both — then configure server verification (allowing
179 * RFC-3820 proxy certs). Any credential/CA failure is fatal: proceeding
180 * anonymously would misrepresent who is connecting. Returns 0, or -1 with *st
181 * set (caller frees ctx). */
182static int
183tls_prepare_ctx(SSL_CTX *ctx, int verify_peer, const char *ca_dir,
184 const char *client_cert, brix_status *st)
185{
186 if (client_cert != NULL && client_cert[0] != '\0') {
187 if (SSL_CTX_use_certificate_chain_file(ctx, client_cert) != 1) {
188 brix_tls_err(st, XRDC_EAUTH, "load client cert");
189 return -1;
190 }
191 if (SSL_CTX_use_PrivateKey_file(ctx, client_cert, SSL_FILETYPE_PEM) != 1) {
192 brix_tls_err(st, XRDC_EAUTH, "load client key");
193 return -1;
194 }
195 if (SSL_CTX_check_private_key(ctx) != 1) {
196 brix_tls_err(st, XRDC_EAUTH, "client cert/key mismatch");
197 return -1;
198 }
199 }
200 if (!verify_peer) {
201 SSL_CTX_set_verify(ctx, SSL_VERIFY_NONE, NULL);
202 return 0;
203 }
204 SSL_CTX_set_verify(ctx, SSL_VERIFY_PEER, NULL);
205 if (ca_dir != NULL && ca_dir[0] != '\0') {
206 if (SSL_CTX_load_verify_locations(ctx, NULL, ca_dir) != 1) {
207 brix_tls_err(st, XRDC_EAUTH, "load CA dir");
208 return -1;
209 }
210 } else {
211 SSL_CTX_set_default_verify_paths(ctx);
212 }
213 X509_VERIFY_PARAM_set_flags(SSL_CTX_get0_param(ctx),
214 X509_V_FLAG_ALLOW_PROXY_CERTS);
215 return 0;
216}

218int
219brix_tls_client(brix_io *io, const char *host, int verify_peer, int verify_host,
220 const char *ca_dir, const char *client_cert, void **out_ctx,
221 brix_status *st)
222{
223 SSL_CTX *ctx;
224 SSL *ssl;
225 int fl;
226
227 *out_ctx = NULL;
228 ctx = SSL_CTX_new(TLS_client_method());
229 if (ctx == NULL) {
230 brix_tls_err(st, XRDC_ESOCK, "SSL_CTX_new");
231 return -1;
232 }
233 SSL_CTX_set_min_proto_version(ctx, TLS1_2_VERSION);
234
235 if (tls_prepare_ctx(ctx, verify_peer, ca_dir, client_cert, st) != 0) {
236 SSL_CTX_free(ctx);
237 return -1;
238 }
239
240 ssl = SSL_new(ctx);
241 if (ssl == NULL) {
242 brix_tls_err(st, XRDC_ESOCK, "SSL_new");
243 SSL_CTX_free(ctx);
244 return -1;
245 }
246 if (host != NULL && host[0] != '\0') {
247 SSL_set_tlsext_host_name(ssl, host); /* SNI */
248 if (verify_peer && verify_host) {
249 SSL_set1_host(ssl, host);
250 }
251 }
252 fl = fcntl(io->fd, F_GETFL, 0);
253 if (fl >= 0) {
254 (void) fcntl(io->fd, F_SETFL, fl | O_NONBLOCK);
255 }
256 SSL_set_fd(ssl, io->fd);
257 SSL_set_connect_state(ssl);
258
259 if (tls_do_connect(ssl, io, verify_peer, st) != 0) {
260 SSL_free(ssl);
261 SSL_CTX_free(ctx);
262 return -1;
263 }
264 io->ssl = ssl;
265 *out_ctx = ctx;
266 return 0;
267}
