6/* WHAT: Return a native user-namespace name or NULL/errno.
7 * WHY: Never reinterpret security/system attributes as user-owned metadata.
8 * HOW: Require a nonempty user. prefix and the portable total-name bound. */
9static const char *freebsd_xattr_name(const char *name)
10{
11 if (name == NULL || strncmp(name, "user.", 5) != 0 || name[5] == '\0') {
12 errno = EOPNOTSUPP; return NULL;
13 }
14 if (strlen(name) > BRIX_XATTR_NAME_MAX) { errno = ENAMETOOLONG; return NULL; }
15 return name + 5;
16}

18/* WHAT: Refuse unsupported atomic create/replace requests.
19 * WHY: extattr_set has no atomic conditional-update flag; a stat-then-set race
20 * must not be advertised as atomic. HOW: Accept zero only, reject unknown bits. */
21static int freebsd_xattr_flags(int flags)
22{
23 if (flags == 0) { return 0; }
24 errno = (flags & ~(XATTR_CREATE | XATTR_REPLACE)) ? EINVAL : ENOTSUP;
25 return -1;
26}

28/* WHAT: Read a path attribute or return required size/ERANGE.
29 * WHY: Native extattr may truncate reads; the PAL must expose undersized buffers.
30 * HOW: Query size first and verify a value read still matches its native size. */
31ssize_t brix_freebsd_getxattr(const char *path, const char *name, void *value,
32 size_t capacity, int nofollow)
33{
34 const char *native = freebsd_xattr_name(name);
35 ssize_t required, count, current;
36
37 if (native == NULL) { return -1; }
38 required = nofollow ? extattr_get_link(path, EXTATTR_NAMESPACE_USER, native, NULL, 0)
39 : extattr_get_file(path, EXTATTR_NAMESPACE_USER, native, NULL, 0);
40 if (required < 0 || capacity == 0) { return required; }
41 if (value == NULL) { errno = EINVAL; return -1; }
42 if ((size_t) required > capacity) { errno = ERANGE; return -1; }
43 count = nofollow ? extattr_get_link(path, EXTATTR_NAMESPACE_USER, native, value, capacity)
44 : extattr_get_file(path, EXTATTR_NAMESPACE_USER, native, value, capacity);
45 if (count < 0) { return -1; }
46 current = nofollow ? extattr_get_link(path, EXTATTR_NAMESPACE_USER, native, NULL, 0)
47 : extattr_get_file(path, EXTATTR_NAMESPACE_USER, native, NULL, 0);
48 if (current != count) { errno = ERANGE; return -1; }
49 return count;
50}

101/* WHAT: Convert length-prefixed native names to NUL-separated user. names.
102 * WHY: Linux listxattr uses a different wire layout; no name may overrun output.
103 * HOW: Validate every native extent, calculate full size, then copy only if fit. */
104static ssize_t freebsd_xattr_names(const unsigned char *native, size_t length,
105 char *output, size_t capacity)
106{
107 size_t cursor = 0, total = 0;
108 while (cursor < length) {
109 size_t count = native[cursor++];
110 if (count == 0 || count > length - cursor || memchr(native + cursor, 0, count)) {
111 errno = EIO; return -1;
112 }
113 if (capacity != 0 && (total > capacity || count + 6 > capacity - total)) {
114 errno = ERANGE; return -1;
115 }
116 if (capacity != 0) {
117 if (output == NULL) { errno = EINVAL; return -1; }
118 memcpy(output + total, "user.", 5);
119 memcpy(output + total + 5, native + cursor, count);
120 output[total + count + 5] = '\0';
121 }
122 total += count + 6;
123 cursor += count;
124 }
125 return (ssize_t) total;
126}

128/* WHAT: Read or size the native list using one explicit target kind.
129 * WHY: Every phase must use the same fd/link/file operation.
130 * HOW: Select fd first, otherwise honor the caller's nofollow policy. */
131static ssize_t
132freebsd_xattr_list_read(const char *path, int descriptor, int nofollow,
133 void *buffer, size_t capacity)

144/* WHAT: Read one native length-prefixed name list into bounded owned storage.
145 * WHY: Conversion requires native content even for a Linux size-only query.
146 * HOW: Query, bound to 1 MiB, allocate, fetch and return actual content length. */
147static unsigned char *freebsd_xattr_list(const char *path, int descriptor,
148 int nofollow, ssize_t *length)
149{
150 unsigned char *buffer;
151 size_t capacity;
152 ssize_t current;
153
154 *length = freebsd_xattr_list_read(path, descriptor, nofollow, NULL, 0);
155 if (*length < 0) { return NULL; }
156 if (*length > 1024 * 1024) { errno = E2BIG; return NULL; }
157 capacity = (size_t) *length;
158 buffer = malloc(capacity + 1);
159 if (buffer == NULL) { return NULL; }
160 /* A zero-capacity call is another size query, not a fetch. Return the
161 * observed empty snapshot instead of decoding a later size as data. */
162 if (capacity == 0) { return buffer; }
163 *length = freebsd_xattr_list_read(path, descriptor, nofollow, buffer, capacity);
164 if (*length < 0) { free(buffer); return NULL; }
165 current = freebsd_xattr_list_read(path, descriptor, nofollow, NULL, 0);
166 if ((size_t) *length > capacity || current != *length) {
167 free(buffer); errno = ERANGE; return NULL;
168 }
169 return buffer;
170}
