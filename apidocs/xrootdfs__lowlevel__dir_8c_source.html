17/* WHAT: Choose an advisory directory inode. WHY: Listing/stat identities should
18 * agree, without creating lookup owners. HOW: 1. Resolve dots explicitly.
19 * 2. Form a checked child key. 3. Ask the shared collision-aware registry. */
20static ino_t
21xfs_dir_ino(xfs_directory *directory, const char *name, const struct stat *attr)
22{
23 if (strcmp(name, ".") == 0) {
24 return (ino_t) directory->ino;
25 }
26 char path[XRDC_PATH_MAX];
27 if (strcmp(name, "..") == 0) {
28 if (xfs_req_path(directory->session, directory->ino, path) != 0) {
29 return (ino_t) -1;
30 }
31 char *slash = strrchr(path, '/');
32 if (slash == path) { slash[1] = '\0'; }
33 else { *slash = '\0'; }
34 xfs_alias *alias = xfs_nodes_alias(&directory->session->nodes, path);
35 return alias ? (ino_t) alias->node->ino : (ino_t) -1;
36 }
37 if (xfs_req_child(directory->session, directory->ino, name, path) != 0) {
38 return (ino_t) -1;
39 }
40 uint64_t candidate = xfs_nodes_ino_hint(&directory->session->nodes, path,
41 attr ? (uint64_t) attr->st_ino : 0);
42 return candidate ? (ino_t) candidate : (ino_t) -1;
43}

45/* WHAT: Append one encoded directory entry. WHY: FUSE owns wire encoding.
46 * HOW: 1. Compute public-API encoded size. 2. Enforce a snapshot budget.
47 * 3. Grow geometrically and emit its byte-exact continuation offset. */
48static int
49xfs_dir_fill(void *userdata, const char *name, const struct stat *attr,
50 off_t offset, enum fuse_fill_dir_flags flags)
51{
52 xfs_directory *directory = userdata;
53 (void) offset; (void) flags;
54 if (directory->error) {
55 return 1;
56 }
57 struct stat entry = {0};
58 if (attr) {
59 entry = *attr;
60 }
61 /* A collision is explicitly unknown (-1), never inode zero (an unused
62 * entry on native BSD). No advisory id authorizes a kernel lookup. */
63 entry.st_ino = xfs_dir_ino(directory, name, attr);
64 size_t bytes = fuse_add_direntry(directory->req, NULL, 0, name, &entry, 0);
65 if (bytes > XFS_DIRECTORY_LIMIT - directory->length) {
66 directory->error = -EOVERFLOW;
67 return 1;
68 }
69 size_t end = directory->length + bytes;
70 if (end > directory->capacity) {
71 size_t capacity = directory->capacity ? directory->capacity * 2 : 4096;
72 if (capacity < end) { capacity = end; }
73 char *replacement = realloc(directory->data, capacity);
74 if (!replacement) {
75 directory->error = -ENOMEM;
76 return 1;
77 }
78 directory->data = replacement;
79 directory->capacity = capacity;
80 }
81 fuse_add_direntry(directory->req, directory->data + directory->length, bytes,
82 name, &entry, (off_t) end);
83 directory->length = end;
84 return 0;
85}

96/* WHAT: Open an authorized directory snapshot. WHY: Offset pagination must be stable.
97 * HOW: 1. Resolve caller path. 2. Build bounded entries. 3. Retain open owner.
98 * 4. Reply with handle or release all resources on failure. */
99void
100xfs_ll_opendir(fuse_req_t req, fuse_ino_t ino, struct fuse_file_info *fi)
101{
102 xfs_session *session = xfs_req_lock(req);
103 char path[XRDC_PATH_MAX];
104 int rc = xfs_req_path(session, ino, path);
105 xfs_directory *directory = calloc(1, sizeof(*directory));
106 if (!directory) {
107 xfs_req_finish(req, session, -ENOMEM);
108 return;
109 }
110 directory->req = req;
111 directory->session = session;
112 directory->ino = ino;
113 if (rc == 0) {
114 rc = xfs_readdir(xfs_req_uid(req), path, directory, xfs_dir_fill, 0, NULL, 0);
115 }
116 if (rc == 0) { rc = directory->error; }
117 if (rc != 0) {
118 xfs_dir_free(directory);
119 xfs_req_finish(req, session, rc);
120 return;
121 }
122 xfs_node *node = xfs_nodes_find(&session->nodes, ino);
123 node->opens++;
124 fi->fh = (uint64_t) (uintptr_t) directory;
125 if (fuse_reply_open(req, fi) != 0) {
126 xfs_dir_free(directory);
127 xfs_nodes_close(&session->nodes, node);
128 }
129 pthread_mutex_unlock(&session->mutex);
130}

132/* WHAT: Return a requested snapshot segment. WHY: Honor kernel pagination.
133 * HOW: 1. Validate handle and cursor. 2. Clamp count to remaining bytes.
134 * 3. Reply without re-querying or acquiring lookup references. */
135void
136xfs_ll_readdir(fuse_req_t req, fuse_ino_t ino, size_t size, off_t offset,
137 struct fuse_file_info *fi)
138{
139 (void) ino;
140 xfs_directory *directory = (xfs_directory *) (uintptr_t) fi->fh;
141 if (!directory || offset < 0 || (uint64_t) offset > directory->length) {
142 fuse_reply_err(req, EINVAL);
143 return;
144 }
145 size_t remaining = directory->length - (size_t) offset;
146 size_t count = size < remaining ? size : remaining;
147 fuse_reply_buf(req, count ? directory->data + offset : NULL, count);
148}
