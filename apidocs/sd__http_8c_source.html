1/*
2 * sd_http.c — read-only-primary HTTP(S) source storage driver: the driver
3 * vtable + instance lifecycle (create/destroy). See sd_http.h.
4 *
5 * A thin driver over the injected brix_s3_transport_t (the same vtable the S3
6 * driver uses): `open`/`stat` HEAD the URL for the size, `pread` issues a byte
7 * Range GET, and a staged whole-object PUT / DELETE make it a writable cache /
8 * stage store. No kernel fd ⇒ memory-served.
9 *
10 * phase-79 file-size split — the driver is four translation units around one
11 * concept each; this file owns only the vtable wiring and create/destroy:
12 * sd_http_select.c — endpoint selection, health scoring, read failover
13 * sd_http_read.c — HEAD/GET read path + per-open credential resolution
14 * sd_http_write.c — staged whole-object PUT + DELETE write path
15 * sd_http_introspect.c — T19/T20 selection + health introspection API
16 * The seams between them are declared in sd_http_internal.h; the driver struct
17 * and sd_http_instance_is stay file-private here (the check is against this
18 * file's driver address).
19 */

28/* Read + write: an HTTP/WebDAV origin as a read source and a writable cache_store /
29 * stage_store (buffered whole-object PUT + DELETE). */
30static const brix_sd_driver_t brix_sd_http_driver = {
31 .name = "http",
32 /* phase-71: read-only primary — no .pwrite slot (writes are staged whole-object
33 * PUTs via .staged_*), so CAP_RANDOM_WRITE is NOT advertised (honest caps).
34 * phase-92: CAP_DIRS advertises WebDAV PROPFIND enumeration (.opendir);
35 * CAP_DIRS_WRITE advertises the mutable catalog — .mkdir (MKCOL), .rename
36 * (MOVE), and .unlink with is_dir (DELETE on a collection). CAP_HARD_RENAME:
37 * MOVE is atomic on the origin, so the VFS never falls back to copy+delete.
38 * CAP_SERVER_COPY: WebDAV COPY duplicates in-origin, so an intra-origin copy
39 * never round-trips the bytes through this host. CAP_XATTR/CAP_XATTR_WRITE:
40 * RFC 4918 dead properties carry the xattrs everything above this driver
41 * (WebDAV LOCK, PROPPATCH, S3 tagging/usermeta, kXR_fattr) already speaks —
42 * see sd_http_xattr.c. */
43 .caps = BRIX_SD_CAP_RANGE_READ | BRIX_SD_CAP_MEMFILE | BRIX_SD_CAP_DIRS
44 | BRIX_SD_CAP_DIRS_WRITE | BRIX_SD_CAP_HARD_RENAME
45 | BRIX_SD_CAP_SERVER_COPY | BRIX_SD_CAP_XATTR
46 | BRIX_SD_CAP_XATTR_WRITE
47 /* C6: PROBED — the cap says this driver speaks the precondition
48 * vocabulary; whether a given origin honours RFC 7232 conditional
49 * PUT is decided by the lazy 412-probe (sd_http_write.c), which
50 * refuses ENOTSUP on an origin that ignores the header. */
51 | BRIX_SD_CAP_PRECOND,
52 .cred_accept = BRIX_SD_CRED_BEARER | BRIX_SD_CRED_PROXY_PEM,
53 .open = sd_http_open,
54 .open_cred = sd_http_open_cred,

125/* sd_http_init_endpoints — fill the endpoint table from the primary host plus
126 * any extra failover origins.
127 *
128 * WHAT: Writes eps[0] from cfg->{host,port,tls,base_path}, then appends up to
129 * SD_HTTP_EP_MAX-1 valid extras, setting is->n_eps.
130 * HOW: Primary is always present (create() validated it); each extra is
131 * skipped when its host/port is missing or out of range. */
132static void
133sd_http_init_endpoints(sd_http_inst_state *is, const brix_sd_http_cfg_t *cfg)
134{
135 snprintf(is->eps[0].host, sizeof(is->eps[0].host), "%s", cfg->host);
136 is->eps[0].port = cfg->port;
137 is->eps[0].tls = cfg->tls;
138 snprintf(is->eps[0].base_path, sizeof(is->eps[0].base_path), "%s",
139 (cfg->base_path != NULL) ? cfg->base_path : "");
140 is->n_eps = 1;
141
142 if (cfg->extra == NULL || cfg->n_extra <= 0) {
143 return;
144 }
145 int i, n = cfg->n_extra;
146
147 if (n > SD_HTTP_EP_MAX - 1) {
148 n = SD_HTTP_EP_MAX - 1;
149 }
150 for (i = 0; i < n; i++) {
151 const brix_sd_http_ep_cfg_t *ec = &cfg->extra[i];
152
153 if (ec->host == NULL || ec->host[0] == '\0' || ec->port <= 0
154 || ec->port > 65535)
155 {
156 continue;
157 }
158 snprintf(is->eps[is->n_eps].host, sizeof(is->eps[0].host), "%s",
159 ec->host);
160 is->eps[is->n_eps].port = ec->port;
161 is->eps[is->n_eps].tls = ec->tls;
162 snprintf(is->eps[is->n_eps].base_path,
163 sizeof(is->eps[0].base_path), "%s",
164 (ec->base_path != NULL) ? ec->base_path : "");
165 is->n_eps++;
166 }
167}

169/* sd_http_init_tctx — resolve the transport TLS context (trust anchor).
170 *
171 * WHAT: An explicit cfg->tctx wins (a caller injecting a custom transport owns
172 * its own context). Otherwise, when an operator CA path is configured,
173 * store a persistent copy and hand it to the curl transport as tctx so
174 * origin TLS verifies against that trust anchor (phase-70 https backend
175 * leg). No CA and no tctx ⇒ NULL ⇒ the transport's system bundle. */
176static void
177sd_http_init_tctx(sd_http_inst_state *is, const brix_sd_http_cfg_t *cfg)
178{
179 if (cfg->tctx != NULL) {
180 is->tctx = cfg->tctx;
181 } else if (cfg->ca_path != NULL && cfg->ca_path[0] != '\0') {
182 snprintf(is->ca_path, sizeof(is->ca_path), "%s", cfg->ca_path);
183 is->tctx = is->ca_path;
184 } else {
185 is->tctx = NULL;
186 }
187}

189brix_sd_instance_t *
190brix_sd_http_create(const brix_sd_http_cfg_t *cfg, ngx_log_t *log)
191{
192 brix_sd_instance_t *inst;
193 sd_http_inst_state *is;
194
195 if (cfg == NULL || cfg->host == NULL || cfg->host[0] == '\0'
196 || cfg->port <= 0 || cfg->port > 65535 || cfg->transport == NULL)
197 {
198 errno = EINVAL;
199 return NULL;
200 }
201 inst = calloc(1, sizeof(*inst));
202 is = calloc(1, sizeof(*is));
203 if (inst == NULL || is == NULL) {
204 free(inst);
205 free(is);
206 errno = ENOMEM;
207 return NULL;
208 }
209 sd_http_init_endpoints(is, cfg);
210 is->transport = cfg->transport;
211 is->failover_note = cfg->failover_note;
212 is->health_note = cfg->health_note;
213 sd_http_init_tctx(is, cfg);
214 is->timeout_ms = (cfg->timeout_ms > 0) ? cfg->timeout_ms
215 : BRIX_SD_HTTP_DEFAULT_TIMEOUT_MS;
216 /* Store only whether selection/failover logging is enabled, NOT a usable log
217 * pointer: this instance is built at postconfiguration and memoized for the
218 * worker's whole life, so a captured log dangles once fork replaces the
219 * config-phase cycle log (its fd is reused — a fill-thread notice would then
220 * land on a live client socket). The fill-thread log sites resolve the CURRENT
221 * process error log via sd_http_live_log() instead. */
222 is->log = log;
223 is->cur_ep = -1;
224 is->put_checksum = cfg->put_checksum ? 1 : 0; /* #12 outbound integrity */
225 if (cfg->bearer_token != NULL && cfg->bearer_token[0] != '\0') {
226 snprintf(is->auth_hdr, sizeof(is->auth_hdr),
227 "Authorization: Bearer %s\r\n", cfg->bearer_token);
228 }
229
230 inst->driver = &brix_sd_http_driver;
231 inst->log = log;
232 inst->pool = NULL;
233 inst->state = is;
234 /* Effective caps default to the descriptor's, exactly as the pool-allocating
235 * brix_sd_instance_create does. This factory bypasses that path (it is called
236 * from tier/registry composition, which has no pool), so without this line
237 * every cap read through brix_sd_caps() came back 0 and cap-gated branches —
238 * cstore's CAP_XATTR cinfo store, the residency/space seams — silently took
239 * their fallback. */
240 inst->caps = brix_sd_http_driver.caps;
241 inst->domain = BRIX_VFS_DOMAIN_EXPORT; /* strict default; composer overrides
242 * for service storage (C9) */
243 /* Nearline is an operator DECLARATION, not a discovery: a configured Tape
244 * REST API base is the only thing that arms the cap, and the composing
245 * registry then requires a cache tier in front of this instance. See
246 * sd_http_nearline.c. */
247 if (sd_http_tape_init(is, cfg->tape_api)) {
248 inst->caps |= BRIX_SD_CAP_NEARLINE;
249 }
250 return inst;
251}
