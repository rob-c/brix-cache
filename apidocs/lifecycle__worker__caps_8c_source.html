13/* ---- Report whether a worker capability remains available ----
14 *
15 * WHAT: Return 1 when `cap` is effective or permitted, otherwise return 0.
16 * WHY: Worker identity transitions must distinguish an ordinary account from
17 * one that can still change uid or gid without duplicating PAL details.
18 * HOW: Read both v3 capability words through the PAL; all capabilities queried
19 * by this subsystem are in word zero. Fail closed to "not held" when the
20 * host has no capability API or the state cannot be read.
21 */
22int
23brix_imp_cap_held(int cap)
24{
25#if BRIX_HAS_CAPABILITIES
26 struct __user_cap_header_struct hdr;
27 struct __user_cap_data_struct data[2];
28
29 ngx_memzero(&hdr, sizeof(hdr));
30 ngx_memzero(data, sizeof(data));
31 hdr.version = _LINUX_CAPABILITY_VERSION_3;
32 hdr.pid = 0;
33 if (brix_plat_capget(&hdr, data) != 0) {
34 return 0;
35 }
36 return (data[0].effective & (1u << cap)) != 0
37 || (data[0].permitted & (1u << cap)) != 0;
38#else
39 (void) cap;
40 return 0;
41#endif
42}
