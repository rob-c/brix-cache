13/* Parse the <off len>... argument pairs of readv into `segs`, allocating a receive
14 * buffer per segment. On error frees any partial allocations, prints the diagnostic,
15 * and returns the shell exit code (>0); on success returns 0 and sets *nseg_out. */
16static int
17readv_parse_segs(int argc, char **argv, brix_readv_seg *segs, size_t *nseg_out)
18{
19 size_t nseg = 0, i;
20 int a;
21
22 for (a = 2; a + 1 < argc && nseg < XRDC_VEC_MAXSEGS; a += 2) {
23 unsigned long long off, len;
24 if (parse_u64_strict(argv[a], &off) != 0
25 || parse_u64_strict(argv[a + 1], &len) != 0) {
26 for (i = 0; i < nseg; i++) { free(segs[i].buf); }
27 fprintf(stderr, "xrdfs: readv: bad offset/length '%s %s'\n",
28 argv[a], argv[a + 1]);
29 return 50;
30 }
31 segs[nseg].offset = (int64_t) off;
32 segs[nseg].len = (size_t) len;
33 segs[nseg].got = 0;
34 segs[nseg].buf = malloc(segs[nseg].len ? segs[nseg].len : 1);
35 if (segs[nseg].buf == NULL) {
36 for (i = 0; i < nseg; i++) { free(segs[i].buf); }
37 fprintf(stderr, "xrdfs: readv: out of memory\n");
38 return 51;
39 }
40 nseg++;
41 }
42 *nseg_out = nseg;
43 return 0;
44}

47/* readv <path> <off1> <len1> [<off2> <len2> ...] — scatter-gather read (kXR_readv);
48 * the requested segments are read in one round-trip and written, concatenated, to
49 * stdout (so the bytes can be verified against the file). */
50int
51do_readv(brix_conn *c, const char *cwd, int argc, char **argv)
52{
53 brix_status st;
54 char path[XRDC_PATH_MAX];
55 brix_file f;
56 brix_readv_seg segs[XRDC_VEC_MAXSEGS];
57 size_t nseg = 0, i;
58 ssize_t got;
59 int rc = 0;
60
61 if (argc < 4 || ((argc - 2) % 2) != 0) {
62 fprintf(stderr, "usage: readv <path> <off len>...\n");
63 return 50;
64 }
65 build_path(cwd, argv[1], path, sizeof(path));
66 rc = readv_parse_segs(argc, argv, segs, &nseg);
67 if (rc != 0) { return rc; }
68 rc = 0;
69 brix_status_clear(&st);
70 if (brix_file_open_read(c, path, &f, &st) != 0) {
71 for (i = 0; i < nseg; i++) { free(segs[i].buf); }
72 return xrdfs_report_err("readv open", path, &st, 0, c);
73 }
74 got = brix_file_readv(c, &f, segs, nseg, &st);
75 if (got < 0) {
76 rc = xrdfs_report_err("readv", path, &st, 0, c);
77 } else {
78 for (i = 0; i < nseg; i++) {
79 fwrite(segs[i].buf, 1, segs[i].got, stdout); /* actual bytes read */
80 }
81 }
82 brix_file_close(c, &f, &st);
83 for (i = 0; i < nseg; i++) { free(segs[i].buf); }
84 return rc;
85}

88/* Decode one <off hexdata> writev pair into segs[nseg]. Frees the segment's own
89 * scratch on failure (the caller frees earlier segments). 0 on success, else the
90 * shell exit code (>0). */
91static int
92writev_parse_seg(char **argv, int a, brix_writev_seg *segs, size_t nseg)
93{
94 const char *hex = argv[a + 1];
95 size_t hl = strlen(hex), n = hl / 2, j;
96 uint8_t *d;
97 unsigned long long off;
98
99 if (hl == 0 || (hl % 2) != 0) {
100 fprintf(stderr, "xrdfs: writev: bad hex data\n");
101 return 50;
102 }
103 d = malloc(n);
104 if (d == NULL) {
105 fprintf(stderr, "xrdfs: writev: out of memory\n");
106 return 51;
107 }
108 for (j = 0; j < n; j++) {
109 unsigned v;
110 if (sscanf(hex + 2 * j, "%2x", &v) != 1) {
111 free(d);
112 fprintf(stderr, "xrdfs: writev: bad hex data\n");
113 return 50;
114 }
115 d[j] = (uint8_t) v;
116 }
117 if (parse_u64_strict(argv[a], &off) != 0) {
118 free(d);
119 fprintf(stderr, "xrdfs: writev: bad offset '%s'\n", argv[a]);
120 return 50;
121 }
122 segs[nseg].offset = (int64_t) off;
123 segs[nseg].len = n;
124 segs[nseg].data = d;
125 return 0;
126}

130 * segments, prints the diagnostic, and returns the shell exit code (>0); on success
