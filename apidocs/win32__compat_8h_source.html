63/* ---- Translate Winsock-specific failures ----
64 * WHAT: Return the matching errno or EINVAL for an unknown failure.
65 * WHY: Socket errors occupy a separate namespace from Win32 file errors.
66 * HOW: 1. Match the supported socket failures. 2. Fall back to EINVAL.
67 */
68static inline int brix_win32_socket_errno(DWORD error)
69{
70 switch (error) {
71 case WSAEWOULDBLOCK: return EAGAIN;
72 case WSAECONNRESET: return ECONNRESET;
73 case WSAENOTSOCK: return ENOTSOCK;
74 default: return EINVAL;
75 }
76}

78/* Convert Windows error to errno */
79static inline void brix_win32_set_errno(DWORD error)
80{
81 switch (error) {
82 case ERROR_SUCCESS:
83 errno = 0;
84 break;
85 case ERROR_FILE_NOT_FOUND:
86 case ERROR_PATH_NOT_FOUND:
87 errno = ENOENT;
88 break;
89 case ERROR_ACCESS_DENIED:
90 case ERROR_SHARING_VIOLATION:
91 errno = EACCES;
92 break;
93 case ERROR_OUTOFMEMORY:
94 errno = ENOMEM;
95 break;
96 case ERROR_FILE_EXISTS:
97 case ERROR_ALREADY_EXISTS:
98 errno = EEXIST;
99 break;
100 case ERROR_NO_MORE_FILES:
101 errno = ENOENT;
102 break;
103 case ERROR_HANDLE_EOF:
104 errno = 0; /* Not an error, just EOF */
105 break;
106 case ERROR_PIPE_BUSY:
107 errno = EAGAIN;
108 break;
109 case ERROR_IO_PENDING:
110 errno = EINPROGRESS;
111 break;
112 default:
113 errno = brix_win32_socket_errno(error);
114 break;
115 }
116}

139/* Check if path is absolute */
140static inline int brix_win32_is_absolute_path(const char *path)
141{
142 /* Check for drive letter (C:\‍) */
143 if (isalpha((unsigned char)path[0]) && path[1] == ':') {
144 return 1;
145 }
146
147 /* Check for UNC path (\\server\share) */
148 if (path[0] == '\\' && path[1] == '\\') {
149 return 1;
150 }
151
152 return 0;
153}
