1/*
2 * sd_frm_arc_seal.c — the dataset archiver's write side (2.0 F3 backup queue)
3 *
4 * WHAT: arc_migrate / arc_create_online / arc_seal — the adapter slots that
5 * turn a completion-marker commit into an archive seal — and the
6 * PENDING state between them.
7 *
8 * WHY: the OssArc backup queue (parity audit §3.5, readiness axis (e) F3).
9 * Sealing composes every member into one archive and ships it; on a
10 * real tape tier that is minutes of work. Before F3 the marker's commit
11 * ran the whole seal inline on the client's close, so the client hung
12 * for the duration and a crash mid-seal left an unsealed dataset that
13 * nothing would ever revisit. Now migrate(marker) only ANSWERS
14 * BRIX_MSS_MIGRATE_DEFERRED and the frm driver (sd_frm_staged.c) hands
15 * the seal to the durable stage engine (kind archive), which drives
16 * arc_seal() off the event loop, retries a transient failure on the
17 * brix_frm_fail_backoff sweep, replays it after a restart and
18 * dead-letters it at brix_frm_fail_retries.
19 *
20 * HOW: a dataset is PENDING from the moment its marker enters the online
21 * buffer (create_online accepted it) until the sidecar index exists
22 * (arc_sealed). Members freeze at the marker's acceptance, not at the
23 * seal: create_online refuses a member of a pending or sealed dataset
24 * EPERM and a second marker EEXIST, so nothing slips in between the
25 * marker and the archive. arc_seal(marker) composes (arc_compose,
26 * sd_frm_arc_store.c); an already-sealed dataset answers 0 (an
27 * idempotent replay), a non-marker key EINVAL, a marker whose online
28 * copy is gone ENOENT (withdrawn by the operator: nothing to seal).
29 */

115
116 if (arc_classify(c, key, &k) != ARC_KEY_MARKER) {

118 return -1;

120 if (arc_sealed(c, k.ds)) {
121 return 0; /* replayed after the seal landed */
122 }
123 if (!arc_online(c, key)) {

125 return -1;
126 }
