147static int
148unclaimed_page_io_refused(xrd_conform_execution *execution,
149 brix_conn *connection)
150{
151 static const uint8_t byte = 0x43;
152 xrd_conform_scratch_object object;
153 brix_file file;
154 brix_status status;
155 int opened;
156 int read_refused;
157 int write_refused;
158
159 if (brix_conform_scratch_path(execution->run->config.scratch,
160 "brix-conform-claim-page", object.path, sizeof(object.path)) != 0
161 || !reserve_claim(execution, 0)) return -1;
162 brix_status_clear(&status);
163 opened = brix_file_open_write(connection, object.path, 1, 0, &file,
164 &status) == 0;
165 if (!opened) return -1;
166 if (!reserve_claim(execution, 1)
167 || brix_file_write(connection, &file, 0, &byte, 1, &status) != 0) {
168 cleanup_claim_object(execution, connection, &file, &object, &status);
169 return -1;
170 }
171 read_refused = unclaimed_pgread_refused(execution, connection, &file,
172 &status);
173 write_refused = unclaimed_pgwrite_refused(execution, connection, &file,
174 &byte, &status);
175 cleanup_claim_object(execution, connection, &file, &object, &status);
176 if (read_refused < 0 || write_refused < 0) return -1;
177 return read_refused && write_refused ? 1 : 0;
178}
