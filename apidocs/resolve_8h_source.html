1/*
2 * resolve.h — the client's ONE name-resolution seam (phase-116).
3 *
4 * WHAT: brix_resolve() turns host:port into an ordered list of socket
5 * addresses; brix_resolve_ntop() / brix_resolve_port() render one back
6 * to numeric text. Every client program — libbrix, the diag tools, the
7 * standalone fault proxy and (through brixcvmfs_curl_pin.c) libcurl —
8 * resolves here and nowhere else.
9 * WHY: getaddrinfo() used to be scattered over eight files, each with its
10 * own hints, its own failure mapping and no memory of the last answer.
11 * One path means one place that honours the process family hint
12 * (netpref.c demotion), one place that tells a transient resolver
13 * failure (EAI_AGAIN) from a permanent one, one bounded answer cache
14 * (what libcurl's private cache gave the cvmfs path before it was
15 * pinned), and one symbol for tools/ci/check_dns_seam.py to pin:
16 * getaddrinfo() appears in resolve.c and nowhere else under client/.
17 * HOW: libc getaddrinfo() under a small mutex-guarded cache. The client is
18 * synchronous and has no event loop, so a blocking lookup is the right
19 * tool — phase-116 forbids the scatter, not the call. libc reads
20 * /etc/resolv.conf itself (glibc re-reads it when its mtime changes),
21 * so the client follows the same runtime file as the server. No
22 * libbrix dependency: the fault proxy compiles this file standalone.
23 */

55/* Resolve host:port into up to `max` candidates, in resolver order.
56 * family: AF_UNSPEC / AF_INET / AF_INET6 (connect paths pass
57 * brix_netpref_family()); socktype SOCK_STREAM / SOCK_DGRAM.
58 * Returns the candidate count (>= 1); 0 when the name has no address (`err`
59 * filled when non-NULL); -1 on a bad argument. Positive answers are cached
60 * for BRIX_RESOLVE_CACHE_S seconds per (host, port, family, socktype, flags);
61 * literals (BRIX_RESOLVE_NUMERIC) and failures are never cached. */

63 unsigned flags, brix_resolve_addr *out, int max,
