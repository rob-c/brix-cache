4/*
5 * netconnect.h — shared OUTBOUND connect/I/O hardening for blocking-thread paths.
6 *
7 * WHAT: two header-only helpers used by every subsystem that opens an outbound
8 * TCP connection from a worker thread (not the event loop):
9 * - brix_apply_socket_io_timeouts() — SO_RCVTIMEO + SO_SNDTIMEO on a fd.
10 * - brix_connect_fd_deadline() — a non-blocking connect(2) bounded by
11 * poll(POLLOUT) + getsockopt(SO_ERROR).
12 *
13 * WHY: native TPC (src/tpc/outbound/connect.c), the read-through cache origin fill
14 * (src/fs/cache/origin_connection.c) and OCSP fetching (src/auth/crypto/
15 * ocsp_transport.c) each grew their OWN copy of the same hardening dance, with
16 * the same Linux caveat:
17 * SO_SNDTIMEO does not reliably bound connect(2), so an unreachable/black-holed
18 * peer must be bounded with a non-blocking connect + poll() deadline or it stalls
19 * the worker thread for a full TCP retransmit window (~2 min). Factoring the
20 * shared logic here keeps ONE audited implementation across protocols and lets
21 * the raw-fd connectors (TPC, cache) drop their near-identical private copies.
22 *
23 * HOW: header-only `static ngx_inline`, compiled into each translation unit (the
24 * netopt.h precedent — no new .c, so the `config` source list is unchanged).
25 * Every setsockopt is best-effort/non-fatal; the connect helper restores the
26 * fd's original blocking mode on success and sets errno to the SO_ERROR cause on
27 * a connect-time failure. OpenSSL/BIO connectors keep their own connect path but
28 * can still share the I/O-timeout helper. Name resolution is NOT here: every
29 * connector resolves through src/net/dns (brix_dns_resolve() on the event loop,
30 * brix_dns_resolve_sync() on a worker thread — phase-116), so this header knows
31 * only sockaddrs.
32 */

58/*
59 * Connect fd to addr with an explicit poll() deadline (timeout_ms), so an
60 * unreachable peer cannot stall the calling thread for a full TCP retransmit
61 * window — SO_SNDTIMEO does not reliably bound connect(2) on Linux.
62 *
63 * Sets O_NONBLOCK, issues connect(2); on EINPROGRESS waits for POLLOUT up to
64 * timeout_ms then checks SO_ERROR. Returns 0 on success (fd restored to its
65 * original blocking mode), -1 on failure (on a connect-time failure errno is set
66 * to the SO_ERROR cause). On the error paths the fd is left non-blocking — the
67 * caller closes it on failure, so this is harmless.
68 */
69static ngx_inline int
70brix_connect_fd_deadline(int fd, const struct sockaddr *addr,
71 socklen_t addrlen, int timeout_ms)
72{
73 struct pollfd pfd;
74 int original_flags;
75 int rc;
76 int socket_error = 0;
77 socklen_t socket_error_len = sizeof(socket_error);
78
79 original_flags = fcntl(fd, F_GETFL, 0);
80 if (original_flags < 0) {
81 return -1;
82 }
83
84 if (fcntl(fd, F_SETFL, original_flags | O_NONBLOCK) != 0) {
85 return -1;
86 }
87
88 rc = connect(fd, addr, addrlen);
89 if (rc == 0) {
90 (void) fcntl(fd, F_SETFL, original_flags);
91 return 0;
92 }
93
94 if (errno != EINPROGRESS) {
95 return -1;
96 }
97
98 pfd.fd = fd;
99 pfd.events = POLLOUT;
100 pfd.revents = 0;
101
102 rc = poll(&pfd, 1, timeout_ms);
103 if (rc <= 0) {
104 return -1;
105 }
106
107 if (getsockopt(fd, SOL_SOCKET, SO_ERROR, &socket_error,
108 &socket_error_len) != 0)
109 {
110 return -1;
111 }
112
113 if (socket_error != 0) {
114 errno = socket_error;
115 return -1;
116 }
117
118 (void) fcntl(fd, F_SETFL, original_flags);
119 return 0;
120}
