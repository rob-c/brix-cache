2 * sd_block_internal.h — internals shared by the block driver's two translation
3 * units.
4 *
5 * WHAT: The per-open extent window (sd_block_obj_t), the window helper the raw
6 * byte ops and the advisory hint both use, and the declarations of the
7 * server-plane namespace slots that sd_block_ns.c defines and the vtable
8 * in sd_block.c wires up.
9 * WHY: sd_block.c carried both planes and reached the 600-line file cap. The
10 * split mirrors the POSIX driver (sd_posix.c / sd_posix_io.c /
11 * sd_posix_ns.c): the raw byte plane compiles into libxrdproto for the
12 * userland clients, the namespace plane is module-only.
13 * HOW: Include-guarded; the namespace declarations are gated on the same
14 * XRDPROTO_NO_NGX switch that gates their definitions, so the client build
15 * never sees a declaration it has no definition for.
16 */

22/* Per-open extent window (obj->state). base/len confine an opened extent to its
23 * slice of the device; a NULL obj->state is the client/unconfined path (a bare
24 * fd wrapped for raw I/O) where offsets are absolute and unclamped. */
25typedef struct {
26 off_t base; /* absolute device offset of this extent */
27 off_t len; /* extent length in bytes (device tail may be short) */
28} sd_block_obj_t;
29
30/* sd_block_read_window — translate a logical read [*off, *len) within an extent
31 * to an absolute device range. Returns 0 when the request starts at/after the
32 * extent end (the caller returns a 0-byte EOF read), else clamps *len to the
33 * extent tail and shifts *off by the extent base. os == NULL (client path) is a
34 * no-op pass-through: the offset stays absolute. */
35int sd_block_read_window(const sd_block_obj_t *os, off_t *off, size_t *len);
