1/*
2 * brixcvmfs.c — CVMFS-brix: a hardened, read-only CVMFS FUSE driver (front-end).
3 *
4 * WHAT: the driver's front-end — CLI entry + brixMount seam, `-o` option
5 * parsing, the clever-overlay/rw-overlay/fuse-launch orchestration, the
6 * opt-in phase-87 sidecar accelerators (G1 negfilter, G6 pathidx, F4
7 * prefetch, G3 dict), and the mount_run lifecycle. Owns the process-global
8 * mounted client (g_cl) and the cat_path/mono_now seam helpers.
9 * WHY: a single small binary "battle-tested against bad/evil networks": every
10 * object is content-hash verified, so any mirror/proxy failure or tampered
11 * reply is retried elsewhere and never trusted unverified.
12 * HOW: the heavy lifting lives in the Phase-38 siblings — brixcvmfs_transport.c
13 * (libcurl fetch seam), brixcvmfs_prefetch.c (F4 worker),
14 * brixcvmfs_ops.c (read-only FUSE ops), brixcvmfs_mount.c (trust-chain
15 * open/check/prewarm) — bound together through brixcvmfs_split.h. This
16 * file wires them into a mount and refuses every mutating op with -EROFS.
17 */

56/* Parsed brix-specific mount options (the rest of `-o` is forwarded to libfuse). */
57typedef struct {
58 int clever; /* overlay cache in <mnt>/.brixcache (default ON) */
59 int quota_mb; /* -o quota=<MB> (0 = from config/unlimited) */
60 int fresh; /* -o fresh: fresh connection per request */
61 int tls; /* -o tls: prefer https:// */
62 int retries; /* -o retries=<N> (-1 = from config/default) */
63 int prefetch; /* -o prefetch=<DEPTH>: subtree readahead (-1 = off) */
64 long prefetch_budget; /* -o prefetch_budget=<BYTES> (0 = unbounded) */
65 int negfilter; /* -o negfilter: in-process negative lookups (G1) */
66 int bundle; /* -o bundle: prefetch batches via .cvmfs-bundle (G2) */
67 int dict; /* -o dict: shared-dictionary transfer coding (G3) */
68 int cache_packed; /* -o cache_format=packed: log-structured cache (G4) */
69 int cache_tiering; /* -o cache_tiering: zstd cold packing + promotion (G5) */
70 int index_mmap; /* -o index=mmap: mmap'd namespace index (G6) */
71 char pin[128]; /* -o pin=<HASH>: pin the root catalog (reproducible mount) */
72 char cache_dir[512]; /* -o cache=<DIR> (implies non-clever) */
73 char writes_dir[512]; /* -o writes=<DIR> (cvmfs-rw overlay location) */
74 char fuse_extra[512]; /* passthrough -o tokens, comma-joined */
75 char *flags[16]; int nflags; /* passthrough flags (-f/-d/-h/...) */
76} brix_opts_t;

78/* Bare boolean -o tokens. 1 = consumed. */
79static int opts_o_flag(const char *t, brix_opts_t *o) {
80 if (strcmp(t, "clever") == 0) o->clever = 1;
81 else if (strcmp(t, "noclever") == 0) o->clever = 0;
82 else if (strcmp(t, "fresh") == 0) o->fresh = 1;
83 else if (strcmp(t, "tls") == 0) o->tls = 1;
84 else if (strcmp(t, "negfilter") == 0) o->negfilter = 1;
85 else if (strcmp(t, "bundle") == 0) o->bundle = 1;
86 else if (strcmp(t, "dict") == 0) o->dict = 1;
87 else if (strcmp(t, "cache_tiering") == 0) o->cache_tiering = 1;
88 else return 0;
89 return 1;
90}

92/* key=value -o tokens. 1 = consumed. */
93static int opts_o_kv(const char *t, brix_opts_t *o) {
94 if (strncmp(t, "cache_format=", 13) == 0) {
95 o->cache_packed = strcmp(t + 13, "packed") == 0;
96 if (!o->cache_packed && strcmp(t + 13, "flat") != 0)
97 fprintf(stderr, "brixcvmfs: unknown cache_format '%s' "
98 "(flat|packed) — using flat\n", t + 13);
99 }
100 else if (strncmp(t, "index=", 6) == 0) {
101 o->index_mmap = strcmp(t + 6, "mmap") == 0;
102 if (!o->index_mmap && strcmp(t + 6, "none") != 0)
103 fprintf(stderr, "brixcvmfs: unknown index '%s' "
104 "(mmap|none) — using none\n", t + 6);
105 }
106 else if (strncmp(t, "quota=", 6) == 0) o->quota_mb = atoi(t + 6);
107 else if (strncmp(t, "retries=", 8) == 0) o->retries = atoi(t + 8);
108 else if (strncmp(t, "prefetch=", 9) == 0) o->prefetch = atoi(t + 9);
109 else if (strncmp(t, "prefetch_budget=", 16) == 0) o->prefetch_budget = atol(t + 16);
110 else if (strncmp(t, "pin=", 4) == 0)
111 snprintf(o->pin, sizeof(o->pin), "%s", t + 4);
112 else if (strncmp(t, "cache=", 6) == 0)
113 snprintf(o->cache_dir, sizeof(o->cache_dir), "%s", t + 6);
114 else if (strncmp(t, "writes=", 7) == 0)
115 snprintf(o->writes_dir, sizeof(o->writes_dir), "%s", t + 7);
116 else return 0;
117 return 1;
118}

120static void opts_o_list(char *list, brix_opts_t *o) {
121 char *save = NULL;
122 for (char *t = strtok_r(list, ",", &save); t; t = strtok_r(NULL, ",", &save)) {
123 if (opts_o_flag(t, o) || opts_o_kv(t, o))
124 continue;
125 /* forward to libfuse */
126 size_t cur = strlen(o->fuse_extra);
127 snprintf(o->fuse_extra + cur, sizeof(o->fuse_extra) - cur,
128 "%s%s", cur ? "," : "", t);
129 }
130}

132static void parse_opts(int argc, char **argv, int start, brix_opts_t *o) {
133 memset(o, 0, sizeof(*o));
134 o->clever = 1; o->retries = -1; o->prefetch = -1;
135 char obuf[512];
136 for (int i = start; i < argc; i++) {
137 if (strcmp(argv[i], "-o") == 0 && i + 1 < argc) {
138 snprintf(obuf, sizeof(obuf), "%s", argv[++i]);
139 opts_o_list(obuf, o);
140 } else if (strncmp(argv[i], "-o", 2) == 0 && argv[i][2] != '\0') {
141 snprintf(obuf, sizeof(obuf), "%s", argv[i] + 2);
142 opts_o_list(obuf, o);
143 } else if (o->nflags < 16) {
144 o->flags[o->nflags++] = argv[i]; /* -f / -d / -h / ... */
145 }
146 }
147}

149/* Resolve the clever-overlay cache: unless an explicit cache (-o cache= or
150 * $BRIXCVMFS_CACHE) opts out, create <mnt>/.brixcache and open a dirfd on it
151 * BEFORE the FUSE mount hides it. Sets `*cache_override` (the explicit-cache
152 * path, else NULL) and returns the overlay dirfd, or -1 (explicit/fallback). */
153static int brixcvmfs_open_clever_cache(const char *mnt, const brix_opts_t *o,
154 const char **cache_override) {
155 *cache_override = NULL;
156 int clever = o->clever;
157 if (o->cache_dir[0]) { *cache_override = o->cache_dir; clever = 0; }
158 else if (getenv("BRIXCVMFS_CACHE")) { clever = 0; }
159 if (!clever) return -1;
160
161 mkdir(mnt, 0755); /* ensure the mountpoint exists */
162 char sub[600];
163 snprintf(sub, sizeof(sub), "%s/.brixcache", mnt);
164 if (mkdir(sub, 0755) != 0 && errno != EEXIST)
165 fprintf(stderr, "brixcvmfs: warning: cannot create overlay cache %s\n", sub);
166 int fd = open(sub, O_RDONLY | O_DIRECTORY);
167 if (fd < 0)
168 fprintf(stderr, "brixcvmfs: overlay cache unavailable, falling back\n");
169 return fd;
170}

172/* Bind the cvmfs-rw writable overlay (<mnt>/.brixwrites or -o writes=) BEFORE
173 * fuse_main hides the mountpoint — same trick as .brixcache. The rw hooks are
174 * weak so a ro-only link (test builds) still works. No-op unless brixcvmfs_rw.
175 * Returns 0 on success (or when not requested), else a process exit code. */
176static int brixcvmfs_setup_rw_overlay(const char *mnt, const brix_opts_t *o) {
177 if (!brixcvmfs_rw) return 0;
178 if (brixcvmfs_setup_rw == NULL || &brixcvmfs_rw_ops == NULL) {
179 fprintf(stderr, "brixcvmfs: rw overlay driver not linked in this build\n");
180 return 2;
181 }
182 if (brixcvmfs_setup_rw(mnt, o->writes_dir) != 0) return 1;
183 return 0;
184}

186/* Hand the mountpoint (+ passthrough flags/opts) to libfuse. Force
187 * single-threaded (-s): the client shares one SQLite catalog handle + lock-free
188 * failover state, so serialised FUSE dispatch is the correct, race-free choice
189 * (reads are cache-served, so throughput is unaffected). Returns fuse_main's rc. */
190static int brixcvmfs_run_fuse(char *arg0, const char *mnt, const brix_opts_t *o) {
191 char *fargv[24]; int fargc = 0;
192 fargv[fargc++] = arg0;
193 fargv[fargc++] = (char *) mnt;
194 fargv[fargc++] = (char *) "-s";
195 /* Flag the kernel mount read-only (matches official cvmfs2) — the pure-ro
196 * build only. The --rw overlay build is genuinely writable, so must not. */
197 if (!brixcvmfs_rw) { fargv[fargc++] = (char *) "-o"; fargv[fargc++] = (char *) "ro"; }
198 for (int i = 0; i < o->nflags && fargc < 20; i++) fargv[fargc++] = o->flags[i];
199 if (o->fuse_extra[0]) { fargv[fargc++] = (char *) "-o"; fargv[fargc++] = (char *) o->fuse_extra; }
200
201 return fuse_main(fargc, fargv,
202 brixcvmfs_rw ? &brixcvmfs_rw_ops : &brixcvmfs_ops, NULL);
203}

205/* ---- G1 negative-lookup filter (phase-87, -o negfilter / $BRIXCVMFS_NEGFILTER)
206 *
207 * Built from the client's OWN verified paths walk (client_negfilter.c) and
208 * persisted as a cache sidecar so later mounts of the same revision skip the
209 * walk. The sidecar is UNTRUSTED input: checksum-verified and root-bound on
210 * load; any defect (tamper, truncation, other revision) falls back to a fresh
211 * verified build. Every failure here is non-fatal — lookups just stay live. */
212
213#define NEGF_SIDECAR "negfilter.bxf"
214#define NEGF_SIDECAR_TMP "negfilter.bxf.tmp"
215#define NEGF_SIDECAR_MAX (256u * 1024u * 1024u) /* load sanity cap */
216
217/* Activate the filter from the sidecar in `dfd`. 0 = active. */
218static int negf_load_sidecar(cvmfs_client_t *cl, int dfd) {
219 int fd = openat(dfd, NEGF_SIDECAR, O_RDONLY);
220 if (fd < 0) return -1;
221 struct stat st;
222 if (fstat(fd, &st) != 0 || st.st_size <= 0
223 || (unsigned long long) st.st_size > NEGF_SIDECAR_MAX) {
224 close(fd);
225 return -1;
226 }
227 size_t len = (size_t) st.st_size;
228 unsigned char *img = malloc(len);
229 if (img == NULL) { close(fd); return -1; }
230 for (size_t off = 0; off < len; ) {
231 ssize_t got = read(fd, img + off, len - off);
232 if (got <= 0) { free(img); close(fd); return -1; }
233 off += (size_t) got;
234 }
235 close(fd);
236
237 cvmfs_xorf_t f;
238 cvmfs_hash_t root;
239 int rc = cvmfs_xorf_deserialize(&f, &root, img, len); /* fail-closed */
240 free(img);
241 if (rc != 0) return -1;
242 if (cvmfs_client_negfilter_adopt(cl, &f, &root) != 0) { /* other revision */
243 cvmfs_xorf_reset(&f);
244 return -1;
245 }
246 return 0;
247}

249/* Persist the active filter atomically (tmp + rename) into `dfd`. */
250static void negf_store_sidecar(cvmfs_client_t *cl, int dfd) {
251 cvmfs_hash_t root;
252 const cvmfs_xorf_t *f = cvmfs_client_negfilter(cl, &root);
253 if (f == NULL) return;
254 size_t need = cvmfs_xorf_size(f), n = 0;
255 unsigned char *img = malloc(need);
256 if (img == NULL || cvmfs_xorf_serialize(f, &root, img, need, &n) != 0) {
257 free(img);
258 return;
259 }
260 int fd = openat(dfd, NEGF_SIDECAR_TMP, O_WRONLY | O_CREAT | O_TRUNC, 0644);
261 int ok = fd >= 0;
262 for (size_t off = 0; ok && off < n; ) {
263 ssize_t w = write(fd, img + off, n - off);
264 if (w <= 0) ok = 0;
265 else off += (size_t) w;
266 }
267 if (fd >= 0) close(fd);
268 free(img);
269 if (ok && renameat(dfd, NEGF_SIDECAR_TMP, dfd, NEGF_SIDECAR) == 0) return;
270 (void) unlinkat(dfd, NEGF_SIDECAR_TMP, 0);
271}

273/* Resolve the cache-dir fd a sidecar setup reads from: the caller's cache_dirfd
274 * passes through (*owned = 0), else the repo's cache dir is opened (*owned = 1,
275 * the caller closes). Returns -1 if neither is available. */
276static int brixcvmfs_sidecar_dirfd(const char *repo, const char *cache_override,
277 int cache_dirfd, int *owned) {
278 char cdir[512] = "";
279 if (cache_dirfd >= 0) {
280 *owned = 0;
281 return cache_dirfd;
282 }
283 brixcvmfs_prepare_cache_dir(repo, cache_override, -1, cdir, sizeof(cdir));
284 *owned = 1;
285 return open(cdir, O_RDONLY | O_DIRECTORY);
286}

288/* Sidecar-or-build bring-up. Runs synchronously before fuse_main (the mount
289 * serves FUSE single-threaded, so no locking), once per served revision. */
290static void brixcvmfs_negfilter_setup(cvmfs_client_t *cl, const char *repo,
291 const char *cache_override, int cache_dirfd) {
292 int owned = 0;
293 int dfd = brixcvmfs_sidecar_dirfd(repo, cache_override, cache_dirfd, &owned);
294 if (dfd >= 0 && negf_load_sidecar(cl, dfd) == 0) {
295 fprintf(stderr, "brixcvmfs: negfilter loaded from sidecar\n");
296 } else if (cvmfs_client_negfilter_build(cl, mono_now()) == 0) {
297 if (dfd >= 0) negf_store_sidecar(cl, dfd);
298 fprintf(stderr, "brixcvmfs: negfilter built from verified walk\n");
299 } else {
300 fprintf(stderr, "brixcvmfs: negfilter unavailable — lookups stay live\n");
301 }
302 if (owned && dfd >= 0) close(dfd);
303}

305/* ---- G6 mmap path index (phase-87, -o index=mmap / $BRIXCVMFS_INDEX=mmap)
306 *
307 * Built from the client's OWN verified paths walk (client_pathidx.c) and
308 * persisted as a cache sidecar the client mmaps back — later mounts of the
309 * same revision skip both the walk and every catalog open. The sidecar is
310 * UNTRUSTED input: geometry-validated + root-bound on load (any other
311 * revision, ABI or truncation defect falls back to a fresh verified build),
312 * and a tampered ENTRY is caught at first read by the CAS verify-fetch, which
313 * drops the index. Every failure here is non-fatal — the catalogs stay live. */
314
315#define PIDX_SIDECAR "pathidx.bxi"
316
317static void brixcvmfs_pathidx_setup(cvmfs_client_t *cl, const char *repo,
318 const char *cache_override, int cache_dirfd) {
319 int owned = 0;
320 int dfd = brixcvmfs_sidecar_dirfd(repo, cache_override, cache_dirfd, &owned);
321 if (dfd >= 0 && cvmfs_client_pathidx_load(cl, dfd, PIDX_SIDECAR) == 0)
322 fprintf(stderr, "brixcvmfs: pathidx loaded from sidecar\n");
323 else if (dfd >= 0
324 && cvmfs_client_pathidx_build(cl, dfd, PIDX_SIDECAR, mono_now()) == 0)
325 fprintf(stderr, "brixcvmfs: pathidx built from verified walk\n");
326 else
327 fprintf(stderr, "brixcvmfs: pathidx unavailable — catalog lookups stay live\n");
328 if (owned && dfd >= 0) close(dfd);
329}

331/* F4 predictive prefetch: -o prefetch=<depth> or $BRIXCVMFS_PREFETCH. */
332static void brixcvmfs_arm_prefetch(cvmfs_client_t *cl, const char *repo,
333 const brix_opts_t *o, const char *cache_override,
334 int cache_dirfd, long quota) {
335 const char *env_pf = getenv("BRIXCVMFS_PREFETCH");
336 int pf_depth = o->prefetch >= 0 ? o->prefetch : (env_pf ? atoi(env_pf) : -1);
337 if (pf_depth < 0) return;
338
339 const char *env_pb = getenv("BRIXCVMFS_PREFETCH_BUDGET");
340 long pf_budget = o->prefetch_budget > 0 ? o->prefetch_budget
341 : (env_pb ? atol(env_pb) : 0);
342 char pf_cache[512] = "";
343 brixcvmfs_prepare_cache_dir(repo, cache_override, cache_dirfd,
344 pf_cache, sizeof(pf_cache));
345 /* G2 batch mode: -o bundle or $BRIXCVMFS_BUNDLE=1 (needs prefetch on;
346 * set before pf_start so the worker thread never sees it change). */
347 const char *env_bd = getenv("BRIXCVMFS_BUNDLE");
348 int pf_bundle = (o->bundle || (env_bd != NULL && atoi(env_bd) != 0)) ? 1 : 0;
349 pf_start(pf_depth, pf_budget, cl->catalog_tmp, pf_cache, cache_dirfd, quota,
350 pf_bundle);
351}

353/* Arm the opt-in phase-87 lookup accelerators + dict coding. */
354static void brixcvmfs_arm_sidecars(cvmfs_client_t *cl, const char *repo,
355 const brix_opts_t *o, const char *cache_override,
356 int cache_dirfd) {
357 /* G1 negative-lookup filter: -o negfilter or $BRIXCVMFS_NEGFILTER=1. */
358 const char *env_nf = getenv("BRIXCVMFS_NEGFILTER");
359 if (o->negfilter || (env_nf != NULL && atoi(env_nf) != 0))
360 brixcvmfs_negfilter_setup(cl, repo, cache_override, cache_dirfd);
361
362 /* G6 mmap path index: -o index=mmap or $BRIXCVMFS_INDEX=mmap. */
363 const char *env_ix = getenv("BRIXCVMFS_INDEX");
364 if (o->index_mmap || (env_ix != NULL && strcmp(env_ix, "mmap") == 0))
365 brixcvmfs_pathidx_setup(cl, repo, cache_override, cache_dirfd);
366
367 /* G3 shared-dictionary coding: -o dict or $BRIXCVMFS_DICT=1 (armed here;
368 * the dict itself is pulled on the first CAS data GET). */
369 const char *env_dc = getenv("BRIXCVMFS_DICT");
370 if (o->dict || (env_dc != NULL && atoi(env_dc) != 0))
371 brixcvmfs_dict_arm();
372}

374/* Full mount bring-up: clever-cache dirfd, rw overlay, verify-mount the repo,
375 * run FUSE, then tear everything down. `cache_dirfd` ownership is local to this
376 * function. Returns the process exit code. */
377static int brixcvmfs_mount_run(char *arg0, const char *repo, const char *mnt,
378 const brix_opts_t *o) {
379 g_tcfg.fresh_connect = o->fresh;
380 g_tcfg.prefer_tls = o->tls;
381 long quota = o->quota_mb > 0 ? (long) o->quota_mb * 1024L * 1024L : 0;
382
383 const char *cache_override = NULL;
384 int cache_dirfd = brixcvmfs_open_clever_cache(mnt, o, &cache_override);
385
386 int rw_rc = brixcvmfs_setup_rw_overlay(mnt, o);
387 if (rw_rc != 0) { if (cache_dirfd >= 0) close(cache_dirfd); return rw_rc; }
388
389 cvmfs_client_t *cl = brixcvmfs_open(repo, cache_override, cache_dirfd, quota, o->retries,
390 o->pin, o->cache_packed, o->cache_tiering);
391 if (cl == NULL) { if (cache_dirfd >= 0) close(cache_dirfd); return 1; }
392 g_cl = cl;
393
394 brixcvmfs_arm_prefetch(cl, repo, o, cache_override, cache_dirfd, quota);
395 brixcvmfs_arm_sidecars(cl, repo, o, cache_override, cache_dirfd);
396
397 int rc = brixcvmfs_run_fuse(arg0, mnt, o);
398
399 if (brixcvmfs_rw && brixcvmfs_teardown_rw != NULL) brixcvmfs_teardown_rw();
400 cvmfs_client_umount(cl);
401 transport_cleanup();
402 brixcvmfs_dict_free();
403 curl_global_cleanup();
404 if (cache_dirfd >= 0) close(cache_dirfd);
405 free(cl);
406 return rc;
407}

409/* brixcvmfs entry — reused by the brixMount umbrella (SP-G).
410 * brixcvmfs <repo> <mountpoint> [fuse-opts] — mount
411 * brixcvmfs --check <repo> — verify + summarise, no mount */
412int brixcvmfs_main(int argc, char **argv) {
413 if (argc >= 3 && strcmp(argv[1], "--check") == 0)
414 return brixcvmfs_check(argv[2]);
415 if (argc >= 3 && strcmp(argv[1], "--prewarm") == 0)
416 return brixcvmfs_prewarm(argv[2]);
417 if (argc >= 2 && strcmp(argv[1], "repo") == 0) {
418 if (brixcvmfs_repo_main == NULL) {
419 fprintf(stderr, "brixcvmfs: repo driver not linked in this build\n");
420 return 2;
421 }
422 return brixcvmfs_repo_main(argc - 1, argv + 1);
423 }
424 if (argc >= 2 && strcmp(argv[1], "ingest") == 0) {
425 if (brixcvmfs_ingest_main == NULL) {
426 fprintf(stderr, "brixcvmfs: ingest driver not linked in this build\n");
427 return 2;
428 }
429 return brixcvmfs_ingest_main(argc - 1, argv + 1);
430 }
431
432 if (argc < 3) {
433 fprintf(stderr,
434 "usage: brixcvmfs <repo.fqrn> <mountpoint> [fuse-opts]\n"
435 " brixcvmfs --check <repo.fqrn>\n"
436 " brixcvmfs --prewarm <repo.fqrn>\n"
437 " brixcvmfs repo mkfs|info|resign ...\n"
438 " brixcvmfs ingest dir|image|prune ...\n");
439 return 2;
440 }
441
442 brix_opts_t o;
443 parse_opts(argc, argv, 3, &o);
444 return brixcvmfs_mount_run(argv[0], argv[1], argv[2], &o);
445}
