1/*
2 * root_prepare.c — shared export-root validation and canonicalization.
3 *
4 * All three protocol surfaces (native root://, WebDAV, S3) need the same
5 * startup check on their configured export root: verify it exists, is a
6 * readable directory with the right permissions, and resolve it through
7 * realpath(3) to eliminate symlinks from the confinement boundary.
8 *
9 * This helper centralises that logic so the behaviour is identical across
10 * all surfaces. It must be called during merge_loc_conf / postconfiguration
11 * so that nginx -t catches misconfigured roots before traffic is accepted.
12 */
13

25char *
26brix_prepare_export_root(ngx_conf_t *cf,
27 const ngx_str_t *root, const brix_export_root_opts_t *opts,
28 char *root_canon)
29{
30 char root_buf[PATH_MAX];
31 int access_mode;
32 ngx_str_t root_str;
33
34 /* Empty root — either silently skip or hard fail depending on opts. */
35 if (root == NULL || root->len == 0) {
36 if (opts->required) {
37 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
38 "%s is required but not set",
39 opts->directive_name);
40 return NGX_CONF_ERROR;
41 }
42 return NGX_CONF_OK;
43 }
44
45 /* Guard against paths that would overflow the canonical buffer. */
46 if (root->len >= opts->canon_size || root->len >= sizeof(root_buf)) {
47 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
48 "%s path is too long", opts->directive_name);
49 return NGX_CONF_ERROR;
50 }
51
52 /* Build a NUL-terminated copy for stat/access/realpath. */
53 ngx_memcpy(root_buf, root->data, root->len);
54 root_buf[root->len] = '\0';
55
56 /* Validate existence, kind (directory), and access permissions. */
57 root_str.data = (u_char *) root_buf;
58 root_str.len = root->len;
59 access_mode = opts->allow_write ? (R_OK | W_OK | X_OK) : (R_OK | X_OK);
60
61 if (brix_validate_path(cf, opts->directive_name, &root_str,
62 BRIX_PATH_DIRECTORY, access_mode) != NGX_OK)
63 {
64 return NGX_CONF_ERROR;
65 }
66
67 /* Resolve symlinks and normalise the path for the confinement boundary. */
68 if (realpath(root_buf, root_canon) == NULL) {
69 ngx_conf_log_error(NGX_LOG_EMERG, cf, errno,
70 "%s: cannot resolve canonical path for \"%s\"",
71 opts->directive_name, root_buf);
72 return NGX_CONF_ERROR;
73 }
74
75 return NGX_CONF_OK;
76}

78char *
79brix_prepare_cache_root(ngx_conf_t *cf, ngx_http_brix_shared_conf_t *common)
80{
81 static const char prefix[] = "posix:";
82 brix_export_root_opts_t cache_opts;
83 u_char *store;
84 size_t root_len;
85
86 if (common->cache_root.len == 0) {
87 return NGX_CONF_OK;
88 }
89 if (common->cache_store.len > 0) {
90 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
91 "brix_cache_root is shorthand for a POSIX brix_cache_store; "
92 "configure only one of them");
93 return NGX_CONF_ERROR;
94 }
95 cache_opts.directive_name = "brix_cache_root";
96 cache_opts.allow_write = 1;
97 cache_opts.required = 0;
98 cache_opts.canon_size = sizeof(common->cache_root_canon);
99 if (brix_prepare_export_root(cf, &common->cache_root, &cache_opts,
100 common->cache_root_canon) != NGX_CONF_OK)
101 {
102 return NGX_CONF_ERROR;
103 }
104
105 /* HARD config guard: the read-through cache root must live OUTSIDE the
106 * export, or cache sidecars would be exposed in the client namespace. */
107 if (brix_assert_dir_outside_export(cf, "brix_cache_root",
108 common->root_canon, common->cache_root_canon) != NGX_OK)
109 {
110 return NGX_CONF_ERROR;
111 }
112
113 /* W9: cache_root is syntax sugar, not a second runtime cache engine.
114 * Canonicalize and confinement-check the directory under its public name,
115 * then lower it to the composable tier grammar before registration. Clear
116 * both legacy fields so VFS opens cannot consult the old by-root cache and
117 * the sd_cache decorator for the same request. */
118 root_len = ngx_strlen(common->cache_root_canon);
119 store = ngx_pnalloc(cf->pool, sizeof(prefix) - 1 + root_len + 1);
120 if (store == NULL) {
121 return NGX_CONF_ERROR;
122 }
123 ngx_memcpy(store, prefix, sizeof(prefix) - 1);
124 ngx_memcpy(store + sizeof(prefix) - 1, common->cache_root_canon,
125 root_len + 1);
126 common->cache_store.data = store;
127 common->cache_store.len = sizeof(prefix) - 1 + root_len;
128 common->cache_root.data = NULL;
129 common->cache_root.len = 0;
130 common->cache_root_canon[0] = '\0';
131 return NGX_CONF_OK;
132}

134/* Validate an explicit brix_vfs_spill_path and canonicalise it into
135 * spill_canon (size PATH_MAX). Split from brix_prepare_spill_scratch to keep
136 * its decision count in budget. */
137static char *
138brix_prepare_spill_path(ngx_conf_t *cf, ngx_http_brix_shared_conf_t *common,
139 char *spill_canon)
140{
141 brix_export_root_opts_t opts;
142
143 /* realpath() would silently resolve a relative path against the
144 * master's cwd — refuse it up front so the boundary is explicit. */
145 if (common->vfs_spill_path.data[0] != '/') {
146 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
147 "brix_vfs_spill_path must be absolute (got \"%V\")",
148 &common->vfs_spill_path);
149 return NGX_CONF_ERROR;
150 }
151 opts.directive_name = "brix_vfs_spill_path";
152 opts.allow_write = 1;
153 opts.required = 0;
154 opts.canon_size = PATH_MAX;
155 if (brix_prepare_export_root(cf, &common->vfs_spill_path, &opts,
156 spill_canon) != NGX_CONF_OK)
157 {
158 return NGX_CONF_ERROR;
159 }
160 /* HARD config guard: scratch inside the export would make service
161 * storage reachable as export storage (and expose spill temps in the
162 * client namespace). */
163 if (brix_assert_dir_outside_export(cf, "brix_vfs_spill_path",
164 common->root_canon, spill_canon) != NGX_OK)
165 {
166 return NGX_CONF_ERROR;
167 }
168 return NGX_CONF_OK;
169}

171char *
172brix_prepare_spill_scratch(ngx_conf_t *cf, ngx_http_brix_shared_conf_t *common,
173 const char *stage_dir_canon)
174{
175 char spill_canon[PATH_MAX];
176 const char *scratch = NULL;
177 size_t max;
178
179 max = (common->vfs_spill_max == (size_t) NGX_CONF_UNSET_SIZE)
180 ? 0 : common->vfs_spill_max;
181 if (max != 0 && max < 1024 * 1024) {
182 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
183 "brix_vfs_spill_max must be 0 or at least 1m (got %uz)", max);
184 return NGX_CONF_ERROR;
185 }
186
187 if (common->vfs_spill_path.len > 0) {
188 if (brix_prepare_spill_path(cf, common, spill_canon) != NGX_CONF_OK) {
189 return NGX_CONF_ERROR;
190 }
191 scratch = spill_canon;
192 } else if (stage_dir_canon != NULL && stage_dir_canon[0] != '\0') {
193 /* Default per the C1 contract: the export's staged temp directory —
194 * already canonical and already guarded outside the export by its own
195 * brix_stage_dir preparation. */
196 scratch = stage_dir_canon;
197 }
198
199 /* Phase-107 C3 rides the same per-export preparation: register a merged
200 * `brix_durable_publish off` so the publish barrier is skipped for this
201 * export. Registered only when OFF — an absent registry entry is durable
202 * (fails safe), and the table stays free of entries for every
203 * default-configured export. */
204 if (common->durable_publish == 0 && common->root_canon[0] != '\0') {
205 brix_vfs_backend_set_durable(common->root_canon, 0);
206 }
207
208 /* Phase-107 C7 rides the same preparation: register a merged
209 * `brix_lock_enforcement advisory|off` so the VFS lock gate relaxes for
210 * this export. Registered only when non-strict — an absent registry entry
211 * is STRICT (fails toward enforcement), and the table stays free of
212 * entries for every default-configured export. */
213 if (common->lock_enforcement != 0 && common->root_canon[0] != '\0') {
214 brix_vfs_backend_set_lock_enforcement(common->root_canon,
215 common->lock_enforcement);
216 }
217
218 if (scratch == NULL || common->root_canon[0] == '\0') {
219 /* No scratch (reordered uploads refuse ENOSPC) or no export anchored
220 * here to key the registry entry on — directives were still validated. */
221 return NGX_CONF_OK;
222 }
223 brix_vfs_backend_set_spill(common->root_canon, scratch, (off_t) max);
224 /* Owned-temp reclaim: a crashed worker's spill is recognised by its
225 * .xrd-tmp.<pid>. name when the reaper walks this registered root. */
226 brix_tmp_reap_register(scratch);
227 return NGX_CONF_OK;
228}
