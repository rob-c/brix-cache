7/*
8 * cert_response.c — GSI round 1: reply to kXGC_certreq with kXGS_cert.
9 *
10 * Generates an ephemeral ffdhe2048 DH key (from the per-worker keypool, so
11 * keygen never runs on the event thread), advertises the keyable cipher/md
12 * lists, optionally signs the DH public (signed-DH) and the client rtag with the
13 * server RSA key, and emits the kXGS_cert bucket list (kXRS_puk/cipher,
14 * cipher_alg, md_alg, x509, main). Each function is documented at its
15 * definition below; the wire layout is described inline at the assembly sites.
16 */

18/*
19 * gsi_certreq_u32 — best-effort read of a 4-byte integer bucket from the
20 * certreq payload. Encoders differ on byte order (our client emits big-endian;
21 * stock XrdSut writes the host-order int), so accept whichever interpretation
22 * lands in the caller's plausible [lo, hi] range, preferring big-endian; 0 when
23 * the bucket is absent, short, or neither reading is plausible. Used for the
24 * kXRS_version (XrdSecgsi version, 5-digit range) and kXRS_clnt_opts (kOpts*
25 * flag word, small bitmask) buckets.
26 */
27static uint32_t
28gsi_certreq_u32(const u_char *payload, size_t plen, uint32_t bucket,
29 uint32_t lo, uint32_t hi)
30{
31 const u_char *vb = NULL;
32 size_t vlen = 0;
33 uint32_t raw, be;
34
35 if (gsi_find_bucket(payload, plen, bucket, &vb, &vlen) != 0
36 || vlen < 4)
37 {
38 return 0;
39 }
40 ngx_memcpy(&raw, vb, 4); /* host-order as written by stock XrdSut */
41 be = ntohl(raw); /* big-endian as written by our client */
42 if (be >= lo && be <= hi) {
43 return be;
44 }
45 if (raw >= lo && raw <= hi) {
46 return raw;
47 }
48 return 0;
49}

51/*
52 * gsi_use_signed_dh — resolve the signed-DH decision for this handshake from
53 * the operator policy (conf->gsi_signed_dh) and the client's advertised
54 * version. REQUIRE always signs; AUTO signs only modern (>=10400) clients;
55 * OFF never signs. An unknown client version (0) is treated as legacy.
56 */
57static int
58gsi_use_signed_dh(ngx_uint_t policy, uint32_t client_version)
59{
60 if (policy == BRIX_GSI_SDH_REQUIRE) {
61 return 1;
62 }
63 if (policy == BRIX_GSI_SDH_AUTO) {
64 return client_version >= BRIX_GSI_VERS_DHSIGNED;
65 }
66 return 0;
67}

69/*
70 * Phase 52 (WS-A): build the advertised kXRS_cipher_alg list into out[] (NUL-
71 * terminated) from the configured preference (or the built-in default, aes-128-
72 * cbc first), keeping ONLY ciphers this build can actually key — so a client
73 * never selects one we cannot decrypt (e.g. bf-cbc without the OpenSSL legacy
74 * provider). Falls back to "aes-128-cbc" if nothing usable remains.
75 */
76static void
77gsi_build_cipher_alg_list(const ngx_stream_brix_srv_conf_t *conf,
78 char *out, size_t outsz)
79{
80 const char *src = (conf->gsi_ciphers.len > 0)
81 ? (const char *) conf->gsi_ciphers.data
82 : brix_gsi_cipher_default_list();
83 const char *p = src;
84 size_t out_n = 0;
85
86 out[0] = '\0';
87 while (*p && out_n + 1 < outsz) {
88 const char *start = p;
89 char name[24];
90 size_t n;
91 brix_gsi_cipher_t tmp;
92
93 while (*p && *p != ':') { p++; }
94 n = (size_t) (p - start);
95 if (n > 0 && n < sizeof(name)) {
96 ngx_memcpy(name, start, n);
97 name[n] = '\0';
98 if (brix_gsi_cipher_lookup(name, &tmp)
99 && out_n + n + 1 < outsz) {
100 if (out_n > 0) { out[out_n++] = ':'; }
101 ngx_memcpy(out + out_n, name, n);
102 out_n += n;
103 out[out_n] = '\0';
104 }
105 }
106 if (*p == ':') { p++; }
107 }
108 if (out_n == 0) {
109 ngx_cpystrn((u_char *) out, (u_char *) "aes-128-cbc", outsz);
110 }
111}

113/*
114 * gsi_cert_chunks_t — the computed byte chunks the kXGS_cert framer emits, one
115 * field per top-level bucket payload. Populated by the build/sign helpers and
116 * consumed by gsi_frame_cert_response() so the wire assembly takes one struct
117 * instead of a long parameter list; every field is a plain pointer+length view

136/*
137 * gsi_record_handshake_opts — WHAT: resolve and record the two per-handshake
138 * policy decisions on the connection (signed-DH variant, client delegation
139 * opts) from the certreq payload + operator config. WHY: round 2
140 * (parse_x509.c) must agree on the signed-DH padding/IV, and a later delegation
141 * round must pick the flow the client advertised, so both are latched onto
142 * ctx->gsi here. HOW: apply the operator policy against the client's advertised
143 * version; a REQUIRE policy with no server key degrades to unsigned rather than
144 * failing the auth. Returns the resolved signed-DH flag; logs the delegation
145 * opts at INFO (byte-identical to the pre-split line).
146 */
147static int
148gsi_record_handshake_opts(brix_ctx_t *ctx, ngx_connection_t *c,
149 const ngx_stream_brix_srv_conf_t *conf)
150{
151 int signed_dh = gsi_use_signed_dh(conf->gsi_signed_dh,
152 gsi_certreq_u32(ctx->recv.payload,
153 ctx->recv.cur_dlen,
154 (uint32_t) kXRS_version,
155 10000, 99999));
156
157 if (signed_dh && conf->gsi_key == NULL) {
158 signed_dh = 0;
159 }
160 ctx->gsi.signed_dh = signed_dh;
161
162 /* §F6: capture the client's advertised delegation mode (kXRS_clnt_opts) so a
163 * later delegation round picks the flow the client actually supports —
164 * kOptsFwdPxy (forward) vs kOptsDlgPxy / kOptsSigReq (sign-request). */
165 ctx->gsi.clnt_opts = gsi_certreq_u32(ctx->recv.payload, ctx->recv.cur_dlen,
166 (uint32_t) kXRS_clnt_opts, 1, 4095);
167 ngx_log_error(NGX_LOG_INFO, c->log, 0,
168 "brix: GSI client delegation opts=0x%02xd (fwd=%d sign=%d dlg=%d)",
169 (unsigned) ctx->gsi.clnt_opts,
170 (ctx->gsi.clnt_opts & 0x2) ? 1 : 0,
171 (ctx->gsi.clnt_opts & 0x4) ? 1 : 0,
172 (ctx->gsi.clnt_opts & 0x1) ? 1 : 0);
173
174 return signed_dh;
175}

177/*
178 * gsi_build_dh_public — WHAT: take/generate an ephemeral ffdhe2048 DH key and
179 * build the kXRS_puk blob ("<PEM params>---BPUB---<pub hex>---EPUB--") into a
180 * c->pool buffer. WHY: this is the DH-public body the kXGS_cert advertises; the
181 * key is latched on ctx->gsi.dh_key for round-2 shared-secret derivation. HOW:
182 * pop a pre-generated key from the per-worker pool (keygen never runs on the
183 * event thread under a certreq burst) or inline-keygen on an empty pool; own the
184 * key on this connection. On any failure the DH key is freed and NGX_ERROR is
185 * returned. Returns NGX_OK with *puk_out / *puk_len_out set on success.
186 */
187static ngx_int_t
188gsi_build_dh_public(brix_ctx_t *ctx, ngx_connection_t *c,
189 const ngx_stream_brix_srv_conf_t *conf,
190 u_char **puk_out, size_t *puk_len_out)
191{
192 EVP_PKEY *dhkey = NULL;
193 BIGNUM *pub_bn = NULL;
194 char *pub_hex = NULL;
195 BIO *bio;
196 BUF_MEM *bptr;
197 char puk_buf[4096];
198 int puk_written;
199 u_char *puk_blob;
200 size_t puk_len;
201
202 if (!brix_gsi_keypool_pop(conf->common.thread_pool, c->log, &dhkey)) {
203 dhkey = brix_gsi_dh_keygen();
204 if (dhkey == NULL) {
205 return NGX_ERROR;
206 }
207 }
208
209 if (!EVP_PKEY_get_bn_param(dhkey, "pub", &pub_bn)) {
210 brix_evp_pkey_free(dhkey);
211 return NGX_ERROR;
212 }
213 pub_hex = BN_bn2hex(pub_bn);
214 BN_free(pub_bn);
215 if (pub_hex == NULL) {
216 brix_evp_pkey_free(dhkey);
217 return NGX_ERROR;
218 }
219
220 bio = BIO_new(BIO_s_mem());
221 if (bio == NULL) {
222 OPENSSL_free(pub_hex);
223 brix_evp_pkey_free(dhkey);
224 return NGX_ERROR;
225 }
226 PEM_write_bio_Parameters(bio, dhkey);
227 ctx->gsi.dh_key = dhkey; /* ownership transfers to the connection */
228 BIO_get_mem_ptr(bio, &bptr);
229
230 puk_written = snprintf(puk_buf, sizeof(puk_buf),
231 "%.*s---BPUB---%s---EPUB--",
232 (int) bptr->length, bptr->data, pub_hex);
233 BIO_free(bio);
234 OPENSSL_free(pub_hex);
235
236 if (puk_written <= 0 || (size_t) puk_written >= sizeof(puk_buf)) {
237 return NGX_ERROR;
238 }
239 puk_len = (size_t) puk_written;
240
241 BRIX_PALLOC_OR_RETURN(puk_blob, c->pool, puk_len, NGX_ERROR);
242 ngx_memcpy(puk_blob, puk_buf, puk_len);
243
244 *puk_out = puk_blob;
245 *puk_len_out = puk_len;
246 return NGX_OK;
247}

249/*
250 * gsi_select_dh_public_bucket — WHAT: choose the DH-public bucket payload+type
251 * for the kXGS_cert. WHY: unsigned (default) advertises the bare Public() blob
252 * as kXRS_puk; signed-DH (>=10400) advertises the SAME blob RSA-signed with the
253 * server key (XrdCryptosslRSA::EncryptPrivate) as kXRS_cipher, which the client
254 * recovers via DecryptPublic against the kXRS_x509 cert we also send —
255 * authenticating the DH parameters in transit. HOW: on the signed path,
256 * RSA-encrypt-private into a c->pool buffer sized for the signature; on failure
257 * WARN and return NGX_ERROR. On entry *ch already carries the unsigned default
258 * (pub_data/pub_len = the bare puk blob); the signed path overwrites it and sets
259 * pub_type to kXRS_cipher, the unsigned path only stamps kXRS_puk. Returns
260 * NGX_OK on success.
261 */
262static ngx_int_t
263gsi_select_dh_public_bucket(ngx_connection_t *c,
264 const ngx_stream_brix_srv_conf_t *conf, int signed_dh,
265 gsi_cert_chunks_t *ch)
266{
267 if (signed_dh) {
268 size_t cap = ch->pub_len
269 + 2 * (size_t) EVP_PKEY_size(conf->gsi_key) + 64;
270 u_char *sig = ngx_palloc(c->pool, cap);
271 size_t pub_len = sig ? brix_gsi_rsa_encrypt_private(conf->gsi_key,
272 ch->pub_data, ch->pub_len, sig, cap) : 0;
273
274 if (pub_len == 0) {
275 ngx_log_error(NGX_LOG_WARN, c->log, 0,
276 "brix: GSI signed-DH: failed to sign DH public");
277 return NGX_ERROR;
278 }
279 ch->pub_data = sig;
280 ch->pub_len = pub_len;
281 ch->pub_type = (uint32_t) kXRS_cipher;
282 return NGX_OK;
283 }
284
285 ch->pub_type = (uint32_t) kXRS_puk;
286 return NGX_OK;
287}

289/*
290 * gsi_sign_client_rtag — WHAT: RSA-PKCS1 sign the client's random tag (kXRS_rtag
291 * inside the certreq's kXRS_main container) with the server key. WHY: the signed
292 * rtag proves possession of the server private key to the client and rides in the
293 * response's nested kXRS_main container. HOW: locate the client rtag; if present
294 * and the whole sign sequence succeeds, emit the signature into a c->pool buffer;
295 * any step failing leaves *rtag_out NULL / *rtag_len_out 0 (unsigned response,
296 * not an error — matches the pre-split fall-through). No return code: the
297 * out-params carry the result.
298 */
299static void
300gsi_sign_client_rtag(brix_ctx_t *ctx, ngx_connection_t *c,
301 const ngx_stream_brix_srv_conf_t *conf,
302 u_char **rtag_out, size_t *rtag_len_out)
303{
304 const u_char *main_data = NULL;
305 size_t main_dlen = 0;
306 const u_char *clnt_rtag = NULL;
307 size_t clnt_rtlen = 0;
308
309 *rtag_out = NULL;
310 *rtag_len_out = 0;
311
312 if (gsi_find_bucket(ctx->recv.payload, ctx->recv.cur_dlen,
313 (uint32_t) kXRS_main, &main_data, &main_dlen) == 0) {
314 gsi_find_bucket(main_data, main_dlen,
315 (uint32_t) kXRS_rtag, &clnt_rtag, &clnt_rtlen);
316 }
317
318 if (clnt_rtag && clnt_rtlen > 0) {
319 EVP_PKEY_CTX *sctx = EVP_PKEY_CTX_new(conf->gsi_key, NULL);
320
321 if (sctx) {
322 size_t slen = (size_t) EVP_PKEY_size(conf->gsi_key);
323 u_char *signed_rtag = ngx_palloc(c->pool, slen);
324
325 if (signed_rtag
326 && EVP_PKEY_sign_init(sctx) > 0
327 && EVP_PKEY_CTX_set_rsa_padding(sctx, RSA_PKCS1_PADDING) > 0
328 && EVP_PKEY_sign(sctx, signed_rtag, &slen,
329 clnt_rtag, clnt_rtlen) > 0)
330 {
331 *rtag_out = signed_rtag;
332 *rtag_len_out = slen;
333 }
334 brix_evp_pkey_ctx_free(sctx);
335 }
336 }
337}

339/*
340 * gsi_build_main_container — WHAT: build the nested kXRS_main container the
341 * kXGS_cert carries as a single top-level kXRS_main bucket. WHY: it is NOT a
342 * flat field — it re-encapsulates its own protocol header plus an inner bucket
343 * list. Inner wire layout (network byte order throughout):
344 * "gsi\0" protocol name (4 bytes, NUL-padded)
345 * kXGS_cert opcode (uint32, 4 bytes)
346 * [ kXRS_signed_rtag bucket ] optional: type(4) + len(4) + signature
347 * kXRS_none terminator (uint32, 4 bytes)
348 * HOW: reserve the fixed 12 bytes (name + opcode + terminator) plus, when the
349 * rtag was signed, the 8-byte bucket header and signature; then walk mp
350 * byte-by-byte to emit the layout. Returns NGX_OK with *buf_out / *len_out set.
351 */
352static ngx_int_t
353gsi_build_main_container(ngx_connection_t *c,
354 const u_char *signed_rtag, size_t signed_rtag_len,
355 u_char **buf_out, size_t *len_out)
356{
357 size_t main_len = 4 + 4 + 4;
358 u_char *main_buf;
359 u_char *mp;
360
361 if (signed_rtag_len > 0) {
362 main_len += 4 + 4 + signed_rtag_len;
363 }
364
365 BRIX_PALLOC_OR_RETURN(main_buf, c->pool, main_len, NGX_ERROR);
366
367 mp = main_buf;
368 mp[0] = 'g';
369 mp[1] = 's';
370 mp[2] = 'i';
371 mp[3] = '\0';
372 mp += 4;
373 *(uint32_t *) mp = htonl(kXGS_cert);
374 mp += 4;
375 if (signed_rtag_len > 0) {
376 *(uint32_t *) mp = htonl(kXRS_signed_rtag);
377 mp += 4;
378 *(uint32_t *) mp = htonl((uint32_t) signed_rtag_len);
379 mp += 4;
380 ngx_memcpy(mp, signed_rtag, signed_rtag_len);
381 mp += signed_rtag_len;
382 }
383 *(uint32_t *) mp = htonl(kXRS_none);
384
385 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
386 "brix: kXGS_cert signed rtag=%uz bytes main_len=%uz",
387 signed_rtag_len, main_len);
388
389 *buf_out = main_buf;
390 *len_out = main_len;
391 return NGX_OK;
392}

394/*
395 * gsi_frame_cert_response — WHAT: assemble and queue the kXGS_cert response frame
396 * from the computed chunks. WHY: this is the single wire-emission site — the
397 * top-level bucket list (kXRS_puk/cipher, kXRS_cipher_alg, kXRS_md_alg, kXRS_x509,
398 * kXRS_main, kXRS_none) wrapped in the kXR_authmore server header. HOW: sum the
399 * body length, allocate the c->pool frame, write the response header, then walk p
400 * byte-by-byte emitting each bucket (type, len, payload) in network byte order;
401 * byte layout is frozen and identical to the pre-split assembly. Returns the
402 * brix_queue_response() result (NGX_OK on queued).
403 */
404/* Store a big-endian uint32 at a possibly-unaligned wire position. The bucket
405 * payloads (pub/calg/malg/cert) are variable length, so `p` is not generally
406 * 4-aligned when the next type/len word is written — a plain `*(uint32_t*)p =`
407 * is a misaligned store (UB; UBSan-flagged; faults on strict-alignment arches).
408 * memcpy is the portable, alignment-safe equivalent and compiles to the same
409 * store on x86. */
410static inline void
411gsi_wire_put_u32(u_char *p, uint32_t v)
412{
413 uint32_t be = htonl(v);
414 ngx_memcpy(p, &be, sizeof(be));
415}
416
417static ngx_int_t
418gsi_frame_cert_response(brix_ctx_t *ctx, ngx_connection_t *c,
419 const gsi_cert_chunks_t *ch)
420{
421 size_t body_len = 4 + 4
422 + 4 + 4 + ch->pub_len
423 + 4 + 4 + ch->calg_len
424 + 4 + 4 + ch->malg_len
425 + 4 + 4 + ch->cert_len
426 + 4 + 4 + ch->main_len
427 + 4;
428 size_t total = XRD_RESPONSE_HDR_LEN + body_len;
429 u_char *buf;
430 u_char *p;
431
432 BRIX_PALLOC_OR_RETURN(buf, c->pool, total, NGX_ERROR);
433
434 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_authmore,
435 (uint32_t) body_len, (ServerResponseHdr *) buf);
436
437 p = buf + XRD_RESPONSE_HDR_LEN;
438
439 p[0] = 'g';
440 p[1] = 's';
441 p[2] = 'i';
442 p[3] = '\0';
443 p += 4;
444 gsi_wire_put_u32(p, kXGS_cert);
445 p += 4;
446
447 gsi_wire_put_u32(p, ch->pub_type);
448 p += 4;
449 gsi_wire_put_u32(p, (uint32_t) ch->pub_len);
450 p += 4;
451 ngx_memcpy(p, ch->pub_data, ch->pub_len);
452 p += ch->pub_len;
453
454 gsi_wire_put_u32(p, kXRS_cipher_alg);
455 p += 4;
456 gsi_wire_put_u32(p, (uint32_t) ch->calg_len);
457 p += 4;
458 ngx_memcpy(p, ch->cipher_alg, ch->calg_len);
459 p += ch->calg_len;
460
461 gsi_wire_put_u32(p, kXRS_md_alg);
462 p += 4;
463 gsi_wire_put_u32(p, (uint32_t) ch->malg_len);
464 p += 4;
465 ngx_memcpy(p, ch->md_alg, ch->malg_len);
466 p += ch->malg_len;
467
468 gsi_wire_put_u32(p, kXRS_x509);
469 p += 4;
470 gsi_wire_put_u32(p, (uint32_t) ch->cert_len);
471 p += 4;
472 ngx_memcpy(p, ch->cert_pem, ch->cert_len);
473 p += ch->cert_len;
474
475 gsi_wire_put_u32(p, kXRS_main);
476 p += 4;
477 gsi_wire_put_u32(p, (uint32_t) ch->main_len);
478 p += 4;
479 ngx_memcpy(p, ch->main_buf, ch->main_len);
480 p += ch->main_len;
481
482 gsi_wire_put_u32(p, kXRS_none);
483
484 ngx_log_debug3(NGX_LOG_DEBUG_STREAM, c->log, 0,
485 "brix: kXGS_cert sent cert_len=%uz puk_len=%uz main_len=%uz",
486 ch->cert_len, ch->puk_len, ch->main_len);
487
488 return brix_queue_response(ctx, c, buf, total);
489}

491/*
492 * Respond to kXGC_certreq with kXGS_cert. Orchestrates the four
493 * equal-behavior phases: latch handshake opts, build the DH-public body, sign
494 * the client rtag into the nested kXRS_main container, and frame+queue the
495 * response. Wire bytes are frozen — see the per-helper doc blocks.
496 */
497
498ngx_int_t
499brix_gsi_send_cert(brix_ctx_t *ctx, ngx_connection_t *c)
500{
501 ngx_stream_brix_srv_conf_t *conf;
502 gsi_cert_chunks_t ch = {0};
503 char cipher_alg[96];
504 u_char *puk_blob = NULL;
505 size_t puk_len = 0;
506 u_char *signed_rtag = NULL;
507 size_t signed_rtag_len = 0;
508 int signed_dh;
509
510 conf = ngx_stream_get_module_srv_conf(ctx->session,
511 ngx_stream_brix_module);
512
513 if (conf->gsi_cert_pem == NULL || conf->gsi_cert_pem_len == 0) {
514 return NGX_ERROR;
515 }
516 ch.cert_pem = conf->gsi_cert_pem;
517 ch.cert_len = conf->gsi_cert_pem_len;
518
519 /* Advertise the kXRS_cipher_alg list (config preference or default, keyable
520 * ciphers only). See gsi_build_cipher_alg_list(). */
521 gsi_build_cipher_alg_list(conf, cipher_alg, sizeof(cipher_alg));
522 ch.cipher_alg = cipher_alg;
523 ch.calg_len = ngx_strlen(cipher_alg);
524
525 ch.md_alg = "sha256:sha1";
526 ch.malg_len = strlen(ch.md_alg);
527
528 signed_dh = gsi_record_handshake_opts(ctx, c, conf);
529
530 if (gsi_build_dh_public(ctx, c, conf, &puk_blob, &puk_len) != NGX_OK) {
531 return NGX_ERROR;
532 }
533 ch.puk_len = puk_len;
534 ch.pub_data = puk_blob; /* unsigned default; signed path overwrites */
535 ch.pub_len = puk_len;
536
537 if (gsi_select_dh_public_bucket(c, conf, signed_dh, &ch) != NGX_OK) {
538 return NGX_ERROR;
539 }
540
541 gsi_sign_client_rtag(ctx, c, conf, &signed_rtag, &signed_rtag_len);
542
543 if (gsi_build_main_container(c, signed_rtag, signed_rtag_len,
544 (u_char **) &ch.main_buf, &ch.main_len)
545 != NGX_OK)
546 {
547 return NGX_ERROR;
548 }
549
550 return gsi_frame_cert_response(ctx, c, &ch);
551}
