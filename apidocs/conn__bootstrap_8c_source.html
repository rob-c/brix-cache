22static void
23fill_username(char out[9])
24{
25 /* The login username is the OS identity advertised for monitoring/default
26 * mapping (real authz comes from the chosen sec protocol). Prefer $LOGNAME/
27 * $USER — set in every interactive/login shell — so the common path skips the
28 * getpwuid() NSS lookup entirely (it lazy-loads libnss_* on first call, a
29 * measurable per-process cost the async manager would otherwise pay once per
30 * parallel stream). Fall back to getpwuid() when the environment is unset. */
31 const char *name = getenv("LOGNAME");
32 struct passwd *pw;
33 size_t n;
34
35 if (name == NULL || name[0] == '\0') { name = getenv("USER"); }
36 if (name == NULL || name[0] == '\0') {
37 pw = getpwuid(geteuid());
38 name = (pw != NULL && pw->pw_name != NULL) ? pw->pw_name : "nobody";
39 }
40 n = strlen(name);
41
42 if (n > 8) { n = 8; } /* the wire field is 8 bytes; truncate */
43 memcpy(out, name, n);
44 out[n] = '\0'; /* NUL-terminated for xrd_pack_login_request */
45}

47/* Read one response frame raw (header + body), bypassing streamid checks; used
48 * for the handshake exchange where the first reply carries streamid {0,0}. */
49static int
50recv_raw(brix_conn *c, uint16_t *sid, uint16_t *status,
51 uint8_t *body, uint32_t bodycap, uint32_t *blen, brix_status *st)
52{
53 uint8_t hdr[XRD_RESPONSE_HDR_LEN];
54 uint32_t dlen;
55
56 if (brix_read_full(&c->io, hdr, sizeof(hdr), st) != 0) {
57 return -1;
58 }
59 xrd_resp_hdr_unpack(hdr, sid, status, &dlen); /* unaligned-safe */
60
61 if (dlen > bodycap) {
62 brix_status_set(st, XRDC_EPROTO, 0,
63 "handshake body too large (%u > %u)", dlen, bodycap);
64 return -1;
65 }
66 if (dlen > 0 && brix_read_full(&c->io, body, dlen, st) != 0) {
67 return -1;
68 }
69 *blen = dlen;
70 if (c->diag.wire_trace) { /* §15: trace handshake/protocol replies too */
71 brix_trace_frame(c, '<', *sid, *status, 0, dlen, body, dlen);
72 }
73 return 0;
74}

76int
77brix_conn_handshake(brix_conn *c, uint16_t proto_sid, int want_tls,
78 brix_status *st)
79{
80 uint8_t seg[XRD_HANDSHAKE_LEN + XRD_REQUEST_HDR_LEN];
81 ClientInitHandShake hs;
82 ClientProtocolRequest pr;
83 uint16_t sid, status;
84 uint8_t body[256];
85 uint32_t blen;
86 int saw_proto = 0;
87 int rounds;
88
89 xrd_pack_handshake(&hs);
90
91 /* The client owns its protocol streamid; ask for the security-requirements
92 * trailer (to learn the signing level) and advertise TLS capability,
93 * requiring TLS for roots:// / --tls. */
94 {
95 const uint8_t sid[2] = { (uint8_t) (proto_sid >> 8),
96 (uint8_t) (proto_sid & 0xff) };
97 uint8_t flags = (uint8_t) (kXR_secreqs | kXR_ableTLS |
98 (want_tls ? kXR_wantTLS : 0));
99 xrd_pack_protocol_request(&pr, sid, flags);
100 }
101
102 memcpy(seg, &hs, XRD_HANDSHAKE_LEN);
103 memcpy(seg + XRD_HANDSHAKE_LEN, &pr, XRD_REQUEST_HDR_LEN);
104
105 if (brix_write_full(&c->io, seg, sizeof(seg), st) != 0) {
106 return -1;
107 }
108 if (c->diag.wire_trace) { /* §15: the 20B init has no streamid/requestid */
109 fprintf(stderr, "> handshake-init (20B) + kXR_protocol sid=%u\n", proto_sid);
110 }
111
112 /* Expect up to two frames: a handshake reply (streamid {0,0}) and the
113 * protocol reply (streamid == proto_sid). Some servers may send only the
114 * protocol reply; key on the streamid rather than assume an ordering. */
115 for (rounds = 0; rounds < 2 && !saw_proto; rounds++) {
116 if (recv_raw(c, &sid, &status, body, sizeof(body), &blen, st) != 0) {
117 return -1;
118 }
119 if (status != kXR_ok) {
120 /* The server completed the framing and EXPLICITLY rejected the
121 * handshake (e.g. kXR_error because we asked for TLS on a non-TLS
122 * port). That is a permanent decision: classify it by the server's
123 * own status code (not the retryable XRDC_EPROTO framing-desync
124 * code) so the resilient loop fails fast instead of re-handshaking
125 * the same rejection until its stall window expires. */
126 brix_status_set(st, (int) status, 0,
127 "handshake: server status %u", status);
128 return -1;
129 }
130 if (sid == proto_sid) {
131 if (blen < sizeof(ServerProtocolBody)) {
132 brix_status_set(st, XRDC_EPROTO, 0,
133 "protocol reply too short (%u bytes)", blen);
134 return -1;
135 }
136 /* ServerProtocolBody = pval[4] flags[4]; capabilities in flags. */
137 c->server_flags = xrd_get_u32_be(body + 4); /* unaligned-safe */
138 /* Optional signing trailer (present because we set kXR_secreqs):
139 * ServerResponseReqs_Protocol immediately after the 8-byte body —
140 * theTag 'S', rsvd, secver, secopt, seclvl, secvsz (then secvsz*2
141 * secvec bytes we do not consume; BriX and default stock servers
142 * send secvsz=0 and express the level in seclvl). Defensive:
143 * default 0 (no signing). */
144 c->sec_level = 0;
145 c->sec_odata = 0;
146 if (blen >= 14 && body[8] == 'S') {
147 c->sec_odata = (body[11] & kXR_secOData) ? 1 : 0;
148 c->sec_level = body[12];
149 }
150 saw_proto = 1;
151 }
152 /* else: handshake reply (streamid {0,0}); keep reading for the protocol. */
153 }
154
155 if (!saw_proto) {
156 brix_status_set(st, XRDC_EPROTO, 0, "no protocol reply from server");
157 return -1;
158 }
159 return 0;
160}

162int
163brix_conn_login(brix_conn *c, const brix_opts *o, brix_status *st)
164{
165 ClientLoginRequest req;
166 uint16_t sid, status;
167 uint8_t *body = NULL;
168 uint32_t blen = 0;
169
170 /* streamid {0,0}: brix_send stamps the real streamid (and dlen) after
171 * packing. The username is the OS identity; advertise async-response
172 * capability. dlen=0 → anonymous (no credential/CGI payload). */
173 {
174 static const uint8_t sid0[2] = { 0, 0 };
175 char uname[9];
176 fill_username(uname);
177 xrd_pack_login_request(&req, sid0, (int32_t) getpid(), uname,
178 (uint8_t) (kXR_ver005 | kXR_asyncap));
179 }
180
181 {
182 brix_resp_out out = { &status, &body, &blen };
183 if (brix_send(c, &req, NULL, &sid, st) != 0) {
184 return -1;
185 }
186 if (brix_recv(c, sid, &out, st) != 0) {
187 return -1;
188 }
189 }
190 if (status != kXR_ok) {
191 brix_status_set(st, XRDC_EPROTO, 0, "login: server status %u", status);
192 free(body);
193 return -1;
194 }
195 if (blen < BRIX_SESSION_ID_LEN) {
196 brix_status_set(st, XRDC_EPROTO, 0,
197 "login reply too short (%u bytes)", blen);
198 free(body);
199 return -1;
200 }
201 memcpy(c->sessid, body, BRIX_SESSION_ID_LEN);
202
203 /* Anything past the 16-byte sessid is a "&P=<proto>,..." security list:
204 * the server demands authentication. Hand off to the auth driver. */
205 if (blen > BRIX_SESSION_ID_LEN) {
206 char sec[256];
207 uint32_t n = blen - BRIX_SESSION_ID_LEN;
208 if (n >= sizeof(sec)) { n = sizeof(sec) - 1; }
209 memcpy(sec, body + BRIX_SESSION_ID_LEN, n);
210 sec[n] = '\0';
211 free(body);
212 snprintf(c->sec_list, sizeof(c->sec_list), "%s", sec); /* §15 explain */
213 return brix_authenticate(c, sec, o, st);
214 }
215
216 free(body);
217 return 0;
218}
