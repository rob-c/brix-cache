1/* sd_frm_adapter.c — MSS adapter dialect selection for the sd_frm backend.
2 *
3 * WHAT: Maps an operator `adapter` string (exec / hpss / cta / lib / stub) onto
4 * a concrete MSS transport and binds it into the sd_frm state: exec-family
5 * dialects resolve a stage command, lib-family dialects resolve a shared
6 * object to dlopen, and the stub is the always-available fallback.
7 *
8 * WHY: Split out of sd_frm.c, which crossed the 600-line cap
9 * (coding-standards §1). Adapter selection is construction-time policy;
10 * the parent TU keeps the runtime driver vtable.
11 *
12 * HOW: Family predicates first (pure string classification), then one
13 * frm_select_* per family returning 0 = bound, 1 = not mine, -1 = hard
14 * failure. brix_sd_frm_create() in sd_frm.c drives them in order. */

32/* ---- exec-family MSS dialects (exec / hpss / cta) ----
33 *
34 * WHAT: Is `adapter` one of the exec-family dialects? All three drive a real MSS
35 * through an operator stage command (the classic FRM model); "hpss" and "cta" are
36 * named HSM dialects over the same transport, differing only in which stage
37 * command they resolve. A NULL adapter is not exec-family.
38 */
39static int
40frm_adapter_is_exec_family(const char *adapter)
41{
42 return adapter != NULL
43 && (ngx_strcmp(adapter, "exec") == 0
44 || ngx_strcmp(adapter, "hpss") == 0
45 || ngx_strcmp(adapter, "cta") == 0);
46}

48/* Resolve the stage command for an exec-family `adapter`.
49 *
50 * WHY: A node can front an HPSS silo (an `hsi`/`pftp` stager) and a CTA silo (an
51 * `eos`/`cta-admin` stager) at once, so each dialect gets its own env override;
52 * all fall back to the generic $BRIX_FRM_STAGECMD. Returns the command, or NULL
53 * when neither the dialect override nor the generic var is set/non-empty. */
54static const char *
55frm_exec_stagecmd(const char *adapter)
56{
57 const char *cmd = NULL;
58
59 if (ngx_strcmp(adapter, "hpss") == 0) {
60 cmd = getenv("BRIX_FRM_HPSS_STAGECMD");
61 } else if (ngx_strcmp(adapter, "cta") == 0) {
62 cmd = getenv("BRIX_FRM_CTA_STAGECMD");
63 }
64 if (cmd == NULL || cmd[0] == '\0') {
65 cmd = getenv("BRIX_FRM_STAGECMD");
66 }
67 return (cmd != NULL && cmd[0] != '\0') ? cmd : NULL;
68}

108/* ---- Select a library-native MSS adapter (lib / libhpss / libcta) ----
109 *
110 * WHAT: If `adapter` names a library-native dialect AND its .so path is
111 * resolvable AND the library loads with the required ABI, points st->mss/mss_ctx
112 * at the dlopen'd adapter. ALWAYS returns 0 (the caller may proceed): a
113 * non-lib-family adapter, an unset path, an absent library, or a missing symbol
114 * all leave st->mss NULL so the exec/stub fallback runs. The vendor library is a
115 * runtime plug-in; its absence is a graceful degrade, never a hard failure.
116 *
117 * WHY: isolates the library-native decision so brix_sd_frm_create stays a flat
118 * orchestration below the complexity cap, and mirrors frm_select_exec_adapter.
119 */
120int
121frm_select_lib_adapter(sd_frm_state *st, const char *adapter,
122 const char *location, ngx_log_t *log)
123{
124 const char *path;
125
126 if (!frm_adapter_is_lib_family(adapter)) {
127 return 0;
128 }
129 path = frm_lib_path(adapter);
130 if (path == NULL) {
131 ngx_log_error(NGX_LOG_WARN, log, 0,
132 "xrootd frm: the \"%s\" MSS adapter needs an HSM library path "
133 "($BRIX_FRM_LIB, or the per-dialect override); "
134 "falling back to the built-in stub", adapter);
135 return 0;
136 }
137 st->mss_ctx = brix_mss_lib_create(location, path, log);
138 if (st->mss_ctx == NULL) {
139 /* brix_mss_lib_create already WARN-logged the dlopen/symbol failure. */
140 return 0; /* graceful fallback to exec/stub */
141 }
142 st->mss = &brix_mss_lib_adapter;
143 ngx_log_error(NGX_LOG_NOTICE, log, 0,
144 "xrootd frm: \"%s\" library-native MSS adapter (lib=%s, online buffer=%s)",
145 adapter, path, location);
146 return 0;
147}

149/* ---- Select an exec-family MSS adapter (exec / hpss / cta) when requested ----
150 *
151 * WHAT: If `adapter` names an exec-family HSM dialect AND its stage command is
152 * resolvable (`frm_exec_stagecmd`), builds the exec MSS context and points
153 * `st->mss`/`st->mss_ctx` at it. Returns 0 whenever the caller may proceed
154 * (adapter not exec-family, or no stage command - in both cases `st->mss` is left
155 * NULL so the stub fallback runs); returns -1 with errno set to ENOMEM only when
156 * the exec context allocation itself fails.
157 *
158 * WHY: Isolates the exec-adapter decision (the classic FRM model over an external
159 * stage command) so brix_sd_frm_create stays a flat orchestration below the
160 * complexity cap. HPSS/CTA are real named dialects here, not stub fallthroughs.
161 *
162 * HOW:
163 * 1. If `adapter` is not exec-family, return 0 (fall through to the stub).
164 * 2. Resolve the stage command; if none, WARN and return 0 (stub fallback).
165 * 3. Create the exec MSS context; on failure set errno=ENOMEM and return -1.
166 * 4. On success, publish the exec adapter, NOTICE-log the dialect, return 0.
167 */
168int
169frm_select_exec_adapter(sd_frm_state *st, const char *adapter,
170 const char *location, ngx_log_t *log)
171{
172 const char *cmd;
173
174 if (!frm_adapter_is_exec_family(adapter)) {
175 return 0;
176 }
177 cmd = frm_exec_stagecmd(adapter);
178 if (cmd == NULL) {
179 ngx_log_error(NGX_LOG_WARN, log, 0,
180 "xrootd frm: the \"%s\" MSS adapter needs a stage command "
181 "($BRIX_FRM_STAGECMD, or the per-dialect override); "
182 "falling back to the built-in stub", adapter);
183 return 0;
184 }
185 st->mss_ctx = brix_mss_exec_create(location, cmd, log);
186 if (st->mss_ctx == NULL) {
187 errno = ENOMEM;
188 return -1;
189 }
190 st->mss = &brix_mss_exec_adapter;
191 ngx_log_error(NGX_LOG_NOTICE, log, 0,
192 "xrootd frm: \"%s\" MSS adapter (stagecmd=%s, online buffer=%s)",
193 adapter, cmd, location);
194 return 0;
195}

197/* ---- Select the built-in local-dir stub MSS adapter (default / fallback) ----
198 *
199 * WHAT: Builds the stub MSS context that simulates tape with local directories and
200 * points `st->mss`/`st->mss_ctx` at it. Returns 0 on success, or -1 with errno set
201 * to ENOMEM when the stub context allocation fails. An `adapter` name that is
202 * neither empty, "stub", nor "exec" is WARN-logged as not-yet-implemented before
203 * the stub is used, matching the original in-place behaviour.
204 *
205 * WHY: The stub is the default and the fallback for every adapter that is not a
206 * working "exec"; factoring it out keeps the create orchestrator flat and under
207 * the complexity cap while preserving the exact warning and errno semantics.
208 *
209 * HOW:
210 * 1. If `adapter` is a non-empty name that is neither "stub" nor an
211 * exec-family dialect (exec/hpss/cta), WARN that it is unrecognized.
212 * 2. Create the stub MSS context; on failure set errno=ENOMEM and return -1.
213 * 3. Publish the stub adapter and return 0.
214 */
215int
216frm_select_stub_adapter(sd_frm_state *st, const char *adapter,
217 const char *location, ngx_log_t *log)
218{
219 if (adapter != NULL && adapter[0] != '\0'
220 && ngx_strcmp(adapter, "stub") != 0
221 && !frm_adapter_is_exec_family(adapter)
222 && !frm_adapter_is_lib_family(adapter))
223 {
224 ngx_log_error(NGX_LOG_WARN, log, 0,
225 "xrootd frm: MSS adapter \"%s\" is not recognized (known: stub, "
226 "exec, hpss, cta, lib, libhpss, libcta); using the built-in stub",
227 adapter);
228 }
229 st->mss_ctx = brix_mss_stub_create(location, log);
230 if (st->mss_ctx == NULL) {
231 errno = ENOMEM;
232 return -1;
233 }
234 st->mss = &brix_mss_stub_adapter;
235 return 0;
236}
