1/* sd_frm_adapter.c — MSS adapter dialect selection for the sd_frm backend.
2 *
3 * WHAT: Maps an operator `adapter` string (exec / hpss / cta / lib / stub) onto
4 * a concrete MSS transport and binds it into the sd_frm state: exec-family
5 * dialects resolve a stage command, lib-family dialects resolve a shared
6 * object to dlopen, and the stub is the always-available fallback.
7 *
8 * WHY: Split out of sd_frm.c, which crossed the 600-line cap
9 * (coding-standards §1). Adapter selection is construction-time policy;
10 * the parent TU keeps the runtime driver vtable.
11 *
12 * HOW: Family predicates first (pure string classification), then one
13 * frm_select_* per family returning 0 = bound, 1 = not mine, -1 = hard
14 * failure. brix_sd_frm_create() in sd_frm.c drives them in order. */

32/* ---- exec-family MSS dialects (exec / hpss / cta) ----
33 *
34 * WHAT: Is `adapter` one of the exec-family dialects? All three drive a real MSS
35 * through an operator stage command (the classic FRM model); "hpss" and "cta" are
36 * named HSM dialects over the same transport, differing only in which stage
37 * command they resolve. A NULL adapter is not exec-family.
38 */
39static int
40frm_adapter_is_exec_family(const char *adapter)
41{
42 return adapter != NULL
43 && (ngx_strcmp(adapter, "exec") == 0
44 || ngx_strcmp(adapter, "hpss") == 0
45 || ngx_strcmp(adapter, "cta") == 0);
46}

48/* ---- master-environ snapshot for the dialect contract vars (W6) ----
49 *
50 * WHAT: frm_env() is getenv() that REMEMBERS: every successful read of one of
51 * the six dialect vars refreshes its slot, and a later miss answers from the
52 * slot. Non-contract names fall through to plain getenv semantics.
53 *
54 * WHY: The stage command/HSM library contract is "read from the MASTER
55 * environment at config time" — and the master's parse-time chain build
56 * honours it. But the registry memo is CYCLE-keyed (vfs_backend_registry.c:
57 * a chain built under an earlier cycle holds that cycle's dead log pointer),
58 * so every worker rebuilds its chain once — AFTER ngx_worker_process_init has
59 * replaced environ with the `env`-directive allowlist. Without this snapshot
60 * that rebuild silently degrades an exec/lib dialect to the stub (caught by
61 * test_frm_scratch when the cycle keying landed). The slots are warmed in
62 * the master during config parse and reach workers by fork inheritance; an
63 * operator `env` directive still works and simply keeps getenv succeeding.
64 * A reload re-parses in the master (whose environ never changes), re-warming
65 * the slots before the new workers fork. */
66typedef struct {
67 const char *name; /* contract var (static literal) */
68 char value[PATH_MAX]; /* last non-empty value seen */
69 unsigned set; /* value[] is meaningful */
70} frm_env_slot_t;
71
72static frm_env_slot_t frm_env_snapshot[] = {
73 { "BRIX_FRM_STAGECMD", "", 0 },
74 { "BRIX_FRM_HPSS_STAGECMD", "", 0 },
75 { "BRIX_FRM_CTA_STAGECMD", "", 0 },
76 { "BRIX_FRM_LIB", "", 0 },
77 { "BRIX_FRM_HPSS_LIB", "", 0 },
78 { "BRIX_FRM_CTA_LIB", "", 0 },
79};

81static const char *
82frm_env(const char *name)
83{
84 const char *v = getenv(name);
85 ngx_uint_t i;
86
87 for (i = 0;
88 i < sizeof(frm_env_snapshot) / sizeof(frm_env_snapshot[0]);
89 i++)
90 {
91 frm_env_slot_t *s = &frm_env_snapshot[i];
92
93 if (ngx_strcmp(s->name, name) != 0) {
94 continue;
95 }
96 if (v != NULL && v[0] != '\0'
97 && ngx_strlen(v) < sizeof(s->value))
98 {
99 ngx_cpystrn((u_char *) s->value, (u_char *) v, sizeof(s->value));
100 s->set = 1;
101 }
102 return s->set ? s->value : NULL;
103 }
104 return (v != NULL && v[0] != '\0') ? v : NULL;
105}

107/* Resolve a per-dialect env override with a generic fallback.
108 *
109 * WHY: a node can front an HPSS silo and a CTA silo at once, so each named
110 * dialect gets its own env var; an unmatched adapter, or an unset/empty
111 * override, falls back to the generic var. Returns the value, or NULL when
112 * nothing is set/non-empty. */
113static const char *
114frm_dialect_env(const char *adapter, const char *hpss_name,
115 const char *hpss_var, const char *cta_name, const char *cta_var,
116 const char *generic_var)
117{
118 const char *v = NULL;
119
120 if (ngx_strcmp(adapter, hpss_name) == 0) {
121 v = frm_env(hpss_var);
122 } else if (ngx_strcmp(adapter, cta_name) == 0) {
123 v = frm_env(cta_var);
124 }
125 if (v == NULL) {
126 v = frm_env(generic_var);
127 }
128 return v;
129}

169/* ---- Select a library-native MSS adapter (lib / libhpss / libcta) ----
170 *
171 * WHAT: If `adapter` names a library-native dialect AND its .so path is
172 * resolvable AND the library loads with the required ABI, points st->mss/mss_ctx
173 * at the dlopen'd adapter. ALWAYS returns 0 (the caller may proceed): a
174 * non-lib-family adapter, an unset path, an absent library, or a missing symbol
175 * all leave st->mss NULL so the exec/stub fallback runs. The vendor library is a
176 * runtime plug-in; its absence is a graceful degrade, never a hard failure.
177 *
178 * WHY: isolates the library-native decision so brix_sd_frm_create stays a flat
179 * orchestration below the complexity cap, and mirrors frm_select_exec_adapter.
180 */
181int
182frm_select_lib_adapter(sd_frm_state *st, const char *adapter,
183 const char *location, ngx_log_t *log)
184{
185 const char *path;
186
187 if (!frm_adapter_is_lib_family(adapter)) {
188 return 0;
189 }
190 path = frm_lib_path(adapter);
191 if (path == NULL) {
192 ngx_log_error(NGX_LOG_WARN, log, 0,
193 "xrootd frm: the \"%s\" MSS adapter needs an HSM library path "
194 "($BRIX_FRM_LIB, or the per-dialect override); "
195 "falling back to the built-in stub", adapter);
196 return 0;
197 }
198 st->mss_ctx = brix_mss_lib_create(location, path, log);
199 if (st->mss_ctx == NULL) {
200 /* brix_mss_lib_create already WARN-logged the dlopen/symbol failure. */
201 return 0; /* graceful fallback to exec/stub */
202 }
203 st->mss = &brix_mss_lib_adapter;
204 ngx_log_error(NGX_LOG_NOTICE, log, 0,
205 "xrootd frm: \"%s\" library-native MSS adapter (lib=%s, online buffer=%s)",
206 adapter, path, location);
207 return 0;
208}

210/* ---- Select an exec-family MSS adapter (exec / hpss / cta) when requested ----
211 *
212 * WHAT: If `adapter` names an exec-family HSM dialect AND its stage command is
213 * resolvable (`frm_exec_stagecmd`), builds the exec MSS context and points
214 * `st->mss`/`st->mss_ctx` at it. Returns 0 whenever the caller may proceed
215 * (adapter not exec-family, or no stage command - in both cases `st->mss` is left
216 * NULL so the stub fallback runs); returns -1 with errno set to ENOMEM only when
217 * the exec context allocation itself fails.
218 *
219 * WHY: Isolates the exec-adapter decision (the classic FRM model over an external
220 * stage command) so brix_sd_frm_create stays a flat orchestration below the
221 * complexity cap. HPSS/CTA are real named dialects here, not stub fallthroughs.
222 *
223 * HOW:
224 * 1. If `adapter` is not exec-family, return 0 (fall through to the stub).
225 * 2. Resolve the stage command; if none, WARN and return 0 (stub fallback).
226 * 3. Create the exec MSS context; on failure set errno=ENOMEM and return -1.
227 * 4. On success, publish the exec adapter, NOTICE-log the dialect, return 0.
228 */
229int
230frm_select_exec_adapter(sd_frm_state *st, const char *adapter,
231 const char *location, ngx_log_t *log)
232{
233 const char *cmd;
234
235 if (!frm_adapter_is_exec_family(adapter)) {
236 return 0;
237 }
238 cmd = frm_exec_stagecmd(adapter);
239 if (cmd == NULL) {
240 ngx_log_error(NGX_LOG_WARN, log, 0,
241 "xrootd frm: the \"%s\" MSS adapter needs a stage command "
242 "($BRIX_FRM_STAGECMD, or the per-dialect override); "
243 "falling back to the built-in stub", adapter);
244 return 0;
245 }
246 st->mss_ctx = brix_mss_exec_create(location, cmd, log);
247 if (st->mss_ctx == NULL) {
248 errno = ENOMEM;
249 return -1;
250 }
251 st->mss = &brix_mss_exec_adapter;
252 ngx_log_error(NGX_LOG_NOTICE, log, 0,
253 "xrootd frm: \"%s\" MSS adapter (stagecmd=%s, online buffer=%s)",
254 adapter, cmd, location);
255 return 0;
256}

258/* ---- Select the built-in local-dir stub MSS adapter (default / fallback) ----
259 *
260 * WHAT: Builds the stub MSS context that simulates tape with local directories and
261 * points `st->mss`/`st->mss_ctx` at it. Returns 0 on success, or -1 with errno set
262 * to ENOMEM when the stub context allocation fails. An `adapter` name that is
263 * neither empty, "stub", nor "exec" is WARN-logged as not-yet-implemented before
264 * the stub is used, matching the original in-place behaviour.
265 *
266 * WHY: The stub is the default and the fallback for every adapter that is not a
267 * working "exec"; factoring it out keeps the create orchestrator flat and under
268 * the complexity cap while preserving the exact warning and errno semantics.
269 *
270 * HOW:
271 * 1. If `adapter` is a non-empty name that is neither "stub" nor an
272 * exec-family dialect (exec/hpss/cta), WARN that it is unrecognized.
273 * 2. Create the stub MSS context; on failure set errno=ENOMEM and return -1.
274 * 3. Publish the stub adapter and return 0.
275 */
276int
277frm_select_stub_adapter(sd_frm_state *st, const char *adapter,
278 const char *location, ngx_log_t *log)
279{
280 if (adapter != NULL && adapter[0] != '\0'
281 && ngx_strcmp(adapter, "stub") != 0
282 && !frm_adapter_is_exec_family(adapter)
283 && !frm_adapter_is_lib_family(adapter))
284 {
285 ngx_log_error(NGX_LOG_WARN, log, 0,
286 "xrootd frm: MSS adapter \"%s\" is not recognized (known: stub, "
287 "exec, hpss, cta, lib, libhpss, libcta); using the built-in stub",
288 adapter);
289 }
290 st->mss_ctx = brix_mss_stub_create(location, log);
291 if (st->mss_ctx == NULL) {
292 errno = ENOMEM;
293 return -1;
294 }
295 st->mss = &brix_mss_stub_adapter;
296 return 0;
297}
