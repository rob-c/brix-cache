32/*
33 * Enumerate all ADS streams on a file
34 *
35 * Uses FindFirstStreamW/FindNextStreamW to enumerate all alternate
36 * data streams associated with the file.
37 */
38ssize_t
39brix_plat_listxattr(const char *path, char *list, size_t size)
40{
41 wchar_t w_filepath[MAX_PATH];
42 HANDLE find_handle;
43 WIN32_FIND_STREAM_DATA stream_data;
44 brix_win32_ads_enum_t enum_ctx;
45 int result;
46
47 if (path == NULL) {
48 errno = EINVAL;
49 return -1;
50 }
51
52 /* Convert path to wide string */
53 result = MultiByteToWideChar(CP_UTF8, 0, path, -1, w_filepath, MAX_PATH);
54 if (result == 0) {
55 brix_win32_set_errno(GetLastError());
56 return -1;
57 }
58
59 /* Initialize enumeration context */
60 enum_ctx.list = list;
61 enum_ctx.size = size;
62 enum_ctx.offset = 0;
63 enum_ctx.count = 0;
64
65 /* Find first stream */
66 find_handle = FindFirstStreamW(w_filepath, FindStreamInfoStandard,
67 &stream_data, 0);
68
69 if (find_handle == INVALID_HANDLE_VALUE) {
70 brix_win32_set_errno(GetLastError());
71 return -1;
72 }
73
74 do {
75 /* Skip the default unnamed stream (::DATA) */
76 if (wcscmp(stream_data.cStreamName, L"::DATA") == 0) {
77 continue;
78 }
79
80 /* Convert stream name to UTF-8 */
81 char stream_name[256];
82 int name_len = WideCharToMultiByte(CP_UTF8, 0, stream_data.cStreamName, -1,
83 stream_name, sizeof(stream_name),
84 NULL, NULL);
85
86 if (name_len == 0) {
87 continue; /* Skip invalid names */
88 }
89
90 name_len--; /* Remove null terminator from length */
91
92 /* If list buffer is NULL, just count */
93 if (list == NULL || size == 0) {
94 enum_ctx.offset += name_len + 1; /* +1 for null separator */
95 enum_ctx.count++;
96 continue;
97 }
98
99 /* Check if buffer has space */
100 if (enum_ctx.offset + name_len + 1 > size) {
101 errno = ERANGE;
102 FindClose(find_handle);
103 return -1;
104 }
105
106 /* Copy stream name to list */
107 memcpy(list + enum_ctx.offset, stream_name, name_len);
108 list[enum_ctx.offset + name_len] = '\0';
109 enum_ctx.offset += name_len + 1;
110 enum_ctx.count++;
111
112 } while (FindNextStreamW(find_handle, &stream_data));
113
114 FindClose(find_handle);
115
116 if (enum_ctx.count == 0) {
117 errno = ENODATA;
118 return -1;
119 }
120
121 return (ssize_t)enum_ctx.offset;
122}
