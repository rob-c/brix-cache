1/*
2 * tier_directives.h — X-macro for the phase-64 composable tier grammar
3 * directive table (<pfx>{cache_store,cache_cold_store,stage,stage_store,
4 * stage_flush,cache_max_object,cache_evict_at,cache_evict_to,
5 * cache_index_cache,cache_meta,cache_slice_size,cache_global_cas,
6 * cache_passthrough,cache_passthrough_max,cache_prefetch,
7 * cache_prefetch_window,cache_only_if_cached,vfs_spill_path,vfs_spill_max,
8 * durable_publish,lock_enforcement}).
9 *
10 * WHAT: BRIX_TIER_DIRECTIVES(pfx, conf_t, ctx, conf_off) expands to the 21
11 * ngx_command_t initializers for the tier grammar, all writing into the
12 * embedded ngx_http_brix_shared_conf_t `common` preamble. The
13 * authoritative name list is the `ngx_string(pfx "...")` tokens in the
14 * macro body below — tools/ci/check_directive_registry.py parses THIS
15 * header for it, so adding an entry here is picked up with no hand-edit.
16 * WHY: root://, WebDAV and S3 once declared byte-identical tables differing
17 * only in prefix, conf struct and context flags — a cross-protocol parity
18 * bug magnet and a triple audit surface. One macro guarantees the grammars
19 * cannot drift.
20 * HOW: Post-unification (2026-07 grammar + phase-101) the macro is instantiated
21 * ONCE per plane with the BARE "brix_" prefix — HTTP via the shared common
22 * module (http_common.c) and stream via directives_tier.h — so every HTTP
23 * protocol inherits the one registration through the common preamble rather
24 * than re-instantiating with its own prefix. The shared sync/async and
25 * meta-mode enum tables are static per including TU. cvmfs deliberately
26 * exposes only cache_store and is NOT converted.
27 */
