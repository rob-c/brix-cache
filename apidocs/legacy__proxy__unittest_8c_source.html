38/* A certificate with `subject`, issued by `issuer_name`, optionally carrying
39 * proxyCertInfo (RFC 3820) with the given policy OID. */
40static X509 *
41make_cert(X509_NAME *subject, X509_NAME *issuer_name, const char *pci_policy_oid)
42{
43 X509 *cert = X509_new();
44
45 X509_set_version(cert, 2);
46 X509_set_subject_name(cert, subject);
47 X509_set_issuer_name(cert, issuer_name);
48 X509_gmtime_adj(X509_getm_notBefore(cert), -60);
49 X509_gmtime_adj(X509_getm_notAfter(cert), 3600);
50 X509_set_pubkey(cert, throwaway_key());
51 if (pci_policy_oid != NULL) {
52 PROXY_CERT_INFO_EXTENSION *pci = PROXY_CERT_INFO_EXTENSION_new();
53
54 pci->proxyPolicy->policyLanguage = OBJ_txt2obj(pci_policy_oid, 1);
55 X509_add1_ext_i2d(cert, NID_proxyCertInfo, pci, 1, 0);
56 PROXY_CERT_INFO_EXTENSION_free(pci);
57 }
58 X509_sign(cert, throwaway_key(), EVP_sha256());
59 return cert;
60}

85static void
86test_shapes(void)
87{
88 X509_NAME *eec = name_of("test", "Alice");
89 X509_NAME *ca = name_of("test", "Test CA");
90 X509_NAME *full = child_name(eec, "proxy");
91 X509_NAME *limited = child_name(eec, "limited proxy");
92 X509_NAME *numeric = child_name(eec, "12345678");
93 X509_NAME *other = child_name(eec, "something");
94 X509_NAME *twodeep = child_name(full, "proxy");
95 X509 *c;
96
97 c = make_cert(eec, ca, NULL);
98 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_NONE); /* an EEC */
99 X509_free(c);
100 c = make_cert(full, eec, NULL);
101 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_FULL);
102 X509_free(c);
103 c = make_cert(limited, eec, NULL);
104 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_LIMITED);
105 X509_free(c);
106 c = make_cert(numeric, eec, NULL);
107 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_FULL);
108 X509_free(c);
109 c = make_cert(other, eec, NULL);
110 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_NONE); /* not a proxy CN */
111 X509_free(c);
112 c = make_cert(twodeep, eec, NULL);
113 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_NONE); /* two CNs added */
114 X509_free(c);
115 c = make_cert(full, ca, NULL);
116 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_NONE); /* issuer is not the prefix */
117 X509_free(c);
118 /* an RFC 3820 proxy is never a GT2 proxy, whatever its CN */
119 c = make_cert(full, eec, RFC_IMPERSONATION);
120 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_NONE);
121 CHECK(brix_px_classify(c) == BRIX_PX_FULL);
122 X509_free(c);
123 c = make_cert(numeric, eec, RFC_LIMITED);
124 CHECK(brix_gt2_proxy_kind(c) == BRIX_PX_NONE);
125 CHECK(brix_px_classify(c) == BRIX_PX_LIMITED);
126 X509_free(c);
127
128 X509_NAME_free(eec); X509_NAME_free(ca); X509_NAME_free(full);
129 X509_NAME_free(limited); X509_NAME_free(numeric); X509_NAME_free(other);
130 X509_NAME_free(twodeep);
131}

133/* Once the verifier marks a GT2 proxy, the classifier reports its kind and
134 * the limited-monotonicity rule applies across GT2 and RFC proxies. */
135static void
136test_marked_chain_rules(void)
137{
138 X509_NAME *eec = name_of("test", "Bob");
139 X509_NAME *ca = name_of("test", "Test CA");
140 X509_NAME *lim = child_name(eec, "limited proxy");
141 X509_NAME *full_below = child_name(lim, "proxy");
142 X509 *root = make_cert(ca, ca, NULL);
143 X509 *user = make_cert(eec, ca, NULL);
144 X509 *limited = make_cert(lim, eec, NULL);
145 X509 *full = make_cert(full_below, lim, RFC_IMPERSONATION);
146 STACK_OF(X509) *chain = sk_X509_new_null();
147
148 CHECK(brix_px_classify(limited) == BRIX_PX_LIMITED); /* by shape, before marking */
149 (void) X509_get_extension_flags(limited);
150 X509_set_proxy_flag(limited);
151 CHECK(brix_px_classify(limited) == BRIX_PX_LIMITED); /* and after the verifier marks it */
152 CHECK(X509_get_extension_flags(limited) & EXFLAG_PROXY);
153
154 /* leaf..root order as X509_STORE_CTX_get0_chain hands it */
155 sk_X509_push(chain, full);
156 sk_X509_push(chain, limited);
157 sk_X509_push(chain, user);
158 sk_X509_push(chain, root);
159 CHECK(!brix_proxy_chain_ok(chain)); /* RFC full beneath GT2 limited: escalation */
160 sk_X509_delete(chain, 0);
161 CHECK(brix_proxy_chain_ok(chain)); /* GT2 limited leaf alone is fine */
162
163 sk_X509_free(chain);
164 X509_free(full); X509_free(limited); X509_free(user); X509_free(root);
165 X509_NAME_free(eec); X509_NAME_free(ca); X509_NAME_free(lim); X509_NAME_free(full_below);
166}
