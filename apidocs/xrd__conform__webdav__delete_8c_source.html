206void xrd_conform_probe_webdav_delete(xrd_conform_execution *execution,
207 const brix_weburl *url, const char *token)
208{
209 brix_weburl collection = { 0 };
210 brix_weburl nested = { 0 };
211 brix_weburl first = { 0 };
212 brix_weburl second = { 0 };
213 dav_delete_job job;
214 pthread_t thread;
215 const char *failure = NULL;
216 char reason[64] = "delete-worker-unavailable";
217 int absent = 0;
218 int partial = 0;
219 int descendants_absent;
220 int pass;
221
222 if (!xrd_conform_row_active(execution, "DAV-006")
223 && !xrd_conform_row_active(execution, "RM-008")) return;
224 if (!prepare_delete_tree(execution, url, token, &collection, &nested,
225 &first, &second)) {
226 set_delete_result(execution, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
227 "fixture-setup:recursive-webdav-collection", NULL, NULL);
228 if (collection.path[0] != '\0')
229 cleanup_tree(execution, &collection, token);
230 return;
231 }
232 if (!start_delete_job(execution, &collection, token, &job, &thread,
233 &failure, reason, sizeof(reason))) {
234 set_delete_result(execution, BRIX_CONFORM_VERDICT_INCONCLUSIVE,
235 job.headers == NULL ? "delete-header-construction-failed"
236 : "delete-worker-unavailable",
237 NULL, failure == NULL ? reason : failure);
238 free(job.headers);
239 cleanup_tree(execution, &collection, token);
240 return;
241 }
242 partial = observe_delete_snapshots(execution, &collection, token, &absent);
243 pass = delete_completed(execution, &job, thread, &collection, &nested,
244 &first, &second, token, partial, &absent, &descendants_absent);
245 set_delete_result(execution,
246 pass ? BRIX_CONFORM_VERDICT_CONFORMANT
247 : BRIX_CONFORM_VERDICT_DIVERGENT,
248 pass ? "recursive-delete-observed-atomically"
249 : "recursive-delete-partial-or-incomplete",
250 "every Depth-1 snapshot is complete or absent; final tree absent",
251 partial ? "partial-tree-observed"
252 : descendants_absent ? job.status.msg
253 : "orphaned-descendant-observed");
254 if (job.rc == 0) brix_http_resp_free(&job.response);
255 free(job.headers);
256 cleanup_tree(execution, &collection, token);
257}
