1/* File: sss_entity.c — SSS v2 entity credential builder + LGID challenge decoder
2 * WHAT: Implements brix_sss_build_entity_credential() (USEDATA entity or the
3 * SNDLID first round) and brix_sss_challenge_lgid() (decrypt the server's
4 * kXR_authmore reply and pull the LGID out of it). See sss_entity.h.
5 * WHY: One ngx-free kernel for the proxy arm and the native client so the
6 * entity wire (field order, 2-byte BE lengths, NUL-included packed strings,
7 * per-field caps, CRC32-IEEE trailer, BF32 body, 16-byte outer header)
8 * cannot drift between the two producers and the server parser.
9 * HOW: The cleartext is assembled in a stack buffer of BRIX_SSS_ENTITY_BLOB_MAX
10 * bytes through a small bounded TLV writer; every string is checked against
11 * its receiver-side cap and the builder fails closed instead of truncating.
12 * The trailer CRC and the cipher pass reuse the v1 kernels (crc32_ieee.c,
13 * sss_bf.c) unchanged, so a NAME-only entity is byte-identical to the v1
14 * brix_sss_build_credential() output for the same nonce and gen_time
15 * (pinned by sss_entity_unittest.c).
16 */

25typedef struct {

30static void
31sss_put_u32(uint8_t *p, uint32_t v)
32{
33 p[0] = (uint8_t) (v >> 24);
34 p[1] = (uint8_t) (v >> 16);
35 p[2] = (uint8_t) (v >> 8);
36 p[3] = (uint8_t) v;
37}

39static uint32_t
40sss_get_u32(const uint8_t *p)
41{
42 return ((uint32_t) p[0] << 24) | ((uint32_t) p[1] << 16)
43 | ((uint32_t) p[2] << 8) | (uint32_t) p[3];
44}

46/* [type][len hi][len lo][value] — refuses anything that would overrun `end`. */
47static int
48sss_tlv_put(sss_tlv_writer_t *w, uint8_t type, const uint8_t *value,
49 size_t len)
50{
51 if (len > 0xFFFF || (size_t) (w->end - w->cur) < 3 + len) {
52 return -1;
53 }
54 *w->cur++ = type;
55 *w->cur++ = (uint8_t) (len >> 8);
56 *w->cur++ = (uint8_t) len;
57 memcpy(w->cur, value, len);
58 w->cur += len;
59 return 0;
60}

62/* Packed string: NUL-terminated, NUL counted in the length. A NULL or empty
63 * string is simply omitted; a string that would not fit the receiver's
64 * `cap`-byte buffer (including its NUL) is refused — never truncated. */
65static int
66sss_tlv_put_str(sss_tlv_writer_t *w, uint8_t type, const char *s, size_t cap)
67{
68 size_t n;
69
70 if (s == NULL || *s == '\0') {
71 return 0;
72 }
73 n = strnlen(s, cap);
74 if (n >= cap) {
75 return -1;
76 }
77 return sss_tlv_put(w, type, (const uint8_t *) s, n + 1);
78}

80/* 40-byte data header: 32 nonce + gen_time(BE) + 3 spare + option byte. */
81static void
82sss_data_header(uint8_t *clear, const uint8_t nonce32[32], uint32_t gen_time,
83 uint8_t opt)
84{
85 memset(clear, 0, BRIX_SSS_DATA_HDR_LEN);
86 memcpy(clear, nonce32, 32);
87 sss_put_u32(clear + 32, gen_time);
88 clear[39] = opt;
89}

91/* 16-byte outer header: magic + version + spare + kn + enc + key_id(8B BE). */
92static void
93sss_outer_header(uint8_t *out, uint64_t key_id)
94{
95 out[0] = 's'; out[1] = 's'; out[2] = 's'; out[3] = '\0';
96 out[4] = 1; /* version */
97 out[5] = 0; /* spare */
98 out[6] = 0; /* kn_size: no named key */
99 out[7] = BRIX_SSS_ENC_BF32;
100 sss_put_u32(out + 8, (uint32_t) (key_id >> 32));
101 sss_put_u32(out + 12, (uint32_t) key_id);
102}

104/* Append the entity TLVs after the data header; *clear_len receives the total
105 * cleartext length. Field order mirrors the tag order so a server walking the
106 * stream sees NAME first, exactly as the v1 wire did. */
107static int
108sss_entity_tlvs(uint8_t *clear, size_t clear_max, const brix_sss_entity_t *ent,
109 size_t *clear_len)
110{
111 sss_tlv_writer_t w = { clear + BRIX_SSS_DATA_HDR_LEN, clear + clear_max };
112 const char *name;
113
114 name = (ent->name != NULL && ent->name[0] != '\0') ? ent->name : "xrd";
115
116 if (sss_tlv_put_str(&w, BRIX_SSS_TYPE_NAME, name, BRIX_SSS_ENT_NAME_MAX) != 0
117 || sss_tlv_put_str(&w, BRIX_SSS_TYPE_VORG, ent->vorg, BRIX_SSS_ENT_VORG_MAX) != 0
118 || sss_tlv_put_str(&w, BRIX_SSS_TYPE_ROLE, ent->role, BRIX_SSS_ENT_ROLE_MAX) != 0
119 || sss_tlv_put_str(&w, BRIX_SSS_TYPE_GRPS, ent->grps, BRIX_SSS_ENT_GRPS_MAX) != 0
120 || sss_tlv_put_str(&w, BRIX_SSS_TYPE_ENDO, ent->endo, BRIX_SSS_ENT_ENDO_MAX) != 0)
121 {
122 return -1;
123 }
124 if (ent->creds != NULL && ent->creds_len > 0) {
125 if (ent->creds_len > BRIX_SSS_ENT_CREDS_MAX
126 || sss_tlv_put(&w, BRIX_SSS_TYPE_CRED, ent->creds, ent->creds_len) != 0)
127 {
128 return -1;
129 }
130 }
131 *clear_len = (size_t) (w.cur - clear);
132 return 0;
133}

135int
136brix_sss_build_entity_credential(const uint8_t *key, size_t key_len,
137 uint64_t key_id, const brix_sss_entity_t *ent, const uint8_t nonce32[32],
138 uint32_t gen_time, uint8_t *out, size_t out_max, size_t *out_len)
139{
140 uint8_t plain[BRIX_SSS_ENTITY_BLOB_MAX]; /* cleartext + CRC32 */
141 size_t clear_len, crypt_out;
142
143 if (key == NULL || key_len == 0 || nonce32 == NULL || out == NULL
144 || out_len == NULL
145 || out_max < BRIX_SSS_HDR_LEN + BRIX_SSS_DATA_HDR_LEN + 4)
146 {
147 return -1;
148 }
149
150 sss_data_header(plain, nonce32, gen_time,
151 ent == NULL ? BRIX_SSS_OPT_SNDLID : BRIX_SSS_OPT_USEDATA);
152 clear_len = BRIX_SSS_DATA_HDR_LEN;
153 if (ent != NULL
154 && sss_entity_tlvs(plain, sizeof(plain) - 4, ent, &clear_len) != 0)
155 {
156 return -1;
157 }
158
159 /* plain = cleartext + IEEE-CRC32 (big-endian). */
160 sss_put_u32(plain + clear_len, brix_crc32_ieee(plain, clear_len));
161
162 if ((size_t) BRIX_SSS_HDR_LEN + clear_len + 4 > out_max) {
163 return -1;
164 }
165 if (brix_sss_bf_crypt(1, key, key_len, plain, clear_len + 4,
166 out + BRIX_SSS_HDR_LEN, out_max - BRIX_SSS_HDR_LEN,
167 &crypt_out) != 0)
168 {
169 return -1;
170 }
171 sss_outer_header(out, key_id);
172 *out_len = (size_t) BRIX_SSS_HDR_LEN + crypt_out;
173 return 0;
174}

176/* Walk the TLV stream for the LGID field; copy it out as a C string. */
177static int
178sss_find_lgid(const uint8_t *p, size_t n, char *lgid, size_t lgid_max)
179{
180 const uint8_t *end = p + n;
181 size_t len, copy;
182
183 while ((size_t) (end - p) >= 3) {
184 uint8_t type = p[0];
185
186 len = ((size_t) p[1] << 8) | (size_t) p[2];
187 p += 3;
188 if ((size_t) (end - p) < len) {
189 return -1;
190 }
191 if (type == BRIX_SSS_TYPE_LGID) {
192 copy = (len > 0 && p[len - 1] == '\0') ? len - 1 : len;
193 if (copy >= lgid_max) {
194 copy = lgid_max - 1;
195 }
196 memcpy(lgid, p, copy);
197 lgid[copy] = '\0';
198 return 0;
199 }
200 p += len;
201 }
202 return -1;
203}

205int
206brix_sss_challenge_lgid(const uint8_t *key, size_t key_len,
207 const uint8_t *body, size_t body_len, char *lgid, size_t lgid_max)
208{
209 uint8_t clear[BRIX_SSS_ENTITY_BLOB_MAX];
210 size_t hdr_len, plain_len, clear_len;
211
212 if (key == NULL || key_len == 0 || body == NULL || lgid == NULL
213 || lgid_max == 0 || body_len < BRIX_SSS_HDR_LEN)
214 {
215 return -1;
216 }
217 if (memcmp(body, "sss", 4) != 0 || body[7] != BRIX_SSS_ENC_BF32) {
218 return -1;
219 }
220 hdr_len = (size_t) BRIX_SSS_HDR_LEN + body[6];
221 if (body_len < hdr_len + BRIX_SSS_DATA_HDR_LEN + 4
222 || body_len - hdr_len > sizeof(clear))
223 {
224 return -1;
225 }
226 if (brix_sss_bf_crypt(0, key, key_len, body + hdr_len, body_len - hdr_len,
227 clear, sizeof(clear), &plain_len) != 0
228 || plain_len < BRIX_SSS_DATA_HDR_LEN + 4)
229 {
230 return -1;
231 }
232 clear_len = plain_len - 4;
233 if (sss_get_u32(clear + clear_len) != brix_crc32_ieee(clear, clear_len)) {
234 return -1;
235 }
236 return sss_find_lgid(clear + BRIX_SSS_DATA_HDR_LEN,
237 clear_len - BRIX_SSS_DATA_HDR_LEN, lgid, lgid_max);
238}
