1/*
2 * resolve_bridge.c — thread-pool -> event-loop resolver bridge (phase-116
3 * Appendix A: "zero blocking sites outside the driver").
4 *
5 * WHAT: Lets a thread-pool task resolve a hostname through the event loop's
6 * nginx resolver (the one `brix_resolver auto` built from resolv.conf)
7 * and block only on a condition variable until the answer is back.
8 * brix_dns_resolve_sync() calls brix_dns_bridge_resolve() and
9 * brix_dns_reverse_sync() calls brix_dns_bridge_reverse(); both return
10 * NGX_DECLINED — "use libc" — when the calling thread IS the event loop,
11 * when the bridge is not armed, or when the policy has no usable
12 * resolver, so the caller always has a path.
13 * WHY: Blocking callers (TPC pins, cache-origin/gsiftp connects, cvmfs
14 * probes, libcurl pinning) would otherwise fall back to libc and drift
15 * from the resolver policy the operator configured: different search
16 * semantics, a different resolv.conf when `path=` is set, no shared
17 * cache. With the bridge every resolution in the worker — async or
18 * blocking — flows through one driver (I-DNS-2/3).
19 * HOW: A per-process singleton (origin_probe precedent) holding an eventfd
20 * registered with the event loop (the uring_bringup connection pattern),
21 * a mutex-guarded pending list and an in-flight queue. A thread
22 * allocates an item on the heap, links it, writes the eventfd and waits
23 * on the item's condvar in 100 ms slices so it can observe the worker's
24 * exit flags (ngx_thread_pool_destroy would otherwise stall behind a
25 * waiting thread). The loop drains the eventfd, splices the pending
26 * list and drives each item through brix_dns_resolve(); the completion
27 * sets `done` and signals. Whoever observes the other side gone frees
28 * the item: a thread that timed out marks it abandoned (or unlinks it
29 * if still pending) and the completion releases it; a completion that
30 * finds `abandoned` releases it itself. Items always carry a usable
31 * resolver so the loop never takes the thread-pool path for them.
32 */

88/* Loop side: a driven request finished (possibly inline in dispatch). */
89static void
90dns_bridge_item_done(dns_bridge_item_t *item)
91{
92 unsigned abandoned;
93
94 ngx_queue_remove(&item->inflight);
95 pthread_mutex_lock(&dns_bridge.mu);
96 abandoned = item->abandoned;
97 item->done = 1;
98 if (!abandoned) {
99 pthread_cond_signal(&item->cv);
100 }
101 pthread_mutex_unlock(&dns_bridge.mu);
102 if (abandoned) {
103 dns_bridge_item_free(item);
104 }
105}

136static void
137dns_bridge_dispatch(dns_bridge_item_t *item)
138{
139 ngx_queue_insert_tail(&dns_bridge.inflight, &item->inflight);
140 if (item->reverse) {
141 dns_bridge_dispatch_reverse(item);
142 return;
143 }
144 item->req.log = ngx_cycle->log;
145 item->req.handler = dns_bridge_done;
146 item->req.data = item;
147 if (brix_dns_resolve(&item->req) != NGX_OK) {
148 item->req.rc = NGX_ERROR;
149 item->req.naddrs = 0;
150 item->req.error = "resolver could not be started";
151 dns_bridge_done(&item->req);
152 }
153}

156static void
157dns_bridge_read(ngx_event_t *rev)
158{
159 ngx_connection_t *c = rev->data;
160 dns_bridge_item_t *item, *next;
161
162 (void) brix_plat_wakefd_drain(c->fd);
163
164 pthread_mutex_lock(&dns_bridge.mu);
165 item = dns_bridge.head;
166 dns_bridge.head = NULL;
167 dns_bridge.tail = NULL;
168 for (next = item; next != NULL; next = next->next) {
169 next->pending = 0;
170 }
171 pthread_mutex_unlock(&dns_bridge.mu);
172
173 for ( ; item != NULL; item = next) {
174 next = item->next;
175 item->next = NULL;
176 dns_bridge_dispatch(item); /* may free `item` */
177 }
178}

191ngx_int_t
192brix_dns_bridge_init_worker(ngx_cycle_t *cycle)
193{
194 ngx_connection_t *c;
195
196 if (dns_bridge.ready) {
197 return NGX_OK;
198 }
199 ngx_queue_init(&dns_bridge.inflight);
200 if (brix_plat_wakefd_open(&dns_bridge.efd, &dns_bridge.efd_write,
201 BRIX_EVENTFD_NONBLOCK | BRIX_EVENTFD_CLOEXEC) != 0) {
202 ngx_log_error(NGX_LOG_ALERT, cycle->log, ngx_errno,
203 "brix dns: eventfd() failed, thread resolves use libc");
204 return NGX_OK; /* degraded, not fatal */
205 }
206 c = ngx_get_connection(dns_bridge.efd, cycle->log);
207 if (c == NULL) {
208 dns_bridge_close_wakefd();
209 return NGX_OK;
210 }
211 c->read->handler = dns_bridge_read;
212 c->read->log = cycle->log;
213 c->data = &dns_bridge;
214 if (ngx_add_event(c->read, NGX_READ_EVENT, 0) != NGX_OK) {
215 ngx_free_connection(c);
216 dns_bridge_close_wakefd();
217 return NGX_OK;
218 }
219 dns_bridge.conn = c;
220 dns_bridge.loop_tid = ngx_thread_tid();
221 dns_bridge.ready = 1;
222 return NGX_OK;
223}

228static ngx_msec_t
229dns_bridge_deadline_ms(const brix_dns_policy_t *policy)
230{
231 ngx_msec_t ms, ncand = 1;
232
233 if (policy->search) {
234 ncand += policy->rc.nsearch;
235 }
236 ms = (ngx_msec_t) policy->rc.timeout * policy->rc.attempts * ncand * BRIX_CMS_SEC_TO_MS_MULTIPLIER
237 + BRIX_DNS_TIMEOUT_FLOOR_MS;
238 if (ms < BRIX_DNS_DEADLINE_MIN_MS) {
239 ms = BRIX_DNS_DEADLINE_MIN_MS;
240 }
241 if (ms > BRIX_DNS_DEADLINE_MAX_MS) {
242 ms = BRIX_DNS_DEADLINE_MAX_MS;
243 }
244 return ms;
245}

248static void
249dns_bridge_enqueue(dns_bridge_item_t *item)
250{
251 pthread_mutex_lock(&dns_bridge.mu);
252 item->pending = 1;
253 if (dns_bridge.tail != NULL) {
254 dns_bridge.tail->next = item;
255 } else {
256 dns_bridge.head = item;
257 }
258 dns_bridge.tail = item;
259 dns_bridge.requests++;
260 pthread_mutex_unlock(&dns_bridge.mu);
261 if (brix_plat_wakefd_signal(dns_bridge.efd_write) != 0) {
262 /* only a saturated eventfd counter refuses a +1: the item is queued
263 * and the next wakeup drains it, so log rather than fail */
264 ngx_log_error(NGX_LOG_ALERT, ngx_cycle->log, ngx_errno,
265 "brix dns bridge: eventfd wakeup write failed");
266 }
267}

270/* Unlink a still-pending item (mu held). 1 if it was unlinked. */
271static unsigned
272dns_bridge_unlink_pending(dns_bridge_item_t *item)
273{
274 dns_bridge_item_t **pp, *prev = NULL;
275
276 for (pp = &dns_bridge.head; *pp != NULL; prev = *pp, pp = &(*pp)->next) {
277 if (*pp == item) {
278 *pp = item->next;
279 if (dns_bridge.tail == item) {
280 dns_bridge.tail = prev;
281 }
282 item->pending = 0;
283 return 1;
284 }
285 }
286 return 0;
287}

290/* Wait for `done` up to deadline_ms; the wait is sliced so a worker exit is
291 * observed promptly. 1 = done, 0 = gave up (item marked/unlinked). */
292static unsigned
293dns_bridge_wait(dns_bridge_item_t *item, ngx_msec_t deadline_ms)
294{
295 struct timespec ts;
296 ngx_msec_t waited = 0;
297
298 pthread_mutex_lock(&dns_bridge.mu);
299 while (!item->done) {
300 if (waited >= deadline_ms || ngx_exiting || ngx_quit || ngx_terminate) {
301 if (item->pending) {
302 (void) dns_bridge_unlink_pending(item);
303 pthread_mutex_unlock(&dns_bridge.mu);
304 dns_bridge_item_free(item);
305 return 0;
306 }
307 item->abandoned = 1;
308 dns_bridge.timeouts++;
309 pthread_mutex_unlock(&dns_bridge.mu);
310 return 0;
311 }
312 clock_gettime(CLOCK_REALTIME, &ts);
313 ts.tv_nsec += DNS_BRIDGE_SLICE_MS * BRIX_DNS_NS_PER_MS;
314 if (ts.tv_nsec >= BRIX_DNS_NS_PER_SEC) {
315 ts.tv_sec += 1;
316 ts.tv_nsec -= BRIX_DNS_NS_PER_SEC;
317 }
318 (void) pthread_cond_timedwait(&item->cv, &dns_bridge.mu, &ts);
319 waited += DNS_BRIDGE_SLICE_MS;
320 }
321 pthread_mutex_unlock(&dns_bridge.mu);
322 return 1;
323}

326ngx_int_t
327brix_dns_bridge_resolve(const brix_dns_policy_t *policy, const char *host,
328 in_port_t port, brix_af_policy_t af, int socktype, brix_dns_addr_t *addrs,
329 ngx_uint_t max, ngx_uint_t *naddrs, char *err, size_t errsz)
330{
331 dns_bridge_item_t *item;
332 ngx_uint_t i, n;
333
334 *naddrs = 0;
335 if (!dns_bridge.ready || brix_dns_policy_resolver(policy) == NULL
336 || ngx_thread_tid() == dns_bridge.loop_tid)
337 {
338 return NGX_DECLINED;
339 }
340 item = ngx_calloc(sizeof(dns_bridge_item_t), ngx_cycle->log);
341 if (item == NULL) {
342 return NGX_ERROR;
343 }
344 pthread_cond_init(&item->cv, NULL);
345 item->req.name.data = (u_char *) host;
346 item->req.name.len = ngx_strlen(host);
347 item->req.port = port;
348 item->req.af = af;
349 item->req.socktype = socktype;
350 item->req.policy = policy;
351
352 dns_bridge_enqueue(item);
353 if (!dns_bridge_wait(item, dns_bridge_deadline_ms(policy))) {
354 if (err != NULL && errsz > 0) {
355 (void) ngx_snprintf((u_char *) err, errsz,
356 "resolver timed out (event loop)%Z");
357 }
358 return NGX_OK; /* answered: failure */
359 }
360
361 n = ngx_min(item->req.naddrs, max);
362 if (item->req.rc == NGX_OK && n > 0) {
363 for (i = 0; i < n; i++) {
364 addrs[i] = item->req.addrs[i];
365 }
366 *naddrs = n;
367 } else if (err != NULL && errsz > 0) {
368 (void) ngx_snprintf((u_char *) err, errsz, "%s%Z",
369 item->req.error ? item->req.error
370 : "no usable address");
371 }
372 dns_bridge_item_free(item);
373 return NGX_OK;
374}

377ngx_int_t
378brix_dns_bridge_reverse(const brix_dns_policy_t *policy,
379 const struct sockaddr *sa, socklen_t len, char *buf, size_t buflen,
380 ngx_int_t *answer)
381{
382 dns_bridge_item_t *item;
383
384 *answer = NGX_ERROR;
385 if (!dns_bridge.ready || brix_dns_policy_resolver(policy) == NULL
386 || ngx_thread_tid() == dns_bridge.loop_tid || len > sizeof(item->rev.ss))
387 {
388 return NGX_DECLINED;
389 }
390 item = ngx_calloc(sizeof(dns_bridge_item_t), ngx_cycle->log);
391 if (item == NULL) {
392 return NGX_DECLINED;
393 }
394 pthread_cond_init(&item->cv, NULL);
395 item->reverse = 1;
396 ngx_memcpy(&item->rev.ss, sa, len);
397 item->rev.len = len;
398 item->rev.policy = policy;
399
400 dns_bridge_enqueue(item);
401 if (!dns_bridge_wait(item, dns_bridge_deadline_ms(policy))) {
402 return NGX_OK; /* answered: failure */
403 }
404 *answer = item->rev.rc;
405 if (item->rev.rc == NGX_OK) {
406 ngx_cpystrn((u_char *) buf, (u_char *) item->rev.name, buflen);
407 }
408 dns_bridge_item_free(item);
409 return NGX_OK;
410}
