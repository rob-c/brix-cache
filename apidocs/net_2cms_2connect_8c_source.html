138/* ngx_brix_cms_schedule_retry — schedule the next reconnect after a failure.
139 *
140 * Two regimes, chosen automatically so every caller (connect.c and recv.c) gets the
141 * right behaviour without having to know which failure path it is on:
142 *
143 * - COLD-START FAST-RETRY: a node that has NEVER yet logged in is racing its
144 * manager's listen socket (most acutely when the whole mesh boots together on
145 * one host). Retry on the short fixed fast-retry interval for a bounded window
146 * so the mesh settles in tens of ms instead of seconds. Quiet (debug) — a brief
147 * race is expected, not WARN-worthy.
148 * - EXPONENTIAL BACKOFF: once the fast-retry window expires, or once this node has
149 * ever logged in (so a failure is a real outage, not a cold start), fall back to
150 * the doubling+jitter backoff. The genuinely-unreachable diagnostic is surfaced
151 * here, sparsely.
152 *
153 * Gating fast-retry on pre-first-login + a bounded window is what guarantees it can
154 * never become a busy-spin (cf. the self-rearming-0ms-timer footgun).
155 */
156
157void
158ngx_brix_cms_schedule_retry(ngx_brix_cms_ctx_t *ctx)
159{
160 ngx_msec_t delay;
161 ngx_msec_t max_backoff;
162 ngx_msec_t now;
163
164 /* Never schedule a reconnect once the worker is draining — a pending retry
165 * timer would otherwise keep the exiting worker alive to the shutdown
166 * timeout for no purpose. */
167 if (ngx_exiting) {
168 return;
169 }
170
171 /* §2.9: while retargeted by a login kYR_try, three consecutive failed
172 * connects to the redirect target fall the link back to the CONFIGURED
173 * manager — a dead supervisor must not orphan the node. */
174 if (ctx->retargeted && !ctx->logged_in) {
175 if (++ctx->retarget_fails >= 3) {
176 ngx_log_error(NGX_LOG_NOTICE, ctx->cycle->log, 0,
177 "brix: CMS retarget %V unreachable — reverting to "
178 "configured manager %V", &ctx->mgr_name, &ctx->orig_name);
179 ctx->mgr_addr = ctx->orig_addr;
180 ctx->mgr_name = ctx->orig_name;
181 ctx->retargeted = 0;
182 ctx->retarget_depth = 0;
183 ctx->retarget_fails = 0;
184 }
185 }
186
187 /* Cold-start fast-retry window. */
188 if (!ctx->ever_logged_in && ctx->fast_window > 0) {
189 now = ngx_current_msec;
190
191 /* Start the window on the first failure of this cold start. */
192 if (ctx->fast_deadline == 0) {
193 ctx->fast_deadline = now + ctx->fast_window;
194 }
195
196 if ((ngx_msec_int_t) (ctx->fast_deadline - now) > 0) {
197 ngx_log_debug2(NGX_LOG_DEBUG_EVENT, ctx->cycle->log, 0,
198 "brix: CMS fast-retry to %V (attempt %ui)",
199 &ctx->mgr_name, ctx->connect_attempts);
200 ngx_brix_cms_schedule(ctx, ctx->fast_retry);
201 return;
202 }
203
204 /* Window expired: the manager is genuinely unreachable for a same-host
205 * cold start — surface the actionable diagnostic, then fall through to the
206 * sparse exponential backoff below. */
207 BRIX_DIAG_WARN(ctx->cycle->log, 0,
208 "xrootd[cms]: cannot reach cluster manager %V",
209 "the cmsd is down, the address/port is wrong, or a firewall blocks "
210 "the connection",
211 "confirm cmsd is listening and that brix_cms_manager matches its "
212 "host:port; this node stays OUT of the cluster until it connects",
213 &ctx->mgr_name);
214 }
215
216 /* Cap max backoff at 10× the heartbeat interval so a short cms_interval
217 * (e.g. 2s for tests) also gives short reconnect windows. */
218 max_backoff = (ngx_msec_t) ctx->conf->cms.interval * 10000;
219 if (max_backoff > NGX_BRIX_CMS_BACKOFF_MAX) {
220 max_backoff = NGX_BRIX_CMS_BACKOFF_MAX;
221 }
222
223 delay = ctx->backoff;
224 if (ctx->backoff < max_backoff) {
225 ctx->backoff *= 2;
226 if (ctx->backoff > max_backoff) {
227 ctx->backoff = max_backoff;
228 }
229 }
230
231 /*
232 * Phase 39 (WS7): add up to +25% random jitter so many workers/nodes that lost
233 * the manager at the same instant do not reconnect in lockstep (a thundering
234 * herd that would re-overload a recovering manager). ngx_random() is nginx's
235 * PRNG; jitter only ever lengthens the delay, never shortens it.
236 */
237 if (delay > 0) {
238 delay += (ngx_msec_t) (ngx_random() % (delay / 4 + 1));
239 }
240
241 ngx_brix_cms_schedule(ctx, delay);
242}

245 * ctx->connection, clear logged_in, and reset inbuf for the next reconnect; called

248void
249ngx_brix_cms_disconnect(ngx_brix_cms_ctx_t *ctx)
250{
251 ngx_connection_t *c;
252
253 c = ctx->connection;
254 if (c == NULL) {
255 return;
256 }
257
258 if (c->read->timer_set) {
259 ngx_del_timer(c->read);
260 }
261
262 if (c->write->timer_set) {
263 ngx_del_timer(c->write);
264 }
265
266 brix_sess_end(ctx->sess, ngx_brix_cms_end_reason(ctx, c));
267 ctx->sess = NULL;
268 ctx->sess_attempt_logged = 0;
269 ctx->sess_end_hint_set = 0;
270
271 ngx_close_connection(c);
272
273 if (ctx->logged_in) {
274 /* This link leaves the cluster now — drop the gauge exactly once per
275 * login/teardown pair (the flag is cleared immediately below). */
276 BRIX_RESIL_METRIC_DEC(cms_registered_links);
277
278 } else {
279 /*
280 * Torn down before LOGIN ever went out: this dial never became a link,
281 * so it is a failed join attempt. Counted HERE rather than at each
282 * failure site because a refused dial can surface on any of three of
283 * them — connect deadline, login write, or (on loopback, the common
284 * one) the read side reporting ECONNREFUSED — and every one of them
285 * funnels through this teardown exactly once.
286 */
287 BRIX_RESIL_METRIC_INC(cms_connect_failures_total);
288 }
289
290 ctx->connection = NULL;
291 ctx->logged_in = 0;
292 ctx->in_pos = 0;
293 ctx->in_need = NGX_BRIX_CMS_HDR_LEN;
294
295 /* Phase-61 W7: flush any kYR_state relays parked on this upward leg —
296 * a late child kYR_have must not be echoed into a dead/reused conn. */
297 brix_cms_state_relay_drop_ctx(ctx);
298}

300static void
301ngx_brix_cms_write_handler(ngx_event_t *ev)
302{
303 ngx_connection_t *c;
304 ngx_brix_cms_ctx_t *ctx;
305
306 c = ev->data;
307 ctx = c->data;
308
309 ngx_log_debug3(NGX_LOG_DEBUG_EVENT, c->write->log, 0,
310 "brix: CMS write handler called timedout=%d "
311 "logged_in=%d c=%p",
312 (int) ev->timedout,
313 ctx ? (int) ctx->logged_in : -1, c);
314
315 if (ev->timedout) {
316 /* A connect/first-write that never completed. Route through the connect
317 * retry policy: pre-first-login it may fast-retry (bounded window), else it
318 * backs off — never the multi-second backoff for a same-host cold start. */
319 ngx_log_debug0(NGX_LOG_DEBUG_EVENT, ev->log, 0,
320 "brix: CMS connect/write timed out");
321 ngx_brix_cms_set_end_hint(ctx, BRIX_SESS_END_TIMEOUT);
322 ngx_brix_cms_disconnect(ctx);
323 ngx_brix_cms_schedule_retry(ctx);
324 return;
325 }
326
327 if (c->write->timer_set) {
328 ngx_del_timer(c->write);
329 }
330
331 if (!ctx->logged_in) {
332 if (ngx_brix_cms_send_login(ctx) != NGX_OK) {
333 /* The TCP connect was refused/half-open (on loopback, ECONNREFUSED
334 * surfaces here as a writable-with-error event, not at connect()).
335 * This is exactly the same-host "manager not listening yet" case → the
336 * fast-retry policy, not the multi-second backoff. */
337 ngx_brix_cms_set_end_hint(ctx, BRIX_SESS_END_ERROR);
338 ngx_brix_cms_disconnect(ctx);
339 ngx_brix_cms_schedule_retry(ctx);
340 return;
341 }
342
343 ctx->logged_in = 1;
344 ctx->retarget_fails = 0; /* §2.9: the retarget (if any) worked */
345 BRIX_RESIL_METRIC_INC(cms_logins_total);
346 BRIX_RESIL_METRIC_INC(cms_registered_links);
347 ctx->backoff = ngx_min((ngx_msec_t) ctx->conf->cms.interval * 1000,
348 (ngx_msec_t) NGX_BRIX_CMS_BACKOFF_INITIAL);
349
350 ngx_log_error(NGX_LOG_NOTICE, ev->log, 0,
351 "brix: CMS login sent to %V",
352 &ctx->mgr_name);
353 brix_cms_log_action(ev->log, "login",
354 (const char *) ctx->mgr_name.data,
355 "out", NULL, 1,
356 ctx->conf->manager_mode
357 ? "sub-manager registering up (Manager bit)"
358 : "leaf/client registering up");
359
360 /* First successful login of this boot: leave fast-retry mode (any future
361 * reconnect is a real outage → backoff) and report the settle time. */
362 if (!ctx->ever_logged_in) {
363 ctx->ever_logged_in = 1;
364 ctx->fast_deadline = 0;
365 ngx_log_error(NGX_LOG_NOTICE, ev->log, 0,
366 "brix: CMS registered with %V after %uL ms "
367 "(%ui connect attempt(s), %s)",
368 &ctx->mgr_name,
369 (brix_phase_now_ns() - ctx->start_ns) / 1000000ull,
370 ctx->connect_attempts,
371 ctx->is_loopback ? "loopback" : "remote");
372 }
373
374 /*
375 * Announce traffic state (Resume|noStage) immediately after login so a
376 * real cmsd manager marks this disk-only node active and eligible for
377 * selection; without it the manager keeps us suspended and never
378 * redirects clients here.
379 */
380 if (ngx_brix_cms_send_status(ctx) != NGX_OK) {
381 ngx_brix_cms_set_end_hint(ctx, BRIX_SESS_END_ERROR);
382 ngx_brix_cms_disconnect(ctx);
383 ngx_brix_cms_schedule_retry(ctx);
384 return;
385 }
386 ngx_brix_cms_log_registration(ctx);
387
388 /*
389 * WS1: arm the manager-silence deadline ONCE, at the login transition.
390 * It is deliberately NOT re-armed on our own heartbeat sends (that would
391 * let our outbound traffic mask a manager that has gone silent); recv.c
392 * resets it only when a frame actually arrives FROM the manager. So it
393 * measures time since the last manager activity and, on expiry, the recv
394 * handler reconnects us to a healthy manager.
395 */
396 ngx_brix_cms_arm_read_deadline(ctx);
397 }
398
399 if (ngx_brix_cms_send_load(ctx) != NGX_OK) {
400 ngx_log_error(NGX_LOG_WARN, ev->log, 0,
401 "brix: CMS write handler: send_load failed");
402 brix_cms_log_action(ev->log, "load",
403 (const char *) ctx->mgr_name.data,
404 "out", NULL, 0, "send_load failed");
405 ngx_brix_cms_set_end_hint(ctx, BRIX_SESS_END_ERROR);
406 ngx_brix_cms_disconnect(ctx);
407 ngx_brix_cms_schedule_retry(ctx);
408 return;
409 }
410
411 if (ngx_handle_read_event(c->read, 0) != NGX_OK) {
412 ngx_brix_cms_set_end_hint(ctx, BRIX_SESS_END_ERROR);
413 ngx_brix_cms_disconnect(ctx);
414 ngx_brix_cms_schedule_retry(ctx);
415 return;
416 }
417
418 ngx_brix_cms_schedule(ctx, (ngx_msec_t) ctx->conf->cms.interval * 1000);
419
420 ngx_log_debug0(NGX_LOG_DEBUG_EVENT, ev->log, 0,
421 "brix: CMS write handler: heartbeat sent");
422}

424/* ngx_brix_cms_connect — start a TCP connect to the CMS manager via
425 * ngx_event_connect_peer, install the read/write handlers, and arm the connect
426 * timeout (NGX_BRIX_CMS_CONNECT_TIMEOUT = 5s); called from the timer when
427 * ctx->connection == NULL. */
428
429static void
430ngx_brix_cms_connect(ngx_brix_cms_ctx_t *ctx)
431{
432 ngx_int_t rc;
433 ngx_connection_t *c;
434
435 ctx->connect_attempts++;
436
437 ngx_memzero(&ctx->peer, sizeof(ctx->peer));
438 ctx->peer.sockaddr = ctx->mgr_addr->sockaddr;
439 ctx->peer.socklen = ctx->mgr_addr->socklen;
440 ctx->peer.name = &ctx->mgr_addr->name;
441 ctx->peer.get = ngx_event_get_peer;
442 ctx->peer.log = ctx->cycle->log;
443 ctx->peer.log_error = NGX_ERROR_ERR;
444
445 rc = ngx_event_connect_peer(&ctx->peer);
446 if (rc == NGX_ERROR || rc == NGX_DECLINED || ctx->peer.connection == NULL) {
447 BRIX_DIAG_WARN(ctx->cycle->log, 0,
448 "xrootd[cms]: cannot reach cluster manager %V",
449 "the cmsd is down, the address/port is wrong, or a firewall "
450 "blocks the connection",
451 "confirm cmsd is listening and that brix_cms_manager matches "
452 "its host:port; this node will keep retrying and stays OUT of the "
453 "cluster until it connects",
454 &ctx->mgr_name);
455 BRIX_RESIL_METRIC_INC(cms_connect_failures_total);
456 ngx_brix_cms_schedule_retry(ctx);
457 return;
458 }
459
460 c = ctx->peer.connection;
461 ctx->connection = c;
462 ctx->logged_in = 0;
463 ctx->in_pos = 0;
464 ctx->in_need = NGX_BRIX_CMS_HDR_LEN;
465
466 c->data = ctx;
467 c->read->handler = ngx_brix_cms_read_handler;
468 c->write->handler = ngx_brix_cms_write_handler;
469 ngx_brix_cms_sess_begin(ctx);
470
471 /*
472 * WS5: OS-level dead-peer reaping on the manager socket, so a silently-
473 * dropped manager is torn down by the kernel even between event-loop
474 * deadlines. Best-effort; failures are non-fatal.
475 */
476 brix_apply_tcp_deadpeer_opts(c->fd, ctx->conf->cms.tcp_keepalive,
477 ctx->conf->cms.tcp_user_timeout);
478
479 if (rc == NGX_AGAIN) {
480 /*
481 * WS2: bound the connect + first-write readiness window with
482 * cms_send_timeout (operator-tunable); the write handler's ev->timedout
483 * path reconnects with backoff. Falls back to the fixed connect timeout
484 * if the knob is disabled (0).
485 */
486 ngx_msec_t connect_tmo = ctx->conf->cms.send_timeout > 0
487 ? ctx->conf->cms.send_timeout
488 : NGX_BRIX_CMS_CONNECT_TIMEOUT;
489 ngx_add_timer(c->write, connect_tmo);
490 return;
491 }
492
493 ngx_brix_cms_write_handler(c->write);
494}

543}
