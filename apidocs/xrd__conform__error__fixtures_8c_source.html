62/* WHAT: Locate the immutable fixture descriptor for one exact errno label.
63 * WHY: CLI parsing and execution must accept the same closed vocabulary.
64 * HOW: Compare bounded input against each literal without requiring a NUL. */
65const error_fixture_spec *
66xrd_conform_error_spec_find(const char *name, size_t length)
67{
68 const error_fixture_spec *specs;
69 size_t count;
70 size_t index;
71
72 specs = xrd_conform_error_specs(&count);
73 for (index = 0; index < count; index++)
74 if (strlen(specs[index].name) == length
75 && memcmp(specs[index].name, name, length) == 0)
76 return &specs[index];
77 return NULL;
78}

81/* WHAT: Validate one --error-fixture errno spelling.

86xrd_conform_error_fixture_name_valid(const char *name, size_t length)
