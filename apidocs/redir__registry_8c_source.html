1/*
2 * redir_registry.c — redirect-collapse registry (§7.11). See redir_registry.h.
3 *
4 * A bounded array of (url -> target) pairs. record() collapses the new target
5 * through the table so stored mappings are always url->FINAL; lookup() still
6 * follows the chain hop-capped as a belt-and-suspenders against a mapping added
7 * before a later one shortened it. Cycles can never be stored (record refuses a
8 * mapping whose collapsed target is the url itself), and both record and lookup
9 * bound their hop counts, so no operation can spin.
10 */
11#include "net/redir_registry.h"
12
13#include <stdlib.h>
14#include <string.h>
15
16#define VREDIR_CAP 256 /* endpoints a single copy touches: small */
17#define VREDIR_MAX_HOPS 32 /* chain-follow bound (cycle/deep-chain guard) */
18
19typedef struct {
20 char *url;
21 char *target;
22 unsigned seq; /* LRU stamp: higher = more recently used */
23} vredir_entry_t;
24
25static vredir_entry_t vredir_tbl[VREDIR_CAP];
26static unsigned vredir_n;
27static unsigned vredir_clock;

29/* Find the slot for `url`, or NULL. */
30static vredir_entry_t *
31vredir_find(const char *url)
32{
33 unsigned i;
34
35 for (i = 0; i < vredir_n; i++) {
36 if (strcmp(vredir_tbl[i].url, url) == 0) {
37 return &vredir_tbl[i];
38 }
39 }
40 return NULL;
41}
42
43/* Follow url through the chain (hop-capped), returning the final target string
44 * or NULL if `url` maps nowhere. */
45static const char *
46vredir_resolve(const char *url)
47{
48 const char *cur = url;
49 const char *last = NULL;
50 int hops;
51
52 for (hops = 0; hops < VREDIR_MAX_HOPS; hops++) {
53 vredir_entry_t *e = vredir_find(cur);
54
55 if (e == NULL) {
56 break;
57 }
58 last = e->target;
59 cur = e->target;
60 }
61 return last;
62}

73/* Evict the least-recently-used entry to make room. */
74static void
75vredir_evict_lru(void)
76{
77 unsigned i, victim = 0;
78
79 for (i = 1; i < vredir_n; i++) {
80 if (vredir_tbl[i].seq < vredir_tbl[victim].seq) {
81 victim = i;
82 }
83 }
84 free(vredir_tbl[victim].url);
85 free(vredir_tbl[victim].target);
86 vredir_tbl[victim] = vredir_tbl[--vredir_n];
87}

89void
90brix_vredir_record(const char *url, const char *target)
91{
92 const char *final;
93 vredir_entry_t *e;
94 char *ku, *kt;
95
96 if (url == NULL || target == NULL || url[0] == '\0' || target[0] == '\0') {
97 return;
98 }
99 /* Collapse the target through any existing chain, then refuse a mapping that
100 * would point url at itself (direct or via the chain) — that is the cycle. */
101 final = vredir_resolve(target);
102 if (final == NULL) {
103 final = target;
104 }
105 if (strcmp(final, url) == 0) {
106 return;
107 }
108
109 e = vredir_find(url);
110 if (e != NULL) { /* update in place */
111 char *nt = strdup(final);
112 if (nt == NULL) {
113 return;
114 }
115 free(e->target);
116 e->target = nt;
117 e->seq = ++vredir_clock;
118 return;
119 }
120
121 if (vredir_n == VREDIR_CAP) {
122 vredir_evict_lru();
123 }
124 ku = strdup(url);
125 kt = strdup(final);
126 if (ku == NULL || kt == NULL) {
127 free(ku);
128 free(kt);
129 return;
130 }
131 vredir_tbl[vredir_n].url = ku;
132 vredir_tbl[vredir_n].target = kt;
133 vredir_tbl[vredir_n].seq = ++vredir_clock;
134 vredir_n++;
135}
