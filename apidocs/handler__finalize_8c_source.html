1/* handler_finalize.c — the cvmfs:// request-finalization observer.
2 *
3 * WHAT: the pool-cleanup observer handler.c registers on every request, plus
4 * its edge helpers: session-log close-out, the optional one-line client
5 * trace, and the T16 fill/hit metric accounting — all keyed off the
6 * FINAL response status.
7 * WHY: handler.c grew past the file-size gate; the finalize family is a
8 * self-contained tail (nothing here touches the serve path) so it moved
9 * here whole. The observer stays the one place every serve path (inline
10 * open, off-loop fill, passthrough) converges — the negative memo (T13)
11 * and the G15 attest observer see every terminal status regardless of
12 * which path produced it.
13 * HOW: cvmfs_finalize_observe is the only cross-file symbol (declared in
14 * cvmfs_module_internal.h; handler.c wires it into the pool cleanup);
15 * everything else is static.
16 */

24/* WHAT: close out the session-log record for this request — result line plus,
25 * for a GET whose transfer started, the byte tally + terminal disposition.
26 * WHY: a success (<400) logs no error; a failure maps the HTTP status to a
27 * sesslog error string. The xfer end classifies COMPLETE vs ABORTED off
28 * the same threshold. Only runs when an attempt was actually logged.
29 * HOW: side-effecting edge helper; guards on sess_attempt_logged and clears
30 * sess_xfer_started so re-entry can't double-count. */
31static void
32cvmfs_finalize_sesslog(ngx_http_request_t *r,
33 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
34 ngx_uint_t status)
35{
36 brix_sess_t *sess;
37 char path[BRIX_SESSLOG_PATH_MAX];
38 char errscratch[BRIX_SESSLOG_ERR_MAX];
39
40 if (lcf == NULL || ctx == NULL || !ctx->sess_attempt_logged) {
41 return;
42 }
43
44 sess = brix_http_sess(r, &lcf->common, BRIX_SESS_PROTO_CVMFS,
45 BRIX_SESS_AM_ANON);
46 brix_sess_result(sess, status < NGX_HTTP_BAD_REQUEST,
47 brix_http_sess_uri(r, path, sizeof(path)),
48 BRIX_SESS_MODE_READ,
49 status < NGX_HTTP_BAD_REQUEST ? NULL
50 : brix_sesslog_err_from_http((int) status,
51 errscratch,
52 sizeof(errscratch)));
53 if (ctx->sess_xfer_started) {
54 if (r->headers_out.content_length_n > 0) {
55 brix_sess_xfer_add(&ctx->sess_xfer,
56 (uint64_t) r->headers_out.content_length_n);
57 }
58 brix_sess_xfer_end(sess, &ctx->sess_xfer,
59 status < NGX_HTTP_BAD_REQUEST ? BRIX_SESS_XFER_COMPLETE
60 : BRIX_SESS_XFER_ABORTED);
61 ctx->sess_xfer_started = 0;
62 }
63}

65/* WHAT: emit the optional one-line client-op trace naming traffic class,
66 * repository, path and final cache disposition + status.
67 * WHY: DEBUG normally (visible under error_log … debug), promoted to INFO by
68 * brix_cvmfs_trace. Correlates with the upstream-request line by path.
69 * HOW: bounds the class/cache indices, sanitizes the non-NUL-terminated uri
70 * span into a stack buffer, and skips the whole build below the level. */
71static void
72cvmfs_finalize_trace(ngx_http_request_t *r,
73 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
74 ngx_uint_t status)
75{
76 static const char *cls_names[] = { "cas", "manifest", "geo", "bundle",
77 "dict", "reject" };
78 static const char *cache_names[] = { "-", "hit", "fill", "neg" };
79 ngx_uint_t level;
80 char safe[1024];
81 char raw[1024];
82 size_t n;
83
84 if (ctx == NULL || lcf == NULL) {
85 return;
86 }
87
88 level = lcf->cvmfs.trace ? NGX_LOG_INFO : NGX_LOG_DEBUG;
89 if (r->connection->log->log_level < level) {
90 return;
91 }
92
93 /* r->uri.data is NOT NUL-terminated (points into the request
94 * buffer); copy the exact uri span before sanitizing. */
95 n = ngx_min(r->uri.len, sizeof(raw) - 1);
96 ngx_memcpy(raw, r->uri.data, n);
97 raw[n] = '\0';
98 brix_sanitize_log_string(raw, safe, sizeof(safe));
99 ngx_log_error(level, r->connection->log, 0,
100 "cvmfs-trace: client id=%uA class=%s repo=%*s path=%s "
101 "cache=%s status=%ui",
102 r->connection->number,
103 cls_names[ctx->url.cls <= CVMFS_URL_REJECT ? ctx->url.cls
104 : CVMFS_URL_REJECT],
105 ctx->url.repo != NULL ? ctx->url.repo_len : (size_t) 0,
106 ctx->url.repo != NULL ? ctx->url.repo : "",
107 safe,
108 cache_names[ctx->cache_status <= BRIX_CVMFS_CACHE_NEG
109 ? ctx->cache_status : 0],
110 status);
111}

113/* WHAT: record the fill-side T16 metrics for a request that missed the cache
114 * and drove an off-loop fill — success counts + bytes, or a fill failure.
115 * WHY: a 200/206 means the fill landed and served; a 502 is a definitive fill
116 * failure. A 504 hold-expiry is NOT counted as a failure — the detached
117 * fill may still publish for the client's retry.
118 * HOW: bumps global + per-repo counters; byte adds use content_length_n which
119 * the serve pipeline set before headers went out. */
120static void
121cvmfs_finalize_metrics_fill(ngx_http_request_t *r,
122 ngx_http_brix_cvmfs_ctx_t *ctx, ngx_uint_t status)
123{
124 if (status == NGX_HTTP_OK || status == NGX_HTTP_PARTIAL_CONTENT) {
125 BRIX_CVMFS_METRIC_INC(fills_total);
126 if (r->headers_out.content_length_n > 0) {
127 BRIX_CVMFS_METRIC_ADD(bytes_served_fill_total,
128 (ngx_atomic_uint_t) r->headers_out.content_length_n);
129 }
130 brix_metric_cache_result(BRIX_PROTO_CVMFS, 0, 0);
131 if (ctx->repo != NULL) {
132 BRIX_ATOMIC_INC(&ctx->repo->fills_total);
133 BRIX_ATOMIC_INC(&ctx->repo->cache_misses_total);
134 if (ctx->url.cls == CVMFS_URL_CAS) {
135 BRIX_ATOMIC_INC(&ctx->repo->files_accessed_total);
136 }
137 if (r->headers_out.content_length_n > 0) {
138 BRIX_ATOMIC_ADD(&ctx->repo->bytes_served_fill_total,
139 (ngx_atomic_uint_t) r->headers_out.content_length_n);
140 }
141 }
142 } else if (status == NGX_HTTP_BAD_GATEWAY) {
143 BRIX_CVMFS_METRIC_INC(fill_failures_total);
144 if (ctx->repo != NULL) {
145 BRIX_ATOMIC_INC(&ctx->repo->fill_failures_total);
146 }
147 }
148 /* a 504 hold-expiry is NOT a definitive fill failure — the
149 * detached fill may still publish for the client's retry */
150}

152/* WHAT: record the hit-side T16 metrics for a request served from a warm cache
153 * (bytes + hit counters) on a 200/206.
154 * WHY: a cache HIT that actually delivered bytes is the success case; other
155 * statuses (e.g. a conditional 304) add nothing.
156 * HOW: bumps global + per-repo hit counters and byte tallies off
157 * content_length_n. */
158static void
159cvmfs_finalize_metrics_hit(ngx_http_request_t *r,
160 ngx_http_brix_cvmfs_ctx_t *ctx)
161{
162 if (r->headers_out.content_length_n > 0) {
163 BRIX_CVMFS_METRIC_ADD(bytes_served_hit_total,
164 (ngx_atomic_uint_t) r->headers_out.content_length_n);
165 }
166 brix_metric_cache_result(BRIX_PROTO_CVMFS, 1, 0);
167 if (ctx->repo != NULL) {
168 BRIX_ATOMIC_INC(&ctx->repo->cache_hits_total);
169 if (ctx->url.cls == CVMFS_URL_CAS) {
170 BRIX_ATOMIC_INC(&ctx->repo->files_accessed_total);
171 }
172 if (r->headers_out.content_length_n > 0) {
173 BRIX_ATOMIC_ADD(&ctx->repo->bytes_served_hit_total,
174 (ngx_atomic_uint_t) r->headers_out.content_length_n);
175 }
176 }
177}

179/* WHAT: dispatch the T16 fill/byte accounting off the FINAL cache disposition.
180 * WHY: the two dispositions (FILL, HIT) have distinct counter families; a
181 * single decision point keeps the accounting off the terminal status.
182 * HOW: routes to the fill- or hit-side helper; NONE/NEG account nothing. */
183static void
184cvmfs_finalize_metrics(ngx_http_request_t *r,
185 ngx_http_brix_cvmfs_ctx_t *ctx, ngx_uint_t status)
186{
187 if (ctx->cache_status == BRIX_CVMFS_CACHE_FILL) {
188 cvmfs_finalize_metrics_fill(r, ctx, status);
189 } else if (ctx->cache_status == BRIX_CVMFS_CACHE_HIT
190 && (status == NGX_HTTP_OK
191 || status == NGX_HTTP_PARTIAL_CONTENT))
192 {
193 cvmfs_finalize_metrics_hit(r, ctx);
194 }
195}

197/* Request-finalization observer: fires once when the request pool is torn
198 * down, with the FINAL response status — the one place every serve path
199 * (inline open, off-loop fill, passthrough) converges, so the negative
200 * memo (T13) sees every 404 regardless of which path produced it.
201 * Non-static: handler.c registers it as the request's pool cleanup. */
202void
203cvmfs_finalize_observe(void *data)
204{
205 ngx_http_request_t *r = data;
206 ngx_http_brix_cvmfs_loc_conf_t *lcf =
207 ngx_http_get_module_loc_conf(r, ngx_http_brix_cvmfs_module);
208 ngx_http_brix_cvmfs_ctx_t *ctx =
209 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
210 ngx_uint_t status =
211 brix_http_effective_status(r, NGX_OK);
212
213 cvmfs_finalize_sesslog(r, lcf, ctx, status);
214
215 if (lcf != NULL) {
216 brix_cvmfs_notify_status(r, lcf, status);
217 }
218
219 cvmfs_finalize_trace(r, lcf, ctx, status);
220
221 if (ctx == NULL) {
222 return;
223 }
224 cvmfs_finalize_metrics(r, ctx, status);
225
226 if (lcf != NULL) {
227 brix_cvmfs_attest_observe(r, lcf, ctx, status); /* G15 */
228 }
229}
