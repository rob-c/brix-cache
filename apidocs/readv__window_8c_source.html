1/*
2 * readv_window.c — bounded-resident kXR_readv response streaming.
3 *
4 * The protocol's logical body is [16-byte segment header][segment bytes] for
5 * every returned range. Stock XrdCl matches each returned header against the
6 * exact requested offset and length, so neither a range nor its header may be
7 * split into extra protocol records. We instead advertise the complete body
8 * length once, then produce its bytes through a reusable bounded window.
9 */
10#include "read.h"
11#include "protocols/root/connection/budget.h"
12#include "protocols/root/connection/write_helpers.h"
13#include "protocols/root/protocol/readv_seg.h"

17{

21/* Select the indexed segment as the shared window pump's current file/range. */
22static void
23readv_window_select(brix_ctx_t *ctx, size_t length)
24{
25 readahead_list *wire = readv_window_wire(ctx);
26 readahead_list *seg = &wire[ctx->rd.win_readv_index];
27 int idx = (int) (unsigned char) seg->fhandle[0];
28
29 ctx->rd.win_idx = idx;
30 ctx->rd.win_fd = ctx->files[idx].fd;
31 ctx->rd.win_offset = (off_t) (int64_t)
32 be64toh((uint64_t) seg->offset);
33 ctx->rd.win_remaining = length;
34}

36void
37brix_readv_window_sizes(brix_ctx_t *ctx, size_t *want,
38 size_t *scratch_need)
39{
40 size_t prefix = ctx->rd.win_readv_seg_started
41 ? 0 : BRIX_READV_SEGSIZE;
42 size_t capacity = (size_t) BRIX_READ_WINDOW - prefix;
43
44 *want = ctx->rd.win_remaining < capacity
45 ? ctx->rd.win_remaining : capacity;
46 *scratch_need = prefix + *want;
47}

49u_char *
50brix_readv_window_payload(brix_ctx_t *ctx, u_char *scratch)
51{
52 readahead_list *wire;
53 uint32_t length_be;
54 uint64_t offset_be;
55 size_t want;
56 size_t scratch_need;
57
58 brix_readv_window_sizes(ctx, &want, &scratch_need);
59 if (ctx->rd.win_readv_seg_started) {
60 return scratch;
61 }
62
63 wire = readv_window_wire(ctx);
64 ngx_memcpy(scratch, wire[ctx->rd.win_readv_index].fhandle, 4);
65 length_be = htonl((uint32_t) ctx->rd.win_remaining);
66 offset_be = htobe64((uint64_t) (int64_t) ctx->rd.win_offset);
67 ngx_memcpy(scratch + 4, &length_be, 4);
68 ngx_memcpy(scratch + 8, &offset_be, 8);
69 return scratch + BRIX_READV_SEGSIZE;
70}

92static ngx_int_t
93readv_window_fail(brix_ctx_t *ctx, ngx_connection_t *c, const char *message)
94{
95 ngx_flag_t started = ctx->rd.win_readv_started;
96
97 ctx->rd.win_active = 0;
98 ctx->rd.win_readv = 0;
99 ctx->rd.win_readv_started = 0;
100 ctx->rd.win_readv_seg_started = 0;
101 ctx->rd.win_readv_wire = NULL;
102 ctx->state = XRD_ST_REQ_HEADER;
103 ctx->recv.hdr_pos = 0;
104 BRIX_OP_ERR(ctx, BRIX_OP_READV);
105 if (started) {
106 /* A response header has already promised the remaining raw body. A
107 * framed error here would be consumed as file data and desynchronise
108 * the connection, so force the normal disconnect funnel instead. */
109 ngx_log_error(NGX_LOG_ERR, c->log, 0,
110 "brix: readv body stream aborted: %s", message);
111 c->close = 1;
112 return NGX_ERROR;
113 }
114 return brix_send_error_sid(ctx, c, ctx->rd.win_streamid, kXR_IOError,
115 message);
116}

118static ngx_flag_t
119readv_window_advance(brix_ctx_t *ctx)
120{
121 ctx->rd.win_readv_index++;
122 if (ctx->rd.win_readv_index >= ctx->rd.win_readv_count) {
123 return 0;
124 }
125
126 return 1;
127}

129ngx_int_t
130brix_readv_window_emit(brix_ctx_t *ctx, ngx_connection_t *c,
131 ssize_t nread, int io_errno)
132{
133 ngx_stream_session_t *session = c->data;
134 ngx_stream_brix_srv_conf_t *rconf;
135 ngx_chain_t *chain;
136 size_t prefix;
137 size_t want;
138 size_t got;
139 ngx_flag_t first;
140
141 rconf = ngx_stream_get_module_srv_conf(session, ngx_stream_brix_module);
142 brix_readv_window_sizes(ctx, &want, &prefix);
143 prefix -= want;
144
145 if (nread < 0 || (size_t) nread != want) {
146 const char *message = nread < 0 && io_errno != 0
147 ? strerror(io_errno) : "readv past EOF";
148 return readv_window_fail(ctx, c, message);
149 }
150
151 got = (size_t) nread;
152 ctx->files[ctx->rd.win_idx].bytes_read += got;
153 ctx->totals.bytes += got;
154 ctx->rd.win_readv_total += got;
155 ctx->rd.win_offset += (off_t) got;
156 ctx->rd.win_remaining -= got;
157 first = !ctx->rd.win_readv_started;
158 ctx->rd.win_readv_seg_started = 1;
159
160 if (ctx->rd.win_remaining == 0) {
161 size_t segment_cap = rconf->readv_segment_size;
162
163 if (readv_window_advance(ctx)) {
164 /* Decode and clamp the next segment now; the next pump iteration
165 * can therefore select a different handle safely. */
166 size_t length;
167 readahead_list *wire = readv_window_wire(ctx);
168
169 length = (size_t) ntohl(
170 (uint32_t) wire[ctx->rd.win_readv_index].rlen);
171 if (length > segment_cap) {
172 length = segment_cap;
173 }
174 readv_window_select(ctx, length);
175 ctx->rd.win_readv_seg_started = 0;
176 }
177 }
178
179 chain = brix_build_body_fragment_chain(ctx, c, ctx->rd.read_scratch,
180 prefix + got,
181 ctx->rd.win_readv_body_size,
182 first);
183 if (chain == NULL) {
184 return readv_window_fail(ctx, c, "readv response framing failed");
185 }
186 ctx->rd.win_readv_started = 1;
187 if (ctx->rd.win_readv_index >= ctx->rd.win_readv_count) {
188 readv_window_finish(ctx, c, rconf);
189 }
190
191 ctx->state = XRD_ST_REQ_HEADER;
192 ctx->recv.hdr_pos = 0;
193 if (first) {
194 brix_queue_response_chain(ctx, c, chain, ctx->rd.read_scratch);
195 } else {
196 brix_queue_response_fragment_chain(ctx, c, chain,
197 ctx->rd.read_scratch);
198 }
199 if (ctx->state != XRD_ST_SENDING) {
200 brix_release_read_buffer(ctx, c, ctx->rd.read_scratch);
201 }
202 return NGX_OK;
203}

206brix_readv_serve_windowed(brix_ctx_t *ctx, ngx_connection_t *c,
