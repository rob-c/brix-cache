1/* handler.c — the cvmfs:// content handler.
2 *
3 * WHAT: entry point for every request on an brix_cvmfs-enabled location:
4 * ctx setup → gate (method/class/reject/geo) → cache-tier open-or-fill
5 * → file response.
6 * WHY: cvmfs:// is a dedicated protocol — this handler owns the request
7 * end-to-end; nothing routes through the WebDAV dispatch.
8 * HOW: everything below the protocol seam is shared machinery: the serve
9 * step is the same offload → coalesced-fill → brix_vfs_open →
10 * brix_http_serve_file_ranged composition the WebDAV GET path

12 * helpers. The proxy-mode (T14) per-upstream override rides on the

38/* CVMFS-flavoured errno → HTTP status: origin-side trouble is a GATEWAY
39 * error (convention #3 — the origin transfer was bad, not the client
40 * request), everything else follows the shared table. */
41static ngx_uint_t
42cvmfs_errno_status(int err)
43{
44 if (err == ENOENT || err == ENOTDIR || err == ENAMETOOLONG) {
45 return NGX_HTTP_NOT_FOUND;
46 }
47 if (err == EACCES || err == EPERM || err == EXDEV || err == ELOOP) {
48 return NGX_HTTP_FORBIDDEN;
49 }
50 if (err == EIO) {
51 return NGX_HTTP_BAD_GATEWAY;
52 }
53 return brix_http_errno_to_status(err);
54}

56/* Re-entry trampoline for the off-loop cache fill: after the fill lands the

66/* Fill-failure interceptor (phase-87 G16): a definitive origin 404 while a
67 * virtual-repo request was parked on a fill advances to the next member and
68 * re-runs the whole handler (same trampoline contract as cvmfs_reenter).
69 * Every other failure — and a 404 on a direct or last-member request —
70 * keeps its status. */
71static ngx_int_t
72cvmfs_fill_fail(ngx_http_request_t *r, void *data, ngx_int_t status)
73{
74 (void) data;
75 if (status == NGX_HTTP_NOT_FOUND
76 && brix_cvmfs_virtual_advance(r) == NGX_OK)
77 {
78 return ngx_http_brix_cvmfs_handler(r);
79 }
80 return status;

100/* WHAT: build the on-disk fs path (export root + request uri) into `path`.
101 * WHY: a "/" root is the pure-cache-node anchor that contributes nothing, so
102 * the path IS the uri; every other root prefixes it. One place keeps the
103 * two overflow checks and the anchor special-case together.
104 * HOW: returns NGX_OK, or NGX_HTTP_REQUEST_URI_TOO_LARGE if the join would
105 * overflow the caller's PATH_MAX buffer. */
106static ngx_int_t
107cvmfs_build_fs_path(ngx_http_request_t *r, const char *root,
108 char *path, size_t path_size)
109{
110 if (root[0] == '/' && root[1] == '\0') {
111 if (brix_str_cbuf(path, path_size, &r->uri) == NULL) {
112 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
113 }
114 return NGX_OK;
115 }
116
117 {
118 size_t rn = ngx_strlen(root);
119
120 if (rn + r->uri.len >= path_size) {
121 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
122 }
123 ngx_memcpy(path, root, rn);
124 ngx_memcpy(path + rn, r->uri.data, r->uri.len);
125 path[rn + r->uri.len] = '\0';
126 }
127 return NGX_OK;
128}

130/* WHAT: fill a shared serve-opts struct with the fixed cvmfs GET descriptor
131 * (anonymous public read, weak etag) plus the per-location compress flag.
132 * WHY: both serve sites (offload and ranged) need the identical options; one
133 * initializer keeps them in lockstep and the values frozen. The compress
134 * flag (phase-85 F12) is the shared `brix_compress` that WebDAV/S3 already
135 * thread into their serve opts — cvmfs simply never opted in, so cvmfs
136 * GETs never transcoded. Off by default => byte-frozen parity; on => the
137 * shared negotiate path serves the client's best Accept-Encoding codec.
138 * Integrity is untouched: F1's CAS verify runs at fill time against the
139 * stored object, and this outbound Content-Encoding is a transparent,
140 * reversible wire transform that never alters the bytes on disk.
141 * HOW: zeroes then sets the constant fields and the caller's compress flag. */
142static void
143cvmfs_serve_opts_init(brix_http_serve_opts_t *opts, ngx_flag_t compress)
144{
145 ngx_memzero(opts, sizeof(*opts));
146 opts->xfer_proto = BRIX_XFER_PROTO_CVMFS;
147 opts->op_name = "GET";
148 opts->identity = "anonymous";
149 opts->etag_flags = BRIX_ETAG_WEAK;
150 opts->compress = compress;
151}

153/* WHAT: run the two off-loop steps of a tier read — socket-wire serve, then a
154 * coalesced miss-fill — and report whether the request is already
155 * resolved.
156 * WHY: socket-wire backends can't open/read on the event loop, so a serve
157 * that dispatches (NGX_DONE) or a fill that parks (NGX_DONE) both end
158 * the current call; only a fall-through means the bytes are now local
159 * and the caller should open + respond. Stale-serve semantics and the
160 * coalescing hold are owned by the shared fill helper, untouched here.
161 * HOW: returns NGX_DECLINED when the caller must proceed to open; otherwise
162 * the terminal rc (NGX_DONE, or an HTTP error status) to return as-is.
163 * On a parked fill it records the FILL disposition, which surfaces as
164 * $brix_cache_status=MISS. */
165static ngx_int_t
166cvmfs_tier_serve_or_fill(ngx_http_request_t *r,
167 ngx_http_brix_cvmfs_loc_conf_t *lcf, brix_sd_instance_t *sd,
168 const char *key, const char *path)
169{
170 ngx_http_brix_cvmfs_ctx_t *ctx =
171 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
172 brix_http_serve_opts_t sopts;
173 ngx_int_t rc;
174
175 /* socket-wire backends can't open/read on the event loop */
176 cvmfs_serve_opts_init(&sopts, lcf->common.compress);
177
178 /* cvmfs is a transparent, anonymous public cache (no per-user
179 * identity or backend credential concept applies) - vctx passed as
180 * NULL skips the phase-2 credential gate entirely, unchanged from
181 * before that gate existed. */
182 rc = brix_http_serve_offload_remote(r, sd, key, path, &sopts,
183 &lcf->common, NULL, NULL);
184 if (rc == NGX_DONE) {
185 return NGX_DONE;
186 }
187 if (rc == NGX_ERROR) {
188 return NGX_HTTP_INTERNAL_SERVER_ERROR;
189 }
190
191 /* remote miss → the fill would hit the origin; charge the requester's
192 * QoS class first (phase-85 F9). Local hits never reach this gate. */
193 if (lcf->qos != NULL && brix_sd_cache_fill_needs_offload(sd, key)) {
194 rc = brix_cvmfs_qos_check(r, lcf);
195 if (rc != NGX_DECLINED) {
196 return rc; /* 429 - fill budget exhausted */
197 }
198 }
199
200 /* miss → one coalesced off-loop fill, re-entering this handler on land */
201 rc = brix_http_cache_fill_if_needed(r, sd, key, &lcf->common, NULL,
202 cvmfs_reenter, NULL,
203 cvmfs_fill_fail);
204 if (rc == NGX_DONE) {
205 if (ctx != NULL) {
206 ctx->cache_status = BRIX_CVMFS_CACHE_FILL; /* → MISS */
207 }
208 return NGX_DONE;
209 }
210 if (rc == NGX_ERROR) {
211 return NGX_HTTP_INTERNAL_SERVER_ERROR;
212 }
213
214 return NGX_DECLINED;
215}

217/* WHAT: open the now-local object, run stat + conditional checks, and serve it
218 * through the shared ranged-file pipeline.
219 * WHY: this is the HIT tail of a tier read (bytes are on disk, whether they
220 * were always there or a fill just landed). Marking HIT here — only when
221 * the disposition is still NONE — preserves a FILL disposition set by a
222 * just-completed fill on the re-entry pass.
223 * HOW: early-return on every open/stat/conditional failure with the mapped
224 * status; closes the handle on each exit that doesn't hand it to the
225 * serve pipeline. */
226static ngx_int_t
227cvmfs_tier_open_respond(ngx_http_request_t *r,
228 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
229 brix_vfs_ctx_t *vctx, const char *path)
230{
231 brix_vfs_file_t *fh;
232 brix_vfs_stat_t vst;
233 int vfs_err = 0;
234 ngx_int_t rc;
235
236 fh = brix_vfs_open(vctx, BRIX_VFS_O_READ, &vfs_err);
237 if (fh == NULL) {
238 return (ngx_int_t) cvmfs_errno_status(vfs_err);
239 }
240 if (ctx != NULL && ctx->cache_status == BRIX_CVMFS_CACHE_NONE) {
241 ctx->cache_status = BRIX_CVMFS_CACHE_HIT; /* → HIT */
242 }
243 if (brix_vfs_file_stat(fh, &vst) != NGX_OK) {
244 brix_vfs_close(fh, r->connection->log);
245 return NGX_HTTP_INTERNAL_SERVER_ERROR;
246 }
247 if (vst.is_directory) {
248 brix_vfs_close(fh, r->connection->log);
249 return NGX_HTTP_FORBIDDEN;
250 }
251
252 rc = brix_http_check_if_modified_since(r, vst.mtime);
253 if (rc == NGX_HTTP_NOT_MODIFIED) {
254 brix_vfs_close(fh, r->connection->log);
255 r->headers_out.status = NGX_HTTP_NOT_MODIFIED;
256 r->headers_out.content_length_n = 0;
257 /* RFC 9110 §15.4.5: a 304 must carry the same validators the 200 would
258 * (the serve pipeline's weak ETag + Last-Modified from vst), else a
259 * cache cannot update its stored representation's metadata. Mirror the
260 * BRIX_ETAG_WEAK path brix_http_set_file_headers takes on the 200. */
261 r->headers_out.last_modified_time = vst.mtime;
262 (void) brix_http_add_etag_header(r, vst.mtime, vst.size,
263 BRIX_ETAG_WEAK, 1);
264 ngx_http_send_header(r);
265 return ngx_http_send_special(r, NGX_HTTP_LAST);
266 }
267 if (rc != NGX_OK) {
268 brix_vfs_close(fh, r->connection->log);
269 return rc;
270 }
271
272 /* phase-87 G10: a CAS GET naming a resident delta base wins over the
273 * dict coding (a same-object base beats any trained dictionary);
274 * NGX_DECLINED leaves fh open and falls through. */
275 rc = brix_cvmfs_delta_try_serve(r, lcf, ctx, fh);
276 if (rc != NGX_DECLINED) {
277 return rc;
278 }
279
280 /* phase-87 G3: a CAS GET opting in with a matching X-Brix-Dict header
281 * may be answered dict-coded; NGX_DECLINED leaves fh open and falls
282 * through to the identity ranged serve. */
283 rc = brix_cvmfs_dict_try_serve(r, lcf, ctx, fh);
284 if (rc != NGX_DECLINED) {
285 return rc;
286 }
287
288 r->allow_ranges = 1;
289
290 {
291 brix_http_serve_opts_t opts;
292 brix_http_serve_result_t result;
293
294 cvmfs_serve_opts_init(&opts, lcf->common.compress);
295 rc = brix_http_serve_file_ranged(r, fh, &vst, path, &opts, &result);
296 }
297 return rc;
298}

300/* Serve a classified CAS GET/HEAD from the tier: offload a socket-wire
301 * serve, coalesce a miss into one off-loop fill, then open + respond via
302 * the shared ranged-file pipeline. */
303static ngx_int_t
304cvmfs_tier_get(ngx_http_request_t *r, ngx_http_brix_cvmfs_loc_conf_t *lcf)
305{
306 ngx_http_brix_cvmfs_ctx_t *ctx =
307 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
308 char path[PATH_MAX];
309 const char *root;
310 char key[PATH_MAX];
311 brix_vfs_ctx_t vctx;
312 int is_tls = 0;
313 ngx_int_t rc;
314
315 root = (ctx != NULL && ctx->up_root != NULL) ? ctx->up_root
316 : lcf->common.root_canon;
317
318 /* fs path = export root + uri */
319 rc = cvmfs_build_fs_path(r, root, path, sizeof(path));
320 if (rc != NGX_OK) {
321 return rc;
322 }
323
324 /* phase-101 W5.2c does NOT gate cvmfs on native brix_authdb: in the common
325 * read-through/proxy mode `root` is the upstream Stratum-1 URL, so the built
326 * path is not a local realpath and cannot match rules finalized against the
327 * local export root. cvmfs objects are content-addressed (hashes), so a
328 * path-prefix ACL is the wrong model here — cvmfs authz belongs on the peer/
329 * URI, a separate design. (s3/webdav/root:// enforce it; see the phase doc.) */
330
331 is_tls = brix_http_request_is_tls(r);
332 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log,
333 BRIX_PROTO_CVMFS, root, "", BRIX_VFS_MUTATION_READ_ONLY, is_tls, NULL,
334 path);
335 brix_http_vfs_bind_no_rules(&lcf->common, &vctx);
336 vctx.sd = cvmfs_resolve_sd(r, lcf);
337 if (vctx.sd == NULL) {
338 ngx_log_error(NGX_LOG_ERR, r->connection->log, 0,
339 "cvmfs: no storage backend registered for \"%s\" - check "
340 "brix_storage_backend", root);
341 return NGX_HTTP_INTERNAL_SERVER_ERROR;
342 }
343 if (brix_path_resolved_to_pfn(&vctx, path, key, sizeof(key)) != NGX_OK) {
344 return NGX_HTTP_INTERNAL_SERVER_ERROR;
345 }
346
347 /* phase-87 G11: passively learn this connection's CAS access sequence
348 * and prewarm the predicted successors (advisory — never touches this
349 * request's own serve path). */
350 brix_cvmfs_learn_note(r, lcf, ctx, vctx.sd, key);
351
352 /* Dashboard live-transfer record from TIER ENTRY, not serve time: the
353 * record then spans a cold fill's whole origin wait, so in-flight cvmfs
354 * fills are visible in the transfer table (the serve pipeline's own
355 * start is idempotent and rebinds this same slot; rejects never got
356 * here, so they never occupy one). Auto-freed with the request pool.
357 * `key` is the client-visible /cvmfs/... path — the synthetic proxy-mode
358 * registry root must not leak into the operator display. */
359 (void) brix_dashboard_http_start_identity(r, key, "anonymous", "",
360 BRIX_XFER_PROTO_CVMFS, BRIX_XFER_DIR_READ, "GET", -1);
361
362 rc = cvmfs_tier_serve_or_fill(r, lcf, vctx.sd, key, path);
363 if (rc != NGX_DECLINED) {
364 return rc;
365 }
366
367 return cvmfs_tier_open_respond(r, lcf, ctx, &vctx, path);
368}

384 if (ctx == NULL) {

386

416 }
417 }
418

423ngx_http_brix_cvmfs_handler(ngx_http_request_t *r)
424{
425 ngx_http_brix_cvmfs_loc_conf_t *lcf =
426 ngx_http_get_module_loc_conf(r, ngx_http_brix_cvmfs_module);
427 ngx_int_t rc;
428
429 /* phase-105 W1: per-client-IP [brix_rate_limit] gate, before any work
430 * (mirrors webdav/access.c + s3/handler.c — reject cheap first). The
431 * directive parsed here pre-105 but was silently inert (webdav-owned). */
432 if (lcf->common.rate_limit.kv != NULL && r->connection != NULL) {
433 ngx_str_t *ip = &r->connection->addr_text;
434
435 if (brix_rate_limit_check(&lcf->common.rate_limit,
436 (const char *) ip->data, ip->len)
437 != NGX_OK)
438 {
439 return NGX_HTTP_TOO_MANY_REQUESTS;
440 }
441 }
442
443 if (cvmfs_handler_ctx(r, lcf) == NULL) {
444 return NGX_HTTP_INTERNAL_SERVER_ERROR;
445 }
446
447 /* GET/HEAD-only protocol — except the phase-87 G2 bundle POST, whose
448 * want-list body the gate's bundle handler reads itself. A POST that
449 * turns out not to be a valid bundle request is rejected by the gate
450 * with the body unread; nginx's special-response path discards it. */
451 if (!(r->method == NGX_HTTP_POST && lcf->cvmfs.bundle)) {
452 rc = ngx_http_discard_request_body(r);
453 if (rc != NGX_OK) {
454 return rc;
455 }
456 }
457
458 if (lcf->scvmfs) {
459 rc = brix_scvmfs_preamble(r, lcf); /* T22: TLS + authz */
460 if (rc != NGX_DECLINED) {
461 return rc;
462 }
463 }
464
465 for ( ;; ) {
466 rc = brix_cvmfs_gate(r, lcf); /* classify + police */
467 if (rc == NGX_DECLINED) {
468 rc = cvmfs_tier_get(r, lcf);
469 }
470 /* phase-87 G16: a definitive 404 on a virtual-repo request tries
471 * the next member (declaration-order precedence, bounded by the
472 * member count); any other answer — including a member's 401/403,
473 * which must never be papered over by a sibling — is final. */
474 if (rc == NGX_HTTP_NOT_FOUND
475 && brix_cvmfs_virtual_advance(r) == NGX_OK)
476 {
477 continue;
478 }
479 return rc; /* reject status, passthrough NGX_DONE … */
480 }
