1/*
2 * connect_upstream.c — upstream TCP connect, XRootD bootstrap buffer,
3 * TLS handshake callback, and lazy-connect lifecycle.
4 *
5 * WHAT: Builds the 68-byte bootstrap payload (client hello + kXR_protocol
6 * + kXR_login), resolves the upstream through the brix DNS driver,
7 * creates a non-blocking socket, connects to an upstream XRootD server,
8 * optionally performs TLS, then transitions into the bootstrap-read
9 * state.
10 *
11 * WHY: The proxy lazily opens backend connections on the first post-login opcode
12 * rather than at client login. This avoids idle upstream sockets during
13 * periods of low traffic and lets per-request auth decisions determine
14 * whether a backend connection is worth keeping.
15 *
16 * HOW: brix_proxy_connect() selects an endpoint from pool/rr/single and starts
17 * an async brix_dns_resolve() under the server block's brix_resolver
18 * policy (phase-116: never getaddrinfo on the event loop). The answer
19 * arrives inline for a literal or a cache hit, otherwise later on the
20 * event loop; either way pc_dns_done() creates the non-blocking socket,
21 * arms the write-event for the async connect, optionally performs TLS,
22 * builds the bootstrap buffer, then sets state to XRD_PX_BOOTSTRAP so
23 * read_handler parses responses. A resolve that fails after
24 * brix_proxy_connect() returned goes through brix_proxy_abort() exactly
25 * like a failed async connect().
26 */

42/*
43 * WHAT: Creates a non-blocking SOCK_STREAM socket for one resolved answer and
44 * records the answer's sockaddr in tgt->addr/addrlen.
45 *
46 * WHY: The resolver returns every family the policy allows; the first family
47 * this host can open a socket for is the one the async connect() uses.
48 *
49 * HOW: ngx_socket + ngx_nonblocking; returns the fd, or NGX_INVALID_FILE with
50 * nothing left open so the caller tries the next answer.
51 */
52static int
53pc_socket_for(const brix_dns_addr_t *answer, brix_proxy_target_t *tgt)
54{
55 int fd = ngx_socket(answer->ss.ss_family, SOCK_STREAM, 0);
56
57 if (fd == (int) NGX_INVALID_FILE) {
58 return fd;
59 }
60 if (ngx_nonblocking(fd) == NGX_ERROR) {
61 ngx_close_socket(fd);
62 return (int) NGX_INVALID_FILE;
63 }
64 ngx_memcpy(&tgt->addr, &answer->ss, answer->len);
65 tgt->addrlen = answer->len;
66 return fd;
67}

71/*
72 * WHAT: Completion handler of the upstream resolve started by
73 * pc_start_resolve(); runs on the event loop, inline or later.
74 *
75 * WHY: brix_proxy_connect() must report an inline outcome to its caller
76 * (which relays a redirect or fails the request itself), while a later
77 * outcome has no caller left to report to and takes the same path as a
78 * failed async connect(): brix_proxy_abort().
79 *
80 * HOW: Clears dns_inflight FIRST so the cleanup that a failed socket phase
81 * triggers does not cancel a request that already completed, drives the
82 * socket phases, then routes the result by dns_inline.
83 */
84static void
85pc_dns_done(brix_dns_req_t *req)
86{
87 brix_proxy_ctx_t *proxy = req->data;
88 ngx_int_t rc;
89
90 proxy->dns_inflight = 0;
91 rc = pc_connect_resolved(proxy);
92 if (proxy->dns_inline) {
93 proxy->dns_rc = rc;
94 return;
95 }
96 if (rc != NGX_OK) {
97 brix_proxy_abort(proxy, "proxy: upstream resolve failed");
98 }
99}

101/*
102 * WHAT: Starts resolving tgt->host:port through the brix DNS driver under the
103 * server block's policy (phase-116 I-DNS-1: nothing on the event loop
104 * calls getaddrinfo).
105 *
106 * WHY: Endpoint selection yields a host string; the async connect() needs a
107 * concrete address, and a name whose answer is not cached must not stall
108 * every other session on this worker while the resolver answers. AF
109 * selection is BRIX_AF_AUTO (every family) — the proxy does not
110 * constrain the upstream family.
111 *
112 * HOW: Fills proxy->dns_req (the name is borrowed from conf / the redirect /
113 * the pin, all of which outlive the request because cleanup cancels it)
114 * and calls brix_dns_resolve(). dns_inline is set across the call so a
115 * handler that runs before it returns stores its outcome in dns_rc
116 * instead of aborting; NGX_AGAIN in dns_rc afterwards means the answer
117 * is still in flight and the connect completes (or aborts) later.
118 * Returns NGX_OK (in flight or connected), else NGX_ERROR with nothing
119 * left open.
120 */
121static ngx_int_t
122pc_start_resolve(brix_proxy_ctx_t *proxy, ngx_connection_t *client_conn,
123 ngx_stream_brix_srv_conf_t *conf, const brix_proxy_target_t *tgt)
124{
125 brix_dns_req_t *req = &proxy->dns_req;
126
127 brix_proxy_cleanup_dns(proxy);
128 ngx_memzero(req, sizeof(*req));
129 req->name = *tgt->host;
130 req->port = (in_port_t) tgt->port;
131 req->af = BRIX_AF_AUTO;
132 req->socktype = SOCK_STREAM;
133 req->policy = conf->common.dns.policy;
134 req->log = client_conn->log;
135 req->handler = pc_dns_done;
136 req->data = proxy;
137
138 proxy->dns_rc = NGX_AGAIN;
139 proxy->dns_inflight = 1;
140 proxy->dns_inline = 1;
141
142 if (brix_dns_resolve(req) != NGX_OK) {
143 proxy->dns_inflight = 0;
144 proxy->dns_inline = 0;
145 BRIX_LOG(NGX_LOG_ERR, client_conn->log, 0,
146 "xrootd proxy: cannot start resolving \"%V\"",
147 tgt->host);
148 return NGX_ERROR;
149 }
150 proxy->dns_inline = 0;
151
152 return proxy->dns_rc == NGX_AGAIN ? NGX_OK : proxy->dns_rc;
153}

155/*
156 * WHAT: Resolves the upstream login username from the configured policy into
157 * user_out (a >= 9-byte buffer).
158 *
159 * WHY: The kXR_login bootstrap request carries a username; policy decides
160 * whether it passes through the client's login name, uses a fixed
161 * configured name, or stays anonymous (empty → "xrd" default downstream).
162 *
163 * HOW: Switches on conf->proxy.login_user; ngx_cpystrn truncates to the buffer
164 * size. Passthrough with no client username falls back to empty.
165 */
166static void
167pc_resolve_login_user(brix_proxy_ctx_t *proxy,
168 ngx_stream_brix_srv_conf_t *conf, char *user_out)
169{
170 switch (conf->proxy.login_user) {
171 case BRIX_PROXY_LOGIN_PASSTHROUGH:
172 if (proxy->client_ctx != NULL
173 && proxy->client_ctx->login.user[0] != '\0')
174 {
175 ngx_cpystrn((u_char *) user_out,
176 (u_char *) proxy->client_ctx->login.user, 9);
177 } else {
178 user_out[0] = '\0'; /* no client username → fall back to "xrd" */
179 }
180 break;
181 case BRIX_PROXY_LOGIN_FIXED:
182 ngx_cpystrn((u_char *) user_out,
183 (u_char *) conf->proxy.login_user_name, 9);
184 break;
185 default: /* BRIX_PROXY_LOGIN_ANONYMOUS */
186 user_out[0] = '\0';
187 break;
188 }
189}

191/*
192 * WHAT: Wraps a connected-ready fd in an nginx connection with its own pool,
193 * builds the bootstrap buffer, and wires up the proxy send/recv handlers.
194 *
195 * WHY: Every upstream connection needs a dedicated pool (for the bootstrap
196 * frames), the standard ngx send/recv vtable, and the proxy read/write
197 * handlers before connect() is armed.
198 *
199 * HOW: ngx_get_connection + ngx_create_pool + ngx_palloc for the 68-byte
200 * bootstrap; fills it via pc_resolve_login_user + brix_proxy_build_bootstrap;
201 * points proxy->wbuf at the pool-owned buffer (wbuf_owned=0). Returns the
202 * connection, or NULL after closing the fd / freeing partial state.
203 */
204static ngx_connection_t *
205pc_open_socket(brix_proxy_ctx_t *proxy, ngx_connection_t *client_conn,
206 ngx_stream_brix_srv_conf_t *conf, int fd)
207{
208 ngx_connection_t *uconn;
209 u_char *bsbuf;
210 char upstream_user[9] = { 0 };
211
212 uconn = ngx_get_connection(fd, client_conn->log);
213 if (uconn == NULL) {
214 ngx_close_socket(fd);
215 return NULL;
216 }
217
218 uconn->pool = ngx_create_pool(BRIX_PROXY_POOL_SIZE, client_conn->log);
219 if (uconn->pool == NULL) {
220 ngx_free_connection(uconn);
221 ngx_close_socket(fd);
222 return NULL;
223 }
224
225 /* Build bootstrap buffer in the upstream connection pool */
226 bsbuf = ngx_palloc(uconn->pool,
227 XRD_HANDSHAKE_LEN
228 + sizeof(ClientProtocolRequest)
229 + sizeof(ClientLoginRequest));
230 if (bsbuf == NULL) {
231 ngx_destroy_pool(uconn->pool);
232 ngx_free_connection(uconn);
233 ngx_close_socket(fd);
234 return NULL;
235 }
236
237 pc_resolve_login_user(proxy, conf, upstream_user);
238
239 uconn->data = proxy;
240 uconn->recv = ngx_recv;
241 uconn->send = ngx_send;
242 uconn->recv_chain = ngx_recv_chain;
243 uconn->send_chain = ngx_send_chain;
244 uconn->log = client_conn->log;
245 uconn->read->handler = brix_proxy_read_handler;
246 uconn->write->handler = brix_proxy_write_handler;
247 uconn->read->log = client_conn->log;
248 uconn->write->log = client_conn->log;
249
250 proxy->conn = uconn;
251 proxy->wbuf = bsbuf;
252 proxy->wbuf_len = brix_proxy_build_bootstrap(bsbuf, upstream_user);
253 proxy->wbuf_pos = 0;
254 proxy->wbuf_owned = 0; /* Phase 39: bsbuf is pool-allocated — pool owns it */
255 proxy->state = XRD_PX_CONNECTING;
256
257 return uconn;
258}

261/*
262 * WHAT: Starts the upstream TLS handshake on an already-connected socket.
263 *
264 * WHY: On an immediate (rc==0) connect to a TLS upstream we begin the client
265 * TLS handshake right away; the async callback finishes the bootstrap.
266 *
267 * HOW: ngx_ssl_create_connection + SNI (explicit name directive, else host) +
268 * ngx_ssl_handshake. NGX_AGAIN leaves the callback to continue; a synchronous
269 * completion invokes brix_proxy_tls_handshake_done() inline. Returns NGX_OK
270 * on success (connect continues via callback/inline), NGX_ERROR after
271 * cleanup on failure.
272 */
273static ngx_int_t
274pc_start_tls(brix_proxy_ctx_t *proxy, ngx_stream_brix_srv_conf_t *conf,
275 ngx_connection_t *uconn)
276{
277 const char *sni;
278
279 if (ngx_ssl_create_connection(conf->proxy.tls_ctx, uconn,
280 NGX_SSL_BUFFER | NGX_SSL_CLIENT)
281 != NGX_OK)
282 {
283 BRIX_PROXY_METRIC_INC(proxy->client_ctx, upstream_connect_errors);
284 BRIX_PROXY_UP_INC(proxy, upstream_connect_errors);
285 brix_proxy_cleanup(proxy);
286 return NGX_ERROR;
287 }
288
289 /* SNI: prefer explicit name directive, fall back to configured host */
290 sni = (conf->proxy.upstream_tls_name.len > 0)
291 ? (const char *) conf->proxy.upstream_tls_name.data
292 : (const char *) conf->proxy.host.data;
293 SSL_set_tlsext_host_name(uconn->ssl->connection, sni);
294
295 uconn->ssl->handler = brix_proxy_tls_handshake_done;
296 proxy->state = XRD_PX_TLS_HANDSHAKE;
297 if (ngx_ssl_handshake(uconn) == NGX_AGAIN) {
298 return NGX_OK; /* TLS callback will continue */
299 }
300 brix_proxy_tls_handshake_done(uconn);
301 return NGX_OK;
302}

305/*
306 * WHAT: Fires the async connect() on fd, arms the write event / connect timer,
307 * and — on an immediate connect — either starts TLS or flushes bootstrap.
308 *
309 * WHY: nginx drives outbound connects under the event loop: connect() returns
310 * EINPROGRESS and the write_handler completes it, unless the peer is local
311 * (rc==0) in which case we proceed to TLS or bootstrap flush immediately.
312 *
313 * HOW: connect() (EINPROGRESS is expected/OK), ngx_handle_write_event, optional
314 * connect timer; on rc==0 dispatches to pc_start_tls (TLS upstreams) or sets
315 * BOOTSTRAP state + brix_proxy_flush. Any failure cleans up. Returns NGX_OK
316 * or NGX_ERROR.
317 */
318static ngx_int_t
319pc_arm_events(brix_proxy_ctx_t *proxy, ngx_connection_t *client_conn,
320 ngx_stream_brix_srv_conf_t *conf, ngx_connection_t *uconn,
321 brix_proxy_target_t *tgt)
322{
323 ngx_int_t rc;
324
325 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, client_conn->log, 0,
326 "xrootd proxy: connect() to %s:%d",
327 tgt->host->data, (int) tgt->port);
328
329 rc = connect(tgt->fd, (struct sockaddr *)(void *) &tgt->addr,
330 tgt->addrlen);
331 if (rc == -1 && ngx_socket_errno != NGX_EINPROGRESS) {
332 BRIX_LOG(NGX_LOG_ERR, client_conn->log, ngx_socket_errno,
333 "xrootd proxy: connect to %s:%d failed",
334 tgt->host->data, (int) tgt->port);
335 BRIX_PROXY_METRIC_INC(proxy->client_ctx, upstream_connect_errors);
336 BRIX_PROXY_UP_INC(proxy, upstream_connect_errors);
337 brix_proxy_cleanup(proxy);
338 return NGX_ERROR;
339 }
340
341 if (ngx_handle_write_event(uconn->write, 0) != NGX_OK) {
342 BRIX_PROXY_METRIC_INC(proxy->client_ctx, upstream_connect_errors);
343 BRIX_PROXY_UP_INC(proxy, upstream_connect_errors);
344 brix_proxy_cleanup(proxy);
345 return NGX_ERROR;
346 }
347
348 /* Arm connect timeout: fires in write_handler as wev->timedout */
349 if (rc == -1 && conf->proxy.connect_timeout > 0) {
350 ngx_add_timer(uconn->write, conf->proxy.connect_timeout);
351 }
352
353 if (rc == 0) {
354 /* Immediate connect (local loopback etc.) */
355#if (NGX_SSL)
356 if (conf->proxy.upstream_tls && conf->proxy.tls_ctx != NULL) {
357 return pc_start_tls(proxy, conf, uconn);
358 }
359#endif
360 proxy->state = XRD_PX_BOOTSTRAP;
361 proxy->bs_phase = XRD_PX_BS_HANDSHAKE;
362 proxy->rhdr_pos = 0;
363
364 if (brix_proxy_flush(proxy) == NGX_ERROR) {
365 BRIX_PROXY_METRIC_INC(proxy->client_ctx, upstream_connect_errors);
366 BRIX_PROXY_UP_INC(proxy, upstream_connect_errors);
367 brix_proxy_cleanup(proxy);
368 return NGX_ERROR;
369 }
370 }
371
372 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, client_conn->log, 0,
373 "xrootd proxy: connecting to %s:%d",
374 tgt->host->data, (int) tgt->port);
375 return NGX_OK;
376}

378/*
379 * WHAT: Turns the finished resolve in proxy->dns_req into a non-blocking
380 * socket + chosen sockaddr and runs the socket phases (open + arm).
381 *
382 * WHY: The resolve completes on the event loop either inline or later; the
383 * socket phases are the same either way, so they live behind the
384 * completion handler rather than in brix_proxy_connect().
385 *
386 * HOW: A resolver failure logs the proxy's own message (nothing is open yet);
387 * the answers are tried in resolver order and the first family that
388 * yields a socket wins, none bumping the connect-error metrics. Then
389 * pc_open_socket + pc_arm_events as before (the latter cleans up on its
390 * own failures). Returns NGX_OK once the connect is in flight.
391 */
392static ngx_int_t
393pc_connect_resolved(brix_proxy_ctx_t *proxy)
394{
395 brix_dns_req_t *req = &proxy->dns_req;
396 ngx_connection_t *client_conn = proxy->client_conn;
397 ngx_stream_brix_srv_conf_t *conf = proxy->conf;
398 ngx_connection_t *uconn;
399 brix_proxy_target_t tgt;
400 ngx_uint_t i;
401
402 ngx_memzero(&tgt, sizeof(tgt));
403 tgt.host = &req->name;
404 tgt.port = (ngx_int_t) req->port;
405 tgt.fd = (int) NGX_INVALID_FILE;
406
407 if (req->rc != NGX_OK || req->naddrs == 0) {
408 BRIX_LOG(NGX_LOG_ERR, client_conn->log, 0,
409 "xrootd proxy: cannot resolve \"%V\": %s", &req->name,
410 req->error != NULL ? req->error : "no address");
411 return NGX_ERROR;
412 }
413
414 for (i = 0; i < req->naddrs && tgt.fd == (int) NGX_INVALID_FILE; i++) {
415 tgt.fd = pc_socket_for(&req->addrs[i], &tgt);
416 }
417 if (tgt.fd == (int) NGX_INVALID_FILE) {
418 BRIX_LOG(NGX_LOG_ERR, client_conn->log, 0,
419 "xrootd proxy: no usable address for \"%V\"",
420 &req->name);
421 BRIX_PROXY_METRIC_INC(proxy->client_ctx, upstream_connect_errors);
422 BRIX_PROXY_UP_INC(proxy, upstream_connect_errors);
423 return NGX_ERROR;
424 }
425
426 uconn = pc_open_socket(proxy, client_conn, conf, tgt.fd);
427 if (uconn == NULL) {
428 return NGX_ERROR;
429 }
430
431 return pc_arm_events(proxy, client_conn, conf, uconn, &tgt);
432}
