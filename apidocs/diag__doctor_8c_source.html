21/*
22 * WHAT: classify why the primary connection could not be set up.
23 * WHY: extracted from doctor_one's connect-failure path (complexity gate).
24 * HOW: map errno / status-message onto a fixed cause+remedy pair; the caller
25 * reports the classified cause, not st->msg (wire text may carry PII).
26 */
27static void
28doctor_classify_conn_error(const brix_status *st, const char **cause,
29 const char **remedy)
30{
31 *cause = "connection setup failed";
32 *remedy = "check the network path and that the server is running";
33 if (st->sys_errno == ECONNREFUSED) {
34 *cause = "no listener on host:port (service down or wrong port)";
35 *remedy = "start the gateway / verify the port and any firewall";
36 } else if (st->sys_errno == ETIMEDOUT || st->sys_errno == EHOSTUNREACH
37 || st->sys_errno == ENETUNREACH) {
38 *cause = "host/network unreachable (routing or firewall drop)";
39 *remedy = "check routing/firewall and that the host is up";
40 } else if (st->msg[0] != '\0' && strstr(st->msg, "resolve") != NULL) {
41 *cause = "DNS resolution failed";
42 *remedy = "check the hostname and DNS resolver";
43 }
44}

47/*
48 * WHAT: verdict-render half of a failed primary connect — auth rejection vs
49 * transport unreachability, plus the standalone auth-suite.
50 * WHY: extracted from doctor_one for the complexity gate.
51 * HOW: distinguish "server reachable but auth failed" from "couldn't reach
52 * it" by the client's error CODE: XRDC_EAUTH / kXR_NotAuthorized /
53 * kXR_AuthFailed mean auth was attempted and rejected; everything else
54 * is transport.
55 */
56static void
57doctor_one_connect_fail(const diag_args *a, const brix_url *u,
58 const brix_status *st, const char *target,
59 doctor_ep *e)
60{
61 if (st->kxr == XRDC_EAUTH || st->kxr == kXR_NotAuthorized
62 || st->kxr == kXR_AuthFailed) {
63 doc_issue(e, DOC_RED, "authentication failed");
64 dx_record(e, &(dx_note){ "auth", DX_FAIL, st->kxr,
65 "could not authenticate (credential rejected, or none usable for the server's auth)",
66 "check the credential's validity/scope and that it matches the server's auth mode" });
67 } else {
68 /* reachability: classify *why* the connection could not be set up. */
69 const char *cause, *remedy;
70 doctor_classify_conn_error(st, &cause, &remedy);
71 /* use the classified cause, not st.msg — wire text may carry PII. */
72 doc_issue(e, DOC_RED, "connect failed: %s", cause);
73 dx_record(e, &(dx_note){ "reachability", DX_FAIL, st->kxr, cause, remedy });
74 }
75 /* the auth-suite is self-contained (its own force_anon session) — run it
76 * even when our credential could not establish the primary connection. */
77 if (a->auth_suite) {
78 doctor_auth_suite(a, u, target, 0, e);
79 }
80}

84 * WHAT: session-fact collection over the freshly opened connection —
85 * transport timings, capabilities, TLS state, chosen auth.
86 * WHY: extracted from doctor_one for the complexity gate.
87 * HOW: read the facts off the live conn into *e and flag the
88 * no-silent-downgrade invariant (gotoTLS advertised but cleartext).

90static void
91doctor_one_session_facts(brix_conn *c, doctor_ep *e)
92{

94
95 /* network + transport facts */
96 brix_netdiag_facts(c, &e->nf);
97 e->caps = (unsigned) c->server_flags;

99 e->tls_active = brix_tls_info(c, &ver, &cipher);

102 snprintf(e->tls_cipher, sizeof(e->tls_cipher), "%s", cipher ? cipher : "?");

105 c->diag.chosen_auth ? c->diag.chosen_auth : "anon");

107 /* no-silent-downgrade: gotoTLS advertised but the session is cleartext */
108 if (e->gototls && !e->tls_active) {
109 doc_issue(e, DOC_RED, "gotoTLS advertised but session is cleartext");

168/* Interrogate ONE endpoint into *e. Bounded by the conn timeout (never hangs). */
169void
170doctor_one(const diag_args *a, const char *url, doctor_ep *e)
171{
172 brix_url u;
173 brix_conn c;
174 brix_status st;
175 char target[XRDC_PATH_MAX];
176 int have_target = 0;
177
178 target[0] = '\0';
179 memset(e, 0, sizeof(*e));
180 e->status = DOC_GREEN;
181 brix_status_clear(&st);
182 if (brix_endpoint_parse(url, &u, &st) != 0) {
183 snprintf(e->host, sizeof(e->host), "%s", url);
184 doc_issue(e, DOC_RED, "unparseable URL (bad scheme/host/port)");
185 return;
186 }
187 snprintf(e->host, sizeof(e->host), "%s", u.host);
188 e->port = u.port;
189
190 if (brix_connect(&c, &u, &a->conn, &st) != 0) {
191 doctor_one_connect_fail(a, &u, &st, target, e);
192 return;
193 }
194 e->connected = 1;
195 c.io.timeout_ms = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
196
197 doctor_one_session_facts(&c, e);
198
199 /* phase-93 config audit: scrape advertised config + capacity while the
200 * connection is live (best-effort, PII-free). --all-servers implies it. */
201 if (a->config_audit || a->all_servers || a->map || a->deep_recon) {
202 doctor_scrape_config(&c, e);
203 }
204
205 /* throughput probe over a resolved file (skip cleanly if the export is empty) */
206 have_target = doctor_one_xfer_probe(&c, &u, target, sizeof(target), e);
207
208 /* active differential diagnosis — exercise subsystems + classify (incl. locate). */
209 doctor_diagnose(a, &c, &u,
210 &(dx_target){ .path = target, .have = have_target }, e);
211
212 /* phase-93 deep-recon: read-only reconnaissance while the connection is
213 * live (stats scrape + full config sweep + authorized-root listing). */
214 if (a->deep_recon) {
215 doctor_recon_probe(a, &c, e);
216 }
217
218 brix_close(&c);
219
220 doctor_one_load_signals(a, e);
221
222 /* phase-93: classify the scraped config/capacity values and promote the
223 * perf/shedding signals into machine-readable diagnosis records. */
224 if (a->config_audit || a->all_servers || a->map || a->deep_recon) {
225 doctor_audit_rules(a, e);
226 doctor_audit_perf(e);
227 }
228}

238/*
239 * WHAT: transfer-path diff of one adjacent endpoint pair.
240 * WHY: extracted from doctor_cross for the complexity gate.
241 * HOW: emit TLS-downgrade (critical), auth-fallback, and address-family
242 * asymmetry lines in the original order; returns the number of
243 * critical diffs found (0 or 1).
244 */
245static int
246cross_diff_pair(const doctor_ep *p, const doctor_ep *q, FILE *out)
247{
248 int crit = 0;
249
250 if (!p->connected || !q->connected) {
251 return 0;
252 }
253 /* TLS-downgrade: a TLS hop followed by a cleartext one */
254 if (p->tls_active && !q->tls_active) {
255 fprintf(out, " %s:%d -> %s:%d TLS DOWNGRADE (encrypted then cleartext)\n",
256 p->host, p->port, q->host, q->port);
257 crit++;
258 }
259 /* auth-fallback: the chosen auth weakens across the hop */
260 if (strcmp(p->auth, q->auth) != 0) {
261 fprintf(out, " %s:%d -> %s:%d auth changed %s -> %s\n",
262 p->host, p->port, q->host, q->port, p->auth, q->auth);
263 }
264 /* v4/v6 asymmetry */
265 if (p->nf.family && q->nf.family && p->nf.family != q->nf.family) {
266 fprintf(out, " %s:%d -> %s:%d address-family asymmetry (%s vs %s)\n",
267 p->host, p->port, q->host, q->port,
268 p->nf.family == 10 ? "IPv6" : "IPv4",
269 q->nf.family == 10 ? "IPv6" : "IPv4");
270 }
271 return crit;
272}

275/* Cross-endpoint diff engine over the transfer path. Returns #critical diffs. */
276int
277doctor_cross(const doctor_ep *eps, int n, FILE *out)
278{
279 int i, crit = 0, connected = 0;
280 if (n < 2) {
281 return 0;
282 }
283 for (i = 0; i < n; i++) {
284 if (eps[i].connected) { connected++; }
285 }
286 if (connected < 2) {
287 return 0; /* fewer than two reachable hops — nothing to compare */
288 }
289 fprintf(out, "Path analysis (%d hops):\n", n);
290 for (i = 1; i < n; i++) {
291 crit += cross_diff_pair(&eps[i - 1], &eps[i], out);
292 }
293 return crit;
294}

319/* Route one URL to its protocol battery by scheme. root:// (and any unrecognized
320 * scheme, for back-compat) goes to the full libbrix battery; the rest to their
321 * deep-dive batteries. */
322void
323doctor_dispatch(const diag_args *a, const char *url, doctor_ep *e)
324{
325 dx_url_t u;
326
327 if (dx_url_parse(url, &u) != 0 || u.proto == DXP_ROOT) {
328 doctor_one(a, url, e);
329 return;
330 }
331 switch (u.proto) {
332 case DXP_HTTP:
333 case DXP_HTTPS:
334 case DXP_DAVS: doctor_http(a, &u, e); break;
335 case DXP_S3: doctor_s3(a, &u, e); break;
336 case DXP_CMS: doctor_cms(a, u.host, u.port, u.path, e); break;
337 default: doctor_one(a, url, e); break;
338 }
339}

389/* Render one endpoint's text report block. WHAT: the [color] header, the
390 * per-protocol fact lines, the issue list and the diagnosis block.
391 * WHY: the per-endpoint renderer is the bulk of do_remote_doctor's text
392 * path; splitting it keeps the orchestrator under the complexity gate.
393 * HOW: verbatim move of the loop body — root/cms print the full libbrix
394 * connect-phase + transport facts, the HTTP family only its TLS line. */
395static void
396remote_doctor_report_ep(const doctor_ep *e)
397{
398 int j;
399 char cms[48];
400
401 if (doctor_eos_report_fst(e)) {
402 return; /* EOS FST enumerated from the MGM — never inbound-probed */
403 }
404 doctor_cms_phrase(&e->cms, cms, sizeof(cms));
405 if (e->skipped) {

419 }

421 * the HTTP-family batteries report TLS facts inline + the diagnosis block. */

426 }
427 doctor_report_config(e); /* phase-93 config/capacity block (no-op if unscraped) */
428 doctor_report_recon(e); /* phase-93 deep-recon block (no-op if unprobed) */
429 for (j = 0; j < e->nissues; j++) { printf(" - %s\n", e->issues[j]); }
430 doctor_print_diagnosis(e);
431}

452/*

454 * data server it locates, and diff the fleet for uniformity.

501static int
502do_remote_doctor_fanout(const diag_args *a)
503{
504 doctor_ep *eps = NULL;
505 int i, n = 0, trunc = 0, worst;
506
507 if (doctor_fanout(a, &eps, &n, &trunc) != 0 || eps == NULL) {
508 fprintf(stderr, "xrddiag: remote-doctor fan-out allocation failed\n");
509 return 50;
510 }

512 * on the plain-text path (not for --json, nor when --map owns the output). */
513 doctor_cross_cluster(eps, n, (a->json || a->map) ? NULL : stdout);
514 worst = fanout_worst(eps, n);
515 fanout_probe_latency(a, eps, n);
516 /* --map: draw the discovered mesh. A machine graph (dot/mermaid) is emitted
517 * ALONE so it pipes cleanly to `dot`/a Mermaid renderer; the ASCII tree is
518 * drawn as a header above the usual per-node report. */
519 if (a->map) {
520 doctor_render_map(eps, n, a->map_format, stdout);
521 if (doctor_map_graph_only(a->map_format)) {
522 return fanout_done(eps, worst);
523 }
524 printf("\n");
525 }
526 if (a->json) {
527 doctor_emit_json(eps, n, stdout);
528 return fanout_done(eps, worst);
529 }
530 printf("remote-doctor: manager + %d data server(s)%s\n", n - 1,
531 trunc ? " (data-server list truncated to cap)" : "");
532 for (i = 0; i < n; i++) {
533 remote_doctor_report_ep(&eps[i]);
534 }
535 if (a->latency) { doctor_render_latency(eps, n, stdout); }
536 remote_doctor_report_creds();
537 printf("\nResult: worst=%s\n", doc_color(worst));
538 return fanout_done(eps, worst);
539}

541int
542do_remote_doctor(const diag_args *a)
543{
544 doctor_ep eps[8];
545 int i, worst = DOC_GREEN, crit;
546
547 if (a->nurls < 1) {
548 fprintf(stderr, "xrddiag: remote-doctor needs at least one URL\n");
549 return 50;
550 }
551 if (a->all_servers || a->map || a->latency) {
552 return do_remote_doctor_fanout(a);
553 }
554 for (i = 0; i < a->nurls; i++) {
555 doctor_dispatch(a, a->urls[i], &eps[i]);
556 if (eps[i].status > worst) {
557 worst = eps[i].status;
558 }
559 }
560
561 if (a->json) {
562 doctor_emit_json(eps, a->nurls, stdout);
563 return (worst == DOC_RED) ? 1 : 0;
564 }
565
566 printf("remote-doctor: %d endpoint(s)\n", a->nurls);
567 for (i = 0; i < a->nurls; i++) {
568 remote_doctor_report_ep(&eps[i]);
569 }
570
571 remote_doctor_report_creds();
572
573 printf("\n");
574 crit = doctor_cross(eps, a->nurls, stdout);
575 printf("Result: worst=%s, %d critical path issue(s)\n", doc_color(worst), crit);
576 return (worst == DOC_RED || crit > 0) ? 1 : 0;
577}
