21static int
22remote_stat(const char *remote, struct stat *stbuf)
23{
24 brix_status st;
25 brix_statinfo si;
26 int rc;
27
28 pthread_mutex_lock(&g_lock);
29 if (ensure_conn() != 0) {
30 pthread_mutex_unlock(&g_lock);
31 errno = EIO;
32 return -1;
33 }
34 brix_status_clear(&st);
35 rc = brix_stat(&g_conn, remote, &si, &st);
36 pthread_mutex_unlock(&g_lock);
37 if (rc != 0) {
38 errno = -brix_kxr_to_errno(&st);
39 return -1;
40 }
41 fill_stat(&si, stbuf);
42 return 0;
43}

67int
68BRIXPOSIX_WRAP(fstat)(int fd, struct stat *stbuf)
69{
70 xfs_slot *s;
71 REAL(fstat);
72 s = slot_of(fd);
73 if (s == NULL) {
74 return real_fstat(fd, stbuf);
75 }
76 memset(stbuf, 0, sizeof(*stbuf));
77 stbuf->st_mode = S_IFREG | 0644;
78 stbuf->st_nlink = 1;
79 stbuf->st_size = (off_t) s->size;
80 return 0;
81}
