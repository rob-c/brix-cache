21static int
22remote_stat(const char *remote, struct stat *stbuf)
23{
24 brix_status st;
25 brix_statinfo si;
26 int rc;
27
28 pthread_mutex_lock(&g_lock);
29 if (ensure_conn() != 0) {
30 pthread_mutex_unlock(&g_lock);
31 errno = EIO;
32 return -1;
33 }
34 brix_status_clear(&st);
35 rc = brix_stat(&g_conn, remote, &si, &st);
36 pthread_mutex_unlock(&g_lock);
37 if (rc != 0) {
38 errno = -brix_kxr_to_errno(&st);
39 return -1;
40 }
41 fill_stat(&si, stbuf);
42 return 0;
43}

45/* WHAT: Route stat through the shared prefix policy or real host ABI.
46 * WHY: Legacy glibc entry points must bypass its inline stat redirects.
47 * HOW: Map once, then use remote_stat or the host's original libc entry. */
48int
49brixposix_stat_impl(const char *path, struct stat *stbuf)

89/* WHAT: Describe a shadow descriptor, or preserve real local fstat.
90 * WHY: Older libc's __fxstat callers must recognize virtual descriptors.
91 * HOW: Look up the existing slot; otherwise use the real host ABI. */
92int
93brixposix_fstat_impl(int fd, struct stat *stbuf)
94{
95 xfs_slot *s;
96 s = slot_of(fd);
97 if (s == NULL) {
98 return brix_plat_preload_fstat(fd, stbuf);
99 }
100 memset(stbuf, 0, sizeof(*stbuf));
101 stbuf->st_mode = S_IFREG | 0644;
102 stbuf->st_nlink = 1;
103 stbuf->st_size = (off_t) s->size;
104 return 0;
105}
