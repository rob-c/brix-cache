1/*
2 * bind_migrate.c — §1.4 cross-worker kXR_bind secondary migration.
3 *
4 * See bind_migrate.h for the WHAT/WHY and the safety argument. Mechanism:
5 *
6 * master (init_module, pre-fork)
7 * one AF_UNIX SOCK_SEQPACKET socketpair per worker — every forked worker
8 * inherits the full array, so any worker can sendmsg to any other.
9 *
10 * worker B (kXR_bind arrives, session owned by worker A)
11 * brix_bind_migrate_try(): eligibility gates, then sendmsg the socket fd
12 * (SCM_RIGHTS) + a fixed-size message {magic, listening index, sessid,
13 * streamid} to A's channel. B then abandons its connection without
14 * writing a byte — closing B's fd reference is safe, the in-flight
15 * SCM_RIGHTS copy keeps the socket open.
16 *
17 * worker A (channel read event)
18 * brix_bind_migrate_read(): recvmsg, validate, then adopt — fabricate the
19 * ngx_connection_t / ngx_stream_session_t / brix ctx exactly as the
20 * accept path would have (the listening index resolves the same
21 * addr_conf in A's inherited cycle), stamp the bind streamid, and run
22 * brix_bind_attach() to assign the pathid, register the offload slot and
23 * send the kXR_ok reply from the owning worker.
24 */

46/* [i][0] = worker i's receive end, [i][1] = the end everyone sends to it on.
47 * Created in the master before fork; each worker keeps its own receive end
48 * plus every send end. s_chan_n == 0 means migration is disabled. */
49static int s_chan[BRIX_BIND_MIGRATE_MAX_WORKERS][2];
50static ngx_uint_t s_chan_n;
51
52/* ---- master: create the per-worker channel pairs (pre-fork) --------------
53 *
54 * WHAT: One SOCK_SEQPACKET socketpair per configured worker.
55 * WHY : SCM_RIGHTS fd passing needs a unix socket that exists in both the
56 * sending and receiving worker — only a pre-fork fd is in both.
57 * HOW : Read worker_processes from the core conf; skip (disabled) for a
58 * single worker or an over-cap count. On reload the master runs this
59 * again: close the previous generation's fds first (old workers hold
60 * their own copies; a new generation never mixes channels with an old
61 * one). Any failure disables migration — never fatal, the fallback is
62 * the pre-§1.4 inline-response behavior.
63 */
64void
65brix_bind_migrate_create_channels(ngx_cycle_t *cycle)
66{
67 ngx_core_conf_t *ccf;
68 ngx_uint_t i, n;
69
70 for (i = 0; i < s_chan_n; i++) {
71 (void) close(s_chan[i][0]);
72 (void) close(s_chan[i][1]);
73 }
74 s_chan_n = 0;
75
76 ccf = (ngx_core_conf_t *) ngx_get_conf(cycle->conf_ctx, ngx_core_module);
77 if (ccf == NULL || ccf->worker_processes <= 1) {
78 return; /* single worker: nothing to migrate to */
79 }
80 n = (ngx_uint_t) ccf->worker_processes;
81 if (n > BRIX_BIND_MIGRATE_MAX_WORKERS) {
82 ngx_log_error(NGX_LOG_WARN, cycle->log, 0,
83 "brix: %ui workers exceed the bind-migration channel "
84 "cap (%d) — cross-worker kXR_bind secondaries will use "
85 "inline responses", n, BRIX_BIND_MIGRATE_MAX_WORKERS);
86 return;
87 }
88
89 for (i = 0; i < n; i++) {
90 if (socketpair(AF_UNIX, SOCK_SEQPACKET, 0, s_chan[i]) != 0
91 || ngx_nonblocking(s_chan[i][0]) == -1
92 || ngx_nonblocking(s_chan[i][1]) == -1)
93 {
94 ngx_log_error(NGX_LOG_WARN, cycle->log, ngx_errno,
95 "brix: bind-migration socketpair failed — "
96 "cross-worker secondaries will use inline responses");
97 while (i-- > 0) {
98 (void) close(s_chan[i][0]);
99 (void) close(s_chan[i][1]);
100 }
101 return;
102 }
103 }
104 s_chan_n = n;
105}

109/* ---- worker: arm this worker's channel read end --------------------------
110 *
111 * WHAT: Register the receive end with the event loop; drop the other
112 * workers' receive ends.
113 * WHY : Only the owning worker may drain its channel; the send ends stay
114 * open everywhere so any worker can forward a bind.
115 * HOW : Same ngx_get_connection + read-handler pattern as the admin socket.
116 * The ngx_worker guard also skips the cache-manager/loader processes
117 * (their ngx_worker is out of range).
118 */
119void
120brix_bind_migrate_init_worker(ngx_cycle_t *cycle)
121{
122 ngx_connection_t *ch;
123 ngx_uint_t i;
124
125 if (s_chan_n == 0 || ngx_worker >= s_chan_n) {
126 return;
127 }
128
129 for (i = 0; i < s_chan_n; i++) {
130 if (i != ngx_worker) {
131 (void) close(s_chan[i][0]);
132 s_chan[i][0] = -1;
133 }
134 }
135
136 ch = ngx_get_connection(s_chan[ngx_worker][0], cycle->log);
137 if (ch == NULL) {
138 ngx_log_error(NGX_LOG_WARN, cycle->log, 0,
139 "brix: no connection slot for the bind-migration "
140 "channel — this worker will not adopt secondaries");
141 return;
142 }
143 ch->read->handler = brix_bind_migrate_read;
144 if (ngx_handle_read_event(ch->read, 0) != NGX_OK) {
145 ngx_close_connection(ch);
146 return;
147 }
148 ngx_log_error(NGX_LOG_INFO, cycle->log, 0,
149 "brix: bind-migration channel armed (worker %ui of %ui)",
150 ngx_worker, s_chan_n);
151}

153/* ---- sender side (worker B) ---------------------------------------------
154 *
155 * WHAT: Decide eligibility and pass the connection fd to the owning worker.
156 * WHY : Each gate protects an invariant: TLS state cannot cross processes;
157 * queued responses must not reorder behind frames the owner will send;
158 * an unknown/local owner means there is nothing to migrate to.
159 * HOW : Fixed-size SEQPACKET message + SCM_RIGHTS. A full channel buffer
160 * (EAGAIN) or any error declines — the caller binds locally, which is
161 * always correct.
162 */
163ngx_int_t
164brix_bind_migrate_try(brix_ctx_t *ctx, ngx_connection_t *c,
165 const u_char sessid[BRIX_SESSION_ID_LEN])
166{
167 brix_bind_migrate_msg_t m;
168 struct msghdr mh;
169 struct iovec iov;
170 union {
171 struct cmsghdr align;
172 u_char buf[CMSG_SPACE(sizeof(int))];
173 } cmsgu;
174 struct cmsghdr *cm;
175 ngx_listening_t *ls_base;
176 ngx_int_t owner;
177 ptrdiff_t idx;
178
179 if (s_chan_n == 0 || c->listening == NULL) {
180 return NGX_DECLINED;
181 }
182#if (NGX_STREAM_SSL)
183 if (c->ssl != NULL) {
184 return NGX_DECLINED; /* TLS state cannot cross processes */
185 }
186#endif
187 if (ctx->out.count != 0) {
188 return NGX_DECLINED; /* never reorder queued replies */
189 }
190 if (ctx->recv.stash_len != ctx->recv.stash_head) {
191 /* Read-ahead stash holds pipelined bytes past the bind frame; they
192 * would not travel with the fd. (Unreachable today — the stash only
193 * engages post-auth and a pre-bind secondary never authenticates —
194 * but the migration safety argument must not depend on that.) */
195 return NGX_DECLINED;
196 }
197
198 owner = brix_session_owner_worker(sessid);
199 if (owner < 0 || (ngx_uint_t) owner == ngx_worker
200 || (ngx_uint_t) owner >= s_chan_n)
201 {
202 return NGX_DECLINED;
203 }
204
205 ls_base = ngx_cycle->listening.elts;
206 idx = c->listening - ls_base;
207 if (idx < 0 || (ngx_uint_t) idx >= ngx_cycle->listening.nelts) {
208 return NGX_DECLINED;
209 }
210
211 ngx_memzero(&m, sizeof(m));
212 m.magic = BRIX_BIND_MIGRATE_MAGIC;
213 m.ls_index = (uint32_t) idx;
214 ngx_memcpy(m.sessid, sessid, BRIX_SESSION_ID_LEN);
215 m.streamid[0] = ctx->recv.cur_streamid[0];
216 m.streamid[1] = ctx->recv.cur_streamid[1];
217
218 ngx_memzero(&mh, sizeof(mh));
219 iov.iov_base = &m;
220 iov.iov_len = sizeof(m);
221 mh.msg_iov = &iov;
222 mh.msg_iovlen = 1;
223 mh.msg_control = cmsgu.buf;
224 mh.msg_controllen = sizeof(cmsgu.buf);
225
226 cm = CMSG_FIRSTHDR(&mh);
227 cm->cmsg_level = SOL_SOCKET;
228 cm->cmsg_type = SCM_RIGHTS;
229 cm->cmsg_len = CMSG_LEN(sizeof(int));
230 ngx_memcpy(CMSG_DATA(cm), &c->fd, sizeof(int));
231
232 if (sendmsg(s_chan[owner][1], &mh, 0) != (ssize_t) sizeof(m)) {
233 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, ngx_errno,
234 "brix: bind migration to worker %i declined at "
235 "sendmsg — binding locally", owner);
236 return NGX_DECLINED;
237 }
238
239 brix_log_access(ctx, c, "BIND", "-", "-", 1, 0, "migrated to owner worker", 0);
240 return NGX_OK;
241}

243/* ---- receiver side (worker A): fabricate the adopted connection ---------
244 *
245 * WHAT: Wrap the received fd in a fully-initialized ngx_connection_t, the
246 * way ngx_event_accept would have.
247 * WHY : ngx_get_connection hands out a BARE connection — I/O vtable, pool,
248 * log, peer/local sockaddrs and the connection number are the accept
249 * path's job and every later brix path relies on them.
250 * HOW : Mirror ngx_event_accept + ngx_stream_init_connection field-for-field
251 * for the naddrs==1 case; on any failure close everything and return
252 * NULL (the client sees EOF on the secondary and keeps using the
253 * control stream — the documented degraded mode).
254 */
255static ngx_connection_t *
256brix_bind_migrate_wrap_fd(int fd, ngx_listening_t *ls)
257{
258 ngx_connection_t *c;
259 ngx_log_t *log;
260 u_char text[NGX_SOCKADDR_STRLEN];
261 struct sockaddr sa;
262 socklen_t salen;
263
264 c = ngx_get_connection(fd, ngx_cycle->log);
265 if (c == NULL) {
266 (void) close(fd);
267 return NULL;
268 }
269
270 c->pool = ngx_create_pool(ls->pool_size, ngx_cycle->log);
271 if (c->pool == NULL) {
272 ngx_close_connection(c);
273 return NULL;
274 }
275
276 log = ngx_palloc(c->pool, sizeof(ngx_log_t));
277 if (log == NULL) {
278 ngx_destroy_pool(c->pool);
279 ngx_close_connection(c);
280 return NULL;
281 }
282 *log = ls->log;
283 c->pool->log = log;
284 c->log = log;
285 c->read->log = log;
286 c->write->log = log;
287
288 c->recv = ngx_recv;
289 c->send = ngx_send;
290 c->recv_chain = ngx_recv_chain;
291 c->send_chain = ngx_send_chain;
292 c->type = SOCK_STREAM;
293 c->listening = ls;
294 c->number = ngx_atomic_fetch_add(ngx_connection_counter, 1);
295 c->log->connection = c->number;
296
297 salen = sizeof(sa);
298 if (getpeername(fd, &sa, &salen) == 0) {
299 c->sockaddr = ngx_palloc(c->pool, salen);
300 if (c->sockaddr != NULL) {
301 ngx_memcpy(c->sockaddr, &sa, salen);
302 c->socklen = salen;
303 c->addr_text.data = ngx_pnalloc(c->pool, NGX_SOCKADDR_STRLEN);
304 if (c->addr_text.data != NULL) {
305 c->addr_text.len = ngx_sock_ntop(c->sockaddr, c->socklen,
306 text, NGX_SOCKADDR_STRLEN, 0);
307 ngx_memcpy(c->addr_text.data, text, c->addr_text.len);
308 }
309 }
310 }
311 salen = sizeof(sa);
312 if (getsockname(fd, &sa, &salen) == 0) {
313 c->local_sockaddr = ngx_palloc(c->pool, salen);
314 if (c->local_sockaddr != NULL) {
315 ngx_memcpy(c->local_sockaddr, &sa, salen);
316 c->local_socklen = salen;
317 }
318 }
319
320 return c;
321}

323/* ---- receiver side: fabricate the stream session -------------------------
324 *
325 * WHAT: The minimal faithful subset of ngx_stream_init_connection — conf
326 * context from the listening address, module ctx array, variables,
327 * start time, per-session error log.
328 * WHY : Every brix path reads conf through s; finalize and the stream log
329 * phase read s->variables; skipping any of these breaks teardown.
330 * HOW : naddrs==1 only (checked by the caller) — the addr_conf is the
331 * first (only) address entry on the listening port.
332 */
333static ngx_stream_session_t *
334brix_bind_migrate_make_session(ngx_connection_t *c, ngx_listening_t *ls)
335{
336 ngx_stream_port_t *port = ls->servers;
337 ngx_stream_addr_conf_t *addr_conf;
338 ngx_stream_session_t *s;
339 ngx_stream_core_srv_conf_t *cscf;
340 ngx_stream_core_main_conf_t *cmcf;
341 ngx_time_t *tp;
342
343 if (ls->sockaddr->sa_family == AF_INET6) {
344#if (NGX_HAVE_INET6)
345 addr_conf = &((ngx_stream_in6_addr_t *) port->addrs)[0].conf;
346#else
347 return NULL;
348#endif
349 } else {
350 addr_conf = &((ngx_stream_in_addr_t *) port->addrs)[0].conf;
351 }
352
353 s = ngx_pcalloc(c->pool, sizeof(ngx_stream_session_t));
354 if (s == NULL) {
355 return NULL;
356 }
357
358 s->signature = NGX_STREAM_MODULE;
359 s->main_conf = addr_conf->default_server->ctx->main_conf;
360 s->srv_conf = addr_conf->default_server->ctx->srv_conf;
361 s->connection = c;
362 c->data = s;
363
364 cscf = ngx_stream_get_module_srv_conf(s, ngx_stream_core_module);
365 ngx_set_connection_log(c, cscf->error_log);
366
367 s->ctx = ngx_pcalloc(c->pool, sizeof(void *) * ngx_stream_max_module);
368 if (s->ctx == NULL) {
369 return NULL;
370 }
371
372 cmcf = ngx_stream_get_module_main_conf(s, ngx_stream_core_module);
373 s->variables = ngx_pcalloc(c->pool, cmcf->variables.nelts
374 * sizeof(ngx_stream_variable_value_t));
375 if (s->variables == NULL) {
376 return NULL;
377 }
378
379 tp = ngx_timeofday();
380 s->start_sec = tp->sec;
381 s->start_msec = tp->msec;
382
383 return s;
384}

386/* ---- receiver side: adopt one migrated secondary -------------------------
387 *
388 * WHAT: Validate the message, build connection + session + brix ctx, stamp
389 * the bind streamid and run the shared attach core.
390 * WHY : After this returns the secondary is indistinguishable from one whose
391 * TCP connection was accepted here — same ctx shape, same handlers,
392 * same teardown paths — so every existing bound-stream invariant
393 * (capability restriction, offload registration, disconnect cleanup)
394 * applies unchanged.
395 * HOW : Any validation or allocation failure closes the socket: the client
396 * sees EOF on the secondary and degrades to control-stream I/O.
397 */
398static void
399brix_bind_migrate_adopt(brix_bind_migrate_msg_t *m, int fd)
400{
401 ngx_listening_t *ls;
402 ngx_connection_t *c;
403 ngx_stream_session_t *s;
404 brix_ctx_t *ctx;
405
406 if (m->magic != BRIX_BIND_MIGRATE_MAGIC
407 || m->ls_index >= ngx_cycle->listening.nelts)
408 {
409 (void) close(fd);
410 return;
411 }
412 ls = &((ngx_listening_t *) ngx_cycle->listening.elts)[m->ls_index];
413 if (ls->servers == NULL
414 || ((ngx_stream_port_t *) ls->servers)->naddrs != 1)
415 {
416 (void) close(fd);
417 return;
418 }
419
420 c = brix_bind_migrate_wrap_fd(fd, ls);
421 if (c == NULL) {
422 return;
423 }
424
425 s = brix_bind_migrate_make_session(c, ls);
426 if (s == NULL) {
427 ngx_destroy_pool(c->pool);
428 ngx_close_connection(c);
429 return;
430 }
431
432 ctx = brix_conn_adopt_attach(s, c);
433 if (ctx == NULL) {
434 return; /* attach refused/failed — session already finalized */
435 }
436
437 /* The handshake happened on the source worker; this connection re-enters
438 * the protocol at the request loop, mid-bind. */
439 ctx->state = XRD_ST_REQ_HEADER;
440 ctx->recv.cur_streamid[0] = m->streamid[0];
441 ctx->recv.cur_streamid[1] = m->streamid[1];
442 c->read->handler = ngx_stream_brix_recv;
443 c->write->handler = ngx_stream_brix_send;
444
445 if (brix_bind_attach(ctx, c, m->sessid) == NGX_ERROR) {
446 brix_on_disconnect(ctx, c);
447 brix_close_all_files(ctx);
448 ngx_stream_finalize_session(s, NGX_STREAM_OK);
449 return;
450 }
451
452 /* Park in the request loop: reads the client's next frame or EAGAINs and
453 * arms the read event — exactly the post-accept pump. */
454 ngx_stream_brix_recv(c->read);
455}

457/* ---- receiver side: channel read event -----------------------------------
458 *
459 * WHAT: Drain every queued migration message, adopting each in turn.
460 * WHY : SEQPACKET preserves message boundaries — one recvmsg per bind; a
461 * short/garbled datagram or one without an fd is dropped whole.
462 * HOW : Loop until EAGAIN, then re-arm. The channel peer set is exactly the
463 * sibling workers (pre-fork socketpair, never exposed on a filesystem),
464 * but every field is validated anyway before use.
465 */
466static void
467brix_bind_migrate_read(ngx_event_t *rev)
468{
469 ngx_connection_t *ch = rev->data;
470 brix_bind_migrate_msg_t m;
471 struct msghdr mh;
472 struct iovec iov;
473 union {
474 struct cmsghdr align;
475 u_char buf[CMSG_SPACE(sizeof(int))];
476 } cmsgu;
477 struct cmsghdr *cm;
478 ssize_t n;
479 int fd;
480
481 for ( ;; ) {
482 ngx_memzero(&mh, sizeof(mh));
483 iov.iov_base = &m;
484 iov.iov_len = sizeof(m);
485 mh.msg_iov = &iov;
486 mh.msg_iovlen = 1;
487 mh.msg_control = cmsgu.buf;
488 mh.msg_controllen = sizeof(cmsgu.buf);
489
490 n = recvmsg(ch->fd, &mh, 0);
491 if (n < 0) {
492 break; /* EAGAIN drained; anything else waits */
493 }
494
495 fd = -1;
496 for (cm = CMSG_FIRSTHDR(&mh); cm != NULL; cm = CMSG_NXTHDR(&mh, cm)) {
497 if (cm->cmsg_level == SOL_SOCKET && cm->cmsg_type == SCM_RIGHTS
498 && cm->cmsg_len == CMSG_LEN(sizeof(int)))
499 {
500 ngx_memcpy(&fd, CMSG_DATA(cm), sizeof(int));
501 }
502 }
503 if (fd < 0) {
504 continue; /* no fd attached — nothing to adopt */
505 }
506 if (n != (ssize_t) sizeof(m) || (mh.msg_flags & (MSG_TRUNC | MSG_CTRUNC))) {
507 (void) close(fd);
508 continue;
509 }
510
511 brix_bind_migrate_adopt(&m, fd);
512 }
513
514 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
515 ngx_log_error(NGX_LOG_ALERT, ch->log, 0,
516 "brix: bind-migration channel read re-arm failed");
517 }
518}
