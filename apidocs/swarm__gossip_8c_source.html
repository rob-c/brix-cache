1/* swarm_gossip.c — the G12 swarm's probe/gossip engine (thread + event loop).
2 *
3 * WHAT: the bounded plain-socket roster probe (thread pool), the live-ring
4 * rebuild + publish, the gossip lifecycle timers, and the per-worker
5 * init that arms them.
6 * WHY: swarm.c grew past the file-size gate; the engine — everything that
7 * runs OFF the request path — moved here whole. The membership core,
8 * roster wire format, and the request-path roster endpoint stay in
9 * swarm.c; the shared seam is swarm_internal.h.
10 * HOW: the probe is a SWIM-style push-pull: the GET's query string
11 * introduces the prober (pull-only gossip can never propagate a new
12 * member upstream), the reply body is merged by swarm.c's roster
13 * merge, and the ALIVE view republishes into the cache fill spine
14 * through brix_sd_cache_ring_swap. Published rings are immutable and
15 * never freed (churn leaks ~4.5 KiB per swap by design).
16 */

33static void
34cvmfs_swarm_thread(void *data, ngx_log_t *log)
35{
36 cvmfs_swarm_ctx_t *sw = data;
37 struct addrinfo hints, *res = NULL, *ai;
38 struct timeval tv = { CVMFS_SWARM_IO_TIMEOUT_S, 0 };
39 char portstr[8], req[512];
40 int fd = -1, n;
41 ssize_t got;
42 size_t off = 0;
43 char *body;
44
45 (void) log;
46 sw->probe_ok = 0;
47 sw->resp_len = 0;
48
49 ngx_memzero(&hints, sizeof(hints));
50 hints.ai_family = AF_UNSPEC;
51 hints.ai_socktype = SOCK_STREAM;
52 (void) snprintf(portstr, sizeof(portstr), "%d", sw->probe_port);
53 if (getaddrinfo(sw->probe_host, portstr, &hints, &res) != 0) {
54 return;
55 }
56 for (ai = res; ai != NULL; ai = ai->ai_next) {
57 fd = socket(ai->ai_family, ai->ai_socktype, ai->ai_protocol);
58 if (fd < 0) {
59 continue;
60 }
61 (void) setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
62 (void) setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
63 if (connect(fd, ai->ai_addr, ai->ai_addrlen) == 0) {
64 break;
65 }
66 close(fd);
67 fd = -1;
68 }
69 freeaddrinfo(res);
70 if (fd < 0) {
71 return;
72 }
73
74 /* Push-pull: the query string INTRODUCES the prober (pull-only gossip
75 * can never propagate a new member upstream — the probed node must
76 * learn who called it). */
77 n = snprintf(req, sizeof(req),
78 "GET " CVMFS_SWARM_ROSTER_PATH "?from=%s&gen=%llu"
79 " HTTP/1.0\r\nHost: %s\r\nConnection: close\r\n\r\n",
80 sw->probe_from, (unsigned long long) sw->probe_gen,
81 sw->probe_host);
82 if (n < 0 || (size_t) n >= sizeof(req)
83 || write(fd, req, (size_t) n) != (ssize_t) n)
84 {
85 close(fd);
86 return;
87 }
88 while (off < sizeof(sw->resp) - 1) {
89 got = read(fd, sw->resp + off, sizeof(sw->resp) - 1 - off);
90 if (got < 0) {
91 close(fd);
92 return; /* timeout / reset: probe miss */
93 }
94 if (got == 0) {
95 break;
96 }
97 off += (size_t) got;
98 }
99 close(fd);
100 sw->resp[off] = '\0';
101
102 if (strncmp(sw->resp, "HTTP/1.", 7) != 0
103 || strncmp(sw->resp + 8, " 200", 4) != 0)
104 {
105 return;
106 }
107 body = strstr(sw->resp, "\r\n\r\n");
108 if (body == NULL) {
109 return;
110 }
111 body += 4;
112 sw->resp_len = off - (size_t) (body - sw->resp);
113 memmove(sw->resp, body, sw->resp_len);
114 sw->resp[sw->resp_len] = '\0';
115 sw->probe_ok = 1;
116}

145/* Build the label-sorted ALIVE index set (self always included). Labels are
146 * sorted so every converged node computes the identical ring — the HRW owner
147 * then agrees swarm-wide with no coordination. Caps at the published-ring
148 * limit with a WARN (no silent caps). */
149static ngx_uint_t
150cvmfs_swarm_ring_alive(cvmfs_swarm_ctx_t *sw, ngx_uint_t *alive,
151 ngx_log_t *log)
152{
153 ngx_uint_t i, j, n = 0;
154
155 for (i = 0; i < sw->n_members; i++) {
156 if ((int) i != sw->self && sw->members[i].dead) {
157 continue;
158 }
159 /* insertion sort by label — deterministic across nodes */
160 for (j = n; j > 0
161 && strcmp(sw->members[alive[j - 1]].label,
162 sw->members[i].label) > 0; j--)
163 {
164 alive[j] = alive[j - 1];
165 }
166 alive[j] = i;
167 n++;
168 }
169 if (n > BRIX_SD_CACHE_MAX_PEERS) {
170 ngx_log_error(NGX_LOG_WARN, log, 0,
171 "cvmfs-swarm (export %s): %ui live members exceed the ring "
172 "cap %d - the lexicographic tail falls through to the origin",
173 sw->reg->root, n, BRIX_SD_CACHE_MAX_PEERS);
174 n = BRIX_SD_CACHE_MAX_PEERS;
175 }
176 return n;
177}

179/* Is the candidate alive set identical to the currently published ring? */
180static int
181cvmfs_swarm_ring_unchanged(cvmfs_swarm_ctx_t *sw, const ngx_uint_t *alive,
182 ngx_uint_t n, int self_slot)
183{
184 ngx_uint_t i;
185
186 if (sw->pub_ring == NULL || sw->pub_ring->n != (int) n
187 || sw->pub_ring->self != self_slot)
188 {
189 return 0;
190 }
191 for (i = 0; i < n; i++) {
192 if (strcmp(sw->pub_ring->peers[i].label,
193 sw->members[alive[i]].label) != 0)
194 {
195 return 0;
196 }
197 }
198 return 1;
199}

201/* Publish the ALIVE membership as the live rendezvous ring iff it changed. */
202void
203cvmfs_swarm_ring_publish(cvmfs_swarm_ctx_t *sw, ngx_log_t *log)
204{
205 ngx_uint_t alive[CVMFS_SWARM_MAX_MEMBERS];
206 ngx_uint_t i, n;
207 int self_slot = -1;
208 brix_sd_cache_ring_t *ring;
209 brix_sd_instance_t *cache;
210
211 n = cvmfs_swarm_ring_alive(sw, alive, log);
212 for (i = 0; i < n; i++) {
213 if ((int) alive[i] == sw->self) {
214 self_slot = (int) i;
215 }
216 }
217 if (n < 2 || self_slot < 0) {
218 return; /* no usable ring — keep current */
219 }
220 if (cvmfs_swarm_ring_unchanged(sw, alive, n, self_slot)) {
221 return;
222 }
223
224 cache = brix_vfs_backend_resolve(sw->reg->root, log);
225 if (cache == NULL) {
226 return;
227 }
228 ring = malloc(sizeof(*ring)); /* published forever, never freed */
229 if (ring == NULL) {
230 return;
231 }
232 ngx_memzero(ring, sizeof(*ring));
233 ring->n = (int) n;
234 ring->self = self_slot;
235 for (i = 0; i < n; i++) {
236 cvmfs_swarm_member_t *m = &sw->members[alive[i]];
237
238 ngx_cpystrn((u_char *) ring->peers[i].label, (u_char *) m->label,
239 sizeof(ring->peers[i].label));
240 ring->peers[i].inst = ((int) i == self_slot)
241 ? NULL : cvmfs_swarm_member_inst(m, log);
242 }
243 brix_sd_cache_ring_swap(cache, ring);
244 sw->pub_ring = ring;
245 ngx_log_error(NGX_LOG_NOTICE, log, 0,
246 "cvmfs-swarm (export %s): published live ring of %ui member(s) "
247 "(self=%s)", sw->reg->root, n, ring->peers[self_slot].label);
248}

252static void
253cvmfs_swarm_done(ngx_event_t *ev)
254{
255 ngx_thread_task_t *task = ev->data;
256 cvmfs_swarm_ctx_t *sw = task->ctx;
257 cvmfs_swarm_member_t *m = &sw->members[sw->probe_idx];
258
259 sw->busy = 0;
260 if (sw->probe_ok) {
261 m->miss = 0;
262 if (m->dead) {
263 m->dead = 0; /* direct proof beats gossip */
264 ngx_log_error(NGX_LOG_NOTICE, ev->log, 0,
265 "cvmfs-swarm (export %s): member %s answered a probe - "
266 "back alive", sw->reg->root, m->label);
267 }
268 (void) cvmfs_swarm_roster_merge(sw, sw->resp, sw->resp_len,
269 ev->log);
270 } else if (!m->dead && ++m->miss >= CVMFS_SWARM_MISS_DEAD) {
271 m->dead = 1;
272 ngx_log_error(NGX_LOG_NOTICE, ev->log, 0,
273 "cvmfs-swarm (export %s): member %s missed %d probe(s) - "
274 "marked dead, its keys route around it", sw->reg->root,
275 m->label, CVMFS_SWARM_MISS_DEAD);
276 }
277 cvmfs_swarm_ring_publish(sw, ev->log);
278
279 if (!ngx_exiting) {
280 ngx_add_timer(&sw->timer,
281 (ngx_msec_t) sw->reg->interval * 1000
282 + (ngx_msec_t) (ngx_random() % 500));
283 }
284}

286static void
287cvmfs_swarm_fire(ngx_event_t *ev)
288{
289 cvmfs_swarm_ctx_t *sw = ev->data;
290 ngx_thread_pool_t *pool;
291 ngx_str_t pname;
292 ngx_uint_t i, idx;
293
294 cvmfs_swarm_seed(sw, ev->log);
295 if (sw->seeded && !sw->busy) {
296 cvmfs_swarm_ring_publish(sw, ev->log); /* seed ring before gossip */
297
298 /* round-robin over probe candidates (everyone but self — dead
299 * members stay probed so a restart is noticed and re-merged). */
300 for (i = 0; i < sw->n_members; i++) {
301 idx = (sw->rr + 1 + i) % sw->n_members;
302 if ((int) idx == sw->self) {
303 continue;
304 }
305 sw->rr = idx;
306 sw->probe_idx = idx;
307 sw->probe_port = sw->members[idx].port;
308 sw->probe_gen = sw->self_gen;
309 ngx_cpystrn((u_char *) sw->probe_host,
310 (u_char *) sw->members[idx].host,
311 sizeof(sw->probe_host));
312 ngx_cpystrn((u_char *) sw->probe_from,
313 (u_char *) sw->members[sw->self].label,
314 sizeof(sw->probe_from));
315
316 if (sw->reg->pool[0] != '\0') {
317 pname.data = (u_char *) sw->reg->pool;
318 pname.len = ngx_strlen(sw->reg->pool);
319 } else {
320 ngx_str_set(&pname, "default");
321 }
322 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname);
323 if (pool != NULL) {
324 sw->busy = 1;
325 if (ngx_thread_task_post(pool, sw->task) == NGX_OK) {
326 return; /* re-armed by cvmfs_swarm_done */
327 }
328 sw->busy = 0;
329 }
330 break;
331 }
332 }
333 if (!ngx_exiting) {
334 ngx_add_timer(ev, (ngx_msec_t) sw->reg->interval * 1000);
335 }
336}

338ngx_int_t
339brix_cvmfs_swarm_init_worker(ngx_cycle_t *cycle)
340{
341 ngx_uint_t i;
342
343 /* EVERY worker gossips: backend instances (and so the published ring)
344 * are per-worker. N workers probing is N× a tiny roster GET. */
345 for (i = 0; i < cvmfs_swarm_regs_n; i++) {
346 ngx_thread_task_t *task;
347 cvmfs_swarm_ctx_t *sw;
348
349 task = ngx_thread_task_alloc(cycle->pool, sizeof(cvmfs_swarm_ctx_t));
350 if (task == NULL) {
351 return NGX_ERROR;
352 }
353 sw = task->ctx;
354 sw->task = task;
355 sw->reg = &cvmfs_swarm_regs[i];
356 sw->self = -1;
357
358 brix_task_bind(task, cvmfs_swarm_thread, cvmfs_swarm_done);
359 task->event.log = cycle->log;
360
361 sw->timer.handler = cvmfs_swarm_fire;
362 sw->timer.data = sw;
363 sw->timer.log = cycle->log;
364 ngx_add_timer(&sw->timer, (ngx_msec_t) sw->reg->interval * 1000
365 + (ngx_msec_t) (ngx_random() % 500));
366
367 cvmfs_swarm_ctxs[i] = sw;
368 }
369 return NGX_OK;
370}
