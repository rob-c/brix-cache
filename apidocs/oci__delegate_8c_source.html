1/*
2 * oci_delegate.c — downstream Basic delegation and the authorize-on-hit gate
3 * (phase-104 D16).
4 *
5 * WHAT: the mirror's delegated-pull mode: read the client's own registry
6 * credential from a downstream Basic header, prove it against the
7 * upstream once per (credential, repository), and refuse to serve any
8 * object route — cache hit or miss alike — without a fresh proof.
9 * WHY: one shared content-addressed cache is the whole point of a mirror,
10 * and it is exactly what turns a private image into a leak: after the
11 * first authorized pull the bytes are local, and a plain cache would
12 * hand them to anyone who can spell the digest. The fix is not to
13 * shard the cache but to keep the UPSTREAM the authorization oracle:
14 * every request on a delegate-mode mirror must first mint (or hold) a
15 * per-principal authorization proof, so revocation upstream propagates
16 * within the proof TTL and no user's credential or token can satisfy
17 * another user's request. The credential itself is replayed ONLY to
18 * the allowlisted token endpoint — never stored, never sent to the
19 * data plane; its sha256 is the one form that outlives the request,
20 * and only ever as an SHM key.
21 * HOW: ident (event loop): decode Basic, refuse cleartext, hash. Gate
22 * (event loop): SHM proof probe; on a miss, one bounded thread-pool
23 * task runs the proof: challenge memo → credential-scoped token mint
24 * (oci_upstream_auth.c) → a HEAD against the object with that bearer,
25 * because a DockerHub-style endpoint answers a denied scope with a
26 * 200 and an empty grant, so "the mint worked" is NOT "authorized".
27 * Denials are answered with one uniform 401 that reveals nothing —
28 * not even whether the repository exists.
29 */

44/* How long a discovered upstream auth challenge is memoized. The challenge
45 * names the realm and service, which change on upstream reconfiguration and
46 * nothing else — but a bound keeps a stale realm from outliving the day. */
47#define OCI_DELEG_CHAL_TTL_MS (3600 * 1000)

71/* ---- keys ---------------------------------------------------------------- */

73/* The proof record's key: sha256("proof" ‖ 0x00 ‖ base_url ‖ 0x00 ‖ scope ‖
74 * 0x00 ‖ cred). Same NUL discipline as the token key, same reason — and the
75 * ASCII prefix is what keeps a proof from ever aliasing a token entry in the
76 * shared zone. 0 = key written. */
77static int
78oci_deleg_proof_key(const brix_oci_upstream_t *up, const char *scope,
79 const u_char cred[32], u_char key[32])
80{
81 char buf[1600];
82 size_t n = 0, b = strlen(up->base_url), s = strlen(scope);
83
84 if (6 + b + 1 + s + 1 + 32 >= sizeof(buf)) {
85 return -1;
86 }
87 memcpy(buf, "proof", 6); n = 6;
88 memcpy(buf + n, up->base_url, b); n += b;
89 buf[n++] = '\0';
90 memcpy(buf + n, scope, s); n += s;
91 buf[n++] = '\0';
92 memcpy(buf + n, cred, 32); n += 32;
93
94 return brix_oci_sha256_key(buf, n, key);
95}

97/* The challenge memo's key: sha256("chal" ‖ 0x00 ‖ base_url). Per upstream,
98 * not per scope — the realm and service are properties of the registry. */
99static int
100oci_deleg_chal_key(const brix_oci_upstream_t *up, u_char key[32])
101{
102 char buf[600];
103 size_t b = strlen(up->base_url);
104
105 if (5 + b >= sizeof(buf)) {
106 return -1;
107 }
108 memcpy(buf, "chal", 5);
109 memcpy(buf + 5, up->base_url, b);
110
111 return brix_oci_sha256_key(buf, 5 + b, key);
112}

116/* The one downstream challenge this surface issues, on every refusal that a
117 * (different) credential could cure. Basic, not Bearer: the client's own
118 * registry credential is the thing being delegated, and docker/podman login
119 * already speaks it with no ceremony. */
120static ngx_int_t
121oci_deleg_challenge_hdr(ngx_http_request_t *r,
122 ngx_http_brix_oci_loc_conf_t *lcf)
123{
124 ngx_table_elt_t *h = ngx_list_push(&r->headers_out.headers);
125 u_char *v;
126 size_t n;
127
128 if (h == NULL) {
129 return NGX_ERROR;
130 }
131 n = sizeof("Basic realm=\"\"") + lcf->delegate_realm.len;
132 v = ngx_pnalloc(r->pool, n);
133 if (v == NULL) {
134 return NGX_ERROR;
135 }
136 h->hash = 1;
137 ngx_str_set(&h->key, "WWW-Authenticate");
138 h->value.data = v;
139 h->value.len = (size_t) (ngx_snprintf(v, n, "Basic realm=\"%V\"",
140 &lcf->delegate_realm) - v);
141
142 return NGX_OK;
143}

145/* The uniform refusal (D16): 401 + Basic challenge + the DENIED envelope with
146 * no detail. One spelling for "bad password", "no such repository" and "not
147 * yours to read", because any difference between them is an enumeration
148 * oracle over somebody else's private namespace. */
149static ngx_int_t
150oci_deleg_refuse(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
151 ngx_http_brix_oci_ctx_t *ctx)
152{
153 ctx->disp = BRIX_OCI_OUT_REFUSED;
154 brix_oci_guard_emit(r, GUARD_R_AUTHFAIL, GUARD_OP_READ,
155 NGX_HTTP_UNAUTHORIZED);
156 if (oci_deleg_challenge_hdr(r, lcf) != NGX_OK) {
157 return NGX_HTTP_INTERNAL_SERVER_ERROR;
158 }
159 return brix_oci_error(r, NGX_HTTP_UNAUTHORIZED, BRIX_OCI_ERR_DENIED,
160 NULL);
161}

163/* Decode the Basic payload into "user:pass" in the request pool and hash it.
164 * NGX_DECLINED = identity established; anything else already answered. */
165static ngx_int_t
166oci_deleg_decode(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
167 ngx_http_brix_oci_ctx_t *ctx, ngx_str_t *b64)
168{
169 ngx_str_t dst;
170 u_char *colon;
171 char *user;
172
173 dst.data = ngx_pnalloc(r->pool, ngx_base64_decoded_length(b64->len) + 1);
174 if (dst.data == NULL) {
175 return NGX_HTTP_INTERNAL_SERVER_ERROR;
176 }
177 if (ngx_decode_base64(&dst, b64) != NGX_OK
178 || dst.len == 0 || dst.len >= BRIX_OCI_BASIC_MAX
179 || memchr(dst.data, '\0', dst.len) != NULL)
180 {
181 return oci_deleg_refuse(r, lcf, ctx);
182 }
183 dst.data[dst.len] = '\0';
184
185 colon = (u_char *) strchr((char *) dst.data, ':');
186 if (colon == NULL || colon == dst.data) {
187 return oci_deleg_refuse(r, lcf, ctx);
188 }
189
190 if (brix_oci_sha256_key(dst.data, dst.len, ctx->deleg_cred) != 0) {
191 return NGX_HTTP_INTERNAL_SERVER_ERROR;
192 }
193 ctx->deleg_basic = (const char *) dst.data;
194
195 /* The username half, for the dashboard/log identity. Never the pair. */
196 user = ngx_pnalloc(r->pool, (size_t) (colon - dst.data) + 1);
197 if (user == NULL) {
198 return NGX_HTTP_INTERNAL_SERVER_ERROR;
199 }
200 ngx_memcpy(user, dst.data, (size_t) (colon - dst.data));
201 user[colon - dst.data] = '\0';
202 ctx->deleg_user = user;
203
204 return NGX_DECLINED;
205}

207ngx_int_t
208brix_oci_delegate_ident(ngx_http_request_t *r,
209 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx)
210{
211 ngx_table_elt_t *auth = r->headers_in.authorization;
212 ngx_str_t b64;
213 int is_tls = 0;
214
215 if (!lcf->delegate || auth == NULL) {
216 return NGX_DECLINED; /* off, or anonymous: cred = 0s */
217 }
218
219#if (NGX_HTTP_SSL)
220 is_tls = (r->connection->ssl != NULL);
221#endif
222 /* A credential on a cleartext connection is already burned; the ONE
223 * thing still in our power is to refuse to act on it — before decoding,
224 * so the secret never even enters this process's data flow. */
225 if (!is_tls && !lcf->deleg_insecure) {
226 ctx->disp = BRIX_OCI_OUT_REFUSED;
227 brix_oci_guard_emit(r, GUARD_R_AUTHFAIL, GUARD_OP_READ,
228 NGX_HTTP_BAD_REQUEST);
229 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST, BRIX_OCI_ERR_DENIED,
230 "credentials require TLS on this mirror");
231 }
232
233 if (auth->value.len < sizeof("Basic ") - 1
234 || ngx_strncasecmp(auth->value.data, (u_char *) "Basic ",
235 sizeof("Basic ") - 1) != 0)
236 {
237 /* Bearer, Negotiate, junk: not a credential this surface can
238 * delegate, and guessing is how a secret goes somewhere wrong. */
239 return oci_deleg_refuse(r, lcf, ctx);
240 }
241 b64.data = auth->value.data + sizeof("Basic ") - 1;
242 b64.len = auth->value.len - (sizeof("Basic ") - 1);
243
244 return oci_deleg_decode(r, lcf, ctx, &b64);
245}

249/* One upstream leg on the worker thread: `method` against `path` (relative
250 * to the upstream base), with an optional extra header block. The HTTP
251 * status, or -1 on transport failure; `challenge` (may be NULL) receives the
252 * WWW-Authenticate of a 401. */
253static int
254oci_deleg_leg(brix_oci_upstream_t *up, const char *method, const char *path,
255 const char *hdrs, char *challenge, size_t challenge_len)
256{
257 const brix_s3_transport_t *tr = &brix_s3_origin_curl_transport;
258 brix_s3_resp_t resp;
259 char full[BRIX_OCI_KEY_MAX + 256];
260 char errbuf[256];
261 int status;
262
263 if (challenge != NULL) {
264 challenge[0] = '\0';
265 }
266 if ((size_t) snprintf(full, sizeof(full), "%s%s", up->base_path, path)
267 >= sizeof(full))
268 {
269 return -1;
270 }
271 if (tr->request(NULL, up->host, up->port, up->tls, method, full, hdrs,
272 NULL, 0, OCI_DELEG_TIMEOUT_MS, &resp,
273 errbuf, sizeof(errbuf)) != 0)
274 {
275 ngx_log_error(NGX_LOG_ERR, up->log, 0,
276 "oci: delegate proof leg unreachable host=%s: %s",
277 up->host, errbuf);
278 return -1;
279 }
280 status = resp.status;
281 if (status == NGX_HTTP_UNAUTHORIZED && challenge != NULL) {
282 (void) tr->resp_header(&resp, "www-authenticate",
283 challenge, challenge_len);
284 }
285 tr->resp_free(&resp);
286
287 return status;
288}

290/* Establish the upstream's auth challenge, through the SHM memo: probe
291 * GET /v2/ once per upstream per memo-TTL, remember either the challenge or
292 * the fact that the upstream challenges nobody. 0 = `chal` holds the
293 * challenge or the open marker; -1 = the upstream could not be asked. */
294static int
295oci_deleg_challenge(brix_oci_upstream_t *up, char *chal, size_t chalsz)
296{
297 u_char key[32];
298 size_t out_len = chalsz - 1;
299 int status;
300
301 if (up->tokens != NULL && oci_deleg_chal_key(up, key) == 0
302 && brix_kv_get(up->tokens, key, sizeof(key), chal, &out_len))
303 {
304 chal[out_len] = '\0';
305 return 0;
306 }
307
308 status = oci_deleg_leg(up, "GET", "/v2/", NULL, chal, chalsz);
309 if (status == NGX_HTTP_UNAUTHORIZED && chal[0] != '\0') {
310 /* keep chal as-is */
311 } else if (status >= 200 && status < 400) {
312 (void) snprintf(chal, chalsz, "%s", OCI_DELEG_OPEN_MARK);
313 } else {
314 return -1;
315 }
316
317 if (up->tokens != NULL && oci_deleg_chal_key(up, key) == 0) {
318 (void) brix_kv_set(up->tokens, key, sizeof(key), chal, strlen(chal),
319 OCI_DELEG_CHAL_TTL_MS);
320 }
321 return 0;
322}

324/* The verify HEAD: does THIS bearer actually read THIS object? 401/403 is
325 * the denial; 404 is "authorized, absent" (the fill will surface it); any
326 * other answer proves the grant. Required because a DockerHub-style token
327 * endpoint mints a 200 with an empty access list for a denied scope. */
328static int
329oci_deleg_verify(oci_deleg_task_t *t, const char *tok)
330{
331 char hdrs[BRIX_OCI_TOKEN_MAX + 64];
332 int status;
333
334 hdrs[0] = '\0';
335 if (tok != NULL
336 && snprintf(hdrs, sizeof(hdrs), "Authorization: Bearer %s\r\n", tok)
337 >= (int) sizeof(hdrs))
338 {
339 return OCI_DELEG_ERROR;
340 }
341 status = oci_deleg_leg(t->up, t->probe_get ? "GET" : "HEAD", t->path,
342 hdrs[0] ? hdrs : NULL, NULL, 0);
343 if (status < 0) {
344 return OCI_DELEG_ERROR;
345 }
346 if (status == NGX_HTTP_UNAUTHORIZED || status == NGX_HTTP_FORBIDDEN) {
347 return OCI_DELEG_DENIED;
348 }
349 return OCI_DELEG_GRANTED;
350}

352/* thread side: challenge → credential-scoped mint → verify → record. */
353static void
354oci_deleg_thread(void *data, ngx_log_t *log)
355{
356 oci_deleg_task_t *t = data;
357 char chal[1024];
358 char tok[BRIX_OCI_TOKEN_MAX];
359 u_char key[32];
360 long expires = 0;
361 int denied = 0;
362
363 (void) log;
364 t->verdict = OCI_DELEG_ERROR;
365
366 if (oci_deleg_challenge(t->up, chal, sizeof(chal)) != 0) {
367 return;
368 }
369
370 if (strcmp(chal, OCI_DELEG_OPEN_MARK) == 0) {
371 /* An upstream with no auth has no private repositories: the HEAD
372 * still runs, so a broken upstream is an error, not a grant. */
373 t->verdict = oci_deleg_verify(t, NULL);
374
375 } else {
376 if (brix_oci_token_get_cred(t->up, t->path, chal,
377 t->basic[0] ? t->basic : NULL, t->cred,
378 tok, sizeof(tok), &expires,
379 &denied) != 0)
380 {
381 t->verdict = denied ? OCI_DELEG_DENIED : OCI_DELEG_ERROR;
382 return;
383 }
384 t->verdict = oci_deleg_verify(t, tok);
385 }
386
387 if (t->verdict != OCI_DELEG_GRANTED) {
388 return;
389 }
390
391 /* Record the proof, then lend the bearer to the credential-blind entry
392 * the coalesced fill's provider probes — bounded by the proof TTL and by
393 * the token's own life, whichever ends first. */
394 if (t->up->tokens != NULL
395 && oci_deleg_proof_key(t->up, t->scope, t->cred, key) == 0)
396 {
397 (void) brix_kv_set(t->up->tokens, key, sizeof(key), "1", 1,
398 (ngx_msec_t) t->proof_ttl * 1000);
399 }
400 if (strcmp(chal, OCI_DELEG_OPEN_MARK) != 0) {
401 long share = (long) t->proof_ttl;
402
403 if (expires > 0 && expires < share) {
404 share = expires;
405 }
406 brix_oci_token_share(t->up, t->path, tok, share);
407 }
408}

410/* event-loop side: turn the verdict into the response or the re-entry. */
411static void
412oci_deleg_done(ngx_event_t *ev)
413{
414 ngx_thread_task_t *task = ev->data;
415 oci_deleg_task_t *t = task->ctx;
416 ngx_http_request_t *r = t->r;
417 ngx_connection_t *c = r->connection;
418 ngx_http_brix_oci_loc_conf_t *lcf =
419 ngx_http_get_module_loc_conf(r, ngx_http_brix_oci_module);
420 ngx_http_brix_oci_ctx_t *ctx =
421 ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
422
423 if (ctx == NULL) {
424 ngx_http_finalize_request(r, NGX_HTTP_INTERNAL_SERVER_ERROR);
425 ngx_http_run_posted_requests(c);
426 return;
427 }
428
429 if (t->verdict == OCI_DELEG_GRANTED) {
430 BRIX_OCI_METRIC_INC(delegate_total[BRIX_OCI_DELEG_GRANTED]);
431 ctx->deleg_proved = 1;
432 ngx_http_finalize_request(r, ngx_http_brix_oci_handler(r));
433
434 } else if (t->verdict == OCI_DELEG_DENIED) {
435 BRIX_OCI_METRIC_INC(delegate_total[BRIX_OCI_DELEG_DENIED]);
436 ngx_http_finalize_request(r, oci_deleg_refuse(r, lcf, ctx));
437
438 } else {
439 BRIX_OCI_METRIC_INC(delegate_total[BRIX_OCI_DELEG_ERROR]);
440 BRIX_OCI_METRIC_INC(upstream_errors_total[BRIX_OCI_UPERR_OTHER]);
441 ctx->disp = BRIX_OCI_OUT_ERROR;
442 ngx_http_finalize_request(r,
443 brix_oci_error(r, NGX_HTTP_BAD_GATEWAY, BRIX_OCI_ERR_UNAVAILABLE,
444 NULL));
445 }
446 ngx_http_run_posted_requests(c);
447}

449/* ---- the gate (event loop) ----------------------------------------------- */

451/* Derive the upstream path the proof leg will probe. The object routes carry
452 * the canonical key and are probed with a HEAD of the object itself. The
453 * listing routes are gated BEFORE any key is built, so the classifier's
454 * validated name is the source and a GET of the (paginated) tags listing is
455 * the probe — same repository:pull scope either way, and the scope string is
456 * derived ONCE, by pull_scope, from whichever probe path is in play.
457 * NGX_OK = probe written (*probe_get set for the listing shape); anything
458 * else is the finalized status the gate must return as-is. */
459static ngx_int_t
460oci_deleg_probe(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
461 char *probe, size_t probesz, int *probe_get)
462{
463 if (ctx->keyed) {
464 if ((size_t) snprintf(probe, probesz, "%s", ctx->key) >= probesz) {
465 return NGX_HTTP_INTERNAL_SERVER_ERROR;
466 }
467 return NGX_OK;
468 }
469
470 if (ctx->classified && ctx->req.name != NULL && ctx->req.name_len > 0) {
471 if ((size_t) snprintf(probe, probesz, "/v2/%.*s/tags/list?n=1",
472 (int) ctx->req.name_len, ctx->req.name)
473 >= probesz)
474 {
475 return NGX_HTTP_INTERNAL_SERVER_ERROR;
476 }
477 *probe_get = 1;
478 return NGX_OK;
479 }
480
481 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
482 BRIX_OCI_ERR_NAME_INVALID, NULL);
483}

485/* Copy everything the worker leg needs OUT of the request into the task —
486 * after the post, only oci_deleg_done may touch `r` again. NGX_OK, or 500
487 * when a component that already fit its source buffer somehow does not fit
488 * the task's (a size-contract violation, not a runtime condition). */
489static ngx_int_t
490oci_deleg_task_fill(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
491 ngx_http_brix_oci_ctx_t *ctx, const char *probe, int probe_get,
492 const char *scope, oci_deleg_task_t *t)
493{
494 t->r = r;
495 t->up = lcf->up;
496 t->proof_ttl = lcf->deleg_proof_ttl;
497 ngx_memcpy(t->cred, ctx->deleg_cred, sizeof(t->cred));
498 t->probe_get = probe_get ? 1 : 0;
499 (void) snprintf(t->scope, sizeof(t->scope), "%s", scope);
500 if ((size_t) snprintf(t->path, sizeof(t->path), "%s", probe)
501 >= sizeof(t->path))
502 {
503 return NGX_HTTP_INTERNAL_SERVER_ERROR;
504 }
505 t->basic[0] = '\0';
506 if (ctx->deleg_basic != NULL
507 && (size_t) snprintf(t->basic, sizeof(t->basic), "%s",
508 ctx->deleg_basic) >= sizeof(t->basic))
509 {
510 return NGX_HTTP_INTERNAL_SERVER_ERROR;
511 }
512
513 return NGX_OK;
514}

516ngx_int_t
517brix_oci_delegate_gate(ngx_http_request_t *r,
518 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx)
519{
520 ngx_thread_task_t *task;
521 ngx_thread_pool_t *pool;
522 ngx_int_t rc;
523 char scope[512];
524 u_char key[32];
525 size_t out_len;
526 char one[2];
527
528 char probe[BRIX_OCI_KEY_MAX + 32];
529 int probe_get = 0;
530
531 if (!lcf->delegate || ctx->deleg_proved) {
532 return NGX_DECLINED;
533 }
534
535 rc = oci_deleg_probe(r, ctx, probe, sizeof(probe), &probe_get);
536 if (rc != NGX_OK) {
537 return rc;
538 }
539
540 if (brix_oci_pull_scope(probe, scope, sizeof(scope)) != 0) {
541 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
542 BRIX_OCI_ERR_NAME_INVALID, NULL);
543 }
544
545 out_len = sizeof(one) - 1;
546 if (lcf->up->tokens != NULL
547 && oci_deleg_proof_key(lcf->up, scope, ctx->deleg_cred, key) == 0
548 && brix_kv_get(lcf->up->tokens, key, sizeof(key), one, &out_len))
549 {
550 BRIX_OCI_METRIC_INC(delegate_total[BRIX_OCI_DELEG_CACHED]);
551 return NGX_DECLINED;
552 }
553
554 brix_oci_up_log_ensure(lcf);
555
556 pool = brix_oci_thread_pool(lcf);
557 if (pool == NULL) {
558 return brix_oci_error(r, NGX_HTTP_SERVICE_UNAVAILABLE,
559 BRIX_OCI_ERR_UNAVAILABLE,
560 "no thread pool configured for delegation");
561 }
562 task = ngx_thread_task_alloc(r->pool, sizeof(oci_deleg_task_t));
563 if (task == NULL) {
564 return NGX_HTTP_INTERNAL_SERVER_ERROR;
565 }
566 rc = oci_deleg_task_fill(r, lcf, ctx, probe, probe_get, scope, task->ctx);
567 if (rc != NGX_OK) {
568 return rc;
569 }
570
571 brix_task_bind(task, oci_deleg_thread, oci_deleg_done);
572 task->event.log = r->connection->log;
573
574 if (ngx_thread_task_post(pool, task) != NGX_OK) {
575 return NGX_HTTP_INTERNAL_SERVER_ERROR;
576 }
577 r->main->count++; /* request survives until oci_deleg_done */
578
579 return NGX_DONE;
