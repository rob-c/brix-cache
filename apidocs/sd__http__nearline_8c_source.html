1/*
2 * sd_http_nearline.c — the nearline (tape/MSS) pair of the http origin driver:
3 * residency (is this path online?) and recall (bring it online), spoken over the
4 * WLCG Tape REST API.
5 *
6 * WHY THIS FILE EXISTS
7 * A WebDAV origin in front of an HSM is the storage shape most of WLCG
8 * actually runs, and until this file existed the http driver had neither
9 * nearline slot: brix_vfs_residency reported ONLINE for every path (its answer
10 * for a driver with no residency model), and the first read of a migrated file
11 * blocked a worker for the length of a tape mount instead of parking the open
12 * and answering "staging, retry later". The protocol planes above already know
13 * how to say that — HTTP 202 + Retry-After, S3 InvalidObjectState, root://
14 * kXR_offline — they just had nothing underneath to say it from.
15 *
16 * THE WIRE
17 * The WLCG Tape REST API is the one interoperable spelling of both halves, and
18 * it splits them exactly as this driver's slots do:
19 * POST {base}/archiveinfo {"paths":["/k"]} → [{"path":…,"locality":…}]
20 * POST {base}/stage {"files":[{"path":"/k"}]} → 201 {"requestId":…}
21 * `archiveinfo` IS the residency slot: it reports locality without touching
22 * the tape system, which is the property the residency contract requires.
23 *
24 * OPT-IN, AND WHY IT MUST BE
25 * Neither slot is reachable unless the instance advertises
26 * BRIX_SD_CAP_NEARLINE, which brix_sd_http_create sets only when the operator
27 * configured a tape-API base. CAP_NEARLINE is a CONTRACT, not a hint:
28 * tier_build refuses to compose a nearline backend without a cache tier in
29 * front (the recall target, §9.4), so inferring it — from a 202, from a header,
30 * from anything — would turn working plain-http configs into startup failures.
31 *
32 * SHAPE
33 * Both slots issue ONE request through the shared no-failover namespace sender
34 * (sd_http_ns_send): a stage submission replayed against a second endpoint
35 * would queue the same tape work twice. Neither slot takes a credential — the
36 * vtable has no _cred twin for either — so both run as the instance identity,
37 * which is correct: a recall is the gateway's own housekeeping, driven by the
38 * cache tier on a miss, not an operation the end user names.
39 */

57/*
58 * WHAT: Write `in` as a complete JSON string literal (quotes included) into
59 * out[cap]. 0, or -1 on overflow.
60 * WHY: The key is the only caller-controlled text in either request body. A
61 * key containing a quote or a backslash would otherwise close the string
62 * early and let the rest of the path be read as JSON structure — a request
63 * for one path becoming a request for several, or a stage submission
64 * carrying fields the driver never wrote. This is an injection defence
65 * before it is an encoding.
66 * HOW: Escape the two characters JSON reserves and emit every control byte as
67 * \u00XX; everything else passes through as-is (a UTF-8 path stays UTF-8).
68 * Overflow is refused, never truncated — a truncated path names a
69 * DIFFERENT object.
70 */
71static int
72sd_http_json_quote(const char *in, char *out, size_t cap)
73{
74 size_t o = 0;
75 size_t i;
76
77 if (in == NULL || out == NULL || cap < 3) {
78 return -1;
79 }
80 out[o++] = '"';
81 for (i = 0; in[i] != '\0'; i++) {
82 unsigned char c = (unsigned char) in[i];
83
84 if (c == '"' || c == '\\') {
85 if (o + 2 >= cap) {
86 return -1;
87 }
88 out[o++] = '\\';
89 out[o++] = (char) c;
90 } else if (c < 0x20) {
91 if (o + 6 >= cap) {
92 return -1;
93 }
94 o += (size_t) snprintf(out + o, cap - o, "\\u%04x", c);
95 } else {
96 if (o + 1 >= cap) {
97 return -1;
98 }
99 out[o++] = (char) c;
100 }
101 }
102 if (o + 2 > cap) {
103 return -1;
104 }
105 out[o++] = '"';
106 out[o] = '\0';
107 return 0;
108}

111/*
112 * WHAT: POST one JSON document to {tape_api}{suffix}; 0 with `resp` holding a
113 * reply the caller must resp_free, or -1 with errno set.
114 * WHY: residency and recall differ only in the suffix and the body; the header
115 * block, the identity and the no-failover rule belong to both.
116 * HOW: Through sd_http_ns_send, so this shares the mutation sender's endpoint-0
117 * pinning rather than opening a second wire path. `auth`/`cert_pem`
118 * (phase-107 C2) override the identity for the recall_cred twin — a
119 * non-empty `auth` replaces the instance's static Authorization line and
120 * `cert_pem` presents the user's x509 proxy; NULL/"" keeps the static
121 * header, exactly the resolve_open_cred fallback every mutation slot uses.
122 */
123static int
124sd_http_tape_post(sd_http_inst_state *is, const char *suffix, const char *body,
125 brix_s3_resp_t *resp, const char *auth, const char *cert_pem)
126{
127 sd_http_ns_req_t rq;
128 char path[SD_HTTP_PATH_MAX];
129 char hdrs[SD_HTTP_AUTH_MAX + 64];
130 char err[256];
131
132 if (snprintf(path, sizeof(path), "%s%s", is->tape_api, suffix)
133 >= (int) sizeof(path))
134 {
135 errno = ENAMETOOLONG;
136 return -1;
137 }
138 (void) snprintf(hdrs, sizeof(hdrs),
139 "Content-Type: application/json\r\n%s",
140 (auth != NULL && auth[0] != '\0') ? auth : is->auth_hdr);
141
142 memset(&rq, 0, sizeof(rq));
143 rq.method = "POST";
144 rq.path = path;
145 rq.hdrs = hdrs;
146 rq.cert_pem = cert_pem;
147 rq.body = body;
148 rq.body_len = strlen(body);
149
150 err[0] = '\0';
151 if (sd_http_ns_send(is, &rq, resp, err, sizeof(err)) != 0) {
152 errno = EIO;
153 return -1;
154 }
155 return 0;
156}

159/*
160 * WHAT: Map one Tape REST API locality token onto a residency class. 0 with
161 * *out set, or -1 with errno set.
162 * WHY: The mapping is the whole semantic content of the residency slot, and
163 * two of its cases are decisions rather than translations: DISK_AND_TAPE
164 * is ONLINE (the disk copy serves the read; the tape copy is irrelevant to
165 * a reader), and NONE is not a residency class at all but the API's way of
166 * saying the path is not in the namespace — reporting it as LOST would
167 * tell every plane above that a file had been DESTROYED because the origin
168 * merely does not have it.
169 * HOW: Exact token match; an unknown token is EIO, never a guess. A new
170 * locality this build has not seen must not be silently read as ONLINE —
171 * that would hand the caller a file the tape system never produced.
172 */
173static int
174sd_http_locality(const char *loc, brix_sd_residency_t *out)
175{
176 if (strcmp(loc, "DISK") == 0 || strcmp(loc, "DISK_AND_TAPE") == 0) {
177 *out = BRIX_SD_RES_ONLINE;
178 return 0;
179 }
180 if (strcmp(loc, "TAPE") == 0) {
181 *out = BRIX_SD_RES_NEARLINE;
182 return 0;
183 }
184 if (strcmp(loc, "UNAVAILABLE") == 0) {
185 *out = BRIX_SD_RES_OFFLINE;
186 return 0;
187 }
188 if (strcmp(loc, "LOST") == 0) {
189 *out = BRIX_SD_RES_LOST;
190 return 0;
191 }
192 if (strcmp(loc, "NONE") == 0) {
193 errno = ENOENT;
194 return -1;
195 }
196 errno = EIO;
197 return -1;
198}

201/*
202 * WHAT: Pull the first element's `locality` out of an archiveinfo reply body.
203 * 0 with *out set, or -1 with errno set.
204 * WHY: Kept separate from the request so the reply grammar — a top-level ARRAY
205 * whose first element describes the one path we asked about — is stated
206 * once, next to the mapping it feeds.
207 * HOW: brix_json_arr_next over the raw body (the span it wants is the array
208 * including its brackets, which is the whole document here), then one
209 * string lookup (brix_json_get_str answers 1 for "found and copied", not
210 * 0 — the codec's convention, not errno's). An empty array is ENOENT: the
211 * API answered, and its answer contains nothing about our path.
212 */
213static int
214sd_http_archiveinfo_parse(const char *body, size_t len,
215 brix_sd_residency_t *out)
216{
217 const char *elem;
218 size_t elemlen;
219 size_t cursor = 0;
220 char loc[SD_HTTP_TAPE_LOC_MAX];
221
222 if (body == NULL || len == 0) {
223 errno = EIO;
224 return -1;
225 }
226 if (brix_json_arr_next(body, len, &cursor, &elem, &elemlen) != 1) {
227 errno = ENOENT;
228 return -1;
229 }
230 if (brix_json_get_str(elem, elemlen, "locality", loc, sizeof(loc)) != 1) {
231 errno = EIO;
232 return -1;
233 }
234 return sd_http_locality(loc, out);
235}

238/* sd_http_residency — vtable residency slot: classify `key` WITHOUT staging it.
239 *
240 * WHAT: NGX_OK with *out set, or NGX_ERROR with errno (ENOENT for a path the
241 * origin does not hold, EIO for an answer this build cannot read).
242 * WHY: Every protocol plane advertises tape state from this seam, and all of
243 * them must be able to say "on tape" without triggering a recall — which
244 * is exactly what `archiveinfo` is for.
245 * HOW: One POST, one JSON array read. Any non-200 is mapped by the shared
246 * WebDAV status verdict so a 401 here reads as EACCES exactly as it does
247 * on every other slot of this driver.
248 */
249static ngx_int_t
250sd_http_residency_auth(brix_sd_instance_t *inst, const char *key,
251 brix_sd_residency_t *out, const char *auth, const char *cert_pem)
252{
253 sd_http_inst_state *is;
254 brix_s3_resp_t resp;
255 char quoted[SD_HTTP_PATH_MAX + 8];
256 char body[SD_HTTP_TAPE_BODY_MAX];
257 const void *rb;
258 size_t rblen = 0;
259 int rc;
260
261 if (inst == NULL || inst->state == NULL || key == NULL || out == NULL) {
262 errno = EINVAL;
263 return NGX_ERROR;
264 }
265 is = inst->state;
266 if (is->tape_api[0] == '\0') {
267 errno = ENOTSUP;
268 return NGX_ERROR;
269 }
270 if (sd_http_json_quote(key, quoted, sizeof(quoted)) != 0) {
271 errno = ENAMETOOLONG;
272 return NGX_ERROR;
273 }
274 (void) snprintf(body, sizeof(body), "{\"paths\":[%s]}", quoted);
275
276 memset(&resp, 0, sizeof(resp));
277 if (sd_http_tape_post(is, "/archiveinfo", body, &resp, auth, cert_pem)
278 != 0)
279 {
280 return NGX_ERROR;
281 }
282 if (resp.status != 200) {
283 errno = sd_http_status_to_errno(resp.status);
284 is->transport->resp_free(&resp);
285 return NGX_ERROR;
286 }
287 rb = is->transport->resp_body(&resp, &rblen);
288 rc = sd_http_archiveinfo_parse(rb, rblen, out);
289 if (rc != 0) {
290 int e = errno;
291
292 is->transport->resp_free(&resp);
293 errno = e;
294 return NGX_ERROR;
295 }
296 is->transport->resp_free(&resp);
297 return NGX_OK;
298}

308/*
309 * WHAT: Submit the stage request for `key` and copy the origin's request id (or
310 * leave "" when it returned none).
311 * WHY: Split from the slot so recall's DECISION table — which residency classes
312 * stage, which do not — reads as one uninterrupted piece.
313 * HOW: 201 is the specified success; 200 and 202 are accepted too, because an
314 * implementation that queues and answers 202 has done precisely what was
315 * asked. The id is advisory: a stage that was accepted without one is
316 * still queued, so a missing `requestId` is not an error.
317 */
318static void
319sd_http_tape_stage(sd_http_inst_state *is, const char *quoted,
320 char reqid_out[40], const char *auth, const char *cert_pem)
321{
322 brix_s3_resp_t resp;
323 char body[SD_HTTP_TAPE_BODY_MAX];
324 const void *rb;
325 size_t rblen = 0;
326
327 (void) snprintf(body, sizeof(body), "{\"files\":[{\"path\":%s}]}", quoted);
328
329 memset(&resp, 0, sizeof(resp));
330 if (sd_http_tape_post(is, "/stage", body, &resp, auth, cert_pem) != 0) {
331 return;
332 }
333 if (resp.status == 200 || resp.status == 201 || resp.status == 202) {
334 rb = is->transport->resp_body(&resp, &rblen);
335 if (rb != NULL && rblen > 0) {
336 (void) brix_json_get_str(rb, rblen, "requestId", reqid_out, 40);
337 }
338 }
339 is->transport->resp_free(&resp);
340}

343/* sd_http_recall — vtable recall slot: bring `key` online, without blocking.
344 *
345 * WHAT: NGX_OK (already online — a normal cache fill follows), NGX_AGAIN
346 * (staging in flight; reqid_out carries the origin's request id, "" if it
347 * gave none), or NGX_ERROR with errno set.
348 * WHY: This is the verb sd_cache drives on every nearline miss. Returning
349 * NGX_AGAIN instead of stalling is the whole point: the open fails soft
350 * with EAGAIN and the plane above answers "staging, retry later" rather
351 * than holding a worker for the length of a tape mount.
352 * HOW: Residency first — asking the tape system to stage a file that is already
353 * on disk queues pointless MSS work on every cache miss of a resident file.
354 * Then, by class: ONLINE is done; NEARLINE stages, and a stage is ALWAYS
355 * NGX_AGAIN because the API is asynchronous by definition — a successful
356 * submit means "queued", never "done", and a failed submit is still
357 * "retry later" for an object that is genuinely not on disk either way;
358 * UNAVAILABLE is NGX_AGAIN with no id, which is what "temporarily
359 * unavailable" means to a client; LOST is a hard ENOENT, because no amount
360 * of retrying will produce a file the tape system has declared destroyed,
361 * and telling the client to keep waiting for it would be a lie.
362 */
363static ngx_int_t
364sd_http_recall_common(brix_sd_instance_t *inst, const char *key,
365 char reqid_out[40], const char *auth, const char *cert_pem)
366{
367 sd_http_inst_state *is;
368 brix_sd_residency_t res;
369 char quoted[SD_HTTP_PATH_MAX + 8];
370
371 if (reqid_out != NULL) {
372 reqid_out[0] = '\0';
373 }
374 if (sd_http_residency_auth(inst, key, &res, auth, cert_pem) != NGX_OK) {
375 return NGX_ERROR; /* errno already set (ENOENT/EACCES/…) */
376 }
377 if (res == BRIX_SD_RES_ONLINE) {
378 return NGX_OK; /* resident — a normal fill follows */
379 }
380 if (res == BRIX_SD_RES_LOST) {
381 errno = ENOENT;
382 return NGX_ERROR;
383 }
384 if (res == BRIX_SD_RES_OFFLINE) {
385 return NGX_AGAIN; /* no request to park on; retry later */
386 }
387
388 is = inst->state;
389 if (sd_http_json_quote(key, quoted, sizeof(quoted)) != 0) {
390 errno = ENAMETOOLONG;
391 return NGX_ERROR;
392 }
393 if (reqid_out != NULL) {
394 sd_http_tape_stage(is, quoted, reqid_out, auth, cert_pem);
395 }
396 return NGX_AGAIN; /* queued — park the open on reqid */
397}

400sd_http_recall(brix_sd_instance_t *inst, const char *key, char reqid_out[40])
401{
402 return sd_http_recall_common(inst, key, reqid_out, NULL, NULL);

405/* recall_cred (phase-107 C2) — the SAME classify-then-stage, with both tape
406 * REST POSTs (archiveinfo + stage) presenting the requesting user's bearer or
407 * x509 proxy instead of the instance's static Authorization line — the same
408 * gate + resolve_open_cred pair every mutation slot of this driver uses, so a
409 * deny-mode prestage is refused here rather than silently signed as the
410 * service (the confused-deputy rule). Both POSTs complete before return: the
411 * STAGING is asynchronous at the origin (the caller parks on reqid), but the
412 * requests themselves are synchronous round trips, so the borrowed cred is
413 * consumed inside the call and never retained — the sd.h copy rule binds only
414 * a driver that keeps issuing requests after it returns. */
415ngx_int_t
416sd_http_recall_cred(brix_sd_instance_t *inst, const char *key,
417 const brix_sd_cred_t *cred, char reqid_out[40])
418{
419 sd_http_inst_state *is = inst->state;
420 char open_auth[SD_HTTP_AUTH_MAX];
421 const char *cert;
422
423 if (sd_http_cred_gate(is, cred) != 0) {
424 return NGX_ERROR; /* errno = EACCES (set by gate) */
425 }
426 cert = sd_http_resolve_open_cred(is, cred, open_auth, sizeof(open_auth));
427 return sd_http_recall_common(inst, key, reqid_out,
428 open_auth[0] ? open_auth : NULL, cert);
429}

432/*
433 * WHAT: 1 iff `c` may appear in the operator's Tape REST API base path.
434 * WHY: The base is concatenated into a request line, so it must BE a path and
435 * nothing else: a stray CR/LF would split the request, and a '?' or '#'
436 * would make "/archiveinfo" part of a query or fragment rather than the
437 * endpoint we meant to address. Refusing the whole base on one bad byte
438 * (the caller leaves the instance un-armed, and the export keeps working
439 * as plain http) is the only safe answer — sanitising it would silently
440 * produce a base the operator never wrote and cannot see in the config.
441 * HOW: RFC 3986's unreserved set plus '/', which is every byte a path segment
442 * needs and nothing that can change what the request line means.
443 */
444static int
445sd_http_path_byte_ok(unsigned char c)
446{
447 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
448 || (c >= '0' && c <= '9') || strchr("/._~-", c) != NULL;
449}

452/*
453 * WHAT: Record the operator's Tape REST API base on the instance; 1 when the
454 * instance is armed for nearline, 0 when it is not.
455 * WHY: The caps bit is a contract the composing registry enforces, so the ONE
456 * condition that sets it belongs beside the code that uses the base — not
457 * spelled out again in the driver's create path where it could drift from
458 * what the path composer actually accepts.
459 * HOW: A base must be absolute (the composer concatenates it with "/stage")
460 * and every byte must pass sd_http_path_byte_ok; a relative or empty
461 * base leaves the instance un-armed rather than building request paths
462 * that would resolve against the export root. A trailing slash is dropped
463 * so the composed path never doubles it.
464 */
465int
466sd_http_tape_init(sd_http_inst_state *is, const char *base)
467{
468 size_t n;
469 size_t i;
470
471 if (base == NULL || base[0] != '/') {
472 return 0;
473 }
474 n = strlen(base);
475 for (i = 0; i < n; i++) {
476 if (!sd_http_path_byte_ok((unsigned char) base[i])) {
477 return 0;
478 }
479 }
480 while (n > 1 && base[n - 1] == '/') {
481 n--;
482 }
483 if (n < 2 || n >= sizeof(is->tape_api)) {
484 return 0;
485 }
486 memcpy(is->tape_api, base, n);
487 is->tape_api[n] = '\0';
488 return 1;
489}
