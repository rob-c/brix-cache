1/*
2 * sd_xroot_nearline.c — the nearline (tape/MSS) pair of the root:// origin
3 * driver: residency (is this path online?) and recall (bring it online).
4 *
5 * WHY THIS FILE EXISTS
6 * An xrootd origin fronting an HSM is the ONE nearline source we can drive
7 * entirely over the wire, because the protocol already spells both halves:
8 * kXR_stat's ASCII flag word carries kXR_offline for a path whose data is not
9 * currently on disk, and kXR_prepare with kXR_stage IS the stage verb, right
10 * down to returning a request id. Until this file existed the driver had
11 * neither slot, so a tape-backed root:// origin reported ONLINE for every path
12 * (brix_vfs_residency's answer for a driver with no residency model) and every
13 * first read of a migrated file blocked a worker for however long the origin's
14 * MSS took — minutes to hours — instead of parking the open and answering
15 * "staging, retry later" (HTTP 202 + Retry-After, S3 InvalidObjectState,
16 * root:// kXR_offline).
17 *
18 * OPT-IN, AND WHY IT MUST BE
19 * Neither slot is reachable unless the instance advertises
20 * BRIX_SD_CAP_NEARLINE, which brix_sd_xroot_create_origin sets only from
21 * cfg->nearline — the `nearline` param on the tier's store line. That is not
22 * timidity: CAP_NEARLINE is a CONTRACT, not a hint. tier_build refuses to
23 * compose a nearline backend without a cache tier in front (the recall target,
24 * §9.4), so auto-advertising it would have turned every existing plain root://
25 * origin config into a startup failure. An origin that is not tape-backed also
26 * never sets kXR_offline, so the operator declaring `nearline` on a disk origin
27 * gets correct-but-pointless behaviour rather than a wrong answer.
28 *
29 * SHAPE
30 * Both slots open a fresh per-op origin session (anonymous/service, or the
31 * caller's credential for the recall_cred twin), issue one request, and tear
32 * it down — the same per-op session shape every path-based op in sd_xroot_ns.c
33 * uses, and for the same reason: these are rare, out-of-band metadata calls, not
34 * a data path worth pooling a connection for.
35 */

43/* sd_xroot_residency — vtable residency slot: classify `key` WITHOUT staging it.
44 *
45 * WHAT: NGX_OK with *out set to BRIX_SD_RES_ONLINE or BRIX_SD_RES_NEARLINE, or
46 * NGX_ERROR with errno set (ENOENT for a path the origin does not have).
47 * WHY: The protocol planes advertise tape state from this seam — the HTTP Tape
48 * REST API, S3 x-amz-storage-class, root:// stat's own kXR_offline bit —
49 * and they must be able to say "on tape" without triggering a recall, so
50 * this is a pure read.
51 * HOW: One kXR_stat by NAME; kXR_offline in the reply's flag word means the
52 * data is not on disk. The base protocol distinguishes only present-and-
53 * online from present-and-offline, so we never report OFFLINE or LOST: a
54 * path the origin cannot describe at all is an error (ENOENT), not a
55 * residency class, and inventing LOST from a missing path would tell the
56 * tape REST API a file was destroyed when the origin merely 404'd. */
57static ngx_int_t
58sd_xroot_residency_as(brix_sd_instance_t *inst, const char *key,
59 brix_sd_residency_t *out, const brix_sd_cred_t *cred)
60{
61 sd_xroot_inst_state *is = inst->state;
62 brix_cache_origin_conn_t oc;
63 brix_cache_fill_t *t;
64 brix_cache_stat_out_t so;
65 int rc, e = 0;
66
67 if (out == NULL) {
68 errno = EINVAL;
69 return NGX_ERROR;
70 }
71 if (sd_xroot_session(is->conf, cred, &oc, &t, &e) != 0) {
72 errno = e;
73 return NGX_ERROR;
74 }
75 rc = brix_cache_origin_stat(t, &oc, key, &so);
76 e = errno;
77 brix_cache_origin_close(&oc);
78 free(t);
79 if (rc != 0) {
80 errno = e;
81 return NGX_ERROR;
82 }
83 *out = (so.flags & kXR_offline) ? BRIX_SD_RES_NEARLINE
84 : BRIX_SD_RES_ONLINE;
85 return NGX_OK;
86}

92 return sd_xroot_residency_as(inst, key, out, NULL);

95/* sd_xroot_recall — vtable recall slot: bring `key` online, without blocking.
96 *
97 * WHAT: NGX_OK (already online — the caller does a normal cache-fill),
98 * NGX_AGAIN (staging in flight; reqid_out carries the origin's request id,
99 * "" if it returned none — the cache tier parks the open on it), or
100 * NGX_ERROR with errno set.
101 * WHY: This is the verb sd_cache drives on every nearline miss. Returning
102 * NGX_AGAIN instead of stalling is the whole point: the open fails soft
103 * with EAGAIN and the protocol plane answers "staging, retry later"
104 * (SP5 §9.2) rather than holding a worker for the length of a tape mount.
105 * HOW: Residency first — an already-online path needs no prepare at all, and
106 * asking anyway would queue pointless MSS work on every cache miss of a
107 * resident file. Only a NEARLINE answer sends kXR_prepare(kXR_stage), and
108 * that is always NGX_AGAIN: prepare is asynchronous by definition, so a
109 * successful send means "queued", never "done". A prepare that FAILS is
110 * still NGX_AGAIN with an empty reqid rather than NGX_ERROR — the object
111 * is genuinely offline, the read cannot be served now either way, and
112 * EAGAIN ("retry later") describes that far better to a client than a hard
113 * error on an advisory hint. Both steps use their own short-lived session
114 * (see the file header). */
115static ngx_int_t
116sd_xroot_recall_common(brix_sd_instance_t *inst, const char *key,
117 char reqid_out[40], const brix_sd_cred_t *cred)
118{
119 sd_xroot_inst_state *is = inst->state;
120 brix_cache_origin_conn_t oc;
121 brix_cache_fill_t *t;
122 brix_sd_residency_t res;
123 int e = 0;
124
125 if (reqid_out != NULL) {
126 reqid_out[0] = '\0';
127 }
128 if (sd_xroot_residency_as(inst, key, &res, cred) != NGX_OK) {
129 return NGX_ERROR; /* errno already set (ENOENT/EIO/…) */
130 }
131 if (res == BRIX_SD_RES_ONLINE) {
132 return NGX_OK; /* resident — a normal fill follows */
133 }
134 if (sd_xroot_session(is->conf, cred, &oc, &t, &e) != 0) {
135 errno = e;
136 return NGX_ERROR;
137 }
138 (void) brix_cache_origin_prepare_stage(t, &oc, key, reqid_out);
139 brix_cache_origin_close(&oc);
140 free(t);
141 return NGX_AGAIN; /* queued — park the open on reqid */
142}

150/* recall_cred (phase-107 C2) — the SAME two-step (kXR_stat probe, then
151 * kXR_prepare(kXR_stage)) with BOTH origin sessions opened under the caller's
152 * credential (sd_xroot_session with a non-NULL cred — the ns_cred wrapper
153 * pattern), so a deny-mode prestage never reaches the origin as the service
154 * identity and the origin's own MSS fair-share sees the real requester. Both
155 * sessions are torn down before return — nothing outlives the call, so the
156 * borrowed cred is never retained (the sd.h copy rule binds only a driver
157 * whose recall keeps running after it returns). */

165/* sd_xroot_evict — vtable evict slot (phase-107 C2): forward the release
166 * upstream as kXR_prepare(optionX=kXR_evict), the wire's own "drop your online
167 * copy" verb — the origin's MSS keeps the durable copy, exactly the shape our
168 * OWN prepare plane now implements on the receiving side. Gated on the SAME
169 * operator declaration as recall/residency (CAP_NEARLINE from cfg->nearline):
170 * evict is top-dispatched with no capability walk, and on a plain disk origin
171 * the origin's copy is the ONLY copy — forwarding a release there would be a
172 * delete wearing a different verb, so it answers ENOTSUP instead (the sd.h
173 * contract) and the caller keeps its advisory no-op. Bytes released are
174 * unknowable over the wire (the reply carries none): *bytes_out stays 0.
175 * Same per-op session shape as the other slots in this file. */
176static ngx_int_t
177sd_xroot_evict_common(brix_sd_instance_t *inst, const char *path,
178 uint64_t *bytes_out, const brix_sd_cred_t *cred)
179{
180 sd_xroot_inst_state *is = inst->state;
181 brix_cache_origin_conn_t oc;
182 brix_cache_fill_t *t;
183 int rc, e = 0;
184
185 if (bytes_out != NULL) {
186 *bytes_out = 0;
187 }
188 if ((brix_sd_caps(inst) & BRIX_SD_CAP_NEARLINE) == 0) {
189 errno = ENOTSUP; /* disk origin: its copy is the only
190 * copy — never release it */
191 return NGX_ERROR;
192 }
193 if (sd_xroot_session(is->conf, cred, &oc, &t, &e) != 0) {
194 errno = e;
195 return NGX_ERROR;
196 }
197 rc = brix_cache_origin_prepare_evict(t, &oc, path);
198 e = errno;
199 brix_cache_origin_close(&oc);
200 free(t);
201 if (rc != 0) {
202 errno = e;
203 return NGX_ERROR;
204 }
205 return NGX_OK;
206}
