19/* Shared task-field init for BOTH pgread post sites (primary slot reuse and
20 * §1.1 offload). Everything except the reply routing: the caller sets the
21 * sec_* trio and the §1.2 pool_send eligibility to its own shape after this
22 * returns. The pool-send reporting flags are cleared here so a uring-path
23 * error completion (which skips both thread fns) can never see a previous
24 * request's stale values in a reused slot task. */
25static void
26brix_pgread_task_init(brix_pgread_aio_t *t, brix_ctx_t *ctx,
27 ngx_connection_t *c, brix_pgread_run_t *run)
28{
29 t->c = c;
30 t->ctx = ctx;
31 t->fd = run->fd;
32 t->handle_idx = run->idx;
33 t->offset = (off_t) run->offset;
34 t->rlen = run->rlen;
35 t->scratch = run->scratch;
36 t->out_size = 0;
37 t->streamid[0] = ctx->recv.cur_streamid[0];
38 t->streamid[1] = ctx->recv.cur_streamid[1];
39 t->obj = ctx->files[run->idx].sd_obj; /* Layer 3: driver obj (or zeroed) */
40 t->start_ns = brix_phase_now_ns(); /* phase-56 D-2 */
41 t->counted = 1; /* single-shot pgread — the per-slot
42 * task may still hold a stale 0 from
43 * a kXR_read windowed post */
44 t->pool_sent = 0;
45 t->pool_sent_all = 0;
46 t->pool_token_held = 0;
47 t->pool_send_errno = 0;
48 t->pool_chunked = 0;
49 t->chunk_error = 0;
50 t->pool_image_len = 0;
51 t->pool_frames = 0;
52}

54/*
55 * brix_pgread_post_aio - offload the read to the thread pool.
56 *
57 * WHAT: Fills the connection's reusable pgread AIO task from `run` and posts
58 * it to the configured thread pool. Returns NGX_OK when posted (the
59 * completion handler sends the response), NGX_DECLINED when the post
60 * failed (caller falls back to the sync path), NGX_ERROR on
61 * allocation failure.
62 *
63 * WHY: The task-population boilerplate is one nameable step of the handler;
64 * extracting it keeps the orchestrator flat (coding-standards §8).
65 *
66 * HOW: Uses the rd_pool slot task backing run->scratch (phase-32 WS3
67 * discipline, shared with read_post_aio): each in-flight pgread carries
68 * an independent task struct, so several can run on worker threads at
69 * once without a later post clobbering a task a worker still owns. A
70 * posted task is counted in rd.aio_inflight (t->counted) so teardown
71 * defers and the recv loop's pipelining/backpressure bounds hold.
72 */
73static ngx_int_t
74brix_pgread_post_aio(brix_ctx_t *ctx, ngx_connection_t *c,
75 ngx_stream_brix_srv_conf_t *rconf, brix_pgread_run_t *run)
76{
77 ngx_thread_task_t *task = NULL;
78 brix_pgread_aio_t *t;
79 ngx_flag_t posted;
80 ngx_uint_t i;
81
82 for (i = 0; i < ctx->out.pipeline_depth; i++) {
83 if (ctx->rd.pool[i].buf == run->scratch) {
84 task = ctx->rd.pool[i].task;
85 if (task == NULL) {
86 /* Sized for either pipelined read opcode — kXR_read posts
87 * share the per-slot task (see brix_rd_slot_aio_u, aio.h). */
88 task = ngx_thread_task_alloc(c->pool,
89 sizeof(brix_rd_slot_aio_u));
90 if (task == NULL) {
91 return NGX_ERROR;
92 }
93 ctx->rd.pool[i].task = task;
94 } else {
95 task->next = NULL;
96 task->event.complete = 0;
97 }
98 break;
99 }
100 }
101
102 /* A scratch that is not a pool slot cannot pipeline safely (no per-slot
103 * task to bind); decline so the caller serves synchronously instead. */
104 if (task == NULL) {
105 return NGX_DECLINED;
106 }
107
108 t = task->ctx;
109 brix_pgread_task_init(t, ctx, c, run);
110 t->sec_c = NULL; /* primary-stream reply (a reused slot
111 * task may hold stale offload fields) */
112 t->sec_ctx = NULL;
113 t->sec_counted = 0;
114 t->pool_send = 0; /* §1.2: never pool-send on the
115 * primary path (reused slot task) */
116
117 brix_task_bind(task, brix_pgread_aio_thread, brix_pgread_aio_done);
118
119 (void) brix_aio_post_task(ctx, c, rconf->common.thread_pool, task,
120 "brix: thread_task_post failed, sync pgread fallback",
121 &posted);
122
123 /* Only a posted task runs on a worker thread, so only then does its
124 * completion decrement rd.aio_inflight — count it here so disconnect
125 * defers teardown until the worker releases the rd_pool buffer, and so
126 * the recv loop's in-flight bounds see this read (mirrors read_post_aio). */
127 if (posted) {
128 ctx->rd.aio_inflight++;
129 }
130
131 return posted ? NGX_OK : NGX_DECLINED;
132}

134/*
135 * brix_pgread_sync_fill - blocking fallback read into the wire buffer.
136 *
137 * WHAT: Runs the pgread VFS job inline (read directly into the gapped wire
138 * buffer, CRC each page in place — no flat-buffer copy). On success
139 * fills run->{out_buf, out_size, flat_buf}, updates run->rlen to the
140 * bytes actually read (accounting), and returns 0; on I/O error
141 * returns the job's errno for the caller's error triplet.
142 *
143 * WHY: Same code path as the AIO worker, kept as a pure produce-or-errno
144 * step so the handler owns the wire error response (side effects at
145 * the edges, coding-standards §8).
146 *
147 * HOW: Skipped by the caller when the warm-cache fast path already produced
148 * the encoding; runs when no thread pool is configured or the post
149 * failed.
150 */
151static int
152brix_pgread_sync_fill(brix_ctx_t *ctx, brix_pgread_run_t *run)
153{
154 brix_vfs_job_t job;
155
156 brix_vfs_job_read_init(&job, run->fd, (off_t) run->offset, run->rlen,
157 run->scratch, run->rlen, 0);
158 job.op = BRIX_VFS_IO_PGREAD;
159 brix_vfs_job_set_obj(&job, &ctx->files[run->idx].sd_obj);
160 brix_vfs_io_execute(&job);
161
162 if (job.io_errno != 0) {
163 return job.io_errno;
164 }
165
166 run->out_size = job.out_size;
167 run->flat_buf = run->scratch;
168 run->out_buf = run->scratch; /* output starts at offset 0 now */
169 run->rlen = (size_t) job.nio; /* actual bytes read (accounting) */
170 return 0;
171}

173/*
174 * brix_pgread_send_response - frame, account, log, and queue the reply.
175 *
176 * WHAT: Builds the kXR_status(4007) response chain over the produced output,
177 * charges the byte accounting / bandwidth limiter, writes the access
178 * log line, and queues the chain. Returns the queue rc (NGX_ERROR on
179 * framing failure), releasing the read buffer on any non-send outcome.
180 *
181 * WHY: Response assembly is the handler's final nameable step; the order of
182 * framing → accounting → log → metric → queue is frozen (byte-identical
183 * wire output and log lines).
184 *
185 * HOW: run->rlen here is the actual byte count (the sync path overwrote it
186 * with job.nio; the warm path read exactly rlen). The buffer is kept
187 * only while the send is in flight (state XRD_ST_SENDING).
188 */
189static ngx_int_t
190brix_pgread_send_response(brix_ctx_t *ctx, ngx_connection_t *c,
191 ngx_stream_brix_srv_conf_t *rconf, brix_pgread_run_t *run)
192{
193 ngx_chain_t *rsp_chain;
194 char detail[64];
195 ngx_int_t rc;
196
197 rsp_chain = brix_build_pgread_chain(ctx, c, run->offset, run->out_buf,
198 (uint32_t) run->out_size);
199 if (rsp_chain == NULL) {
200 brix_release_read_buffer(ctx, c, run->flat_buf);
201 return NGX_ERROR;
202 }
203
204 ctx->files[run->idx].bytes_read += run->rlen;
205 ctx->totals.bytes += run->rlen;
206 brix_rl_charge_ctx(ctx, run->rlen); /* Phase 25 bandwidth */
207
208 if (rconf->access_log_fd != NGX_INVALID_FILE) {
209 snprintf(detail, sizeof(detail), "%lld+%zu",
210 (long long) run->offset, run->rlen);
211 brix_log_access(ctx, c, "PGREAD", ctx->files[run->idx].path,
212 detail, 1, 0, NULL, run->rlen);
213 }
214 BRIX_OP_OK(ctx, BRIX_OP_PGREAD);
215
216 /* Self-contained frame (per-response palloc'd header, data in this
217 * request's own rd_pool slot): if it parks, the next pgread may safely
218 * queue behind it while it drains (brix_recv_try_pipeline_read). */
219 ctx->out.resp_pipelinable = 1;
220
221 rc = brix_queue_response_chain(ctx, c, rsp_chain, run->flat_buf);
222 if (rc != NGX_OK || ctx->state != XRD_ST_SENDING) {
223 brix_release_read_buffer(ctx, c, run->flat_buf);
224 }
225 return rc;
226}

228/*
229 * brix_pgread_post_aio_offload — post a large pgread to the thread pool with
230 * the reply targeted at a bound same-worker SECONDARY channel (§1.1).
231 *
232 * WHAT: The offload analog of brix_pgread_post_aio: binds the SECONDARY's
233 * per-slot task (fbuf is its rd_pool slot; the encode region starts 32
234 * bytes in, after the pgRead status header the done handler stamps),
235 * fills it from `run`, and posts. Counted in BOTH connections'
236 * rd.aio_inflight — the primary owns the request/recv state, the
237 * secondary owns the buffer and the out-ring slot the reply will take —
238 * so either side's teardown defers until the completion runs.
239 *
240 * WHY: One TCP socket tops out well below what a striped -S4 client can
241 * sink; routing each large pgread's reply over the substream the client
242 * asked for (pathid) engages every bound socket in parallel, which is
243 * exactly how XRootD's do_Offload wins multi-stream benchmarks.
244 *
245 * HOW: Returns NGX_OK when posted (brix_pgread_aio_done routes to the
246 * offload epilogue via t->sec_c) or NGX_DECLINED when no per-slot task
247 * could be bound / the pool refused — the caller releases fbuf and falls
248 * through to the primary producer paths.
249 */
250static ngx_int_t
251brix_pgread_post_aio_offload(brix_ctx_t *ctx, ngx_connection_t *c,
252 ngx_stream_brix_srv_conf_t *rconf, brix_pgread_run_t *run,
253 brix_ctx_t *sec_ctx, ngx_connection_t *sec_c, u_char *fbuf)
254{
255 ngx_thread_task_t *task = NULL;
256 brix_pgread_aio_t *t;
257 ngx_flag_t posted;
258 ngx_uint_t i;
259
260 for (i = 0; i < sec_ctx->out.pipeline_depth; i++) {
261 if (sec_ctx->rd.pool[i].buf == fbuf) {
262 task = sec_ctx->rd.pool[i].task;
263 if (task == NULL) {
264 task = ngx_thread_task_alloc(sec_c->pool,
265 sizeof(brix_rd_slot_aio_u));
266 if (task == NULL) {
267 return NGX_DECLINED;
268 }
269 sec_ctx->rd.pool[i].task = task;
270 } else {
271 task->next = NULL;
272 task->event.complete = 0;
273 }
274 break;
275 }
276 }
277
278 if (task == NULL) {
279 return NGX_DECLINED;
280 }
281
282 t = task->ctx;
283 brix_pgread_task_init(t, ctx, c, run); /* run->scratch = fbuf + hdr off */
284 t->sec_c = sec_c;
285 t->sec_ctx = sec_ctx;
286 t->sec_counted = 1;
287
288 /* §1.2 pool-send eligibility: any cleartext secondary (a worker thread
289 * cannot enter the TLS filter, invariant #2). Ring state is NOT checked
290 * here — the thread's send-time gate (token CAS + send_busy) is the
291 * authoritative throttle; a post-time out.count check would cascade (one
292 * parked frame routes every task posted during its drain to the event
293 * loop, which keeps the ring busy, which extends the cascade). */
294 t->pool_send = (sec_c->ssl == NULL) ? 1 : 0;
295 if (t->pool_send) {
296 sec_ctx->out.pool_send_active = 1; /* engage the send-token gates */
297 }
298
299 brix_task_bind(task, brix_pgread_aio_thread, brix_pgread_aio_done);
300
301 (void) brix_aio_post_task(ctx, c, rconf->common.thread_pool, task,
302 "brix: thread_task_post failed, primary pgread fallback",
303 &posted);
304
305 if (posted) {
306 ctx->rd.aio_inflight++;
307 sec_ctx->rd.aio_inflight++;
308 }
309
310 return posted ? NGX_OK : NGX_DECLINED;
311}

313/*
314 * brix_pgread_try_offload - §1.1 response offloading (do_Offload parity) for
315 * kXR_pgread: the pgread analog of brix_read_try_offload (read.c) — see there
316 * for the full safety rationale. pgread already VALIDATES its pathid in
317 * brix_pgread_parse_validate (§1.2); this routes an eligible response over the
318 * bound secondary. The pgread reply is [32B kXR_status frame | CRC-interleaved
319 * page data], so it is the same secondary-owned flat-buffer shape as read/readv,
320 * only with a 32-byte header built by brix_build_pgread_status instead of 8.
321 *
322 * When the pathid names a same-worker, quiescent secondary and the encoded reply
323 * fits one streaming window: acquire the frame buffer from the SECONDARY's pool
324 * (32B header + the gapped [CRC32c][page] encoding), encode the pages straight
325 * into buf+32 via the shared brix_pgread_sync_fill, stamp the kXR_status frame
326 * (carrying the PRIMARY request's streamid) into buf[0..32), and queue it on the
327 * secondary's out-ring — acquire+release both on the secondary's ctx, no cross-
328 * connection lifetime tangle. Returns 1 when routed (*rc set), 0 to fall through
329 * to the normal primary-stream producer paths (warm / AIO / sync).
330 */
331static ngx_flag_t
332brix_pgread_try_offload(brix_ctx_t *ctx, ngx_connection_t *c,
333 ngx_stream_brix_srv_conf_t *rconf, brix_pgread_run_t *run, ngx_int_t *rc)
334{
335 ngx_connection_t *sec_c;
336 brix_ctx_t *sec_ctx;
337 size_t n_pages_max, enc_max, hdr_space;
338 u_char *buf;
339 int io_errno;
340
341 sec_ctx = brix_read_offload_secondary(ctx, c, run->pathid, &sec_c);
342 if (sec_ctx == NULL) {
343 return 0;
344 }
345
346 /* Encoded-reply upper bound — the gapped [CRC32c(4)][page] size, identical
347 * to brix_pgread_scratch_size's math. A reply larger than one streaming
348 * window belongs on the (thread-pool-capable) primary path. */
349 n_pages_max = ((size_t) (run->offset & (kXR_pgPageSZ - 1)) + run->rlen
350 + kXR_pgPageSZ - 1) / kXR_pgPageSZ;
351 if (n_pages_max == 0) {
352 n_pages_max = 1;
353 }
354 enc_max = run->rlen + n_pages_max * BRIX_PG_CKSZ;
355 if (enc_max > (size_t) BRIX_READ_WINDOW) {
356 /* Large reply: too big for the inline sync fill (it would stall the
357 * event loop for milliseconds), but exactly what the thread pool is
358 * for — post the normal pgread AIO with the response targeted at the
359 * SECONDARY (multi-socket parallelism for striped -S clients). Any
360 * ineligibility falls through to the unchanged primary paths. */
361 if (rconf->common.thread_pool == NULL) {
362 return 0;
363 }
364 /* §1.3 chunked streaming lays one status header per chunk frame
365 * back-to-back in this buffer — size for the worst-case train. */
366 hdr_space = sizeof(ServerStatusResponse_pgRead)
367 * (1 + (run->rlen + BRIX_PGREAD_STREAM_CHUNK - 1)
368 / BRIX_PGREAD_STREAM_CHUNK);
369 if (!brix_budget_admit(ctx, rconf->memory_budget,
370 hdr_space + enc_max)) {
371 return 0; /* over budget — the primary path issues the kXR_wait */
372 }
373 buf = brix_acquire_read_buffer(sec_ctx, sec_c, hdr_space + enc_max);
374 if (buf == NULL) {
375 return 0; /* secondary pool exhausted — primary path serves it */
376 }
377 brix_budget_sync(sec_ctx); /* charge the (possibly grown) slot */
378 run->scratch = buf + sizeof(ServerStatusResponse_pgRead);
379 if (brix_pgread_post_aio_offload(ctx, c, rconf, run, sec_ctx, sec_c,
380 buf) != NGX_OK)
381 {
382 brix_release_read_buffer(sec_ctx, sec_c, buf);
383 run->scratch = NULL;
384 return 0;
385 }
386 *rc = NGX_OK;
387 return 1;
388 }
389
390 /* Frame buffer from the SECONDARY's pool: 32B status frame + encoding. */
391 buf = brix_acquire_read_buffer(sec_ctx, sec_c,
392 sizeof(ServerStatusResponse_pgRead) + enc_max);
393 if (buf == NULL) {
394 return 0; /* secondary pool exhausted — fall back rather than fail */
395 }
396
397 /* Encode the pages straight into the frame's data region [32 .. 32+enc). */
398 run->scratch = buf + sizeof(ServerStatusResponse_pgRead);
399 io_errno = brix_pgread_sync_fill(ctx, run); /* fills out_size + rlen(=nio) */
400 if (io_errno != 0) {
401 /* I/O failure: the secondary wire is untouched, so the error rides the
402 * PRIMARY control stream exactly like the normal path. */
403 brix_release_read_buffer(sec_ctx, sec_c, buf);
404 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
405 *rc = brix_send_error(ctx, c, kXR_IOError, strerror(io_errno));
406 return 1;
407 }
408
409 /* Accounting stays on the PRIMARY ctx (the request's owner). */
410 ctx->files[run->idx].bytes_read += run->rlen;
411 ctx->totals.bytes += run->rlen;
412 brix_rl_charge_ctx(ctx, run->rlen);
413 if (rconf->access_log_fd != NGX_INVALID_FILE) {
414 char detail[64];
415 snprintf(detail, sizeof(detail), "%lld+%zu",
416 (long long) run->offset, run->rlen);
417 brix_log_access(ctx, c, "PGREAD", ctx->files[run->idx].path,
418 detail, 1, 0, NULL, run->rlen);
419 }
420 BRIX_OP_OK(ctx, BRIX_OP_PGREAD);
421
422 /* Stamp the kXR_status frame (PRIMARY streamid) into the 32-byte header, then
423 * queue the contiguous [status|data] frame on the SECONDARY (it owns buf). */
424 brix_build_pgread_status(ctx, run->offset, (uint32_t) run->out_size,
425 (ServerStatusResponse_pgRead *) buf);
426 *rc = brix_queue_response_base(sec_ctx, sec_c, buf,
427 sizeof(ServerStatusResponse_pgRead)
428 + run->out_size, buf);
429 if (*rc == NGX_OK) {
430 brix_metric_offload(BRIX_PROTO_ROOT); /* §1.1 observability */
431 }
432
433 if (*rc != NGX_OK || sec_ctx->out.count == 0) {
434 brix_release_read_buffer(sec_ctx, sec_c, buf);
435 }
436 return 1;
437}

439/*
440 * brix_handle_pgread - kXR_pgread orchestrator.
441 *
442 * WHAT: Decodes and validates the request, sizes the wire buffer, then tries
443 * the producer paths in order — warm-cache inline, thread-pool offload,
444 * sync fallback — and frames whatever output was produced.
445 *
446 * WHY: Flat sequence of named steps per coding-standards §8; the complexity
447 * lives in the helpers above.
448 *
449 * HOW: run.{out_buf, flat_buf, out_size} start NULL/0 (phase-72.A) and are
450 * set only by a producer; the out_buf==NULL guard before framing makes
451 * the exactly-one-producer invariant enforceable.
452 */
453ngx_int_t
454brix_handle_pgread(brix_ctx_t *ctx, ngx_connection_t *c)
455{
456 /* phase-42 W4 invariant: pgread is ALWAYS plaintext — it never consults
457 * ctx->files[idx].read_codec. Inline read compression is a kXR_read-only
458 * handle property; pgread's kXR_status(4007) framing + per-page CRC32c must
459 * stay byte-for-byte intact, so compression is deliberately not applied here. */
460 brix_pgread_run_t run;
461 ngx_stream_brix_srv_conf_t *rconf;
462 ngx_int_t rc;
463 ngx_flag_t warm_hit;
464
465 ngx_memzero(&run, sizeof(run)); /* out_buf/flat_buf NULL, out_size 0 */
466
467 if (!brix_pgread_parse_validate(ctx, c, &run, &rc)) {
468 return rc;
469 }
470
471 rconf = ngx_stream_get_module_srv_conf(
472 (ngx_stream_session_t *) c->data, ngx_stream_brix_module);
473
474 /*
475 * §1.1 response offloading: route an eligible pgread reply over the bound
476 * secondary channel (the pathid was already validated in parse_validate).
477 * Ineligible requests (pathid 0, cross-worker, busy secondary, large) fall
478 * through to the normal warm/AIO/sync producer paths below, unchanged.
479 */
480 if (brix_pgread_try_offload(ctx, c, rconf, &run, &rc)) {
481 return rc;
482 }
483
484 /*
485 * Windowed streaming (pgread_window.c): a primary-path request larger
486 * than one window is served as a kXR_PartialResult train produced
487 * window-by-window into the hot read_scratch slot instead of a
488 * request-sized rd_pool fill — LLC-hot copy destination, one window of
489 * memory budget, first frame on the wire after one window. Offload-
490 * eligible requests were consumed above, so everything reaching this
491 * gate rides the primary stream.
492 */
493 if (run.rlen > BRIX_PGREAD_WARM_INLINE_MAX) {
494 return brix_pgread_serve_windowed(ctx, c, rconf, &run);
495 }
496
497 /* Memory-budget admission (phase-31 W4, parity with read_serve_buffered):
498 * this request will hold an rd_pool buffer of the encoded size until its
499 * response drains; over budget it is deferred with kXR_wait. */
500 if (!brix_budget_admit(ctx, rconf->memory_budget,
501 brix_pgread_scratch_size(&run))) {
502 return brix_fsoverload_backoff(ctx, c, rconf);
503 }
504
505 /*
506 * Per-in-flight gapped wire buffer (pgread pipelining): each outstanding
507 * pgread encodes into its OWN rd_pool slot rather than the shared
508 * read_scratch, so this response can keep draining while the recv loop
509 * admits the next pgread into a different buffer (and a pool thread reads
510 * it concurrently). Released back to the pool when the response's
511 * out_ring slot drains, or below on the error paths.
512 */
513 run.scratch = brix_acquire_read_buffer(ctx, c,
514 brix_pgread_scratch_size(&run));
515 if (run.scratch == NULL) {
516 /* Pool exhausted despite the recv-side depth bound — slot accounting
517 * has gone wrong upstream. Never fatal: a bare NGX_ERROR here tears
518 * down a healthy pipelined connection (silent RST, client re-login
519 * storm); defer the request with the standard overload backoff. */
520 ngx_log_error(NGX_LOG_ERR, c->log, 0,
521 "brix: pgread rd_pool exhausted (depth %ui, "
522 "rd_inflight %ui, aio_inflight %ui) — deferring",
523 ctx->out.pipeline_depth, ctx->rd.inflight,
524 ctx->rd.aio_inflight);
525 return brix_fsoverload_backoff(ctx, c, rconf);
526 }
527 brix_budget_sync(ctx); /* charge the (possibly grown) slot promptly */
528
529 /* Large requests skip the inline warm probe even when resident: posting
530 * to the pool overlaps this request's read+CRC with the previous
531 * response's socket writes (see BRIX_PGREAD_WARM_INLINE_MAX). */
532 warm_hit = run.rlen <= BRIX_PGREAD_WARM_INLINE_MAX
533 ? brix_pgread_try_warm(ctx, rconf, &run) : 0;
534
535 if (!warm_hit && rconf->common.thread_pool != NULL) {

543 }

545 }

554 kXR_IOError, strerror(io_errno));
555 }
556 }

559 * fallback) must have filled the output; the AIO path returned above. */
560 if (run.out_buf == NULL) {
561 brix_release_read_buffer(ctx, c, run.scratch);
562 BRIX_RETURN_ERR(ctx, c, BRIX_OP_PGREAD, "PGREAD",

564 kXR_ServerError, "pgread: no output produced");
565 }
566
567 return brix_pgread_send_response(ctx, c, rconf, &run);
