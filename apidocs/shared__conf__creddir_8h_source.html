20/*
21 * brix_shared_credential_dir_ensure() — Config-time guarantee for
22 * brix_storage_credential_dir: create the directory 0700 (chown'd to the
23 * worker user when the master runs as root) if it is missing, and shout —
24 * NGX_LOG_WARN at parse time — when it cannot be created, is not a
25 * directory, is group/other-accessible, or is owned by a user the workers
26 * do not run as.
27 *
28 * WHY: The store holds delegated PRIVATE KEYS, and the default lives on
29 * tmpfs (/dev/shm) precisely so nothing persists across reboots or lands in
30 * backups — but /dev/shm is world-writable (1777), so the entire security
31 * boundary is this directory's 0700 mode + ownership, which must therefore
32 * be enforced here rather than left to operator setup. A broken or
33 * foreign-owned path must NOT kill startup (the store may be unused, and
34 * fallback=allow keeps requests on the service credential), but the admin
35 * must be told that credential delegation will not work until it is fixed.
36 *
37 * HOW: stat(); on ENOENT mkdir(0700) + chown to ccf->user when euid==0
38 * (mirroring ngx_create_paths). Every failure and every unsafe pre-existing
39 * state warns via ngx_conf_log_error and returns — never fatal. Called from
40 * every shared merge (per server/location), so a broken path is shouted per
41 * context; the healthy path prints nothing and costs one stat().
42 */

50/* The uid/gid worker-writable provisioned dirs must be handed to: the runtime
51 * worker identity when the master runs as root, else the invoking user. */
52static inline void
53brix_shared_worker_dir_ids(ngx_conf_t *cf, uid_t *uid_out, gid_t *gid_out)
54{
55 ngx_core_conf_t *ccf = (ngx_core_conf_t *)
56 ngx_get_conf(cf->cycle->conf_ctx, ngx_core_module);
57
58 *uid_out = geteuid();
59 *gid_out = (gid_t) -1;
60 if (geteuid() == 0) {
61 (void) brix_imp_worker_runtime_ids(
62 (ccf != NULL) ? ccf->user : (ngx_uid_t) NGX_CONF_UNSET_UINT,
63 (ccf != NULL) ? (ngx_gid_t) ccf->group
64 : (ngx_gid_t) NGX_CONF_UNSET_UINT,
65 uid_out, gid_out); /* on failure the invoking-root ids stay */
66 }
67}

69/* Rewrite the COMPILED default store path to its worker-uid-scoped form
70 * (BRIX_CREDENTIAL_DIR_DEFAULT ".<uid>", the cred_stage.c convention). Two
71 * services sharing one host — the distro's own www-data nginx and a user's
72 * test fleet both defaulting to the same 0700 tmpfs dir — otherwise fight
73 * over ownership: whichever creates it first locks every other identity out
74 * of credential delegation (seen live: a www-data-owned /dev/shm/brix-creds
75 * broke delegation for the whole unprivileged test lane). An operator's
76 * EXPLICIT brix_storage_credential_dir is never touched. */
77static inline void
78brix_shared_credential_dir_default_scope(ngx_conf_t *cf, ngx_str_t *dir)
79{
80 uid_t want_uid;
81 gid_t want_gid;
82 u_char *p;
83
84 if (dir->len != sizeof(BRIX_CREDENTIAL_DIR_DEFAULT) - 1
85 || ngx_strncmp(dir->data, BRIX_CREDENTIAL_DIR_DEFAULT, dir->len) != 0)
86 {
87 return; /* operator-chosen path — leave it alone */
88 }
89 brix_shared_worker_dir_ids(cf, &want_uid, &want_gid);
90 p = ngx_pnalloc(cf->pool,
91 sizeof(BRIX_CREDENTIAL_DIR_DEFAULT) + NGX_INT64_LEN + 2);
92 if (p == NULL) {
93 return; /* keep the shared default; ensure still runs */
94 }
95 dir->len = ngx_sprintf(p, BRIX_CREDENTIAL_DIR_DEFAULT ".%d",
96 (int) want_uid) - p;
97 dir->data = p;
98 p[dir->len] = '\0'; /* conf strings are read as C strings below */
99}

101static inline void
102brix_shared_credential_dir_ensure(ngx_conf_t *cf, const ngx_str_t *dir)
103{
104 struct stat st;
105 const char *path;
106 uid_t want_uid;
107 gid_t want_gid;
108
109 if (dir == NULL || dir->len == 0) {
110 return; /* explicit "" = per-user store disabled */
111 }
112
113 path = (const char *) dir->data; /* conf tokens are NUL-terminated */
114 brix_shared_worker_dir_ids(cf, &want_uid, &want_gid);
115
116 if (stat(path, &st) != 0) {
117 if (errno != ENOENT) {
118 ngx_conf_log_error(NGX_LOG_WARN, cf, errno,
119 "brix: credential store \"%s\" is not accessible — "
120 "credential delegation will not work until "
121 "brix_storage_credential_dir is fixed", path);
122 return;
123 }
124
125 if (mkdir(path, 0700) != 0 && errno != EEXIST) {
126 ngx_conf_log_error(NGX_LOG_WARN, cf, errno,
127 "brix: cannot create credential store \"%s\" — "
128 "credential delegation will not work until "
129 "brix_storage_credential_dir is fixed", path);
130 return;
131 }
132
133 if (stat(path, &st) != 0) {
134 ngx_conf_log_error(NGX_LOG_WARN, cf, errno,
135 "brix: credential store \"%s\" vanished after create — "
136 "credential delegation will not work", path);
137 return;
138 }
139
140 /* The master parses config as root but the workers write the store
141 * as the RUNTIME worker identity (the `user` account, or the
142 * de-escalation target for a root-capable worker) — hand the fresh
143 * directory to them, as ngx_create_paths does for the temp paths. */
144 if (geteuid() == 0 && st.st_uid != want_uid
145 && chown(path, want_uid, want_gid) != 0)
146 {
147 ngx_conf_log_error(NGX_LOG_WARN, cf, errno,
148 "brix: cannot chown credential store \"%s\" to the worker "
149 "user — credential delegation will not work", path);
150 }
151 return;
152 }
153
154 if (!S_ISDIR(st.st_mode)) {
155 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
156 "brix: credential store \"%s\" is not a directory — "
157 "credential delegation will not work until "
158 "brix_storage_credential_dir is fixed", path);
159 return;
160 }
161
162 if ((st.st_mode & 0077) != 0) {
163 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
164 "brix: credential store \"%s\" is group/other-accessible "
165 "(mode %04uo) — delegated private keys may be exposed; "
166 "chmod 0700", path, (ngx_uint_t) (st.st_mode & 07777));
167 }
168
169 if (st.st_uid != want_uid) {
170 ngx_conf_log_error(NGX_LOG_WARN, cf, 0,
171 "brix: credential store \"%s\" is owned by uid %ud but the "
172 "workers run as uid %ud — credential delegation will not work "
173 "until ownership or brix_storage_credential_dir is fixed",
174 path, (ngx_uint_t) st.st_uid, (ngx_uint_t) want_uid);
175 }
176}
