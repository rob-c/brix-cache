24/*
25 * Confinement (Phase 8): the temp file and its final destination always live
26 * under root_canon (brix_make_tmp_path derives the temp name next to
27 * final_path, which the caller already confined to the export root). We open a
28 * kernel-confinement rootfd on root_canon and route the temp create / rename /
29 * unlink through the beneath API so the operation is bounded by the kernel
30 * regardless of how the caller derived the path. A path that does not strip
31 * cleanly under root_canon is refused (EXDEV) rather than touched raw.
32 */
33
34/*
35 * WHAT: Open a unique temporary file inside the confined root_canon for atomic write.
36 *
37 * WHY: S3 PUT, WebDAV PUT, and other operations need to write data safely without risking
38 * corruption of the final path if the process crashes mid-write. A temp file with O_EXCL
39 * guarantees atomicity — either the rename succeeds (final path appears) or it doesn't
40 * (temp file remains for cleanup).
41 *
42 * HOW: Generate a unique tmp_path via brix_make_tmp_path(). Open with
43 * open_flags | O_CREAT | O_EXCL inside root_canon via brix_open_confined_canon().
44 * Loop up to 'attempts' times (default 16) on EEXIST. On success: set staged->active=1,
45 * store fd and tmp_path, return NGX_OK. On non-EEXIST failure or exhaustion: errno set,
46 * return NGX_ERROR.
47 *
48 * Parameters:
49 * log — nginx log for error reporting
50 * req — request description: root_canon / final_path / open_flags / mode /
51 * attempts (see brix_staged_open_req_t)
52 * staged — output struct: fd, tmp_path, active flag populated on success
53 */
54ngx_int_t
55brix_staged_open(ngx_log_t *log, const brix_staged_open_req_t *req,
56 brix_staged_file_t *staged)
57{
58 ngx_uint_t attempts;
59 ngx_uint_t i;
60 int rootfd;
61
62 (void) log;
63
64 if (staged == NULL || req == NULL || req->final_path == NULL) {
65 errno = EINVAL;
66 return NGX_ERROR;
67 }
68
69 ngx_memzero(staged, sizeof(*staged));
70 staged->fd = NGX_INVALID_FILE;
71 /* SECURITY: the temp is created PRIVATE (0600) so another mapped uid on a
72 * shared filesystem cannot read an in-progress upload; the caller's intended
73 * final mode is restored at commit (staged_commit_internal). */
74 staged->final_mode = req->mode;
75
76 attempts = req->attempts;
77 if (attempts == 0) {
78 attempts = 16;
79 }
80
81 rootfd = brix_beneath_open_root(req->root_canon);
82 if (rootfd < 0) {
83 return NGX_ERROR;
84 }
85
86 for (i = 0; i < attempts; i++) {
87 const char *rel;
88
89 if (brix_make_tmp_path(req->final_path, staged->tmp_path,
90 sizeof(staged->tmp_path)) != NGX_OK)
91 {
92 errno = ENAMETOOLONG;
93 close(rootfd);
94 return NGX_ERROR;
95 }
96
97 rel = brix_beneath_strip_root(req->root_canon, staged->tmp_path);
98 if (rel == NULL) {
99 errno = EXDEV;
100 close(rootfd);
101 return NGX_ERROR;
102 }
103
104 staged->fd = brix_open_beneath(rootfd, rel,
105 req->open_flags | O_CREAT | O_EXCL,
106 0600);
107 if (staged->fd != NGX_INVALID_FILE) {
108 staged->active = 1;
109 close(rootfd);
110 return NGX_OK;
111 }
112
113 if (errno != EEXIST) {
114 close(rootfd);
115 return NGX_ERROR;
116 }
117 }
118
119 close(rootfd);
120 errno = EEXIST;
121 return NGX_ERROR;
122}

124/*
125 * WHAT: Open the DETERMINISTIC, identity-keyed upload-resume partial for a final
126 * path (confined inside root_canon), creating it if absent and PRESERVING
127 * any existing bytes (no O_TRUNC, no O_EXCL). Reports the current partial
128 * size in *cur_size so the caller / client can resume at that offset.
129 *
130 * WHY: WebDAV resumable PUT (Content-Range) needs a chunk to land at an absolute
131 * offset on a partial that survives across separate PUT requests and a
132 * server restart, then commit (rename) only when complete — the HTTP
133 * analogue of the root:// resume staging. Reuses the same name scheme
134 * (brix_make_resume_path) and confinement (brix_open_beneath) as the
135 * random staged_open so security and glob-clean are identical.
136 *
137 * Returns NGX_OK with staged->active=1, or NGX_ERROR (errno set).
138 *
139 * Parameters:
140 * log — nginx log for error reporting
141 * req — request description: root_canon / final_path / principal / stage_dir /
142 * mode (see brix_staged_open_req_t)
143 * staged — output struct populated on success
144 * cur_size — output: current partial size (resume offset), 0 if fresh
145 */
146ngx_int_t
147brix_staged_open_resume(ngx_log_t *log, const brix_staged_open_req_t *req,
148 brix_staged_file_t *staged, off_t *cur_size)
149{
150 const char *stage_dir;
151 int rootfd, fd;
152 const char *rel;
153 struct stat sb;
154
155 (void) log;
156
157 if (staged == NULL || req == NULL || req->final_path == NULL) {
158 errno = EINVAL;
159 return NGX_ERROR;
160 }
161 ngx_memzero(staged, sizeof(*staged));
162 staged->fd = NGX_INVALID_FILE;
163 /* SECURITY: the resume partial is created PRIVATE (0600) and stays private
164 * across requests/restarts (it persists between range chunks); the intended
165 * final mode is restored at commit. */
166 staged->final_mode = req->mode;
167 if (cur_size != NULL) {
168 *cur_size = 0;
169 }
170
171 if (brix_make_resume_path(req->final_path, req->principal, req->stage_dir,
172 staged->tmp_path, sizeof(staged->tmp_path))
173 != NGX_OK)
174 {
175 errno = ENAMETOOLONG;
176 return NGX_ERROR;
177 }
178
179 stage_dir = req->stage_dir;
180 if (stage_dir != NULL && stage_dir[0] != '\0') {
181 /* Partial lives on the configured fast device (outside root_canon). The
182 * basename is a server-generated hash inside the operator-trusted stage
183 * dir, so a direct O_NOFOLLOW open is safe; commit moves it to storage. */
184 fd = open(staged->tmp_path, O_RDWR | O_CREAT | O_NOFOLLOW | O_CLOEXEC,
185 0600);
186 if (fd == NGX_INVALID_FILE) {
187 return NGX_ERROR;
188 }
189 } else {
190 rootfd = brix_beneath_open_root(req->root_canon);
191 if (rootfd < 0) {
192 return NGX_ERROR;
193 }
194 rel = brix_beneath_strip_root(req->root_canon, staged->tmp_path);
195 if (rel == NULL) {
196 close(rootfd);
197 errno = EXDEV;
198 return NGX_ERROR;
199 }
200 /* O_CREAT but NOT O_EXCL / O_TRUNC: create-or-resume, preserving bytes. */
201 fd = brix_open_beneath(rootfd, rel, O_RDWR | O_CREAT, 0600);
202 close(rootfd);
203 if (fd == NGX_INVALID_FILE) {
204 return NGX_ERROR;
205 }
206 }
207
208 if (cur_size != NULL && fstat(fd, &sb) == 0) {
209 *cur_size = sb.st_size;
210 }
211 staged->fd = fd;
212 staged->active = 1;
213 return NGX_OK;
214}

217/*
218 * staged_seal_temp — make the staged TEMP publishable: flush its data, restore
219 * the caller's intended mode, close the fd.
220 *
221 * Phase 51 (C1): flush the staged data to stable storage BEFORE the rename
222 * publishes it, so a crash / power loss / ENOSPC mid-write cannot expose a
223 * torn object. A failed fsync means the data is NOT durable — fail the
224 * commit (unlink the temp, leave the final path untouched) rather than
225 * publish possibly-incomplete data. (close() alone does not flush.)
226 *
227 * SECURITY (mode restore): the temp was written 0600 (private); the committed
228 * object carries its client-intended bits (e.g. 0644) with no world-readable
229 * in-flight window. Under impersonation the fd fchmod runs as the
230 * unprivileged worker on the MAPPED USER's file and EPERMs (silently),
231 * leaving the object 0600 and blocking group DAC on a shared/setgid-dir
232 * upload — so re-apply the mode AS THE MAPPED USER via the broker
233 * (path-based), which the fd fchmod cannot reach.
234 *
235 * A no-op when the fd is already closed. On failure: fd closed, temp
236 * unlinked, staged deactivated, errno set — the caller only closes rootfd.
237 */
238static ngx_int_t
239staged_seal_temp(ngx_log_t *log, const char *root_canon,
240 brix_staged_file_t *staged, int rootfd, const char *tmp_rel,
241 const char *final_path)
242{
243 if (staged->fd == NGX_INVALID_FILE) {
244 return NGX_OK;
245 }
246 if (fsync(staged->fd) != 0) {
247 int e = errno;
248 ngx_log_error(NGX_LOG_ERR, log, e,
249 "brix: staged commit fsync failed — not publishing "
250 "\"%s\"", final_path);
251 ngx_close_file(staged->fd);
252 staged->fd = NGX_INVALID_FILE;
253 (void) brix_unlink_beneath(rootfd, tmp_rel, 0);
254 staged->active = 0;
255 errno = e;
256 return NGX_ERROR;
257 }
258 if (staged->final_mode != 0 && brix_imp_client_active()) {
259 (void) brix_chmod_confined_canon(log, root_canon, staged->tmp_path,
260 staged->final_mode);
261 } else {
262 (void) fchmod(staged->fd, staged->final_mode);
263 }
264 ngx_close_file(staged->fd);
265 staged->fd = NGX_INVALID_FILE;
266 return NGX_OK;
267}

269/*
270 * WHAT: Carry a live WebDAV lock record across a replace-publish.
271 *
272 * WHY: Phase-107 C7 — the lock record (BRIX_LOCK_XATTR_KEY) lives in an xattr
273 * ON the destination inode, and a rename publish REPLACES that inode. An
274 * ADMITTED write under a live lock (the owner presenting the If: token
275 * over WebDAV, or any write on an `advisory`-enforcement export) must
276 * not silently discharge the lock: RFC 4918 §7.4 — a write does not
277 * remove a lock. Only the lock STATE MACHINE (UNLOCK/expiry reap) may.
278 *
279 * HOW: Quiet lgetxattr on the current destination; the absent class (ENOENT,
280 * ENODATA, ENOTSUP, an over-cap value) means nothing to carry (NGX_OK).
281 * A record found is copied VERBATIM onto the about-to-publish temp —
282 * expired records included: expiry is the gate's question, and the reap
283 * belongs to the WebDAV edge, never to a commit. A failed copy FAILS
284 * the commit (errno preserved): publishing would strip the lock.
285 */
286ngx_int_t
287brix_staged_lock_carry(ngx_log_t *log, const char *final_path,
288 const char *tmp_path)
289{
290 char buf[BRIX_LOCK_XATTR_MAXLEN];
291 ssize_t n;
292
293 n = lgetxattr(final_path, BRIX_LOCK_XATTR_KEY, buf, sizeof(buf));
294 if (n <= 0) {
295 return NGX_OK; /* absent class: nothing to carry */
296 }
297 if (lsetxattr(tmp_path, BRIX_LOCK_XATTR_KEY, buf, (size_t) n, 0) != 0) {
298 int e = errno;
299
300 ngx_log_error(NGX_LOG_ERR, log, e,
301 "brix: staged publish could not carry the lock record "
302 "onto \"%s\" — refusing a lock-stripping publish",
303 tmp_path);
304 errno = e;
305 return NGX_ERROR;
306 }
307 return NGX_OK;
308}

310/*
311 * WHAT: Atomically rename the temp file to its final path and clean up.
312 *
313 * WHY: After all data has been written to the staged temp file, commit makes it visible at
314 * the target location. The rename is atomic on POSIX filesystems — readers see either
315 * the old file or the new one, never a partial write.
316 *
317 * HOW: Close the fd if still open (data should be flushed). Rename tmp_path → final_path via
318 * brix_rename_confined_canon(). On rename failure: unlink the temp file as cleanup. Set
319 * staged->active=0 and clear tmp_path buffer. Return NGX_OK on success, NGX_ERROR on fail.
320 *
321 * Parameters:
322 * log — nginx log for error/cleanup reporting
323 * root_canon — canonical root for confined rename
324 * staged — the staged file struct (must be active)
325 * final_path — destination path to rename into
326 */
327/* Open the confinement root and strip both paths to their root-relative
328 * forms. Returns the rootfd, or -1 after full cleanup (temp fd closed,
329 * staged deactivated; errno = EXDEV on a path that does not strip). Split
330 * from staged_commit_internal to keep its decision count in budget. */
331static int
332staged_commit_confine(const char *root_canon, brix_staged_file_t *staged,
333 const char *final_path, const char **tmp_rel, const char **final_rel)
334{
335 int rootfd;
336
337 /* Open the confinement root first so the fsync-failure cleanup path (C1) can
338 * unlink the temp without re-opening it. */
339 rootfd = brix_beneath_open_root(root_canon);
340 if (rootfd < 0) {
341 if (staged->fd != NGX_INVALID_FILE) {
342 ngx_close_file(staged->fd);
343 staged->fd = NGX_INVALID_FILE;
344 }
345 staged->active = 0;
346 return -1;
347 }
348 *tmp_rel = brix_beneath_strip_root(root_canon, staged->tmp_path);
349 *final_rel = brix_beneath_strip_root(root_canon, final_path);
350 if (*tmp_rel == NULL || *final_rel == NULL) {
351 if (staged->fd != NGX_INVALID_FILE) {
352 ngx_close_file(staged->fd);
353 staged->fd = NGX_INVALID_FILE;
354 }
355 close(rootfd);
356 staged->active = 0;
357 errno = EXDEV;
358 return -1;
359 }
360 return rootfd;
361}

363static ngx_int_t
364staged_commit_internal(ngx_log_t *log, const char *root_canon,
365 brix_staged_file_t *staged, const char *final_path, int exclusive)
366{
367 if (staged == NULL || !staged->active) {
368 errno = EINVAL;
369 return NGX_ERROR;
370 }
371
372 int rootfd;
373 int rc;
374 const char *tmp_rel, *final_rel;
375
376 rootfd = staged_commit_confine(root_canon, staged, final_path,
377 &tmp_rel, &final_rel);
378 if (rootfd < 0) {
379 return NGX_ERROR;
380 }
381
382 if (staged_seal_temp(log, root_canon, staged, rootfd, tmp_rel,
383 final_path) != NGX_OK)
384 {
385 int e = errno;
386 close(rootfd);
387 errno = e;
388 return NGX_ERROR;
389 }
390
391 /* Phase-107 C7: the replace-publish swaps the destination inode — carry a
392 * live lock record onto the temp first so an ADMITTED write under a lock
393 * does not discharge it (brix_staged_lock_carry). The exclusive commit
394 * (RENAME_NOREPLACE) has no destination inode to carry from. */
395 if (!exclusive
396 && brix_staged_lock_carry(log, final_path, staged->tmp_path) != NGX_OK)
397 {
398 int e = errno;
399
400 (void) brix_unlink_beneath(rootfd, tmp_rel, 0);
401 close(rootfd);
402 staged->active = 0;
403 errno = e;
404 return NGX_ERROR;
405 }
406
407 rc = exclusive ? brix_rename_beneath_excl(rootfd, tmp_rel, final_rel)
408 : brix_rename_beneath(rootfd, tmp_rel, final_rel);
409 if (rc != 0) {
410 int e = errno; /* preserve EEXIST for the caller */
411 (void) brix_unlink_beneath(rootfd, tmp_rel, 0);
412 close(rootfd);
413 staged->active = 0;
414 errno = e;
415 return NGX_ERROR;
416 }
417
418 /* Phase-107 C3: persist the DIRECTORY ENTRY so the rename itself survives
419 * a crash. This replaces an inert (void) fsync(rootfd): rootfd is O_PATH
420 * (fsync = EBADF, discarded) and the export ROOT is the wrong directory
421 * anyway — the publish to a/b/c needs a/b flushed. A failed barrier FAILS
422 * the commit: the name is already visible and cannot be un-renamed, but a
423 * publish that reports success without durability is the exact bug this
424 * barrier exists to remove (the caller sees EIO; dirsync logged at crit).
425 * Gated per export by brix_durable_publish (absent/unregistered = on). */
426 if (brix_vfs_backend_durable(root_canon)
427 && brix_publish_dirsync(log, rootfd, root_canon, final_rel) != NGX_OK)
428 {
429 int e = errno ? errno : EIO;
430
431 close(rootfd);
432 staged->active = 0;
433 staged->tmp_path[0] = '\0'; /* the temp was consumed by the rename */
434 errno = e;
435 return NGX_ERROR;
436 }
437
438 close(rootfd);
439 staged->active = 0;
440 staged->tmp_path[0] = '\0';
441 return NGX_OK;
442}

445/*
446 * brix_publish_dirsync — see staged_file.h.
447 *
448 * HOW: derive the parent as everything before the last '/' of the
449 * root-relative path ("." when the object sits directly under the root),
450 * open it beneath the anchor with O_RDONLY — the one flag combination an
451 * fsync accepts — and flush. The parent fd is derived from the SAME
452 * rootfd the rename used, so a symlink swapped in after the rename can
453 * redirect the flush only chroot-style within the export, never outside
454 * it (RESOLVE_IN_ROOT).
455 */
456ngx_int_t
457brix_publish_dirsync(ngx_log_t *log, int rootfd, const char *root_canon,
458 const char *final_path)
459{
460 char parent[PATH_MAX];
461 const char *rel, *slash;
462 size_t n;
463 int anchor = rootfd, dirfd, e;
464
465 if (final_path == NULL) {
466 errno = EINVAL;
467 return NGX_ERROR;
468 }
469 /* Accept both forms the callers hand in: an ABSOLUTE canonical path under
470 * root_canon (vfs_rename's resolved dst), or a ROOT-RELATIVE tail — which
471 * strip_root itself produces WITH its leading '/' kept (staged commit's
472 * final_rel), so a failed strip on a '/'-path is not an escape, it is the
473 * already-relative form. beneath_rel then drops the slashes either way;
474 * RESOLVE_BENEATH keeps every interpretation confined. */
475 rel = final_path;
476 if (rel[0] == '/') {
477 const char *stripped = brix_beneath_strip_root(root_canon, rel);
478
479 if (stripped != NULL) {
480 rel = stripped;
481 }
482 }
483 rel = brix_beneath_rel(rel);
484 slash = strrchr(rel, '/');
485 if (slash == NULL) {
486 parent[0] = '.'; parent[1] = '\0';
487 } else {
488 n = (size_t) (slash - rel);
489 if (n == 0 || n >= sizeof(parent)) {
490 errno = EINVAL;
491 return NGX_ERROR;
492 }
493 memcpy(parent, rel, n);
494 parent[n] = '\0';
495 }
496
497 if (anchor < 0) {
498 anchor = brix_beneath_open_root(root_canon);
499 if (anchor < 0) {
500 return NGX_ERROR;
501 }
502 }
503 dirfd = brix_open_beneath(anchor, parent,
504 O_RDONLY | O_DIRECTORY | O_CLOEXEC, 0);
505 e = errno;
506 if (anchor != rootfd) {
507 close(anchor);
508 }
509 if (dirfd < 0) {
510 errno = e;
511 return NGX_ERROR;
512 }
513 if (fsync(dirfd) != 0) {
514 e = errno;
515 close(dirfd);
516 ngx_log_error(NGX_LOG_CRIT, log, e,
517 "brix: durable publish: parent dirsync of \"%s\" "
518 "failed — the published name is NOT durable", final_path);
519 errno = e ? e : EIO;
520 return NGX_ERROR;
521 }
522 close(dirfd);
523 return NGX_OK;
524}

540/*
541 * WHAT: Close the temp file fd and optionally unlink it from disk.
542 *
543 * WHY: On write failure or client disconnect, abort cleans up the staged temp file so it
544 * doesn't leak on disk. The caller decides whether to remove the temp file (remove_tmp=1)
545 * or leave it for later inspection (remove_tmp=0).
546 *
547 * HOW: Close fd if open and valid. If remove_tmp is set AND staged is active AND tmp_path is
548 * non-empty: unlink via brix_unlink_confined_canon(). Always set active=0 and clear
549 * tmp_path buffer.
550 *
551 * Parameters:
552 * log — nginx log for cleanup error reporting
553 * root_canon — canonical root for confined unlink
554 * staged — the staged file struct (NULL-safe)
555 * remove_tmp — 1 to delete temp file, 0 to leave it on disk
556 */
557void
558brix_staged_abort(ngx_log_t *log, const char *root_canon,
559 brix_staged_file_t *staged, ngx_flag_t remove_tmp)
560{
561 if (staged == NULL) {
562 return;
563 }
564
565 (void) log;
566
567 if (staged->fd != NGX_INVALID_FILE) {
568 ngx_close_file(staged->fd);
569 staged->fd = NGX_INVALID_FILE;
570 }
571
572 if (remove_tmp && staged->active && staged->tmp_path[0] != '\0') {
573 int rootfd = brix_beneath_open_root(root_canon);
574 const char *tmp_rel;
575
576 if (rootfd >= 0) {
577 tmp_rel = brix_beneath_strip_root(root_canon, staged->tmp_path);
578 if (tmp_rel != NULL) {
579 (void) brix_unlink_beneath(rootfd, tmp_rel, 0);
580 }
581 close(rootfd);
582 }
583 }
584
585 staged->active = 0;
586 staged->tmp_path[0] = '\0';
587}
