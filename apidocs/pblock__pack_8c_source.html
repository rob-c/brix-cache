1/*
2 * pblock_pack.c — phase-88 W2: the packed small-blob arena. See pblock_pack.h.
3 *
4 * WHAT: Implements the arena: catalog `pack` table (blob_id → seg/off/len),
5 * flock-serialised record appends into <root>/pack/seg-<n>.dat, memfd
6 * read-serving, materialise-back, and row/segment reaping.
7 *
8 * HOW: Records are the shared "BXS1" layout (shared/cache/cas_pack_format.h),
9 * keyed by blob_id, always fmt 0 (raw — a packed blob requires xform
10 * NONE). The append protocol under pack/.lock is: resolve the active
11 * segment (highest seg-<n>.dat, rolling at PBLOCK_PACK_SEG_BYTES) →
12 * pwrite header+key+data at its tail → fdatasync → INSERT the pack row →
13 * unlock. Because the row lands before the lock releases, the reap check
14 * (count(seg)==0, also under the lock) can never unlink a segment an
15 * in-flight append is targeting. Whole-record crc32 is verified on every
16 * decode (memfd open / materialise) — damage is EIO, never wrong bytes.
17 * ngx-free (libc + sqlite3 + zlib); BRIX_HAVE_SQLITE-gated.
18 */

80int
81pblock_pack_find(const pblock_state_t *st, const char *blob_id,
82 pblock_pack_loc_t *out)
83{
84 sqlite3_stmt *q;
85 int rc;
86
87 q = cat_prepare(st->cat,
88 "SELECT seg, off, len FROM pack WHERE blob_id = ?1;");
89 if (q == NULL) {
90 return -1;
91 }
92 sqlite3_bind_text(q, 1, blob_id, -1, SQLITE_STATIC);
93 rc = sqlite3_step(q);
94 if (rc == SQLITE_ROW) {
95 out->seg = sqlite3_column_int64(q, 0);
96 out->off = sqlite3_column_int64(q, 1);
97 out->len = sqlite3_column_int64(q, 2);
98 sqlite3_finalize(q);
99 return 0;
100 }
101 sqlite3_finalize(q);
102 return rc == SQLITE_DONE ? 1 : cat_fail(EIO);
103}

113 int rc;
114
115 if (q == NULL) {
116 return -1;
117 }
118 sqlite3_bind_text(q, 1, blob_id, -1, SQLITE_STATIC);
119 sqlite3_bind_int64(q, 2, seg);
120 sqlite3_bind_int64(q, 3, off);

122 rc = sqlite3_step(q);
123 sqlite3_finalize(q);
124 return rc == SQLITE_DONE ? 0 : cat_fail(EIO);

141/* Rows still referencing `seg` (-1 on a catalog error — treated as "live" so
142 * a reap can never race a failing count into data loss). */
143static int64_t
144pack_seg_live(const pblock_state_t *st, int64_t seg)
145{
146 sqlite3_stmt *q = cat_prepare(st->cat,
147 "SELECT COUNT(*) FROM pack WHERE seg = ?1;");
148 int64_t live = -1;
149
150 if (q == NULL) {
151 return -1;
152 }
153 sqlite3_bind_int64(q, 1, seg);
154 if (sqlite3_step(q) == SQLITE_ROW) {
155 live = sqlite3_column_int64(q, 0);
156 }
157 sqlite3_finalize(q);

161/* pack_seg_reap — under the arena lock, unlink `seg` once no rows reference
162 * it. Safe against appends: they insert their row before unlocking. */
163static void
164pack_seg_reap(const pblock_state_t *st, int64_t seg)
165{
166 char segp[PATH_MAX];
167 int lockfd = pack_lock(st);
168
169 if (lockfd < 0) {
170 return;
171 }
172 if (pack_seg_live(st, seg) == 0
173 && pack_seg_path(st, seg, segp, sizeof(segp)) == 0)
174 {
175 (void) unlink(segp);
176 }
177 close(lockfd); /* closing releases the flock */
178}

182/* pack_hdr_check — decode a just-read record header and shape-check it against
183 * the expected blob (fmt 0, matching klen, raw==stored==expect_len, key match);
184 * fills *rec. 0 when sound, -1 on any mismatch (caller maps to EIO). */
185static int
186pack_hdr_check(const unsigned char *hdr, const char *blob_id, size_t klen,
187 int64_t expect_len, brix_pack_rec_t *rec)
188{
189 if (brix_pack_seg_decode(hdr, rec) != 0
190 || rec->klen != klen
191 || rec->fmt != 0
192 || rec->stored != rec->raw
193 || (int64_t) rec->raw != expect_len
194 || memcmp(hdr + SEG_HDR, blob_id, klen) != 0)
195 {
196 return -1;
197 }
198 return 0;
199}

201/* pack_read_record — open the record's segment, decode + shape-check its
202 * header against the expected blob, then read + crc-verify the data into a
203 * malloc'd buffer. Returns the buffer (caller frees) or NULL/errno (EIO for
204 * any damage — a packed record is served verified or not at all). */
205static unsigned char *
206pack_read_record(const pblock_state_t *st, const char *blob_id,
207 const pblock_pack_loc_t *loc, int64_t expect_len)
208{
209 char segp[PATH_MAX];
210 unsigned char hdr[SEG_HDR + PACK_KMAX];
211 brix_pack_rec_t rec;
212 unsigned char *data;
213 size_t klen = strlen(blob_id);
214 int segfd;
215
216 if (klen == 0 || klen > PACK_KMAX
217 || pack_seg_path(st, loc->seg, segp, sizeof(segp)) != 0)
218 {
219 errno = EINVAL;
220 return NULL;
221 }
222 segfd = open(segp, O_RDONLY | O_CLOEXEC);
223 if (segfd < 0) {
224 errno = EIO; /* row without a segment = damage, not ENOENT */
225 return NULL;
226 }
227 if (pack_pread_full(segfd, hdr, SEG_HDR + klen, (off_t) loc->off) != 0
228 || pack_hdr_check(hdr, blob_id, klen, expect_len, &rec) != 0)
229 {
230 close(segfd);
231 errno = EIO;
232 return NULL;
233 }
234
235 data = malloc(rec.stored ? (size_t) rec.stored : 1);
236 if (data == NULL) {
237 close(segfd);
238 errno = ENOMEM;
239 return NULL;
240 }
241 if (pack_pread_full(segfd, data, (size_t) rec.stored,
242 (off_t) (loc->off + SEG_HDR + (int64_t) klen)) != 0
243 || crc_of(data, (size_t) rec.stored) != rec.crc)
244 {
245 close(segfd);
246 free(data);
247 errno = EIO;
248 return NULL;
249 }
250 close(segfd);
251 return data;
252}

256/* pack_open_admit_seg — open the segment that will receive a `size`-byte record
257 * for a `klen`-byte key, first rolling to the next segment when appending here
258 * would push it past PBLOCK_PACK_SEG_BYTES. On success returns the open (O_RDWR)
259 * segfd, writing the chosen segment number to *seg and its tail offset to *off;
260 * -1/errno on failure. Caller holds the arena lock (active seg is stable). */
261static int
262pack_open_admit_seg(const pblock_state_t *st, size_t klen, int64_t size,
263 int64_t *seg, off_t *off)
264{
265 char segp[PATH_MAX];
266 struct stat segst;
267 int64_t s = pack_active_seg(st);
268 int segfd;
269
270 if (s == 0) {
271 s = 1;
272 }
273 if (pack_seg_path(st, s, segp, sizeof(segp)) != 0) {
274 errno = ENAMETOOLONG;
275 return -1;
276 }
277 segfd = open(segp, O_RDWR | O_CREAT | O_CLOEXEC, 0600);
278 if (segfd >= 0 && fstat(segfd, &segst) == 0
279 && segst.st_size > 0
280 && segst.st_size + (off_t) (SEG_HDR + klen) + (off_t) size
281 > (off_t) PBLOCK_PACK_SEG_BYTES)
282 {
283 /* Roll to the next segment before this record would overgrow it. */
284 close(segfd);
285 segfd = -1;
286 s++;
287 if (pack_seg_path(st, s, segp, sizeof(segp)) == 0) {
288 segfd = open(segp, O_RDWR | O_CREAT | O_CLOEXEC, 0600);
289 }
290 }
291 if (segfd < 0 || fstat(segfd, &segst) != 0) {
292 int err = errno;
293
294 if (segfd >= 0) { close(segfd); }
295 errno = err;
296 return -1;
297 }
298 *seg = s;
299 *off = segst.st_size;
300 return segfd;
301}

303/* pack_write_record — encode and durably append one record (header+key+data) at
304 * `off` in segfd, retracting a torn tail on failure. 0 on success, -1/errno.
305 * Caller holds the arena lock. */
306static int
307pack_write_record(int segfd, const char *blob_id, const unsigned char *data,
308 int64_t size, size_t klen, off_t off)
309{
310 unsigned char hdr[SEG_HDR + PACK_KMAX];
311 brix_pack_rec_t rec;
312
313 rec.klen = klen;
314 rec.fmt = 0;
315 rec.crc = crc_of(data, (size_t) size);
316 rec.stored = (uint64_t) size;
317 rec.raw = (uint64_t) size;
318 brix_pack_seg_encode(hdr, blob_id, &rec);
319
320 if (pack_pwrite_full(segfd, hdr, SEG_HDR + klen, off) != 0
321 || pack_pwrite_full(segfd, data, (size_t) size,
322 off + (off_t) (SEG_HDR + klen)) != 0
323 || fdatasync(segfd) != 0)
324 {
325 int err = errno;
326
327 if (ftruncate(segfd, off) != 0) {
328 /* best-effort retract of the torn tail — the original pwrite/sync
329 * error verdict (err) stands either way */
330 }
331 errno = err;
332 return -1;
333 }
334 return 0;
335}

337int
338pblock_pack_admit(pblock_state_t *st, const char *blob_id, int64_t size,
339 int64_t bs)
340{
341 unsigned char *data;
342 size_t klen = strlen(blob_id);
343 int64_t seg;
344 off_t off;
345 ssize_t got;
346 int lockfd, segfd;
347
348 if (size <= 0 || klen == 0 || klen > PACK_KMAX) {
349 errno = EINVAL;
350 return -1;
351 }
352 data = malloc((size_t) size);
353 if (data == NULL) {
354 errno = ENOMEM;
355 return -1;
356 }
357 /* Read through the block engine (holes read as zeros), never raw off the
358 * block-0 file — the blob's logical bytes are what the record must carry. */
359 got = pblock_read_blocks(st, blob_id, bs, -1, data, (size_t) size, 0);
360 if (got != (ssize_t) size) {
361 free(data);
362 errno = got < 0 ? errno : EIO;
363 return -1;
364 }
365
366 lockfd = pack_lock(st);
367 if (lockfd < 0) {
368 free(data);
369 return -1;
370 }
371
372 segfd = pack_open_admit_seg(st, klen, size, &seg, &off);
373 if (segfd < 0) {
374 int err = errno;
375
376 close(lockfd);
377 free(data);
378 errno = err;
379 return -1;
380 }
381
382 if (pack_write_record(segfd, blob_id, data, size, klen, off) != 0) {
383 int err = errno;
384
385 close(segfd);
386 close(lockfd);
387 free(data);
388 errno = err;
389 return -1;
390 }
391 close(segfd);
392 free(data);
393
394 if (pack_row_insert(st, blob_id, seg, (int64_t) off, size) != 0) {
395 close(lockfd);
396 return -1; /* record is an fsck-collectable orphan */
397 }
398 close(lockfd);
399
400 /* The record is durable and indexed — the striped copy is now redundant. */
401 pblock_remove_blocks(st, blob_id, size, bs);
402 return 0;
403}

405int
406pblock_pack_open_memfd(const pblock_state_t *st, const pblock_meta *meta)
407{
408 pblock_pack_loc_t loc;
409 unsigned char *data;
410 int fd, rc;
411
412 rc = pblock_pack_find(st, meta->blob_id, &loc);
413 if (rc != 0) {
414 errno = rc > 0 ? ENOENT : EIO;
415 return -1;
416 }
417 data = pack_read_record(st, meta->blob_id, &loc, meta->size);
418 if (data == NULL) {
419 return -1; /* errno set (EIO family) */
420 }
421
422 fd = memfd_create(meta->blob_id, MFD_CLOEXEC | MFD_ALLOW_SEALING);
423 if (fd < 0) {
424 free(data);
425 return -1;
426 }
427 if (pack_pwrite_full(fd, data, (size_t) meta->size, 0) != 0) {
428 int err = errno;
429
430 close(fd);
431 free(data);
432 errno = err;
433 return -1;
434 }
435 free(data);
436 /* Best-effort seals: the handle is read-intent, so freezing the bytes is
437 * pure hardening — never a functional dependency. */
438 (void) fcntl(fd, F_ADD_SEALS, F_SEAL_GROW | F_SEAL_SHRINK | F_SEAL_WRITE);
439 return fd;
440}

442int
443pblock_pack_materialize(pblock_state_t *st, const pblock_meta *meta)
444{
445 pblock_pack_loc_t loc;
446 unsigned char *data;
447 ssize_t put;
448 int rc;
449
450 if (!st->pack) {
451 return 0;
452 }
453 rc = pblock_pack_find(st, meta->blob_id, &loc);
454 if (rc != 0) {
455 if (rc > 0) {
456 return 0; /* not packed — nothing to do */
457 }
458 errno = EIO;
459 return -1;
460 }
461 data = pack_read_record(st, meta->blob_id, &loc, meta->size);
462 if (data == NULL) {
463 return -1;
464 }
465 if (pblock_ensure_obj_dir(st, meta->blob_id) != 0) {
466 free(data);
467 return -1;
468 }
469 put = pblock_write_blocks(st, meta->blob_id, meta->block_size, -1, data,
470 (size_t) meta->size, 0);
471 free(data);
472 if (put != (ssize_t) meta->size) {
473 errno = put < 0 ? errno : EIO;
474 return -1;
475 }
476 pack_row_delete(st, meta->blob_id);
477 pack_seg_reap(st, loc.seg);
478 return 0;
479}

494pblock_pack_or_block_read(const pblock_state_t *st, const char *blob_id,
495 int64_t bs, void *buf, size_t len, off_t off)
496{
497 pblock_pack_loc_t loc;
498 char segp[PATH_MAX];
499 size_t klen = strlen(blob_id);
500 size_t want = len;
501 int segfd, rc;
502
503 if (!st->pack || (rc = pblock_pack_find(st, blob_id, &loc)) > 0) {
504 return pblock_read_blocks(st, blob_id, bs, -1, buf, len, off);
505 }
506 if (rc < 0) {
507 errno = EIO;
508 return -1;
509 }
510 if (off >= (off_t) loc.len) {
511 return 0;
512 }
513 if ((off_t) want > (off_t) loc.len - off) {
514 want = (size_t) ((off_t) loc.len - off);
515 }
516 if (pack_seg_path(st, loc.seg, segp, sizeof(segp)) != 0) {
517 errno = ENAMETOOLONG;
518 return -1;
519 }
520 segfd = open(segp, O_RDONLY | O_CLOEXEC);
521 if (segfd < 0) {
522 errno = EIO;
523 return -1;
524 }
525 if (pack_pread_full(segfd, buf, want,
526 (off_t) (loc.off + SEG_HDR + (int64_t) klen) + off) != 0)
527 {
528 int err = errno;
529
530 close(segfd);
531 errno = err;
532 return -1;
533 }
534 close(segfd);
535 return (ssize_t) want;
536}
