23/*
24 * WHAT: The immutable parameter set of one collection (directory) COPY — the
25 * confinement root, the source and destination paths, the source directory
26 * mode, and the overwrite/depth flags derived from the request.
27 *
28 * WHY: The collection copy is executed either inline on the event loop or on a
29 * thread-pool worker, and it is described by eight loosely-related values. A
30 * struct threads them as a single unit through the execute / post-task /
31 * per-phase helpers — one job, one owner — instead of an eight-argument
32 * signature that trips the complexity gate and is easy to mis-order at a call
33 * site. It is thread-safe by construction: it holds copied stack-buffer paths
34 * (no ngx_http_request_t handle), so a worker thread reads only its own copy.
35 *
36 * HOW: Populated once by the caller from resolved paths and header-derived
37 * flags, then passed by const pointer to the execute helpers. The async task
38 * struct embeds a job plus the fields the event loop needs to reply.
39 */
40typedef struct {
41 ngx_log_t *log;
42 char root_canon[WEBDAV_MAX_PATH];
43 char src_path[WEBDAV_MAX_PATH];
44 char dst_path[WEBDAV_MAX_PATH];
45 mode_t src_mode;
46 ngx_flag_t dst_existed;
47 ngx_flag_t dst_was_dir;
48 ngx_flag_t depth_infinity;
49 /* Phase-105: the endpoint write posture, copied by VALUE at job build. The
50 * job may run on a thread-pool worker long after the request that made it,
51 * so it must never re-derive the posture from configuration; and because
52 * the whole job is copied into the task context, the posture travels with
53 * it into every child of the recursion (Appendix D.7). */
54 brix_vfs_mutation_policy_t policy;
55} webdav_copy_job_t;

63/*
64 * WHAT: Populate a webdav_copy_job_t from a resolved COPY request and the
65 * export root, deep-copying the stack-buffer paths into the job's own storage.
66 *
67 * WHY: The collection-copy job derives entirely from the resolved request; a
68 * single builder keeps the inline and threaded paths in lock step and
69 * guarantees the job never aliases a caller stack buffer that could go out of
70 * scope before a worker thread reads it.
71 *
72 * HOW: Assigns the scalar flags/mode from req (dst_was_dir = existed && dir),
73 * and uses ngx_cpystrn (bounded, always NUL-terminating) for each path,
74 * matching the prior per-field copies.
75 */
76static void
77webdav_copy_job_init(webdav_copy_job_t *job, ngx_log_t *log,
78 const char *root_canon, brix_vfs_mutation_policy_t policy,
79 const webdav_copy_req_t *req)
80{
81 job->log = log;
82 job->policy = policy;
83 job->src_mode = req->src_sb.st_mode;
84 job->dst_existed = req->dst_existed;
85 job->dst_was_dir = req->dst_existed && S_ISDIR(req->dst_sb.st_mode);
86 job->depth_infinity = req->depth_infinity;
87
88 ngx_cpystrn((u_char *) job->root_canon, (u_char *) root_canon,
89 sizeof(job->root_canon));
90 ngx_cpystrn((u_char *) job->src_path, (u_char *) req->src_path,
91 sizeof(job->src_path));
92 ngx_cpystrn((u_char *) job->dst_path, (u_char *) req->dst_path,
93 sizeof(job->dst_path));
94}

107/*
108 * WHAT: Stage the copied tree into a sibling temp directory: make the temp dir,
109 * clone dead props + fattrs, and (depth:infinity) recurse the contents.
110 *
111 * WHY: The atomic-rename strategy needs the whole copy built out of band before
112 * the final rename; isolating the build keeps webdav_copy_collection_execute a
113 * flat sequence and gives one place that reports the build-phase HTTP status.
114 *
115 * HOW: mkdir maps ENOENT-on-parent to 409 Conflict (any other failure to 500);
116 * on a recursion failure it removes the temp tree and returns 500. On success
117 * it returns NGX_OK with tmp_path filled for the commit phase.
118 */
119static ngx_int_t
120webdav_copy_collection_stage(const webdav_copy_job_t *job, char *tmp_path,
121 size_t tmp_path_size)
122{
123 brix_vfs_export_op_ctx_t opctx;
124
125 webdav_copy_job_opctx(job, &opctx);
126
127 if (brix_make_tmp_path(job->dst_path, tmp_path, tmp_path_size) != NGX_OK) {
128 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
129 }
130
131 /* The staging directory is a sibling of the destination and therefore lives
132 * ON THE EXPORT, not in a service-owned area — so it is the first mutation
133 * this COPY makes and the first thing the endpoint gate refuses. Nothing is
134 * created before this point (brix_make_tmp_path only builds a name), which
135 * is what makes an EROFS refusal here side-effect-free. */
136 if (brix_vfs_export_mkdir(&opctx, tmp_path, job->src_mode & 0777) != 0) {
137 if (errno == ENOENT) {
138 return NGX_HTTP_CONFLICT;
139 }
140 return brix_http_errno_to_status(errno);
141 }
142
143 webdav_dead_props_copy(job->log, job->src_path, tmp_path);
144 brix_ns_copy_fattrs(job->log, job->src_path, tmp_path);
145
146 if (job->depth_infinity
147 && webdav_copy_dir_recursive(&opctx, job->src_path, tmp_path) != NGX_OK)
148 {
149 (void) webdav_delete_path_recursive(job->log, job->root_canon, tmp_path);
150 return NGX_HTTP_INTERNAL_SERVER_ERROR;
151 }
152
153 return NGX_OK;
154}

156/*
157 * WHAT: Commit a fully-staged temp tree onto the destination: drop a pre-existing
158 * destination directory (rename cannot replace a non-empty dir) then rename the
159 * temp tree into place.
160 *
161 * WHY: Splitting the commit from the stage keeps each phase single-purpose and
162 * confines the "remove temp on any failure" cleanup to one linear path.
163 *
164 * HOW: If the destination existed as a directory it is removed first; a failure
165 * there or in the rename removes the temp tree and returns 500. On success it
166 * returns 204 (replaced) or 201 (created) matching the prior end-of-function
167 * status.
168 */
169static ngx_int_t
170webdav_copy_collection_commit(const webdav_copy_job_t *job, const char *tmp_path)
171{
172 if (job->dst_existed && job->dst_was_dir) {
173 if (webdav_delete_path_recursive(job->log, job->root_canon,
174 job->dst_path) != NGX_OK)
175 {
176 (void) webdav_delete_path_recursive(job->log, job->root_canon,
177 tmp_path);
178 return NGX_HTTP_INTERNAL_SERVER_ERROR;
179 }
180 }
181
182 if (brix_rename_confined_canon(job->log, job->root_canon, tmp_path,
183 job->dst_path) != 0)
184 {
185 brix_log_safe_path(job->log, NGX_LOG_ERR, ngx_errno,
186 "brix_webdav COPY: dir rename failed: \"%s\"",
187 job->dst_path);
188 (void) webdav_delete_path_recursive(job->log, job->root_canon, tmp_path);
189 return NGX_HTTP_INTERNAL_SERVER_ERROR;
190 }
191
192 return job->dst_existed ? NGX_HTTP_NO_CONTENT : NGX_HTTP_CREATED;
193}

195/*
196 * Copy a collection (directory) atomically.
197 * Strategy: build the whole copy in a sibling temp directory, then rename it
198 * into place — so a concurrent reader never sees a half-copied tree
199 * (stage → commit). Every failure after the temp dir exists removes it, so no
200 * orphan is left. Returns 201/204 on success or an HTTP error status. May run
201 * on a worker thread (see webdav_copy_collection_thread), so it uses only its
202 * job parameters and thread-safe confined-FS helpers — no ngx_http_request_t
203 * access.
204 */
205static ngx_int_t
206webdav_copy_collection_execute(const webdav_copy_job_t *job)
207{
208 char tmp_path[WEBDAV_MAX_PATH];
209 ngx_int_t rc;
210
211 rc = webdav_copy_collection_stage(job, tmp_path, sizeof(tmp_path));
212 if (rc != NGX_OK) {
213 return rc;
214 }
215
216 return webdav_copy_collection_commit(job, tmp_path);
217}

234/*
235 * Completion handler (event loop): the async re-entry point after the worker
236 * finishes. Sends 201/204 on success or finalizes with the error status, and
237 * releases the request reference taken in webdav_copy_collection_post_task.
238 */
239static void
240webdav_copy_collection_done(ngx_event_t *ev)
241{
242 ngx_thread_task_t *task = ev->data;
243 webdav_copy_collection_task_t *t = task->ctx;
244 ngx_http_request_t *r = t->r;
245 ngx_int_t status = t->http_status;
246
247 if (status == NGX_HTTP_CREATED || status == NGX_HTTP_NO_CONTENT) {
248 /* The copy ran on a pool thread; this is its event-loop completion, the
249 * only place the worker's CMS connection may be written. */
250 webdav_cns_note_written(r, t->job.dst_path);
251 webdav_send_status_only(r, (ngx_uint_t) status);
252 return;
253 }
254
255 webdav_metrics_finalize_request(r, status);
256}

276/*
277 * Try to offload a collection COPY to the thread pool.
278 * Returns NGX_DONE if queued (request kept alive via r->main->count++),
279 * NGX_DECLINED if no thread pool is available (caller runs it synchronously),
280 * NGX_ERROR on allocation/post failure. The job's stack-buffer paths are
281 * copied into the task context so they remain valid after this returns.
282 */
283static ngx_int_t
284webdav_copy_collection_post_task(ngx_http_request_t *r,
285 ngx_http_brix_webdav_loc_conf_t *conf, const webdav_copy_job_t *job)
286{
287 ngx_thread_task_t *task;
288 webdav_copy_collection_task_t *t;
289 ngx_thread_pool_t *pool;
290
291 /*
292 * Under impersonation the per-worker broker socket is a single fd shared by
293 * the event-loop thread; a thread-pool task issuing confined ops would use
294 * it concurrently and corrupt the request/reply framing, wedging the whole
295 * worker's broker channel. The thread also lacks the per-worker principal.
296 * Force the synchronous (NGX_DECLINED) path so the recursive copy runs on
297 * the event-loop thread where the principal is set and the broker socket
298 * has exactly one user.
299 */
300 if (brix_imp_enabled()) {
301 return NGX_DECLINED;
302 }
303
304 pool = webdav_copy_thread_pool(conf);
305 if (pool == NULL) {
306 return NGX_DECLINED;
307 }
308
309 task = ngx_thread_task_alloc(r->pool, sizeof(webdav_copy_collection_task_t));
310 if (task == NULL) {
311 return NGX_ERROR;
312 }
313
314 t = task->ctx;
315 t->r = r;
316 t->job = *job;
317 t->job.log = r->connection->log;
318 t->http_status = NGX_HTTP_INTERNAL_SERVER_ERROR;
319
320 brix_task_bind(task, webdav_copy_collection_thread, webdav_copy_collection_done);
321 task->event.log = r->connection->log;
322
323 if (ngx_thread_task_post(pool, task) != NGX_OK) {
324 return NGX_ERROR;
325 }
326
327 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
328 "brix_webdav: offloaded collection COPY to thread pool");
329
330 r->main->count++;
331 return NGX_DONE;
332}

334/*
335 * WHAT: Execute a collection (directory) COPY for a resolved request — offload
336 * to the thread pool when possible, else run it inline.
337 *
338 * WHY: Isolates the S_ISDIR branch of the handler, keeping the depth:infinity
339 * offload / fall-through / inline-run decision in one place.
340 *
341 * HOW: Builds a webdav_copy_job_t from the resolved request. For depth:infinity
342 * it tries the thread pool: NGX_DONE means queued (returned as-is so the handler
343 * finishes), NGX_ERROR maps to 500, NGX_DECLINED (no pool) falls through. It
344 * then runs the copy inline; a non-201/204 status is returned as the failure,
345 * NGX_OK means success (the handler emits the final status). Depth:0 always runs
346 * inline (it only creates the top directory).
347 */
348ngx_int_t
349webdav_copy_do_collection(ngx_http_request_t *r,
350 ngx_http_brix_webdav_loc_conf_t *conf, const webdav_copy_req_t *req)
351{
352 webdav_copy_job_t job;
353 ngx_int_t rc;
354
355 webdav_copy_job_init(&job, r->connection->log, conf->common.root_canon,
356 brix_vfs_policy_from_write_enable(conf->common.allow_write), req);
357
358 if (req->depth_infinity) {
359 rc = webdav_copy_collection_post_task(r, conf, &job);
360 if (rc == NGX_DONE) {
361 return NGX_DONE;
362 }
363 if (rc == NGX_ERROR) {
364 return NGX_HTTP_INTERNAL_SERVER_ERROR;
365 }
366 /* rc == NGX_DECLINED: no thread pool, fall through to run inline. */
367 }
368
369 rc = webdav_copy_collection_execute(&job);
370 if (rc != NGX_HTTP_CREATED && rc != NGX_HTTP_NO_CONTENT) {
371 return rc;
372 }
373
374 return NGX_OK;
375}
