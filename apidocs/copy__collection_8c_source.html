24/*
25 * WHAT: The immutable parameter set of one collection (directory) COPY — the
26 * confinement root, the source and destination paths, the source directory
27 * mode, and the overwrite/depth flags derived from the request.
28 *
29 * WHY: The collection copy is executed either inline on the event loop or on a
30 * thread-pool worker, and it is described by eight loosely-related values. A
31 * struct threads them as a single unit through the execute / post-task /
32 * per-phase helpers — one job, one owner — instead of an eight-argument
33 * signature that trips the complexity gate and is easy to mis-order at a call
34 * site. It is thread-safe by construction: it holds copied stack-buffer paths
35 * (no ngx_http_request_t handle), so a worker thread reads only its own copy.
36 *
37 * HOW: Populated once by the caller from resolved paths and header-derived
38 * flags, then passed by const pointer to the execute helpers. The async task
39 * struct embeds a job plus the fields the event loop needs to reply.
40 */
41typedef struct {
42 ngx_log_t *log;
43 char root_canon[WEBDAV_MAX_PATH];
44 char src_path[WEBDAV_MAX_PATH];
45 char dst_path[WEBDAV_MAX_PATH];
46 mode_t src_mode;
47 ngx_flag_t dst_existed;
48 ngx_flag_t dst_was_dir;
49 ngx_flag_t depth_infinity;
50 ngx_flag_t authz_required;
51 webdav_copy_authz_ctx_t authz;
52 /* Phase-105: the endpoint write posture, copied by VALUE at job build. The
53 * job may run on a thread-pool worker long after the request that made it,
54 * so it must never re-derive the posture from configuration; and because
55 * the whole job is copied into the task context, the posture travels with
56 * it into every child of the recursion (Appendix D.7). */
57 brix_vfs_mutation_policy_t policy;
58} webdav_copy_job_t;

66/*
67 * WHAT: Populate a webdav_copy_job_t from a resolved COPY request and the
68 * export root, deep-copying the stack-buffer paths into the job's own storage.
69 *
70 * WHY: The collection-copy job derives entirely from the resolved request; a
71 * single builder keeps the inline and threaded paths in lock step and
72 * guarantees the job never aliases a caller stack buffer that could go out of
73 * scope before a worker thread reads it.
74 *
75 * HOW: Assigns the scalar flags/mode from req (dst_was_dir = existed && dir),
76 * and uses ngx_cpystrn (bounded, always NUL-terminating) for each path,
77 * matching the prior per-field copies.
78 */
79static void
80webdav_copy_job_init(webdav_copy_job_t *job, ngx_http_request_t *r,
81 ngx_http_brix_webdav_loc_conf_t *conf, const webdav_copy_req_t *req)
82{
83 ngx_memzero(job, sizeof(*job));
84 job->log = r->connection->log;
85 job->policy = brix_vfs_policy_from_write_enable(
86 conf->common.allow_write);
87 job->src_mode = req->src_sb.st_mode;
88 job->dst_existed = req->dst_existed;
89 job->dst_was_dir = req->dst_existed && S_ISDIR(req->dst_sb.st_mode);
90 job->depth_infinity = req->depth_infinity;
91
92 ngx_cpystrn((u_char *) job->root_canon,
93 (u_char *) conf->common.root_canon,
94 sizeof(job->root_canon));
95 ngx_cpystrn((u_char *) job->src_path, (u_char *) req->src_path,
96 sizeof(job->src_path));
97 ngx_cpystrn((u_char *) job->dst_path, (u_char *) req->dst_path,
98 sizeof(job->dst_path));
99 webdav_copy_authz_init(&job->authz, r, conf, req);
100 job->authz_required = webdav_copy_authz_has_effective_rules(&job->authz);
101}

114/*
115 * WHAT: Stage the copied tree into a sibling temp directory: make the temp dir,
116 * clone dead props + fattrs, and (depth:infinity) recurse the contents.
117 *
118 * WHY: The atomic-rename strategy needs the whole copy built out of band before
119 * the final rename; isolating the build keeps webdav_copy_collection_execute a
120 * flat sequence and gives one place that reports the build-phase HTTP status.
121 *
122 * HOW: mkdir maps ENOENT-on-parent to 409 Conflict (any other failure to 500);
123 * on a recursion failure it removes the temp tree and returns 500. On success
124 * it returns NGX_OK with tmp_path filled for the commit phase.
125 */
126static ngx_int_t
127webdav_copy_collection_stage(const webdav_copy_job_t *job, char *tmp_path,
128 size_t tmp_path_size)
129{
130 brix_vfs_export_op_ctx_t export_op_ctx;
131 ngx_int_t rc;
132
133 webdav_copy_job_export_op_ctx(job, &export_op_ctx);
134
135 if (brix_make_tmp_path(job->dst_path, tmp_path, tmp_path_size) != NGX_OK) {
136 return NGX_HTTP_REQUEST_URI_TOO_LARGE;
137 }
138
139 /* The staging directory is a sibling of the destination and therefore lives
140 * ON THE EXPORT, not in a service-owned area — so it is the first mutation
141 * this COPY makes and the first thing the endpoint gate refuses. Nothing is
142 * created before this point (brix_make_tmp_path only builds a name), which
143 * is what makes an EROFS refusal here side-effect-free. */
144 if (brix_vfs_export_mkdir(&export_op_ctx, tmp_path, job->src_mode & BRIX_PERM_MASK_FULL) != 0) {
145 if (errno == ENOENT) {
146 return NGX_HTTP_CONFLICT;
147 }
148 return brix_http_errno_to_status(errno);
149 }
150
151 webdav_dead_props_copy(job->log, job->src_path, tmp_path);
152 brix_ns_copy_fattrs(job->log, job->src_path, tmp_path);
153
154 if (!job->depth_infinity) {
155 return NGX_OK;
156 }
157
158 rc = webdav_copy_dir_recursive(&export_op_ctx, job->src_path, tmp_path,
159 job->authz_required ? webdav_copy_authorize_child : NULL,
160 job->authz_required ? (void *) &job->authz : NULL);
161 if (rc != NGX_OK) {
162 (void) webdav_delete_path_recursive(job->log, job->root_canon, tmp_path);
163 return rc >= NGX_HTTP_BAD_REQUEST
164 ? rc : NGX_HTTP_INTERNAL_SERVER_ERROR;
165 }
166
167 return NGX_OK;
168}

170/*
171 * WHAT: Commit a fully-staged temp tree onto the destination: drop a pre-existing
172 * destination directory (rename cannot replace a non-empty dir) then rename the
173 * temp tree into place.
174 *
175 * WHY: Splitting the commit from the stage keeps each phase single-purpose and
176 * confines the "remove temp on any failure" cleanup to one linear path.
177 *
178 * HOW: If the destination existed as a directory it is removed first; a failure
179 * there or in the rename removes the temp tree and returns 500. On success it
180 * returns 204 (replaced) or 201 (created) matching the prior end-of-function
181 * status.
182 */
183static ngx_int_t
184webdav_copy_collection_commit(const webdav_copy_job_t *job, const char *tmp_path)
185{
186 if (job->dst_existed && job->dst_was_dir) {
187 ngx_int_t delete_rc;
188
189 if (job->authz_required) {
190 delete_rc = brix_fs_guard_tree_confined(job->log,
191 job->root_canon, job->dst_path,
192 webdav_copy_authorize_delete, (void *) &job->authz);
193 if (delete_rc != NGX_OK) {
194 (void) webdav_delete_path_recursive(job->log,
195 job->root_canon, tmp_path);
196 return delete_rc >= NGX_HTTP_BAD_REQUEST
197 ? delete_rc : NGX_HTTP_INTERNAL_SERVER_ERROR;
198 }
199 }
200
201 delete_rc = webdav_delete_path_recursive_checked(job->log,
202 job->root_canon, job->dst_path,
203 job->authz_required ? webdav_copy_authorize_delete : NULL,
204 job->authz_required ? (void *) &job->authz : NULL);
205 if (delete_rc != NGX_OK)
206 {
207 (void) webdav_delete_path_recursive(job->log, job->root_canon,
208 tmp_path);
209 return delete_rc >= NGX_HTTP_BAD_REQUEST
210 ? delete_rc : NGX_HTTP_INTERNAL_SERVER_ERROR;
211 }
212 }
213
214 if (brix_rename_confined_canon(job->log, job->root_canon, tmp_path,
215 job->dst_path) != 0)
216 {
217 brix_log_safe_path(job->log, NGX_LOG_ERR, ngx_errno,
218 "brix_webdav COPY: dir rename failed: \"%s\"",
219 job->dst_path);
220 (void) webdav_delete_path_recursive(job->log, job->root_canon, tmp_path);
221 return NGX_HTTP_INTERNAL_SERVER_ERROR;
222 }
223
224 return job->dst_existed ? NGX_HTTP_NO_CONTENT : NGX_HTTP_CREATED;
225}

227/*
228 * Copy a collection (directory) atomically.
229 * Strategy: build the whole copy in a sibling temp directory, then rename it
230 * into place — so a concurrent reader never sees a half-copied tree
231 * (stage → commit). Every failure after the temp dir exists removes it, so no
232 * orphan is left. Returns 201/204 on success or an HTTP error status. May run
233 * on a worker thread (see webdav_copy_collection_thread), so it uses only its
234 * job parameters and thread-safe confined-FS helpers — no ngx_http_request_t
235 * access.
236 */
237static ngx_int_t
238webdav_copy_collection_execute(const webdav_copy_job_t *job)
239{
240 char tmp_path[WEBDAV_MAX_PATH];
241 ngx_int_t rc;
242
243 rc = webdav_copy_collection_stage(job, tmp_path, sizeof(tmp_path));
244 if (rc != NGX_OK) {
245 return rc;
246 }
247
248 return webdav_copy_collection_commit(job, tmp_path);
249}

266/*
267 * Completion handler (event loop): the async re-entry point after the worker
268 * finishes. Sends 201/204 on success or finalizes with the error status, and
269 * releases the request reference taken in webdav_copy_collection_post_task.
270 */
271static void
272webdav_copy_collection_done(ngx_event_t *ev)
273{
274 ngx_thread_task_t *task = ev->data;
275 webdav_copy_collection_task_t *t = task->ctx;
276 ngx_http_request_t *r = t->r;
277 ngx_int_t status = t->http_status;
278
279 if (status == NGX_HTTP_CREATED || status == NGX_HTTP_NO_CONTENT) {
280 /* The copy ran on a pool thread; this is its event-loop completion, the
281 * only place the worker's CMS connection may be written. */
282 webdav_cns_note_written(r, t->job.dst_path);
283 webdav_send_status_only(r, (ngx_uint_t) status);
284 return;
285 }
286
287 webdav_metrics_finalize_request(r, status);
288}

308/*
309 * Try to offload a collection COPY to the thread pool.
310 * Returns NGX_DONE if queued (request kept alive via r->main->count++),
311 * NGX_DECLINED if no thread pool is available (caller runs it synchronously),
312 * NGX_ERROR on allocation/post failure. The job's stack-buffer paths are
313 * copied into the task context so they remain valid after this returns.
314 */
315static ngx_int_t
316webdav_copy_collection_post_task(ngx_http_request_t *r,
317 ngx_http_brix_webdav_loc_conf_t *conf, const webdav_copy_job_t *job)
318{
319 ngx_thread_task_t *task;
320 webdav_copy_collection_task_t *t;
321 ngx_thread_pool_t *pool;
322
323 /*
324 * Under impersonation the per-worker broker socket is a single fd shared by
325 * the event-loop thread; a thread-pool task issuing confined ops would use
326 * it concurrently and corrupt the request/reply framing, wedging the whole
327 * worker's broker channel. The thread also lacks the per-worker principal.
328 * Force the synchronous (NGX_DECLINED) path so the recursive copy runs on
329 * the event-loop thread where the principal is set and the broker socket
330 * has exactly one user.
331 */
332 if (brix_imp_enabled() || job->authz_required) {
333 return NGX_DECLINED;
334 }
335
336 pool = webdav_copy_thread_pool(conf);
337 if (pool == NULL) {
338 return NGX_DECLINED;
339 }
340
341 task = ngx_thread_task_alloc(r->pool, sizeof(webdav_copy_collection_task_t));
342 if (task == NULL) {
343 return NGX_ERROR;
344 }
345
346 t = task->ctx;
347 t->r = r;
348 t->job = *job;
349 t->job.log = r->connection->log;
350 t->http_status = NGX_HTTP_INTERNAL_SERVER_ERROR;
351
352 brix_task_bind(task, webdav_copy_collection_thread, webdav_copy_collection_done);
353 task->event.log = r->connection->log;
354
355 if (ngx_thread_task_post(pool, task) != NGX_OK) {
356 return NGX_ERROR;
357 }
358
359 BRIX_LOG(NGX_LOG_INFO, r->connection->log, 0,
360 "brix_webdav: offloaded collection COPY to thread pool");
361
362 r->main->count++;
363 return NGX_DONE;
364}

366/*
367 * WHAT: Execute a collection (directory) COPY for a resolved request — offload
368 * to the thread pool when possible, else run it inline.
369 *
370 * WHY: Isolates the S_ISDIR branch of the handler, keeping the depth:infinity
371 * offload / fall-through / inline-run decision in one place.
372 *
373 * HOW: Builds a webdav_copy_job_t from the resolved request. For depth:infinity
374 * it tries the thread pool: NGX_DONE means queued (returned as-is so the handler
375 * finishes), NGX_ERROR maps to 500, NGX_DECLINED (no pool) falls through. It
376 * then runs the copy inline; a non-201/204 status is returned as the failure,
377 * NGX_OK means success (the handler emits the final status). Depth:0 always runs
378 * inline (it only creates the top directory).
379 */
380ngx_int_t
381webdav_copy_do_collection(ngx_http_request_t *r,
382 ngx_http_brix_webdav_loc_conf_t *conf, const webdav_copy_req_t *req)
383{
384 webdav_copy_job_t job;
385 ngx_int_t rc;
386
387 webdav_copy_job_init(&job, r, conf, req);
388
389 if (req->depth_infinity) {
390 rc = webdav_copy_collection_post_task(r, conf, &job);
391 if (rc == NGX_DONE) {
392 return NGX_DONE;
393 }
394 if (rc == NGX_ERROR) {
395 return NGX_HTTP_INTERNAL_SERVER_ERROR;
396 }
397 /* rc == NGX_DECLINED: no thread pool, fall through to run inline. */
398 }
399
400 rc = webdav_copy_collection_execute(&job);
401 if (rc != NGX_HTTP_CREATED && rc != NGX_HTTP_NO_CONTENT) {
402 return rc;
403 }
404
405 return NGX_OK;
406}
