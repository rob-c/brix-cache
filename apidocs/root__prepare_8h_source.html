21/*
22 * brix_prepare_export_root — validate and canonicalize an export root path.
23 *
24 * Called from protocol merge_loc_conf callbacks. The helper:
25 * 1. Returns NGX_CONF_OK immediately if root is empty and required == 0.
26 * 2. Returns NGX_CONF_ERROR with NGX_LOG_EMERG if root is empty and required != 0.
27 * 3. Rejects root.len >= opts->canon_size (path too long).
28 * 4. Calls brix_validate_path() to verify the path is a readable directory
29 * with write access when opts->allow_write is set.
30 * 5. Calls realpath(3) to resolve symlinks and store the canonical form.
31 * 6. Writes the NUL-terminated canonical path into root_canon.
32 *
33 * Returns NGX_CONF_OK on success, NGX_CONF_ERROR on any failure (after emitting
34 * an NGX_LOG_EMERG log message via ngx_conf_log_error).
35 */

37 const ngx_str_t *root, const brix_export_root_opts_t *opts,
