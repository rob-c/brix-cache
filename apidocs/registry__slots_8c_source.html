15/* ---- Per-source quota key for one registrant (W5 / P90-27.2) --------------
16 *
17 * WHAT: Renders the registrant's ratelimit-vocabulary bucket id into `out`
18 * (BRIX_SESSION_SRC_KEY_LEN bytes): "sub:<8-hex>" for token logins,
19 * "dn:<8-hex>" for certificate/system logins, "" when the login carried no DN
20 * (un-keyed — such sessions are exempt from the per-source cap and only ever
21 * subject to the F4 global LRU, exactly the pre-W5 regime).
22 *
23 * WHY: The registry only receives (dn, token_auth), not the full connection
24 * ctx, so the key is derived here from the same identity string every caller
25 * already passes — via the SHARED brix_rl_key_* formatters, so one principal
26 * maps to the same no-PII bucket id in the limiter and in this quota. */
27void
28brix_session_src_key(const char *dn, ngx_uint_t token_auth,
29 char out[BRIX_SESSION_SRC_KEY_LEN])
30{
31 out[0] = '\0';
32 if (dn == NULL || dn[0] == '\0') {
33 return;
34 }
35 if (token_auth) {
36 brix_rl_key_sub_hash((const u_char *) dn, ngx_strlen(dn),
37 out, BRIX_SESSION_SRC_KEY_LEN);
38 } else {
39 brix_rl_key_dn_hash((const u_char *) dn, ngx_strlen(dn),
40 out, BRIX_SESSION_SRC_KEY_LEN);
41 }
42}

44/* ---- Scan the session table for a sessid, tracking free/LRU/quota slots ----
45 *
46 * WHAT: Walks every occupied slot looking for `sessid`. On a hit it refreshes
47 * that slot's last_seen to `now` and returns 1. On a miss it returns 0 with
48 * `sc` reporting the first free slot, the global-LRU occupied slot (F4
49 * reap-on-full), and — when `src_key` is non-empty — how many slots the
50 * registrant already owns plus its own-LRU slot (W5 per-source quota).
51 *
52 * WHY: Isolates the single linear pass over the SHM table so the caller stays a
53 * flat sequence of decisions. Must run with brix_session_mutex held — it reads
54 * and (on a hit) writes shared slot state.
55 *
56 * The pass covers the LIVE prefix only: high_water bounds every occupied slot,
57 * so nothing beyond it can match the sessid, contribute to an LRU minimum or
58 * count against the per-source quota, and everything beyond it is free. The
59 * frontier slot itself is the free-slot fallback when the prefix has no hole.
60 *
61 * HOW:
62 * 1. Seed `sc` to its empty-table defaults.
63 * 2. For each slot in the live prefix: record the first free one and skip it.
64 * 3. On a sessid match, stamp last_seen and return 1.
65 * 4. Otherwise fold the occupied slot into the global LRU minimum, and — on
66 * a src_key match — into the registrant's own count + own-LRU minimum.
67 * 5. Fall the free slot back to the frontier, then return 0 (no match).
68 */
69int
70brix_session_scan(brix_session_table_t *tbl,
71 const u_char sessid[BRIX_SESSION_ID_LEN], ngx_msec_t now,
72 const char *src_key, brix_session_scan_t *sc)
73{
74 brix_session_entry_t *e;
75 ngx_uint_t i;
76
77 sc->free_slot = tbl->capacity;
78 sc->lru_slot = tbl->capacity;
79 sc->lru_seen = 0;
80 sc->src_count = 0;
81 sc->src_lru_slot = tbl->capacity;
82 sc->src_lru_seen = 0;
83 sc->match_slot = tbl->capacity;
84
85 for (i = 0; i < tbl->high_water; i++) {
86 e = &tbl->slots[i];
87 if (!e->in_use) {
88 if (sc->free_slot == tbl->capacity) {
89 sc->free_slot = i;
90 }
91 continue;
92 }
93 if (ngx_memcmp(e->sessid, sessid, BRIX_SESSION_ID_LEN) == 0) {
94 e->last_seen = now; /* refresh activity on re-register */
95 sc->match_slot = i; /* round 15: the caller's slot hint */
96 return 1;
97 }
98 /* Track the global-LRU occupied slot for reap-on-full (F4). */
99 if (sc->lru_slot == tbl->capacity || e->last_seen < sc->lru_seen) {
100 sc->lru_slot = i;
101 sc->lru_seen = e->last_seen;
102 }
103 /* Track the registrant's OWN slot count + own-LRU (W5 quota). */
104 if (src_key[0] != '\0' && ngx_strcmp(e->src_key, src_key) == 0) {
105 sc->src_count++;
106 if (sc->src_lru_slot == tbl->capacity
107 || e->last_seen < sc->src_lru_seen)
108 {
109 sc->src_lru_slot = i;
110 sc->src_lru_seen = e->last_seen;
111 }
112 }
113 }
114
115 if (sc->free_slot == tbl->capacity && tbl->high_water < tbl->capacity) {
116 sc->free_slot = tbl->high_water; /* first slot past the live prefix */
117 }
118
119 return 0;
120}

122/* ---- Retire a freed top run so the mark tracks the peak LIVE population ----
123 *
124 * WHAT: Walks high_water down over any trailing free slots.
125 *
126 * WHY: Without this the mark would only ever grow, so one transient peak would
127 * tax every later scan for the process's lifetime — exactly the full-table cost
128 * the mark exists to remove. Must run with brix_session_mutex held.
129 *
130 * WHERE: brix_session_unregister only. The other two slot clears (the F4 reap
131 * and the W5 self-eviction) hand their freed index straight to fill_slot, so the
132 * slot is occupied again before any scan can observe it; retiring the frontier
133 * there would only be undone on the next line.
134 */
135void
136brix_session_shrink(brix_session_table_t *tbl)
137{
138 while (tbl->high_water > 0 && !tbl->slots[tbl->high_water - 1].in_use) {
139 tbl->high_water--;
140 }
141}

143/* ---- Reap the LRU slot when the table is full, else count the rejection ----
144 *
145 * WHAT: Phase 27 F4 slot-exhaustion defence, invoked only when no free slot and
146 * no matching session were found. If the LRU slot is older than the minimum reap
147 * age it copies that slot's sessid into `victim`, clears the slot, publishes its
148 * index via `free_slot_out`, and returns 1. Otherwise it increments the
149 * registry-full metric and returns 0 (leaving `victim`/`free_slot_out` untouched).
150 *
151 * WHY: A slot-exhaustion attacker must not be able to permanently deny new
152 * logins; reaping the least-recently-seen aged session bounds that. Must run
153 * with brix_session_mutex held — it mutates shared slot state.
154 *
155 * HOW:
156 * 1. If an aged LRU slot exists, snapshot its sessid, zero the slot, hand its
157 * index back as the free slot, and return 1.
158 * 2. Otherwise bump session_registry_full_total (if metrics are up) and
159 * return 0.
160 */
161int
162brix_session_reap_lru(brix_session_table_t *tbl, ngx_msec_t now,
163 ngx_uint_t lru_slot, ngx_msec_t lru_seen,
164 ngx_uint_t *free_slot_out, u_char victim[BRIX_SESSION_ID_LEN])
165{
166 ngx_brix_metrics_t *m;
167
168 if (lru_slot < tbl->capacity
169 && (now - lru_seen) >= BRIX_SESSION_REAP_MIN_AGE_MS)
170 {
171 ngx_memcpy(victim, tbl->slots[lru_slot].sessid, BRIX_SESSION_ID_LEN);
172 ngx_memzero(&tbl->slots[lru_slot], sizeof(tbl->slots[lru_slot]));
173 *free_slot_out = lru_slot;
174 return 1;
175 }
176
177 m = brix_metrics_shared();
178 if (m != NULL) {
179 (void) ngx_atomic_fetch_add(&m->session_registry_full_total, 1);
180 }
181 return 0;
182}

184/* ---- Populate a free session slot with a login's metadata ----
185 *
186 * WHAT: Writes sessid, DN, VO list and token_auth into slot `slot` and marks it
187 * in_use with last_seen = now. No return value.
188 *
189 * WHY: Keeps the field-by-field copy (with NULL-string coalescing and bounded
190 * ngx_cpystrn) in one named place off the register orchestrator. Must run with
191 * brix_session_mutex held — it writes shared slot state.
192 *
193 * HOW:
194 * 1. Copy the fixed-length sessid.
195 * 2. Bounded-copy DN and VO list, substituting "" for NULL inputs.
196 * 3. Store token_auth and last_seen, then flag the slot in_use.
197 * 4. Extend the live prefix if this slot sits past the frontier.
198 */
199void
200brix_session_fill_slot(brix_session_table_t *tbl, ngx_uint_t slot,
201 const u_char sessid[BRIX_SESSION_ID_LEN],
202 const char *dn, const char *vo_list, ngx_uint_t token_auth,
203 const char *src_key, ngx_msec_t now)
204{
205 brix_session_entry_t *e = &tbl->slots[slot];
206
207 ngx_memcpy(e->sessid, sessid, BRIX_SESSION_ID_LEN);
208 ngx_cpystrn((u_char *) e->dn, (u_char *) (dn ? dn : ""), sizeof(e->dn));
209 ngx_cpystrn((u_char *) e->vo_list, (u_char *) (vo_list ? vo_list : ""),
210 sizeof(e->vo_list));
211 ngx_cpystrn((u_char *) e->src_key, (u_char *) src_key,
212 sizeof(e->src_key));
213 e->token_auth = token_auth;
214 e->last_seen = now;
215 e->in_use = 1;
216 /* §1.4: the registering worker owns the primary connection — a kXR_bind
217 * arriving on any OTHER worker migrates its secondary here (bind_migrate.c). */
218 e->owner_worker = (ngx_int_t) ngx_worker;
219 /* A recycled slot must not inherit the previous session's bound paths. */
220 ngx_memzero(e->pathid_map, sizeof(e->pathid_map));
221
222 if (slot >= tbl->high_water) {
223 tbl->high_water = slot + 1; /* extend the live prefix */
224 }
225}

227/* ---- Locate a live slot by sessid (registry mutex held) ----
228 *
229 * WHAT: Returns the in-use entry whose sessid matches, or NULL.
230 *
231 * WHY: The three pathid-bitmap operations share this scan; keeping it a
232 * helper keeps each public function a flat lock→find→bit-op→unlock
233 * sequence.
234 *
235 * HOW: Linear scan over the LIVE prefix (high_water bounds every occupied slot,
236 * so no match can exist beyond it), ngx_memcmp on the 16-byte id.
237 */
238brix_session_entry_t *
239brix_session_find_locked(brix_session_table_t *tbl,
240 const u_char sessid[BRIX_SESSION_ID_LEN])
241{
242 brix_session_entry_t *slots = tbl->slots;
243 ngx_uint_t i, live = tbl->high_water;
244
245 for (i = 0; i < live; i++) {
246 if (slots[i].in_use
247 && ngx_memcmp(slots[i].sessid, sessid,
248 BRIX_SESSION_ID_LEN) == 0)
249 {
250 return &slots[i];
251 }
252 }
253 return NULL;
254}

257/* ---- Self-evict an over-quota identity's own LRU slot (W5 / P90-27.2) -----
258 *
259 * WHAT: Invoked when the registrant already owns >= the per-source soft cap of
260 * live slots. Snapshots the registrant's OWN least-recently-seen session into
261 * `victim`, clears that slot, publishes its index via `free_slot_out`, bumps
262 * the src-cap metric, and returns 1. Returns 0 (nothing evicted) if the scan
263 * found no own-LRU slot (defensive; count >= cap implies one exists).
264 *
265 * WHY: The soft cap must bite BEFORE free-slot consumption and BEFORE the F4
266 * global reap: an over-quota identity recycles its own sessions, so it can
267 * neither fill the table nor pressure OTHER identities' slots. Only the
268 * over-quota principal pays — its oldest session dies, everyone else is
269 * untouched. No minimum-age gate (unlike F4): self-eviction harms no third
270 * party, and an identity churning through its own quota is exactly the
271 * behaviour the cap exists to bound. Must run with brix_session_mutex held.
272 */
273int
274brix_session_src_cap_evict(brix_session_table_t *tbl,
275 ngx_uint_t src_lru_slot, ngx_uint_t *free_slot_out,
276 u_char victim[BRIX_SESSION_ID_LEN])
277{
278 ngx_brix_metrics_t *m;
279
280 if (src_lru_slot >= tbl->capacity) {
281 return 0;
282 }
283
284 ngx_memcpy(victim, tbl->slots[src_lru_slot].sessid, BRIX_SESSION_ID_LEN);
285 ngx_memzero(&tbl->slots[src_lru_slot], sizeof(tbl->slots[src_lru_slot]));
286 *free_slot_out = src_lru_slot;
287
288 m = brix_metrics_shared();
289 if (m != NULL) {
290 (void) ngx_atomic_fetch_add(&m->session_src_cap_evict_total, 1);
291 }
292 return 1;
293}

295/* ---- Finish an eviction after the session mutex is released ----
296 *
297 * WHAT: Increments the eviction metric and unpublishes every handle owned by the
298 * reaped `victim` session. No return value.
299 *
300 * WHY: Handle unpublish MUST happen AFTER releasing brix_session_mutex to mirror
301 * brix_session_unregister's lock order (session then handle) and avoid a lock
302 * inversion. Keeping it in its own helper makes that ordering contract explicit.
303 *
304 * HOW:
305 * 1. Bump session_evict_total (if metrics are up).
306 * 2. Unpublish all of the victim's handles via the handle table.
307 */
308void
309brix_session_finish_eviction(const u_char victim[BRIX_SESSION_ID_LEN])
310{
311 ngx_brix_metrics_t *m = brix_metrics_shared();
312
313 if (m != NULL) {
314 (void) ngx_atomic_fetch_add(&m->session_evict_total, 1);
315 }
316 brix_session_handle_unpublish_all(victim);
317}
