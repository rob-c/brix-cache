21/* GET range/bytes metrics — shared by the serve phases (get_serve.c) and the
22 * off-loop serve completion (brix_http_serve_offload), so all report
23 * identically. Prototype in get_internal.h. */
24void
25webdav_serve_metrics(ngx_http_request_t *r,
26 const brix_http_serve_result_t *result)
27{
28 if (result->range_result == BRIX_SERVE_RANGE_UNSATISFIED) {
29 BRIX_WEBDAV_METRIC_INC(range_total[BRIX_WEBDAV_RANGE_UNSATISFIED]);
30 } else if (result->range_result == BRIX_SERVE_RANGE_PARTIAL) {
31 BRIX_WEBDAV_METRIC_INC(range_total[BRIX_WEBDAV_RANGE_PARTIAL]);
32 } else {
33 BRIX_WEBDAV_METRIC_INC(range_total[BRIX_WEBDAV_RANGE_FULL]);
34 }
35 if (result->bytes_sent > 0) {
36 /* phase-106 W1: the authoritative served-byte count for
37 * $brix_bytes_served (the serve is zero-copy, off the VFS observer). */
38 brix_http_monitor_record_served(r, result->bytes_sent);
39 BRIX_WEBDAV_METRIC_ADD(bytes_tx_total, (size_t) result->bytes_sent);
40 if (r->connection && r->connection->sockaddr) {
41 if (r->connection->sockaddr->sa_family == AF_INET6) {
42 BRIX_WEBDAV_METRIC_ADD(bytes_tx_ipv6_total,
43 (size_t) result->bytes_sent);
44 } else {
45 BRIX_WEBDAV_METRIC_ADD(bytes_tx_ipv4_total,
46 (size_t) result->bytes_sent);
47 }
48 }
49 }
50}

52/* Re-entry trampoline for the off-event-loop cache fill: after the fill lands the
53 * completion event re-runs the GET handler, which now finds a cache HIT and serves
54 * it zero-copy. The fill helper carries no per-handler state (the request re-
55 * resolves from r), so `data` is unused.
56 *
57 * The result goes back through webdav_metrics_return for the same reason the
58 * synchronous dispatch does: the first pass parked with NGX_DONE, and
59 * webdav_metrics_response deliberately books nothing for a request that has not
60 * finished. Without this the ONE request that actually paid for a cache fill
61 * was the one missing from brix_webdav_responses_total and from
62 * brix_io_ops_total{proto="webdav",op="read"} — its bytes still landed (they
63 * come from the scrape-time ledger fold), so ops and bytes disagreed on exactly
64 * the remote-backed exports where fills happen. */
65static ngx_int_t
66webdav_get_reenter(ngx_http_request_t *r, void *data)
67{
68 (void) data;
69 return webdav_metrics_return(r, webdav_handle_get(r));
70}

72/* Failure tail of the same park: a fill that ends 404/403/502 never re-enters
73 * the handler, so this is the only place the parked request can book its
74 * response. Books and returns the status unchanged (the hook also exists to
75 * re-drive against another source; that is not this handler's policy). */
76static ngx_int_t
77webdav_get_fill_failed(ngx_http_request_t *r, void *data, ngx_int_t rc)
78{
79 (void) data;
80 webdav_metrics_response(r, rc);
81 return rc;
82}

84/*
85 * get_zip_member_serve — Phase-57 W2 ZIP member access over HTTP GET.
86 *
87 * WHAT: when the location is a ZIP-archive export, serve the requested member
88 * of the archive at `path` rather than the whole archive file.
89 * WHY: archive auth already ran in the access phase; a GET on a zip export
90 * selects one member by argument, and this must short-circuit the normal
91 * file-serving pipeline.
92 * HOW: parse the member argument; a negative result is a malformed request
93 * (400), a positive result serves the member and yields a terminal status,
94 * and zero means "no member requested" — signalled by NGX_DECLINED so the
95 * caller falls through to the normal whole-file path.
96 */
97static ngx_int_t
98get_zip_member_serve(ngx_http_request_t *r,
99 ngx_http_brix_webdav_loc_conf_t *conf, const char *path,
100 const brix_vfs_ctx_t *vfs_scope)
101{
102 char member[WEBDAV_MAX_PATH];
103 int zr;
104
105 if (!conf->common.zip_access) {
106 return NGX_DECLINED;
107 }
108
109 zr = brix_zip_http_member_arg(r, member, sizeof(member));
110 if (zr < 0) {
111 return NGX_HTTP_BAD_REQUEST;
112 }
113 if (zr > 0) {
114 return brix_zip_http_serve(r, vfs_scope,
115 conf->common.zip_cd_max_bytes, path, member);
116 }
117 return NGX_DECLINED;
118}

120/*
121 * get_offload_or_fill — run the off-loop serve/fill fast paths.
122 *
123 * WHAT: give the socket-wire serve offload (SP3) and the remote cache-fill
124 * offload (SP2) a chance to handle the request off the event loop.
125 * WHY: a root://-backed or remote-store export cannot open/read on the worker
126 * loop, and a remote cache MISS fill would stall it; both are pushed to the
127 * thread pool and completed (or re-entered) asynchronously.
128 * HOW: returns NGX_DONE when a path took over (async in flight), a terminal
129 * HTTP status on error, or NGX_DECLINED when neither applies and the caller
130 * should open inline. EACCES/EPERM from the offload is the per-user backend
131 * credential gate → 403; any other offload error → 500.
132 */
133static ngx_int_t
134get_offload_or_fill(ngx_http_request_t *r,
135 ngx_http_brix_webdav_loc_conf_t *conf,
136 ngx_http_brix_webdav_req_ctx_t *wctx, const char *path,
137 brix_vfs_ctx_t *vctx)
138{
139 const char *identity =
140 (wctx != NULL && wctx->dn[0] != '\0') ? wctx->dn : "anonymous";
141 brix_http_serve_opts_t sopts;
142 ngx_int_t sr;
143 ngx_int_t fr;
144 char physical[PATH_MAX];
145
146 if (brix_path_resolved_to_pfn(vctx, path, physical,
147 sizeof(physical)) != NGX_OK)
148 {
149 return NGX_HTTP_INTERNAL_SERVER_ERROR;
150 }
151
152 ngx_memzero(&sopts, sizeof(sopts));
153 sopts.xfer_proto = BRIX_XFER_PROTO_WEBDAV;
154 sopts.op_name = "GET";
155 sopts.identity = identity;
156 sopts.etag_flags = BRIX_ETAG_WEAK;
157 sopts.compress = conf->common.compress;
158
159 sr = brix_http_serve_offload_remote(r, vctx->sd, physical, path, &sopts,
160 &conf->common, vctx, webdav_serve_metrics);
161 if (sr == NGX_DONE) {
162 return NGX_DONE;
163 }
164 if (sr == NGX_ERROR) {
165 return (errno == EACCES || errno == EPERM)
166 ? NGX_HTTP_FORBIDDEN : NGX_HTTP_INTERNAL_SERVER_ERROR;
167 }
168
169 fr = brix_http_cache_fill_if_needed(r, vctx->sd, physical, &conf->common,
170 vctx,
171 webdav_get_reenter, NULL, webdav_get_fill_failed);
172 if (fr == NGX_DONE) {
173 return NGX_DONE;
174 }
175 if (fr == NGX_ERROR) {
176 return (errno == EACCES || errno == EPERM)
177 ? NGX_HTTP_FORBIDDEN : NGX_HTTP_INTERNAL_SERVER_ERROR;
178 }
179 return NGX_DECLINED;
180}

182/*
183 * webdav_handle_get — serve a file via HTTP GET with Range support.
184 *
185 * Fast path: if the fd-cache already holds an open fd for the requested URI
186 * hash, the stat and open system calls are skipped entirely. The cached fd
187 * remains owned by the fd-cache; the cleanup handler registered below uses
188 * NGX_INVALID_FILE so it does not close it a second time.
189 *
190 * Range handling: a single "bytes=start-end" or "bytes=-suffix" range is
191 * parsed and served as 206 Partial Content. Multi-range requests and
192 * overlapping ranges are not supported; clients that send them receive the
193 * full file (200 OK).
194 *
195 * ngx_http_send_header + r->header_only: after calling ngx_http_send_header(),
196 * always check r->header_only. If true, the client sent HEAD — return
197 * immediately without sending a body. The check inside the serve phases
198 * handles this.
199 *
200 * Pool allocation: ngx_pcalloc(r->pool, ...) for ngx_buf_t and ngx_file_t —
201 * both are freed when the request pool is destroyed after the response
202 * is sent.
203 *
204 * Ownership of fd:
205 * - If fd came from the fd-cache (fd_from_table=1), the cleanup handler
206 * stores NGX_INVALID_FILE so the fd-cache retains ownership.
207 * - If fd was opened here (fd_from_table=0), the cleanup handler closes it.
208 *
209 * Flow: resolve path → (zip member?) → build VFS ctx → off-loop serve/fill →
210 * open+stat (webdav_get_resolve_and_stat) → multi-range (webdav_get_serve_range) or
211 * conditional (webdav_get_eval_conditionals) + full/single-range (webdav_get_serve_full).
212 */
213ngx_int_t
214webdav_handle_get(ngx_http_request_t *r)
215{
216 ngx_http_brix_webdav_loc_conf_t *conf;
217 char path[WEBDAV_MAX_PATH];
218 ngx_int_t rc;
219 ngx_http_brix_webdav_req_ctx_t *wctx;
220 brix_vfs_ctx_t vctx;
221 get_serve_state_t st = {0};
222
223 conf = ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
224
225 rc = ngx_http_brix_webdav_resolve_path(r, conf->common.root_canon, path,
226 sizeof(path));
227 if (rc != NGX_OK) {
228 return rc;
229 }
230
231 wctx = ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
232 webdav_vfs_ctx_build_data(r, conf, path, &vctx);
233 /* Route through the export's selected storage backend (NULL ⇒ default POSIX). */
234 vctx.sd = brix_webdav_backend_instance(conf, r->connection->log);
235
236 /* ZIP archive reads derive from the same fully bound request context as a
237 * normal GET, preserving identity, authorization, credentials and backend. */
238 rc = get_zip_member_serve(r, conf, path, &vctx);
239 if (rc != NGX_DECLINED) {
240 return rc;
241 }
242
243 /* phase-64 SP3/SP2: socket-wire serve offload and remote cache-fill offload
244 * (both off the event loop). NGX_DONE ⇒ async took over; a terminal HTTP
245 * status ⇒ error; NGX_DECLINED ⇒ open inline below. */
246 rc = get_offload_or_fill(r, conf, wctx, path, &vctx);
247 if (rc != NGX_DECLINED) {
248 return rc;
249 }
250
251 rc = webdav_get_resolve_and_stat(r, &vctx, path, &st);
252 if (rc == NGX_DECLINED) {
253 /* §6.6: the target is a directory — listingredir / HTML listing /
254 * listingdeny per config. This owns the response. */
255 return webdav_get_serve_directory(r, conf, path);
256 }
257 if (rc != NGX_OK) {
258 return rc;
259 }
260
261 /* XrdHttp: multi-range vector read (kXR_readv equivalent over HTTP).
262 * A comma in the Range: value indicates multiple byte ranges — delegate
263 * to the multipart/byteranges handler rather than the single-range path. */
264 if (xrdhttp_request_is_multirange(r)) {
265 return webdav_get_serve_range(r, &st, path);
266 }
267
268 rc = webdav_get_eval_conditionals(r, &st);
269 if (rc != NGX_OK) {
270 return rc;
271 }
272
273 return webdav_get_serve_full(r, conf, wctx, &st, path);
274}
