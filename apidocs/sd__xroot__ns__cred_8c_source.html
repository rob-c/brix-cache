24/* ---- credential-scoped namespace wrappers (Phase 2 Task 1) -----------------
25 *
26 * WHAT: One wrapper per supported ns op that opens a per-user origin session
27 * (via sd_xroot_session with a non-NULL cred) instead of the anonymous /
28 * service-credential session the plain slots use.
29 *
30 * WHY: A deny-mode request whose data-plane credential gate fires must also
31 * present the user proxy for every pre-flight probe and metadata mutation
32 * (unlink, rename, copy, xattr). Without these wrappers the probe stat
33 * still runs under the static service credential even when the protocol
34 * handler correctly denied the operation.
35 *
36 * HOW: Each op opens the origin session through the shared ns_open/ns_close
37 * helpers below, passing the caller's cred to sd_xroot_session (NULL =
38 * anonymous / service credential — the plain vtable slots in
39 * sd_xroot_ns.c delegate here with cred=NULL, so these bodies are the
40 * single implementation of every path-based ns op).
41 * stat_cred is implemented in sd_xroot.c (alongside sd_xroot_stat, which
42 * reuses sd_xroot_origin_open — a file-private type). */
43
44/* One fresh origin session per path-based op: connection + fill task. */
45typedef struct {
46 brix_cache_origin_conn_t oc;
47 brix_cache_fill_t *t;
48} ns_session_t;

50/* Open a fresh origin session under cred (NULL = service credential).
51 * On failure sets errno and returns NGX_ERROR. */
52static ngx_int_t
53ns_open(brix_sd_instance_t *inst, const brix_sd_cred_t *cred, ns_session_t *s)
54{
55 sd_xroot_inst_state *is = inst->state;
56 int e = 0;
57
58 if (sd_xroot_session(is->conf, cred, &s->oc, &s->t, &e) != 0) {
59 errno = e;
60 return NGX_ERROR;
61 }
62 return NGX_OK;
63}
64
65/* Tear the session down, preserving errno across close/free. */
66static void
67ns_close(ns_session_t *s)
68{
69 int e = errno;
70
71 brix_cache_origin_close(&s->oc);
72 free(s->t);
73 errno = e;
74}

76/* Close the session and map a 0/-1 origin rc to NGX_OK/NGX_ERROR (errno is
77 * already set by the origin call, or by the caller for mapped kXR errors). */
78static ngx_int_t
79ns_result(ns_session_t *s, int rc)
80{
81 ns_close(s);
82 return rc == 0 ? NGX_OK : NGX_ERROR;
83}

127/* setattr_cred: apply a metadata mutation under the user's credential.
128 *
129 * WHAT: The union slot (mode + times + owner) against a root:// export. Only
130 * the mode group travels — as kXR_chmod, the one metadata mutation the
131 * base XRootD protocol defines.
132 * WHY: With no setattr slot at all the VFS treats the driver as having no
133 * mutable metadata and returns success WITHOUT touching the origin, so a
134 * chmod over a root:// export silently did nothing while reporting OK.
135 * Wiring the slot makes the mode case real and the rest honest.
136 * HOW: set_times / set_owner are accepted-and-ignored rather than refused: the
137 * times+owner opcode (kXR_setattr, 3500) is this project's capability-
138 * negotiated vendor extension, which a stock origin does not implement, and
139 * the driver has no negotiation to lean on here. Failing a `cp -p` on an
140 * otherwise-applied chmod would be worse than the documented sd.h contract
141 * ("a driver applies what its namespace can represent"), which sd_pblock
142 * reads the same way for atime. A request with no group set at all is an
143 * immediate no-op success: opening an origin session to send nothing would
144 * be pure latency. errno comes straight from brix_cache_origin_chmod (which
145 * maps the kXR status itself, like rm/rename/mkdir — unlike the truncate
146 * path, which records its error on the fill task for sd_xroot_errno). */
147ngx_int_t
148sd_xroot_setattr_cred(brix_sd_instance_t *inst, const char *path,
149 const brix_sd_setattr_t *attr, const brix_sd_cred_t *cred)
150{
151 ns_session_t s;
152 int rc;
153
154 if (attr == NULL) {
155 errno = EINVAL;
156 return NGX_ERROR;
157 }
158 if (!attr->set_mode) {
159 return NGX_OK; /* nothing this namespace can represent */
160 }
161 if (ns_open(inst, cred, &s) != NGX_OK) { return NGX_ERROR; }
162 rc = brix_cache_origin_chmod(s.t, &s.oc, path, attr->mode);
163 return ns_result(&s, rc);
164}

182/* server_copy_cred: server-side byte copy under the user's credential. */
183ngx_int_t
184sd_xroot_server_copy_cred(brix_sd_instance_t *inst, const char *src,
185 const char *dst, off_t *bytes_out, const brix_sd_cred_t *cred)
186{
187 ns_session_t s;
188 u_char src_fh[XRD_FHANDLE_LEN], dst_fh[XRD_FHANDLE_LEN];
189 ngx_int_t rc;
190 int e;
191
192 if (ns_open(inst, cred, &s) != NGX_OK) { return NGX_ERROR; }
193 ngx_cpystrn((u_char *) s.t->clean_path, (u_char *) src,
194 sizeof(s.t->clean_path));
195 if (brix_cache_origin_open(s.t, &s.oc, src_fh) != 0) {
196 errno = sd_xroot_errno(s.t);
197 ns_close(&s);
198 return NGX_ERROR;
199 }
200 if (brix_cache_origin_open_write(s.t, &s.oc, dst, 0644, dst_fh) != 0) {
201 errno = sd_xroot_errno(s.t);
202 brix_cache_origin_close_file(&s.oc, src_fh);
203 ns_close(&s);
204 return NGX_ERROR;
205 }
206 rc = sd_xroot_copy_body(s.t, &s.oc, src_fh, dst_fh, bytes_out);
207 e = errno;
208 brix_cache_origin_close_file(&s.oc, dst_fh);
209 brix_cache_origin_close_file(&s.oc, src_fh);
210 errno = e;
211 ns_close(&s);
212 return rc;
213}

230/* listxattr_cred: enumerate extended attributes under the user's credential.
231 * The origin returns its user attrs as a NUL-separated list of CLIENT names
232 * (its own "user.U." stripped). Re-add the "user.U." prefix to each so the
233 * kXR_fattr list handler — which keeps "user.U.*" keys — recognizes them. */
234ssize_t
235sd_xroot_listxattr_cred(brix_sd_instance_t *inst, const char *path,
236 void *buf, size_t cap, const brix_sd_cred_t *cred)
237{
238 ns_session_t s;
239 char *raw;
240 const size_t rawcap = 65536;
241 ssize_t n;
242 size_t out = 0, i;
243
244 raw = malloc(rawcap);
245 if (raw == NULL) { errno = ENOMEM; return -1; }
246 if (ns_open(inst, cred, &s) != NGX_OK) { free(raw); return -1; }
247 n = brix_cache_origin_listfattr(s.t, &s.oc, path, raw, rawcap);
248 ns_close(&s);
249 if (n < 0) { free(raw); return -1; }
250
251 for (i = 0; i < (size_t) n; ) {
252 size_t nl = strnlen(raw + i, (size_t) n - i);
253
254 if (nl == 0) { i += 1; continue; } /* skip an empty entry */
255 if (buf != NULL && cap > 0) {
256 if (out + SD_XROOT_FATTR_PFX_LEN + nl + 1 > cap) {
257 free(raw); errno = ERANGE; return -1;
258 }
259 ngx_memcpy((char *) buf + out, SD_XROOT_FATTR_PFX,
260 SD_XROOT_FATTR_PFX_LEN);
261 ngx_memcpy((char *) buf + out + SD_XROOT_FATTR_PFX_LEN, raw + i, nl);
262 ((char *) buf)[out + SD_XROOT_FATTR_PFX_LEN + nl] = '\0';
263 }
264 out += SD_XROOT_FATTR_PFX_LEN + nl + 1;
265 i += nl + 1;
266 }
267 free(raw);
268 return (ssize_t) out;
269}
