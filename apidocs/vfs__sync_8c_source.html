1/*
2 * vfs_sync.c — VFS handle-level truncate and durability.
3 *
4 * WHAT: Implements brix_vfs_truncate() (resize an open handle to `length`),
5 * brix_vfs_sync() (flush an open handle to stable storage), and
6 * brix_vfs_file_read_advise() (advisory read-ahead hint on the handle).
7 *
8 * WHY: kXR_truncate and the sync/commit step of writes (kXR_sync, WebDAV PUT
9 * finalisation) operate on an already-open handle rather than a path, so
10 * they live with the file-handle ops; truncate must also keep the handle's
11 * cached size in step with the file.
12 *
13 * HOW: truncate validates the handle/fd and a non-negative length, runs a VFS
14 * I/O-core TRUNCATE job, and on success updates fh->size so later reads
15 * see the new length. sync validates the handle and runs a VFS I/O-core
16 * SYNC job. Both are unmetered handle operations (the surrounding write
17 * op records the metric) returning NGX_OK / NGX_ERROR with errno set.
18 */

23 * own cache invalidation */

25/* Resize the open handle to `length` (ftruncate) and update the cached
26 * fh->size. NGX_ERROR with errno set on a bad handle or negative length. */
27ngx_int_t
28brix_vfs_truncate(brix_vfs_file_t *fh, off_t length)
29{
30 brix_vfs_job_t job;
31
32 if (fh == NULL || fh->obj.fd == NGX_INVALID_FILE || length < 0) {
33 errno = EINVAL;
34 return NGX_ERROR;
35 }
36
37 /* phase-105: the endpoint gate, from the policy the handle copied at open
38 * and BEFORE the capability question — a read-only export answers EROFS
39 * whatever the backend can do (Appendix I.5 precedence). */
40 if (brix_vfs_gate_file_mutation(fh, BRIX_VFS_MUTATE_TRUNCATE) != NGX_OK)
41 {
42 return NGX_ERROR;
43 }
44
45 /* phase-71: capability gate — a backend without CAP_TRUNCATE rejects resize
46 * uniformly rather than issuing an ftruncate the driver cannot honor. */
47 if (fh->obj.inst != NULL
48 && !(brix_sd_caps(fh->obj.inst) & BRIX_SD_CAP_TRUNCATE))
49 {
50 errno = ENOTSUP;
51 return NGX_ERROR;
52 }
53
54 brix_vfs_job_truncate_init(&job, fh->obj.fd, length);
55 brix_vfs_io_execute(&job);
56 if (job.io_errno != 0) {
57 errno = job.io_errno;
58 return NGX_ERROR;
59 }
60
61 fh->size = length;
62 return NGX_OK;
63}

65/* Path-based truncate: resize the resolved ctx path to `length` WITHOUT opening a
66 * write handle when the backend offers a path-native truncate. Write-gated.
67 *
68 * WHY: Over a remote (root://) backend BriX auto-composes a write-stage tier; the
69 * old handler opened a WRITE handle to truncate, which staged the file (whole-
70 * file RECALL) and took an origin write-open that self-collides on commit —
71 * surfacing as kXR_Unsupported. A path-native truncate resizes the origin by
72 * name in one round-trip, no staging.
73 *
74 * HOW: Dispatch on the LEAF instance through brix_sd_truncate_path_maybe_cred so
75 * per-user credentials reach the leaf's truncate_path_cred slot (decorators
76 * carry only plain relays). The gate asks THE LEAF whether it has the slot,
77 * for the same reason: asking the top driver made the answer depend on which
78 * decorator happened to be composed on top — the stage tier relays
79 * truncate_path and the cache tier did not, so a cache-fronted root:// export
80 * skipped the path-native branch entirely and fell back into exactly the
81 * staging round trip this function exists to avoid. Backends with no
82 * path-native truncate (POSIX, http, s3) still take the original
83 * open(O_WRITE)+ftruncate+close fallback, which is also why the gate cannot
84 * just be the decorator's forwarder returning ENOTSUP.
85 * Unmetered like brix_vfs_truncate — the kXR_truncate handler logs access. */
86/* vfs_truncate_driver_path -- execute the native path-truncate branch.
87 *
88 * WHAT: Applies the lock and credential gates, translates the confined path,
89 * invokes the leaf truncate slot and invalidates a stale cache entry.
90 * WHY: The driver branch is independently coherent and keeping it outside the
91 * public router bounds the latter's control-flow complexity.
92 * HOW: Gate lock, acquire a credential when configured, derive the PFN, call
93 * the cred-aware slot, wipe the credential, then evict on success. */
94static ngx_int_t
95vfs_truncate_driver_path(brix_vfs_ctx_t *ctx, brix_sd_instance_t *leaf,
96 const char *path, off_t length)
97{
98 brix_sd_ucred_t store;
99 brix_sd_cred_t cred;
100 ngx_int_t rc;
101 int use_cred = 0, cred_err = 0, saved_errno;
102 char key[PATH_MAX];
103
104 if (brix_vfs_require_unlocked(ctx, BRIX_VFS_MUTATE_TRUNCATE) != NGX_OK) {
105 return NGX_ERROR;
106 }
107 ngx_memzero(&store, sizeof(store));
108 ngx_memzero(&cred, sizeof(cred));
109 if (brix_vfs_cred_gate_active(ctx)
110 && brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err) != NGX_OK)
111 {
112 errno = cred_err ? cred_err : EACCES;
113 return NGX_ERROR;
114 }
115 if (brix_path_resolved_to_pfn(ctx, path, key, sizeof(key)) != NGX_OK) {
116 brix_sd_ucred_wipe(&store);
117 return NGX_ERROR;
118 }
119 rc = brix_sd_truncate_path_maybe_cred(leaf, key, length,
120 use_cred ? &cred : NULL);
121 saved_errno = errno;
122 brix_sd_ucred_wipe(&store);
123 if (rc == NGX_OK) {
124 brix_metric_cache_evicted(brix_vfs_metrics_proto(ctx),
125 brix_sd_cache_evict(ctx->sd, key));
126 }
127 errno = saved_errno;
128 return rc;
129}

131/* brix_vfs_truncate_path -- resize the resolved path by its best backend verb.
132 *
133 * WHAT: Uses path-native truncate when available, otherwise a confined
134 * write-open/ftruncate/close. Returns NGX_OK or NGX_ERROR with errno.
135 * WHY: Remote drivers avoid a whole-file staging round trip while drivers
136 * without a path slot retain the uniform handle fallback.
137 * HOW: Validate and gate, select the leaf capability, then dispatch either
138 * vfs_truncate_driver_path or the VFS handle sequence. */
139ngx_int_t
140brix_vfs_truncate_path(brix_vfs_ctx_t *ctx, off_t length)
141{
142 brix_sd_instance_t *leaf;
143 const brix_sd_driver_t *leaf_drv;
144 const char *path;
145
146 if (length < 0) {
147 errno = EINVAL;
148 return NGX_ERROR;
149 }
150 if (brix_vfs_confined_mutation_checked(ctx,
151 BRIX_VFS_MUTATE_TRUNCATE) != NGX_OK)
152 {
153 return NGX_ERROR;
154 }
155
156 path = brix_vfs_ctx_path(ctx);
157 leaf = brix_vfs_ns_leaf(ctx->sd);
158 leaf_drv = (leaf != NULL) ? leaf->driver : NULL;
159
160 if (leaf_drv != NULL
161 && (leaf_drv->truncate_path != NULL
162 || leaf_drv->truncate_path_cred != NULL))
163 {
164 return vfs_truncate_driver_path(ctx, leaf, path, length);
165 }
166
167 /* Fallback: no path-native truncate — open a write handle, resize, close. */
168 {
169 brix_vfs_file_t *fh;
170 int vfs_err = 0;
171
172 fh = brix_vfs_open(ctx, BRIX_VFS_O_WRITE, &vfs_err);
173 if (fh == NULL) {
174 errno = vfs_err;
175 return NGX_ERROR;
176 }
177 if (brix_vfs_truncate(fh, length) != NGX_OK) {
178 int e = errno;
179 brix_vfs_close(fh, ctx->log);
180 errno = e;
181 return NGX_ERROR;
182 }
183 brix_vfs_close(fh, ctx->log);
184 return NGX_OK;
185 }
186}

188/* Flush the open handle to stable storage (fsync). NGX_ERROR with errno set on
189 * a bad handle or fsync failure. */
190ngx_int_t
191brix_vfs_sync(brix_vfs_file_t *fh)
192{
193 brix_vfs_job_t job;
194
195 /* A memory-served handle (no kernel fd) can still honor a durability
196 * barrier through its driver's fsync slot — the root:// backend maps it
197 * to kXR_sync on the origin write handle. Only a handle with neither an
198 * fd nor a dispatchable driver object is invalid here. */
199 if (fh == NULL
200 || (fh->obj.fd == NGX_INVALID_FILE
201 && (fh->obj.driver == NULL || fh->obj.driver->fsync == NULL
202 || fh->obj.state == NULL)))
203 {
204 errno = EINVAL;
205 return NGX_ERROR;
206 }
207
208 /* phase-105: a durability barrier is only ever owed for bytes this endpoint
209 * was allowed to write, so it is gated with them (never EACCES). */
210 if (brix_vfs_gate_file_mutation(fh, BRIX_VFS_MUTATE_SYNC) != NGX_OK)
211 {
212 return NGX_ERROR;
213 }
214
215 brix_vfs_job_sync_init(&job, fh->obj.fd);
216 if (fh->obj.fd == NGX_INVALID_FILE) {
217 /* Bind the driver object so the executor dispatches driver->fsync
218 * instead of wrapping the (invalid) fd. */
219 brix_vfs_job_set_obj(&job, &fh->obj);
220 }
221 brix_vfs_io_execute(&job);
222 if (job.io_errno != 0) {
223 errno = job.io_errno;
224 return NGX_ERROR;
225 }
226
227 return NGX_OK;
228}

230/* Advisory read-ahead hint on the open handle (phase-56 B-2). Dispatches to the
231 * driver's read_advise slot; a backend without the slot succeeds as a no-op, so
232 * callers hint unconditionally and never branch on backend type. */
233ngx_int_t
234brix_vfs_file_read_advise(brix_vfs_file_t *fh, off_t off, size_t len,
235 int advice)
236{
237 if (fh == NULL || fh->obj.driver == NULL || off < 0) {
238 errno = EINVAL;
239 return NGX_ERROR;
240 }
241
242 if (fh->obj.driver->read_advise == NULL) {
243 return NGX_OK;
244 }
245
246 return fh->obj.driver->read_advise(&fh->obj, off, len, advice);
247}
