1/*
2 * vfs_sync.c — VFS handle-level truncate and durability.
3 *
4 * WHAT: Implements brix_vfs_truncate() (resize an open handle to `length`),
5 * brix_vfs_sync() (flush an open handle to stable storage), and
6 * brix_vfs_file_read_advise() (advisory read-ahead hint on the handle).
7 *
8 * WHY: kXR_truncate and the sync/commit step of writes (kXR_sync, WebDAV PUT
9 * finalisation) operate on an already-open handle rather than a path, so
10 * they live with the file-handle ops; truncate must also keep the handle's
11 * cached size in step with the file.
12 *
13 * HOW: truncate validates the handle/fd and a non-negative length, runs a VFS
14 * I/O-core TRUNCATE job, and on success updates fh->size so later reads
15 * see the new length. sync validates the handle and runs a VFS I/O-core
16 * SYNC job. Both are unmetered handle operations (the surrounding write
17 * op records the metric) returning NGX_OK / NGX_ERROR with errno set.
18 */
19#include "vfs_internal.h"
20#include "vfs_io_core.h"

22/* Resize the open handle to `length` (ftruncate) and update the cached
23 * fh->size. NGX_ERROR with errno set on a bad handle or negative length. */
24ngx_int_t
25brix_vfs_truncate(brix_vfs_file_t *fh, off_t length)
26{
27 brix_vfs_job_t job;
28
29 if (fh == NULL || fh->obj.fd == NGX_INVALID_FILE || length < 0) {
30 errno = EINVAL;
31 return NGX_ERROR;
32 }
33
34 /* phase-71: capability gate — a backend without CAP_TRUNCATE rejects resize
35 * uniformly rather than issuing an ftruncate the driver cannot honor. */
36 if (fh->obj.inst != NULL
37 && !(brix_sd_caps(fh->obj.inst) & BRIX_SD_CAP_TRUNCATE))
38 {
39 errno = ENOTSUP;
40 return NGX_ERROR;
41 }
42
43 brix_vfs_job_truncate_init(&job, fh->obj.fd, length);
44 brix_vfs_io_execute(&job);
45 if (job.io_errno != 0) {
46 errno = job.io_errno;
47 return NGX_ERROR;
48 }
49
50 fh->size = length;
51 return NGX_OK;
52}

54/* Path-based truncate: resize the resolved ctx path to `length` WITHOUT opening a
55 * write handle when the backend offers a path-native truncate. Write-gated.
56 *
57 * WHY: Over a remote (root://) backend BriX auto-composes a write-stage tier; the
58 * old handler opened a WRITE handle to truncate, which staged the file (whole-
59 * file RECALL) and took an origin write-open that self-collides on commit —
60 * surfacing as kXR_Unsupported. A path-native truncate resizes the origin by
61 * name in one round-trip, no staging.
62 *
63 * HOW: When the bound driver advertises truncate_path (remote xroot, directly or
64 * via the stage decorator's forwarder), dispatch on the LEAF instance through
65 * brix_sd_truncate_path_maybe_cred so per-user credentials reach the leaf's
66 * truncate_path_cred slot (decorators carry only plain relays). Otherwise
67 * (POSIX and any backend without the slot) fall back to the original
68 * open(O_WRITE)+ftruncate+close, preserving prior behavior exactly.
69 * Unmetered like brix_vfs_truncate — the kXR_truncate handler logs access. */
70ngx_int_t
71brix_vfs_truncate_path(brix_vfs_ctx_t *ctx, off_t length)
72{
73 const brix_sd_driver_t *drv;
74 const char *path;
75
76 if (length < 0) {
77 errno = EINVAL;
78 return NGX_ERROR;
79 }
80 if (brix_vfs_require_write(ctx) != NGX_OK) {
81 return NGX_ERROR;
82 }
83
84 path = brix_vfs_ctx_path(ctx);
85 drv = brix_vfs_ctx_driver(ctx);
86
87 if (drv != NULL && drv->truncate_path != NULL) {
88 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(ctx->sd);
89 brix_sd_ucred_t store;
90 brix_sd_cred_t cred;
91 ngx_int_t rc;
92 int use_cred = 0, cred_err = 0, saved_errno;
93 const char *key;
94
95 /* Zero before the gate: it fills only the active credential kind; an
96 * unzeroed cred hands a garbage inactive pointer to the driver's slot. */
97 ngx_memzero(&cred, sizeof(cred));
98
99 if (brix_vfs_cred_gate_active(ctx)) {
100 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
101 != NGX_OK)
102 {
103 errno = cred_err ? cred_err : EACCES;
104 return NGX_ERROR;
105 }
106 }
107
108 key = brix_vfs_export_relative(ctx, path);
109 rc = brix_sd_truncate_path_maybe_cred(leaf, key, length,
110 use_cred ? &cred : NULL);
111 saved_errno = errno;
112 brix_sd_ucred_wipe(&store); /* secret consumed; erase (A-4/T4) */
113 errno = saved_errno;
114 return rc;
115 }
116
117 /* Fallback: no path-native truncate — open a write handle, resize, close. */
118 {
119 brix_vfs_file_t *fh;
120 int vfs_err = 0;
121
122 fh = brix_vfs_open(ctx, BRIX_VFS_O_WRITE, &vfs_err);
123 if (fh == NULL) {
124 errno = vfs_err;
125 return NGX_ERROR;
126 }
127 if (brix_vfs_truncate(fh, length) != NGX_OK) {
128 int e = errno;
129 brix_vfs_close(fh, ctx->log);
130 errno = e;
131 return NGX_ERROR;
132 }
133 brix_vfs_close(fh, ctx->log);
134 return NGX_OK;
135 }
136}
