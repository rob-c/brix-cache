1/*
2 * io_monitor.h — the per-request / per-session I/O monitor record.
3 *
4 * WHAT: One small protocol-neutral record that accumulates, across every brix
5 * VFS operation of ONE HTTP request or ONE root:// session, the facts the
6 * uniform $brix_* monitoring surface reports at log time:
7 *
8 * bytes served / received $brix_bytes_served / $brix_bytes_received
9 * summed backend I/O time $brix_backend_time
10 * cache disposition $brix_cache_status
11 * primary operation + path $brix_op / $brix_path
12 * outcome class $brix_status
13 * operation count $brix_ops
14 * reported file checksum $brix_checksum
15 *
16 * WHY: Every value above already existed somewhere in brix (the JSON access
17 * log proves it) but nothing retained them per request/session, so the
18 * variables could not exist as pure handlers (phase-106 Appendix G, the
19 * "data-plane" rows). This record is that retention layer, kept
20 * deliberately small and free of protocol or HTTP knowledge so the VFS
21 * layer can fold into it through a bare pointer without depending on the
22 * plane that owns its lifetime. phase-110 W1-W4 extend it from the three
23 * phase-106 counters to the full uniform vocabulary.
24 *
25 * HOW: The owning plane allocates it ON THE EVENT LOOP — an HTTP plane on the
26 * request pool (stored in ngx_http_brix_common_module's ctx slot), the
27 * root plane embedded in its per-connection brix_ctx_t — and points every
28 * data-plane brix_vfs_ctx_t->io_monitor at it. The VFS layer folds:
29 * * the post-op observer (vfs_internal.h): latency, op/path/err by the
30 * weight rule below, op count;
31 * * brix_vfs_adopt_fd: the open as a read/write op (the client-facing
32 * serve is zero-copy and never reaches the observer, so the OPEN is
33 * what identifies a GET as "read");
34 * * brix_vfs_observe_cache: HIT/MISS/BYPASS beside the unified metric;
35 * * the mutation gate: FORBIDDEN on a read-only refusal;
36 * and the planes fold the two facts only they know: served bytes
37 * (result->bytes_sent) and the checksum they reported to the client.
38 *
39 * THREADING: single-writer. Exactly one op runs at a time for a request or
40 * session; the GET/PUT offload thread only WRITES scalar fields / memcpys
41 * into the pre-allocated fixed buffers below (never allocates), and the
42 * task done-handler orders those writes before the event-loop log phase
43 * reads them. Do not add a reader on any other thread, and never give
44 * this struct a pointer into a pool the thread could free.
45 *
46 * VALUE RULES (the "-" discipline): a field is "-" when the event did not
47 * happen, never a measured zero. `any` = at least one op observed;
48 * `have_op` = an op was recorded; `have_checksum` = one was reported.
49 */

60/* Bounded copy of the primary op's export-relative path. Matches the JSON
61 * access log's own bound (access_log.c path_json[1024]); a longer path is
62 * truncated, never dropped — the log shows a prefix rather than "-". */
63#define BRIX_IO_MONITOR_PATH_MAX 1024
64/* "alg:hex": brix_integrity_info_t alg_name[16] + ':' + hex[129]. */
65#define BRIX_IO_MONITOR_CKSUM_MAX (16 + 1 + 129)

67/* The tag is what brix_vfs_ctx_t forward-declares; keep it stable. */
68typedef struct brix_io_monitor_s {
69 uint64_t bytes; /* bytes SERVED to the client */
70 uint64_t bytes_received; /* bytes RECEIVED from the client (writes)*/
71 ngx_msec_t backend_usec; /* summed VFS op latency, microseconds */
72 uint32_t ops; /* observed VFS ops */
73 brix_metric_op_t op; /* primary op (valid iff have_op) */
74 brix_err_class_t err; /* outcome class of the primary op */
75 brix_cache_status_e cache; /* disposition; NONE = no decision */
76 unsigned op_weight; /* weight of the recorded primary op */
77 size_t path_len; /* 0 = none */
78 size_t checksum_len; /* 0 = none */
79 unsigned any:1; /* at least one brix I/O op was observed */
80 unsigned have_op:1;
81 unsigned have_checksum:1;
82 char path[BRIX_IO_MONITOR_PATH_MAX];
83 char checksum[BRIX_IO_MONITOR_CKSUM_MAX];
84} brix_io_monitor_t;

87/*
88 * brix_io_monitor_op_weight — which op describes the request.
89 *
90 * A GET is stat + open-for-read; a PROPFIND is stat + dirlist + stats; a COPY
91 * is copy + the reads and writes that implement it. The op the operator wants
92 * to see is the one with the highest weight; on equal weight the LATER op wins
93 * (a PROPPATCH is stat then xattr → xattr). Metadata probes weigh nothing so
94 * they never displace a data op, and the composite ops (copy/tpc) outweigh the
95 * primitive reads/writes they are made of.
96 */
97static ngx_inline unsigned
98brix_io_monitor_op_weight(brix_metric_op_t op)
99{
100 switch (op) {
101 case BRIX_METRIC_OP_COPY:
102 case BRIX_METRIC_OP_TPC:
103 return 3;
104 case BRIX_METRIC_OP_READ:
105 case BRIX_METRIC_OP_WRITE:
106 case BRIX_METRIC_OP_DELETE:
107 case BRIX_METRIC_OP_MKDIR:
108 case BRIX_METRIC_OP_RENAME:
109 return 2;
110 case BRIX_METRIC_OP_DIRLIST:
111 return 1;
112 case BRIX_METRIC_OP_STAT:
113 case BRIX_METRIC_OP_XATTR:
114 case BRIX_METRIC_OP_COUNT:
115 default:
116 return 0;
117 }
118}

159/*
160 * brix_io_monitor_record_op — candidate primary op (weight rule above).
161 *
162 * Copies `path` into the monitor's own buffer: the observer's path argument may
163 * be a caller STACK buffer (lock_check.c hands it a char[PATH_MAX]), so a stored
164 * pointer would dangle by log time. memcpy into a preallocated buffer is safe
165 * from the offload thread. The err class rides with the op so $brix_status is
166 * the outcome of the op $brix_op names, not of an incidental probe.
167 */
168static ngx_inline void
169brix_io_monitor_record_op(brix_io_monitor_t *m, brix_metric_op_t op,
170 const char *path, brix_err_class_t err)
171{
172 unsigned w;
173 size_t n;
174
175 if (m == NULL) {
176 return;
177 }
178 m->any = 1;
179 m->ops++;
180 w = brix_io_monitor_op_weight(op);
181 if (m->have_op && w < m->op_weight) {
182 return;
183 }
184 m->have_op = 1;
185 m->op = op;
186 m->op_weight = w;
187 m->err = err;
188 if (path == NULL || path[0] == '\0') {
189 m->path_len = 0;
190 return;
191 }
192 n = strlen(path);
193 if (n > sizeof(m->path) - 1) {
194 n = sizeof(m->path) - 1;
195 }
196 memcpy(m->path, path, n);
197 m->path[n] = '\0';
198 m->path_len = n;
199}

202/* Record an outcome class WITHOUT an op — the mutation gate's read-only
203 * refusal happens before any VFS op runs, and an HTTP plane's own refusal
204 * (401/403/404) may never reach the VFS at all. A recorded op's own class is
205 * not overwritten: the primary op's outcome stays authoritative. */
206static ngx_inline void
207brix_io_monitor_record_err(brix_io_monitor_t *m, brix_err_class_t err)
208{
209 if (m == NULL) {
210 return;
211 }
212 m->any = 1;
213 if (!m->have_op || m->err == BRIX_ERR_NONE) {
214 m->err = err;
215 }
216}

219/*
220 * brix_io_monitor_record_cache — fold a cache decision.
221 *
222 * MISS dominates HIT: a request that had to reach the origin for any part of
223 * its data was, for hit-rate purposes, a miss. BYPASS/NEGHIT only fill an
224 * undecided slot (a deliberate skip beside a real hit is still a hit).
225 */
226static ngx_inline void
227brix_io_monitor_record_cache(brix_io_monitor_t *m, brix_cache_status_e status)
228{
229 if (m == NULL || status == BRIX_CACHE_STATUS_NONE) {
230 return;
231 }
232 m->any = 1;
233 if (m->cache == BRIX_CACHE_STATUS_NONE || status == BRIX_CACHE_STATUS_MISS) {
234 m->cache = status;
235 }
236}

239/* Record the file checksum the plane REPORTED to the client, rendered
240 * "alg:hex" (INVARIANT #9: the algorithm travels with the digits so the value
241 * is never misread). Last one wins — a request reports at most one. */
242static ngx_inline void
243brix_io_monitor_record_checksum(brix_io_monitor_t *m, const char *alg,
244 const char *hex)
245{
246 size_t la, lh;
247
248 if (m == NULL || alg == NULL || hex == NULL || alg[0] == '\0'
249 || hex[0] == '\0')
250 {
251 return;
252 }
253 la = strlen(alg);
254 lh = strlen(hex);
255 if (la + 1 + lh > sizeof(m->checksum) - 1) {
256 return; /* not representable: leave "-" */
257 }
258 memcpy(m->checksum, alg, la);
259 m->checksum[la] = ':';
260 memcpy(m->checksum + la + 1, hex, lh);
261 m->checksum[la + 1 + lh] = '\0';
262 m->checksum_len = la + 1 + lh;
263 m->have_checksum = 1;
264 m->any = 1;
265}
