28ngx_int_t
29brix_krb5_deleg_credbytes(const u_char *payload, size_t dlen,
30 const u_char **cred, size_t *credlen)
31{
32 size_t off;
33
34 if (payload == NULL || cred == NULL || credlen == NULL
35 || dlen <= 4 || ngx_strncmp(payload, "krb5", 4) != 0)
36 {
37 return NGX_ERROR;
38 }
39
40 /* The official XrdSeckrb5 client NUL-terminates the "krb5" prefix; the native
41 * client emits a bare "krb5". A KRB_CRED begins with its ASN.1 APPLICATION
42 * tag (0x76), never 0x00, so skipping the optional NUL cannot truncate it.
43 * dlen > 4 guarantees payload[4] is in-bounds; a body-less "krb5\0" then
44 * skips to off == 5 and fails the emptiness check below. */
45 off = 4;
46 if (payload[4] == '\0') {
47 off = 5;
48 }
49
50 if (dlen <= off) {
51 return NGX_ERROR;
52 }
53
54 *cred = payload + off;
55 *credlen = dlen - off;
56 return NGX_OK;
57}

59ngx_int_t
60brix_krb5_send_fwdtgt(brix_ctx_t *ctx, ngx_connection_t *c)
61{
62 brix_gbuf g;
63 u_char *buf;
64 size_t total;
65
66 brix_gbuf_init(&g);
67 brix_gbuf_raw(&g, "krb5", 5); /* protocol name + NUL */
68 brix_gbuf_raw(&g, "fwdtgt", 7); /* forward-TGT continuation marker + NUL */
69 brix_gbuf_end(&g);
70
71 if (g.err) {
72 brix_gbuf_free(&g);
73 return brix_send_error(ctx, c, kXR_NoMemory, "krb5: out of memory");
74 }
75
76 total = XRD_RESPONSE_HDR_LEN + g.len;
77 buf = ngx_palloc(c->pool, total);
78 if (buf == NULL) {
79 brix_gbuf_free(&g);
80 return brix_send_error(ctx, c, kXR_NoMemory, "krb5: out of memory");
81 }
82
83 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_authmore,
84 (uint32_t) g.len, (ServerResponseHdr *) buf);
85 ngx_memcpy(buf + XRD_RESPONSE_HDR_LEN, g.p, g.len);
86 brix_gbuf_free(&g);
87
88 ngx_log_debug0(NGX_LOG_DEBUG_STREAM, c->log, 0,
89 "brix: krb5 round 1 -> fwdtgt (delegation requested)");
90 return brix_queue_response(ctx, c, buf, total);
91}

93ngx_int_t
94brix_krb5_deleg_origin_spn(const ngx_str_t *ccache, int forwardable,
95 const ngx_str_t *origin_host, const ngx_str_t *gateway_princ,
96 ngx_pool_t *pool, ngx_str_t *out_spn)
97{
98 char host[256];
99 char gwp[512];
100 u_char *spn;
101 size_t spnlen;
102
103 if (ccache == NULL || ccache->len == 0 || !forwardable
104 || origin_host == NULL || origin_host->len == 0
105 || origin_host->len >= sizeof(host)
106 || gateway_princ == NULL || gateway_princ->len == 0
107 || gateway_princ->len >= sizeof(gwp))
108 {
109 return NGX_DECLINED; /* a gate is unmet — nothing to bind */
110 }
111
112 ngx_memcpy(host, origin_host->data, origin_host->len);
113 host[origin_host->len] = '\0';
114 ngx_memcpy(gwp, gateway_princ->data, gateway_princ->len);
115 gwp[gateway_princ->len] = '\0';
116
117 spn = ngx_pnalloc(pool, 512);
118 if (spn == NULL) {
119 return NGX_ERROR;
120 }
121
122 if (brix_krb5_origin_princ_from_host(host, gwp, (char *) spn, 512) != NGX_OK) {
123 return NGX_ERROR; /* fail closed — do not bind a bad SPN */
124 }
125
126 spnlen = ngx_strlen(spn);
127 if (spnlen == 0) {
128 return NGX_ERROR;
129 }
130
131 out_spn->data = spn;
132 out_spn->len = spnlen;
133 return NGX_OK;
134}

180brix_krb5_deleg_park(brix_ctx_t *ctx, ngx_connection_t *c, krb5_context kctx,
181 krb5_auth_context auth_ctx, krb5_principal client, const char *cname)
182{

189 }
190
191 cln = ngx_pool_cleanup_add(c->pool, sizeof(*cl));

193 krb5_free_principal(kctx, cli_copy);
194 return NGX_ERROR;
195 }
196
197 ctx->krb5.auth_ctx = auth_ctx;
198 ctx->krb5.client = cli_copy;

201 sizeof(ctx->krb5.cname));
202
203 cl = cln->data;
204 cl->ctx = ctx;
205 cl->kctx = kctx;
206 cln->handler = brix_krb5_deleg_cleanup;
207 return NGX_OK;
208}

210/* Create a fresh EMPTY 0600 file to back the captured forwarded-TGT ccache,
211 * through the shared credential-write verb (phase-108 C11). Before C11 this
212 * honored $TMPDIR with a /tmp fallback — a world-traversable parent, the exact
213 * CWE-377 co-tenant race cred_stage.h exists to prevent, on a forwarded TGT.
214 * The fix is the unreachable 0700 per-uid parent, not the file's own mode:
215 * libkrb5 rewrites the FILE by name (keeping 0600), so the empty pre-created
216 * file only reserves the name — len == 0 is the intended shape. */
217static ngx_int_t
218brix_krb5_deleg_mkccache(ngx_connection_t *c, char *path, size_t pathlen)
219{
220 brix_cred_write_req_t req;
221
222 ngx_memzero(&req, sizeof(req));
223 req.log = c->log;
224 req.arm = BRIX_CRED_ARM_VOLATILE;
225 req.kind = BRIX_CRED_KIND_CCACHE;
226 req.prefix = "brix-krb5-fwd-";
227
228 if (brix_cred_write(&req, NULL, 0, path, pathlen) != 0) {
229 ngx_log_error(NGX_LOG_ERR, c->log, ngx_errno,
230 "brix: krb5 fwdtgt: cannot create temp ccache");
231 return NGX_ERROR;
232 }
233 return NGX_OK;
234}

236/* Release the capture entry's output handles (its private MEMORY ccache + the

252/* The serialising core of round 2: decode the KRB_CRED, run the production
253 * capture, and export the forwarded TGT to a 0600 FILE ccache at *path. Split
254 * from the entry so each stays within the CCN ceiling; the entry owns handle
255 * release regardless of the outcome here. */
256static ngx_int_t
257brix_krb5_capture_do(brix_ctx_t *ctx, ngx_connection_t *c, krb5_context kctx,
258 void **gss_cred, void **cap_cc, char *path, size_t pathlen)
259{
260 const u_char *cred;
261 size_t credlen;
262
263 if (brix_krb5_deleg_credbytes(ctx->recv.payload, ctx->recv.cur_dlen,
264 &cred, &credlen) != NGX_OK)
265 {
266 ngx_log_error(NGX_LOG_WARN, c->log, 0,
267 "brix: krb5 fwdtgt round 2: malformed forwarded credential");
268 return NGX_ERROR;
269 }
270
271 if (brix_krb5_capture_fwd_cred(kctx, ctx->krb5.auth_ctx, ctx->krb5.client,
272 cred, credlen, gss_cred, cap_cc, c->log)
273 != NGX_OK)
274 {
275 return NGX_ERROR; /* capture already logged the krb5 detail */
276 }
277
278 if (brix_krb5_deleg_mkccache(c, path, pathlen) != NGX_OK) {
279 return NGX_ERROR;
280 }
281
282 if (brix_krb5_cred_to_ccache(*gss_cred, path, c->log) != NGX_OK) {
283 (void) unlink(path);
284 return NGX_ERROR;
285 }
286 return NGX_OK;
287}

289ngx_int_t
290brix_krb5_deleg_capture(brix_ctx_t *ctx, ngx_connection_t *c,
291 ngx_stream_brix_srv_conf_t *conf)
292{
293 krb5_context kctx = conf->krb5.context;
294 void *gss_cred = NULL;
295 void *cap_cc = NULL;
296 char path[1024];
297 ngx_int_t rc;
298
299 if (ctx->krb5.round != 1 || ctx->krb5.auth_ctx == NULL
300 || ctx->krb5.client == NULL || kctx == NULL)
301 {
302 return NGX_ERROR;
303 }
304
305 rc = brix_krb5_capture_do(ctx, c, kctx, &gss_cred, &cap_cc,
306 path, sizeof(path));
307
308 /* The round-1 handles have done their job on both paths — release them and
309 * drop the round state; likewise the capture entry's output handles. */
310 brix_krb5_capture_out_release(kctx, gss_cred, cap_cc);
311 brix_krb5_deleg_release(ctx, kctx);
312
313 if (rc != NGX_OK) {
314 return NGX_ERROR;
315 }
316
317 ngx_cpystrn((u_char *) ctx->krb5.ccache, (u_char *) path,
318 sizeof(ctx->krb5.ccache));
319
320 /* Operator-visible confirmation the inbound leg completed: the mapped identity
321 * only (cname is already logged by the session grant), never the TGT bytes. */
322 ngx_log_error(NGX_LOG_INFO, c->log, 0,
323 "brix: krb5 delegation captured forwarded TGT for \"%s\"",
324 ctx->krb5.cname);
325 return NGX_OK;
326}
