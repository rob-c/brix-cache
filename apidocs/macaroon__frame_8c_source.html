17int
18brix_macaroon_packet_len(const unsigned char *p)
19/* WHAT: parse a 4-character hex-encoded packet length from macaroon binary data.
20 * WHY: macaroon packets are prefixed with a hex-encoded length field; this
21 * converts the first 4 hex characters into an integer for bounds checking
22 * before reading packet data.
23 * HOW: brix_hex_from_char() on each of p[0..3]; reject if any nibble is invalid
24 * (<0); combine via (v0<<12)|(v1<<8)|(v2<<4)|v3; return value or -1. */
25{
26 int v0, v1, v2, v3;
27 v0 = brix_hex_from_char(p[0]);
28 v1 = brix_hex_from_char(p[1]);
29 v2 = brix_hex_from_char(p[2]);
30 v3 = brix_hex_from_char(p[3]);
31 if (v0 < 0 || v1 < 0 || v2 < 0 || v3 < 0) return -1;
32 return (v0 << 12) | (v1 << 8) | (v2 << 4) | v3;
33}

35int
36brix_macaroon_scan_frames(const unsigned char *bin, size_t len)
37/* WHAT: walk the macaroon binary as length-prefixed packets, enforcing the
38 * per-packet framing bounds and counting well-framed packets.
39 * WHY: the length-prefix walk is the pre-auth framing surface; expressing it
40 * purely (no HMAC/claims) makes the `plen < 4 || p + plen > end` bound
41 * machine-checkable under ASan+UBSan on arbitrary bytes.
42 * HOW: loop while at least 4 header bytes remain; decode plen; reject on an
43 * out-of-range length prefix; strip an optional trailing '\n' from the
44 * body span (as the parser does) without dereferencing past the packet;
45 * advance by plen; return the packet count or -1 on the first bad frame. */
46{
47 const unsigned char *p = bin;
48 const unsigned char *end = bin + len;
49 int count = 0;
50
51 while (p + 4 <= end) {
52 int plen = brix_macaroon_packet_len(p);
53 size_t dlen;
54
55 if (plen < 4 || p + plen > end) {
56 return -1;
57 }
58
59 /* Body span is [p+4, p+plen); the parser strips one trailing '\n'. This
60 * touches only bytes already proven in-bounds by the check above. */
61 dlen = (size_t) (plen - 4);
62 if (dlen > 0 && p[4 + dlen - 1] == '\n') {
63 dlen--;
64 }
65 (void) dlen;
66
67 p += plen;
68 count++;
69 }
70
71 return count;
72}
