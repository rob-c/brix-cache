1/*
2 * src/platform/darwin/fs_watcher.c - macOS filesystem monitoring
3 *
4 * Phase 2: Uses kqueue EVFILT_VNODE for file monitoring.
5 * Phase 4: Can be extended to use FSEvents for directory-wide monitoring.
6 *
7 * kqueue EVFILT_VNODE advantages:
8 * - Simple API, already using kqueue for events
9 * - Per-file watching (like inotify)
10 * - No additional frameworks needed
11 *
12 * kqueue EVFILT_VNODE limitations:
13 * - Must have file open to watch it
14 * - Doesn't work well for directories
15 * - No recursive watching
16 *
17 * FSEvents (Phase 4) advantages:
18 * - Directory-wide monitoring
19 * - Recursive watching support
20 * - Efficient for large directory trees
21 */

59int
60brix_plat_fs_watcher_init(brix_plat_fs_watcher_t *watcher)
61{
62 int flags;
63
64 if (watcher == NULL) {
65 errno = EINVAL;
66 return -1;
67 }
68
69 memset(watcher, 0, sizeof(brix_plat_fs_watcher_t));
70
71 /* Create kqueue with CLOEXEC */
72 watcher->kqueue_fd = kqueue();
73 if (watcher->kqueue_fd < 0) {
74 return -1;
75 }
76
77 /* Set CLOEXEC */
78 flags = fcntl(watcher->kqueue_fd, F_GETFD);
79 if (flags != -1) {
80 fcntl(watcher->kqueue_fd, F_SETFD, flags | FD_CLOEXEC);
81 }
82
83 watcher->watches = NULL;
84 watcher->watch_count = 0;
85
86 return 0;
87}

89void
90brix_plat_fs_watcher_destroy(brix_plat_fs_watcher_t *watcher)
91{
92 brix_watch_desc_t *w, *next;
93
94 if (watcher != NULL) {
95 /* Close all watched files and free watch descriptors */
96 w = watcher->watches;
97 while (w != NULL) {
98 next = w->next;
99 if (w->fd >= 0) {
100 close(w->fd);
101 }
102 free(w);
103 w = next;
104 }
105
106 if (watcher->kqueue_fd >= 0) {
107 close(watcher->kqueue_fd);
108 }
109
110 /* Note: Caller allocated watcher, so we don't free it here */
111 }
112}

117static brix_watch_desc_t *
118brix_watch_desc_create(brix_plat_fs_watcher_t *watcher, int fd, const char *path, uint32_t events)
119{
120 brix_watch_desc_t *wd;
121
122 wd = calloc(1, sizeof(brix_watch_desc_t));
123 if (wd == NULL) {
124 return NULL;
125 }
126
127 wd->fd = fd;
128 wd->wd = watcher->watch_count + 1; /* Assign unique ID */
129 wd->events = events;
130 strncpy(wd->path, path, sizeof(wd->path) - 1);
131 wd->path[sizeof(wd->path) - 1] = '\0';
132 wd->next = NULL;
133
134 return wd;
135}

148/*
149 * Helper: Remove and free watch descriptor
150 */
151static void
152brix_watch_desc_remove(brix_plat_fs_watcher_t *watcher, brix_watch_desc_t *wd, brix_watch_desc_t *prev)
153{
154 if (prev == NULL) {
155 watcher->watches = wd->next;
156 } else {
157 prev->next = wd->next;
158 }
159
160 if (wd->fd >= 0) {
161 /* Remove from kqueue before closing */
162 struct kevent ev;
163 EV_SET(&ev, (uintptr_t)wd->fd, EVFILT_VNODE, EV_DELETE, 0, 0, NULL);
164 kevent(watcher->kqueue_fd, &ev, 1, NULL, 0, NULL);
165
166 close(wd->fd);
167 }
168
169 free(wd);
170 watcher->watch_count--;
171}

173int
174brix_plat_fs_watcher_add(brix_plat_fs_watcher_t *watcher, const char *path, uint32_t events)
175{
176 brix_watch_desc_t *wd;
177 struct kevent ev;
178 int fd;
179 int filter_flags;
180 uint32_t vnode_flags = 0;
181
182 if (watcher == NULL || path == NULL) {
183 errno = EINVAL;
184 return -1;
185 }
186
187 /* kqueue EVFILT_VNODE requires the file to be open */
188 fd = open(path, O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: SEAM_CORRECT - PAL kqueue watch descriptor */
189 if (fd < 0) {
190 return -1; /* errno set by open() */
191 }
192
193 /* Check if it's a directory - kqueue vnode filters work differently for dirs */
194 struct stat st;
195 if (fstat(fd, &st) < 0) {
196 close(fd);
197 return -1;
198 }
199
200 if (S_ISDIR(st.st_mode)) {
201 /* kqueue EVFILT_VNODE doesn't work well for directories.
202 * For Phase 2, we'll skip directory watches.
203 * Phase 4 can use FSEvents for directory monitoring.
204 */
205 close(fd);
206 errno = EISDIR;
207 return -1;
208 }
209
210 /* Translate event mask to kqueue vnode flags */
211 filter_flags = EV_ADD | EV_ENABLE | EV_CLEAR;
212
213 /* Note: We watch for all common events and filter in brix_plat_fs_watcher_next() */
214 vnode_flags = NOTE_DELETE | NOTE_WRITE | NOTE_EXTEND |
215#if defined(NOTE_TRUNCATE)
216 NOTE_TRUNCATE |
217#endif
218 NOTE_ATTRIB | NOTE_RENAME;
219
220 /* Register with kqueue */
221 EV_SET(&ev, (uintptr_t)fd, EVFILT_VNODE, filter_flags, vnode_flags, 0, NULL);
222
223 if (kevent(watcher->kqueue_fd, &ev, 1, NULL, 0, NULL) < 0) {
224 close(fd);
225 return -1; /* errno set by kevent() */
226 }
227
228 /* Create and add watch descriptor */
229 wd = brix_watch_desc_create(watcher, fd, path, 0);
230 if (wd == NULL) {
231 /* Remove from kqueue */
232 EV_SET(&ev, (uintptr_t)fd, EVFILT_VNODE, EV_DELETE, 0, 0, NULL);
233 kevent(watcher->kqueue_fd, &ev, 1, NULL, 0, NULL);
234 close(fd);
235 errno = ENOMEM;
236 return -1;
237 }
238
239 brix_watch_desc_add(watcher, wd);
240 return wd->wd; /* Return watch descriptor ID */
241}

243int
244brix_plat_fs_watcher_rm(brix_plat_fs_watcher_t *watcher, int wd)
245{
246 brix_watch_desc_t *current, *prev;
247
248 if (watcher == NULL || wd < 0) {
249 errno = EINVAL;
250 return -1;
251 }
252
253 /* Find the watch descriptor by ID */
254 prev = NULL;
255 current = watcher->watches;
256
257 while (current != NULL) {
258 if (current->wd == wd) {
259 brix_watch_desc_remove(watcher, current, prev);
260 return 0;
261 }
262 prev = current;
263 current = current->next;
264 }
265
266 /* Watch descriptor not found */
267 errno = ENOENT;
268 return -1;
269}

271/* ---- Translate native filesystem notification flags ----
272 * WHAT: Return the corresponding public PAL event mask.
273 * WHY: Keep decoding independent of event wait and descriptor ownership.
274 * HOW: 1. Start with no events. 2. Accumulate each matching native flag.
275 */
276static uint32_t
277brix_native_watch_events(uint32_t mask)
278{
279 /* Translate kqueue vnode flags to platform-independent mask */
280 uint32_t events = 0;
281 if (mask & NOTE_DELETE) {
282 events |= BRIX_FS_EVENT_DELETE;
283 }
284 if (mask & NOTE_WRITE) {
285 events |= BRIX_FS_EVENT_WRITE;
286 }
287 if (mask & NOTE_EXTEND) {
288 events |= BRIX_FS_EVENT_CREATE; /* File extended/created */
289 }
290 if (mask & NOTE_ATTRIB) {
291 events |= BRIX_FS_EVENT_ATTRIB;
292 }
293 if (mask & NOTE_RENAME) {
294 events |= BRIX_FS_EVENT_RENAME;
295 }
296
297 return events;
298}

300int
301brix_plat_fs_watcher_next(brix_plat_fs_watcher_t *watcher, brix_plat_fs_event_t *event, int timeout_ms)
302{
303 struct kevent ev;
304 struct timespec ts;
305 struct timespec *tsp;
306 brix_watch_desc_t *wd;
307 int nready;
308
309 if (watcher == NULL || event == NULL) {
310 errno = EINVAL;
311 return -1;
312 }
313
314 /* Set up timeout */
315 if (timeout_ms < 0) {
316 tsp = NULL; /* Infinite */
317 } else {
318 ts.tv_sec = timeout_ms / 1000;
319 ts.tv_nsec = (timeout_ms % 1000) * 1000000;
320 tsp = &ts;
321 }
322
323 /* Wait for kqueue events */
324 nready = kevent(watcher->kqueue_fd, NULL, 0, &ev, 1, tsp);
325
326 if (nready < 0) {
327 if (errno == EINTR) {
328 return 0; /* Interrupted by signal */
329 }
330 return -1; /* Real error */
331 }
332
333 if (nready == 0) {
334 return 0; /* Timeout */
335 }
336
337 /* Check that it's a vnode event */
338 if (ev.filter != EVFILT_VNODE) {
339 errno = EINVAL;
340 return -1;
341 }
342
343 /* Find the watch descriptor for this fd */
344 wd = watcher->watches;
345 while (wd != NULL) {
346 if ((uintptr_t)wd->fd == (uintptr_t)ev.ident) {
347 break;
348 }
349 wd = wd->next;
350 }
351
352 if (wd == NULL) {
353 /* Orphaned event - shouldn't happen */
354 errno = ENOENT;
355 return -1;
356 }
357
358 /* Copy path from watch descriptor */
359 strncpy(event->path, wd->path, sizeof(event->path) - 1);
360 event->path[sizeof(event->path) - 1] = '\0';
361
362 event->events = brix_native_watch_events(ev.fflags);
363
364 event->cookie = 0; /* kqueue doesn't provide cookies like inotify */
365 /* Timestamp: kqueue EVFILT_VNODE doesn't provide event timestamps.
366 * Could use clock_gettime(CLOCK_REALTIME) for approximate time.
367 * Current: 0 (not available in Phase 2 implementation)
368 */
369 event->timestamp = 0;
370
371 return 1; /* Event returned */
372}
