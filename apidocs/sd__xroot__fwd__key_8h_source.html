1/*
2 * sd_xroot_fwd_key.h — parse a forwarding-proxy key into an origin target.
3 *
4 * WHAT: The ngx-free leaf of the forwarding root:// driver (sd_xroot_fwd.c).
5 * A client of a `forward://` export names the origin INSIDE the path
6 * (`/root://host:port//file`, the XrdPss forwarding convention); this
7 * header turns that key into {host, port, tls, remote path} and answers
8 * the operator's `permit=` host allowlist.
9 * WHY: The verdicts here decide which remote host a proxy will dial on a
10 * client's say-so, so they live in a pure function with a standalone
11 * unit test (tests/test_sd_xroot_fwd_key.c) rather than inside the
12 * nginx-coupled driver.
13 * HOW: libc string scanning over caller buffers; the host:port split is the
14 * shared core/compat/host_split.h and the host-pattern match is the TPC
15 * egress guard's brix_tpc_host_pattern_match, so a forwarded open and a
16 * TPC pull agree on what ".example.org" permits.
17 */

34/*
35 * Parse `key` — either the wire shape `/root://host[:port]//path` or the
36 * VFS-shaped `/root:/host[:port]/path` (brix_candidate_append_req collapses
37 * slash runs) — into *t. Returns 0 on success or an errno:
38 * ENOENT the first component names no "<scheme>:" — the client did
39 * not name an origin, so there is nothing to forward to
40 * ENOTSUP a scheme other than root / roots
41 * EINVAL an empty or malformed authority (host_split rules, or a
42 * character outside [A-Za-z0-9.-:[]])
43 * ENAMETOOLONG the host or the remote path overflows the target buffers
44 * The remote path is always absolute ("/" when the key ends at the host).
45 */
46int brix_sd_xroot_fwd_parse_key(const char *key,
47 brix_sd_xroot_fwd_target_t *t);

49/*
50 * Is `host` allowed by the space-separated `permit` list? Each entry is
51 * either an exact host or a leading-dot domain suffix (".example.org"), as
52 * brix_tpc_host_pattern_match defines them. Fails closed: an empty or NULL
53 * list permits nothing. Returns 1 = permitted, 0 = refused.
54 */
55int brix_sd_xroot_fwd_host_permitted(const char *permit, const char *host);
