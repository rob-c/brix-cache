9void
10dx_probe_auth(const brix_conn *c, doctor_ep *e)
11{
12 int anon = (c->diag.chosen_auth == NULL);

14 dx_record(e, &(dx_note){ "auth", DX_WARN, 0,

16 "provide matching credentials (--auth + token/proxy) if operations are denied" });
17 } else {
18 dx_record(e, &(dx_note){ "auth", DX_OK, 0,
19 anon ? "anonymous (server offered no auth)" : "authenticated", "" });

24/* namespace: the export root must stat as a directory and be listable. */
25void
26dx_probe_namespace(brix_conn *c, doctor_ep *e)
27{
28 brix_statinfo si;
29 brix_status st;
30 brix_dirent *ents = NULL;
31 size_t n = 0;
32
33 brix_status_clear(&st);
34 if (brix_stat(c, "/", &si, &st) != 0) {
35 dx_record_status(e, "namespace", &st);
36 return;
37 }
38 brix_status_clear(&st);
39 if (brix_dirlist(c, "/", 0, &ents, &n, &st) != 0) {
40 dx_record_status(e, "namespace", &st);
41 return;
42 }
43 {
44 /* Count visible (non-dot) entries — the server keeps its own dotfiles
45 * (e.g. a checkpoint-recovery lock) in the root, so a naive count is
46 * never zero; only an absence of real data is a meaningful signal. */
47 size_t i, visible = 0;
48 for (i = 0; i < n; i++) {
49 if (ents[i].name[0] != '.') { visible++; }
50 }
51 free(ents);
52 if (visible == 0) {
53 dx_record(e, &(dx_note){ "namespace", DX_WARN, 0,
54 "export root has no visible files (empty or wrong brix_root)",
55 "confirm data is present under the configured export root" });
56 } else {
57 dx_record(e, &(dx_note){ "namespace", DX_OK, 0, "export root listable", "" });
58 }
59 }
60}

63/*
64 * read path: stat the target (note kXR_offline = on tape), then open it and read one
65 * block. A failure is classified by the (read, kxr) rule; offline is a tape signal.
66 */
67void
68dx_probe_read(brix_conn *c, const char *target, doctor_ep *e)
69{
70 brix_statinfo si;
71 brix_status st;
72 brix_file f;
73
74 brix_status_clear(&st);
75 if (brix_stat(c, target, &si, &st) != 0) {
76 dx_record_status(e, "read", &st);
77 return;
78 }
79 if (si.flags & kXR_offline) {
80 e->offline_seen = 1;
81 dx_record(e, &(dx_note){ "read", DX_WARN, 0, "file is offline (on tape/cache, not staged)",
82 "issue a stage/prepare and retry after recall (use --allow-write for an active stage probe)" });
83 return;
84 }
85 brix_status_clear(&st);
86 if (brix_file_open_read(c, target, &f, &st) != 0) {
87 dx_record_status(e, "read", &st);
88 return;
89 }
90 {
91 uint8_t buf[4096]; /* stack-backed: a read probe proves the path, no malloc */
92 brix_status rst;
93 ssize_t r;
94 brix_status_clear(&rst);
95 r = brix_file_read(c, &f, 0, buf, sizeof(buf), &rst);
96 if (r < 0) {
97 brix_file_close(c, &f, &st);
98 dx_record_status(e, "read", &rst);
99 return;
100 }
101 }
102 brix_file_close(c, &f, &st);
103 dx_record(e, &(dx_note){ "read", DX_OK, 0, "read path healthy", "" });
104}

107/*
108 * checksum integrity: compare the server's advertised checksum against one recomputed
109 * locally from the bytes we downloaded. A disagreement means a stale checksum DB or
110 * on-disk corruption — a class of bug only a cross-check can surface.
111 */
112void
113dx_probe_checksum(brix_conn *c, const char *target, doctor_ep *e)
114{
115 char srv[160], loc[160];
116 brix_status st;
117 int fd;
118 char tmpl[] = "/tmp/xrddiag-dx.XXXXXX";
119 int64_t got = 0;
120
121 /* Request adler32 (the client can recompute it) and compare the bare hex,
122 * mirroring the proven `check` checksum-works probe. */
123 brix_status_clear(&st);
124 if (brix_query_cksum(c, target, "adler32", srv, sizeof(srv), &st) != 0) {
125 if (st.kxr == kXR_Unsupported) {
126 dx_record_status(e, "checksum", &st);
127 }
128 return; /* server simply doesn't expose a checksum — not a problem */
129 }
130 fd = mkstemp(tmpl);
131 if (fd < 0) {
132 return;
133 }
134 brix_status_clear(&st);
135 if (download_to_fd(c, target, fd, &got, &st) == 0
136 && brix_cksum_fd(fd, XRDC_CK_ADLER32, loc, sizeof(loc), &st) == 0) {
137 if (strcmp(srv, loc) == 0) {
138 dx_record(e, &(dx_note){ "checksum", DX_OK, 0, "server checksum matches read data", "" });
139 } else {
140 dx_record(e, &(dx_note){ "checksum", DX_FAIL, 0,
141 "server checksum disagrees with the bytes read (stale checksum DB or data corruption)",
142 "recompute/repair the server checksum; verify storage integrity" });
143 }
144 }
145 close(fd);
146 unlink(tmpl);
147}

150/*
151 * write path (GATED): create a unique temp dir, write+read-back a small object, verify
152 * byte-exactness, then clean up. The failure code pins the cause precisely
153 * (kXR_fsReadOnly = read-only export, kXR_NotAuthorized = no write scope, quota/space).
154 * Always reverses its own mutations; bounded; never touches user data paths.
155 */
156void
157dx_probe_write(brix_conn *c, doctor_ep *e)
158{
159 char dir[96], path[160];
160 brix_status st;
161 brix_file f;
162 const char payload[] = "xrddiag-remote-doctor-write-probe\n";
163 int wrote = 0;
164
165 /* pid + monotonic clock makes the temp namespace collision-proof across runs
166 * and pid reuse, so the probe never reuses or fights a pre-existing directory. */
167 snprintf(dir, sizeof(dir), "/.xrddiag-dx-%ld-%llx", (long) getpid(),
168 (unsigned long long) brix_mono_ns());
169 snprintf(path, sizeof(path), "%s/probe.tmp", dir);
170
171 brix_status_clear(&st);
172 if (brix_mkdir(c, dir, 0700, 0, &st) != 0) {
173 dx_record_status(e, "write", &st);
174 return;
175 }
176 brix_status_clear(&st);
177 if (brix_file_open_write(c, path, 1 /*force*/, 0 /*posc*/, &f, &st) != 0) {
178 dx_record_status(e, "write", &st);
179 brix_rmdir(c, dir, &st);
180 return;
181 }
182 brix_status_clear(&st);
183 if (brix_file_write(c, &f, 0, payload, sizeof(payload) - 1, &st) == 0) {
184 wrote = 1;
185 }
186 {
187 /* a close failure means the server never durably committed the write. */
188 brix_status cst;
189 brix_status_clear(&cst);
190 if (brix_file_close(c, &f, &cst) != 0 && wrote) {
191 wrote = 0;
192 st = cst;
193 }
194 }
195
196 if (!wrote) {
197 dx_record_status(e, "write", &st);
198 } else {
199 /* read back + verify byte-exactness */
200 brix_file rf;
201 uint8_t rb[64];
202 ssize_t rn = -1;
203 brix_status rst;
204 brix_status_clear(&rst);
205 if (brix_file_open_read(c, path, &rf, &rst) == 0) {
206 rn = brix_file_read(c, &rf, 0, rb, sizeof(rb), &rst);
207 brix_file_close(c, &rf, &rst);
208 }
209 if (rn == (ssize_t) (sizeof(payload) - 1)
210 && memcmp(rb, payload, sizeof(payload) - 1) == 0) {
211 dx_record(e, &(dx_note){ "write", DX_OK, 0, "write path healthy (write/read-back verified)", "" });
212 } else {
213 dx_record(e, &(dx_note){ "write", DX_FAIL, 0,
214 "write succeeded but read-back did not match (durability/consistency fault)",
215 "check the server write-through/cache flush path and storage backend" });
216 }
217 }
218 /* always clean up our mutations; warn (no silent residue) if the dir survives. */
219 brix_status_clear(&st);
220 brix_rm(c, path, &st);
221 brix_status_clear(&st);
222 if (brix_rmdir(c, dir, &st) != 0) {
223 dx_record(e, &(dx_note){ "write", DX_WARN, st.kxr,
224 "write-probe test directory could not be removed",
225 "remove the leftover write-probe directory under the export root" });
226 }
227}

252/*
253 * authz-anon: open a force_anon session (login, NO credential), learn the server's
254 * advertised auth from its &P= list, and on an auth-REQUIRED server assert that
255 * unauthenticated stat/read is DENIED. A served op on an auth-advertising server is
256 * the auth-bypass smoking gun. Writes the discovered sec list to *sec_out so the
257 * caller can run the token tests. Returns 1 if the session was established.
258 */
259int
260dx_authz_anon(const diag_args *a, const brix_url *u, const char *target,
261 int have_target, char *sec_out, size_t sec_sz, doctor_ep *e)
262{
263 brix_conn c;
264 brix_status st;
265 brix_statinfo si;
266 int served = 0;
267
268 if (sec_out != NULL && sec_sz > 0) { sec_out[0] = '\0'; }
269 if (dx_connect_as(a, u, &(dx_cred_sel){ .force_anon = 1 }, &c, &st) != 0) {
270 dx_record(e, &(dx_note){ "authz-anon", DX_WARN, st.kxr,
271 "could not establish even an unauthenticated session (cannot assess auth posture)",
272 "check reachability and retry when the server is up" });
273 return 0;
274 }
275 if (sec_out != NULL && sec_sz > 0) {
276 snprintf(sec_out, sec_sz, "%s", c.sec_list);
277 }
278 if (c.sec_list[0] == '\0') {
279 brix_close(&c);
280 dx_record(e, &(dx_note){ "authz-anon", DX_OK, 0,
281 "server requires no authentication (anonymous by design)", "" });
282 return 1;
283 }
284 brix_status_clear(&st);
285 if (brix_stat(&c, "/", &si, &st) == 0) {
286 served = 1;
287 }
288 if (!served && have_target) {
289 brix_file f;
290 brix_status ost;
291 brix_status_clear(&ost);
292 if (brix_file_open_read(&c, target, &f, &ost) == 0) {
293 served = 1;
294 brix_file_close(&c, &f, &ost);
295 }
296 }
297 brix_close(&c);
298 if (served) {
299 dx_record(e, &(dx_note){ "authz-anon", DX_FAIL, 0,
300 "an unauthenticated client was served data/metadata on an auth-required server (auth bypass)",
301 "the server is not enforcing authentication — audit the auth config and the server build" });
302 } else {
303 dx_record(e, &(dx_note){ "authz-anon", DX_OK, st.kxr,
304 "unauthenticated access correctly denied", "" });
305 }
306 return 1;
307}

311 * authz-forged: present a structurally-valid but cryptographically-invalid bearer
312 * token (garbage signature, or alg:none). A correct server rejects it at kXR_auth
313 * (connect fails). A connect SUCCESS means the server accepted an unverifiable
314 * token — exactly the broken-signature-verification class of regression.

316void
317dx_authz_forged(const diag_args *a, const brix_url *u, const char *probe,

327 "server ACCEPTED an invalid bearer token (broken token verification)",

372/*
373 * authz-scope (GATED --allow-write): present a read-only token and attempt a write
374 * in a unique temp namespace. A correct server denies (kXR_NotAuthorized). A write
375 * that SUCCEEDS means token write-scope is not enforced — privilege escalation.
376 * Always reverses any mutation.
377 */
378void
379dx_authz_scope(const diag_args *a, const brix_url *u, const char *tok, doctor_ep *e)
380{
381 brix_conn c;
382 brix_status st;
383 char dir[96];
384
385 if (dx_connect_as(a, u, &(dx_cred_sel){ .token_override = tok,
386 .auth_force = "ztn" }, &c, &st) != 0) {
387 dx_record(e, &(dx_note){ "authz-scope", DX_WARN, st.kxr,
388 "read-only token did not authenticate; cannot test write-scope enforcement", "" });
389 return;
390 }
391 snprintf(dir, sizeof(dir), "/.xrddiag-az-%ld-%llx", (long) getpid(),
392 (unsigned long long) brix_mono_ns());
393
394 /*
395 * mkdir IS a write operation: a read-only token must be denied it. Testing it
396 * directly avoids the mkdir-then-open ambiguity (if mkdir is denied, a later
397 * open-write fails with NotFound, not a scope verdict). The DENIAL CODE is
398 * decisive: kXR_NotAuthorized = scope/ACL enforced (correct); kXR_fsReadOnly =
399 * read-only export, so we CANNOT isolate scope enforcement (inconclusive);
400 * SUCCESS = a read-only token mutated the namespace (scope NOT enforced).
401 */
402 brix_status_clear(&st);
403 if (brix_mkdir(&c, dir, 0700, 0, &st) == 0) {
404 dx_record(e, &(dx_note){ "authz-scope", DX_FAIL, 0,
405 "a read-only token was allowed to create a directory (token scope not enforced)",
406 "CRITICAL: the server is not enforcing token write-scope" });
407 brix_status_clear(&st);
408 brix_rmdir(&c, dir, &st); /* reverse our mutation */
409 } else if (st.kxr == kXR_NotAuthorized || st.kxr == kXR_AuthFailed
410 || st.kxr == XRDC_EAUTH) {
411 dx_record(e, &(dx_note){ "authz-scope", DX_OK, st.kxr,
412 "write (mkdir) correctly denied for a read-only token (scope/ACL enforced)", "" });
413 } else if (st.kxr == kXR_fsReadOnly) {
414 dx_record(e, &(dx_note){ "authz-scope", DX_WARN, st.kxr,
415 "export is read-only (allow_write off) — cannot isolate token write-scope enforcement",
416 "re-run against a read-write export to test token write-scope" });
417 } else {
418 dx_record(e, &(dx_note){ "authz-scope", DX_WARN, st.kxr,
419 "write probe failed for an unexpected reason; scope enforcement unclear",
420 "inspect the server logs for this operation" });
421 }
422 brix_close(&c);
423}
