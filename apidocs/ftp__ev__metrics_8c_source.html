3/*
4 * ftp_ev_metrics.c — the GridFTP gateway's data-plane seam into the shared
5 * {proto="gridftp"} metrics zone.
6 *
7 * WHAT: the FTP data verb → brix_metric_op_t map, the per-transfer byte/duration
8 * accounting, and the two record points every transfer passes through — one for a
9 * transfer that reached the data channel, one for a transfer refused before it.
10 *
11 * WHY: the VFS meters what it owns. Every namespace verb this gateway issues
12 * (SIZE/MDTM/MLST → STAT, MKD → MKDIR, DELE/RMD → DELETE, RNFR+RNTO → RENAME,
13 * LIST/NLST/MLSD → DIRLIST) is observed exactly once inside brix_vfs_* under the
14 * ctx built by brix_ftp_ev_vfs_ctx(), so it already lands under this label and a
15 * second op row here would double-count it. What the VFS deliberately does NOT
16 * meter is the data plane: brix_vfs_file_pread / brix_vfs_writer_write book only
17 * per-backend bytes, and brix_vfs_staged_commit leaves "the unified WRITE op row
18 * to the owning protocol". That row is this file's job — the same split WebDAV
19 * makes (GET→READ, PUT→WRITE recorded protocol-side, everything else left to the
20 * VFS).
21 *
22 * HOW: the transfer's own state carries the accounting, so the hot pumps stay
23 * untouched — no per-chunk counter, no extra branch in the send/recv loops. A
24 * transfer's byte total is derived at completion from the offsets the pumps
25 * already maintain, and its duration from ngx_current_msec sampled when the verb
26 * started (a signed diff, so a clock step or msec wrap reads as 0, never as a
27 * bogus multi-hour sample).
28 */

48/* Payload bytes this transfer actually moved.
49 *
50 * MODE E STOR commits out-of-order blocks at their own offsets, so ->off never
51 * advances and eb_received is its committed-byte counter. Every other shape
52 * advances ->off from the transfer's start offset (non-zero after a REST resume),
53 * and for RETR that is exactly the payload pulled from the VFS — the same volume
54 * brix_metric_backend_bytes booked, so the per-proto and per-backend series stay
55 * comparable. A failed transfer reports what it moved before it failed: those
56 * bytes were read or written for real, and the backend ledger already counted
57 * them. */
58static off_t
59ev_metric_bytes(const ftp_ev_dc_t *dc)
60{
61 if (dc->writing && dc->mode_e) {
62 return dc->eb_received;
63 }
64 return (dc->off > dc->start_off) ? dc->off - dc->start_off : 0;
65}

68void
69brix_ftp_ev_metric_xfer(const ftp_ev_dc_t *dc, ngx_int_t rc)
70{
71 brix_metric_op_t op = brix_ftp_ev_metric_op(dc->op);
72 ngx_msec_int_t elapsed;
73
74 if (op != BRIX_METRIC_OP_READ && op != BRIX_METRIC_OP_WRITE) {
75 return; /* a listing: the VFS owns that op row */
76 }
77
78 /* Measured from the verb, so it spans the data-channel set-up (PASV accept
79 * or active connect) as well as the bytes — that wait is part of what the
80 * client experienced as the transfer. */
81 elapsed = (ngx_msec_int_t) (ngx_current_msec - dc->start_msec);
82 if (elapsed < 0) {
83 elapsed = 0;
84 }
85
86 /* The class matches the reply the client just received: a transfer that got
87 * this far and failed did so on the data channel or in the VFS. The refusals
88 * that carry a permission or not-found verdict never reach here — they are
89 * recorded by brix_ftp_ev_metric_refused() before any dc exists. */
90 brix_metric_op_done(BRIX_PROTO_GRIDFTP, op, (size_t) ev_metric_bytes(dc),
91 (ngx_msec_t) elapsed * 1000,
92 (rc == NGX_OK) ? BRIX_ERR_NONE : BRIX_ERR_IO);
93}

96/* A refused listing IS recorded here, unlike a successful one: the refusal
97 * happens before brix_vfs_opendir, so the VFS never saw the op and there is

99void
