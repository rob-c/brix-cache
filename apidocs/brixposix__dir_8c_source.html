1/*
2 * brixposix_dir.c — §7.8 readdir family for the POSIX preload shim.
3 *
4 * WHAT: opendir/readdir/closedir (+ rewinddir/telldir/seekdir/dirfd/
5 * readdir_r; the glibc *64 spellings forward here from
6 * lib/platform/linux/preload_lfs.c) for paths under the BRIX_VMP
7 * prefix, so `ls`, `find` and anything else that walks a directory
8 * sees the remote namespace instead of ENOENT.
9 * WHY: The shim could open and read a remote FILE but could not LIST one,
10 * which meant every tool that discovers its inputs — as opposed to
11 * being handed them — fell straight through to a local path that does
12 * not exist. Read-only access you cannot enumerate is barely access.
13 * HOW: One kXR_dirlist at opendir time, materialised into a snapshot the
14 * handle owns; readdir walks the snapshot. The DIR* we hand back is
15 * our own object, so every wrapper first asks the registry "is this
16 * one of mine?" by POINTER IDENTITY and passes anything else to the
17 * real libc function untouched — a foreign DIR* is never dereferenced
18 * by this code.
19 *
20 * Deliberately narrow, symmetric with the rest of the shim: scandir(),
21 * glob() and ftw() are not interposed (they are libc code paths that call
22 * opendir/readdir, so they work through THESE wrappers when the entry point
23 * is the one we replaced, and not otherwise).
24 */

43static xfs_dir *g_dirs[XFS_DIR_MAX];

45/* ---- registry ----------------------------------------------------------
46 * A DIR* is ours only if this table holds it. Pointer identity is the only
47 * safe test: a magic field would mean dereferencing libc's DIR, whose layout
48 * is opaque and whose memory we do not own.
49 */
50static int
51dir_register(xfs_dir *d)
52{
53 int i;
54
55 pthread_mutex_lock(&g_dir_lock);
56 for (i = 0; i < XFS_DIR_MAX; i++) {
57 if (g_dirs[i] == NULL) {
58 g_dirs[i] = d;
59 pthread_mutex_unlock(&g_dir_lock);
60 return 0;
61 }
62 }
63 pthread_mutex_unlock(&g_dir_lock);
64 return -1;
65}

67static xfs_dir *
68dir_of(DIR *dirp)
69{
70 int i;
71 xfs_dir *found = NULL;
72
73 if (dirp == NULL) {
74 return NULL;
75 }
76 pthread_mutex_lock(&g_dir_lock);
77 for (i = 0; i < XFS_DIR_MAX; i++) {
78 if (g_dirs[i] == (xfs_dir *) dirp) {
79 found = g_dirs[i];
80 break;
81 }
82 }
83 pthread_mutex_unlock(&g_dir_lock);
84 return found;
85}

102/* ---- entry synthesis ---------------------------------------------------
103 * d_ino must be NON-ZERO: a zero inode means "deleted" to readdir consumers
104 * (glibc's own readdir_r documentation, and every tool that copies that
105 * convention), so an entry carrying one is skipped by code that has nothing
106 * to do with us. The server's file id is used when it sent one, and a
107 * name-derived FNV-1a fills in when it did not.
108 */
109static uint64_t
110dir_ino(const brix_dirent *e)
111{
112 uint64_t h = 1469598103934665603ULL;
113 const unsigned char *p;
114
115 if (e->have_stat && e->st.id != 0) {
116 return (uint64_t) e->st.id;
117 }
118 for (p = (const unsigned char *) e->name; *p != '\0'; p++) {
119 h = (h ^ *p) * 1099511628211ULL;
120 }
121 return h | 1ULL; /* never 0, whatever the hash lands on */
122}

124/* DT_DIR/DT_REG when the listing carried a stat, DT_UNKNOWN otherwise —
125 * which is a legal answer that sends the caller to stat(), and the shim
126 * interposes that too. */
127static unsigned char
128dir_type(const brix_dirent *e)
129{
130 struct stat sb;
131
132 if (!e->have_stat) {
133 return DT_UNKNOWN;
134 }
135 fill_stat(&e->st, &sb);
136 if (S_ISDIR(sb.st_mode)) {
137 return DT_DIR;
138 }
139 if (S_ISLNK(sb.st_mode)) {
140 return DT_LNK;
141 }
142 if (S_ISREG(sb.st_mode)) {
143 return DT_REG;
144 }
145 return DT_UNKNOWN;
146}

148/* Fill one entry from the snapshot position, or return 0 at end-of-stream.
149 * Positions 0 and 1 are "." and "..": the wire listing does not carry them,
150 * and a directory without them is not a POSIX directory — `find` prunes on
151 * their absence and `rm -r` has been known to loop. */
152static int
153dir_fill(xfs_dir *d, uint64_t *ino, unsigned char *type, const char **name)
154{
155 const brix_dirent *e;
156
157 if (d->pos == 0) {
158 *ino = 1; *type = DT_DIR; *name = ".";
159 return 1;
160 }
161 if (d->pos == 1) {
162 *ino = 2; *type = DT_DIR; *name = "..";
163 return 1;
164 }
165 if (d->pos - 2 >= d->count) {
166 return 0;
167 }
168 e = &d->ents[d->pos - 2];
169 *ino = dir_ino(e); *type = dir_type(e); *name = e->name;
170 return 1;
171}

173static int
174dir_next(xfs_dir *d, struct dirent *out)
175{
176 uint64_t ino;
177 unsigned char type;
178 const char *name;
179 size_t len;
180
181 if (!dir_fill(d, &ino, &type, &name)) {
182 return 0;
183 }
184 memset(out, 0, sizeof(*out));
185 out->d_ino = (ino_t) ino;
186 out->d_reclen = (unsigned short) sizeof(*out);
187 out->d_type = type;
188 len = strlen(name);
189 if (len >= sizeof(out->d_name)) {
190 d->pos++;
191 return dir_next(d, out); /* unrepresentable name: skip, never crop */
192 }
193 memcpy(out->d_name, name, len + 1);
194 /* the host's own cursor / length fields (d_off; d_seekoff + d_namlen) */
195 BRIXPOSIX_DIRENT_FINISH(out, d->pos + 1, len);
196 d->pos++;
197 return 1;
198}

202/* One listing into a freshly allocated handle. Returns 0, or -1 with errno
203 * already carrying the server's own answer: a listing refused for want of
204 * authorization must not reach the caller as ENOENT, which would read as
205 * "empty directory, nothing to see" and hide the refusal. */
206static int
207dir_fetch(const char *remote, xfs_dir *d)
208{
209 brix_status st;
210 int rc;
211
212 pthread_mutex_lock(&g_lock);
213 if (ensure_conn() != 0) {
214 pthread_mutex_unlock(&g_lock);
215 errno = EIO;
216 return -1;
217 }
218 brix_status_clear(&st);
219 rc = brix_dirlist(&g_conn, remote, 1, &d->ents, &d->count, &st);
220 pthread_mutex_unlock(&g_lock);
221 if (rc != 0) {
222 errno = -brix_kxr_to_errno(&st);
223 return -1;
224 }
225 return 0;
226}

228DIR *
229BRIXPOSIX_WRAP(opendir)(const char *name)
230{
231 char remote[XRDC_PATH_MAX];
232 xfs_dir *d;
233
234 REAL(opendir);
235 if (!map_path(name, remote, sizeof(remote))) {
236 return real_opendir(name); /* map_path is the NULL gate too */
237 }
238 d = calloc(1, sizeof(*d));
239 if (d == NULL) {
240 errno = ENOMEM;
241 return NULL;
242 }
243 if (dir_fetch(remote, d) != 0) {
244 free(d);
245 return NULL;
246 }
247 if (dir_register(d) != 0) {
248 free(d->ents);
249 free(d);
250 errno = EMFILE;
251 return NULL;
252 }
253 return (DIR *) d;
254}

257BRIXPOSIX_WRAP(readdir)(DIR *dirp)
258{
259 xfs_dir *d = dir_of(dirp);
260
261 REAL(readdir);
262 if (d == NULL) {
263 return real_readdir(dirp);
264 }
265 if (!dir_next(d, &d->de)) {
266 return NULL; /* end of directory: NULL with errno untouched */
267 }
