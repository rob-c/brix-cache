8/* File: push.c — F16 native root:// TPC PUSH, the source-side driver.
9 *
10 * WHAT: tpc_push_to_dest() runs the whole leg-2 worker sequence on an already
11 * connected + bootstrapped socket to the REMOTE DESTINATION: open t->push_lfn
12 * there for update carrying the rendezvous opaque, bind the sub-streams, stream
13 * the local file out as kXR_write frames, sync the remote copy, then close the
14 * remote handle and release the sub-streams on either outcome.
15 *
16 * WHY: stock xrootd has no push dialect — its native TPC is destination-side
17 * pull only — so a source that can only reach the destination outbound (a
18 * one-way firewall, an egress-only site) could not participate in a native
19 * copy at all. F16 adds the mirrored leg. It is deliberately assembled from the
20 * pull's own parts (tpc_open_remote, tpc_substreams_open/close,
21 * tpc_close_source) rather than a parallel implementation: the async open
22 * resolution, the redirect handling and the bind degradation are subtle and
23 * must not exist twice.
24 *
25 * HOW: the opaque is "?tpc.key=K&tpc.org=<us>&tpc.stage=push". Presenting the
26 * key is what CONSUMES it at the destination (single-use, replay-protected),
27 * and tpc.stage=push is what tells the destination this write-open is leg 3 of
28 * a push rather than an ordinary client write. The open options are
29 * kXR_open_updt ONLY — see tpc_open_spec_t: leg 1 already created the file, so
30 * an update-only open means a push can never write anywhere a client did not
31 * explicitly ask for. A remote sync before close makes the destination's own
32 * durability guarantee part of this transfer's success.
33 */
34
35/*
36 * WHAT: render the leg-3 opaque suffix.
37 * WHY: the destination authorizes this open on three facts — a key it minted
38 * for a client (single-use), the identity of the server presenting it, and an
39 * explicit statement that this is the push dialect. All three must be present:
40 * a missing stage would be parsed as an ordinary keyed write, and a missing org
41 * would leave the destination unable to attribute the bytes.
42 * HOW: snprintf, overflow-checked. Returns 0 with *opqlen set, or -1 with
43 * t->err_msg / t->xrd_error set.
44 */
45static int
46tpc_push_build_opaque(brix_tpc_pull_t *t, char *opaque, size_t opaque_sz,
47 size_t *opqlen)
48{
49 size_t n;
50

52 snprintf(t->err_msg, sizeof(t->err_msg),
53 "TPC push has no rendezvous key");
54 t->xrd_error = kXR_NotAuthorized;
55 return -1;
56 }
57
58 n = (size_t) snprintf(opaque, opaque_sz,

64 t->xrd_error = kXR_ArgTooLong;
65 return -1;
66 }
67

69 return 0;
70}

72/*
73 * WHAT: kXR_sync the remote handle and require a clean answer.
74 * WHY: unlike the pull — where this server owns the destination fd and fsyncs
75 * it itself — a push hands every byte to a peer. Without an acknowledged remote
76 * sync a "successful" push would only mean "the destination accepted the last
77 * frame into its page cache", and the client would be told the copy is durable
78 * when it may not be. This is the push's equivalent of tpc_stream_sync_dst, so
79 * it is NOT best-effort: a refused or failed sync fails the transfer.
80 * HOW: ClientSyncRequest on the open tag (streamid[1]=2), then one response
81 * frame; anything but kXR_ok is an error carrying the remote's own code.
82 */
83static int
84tpc_push_sync_dest(brix_tpc_pull_t *t, int fd, const u_char *fhandle)
85{
86 ClientSyncRequest sreq;
87 uint16_t status = 0;
88 uint32_t dlen = 0;

90
91 ngx_memzero(&sreq, sizeof(sreq));
92 sreq.streamid[1] = 2;
93 sreq.requestid = htons(kXR_sync);
94 ngx_memcpy(sreq.fhandle, fhandle, XRD_FHANDLE_LEN);
95
96 if (tpc_send_all(t, fd, &sreq, sizeof(sreq)) != 0) {
97 snprintf(t->err_msg, sizeof(t->err_msg),
98 "TPC push kXR_sync send failed");

100 return -1;
101 }

106 return -1;
107 }

111 const char *msg = "";

113
114 if (status == kXR_error) {

116 }

118 snprintf(t->err_msg, sizeof(t->err_msg),

123 t->xrd_error = code;
124 return -1;
125 }

128 return 0;
