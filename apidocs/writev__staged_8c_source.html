2 * writev_staged.c — whole-object staged-handle support for kXR_writev.
3 *
4 * A staged writer is a live writable handle with fd == -1. Vector writes must
5 * therefore append through brix_vfs_writer instead of entering the fd/AIO path,
6 * and kXR_wv_doSync must commit it with the same checksum gate as kXR_sync.

35writev_staged_segment(brix_ctx_t *ctx, ngx_connection_t *c,
36 const write_list *seg, const u_char **data_ptr,

38{
39 int idx = (int) (unsigned char) seg->fhandle[0];
40 int64_t offset = (int64_t) brix_plat_be64toh((uint64_t) seg->offset);
41 uint32_t wlen = (uint32_t) ntohl((uint32_t) seg->wlen);
42 int append_rc;
43
44 append_rc = brix_staged_append_raw(ctx, idx, offset, *data_ptr, wlen);
45 if (append_rc != BRIX_STAGED_APPEND_OK) {
46 uint16_t code = brix_kxr_from_errno(errno);
47 ngx_int_t rc;
48
49 BRIX_OP_ERR(ctx, BRIX_OP_WRITEV);
50 rc = brix_send_error(ctx, c, code, strerror(errno));
51 return rc == NGX_OK ? NGX_DONE : rc;
52 }
53 *data_ptr += wlen;
54 *bytes_written_total += wlen;
