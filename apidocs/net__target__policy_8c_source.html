1/*
2 * net_target_policy.c — transport policy and port selection for net targets.
3 *
4 * WHAT: enforces the supported outbound URL schemes and derives the single
5 * effective port shared by DNS validation, pinning and connection.
6 * WHY: URL parsing and transfer libraries must make the same transport
7 * decision; treating policy flags as mere port hints creates SSRF gaps.
8 * HOW: exact bounded scheme comparisons, followed by explicit-or-default

18/*
19 * net_scheme_equal — compare a parsed scheme to one supported spelling.
20 *
21 * WHAT: returns 1 when scheme is an exact, case-insensitive match; 0 otherwise.
22 * WHY: URL schemes are case-insensitive, but prefix matches would let an
23 * attacker smuggle a different transport through a policy decision.
24 * HOW: require equal lengths before applying nginx's ASCII case fold.
25 */
26static int
27net_scheme_equal(const ngx_str_t *scheme, const char *expected, size_t length)
28{
29 return scheme->len == length
30 && ngx_strncasecmp(scheme->data, (u_char *) expected, length) == 0;
31}

33/*
34 * brix_net_target_check_scheme — enforce the outbound transport policy.
35 *
36 * WHAT: returns NGX_OK for https, for http when TLS is not required, and for
37 * root only when explicitly enabled; returns NGX_ERROR with err for all
38 * other parsed schemes. A scheme-less, caller-built host target remains
39 * valid only when the policy does not require HTTPS.
40 * WHY: require_https and allow_root_scheme are security policy, not default-
41 * port hints. Enforcing them before DNS prevents an unsupported URL from
42 * reaching a resolver or transfer library with different parsing rules.
43 * HOW: exact case-insensitive comparisons over the bounded scheme slice.
44 */
45ngx_int_t
46brix_net_target_check_scheme(const brix_net_target_t *target,
47 const brix_net_target_policy_t *policy, char *err, size_t errsz)
48{
49 if (target->scheme.len == 0) {
50 if (!policy->require_https) {
51 return NGX_OK;
52 }
53 snprintf(err, errsz, "target URL scheme is required");
54 return NGX_ERROR;
55 }
56
57 if (net_scheme_equal(&target->scheme, "https", sizeof("https") - 1)) {
58 return NGX_OK;
59 }
60 if (net_scheme_equal(&target->scheme, "http", sizeof("http") - 1)) {
61 if (!policy->require_https) {
62 return NGX_OK;
63 }
64 snprintf(err, errsz, "target URL must use https");
65 return NGX_ERROR;
66 }
67 if (net_scheme_equal(&target->scheme, "root", sizeof("root") - 1)) {
68 if (!policy->require_https && policy->allow_root_scheme) {
69 return NGX_OK;
70 }
71 snprintf(err, errsz, "target URL root scheme is not allowed");
72 return NGX_ERROR;
73 }
74
75 snprintf(err, errsz, "unsupported target URL scheme");
76 return NGX_ERROR;
77}

79/*
80 * brix_net_target_effective_port — validate and select a target's dial port.
81 *
82 * WHAT: returns NGX_OK with one nonzero port for a policy-permitted target;
83 * returns NGX_ERROR with err for a refused scheme or invalid output.
84 * WHY: the security check, DNS lookup and final connection must all use the
85 * same port. Re-deriving defaults at each edge creates pinning gaps.
86 * HOW: validate the scheme, preserve an explicit port, otherwise select the
87 * exact scheme's configured or standard default.
88 */
89ngx_int_t
90brix_net_target_effective_port(const brix_net_target_t *target,
91 const brix_net_target_policy_t *policy, uint16_t *port,
92 char *err, size_t errsz)
93{
94 if (port == NULL) {
95 snprintf(err, errsz, "target port output is missing");
96 return NGX_ERROR;
97 }
98 if (brix_net_target_check_scheme(target, policy, err, errsz) != NGX_OK) {
99 return NGX_ERROR;
100 }
101 if (target->has_port) {
102 if (target->port == 0) {
103 snprintf(err, errsz, "target port is invalid");
104 return NGX_ERROR;
105 }
106 *port = target->port;
107 return NGX_OK;
108 }
109 if (net_scheme_equal(&target->scheme, "https", sizeof("https") - 1)) {
110 *port = policy->default_https_port
111 ? policy->default_https_port : BRIX_PORT_HTTPS_DEFAULT;
112 return NGX_OK;
113 }
114 if (net_scheme_equal(&target->scheme, "http", sizeof("http") - 1)) {
115 *port = BRIX_PORT_HTTP_DEFAULT;
116 return NGX_OK;
117 }
118
119 *port = policy->default_root_port
120 ? policy->default_root_port : BRIX_PORT_ROOT_DEFAULT;
121 return NGX_OK;
122}

125/*
126 * net_target_same_origin — compare URL origins after default-port expansion.
127 *
128 * WHAT: returns non-zero only when scheme and host match case-insensitively
129 * and both targets select the same effective port.
130 * WHY: textual explicit/default port differences are the same origin, while
131 * a host, scheme or actual dial-port change must not inherit authority.
132 * HOW: use the shared effective-port decision, then bounded comparisons.
133 */
134static int
135net_target_same_origin(const brix_net_target_t *first,
136 const brix_net_target_t *second)
137{
138 brix_net_target_policy_t policy;
139 uint16_t first_port, second_port;
140 char ignored[64];
141
142 ngx_memzero(&policy, sizeof(policy));
143 if (brix_net_target_effective_port(first, &policy, &first_port,
144 ignored, sizeof(ignored)) != NGX_OK
145 || brix_net_target_effective_port(second, &policy, &second_port,
146 ignored, sizeof(ignored)) != NGX_OK)
147 {
148 return 0;
149 }
150 return first->scheme.len == second->scheme.len
151 && first->host.len == second->host.len
152 && ngx_strncasecmp(first->scheme.data, second->scheme.data,
153 first->scheme.len) == 0
154 && ngx_strncasecmp(first->host.data, second->host.data,
155 first->host.len) == 0
156 && first_port == second_port;
157}

160/*
161 * brix_net_target_check_redirect — constrain authority across redirects.
162 *
163 * WHAT: rejects HTTPS-to-non-HTTPS redirects and, by default, every origin
164 * change; returns NGX_OK for a permitted next target.
165 * WHY: the pinned curl loop reuses the caller's method, body and explicit
166 * header list. Following an attacker-selected Location without this
167 * gate can replay bearer tokens or authenticated POST bodies to another
168 * origin even though DNS pinning itself succeeds.
169 * HOW: validate the next scheme, preserve TLS once established, then compare
170 * scheme/host/effective-port unless the caller explicitly declares its
171 * request safe for cross-origin replay.
172 */
173ngx_int_t
174brix_net_target_check_redirect(const brix_net_target_t *initial,
175 const brix_net_target_t *next, ngx_flag_t allow_cross_origin,
176 char *err, size_t errsz)
177{
178 brix_net_target_policy_t policy;
179
180 ngx_memzero(&policy, sizeof(policy));
181 if (brix_net_target_check_scheme(next, &policy, err, errsz) != NGX_OK) {
182 return NGX_ERROR;
183 }
184 if (net_scheme_equal(&initial->scheme, "https", sizeof("https") - 1)
185 && !net_scheme_equal(&next->scheme, "https", sizeof("https") - 1))
186 {
187 snprintf(err, errsz, "HTTPS redirect downgrade is not allowed");
188 return NGX_ERROR;
189 }
190 if (!allow_cross_origin && !net_target_same_origin(initial, next)) {
191 snprintf(err, errsz, "cross-origin redirect is not allowed");
192 return NGX_ERROR;
193 }
194 return NGX_OK;
195}
