29/* WHAT: Plant one bounded PRF object below the authorized web scratch prefix.
30 * WHY: HTR-001/S3-001 require an independent oracle, not adopted-body echo.
31 * HOW: Build the shared scratch URL, fill 64 KiB from the run key, and PUT it. */
32static int
33plant_http_baseline(xrd_conform_execution *execution,
34 const brix_weburl *base, const char *token, brix_weburl *planted)
35{
36 uint8_t *body;
37 brix_http_resp response;
38 brix_status status;
39 int rc;
40 int success;
41
42 if (!execution->run->config.allow_mutate
43 || !baseline_row_active(execution, base->is_s3)
44 || xrd_conform_webdav_scratch_url(execution, base,
45 "brix-conform-http-prf", planted) != 0) return 0;
46 body = malloc(CONFORM_HTTP_PLANTED_SIZE);
47 if (body == NULL) return -1;
48 brix_conform_fill(&execution->run->key, 0, body,
49 CONFORM_HTTP_PLANTED_SIZE);
50 memset(&response, 0, sizeof(response));
51 brix_status_clear(&status);
52 rc = xrd_conform_http_request_body(execution, planted, "PUT", token,
53 NULL, body, CONFORM_HTTP_PLANTED_SIZE, 4096, &response, &status);
54 free(body);
55 success = rc == 0 && response.status >= 200 && response.status < 300;
56 if (rc == 0) brix_http_resp_free(&response);
57 return success ? 1 : -1;
58}

81/* ---- Validate a bearer value before forming an HTTP header ----
82 *
83 * WHAT: Rejects controls, whitespace and oversized discovered token values.
84 * WHY: Credential discovery must never become a request-header injection seam.
85 * HOW: Accept visible non-space ASCII only, with a conservative 8 KiB bound.
86 */
87int
88xrd_conform_credential_safe(const char *token)
89{
90 const unsigned char *cursor = (const unsigned char *) token;
91 size_t length = 0;
92
93 if (token == NULL) return 1;
94 while (*cursor != '\0') {
95 if (*cursor <= 0x20 || *cursor >= 0x7f || ++length > 8192) return 0;
96 cursor++;
97 }
98 return length > 0;
99}

101/* ---- Build private authenticated request headers ----
102 *
103 * WHAT: Returns an owned bearer-or-SigV4 header block plus optional Range.
104 * WHY: S3 credentials are not bearer tokens and every method has its own SigV4.
105 * HOW: Validate discovered values, sign the exact host/path/method, then append
106 * Range outside the signed-header set used by the shared signer.
107 */
108char *
109xrd_conform_http_headers(const brix_weburl *url, const char *method,
110 const char *token,
111 const char *fields, const char **failure)
112{
113 char authentication[2048] = "";
114 size_t token_len = token == NULL ? 0 : strlen(token);
115 size_t fields_len = fields == NULL ? 0 : strlen(fields);
116 size_t size;
117 char *headers;
118
119 *failure = "invalid bearer credential";
120 if (!xrd_conform_credential_safe(token)) return NULL;
121 if (url->is_s3) {
122 int signed_rc = xrd_conform_s3_auth_headers(url, method,
123 authentication);
124
125 if (signed_rc != 0) {
126 *failure = signed_rc == -2 ? "partial S3 credential pair"
127 : signed_rc == -3 ? "unsafe S3 credential value"
128 : signed_rc == -4 ? "S3 URL query cannot be header-signed"
129 : "S3 SigV4 construction failed";
130 return NULL;
131 }
132 } else if (token != NULL) {
133 snprintf(authentication, sizeof(authentication),
134 "Authorization: Bearer %s\r\n", token);
135 }
136 size = strlen(authentication) + token_len + fields_len + 1;
137 headers = malloc(size);
138 if (headers == NULL) return NULL;
139 snprintf(headers, size, "%s", authentication);
140 if (fields != NULL) {
141 size_t used = strlen(headers);
142
143 snprintf(headers + used, size - used, "%s", fields);
144 }
145 *failure = NULL;
146 return headers;
147}

149/* ---- Issue one bounded HTTP request ----
150 *
151 * WHAT: Charges a request, executes it through the shared client and returns it.
152 * WHY: Every network operation must pass the common resource ceilings.
153 * HOW: Reserve predicted response bytes, call brix_http_req with shared TLS opts.
154 */
155int
156xrd_conform_http_request_custom(xrd_conform_execution *execution,
157 const brix_weburl *url, const char *method, const char *headers,
158 const void *body, size_t body_length,
159 size_t expected_bytes, brix_http_resp *response, brix_status *status)
160{
161 char reason[64];
162 char proxy[XRDC_PATH_MAX];
163 char wire_path[XRDC_PATH_MAX * 4];
164 const char *request_path = url->path;
165 const char *client_cert = brix_web_proxy_pem(proxy, sizeof(proxy));
166
167 if (body_length > SIZE_MAX - expected_bytes) {
168 brix_status_set(status, XRDC_EUSAGE, 0,
169 "HTTP request/response byte count overflow");
170 return -1;
171 }
172 if (!xrd_conform_reserve(execution, body_length + expected_bytes,
173 reason, sizeof(reason))) {
174 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
175 return -1;
176 }
177 if (url->is_s3) {
178 if (xrd_conform_s3_wire_path(url, wire_path, sizeof(wire_path)) != 0) {
179 brix_status_set(status, XRDC_EUSAGE, 0,
180 "S3 request path cannot be encoded safely");
181 return -1;
182 }
183 request_path = wire_path;
184 }
185 {
186 return brix_http_req(url->host, url->port, url->tls, method,
187 request_path,
188 headers, body, body_length, 10000, execution->run->opts.verify_host,
189 brix_resolve_ca_dir(execution->run->opts.ca_dir), client_cert, response,
190 status);
191 }
192}

307/* WHAT: Observe whether a generated sibling is absent on this web plane.
308 * WHY: XPR-007 must compare actual namespace answers, not successful baselines.
309 * HOW: Record 404 as absent, 2xx as exposed, and retain uncertainty otherwise. */
310static void
311probe_cross_plane_missing(xrd_conform_execution *execution,
312 const brix_weburl *url, const char *token)
313{
314 const brix_conform_result *row = xrd_conform_result_find(execution,
315 "XPR-007");
316 brix_weburl missing = *url;
317 brix_http_resp response;
318 brix_status status;
319 int rc;
320 int state = 0;
321
322 if (row == NULL || row->verdict == BRIX_CONFORM_VERDICT_SKIPPED) return;
323 if ((size_t) snprintf(missing.path, sizeof(missing.path),
324 "%s.brix-conform-missing-%.8s", url->path,
325 execution->run->key_hex) >= sizeof(missing.path)) return;
326 memset(&response, 0, sizeof(response));
327 brix_status_clear(&status);
328 rc = xrd_conform_http_request_fields(execution, &missing, "GET", token,
329 NULL, 4096, &response, &status);
330 if (rc == 0 && response.status == 404) state = 1;
331 else if (rc == 0 && response.status >= 200 && response.status < 300)
332 state = -1;
333 if (url->is_s3) execution->s3_missing_state = state;
334 else execution->http_missing_state = state;
335 if (rc == 0) brix_http_resp_free(&response);

407/* ---- Fetch and grade the adopted HTTP baseline ----
408 *
409 * WHAT: Returns zero after a complete GET and all supported boundary probes.
410 * WHY: Isolating response ownership keeps top-level failure cleanup simple.
411 * HOW: GET exact HEAD size, record plane digest, run boundary comparisons.
412 */
413static int
414http_baseline_grade(xrd_conform_execution *execution, const brix_weburl *url,
415 const char *token, uint64_t size, int prf_proven, brix_status *status)
416{
417 brix_http_resp response;
418 int rc;
419
420 memset(&response, 0, sizeof(response));
421 rc = xrd_conform_http_request(execution, url, "GET", token, NULL,
422 (size_t) size,
423 &response, status);
424 if (rc != 0) {
425 if (!url->is_s3 && strstr(status->msg,
426 "Content-Length body truncated") != NULL) {
427 xrd_conform_result_set(execution, "HTR-015",
428 BRIX_CONFORM_VERDICT_DIVERGENT,
429 "GET-content-length-truncated", "advertised bytes delivered",
430 status->msg);
431 }
432 return -1;
433 }
434 if (!response.framing_valid) {
435 xrd_conform_result_set(execution, "HTR-016",
436 BRIX_CONFORM_VERDICT_DIVERGENT, "chunked-body-framing-invalid",
437 "complete chunks followed by zero chunk", NULL);
438 brix_http_resp_free(&response);
439 return -1;
440 }
441 if (response.status < 200 || response.status >= 300
442 || response.body_len != (size_t) size) {
443 brix_http_resp_free(&response);
444 return -1;
445 }
446 observe_web_baseline(execution, url, &response);
447 run_web_baseline_probes(execution, url, token, &response);
448 grade_web_prf(execution, url, &response, prf_proven);
449 if (!url->is_s3) execution->http_size = response.body_len;
450 brix_http_resp_free(&response);
451 return 0;
452}

454/* ---- Run an adopted HTTP/WebDAV/S3 read-plane assessment ----
455 *
456 * WHAT: Returns zero after a bounded usable GET, otherwise incomplete code 3.
457 * WHY: Without a complete baseline no later range answer has a byte oracle.
458 * HOW: HEAD-gate size, GET once, grade observable rows, retain unknowns as such.
459 */
460int
461xrd_conform_run_http(xrd_conform_execution *execution)
462{
463 brix_weburl url;
464 brix_weburl planted_url;
465 const brix_weburl *baseline_url = &url;
466 brix_status status;
467 char *token = NULL;
468 uint64_t size;
469 int planted = 0;
470 int rc = 3;
471
472 if (brix_weburl_parse(execution->target, &url) != 0) {
473 BRIX_CLIENT_LOG("ERROR",
474 "xrd conform: target is not a valid web URL\n");
475 return 3;
476 }
477 if (!url.is_s3) token = brix_token_discover();
478 planted = plant_http_baseline(execution, &url, token, &planted_url);
479 if (planted > 0) baseline_url = &planted_url;
480 brix_status_clear(&status);
481 if (planted < 0) {
482 BRIX_CLIENT_LOG("ERROR",
483 "xrd conform: HTTP PRF baseline plant failed\n");
484 } else if (http_baseline_size(execution, baseline_url, token, &size,
485 &status) != 0) {
486 BRIX_CLIENT_LOG("ERROR",
487 "xrd conform: HTTP baseline HEAD failed or exceeded ceiling: %s\n",
488 status.msg);
489 } else {
490 brix_status_clear(&status);
491 if (http_baseline_grade(execution, baseline_url, token, size,
492 planted > 0, &status) == 0) {
493 rc = 0;
494 } else if (execution->run->config.only != NULL
495 && (strcmp(execution->run->config.only, "HTR-015") == 0
496 || strcmp(execution->run->config.only, "HTR-016") == 0)) {
497 brix_conform_result *row = xrd_conform_result_find(execution,
498 execution->run->config.only);
499
500 if (row != NULL
501 && row->verdict == BRIX_CONFORM_VERDICT_DIVERGENT) rc = 0;
502 } else {
503 BRIX_CLIENT_LOG("ERROR",
504 "xrd conform: HTTP baseline GET failed: %s\n",
505 status.msg);
506 }
507 }
508 if (planted > 0
509 && cleanup_http_baseline(execution, &planted_url, token) != 0) rc = 3;
510 free(token);
511 return rc;
512}
