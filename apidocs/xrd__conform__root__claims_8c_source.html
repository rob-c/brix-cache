4/* WHAT: Fetch and normalize one Qconfig value under the run ceilings.
5 * WHY: Stock peers may answer either VALUE or KEY=VALUE with a final newline.
6 * HOW: Reserve one exchange, strip line endings, then remove an exact key=. */
7static int
8query_value(xrd_conform_execution *execution, brix_conn *connection,
9 const char *key, char *value, size_t value_size, brix_status *status)
10{
11 char reason[64];
12 char *start;
13 size_t key_length = strlen(key);
14 size_t length;
15
16 brix_status_clear(status);
17 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return -2;
18 if (brix_query(connection, kXR_Qconfig, key, value, value_size, status)
19 != 0) return -1;
20 length = strlen(value);
21 while (length > 0
22 && (value[length - 1] == '\n' || value[length - 1] == '\r'))
23 value[--length] = '\0';
24 start = value;
25 if (length > key_length && memcmp(value, key, key_length) == 0
26 && value[key_length] == '=') start += key_length + 1;

28 return value[0] == '\0' || strcmp(value, key) == 0 ? -1 : 0;

31/* WHAT: Parse a positive decimal Qconfig limit without truncation.
32 * WHY: atoi silently accepts suffixes and wraps values used for allocation.
33 * HOW: Require digits only, use strtoull, and constrain to readv's wire u32. */
34static int
35parse_readv_limit(const char *value, size_t *limit)
36{
37 const unsigned char *cursor = (const unsigned char *) value;
38 char *end = NULL;
39 unsigned long long parsed;
40
41 if (*cursor == '\0') return 0;
42 while (*cursor != '\0') {
43 if (!isdigit(*cursor)) return 0;
44 cursor++;
45 }
46 errno = 0;
47 parsed = strtoull(value, &end, 10);
48 if (errno != 0 || end == value || *end != '\0' || parsed == 0
49 || parsed >= UINT32_MAX || parsed >= SIZE_MAX) return 0;
50 *limit = (size_t) parsed;
51 return 1;
52}

54/* WHAT: Decide whether one advertised checksum name is safe to re-query.
55 * WHY: A hostile Qconfig response must not inject CGI syntax into the path.
56 * HOW: Accept a short identifier alphabet only. */
57static int
58algorithm_name_valid(const char *name)
59{
60 const unsigned char *cursor = (const unsigned char *) name;
61 size_t length = 0;
62
63 while (*cursor != '\0') {
64 if (!isalnum(*cursor) && *cursor != '_' && *cursor != '-') return 0;
65 cursor++;
66 length++;
67 }
68 return length > 0 && length <= 31;

85/* WHAT: Prove each advertised checksum algorithm has a working Qcksum path.
86 * WHY: A capability list that names an unusable algorithm misleads clients.
87 * HOW: Parse bounded CSV tokens and issue one typed query for every token. */
88static int
89probe_checksum_claims(xrd_conform_execution *execution,
90 brix_conn *connection, const char *path, char evidence[192])
91{
92 char value[512];
93 char digest[256];
94 char reason[64];

97 unsigned count = 0;
98 int rc;
99
100 rc = query_value(execution, connection, "chksum", value, sizeof(value),
101 &status);
102 if (rc != 0 || strcmp(value, "0") == 0) {
103 snprintf(evidence, 192, "chksum-unavailable: %.140s", status.msg);
104 return 0;
105 }
106 cursor = value;
107 while (*cursor != '\0') {
108 char *comma = strchr(cursor, ',');
109
110 if (comma != NULL) *comma = '\0';
111 if (!algorithm_name_valid(cursor)) {
112 snprintf(evidence, 192, "invalid-chksum-token: %.120s", cursor);
113 return -1;
114 }
115 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) {
116 snprintf(evidence, 192, "%s", reason);
117 return 0;
118 }
119 brix_status_clear(&status);
120 if (brix_query_cksum(connection, path, cursor, digest,
121 sizeof(digest), &status) != 0 || !digest_is_hex(digest)) {
122 snprintf(evidence, 192, "advertised-%.31s-failed: kxr=%d %.100s",
123 cursor, status.kxr, status.msg);
124 return -1;
125 }
126 count++;
127 if (comma == NULL) break;
128 cursor = comma + 1;
129 }
130 if (count == 0) return 0;
131 snprintf(evidence, 192, "advertised-checksums-served=%u", count);
132 return 1;
133}

135/* WHAT: Prove the advertised per-element readv cap at and above its boundary.
136 * WHY: A small target cannot distinguish an honest cap from ordinary EOF.
137 * HOW: On a large target, compare cap bytes then require cap+1 to error or
138 * return no more than cap bytes, byte-exact against the adopted baseline. */
139static int
140probe_readv_claim(xrd_conform_execution *execution, brix_conn *connection,
141 brix_file *file, const uint8_t *baseline, uint64_t size,
142 char evidence[192])

144 char value[64];

156 snprintf(evidence, 192, "readv_ior_max-unavailable-or-invalid");

176 }
177 brix_status_clear(&status);
178 got = brix_file_readv(connection, file, &segment, 1, &status);
179 if (got != (ssize_t) limit || segment.got != limit
180 || memcmp(buffer, baseline, limit) != 0) {

183 status.kxr, got);

193 }

196 rc = got < 0 || (got == (ssize_t) segment.got && segment.got <= limit

198 free(buffer);
