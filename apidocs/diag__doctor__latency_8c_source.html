1/*
2 * diag_doctor_latency.c — mesh round-trip latency probe (phase-93 extension).
3 *
4 * WHAT: for every server the fan-out discovered, measure round-trip latency over
5 * the two XRootD control planes — the data-server plane (a kXR_stat "/"
6 * round-trip) and the CMS redirect plane (a kXR_locate "/" round-trip, the
7 * query the CMSD answers) — reporting min/avg/max ms per plane. Also the
8 * local-IPv6 capability probe used to skip IPv6-only nodes.
9 * WHY: --latency answers "how far is each node, and is the redirect plane as
10 * responsive as the data plane?" Each sample is a full request→reply, so
11 * the figure is bi-directional (out and back) by construction. Comparing
12 * the two planes surfaces an overloaded manager/CMSD vs a healthy DS.
13 * HOW: pure composition of the public libbrix API (stat/locate) timed with
14 * CLOCK_MONOTONIC — no new wire. PII-free: only cluster-member authorities
15 * and timing scalars, never a path, token, or credential. No goto.
16 */

23/* 1 when this host can route to the global IPv6 internet. A *connected* UDP
24 * socket does a route lookup without emitting a packet; no route → ENETUNREACH.
25 * Used to skip IPv6-only mesh nodes that would otherwise report a false DOWN. */
26int
27doctor_have_ipv6(void)
28{
29 struct sockaddr_in6 sa;
30 int fd, rc, err;
31
32 fd = socket(AF_INET6, SOCK_DGRAM, 0);
33 if (fd < 0) {
34 return 0; /* no AF_INET6 at all */
35 }
36 memset(&sa, 0, sizeof(sa));
37 sa.sin6_family = AF_INET6;
38 sa.sin6_port = htons(53);
39 inet_pton(AF_INET6, "2001:4860:4860::8888", &sa.sin6_addr); /* no packet sent */
40 rc = connect(fd, (struct sockaddr *) &sa, sizeof(sa));
41 err = errno;
42 close(fd);
43 if (rc != 0 && (err == ENETUNREACH || err == EHOSTUNREACH
44 || err == EADDRNOTAVAIL)) {
45 return 0;
46 }
47 return 1;
48}

51/* 1 when `host` resolves to IPv6 only (or is a v6 literal). Callers pair this
52 * with !doctor_have_ipv6() to decide a node is unreachable-by-design here.
53 * Unknown/failed lookups return 0 (don't skip — let the real connect speak). */
54int
55doctor_host_ipv6_only(const char *host)
56{
57 struct addrinfo hints, *res, *ai;
58 int have4 = 0, have6 = 0;
59
60 if (host == NULL || host[0] == '\0') {
61 return 0;
62 }
63 if (strchr(host, ':') != NULL) {
64 return 1; /* bare IPv6 literal */
65 }
66 memset(&hints, 0, sizeof(hints));
67 hints.ai_family = AF_UNSPEC;
68 hints.ai_socktype = SOCK_STREAM;
69 if (getaddrinfo(host, NULL, &hints, &res) != 0) {
70 return 0;
71 }
72 for (ai = res; ai != NULL; ai = ai->ai_next) {
73 if (ai->ai_family == AF_INET) { have4 = 1; }
74 if (ai->ai_family == AF_INET6) { have6 = 1; }
75 }
76 freeaddrinfo(res);
77 return have6 && !have4;
78}

81static double
82mono_ms(void)
83{
84 struct timespec t;
85 clock_gettime(CLOCK_MONOTONIC, &t);
86 return (double) t.tv_sec * 1000.0 + (double) t.tv_nsec / 1e6;
87}

90/* Fold one timing sample into a min/sum/max accumulator triple. */
91static void
92lat_fold(double dt, double *sum, int *ok, double *min, double *max)
93{
94 *sum += dt;
95 (*ok)++;
96 if (dt < *min) { *min = dt; }
97 if (dt > *max) { *max = dt; }
98}

101/* Data plane: `count` timed stat("/") round-trips. Failures are simply not
102 * sampled — l->xr_ok records how many landed. */
103static void
104lat_sample_stat(brix_conn *c, int count, doctor_lat *l)
105{
106 int i;
107
108 for (i = 0; i < count; i++) {
109 brix_statinfo si;
110 brix_status ps;
111 double t0;
112
113 brix_status_clear(&ps);
114 t0 = mono_ms();
115 if (brix_stat(c, "/", &si, &ps) == 0) {
116 lat_fold(mono_ms() - t0, &l->xr_avg, &l->xr_ok,
117 &l->xr_min, &l->xr_max);
118 }
119 }
120}

123/* CMS plane: `count` timed locate("/") round-trips on the same connection. */
124static void
125lat_sample_locate(brix_conn *c, int count, doctor_lat *l)
126{
127 int i;
128

130 char loc[4096];
131 brix_status ps;
132 double t0;
133
134 brix_status_clear(&ps);
135 t0 = mono_ms();
136 if (brix_locate(c, "/", loc, sizeof(loc), &ps) == 0) {
137 lat_fold(mono_ms() - t0, &l->cms_avg, &l->cms_ok,
138 &l->cms_min, &l->cms_max);
139 }
140 }
141}

144/* Probe one reachable node: time `count` stat("/") round-trips (data plane) and
145 * `count` locate("/") round-trips (cms plane) on one fresh connection. Skipped or
146 * unconnected nodes are left untouched (probed stays 0). */
147void
148doctor_latency_probe(const diag_args *a, doctor_ep *e)
149{
150 char url[320];
151 brix_url u;
152 brix_conn c;
153 brix_status st;
154 doctor_lat *l = &e->lat;
155 int count = a->latency_count > 0 ? a->latency_count : 5;
156 int v6;
157
158 if (e->skipped || !e->connected) {
159 return;
160 }
161 l->probed = 1;
162 l->samples = count;
163
164 v6 = (strchr(e->host, ':') != NULL && e->host[0] != '[');
165 snprintf(url, sizeof(url), "root://%s%s%s:%d/",
166 v6 ? "[" : "", e->host, v6 ? "]" : "", e->port);
167
168 brix_status_clear(&st);
169 if (brix_endpoint_parse(url, &u, &st) != 0
170 || brix_connect(&c, &u, &a->conn, &st) != 0) {
171 return; /* probed=1, *_ok=0 → renders "unreachable" */
172 }
173 c.io.timeout_ms = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
174 l->xr_min = l->cms_min = 1e30;
175
176 lat_sample_stat(&c, count, l);
177 lat_sample_locate(&c, count, l);
178 if (l->xr_ok) { l->xr_avg /= l->xr_ok; } else { l->xr_min = 0; }
179 if (l->cms_ok) { l->cms_avg /= l->cms_ok; } else { l->cms_min = 0; }
180 brix_close(&c);
181}

197void
198doctor_render_latency(const doctor_ep *eps, int n, FILE *out)
199{
200 int i, any = 0;
201
202 for (i = 0; i < n; i++) {
203 if (eps[i].lat.probed || eps[i].skipped) { any = 1; break; }
204 }
205 if (!any) {
206 return;
207 }
208 fprintf(out, "\nMesh latency — bi-directional round-trip per node "
209 "(min/avg/max ms):\n");
210 fprintf(out, " %-42s %-20s %-20s\n", "node",
211 "xrootd (data plane)", "cms (redirect plane)");
212 for (i = 0; i < n; i++) {
213 const doctor_lat *l = &eps[i].lat;
214 char node[64], xr[24], cms[24];
215
216 snprintf(node, sizeof(node), "%.48s:%d", eps[i].host, eps[i].port);
217 if (eps[i].skipped) {
218 fprintf(out, " %-42s %s\n", node, "skipped — no local IPv6 route");
219 continue;
220 }
221 if (!l->probed) {
222 continue;
223 }
224 lat_cell(l->xr_ok, l->xr_min, l->xr_avg, l->xr_max, xr, sizeof(xr), "unreachable");
225 lat_cell(l->cms_ok, l->cms_min, l->cms_avg, l->cms_max, cms, sizeof(cms), "n/a");
226 fprintf(out, " %-42s %-20s %-20s\n", node, xr, cms);
227 }
228 fprintf(out, " legend: xrootd = kXR_stat round-trip (data-server plane); "
229 "cms = kXR_locate round-trip (CMSD-answered redirect plane).\n");
230}
