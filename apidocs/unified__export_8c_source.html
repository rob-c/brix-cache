3/*
4 * unified_export.c — scrape-time exporter for the non-io unified families and
5 * the top-level brix_export_unified_metrics orchestrator.
6 *
7 * WHAT: Renders the cred_select, cache (hits/misses/evicted + watermark reaper),
8 * write-back staging, auth, and tpc Prometheus families, and hosts the
9 * brix_export_unified_metrics entry point that fans out over every
10 * unified_emit_<family> helper (the io families live in
11 * unified_export_io.c, the phase-107 VFS-verb families in
12 * unified_export_vfs.c). Also provides unified_emit_proto_counter, the
13 * generic per-proto counter renderer these families share.
14 * WHY: The exporter half of unified.c exceeded the file-size budget; this file
15 * owns the credential/cache/auth/tpc families and the orchestrator, while
16 * unified_export_io.c owns the byte/op/latency families and the legacy
17 * fold — two cohesive clusters, each in its own file (coding-standards §1).
18 * HOW: Each emitter reads its region of *shm via brix_metric_value and prints
19 * HELP/TYPE + per-label lines; labels stay low-cardinality (INVARIANT #8).

21 * orchestrator is a flat call sequence with frozen emission order/bytes.

24/*
25 * unified_emit_proto_counter — emit a single HELP/TYPE header followed by one
26 * per-protocol line reading counter values from `field`.
27 *
28 * WHAT: Generic per-proto counter renderer (proto label only, no fold).
29 * WHY: The cred_select and cache hit/miss/evicted families are all identical
30 * "HELP/TYPE + per-proto value" shapes; a shared emitter removes the
31 * copy-paste while keeping exposition bytes frozen.
32 * HOW: Caller passes the pre-formatted HELP/TYPE block and the per-proto
33 * counter array base; we print `metric_name{proto="…"} <value>` per proto.
34 */
35static void
36unified_emit_proto_counter(metrics_writer_t *mw, const char *help_type,
37 const char *metric_name, ngx_atomic_t *field)
38{
39 ngx_uint_t proto;
40
41 mw_printf(mw, "%s", help_type);
42 for (proto = 0; proto < BRIX_PROTO_COUNT; proto++) {
43 mw_printf(mw, "%s{proto=\"%s\"} %llu\n", metric_name,
44 brix_metric_proto_name((brix_proto_t) proto),
45 brix_metric_value(&field[proto]));
46 }
47}

287 * unified_emit_cache — render the per-protocol cache families
288 * (requests-by-disposition / bytes_evicted).
289 *
290 * WHAT: Emits the labelled cache-lookup counter and the eviction counter.
291 * WHY: Groups the cache-lookup outcome concern behind one call.
292 * HOW: One proto x disposition loop, then one unified_emit_proto_counter.
293 */
294static void
295unified_emit_cache(metrics_writer_t *mw, ngx_brix_metrics_t *shm)
296{
297 ngx_uint_t proto;
298
299 /* phase-110 W1: cache lookups carry the disposition as a LABEL VALUE from
300 * brix_metric_cache_status_name(), i.e. the identical word
301 * $brix_cache_status logs and the JSON "cache_status" key prints, so a
302 * PromQL selector and a log grep share one string. Rendered from the SAME
303 * SHM fields the removed brix_cache_{hits,misses}_total counters read (no
304 * new counter, no layout change): HIT ← cache_hits, MISS ← cache_misses.
305 * BYPASS has no counter and emits no series — an absent series is honest,
306 * a zero one would claim a measurement. Phase 112 removed the two
307 * one-fact-per-family counters this view replaced. */
308 mw_printf(mw, "%s",
309 "# HELP brix_cache_requests_total Cache lookups by protocol and "
310 "disposition (HIT/MISS — the $brix_cache_status vocabulary).\n"
311 "# TYPE brix_cache_requests_total counter\n");
312 for (proto = 0; proto < BRIX_PROTO_COUNT; proto++) {
313 const char *pn = brix_metric_proto_name((brix_proto_t) proto);
314
315 mw_printf(mw, "brix_cache_requests_total{proto=\"%s\","
316 "cache_status=\"%s\"} %llu\n", pn,
317 brix_metric_cache_status_name(BRIX_CACHE_STATUS_HIT),
318 brix_metric_value(&shm->unified.cache_hits[proto]));
319 mw_printf(mw, "brix_cache_requests_total{proto=\"%s\","
320 "cache_status=\"%s\"} %llu\n", pn,
321 brix_metric_cache_status_name(BRIX_CACHE_STATUS_MISS),
322 brix_metric_value(&shm->unified.cache_misses[proto]));
323 /* phase-110 W10: the NEGHIT series, from the unified neghit slot —
324 * so a fleet-wide negative-hit rate is one query across every plane
325 * (was only cvmfs's own negative_hits_total). */
326 mw_printf(mw, "brix_cache_requests_total{proto=\"%s\","
327 "cache_status=\"%s\"} %llu\n", pn,
