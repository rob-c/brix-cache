22int
23xrd_conform_s3_credentials(const char **access, const char **secret,
24 const char **region)
25{
26 *access = getenv("AWS_ACCESS_KEY_ID");
27 *secret = getenv("AWS_SECRET_ACCESS_KEY");
28 *region = getenv("AWS_DEFAULT_REGION");
29 if ((*access == NULL) != (*secret == NULL)) return -2;
30 if (*access == NULL) return 0;
31 if (!xrd_conform_credential_safe(*access)
32 || !xrd_conform_credential_safe(*secret)
33 || (*region != NULL && !xrd_conform_credential_safe(*region))) return -3;
34 if (*region == NULL) *region = "us-east-1";

39s3_request_parts(const brix_weburl *url, char uri[XRDC_PATH_MAX],
40 const char **query)
41{

47 if (uri_length == 0 || uri_length >= XRDC_PATH_MAX) return -1;

368static int
369list_page_url(const s3_list_probe *probe, unsigned max_keys,
370 const char *token, brix_weburl *out)
371{
372 const char *bucket_end = strchr(probe->prefix_path + 1, '/');
373 const char *prefix;
374 char encoded_prefix[XRDC_PATH_MAX * 3];
375 char encoded_token[S3_LIST_TOKEN_MAX * 3];
376 char query[XRDC_PATH_MAX * 4];
377 size_t bucket_length;
378 int written;
379
380 if (probe->prefix_path[0] != '/' || bucket_end == NULL) return -1;
381 bucket_length = (size_t) (bucket_end - probe->prefix_path);
382 prefix = bucket_end + 1;
383 if (brix_http_urlencode((const unsigned char *) prefix, strlen(prefix),
384 encoded_prefix, sizeof(encoded_prefix), "") < 0) return -1;
385 if (token[0] != '\0'
386 && brix_http_urlencode((const unsigned char *) token, strlen(token),
387 encoded_token, sizeof(encoded_token), "") < 0) return -1;
388 written = token[0] == '\0'
389 ? snprintf(query, sizeof(query), "list-type=2%s%sprefix=%s",
390 max_keys == 0 ? "&" : "&max-keys=", max_keys == 0 ? "" : "1&",
391 encoded_prefix)
392 : snprintf(query, sizeof(query),
393 "continuation-token=%s&list-type=2%s%sprefix=%s", encoded_token,
394 max_keys == 0 ? "&" : "&max-keys=", max_keys == 0 ? "" : "1&",
395 encoded_prefix);
396 if (written < 0 || (size_t) written >= sizeof(query)) return -1;
397 *out = *probe->base;
398 if (bucket_length + 1U + (size_t) written >= sizeof(out->path)) return -1;
399 memcpy(out->path, probe->prefix_path, bucket_length);
400 out->path[bucket_length] = '?';
401 memcpy(out->path + bucket_length + 1, query, (size_t) written + 1);
402 return 0;
403}

420static int
421scan_list_page(s3_list_probe *probe, const brix_http_resp *response,
422 char *next, size_t next_size, int *more)
423{
424 const char *cursor = response->body == NULL ? "" : response->body;
425 size_t page_count = 0;
426
427 while ((cursor = strstr(cursor, "<Key>")) != NULL) {
428 char key[XRDC_PATH_MAX];
429 int match = -1;
430 size_t index;
431
432 if (xml_value(cursor, "<Key>", "</Key>", key, sizeof(key)) != 1)
433 return -1;
434 for (index = 0; index < 3; index++) {
435 const char *expected = s3_object_key(probe->objects[index].path);
436
437 if (expected != NULL && strcmp(key, expected) == 0)
438 match = (int) index;
439 }
440 if (match < 0 || probe->seen[match]) probe->extra = 1;
441 else probe->seen[match] = 1;
442 page_count++;
443 cursor += 5;
444 }
445 *more = response->body != NULL
446 && strstr(response->body, "<IsTruncated>true</IsTruncated>") != NULL;
447 next[0] = '\0';
448 if (*more && xml_value(response->body, "<NextContinuationToken>",
449 "</NextContinuationToken>", next, next_size) != 1) return -1;
450 return (int) page_count;

453static int
454collect_listing(s3_list_probe *probe, unsigned max_keys, int replay)
455{
456 char token[S3_LIST_TOKEN_MAX] = "";
457 unsigned round;
458
459 memset(probe->seen, 0, sizeof(probe->seen));
460 probe->extra = 0;
461 probe->replay_ok = 0;
462 for (round = 0; round < 8; round++) {
463 brix_http_resp response;
464 brix_http_resp repeated;
465 char next[S3_LIST_TOKEN_MAX];
466 int more;
467 int count;
468
469 if (fetch_list_page(probe, max_keys, token, &response) != 0) return 0;
470 if (replay && round == 1) {
471 if (fetch_list_page(probe, max_keys, token, &repeated) == 0) {
472 probe->replay_ok = repeated.status == response.status
473 && repeated.body_len == response.body_len
474 && memcmp(repeated.body, response.body,
475 response.body_len) == 0;
476 brix_http_resp_free(&repeated);
477 }
478 }
479 count = scan_list_page(probe, &response, next, sizeof(next), &more);
480 brix_http_resp_free(&response);
481 if (count < 0 || (max_keys == 1 && count > 1)) return 0;
482 if (!more)
483 return !probe->extra && probe->seen[0] && probe->seen[1]
484 && probe->seen[2];
485 snprintf(token, sizeof(token), "%s", next);
486 }
487 return 0;
