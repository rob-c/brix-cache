2 * recv_frame_bounds.c — pure XRootD request-framing bounds (see the header).
3 *
4 * brix_max_payload_for_request() was carved out of recv_process.c so the
5 * per-opcode dlen cap — and the "reject before allocation" invariant built on it
6 * — can be fuzzed standalone (hyper-hardening C-2). Behaviour is byte-identical;
7 * recv_process.c now calls this and keeps its allocation path unchanged.
8 *
9 * Depends only on the wire opcode numbers (opcodes.h), the readv geometry

19uint32_t
20brix_max_payload_for_request(uint16_t reqid)
21/* WHAT: per-opcode payload size limit, checked BEFORE any allocation so an
22 * oversized dlen is rejected without allocating.
23 * WHY: each opcode has a legitimate maximum body; a hostile dlen beyond it must
24 * never reach the payload allocator.
25 * HOW: table of the opcodes that carry large bodies (writes, readv segments,
26 * auth, prepare); everything else is a path plus a small fixed body. */
27{
28 /* A plain kXR_write is delivered in bounded chunks (see BRIX_WRITE_STREAM_*),
29 * so its dlen may legitimately exceed the buffered 16 MiB cap; bound it only
30 * by the streaming ceiling. pgwrite/writev/chkpoint still buffer the whole
31 * body (per-page CRC / descriptor structure), so they keep the 16 MiB cap. */
32 if (reqid == kXR_write) {
33 return BRIX_MAX_WRITE_STREAM;
34 }
35
36 if (reqid == kXR_pgwrite || reqid == kXR_writev
37 || reqid == kXR_chkpoint) {
38 return BRIX_MAX_WRITE_PAYLOAD;
39 }
40
41 if (reqid == kXR_readv) {
42 /* Each segment is BRIX_READV_SEGSIZE (16) bytes. */
43 return BRIX_READV_MAXSEGS * BRIX_READV_SEGSIZE;
44 }
45
46 if (reqid == kXR_auth) {
47 return BRIX_MAX_AUTH_PAYLOAD;
48 }
49
50 if (reqid == kXR_prepare) {
51 return BRIX_MAX_PREPARE_PAYLOAD;
52 }
53
54 /* All other requests carry only a path (BRIX_MAX_PATH) plus a small
55 * fixed-size body. The +64 covers opcode-specific extras (e.g. the
56 * kXR_login info field that follows the username in the payload). */
57 return BRIX_MAX_PATH + 64;
58}

60int
61brix_root_frame_dlen_ok(const unsigned char *hdr, size_t len,
62 uint16_t *reqid_out, uint32_t *dlen_out)
63/* WHAT: decode the 24-byte ClientRequestHdr and test its dlen against the
64 * per-opcode cap — the executable form of "reject an oversized dlen before
65 * allocating a payload buffer".
66 * WHY: this is the exact bound recv_process.c enforces on the first untrusted
67 * bytes of every request; a harness driving arbitrary headers proves no
68 * reqid/dlen combination overflows the cap arithmetic or slips past.
69 * HOW: require a full 24-byte header; read requestid (BE16 @2) and dlen (BE32
70 * @20) exactly as the wire decoder does (streamid @0 is opaque, ignored);
71 * return dlen <= brix_max_payload_for_request(reqid). Pure. */
72{
73 uint16_t reqid;
74 uint32_t dlen;
75
76 if (hdr == NULL || len < 24) {
77 return 0;
78 }
79
80 reqid = (uint16_t) ((hdr[2] << 8) | hdr[3]);
81 dlen = ((uint32_t) hdr[20] << 24) | ((uint32_t) hdr[21] << 16)
82 | ((uint32_t) hdr[22] << 8) | (uint32_t) hdr[23];
83
84 if (reqid_out != NULL) { *reqid_out = reqid; }
85 if (dlen_out != NULL) { *dlen_out = dlen; }
86
87 return dlen <= brix_max_payload_for_request(reqid);
88}
