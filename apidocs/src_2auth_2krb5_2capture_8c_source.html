33/*
34 * Park a forwarded TGT in a fresh private MEMORY ccache keyed by the client
35 * principal. On any failure the half-built ccache is destroyed and the krb5
36 * error is returned (already logged); on success *out_cc owns the ccache. Split
37 * from the capture entry so each stays within the CCN ceiling (standards §4/§8).
38 */
39static krb5_error_code
40brix_krb5_stash_tgt_ccache(krb5_context kctx, krb5_principal client,
41 krb5_creds *cred, krb5_ccache *out_cc, ngx_log_t *log)
42{
43 krb5_ccache cc = NULL;
44 krb5_error_code krc;
45
46 krc = krb5_cc_new_unique(kctx, "MEMORY", NULL, &cc);
47 if (krc != 0) {
48 brix_krb5_cap_log(kctx, krc, "krb5_cc_new_unique", log);
49 return krc;
50 }
51 krc = krb5_cc_initialize(kctx, cc, client);
52 if (krc != 0) {
53 brix_krb5_cap_log(kctx, krc, "krb5_cc_initialize", log);
54 krb5_cc_destroy(kctx, cc);
55 return krc;
56 }
57 krc = krb5_cc_store_cred(kctx, cc, cred);
58 if (krc != 0) {
59 brix_krb5_cap_log(kctx, krc, "krb5_cc_store_cred", log);
60 krb5_cc_destroy(kctx, cc);
61 return krc;
62 }
63 *out_cc = cc;
64 return 0;
65}

68brix_krb5_capture_fwd_cred(void *kctx_v, void *auth_ctx_v, void *client_v,
69 const u_char *krb_cred, size_t krb_cred_len,
70 void **out_gss_cred, void **out_ccache, ngx_log_t *log)

78 krb5_error_code krc;

83 || krb_cred == NULL || krb_cred_len == 0
84 || out_gss_cred == NULL || out_ccache == NULL)
85 {
86 ngx_log_error(NGX_LOG_WARN, log, 0,
87 "brix: krb5 capture: invalid arguments");
88 return NGX_ERROR;
89 }
90
91 /*
92 * Read the forwarded credential. krb5_rd_cred checks per-message time/replay
93 * and addresses only when the auth context still carries those flags from
94 * round-1 krb5_rd_req; the peer is already authenticated by the AP exchange,
95 * and requiring a replay cache here (or matching addresses on an addressless
96 * forwarded ticket) buys nothing, so clear the flags for this single read.
97 */
98 (void) krb5_auth_con_setflags(kctx, auth_ctx, 0);
99
100 fwd.data = (char *) krb_cred;
101 fwd.length = (unsigned int) krb_cred_len;
102
103 krc = krb5_rd_cred(kctx, auth_ctx, &fwd, &creds, NULL);
104 if (krc != 0) {
105 brix_krb5_cap_log(kctx, krc, "krb5_rd_cred", log);
106 return NGX_ERROR;
107 }
108 if (creds == NULL || creds[0] == NULL) {
109 ngx_log_error(NGX_LOG_WARN, log, 0,
110 "brix: krb5 capture: KRB_CRED carried no credential");
111 if (creds) {
112 krb5_free_tgt_creds(kctx, creds);
113 }
114 return NGX_ERROR;
115 }
116
117 /* Park the forwarded TGT in a private in-memory ccache keyed by the user. */
118 krc = brix_krb5_stash_tgt_ccache(kctx, client, creds[0], &cc, log);
119 krb5_free_tgt_creds(kctx, creds);
120 if (krc != 0) {
121 return NGX_ERROR;
122 }
123
124 /* Import the ccache as a GSS initiator credential (acts AS the user). */
125 maj = gss_krb5_import_cred(&min, cc, NULL, NULL, &gcred);
126 if (GSS_ERROR(maj)) {
127 ngx_log_error(NGX_LOG_WARN, log, 0,
128 "brix: krb5 capture: gss_krb5_import_cred failed"
129 " (major=0x%xL minor=0x%xL)",
130 (unsigned long) maj, (unsigned long) min);
131 krb5_cc_destroy(kctx, cc);
132 return NGX_ERROR;
133 }
134
135 /* The GSS cred references the ccache; ownership of both passes to caller. */
136 *out_gss_cred = gcred;
137 *out_ccache = cc;
138 return NGX_OK;
139}
