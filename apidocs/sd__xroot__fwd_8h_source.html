4/*
5 * sd_xroot_fwd.h — forwarding root:// proxy driver (XrdPss forwarding mode).
6 *
7 * WHAT: A storage backend whose ORIGIN is named by the CLIENT, per open: a
8 * `brix_storage_backend forward://root[,roots] permit=<host|.suffix>...`
9 * export serves `/root://host[:port]//path` by dialling `host` with the
10 * export's service credential and forwarding the operation. It is the
11 * one-to-many sibling of sd_xroot: every op parses the key
12 * (sd_xroot_fwd_key.h), checks the protocol list and the permit
13 * allowlist, then relays to a per-host sd_xroot child instance.
14 * WHY: Upstream XCache/XrdPss forwarding mode (`pss.origin =` absent,
15 * `pss.permit`) is what lets one proxy front a whole federation without
16 * an origin per export — 2.0 F5, release-2.0-readiness.md axis (e).
17 * The client names the host, so the driver is a network-egress surface:
18 * the permit list is mandatory (an empty list forwards to nothing) and
19 * the protocol list is closed (root / roots only — never http, never a
20 * local path).
21 * HOW: Children are created lazily under a mutex (metadata ops run on the
22 * thread pool) from the template credentials the registry stamped, keyed
23 * by host:port:tls, capped at SD_XROOT_FWD_MAX_CHILDREN. Objects, dirs
24 * and staged handles belong to the child (obj->inst / obj->driver are the
25 * child's), so the VFS talks to sd_xroot directly for byte I/O; the
26 * object-keyed slots here exist for the seams that dispatch through the
27 * export's instance (ctx->sd->driver->staged_write, ->close, ...) and
28 * relay to the object's own driver. The permit verdict shares
29 * brix_tpc_host_pattern_match with the TPC egress guard so a forwarded
30 * open and a TPC pull agree on what ".example.org" means.
31 */

57/* The admission verdict for a client-named key, WITHOUT parsing a host into
58 * existence: 0 when the key is admitted -- or when `inst` is not a forwarding
59 * export, so there is nothing to admit -- ENOTSUP for a scheme outside the
60 * protocol list, EACCES for a host the permit list refuses, and the key
61 * parser's own errno (ENOENT / EINVAL / ENAMETOOLONG) for a key that names no
62 * origin at all.
63 *
64 * The read-open existence probe reports EVERY driver-stat failure as a miss
65 * (the no-existence-oracle rule), which would turn both refusals into
66 * kXR_NotFound and hide the reason from the client. The protocol layer calls
67 * this before the probe so the real verdict is reported: the origin is the
68 * client's own words, so naming it back leaks nothing about the namespace.
69 * No child is created and no host is dialled. */
70int brix_sd_xroot_fwd_admit_key(brix_sd_instance_t *inst, const char *key);
