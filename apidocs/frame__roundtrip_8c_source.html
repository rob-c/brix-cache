18/* One parsed ServerRedirectBody: connectable host, port, and any capability
19 * opaque split off the host field (see parse_redirect). Sized so a long EOS
20 * cap.sym/cap.msg opaque is never truncated by the host buffer. */
21typedef struct {
22 char host[256];
23 int port;
24 char opaque[4096];
25} redir_tgt_t;
26
27/* ServerRedirectBody = port[4 BE] + host[NUL] (host already IPv6-bracketed).
28 * The host field may carry a "?<opaque>" tail: a redirector (notably EOS/cmsd)
29 * appends the open CAPABILITY (cap.sym/cap.msg) that the open MUST replay to the
30 * chosen data server, else the DS cannot authorize it and bounces the open back
31 * (an endless manager↔DS redirect loop). We split it off so `host` is connectable
32 * and `opaque` can be re-attached to the open's path. */
33static int
34parse_redirect(const uint8_t *body, uint32_t blen, redir_tgt_t *t)
35{
36 const char *field, *qmark;
37 uint32_t flen, hlen;
38 if (blen < 5) {
39 return -1;
40 }
41 t->port = (int) xrd_get_u32_be(body);
42
43 /* The host field runs from body+4 to the first NUL/CR/LF (or end of body). We
44 * scan it IN PLACE and split host vs opaque straight into their own buffers, so
45 * a long capability opaque (EOS cap.sym/cap.msg, often >256B) is NOT truncated
46 * by the host buffer — the bug that made the DS reject the capability. */
47 field = (const char *) body + 4;
48 flen = blen - 4;
49 {
50 const char *nl = memchr(field, '\0', flen);
51 if (nl == NULL) { nl = memchr(field, '\r', flen); }
52 if (nl == NULL) { nl = memchr(field, '\n', flen); }
53 if (nl != NULL) {
54 flen = (uint32_t) (nl - field);
55 }
56 }
57
58 qmark = memchr(field, '?', flen);
59 hlen = qmark != NULL ? (uint32_t) (qmark - field) : flen;
60 if (hlen >= sizeof(t->host)) {
61 hlen = (uint32_t) sizeof(t->host) - 1;
62 }
63 memcpy(t->host, field, hlen);
64 t->host[hlen] = '\0';
65
66 t->opaque[0] = '\0';
67 if (qmark != NULL) {
68 const char *o = qmark + 1;
69 uint32_t olen = (uint32_t) (field + flen - o);
70 while (olen > 0 && (*o == '&' || *o == '?')) { /* EOS sends "?&cap.sym=" */
71 o++;
72 olen--;
73 }
74 if (olen >= sizeof(t->opaque)) {
75 olen = (uint32_t) sizeof(t->opaque) - 1;
76 }
77 memcpy(t->opaque, o, olen);
78 t->opaque[olen] = '\0';
79 }
80 return 0;
81}

95/*
96 * The tried=/triedrc= failure feedback accumulated across ONE logical
97 * operation's redirect chase (audit §7.4).
98 *
99 * WHAT: two parallel comma-separated lists — the endpoints this op tried and
100 * FAILED on, and the stock reason token for each.
101 * WHY: a redirector that is never told which data server just failed keeps
102 * handing back the same dead one; the client then burns its redirect
103 * budget and reports a confusing loop error instead of converging. BriX's
104 * own manager already PARSES this protocol (brix_manager_tried_exhausted,
105 * registry_select.c) to answer kXR_NotFound once the client has visited
106 * every candidate — but no BriX client ever emitted it, so that
107 * convergence path was unreachable from our own tools.
108 * HOW: fixed buffers sized for a realistic federation; once either fills, no
109 * further entry is recorded (the list is a hint, never load-bearing, so
110 * truncating is always safe). `n` gates emission — zero means the request
111 * goes out byte-identical to today.
112 */
113#define XRDC_TRIED_HOSTS_MAX 1024
114#define XRDC_TRIED_RCS_MAX 128
115typedef struct {
116 char hosts[XRDC_TRIED_HOSTS_MAX]; /* "h1:1094,h2:1094" */
117 char rcs[XRDC_TRIED_RCS_MAX]; /* "ioerr,enoent" */
118 int n;
119} rt_tried_t;
120
121/*
122 * WHAT: the stock triedrc reason token for a failed attempt's status.
123 * WHY: reference XrdCl reports one of four fixed spellings; a redirector may
124 * weight its re-selection on them, so inventing our own would be noise on
125 * the wire.
126 * HOW: the wire kXR code selects the token; every transport/local failure
127 * (negative XRDC_E* sentinels) reads as an I/O failure to the redirector,
128 * which is what a dead endpoint is from its point of view.
129 */
130static const char *
131tried_rc_token(const brix_status *st)
132{
133 if (st == NULL) {
134 return "srverr";
135 }
136 switch (st->kxr) {
137 case kXR_NotFound: return "enoent";
138 case kXR_FSError: return "fserr";
139 case kXR_IOError: return "ioerr";
140 default:
141 /* Local sentinels are negative; a dead/unreachable endpoint is an I/O
142 * failure from the redirector's perspective, anything else server-side. */
143 return (st->kxr < 0) ? "ioerr" : "srverr";
144 }
145}

147/*
148 * WHAT: record one failed endpoint + reason into the tried lists.
149 * WHY: keeps the append-with-bounds rule (and the "full → silently stop"
150 * policy) in one place instead of at each failure site.
151 * HOW: comma-separates after the first entry; a would-be overflow leaves both
152 * lists exactly as they were so they can never desynchronise in length.
153 */
154static void
155tried_record(rt_tried_t *tr, const char *hostport, const brix_status *st)
156{
157 const char *tok = tried_rc_token(st);
158 size_t hl = strlen(tr->hosts), rl = strlen(tr->rcs);
159 size_t need_h = strlen(hostport) + (tr->n ? 1 : 0);
160 size_t need_r = strlen(tok) + (tr->n ? 1 : 0);
161
162 if (hl + need_h >= sizeof(tr->hosts) || rl + need_r >= sizeof(tr->rcs)) {
163 return; /* full: the list is a hint, truncation is safe */
164 }
165 if (tr->n) {
166 tr->hosts[hl++] = ',';
167 tr->rcs[rl++] = ',';
168 }
169 memcpy(tr->hosts + hl, hostport, strlen(hostport) + 1);
170 memcpy(tr->rcs + rl, tok, strlen(tok) + 1);
171 tr->n++;
172}

174/*
175 * WHAT: 1 when this opcode's payload is a bare path that may carry the
176 * tried=/triedrc= CGI.
177 * WHY: appending CGI to a payload that is NOT a single path would corrupt it
178 * (kXR_mv carries two paths, the data opcodes carry no path at all). This
179 * list is exactly the set BriX's own manager parses tried= for —
180 * open_manager.c, stat_manager.c and checksum_qcksum_path.c — so emission
181 * and consumption cannot drift apart.
182 * HOW: pure opcode test.
183 */
184static int
185tried_op_carries_cgi(uint16_t reqid)
186{
187 return reqid == kXR_open || reqid == kXR_stat || reqid == kXR_query;

190/*
191 * Phase 40 (a): reconnect to a redirect target, surviving a DEAD target.
192 *
193 * WHAT: bring up a session against rhost:rport; if that target is unreachable,
194 * fall back ONCE to the home manager (c->home_host/home_port) so it can
195 * re-select a live data server.
196 * WHY: official-parity hard-fail (`brix_reconnect != 0 → give up`) turns one
197 * dead replica into a failed op even when other replicas are healthy, and
198 * surfaces a confusing connect error against a host the user never typed.
199 * HOW: the dead target is already recorded in tried[] by the caller, so the
200 * loop guard prevents the manager bouncing us straight back to it. Returns
201 * 0 if a session is up (target or manager), -1 with a clear combined error.
202 * NEVER calls brix_close between attempts — brix_reconnect/bringup own their
203 * own teardown-on-failure, and a close on a torn-down socket would misfire.
204 * Audit §7.4: a target that failed is recorded in `tr` so the replayed
205 * request tells the manager WHICH endpoint died and why — without that the
206 * manager can only re-select blindly and may hand back the same one.
207 */
208static int
209follow_redirect(brix_conn *c, const char *rhost, int rport, rt_tried_t *tr,
210 brix_status *st)
211{
212 char tgt_msg[XRDC_MSG_MAX];
213 char hp[XRDC_HOSTPORT_MAX];
214
215 if (brix_reconnect(c, rhost, rport, st) == 0) {
216 return 0;
217 }
218 snprintf(tgt_msg, sizeof(tgt_msg), "%s", st->msg); /* keep target error */
219
220 /* The target is dead: tell the manager on the replay (tried=/triedrc=). */
221 brix_format_host_port(rhost, (uint16_t) rport, hp, sizeof(hp));
222 tried_record(tr, hp, st);
223
224 if (c->home_host[0] == '\0'
225 || (strcmp(rhost, c->home_host) == 0 && rport == c->home_port)) {
226 brix_status_set(st, XRDC_ESOCK, 0,
227 "redirect target %s:%d unreachable: %s",
228 rhost, rport, tgt_msg);
229 return -1;
230 }
231 if (brix_reconnect(c, c->home_host, c->home_port, st) == 0) {
232 return 0; /* manager re-selects; the dead target is in tried[] */
233 }
234 brix_status_set(st, XRDC_ESOCK, 0,
235 "redirect target %s:%d unreachable (%s); manager %s:%d "
236 "fallback also failed", rhost, rport, tgt_msg,
237 c->home_host, c->home_port);
238 return -1;
239}

241/* One in-flight roundtrip request: the immutable original payload (opaque
242 * rebuilds always start from it so successive redirects swap rather than
243 * accumulate), the payload actually sent this iteration, and the rebuilt
244 * buffer (owned + freed once by brix_roundtrip — so the in-loop early
245 * returns need no per-exit cleanup). */
246typedef struct {
247 uint16_t reqid; /* request opcode (kXR_open gets opaque replay) */
248 const void *orig_pl; /* original payload, never rewritten */
249 uint32_t orig_len;
250 const void *cur_pl; /* payload actually sent (may be rebuilt) */
251 uint32_t cur_len;
252 char *rebuilt; /* opaque-carrying open payload, if any */
253 char redir_opaque[4096]; /* last redirect capability opaque ("" = none) */
254 rt_tried_t tried; /* failed endpoints for the manager (audit §7.4) */
255} rt_req_t;
256
257/*
258 * WHAT: Rebuild the request payload as "<original-path>?<extras>" where the
259 * extras are the redirector's capability opaque (kXR_open) and/or the
260 * tried=/triedrc= failure feedback. Points rq->cur_pl/cur_len at the new
261 * buffer; 0 / -1 (st set).
262 * WHY: Both carriers append CGI to the SAME payload, and both can be live at
263 * once (an open redirected with a capability that then fails over to the
264 * manager). Rebuilding from the ORIGINAL path each time — never from an
265 * earlier rebuild — is what stops successive redirects accumulating stale
266 * opaques, which is why this is one function rather than two appenders.
267 * HOW: Compose the extras, size the buffer, join with '?' or '&' depending on
268 * whether the original path already carries CGI. rq->rebuilt is owned and
269 * freed once by brix_roundtrip, so callers need no cleanup.
270 */
271static int
272rt_rebuild_payload(rt_req_t *rq, brix_status *st)
273{
274 const char *p = (const char *) rq->orig_pl;
275 int hasq = (rq->orig_len > 0 && memchr(p, '?', rq->orig_len) != NULL);
276 char extras[4096 + XRDC_TRIED_HOSTS_MAX + XRDC_TRIED_RCS_MAX + 32];
277 size_t el, need;
278 char *nb;
279
280 extras[0] = '\0';
281 if (rq->redir_opaque[0] != '\0') {
282 snprintf(extras, sizeof(extras), "%s", rq->redir_opaque);
283 }
284 if (rq->tried.n > 0 && tried_op_carries_cgi(rq->reqid)) {
285 el = strlen(extras);
286 snprintf(extras + el, sizeof(extras) - el, "%stried=%s&triedrc=%s",
287 el ? "&" : "", rq->tried.hosts, rq->tried.rcs);
288 }
289 el = strlen(extras);
290 if (el == 0) {
291 rq->cur_pl = rq->orig_pl; /* nothing to carry — send it verbatim */
292 rq->cur_len = rq->orig_len;
293 return 0;
294 }
295
296 need = (size_t) rq->orig_len + 1 + el + 1;
297 nb = (char *) malloc(need);
298 if (nb == NULL) {
299 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (redirect opaque)");
300 return -1;
301 }
302 memcpy(nb, p, rq->orig_len);
303 nb[rq->orig_len] = hasq ? '&' : '?';
304 memcpy(nb + rq->orig_len + 1, extras, el + 1); /* includes the NUL */
305 free(rq->rebuilt);
306 rq->rebuilt = nb;
307 rq->cur_pl = nb;
308 rq->cur_len = (uint32_t) (rq->orig_len + 1 + el);
309 return 0;
310}

312/*
313 * WHAT: process one kXR_redirect frame — parse the target, run the loop guards
314 * (budget / self-loop / already-tried), carry an open capability opaque,
315 * and reconnect to the target (with manager fallback).
316 * WHY: this is the entire redirect policy; hoisting it leaves roundtrip_loop
317 * as the bare retry state machine.
318 * HOW: consumes bd on every path. Records the target in tried[] BEFORE
319 * reconnecting so a manager fallback can't bounce us straight back.
320 * Returns 0 = session up, replay the request; -1 = fail (st set).
321 */
322static int
323rt_handle_redirect(brix_conn *c, rt_req_t *rq, uint8_t *bd, uint32_t bl,
324 brix_status *st)
325{
326 redir_tgt_t t;
327 char hp[XRDC_HOSTPORT_MAX];
328
329 if (parse_redirect(bd, bl, &t) != 0) {
330 free(bd);
331 brix_status_set(st, XRDC_EPROTO, 0, "malformed redirect");
332 return -1;
333 }
334 free(bd);
335 if (c->diag.redir_trace) { /* §15.4: surface each hop on stderr */
336 fprintf(stderr, "redirect[%d] -> %s:%d%s\n", c->redir_depth + 1,
337 t.host, t.port, t.opaque[0] ? " (+opaque)" : "");
338 }
339 if (++c->redir_depth > XRDC_REDIR_MAX) {
340 if (c->diag.redir_trace) {
341 fprintf(stderr, "redirect: budget exhausted (>%d hops)\n",
342 XRDC_REDIR_MAX);
343 }
344 brix_status_set(st, XRDC_EREDIRECT, 0,
345 "too many redirects (>%d)", XRDC_REDIR_MAX);
346 return -1;
347 }
348 brix_format_host_port(t.host, (uint16_t) t.port, hp, sizeof(hp));
349 /* Immediate self-redirect: the server we are talking to right now
350 * bounced us straight back to itself. That is an unambiguous loop;
351 * fail fast here rather than reconnecting to it (which would chase
352 * the loop until the connect timeout — up to 15s per hop). */
353 if (t.port == c->port && strcmp(t.host, c->host) == 0) {
354 if (c->diag.redir_trace) {
355 fprintf(stderr, "redirect: self-loop to %s\n", hp);
356 }
357 brix_status_set(st, XRDC_EREDIRECT, 0,
358 "redirect loop: server redirected to itself (%s)",
359 hp);
360 return -1;
361 }
362 if (tried_seen(c, hp)) {
363 if (c->diag.redir_trace) {
364 fprintf(stderr, "redirect: LOOP to already-tried %s\n", hp);
365 }
366 brix_status_set(st, XRDC_EREDIRECT, 0,
367 "redirect loop to already-tried %s", hp);
368 return -1;
369 }
370 if (c->tried_n < XRDC_REDIR_MAX) {
371 snprintf(c->tried[c->tried_n++], sizeof(c->tried[0]), "%s", hp);
372 }
373 /* EOS/cmsd redirects an open to a data server with a one-shot capability
374 * opaque; carry it onto the open's path so the DS authorizes the open
375 * (else it bounces back → the endless loop the guard above trips on). */
376 if (rq->reqid == kXR_open) {
377 snprintf(rq->redir_opaque, sizeof(rq->redir_opaque), "%s", t.opaque);
378 }
379 /* 0 = target or manager up. A dead target is recorded into rq->tried, so
380 * the rebuild below carries the failure feedback on the replay. */
381 if (follow_redirect(c, t.host, t.port, &rq->tried, st) != 0) {
382 return -1;
383 }
384 return rt_rebuild_payload(rq, st);
385}

387/*
388 * WHAT: honor one kXR_wait frame — sleep for the server's advised delay
389 * (clamped [1,30]s) plus additive sub-second jitter.
390 * WHY: keeps the backoff policy out of the retry state machine.
391 * HOW: consumes bd; bumps *waits against XRDC_REDIR_MAX so a server that
392 * stalls forever fails cleanly. Phase 40 (a): the jitter is ADDITIVE so
393 * a fleet handed the same advised wait doesn't resend in lockstep —
394 * never shorten the server's requested delay. 0 = resend / -1 (st set).
395 */
396static int
397rt_handle_wait(brix_conn *c, uint8_t *bd, uint32_t bl, int *waits,
398 brix_status *st)
399{
400 unsigned secs = xrd_wait_secs_parse(bd, bl, 1, 30); /* clamp [1,30] */
401 unsigned jms;
402
403 (void) c;
404 free(bd);
405 if (++(*waits) > XRDC_REDIR_MAX) {
406 brix_status_set(st, XRDC_EPROTO, 0, "server kept asking to wait");
407 return -1;
408 }
409 sleep(secs);
410 jms = brix_jitter_ms(secs >= 1 ? 1000u : 250u);
411 if (jms > 0) {
412 struct timespec ts;
413 ts.tv_sec = 0;
414 ts.tv_nsec = (long) jms * 1000000L;
415 (void) nanosleep(&ts, NULL);
416 }
417 return 0;
418}

420/* The redirect/wait-following request loop: send, receive, and either deliver
421 * the reply or hand kXR_redirect / kXR_wait to their policy helpers and replay.
422 * rq->rebuilt is owned + freed once by brix_roundtrip, so the in-loop early
423 * returns need no per-exit cleanup. */
424static int
425roundtrip_loop(brix_conn *c, void *hdr24, rt_req_t *rq, brix_resp_out *out,
426 brix_status *st)
427{
428 int waits = 0;
429
430 /* Each top-level op gets a fresh redirect budget + loop-guard. */
431 c->redir_depth = 0;
432 c->tried_n = 0;
433
434 for (;;) {
435 uint16_t sid, stt;
436 uint8_t *bd = NULL;
437 uint32_t bl = 0;
438 brix_payload pl = { rq->cur_pl, rq->cur_len };
439 brix_resp_out rr = { &stt, &bd, &bl };
440
441 if (brix_send(c, hdr24, &pl, &sid, st) != 0) {
442 return -1;
443 }
444 if (brix_recv(c, sid, &rr, st) != 0) {
445 return -1; /* kXR_error / transport → st already set */
446 }
447
448 if (stt == kXR_redirect) {
449 if (rt_handle_redirect(c, rq, bd, bl, st) != 0) {
450 return -1;
451 }
452 continue; /* replay against the redirect target (or manager) */
453 }
454 if (stt == kXR_wait) {
455 if (rt_handle_wait(c, bd, bl, &waits, st) != 0) {
456 return -1;
457 }
458 continue; /* resend the same request to the same server */
459 }
460
461 /* kXR_ok / kXR_oksofar */
462 if (out->status != NULL) { *out->status = stt; }
463 *out->body = bd;
464 *out->blen = bl;
465 return 0;
466 }
467}

469int
470brix_roundtrip(brix_conn *c, void *hdr24, const brix_payload *pl,
471 brix_resp_out *out, brix_status *st)
472{
473 rt_req_t rq;
474 int rc;
475
476 memset(&rq, 0, sizeof(rq)); /* redir_opaque + tried start empty */
477 rq.reqid = xrd_get_u16_be((uint8_t *) hdr24 + 2);
478 rq.orig_pl = (pl != NULL) ? pl->data : NULL;
479 rq.orig_len = (pl != NULL) ? pl->len : 0;
480 rq.cur_pl = rq.orig_pl;
481 rq.cur_len = rq.orig_len;
482 rq.rebuilt = NULL; /* opaque-carrying open payload, if a redirect needs it */
483
484 rc = roundtrip_loop(c, hdr24, &rq, out, st);
485 free(rq.rebuilt);
486 return rc;
487}
