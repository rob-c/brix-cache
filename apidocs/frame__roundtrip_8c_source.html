18/* One parsed ServerRedirectBody: connectable host, port, and any capability
19 * opaque split off the host field (see parse_redirect). Sized so a long EOS
20 * cap.sym/cap.msg opaque is never truncated by the host buffer. */
21typedef struct {
22 char host[256];
23 int port;
24 char opaque[4096];
25} redir_tgt_t;
26
27/* ServerRedirectBody = port[4 BE] + host[NUL] (host already IPv6-bracketed).
28 * The host field may carry a "?<opaque>" tail: a redirector (notably EOS/cmsd)
29 * appends the open CAPABILITY (cap.sym/cap.msg) that the open MUST replay to the
30 * chosen data server, else the DS cannot authorize it and bounces the open back
31 * (an endless manager↔DS redirect loop). We split it off so `host` is connectable
32 * and `opaque` can be re-attached to the open's path. */
33static int
34parse_redirect(const uint8_t *body, uint32_t blen, redir_tgt_t *t)
35{
36 const char *field, *qmark;
37 uint32_t flen, hlen;
38 if (blen < 5) {
39 return -1;
40 }
41 t->port = (int) xrd_get_u32_be(body);
42
43 /* The host field runs from body+4 to the first NUL/CR/LF (or end of body). We
44 * scan it IN PLACE and split host vs opaque straight into their own buffers, so
45 * a long capability opaque (EOS cap.sym/cap.msg, often >256B) is NOT truncated
46 * by the host buffer — the bug that made the DS reject the capability. */
47 field = (const char *) body + 4;
48 flen = blen - 4;
49 {
50 const char *nl = memchr(field, '\0', flen);
51 if (nl == NULL) { nl = memchr(field, '\r', flen); }
52 if (nl == NULL) { nl = memchr(field, '\n', flen); }
53 if (nl != NULL) {
54 flen = (uint32_t) (nl - field);
55 }
56 }
57
58 qmark = memchr(field, '?', flen);
59 hlen = qmark != NULL ? (uint32_t) (qmark - field) : flen;
60 if (hlen >= sizeof(t->host)) {
61 hlen = (uint32_t) sizeof(t->host) - 1;
62 }
63 memcpy(t->host, field, hlen);
64 t->host[hlen] = '\0';
65
66 t->opaque[0] = '\0';
67 if (qmark != NULL) {
68 const char *o = qmark + 1;
69 uint32_t olen = (uint32_t) (field + flen - o);
70 while (olen > 0 && (*o == '&' || *o == '?')) { /* EOS sends "?&cap.sym=" */
71 o++;
72 olen--;
73 }
74 if (olen >= sizeof(t->opaque)) {
75 olen = (uint32_t) sizeof(t->opaque) - 1;
76 }
77 memcpy(t->opaque, o, olen);
78 t->opaque[olen] = '\0';
79 }
80 return 0;
81}

95/*
96 * Phase 40 (a): reconnect to a redirect target, surviving a DEAD target.
97 *
98 * WHAT: bring up a session against rhost:rport; if that target is unreachable,
99 * fall back ONCE to the home manager (c->home_host/home_port) so it can
100 * re-select a live data server.
101 * WHY: official-parity hard-fail (`brix_reconnect != 0 → give up`) turns one
102 * dead replica into a failed op even when other replicas are healthy, and
103 * surfaces a confusing connect error against a host the user never typed.
104 * HOW: the dead target is already recorded in tried[] by the caller, so the
105 * loop guard prevents the manager bouncing us straight back to it. Returns
106 * 0 if a session is up (target or manager), -1 with a clear combined error.
107 * NEVER calls brix_close between attempts — brix_reconnect/bringup own their
108 * own teardown-on-failure, and a close on a torn-down socket would misfire.
109 */
110static int
111follow_redirect(brix_conn *c, const char *rhost, int rport, brix_status *st)
112{
113 char tgt_msg[XRDC_MSG_MAX];
114
115 if (brix_reconnect(c, rhost, rport, st) == 0) {
116 return 0;
117 }
118 snprintf(tgt_msg, sizeof(tgt_msg), "%s", st->msg); /* keep target error */
119
120 if (c->home_host[0] == '\0'
121 || (strcmp(rhost, c->home_host) == 0 && rport == c->home_port)) {
122 brix_status_set(st, XRDC_ESOCK, 0,
123 "redirect target %s:%d unreachable: %s",
124 rhost, rport, tgt_msg);
125 return -1;
126 }
127 if (brix_reconnect(c, c->home_host, c->home_port, st) == 0) {
128 return 0; /* manager re-selects; the dead target is in tried[] */
129 }
130 brix_status_set(st, XRDC_ESOCK, 0,
131 "redirect target %s:%d unreachable (%s); manager %s:%d "
132 "fallback also failed", rhost, rport, tgt_msg,
133 c->home_host, c->home_port);
134 return -1;
135}

137/* One in-flight roundtrip request: the immutable original payload (opaque
138 * rebuilds always start from it so successive redirects swap rather than
139 * accumulate), the payload actually sent this iteration, and the rebuilt
140 * buffer (owned + freed once by brix_roundtrip — so the in-loop early
141 * returns need no per-exit cleanup). */
142typedef struct {
143 uint16_t reqid; /* request opcode (kXR_open gets opaque replay) */
144 const void *orig_pl; /* original payload, never rewritten */
145 uint32_t orig_len;
146 const void *cur_pl; /* payload actually sent (may be rebuilt) */
147 uint32_t cur_len;
148 char *rebuilt; /* opaque-carrying open payload, if any */
149} rt_req_t;
150
151/* Rebuild a kXR_open payload as "<original-path><sep><opaque>" so a redirected
152 * open replays the redirector's capability to the data server. Always built from
153 * the ORIGINAL path (never an earlier rebuild) so successive redirects swap rather
154 * than accumulate opaques. rq->rebuilt is freed once by the wrapper. On success
155 * points rq->cur_pl and rq->cur_len at the new buffer. 0 / -1 (st set). */
156static int
157open_payload_with_opaque(const char *opaque, rt_req_t *rq, brix_status *st)
158{
159 const char *p = (const char *) rq->orig_pl;
160 int hasq = (rq->orig_len > 0 && memchr(p, '?', rq->orig_len) != NULL);
161 size_t ol = strlen(opaque);
162 size_t need = (size_t) rq->orig_len + 1 + ol + 1;
163 char *nb = (char *) malloc(need);
164 if (nb == NULL) {
165 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (redirect opaque)");
166 return -1;
167 }
168 memcpy(nb, p, rq->orig_len);
169 nb[rq->orig_len] = hasq ? '&' : '?';
170 memcpy(nb + rq->orig_len + 1, opaque, ol + 1); /* includes the NUL */
171 free(rq->rebuilt);
172 rq->rebuilt = nb;
173 rq->cur_pl = nb;
174 rq->cur_len = (uint32_t) (rq->orig_len + 1 + ol);
175 return 0;
176}

178/*
179 * WHAT: process one kXR_redirect frame — parse the target, run the loop guards
180 * (budget / self-loop / already-tried), carry an open capability opaque,
181 * and reconnect to the target (with manager fallback).
182 * WHY: this is the entire redirect policy; hoisting it leaves roundtrip_loop
183 * as the bare retry state machine.
184 * HOW: consumes bd on every path. Records the target in tried[] BEFORE
185 * reconnecting so a manager fallback can't bounce us straight back.
186 * Returns 0 = session up, replay the request; -1 = fail (st set).
187 */
188static int
189rt_handle_redirect(brix_conn *c, rt_req_t *rq, uint8_t *bd, uint32_t bl,
190 brix_status *st)
191{
192 redir_tgt_t t;
193 char hp[XRDC_HOSTPORT_MAX];
194
195 if (parse_redirect(bd, bl, &t) != 0) {
196 free(bd);
197 brix_status_set(st, XRDC_EPROTO, 0, "malformed redirect");
198 return -1;
199 }
200 free(bd);
201 if (c->diag.redir_trace) { /* §15.4: surface each hop on stderr */
202 fprintf(stderr, "redirect[%d] -> %s:%d%s\n", c->redir_depth + 1,
203 t.host, t.port, t.opaque[0] ? " (+opaque)" : "");
204 }
205 if (++c->redir_depth > XRDC_REDIR_MAX) {
206 if (c->diag.redir_trace) {
207 fprintf(stderr, "redirect: budget exhausted (>%d hops)\n",
208 XRDC_REDIR_MAX);
209 }
210 brix_status_set(st, XRDC_EREDIRECT, 0,
211 "too many redirects (>%d)", XRDC_REDIR_MAX);
212 return -1;
213 }
214 brix_format_host_port(t.host, (uint16_t) t.port, hp, sizeof(hp));
215 /* Immediate self-redirect: the server we are talking to right now
216 * bounced us straight back to itself. That is an unambiguous loop;
217 * fail fast here rather than reconnecting to it (which would chase
218 * the loop until the connect timeout — up to 15s per hop). */
219 if (t.port == c->port && strcmp(t.host, c->host) == 0) {
220 if (c->diag.redir_trace) {
221 fprintf(stderr, "redirect: self-loop to %s\n", hp);
222 }
223 brix_status_set(st, XRDC_EREDIRECT, 0,
224 "redirect loop: server redirected to itself (%s)",
225 hp);
226 return -1;
227 }
228 if (tried_seen(c, hp)) {
229 if (c->diag.redir_trace) {
230 fprintf(stderr, "redirect: LOOP to already-tried %s\n", hp);
231 }
232 brix_status_set(st, XRDC_EREDIRECT, 0,
233 "redirect loop to already-tried %s", hp);
234 return -1;
235 }
236 if (c->tried_n < XRDC_REDIR_MAX) {
237 snprintf(c->tried[c->tried_n++], sizeof(c->tried[0]), "%s", hp);
238 }
239 /* EOS/cmsd redirects an open to a data server with a one-shot capability
240 * opaque; carry it onto the open's path so the DS authorizes the open
241 * (else it bounces back → the endless loop the guard above trips on). */
242 if (rq->reqid == kXR_open && t.opaque[0] != '\0'
243 && open_payload_with_opaque(t.opaque, rq, st) != 0) {
244 return -1;
245 }
246 return follow_redirect(c, t.host, t.port, st); /* 0 = target or manager up */
247}

249/*
250 * WHAT: honor one kXR_wait frame — sleep for the server's advised delay
251 * (clamped [1,30]s) plus additive sub-second jitter.
252 * WHY: keeps the backoff policy out of the retry state machine.
253 * HOW: consumes bd; bumps *waits against XRDC_REDIR_MAX so a server that
254 * stalls forever fails cleanly. Phase 40 (a): the jitter is ADDITIVE so
255 * a fleet handed the same advised wait doesn't resend in lockstep —
256 * never shorten the server's requested delay. 0 = resend / -1 (st set).
257 */
258static int
259rt_handle_wait(brix_conn *c, uint8_t *bd, uint32_t bl, int *waits,
260 brix_status *st)
261{
262 unsigned secs = xrd_wait_secs_parse(bd, bl, 1, 30); /* clamp [1,30] */
263 unsigned jms;
264
265 (void) c;
266 free(bd);
267 if (++(*waits) > XRDC_REDIR_MAX) {
268 brix_status_set(st, XRDC_EPROTO, 0, "server kept asking to wait");
269 return -1;
270 }
271 sleep(secs);
272 jms = brix_jitter_ms(secs >= 1 ? 1000u : 250u);
273 if (jms > 0) {
274 struct timespec ts;
275 ts.tv_sec = 0;
276 ts.tv_nsec = (long) jms * 1000000L;
277 (void) nanosleep(&ts, NULL);
278 }
279 return 0;
280}

282/* The redirect/wait-following request loop: send, receive, and either deliver
283 * the reply or hand kXR_redirect / kXR_wait to their policy helpers and replay.
284 * rq->rebuilt is owned + freed once by brix_roundtrip, so the in-loop early
285 * returns need no per-exit cleanup. */
286static int
287roundtrip_loop(brix_conn *c, void *hdr24, rt_req_t *rq, brix_resp_out *out,
288 brix_status *st)
289{
290 int waits = 0;
291
292 /* Each top-level op gets a fresh redirect budget + loop-guard. */
293 c->redir_depth = 0;
294 c->tried_n = 0;
295
296 for (;;) {
297 uint16_t sid, stt;
298 uint8_t *bd = NULL;
299 uint32_t bl = 0;
300 brix_payload pl = { rq->cur_pl, rq->cur_len };
301 brix_resp_out rr = { &stt, &bd, &bl };
302
303 if (brix_send(c, hdr24, &pl, &sid, st) != 0) {
304 return -1;
305 }
306 if (brix_recv(c, sid, &rr, st) != 0) {
307 return -1; /* kXR_error / transport → st already set */
308 }
309
310 if (stt == kXR_redirect) {
311 if (rt_handle_redirect(c, rq, bd, bl, st) != 0) {
312 return -1;
313 }
314 continue; /* replay against the redirect target (or manager) */
315 }
316 if (stt == kXR_wait) {
317 if (rt_handle_wait(c, bd, bl, &waits, st) != 0) {
318 return -1;
319 }
320 continue; /* resend the same request to the same server */
321 }
322
323 /* kXR_ok / kXR_oksofar */
324 if (out->status != NULL) { *out->status = stt; }
325 *out->body = bd;
326 *out->blen = bl;
327 return 0;
328 }
329}

331int
332brix_roundtrip(brix_conn *c, void *hdr24, const brix_payload *pl,
333 brix_resp_out *out, brix_status *st)
334{
335 rt_req_t rq;
336 int rc;
337
338 rq.reqid = xrd_get_u16_be((uint8_t *) hdr24 + 2);
339 rq.orig_pl = (pl != NULL) ? pl->data : NULL;
340 rq.orig_len = (pl != NULL) ? pl->len : 0;
341 rq.cur_pl = rq.orig_pl;
342 rq.cur_len = rq.orig_len;
343 rq.rebuilt = NULL; /* opaque-carrying open payload, if a redirect needs it */
344
345 rc = roundtrip_loop(c, hdr24, &rq, out, st);
346 free(rq.rebuilt);
347 return rc;
348}
