15void
16brix_statinfo_to_stat(const brix_statinfo *si, int allow_symlink,
17 struct stat *stbuf)
18{
19 memset(stbuf, 0, sizeof(*stbuf));
20
21 /* File-type + permission bits come from the shared stat-flags semantics, the
22 * inverse-by-spec of the server's st_mode -> kXR flags encode. The FUSE-facing
23 * link count / size policy stays here: a directory advertises nlink 2 and no
24 * size; a symlink (lstat) or regular file advertises nlink 1 and the wire
25 * size (for a symlink that is the target length, which drives readlink). */
26 stbuf->st_mode = brix_stat_mode_from_flags(si->flags, allow_symlink);
27 if (si->have_ext) {
28 stbuf->st_mode = (stbuf->st_mode & S_IFMT)
29 | (mode_t) (si->mode & 07777);
30 }
31 if (S_ISDIR(stbuf->st_mode)) {
32 stbuf->st_nlink = 2;
33 } else {
34 stbuf->st_nlink = 1;
35 stbuf->st_size = (off_t) si->size;
36 }
37 stbuf->st_mtime = (time_t) si->mtime;
38 stbuf->st_atime = (time_t) (si->have_ext ? si->atime : si->mtime);
39 stbuf->st_ctime = (time_t) (si->have_ext ? si->ctime : si->mtime);
40 /* Stable inode from the server's file id so inode-tracking tools (find
41 * -samefile, rsync, tar) see a consistent identity across calls. */
42 stbuf->st_ino = (ino_t) si->id;
43 stbuf->st_blksize = 1024 * 1024; /* hint a large I/O unit for cp/dd */
44 stbuf->st_blocks = (blkcnt_t) ((stbuf->st_size + 511) / 512);
45}

55int
56brix_fattr_listxattr_xlate(const char *raw, size_t rawlen, char *list, size_t size)
57{
58 size_t total = 0, i = 0;
59
60 while (i < rawlen) {
61 const char *nm = raw + i;
62 size_t nl = strnlen(nm, rawlen - i);
63 const char *body = nm;
64 size_t blen = nl;
65 size_t need;
66
67 if (nl == 0) {
68 i += 1;
69 continue;
70 }
71 if (blen >= 2 && body[0] == 'U' && body[1] == '.') {
72 body += 2;
73 blen -= 2;
74 }
75 need = 5 + blen + 1; /* "user." + body + NUL */
76 if (size > 0) {
77 if (total + need > size) {
78 return -ERANGE;
79 }
80 memcpy(list + total, "user.", 5);
81 memcpy(list + total + 5, body, blen);
82 list[total + 5 + blen] = '\0';
83 }
84 total += need;
85 i += nl + 1; /* advance past the name and its NUL */
86 }
87 return (int) total;
88}
