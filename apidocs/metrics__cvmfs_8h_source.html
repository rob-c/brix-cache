17/*

33/* ---- per-repository (fqrn) counters --------------------------------------
34 * A site serves O(20) repositories, so the fqrn is a usable label — but the
35 * name arrives FROM THE WIRE, so the label set must be BOUNDED or a scanner
36 * minting random repo names explodes both the SHM and the Prometheus series
37 * space. A fixed slot table holds the first BRIX_CVMFS_REPO_SLOTS-1
38 * distinct fqrns seen; everything past capacity folds into the reserved
39 * last slot, exported as repo="_other". Slots register lock-free
40 * (state: EMPTY -> CAS -> CLAIMED(write name) -> READY); a claim race can
41 * duplicate a name across two slots, so RESOLUTION always returns the
42 * LOWEST-index READY match and the exporter skips later duplicates — all
43 * increments and all output converge on one slot per name. */
44#define BRIX_CVMFS_REPO_SLOTS 32
45#define BRIX_CVMFS_REPO_NAME_MAX 64
46
47#define BRIX_CVMFS_REPO_EMPTY 0u
48#define BRIX_CVMFS_REPO_CLAIMED 1u
49#define BRIX_CVMFS_REPO_READY 2u
50
51typedef struct {
52 ngx_atomic_t state; /* EMPTY / CLAIMED / READY */
53 char name[BRIX_CVMFS_REPO_NAME_MAX]; /* fqrn, NUL-terminated */
54
55 ngx_atomic_t requests_total[BRIX_CVMFS_CLASS_COUNT]; /* by class */
56 ngx_atomic_t files_accessed_total; /* CAS objects served OK (hit+fill) */
57 ngx_atomic_t cache_hits_total; /* served from the local store */
58 ngx_atomic_t cache_misses_total; /* needed an origin fill */
59 ngx_atomic_t fills_total; /* fills that published */
60 ngx_atomic_t fill_failures_total; /* fills that failed definitively */
61 ngx_atomic_t verify_failures_total; /* CAS mismatches (quarantined) */
62 ngx_atomic_t negative_hits_total; /* 404s absorbed by the worker memo */
63 ngx_atomic_t bytes_served_hit_total; /* LAN out, from cache */
64 ngx_atomic_t bytes_served_fill_total; /* LAN out, via a fresh fill */
65 ngx_atomic_t origin_bytes_total; /* WAN in, pulled from Stratum-1s */
66} ngx_brix_cvmfs_repo_metrics_t;

68/* ---- per-upstream (Stratum-1) counters -----------------------------------
69 * "Which of my origins am I pulling from, and how much?" — the RAL-vs-CERN
70 * question. The upstream "host:port" arrives FROM THE WIRE / config, so the
71 * label set is BOUNDED exactly like the repo table (INVARIANT #8): a fixed
72 * slot table holds the first BRIX_CVMFS_UPSTREAM_SLOTS-1 distinct hosts;
73 * the reserved last slot is exported upstream="_other". Registration and
74 * convergence use the same lock-free EMPTY->CLAIMED->READY lowest-index
75 * scheme as the repo table. */
76#define BRIX_CVMFS_UPSTREAM_SLOTS 16 /* >= a site's Stratum-1 allowlist */
77#define BRIX_CVMFS_UPSTREAM_NAME_MAX 300 /* matches sd_http last_origin buf */
78
79/* Fill-duration histogram: cumulative-le upper bounds in MILLISECONDS. The
80 * exporter renders them as Prometheus seconds buckets (/1000) plus an
81 * implicit +Inf. They bracket a LAN-adjacent fill (ms) through a cold-WAN
82 * fill (seconds). */
83#define BRIX_CVMFS_UP_HBUCKETS 6
84static const long brix_cvmfs_up_bucket_ms[BRIX_CVMFS_UP_HBUCKETS] = {
85 5, 25, 100, 500, 2000, 10000
86};

127/*

131 * served it. No-op when the metrics SHM is unmapped or host is NULL/empty.
132 * Lock-free; safe from fill worker threads. (metrics/cvmfs.c)
133 */
