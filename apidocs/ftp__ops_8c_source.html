1/*
2 * ftp_ops.c — GridFTP client session setup and metadata queries.
3 *
4 * WHAT: bring a control session from "connected" to "ready to transfer", and
5 * answer the one metadata question a copy needs — size and modification
6 * time of a remote path.
7 * WHY: the negotiation preamble (protection buffer size, data-channel
8 * protection, authentication of the data channel, binary type) is fixed
9 * boilerplate every transfer needs and no caller should repeat; and --sync
10 * needs remote metadata through the same session machinery the copy uses.
11 * HOW: PBSZ/PROT/DCAU are advisory — a server that rejects them still transfers,
12 * so their replies are tolerated, while TYPE I is mandatory (a text-mode
13 * transfer would corrupt data). Metadata comes from SIZE and MDTM (RFC 3659
14 * §4/§3), and the MDTM timestamp is converted by the MLSx kernel, which
15 * already implements the timezone-free `modify=` conversion — MLST itself is
16 * unusable here because its payload is a multiline block the reply contract
17 * does not expose.
18 */

27int
28brix_ftp_session_open(brix_ftp_sess *s, const brix_ftpurl *u,
29 const brix_opts *co, brix_status *st)
30{
31 int timeout_ms = (co != NULL && co->max_stall_ms > 0) ? co->max_stall_ms
32 : 60000;
33
34 if (brix_ftp_connect(s, u->host, u->port, timeout_ms, st) != 0) {
35 return -1;
36 }
37 if (brix_ftp_login(s, u, co, st) != 0) {
38 brix_ftp_close(s);
39 return -1;
40 }
41 if (s->secure) {
42 /* Clear-text data channel with an authenticated control channel: the
43 * gateway's default, and what globus-url-copy negotiates without -dcpriv.
44 * Advisory — a refusal does not stop the transfer. */
45 (void) brix_ftp_cmd(s, st, "PBSZ 0");
46 (void) brix_ftp_cmd(s, st, "PROT C");
47 (void) brix_ftp_cmd(s, st, "DCAU N");
48 }
49 if (brix_ftp_cmd_expect(s, 200, 299, st, "TYPE I") != 0) {
50 brix_ftp_close(s);
51 return -1;
52 }
53 return 0;
54}

57/* Parse a "213 <digits>" SIZE reply. */
58static int
59parse_size(const char *text, int64_t *size)
60{
61 char *end = NULL;
62 long long v;
63 const char *p = text;
64
65 while (*p == ' ') {
66 p++;
67 }
68 v = strtoll(p, &end, 10);
69 if (end == p || v < 0) {
70 return -1;
71 }
72 *size = (int64_t) v;
73 return 0;
74}

77/*
78 * Convert a "213 YYYYMMDDHHMMSS[.frac]" MDTM reply to an epoch time by handing it
79 * to the MLSx fact parser: MDTM's value and MLSx's `modify=` fact are the same
80 * production (RFC 3659 §2.3), so the conversion exists once, in one kernel.
81 */
82static int
83parse_mdtm(const char *text, int64_t *mtime)
84{
85 char line[96];
86 gftp_mlsx_ent_t ent;
87 const char *p = text;
88 size_t n = 0;
89
90 while (*p == ' ') {
91 p++;
92 }
93 while (p[n] != '\0' && p[n] != ' ' && p[n] != '\r' && n < 32) {
94 n++;
95 }
96 if (n < 14) {
97 return -1;
98 }
99 if (snprintf(line, sizeof(line), "modify=%.*s; x", (int) n, p)
100 >= (int) sizeof(line)) {
101 return -1;
102 }
103 if (gftp_mlsx_parse(line, strlen(line), &ent) != 0 || !ent.has_mtime) {
104 return -1;
105 }
106 *mtime = (int64_t) ent.mtime;
107 return 0;
108}

111int
112brix_ftp_stat(brix_ftp_sess *s, const char *path, int64_t *size,
113 int64_t *mtime, brix_status *st)
114{
115 if (size != NULL) {
116 *size = -1;
117 }
118 if (mtime != NULL) {
119 *mtime = -1;
120 }
121 if (brix_ftp_cmd(s, st, "SIZE %s", path) != 0) {
122 return -1;
123 }
124 if (s->code == 550) {
125 brix_status_set(st, XRDC_ENOENT, 0, "gsiftp: %s: no such file", path);
126 return -1;
127 }
128 if (s->code == 213 && size != NULL && parse_size(s->text, size) != 0) {
129 *size = -1;
130 }
131 if (brix_ftp_cmd(s, st, "MDTM %s", path) != 0) {
132 return -1;
133 }
134 if (s->code == 213 && mtime != NULL && parse_mdtm(s->text, mtime) != 0) {
135 *mtime = -1;
136 }
137 return 0;
138}

141int
142brix_ftp_url_stat(const char *url, const brix_opts *co, int64_t *size,
143 int64_t *mtime, brix_status *st)
144{
145 brix_ftpurl u;
146 brix_ftp_sess *s;
147 int rc;
148
149 if (brix_ftpurl_parse(url, &u) != 0) {
150 brix_status_set(st, XRDC_EUSAGE, 0, "gsiftp: bad URL: %s", url);
151 return -1;
152 }
153 s = calloc(1, sizeof(*s)); /* the session carries 64 KiB of buffers */
154 if (s == NULL) {
155 brix_status_set(st, XRDC_EIO, 0, "gsiftp: out of memory");
156 return -1;
157 }
158 if (brix_ftp_session_open(s, &u, co, st) != 0) {
159 free(s);
160 return -1;
161 }
162 rc = brix_ftp_stat(s, u.path, size, mtime, st);
163 brix_ftp_close(s);
164 free(s);
165 return rc;
166}
