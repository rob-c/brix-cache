1/*
2 * rpm_merge.c — the RPM location merge: refusals first, then the export build.
3 *
4 * WHAT: ngx_http_brix_rpm_merge_loc_conf() and its three helpers — scalar
5 * inheritance with the verify pre-seed, the config-load refusal matrix,
6 * and the export/backend/tier build.
7 * WHY: the whole mirror is named by ONE directive, so almost everything here
8 * is turning that one URL into shapes the existing machinery already
9 * understands: a storage-backend string for sd_http, an export root for
10 * the confinement fd, a verify mode for the fill, a manifest TTL for the
11 * mutable half of the repository. What is genuinely new is the refusal
12 * matrix. Each entry below is a config an operator can write and would
13 * otherwise get a silently WRONG deployment from — a "mirror" that
14 * accepts uploads, a cleartext upstream, two directives naming the same
15 * upstream differently, or a verify mode that quietly turns the
16 * self-addressing check off. They are nginx -t failures on purpose.
17 * HOW: the order is load-bearing and matches the OCI merge it is modelled on:
18 * adopt the unified directives → merge our scalars → pre-seed the verify
19 * default while the sentinel is still readable → shared merge → refuse →
20 * build. A location without brix_rpm_mirror returns after the shared
21 * merge having allocated nothing.
22 */

36/* Inherit every RPM scalar and pre-seed the verify default.
37 *
38 * The pre-seed must land BEFORE the shared merge: the sentinel is what
39 * distinguishes "the operator said nothing" from "the operator said off", and
40 * the shared merge collapses the former to the latter. Every metadata file a
41 * createrepo repository publishes carries its own checksum in its name, so
42 * the self-verifying mode is the only default that makes a mirror
43 * trustworthy by construction. */
44static char *
45rpm_merge_scalars(ngx_conf_t *cf, ngx_http_brix_rpm_loc_conf_t *prev,
46 ngx_http_brix_rpm_loc_conf_t *conf)
47{
48 brix_http_common_adopt(cf, &conf->common);
49
50 ngx_conf_merge_value(conf->mirror, prev->mirror, 0);
51 ngx_conf_merge_str_value(conf->mirror_url, prev->mirror_url, "");
52 ngx_conf_merge_value(conf->insecure, prev->insecure, 0);
53 ngx_conf_merge_value(conf->prefetch, prev->prefetch, 0);
54 ngx_conf_merge_sec_value(conf->metadata_ttl, prev->metadata_ttl,
55 BRIX_RPM_METADATA_TTL_DEFAULT);
56
57 if (conf->mirror && conf->common.cache_verify_mode == NGX_CONF_UNSET_UINT) {
58 conf->common.cache_verify_mode = BRIX_CACHE_VERIFY_RPM_REPODATA;
59 }
60
61 return ngx_http_brix_shared_merge(cf, &prev->common, &conf->common, "");
62}

65/* The refusal matrix. */
66static char *
67rpm_reject(ngx_conf_t *cf, ngx_http_brix_rpm_loc_conf_t *conf,
68 const brix_oci_url_t *url)
69{
70 if (conf->common.allow_write == 1) {
71 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
72 "brix_allow_write on: an RPM pull-through mirror is read-only by "
73 "construction; publish a repository with brixrpm createrepo "
74 "instead");
75 return NGX_CONF_ERROR;
76 }
77
78 if (conf->common.stage_store.len > 0) {
79 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
80 "brix_stage_store: an RPM pull-through mirror never stages "
81 "writes; remove it from this block");
82 return NGX_CONF_ERROR;
83 }
84
85 if (!url->tls && !conf->insecure) {
86 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
87 "brix_rpm_mirror \"%V\": a cleartext upstream lets anyone on the "
88 "path replace a package before this mirror ever hashes it; use "
89 "https, or brix_rpm_mirror_insecure on for a test fixture",
90 &conf->mirror_url);
91 return NGX_CONF_ERROR;
92 }
93
94 /* The rpm-repodata verify is not a preference here, it is the reason this
95 * plane exists: the classifier hands the cache tier a per-instance
96 * personality, and every OTHER mode would tell the tier this cache is a
97 * CVMFS or OCI one — which would then read RPM keys with the wrong
98 * grammar and stamp the wrong freshness window on them. An operator who
99 * wants no verification at all wants nginx's own proxy_cache, not this. */
100 if (conf->common.cache_verify_mode != BRIX_CACHE_VERIFY_RPM_REPODATA) {
101 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
102 "brix_cache_verify %s: a brix_rpm_mirror location must run the "
103 "rpm-repodata verification - it is what checks each metadata file "
104 "against the checksum its own name carries",
105 brix_cache_verify_mode_str(conf->common.cache_verify_mode));
106 return NGX_CONF_ERROR;
107 }
108
109 if (conf->common.storage_backend.len > 0) {
110 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
111 "brix_storage_backend and brix_rpm_mirror both name the upstream "
112 "for this location; brix_rpm_mirror is the RPM spelling - "
113 "configure exactly one");
114 return NGX_CONF_ERROR;
115 }
116
117 return NGX_CONF_OK;
118}

121/* Turn `brix_rpm_mirror <base-url>` into the storage-backend string sd_http is
122 * configured from. The cache key is the request URI verbatim (an absolute
123 * repository path), so the backend base carries the scheme, authority and any
124 * path prefix, and nothing else — exactly the composition the stock
125 * `proxy_cache_key $uri; proxy_pass http://origin;` recipe performs. */
126static char *
127rpm_build_upstream(ngx_conf_t *cf, ngx_http_brix_rpm_loc_conf_t *conf)
128{
129 brix_oci_url_t url;
130 u_char *backend;
131 char base[512];
132 size_t len;
133
134 if (conf->mirror_url.len == 0) {
135 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
136 "brix_rpm_mirror requires an upstream base URL");
137 return NGX_CONF_ERROR;
138 }
139
140 if (brix_oci_url_parse((const char *) conf->mirror_url.data,
141 conf->mirror_url.len, &url) != 0)
142 {
143 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
144 "brix_rpm_mirror \"%V\" is not an absolute http(s) URL with a "
145 "plain host and path", &conf->mirror_url);
146 return NGX_CONF_ERROR;
147 }
148
149 if (rpm_reject(cf, conf, &url) != NGX_CONF_OK) {
150 return NGX_CONF_ERROR;
151 }
152
153 len = (size_t) snprintf(base, sizeof(base), "%s://%s%s%s:%d%s",
154 url.tls ? "https" : "http",
155 strchr(url.host, ':') ? "[" : "",
156 url.host,
157 strchr(url.host, ':') ? "]" : "",
158 url.port, url.path);
159 if (len >= sizeof(base)) {
160 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
161 "brix_rpm_mirror \"%V\" is too long", &conf->mirror_url);
162 return NGX_CONF_ERROR;
163 }
164
165 backend = ngx_pnalloc(cf->pool, len + 1);
166 if (backend == NULL) {
167 return NGX_CONF_ERROR;
168 }
169 ngx_memcpy(backend, base, len + 1);
170 conf->common.storage_backend.data = backend;
171 conf->common.storage_backend.len = len;
172
173 return NGX_CONF_OK;
174}

177/* Anchor the namespace, open the confinement fd, register the backend and
178 * compose the cache tiers — the build order every brix HTTP protocol shares.
179 * A mirror is a pure cache node: no local export tree, so the namespace
180 * anchors at "/" and every object lives in the cache store. */
181static char *
182rpm_merge_export(ngx_conf_t *cf, ngx_http_brix_rpm_loc_conf_t *conf)
183{
184 conf->common.allow_write = 0;
185
186 brix_storage_backend_posix_root(&conf->common);
187
188 if (brix_http_merge_export_anchor(cf, &conf->common, "brix_rpm_mirror", 0)
189 != NGX_CONF_OK)
190 {
191 return NGX_CONF_ERROR;
192 }
193
194 /* The mutable half of a repository — repomd.xml and anything createrepo
195 * did not name after its own checksum — expires on this window; every
196 * digest-named metadata file and every package is immutable and ignores
197 * it. Which is which is the classifier's answer, read back through the
198 * instance's verify personality (sd_cache_is_manifest_key). */
199 conf->common.cache_manifest_ttl = conf->metadata_ttl;
200
201 if (brix_tier_register_stores(cf, &conf->common) != NGX_OK) {
202 return NGX_CONF_ERROR;
203 }
204
205 return NGX_CONF_OK;
206}

209char *
210ngx_http_brix_rpm_merge_loc_conf(ngx_conf_t *cf, void *parent, void *child)
211{
212 ngx_http_brix_rpm_loc_conf_t *prev = parent;
213 ngx_http_brix_rpm_loc_conf_t *conf = child;
214
215 if (rpm_merge_scalars(cf, prev, conf) != NGX_CONF_OK) {
216 return NGX_CONF_ERROR;
217 }
218
219 if (!conf->mirror) {
220 return NGX_CONF_OK; /* surface off: nothing is built */
221 }
222
223 if (rpm_build_upstream(cf, conf) != NGX_CONF_OK) {
224 return NGX_CONF_ERROR;
225 }
226
227 return rpm_merge_export(cf, conf);
228}
