14 * WHAT: Return an owned EVP_PKEY or NULL, leaving the builder with its caller.
15 * WHY: RSA and EC imports must share complete cleanup on every provider error.
16 * HOW: 1. Build parameters. 2. Import into the requested type. 3. Free temporaries.

18static inline EVP_PKEY *

20{
21 OSSL_PARAM *parameters = OSSL_PARAM_BLD_to_param(builder);
22 EVP_PKEY_CTX *context = EVP_PKEY_CTX_new_from_name(NULL, type, NULL);
23 EVP_PKEY *key = NULL;
24

27 if (EVP_PKEY_fromdata(context, &key, EVP_PKEY_PUBLIC_KEY, parameters) != 1) {
28 EVP_PKEY_free(key);
29 key = NULL;
30 }
31 }
32 EVP_PKEY_CTX_free(context);

38/* ---- Construct an RSA public key without consuming the input numbers ----
39 * WHAT: Return an owned EVP_PKEY or NULL; modulus/exponent stay caller-owned.
40 * WHY: RSA_set0_key consumes numbers but provider builders borrow them.
41 * HOW: 1. Validate inputs. 2. Build with native ownership rules. 3. Clean up.
42 */
43static inline EVP_PKEY *
44brix_crypto_rsa_public(const BIGNUM *modulus, const BIGNUM *exponent)
45{
46 EVP_PKEY *key = NULL;
47
48 if (modulus == NULL || exponent == NULL) {
49 return NULL;
50 }
51#if OPENSSL_VERSION_NUMBER >= 0x30000000L
52 OSSL_PARAM_BLD *builder = OSSL_PARAM_BLD_new();
53
54 if (builder != NULL
55 && OSSL_PARAM_BLD_push_BN(builder, OSSL_PKEY_PARAM_RSA_N, modulus) == 1
56 && OSSL_PARAM_BLD_push_BN(builder, OSSL_PKEY_PARAM_RSA_E, exponent) == 1) {
57 key = brix_crypto_import_public("RSA", builder);
58 }
59 OSSL_PARAM_BLD_free(builder);
60#else
61 RSA *rsa = RSA_new();
62 BIGNUM *owned_n = BN_dup(modulus), *owned_e = BN_dup(exponent);
63
64 if (rsa != NULL && owned_n != NULL && owned_e != NULL
65 && RSA_set0_key(rsa, owned_n, owned_e, NULL) == 1) {
66 owned_n = NULL;
67 owned_e = NULL;
68 key = EVP_PKEY_new();
69 if (key != NULL && EVP_PKEY_set1_RSA(key, rsa) != 1) {
70 EVP_PKEY_free(key);
71 key = NULL;
72 }
73 }
74 BN_free(owned_n);
75 BN_free(owned_e);
76 RSA_free(rsa);
77#endif
78 return key;
79}
