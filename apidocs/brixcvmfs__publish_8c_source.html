1/* brixcvmfs_publish.c — `brixcvmfs repo` transaction family (phase-96 S4).
2 * Unprivileged, tool-surface only (G14): no FUSE, no root.
3 *
4 * brixcvmfs repo transaction <repo_dir>
5 * — open a transaction: take the lock, create the upper tree
6 * (<repo_dir>/.brixtxn/upper) that overlay writes populate.
7 * brixcvmfs repo abort <repo_dir>
8 * — discard the transaction (recursive removal of .brixtxn).
9 * brixcvmfs repo publish <repo_dir> [keys_dir] [--chunk-size N] [--dirtab F]
10 * — scan the upper tree into a changeset, run the publish engine,
11 * then retire the transaction. The manifest swap is the engine's
12 * last act, so a crash mid-publish leaves the old revision live and
13 * the transaction intact for a clean re-run.
14 *
15 * The lock (.brixtxn/lock, O_CREAT|O_EXCL) records "pid:\nboot:\n" for
16 * forensics. A transaction is durable state (the staged writes live in the
17 * upper tree), so an existing lock is NEVER broken automatically — it is
18 * reported with its opener's pid and retired only by `abort` or `publish`.
19 */

54int brixcvmfs_tx_lock_pid(const char *lockpath) {
55 FILE *f = fopen(lockpath, "r");
56 int pid = 0;
57 if (f != NULL) {
58 if (fscanf(f, "pid:%d", &pid) != 1) pid = 0;

64/* 1 = the lock's opener is provably gone: the recorded boot differs from
65 * the running boot, or the pid no longer exists on this boot. Callers must
66 * only judge locks with no durable upper tree behind them (a crashed
67 * ingest); a `repo transaction` lock guards real staged state and is never
68 * broken automatically. */
69int brixcvmfs_tx_lock_stale(const char *lockpath) {
70 char boot[64], cur[64];
71 int pid = 0;
72 FILE *f = fopen(lockpath, "r");
73 if (f == NULL) return 0;
74 boot[0] = '\0';
75 int fields = fscanf(f, "pid:%d\nboot:%63s", &pid, boot);
76 fclose(f);
77 if (fields != 2) return 0;
78 tx_boot_id(cur, sizeof(cur));
79 if (strcmp(boot, cur) != 0) return 1;
80 return pid > 0 && kill(pid, 0) != 0 && errno == ESRCH;
81}

83int brixcvmfs_tx_lock_take(const char *lockpath) {
84 char boot[64], body[128];
85 tx_boot_id(boot, sizeof(boot));
86 int fd = open(lockpath, O_WRONLY | O_CREAT | O_EXCL | O_CLOEXEC, 0644);
87 if (fd < 0) return -1;
88 int len = snprintf(body, sizeof(body), "pid:%d\nboot:%s\n", (int) getpid(), boot);
89 int rc = write(fd, body, (size_t) len) == len ? 0 : -1;
90 close(fd);
91 return rc;
92}

96int brixcvmfs_tx_rm_tree(const char *path) {
97 struct stat st;
98 if (lstat(path, &st) != 0)
99 return errno == ENOENT ? 0 : -1;
100 if (!S_ISDIR(st.st_mode))
101 return unlink(path);
102 DIR *d = opendir(path);
103 if (d == NULL) return -1;
104 struct dirent *e;
105 int rc = 0;
106 while (rc == 0 && (e = readdir(d)) != NULL) {
107 if (strcmp(e->d_name, ".") == 0 || strcmp(e->d_name, "..") == 0)
108 continue;
109 char sub[TX_PATH * 2];
110 if (snprintf(sub, sizeof(sub), "%s/%s", path, e->d_name)
111 >= (int) sizeof(sub))
112 rc = -1;
113 else
114 rc = brixcvmfs_tx_rm_tree(sub);
115 }
116 closedir(d);
117 return rc == 0 ? rmdir(path) : rc;
118}

122static int tx_transaction(const char *repo_dir) {
123 char txn[TX_PATH], lock[TX_PATH + 16], upper[TX_PATH + 16];
124 snprintf(txn, sizeof(txn), "%s/.brixtxn", repo_dir);
125 snprintf(lock, sizeof(lock), "%s/lock", txn);
126 snprintf(upper, sizeof(upper), "%s/upper", txn);
127 if (mkdir(txn, 0755) != 0 && errno != EEXIST)
128 return tx_err("cannot create transaction dir", txn);
129 if (brixcvmfs_tx_lock_take(lock) != 0) {
130 if (errno != EEXIST)
131 return tx_err("cannot take transaction lock", lock);
132 char who[64];
133 snprintf(who, sizeof(who), "%d", brixcvmfs_tx_lock_pid(lock));
134 return tx_err("repository is in a transaction (pid)", who);
135 }
136 if (mkdir(upper, 0755) != 0 && errno != EEXIST)
137 return tx_err("cannot create upper tree", upper);
138 printf("transaction open: %s\n", upper);
139 return 0;
140}

154static int tx_publish(const char *repo_dir, int argc, char **argv) {
155 cvmfs_publish_opts_t o;
156 memset(&o, 0, sizeof(o));
157 o.repo_dir = repo_dir;
158 for (int i = 0; i < argc; i++) {
159 if (strcmp(argv[i], "--chunk-size") == 0 && i + 1 < argc)
160 o.chunk_size = atol(argv[++i]);
161 else if (strcmp(argv[i], "--dirtab") == 0 && i + 1 < argc)
162 o.dirtab = argv[++i];
163 else if (argv[i][0] != '-' && o.keys_dir == NULL)
164 o.keys_dir = argv[i];
165 else
166 return tx_err("unknown publish option", argv[i]);
167 }
168
169 char txn[TX_PATH], upper[TX_PATH + 16], err[1024];
170 struct stat st;
171 snprintf(txn, sizeof(txn), "%s/.brixtxn", repo_dir);
172 snprintf(upper, sizeof(upper), "%s/upper", txn);
173 if (lstat(upper, &st) != 0 || !S_ISDIR(st.st_mode))
174 return tx_err("no open transaction under", repo_dir);
175
176 cvmfs_changeset_t cs;
177 if (cvmfs_changeset_scan(upper, &cs, err, sizeof(err)) != 0)
178 return tx_err("changeset scan failed", err);
179 long new_rev = 0;
180 int rc = cvmfs_publish_run(&o, &cs, &new_rev, err, sizeof(err));
181 cvmfs_changeset_free(&cs);
182 if (rc != 0)
183 return tx_err("publish failed", err);
184 if (brixcvmfs_tx_rm_tree(txn) != 0)
185 return tx_err("published, but cannot retire transaction dir", txn);
186 printf("published revision %ld\n", new_rev);
187 return 0;
188}

198int brixcvmfs_txn_main(int argc, char **argv) {
199 /* argv[0] = "repo" after the front-end shift. */
200 const char *cmd = argc >= 2 ? argv[1] : "";
201 if (strcmp(cmd, "transaction") == 0 && argc == 3)
202 return tx_transaction(argv[2]);
203 if (strcmp(cmd, "abort") == 0 && argc == 3)
204 return tx_abort(argv[2]);
205 if (strcmp(cmd, "publish") == 0 && argc >= 3)
206 return tx_publish(argv[2], argc - 3, argv + 3);
207 if (strcmp(cmd, "fsck") == 0 && argc == 3)
208 return tx_fsck(argv[2], 0);
209 if (strcmp(cmd, "fsck") == 0 && argc == 4 && strcmp(argv[3], "--data") == 0)
210 return tx_fsck(argv[2], 1);
211 fprintf(stderr,
212 "usage: brixcvmfs repo transaction <repo_dir>\n"
213 " brixcvmfs repo abort <repo_dir>\n"
214 " brixcvmfs repo publish <repo_dir> [keys_dir]"
215 " [--chunk-size N] [--dirtab F]\n"
216 " brixcvmfs repo fsck <repo_dir> [--data]\n");
217 return 2;
218}
