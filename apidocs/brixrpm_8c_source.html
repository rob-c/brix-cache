1/* brixrpm — clean-room createrepo + RPM inspector (phase-104 D12.1), an
2 * argv[0] personality of brixMount (dispatch in apps/fs/brixmount.c).
3 *
4 * WHAT: `createrepo <dir>` builds dnf-consumable repodata/ from a directory
5 * of .rpm files; `inspect <pkg.rpm>` prints NEVRA + deps + files as a
6 * debug/verify aid (the D12.4 oracle leg diffs it against `rpm -qp`).
7 * WHY: sites publishing RPM repos into CVMFS (D13) need a repo builder with
8 * no createrepo_c/librpm dependency; the parser and emitter live in
9 * shared/rpm/, this file is only the verb surface.
10 * HOW: exit codes follow the brixcvmfs table subset: 0 ok · 1 operation

22static void
23rpm_usage(FILE *out)
24{
25 fprintf(out,
26 "usage: brixrpm createrepo <dir> [--update] [--paranoid] [--strict]\n"
27 " [--baseurl-relative] [--compress gz]\n"
28 " brixrpm inspect <pkg.rpm> [--json]\n"
29 "\n"
30 " createrepo scans <dir> recursively for *.rpm and writes\n"
31 " repodata/ (primary/filelists/other + repomd.xml, repomd last).\n"
32 " --update reuses the .brixrpm-cache memo for packages whose\n"
33 " (size, mtime) are unchanged; --paranoid re-hashes each memo hit\n"
34 " instead, so a package rewritten in place under an unchanged\n"
35 " (size, mtime) is caught rather than republished stale (without\n"
36 " --update every package is parsed anyway, so it adds nothing).\n"
37 " --strict makes an unreadable .rpm fatal instead of a\n"
38 " skip-with-warning. Hrefs are always emitted repo-relative and\n"
39 " gzip is the only compression, so the last two flags just name\n"
40 " the defaults.\n"
41 "%s", BRIX_USAGE_FOOTER("brixrpm"));
42}

44/* Sense bits → the rpm -qp operator spelling ("" when unversioned). */
45static const char *
46rpm_sense_op(uint32_t flags)
47{
48 uint32_t s = flags & (BRIX_RPMSENSE_LT | BRIX_RPMSENSE_GT |
49 BRIX_RPMSENSE_EQ);
50
51 switch (s) {
52 case BRIX_RPMSENSE_LT: return "<";
53 case BRIX_RPMSENSE_GT: return ">";
54 case BRIX_RPMSENSE_EQ: return "=";
55 case BRIX_RPMSENSE_LT | BRIX_RPMSENSE_EQ: return "<=";
56 case BRIX_RPMSENSE_GT | BRIX_RPMSENSE_EQ: return ">=";
57 default: return "";
58 }
59}

61/* One dependency block (provides/requires/...) in text or JSON form.
62 * inspect shows the whole truth — rpmlib() tracking deps included — so its
63 * output diffs cleanly against `rpm -qp --requires`; only primary.xml
64 * filters them (repomd_write.c). */
65static void
66rpm_deps(brix_rpm_pkg_t *p, const char *label, uint32_t names_tag,
67 uint32_t flags_tag, uint32_t vers_tag, int json, int *jfirst)
68{
69 uint32_t n = brix_rpm_count(p, names_tag);
70 uint32_t i;
71
72 if (json) {
73 printf("%s \"%s\": [", *jfirst ? "" : ",", label);
74 *jfirst = 0;
75 } else {
76 printf("%s:\n", label);
77 }
78 for (i = 0; i < n; i++) {
79 const char *name = brix_rpm_stra(p, names_tag, i);
80 const char *ver = brix_rpm_stra(p, vers_tag, i);
81 uint32_t fl = 0;
82 const char *op;
83
84 if (name == NULL) {
85 break;
86 }
87 (void) brix_rpm_u32(p, flags_tag, i, &fl);
88 op = rpm_sense_op(fl);
89 if (json) {
90 printf("%s{", i > 0 ? "," : "");
91 brix_json_kv_str(stdout, "name", name, 0);
92 if (op[0] != '\0' && ver != NULL && ver[0] != '\0') {
93 printf(",");
94 brix_json_kv_str(stdout, "op", op, 0);
95 printf(",");
96 brix_json_kv_str(stdout, "version", ver, 0);
97 }
98 printf("}");
99 } else if (op[0] != '\0' && ver != NULL && ver[0] != '\0') {
100 printf(" %s %s %s\n", name, op, ver);
101 } else {
102 printf(" %s\n", name);
103 }
104 }
105 if (json) {
106 printf("]");
107 }
108}

112static void
113rpm_files(brix_rpm_pkg_t *p, int json)
114{
115 uint32_t nf = brix_rpm_nfiles(p);
116 uint32_t i;
117
118 if (json) {
119 printf(", \"files\": [");
120 } else {
121 printf("files:\n");
122 }
123 for (i = 0; i < nf; i++) {
124 char path[4096];
125
126 if (brix_rpm_file(p, i, path, sizeof(path), NULL, NULL) != 0) {
127 break;
128 }
129 if (json) {
130 printf("%s", i > 0 ? "," : "");
131 brix_json_fputs(stdout, path);
132 } else {
133 printf(" %s%s\n", path,
134 brix_rpm_path_sane(path) ? "" : " [insane path]");
135 }
136 }
137 if (json) {
138 printf("]");
139 }

206static int
207rpm_cmd_inspect(const char *path, int json)
208{
209 char err[512] = "";
210 char nevra[1024];
211 uint32_t epoch = 0;
212 brix_rpm_pkg_t *p = brix_rpm_open(path, err, sizeof(err));
213
214 if (p == NULL) {
215 fprintf(stderr, "brixrpm: %s\n", err);
216 return 1;
217 }
218 (void) brix_rpm_u32(p, BRIX_RPMTAG_EPOCH, 0, &epoch);
219 rpm_nevra(p, epoch, nevra, sizeof(nevra));
220 if (json) {
221 rpm_inspect_json(p, epoch, nevra);
222 } else {
223 rpm_inspect_text(p, epoch, nevra);
224 }
225 brix_rpm_close(p);
226 return 0;
227}

229/* createrepo flag walk: one positional (the dir). -1 = usage error. */
230static int
231rpm_createrepo_args(int argc, char **argv, brixrpm_cr_opts_t *o)
232{
233 int i;
234
235 memset(o, 0, sizeof(*o));
236 for (i = 2; i < argc; i++) {
237 const char *a = argv[i];
238
239 if (a[0] != '-') {
240 if (o->dir != NULL) {
241 return -1;
242 }
243 o->dir = a;
244 } else if (strcmp(a, "--update") == 0) {
245 o->update = 1;
246 } else if (strcmp(a, "--strict") == 0) {
247 o->strict = 1;
248 } else if (strcmp(a, "--paranoid") == 0) {
249 o->paranoid = 1;
250 } else if (strcmp(a, "--baseurl-relative") == 0) {
251 /* the default (and only) href mode — accepted, nothing to set */
252 } else if (strcmp(a, "--compress") == 0) {
253 if (i + 1 >= argc || strcmp(argv[++i], "gz") != 0) {
254 fprintf(stderr, "brixrpm: --compress supports only gz\n");
255 return -1;
256 }
257 } else {
258 fprintf(stderr, "brixrpm: unknown option %s\n", a);
259 return -1;
260 }
261 }
262 return o->dir != NULL ? 0 : -1;
263}

281/* `inspect <rpm> [--json]` — the only optional argument this tool takes, so
282 * its shape is checked here rather than in the dispatch table. */
283static int
284rpm_run_inspect(int argc, char **argv)
285{
286 int json = (argc == 4 && strcmp(argv[3], "--json") == 0);
287
288 if (argc < 3 || argc > 4 || (argc == 4 && !json)) {
289 rpm_usage(stderr);
290 return 2;
291 }
292 return rpm_cmd_inspect(argv[2], json);

296int
297brixrpm_main(int argc, char **argv)
298{
299 if (argc < 2) {
300 rpm_usage(stderr);
301 return 2;
302 }
303 if (strcmp(argv[1], "--version") == 0) {
304 printf("brixrpm (BriX-Cache client) %s\n", brix_client_version());
305 return 0;
306 }
307 if (strcmp(argv[1], "--help") == 0 || strcmp(argv[1], "-h") == 0) {
308 rpm_usage(stdout);
309 return 0;
310 }
311 if (strcmp(argv[1], "createrepo") == 0) {
312 return rpm_run_createrepo(argc, argv);
313 }
314 if (strcmp(argv[1], "inspect") == 0) {
315 return rpm_run_inspect(argc, argv);
316 }
317 fprintf(stderr, "brixrpm: unknown command \"%s\"\n", argv[1]);
318 rpm_usage(stderr);
319 return 2;
320}
