1/*
2 * oci_upload_seal.c — where the pushed bytes actually land (§D4.2, App. J.7).
3 *
4 * WHAT: the body half of the upload machine: append the request body to the
5 * staged part-file, and — on a seal — hash it, compare it against the
6 * digest the client claimed, and publish it into the content store.
7 * WHY: a registry that stores what it was TOLD it received rather than what
8 * it verifiably did receive is a corruption pump: every later pull of
9 * that layer hands a client bytes that hash to something else, and the
10 * client is the one that looks broken. The seal is therefore the single
11 * point where bytes become an object, and it is the only place in this
12 * module allowed to say so.
13 * HOW: the routing half (oci_upload.c) leaves a decision in ctx->reg; nginx
14 * re-enters here once the body is on hand. Appends go in at the
15 * part-file's current length under an advisory lock, so two writers on
16 * one session id collide loudly rather than interleaving.
17 *
18 * The raw namespace calls carry per-line vfs-seam-allow markers for the
19 * reason oci_store.c documents at its head: this is the registry's own
20 * staging area, not a VFS export.
21 */

33/* Append the request body onto the part-file at `u->base`, under the
34 * session's advisory lock. Returns the new length, or -1 with *status set. */
35static off_t
36oci_append_body(ngx_http_request_t *r, const oci_upload_ctx_t *u,
37 size_t max_blob, ngx_uint_t *status)
38{
39 brix_http_body_summary_t sum;
40 char path[PATH_MAX];
41 off_t end;
42 int fd;
43
44 *status = NGX_HTTP_INTERNAL_SERVER_ERROR;
45
46 if (brix_oci_store_upload_path(&u->st, u->session, u->session_len, "part",
47 path, sizeof(path)) != 0)
48 {
49 return -1;
50 }
51 fd = open(path, O_WRONLY | O_CLOEXEC); /* vfs-seam-allow: DOMAIN_REGISTRY — registry staging area; the received bytes are not an object until the seal */
52 if (fd < 0) {
53 *status = NGX_HTTP_NOT_FOUND;
54 return -1;
55 }
56
57 /* One writer per session. A second concurrent PATCH is a client bug (the
58 * id is issued to exactly one client), and interleaving its bytes would
59 * corrupt the layer silently — so it is refused, loudly. */
60 if (flock(fd, LOCK_EX | LOCK_NB) != 0) {
61 (void) close(fd);
62 *status = NGX_HTTP_CONFLICT;
63 return -1;
64 }
65
66 if (brix_http_body_write_to_fd_at(r, fd, path, &sum, u->base) != NGX_OK) {
67 (void) close(fd);
68 return -1;
69 }
70 (void) close(fd);
71
72 end = u->base + (off_t) sum.bytes;
73
74 /* The cap is enforced against the RUNNING COUNT, never a header: real
75 * layer PATCHes arrive chunked with no Content-Length at all (App. Y-2),
76 * so a header-based cap would be no cap. */
77 if (max_blob > 0 && (size_t) end > max_blob) {
78 *status = NGX_HTTP_REQUEST_ENTITY_TOO_LARGE;
79 return -1;
80 }
81 return end;
82}

85/* Move the sealed part-file into the CAS and record the repo's reference. */
86static ngx_int_t
87oci_seal_commit(ngx_http_request_t *r, const oci_upload_ctx_t *u,
88 const brix_oci_req_t *req, const brix_oci_digest_t *d)
89{
90 char part[PATH_MAX];
91 char blob[PATH_MAX];
92 char dir[PATH_MAX];
93 ngx_log_t *log = r->connection->log;
94
95 if (brix_oci_store_upload_path(&u->st, u->session, u->session_len, "part",
96 part, sizeof(part)) != 0
97 || brix_oci_store_upload_path(&u->st, u->session, u->session_len, NULL,
98 dir, sizeof(dir)) != 0
99 || brix_oci_store_blob_path(&u->st, d, blob, sizeof(blob)) != 0)
100 {
101 return NGX_HTTP_INTERNAL_SERVER_ERROR;
102 }
103
104 /* Content addressing makes a re-upload a no-op rather than a conflict: the
105 * bytes already in the store hash to the same digest by definition, so the
106 * winner of a race is irrelevant. The exclusive publish makes that atomic —
107 * EEXIST is the "already have these bytes" answer, not a failure — and
108 * closes the exists()-then-publish window the previous two-step left open. */
109 if (brix_oci_store_publish_staged(&u->st, part, blob, log) != NGX_OK
110 && errno != EEXIST)
111 {
112 return NGX_HTTP_INTERNAL_SERVER_ERROR;
113 }
114 if (brix_oci_store_mark_layer(&u->st, req->name, req->name_len, d,
115 log) != NGX_OK)
116 {
117 return NGX_HTTP_INTERNAL_SERVER_ERROR;
118 }
119 brix_oci_store_drop_dir(dir, log);
120
121 return brix_oci_upload_created(r, req, d);
122}

127static ngx_int_t
128oci_body_monolithic(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
129 oci_upload_ctx_t *u, size_t max_blob)
130{
131 ngx_uint_t status;
132 ngx_int_t rc;
133 char part[PATH_MAX];
134
135 if (oci_append_body(r, u, max_blob, &status) < 0) {
136 return brix_oci_error(r, status,
137 (status == NGX_HTTP_REQUEST_ENTITY_TOO_LARGE)
138 ? BRIX_OCI_ERR_SIZE_INVALID
139 : BRIX_OCI_ERR_BLOB_UPLOAD_INVALID, NULL);
140 }
141 if (brix_oci_store_upload_path(&u->st, u->session, u->session_len, "part",
142 part, sizeof(part)) != 0)
143 {
144 return NGX_HTTP_INTERNAL_SERVER_ERROR;
145 }
146
147 rc = brix_oci_store_verify(part, &u->want, r->connection->log);
148 if (rc != NGX_OK) {
149 char dir[PATH_MAX];
150
151 if (brix_oci_store_upload_path(&u->st, u->session, u->session_len,
152 NULL, dir, sizeof(dir)) == 0)
153 {
154 brix_oci_store_drop_dir(dir, r->connection->log);
155 }
156 ctx->disp = BRIX_OCI_OUT_REFUSED;
157 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
158 BRIX_OCI_ERR_DIGEST_INVALID,
159 "the uploaded bytes do not hash to the "
160 "digest the request named");
161 }
162 return oci_seal_commit(r, u, &ctx->req, &u->want);
163}

166static ngx_int_t
167oci_body_seal(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
168 oci_upload_ctx_t *u, size_t max_blob)
169{
170 char part[PATH_MAX];
171 ngx_int_t rc;
172
173 /* A PUT may carry a final chunk. Appending it before the hash is what
174 * makes the two-request push (POST, then PUT-with-everything) work. */
175 if (r->request_body != NULL && r->request_body->bufs != NULL) {
176 ngx_uint_t status;
177
178 if (oci_append_body(r, u, max_blob, &status) < 0) {
179 return brix_oci_error(r, status,
180 (status == NGX_HTTP_REQUEST_ENTITY_TOO_LARGE)
181 ? BRIX_OCI_ERR_SIZE_INVALID
182 : BRIX_OCI_ERR_BLOB_UPLOAD_INVALID, NULL);
183 }
184 }
185 if (brix_oci_store_upload_path(&u->st, u->session, u->session_len, "part",
186 part, sizeof(part)) != 0)
187 {
188 return NGX_HTTP_INTERNAL_SERVER_ERROR;
189 }
190
191 rc = brix_oci_store_verify(part, &u->want, r->connection->log);
192 if (rc == NGX_ERROR) {
193 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
194 BRIX_OCI_ERR_BLOB_UPLOAD_UNKNOWN, NULL);
195 }
196 if (rc == NGX_DECLINED) {
197 /* J.7: the session STAYS ACTIVE. The client may have mis-stated the
198 * digest, or lost a chunk it can still re-send; destroying its work
199 * would turn a recoverable mistake into a full re-upload. */
200 ctx->disp = BRIX_OCI_OUT_REFUSED;
201 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
202 BRIX_OCI_ERR_DIGEST_INVALID,
203 "the staged bytes do not hash to the digest "
204 "the seal named");
205 }
206 return oci_seal_commit(r, u, &ctx->req, &u->want);
207}

210void
211brix_oci_upload_body_handler(ngx_http_request_t *r)
212{
213 ngx_http_brix_oci_loc_conf_t *lcf =
214 ngx_http_get_module_loc_conf(r, ngx_http_brix_oci_module);
215 ngx_http_brix_oci_ctx_t *ctx =
216 ngx_http_get_module_ctx(r, ngx_http_brix_oci_module);
217 oci_upload_ctx_t *u = (ctx != NULL) ? ctx->reg : NULL;
218 ngx_uint_t status;
219 off_t end;
220 ngx_int_t rc;
221
222 if (u == NULL) {
223 ngx_http_finalize_request(r, NGX_HTTP_INTERNAL_SERVER_ERROR);
224 return;
225 }
226
227 switch (u->act) {
228
229 case OCI_UP_MONOLITHIC:
230 rc = oci_body_monolithic(r, ctx, u, lcf->max_blob);
231 break;
232
233 case OCI_UP_SEAL:
234 rc = oci_body_seal(r, ctx, u, lcf->max_blob);
235 break;
236
237 default:
238 end = oci_append_body(r, u, lcf->max_blob, &status);
239 if (end < 0) {
240 rc = brix_oci_error(r, status,
241 (status == NGX_HTTP_REQUEST_ENTITY_TOO_LARGE)
242 ? BRIX_OCI_ERR_SIZE_INVALID
243 : BRIX_OCI_ERR_BLOB_UPLOAD_UNKNOWN, NULL);
244 break;
245 }
246 rc = (brix_oci_upload_headers(r, &ctx->req, u->session, end) == NGX_OK)
247 ? brix_oci_reply_empty(r, NGX_HTTP_ACCEPTED)
248 : NGX_HTTP_INTERNAL_SERVER_ERROR;
249 break;
250 }
251
252 ngx_http_finalize_request(r, rc);
253}
