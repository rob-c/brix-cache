37/* ---- Parse a stock RATE[k|m|g] string into bytes/sec ----
38 *
39 * WHAT: strtoll + one optional case-insensitive binary suffix; returns the
40 * rate in bytes/sec, or -1 for garbage, non-positive values, trailing
41 * junk, or multiplication overflow.
42 *
43 * WHY: --xrate/--xrate-threshold accept the stock spellings ("512k", "10m");
44 * a hostile value must fail the parse, never wrap into a tiny or
45 * negative rate.
46 *
47 * HOW: errno/end checks, single-suffix switch, INT64_MAX/mult overflow guard.
48 */
49static int64_t
50xrdcp_parse_rate(const char *s)
51{
52 char *end;
53 long long v;
54 int64_t mult = 1;
55
56 if (s == NULL || *s == '\0') {
57 return -1;
58 }
59 errno = 0;
60 v = strtoll(s, &end, 10);
61 if (errno != 0 || end == s || v <= 0) {
62 return -1;
63 }
64 if (*end != '\0') {
65 switch (*end | 0x20) {
66 case 'k': mult = 1024LL; break;
67 case 'm': mult = 1024LL * 1024; break;
68 case 'g': mult = 1024LL * 1024 * 1024; break;
69 default: return -1;
70 }
71 if (end[1] != '\0') {
72 return -1;
73 }
74 }
75 if (v > (long long) (INT64_MAX / mult)) {
76 return -1;
77 }
78 return (int64_t) v * mult;
79}

143static int
144xrdcp_parse_retry_option(const char *a, xrdcp_opts_t *o, int argc,
145 char **argv, size_t *i)
146{
147
148 if (strcmp(a, "--retry-policy") == 0 && *i + 1 < (size_t) argc) {
149 const char *policy = argv[++(*i)];
150
151 /* §7.13 stock semantics: how a RETRY treats the partial destination.
152 * "continue" = resume at the partial's size — exactly the --continue
153 * write mode, so it simply arms that engine (each retry then picks up
154 * where the failed attempt stopped); "force" = restart from scratch,
155 * the pre-existing default. */
156 if (strcmp(policy, "continue") == 0) {
157 o->copt->cont = 1;
158 } else if (strcmp(policy, "force") != 0) {
159 fprintf(stderr, "xrdcp: --retry-policy expects 'force' or "
160 "'continue', got '%s'\n", policy);
161 return 50;
162 }
163 return 1;
164 }
165 if (strcmp(a, "--retry") == 0 && *i + 1 < (size_t) argc) {
166 o->retries = atoi(argv[++(*i)]);
167 if (o->retries <= 0) {
168 o->retries = 0;
169 o->copt->no_retry = 1;
170 }
171 o->copt->retry_count = o->retries;
172 return 1;
173 }
174 if (strcmp(a, "--no-retry") == 0) {
175 o->retries = 0;
176 o->copt->retry_count = 0;
177 o->copt->no_retry = 1;
178 return 1;
179 }
180 if ((strcmp(a, "-j") == 0 || strcmp(a, "--jobs") == 0)
181 && *i + 1 < (size_t) argc) {
182 o->jobs = atoi(argv[++(*i)]);
183 return 1;
184 }
185 return 0;

358 * already satisfy unconditionally.
359 *
360 * WHAT: Accepts -A/--allow-http and does nothing with it.
361 * WHY: In stock xrdcp that flag is the gate on the XrdClHttp plugin — the
362 * client refuses http/davs URLs until it is given. This client has no
363 * plugin layer and no such gate: every scheme its transport understands
364 * is available on every invocation, so the permission is already

370 * read as touching TLS posture or host verification.

372 * stays under the complexity cap, and so the compat spellings have one

389xrdcp_parse_option(xrdcp_cli_state *s, int argc, char **argv, size_t *i)
390{
391 int oi = (int) *i;
392 int pr = brix_opts_parse_arg(s->o->conn, argc, argv, &oi);
393 int rc;
394 const char *a = argv[*i];
395 xrdcp_opts_t *o = s->o;
396
397 if (pr == 2) { usage_fp(stdout, argv[0]); return 2; }
398 if (pr) { *i = (size_t) oi; return 1; }
399 rc = xrdcp_parse_basic_option(s, argc, argv, i);
400 if (rc) { return rc; }
401 rc = xrdcp_parse_compat_option(argc, argv, i);
402 if (rc) { return rc; }
403 rc = xrdcp_parse_manifest_option(a, o, argc, argv, i);
404 if (rc) { return rc; }
405 rc = xrdcp_parse_sync_filter_option(s, argc, argv, i);
406 if (rc) { return rc; }
407 rc = xrdcp_parse_auth_data_option(a, o, argc, argv, i);
408 if (rc) { return rc; }
409 rc = xrdcp_parse_transport_option(s->o->copt, argc, argv, i);
410 if (rc) { return rc; }
411 rc = xrdcp_parse_remote_auth_option(s, argc, argv, i);
412 if (rc) { return rc; }
413 if (strcmp(argv[*i], "-V") == 0) {
414 printf("%s (BriX-Cache client) %s\n", brix_prog_base(argv[0]),
415 brix_client_version());
416 return 2;
417 }
418 if (strcmp(argv[*i], "-h") == 0) { usage(argv[0]); return 2; }
419
420 fprintf(stderr, "xrdcp: unknown option '%s'\n", argv[*i]);
421 usage(argv[0]);
422 return 50;
423}

426/*
427 * WHAT: Parse and validate command-line arguments.
428 *
429 * WHY: The main() function is CCN 187 (527 lines). Extracting the argument
430 * parsing and validation logic reduces complexity and improves testability.
431 *
432 * HOW: Parse all CLI flags into the provided structures, validate flag
433 * interactions (--sync implies --force, --delete requires -r + --sync,
434 * etc.), build positional/exclusion/inclusion lists, and derive defaults.
435 * Returns 0 on success, 50 on usage error, 51 on OOM.
436 */
437int
438parse_and_validate_args(int argc, char **argv, xrdcp_opts_t *o, xrdcp_lists_t *l)
439{
440 size_t i;
441 xrdcp_cli_state state;
442
443 memset(&state, 0, sizeof(state));
444 state.o = o;
445 state.l = l;
446
447 /* Phase 44: XRDC_IO_URING env is the default (auto) for the local-disk
448 * overlap ring; --io-uring overrides it below. auto = 0 = memset default. */
449 {
450 const char *e = getenv("XRDC_IO_URING");
451 if (e != NULL) {
452 if (strcmp(e, "on") == 0) { o->copt->io_uring = XRDC_IO_URING_ON; }
453 else if (strcmp(e, "off") == 0) { o->copt->io_uring = XRDC_IO_URING_OFF; }
454 else { o->copt->io_uring = XRDC_IO_URING_AUTO; }
455 }
456 }
457 /* XRDC_IO_URING_DIRECT env is the default for the O_DIRECT tier (off unless
458 * set); --io-uring-direct overrides it below. */
459 {
460 const char *e = getenv("XRDC_IO_URING_DIRECT");
461 if (e != NULL) {
462 o->copt->io_uring_direct =
463 (strcmp(e, "on") == 0 || strcmp(e, "1") == 0) ? 1 : 0;
464 }
465 }
466
467 for (i = 1; i < (size_t) argc; i++) {
468 const char *a = argv[i];
469 if (a[0] == '-' && a[1] != '\0' && strcmp(a, "-") != 0) {
470 int parsed = xrdcp_parse_option(&state, argc, argv, &i);
471 if (parsed == 2) { return XRDCP_PARSE_EXIT_OK; }
472 if (parsed == 50) { return 50; }
473 } else if (str_append(&l->pos.items, &l->pos.n, &l->pos.cap, a) != 0) {
474 fprintf(stderr, "xrdcp: out of memory\n");
475 return 51;
476 }
477 }
478 if (state.oom) {
479 fprintf(stderr, "xrdcp: out of memory\n");
480 return 51;
481 }
482
483 o->copt->excludes = (const char *const *) l->excl.items;
484 o->copt->n_excludes = l->excl.n;
485 o->copt->includes = (const char *const *) l->incl.items;
486 o->copt->n_includes = l->incl.n;
487
488 return xrdcp_validate_and_finalize_args(o, l, argv[0]);
489}
