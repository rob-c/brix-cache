2 *
3 * WHAT: Three static helper functions + one public API function fetch OAuth2/OIDC
4 * access tokens for native XRootD TPC outbound authentication.
5 *
6 * Two delegation modes:
7 * - oidc-agent: fork/exec UNIX-socket JSON IPC to local oidc-agent daemon
8 * (tries dedicated helper binary first, falls back to oidc-token CLI)
9 * - token-exchange: RFC 8693 POST to external OAuth2 token endpoint via
10 * fork/exec curl with subject_token from tpc_outbound_bearer_file
11 *
12 * tpc_fetch_delegated_token (public API) dispatches based on t->token_mode:
13 * - "none": skip token fetch
14 * - "oidc-agent": use oidc-agent daemon
15 * - "token-exchange": use RFC 8693 token exchange (validates token_endpoint)
16 *
17 * WHY: TPC source authentication may require delegated OAuth2/OIDC access tokens
18 * when the destination server needs to authenticate as a different identity
19 * to the remote origin.
20 *
21 * - oidc-agent mode: fetches tokens from local agent daemon (CMS/Fermilab)
22 * - token-exchange mode: RFC 8693 exchange using bearer file subject token
23 * - Both modes: fork-exec subprocesses, read stdout via pipe (EINTR-safe)
24 * - JSON parsing: delegates to brix_oauth2_parse_access_token()
25 * - Plain tokens: copy directly into t->delegated_token with size guard
26 *
27 * HOW: fetch_delegated_token dispatches on token_mode:
28 * - "none"/empty: return 0
29 * - "oidc-agent": tpc_token_oidc_agent(pipe, fork, exec, read, parse)
30 * - "token-exchange": validate endpoint → tpc_token_rfc8693(build POST,
31 * mkstemp, curl -X POST, pipe read, parse JSON)

34 * - rfc8693: read bearer file → snprintf POST body → mkstemp temp →
35 * write → curl -s -S -f -X POST -H Content-Type -u client_id:secret →

37 * - unknown mode: snprintf err_msg, xrd_error=kXR_ArgInvalid, return -1

76/*
77 * WHAT: Trims trailing whitespace (space, tab, newline, carriage return) from s.
78 * - Returns early if s is NULL or already empty
79 * - Enforces strlen() boundary via while(len > 0) guard
80 * - Prevents underflow when trimming all-whitespace string
81 *
82 * WHY: TPC token helper outputs may have trailing whitespace from subprocess
83 * execution. This cleanup ensures clean token strings before parsing.
84 * Prevents validation failures from invisible trailing characters.
85 *
86 * HOW: NULL/empty check → strlen(s) → while loop decrementing len from end,
87 * replacing each whitespace/newline/CR byte with '\0' → stops when first
88 * non-trimmable char reached or len=0 (all-whitespace case)
89 */
90
91static void
92tpc_trim_trailing(char *s)
93{
94 size_t len;
95
96 if (s == NULL || *s == '\0') {
97 return;
98 }
99
100 len = strlen(s);
101 while (len > 0 && (isspace((unsigned char) s[len - 1]) || s[len - 1] == '\n'
102 || s[len - 1] == '\r')) {
103 s[--len] = '\0';
104 }
105}

107/* WHAT: Extract the "access_token" field from an OAuth2 JSON response into out.
108 * WHY: Thin wrapper over the shared oauth2 parser used by both delegation
109 * modes; on failure it overwrites `out` with the parser's error text so the
110 * caller can copy it into t->err_msg (out doubles as the error scratch buffer).

129/* WHAT: Sanity-check a freshly fetched delegated token before it is used to
130 * authenticate the outbound TPC pull.
131 * WHY: A delegation backend (oidc-agent / token-exchange) can return a string
132 * that is well-formed JSON-wise but not a usable credential; validating here
133 * fails fast with kXR_AuthFailed rather than letting a bad token surface as an
134 * opaque remote-open rejection later. An empty token is treated as "no
135 * delegation requested" and accepted (returns 0).
136 * HOW: Wrap delegated_token in an ngx_str_t, parse+validate it as a
137 * BRIX_TPC_CREDENTIAL_TOKEN; on any failure set err_msg/xrd_error and -1. */
138static int
139tpc_token_validate_delegated(brix_tpc_pull_t *t)
140{
141 ngx_str_t raw;
142 brix_tpc_credential_t cred;
143
144 /* Empty token => caller did not request delegation; nothing to validate. */
145 if (t->delegated_token[0] == '\0') {
146 return 0;
147 }
148
149 raw.data = (u_char *) t->delegated_token;
150 raw.len = ngx_strlen(t->delegated_token);
151
152 if (brix_tpc_credential_parse(&raw, BRIX_TPC_CREDENTIAL_TOKEN,
153 &cred, NULL,
154 t->c != NULL ? t->c->log : NULL)
155 != NGX_OK
156 || brix_tpc_credential_validate(
157 &cred, t->c != NULL ? t->c->log : NULL) != NGX_OK)
158 {
159 snprintf(t->err_msg, sizeof(t->err_msg),
160 "TPC token: delegated token validation failed");
161 t->xrd_error = kXR_AuthFailed;
162 return -1;
163 }
164
165 return 0;
166}

168/* WHAT: Resolve the oidc-token binary to an absolute path — first honours an
169 * explicit override env var, then probes standard install locations.
170 * WHY: The fallback exec path must not be subject to $PATH substitution in the
171 * daemon's environment (a compromised PATH entry could shadow the real binary).
172 * HOW: secure_getenv("BRIX_OIDC_TOKEN_BIN") if access(X_OK) passes; else
173 * walk a fixed candidate list; return NULL if none executable → caller _exit(127). */
174static const char *
175resolve_oidc_token_binary(void)
176{
177 static const char *const candidates[] = {
178 "/usr/bin/oidc-token", "/usr/local/bin/oidc-token", NULL
179 };
180 const char *const *p;
181 const char *override = secure_getenv("BRIX_OIDC_TOKEN_BIN");
182
183 if (override != NULL && access(override, X_OK) == 0) {
184 return override;
185 }
186 for (p = candidates; *p != NULL; p++) {
187 if (access(*p, X_OK) == 0) {
188 return *p;
189 }
190 }
191 return NULL;
192}

194/* WHAT: In the forked child, resolve the generic oidc-token CLI to an absolute
195 * path and execve it with a minimal controlled environment; returns only if the
196 * binary is unresolved or execve fails (caller then _exit(127)).
197 * WHY: The dedicated helper binary may be absent, so oidc-token is the fallback
198 * token source. It must run with an absolute path (no $PATH substitution — the
199 * daemon's PATH is untrusted) and a curated environment: PATH + HOME + OIDC_SOCK
200 * + XDG_RUNTIME_DIR (the latter two let oidc-token locate its config and agent
201 * socket); the full environ is deliberately NOT inherited.
202 * HOW: resolve_oidc_token_binary(); if found, read HOME/XDG_RUNTIME_DIR, build a
203 * fixed argv (-c default) and a curated envp (PATH always, HOME/OIDC_SOCK/
204 * XDG_RUNTIME_DIR only when set), then execve. */
205static void
206tpc_oidc_exec_fallback(const char *sock_env)
207{
208 const char *fb_bin = resolve_oidc_token_binary();
209
210 if (fb_bin != NULL) {
211 const char *home_val = getenv("HOME");
212 const char *xdg_val = getenv("XDG_RUNTIME_DIR");
213 char home_buf[280], oidc_buf[280], xdg_buf[280];
214 char *fb_argv[4] = { (char *) fb_bin, "-c", "default", NULL };
215 char *fb_envp[5];
216 int ei = 0;
217
218 fb_envp[ei++] = "PATH=/usr/bin:/bin";
219 if (home_val != NULL) {
220 snprintf(home_buf, sizeof(home_buf), "HOME=%s", home_val);
221 fb_envp[ei++] = home_buf;
222 }
223 if (sock_env != NULL) {
224 snprintf(oidc_buf, sizeof(oidc_buf), "OIDC_SOCK=%s", sock_env);
225 fb_envp[ei++] = oidc_buf;
226 }
227 if (xdg_val != NULL) {
228 snprintf(xdg_buf, sizeof(xdg_buf), "XDG_RUNTIME_DIR=%s", xdg_val);
229 fb_envp[ei++] = xdg_buf;
230 }
231 fb_envp[ei] = NULL;
232 execve(fb_bin, fb_argv, fb_envp);
233 }
234}

236/* WHAT: Child-process body of the oidc-agent fetch — redirects stdout to the
237 * pipe write end and execs the token helper chain; never returns (ends in
238 * _exit(127) if both execs fail).
239 * WHY: Isolating the post-fork child path keeps the orchestrator flat and puts
240 * the exec fallback chain (dedicated helper → generic oidc-token) in one place.
241 * The dedicated helper is tried first via execve with an absolute path (avoiding
242 * a TOCTOU access()-before-exec race), then the CLI fallback.
243 * HOW: close read end; dup2 write end onto STDOUT; close write end; read
244 * OIDC_SOCK (default the standard user socket path); execve the dedicated helper
245 * with a two-arg argv; on return call tpc_oidc_exec_fallback(); _exit(127). */
246static void
247tpc_oidc_child_run(int *pipefd)
248{
249 const char *sock_env;
250 char *argv[3];
251
252 /* redirect stdout into the write end of the pipe so the parent can read the
253 * helper's token output, then close both raw fds. */
254 close(pipefd[0]);
255 dup2(pipefd[1], STDOUT_FILENO);
256 close(pipefd[1]);
257
258 /* Try the dedicated helper first; fall through to oidc-token if execve fails
259 * (binary absent / not executable). Avoid access() before execve — that is a
260 * TOCTOU race on the executable path. */
261 sock_env = getenv("OIDC_SOCK");
262 argv[0] = (char *) TPC_TOKEN_HELPER_PATH;
263 argv[1] = sock_env ? (char *) sock_env
264 : (char *) "/run/user/1000/oidc/oidc_agent.sock";
265 argv[2] = NULL;
266 execve(argv[0], argv, NULL);
267 /* execve returned — helper not found or not executable; try the fallback. */
268
269 tpc_oidc_exec_fallback(sock_env);
270 _exit(127); /* both execs failed: 127 == "command not found" */
271}

273/* WHAT: Parent-side drain of the child's stdout — reads up to bufsz-1 bytes from
274 * fd into buf and NUL-terminates it. Returns the byte count read.
275 * WHY: A single read() may return short, so the token output must be drained in
276 * a loop until EOF/error or the buffer is full (minus the NUL slot). Factoring
277 * this out keeps the orchestrator's control flow linear.
278 * HOW: loop read()ing into buf at the running offset while space remains; stop on
279 * EOF/error (nr <= 0); NUL-terminate at the accumulated length. */
280static ssize_t
281tpc_oidc_read_all(int fd, char *buf, size_t bufsz)
282{
283 ssize_t nread = 0;
284
285 while ((size_t) nread < bufsz - 1) {
286 ssize_t nr = read(fd, buf + nread, bufsz - 1 - nread);
287 if (nr <= 0) {
288 break;
289 }
290 nread += nr;
291 }
292 buf[nread] = '\0';
293 return nread;
294}

296/* WHAT: Extracts the usable token from the (already whitespace-trimmed) helper
297 * output buf into token_out. Returns 0 on success, -1 (with err_msg/xrd_error
298 * set) on parse failure or oversize plain token.
299 * WHY: The two token producers differ in shape — the oidc-agent daemon emits a
300 * JSON object, the oidc-token CLI emits a bare token — and this discrimination
301 * belongs in one pure helper so the orchestrator does not branch on output form.
302 * HOW: a leading '{' means JSON → tpc_token_parse_access_token(); otherwise a
303 * bare token → length-guard against token_out_sz and ngx_memcpy verbatim. */
304static int
305tpc_oidc_extract_token(brix_tpc_pull_t *t, const char *buf,
306 char *token_out, size_t token_out_sz)
307{
308 if (buf[0] == '{') {
309 if (tpc_token_parse_access_token(buf, token_out, token_out_sz) != 0) {
310 t->xrd_error = kXR_AuthFailed;
311 return -1;
312 }
313 return 0;
314 }
315
316 {
317 size_t tok_len = strlen(buf);
318 if (tok_len >= token_out_sz) {
319 snprintf(t->err_msg, sizeof(t->err_msg),
320 "TPC token: token too long");
321 t->xrd_error = kXR_AuthFailed;
322 return -1;
323 }
324 ngx_memcpy(token_out, buf, tok_len + 1);
325 }
326
327 return 0;
328}

330/* WHAT: Fetches an OAuth2/OIDC access token via fork/exec to a local oidc-agent
331 * (dedicated helper binary, falling back to the oidc-token CLI) using pipe IPC.
332 * Returns 0 on success (token in token_out), -1 with err_msg/xrd_error set on
333 * pipe/fork failure, non-zero child exit, parse failure, or oversize token.
334 * WHY: CMS/Fermilab environments use oidc-agent daemons for OIDC token
335 * management. The fallback chain (dedicated helper → generic oidc-token CLI)
336 * ensures token fetch works even when the custom binary is absent. Pipe-based
337 * IPC with execve (no access() pre-check) avoids a TOCTOU race on the exec path.
338 * HOW: pipe() then fork(); child runs tpc_oidc_child_run(); parent closes the
339 * write end, drains stdout via tpc_oidc_read_all(), waitpid()s and checks the
340 * exit status, trims trailing whitespace, then tpc_oidc_extract_token(). */
341static int
342tpc_token_oidc_agent(brix_tpc_pull_t *t, char *token_out, size_t token_out_sz)
343{
344 int pipefd[2];
345 pid_t pid;
346 char buf[BRIX_TPC_TOKEN_MAX + BRIX_TPC_TOKEN_ERR_MAX];
347 int wstatus;
348
349 pipefd[0] = -1;
350 pipefd[1] = -1;
351
352 if (pipe(pipefd) == -1) {
353 snprintf(t->err_msg, sizeof(t->err_msg),
354 "TPC token: pipe() failed: %s", strerror(errno));
355 t->xrd_error = kXR_ServerError;
356 return -1;
357 }
358
359 pid = fork();
360 if (pid == -1) {
361 snprintf(t->err_msg, sizeof(t->err_msg),
362 "TPC token: fork() failed: %s", strerror(errno));
363 close(pipefd[0]);
364 close(pipefd[1]);
365 t->xrd_error = kXR_ServerError;
366 return -1;
367 }
368
369 if (pid == 0) {
370 tpc_oidc_child_run(pipefd);
371 }
372
373 /* parent --- : close the write end so read() sees EOF when the child exits,
374 * then drain all stdout into buf. */
375 close(pipefd[1]);
376 tpc_oidc_read_all(pipefd[0], buf, sizeof(buf));
377 close(pipefd[0]);
378
379 waitpid(pid, &wstatus, 0);
380 if (!WIFEXITED(wstatus) || WEXITSTATUS(wstatus) != 0) {
381 snprintf(t->err_msg, sizeof(t->err_msg),
382 "TPC token: oidc-agent helper exited with status %d",
383 WIFEXITED(wstatus) ? WEXITSTATUS(wstatus) : -1);
384 t->xrd_error = kXR_AuthFailed;
385 return -1;
386 }
387
388 tpc_trim_trailing(buf);
389
390 return tpc_oidc_extract_token(t, buf, token_out, token_out_sz);
391}

393/*
394 * WHAT: Public entry point dispatching delegated token fetching by t->token_mode.
395 *
396 * Mode behaviors:
397 * - "none"/empty: return 0 immediately (no token fetch)
398 * - "passthrough" (STRICT): validate inbound bearer JWT in t->delegated_token
399 * (empty → kXR_AuthFailed, no fallback)
400 * - "passthrough-opt" (OPPORTUNISTIC): validate inbound JWT if present,
401 * return 0 if absent (allows fallback to bearer-file/GSI/anon)
402 * - "oidc-agent": delegate to tpc_token_oidc_agent() (UNIX-socket JSON IPC)
403 * - "token-exchange": delegate to tpc_token_rfc8693() (RFC 8693 POST,
404 * validates token_endpoint configured first)
405 * - unknown: snprintf err_msg, xrd_error=kXR_ArgInvalid, return -1
406 *
407 * WHY: TPC source authentication requires delegated tokens when destination server
408 * authenticates as different identity to remote origin.
409 * This dispatcher centralizes mode selection — callers pass t->token_mode and
410 * receive fetched token in t->delegated_token without knowing backend mechanism.
411 * Prevents callers from duplicating mode-switch logic across launch.c/thread.c.
412 *
413 * Passthrough modes differ:
414 * - No fetch happens here — token captured on event loop (launch.c)
415 * - "passthrough" (client asked for it): fails closed when empty
416 * - "passthrough-opt" (server default): returns 0 so outbound auth path
417 * can fall back (never denies previously-anonymous/GSI-only pull)
418 *
419 * HOW: token_mode dispatch:
420 * - "none"/empty: return 0
421 * - "passthrough": require t->delegated_token[0] set
422 * (else err_msg/xrd_error=kXR_AuthFailed) → tpc_token_validate_delegated
423 * - "passthrough-opt": empty token → return 0, else tpc_token_validate_delegated
424 * - "oidc-agent": call tpc_token_oidc_agent(t, delegated_token, sizeof)
425 * - "token-exchange": validate conf->common.tpc_outbound_token_endpoint.len>0
426 * (else error) → call tpc_token_rfc8693(t, delegated_token, sizeof)
427 * - unknown: snprintf err_msg, xrd_error=kXR_ArgInvalid, return -1
428 */
429int
430tpc_fetch_delegated_token(brix_tpc_pull_t *t)
431{
432 if (t->token_mode[0] == '\0'
433 || ngx_strcmp(t->token_mode, "none") == 0) {
434 return 0;
435 }
436
437 /*
438 * passthrough (STRICT) — an explicit client tpc.token_mode=passthrough. The
439 * client's own inbound bearer JWT was already snapshotted into
440 * t->delegated_token on the event loop (tpc_pull_capture_passthrough_token in
441 * launch.c). There is nothing to fetch; just require the token to be present
442 * and valid so the source authenticates the end user. An empty token (no
443 * inbound bearer / one too large to forward) is a clean kXR_AuthFailed —
444 * never a silent anonymous/service fallback, because the client asked for it.
445 */
446 if (ngx_strcmp(t->token_mode, "passthrough") == 0) {
447 if (t->delegated_token[0] == '\0') {
448 snprintf(t->err_msg, sizeof(t->err_msg),
449 "TPC token: passthrough requested but no inbound bearer "
450 "token was presented by the client");
451 t->xrd_error = kXR_AuthFailed;
452 return -1;
453 }
454 return tpc_token_validate_delegated(t);
455 }
456
457 /*
458 * passthrough-opt (OPPORTUNISTIC) — the default-on brix_tpc_outbound_passthrough
459 * mode. Same event-loop capture as strict passthrough, but the ABSENCE of an
460 * inbound token is NOT an error: return 0 with delegated_token left empty so
461 * the outbound auth path (gsi_outbound_finish) falls back to the static bearer
462 * file, then GSI proxy delegation, then anonymous — exactly as it did before
463 * this mode became the default. This guarantees a flipped default never turns
464 * a previously-succeeding GSI-only or anonymous TPC pull into a new denial.
465 * When an inbound token IS present, use it (validate + forward via ztn).
466 */
467 if (ngx_strcmp(t->token_mode, "passthrough-opt") == 0) {
468 if (t->delegated_token[0] == '\0') {
469 return 0;
470 }
471 return tpc_token_validate_delegated(t);
472 }
473
474 if (ngx_strcmp(t->token_mode, "oidc-agent") == 0) {
475 if (tpc_token_oidc_agent(t, t->delegated_token,
476 sizeof(t->delegated_token)) != 0) {
477 return -1;
478 }
479 return tpc_token_validate_delegated(t);
480 }
481
482 if (ngx_strcmp(t->token_mode, "token-exchange") == 0) {
483 if (t->conf->common.tpc_outbound_token_endpoint.len == 0) {
484 snprintf(t->err_msg, sizeof(t->err_msg),
485 "TPC token: token_endpoint not configured for "
486 "token-exchange mode");
487 t->xrd_error = kXR_ArgInvalid;
488 return -1;
489 }
490 if (tpc_token_rfc8693(t, t->delegated_token,
491 sizeof(t->delegated_token)) != 0) {
492 return -1;
493 }
494 return tpc_token_validate_delegated(t);
495 }
496
497 snprintf(t->err_msg, sizeof(t->err_msg),
498 "TPC token: unknown delegation mode \"%s\"", t->token_mode);
499 t->xrd_error = kXR_ArgInvalid;
500 return -1;
501}
