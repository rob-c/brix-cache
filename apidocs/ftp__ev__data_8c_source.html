12/*
13 * ftp_ev_data.c — non-blocking data-channel lifecycle for the event engine.
14 *
15 * WHAT: the event-driven bring-up of a data connection (accept a passive client,
16 * or connect out in active mode) and the idempotent transfer teardown. The
17 * verbs that nominate the endpoint live in ftp_ev_data_setup.c.
18 *
19 * WHY: the sync engine blocks in accept()/connect() and then in the read/write
20 * loop, monopolising the worker for one transfer. Here the listener and the
21 * connect socket are wrapped in nginx connections and driven by their own
22 * read/write events, so the worker keeps serving other sessions while a transfer
23 * proceeds.
24 *
25 * HOW: brix_ftp_ev_data_open() is called once a transfer verb has sent its 150:
26 * it wraps the listener PASV/EPSV left in fc->pasv_fd (passive) or a fresh
27 * connect() socket aimed at fc->active_sa (active) in an ngx_connection_t and
28 * arms the accept/connect event. When the data socket is ready the handler
29 * hands off to brix_ftp_ev_data_ready() (ftp_ev_xfer.c), which opens the VFS
30 * side and starts the pump. Every transfer, whatever its shape, ends in
31 * brix_ftp_ev_data_finish() — the single place the CNS note and the
32 * {proto="gridftp"} metrics record are emitted.
33 */

40/* Wrap an already-open data fd in an nginx connection with the standard
41 * send/recv vtable, ready for the pump to arm read/write events on it. Exported
42 * so the MODE E receiver (ftp_ev_mode_e.c) wraps its child streams identically. */
43ngx_connection_t *
44brix_ftp_ev_wrap_conn(ftp_ev_t *fc, int fd)
45{
46 ngx_connection_t *c = ngx_get_connection(fd, fc->c->log);
47
48 if (c == NULL) {
49 return NULL;
50 }
51 c->recv = ngx_recv;
52 c->send = ngx_send;
53 c->recv_chain = ngx_recv_chain;
54 c->send_chain = ngx_send_chain;
55 c->log = fc->c->log;
56 c->read->log = fc->c->log;
57 c->write->log = fc->c->log;
58 return c;
59}

62/* Passive listener read event: a client has connected — accept it, retire the
63 * listener, and hand the data socket to the pump. */
64static void
65ev_accept_handler(ngx_event_t *rev)
66{
67 ngx_connection_t *lc = rev->data;
68 ftp_ev_dc_t *dc = lc->data;
69 ftp_ev_t *fc = dc->fc;
70 int dfd;
71
72 if (rev->timedout) {
73 ngx_log_error(NGX_LOG_INFO, fc->c->log, NGX_ETIMEDOUT,
74 "brix: GridFTP(ev) passive data accept timeout");
75 brix_ftp_ev_data_finish(dc, NGX_ERROR);
76 return;
77 }
78
79 dfd = accept(lc->fd, NULL, NULL);
80 if (dfd < 0) {
81 if (ngx_socket_errno == NGX_EAGAIN) {
82 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
83 brix_ftp_ev_data_finish(dc, NGX_ERROR);
84 }
85 return; /* spurious wakeup — wait */
86 }
87 ngx_log_error(NGX_LOG_ERR, fc->c->log, ngx_socket_errno,
88 "brix: GridFTP(ev) passive accept failed");
89 brix_ftp_ev_data_finish(dc, NGX_ERROR);
90 return;
91 }
92
93 /* Retire the listener (closes fc->pasv_fd) — one transfer, one accept. */
94 if (rev->timer_set) { ngx_del_timer(rev); }
95 ngx_close_connection(lc);
96 fc->pasv_fd = -1;
97 dc->lc = NULL;
98
99 if (ngx_nonblocking(dfd) == -1) {
100 (void) close(dfd);
101 brix_ftp_ev_data_finish(dc, NGX_ERROR);
102 return;
103 }
104 dc->dconn = brix_ftp_ev_wrap_conn(fc, dfd);
105 if (dc->dconn == NULL) {
106 (void) close(dfd);
107 brix_ftp_ev_data_finish(dc, NGX_ERROR);
108 return;
109 }
110 dc->dconn->data = dc;
111 if (fc->prot == 'P') {
112 brix_ftp_ev_dc_start_tls(dc); /* GSI-secure the data channel */
113 } else {
114 brix_ftp_ev_data_ready(dc);
115 }
116}

119/* Active-mode connect completion: confirm the non-blocking connect() succeeded,
120 * then hand the data socket to the pump. */
121static void
122ev_connect_handler(ngx_event_t *wev)
123{
124 ngx_connection_t *c = wev->data;
125 ftp_ev_dc_t *dc = c->data;
126 ftp_ev_t *fc = dc->fc;
127 int soerr = 0;
128 socklen_t slen = sizeof(soerr);
129
130 if (wev->timedout) {
131 ngx_log_error(NGX_LOG_INFO, fc->c->log, NGX_ETIMEDOUT,
132 "brix: GridFTP(ev) active data connect timeout");
133 brix_ftp_ev_data_finish(dc, NGX_ERROR);
134 return;
135 }
136 if (getsockopt(c->fd, SOL_SOCKET, SO_ERROR, &soerr, &slen) != 0
137 || soerr != 0)
138 {
139 ngx_log_error(NGX_LOG_ERR, fc->c->log, soerr,
140 "brix: GridFTP(ev) active data connect failed");
141 brix_ftp_ev_data_finish(dc, NGX_ERROR);
142 return;
143 }
144
145 if (wev->timer_set) { ngx_del_timer(wev); }
146 dc->connecting = 0;
147 if (fc->prot == 'P') {
148 brix_ftp_ev_dc_start_tls(dc); /* GSI-secure the data channel */
149 } else {
150 brix_ftp_ev_data_ready(dc);
151 }
152}

155/* ev_data_open_passive — PASV/EPSV: the listener fd is already open and
156 * non-blocking, so wrap it in a connection and arm its accept event. */
157static ngx_int_t
158ev_data_open_passive(ftp_ev_dc_t *dc)
159{
160 ftp_ev_t *fc = dc->fc;
161
162 if (fc->pasv_fd < 0) {
163 return NGX_ERROR;
164 }
165 dc->lc = ngx_get_connection(fc->pasv_fd, fc->c->log);
166 if (dc->lc == NULL) {
167 (void) close(fc->pasv_fd);
168 fc->pasv_fd = -1;
169 return NGX_ERROR;
170 }
171 dc->lc->log = fc->c->log;
172 dc->lc->read->log = fc->c->log;
173 dc->lc->data = dc;
174 /* MODE E STOR keeps the listener open to accept every parallel stream; all
175 * other transfers take a single connection and retire the listener. */
176 dc->lc->read->handler = (dc->mode_e && dc->writing)
177 ? brix_ftp_ev_eb_accept
178 : ev_accept_handler;
179 if (ngx_handle_read_event(dc->lc->read, 0) != NGX_OK) {
180 ngx_close_connection(dc->lc); /* also closes pasv_fd */
181 fc->pasv_fd = -1;
182 dc->lc = NULL;
183 return NGX_ERROR;
184 }
185 ngx_add_timer(dc->lc->read, BRIX_FTP_EV_IO_TIMEO);
186 return NGX_OK;
187}

190/* ev_data_open_active — PORT/EPRT: dial the armed, peer-pinned target from a
191 * fresh non-blocking socket sourced on the control channel's local IP. */
192static ngx_int_t
193ev_data_open_active(ftp_ev_dc_t *dc)
194{
195 ftp_ev_t *fc = dc->fc;
196 struct sockaddr_in local;
197 socklen_t llen = sizeof(local);
198 int dfd;
199 int rc;
200
201 fc->active = 0; /* one-shot: consume the arm */
202 ngx_memzero(&local, sizeof(local)); /* getsockname may leave it */
203
204 dfd = socket(AF_INET, SOCK_STREAM, 0);
205 if (dfd < 0) {
206 return NGX_ERROR;
207 }
208 if (ngx_nonblocking(dfd) == -1) {
209 (void) close(dfd);
210 return NGX_ERROR;
211 }
212 if (getsockname(fc->c->fd, (struct sockaddr *) &local, &llen) == 0
213 && local.sin_family == AF_INET)
214 {
215 local.sin_port = 0; /* ephemeral source port */
216 /* Best-effort source pinning: if the bind fails the kernel picks
217 * the source address itself, which is still a usable data leg —
218 * so log it and carry on rather than failing the transfer. */
219 if (bind(dfd, (struct sockaddr *) &local, sizeof(local)) != 0) {
220 ngx_log_error(NGX_LOG_INFO, fc->c->log, ngx_errno,
221 "brix: gsiftp(ev) data-leg source bind failed; "
222 "letting the kernel choose");
223 }
224 }
225
226 dc->dconn = brix_ftp_ev_wrap_conn(fc, dfd);
227 if (dc->dconn == NULL) {
228 (void) close(dfd);
229 return NGX_ERROR;
230 }
231 dc->dconn->data = dc;
232 dc->dconn->write->handler = ev_connect_handler;
233 dc->connecting = 1;
234 dc->tls_client = 1; /* we dialled out: TLS client role */
235
236 rc = connect(dfd, (struct sockaddr *) &fc->active_sa,
237 sizeof(fc->active_sa));
238 if (rc == -1 && ngx_socket_errno != NGX_EINPROGRESS) {
239 ngx_log_error(NGX_LOG_ERR, fc->c->log, ngx_socket_errno,
240 "brix: GridFTP(ev) active data connect() failed");
241 ngx_close_connection(dc->dconn);
242 dc->dconn = NULL;
243 return NGX_ERROR;
244 }
245 if (ngx_handle_write_event(dc->dconn->write, 0) != NGX_OK) {
246 ngx_close_connection(dc->dconn);
247 dc->dconn = NULL;
248 return NGX_ERROR;
249 }
250 /* Whether connect() returned 0 (immediate, loopback) or EINPROGRESS, let
251 * the write event drive ev_connect_handler — running the hand-off out of
252 * this stack keeps us from re-entering the control loop synchronously. */
253 ngx_add_timer(dc->dconn->write, BRIX_FTP_EV_IO_TIMEO);
254 return NGX_OK;
255}

258/* Bring the data connection up under the event loop; NGX_OK once the relevant
259 * event is armed (completion is asynchronous), NGX_ERROR if the socket could
260 * not be set up. The arm is one-shot: ev_data_open_active consumes fc->active

262ngx_int_t
263brix_ftp_ev_data_open(ftp_ev_dc_t *dc)

265 return dc->fc->active ? ev_data_open_active(dc) : ev_data_open_passive(dc);

269/* ---- teardown -------------------------------------------------------------- */
270
271/* Finish a transfer: release the VFS side, close the data socket and any pending
272 * listener, emit the control-channel result (226 / 550), and resume the control
273 * state machine. Idempotent-safe: the pump only calls it once per transfer. */
274void
275brix_ftp_ev_data_finish(ftp_ev_dc_t *dc, ngx_int_t rc)
276{
277 ftp_ev_t *fc = dc->fc;
278 int committed = (rc == NGX_OK);
279
280 if (dc->eb_conns != NULL) {
281 brix_ftp_ev_eb_teardown(dc); /* close live MODE E streams */
282 }
283 if (dc->fh != NULL) {
284 (void) brix_vfs_close(dc->fh, fc->c->log);
285 dc->fh = NULL;
286 }
287 if (dc->dh != NULL) {
288 (void) brix_vfs_closedir(dc->dh, fc->c->log);
289 dc->dh = NULL;
290 }
291 if (dc->writer != NULL) {
292 /* A writer still open here means the transfer failed before commit. */
293 brix_vfs_writer_abort(dc->writer);
294 dc->writer = NULL;
295 committed = 0; /* nothing was published */
296 }
297
298 /* phase-97 §5: this is the single completion for every transfer shape
299 * (stream, MODE E, active, passive) and it runs on the event loop, so one
300 * report here covers a committed STOR/APPE however it was carried. Only for
301 * a transfer that both succeeded and got past the commit — an aborted writer
302 * left the path untouched. */
303 if (committed && dc->writing) {
304 brix_ftp_ev_cns_note_stored(fc, dc->abs);
305 }
306
307 /* The one unified-metrics record per transfer, for the same reason: every
308 * shape lands here exactly once, with the offsets still intact. */
309 brix_ftp_ev_metric_xfer(dc, rc);
310
311 if (dc->dconn != NULL) {
312 if (dc->dconn->ssl != NULL) {
313 /* Quiet, non-blocking teardown: free the SSL synchronously (no
314 * close_notify round trip) before closing the socket. */
315 dc->dconn->ssl->no_wait_shutdown = 1;
316 dc->dconn->ssl->no_send_shutdown = 1;
317 (void) ngx_ssl_shutdown(dc->dconn);
318 }
319 ngx_close_connection(dc->dconn);
320 dc->dconn = NULL;
321 }
322 if (dc->lc != NULL) {
323 ngx_close_connection(dc->lc); /* also closes fc->pasv_fd */
324 fc->pasv_fd = -1;
325 dc->lc = NULL;
326 }
327 if (dc->dpool != NULL) {
328 ngx_destroy_pool(dc->dpool); /* the TLS conn's pool */
329 dc->dpool = NULL;
330 }
331
332 fc->dc = NULL;
333 fc->state = FTP_EV_ST_CMD;
334
335 /* phase-107 C7: a failure site that knows a more precise RFC-959 class
336 * (the lock gate's EBUSY → 450 "file busy") stamps dc->fail_reply; the
337 * generic 550 stays the default for everything else. */
338 (void) brix_ftp_ev_reply(fc, (rc == NGX_OK)
339 ? "226 Transfer complete\r\n"
340 : (dc->fail_reply != NULL)
341 ? dc->fail_reply
342 : "550 Transfer failed\r\n");
343
344 /* Resume the control channel: flush the result and frame the next command. */
345 brix_ftp_ev_resume(fc);
346}

349/* brix_ftp_ev_send_drain — shared send-side pump head for the data-channel
350 * write handlers (stream RETR and MODE E RETR): handle the idle timeout,
351 * clear a pending timer, then push dc->buf[buf_pos..buf_len) to the socket.
352 * NGX_OK: buffer fully drained — the caller refills and calls again.
353 * NGX_AGAIN: backpressure — write event re-armed + IO timer set, caller
354 * returns. NGX_ERROR: the transfer was already finished with an error
355 * (send failure, peer EOF, or event-arm failure), caller returns. */
356ngx_int_t
357brix_ftp_ev_send_drain(ngx_event_t *wev)
358{
359 ngx_connection_t *c = wev->data;
360 ftp_ev_dc_t *dc = c->data;
361
362 if (wev->timedout) {
363 brix_ftp_ev_data_finish(dc, NGX_ERROR);
364 return NGX_ERROR;
365 }
366 if (wev->timer_set) {
367 ngx_del_timer(wev);
368 }
369
370 while (dc->buf_pos < dc->buf_len) {
371 ssize_t n = c->send(c, dc->buf + dc->buf_pos,
372 dc->buf_len - dc->buf_pos);
373
374 if (n > 0) {
375 dc->buf_pos += (size_t) n;
376 continue;
377 }
378 if (n == NGX_AGAIN) {
379 if (ngx_handle_write_event(wev, 0) != NGX_OK) {
380 brix_ftp_ev_data_finish(dc, NGX_ERROR);
381 return NGX_ERROR;
382 }
383 ngx_add_timer(wev, BRIX_FTP_EV_IO_TIMEO);
384 return NGX_AGAIN;
385 }
386 brix_ftp_ev_data_finish(dc, NGX_ERROR); /* NGX_ERROR or peer EOF */
387 return NGX_ERROR;
388 }
389
390 return NGX_OK;
391}
