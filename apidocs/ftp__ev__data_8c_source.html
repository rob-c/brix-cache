15/*
16 * ftp_ev_data.c — non-blocking data-channel lifecycle for the event engine.
17 *
18 * WHAT: the event-driven bring-up of a data connection (accept a passive client,
19 * or connect out in active mode) and the idempotent transfer teardown. The
20 * verbs that nominate the endpoint live in ftp_ev_data_setup.c.
21 *
22 * WHY: the sync engine blocks in accept()/connect() and then in the read/write
23 * loop, monopolising the worker for one transfer. Here the listener and the
24 * connect socket are wrapped in nginx connections and driven by their own
25 * read/write events, so the worker keeps serving other sessions while a transfer
26 * proceeds.
27 *
28 * HOW: brix_ftp_ev_data_open() is called once a transfer verb has sent its 150:
29 * it wraps the listener PASV/EPSV left in fc->pasv_fd (passive) or a fresh
30 * connect() socket aimed at fc->active_sa (active) in an ngx_connection_t and
31 * arms the accept/connect event. When the data socket is ready the handler
32 * hands off to brix_ftp_ev_data_ready() (ftp_ev_xfer.c), which opens the VFS
33 * side and starts the pump. Every transfer, whatever its shape, ends in
34 * brix_ftp_ev_data_finish() — the single place the CNS note and the
35 * {proto="gridftp"} metrics record are emitted.
36 */

65/* ---- Admit a passive data socket to the control session ----
66 *
67 * WHAT: Returns NGX_OK when a protected socket will undergo the GSI identity
68 * handshake or a clear socket comes from the control transport peer; returns
69 * NGX_DECLINED for every other socket or when the peer cannot be established.
70 *
71 * WHY: possession of an advertised passive port is not authentication. This
72 * is the shared boundary used by ordinary and MODE E accepts before either can
73 * open the VFS side or attach transfer authority to the new descriptor.
74 *
75 * HOW: 1. Let PROT P continue only to its mandatory TLS/DN verification.
76 * 2. Reject unknown protection modes.
77 * 3. Read the kernel control peer and compare canonical host addresses.
78 */
79ngx_int_t
80brix_ftp_ev_passive_peer_admit(ftp_ev_t *fc, const struct sockaddr *peer,
81 socklen_t peer_len)
82{
83 struct sockaddr_storage control_peer;
84 socklen_t control_peer_len = sizeof(control_peer);
85
86 if (fc->prot == 'P') {
87 return NGX_OK;
88 }
89 if (fc->prot != 'C') {
90 return NGX_DECLINED;
91 }
92
93 ngx_memzero(&control_peer, sizeof(control_peer));
94 if (getpeername(fc->c->fd, (struct sockaddr *) &control_peer,
95 &control_peer_len) != 0)
96 {
97 return NGX_DECLINED;
98 }
99 return brix_ftp_peer_addr_equal(
100 (const struct sockaddr *) &control_peer, control_peer_len,
101 peer, peer_len)
102 ? NGX_OK : NGX_DECLINED;
103}

106/* ---- Accept the ordinary passive data peer ----
107 *
108 * WHAT: Accepts the first admissible passive data connection, retires the
109 * one-shot listener, and starts TLS or the transfer pump; timeouts and fatal
110 * socket errors fail the transfer.
111 *
112 * WHY: an off-peer connector must be discarded without consuming the victim's
113 * listener, while a bounded accept batch prevents a rejection flood from
114 * monopolizing the worker event loop.
115 *
116 * HOW: 1. Accept at most one listener-backlog-sized batch and capture peers.
117 * 2. Close rejected peers before any transfer/VFS action and keep waiting.
118 * 3. Re-post a saturated batch, or retire the listener for an admitted fd.
119 * 4. Make the fd nonblocking and enter TLS or the clear transfer pump.
120 */
121static void
122ev_accept_handler(ngx_event_t *rev)
123{
124 ngx_connection_t *lc = rev->data;
125 ftp_ev_dc_t *dc = lc->data;
126 ftp_ev_t *fc = dc->fc;
127 struct sockaddr_storage peer;
128 socklen_t peer_len;
129 ngx_uint_t attempts;
130 int dfd = -1;
131
132 if (rev->timedout) {
133 BRIX_LOG(NGX_LOG_INFO, fc->c->log, NGX_ETIMEDOUT,
134 "brix: GridFTP(ev) passive data accept timeout");
135 brix_ftp_ev_data_finish(dc, NGX_ERROR);
136 return;
137 }
138
139 for (attempts = 0; attempts < BRIX_FTP_EV_DATA_BACKLOG; attempts++) {
140 ngx_memzero(&peer, sizeof(peer));
141 peer_len = sizeof(peer);
142 dfd = accept(lc->fd, (struct sockaddr *) &peer, &peer_len);
143 if (dfd < 0) {
144 if (ngx_socket_errno == NGX_EAGAIN) {
145 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
146 brix_ftp_ev_data_finish(dc, NGX_ERROR);
147 }
148 return; /* queue drained — wait */
149 }
150 BRIX_LOG(NGX_LOG_ERR, fc->c->log, ngx_socket_errno,
151 "brix: GridFTP(ev) passive accept failed");
152 brix_ftp_ev_data_finish(dc, NGX_ERROR);
153 return;
154 }
155
156 if (brix_ftp_ev_passive_peer_admit(fc,
157 (const struct sockaddr *) &peer, peer_len) == NGX_OK)
158 {
159 break;
160 }
161 BRIX_LOG(NGX_LOG_WARN, fc->c->log, 0,
162 "brix: GridFTP(ev) rejected passive data peer not "
163 "bound to control peer");
164 (void) close(dfd);
165 dfd = -1;
166 }
167
168 if (dfd < 0) {
169 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
170 brix_ftp_ev_data_finish(dc, NGX_ERROR);
171 return;
172 }
173 ngx_post_event(rev, &ngx_posted_events);
174 return;
175 }
176
177 /* Retire the listener (closes fc->pasv_fd) — one transfer, one accept. */
178 if (rev->timer_set) { ngx_del_timer(rev); }
179 ngx_close_connection(lc);
180 fc->pasv_fd = -1;
181 dc->lc = NULL;
182
183 if (ngx_nonblocking(dfd) == -1) {
184 (void) close(dfd);
185 brix_ftp_ev_data_finish(dc, NGX_ERROR);
186 return;
187 }
188 dc->dconn = brix_ftp_ev_wrap_conn(fc, dfd);
189 if (dc->dconn == NULL) {
190 (void) close(dfd);
191 brix_ftp_ev_data_finish(dc, NGX_ERROR);
192 return;
193 }
194 dc->dconn->data = dc;
195 if (fc->prot == 'P') {
196 brix_ftp_ev_dc_start_tls(dc); /* GSI-secure the data channel */
197 } else {
198 brix_ftp_ev_data_ready(dc);
199 }
200}

242ev_data_open_passive(ftp_ev_dc_t *dc)
243{

274/* ev_data_open_active — PORT/EPRT: dial the armed, peer-pinned target from a
275 * fresh non-blocking socket sourced on the control channel's local IP. */
276static ngx_int_t
277ev_data_open_active(ftp_ev_dc_t *dc)
278{
279 ftp_ev_t *fc = dc->fc;
280 struct sockaddr_in local;
281 socklen_t llen = sizeof(local);
282 int dfd;
283 int rc;
284
285 fc->active = 0; /* one-shot: consume the arm */
286 ngx_memzero(&local, sizeof(local)); /* getsockname may leave it */
287
288 dfd = socket(AF_INET, SOCK_STREAM, 0);
289 if (dfd < 0) {
290 return NGX_ERROR;
291 }
292 if (ngx_nonblocking(dfd) == -1) {
293 (void) close(dfd);
294 return NGX_ERROR;
295 }
296 if (getsockname(fc->c->fd, (struct sockaddr *) &local, &llen) == 0
297 && local.sin_family == AF_INET)
298 {
299 local.sin_port = 0; /* ephemeral source port */
300 /* Best-effort source pinning: if the bind fails the kernel picks
301 * the source address itself, which is still a usable data leg —
302 * so log it and carry on rather than failing the transfer. */
303 if (bind(dfd, (struct sockaddr *) &local, sizeof(local)) != 0) {
304 BRIX_LOG(NGX_LOG_INFO, fc->c->log, ngx_errno,
305 "brix: gsiftp(ev) data-leg source bind failed; "
306 "letting the kernel choose");
307 }
308 }
309
310 dc->dconn = brix_ftp_ev_wrap_conn(fc, dfd);
311 if (dc->dconn == NULL) {
312 (void) close(dfd);
313 return NGX_ERROR;
314 }
315 dc->dconn->data = dc;
316 dc->dconn->write->handler = ev_connect_handler;
317 dc->connecting = 1;
318 dc->tls_client = 1; /* we dialled out: TLS client role */
319
320 rc = connect(dfd, (struct sockaddr *) &fc->active_sa,
321 sizeof(fc->active_sa));
322 if (rc == -1 && ngx_socket_errno != NGX_EINPROGRESS) {
323 BRIX_LOG(NGX_LOG_ERR, fc->c->log, ngx_socket_errno,
324 "brix: GridFTP(ev) active data connect() failed");
325 ngx_close_connection(dc->dconn);
326 dc->dconn = NULL;
327 return NGX_ERROR;
328 }
329 if (ngx_handle_write_event(dc->dconn->write, 0) != NGX_OK) {
330 ngx_close_connection(dc->dconn);
331 dc->dconn = NULL;
332 return NGX_ERROR;
333 }
334 /* Whether connect() returned 0 (immediate, loopback) or EINPROGRESS, let
335 * the write event drive ev_connect_handler — running the hand-off out of
336 * this stack keeps us from re-entering the control loop synchronously. */
337 ngx_add_timer(dc->dconn->write, BRIX_FTP_EV_IO_TIMEO);
338 return NGX_OK;
339}

353/* ---- teardown -------------------------------------------------------------- */
354
355/* Finish a transfer: release the VFS side, close the data socket and any pending
356 * listener, emit the control-channel result (226 / 550), and resume the control
357 * state machine. Idempotent-safe: the pump only calls it once per transfer. */
358void
359brix_ftp_ev_data_finish(ftp_ev_dc_t *dc, ngx_int_t rc)
360{
361 ftp_ev_t *fc = dc->fc;
362 int committed = (rc == NGX_OK);
363
364 if (dc->eb_conns != NULL) {
365 brix_ftp_ev_eb_teardown(dc); /* close live MODE E streams */
366 }
367 if (dc->fh != NULL) {
368 (void) brix_vfs_close(dc->fh, fc->c->log);
369 dc->fh = NULL;
370 }
371 if (dc->dh != NULL) {
372 (void) brix_vfs_closedir(dc->dh, fc->c->log);
373 dc->dh = NULL;
374 }
375 if (dc->writer != NULL) {
376 /* A writer still open here means the transfer failed before commit. */
377 brix_vfs_writer_abort(dc->writer);
378 dc->writer = NULL;
379 committed = 0; /* nothing was published */
380 }
381
382 /* phase-97 §5: this is the single completion for every transfer shape
383 * (stream, MODE E, active, passive) and it runs on the event loop, so one
384 * report here covers a committed STOR/APPE however it was carried. Only for
385 * a transfer that both succeeded and got past the commit — an aborted writer
386 * left the path untouched. */
387 if (committed && dc->writing) {
388 brix_ftp_ev_cns_note_stored(fc, dc->abs);
389 }
390
391 /* The one unified-metrics record per transfer, for the same reason: every
392 * shape lands here exactly once, with the offsets still intact. */
393 brix_ftp_ev_metric_xfer(dc, rc);
394
395 if (dc->dconn != NULL) {
396 if (dc->dconn->ssl != NULL) {
397 /* Quiet, non-blocking teardown: free the SSL synchronously (no
398 * close_notify round trip) before closing the socket. */
399 dc->dconn->ssl->no_wait_shutdown = 1;
400 dc->dconn->ssl->no_send_shutdown = 1;
401 (void) ngx_ssl_shutdown(dc->dconn);
402 }
403 ngx_close_connection(dc->dconn);
404 dc->dconn = NULL;
405 }
406 if (dc->lc != NULL) {
407 ngx_close_connection(dc->lc); /* also closes fc->pasv_fd */
408 fc->pasv_fd = -1;
409 dc->lc = NULL;
410 }
411 if (dc->dpool != NULL) {
412 ngx_destroy_pool(dc->dpool); /* the TLS conn's pool */
413 dc->dpool = NULL;
414 }
415
416 fc->dc = NULL;
417 fc->state = FTP_EV_ST_CMD;
418
419 /* phase-107 C7: a failure site that knows a more precise RFC-959 class
420 * (the lock gate's EBUSY → 450 "file busy") stamps dc->fail_reply; the
421 * generic 550 stays the default for everything else. */
422 (void) brix_ftp_ev_reply(fc, (rc == NGX_OK)
423 ? "226 Transfer complete\r\n"
424 : (dc->fail_reply != NULL)
425 ? dc->fail_reply
426 : "550 Transfer failed\r\n");
427
428 /* Resume the control channel: flush the result and frame the next command. */
429 brix_ftp_ev_resume(fc);
430}

433/* brix_ftp_ev_send_drain — shared send-side pump head for the data-channel
434 * write handlers (stream RETR and MODE E RETR): handle the idle timeout,
435 * clear a pending timer, then push dc->buf[buf_pos..buf_len) to the socket.
436 * NGX_OK: buffer fully drained — the caller refills and calls again.
437 * NGX_AGAIN: backpressure — write event re-armed + IO timer set, caller
438 * returns. NGX_ERROR: the transfer was already finished with an error
439 * (send failure, peer EOF, or event-arm failure), caller returns. */
440ngx_int_t
441brix_ftp_ev_send_drain(ngx_event_t *wev)
442{
443 ngx_connection_t *c = wev->data;
444 ftp_ev_dc_t *dc = c->data;
445
446 if (wev->timedout) {
447 brix_ftp_ev_data_finish(dc, NGX_ERROR);
448 return NGX_ERROR;
449 }
450 if (wev->timer_set) {
451 ngx_del_timer(wev);
452 }
453
454 while (dc->buf_pos < dc->buf_len) {
455 ssize_t n = c->send(c, dc->buf + dc->buf_pos,
456 dc->buf_len - dc->buf_pos);
457
458 if (n > 0) {
459 dc->buf_pos += (size_t) n;
460 continue;
461 }
462 if (n == NGX_AGAIN) {
463 if (ngx_handle_write_event(wev, 0) != NGX_OK) {
464 brix_ftp_ev_data_finish(dc, NGX_ERROR);
465 return NGX_ERROR;
466 }
467 ngx_add_timer(wev, BRIX_FTP_EV_IO_TIMEO);
468 return NGX_AGAIN;
469 }
470 brix_ftp_ev_data_finish(dc, NGX_ERROR); /* NGX_ERROR or peer EOF */
471 return NGX_ERROR;
472 }
473
474 return NGX_OK;
475}
