1/* brix_fault_pump_dir.c — the per-direction transform chain and pump step.
2 *
3 * WHAT: One direction's journey from read() to write(): the ping-pong buffer
4 * chain, the payload rewrite/inject and bit-mutation stages, the HTTP 100-
5 * continue eater, the TLS ClientHello reset trigger, the DPI/middlebox
6 * stall levers, and relay_pump_dir() which drives them for one poll hit.
7 *
8 * WHY: Split out of brix_fault_pump.c on the 600-line file cap
9 * (coding-standards §1). The transform chain is one concept — it never
10 * touches poll state, timers or teardown; the sibling TU owns the loop
11 * that calls it once per readable direction.
12 *
13 * HOW: Same behaviour as before the split — this is a pure move. The shared
14 * state and relay_pump_dir's prototype stayed in brix_fault_proxy_state.h,
15 * so the seam needed no new header. Levers are read lock-free; wide

32/* One direction's bytes as they travel the transform chain. `cur`/`n` are the
33 * current payload; `dst`/`alt` are the two ping-pong scratch buffers, so a
34 * length-changing stage never overwrites its own input. */
35typedef struct {
36 const char *cur;
37 ssize_t n;
38 unsigned char *dst;
39 unsigned char *alt;
40} pump_chain;
41
42/* Adopt the `n` bytes a stage just wrote to `dst` as the new payload, and flip
43 * the scratch buffers so the next stage writes to the other one. */
44static void
45chain_take(pump_chain *ch, size_t n)
46{
47 ch->n = (ssize_t) n;
48 ch->cur = (const char *) ch->dst;
49 FP_PSWAP(ch->dst, ch->alt);
50}

52/* The protocol-aware rewriters: TLS record surgery, then HTTP request
53 * smuggling. Both may change the payload length. */
54static void
55pump_rewrite(int is_up, pump_chain *ch)
56{
57 if (fp_tls_active(is_up ? &g_tls_up : &g_tls_down)) {
58 chain_take(ch, apply_tls(is_up, ch->cur, ch->n, ch->dst));
59 }
60 if (fp_http_active(is_up ? &g_http_up : &g_http_down)) {
61 int applied = 0;
62 size_t on = apply_http(is_up, ch->cur, ch->n, ch->dst, &applied);
63 if (applied) {
64 chain_take(ch, on);
65 }
66 }
67}

69/* Byte-level MITM mutation (drop/repeat/inject/replace). Returns 1 when every
70 * byte was dropped, i.e. there is nothing left to forward. */
71static int
72pump_mutate(int is_up, volatile lever_t *L, pump_chain *ch, unsigned *seed)
73{
74 fp_ext_mut mut;
75 unsigned char fbuf[128], rbuf[256], ibuf[512];
76
77 if (!ext_snapshot(is_up, L, &mut, fbuf, rbuf, ibuf)) {
78 return 0;
79 }
80 fp_ext_stats st = { 0, 0, 0, 0 };
81 size_t on = fp_ext_mutate((const unsigned char *) ch->cur, (size_t) ch->n,
82 ch->dst, FP_SCRATCH, &mut, seed, &st);
83 chain_take(ch, on);
84 if (st.dropped) { CBUMP(dropped, st.dropped); }
85 if (st.repeated) { CBUMP(repeated, st.repeated); }
86 if (st.injected) { CBUMP(injected, st.injected); }
87 if (st.replaced) { CBUMP(replaced, st.replaced); }
88 return ch->n == 0;
89}

91/* eat-100-continue: on the down path, splice out a leading "HTTP/1.x 100 ...
92 * CRLFCRLF" interim response so an Expect: 100-continue upload hangs waiting for
93 * a 100 the middlebox ate. Returns 1 when a 100 was removed (ch flipped). */
94static int
95pump_eat_100(pump_chain *ch)
96{
97 const unsigned char *in = (const unsigned char *) ch->cur;
98 size_t n = (size_t) ch->n;
99 /* "HTTP/1.x 100": "HTTP/1." at [0..6], minor version at [7], SP at [8],
100 * status "100" at [9..11]. */
101 if (n < 12 || memcmp(in, "HTTP/1.", 7) != 0 || in[8] != ' ' ||
102 memcmp(in + 9, "100", 3) != 0) {
103 return 0;
104 }
105 for (size_t i = 0; i + 3 < n; i++) {
106 if (in[i] == '\r' && in[i + 1] == '\n' &&
107 in[i + 2] == '\r' && in[i + 3] == '\n') {
108 size_t drop = i + 4;
109 memcpy(ch->dst, in + drop, n - drop);
110 chain_take(ch, n - drop);
111 return 1;
112 }
113 }
114 return 0;
115}

117/* hello-split-reset: true when `buf` opens a TLS handshake ClientHello whose
118 * DECLARED record length is >= `thresh` — caught from the record header even
119 * when the ClientHello is split across TCP segments (post-quantum keyshares,
120 * fat SNI/cert lists blow past one MSS and a first-segment-only DPI resets). */
121static int
122tls_hello_oversized(const char *buf, ssize_t nr, int thresh)
123{
124 const unsigned char *b = (const unsigned char *) buf;
125 if (thresh <= 0 || nr < 6) {
126 return 0;
127 }
128 if (b[0] != 0x16 || b[5] != 0x01) { /* handshake record, ClientHello */
129 return 0;
130 }
131 int rec_len = (b[3] << 8) | b[4]; /* TLSPlaintext.length (BE16) */
132 return rec_len >= thresh;
133}
134
135/* hello-split-reset: an oversized TLS ClientHello on the request path is RST by
136 * a DPI that only inspects the first segment. Returns 2 (severed + CDEC) or 0. */
137static int
138pump_hello_reset(int is_up, const char *buf, ssize_t nr, int cfd, int ufd)
139{
140 if (is_up && tls_hello_oversized(buf, nr, g_hello_reset_thresh)) {
141 CBUMP(hello_reset, 1);
142 sever(cfd, ufd, 1);
143 CDEC(active);
144 return 2;
145 }
146 return 0;
147}

174/* One store-and-forward hold lever (header-size or body-size): if armed and
175 * triggered, forward the withheld prefix, sleep `hold_ms`, then trim `ch` to
176 * the remainder. Returns 2 if a forward error severed the pair (already closed
177 * + CDEC), 1 if the hold fired, 0 if it did not apply. */
178static int
179pump_apply_hold(fp_http_cfg *HC, int (*active)(const fp_http_cfg *),
180 int (*decide)(const fp_http_cfg *, const unsigned char *, size_t, size_t *),
181 int hold_ms, pump_chain *ch, int to, int cfd, int ufd,
182 unsigned epoch, volatile lever_t *L, unsigned *seed,
183 unsigned long *conn_ctr, unsigned long *glob_ctr)
184{
185 size_t rel = 0;
186 if (!active(HC) ||
187 !decide(HC, (const unsigned char *) ch->cur, (size_t) ch->n, &rel)) {
188 return 0;
189 }
190 if (rel > 0 &&
191 forward_faulted(to, (char *) ch->cur, (ssize_t) rel, epoch, L,
192 seed, conn_ctr, glob_ctr) != 0) {
193 sever(cfd, ufd, g_abortive);
194 CDEC(active);
195 return 2;
196 }
197 usleep((useconds_t) hold_ms * 1000);
198 CBUMP(held, 1);
199 ch->cur += rel;
200 ch->n -= (ssize_t) rel; /* forward the withheld remainder below */
201 return 1;
202}

204/* DPI stall levers applied just before forwarding: HTTP header-size hold, body
205 * hold (store-and-forward), and volume classify-throttle. Each hold forwards
206 * any withheld prefix itself and trims `ch` to the remainder. Returns 2 if a
207 * forward error severed the pair (already closed + CDEC), else 0. */
208static int
209pump_dpi_stalls(int is_up, pump_chain *ch, int to, int cfd, int ufd,
210 unsigned epoch, volatile lever_t *L, unsigned *seed,
211 unsigned long *conn_ctr, unsigned long *glob_ctr)
212{
213 fp_http_cfg *HC = is_up ? &g_http_up : &g_http_down;
214 if (pump_apply_hold(HC, fp_http_hold_active, fp_http_hold_decide,
215 HC->hold_ms, ch, to, cfd, ufd, epoch, L, seed,
216 conn_ctr, glob_ctr) == 2) {
217 return 2;
218 }
219 if (pump_apply_hold(HC, fp_http_body_hold_active, fp_http_body_hold_decide,
220 HC->body_hold_ms, ch, to, cfd, ufd, epoch, L, seed,
221 conn_ctr, glob_ctr) == 2) {
222 return 2;
223 }
224 /* classify-throttle: once a direction crosses the volume heuristic the flow
225 * is shunted to a <kbps> slow lane (misclassified bulk = "exfiltration"). */
226 if (g_classify_kbps > 0 && g_classify_bytes > 0 && ch->n > 0 &&
227 *conn_ctr + (unsigned long) ch->n > (unsigned long) g_classify_bytes) {
228 long us = (long) ((double) ch->n * 1000000.0 /
229 ((double) g_classify_kbps * 1024.0));
230 CBUMP(throttled, 1);
231 if (us > 0) {
232 usleep((useconds_t) us);
233 }
234 }
235 return 0;
236}

238/* Relay one poll-ready direction through the fault engine. Returns 0 to keep
239 * looping, 1 on EOF/read error (caller closes both ends), 2 if a fault severed
240 * the pair (already closed + CDEC, caller just returns). */
241int
242relay_pump_dir(int i, struct pollfd *pfd, int cfd, int ufd,
243 char *buf, size_t bufsz, unsigned char *scratch,
244 unsigned char *scratch2, unsigned epoch,
245 unsigned *seed, unsigned long *up_ctr, unsigned long *down_ctr,
246 int *firstflag)
247{
248 if (!(pfd[i].revents & (POLLIN | POLLHUP | POLLERR))) {
249 return 0;
250 }
251 int from = pfd[i].fd;
252 int to = pfd[i ^ 1].fd;
253 ssize_t nr = read(from, buf, bufsz);
254 if (nr <= 0) {
255 return 1;
256 }
257 int is_up = (i == 0);
258 t_fwd_up = is_up; /* so a deep truncate event can name its direction */
259 volatile lever_t *L = is_up ? &g_up : &g_down;
260 unsigned long *conn_ctr = is_up ? up_ctr : down_ctr;
261
262 /* toxicity: an unafflicted direction gets a clean-lever pass-through, so
263 * armed byte levers (corrupt/latency/…) do not fire on this connection. */
264 static const lever_t g_clean_lever;
265 if (!(is_up ? t_afflict_up : t_afflict_down)) {
266 L = (volatile lever_t *) &g_clean_lever;
267 }
268
269 if (pump_hello_reset(is_up, buf, nr, cfd, ufd) == 2) {
270 return 2;
271 }
272
273 /* Stateful, content-addressed faults on the RAW bytes (before any mutation
274 * changes their length/offset). */
275 trig_check(is_up, buf, nr);
276 mangle_apply(is_up, buf, nr, *conn_ctr);
277
278 pump_delay_first(L, firstflag);
279
280 /* Transform chain: each length-changing stage ping-pongs between scratch A
281 * and B so it never overwrites its own input — TLS record surgery, then HTTP
282 * request smuggling, then byte-level MITM mutation. */
283 pump_chain ch = { buf, nr, scratch, scratch2 };
284
285 pump_rewrite(is_up, &ch);
286 if (pump_mutate(is_up, L, &ch, seed)) {
287 return 0; /* every byte was dropped — nothing to forward */
288 }
289
290 if (pump_eat_100_dir(is_up, &ch)) {
291 return 0; /* nothing left after the 100 was removed */
292 }
293
294 /* Session recording: capture exactly what we are about to forward. */
295 if (fp_replay_recording()) {
296 fp_replay_record(is_up, now_ms_since(g_t0),
297 (const unsigned char *) ch.cur, (size_t) ch.n);
298 CBUMP(recorded, (unsigned long) ch.n);
299 }
300
301 unsigned long *glob_ctr = is_up ? &C.up_bytes : &C.down_bytes;
302
303 /* DPI header/body-size holds and classify-throttle — a complete,
304 * over-threshold header block (e.g. a fat client-cert PEM in an XrdHttp
305 * request) stalls, as does bulk that crosses the volume heuristic. */
306 if (pump_dpi_stalls(is_up, &ch, to, cfd, ufd, epoch, L, seed,
307 conn_ctr, glob_ctr) == 2) {
308 return 2;
309 }
310
311 if (forward_faulted(to, (char *) ch.cur, ch.n, epoch, L, seed,
312 conn_ctr, glob_ctr) != 0) {
313 sever(cfd, ufd, g_abortive);
314 CDEC(active);
315 return 2;
316 }
317 return 0;
318}
