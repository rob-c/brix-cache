28/* brix_ensure_payload_buffer: ensure payload_buf holds dlen (+1 NUL) bytes at
29 * request start (free-then-alloc on resize — the buffer is empty here). */
30static ngx_int_t
31brix_ensure_payload_buffer(brix_ctx_t *ctx, ngx_connection_t *c,
32 uint32_t dlen)
33{
34 u_char *buf;
35 size_t need;
36
37 if (dlen > (uint32_t) (SIZE_MAX - 1)) {
38 return NGX_ERROR;
39 }
40 need = (size_t) dlen + 1;
41
42 if (ctx->recv.payload_buf != NULL && ctx->recv.payload_buf_size >= need) {
43 ctx->recv.payload = ctx->recv.payload_buf;
44 ctx->recv.payload[dlen] = '\0';
45 return NGX_OK;
46 }
47
48 buf = ngx_alloc(need, c->log);
49 if (buf == NULL) {
50 return NGX_ERROR;
51 }
52
53 if (ctx->recv.payload_buf != NULL) {
54 ngx_free(ctx->recv.payload_buf);
55 }
56
57 ctx->recv.payload_buf = buf;
58 ctx->recv.payload_buf_size = need;
59 ctx->recv.payload = buf;
60 ctx->recv.payload[dlen] = '\0';
61
62 return NGX_OK;
63}

104/*
105 * Phase 29 drain barrier condition: a non-read/write opcode arriving while
106 * reads (out.count) or writes (wr_inflight) are still in flight must run with
107 * the connection quiescent (a kXR_close could free a handle an in-flight
108 * sendfile chain or pwrite still references). kXR_read and kXR_write both
109 * pipeline and are never deferred.
110 *
111 * phase-32 WS3: a cold kXR_read now stays in flight on a worker thread
112 * (rd.aio_inflight) with recv still receiving — a following kXR_close would
113 * otherwise retire the very handle that read's pread is using. Defer on
114 * rd.aio_inflight too so the barrier waits for the read worker to finish.
115 */
116static ngx_flag_t
117brix_recv_should_defer(brix_ctx_t *ctx)
118{
119 return (ctx->out.count > 0 || ctx->out.wr_inflight > 0
120 || ctx->rd.aio_inflight > 0)
121 && ctx->recv.cur_reqid != kXR_read
122 && ctx->recv.cur_reqid != kXR_write;
123}

125/*
126 * Phase 29 pipelining condition for a parked cleartext sendfile kXR_read: keep
127 * reading so the next read's sendfile span queues behind this one while the
128 * prior response drains. All four conjuncts are load-bearing: resp_pipelinable
129 * = a single self-contained sendfile span (safe to queue another behind);
130 * !rd.win_active excludes a multi-window read still streaming out of the shared
131 * read_scratch (which must stay serial); out.count < depth is the in-flight cap.
132 */
133static ngx_flag_t
134brix_recv_try_pipeline_read(brix_ctx_t *ctx)
135{
136 return ctx->state == XRD_ST_SENDING
137 && ctx->recv.cur_reqid == kXR_read
138 && ctx->out.resp_pipelinable
139 && !ctx->rd.win_active
140 && ctx->out.count < ctx->out.pipeline_depth;
141}

143/*
144 * kXR_writev / kXR_chkpoint stock wire framing: the header dlen covers only the
145 * descriptor block (writev) or the embedded 24-byte sub-header (chkpoint/ckpXeq);
146 * the trailing segment/sub-request data streams after the frame. Just after
147 * that first block lands, raise the read obligation by the trailing length so
148 * descriptors + data land contiguously before dispatch. A malformed block is
149 * deliberately NOT rejected here — dispatch runs the login/auth/write gates
150 * first and the handler emits the stock-parity error and drops the link. Both
151 * stages are bounded (BRIX_MAX_WRITE_PAYLOAD), so the grow is safe. CONTINUE to
152 * keep receiving the extended body, NEXT when no extension applies, BREAK on a
153 * grow failure.
154 */
155static brix_recv_step_t
156brix_recv_extend_body(brix_ctx_t *ctx, ngx_connection_t *c)
157{
158 if (ctx->recv.cur_reqid == kXR_writev && !ctx->recv.cur_body_extended) {
159 uint32_t extra;
160
161 ctx->recv.cur_body_extended = 2;
162
163 if (brix_writev_body_extra(ctx->recv.payload, ctx->recv.cur_dlen,
164 &extra) == NGX_OK
165 && extra > 0)
166 {
167 if (brix_grow_payload_buffer(ctx, c,
168 ctx->recv.cur_dlen + extra) != NGX_OK)
169 {
170 return BRIX_RECV_STEP_BREAK;
171 }
172 ctx->recv.cur_body_extra = extra;
173 return BRIX_RECV_STEP_CONTINUE; /* receive the streamed segments */
174 }
175 }
176
177 if (ctx->recv.cur_reqid == kXR_chkpoint
178 && ctx->recv.cur_body[15] == kXR_ckpXeq
179 && ctx->recv.cur_dlen == XRD_REQUEST_HDR_LEN
180 && ctx->recv.cur_body_extended < 2)
181 {
182 uint32_t extra = 0;
183 unsigned final = 1;
184
185 (void) brix_ckpxeq_body_extra(ctx->recv.payload,
186 ctx->recv.cur_dlen + ctx->recv.cur_body_extra,
187 &extra, &final);
188 ctx->recv.cur_body_extended = final ? 2 : 1;
189
190 if (extra > 0) {
191 if (brix_grow_payload_buffer(ctx, c,
192 ctx->recv.cur_dlen + ctx->recv.cur_body_extra + extra)
193 != NGX_OK)
194 {
195 return BRIX_RECV_STEP_BREAK;
196 }
197 ctx->recv.cur_body_extra += extra;
198 return BRIX_RECV_STEP_CONTINUE; /* receive the streamed sub-body */
199 }
200 }
201
202 return BRIX_RECV_STEP_NEXT;
203}

205/*
206 * Phase 94 kXR_write header handling on THIS connection, before the payload-size
207 * path. Returns 1 (with *step set) when it fully handled the frame, else 0 (fall
208 * through to the normal payload path). Two cases:
209 * - a bound secondary carrying a plain write reopens the primary's published
210 * writable handle in this worker BEFORE the streaming/buffered paths inspect
211 * file->fd (brix_write_stream_begin runs at the header phase); the payload
212 * follows inline on this connection, so it falls through.
213 * - a primary-connection non-zero-pathid write is header-only (its data rides a
214 * bound secondary, which instead carries the WHOLE write itself), a
215 * cross-connection data-path BriX does not service — refuse it cleanly WITHOUT
216 * reading cur_dlen bytes (there are none; consuming them would desync framing).
217 */
218static int
219brix_recv_write_hdr_hook(ngx_connection_t *c, brix_ctx_t *ctx,
220 size_t *rx_pending, brix_recv_step_t *step)
221{
222 ngx_int_t serr;
223
224 if (ctx->recv.cur_reqid != kXR_write) {
225 return 0;
226 }
227 if (ctx->is_bound) {
228 (void) brix_ensure_write_handle(ctx, c,
229 (int) (unsigned char) ctx->recv.cur_body[0]);
230 return 0;
231 }
232 if (ctx->recv.cur_body[12] == 0) { /* cur_body: fhandle[4] offset[8] pathid[1] */
233 return 0;
234 }
235 ctx->recv.cur_dlen = 0;
236 ctx->recv.payload = NULL;
237 BRIX_OP_ERR(ctx, BRIX_OP_WRITE);
238 serr = brix_send_error(ctx, c, kXR_Unsupported,
239 "write data-path (pathid) unsupported; send data inline (pathid 0)");
240 if (serr == NGX_ERROR) {
241 *step = BRIX_RECV_STEP_BREAK;
242 return 1;
243 }
244 ctx->out.resp_pipelinable = 0;
245 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
246 *rx_pending = 0;
247 if (ctx->state == XRD_ST_SENDING) {
248 *step = BRIX_RECV_STEP_RETURN;
249 return 1;
250 }
251 ctx->state = XRD_ST_REQ_HEADER;
252 ctx->recv.hdr_pos = 0;
253 *step = BRIX_RECV_STEP_CONTINUE;
254 return 1;
255}

257/*
258 * The post-dispatch tail of the dlen==0 header path: a freshly posted read-AIO
259 * either pipelines a cold read (keep receiving) or suspends; a cleartext sendfile
260 * read pipelines; a pending TLS handshake is handed off; otherwise return to the
261 * request-header state. CONTINUE / RETURN / BREAK.
262 */
263static brix_recv_step_t
264brix_recv_after_dispatch_hdr(ngx_connection_t *c, brix_ctx_t *ctx,
265 ngx_stream_brix_srv_conf_t *conf, ngx_event_t *rev)
266{
267 if (ctx->state == XRD_ST_AIO) {
268 if (ctx->recv.cur_reqid == kXR_read && ctx->rd.aio_inflight > 0
269 && !ctx->rd.win_active)
270 {
271 ctx->state = XRD_ST_REQ_HEADER;
272 ctx->recv.hdr_pos = 0;
273 return BRIX_RECV_STEP_CONTINUE;
274 }
275 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
276 return BRIX_RECV_STEP_BREAK;
277 }
278 return BRIX_RECV_STEP_RETURN;
279 }
280 if (brix_recv_try_pipeline_read(ctx)) {
281 ctx->state = XRD_ST_REQ_HEADER;
282 ctx->recv.hdr_pos = 0;
283 }
284 if (ctx->state != XRD_ST_SENDING) {
285 if (ctx->tls_pending) {
286 brix_start_tls(ctx, c, conf);
287 return BRIX_RECV_STEP_RETURN;
288 }
289 ctx->state = XRD_ST_REQ_HEADER;
290 ctx->recv.hdr_pos = 0;
291 }
292 return BRIX_RECV_STEP_CONTINUE;
293}

295/*
296 * A fully-received 24-byte ClientRequestHdr just landed: decode it, reject an
297 * oversized dlen before any allocation, and either accept a payload-bearing
298 * request (→ REQ_PAYLOAD) or dispatch a dlen==0 request. CONTINUE/RETURN/BREAK.
299 */
300static brix_recv_step_t
301brix_recv_after_header(ngx_stream_session_t *s, ngx_connection_t *c,
302 ngx_stream_brix_srv_conf_t *conf, brix_ctx_t *ctx, ngx_event_t *rev,
303 size_t *rx_pending)
304{
305 ClientRequestHdr *hdr = (ClientRequestHdr *) ctx->recv.hdr_buf;
306 uint32_t max_pl;
307 ngx_int_t rc;
308
309 /*
310 * Decode the fixed 24-byte ClientRequestHdr (wire layout: streamid[2] @0,
311 * requestid @2, body[16] @4, dlen @20). streamid is an opaque client tag
312 * echoed verbatim — copied byte-for-byte, never byte-swapped. requestid and
313 * dlen are big-endian on the wire, so ntohs/ntohl to host order. cur_body is
314 * the raw 16-byte opcode argument block, handed to dispatch untouched.
315 */
316 ctx->recv.cur_streamid[0] = hdr->streamid[0];
317 ctx->recv.cur_streamid[1] = hdr->streamid[1];
318 ctx->recv.cur_reqid = ntohs(hdr->requestid);
319 ngx_memcpy(ctx->recv.cur_body, hdr->body, 16);
320 ctx->recv.cur_dlen = (uint32_t) ntohl(hdr->dlen);
321 ctx->recv.cur_body_extra = 0;
322 ctx->recv.cur_body_extended = 0;
323 BRIX_SRV_METRIC_INC(ctx, request_frames_total);
324 BRIX_SRV_METRIC_ADD(ctx, request_payload_bytes_total, ctx->recv.cur_dlen);
325
326 ngx_log_debug(NGX_LOG_DEBUG_STREAM, c->log, 0,
327 "brix: req sid=[%02xd%02xd] reqid=%04xd dlen=%uz"
328 " avail=%d ready=%d",
329 (int) ctx->recv.cur_streamid[0],
330 (int) ctx->recv.cur_streamid[1],
331 (int) ctx->recv.cur_reqid, (size_t) ctx->recv.cur_dlen,
332 c->read->available, (int) c->read->ready);
333
334 /*
335 * Phase 94 kXR_write header handling: bound-secondary write-handle reopen, and
336 * clean refusal of a header-only non-zero-pathid write on the primary. Returns
337 * 1 when it fully handled the frame (step set); otherwise falls through.
338 */
339 {
340 brix_recv_step_t wstep;
341 if (brix_recv_write_hdr_hook(c, ctx, rx_pending, &wstep)) {
342 return wstep;
343 }
344 }
345
346 /* dlen is untrusted client input — reject before any allocation. */
347 max_pl = brix_max_payload_for_request(ctx->recv.cur_reqid);
348 if (ctx->recv.cur_dlen > max_pl) {
349 ngx_log_error(NGX_LOG_WARN, c->log, 0,
350 "brix: payload too large (%uz), closing",
351 (size_t) ctx->recv.cur_dlen);
352 BRIX_SRV_METRIC_INC(ctx, oversized_payloads_total);
353 return BRIX_RECV_STEP_BREAK;
354 }
355
356 /*
357 * Streaming large plain kXR_write: a single write whose dlen exceeds the
358 * chunk size is delivered to the fd / staged writer in bounded chunks rather
359 * than buffered whole (memory stays at one chunk). brix_write_stream_begin
360 * validates the handle, sets recv.sw_* state, and REWRITES cur_dlen to the
361 * first chunk length so the shared payload-buffer setup below allocates one
362 * chunk; brix_recv_after_payload then applies each chunk and re-arms the next.
363 * If the write is not streamable (SSI / codec / require_pgwrite handle) and
364 * is larger than the buffered write cap, reject it before allocating.
365 */
366 if (ctx->recv.cur_reqid == kXR_write
367 && ctx->recv.cur_dlen > BRIX_WRITE_STREAM_CHUNK)
368 {
369 if (!brix_write_stream_begin(ctx, c, conf)
370 && ctx->recv.cur_dlen > BRIX_MAX_WRITE_PAYLOAD)
371 {
372 ngx_log_error(NGX_LOG_WARN, c->log, 0,
373 "brix: unstreamable write payload too large (%uz),"
374 " closing", (size_t) ctx->recv.cur_dlen);
375 BRIX_SRV_METRIC_INC(ctx, oversized_payloads_total);
376 return BRIX_RECV_STEP_BREAK;
377 }
378 }
379
380 if (ctx->recv.cur_dlen > 0) {
381 if (brix_ensure_payload_buffer(ctx, c, ctx->recv.cur_dlen) != NGX_OK) {
382 return BRIX_RECV_STEP_BREAK;
383 }
384 ctx->recv.payload_pos = 0;
385 ctx->state = XRD_ST_REQ_PAYLOAD;
386 ctx->recv.hdr_pos = 0;
387 return BRIX_RECV_STEP_CONTINUE;
388 }
389
390 ctx->recv.payload = NULL;
391
392 /*
393 * Phase 29 drain barrier (extended for write pipelining): a non-read/write
394 * opcode arriving while reads or writes are still in flight must run with the
395 * connection quiescent. Defer it until both out.count and wr_inflight drain;
396 * the recv loop re-dispatches it then.
397 */
398 if (brix_recv_should_defer(ctx)) {
399 ctx->out.recv_deferred = 1;
400 ctx->out.deferred_streamid[0] = ctx->recv.cur_streamid[0];
401 ctx->out.deferred_streamid[1] = ctx->recv.cur_streamid[1];
402 ctx->state = XRD_ST_SENDING;
403 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
404 return BRIX_RECV_STEP_RETURN;
405 }
406
407 /*
408 * Reset the pipelinable marker before dispatch; only the single-chunk
409 * sendfile read builder sets it back to 1. A read served from the
410 * memory/window path thus stays non-pipelinable (its header/data live in the
411 * shared scratch buffers).
412 */
413 ctx->out.resp_pipelinable = 0;
414
415 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
416 *rx_pending = 0;
417 rc = brix_dispatch(ctx, c, conf);
418 if (rc == NGX_ERROR) {
419 return BRIX_RECV_STEP_BREAK;
420 }
421
422 return brix_recv_after_dispatch_hdr(c, ctx, conf, rev);
423}

425/*
426 * Streaming large plain kXR_write (caller ensures recv.sw_active): one bounded
427 * chunk (payload[0..cur_dlen)) just landed. Apply it to the fd / staged writer at
428 * sw_base_off+sw_done and either re-arm cur_dlen for the next chunk (keep
429 * receiving) or, once the whole logical write has been applied, send the single
430 * ack and return to the request-header state. A streaming write is never deferred
431 * / dispatched: it short-circuits the extend-body, drain-barrier, and dispatch.
432 */
433static brix_recv_step_t
434brix_recv_stream_write_chunk(ngx_connection_t *c, brix_ctx_t *ctx,
435 size_t *rx_pending)
436{
437 brix_write_stream_apply_chunk(ctx, c);
438 ctx->recv.sw_done += ctx->recv.cur_dlen;
439
440 if (ctx->recv.sw_done < ctx->recv.sw_total) {
441 uint32_t remain = ctx->recv.sw_total - ctx->recv.sw_done;
442
443 ctx->recv.cur_dlen = remain < BRIX_WRITE_STREAM_CHUNK
444 ? remain : BRIX_WRITE_STREAM_CHUNK;
445 ctx->recv.payload_pos = 0;
446 ctx->state = XRD_ST_REQ_PAYLOAD;
447 ctx->recv.hdr_pos = 0;
448 return BRIX_RECV_STEP_CONTINUE; /* receive the next chunk */
449 }
450
451 /* Last chunk applied — emit exactly one reply for the logical write. */
452 (void) brix_write_stream_finish(ctx, c);
453 ctx->out.resp_pipelinable = 0;
454 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
455 *rx_pending = 0;
456
457 if (ctx->state == XRD_ST_SENDING) {
458 return BRIX_RECV_STEP_RETURN;
459 }
460 ctx->state = XRD_ST_REQ_HEADER;
461 ctx->recv.hdr_pos = 0;
462 return BRIX_RECV_STEP_CONTINUE;
463}

465/*
466 * A fully-received payload body just landed: run the kXR_writev / kXR_chkpoint
467 * body extension (which may keep receiving), apply the drain barrier, then
468 * dispatch — handling the write-pipelining and read-pipelining continuations.
469 * CONTINUE/RETURN/BREAK.
470 */
471static brix_recv_step_t
472brix_recv_after_payload(ngx_stream_session_t *s, ngx_connection_t *c,
473 ngx_stream_brix_srv_conf_t *conf, brix_ctx_t *ctx, ngx_event_t *rev,
474 size_t *rx_pending)
475{
476 brix_recv_step_t ext;
477 ngx_int_t rc;
478
479 if (ctx->recv.sw_active) {
480 return brix_recv_stream_write_chunk(c, ctx, rx_pending);
481 }
482
483 ext = brix_recv_extend_body(ctx, c);
484 if (ext != BRIX_RECV_STEP_NEXT) {
485 return ext; /* CONTINUE (receiving the streamed body) or BREAK */
486 }
487
488 /*
489 * Phase 29 drain barrier (extended for write pipelining): every opcode other
490 * than kXR_read / kXR_write must run with the connection quiescent, so defer
491 * it until both the response/ack queue and the in-flight pwrites have drained
492 * — e.g. a kXR_close must not retire a handle a pwrite is still writing.
493 */
494 if (brix_recv_should_defer(ctx)) {
495 ctx->out.recv_deferred = 1;
496 ctx->out.deferred_streamid[0] = ctx->recv.cur_streamid[0];
497 ctx->out.deferred_streamid[1] = ctx->recv.cur_streamid[1];
498 ctx->state = XRD_ST_SENDING;
499 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
500 return BRIX_RECV_STEP_RETURN;
501 }
502
503 ctx->state = XRD_ST_REQ_HEADER;
504 ctx->recv.hdr_pos = 0;
505
506 /* Only the single-chunk sendfile read builder re-sets this. */
507 ctx->out.resp_pipelinable = 0;
508
509 BRIX_SRV_METRIC_ADD(ctx, wire_bytes_rx_total, *rx_pending);
510 *rx_pending = 0;
511 rc = brix_dispatch(ctx, c, conf);
512 if (rc == NGX_ERROR) {
513 return BRIX_RECV_STEP_BREAK;
514 }
515
516 if (ctx->state == XRD_ST_AIO) {
517 /*
518 * Write pipelining: a plain kXR_write just posted its pwrite to the
519 * thread pool (wr_inflight bumped in brix_handle_write). Instead of
520 * suspending until it completes — which would serialize the next chunk's
521 * network receive behind this chunk's disk write — keep receiving. The
522 * backpressure boundary caps the depth; the completion callback queues
523 * the ack asynchronously. Every other AIO op still suspends.
524 */
525 if (ctx->recv.cur_reqid == kXR_write && ctx->out.wr_inflight > 0) {
526 ctx->state = XRD_ST_REQ_HEADER;
527 ctx->recv.hdr_pos = 0;
528 return BRIX_RECV_STEP_CONTINUE;
529 }
530 /*
531 * phase-32 WS3: a payload-bearing kXR_read (read-ahead list) whose cold
532 * pread just posted — pipeline it exactly like the dlen==0 read path in
533 * brix_recv_after_header rather than suspending on its AIO.
534 */
535 if (ctx->recv.cur_reqid == kXR_read && ctx->rd.aio_inflight > 0
536 && !ctx->rd.win_active)
537 {
538 ctx->state = XRD_ST_REQ_HEADER;
539 ctx->recv.hdr_pos = 0;
540 return BRIX_RECV_STEP_CONTINUE;
541 }
542 if (ngx_handle_read_event(rev, 0) != NGX_OK) {
543 return BRIX_RECV_STEP_BREAK;
544 }
545 return BRIX_RECV_STEP_RETURN;
546 }
547
548 /*
549 * Phase 29 pipelining (payload-bearing kXR_read with a read-ahead list):
550 * identical to the dlen==0 read path — keep reading so the next read's
551 * sendfile span queues behind this one.
552 */
553 if (brix_recv_try_pipeline_read(ctx)) {
554 ctx->state = XRD_ST_REQ_HEADER;
555 ctx->recv.hdr_pos = 0;
556 return BRIX_RECV_STEP_CONTINUE;
557 }
558
559 if (ctx->state == XRD_ST_SENDING) {
560 return BRIX_RECV_STEP_RETURN;
561 }
562
563 return BRIX_RECV_STEP_CONTINUE;
564}
