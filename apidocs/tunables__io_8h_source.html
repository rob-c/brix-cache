9/*
10 * Read sizing.
11 *
12 * BRIX_READ_MAX is the per-vector element cap used when normalising
13 * client readv requests. Large contiguous reads may still return up to
14 * BRIX_READ_REQUEST_MAX, but the response is split into larger 32 MiB
15 * wire chunks so cleartext sendfile responses need fewer header/file
16 * boundaries and therefore fewer writev/sendfile calls.
17 *
18 * BRIX_READ_CHUNK_MAX — phase-33 P3-B1 (sendfile-span). The XRootD wire
19 * interleaves a kXR response header between successive wire chunks, so a
20 * single sendfile(2) cannot span two chunks: this constant is the largest
21 * contiguous span one sendfile can move. Raising it 16→32 MiB halves the
22 * header interleave (and sendfile/writev syscalls) for a maximum-size
23 * BRIX_READ_REQUEST_MAX (64 MiB) request — a throughput-only geometry
24 * change: reads are byte-identical, only the framing coarsens. It is NOT
25 * pushed to the full 64 MiB request cap so a single sendfile still cannot
26 * monopolise the worker for a whole request (event-loop fairness). Only
27 * the cleartext/kTLS sendfile path is affected — the userspace-TLS and
28 * buffered paths stream through BRIX_READ_WINDOW-sized slices regardless.
29 */
30#define BRIX_READ_MAX (4 * 1024 * 1024)
31#define BRIX_READ_CHUNK_MAX (32 * 1024 * 1024)
32#define BRIX_READ_REQUEST_MAX (64 * 1024 * 1024)
33
34/*
35 * Small-read sendfile floor. A [memory header][in_file body] chain makes
36 * ngx_linux_sendfile_chain bracket every response with four setsockopt(2)
37 * calls (TCP_NODELAY off, TCP_CORK on ... CORK off, NODELAY on) around the
38 * writev+sendfile pair — six syscalls to move a 1 KiB frame. Below this
39 * floor a kXR_read takes the memory path instead: one preadv2 (warm-cache
40 * probe) plus one writev of [hdr|data], zero setsockopts, and the copy cost
41 * of so few bytes is far below the setsockopt bracket it saves. At and above
42 * the floor zero-copy sendfile wins and keeps the bracket.
43 */
44#define BRIX_READ_SENDFILE_MIN (32 * 1024)
45
46/*
47 * Post-auth request read-ahead stash. Reading exact frame sizes makes every
48 * metadata request cost four socket syscalls (header recv, payload recv, and
49 * an ioctl(FIONREAD) after each exact fill inside ngx_unix_recv); asking for
50 * this much instead pulls a whole small request — usually header AND payload,
51 * often several pipelined requests — in ONE partial recv that nginx never
52 * follows with an ioctl. Reads at least this large bypass the stash and go
53 * straight to the destination buffer (bulk write payloads must not pay a
54 * bounce copy). Sized to hold any metadata request (paths cap at PATH_MAX)
55 * while staying a trivial per-connection allocation.
56 */
57#define BRIX_RECV_STASH_SIZE (8 * 1024)

60 * io_uring maximum submission queue depth.
61 * Upper bound for uring ring depth to prevent excessive memory allocation.
62 * 4096 entries is the practical maximum for most workloads.

64#define BRIX_URING_MAX_DEPTH 4096

66/*
67 * Memory-budget streaming (Phase 31).
68 *
69 * BRIX_READ_WINDOW caps the *resident heap* a single in-flight read may hold
70 * at once, independently of the logical request size. A 64 MiB TLS read (which
71 * cannot use sendfile) is served as a sequence of fill -> encrypt -> drain
72 * cycles over a buffer of at most this size, instead of buffering the whole
73 * request in heap. The wire framing (BRIX_READ_CHUNK_MAX chunks) is unchanged
74 * — only the in-memory slice shrinks.
75 *
76 * BRIX_SCRATCH_TRIM_THRESHOLD is the high-water mark above which a per-session
77 * reusable scratch buffer is shrunk back to BRIX_READ_WINDOW once a request
78 * has fully drained. Hysteresis (2x window) avoids realloc thrash on sessions
79 * that legitimately oscillate around the window size, and the trim is hot-
80 * deferred (brix_trim_scratch): a buffer used since the previous pass is kept
81 * warm for one more cycle so streaming transfers never free/realloc per
82 * request (equal-size churn made glibc mmap/munmap the block every request).
83 *
84 * BRIX_CONN_XFER_HEAP_MAX bounds the combined size of one connection's
85 * transfer scratch buffers (read_scratch + read_hdr_scratch + write_scratch +
86 * payload_buf). With the read/write windows in place this ceiling is normally
87 * never reached; it is the per-connection backstop that turns a runaway buffer
88 * into a clean kXR_NoMemory rather than unbounded growth.
89 */
90#define BRIX_READ_WINDOW (2 * 1024 * 1024)
91#define BRIX_SCRATCH_TRIM_THRESHOLD (2 * BRIX_READ_WINDOW)
92#define BRIX_CONN_XFER_HEAP_MAX (4 * BRIX_READ_WINDOW)

94/*
95 * Optional io_uring disk-I/O backend (Phase 44).
96 *
97 * The backend is a third AIO dispatch tier (io_uring -> thread pool -> inline
98 * sync) selected per server block by `brix_io_uring on|off|auto`. These are
99 * compile-time, header-only constants; the runtime verdict is decided by the
100 * authoritative opcode probe in src/aio/uring.c, never by parsing `uname`.
101 *
102 * Mode enum (stored in ngx_stream_brix_srv_conf_t.io_uring, set via an
103 * ngx_conf_enum_t slot):
104 * OFF = never use io_uring (thread pool / inline only)
105 * ON = require io_uring — startup FAILS if it is not compiled in or the
106 * runtime probe fails (see §32 fail-fast; brix_uring_validate_conf)
107 * AUTO = enable iff the probe passes this process, else silent fallback
108 *
109 * QUEUE_DEPTH is the per-worker ring's SQ/CQ entry count = the ceiling on
110 * concurrent in-flight SQEs. Each read submits one SQE (windowed reads stream
111 * as one contiguous IORING_OP_READ per wire chunk), so depth tracks connection
112 * concurrency, not request size; get_sqe -> NULL simply falls back to the pool.
113 *
114 * The MIN_KERNEL_* values are a fast pre-filter only (5.6 = reliable
115 * register_eventfd for the completion bridge); RESTRICT_KERNEL_MINOR (5.10) is
116 * where io_uring_register_restrictions() becomes available (best-effort).
117 */
118#define BRIX_IO_URING_OFF 0
119#define BRIX_IO_URING_ON 1
120#define BRIX_IO_URING_AUTO 2
121
122#define BRIX_IO_URING_QUEUE_DEPTH 256
123#define BRIX_IO_URING_MIN_KERNEL_MAJOR 5
124#define BRIX_IO_URING_MIN_KERNEL_MINOR 6
125#define BRIX_IO_URING_RESTRICT_KERNEL_MINOR 10

127/*
128 * Output-queue depth (Phase 29 pipelining; runtime-configurable).
129 *
130 * The per-connection output ring (brix_ctx_t.out_ring) and read-buffer pool
131 * (rd_pool) are heap-allocated at connection time to ctx->out.pipeline_depth — the
132 * number of in-flight responses that may be outstanding before the recv loop
133 * applies backpressure. Each slot owns one in-flight response's send state
134 * (flat-buffer tail, chain tail, reusable header/data/file chain structs), so a
135 * DEEPER pipeline absorbs more wire latency/jitter (packet reordering, high-BDP
136 * links) — a momentarily-slow drain no longer empties the in-flight window and
137 * stalls the recv->process->send loop — at a per-slot memory cost.
138 *
139 * Set via `brix_pipeline_depth N`; merged/clamped to [MIN, MAX]. Sizing the
140 * rings to the configured depth keeps a small default cheap while leaving a large
141 * value available for lossy/jittery WANs. Was a fixed #define of 4 (Phase 29/32).
142 */
143#define BRIX_PIPELINE_DEPTH_DEFAULT 8
144#define BRIX_PIPELINE_DEPTH_MIN 1
145#define BRIX_PIPELINE_DEPTH_MAX 64

147/*
148 * Per-slot wire-header capacity (Phase 32 WS2). A multi-chunk sendfile read
149 * emits one XRD_RESPONSE_HDR_LEN header per 32 MiB wire chunk (phase-33 P3-B1);
150 * the worst case is BRIX_READ_REQUEST_MAX/BRIX_READ_CHUNK_MAX chunks (= 2 → 16
151 * bytes). Sizing
152 * the per-slot header buffer to this lets a multi-chunk response keep its headers
153 * in its own slot (not the shared read_hdr_scratch), so multiple multi-chunk
154 * reads can be pipelined without their headers aliasing.
155 */
156#define BRIX_SLOT_HDR_MAX \
157 (((BRIX_READ_REQUEST_MAX + BRIX_READ_CHUNK_MAX - 1) \
158 / BRIX_READ_CHUNK_MAX) * XRD_RESPONSE_HDR_LEN)

174/*
175 * Maximum write payload per request. xrdcp v5 uses 8 MiB chunks by default;
176 * each pgwrite payload adds 4-byte CRC per 4096-byte page (~0.1% overhead).
177 * Cap at 16 MiB to handle non-default chunk sizes with headroom.
178 */
179#define BRIX_MAX_WRITE_PAYLOAD (16 * 1024 * 1024)

181/*
182 * Streaming large plain kXR_write. A single kXR_write whose dlen exceeds
183 * BRIX_WRITE_STREAM_CHUNK is delivered to the file / staged writer in bounded
184 * BRIX_WRITE_STREAM_CHUNK-sized installments instead of being buffered whole,
185 * so a client that sends one very large write (e.g. go-hep's 64 MiB inline
186 * WriteAtContext) is accepted with memory bounded to one chunk rather than the
187 * whole payload. BRIX_MAX_WRITE_STREAM is the absolute per-write ceiling (a
188 * sanity bound on the dlen field, still well below the 4 GiB the wire allows) so
189 * a hostile dlen is rejected before streaming begins; the buffered fast path for
190 * writes <= BRIX_WRITE_STREAM_CHUNK is unchanged and still capped at
191 * BRIX_MAX_WRITE_PAYLOAD. pgwrite/writev/chkpoint keep the 16 MiB cap — they
192 * carry per-page CRC / descriptor structure that the chunk streamer does not
193 * split, and stock clients already chunk them below the cap.
194 */
195#define BRIX_WRITE_STREAM_CHUNK (8 * 1024 * 1024)
196#define BRIX_MAX_WRITE_STREAM (1024u * 1024u * 1024u)
