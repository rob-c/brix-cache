152/*
153 * A directory rename carries the whole recorded subtree. This is the property a
154 * DEL+ADD pair cannot express, and its absence was the bug: the children stayed
155 * at a path the cluster no longer served.
156 */
157static void
158test_rename_directory_subtree(void)
159{
160 brix_cns_inv_t *inv = mk(16);
161 uint64_t sz = 0;
162 int dir = 0;
163
164 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_MKDIR, "/run/a", 0, 1, 3) == 0);
165 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_ADD, "/run/a/f1", 11, 1, 3) == 0);
166 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_MKDIR, "/run/a/sub", 0, 1, 3) == 0);
167 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_ADD, "/run/a/sub/f2", 22, 1, 3) == 0);
168 /* Shares the "/run/a" prefix but NOT at a component boundary — must not move. */
169 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_ADD, "/run/abc", 33, 1, 3) == 0);
170
171 CHECK(brix_cns_inv_rename(inv, "/run/a", "/run/b", 0, 999, 1, 3) == 0);
172
173 CHECK(brix_cns_inv_stat(inv, "/run/b", NULL, NULL, &dir) == 0 && dir == 1);
174 CHECK(brix_cns_inv_stat(inv, "/run/b/f1", &sz, NULL, NULL) == 0 && sz == 11);
175 CHECK(brix_cns_inv_stat(inv, "/run/b/sub/f2", &sz, NULL, NULL) == 0 && sz == 22);
176 CHECK(brix_cns_inv_stat(inv, "/run/a/f1", NULL, NULL, NULL) == 1);
177 CHECK(brix_cns_inv_stat(inv, "/run/a/sub/f2", NULL, NULL, NULL) == 1);
178 /* the boundary non-match survives untouched */
179 CHECK(brix_cns_inv_stat(inv, "/run/abc", &sz, NULL, NULL) == 0 && sz == 33);
180 CHECK(brix_cns_inv_count(inv) == 5);
181 free(inv);
182}

218/*
219 * A child whose re-prefixed path would not fit is DROPPED, never truncated — a
220 * truncated path would name a file the cluster does not have, and the manager
221 * answers stat from this table, so a wrong hit is worse than a miss.
222 */
223static void
224test_rename_overlong_child_dropped(void)
225{
226 brix_cns_inv_t *inv = mk(16);
227 char longdst[BRIX_CNS_PATH_MAX + 1];
228 char child[BRIX_CNS_PATH_MAX + 1];
229
230 /* "/a" + "/" + 8 tail chars; the destination is sized so dst+tail overflows. */
231 snprintf(child, sizeof(child), "/a/deepfile");
232 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_MKDIR, "/a", 0, 1, 1) == 0);
233 CHECK(brix_cns_inv_apply(inv, BRIX_CNS_ADD, child, 5, 1, 1) == 0);
234
235 memset(longdst, 'z', sizeof(longdst));
236 longdst[0] = '/';
237 longdst[BRIX_CNS_PATH_MAX] = '\0'; /* exactly the maximum length */
238
239 CHECK(brix_cns_inv_rename(inv, "/a", longdst, 0, 1, 1, 1) == 0);
240 CHECK(brix_cns_inv_stat(inv, longdst, NULL, NULL, NULL) == 0); /* dir moved */
241 CHECK(brix_cns_inv_stat(inv, child, NULL, NULL, NULL) == 1); /* old gone */
242 CHECK(brix_cns_inv_count(inv) == 1); /* child dropped */
243 free(inv);
244}
