1/* brixcvmfs_ingest_image.c — `brixcvmfs ingest image <ref>` (phase-104 D8):
2 * pull → flatten → publish, incrementally. Tool surface only (G14).
3 *
4 * brixcvmfs ingest image <ref> --repo <repo_dir> [--prefix /images]
5 * [--platform os/arch] [--tag-path name:tag-dir] [--squash-owner U:G]
6 * [--strict] [--max-bytes N] [--keys-dir D] [--chunk-size N]
7 * [--insecure] [--token-file F] [--prune-old] [--force-overlap]
8 * [--verify-diffids] [--layout flat|layered] [--dry-run] [--no-wait]
9 *
10 * Published namespace (the DUCC-compatible flat layout, D8.2):
11 * <prefix>/.images/<alg>/<manifest-hex>/ the rootfs
12 * <prefix>/.images/<alg>/<manifest-hex>/.config.json image config
13 * <prefix>/.images/<alg>/<manifest-hex>/.manifest.json manifest, verbatim
14 * <prefix>/<host>/<name>:<tag> → relative symlink → the digest root
15 *
16 * --layout layered (D15.6) publishes each layer at its own content-addressed
17 * root instead of merging them:
18 * <prefix>/.layers/<alg>/<layer-hex>/ one layer, verbatim
19 * <prefix>/.images/<alg>/<manifest-hex>/.layers the composition, in order
20 * A published layer is never fetched, decompressed or scanned again, so the
21 * second image off a shared base costs only its own top layers. The image root
22 * stops being a runnable rootfs there — composing the layer roots (overlayfs
23 * lowerdirs, lowest last) becomes the consumer's job — which is why this is a
24 * flag and flat stays the default.
25 *
26 * Pipeline: resolve → memo check (<repo>/.brix-ingest/memo<flat-path>; same
27 * manifest digest = no-op) → take the transaction lock → fetch layers
28 * hash-on-stream → flatten (D7) into a scratch upper → sidecars + tag
29 * symlink → scan + reprefix → publish → memo/ledger write. A crash
30 * pre-publish leaves only scratch (reaped on the next run); the publish
31 * engine's manifest-swap-last gives crash-safety for step 7 for free.
32 *
33 * --verify-diffids (D8.e): the layer blob digests are verified on fetch (that
34 * is the transport identity); the image *config* additionally names each
35 * layer's diff_id — the sha256 of the UNCOMPRESSED tar. The flattener already
36 * decompresses, so capturing that hash costs no second inflate; the compare
37 * happens once the config sidecar is in hand and catches a registry whose
38 * manifest and config disagree about which bytes the image is made of.
39 *
40 * Foreign-path safety: every structural ADD outside the image's own digest
41 * root carries no_clobber, so a published non-dir at <prefix>/<host>/… fails
42 * the publish instead of being silently retyped (--force-overlap lifts it).
43 */

166static int img_parse_val(ing_img_opts_t *o, const char *a, const char *v) {
167 if (img_parse_str(o, a, v) == 0) return 0;
168 else if (strcmp(a, "--chunk-size") == 0) o->chunk_size = atol(v);
169 else if (strcmp(a, "--max-bytes") == 0) o->max_bytes = atoll(v);
170 else if (strcmp(a, "--layout") == 0) {
171 if (strcmp(v, "layered") != 0 && strcmp(v, "flat") != 0) return -1;
172 o->layered = strcmp(v, "layered") == 0;
173 }
174 else if (strcmp(a, "--prefix") == 0) {
175 size_t n = snprintf(o->prefix, sizeof(o->prefix), "%s", v);
176 if (n >= sizeof(o->prefix)) return -1;
177 while (n > 1 && o->prefix[n - 1] == '/') o->prefix[--n] = '\0';
178 } else if (strcmp(a, "--squash-owner") == 0) {
179 if (sscanf(v, "%u:%u", &o->squash_uid, &o->squash_gid) != 2) return -1;
180 o->squash = 1;
181 } else
182 return -1;
183 return 0;
184}

186static int img_parse(int argc, char **argv, ing_img_opts_t *o) {
187 memset(o, 0, sizeof(*o));
188 snprintf(o->prefix, sizeof(o->prefix), "/images");
189 if (argc < 2 || argv[1][0] == '-') return img_usage();
190 o->ref_text = argv[1];
191 for (int i = 2; i < argc; i++) {
192 if (img_parse_flag(o, argv[i]) == 0) continue;
193 if (i + 1 < argc && img_parse_val(o, argv[i], argv[i + 1]) == 0) {
194 i++;
195 continue;
196 }
197 return img_usage();
198 }
199 return o->repo != NULL ? ING_OK : img_usage();
200}

204/* --require-digest (App. L): the digest chain below proves the tree matches
205 * the manifest we RESOLVED, never that the manifest is the one the operator
206 * meant — a tag can be repointed between two runs. Only the operator knows
207 * which this ingest is, so it is a flag, and it fires before the first
208 * network byte. */
209static int img_resolve(ing_img_ctx_t *c) {
210 char err[512];
211 if (brix_oci_ref_parse(c->o->ref_text, &c->ref, err, sizeof(err)) != 0)
212 return bci_fail(ING_USAGE, "bad image ref", err);
213 if (c->o->require_digest && !c->ref.has_digest)
214 return bci_fail(ING_USAGE,
215 "--require-digest: ref is not digest-pinned "
216 "(write <name>@sha256:<hex>)", c->o->ref_text);
217 int rc = brix_oci_reg_from_ref(&c->reg, &c->ref, c->o->insecure,
218 c->name, sizeof(c->name), err, sizeof(err));
219 if (rc != 0) return img_map(rc, "registry setup failed", err);
220 if (c->o->token_file != NULL
221 && bci_read_line(c->o->token_file, c->reg.bearer,
222 sizeof(c->reg.bearer)) != 0)
223 return bci_fail(ING_USAGE, "cannot read --token-file", c->o->token_file);
224 rc = brix_oci_reg_resolve(&c->reg, &c->ref, c->o->platform, &c->desc,
225 err, sizeof(err));
226 if (rc != 0) return img_map(rc, "manifest resolve failed", err);
227 /* The resolved descriptor names its own algorithm; the on-disk tree is
228 * keyed by it, so read it out of the grammar rather than stepping over a
229 * prefix whose length is only incidentally the same for both. */
230 if (brix_oci_digest_parse(c->desc.digest, strlen(c->desc.digest),
231 &c->dig) != 0)
232 return bci_fail(ING_FAIL, "registry returned an unusable digest",
233 c->desc.digest);
234 c->hex = c->dig.hex;
235 return ING_OK;
236}

238static int img_paths(ing_img_ctx_t *c) {
239 const char *host = c->ref.host[0] != '\0' ? c->ref.host : "docker.io";
240 int rc = c->o->tag_path != NULL
241 ? img_path(c->tagrel, sizeof(c->tagrel), "%s", c->o->tag_path)
242 : img_path(c->tagrel, sizeof(c->tagrel), "%s/%s:%s",
243 host, c->name, c->ref.tag);
244 if (rc != ING_OK) return rc;
245 char probe[ING_PATH_MAX + 1]; /* tagrel gets the prefix grammar */
246 if (img_path(probe, sizeof(probe), "/%s", c->tagrel) != ING_OK
247 || bci_prefix_check(probe) != ING_OK)
248 return bci_fail(ING_USAGE, "bad tag path", c->tagrel);
249 rc = img_path(c->root_rel, sizeof(c->root_rel),
250 ".images/%s/%s", brix_oci_alg_name(c->dig.alg), c->hex);
251 if (rc == ING_OK)
252 rc = img_path(c->root_path, sizeof(c->root_path), "%s/%s",
253 img_pfx(c), c->root_rel);
254 if (rc == ING_OK)
255 rc = img_path(c->flat_path, sizeof(c->flat_path), "%s/%s",
256 img_pfx(c), c->tagrel);
257 if (rc == ING_OK)
258 rc = img_path(c->scratch, sizeof(c->scratch),
259 "%s/.brix-ingest/scratch", c->o->repo);
260 if (rc == ING_OK)
261 rc = img_path(c->upper, sizeof(c->upper), "%s/upper", c->scratch);
262 if (rc == ING_OK)
263 rc = img_path(c->root_abs, sizeof(c->root_abs), "%s/%s",
264 c->upper, c->root_rel);
265 if (rc == ING_OK)
266 rc = img_path(c->layers_path, sizeof(c->layers_path), "%s/.layers",
267 img_pfx(c));
268 if (rc == ING_OK)
269 rc = img_path(c->memo_path, sizeof(c->memo_path),
270 "%s/.brix-ingest/memo%s", c->o->repo, c->flat_path);
271 return rc;
272}

291static int img_layers_foreach(ing_img_ctx_t *c, img_layer_cb cb, int *out_n) {
292 const char *arr, *el;
293 size_t an, en, cur = 0;
294 if (brix_json_get_raw(c->desc.body, c->desc.body_len, "layers",
295 &arr, &an) != 1)
296 return bci_fail(ING_FAIL, "manifest has no layers array", NULL);
297 int idx = 0, st;
298 while ((st = brix_json_arr_next(arr, an, &cur, &el, &en)) == 1) {
299 char dig[BRIX_OCI_DIGEST_STRLEN];
300 if (brix_json_get_str(el, en, "digest", dig, sizeof(dig)) != 1)
301 return bci_fail(ING_FAIL, "manifest layer without digest", NULL);
302 if (cb != NULL) {
303 int rc = cb(c, idx, dig);
304 if (rc != ING_OK) return rc;
305 }
306 idx++;
307 }
308 if (st < 0)
309 return bci_fail(ING_FAIL, "malformed manifest layers array", NULL);
310 if (out_n != NULL) *out_n = idx;
311 return ING_OK;
312}

314static int img_layer_apply(ing_img_ctx_t *c, int idx, const char *digest) {
315 char lpath[ING_PATH_MAX + 32], err[512], updir[ING_PATH_MAX];
316 const char *upper_dir = c->root_abs;
317 if (c->o->layered) {
318 int st = bci_layer_stage(c->o->repo, c->o->prefix, c->upper, digest,
319 c->o->verify_diffids, &c->layers[idx],
320 updir, sizeof(updir));
321 if (st != ING_OK) return st;
322 if (c->layers[idx].reused) {
323 /* --verify-diffids still has to have something to compare: the
324 * ledger recorded it when the layer was first materialized. */
325 snprintf(c->diffids[idx], sizeof(c->diffids[idx]), "%s",
326 c->layers[idx].diffid);
327 return ING_OK;
328 }
329 upper_dir = updir;
330 }
331 snprintf(lpath, sizeof(lpath), "%s/layer.%d", c->scratch, idx);
332 int fd = open(lpath, O_RDWR | O_CREAT | O_TRUNC | O_CLOEXEC, 0600);
333 if (fd < 0)
334 return bci_fail(ING_FAIL, "cannot create layer temp", lpath);
335 int rc = brix_oci_reg_blob_fetch(&c->reg, c->name, digest, fd,
336 err, sizeof(err));
337 if (rc != 0) {
338 close(fd);
339 unlink(lpath);
340 return img_map(rc, "layer fetch failed", err);
341 }
342 brix_flatten_opts_t fo;
343 memset(&fo, 0, sizeof(fo));
344 fo.upper_dir = upper_dir;
345 fo.strict = c->o->strict;
346 fo.max_total_bytes = c->o->max_bytes;
347 fo.squash = c->o->squash;
348 fo.squash_uid = (uid_t) c->o->squash_uid;
349 fo.squash_gid = (gid_t) c->o->squash_gid;
350 if (c->diffids != NULL && idx < c->ndiffids) {
351 fo.diffid_hex = c->diffids[idx];
352 fo.diffid_hexlen = sizeof(c->diffids[idx]);
353 }
354 rc = lseek(fd, 0, SEEK_SET) == 0
355 && brix_flatten_layer(&fo, fd, &c->st, err, sizeof(err)) == 0
356 ? ING_OK : bci_fail(ING_FAIL, "layer refused", err);
357 close(fd);
358 unlink(lpath);
359 if (rc == ING_OK && c->layers != NULL && c->diffids != NULL)
360 snprintf(c->layers[idx].diffid, sizeof(c->layers[idx].diffid), "%s",
361 c->diffids[idx]);
362 return rc;
363}

367static int img_sidecars(ing_img_ctx_t *c) {
368 char dig[BRIX_OCI_DIGEST_STRLEN], path[ING_PATH_MAX + 32], err[512];
369 const char *cel;
370 size_t cen;
371 if (brix_json_get_raw(c->desc.body, c->desc.body_len, "config",
372 &cel, &cen) != 1
373 || brix_json_get_str(cel, cen, "digest", dig, sizeof(dig)) != 1)
374 return bci_fail(ING_FAIL, "manifest has no config digest", NULL);
375 snprintf(path, sizeof(path), "%s/.config.json", c->root_abs);
376 int fd = open(path, O_RDWR | O_CREAT | O_TRUNC | O_CLOEXEC, 0644);
377 if (fd < 0)
378 return bci_fail(ING_FAIL, "cannot create config sidecar", path);
379 int rc = brix_oci_reg_blob_fetch(&c->reg, c->name, dig, fd,
380 err, sizeof(err));
381 close(fd);
382 if (rc != 0) {
383 unlink(path);
384 return img_map(rc, "config fetch failed", err);
385 }
386 snprintf(path, sizeof(path), "%s/.manifest.json", c->root_abs);
387 FILE *f = fopen(path, "w");
388 int ok = f != NULL
389 && fwrite(c->desc.body, 1, c->desc.body_len, f) == c->desc.body_len;
390 if (f != NULL) ok = fclose(f) == 0 && ok;
391 if (!ok)
392 return bci_fail(ING_FAIL, "cannot write manifest sidecar", path);
393 return c->o->layered
394 ? bci_layers_sidecar(c->root_abs, c->root_rel, c->layers, c->nlayers)
395 : ING_OK;
396}

412/* Structural entries (everything outside the image's own digest root) must
413 * not retype foreign published content — see the header comment. */
414static int img_own(const char *path, const char *root) {
415 size_t n = strlen(root);
416 return strncmp(path, root, n) == 0 && (path[n] == '\0' || path[n] == '/');
417}

419/* The layer namespace is this tool's own too: two images sharing a base both
420 * ADD the same layer root, and no_clobber there would turn a correct second
421 * publish into a refusal. */
422static void img_guard_structural(ing_img_ctx_t *c, cvmfs_changeset_t *cs) {
423 for (size_t i = 0; i < cs->n; i++) {
424 cvmfs_change_t *ch = &cs->v[i];
425 if (img_own(ch->path, c->root_path)
426 || (c->o->layered && img_own(ch->path, c->layers_path)))
427 continue;
428 if (c->o->force_overlap)
429 ch->no_clobber = 0;
430 else if (ch->op == CVMFS_CH_ADD_DIR || ch->op == CVMFS_CH_ADD_LINK)
431 ch->no_clobber = 1;
432 }
433}

435/* Step 8: memo + roots ledger — advisory state, so failures warn, never
436 * roll back a publish that already happened. */
437static void img_record(ing_img_ctx_t *c, long rev) {
438 char utc[40], line[2 * ING_PATH_MAX], lpath[ING_PATH_MAX + 96];
439 bci_utc_now(utc, sizeof(utc));
440 snprintf(line, sizeof(line), "%s %s %ld %s\n",
441 c->flat_path, c->desc.digest, rev, utc);
442 if (bci_write_atomic(c->memo_path, line) != 0)
443 fprintf(stderr, "brixcvmfs ingest: warning: cannot write memo %s\n",
444 c->memo_path);
445 snprintf(lpath, sizeof(lpath), "%s/.brix-ingest/roots%s/%s",
446 c->o->repo, img_pfx(c), c->hex);
447 snprintf(line, sizeof(line), "%s %s %ld %s\n",
448 c->o->prefix, c->desc.digest, rev, utc);
449 if (bci_write_atomic(lpath, line) != 0)
450 fprintf(stderr, "brixcvmfs ingest: warning: cannot write ledger %s\n",
451 lpath);
452 if (c->o->layered) {
453 bci_layer_record(c->o->repo, c->o->prefix, c->layers, c->nlayers, rev);
454 bci_imglayers_record(c->o->repo, c->o->prefix, c->hex, c->layers,
455 c->nlayers);
456 }
457 if (c->pruned_old)
458 bci_root_forget(c->o->repo, c->o->prefix, c->old_digest);
459}

461static int img_publish(ing_img_ctx_t *c, int nlayers) {
462 cvmfs_changeset_t cs;
463 char err[1024];
464 long rev = 0;
465 if (cvmfs_changeset_scan(c->upper, &cs, err, sizeof(err)) != 0)
466 return bci_fail(ING_FAIL, "scratch scan failed", err);
467 int rc = cvmfs_changeset_reprefix(&cs, c->o->prefix, err, sizeof(err)) == 0
468 ? ING_OK : bci_fail(ING_FAIL, "prefix remap failed", err);
469 if (rc == ING_OK) {
470 img_guard_structural(c, &cs);
471 int marked = c->o->prune_old
472 ? bci_prune_old_mark(c->o->repo, c->o->prefix, c->memo_path,
473 c->old_digest, c->desc.digest, &cs) : 0;
474 rc = marked < 0 ? ING_FAIL : ING_OK;
475 c->pruned_old = marked == 1;
476 }
477 size_t nch = cs.n;
478 if (rc == ING_OK) {
479 cvmfs_publish_opts_t po;
480 memset(&po, 0, sizeof(po));
481 po.repo_dir = c->o->repo;
482 po.keys_dir = c->o->keys_dir;
483 po.chunk_size = c->o->chunk_size;
484 rc = cvmfs_publish_run(&po, &cs, &rev, err, sizeof(err)) == 0
485 ? ING_OK : bci_fail(ING_FAIL, "publish failed", err);
486 }
487 cvmfs_changeset_free(&cs);
488 if (rc != ING_OK) return rc;
489 img_record(c, rev);
490 printf("ingested %s -> %s (revision %ld, %d layers, %zu changes,"
491 " %lld files, %lld bytes)%s\n",
492 c->o->ref_text, c->flat_path, rev, nlayers, nch,
493 (long long) c->st.files, (long long) c->st.bytes,
494 c->pruned_old ? " [pruned old root]" : "");
495 return ING_OK;
496}

500static int img_dry_run(ing_img_ctx_t *c) {
501 int n = 0;
502 int rc = img_layers_foreach(c, NULL, &n);
503 if (rc != ING_OK) return rc;
504 printf("dry-run: %s\n manifest: %s (%d layers)\n"
505 " root: %s\n tag: %s -> %s\n",

511/* 1 = some other tag under THIS prefix already published this manifest's
512 * digest root (memos are written only after a successful publish) — a retag
513 * needs no layer bytes, only the tag symlink. */
514static int img_root_published(ing_img_ctx_t *c) {
515 char memo_dir[ING_PATH_MAX + 32];
516 snprintf(memo_dir, sizeof(memo_dir), "%s/.brix-ingest/memo%s",
517 c->o->repo, img_pfx(c));
518 return bci_memo_refs(memo_dir, c->desc.digest, c->memo_path) > 0;
519}
520
521/* Per-layer bookkeeping needs a slot per layer before the apply loop runs, so
522 * the manifest is walked once for the count. --verify-diffids wants the
523 * captured hashes; --layout layered wants the digests, and takes the diff_id
524 * capture with it because the flattener decompresses anyway and a recorded
525 * diff_id is what lets a REUSED layer still be verified on a later run. With
526 * neither flag nothing is allocated and both stay off end to end. */
527static int img_layers_alloc(ing_img_ctx_t *c) {
528 int n = 0, rc;
529
530 if (!c->o->verify_diffids && !c->o->layered)
531 return ING_OK;
532 rc = img_layers_foreach(c, NULL, &n);
533 if (rc != ING_OK)
534 return rc;
535 if (n <= 0)
536 return bci_fail(ING_FAIL, "manifest has no layers", NULL);
537 c->diffids = calloc((size_t) n, sizeof(*c->diffids));
538 if (c->diffids == NULL)
539 return bci_fail(ING_FAIL, "out of memory", NULL);
540 c->ndiffids = n;
541 if (c->o->layered
542 && (c->layers = calloc((size_t) n, sizeof(*c->layers))) == NULL)
543 return bci_fail(ING_FAIL, "out of memory", NULL);
544 c->nlayers = c->o->layered ? n : 0;
545 return ING_OK;
546}

548static int img_locked(ing_img_ctx_t *c) {
549 int nlayers = 0;
550 if (brixcvmfs_tx_rm_tree(c->scratch) != 0)
551 return bci_fail(ING_FAIL, "cannot reap stale scratch", c->scratch);
552 int rc = ING_OK;
553 if (img_root_published(c)) {
554 /* symlink-only upper: bci_tag_symlink creates the chain */
555 rc = bci_mkdir_p(c->upper, 0755) == 0
556 ? ING_OK : bci_fail(ING_FAIL, "cannot create scratch tree", c->upper);
557 } else {
558 if (bci_mkdir_p(c->root_abs, 0755) != 0)
559 return bci_fail(ING_FAIL, "cannot create scratch tree", c->root_abs);
560 rc = img_layers_alloc(c);
561 if (rc == ING_OK) rc = img_layers_foreach(c, img_layer_apply, &nlayers);
562 if (rc == ING_OK) rc = img_sidecars(c);
563 if (rc == ING_OK) rc = img_verify(c);
564 }
565 if (rc == ING_OK) rc = bci_tag_symlink(c->upper, c->tagrel, c->root_rel);
566 if (rc == ING_OK) rc = img_publish(c, nlayers);
567 if (rc == ING_OK) brixcvmfs_tx_rm_tree(c->scratch);
568 /* on failure scratch stays for forensics; the next run reaps it */
569 return rc;
570}

572int bci_image_main(int argc, char **argv) {
573 ing_img_opts_t o;
574 ing_img_ctx_t c;
575 int rc = img_parse(argc, argv, &o);
576 if (rc != ING_OK) return rc;
577 rc = bci_prefix_check(o.prefix);
578 if (rc != ING_OK) return rc;
579 memset(&c, 0, sizeof(c));
580 c.o = &o;
581 rc = img_resolve(&c);
582 if (rc == ING_OK) rc = img_paths(&c);
583 if (rc == ING_OK && img_memo_fresh(&c)) {
584 printf("up to date: %s at %s\n", c.flat_path, c.desc.digest);
585 } else if (rc == ING_OK && o.dry_run) {
586 rc = img_dry_run(&c);
587 } else if (rc == ING_OK) {
588 rc = bci_lock_acquire(o.repo, o.no_wait);
589 if (rc == ING_OK) {
590 rc = img_locked(&c);
591 bci_lock_release(o.repo);
592 }
593 }
594 brix_oci_desc_free(&c.desc);
595 free(c.diffids);
596 free(c.layers);
597 return rc;
598}
