1/*
2 * conn.c — connection / session bring-up and teardown.
3 *
4 * WHAT: connect → (20B handshake + kXR_protocol, pipelined) → kXR_login (anon) →
5 * ready; and a best-effort kXR_endsess + close on teardown.
6 * WHY: Everything else (stat/ls/get/put) needs a logged-in session; this is the
7 * one place that drives the handshake state machine.
8 * HOW: We send the 20-byte ClientInitHandShake and the 24-byte protocol request
9 * as one 44-byte segment (as modern clients do). The server replies with a
10 * handshake frame (streamid {0,0}) AND a protocol frame (streamid = our
11 * protocol request's id); we tolerate either ordering / a combined reply by
12 * keying on the streamid. Anonymous login sends no credential payload; if
13 * the server demands a security protocol we fail cleanly (auth is M4).
14 *
15 * wire: XProtocol.hh ClientInitHandShake — {0,0,0,htonl(4),htonl(2012=ROOTD_PQ)}.
16 * wire: XProtocol.hh ServerProtocolBody — pval[4] flags[4]; flags carry server caps.
17 * wire: XProtocol.hh ServerLoginBody — sessid[16] [+ "&P=..." security list].
18 */

41/*
42 * Resolve the CA trust directory for TLS peer verification using the same search
43 * order as the stock xrootd/globus tooling, so the client trusts grid CAs out of
44 * the box:
45 * 1. explicit caller value (e.g. --ca-dir) 2. $X509_CERT_DIR
46 * 3. /etc/grid-security/certificates (if present and readable)
47 * 4. NULL ⇒ caller falls back to OpenSSL system defaults
48 * Returns a borrowed string (env/literal/argument); never allocates.
49 */
50const char *
51brix_resolve_ca_dir(const char *opt_ca_dir)
52{
53 const char *env;
54
55 if (opt_ca_dir != NULL && opt_ca_dir[0] != '\0') {
56 return opt_ca_dir;
57 }
58
59 env = getenv("X509_CERT_DIR");
60 if (env != NULL && env[0] != '\0') {
61 return env;
62 }
63
64 if (access(XRDC_GRID_CA_DIR, R_OK | X_OK) == 0) {
65 return XRDC_GRID_CA_DIR;
66 }
67
68 return NULL;
69}

71/*
72 * Resolve the X.509 proxy PEM to present as the CLIENT certificate on a davs/
73 * https mutual-TLS handshake — the HTTP-client analogue of the root:// GSI
74 * resolver (sec_gsi.c). Same precedence:
75 * 1. $X509_USER_PROXY (xrdcp exports --proxy into it)
76 * 2. /tmp/x509up_u<euid> (the grid default proxy location)
77 * A proxy PEM carries the proxy cert, its private key, and the EEC chain in one
78 * file, so the single returned path feeds both the cert-chain and the key.
79 * Writes into `buf` and returns it ONLY when the file exists and is readable;
80 * returns NULL otherwise (no proxy → the client presents no cert, and TLS only
81 * sends one when the server actually requests client auth, so this is safe for
82 * plain https endpoints). Never allocates.
83 */
84const char *
85brix_web_proxy_pem(char *buf, size_t buflen)
86{
87 const char *env;
88
89 if (buf == NULL || buflen == 0) {
90 return NULL;
91 }
92
93 env = getenv("X509_USER_PROXY");
94 if (env != NULL && env[0] != '\0') {
95 snprintf(buf, buflen, "%s", env);
96 } else {
97 snprintf(buf, buflen, "/tmp/x509up_u%u", (unsigned) geteuid());
98 }
99
100 if (access(buf, R_OK) != 0) {
101 return NULL;
102 }
103 return buf;
104}

106/* Establish the session on the already-set c->host/c->port using c->want_tls +
107 * c->opts: connect → handshake → [TLS] → login → auth. Resets per-connection state
108 * so it is safe to call again from brix_reconnect after a redirect. */
109static int
110brix_bringup_ex(brix_conn *c, int want_login, brix_status *st)
111{
112 c->io.fd = -1;
113 c->io.ssl = NULL;
114 c->ssl_ctx = NULL;
115 c->next_sid = 1;
116 c->server_flags = 0;
117 c->sec_level = 0;
118 c->sec_odata = 0;
119 c->signing_active = 0;
120 c->sig_seqno = 0;
121
122 /* §15: arm trace/timing from opts (the per-opcode RTT table on c persists
123 * across a redirect's reconnect, so the final summary aggregates all hops). */
124 c->diag.wire_trace = c->opts.wire_trace;
125 c->diag.timing = c->opts.timing;
126 c->diag.redir_trace = c->opts.redir_trace;
127
128 /* Bound the whole bring-up (connect + handshake + TLS + login) with the SHORT
129 * connect timeout, not the long steady-state I/O timeout: a firewall that
130 * completes the TCP handshake then black-holes the protocol bytes must fail
131 * promptly so the reconnect machinery can ride over it, rather than hanging
132 * the caller. The steady-state timeout is restored once the session is up. */
133 c->io.timeout_ms = brix_tmo_connect_ms();
134
135 c->diag.phase_ns[0] = brix_mono_ns(); /* §15.3: connect-phase breakdown */
136 c->io.fd = brix_tcp_connect(c->host, c->port, c->io.timeout_ms, st);
137 if (c->io.fd < 0) {
138 return -1;
139 }
140 c->diag.phase_ns[1] = brix_mono_ns(); /* tcp connected */
141
142 /* Reserve streamid 1 for the protocol request; subsequent ops start at 2. */
143 c->next_sid = 2;
144 if (brix_conn_handshake(c, 1, c->want_tls, st) != 0) {
145 brix_close(c);
146 return -1;
147 }
148
149 /* TLS decision — never silently downgrade. */
150 {
151 int have_tls = (c->server_flags & kXR_haveTLS) != 0;
152 int goto_tls = (c->server_flags & kXR_gotoTLS) != 0;
153 if (goto_tls || (c->want_tls && have_tls)) {
154 const char *ca = brix_resolve_ca_dir(c->opts.ca_dir);
155 if (brix_tls_upgrade(c, !c->opts.insecure_tls, c->opts.verify_host, ca, st) != 0) {
156 brix_close(c);
157 return -1;
158 }
159 } else if (c->want_tls && !have_tls) {
160 if (c->tls_strict || !c->opts.notlsok) {
161 brix_status_set(st, XRDC_EAUTH, 0,
162 "server offers no TLS; refusing cleartext "
163 "(use --notlsok with root:// to override)");
164 brix_close(c);
165 return -1;
166 }
167 /* root:// + --notlsok: proceed cleartext. */
168 }
169 }
170 c->diag.phase_ns[2] = brix_mono_ns(); /* tls negotiated (==tcp if cleartext) */
171
172 if (want_login && brix_conn_login(c, &c->opts, st) != 0) {
173 brix_close(c);
174 return -1;
175 }
176 c->diag.phase_ns[3] = brix_mono_ns(); /* login + auth done */
177
178 /* Session is up: switch from the short bring-up cap to the steady-state I/O
179 * timeout so a legitimately long read/write is not cut off. Also arm the
180 * opt-in slow-drip completion deadline (0 = disabled) now that bulk reads
181 * begin — a dribbling peer cannot hold a steady-state read open past it. */
182 c->io.timeout_ms = brix_tmo_io_ms();
183 c->io.stall_deadline_ms = brix_tmo_stall_ms();
184 return 0;
185}

191 return brix_bringup_ex(c, 1, st);

194int
195brix_bind(brix_conn *sec, const brix_conn *primary, brix_status *st)
196{
197 ClientBindRequest req;
198 uint16_t sid, status;
199 uint8_t *body = NULL;
200 uint32_t blen = 0;
201
202 /* A secondary stream re-runs handshake + kXR_protocol [+ TLS] against the
203 * SAME target but SKIPS kXR_login; the server inherits identity from the
204 * primary's session via kXR_bind{sessid}. (src/protocols/root/session/bind.c) */
205 memset(sec, 0, sizeof(*sec));
206 sec->io.timeout_ms = primary->io.timeout_ms;
207 sec->io.stall_deadline_ms = primary->io.stall_deadline_ms;
208 sec->opts = primary->opts;
209 sec->want_tls = primary->want_tls;
210 sec->tls_strict = primary->tls_strict;
211 snprintf(sec->host, sizeof(sec->host), "%s", primary->host);
212 sec->port = primary->port;
213
214 if (brix_bringup_ex(sec, 0, st) != 0) {
215 return -1;
216 }
217
218 memset(&req, 0, sizeof(req));
219 req.requestid = htons(kXR_bind);
220 {
221 xrdw_sessid_req_t b;
222 memcpy(b.sessid, primary->sessid, BRIX_SESSION_ID_LEN);
223 xrdw_sessid_req_pack(&b, ((ClientRequestHdr *) &req)->body);
224 }
225
226 {
227 brix_resp_out out = { &status, &body, &blen };
228 if (brix_send(sec, &req, NULL, &sid, st) != 0
229 || brix_recv(sec, sid, &out, st) != 0) {
230 /* Quiet teardown: a bound stream has no session of its own to end. */
231 brix_tls_free(sec);
232 if (sec->io.fd >= 0) { close(sec->io.fd); sec->io.fd = -1; }
233 return -1;
234 }
235 }
236 free(body); /* reply body = 1-byte pathid (server bookkeeping) */
237 return 0;
238}

258int
259brix_connect(brix_conn *c, const brix_url *u, const brix_opts *o, brix_status *st)
260{
261 brix_crypto_once();
262 memset(c, 0, sizeof(*c));
263 c->io.timeout_ms = brix_tmo_io_ms(); /* steady-state; bring-up uses the short cap */
264
265 if (o != NULL) {
266 c->opts = *o;
267 } else {
268 c->opts.verify_host = 1;
269 }
270
271 if (u->scheme != XRDC_SCHEME_ROOT && u->scheme != XRDC_SCHEME_ROOTS) {
272 brix_status_set(st, XRDC_EUSAGE, 0,
273 "native client speaks root:// / roots:// only (scheme %d)",
274 (int) u->scheme);
275 return -1;
276 }
277
278 snprintf(c->host, sizeof(c->host), "%s", u->host);
279 c->port = u->port;
280 /* Phase 40 (a): remember the ORIGINAL endpoint (the manager) so a dead
281 * redirect target can fall back here for a fresh server selection. Set once
282 * at connect; brix_reconnect deliberately does NOT touch it. */
283 snprintf(c->home_host, sizeof(c->home_host), "%s", u->host);
284 c->home_port = u->port;
285 /* WS-3: carry the single-slash flag forward so per-op error sites can fire
286 * the double-slash hint without re-parsing the original URL. */
287 c->single_slash_path = u->single_slash_path;
288 c->tls_strict = (u->scheme == XRDC_SCHEME_ROOTS);
289 c->want_tls = c->tls_strict || c->opts.want_tls;
290
291 /* §15.1: open the capture sink ONCE (here, not in bringup — so a redirect's
292 * reconnect appends rather than truncates). Frames are recorded by frame.c. */
293 if (c->opts.capture != NULL && c->opts.capture[0] != '\0') {
294 char ep[320];
295 c->diag.cap = brix_capture_open(c->opts.capture);
296 brix_format_host_port(c->host, (uint16_t) c->port, ep, sizeof(ep));
297 brix_capture_meta(c->diag.cap, "endpoint", ep);
298 }
299
300 {
301 int rc = brix_bringup(c, st);
302 if (rc == 0 && c->diag.cap != NULL) { /* snapshot negotiated session */
303 char buf[64], sx[2 * BRIX_SESSION_ID_LEN + 1];
304 int i;
305 snprintf(buf, sizeof(buf), "0x%x", (unsigned) c->server_flags);
306 brix_capture_meta(c->diag.cap, "caps", buf);
307 brix_capture_meta(c->diag.cap, "auth",
308 c->diag.chosen_auth ? c->diag.chosen_auth : "anon");
309 brix_capture_meta(c->diag.cap, "seclist",
310 c->sec_list[0] ? c->sec_list : "(none)");
311 for (i = 0; i < BRIX_SESSION_ID_LEN; i++) {
312 snprintf(sx + i * 2, 3, "%02x", c->sessid[i]);
313 }
314 brix_capture_meta(c->diag.cap, "sessid", sx);
315 }
316 return rc;
317 }
318}

321brix_connect_no_login(brix_conn *c, const brix_url *u, const brix_opts *o,
322 brix_status *st)
323{
324 brix_crypto_once();
325 memset(c, 0, sizeof(*c));
326 c->io.timeout_ms = brix_tmo_io_ms(); /* steady-state; bring-up uses the short cap */
327 if (o != NULL) { c->opts = *o; } else { c->opts.verify_host = 1; }
328
329 if (u->scheme != XRDC_SCHEME_ROOT && u->scheme != XRDC_SCHEME_ROOTS) {
330 brix_status_set(st, XRDC_EUSAGE, 0,
331 "native client speaks root:// / roots:// only");
332 return -1;
333 }
334 snprintf(c->host, sizeof(c->host), "%s", u->host);
335 c->port = u->port;
336 snprintf(c->home_host, sizeof(c->home_host), "%s", u->host);
337 c->home_port = u->port;
338 c->single_slash_path = u->single_slash_path; /* WS-3: propagate for hint */
339 c->tls_strict = (u->scheme == XRDC_SCHEME_ROOTS);
340 /* TLS per the scheme: roots:// negotiates TLS (cert presented); a plain root://
341 * server that does not mandate TLS stays cleartext (then there is simply no cert,
342 * rather than a rejected wantTLS handshake). No kXR_login: cert inspection needs
343 * no identity, so this works even where anon login would be rejected. A server
344 * that sends kXR_gotoTLS still upgrades regardless. */
345 c->want_tls = c->tls_strict;
346 c->opts.notlsok = 1;
347 return brix_bringup_ex(c, 0 /*want_login*/, st);
348}

350int
351brix_reconnect(brix_conn *c, const char *host, int port, brix_status *st)
352{
353 /* Abandon the current transport (no endsess — we are leaving this server) but
354 * keep opts/want_tls/redirect-state, then re-establish against the new target. */
355 brix_tls_free(c);
356 if (c->io.fd >= 0) {
357 close(c->io.fd);
358 c->io.fd = -1;
359 }
360 snprintf(c->host, sizeof(c->host), "%s", host);
361 c->port = port;
362 return brix_bringup(c, st);
363}

379void
380brix_close(brix_conn *c)
381{
382 if (c != NULL && c->diag.timing) { /* §15: one summary per session at exit */
383 brix_timing_report(c);
384 }
385 if (c != NULL && c->diag.cap != NULL) { /* §15.1: flush + close the capture */
386 brix_capture_close(c->diag.cap);
387 c->diag.cap = NULL; /* idempotent: safe on double close */
388 }
389 if (c == NULL || c->io.fd < 0) {
390 return;
391 }
392
393 /* Best-effort graceful end-of-session: kXR_endsess{sessid[16]}, FIRE-AND-
394 * FORGET. We deliberately do NOT wait for the reply: against a black-holing
395 * peer (a misbehaving inline firewall) reading a reply that never comes would
396 * stall teardown for a full timeout, and the server tears the session down on
397 * socket close regardless. Skip it entirely when no session was established
398 * (sessid still zero — e.g. a connection that failed during handshake), and
399 * cap the send so even a wedged socket cannot block the close. */
400 if (!sessid_is_zero(c->sessid)) {
401 uint8_t req[XRD_REQUEST_HDR_LEN];
402 uint16_t sid;
403 brix_status throwaway;
404
405 if (c->io.timeout_ms <= 0 || c->io.timeout_ms > 2000) {
406 c->io.timeout_ms = 2000; /* teardown send must not hang */
407 }
408 memset(req, 0, sizeof(req));
409 req[2] = (uint8_t) (kXR_endsess >> 8);
410 req[3] = (uint8_t) (kXR_endsess & 0xff);
411 {
412 xrdw_sessid_req_t b; /* body[16] = sessid (shared codec) */
413 memcpy(b.sessid, c->sessid, BRIX_SESSION_ID_LEN);
414 xrdw_sessid_req_pack(&b, req + 4);
415 }
416 brix_status_clear(&throwaway);
417 (void) brix_send(c, req, NULL, &sid, &throwaway);
418 }
419
420 brix_tls_free(c); /* SSL_shutdown/free + SSL_CTX_free (no-op if none) */
421 close(c->io.fd);
422 c->io.fd = -1;
423 OPENSSL_cleanse(c->gsi_deleg_key, sizeof(c->gsi_deleg_key));
424 c->gsi_deleg_keylen = 0;
425 c->gsi_deleg_ready = 0;
426 c->signing_active = 0;
427}
