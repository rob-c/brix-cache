52/* Count non-overlapping occurrences of needle in hay. */
53static int
54count(const char *hay, const char *needle)
55{
56 int n = 0;
57 const char *p = hay;
58 size_t nl = strlen(needle);
59
60 while ((p = strstr(p, needle)) != NULL) { n++; p += nl; }
61 return n;
62}

64/* A throwaway self-signed CN=nobody certificate, optionally carrying `ext_der`
65 * as its VOMS extension. Never the fixture AC's holder. */
66static X509 *
67make_cert(const unsigned char *ext_der, int ext_len)
68{
69 EVP_PKEY *key = EVP_RSA_gen(2048);
70 X509 *cert = X509_new();
71 X509_NAME *name = X509_NAME_new();
72
73 CHECK(key != NULL && cert != NULL && name != NULL);
74 X509_NAME_add_entry_by_txt(name, "CN", MBSTRING_ASC,
75 (const unsigned char *) "nobody", -1, -1, 0);
76 X509_set_version(cert, 2);
77 X509_set_subject_name(cert, name);
78 X509_set_issuer_name(cert, name);
79 ASN1_INTEGER_set(X509_get_serialNumber(cert), 1);
80 X509_gmtime_adj(X509_getm_notBefore(cert), -3600);
81 X509_gmtime_adj(X509_getm_notAfter(cert), 3600);
82 X509_set_pubkey(cert, key);
83 if (ext_der != NULL) {
84 ASN1_OCTET_STRING *data = ASN1_OCTET_STRING_new();
85 ASN1_OBJECT *obj = OBJ_txt2obj(BRIX_VOMS_OID_ACSEQ, 1);
86 X509_EXTENSION *ext;
87
88 ASN1_OCTET_STRING_set(data, ext_der, ext_len);
89 ext = X509_EXTENSION_create_by_OBJ(NULL, obj, 0, data);
90 CHECK(ext != NULL && X509_add_ext(cert, ext, -1) == 1);
91 X509_EXTENSION_free(ext);
92 ASN1_OBJECT_free(obj);
93 ASN1_OCTET_STRING_free(data);
94 }
95 CHECK(X509_sign(cert, key, EVP_sha256()) > 0);
96 X509_NAME_free(name);
97 EVP_PKEY_free(key);
98 return cert;
99}

114/* The decoded-facts pins every fixture rendering must satisfy. */
115static void
116check_fixture_facts(const char *s)
117{
118 CHECK(count(s, FQAN_USER) == 1);
119 CHECK(count(s, FQAN_NULL) == 1);
120 CHECK(count(s, " VOMS: /") == 2); /* exactly two FQAN lines */
121 CHECK(count(s, " VOMS: vo=lhcb server=voms-lhcb-auth.cern.ch:443 "
122 "issuer=/DC=ch/DC=cern/OU=computers/CN=lhcb-auth.cern.ch "
123 "valid 2026-08-03T10:34:22Z..2026-08-10T10:34:22Z\n") == 1);
124 /* none of the blind-scan noise the old byte parser emitted */
125 CHECK(!has(s, "cafiles.cern.ch")); /* signer CRL/AIA URI */
126 CHECK(!has(s, "ocsp.cern.ch")); /* signer OCSP URI */
127 CHECK(!has(s, "ldap:")); /* signer LDAP CRL URI */
128 CHECK(!has(s, "Capability=NULL0")); /* trailing tag byte */
129 CHECK(!has(s, ":4430")); /* URI port over-read */
130 CHECK(!has(s, "undecodable"));
131 CHECK(!has(s, "VOMS: none"));
132}

134/* Success: the fixture on the leaf, no trust dirs → decoded + "unverified". */
135static void
136test_leaf_decodes(X509 *with_ext)
137{
138 char *s;
139
140 setenv("X509_CERT_DIR", "/nonexistent/certificates", 1);
141 setenv("X509_VOMS_DIR", "/nonexistent/vomsdir", 1);
142 s = render(with_ext, NULL);
143 check_fixture_facts(s);
144 CHECK(count(s, " VOMS: unverified (no /nonexistent/certificates)\n") == 1);
145 CHECK(count(s, " VOMS: ") == 4); /* 2 FQAN + summary + verdict */
146 CHECK(!has(s, " VOMS: verified\n"));
147 CHECK(!has(s, "NOT verified"));
148 free(s);
149}

189/* Security-negative: trust directories present, but the certificate is not
190 * the AC's holder → "NOT verified: ..." — the verdict is never "verified". */
191static void
192test_not_holder_is_rejected(X509 *with_ext)
193{
194 const char *tmp = getenv("TMPDIR");
195 char cert_dir[256], voms_dir[256];
196 char *s;
197
198 snprintf(cert_dir, sizeof(cert_dir), "%s/credinfo_voms_certs_XXXXXX",
199 (tmp != NULL && tmp[0] != '\0') ? tmp : "/tmp");
200 snprintf(voms_dir, sizeof(voms_dir), "%s/credinfo_voms_vomsdir_XXXXXX",
201 (tmp != NULL && tmp[0] != '\0') ? tmp : "/tmp");
202 CHECK(mkdtemp(cert_dir) != NULL && mkdtemp(voms_dir) != NULL);
203 setenv("X509_CERT_DIR", cert_dir, 1);
204 setenv("X509_VOMS_DIR", voms_dir, 1);
205 s = render(with_ext, NULL);
206 check_fixture_facts(s);
207 CHECK(count(s, " VOMS: NOT verified: ") == 1);
208 CHECK(!has(s, " VOMS: verified\n"));
209 CHECK(!has(s, "unverified"));
210 free(s);
211 rmdir(cert_dir);
212 rmdir(voms_dir);
213}
