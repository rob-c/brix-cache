9/*
10 * ftp_ev_sec.c — RFC 2228 GSI security handshake on the control channel:
11 * AUTH GSSAPI, the ADAT token exchange, and the MIC/CONF/ENC protected-command
12 * unwrap.
13 *
14 * WHAT: stand up the mem-BIO GSSAPI engine on AUTH, feed each ADAT token through
15 * it until the handshake completes, then bind the verified proxy DN into a
16 * brix_identity_t and switch the control channel to GSS-wrapped replies.
17 *
18 * WHY: the security handshake is the one part of the protocol that is *already*
19 * event-compatible — brix_gssapi_srv_step() is token-driven, not socket-driven:
20 * each ADAT command line carries one handshake token and the reply carries the
21 * next, so no inline TLS I/O is needed on the control channel. That lets the
22 * event engine reach GSI parity here with no state-machine surgery; only the
23 * PROT P *data* channel needs event-driven TLS (P82.3).
24 *
25 * HOW: ported from the sync engine's ftp_cmd_auth/adat/protected + ftp_gss_finalize,
26 * writing through the event reply queue. The pre-auth handshake replies
27 * (334/335/235) go out cleartext because ->sec_active is still 0; finalize flips
28 * it so every subsequent reply is wrapped.
29 */

32/* Lift the client proxy's VOMS FQANs into the session identity so an authorized
33 * VO can *satisfy* a brix_gridftp_require_vo rule (not merely be denied by it),
34 * exactly as webdav_extract_and_set_voms_identity() does on the HTTPS plane. No
35 * vomsdir/voms_cert_dir (or no VOMS build) ⇒ no carry, so a require_vo export
36 * stays fail-closed: a proxy's VOMS AC is ignored and the gate denies. An absent
37 * VOMS extension on the proxy is normal (plain GSI) and non-fatal. */
38static void
39ev_gss_carry_voms(ftp_ev_t *fc)
40{
41 X509 *leaf = NULL;
42 STACK_OF(X509) *chain = NULL;
43 char primary_vo[256] = "";
44 char vo_list[1024] = "";
45
46 if (fc->identity == NULL
47 || fc->conf->common.vomsdir.len == 0 || fc->conf->common.voms_cert_dir.len == 0
48 || !brix_voms_available())
49 {
50 return;
51 }
52 if (brix_gssapi_srv_peer_x509(fc->gss, &leaf, &chain) != NGX_OK) {
53 return;
54 }
55 (void) brix_extract_voms_info(fc->c->log, leaf, chain,
56 &fc->conf->common.vomsdir, &fc->conf->common.voms_cert_dir,
57 primary_vo, sizeof(primary_vo),
58 vo_list, sizeof(vo_list));
59 if (vo_list[0] != '\0'
60 && brix_identity_set_vos_csv(fc->identity, fc->c->pool, vo_list)
61 == NGX_OK)
62 {
63 BRIX_LOG(NGX_LOG_INFO, fc->c->log, 0,
64 "brix: GridFTP(ev) VOMS carry primary_vo=\"%s\"",
65 primary_vo);
66 }
67 X509_free(leaf);
68}

71/* After AUTH GSSAPI completes, bind the verified proxy DN into a brix_identity_t
72 * (so the VFS ctx and audit see a real GSI principal), capture any delegated
73 * credential, and activate the wrapped control channel. */
74static ngx_int_t
75ev_gss_finalize(ftp_ev_t *fc)
76{
77 ngx_str_t dn = { 0, NULL }, proxy = { 0, NULL };
78 brix_identity_t *id;
79 char *dn_text;
80
81 if (brix_gssapi_srv_peer(fc->gss, &dn, &proxy) != NGX_OK || dn.len == 0) {
82 brix_metric_auth(BRIX_PROTO_GRIDFTP, BRIX_AUTHN_GSI, 0);
83 BRIX_LOG(NGX_LOG_WARN, fc->c->log, 0,
84 "brix: GridFTP(ev) GSI handshake produced no identity");
85 return NGX_ERROR;
86 }
87
88 id = brix_identity_alloc(fc->c->pool);
89 dn_text = ngx_pnalloc(fc->c->pool, dn.len + 1);
90 if (id == NULL || dn_text == NULL) {
91 brix_metric_auth(BRIX_PROTO_GRIDFTP, BRIX_AUTHN_GSI, 0);
92 BRIX_LOG(NGX_LOG_ERR, fc->c->log, 0,
93 "brix: GridFTP(ev) cannot allocate GSI identity");
94 return NGX_ERROR;
95 }
96 ngx_memcpy(dn_text, dn.data, dn.len);
97 dn_text[dn.len] = '\0';
98 if (brix_identity_set_dn(id, fc->c->pool, dn_text, BRIX_AUTHN_GSI)
99 != NGX_OK)
100 {
101 brix_metric_auth(BRIX_PROTO_GRIDFTP, BRIX_AUTHN_GSI, 0);
102 BRIX_LOG(NGX_LOG_ERR, fc->c->log, 0,
103 "brix: GridFTP(ev) cannot materialize GSI identity");
104 return NGX_ERROR;
105 }
106
107 fc->identity = id;
108 fc->deleg_proxy = proxy;
109 fc->ctrl_dn.data = ngx_pnalloc(fc->c->pool, dn.len);
110 if (fc->ctrl_dn.data != NULL) {
111 ngx_memcpy(fc->ctrl_dn.data, dn.data, dn.len);
112 fc->ctrl_dn.len = dn.len;
113 }
114 /* the client's control leaf is the delegated proxy's direct issuer;
115 * needed to present a complete chain on a PROT P data channel. */
116 (void) brix_gssapi_srv_peer_cert_pem(fc->gss, &fc->ctrl_leaf_pem);
117 ev_gss_carry_voms(fc); /* lift VOMS FQANs into fc->identity */
118 fc->authed = 1;
119 brix_metric_auth(BRIX_PROTO_GRIDFTP, BRIX_AUTHN_GSI, 1);
120 BRIX_LOG(NGX_LOG_INFO, fc->c->log, 0,
121 "brix: GridFTP(ev) gsiftp authenticated dn=\"%V\" deleg=%uz",
122 &dn, proxy.len);
123 return NGX_OK;
124}

127/* AUTH GSSAPI: stand up the mem-BIO engine and invite ADAT tokens (334). */
128ngx_int_t
129brix_ftp_ev_cmd_auth(ftp_ev_t *fc, const char *arg)
130{
131 if (!fc->conf->gsi || fc->conf->tls_ctx == NULL
132 || fc->conf->tls_ctx->ctx == NULL)
133 {
134 return brix_ftp_ev_reply(fc, "534 Security mechanism not available\r\n");
135 }
136 if (strcasecmp(arg, "GSSAPI") != 0) {
137 return brix_ftp_ev_reply(fc, "504 Unknown security mechanism %s\r\n", arg);
138 }
139 if (fc->gss != NULL) {
140 return brix_ftp_ev_reply(fc,
141 "534 Security handshake already in progress\r\n");
142 }
143 fc->gss = brix_gssapi_srv_create(fc->c->pool, fc->c->log,
144 fc->conf->tls_ctx->ctx, fc->conf->ca_store, 1 /* accept delegation */);
145 if (fc->gss == NULL) {
146 return brix_ftp_ev_reply(fc, "431 GSSAPI context init failed\r\n");
147 }
148 return brix_ftp_ev_reply(fc,
149 "334 Using authentication type GSSAPI; ADAT must follow\r\n");
150}

153/* ADAT: feed one decoded token to the engine and reply 335 (continue) or 235
154 * (complete, then switch on the wrapped control channel). */
155ngx_int_t
156brix_ftp_ev_cmd_adat(ftp_ev_t *fc, const char *arg)
157{
158 ngx_str_t tok, out;
159 brix_gss_status_e st;
160
161 if (fc->gss == NULL) {
162 return brix_ftp_ev_reply(fc, "503 Send AUTH GSSAPI first\r\n");
163 }
164 if (brix_ftp_ev_b64_decode(fc->c->pool, arg, &tok) != NGX_OK) {
165 return brix_ftp_ev_reply(fc, "501 Malformed ADAT token\r\n");
166 }
167
168 st = brix_gssapi_srv_step(fc->gss, tok.data, tok.len, &out);
169 if (st == BRIX_GSS_FAILED) {
170 brix_metric_auth(BRIX_PROTO_GRIDFTP, BRIX_AUTHN_GSI, 0);
171 return brix_ftp_ev_reply(fc, "535 GSSAPI authentication failed\r\n");
172 }
173 if (st == BRIX_GSS_CONTINUE) {
174 return brix_ftp_ev_send_adat(fc, 335, &out);
175 }
176
177 /* COMPLETE: materialize the authenticated principal before reporting
178 * success, emit the final 235 cleartext, THEN activate wrapping. A final
179 * token is framed "235 ADAT=<b64>"; with none, send a bare completion line
180 * (an empty "ADAT=" would decode to a zero-length token and fail the client). */
181 if (ev_gss_finalize(fc) != NGX_OK) {
182 return brix_ftp_ev_reply(fc,
183 "535 GSSAPI authentication produced no usable identity\r\n");
184 }
185 if (out.len > 0) {
186 if (brix_ftp_ev_send_adat(fc, 235, &out) != NGX_OK) {
187 return NGX_ERROR;
188 }
189 } else if (brix_ftp_ev_reply(fc, "235 GSSAPI authentication succeeded\r\n")
190 != NGX_OK)
191 {
192 return NGX_ERROR;
193 }
194 fc->wrap_code = "633"; /* default: private (ENC) replies */
195 fc->sec_active = 1;
196 return NGX_OK;
197}

200/* MIC/CONF/ENC: unwrap a protected command and re-dispatch its plaintext. The
201 * safety code chosen here also frames the reply(ies) that command emits. */
202ngx_int_t
203brix_ftp_ev_cmd_protected(ftp_ev_t *fc, const char *code, const char *arg)
204{
205 ngx_str_t tok, inner;
206 char *innerline;
207 size_t n;
208
209 if (fc->gss == NULL || !fc->sec_active) {
210 return brix_ftp_ev_reply(fc, "503 Not authenticated\r\n");
211 }
212 fc->wrap_code = code;
213 if (brix_ftp_ev_b64_decode(fc->c->pool, arg, &tok) != NGX_OK) {
214 return brix_ftp_ev_reply(fc, "501 Malformed protected token\r\n");
215 }
216 if (brix_gssapi_unwrap(fc->gss, tok.data, tok.len, &inner) != NGX_OK
217 || inner.len == 0)
218 {
219 return brix_ftp_ev_reply(fc, "535 Cannot decode protected command\r\n");
220 }
221 innerline = ngx_pnalloc(fc->c->pool, BRIX_FTP_EV_CMD_MAX);
222 if (innerline == NULL) {
223 return NGX_ERROR;
224 }
225 n = ngx_min(inner.len, (size_t) BRIX_FTP_EV_CMD_MAX - 1);
226 ngx_memcpy(innerline, inner.data, n);
227 while (n > 0 && (innerline[n - 1] == '\r' || innerline[n - 1] == '\n')) {
228 n--;
229 }
230 innerline[n] = '\0';
231 return brix_ftp_ev_dispatch_protected(fc, innerline);
232}
