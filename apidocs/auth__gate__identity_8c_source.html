12static void
13authz_resolve_fqan_name(brix_identity_t *id)
14{
15 ngx_uint_t k;
16 ngx_str_t *fqans;
17 char fqan[512];
18
19 fqans = id->vo_list != NULL ? id->vo_list->elts : NULL;
20 for (k = 0; fqans != NULL && k < id->vo_list->nelts; k++) {
21 if (brix_str_cbuf(fqan, sizeof(fqan), &fqans[k]) != NULL
22 && brix_idmap_gate_username(fqan, id->mapped_user,
23 sizeof(id->mapped_user)))
24 {
25 return;
26 }
27 id->mapped_user[0] = '\0';
28 }
29}

31static void
32authz_resolve_mapped_name(brix_identity_t *id, const char *dn)
33{
34 id->mapped_resolved = 1;
35 id->mapped_user[0] = '\0';
36 authz_resolve_fqan_name(id);
37 if (id->mapped_user[0] == '\0' && dn != NULL && dn[0] != '\0') {
38 if (!brix_idmap_gate_username(dn, id->mapped_user,
39 sizeof(id->mapped_user)))
40 {
41 id->mapped_user[0] = '\0';
42 }
43 }
44}

46const char *
47brix_authz_mapped_name(brix_identity_t *id, const char *dn)
48{
49 if (id == NULL || !brix_idmap_gate_enabled()) {
50 return (dn != NULL) ? dn : "";
51 }
52 if (!id->mapped_resolved) {
53 authz_resolve_mapped_name(id, dn);
54 }
55 return id->mapped_user[0] != '\0' ? id->mapped_user
56 : (dn != NULL ? dn : "");
57}

84static ngx_int_t
85authz_identity_primary(const brix_authz_identity_query_t *q)
86{
87 if (q->acc_format == BRIX_AUTHDB_FORMAT_XRDACC) {
88 brix_acc_entity_t *ent = q->acc_entity;
89
90 if (ent == NULL) {
91 ent = brix_authz_acc_entity(q->pool, q->identity, q->peer_ip);
92 }
93
94 if (q->acc_tables == NULL || ent == NULL) {
95 return NGX_ERROR;
96 }
97 return brix_acc_access((brix_acc_tables_t *) q->acc_tables, ent,
98 q->logical_path, q->acc_op) != BRIX_ACC_PRIV_NONE
99 ? NGX_OK : NGX_ERROR;
100 }
101
102 {

104 .rules = q->authdb_rules,
105 .identity = q->identity,
106 .peer_ip = q->peer_ip,
107 .resolved_path = q->resolved_path,
108 .needed_privs = q->needed_privs,
109 };
