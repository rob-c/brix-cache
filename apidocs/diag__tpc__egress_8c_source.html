1/*
2 * diag_tpc_egress.c — TPC egress (SSRF-control) self-test.
3 *
4 * `xrddiag tpc-egress <your-gateway-url> --tpc-target host[:port]`
5 *
6 * Points at YOUR OWN root:// gateway and asks it to originate a third-party-copy
7 * *pull* from the named source. In the TPC-pull model the destination server —
8 * your gateway — is the party that dials the source host, so this is precisely
9 * the surface an attacker abuses for SSRF: coerce the gateway into connecting to
10 * an internal address it would never expose directly.
11 *
12 * The self-test issues only the destination *arm* (a cgiC2Dst kXR_open naming the
13 * target as tpc.src) and, if the arm is accepted, a single bounded trigger. It
14 * classifies the outcome:
15 *
16 * REFUSED (policy) the gateway's egress guard declined to originate → SAFE
17 * conn-refused egress permitted; the source port answered RST → RISK
18 * filtered/timeout egress permitted; the source stayed silent → RISK
19 * reached (no lfn) egress permitted; the source is up, lfn absent → RISK
20 * reached (error) egress permitted; the source answered an error → RISK
21 * ACCEPTED egress permitted; the pull ran to completion → RISK
22 *
23 * Read-only against the source (a self-test lfn that will not exist) and strictly
24 * PII-free in its report: verdict / kXR code / milliseconds / the operator-named
25 * target only — never a resolved address, a path body, or a credential.
26 *
27 * Pure composition of the public libbrix client API — no new wire.
28 */

43/* The self-test source path. Deliberately implausible so a *reachable* source
44 * answers "no such file" fast instead of streaming real data back to us. */
45#define TPCE_SELFTEST_LFN "/.brix-egress-selftest-nonexistent"

47/* Default trigger budget when --probe-timeout is not given (ms). Short: a live
48 * pull we do not want; we only need long enough to tell RST from a dropped SYN. */
49#define TPCE_DEFAULT_BUDGET_MS 4000
50
51/* A returned outbound-connect failure faster than this reflects an active
52 * refusal (an RST came straight back); slower means the gateway sat through its
53 * own connect timeout on a dropped SYN — i.e. the source is filtered. */
54#define TPCE_FAST_FAIL_MS 1000.0

58tpce_mono_ms(void)

66/* Case-insensitive substring test (needle assumed lowercase). */
67static int
68tpce_has(const char *hay, const char *needle)
69{
70 size_t nlen = strlen(needle);
71 size_t i;
72
73 if (hay == NULL) {
74 return 0;
75 }
76 for (; *hay != '\0'; hay++) {
77 for (i = 0; i < nlen; i++) {
78 if (tolower((unsigned char) hay[i]) != needle[i]) {
79 break;
80 }
81 }
82 if (i == nlen) {
83 return 1;
84 }
85 }
86 return 0;
87}

90/*
91 * A kXR_NotAuthorized from the arm is only an *egress* refusal when its text
92 * names the source/egress policy — the gateway also returns NotAuthorized for
93 * unrelated denials (read-only export, token scope), which must NOT read as a
94 * working SSRF guard. Matches the address-range gate ("prohibited"/"private"/
95 * "loopback") and the host-allowlist guard ("source host not permitted").
96 */
97static int
98tpce_msg_is_egress_denial(const char *msg)
99{
100 return tpce_has(msg, "source")
101 || tpce_has(msg, "egress")
102 || tpce_has(msg, "not permitted")
103 || tpce_has(msg, "not allowed")
104 || tpce_has(msg, "prohibited")
105 || tpce_has(msg, "private")
106 || tpce_has(msg, "loopback")
107 || tpce_has(msg, "ssrf");
108}

122/*
123 * tpce_classify_trigger — map the trigger (sync) outcome to a verdict. Public so
124 * the offline unit suite can prove every branch without a live gateway. `st` is
125 * the status the trigger returned (rc==0 → st is success), `elapsed_ms` the
126 * measured wall time, `budget_ms` the timeout we imposed.
127 */
128tpce_verdict
129tpce_classify_trigger(const brix_status *st, double elapsed_ms,
130 double budget_ms, char *detail, size_t dsz)
131{
132 /* Success: the gateway connected to the source and completed the pull. */
133 if (st->kxr == 0) {
134 snprintf(detail, dsz, "pull completed — full egress + transfer");
135 return TPCE_ACCEPTED;
136 }
137 /* Explicit reset — the source port answered, reachable but not serving. */
138 if (st->sys_errno == ECONNREFUSED || tpce_has(st->msg, "refused")) {
139 snprintf(detail, dsz, "source port answered RST (connection refused)");
140 return TPCE_CONN_REFUSED;
141 }
142 /* Our bounded wait elapsed with no reply, or an explicit timeout: the pull
143 * is still hanging server-side because the SYN was dropped. */
144 if (st->sys_errno == ETIMEDOUT || tpce_has(st->msg, "timed out")
145 || tpce_has(st->msg, "timeout") || elapsed_ms >= budget_ms * 0.9) {
146 snprintf(detail, dsz, "source silent — SYN dropped (filtered)");
147 return TPCE_FILTERED;
148 }
149 /* The gateway reported an outbound connect failure (its text carries no
150 * errno): a fast reply is an active refusal, a slow one is the gateway's own
151 * connect timeout expiring on a dropped SYN. */
152 if (tpce_has(st->msg, "connect") && tpce_has(st->msg, "fail")) {
153 if (elapsed_ms < TPCE_FAST_FAIL_MS) {
154 snprintf(detail, dsz, "gateway's outbound connect failed fast "
155 "(source refused the connection)");
156 return TPCE_CONN_REFUSED;
157 }
158 snprintf(detail, dsz, "gateway's outbound connect timed out "
159 "(source filtered)");
160 return TPCE_FILTERED;
161 }
162 /* The source is up and spoke XRootD back: the self-test lfn is absent. */
163 if (st->kxr == kXR_NotFound || tpce_has(st->msg, "not found")
164 || tpce_has(st->msg, "no such") || tpce_has(st->msg, "does not exist")) {
165 snprintf(detail, dsz, "source reachable — self-test lfn absent (expected)");
166 return TPCE_REACHED_NOENT;
167 }
168 snprintf(detail, dsz, "source reachable — answered with an error");
169 return TPCE_REACHED_ERROR;
170}

173/*
174 * Drive the destination through the two-sync TPC rendezvous and classify. The
175 * server arms on the FIRST kXR_sync (returns OK immediately, sets tpc_armed) and
176 * only originates the outbound pull on the SECOND — see brix_handle_sync /
177 * brix_tpc_start_pull. So the arming sync is issued and discarded; the trigger
178 * sync, run under the raised budget, is the one whose outcome names the egress
179 * decision (its reply is deferred via kXR_waitresp until the pull settles).
180 */
181static void
182tpce_trigger(brix_conn *c, brix_file *f, double budget_ms, tpce_result *r)
183{
184 brix_status st;
185 double t0;
186
187 brix_status_clear(&st);
188 (void) brix_file_sync(c, f, &st); /* sync #1: arm (immediate OK) */
189
190 brix_status_clear(&st);
191 if (c->io.timeout_ms < (int) budget_ms) {
192 c->io.timeout_ms = (int) budget_ms;
193 }
194 t0 = tpce_mono_ms();
195 (void) brix_file_sync(c, f, &st); /* sync #2: trigger the outbound pull */
196 r->trig_ms = tpce_mono_ms() - t0;
197 r->trig_kxr = st.kxr;
198 r->verdict = tpce_classify_trigger(&st, r->trig_ms, budget_ms,
199 r->detail, sizeof(r->detail));
200}

203/* Interpret the arm (destination open) result and drive the trigger when the
204 * gateway agreed to originate. Returns with r->verdict + r->egress_permitted set. */
205static void
206tpce_after_arm(brix_conn *c, brix_file *f, int arm_rc, const brix_status *arm_st,
207 double budget_ms, tpce_result *r)
208{
209 r->arm_kxr = arm_st->kxr;
210
211 if (arm_rc == 0) {
212 r->egress_permitted = 1;
213 tpce_trigger(c, f, budget_ms, r);
214 return;
215 }
216 if (arm_st->kxr == kXR_NotAuthorized && tpce_msg_is_egress_denial(arm_st->msg)) {
217 r->verdict = TPCE_REFUSED_POLICY;
218 snprintf(r->detail, sizeof(r->detail),
219 "gateway egress guard declined to originate");
220 return;
221 }
222 r->verdict = TPCE_ARM_ERROR;
223 snprintf(r->detail, sizeof(r->detail),
224 "gateway refused the arm (kXR=%d) — not an egress-policy denial",
225 arm_st->kxr);
226}

229/*
230 * tpce_run — execute the self-test: connect to the gateway, arm a TPC-pull that
231 * names `target` as the source, and classify. `out` is fully populated (verdict,
232 * timings, kXR codes, PII-free detail) on every non-usage return.
233 */
234int
235tpce_run(const diag_args *a, const char *gw_url, const char *target,
236 tpce_result *out)
237{
238 brix_url u;
239 brix_conn c;
240 brix_file f;
241 brix_status st;
242 char opaque[1024];
243 char key[48];
244 double budget = a->probe_timeout_ms > 0
245 ? (double) a->probe_timeout_ms : TPCE_DEFAULT_BUDGET_MS;
246 const char *dst_path;
247 double t0;
248 int arm_rc;
249
250 memset(out, 0, sizeof(*out));
251 brix_status_clear(&st);
252 if (brix_url_parse(gw_url, &u, &st) != 0) {
253 out->verdict = TPCE_ERR_CONNECT;
254 snprintf(out->detail, sizeof(out->detail), "bad gateway url");
255 return -1;
256 }
257 snprintf(out->gw_host, sizeof(out->gw_host), "%s", u.host);
258 snprintf(out->target, sizeof(out->target), "%s", target);
259
260 memset(&c, 0, sizeof(c));
261 if (brix_connect(&c, &u, &a->conn, &st) != 0) {
262 out->verdict = TPCE_ERR_CONNECT;
263 out->arm_kxr = st.kxr;
264 snprintf(out->detail, sizeof(out->detail),
265 "cannot reach/log in to the gateway (kXR=%d)", st.kxr);
266 return -1;
267 }
268
269 if (gen_tpc_key(key, sizeof(key)) != 0) {
270 brix_close(&c);
271 out->verdict = TPCE_ERR_CONNECT;
272 snprintf(out->detail, sizeof(out->detail), "cannot mint rendezvous key");
273 return -1;
274 }
275 tpce_build_opaque(opaque, sizeof(opaque), key, target);
276 dst_path = (u.path[0] != '\0' && strcmp(u.path, "/") != 0)
277 ? u.path : TPCE_SELFTEST_LFN;
278
279 brix_status_clear(&st);
280 memset(&f, 0, sizeof(f));
281 t0 = tpce_mono_ms();
282 arm_rc = brix_file_open_opaque(&c, dst_path, opaque, 1, 1, 0, &f, &st);
283 out->arm_ms = tpce_mono_ms() - t0;
284
285 tpce_after_arm(&c, &f, arm_rc, &st, budget, out);
286
287 brix_close(&c);
288 return 0;
289}

294/* Short verdict label + the "is egress permitted?" summary word. */
295static const char *
296tpce_verdict_label(tpce_verdict v)
297{
298 switch (v) {
299 case TPCE_ERR_CONNECT: return "GATEWAY-UNREACHABLE";
300 case TPCE_REFUSED_POLICY: return "REFUSED (policy)";
301 case TPCE_CONN_REFUSED: return "PERMITTED (conn-refused)";
302 case TPCE_FILTERED: return "PERMITTED (filtered/timeout)";
303 case TPCE_REACHED_NOENT: return "PERMITTED (source reachable)";
304 case TPCE_REACHED_ERROR: return "PERMITTED (source reachable)";
305 case TPCE_ACCEPTED: return "PERMITTED (transfer completed)";
306 case TPCE_ARM_ERROR: return "INCONCLUSIVE (arm error)";
307 }
308 return "UNKNOWN";
309}

314{

329 " operator-named source — add a brix_tpc_source_guard\n"

354/*
355 * do_tpc_egress — subcommand entry. Exit codes: 0 = egress refused (the SSRF-safe
356 * outcome the guard produces); 3 = egress permitted (a finding); 2 = gateway
357 * unreachable / inconclusive; 50 = usage.
358 */
359int
360do_tpc_egress(const diag_args *a)
361{
362 tpce_result r;
363
364 if (a->url == NULL || a->tpc_target == NULL || a->tpc_target[0] == '\0') {
365 fprintf(stderr, "usage: xrddiag tpc-egress <your-gateway-url> "
366 "--tpc-target host[:port] [--json] [--probe-timeout ms]\n");
367 return 50;
368 }
369
370 tpce_run(a, a->url, a->tpc_target, &r);
371
372 if (a->json) {
373 tpce_emit_json(&r, stdout);
374 } else {
375 tpce_report(&r);
376 }
377
378 if (r.verdict == TPCE_REFUSED_POLICY) {
379 return 0;
380 }
381 if (r.egress_permitted) {
382 return 3;
383 }
384 return 2;
385}
