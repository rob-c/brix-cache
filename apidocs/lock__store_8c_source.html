13/* WHAT: Compare the current persisted value with the observed snapshot.
14 * WHY: Stale refresh/unlock/expiry must not overwrite a successor's record.
15 * HOW: NULL means absence; otherwise require exact length and bytes. A
16 * truncated or unreadable value fails closed, never means an absent lock.
17 */
18static int
19lock_store_compare(int descriptor, const void *expected, size_t length)
20{
21 unsigned char current[BRIX_LOCK_XATTR_MAXLEN];
22 ssize_t count = fgetxattr(descriptor, BRIX_LOCK_XATTR_KEY,
23 current, sizeof(current));
24
25 if (count < 0) {
26 if (errno == ENODATA && expected == NULL) {
27 return 0;
28 }
29 return -1;
30 }
31 if (expected == NULL) {
32 errno = EEXIST;
33 return -1;
34 }
35 if ((size_t) count != length || memcmp(current, expected, length) != 0) {
36 errno = EBUSY;
37 return -1;
38 }
39 return 0;
40}

42/* WHAT: Apply a transaction while the inode's kernel lock is held.
43 * WHY: No native conditional extattr syscall is needed by this store.
44 * HOW: Compare, then perform one unconditional set or remove operation.
45 */
46static int
47lock_store_apply(int descriptor, const void *expected, size_t expected_length,
48 const void *replacement, size_t replacement_length)
49{
50 if (lock_store_compare(descriptor, expected, expected_length) != 0) {
51 return -1;
52 }
53 if (replacement == NULL) {
54 return fremovexattr(descriptor, BRIX_LOCK_XATTR_KEY);
55 }
56 return fsetxattr(descriptor, BRIX_LOCK_XATTR_KEY, replacement,
57 replacement_length, 0);
58}

60/* WHAT: Serialize one compare/exchange or refuse immediately on contention.
61 * WHY: A process crash releases the lock; an untrusted contender cannot make
62 * the event loop wait. An unrelated file lock must never be forcibly stolen.
63 * HOW: Validate, reject special files, try-lock, compare/mutate and unlock.
64 * Preserve the transaction errno; the caller closes its private descriptor.
65 */
66int
67brix_posix_lock_store_update_fd(int descriptor,
68 const void *expected, size_t expected_length,
69 const void *replacement, size_t replacement_length)
70{
71 struct stat metadata;
72 int result, saved_errno;
73
74 if ((expected == NULL && replacement == NULL)
75 || (expected == NULL && expected_length != 0)
76 || (replacement == NULL && replacement_length != 0)
77 || expected_length >= BRIX_LOCK_XATTR_MAXLEN
78 || replacement_length >= BRIX_LOCK_XATTR_MAXLEN)
79 {
80 errno = EINVAL;
81 return -1;
82 }
83 if (fstat(descriptor, &metadata) != 0) {
84 return -1;
85 }
86 if (!S_ISREG(metadata.st_mode) && !S_ISDIR(metadata.st_mode)) {
87 errno = EINVAL;
88 return -1;
89 }
90 if (flock(descriptor, LOCK_EX | LOCK_NB) != 0) {
91 return -1;
92 }
93 result = lock_store_apply(descriptor, expected, expected_length,
94 replacement, replacement_length);
95 saved_errno = errno;
96 (void) flock(descriptor, LOCK_UN);
97 errno = saved_errno;
98 return result;
99}
