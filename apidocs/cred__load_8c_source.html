1/* File: cred_load.c — outbound X.509 credential file loaders (PEM chain + key)
2 * WHAT: brix_gsi_cred_load_pem() reads every CERTIFICATE block of an operator-configured PEM file and returns them re-serialised as one malloc'd PEM blob; brix_gsi_cred_load_key() reads the first PRIVATE KEY block as an EVP_PKEY. Both open O_RDONLY|O_NOFOLLOW|O_CLOEXEC and never follow a symlink.
3 * WHY: three server-side GSI clients (cache-fill origin, TPC destination, transparent upstream) each need the same two reads; before this file the cache origin carried private copies and the upstream would have needed a third. One implementation keeps the symlink hardening and the "certs only" contract identical everywhere.
4 * HOW: fd → BIO_new_fd(BIO_CLOSE) → PEM_read_bio_X509 loop re-written into a memory BIO (so a key interleaved in a proxy file is dropped) → malloc copy of the BUF_MEM; the key path is a single PEM_read_bio_PrivateKey. ERR_clear_error() after the cert loop drops the benign PEM_R_NO_START_LINE the loop end always leaves.
5 */

18/* Open `path` read-only without following symlinks and wrap it in a BIO that
19 * owns the descriptor. NULL on either failure (the fd is closed). */
20static BIO *
21cred_open_bio(const char *path)
22{
23 int fd;
24 BIO *in;
25
26 fd = open(path, O_RDONLY | O_NOFOLLOW | O_CLOEXEC); /* vfs-seam-allow: DOMAIN_CREDENTIAL — operator-configured X.509 credential file (not export storage) */
27 if (fd < 0) {
28 return NULL;
29 }
30 in = BIO_new_fd(fd, BIO_CLOSE);
31 if (in == NULL) {
32 close(fd);
33 }
34 return in;
35}

37uint8_t *
38brix_gsi_cred_load_pem(const char *path, size_t *outlen)
39{
40 BIO *in, *out;
41 X509 *cert;
42 BUF_MEM *bm;
43 uint8_t *buf = NULL;
44 int n = 0;
45
46 in = cred_open_bio(path);
47 if (in == NULL) {
48 return NULL;
49 }
50 out = BIO_new(BIO_s_mem());
51 if (out == NULL) {
52 BIO_free(in);
53 return NULL;
54 }
55 while ((cert = PEM_read_bio_X509(in, NULL, NULL, NULL)) != NULL) {
56 PEM_write_bio_X509(out, cert);
57 X509_free(cert);
58 n++;
59 }
60 ERR_clear_error(); /* benign PEM EOF on the loop end */
61 BIO_free(in);
62
63 if (n > 0) {
64 BIO_get_mem_ptr(out, &bm);
65 buf = malloc(bm->length);
66 if (buf != NULL) {
67 memcpy(buf, bm->data, bm->length);
68 *outlen = bm->length;
69 }
70 }
71 BIO_free(out);
72 return buf;
73}
