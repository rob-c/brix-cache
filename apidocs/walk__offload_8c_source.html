1/*
2 * walk_offload.c — thread-offload for blocking metadata walks (phase 109 W1).
3 *
4 * WHAT: Runs the PROPFIND build phase (propfind_build: resolve/stat, the
5 * recursive walk's opendir/readdir, per-prop residency queries, XML
6 * assembly) on the shared thread pool, and the send phase
7 * (propfind_send) on the event loop, so a slow REMOTE backend no longer
8 * stalls every connection on the worker.
9 *
10 * WHY: The metadata methods were the only webdav handlers still doing backend
11 * I/O inline (GET/PUT/COPY/MOVE already offload). Against a remote
12 * backend, brix_vfs_opendir is an origin PROPFIND through the blocking
13 * curl transport — bounded, but on the event loop (phase-106 W5, R-7).
14 *
15 * HOW: Three load-bearing decisions:
16 *
17 * 1. GATE, not blanket. The offload fires only when the backend is
18 * remote (brix_storage_backend_is_remote), a thread pool is
19 * configured, and impersonation is OFF. A local-POSIX opendir is a
20 * fast syscall — offloading it would add a thread hop to the common
21 * case for nothing, so the local path stays inline and byte-
22 * identical.
23 *
24 * 2. IMPERSONATION DECLINES, exactly as copy_collection.c:300: the
25 * per-worker broker socket is a single fd owned by the event-loop
26 * thread (concurrent use from a task corrupts the request/reply
27 * framing and wedges the worker's broker channel), and the thread
28 * lacks the per-worker principal. Under brix_imp_enabled() the walk
29 * runs inline, where the principal is set — today's behaviour,
30 * which the phase-106 authz work already tests. This is a SECURITY
31 * decline: an offloaded walk without the principal would enumerate
32 * as the WORKER, not the mapped user.
33 *
34 * 3. TASK-PRIVATE POOL. nginx pools are not thread-safe, and the event
35 * loop can run abort/teardown handlers touching r->pool while the
36 * task is in flight. The build therefore allocates exclusively from
37 * a pool created here (routed via propfind_pool()), and that pool is
38 * registered as a cleanup on r->pool so the response chains survive
39 * until the request is torn down — never freed under a draining
40 * send. The request itself is held alive across the dispatch with
41 * r->main->count++, balanced in the done handler (the PUT pattern,
42 * put_body.c).
43 */

65/* Thread body: the method's build. No r->pool, no event-loop calls — the
66 * build allocates through webdav_req_pool(r), which the dispatcher pointed at
67 * the task-private pool before posting. */
68static void
69webdav_walk_thread(void *data, ngx_log_t *log)
70{
71 webdav_walk_task_t *t = data;
72
73 (void) log;
74 t->rc = t->build(t->r, &t->head, &t->tail, &t->total_len);
75}

78/* Event-loop completion: send (or fail) and finalize with the same metrics
79 * accounting the inline body handler performs. */
80static void
81webdav_walk_done(ngx_event_t *ev)
82{
83 ngx_thread_task_t *task = ev->data;
84 webdav_walk_task_t *t = task->ctx;
85 ngx_http_request_t *r = t->r;
86 ngx_int_t rc = t->rc;
87
88 /* Balance the count++ from the dispatch that kept the request alive. */
89 r->main->count--;
90
91 if (rc == NGX_OK) {
92 rc = t->send(r, t->head, t->tail, t->total_len);
93 }
94 webdav_metrics_finalize_request(r, rc);
95}

107static ngx_int_t
108webdav_walk_offload_wanted(ngx_http_request_t *r,
109 ngx_http_brix_webdav_loc_conf_t *conf, ngx_thread_pool_t **pool)
110{
111 /* Security decline first — see HOW(2) above. */
112 if (brix_imp_enabled()) {
113 return 0;
114 }
115 /* Local backend: fast syscalls, nothing to offload — see HOW(1).
116 * EXCEPTION: EXCHANGE-mode delegation mints an RFC-8693 token through a
117 * blocking POST inside the walk's cred gate even when the storage itself
118 * is local, so that combination offloads too (phase-109 W3 — it is the
119 * exact event-loop stall phase-106 R-7 traced). */
120 if (!brix_storage_backend_is_remote(&conf->common)
121 && conf->common.backend_delegation != BRIX_CRED_EXCHANGE)
122 {
123 return 0;
124 }
125 *pool = brix_shared_thread_pool(&conf->common);
126 return *pool != NULL;
127}

130ngx_int_t
131webdav_walk_offload(ngx_http_request_t *r, webdav_walk_build_pt build,
132 webdav_walk_send_pt send)
133{
134 ngx_http_brix_webdav_loc_conf_t *conf =
135 ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
136 ngx_http_brix_webdav_req_ctx_t *rx =
137 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
138 ngx_thread_pool_t *pool = NULL;
139 ngx_pool_t *wpool;
140 ngx_pool_cleanup_t *cln;
141 ngx_thread_task_t *task;
142 webdav_walk_task_t *t;
143
144 if (rx == NULL || !webdav_walk_offload_wanted(r, conf, &pool)) {
145 return NGX_DECLINED;
146 }
147
148 wpool = ngx_create_pool(4096, r->connection->log);
149 if (wpool == NULL) {
150 return NGX_DECLINED; /* inline path still works */
151 }
152 cln = ngx_pool_cleanup_add(r->pool, 0);
153 if (cln == NULL) {
154 ngx_destroy_pool(wpool);
155 return NGX_DECLINED;
156 }
157 cln->handler = webdav_walk_pool_cleanup;
158 cln->data = wpool;
159
160 task = ngx_thread_task_alloc(r->pool, sizeof(webdav_walk_task_t));
161 if (task == NULL) {
162 return NGX_DECLINED; /* cleanup owns wpool now */
163 }
164 t = task->ctx;
165 t->r = r;
166 t->build = build;
167 t->send = send;
168 t->head = NULL;
169 t->tail = NULL;
170 t->total_len = 0;
171 t->rc = NGX_HTTP_INTERNAL_SERVER_ERROR;
172
173 /* Route every build-phase allocation to the task pool (propfind_pool). */
174 rx->walk_pool = wpool;
175
176 brix_task_bind(task, webdav_walk_thread, webdav_walk_done);
177 if (ngx_thread_task_post(pool, task) != NGX_OK) {
178 rx->walk_pool = NULL; /* inline path, back on r->pool */
179 return NGX_DECLINED;
180 }
181
182 r->main->count++; /* balanced in webdav_walk_done */
183 return NGX_DONE;
184}
