48ngx_int_t
49brix_space_group_usage(ngx_log_t *log, ngx_stream_brix_srv_conf_t *conf,
50 brix_oss_space_t *g, unsigned long long *used)
51{
52 brix_vfs_walk_opts_t opts;
53 unsigned long long sum = 0;
54 char err[128];
55 ngx_int_t rc;
56
57 if (g->used_valid
58 && ngx_current_msec - g->used_at < BRIX_SPACE_GROUP_TTL_MS)
59 {
60 *used = g->used_bytes;
61 return NGX_OK;
62 }
63
64 ngx_memzero(&opts, sizeof(opts));
65 opts.max_depth = SPACE_GROUP_WALK_DEPTH;
66 rc = brix_vfs_walk(log, conf->rootfd, (const char *) g->prefix.data,
67 &opts, space_walk_add_file, &sum, NULL, err, sizeof(err));
68 if (rc == NGX_ERROR) {
69 g->used_valid = 0;
70 ngx_log_error(NGX_LOG_INFO, log, 0,
71 "brix: space group \"%V\" (%V): usage walk failed: %s",
72 &g->name, &g->prefix, err);
73 return NGX_ERROR;
74 }
75 /* NGX_DECLINED = the prefix does not exist yet: an empty group. */
76 g->used_bytes = sum;
77 g->used_at = ngx_current_msec;
78 g->used_valid = 1;
79 *used = sum;
80 return NGX_OK;
81}

83int
84brix_write_space_group_admit(brix_ctx_t *ctx, ngx_connection_t *c,
85 ngx_stream_brix_srv_conf_t *conf, int idx, size_t len,
86 ngx_uint_t op_id, const char *op, ngx_int_t *rc)
87{
88 brix_oss_space_t *g;
89 unsigned long long used;
90 char msg[160];
91
92 g = brix_space_group_of_handle(conf, &ctx->files[idx]);
93 if (g == NULL) {
94 return -1; /* export-wide rules apply */

96 if (g->quota < 0 || conf->oss_quota_enforce != 1) {
97 return 0; /* accounting only */
98 }
99 if (brix_space_group_usage(c->log, conf, g, &used) != NGX_OK) {
100 return 0; /* fail-open, logged by the walk */
101 }
102 if (used + (unsigned long long) len > (unsigned long long) g->quota) {
103 snprintf(msg, sizeof(msg),
104 "write exceeds the quota of space group %.*s (brix_oss_space)",
105 (int) g->name.len, (const char *) g->name.data);
106 brix_log_access(ctx, c, op, ctx->files[idx].path, "-", 0,
107 kXR_overQuota, msg, 0);
108 BRIX_OP_ERR(ctx, op_id);
109 *rc = brix_send_error(ctx, c, kXR_overQuota, msg);
110 return 1;
111 }
112 g->used_bytes = used + (unsigned long long) len; /* charge before pwrite */
113 return 0;
