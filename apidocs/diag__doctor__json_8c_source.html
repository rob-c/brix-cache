1/*
2 * diag_doctor_json.c — the `xrddiag doctor --json` machine report.
3 *
4 * WHAT: renders the doctor_ep array as one JSON document: per-endpoint facts,
5 * issue list, classified diagnosis, CMS locate-plane verdict, plus the
6 * config/latency/EOS sub-objects each owned by its own translation unit.
7 * WHY: the emitter is pure formatting with no probing logic, and it was the
8 * single largest branch cluster in diag_doctor.c — hoisting it out keeps
9 * both files under the size and complexity gates.
10 * HOW: one emitter per JSON object, each writing its own leading comma so the
11 * orchestrator stays a flat loop (coding-standards §4/§8).
12 */
13#include "diag_internal.h"

16/* `{"protocol":...,"host":...,"port":...,"status":...,"facts":{...},"issues":[`
17 * — the endpoint header through to the (still open) issues array. */
18static void
19ep_emit_facts(const doctor_ep *e, int first, FILE *out)
20{
21 fprintf(out, "%s{\"protocol\":\"%s\",\"host\":", first ? "" : ",",
22 dx_proto_name(e->proto));
23 fjson_str(out, e->host);
24 fprintf(out, ",\"port\":%d,\"status\":\"%s\","
25 "\"connected\":%s,\"skipped\":%s,\"facts\":{\"family\":\"%s\","
26 "\"tcp_ms\":%.3f,\"tls_ms\":%.3f,\"auth_ms\":%.3f,\"total_ms\":%.3f,"
27 "\"rtt_us\":%u,\"retrans\":%u,\"tls\":\"%s\",\"auth\":\"%s\","
28 "\"caps\":\"0x%x\",\"ttfb_ms\":%.3f,\"mbps\":%.1f,\"holders\":%d,"
29 "\"metrics_http\":%d,\"shedding\":%s},\"issues\":[",
30 e->port, doc_color(e->status),
31 e->connected ? "true" : "false",
32 e->skipped ? "true" : "false",
33 e->nf.family == 10 ? "IPv6" : e->nf.family == 2 ? "IPv4" : "none",
34 e->nf.tcp_ms, e->nf.tls_ms, e->nf.auth_ms, e->nf.total_ms,
35 e->nf.rtt_us, e->nf.retrans,
36 e->tls_active ? e->tls_ver : "none", e->auth, e->caps,
37 e->ttfb_ms, e->mbps, e->holders, e->metrics_http,
38 e->shedding ? "true" : "false");
39}

43static void
44ep_emit_issues(const doctor_ep *e, FILE *out)
45{

58/* The classified per-probe findings: verdict, kXR code, cause and remedy. */
59static void
60ep_emit_diagnosis(const doctor_ep *e, FILE *out)
61{
62 int j;
63
64 for (j = 0; j < e->ndx; j++) {
65 const dx_finding *d = &e->dx[j];
66
67 fprintf(out, "%s{\"probe\":", j ? "," : "");
68 fjson_str(out, d->probe);
69 fprintf(out, ",\"verdict\":\"%s\",\"kxr\":%d,\"cause\":",
70 dx_verdict_name(d->verdict), d->kxr);
71 fjson_str(out, d->cause);
72 fprintf(out, ",\"remedy\":");
73 fjson_str(out, d->remedy);
74 fputc('}', out);
75 }
76 fprintf(out, "]");
77}

80/* CMS locate-plane classification (mesh map) — role/access as the manager
81 * reported it, present even for un-connectable (skipped) holders. */
82static void
83ep_emit_cms(const doctor_cmsloc *m, FILE *out)
84{
85 const char *role = m->role == DOC_CMS_MANAGER ? "manager"
86 : m->role == DOC_CMS_SERVER ? "server" : "none";
87 const char *access = "none";
88
89 if (m->reported && m->role == DOC_CMS_SERVER) {
90 access = m->write ? "rw" : "ro";
91 }
92 fprintf(out, ",\"cms\":{\"reported\":%s,\"role\":\"%s\",\"access\":\"%s\","
93 "\"pending\":%s}",
94 m->reported ? "true" : "false", role, access,
95 m->pending ? "true" : "false");
96}

99void
100doctor_emit_json(const doctor_ep *eps, int n, FILE *out)
101{
102 int i;
103
104 fprintf(out, "{\"remote_doctor\":{\"endpoints\":[");
105 for (i = 0; i < n; i++) {
106 const doctor_ep *e = &eps[i];
107
108 ep_emit_facts(e, i == 0, out);
109 ep_emit_issues(e, out);
110 ep_emit_diagnosis(e, out);
111 doctor_emit_config_json(e, out); /* phase-93 "config" object (or null) */
112 doctor_emit_latency_json(e, out); /* phase-93 "latency" object (or nothing) */
113 doctor_emit_recon_json(e, out); /* phase-93 "recon" object (or nothing) */
114 ep_emit_cms(&e->cms, out);
115 doctor_eos_emit_json(e, out); /* EOS MGM banner / enumerated FST */
116 fprintf(out, "}");
117 }
118 fprintf(out, "],\"cross_endpoint_analysis\":{\"hops\":%d}}}\n", n > 1 ? n - 1 : 0);
119}
