1/*
2 * sd_pblock_catalog_nsidx.c — phase-88 W4: the shared-mmap namespace cache.
3 *
4 * WHAT: Promotes the worker-local heap nscache (sd_pblock_catalog.c) to a
5 * cross-process table: the same direct-mapped, positive-entry
6 * path→pblock_meta cache, but living in a mmap'd <root>/catalog.bxi
7 * every worker maps MAP_SHARED. One worker's fill warms all workers;
8 * one worker's invalidation is seen by all workers — a strict coherence
9 * upgrade over the heap table, whose invalidations never crossed the
10 * process boundary.
11 *
12 * WHY: The metadata characterisation (pblock-metadata-performance.md) pinned
13 * pblock_catalog_lookup as the #1 SQLite consumer; the heap cache halved
14 * it per worker but stays cold per process and incoherent across them.
15 * The cvmfs pathidx (shared/cvmfs/index/pathidx.c) proved the mmap
16 * mechanics — fixed entries, FNV addressing, crash-tolerant validation —
17 * for a FROZEN revision; a live-mutable catalog needs the coherent-cache
18 * variant of that idea, not a snapshot.
19 *
20 * HOW: Lock-free throughout (no mutex, no nginx SHM zone — INVARIANT 10 has
21 * no surface here): each entry carries a seqlock (CAS even→odd claims
22 * the writer role, +1 releases; a contended claim just skips — the cache
23 * is best-effort), the header carries two atomic u64s: `gen` (the
24 * fill-after-miss guard the heap cache already used) and `epoch`
25 * (clear-all in O(1): entries stamp the epoch they were written under;
26 * a stale stamp reads as invalid). First live opener — flock(EX|NB)
27 * succeeds on the sidecar, kernel drops flocks on death so a crashed
28 * fleet never wedges — bumps the epoch, killing any residue from an
29 * unclean shutdown; every opener then holds a SH flock as its liveness
30 * signal. Any init/validation failure disarms silently: the catalog
31 * falls back to the heap cache, correctness unchanged.
32 * ngx-free (libc + atomics); BRIX_HAVE_SQLITE-gated like its includers.
33 */

65/* One shared entry. `seq` is the per-entry seqlock (even = stable, odd = a
66 * writer is inside); `epoch` must match the header's for the entry to be
67 * live, which makes clear-all a single atomic increment. */
68typedef struct {
69 uint32_t seq;
70 uint32_t valid;
71 uint64_t epoch;
72 char path[PBLOCK_NSCACHE_PATHMAX];
73 pblock_meta meta;
74} nsidx_ent_t;

76typedef struct {
77 uint32_t magic;
78 uint32_t version;
79 uint32_t buckets;
80 uint32_t entsize; /* ABI guard across mixed builds */
81 uint64_t gen; /* fill-after-miss guard */
82 uint64_t epoch; /* clear-all generation */
83 uint64_t reserved[4];
84} nsidx_hdr_t;

100static nsidx_ent_t *
101nsidx_ent(pblock_catalog *cat, const char *path)
102{
103 uint64_t h = 1469598103934665603ull;
104 const unsigned char *cursor;
105
106 for (cursor = (const unsigned char *) path; *cursor != '\0'; cursor++) {
107 h ^= *cursor;
108 h *= 1099511628211ull;
109 }
110 return (nsidx_ent_t *) ((char *) cat->nsidx + sizeof(nsidx_hdr_t))
111 + (h & (PBLOCK_NSCACHE_BUCKETS - 1));
112}

114/* nsidx_claim — CAS the entry's seqlock even→odd (the writer role). Returns
115 * the claimed (odd) value, or 0 when the entry is contended — the caller just
116 * skips its best-effort write. */
117static uint32_t
118nsidx_claim(nsidx_ent_t *e)
119{
120 uint32_t s = __atomic_load_n(&e->seq, __ATOMIC_ACQUIRE);
121
122 if (s & 1u) {
123 return 0;
124 }
125 if (!__atomic_compare_exchange_n(&e->seq, &s, s + 1u, 0,
126 __ATOMIC_ACQ_REL, __ATOMIC_RELAXED))
127 {
128 return 0;
129 }
130 return s + 1u;
131}
132
133/* nsidx_release — publish the write: seqlock odd→even. */
134static void
135nsidx_release(nsidx_ent_t *e, uint32_t claimed)
136{
137 __atomic_store_n(&e->seq, claimed + 1u, __ATOMIC_RELEASE);
138}
139
140/* nsidx_install — claim + write a positive entry under the current epoch.
141 * Best-effort: a contended entry is simply skipped. */
142static void
143nsidx_install(pblock_catalog *cat, const char *path, const pblock_meta *meta)
144{
145 nsidx_ent_t *e = nsidx_ent(cat, path);
146 uint32_t claimed = nsidx_claim(e);
147
148 if (claimed == 0) {
149 return;
150 }
151 snprintf(e->path, sizeof(e->path), "%s", path);
152 e->meta = *meta;
153 e->valid = 1;
154 __atomic_store_n(&e->epoch,
155 __atomic_load_n(&nsidx_hdr(cat)->epoch, __ATOMIC_ACQUIRE),
156 __ATOMIC_RELEASE);
157 nsidx_release(e, claimed);
158}

160/* nsidx_drop — claim + invalidate whatever occupies the bucket (direct-mapped:
161 * dropping a colliding tenant only costs it a refill). */
162static void
163nsidx_drop(pblock_catalog *cat, const char *path)
164{
165 nsidx_ent_t *e = nsidx_ent(cat, path);
166 uint32_t claimed = nsidx_claim(e);
167
168 if (claimed == 0) {
169 return;
170 }
171 e->valid = 0;
172 nsidx_release(e, claimed);
173}

177int
178nsidx_get(pblock_catalog *cat, const char *path, pblock_meta *out)
179{
180 nsidx_ent_t *e = nsidx_ent(cat, path);
181 nsidx_ent_t snap;
182 uint32_t s1, s2;
183
184 s1 = __atomic_load_n(&e->seq, __ATOMIC_ACQUIRE);
185 if (s1 & 1u) {
186 return 0; /* a writer is inside — miss */
187 }
188 memcpy(&snap, e, sizeof(snap));
189 __atomic_thread_fence(__ATOMIC_ACQUIRE);
190 s2 = __atomic_load_n(&e->seq, __ATOMIC_ACQUIRE);
191 if (s1 != s2 || !snap.valid
192 || snap.epoch != __atomic_load_n(&nsidx_hdr(cat)->epoch,
193 __ATOMIC_ACQUIRE)
194 || strncmp(snap.path, path, sizeof(snap.path)) != 0)
195 {
196 return 0;
197 }
198 if (out != NULL) {
199 *out = snap.meta;
200 }
201 return 1;
202}

210void
211nsidx_store(pblock_catalog *cat, const char *path, const pblock_meta *meta,
212 uint64_t gen)
213{
214 /* Fill-after-miss: install only while no invalidation landed since the
215 * caller snapshotted `gen`; re-check afterwards and retract on a race —
216 * the row read under the old generation may already be stale. */
217 if (nsidx_gen(cat) != gen) {
218 return;
219 }
220 nsidx_install(cat, path, meta);
221 if (nsidx_gen(cat) != gen) {
222 nsidx_drop(cat, path);
223 }
224}

250/* nsidx_init_hdr — first-live-opener initialisation: lay down (or revalidate)
251 * the header and bump the epoch so residue from an unclean shutdown — e.g. a
252 * committed write whose cache install never happened — can never be served. */
253static int
254nsidx_init_hdr(void *map)
255{
256 nsidx_hdr_t *hdr = map;
257
258 if (hdr->magic != 0 && (hdr->magic != NSIDX_MAGIC
259 || hdr->version != NSIDX_VERSION
260 || hdr->buckets != PBLOCK_NSCACHE_BUCKETS
261 || hdr->entsize != sizeof(nsidx_ent_t)))
262 {
263 return -1; /* an alien/mixed-build sidecar */
264 }
265 hdr->buckets = PBLOCK_NSCACHE_BUCKETS;
266 hdr->entsize = (uint32_t) sizeof(nsidx_ent_t);
267 hdr->version = NSIDX_VERSION;
268 __atomic_store_n(&hdr->magic, NSIDX_MAGIC, __ATOMIC_RELEASE);
269 __atomic_add_fetch(&hdr->epoch, 1, __ATOMIC_ACQ_REL);
270 __atomic_add_fetch(&hdr->gen, 1, __ATOMIC_ACQ_REL);
271 return 0;
272}

274/* First opener resets the header; a joiner verifies the existing one is a
275 * compatible layout (magic/version/geometry). Returns 0, -1 on a bad layout. */
276static int
277nsidx_init_or_verify(void *map, int first)
278{
279 const nsidx_hdr_t *hdr;
280
281 if (first) {
282 return nsidx_init_hdr(map) == 0 ? 0 : -1;
283 }
284 hdr = map;
285 if (__atomic_load_n(&hdr->magic, __ATOMIC_ACQUIRE) != NSIDX_MAGIC
286 || hdr->version != NSIDX_VERSION
287 || hdr->buckets != PBLOCK_NSCACHE_BUCKETS
288 || hdr->entsize != sizeof(nsidx_ent_t))
289 {
290 return -1;
291 }
292 return 0;
293}

295int
296pblock_catalog_nsidx_arm(pblock_catalog *cat, const char *root)
297{
298 char path[PATH_MAX];
299 void *map;
300 struct stat stx;
301 int fd, first;
302
303 if (cat == NULL || root == NULL || cat->nsidx != NULL) {
304 errno = EINVAL;
305 return -1;
306 }
307 if (snprintf(path, sizeof(path), "%s/catalog.bxi", root)
308 >= (int) sizeof(path))
309 {
310 errno = ENAMETOOLONG;
311 return -1;
312 }
313 fd = open(path, O_RDWR | O_CREAT | O_CLOEXEC, 0600);
314 if (fd < 0) {
315 return -1;
316 }
317
318 /* flock(EX|NB) succeeding means no other live opener exists (the kernel
319 * drops flocks on process death, so this is crash-honest); the winner
320 * resets, everyone converges on a SH lock as their liveness mark. */
321 first = flock(fd, LOCK_EX | LOCK_NB) == 0;
322
323 if (ftruncate(fd, (off_t) NSIDX_FILE_SIZE) != 0
324 || fstat(fd, &stx) != 0 || stx.st_size < (off_t) NSIDX_FILE_SIZE)
325 {
326 close(fd);
327 return -1;
328 }
329 map = mmap(NULL, NSIDX_FILE_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED,
330 fd, 0);
331 if (map == MAP_FAILED) {
332 close(fd);
333 return -1;
334 }
335
336 if (nsidx_init_or_verify(map, first) != 0) {
337 munmap(map, NSIDX_FILE_SIZE);
338 close(fd);
339 errno = EINVAL;
340 return -1;
341 }
342
343 if (flock(fd, LOCK_SH) != 0) { /* EX holders downgrade; joiners take SH */
344 munmap(map, NSIDX_FILE_SIZE);
345 close(fd);
346 return -1;
347 }
348
349 cat->nsidx = map;
350 cat->nsidx_fd = fd;
351 return 0;
352}

356{
357 if (cat == NULL || cat->nsidx == NULL) {
358 return;

362 cat->nsidx = NULL;
363 cat->nsidx_fd = -1;
364}
