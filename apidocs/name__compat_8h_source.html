47/* ---- Escape value delimiters, never structural name boundaries ----
48 * WHAT: Append one rendered value; return 0 on allocation/write error.
49 * WHY: Literal '/' and '+' cannot impersonate additional identity components.
50 * HOW: 1. Walk the ASCII rendering. 2. Prefix only literal value delimiters.
51 */
52static inline int
53brix_crypto_name_write_value(BIO *output, const char *value)
54{
55 for (const char *cursor = value; *cursor != '\0'; cursor++) {
56 if ((*cursor == '/' || *cursor == '+') && BIO_write(output, "\\", 1) != 1) {
57 return 0;
58 }
59 if (BIO_write(output, cursor, 1) != 1) {
60 return 0;
61 }
62 }
63 return 1;
64}

66/* ---- Render one AVA with the library's existing OID and ASN.1 rules ----
67 * WHAT: Append an attribute and its original RDN delimiter; 1/0 verdict.
68 * WHY: Splitting a complete legacy DN cannot distinguish embedded slashes.
69 * HOW: 1. Render a single-entry name. 2. Separate its field. 3. Escape its value.
70 */
71static inline int
72brix_crypto_name_write_entry(BIO *output, X509_NAME_ENTRY *entry, const char *separator)
73{
74 X509_NAME *single = X509_NAME_new();
75 char *text = NULL, *value;
76 int result = 0;
77
78 if (single != NULL && X509_NAME_add_entry(single, entry, -1, 0)) {
79 text = X509_NAME_oneline(single, NULL, 0);
80 }
81 X509_NAME_free(single);
82 if (text == NULL) {
83 return 0;
84 }
85 value = strchr(text, '=');
86 if (text[0] == '/' && value != NULL && value - text < INT_MAX) {
87 int field_length = (int) (value - text);

90 && brix_crypto_name_write_value(output, value + 1);

92 OPENSSL_free(text);
93 return result;
94}

97 * WHAT: Return an OPENSSL_free-owned string, or NULL on rendering failure.
98 * WHY: Legacy identity mapping and signing policy need the same representation.
99 * HOW: 1. Detect fixed builds. 2. Render each RDN/AVA. 3. Copy owned bytes.

160/* ---- Recognize a literal backslash code unit, not a low byte of another glyph ----
161 * WHAT: Return 1 only for U+005C in a one-, two- or four-byte name string.
162 * WHY: BMP/UniversalString values must not mistake another code point for '\\'.
163 * HOW: 1. Require zero high bytes. 2. Compare the final byte.
164 */
165static inline int
166brix_crypto_name_backslash_unit(const unsigned char *unit, size_t width)
167{
168 for (size_t index = 0; index + 1 < width; index++) {
169 if (unit[index] != 0) {
170 return 0;
171 }
172 }
173 return unit[width - 1] == '\\';
174}

176/* ---- Inspect the original ASN.1 value without converting its encoding ----
177 * WHAT: Return a literal-backslash count, or -1 for malformed code-unit bounds.
178 * WHY: Existing non-ASCII rendering must remain unchanged on both libraries.
179 * HOW: 1. Select the ASN.1 unit width. 2. Validate length. 3. Count U+005C.
180 */
181static inline int
182brix_crypto_name_backslashes(const ASN1_STRING *value, size_t *width)
183{
184 int type, length;
185 const unsigned char *data;
186 int count = 0;
187
188 if (value == NULL) {
189 return -1;
190 }
191 type = ASN1_STRING_type(value);
192 length = ASN1_STRING_length(value);
193 data = ASN1_STRING_get0_data(value);
194 *width = type == V_ASN1_BMPSTRING ? 2 : type == V_ASN1_UNIVERSALSTRING ? 4 : 1;
195 if (length < 0 || (size_t) length % *width != 0 || (length > 0 && data == NULL)) {
196 return -1;
197 }
198 for (size_t offset = 0; offset < (size_t) length; offset += *width) {
199 count += brix_crypto_name_backslash_unit(data + offset, *width);
200 }
201 return count;
202}

233/* ---- Quote backslashes only in an owned temporary rendering entry ----
234 * WHAT: Double U+005C code units, preserving the value's ASN.1 type; 1/0 verdict.
235 * WHY: A trailing literal backslash cannot escape the next structural slash.
236 * HOW: 1. Bound expansion. 2. Duplicate complete code units. 3. Replace owned data.
237 */
238static inline int
239brix_crypto_name_quote_entry(X509_NAME_ENTRY *entry)
240{
241 const ASN1_STRING *value = X509_NAME_ENTRY_get_data(entry);
242 size_t width, cursor = 0;
243 int count = brix_crypto_name_backslashes(value, &width);
244 int length;
245 const unsigned char *data;
246 unsigned char *quoted;
247 size_t expanded;
248 int result;
249
250 if (count <= 0) {
251 return count == 0;
252 }
253 length = ASN1_STRING_length(value);
254 data = ASN1_STRING_get0_data(value);
255 expanded = (size_t) length + (size_t) count * width;
256 if (expanded > INT_MAX) {
257 return 0;
258 }
259 quoted = OPENSSL_malloc(expanded);
260 if (quoted == NULL) {
261 return 0;
262 }
263 for (size_t offset = 0; offset < (size_t) length; offset += width) {
264 memcpy(quoted + cursor, data + offset, width);
265 cursor += width;
266 if (brix_crypto_name_backslash_unit(data + offset, width)) {
267 memcpy(quoted + cursor, data + offset, width);
268 cursor += width;
269 }
270 }
271 result = X509_NAME_ENTRY_set_data(entry, ASN1_STRING_type(value), quoted,
272 (int) expanded);
273 OPENSSL_free(quoted);
274 return result;
275}

277/* ---- Make an owned rendering-only name with unambiguous literal backslashes ----
278 * WHAT: Return a temporary name or NULL, retaining original RDN/AVA grouping.
279 * WHY: The verified credential and its encoded signature input must never change.
280 * HOW: 1. Copy each entry. 2. Preserve grouping. 3. Quote only the owned entries.
281 */
282static inline X509_NAME *
283brix_crypto_name_quoted_copy(const X509_NAME *name)
284{
285 X509_NAME *copy = X509_NAME_new();
286 int previous_set = -1;
287
288 if (copy == NULL) {
289 return NULL;
290 }

292 const X509_NAME_ENTRY *entry = X509_NAME_get_entry(name, index);
293 /* Legacy dup takes mutable input; each ABI's getter matches its dup. */
294 X509_NAME_ENTRY *quoted = X509_NAME_ENTRY_dup(X509_NAME_get_entry(name, index));

301 X509_NAME_free(copy);
302 return NULL;
303 }
304 X509_NAME_ENTRY_free(quoted);
305 previous_set = current_set;
306 }

329/* ---- Return an owned canonical name with literal and structural escapes distinct ----
330 * WHAT: Return complete text or NULL; the caller frees it with OPENSSL_free.
331 * WHY: Legacy oneline APIs alias a value slash with a backslash/RDN boundary.
332 * HOW: 1. Inspect original values. 2. Normalize only when needed. 3. Render/free copy.
333 */
334static inline char *
335brix_crypto_name_string(const X509_NAME *name)
336{
337 int needs_quote, factor;
338 X509_NAME *copy;
339 char *text;
340
341 if (name == NULL) {
342 return NULL;
343 }
344 needs_quote = brix_crypto_name_has_backslash(name);
345 if (needs_quote < 0) {
346 return NULL;
347 }
348 if (needs_quote == 0) {
349 return brix_crypto_name_native_string(name);
350 }
351 factor = brix_crypto_name_backslash_factor();
352 if (factor != 2) {
353 return factor == 1 ? brix_crypto_name_native_string(name) : NULL;
354 }
355 copy = brix_crypto_name_quoted_copy(name);
356 if (copy == NULL) {
357 return NULL;
358 }
359 text = brix_crypto_name_native_string(copy);
360 X509_NAME_free(copy);
361 return text;
362}

365 * WHAT: Return owned text for NULL buffer, otherwise buffer or NULL on error.
