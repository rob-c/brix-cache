1/*
2 * server_conf_merge_proxy_net.c — server-block merge for the upstream/proxy and
3 * network-fault configuration area.
4 *
5 * WHAT: Owns brix_merge_srv_proxy_net() (upstream TLS + token, transparent proxy
6 * mode, the write-through origin + prefix rules + decision struct, OCSP,
7 * the Phase-39 network-fault deadlines, and rate-limit rule inheritance),
8 * together with the file-local per-concern helpers it delegates to.
9 * WHY: Split (phase-79 file-size cap) out of the former 1249-line
10 * server_conf.c. The entry point is non-static (declared in
11 * server_conf_internal.h) because the top-level orchestrator in
12 * server_conf.c calls it last in linear order — after the cache fields it
13 * depends on (origin host/port, cache size cap, include-regex) have
14 * settled; every sub-helper stays file-local.
15 * HOW: Standard nginx parent->child inheritance via ngx_conf_merge_* and the
16 * BRIX_MERGE_* macros. The write-through decision struct rebuilds from the
17 * just-merged prefix arrays/size cap/regex; the manager-staleness publish
18 * is a config-time side effect that runs once its input settles, exactly
19 * as before. No behaviour change from the split.
20 */

91/*
92 * WHAT: merge the write-through group — enable/mode, origin host/port, the
93 * deny/allow prefix arrays, and rebuild the write-through decision struct
94 * from the merged inputs.
95 * WHY: the decision struct references the just-merged prefix arrays, cache size
96 * cap, and include-regex, so it must rebuild after they settle.
97 * HOW: merge the wt scalars + prefix arrays (NULL result + non-empty parent/
98 * child ⇒ NGX_CONF_ERROR), then point the decision fn/data/limits at them.
99 */
100static char *
101brix_merge_srv_writethrough(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
102 ngx_stream_brix_srv_conf_t *prev)
103{
104 ngx_array_t *child_wt_deny_prefixes;
105 ngx_array_t *child_wt_allow_prefixes;
106
107 ngx_conf_merge_value(conf->wt.enable, prev->wt.enable, 0);
108 BRIX_MERGE_ENUM(conf, prev, wt.mode, BRIX_WT_MODE_UNSET, BRIX_WT_MODE_SYNC);
109 /* 2.0: the fallback to cache_origin_host that used to stand here was dead —
110 * brix_cache_origin was retired in phase-64 §14, so no directive writes that
111 * field on a real server conf and the branch could never be taken. A
112 * write-through server names its flush target with brix_wt_origin. */
113 BRIX_MERGE_HOSTPORT(conf, prev, wt.origin_host, wt.origin_port);
114
115 child_wt_deny_prefixes = conf->wt.deny_prefixes;
116 conf->wt.deny_prefixes = brix_merge_arrays(cf, prev->wt.deny_prefixes,
117 child_wt_deny_prefixes,
118 sizeof(brix_wt_prefix_entry_t));
119 if (conf->wt.deny_prefixes == NULL
120 && (prev->wt.deny_prefixes != NULL || child_wt_deny_prefixes != NULL))
121 {
122 return NGX_CONF_ERROR;
123 }
124
125 child_wt_allow_prefixes = conf->wt.allow_prefixes;
126 conf->wt.allow_prefixes = brix_merge_arrays(cf, prev->wt.allow_prefixes,
127 child_wt_allow_prefixes,
128 sizeof(brix_wt_prefix_entry_t));
129 if (conf->wt.allow_prefixes == NULL
130 && (prev->wt.allow_prefixes != NULL || child_wt_allow_prefixes != NULL))
131 {
132 return NGX_CONF_ERROR;
133 }
134
135 conf->wt.decision.fn = brix_wt_default_decide;
136 conf->wt.decision.user_data = &conf->wt.decision;
137 conf->wt.decision.deny_prefixes = conf->wt.deny_prefixes;
138 conf->wt.decision.allow_prefixes = conf->wt.allow_prefixes;
139 conf->wt.decision.max_write_through_bytes = conf->cache_max_file_size;
140 conf->wt.decision.include_regex = conf->include_regex.re;
141 conf->wt.decision.include_regex_set = conf->include_regex.set;
142
143 return NGX_CONF_OK;
144}

146/*
147 * WHAT: merge OCSP, the Phase-39 network-fault deadlines, and inherit the
148 * advanced rate-limit rule array; publish the manager staleness threshold.
149 * WHY: the staleness publish is a config-time side effect (before fork) that
150 * must run once inputs settle; grouping keeps it beside its merge.
151 * HOW: delegate OCSP, inherit the deadline scalars (0 == disabled), publish the
152 * staleness threshold when > 0, and inherit rl_rules when the child is NULL.
153 */
154static void
155brix_merge_srv_net_fault(ngx_stream_brix_srv_conf_t *conf,
156 ngx_stream_brix_srv_conf_t *prev)
157{
158 brix_ocsp_conf_merge(&conf->ocsp, &prev->ocsp);
159
160 /* ocsp.staple_data/len are populated at init_process, not here */
161
162 /* ofs.maxdelay analog: stock default 60 s. Clamps at brix_send_wait, the
163 * single kXR_wait emission choke point; 0 disables the clamp. */
164 ngx_conf_merge_sec_value(conf->max_delay, prev->max_delay, 60);
165 /* fsoverload stall: default 1 s (the historical hardcoded budget-overload
166 * backoff); still clamped by max_delay at the emission choke point. */
167 ngx_conf_merge_sec_value(conf->fsoverload_stall, prev->fsoverload_stall, 1);
168 /* fsoverload redirect host+port inherit together as a unit (like upstream_host):
169 * a child that set neither takes the parent's pair; "" = off (stall). */
170 if (conf->fsoverload_redirect_host.len == 0
171 && prev->fsoverload_redirect_host.len > 0)
172 {
173 conf->fsoverload_redirect_host = prev->fsoverload_redirect_host;
174 conf->fsoverload_redirect_port = prev->fsoverload_redirect_port;
175 }
176
177 /* §3.1: the advertised oss.quota — default -1 (unlimited). NGX_CONF_UNSET is
178 * already (off_t)-1, so an unconfigured server reports the stock -1. */
179 ngx_conf_merge_off_value(conf->oss_quota, prev->oss_quota, -1);
180 ngx_conf_merge_value(conf->oss_quota_enforce, prev->oss_quota_enforce, 0);
181
182 /* Phase 39: network-fault resilience — default OFF (0) = no behaviour change.
183 * 0 means "disabled"; arm sites all guard on `> 0` (ngx_msec_t is unsigned). */
184 ngx_conf_merge_msec_value(conf->read_timeout, prev->read_timeout, 0);
185 ngx_conf_merge_msec_value(conf->handshake_timeout, prev->handshake_timeout, 0);
186 ngx_conf_merge_msec_value(conf->send_timeout, prev->send_timeout, 0);
187 ngx_conf_merge_msec_value(conf->tcp_user_timeout, prev->tcp_user_timeout, 0);
188 ngx_conf_merge_value(conf->tcp_keepalive, prev->tcp_keepalive, 0);
189 ngx_conf_merge_str_value(conf->tcp_congestion, prev->tcp_congestion, "");
190 ngx_conf_merge_size_value(conf->socket_sndbuf, prev->socket_sndbuf, 0);
191 ngx_conf_merge_size_value(conf->socket_rcvbuf, prev->socket_rcvbuf, 0);
192 ngx_conf_merge_uint_value(conf->max_connections, prev->max_connections, 0);
193 /* §3.15 slowop threshold (µs); 0 = classifier disabled (default). */
194 ngx_conf_merge_value(conf->slowop_usec, prev->slowop_usec, 0);
195 ngx_conf_merge_msec_value(conf->manager_stale_after,
196 prev->manager_stale_after, 0);
197 /* Phase 39 (WS7): publish the data-server staleness threshold to the shared
198 * cluster registry (config-time, before fork). Guarded so a 0-default block
199 * does not disable a sibling that enabled it. */
200 if (conf->manager_stale_after > 0) {
201 brix_srv_set_stale_after(conf->manager_stale_after);
202 }
203
204 /* Phase 39 (WS9): a child server block with no advanced rate-limit rules of
205 * its own inherits the parent's (the Phase-20 kv caches above already do;
206 * the Phase-25 rl_rules array was previously never inherited). */
207 if (conf->rl_rules == NULL) {
208 conf->rl_rules = prev->rl_rules;
209 }
210}

212/* Upstream/proxy & network: upstream TLS + token, transparent proxy mode
213 * (auth/login/timeouts/rewrite), the write-through origin + prefix rules +
214 * decision struct, OCSP, the Phase-39 network-fault deadlines, and rate-limit
215 * rule inheritance. */
216char *
217brix_merge_srv_proxy_net(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
218 ngx_stream_brix_srv_conf_t *prev)
219{
220 brix_merge_srv_proxy(conf, prev);
221 if (brix_merge_srv_writethrough(cf, conf, prev) != NGX_CONF_OK) {
222 return NGX_CONF_ERROR;
223 }
224 brix_merge_srv_net_fault(conf, prev);
225
226 return NGX_CONF_OK;
227}
