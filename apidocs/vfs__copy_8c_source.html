1/*
2 * vfs_copy.c — VFS single-file copy (server-side COPY / CopyObject).
3 *
4 * WHAT: Implements brix_vfs_copy() — copy the resolved ctx (source) regular
5 * file to a destination path within the same export root, the data-mover
6 * behind WebDAV COPY and S3 CopyObject.
7 *
8 * WHY: The copy engines reached brix_ns_local_copy() (copy_file_range with a
9 * read/write fallback) directly, so the bytes moved were confined but
10 * unmetered. Routing through here books one BRIX_METRIC_OP_COPY metric +
11 * access-log line per copy, with the byte count taken from the resulting
12 * destination size, while still delegating the confined data move to the
13 * namespace layer.
14 *
15 * HOW: brix_vfs_copy() mutation-gates (brix_vfs_require_confined_mutation
16 * with MUTATE_COPY — a copy creates a new object), translates the public brix_vfs_copy_opts_t into an
17 * brix_ns_copy_opts_t, and calls brix_ns_local_copy(ctx->root_canon,
18 * src=ctx path, dst). The namespace status is mapped back to errno and
19 * observed as OP_COPY; bytes are best-effort from a post-copy lstat of the
20 * destination (0 if that stat fails — it never affects the return value).
21 */
22#include "vfs_internal.h"
23#include "fs/path/beneath.h" /* NOREPLACE degradation latch (C6 atomic verdict) */
24#include "fs/backend/cache/sd_cache.h" /* brix_sd_cache_evict: the leaf
25 * dispatch bypasses the decorator's
26 * own invalidation of the copy dst */

28/*
29 * brix_vfs_copy_fail — book a failed OP_COPY observation and return NGX_ERROR.
30 *
31 * WHAT: Emit one BRIX_METRIC_OP_COPY error observation (zero bytes) for `src`
32 * with the caller-chosen errno, set errno to that value, and hand back
33 * NGX_ERROR so callers can `return brix_vfs_copy_fail(...)`.
34 * WHY: Every failure exit in brix_vfs_copy repeated the same errno-set +
35 * observe + return triad; centralising it keeps the mapping and the
36 * metric label identical across all error paths.
37 * HOW: Assign errno from `err`, forward it as the observed status errno with
38 * NGX_ERROR, and return NGX_ERROR.
39 */
40static ngx_int_t
41brix_vfs_copy_fail(brix_vfs_ctx_t *ctx, const char *src, int err,
42 uint64_t start)
43{
44 errno = err;
45 brix_vfs_observe_ctx_op(ctx, src, BRIX_METRIC_OP_COPY, NULL, 0,
46 NGX_ERROR, err, start);
47 return NGX_ERROR;
48}

50/*
51 * vfs_copy_driver_precond — evaluate a carried publish precondition against
52 * the DESTINATION before the driver's server-copy replaces it (phase-107 C6).
53 *
54 * WHAT: For ABSENT, refuses with EEXIST when the destination exists; for
55 * MATCH_*, stats the destination and runs the shared stat-grammar
56 * evaluator (a missing target is a failed match, ECANCELED). Returns 0
57 * to proceed, -1 with errno set to refuse.
58 * WHY: server_copy has no precondition parameter (the C6 ABI change covered
59 * staged_commit only), so the honest evaluation here is check-then-act
60 * against the driver's own stat — never atomic, but a backend that
61 * silently IGNORED If-Match instead of answering 412 would be worse
62 * (§B.2: "ignores If-Match … must not be advertised"). A driver with no
63 * stat slot cannot evaluate at all and refuses ENOTSUP rather than
64 * pretending.
65 */
66static int
67vfs_copy_driver_precond(brix_sd_instance_t *leaf, const brix_sd_driver_t *drv,
68 const char *d, const brix_sd_precond_t *pre, const brix_sd_cred_t *cred)
69{
70 brix_sd_stat_t dst_st;
71 int exists;
72
73 if (pre == NULL || pre->kind == BRIX_SD_PRECOND_NONE) {
74 return 0;
75 }
76 if (drv->stat == NULL) {
77 errno = ENOTSUP;
78 return -1;
79 }
80 exists = brix_sd_stat_maybe_cred(leaf, d, &dst_st, cred) == NGX_OK;
81
82 if (pre->kind == BRIX_SD_PRECOND_ABSENT) {
83 if (exists) {
84 errno = EEXIST;
85 return -1;
86 }
87 return 0;
88 }
89 if (!exists) {
90 errno = ECANCELED;
91 return -1;
92 }
93 return brix_sd_precond_eval_stat(pre, dst_st.size, dst_st.mtime) != 0
94 ? -1 : 0;
95}

97/*
98 * brix_vfs_copy_driver — non-POSIX backend server-side copy path.
99 *
100 * WHAT: Copy the export-relative source `src` to `dst_resolved` through the
101 * driver's server-copy slot, enforcing the overwrite gate via a pre-stat
102 * and forwarding a per-open credential when the credential gate is armed.
103 * WHY: Object/S3-style backends replace bytes in-store, so the confined POSIX
104 * copy_file_range path does not apply; this branch dispatches on the leaf
105 * instance so *_maybe_cred forwarders reach the leaf driver's *_cred
106 * slots (decorators carry only plain relays).
107 * HOW: Re-fetch the leaf driver (non-NULL by the caller's guard), zero the
108 * cred (the gate fills only the active kind), resolve the credential,
109 * reject an existing destination when overwrite is unset, then call the
110 * server_copy slot (ENOTSUP when absent) and book the OP_COPY observation
111 * with the copied byte count.
112 */
113/* The destination gates for a driver server-copy: the overwrite pre-stat
114 * (EEXIST when the destination exists and overwrite is unset) and the carried
115 * publish precondition (C6) — both evaluated before server_copy replaces the
116 * destination's bytes, both check-then-act, which the protocol layer already
117 * treats as non-atomic on this arm. On refusal: wipes the credential store,
118 * books the C6 refusal telemetry (precondition arm), and returns the
119 * copy-fail verdict; the caller just propagates. */
120static ngx_int_t
121vfs_copy_driver_dst_gate(brix_vfs_ctx_t *ctx, brix_sd_instance_t *leaf,
122 const brix_sd_driver_t *drv, const char *src, const char *d,
123 const brix_vfs_copy_opts_t *opts, brix_sd_ucred_t *store,
124 const brix_sd_cred_t *cred, uint64_t start)
125{
126 brix_sd_stat_t dst_st;
127 int saved_errno;
128
129 if (opts == NULL) {
130 return NGX_OK;
131 }
132 if (!opts->overwrite && drv->stat != NULL
133 && brix_sd_stat_maybe_cred(leaf, d, &dst_st, cred) == NGX_OK)
134 {
135 brix_sd_ucred_wipe(store); /* secret consumed; erase (A-4/T4) */
136 return brix_vfs_copy_fail(ctx, src, EEXIST, start);
137 }
138 if (vfs_copy_driver_precond(leaf, drv, d, opts->precond, cred) != 0) {
139 saved_errno = errno;
140 brix_sd_ucred_wipe(store); /* secret consumed; erase (A-4/T4) */
141 /* C6 refusal telemetry: this arm's evaluation is check-then-act by
142 * construction (see the helper), so pre->atomic is 0 and a refusal
143 * books the advisory row too. */
144 brix_vfs_precond_refused_observe(opts->precond, saved_errno,
145 brix_sd_backend_name(leaf));
146 return brix_vfs_copy_fail(ctx, src, saved_errno, start);
147 }
148 return NGX_OK;
149}

151static ngx_int_t
152brix_vfs_copy_driver(brix_vfs_ctx_t *ctx, const char *src,
153 const char *dst_resolved, const brix_vfs_copy_opts_t *opts,
154 uint64_t start)
155{
156 const brix_sd_driver_t *drv = brix_vfs_ctx_driver(ctx);
157 char s[PATH_MAX];
158 char d[PATH_MAX];
159 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(ctx->sd);
160 brix_sd_ucred_t store;
161 brix_sd_cred_t cred;
162 off_t copied = 0;
163 ngx_int_t rc;
164 int saved_errno;
165 int use_cred = 0, cred_err = 0;
166
167 /* The sole caller checks the driver before dispatching here, but the
168 * accessor legitimately returns NULL for the default POSIX driver — keep
169 * the invariant local so a future caller cannot hand us a NULL vtable. */
170 if (drv == NULL) {
171 return brix_vfs_copy_fail(ctx, src, ENOTSUP, start);
172 }
173 if (brix_path_resolved_to_pfn(ctx, src, s, sizeof(s)) != NGX_OK
174 || brix_path_resolved_to_pfn(ctx, dst_resolved, d, sizeof(d)) != NGX_OK)
175 {
176 return brix_vfs_copy_fail(ctx, src, errno, start);
177 }
178
179 /* Zero before the gate: it fills only the active credential kind; an
180 * unzeroed cred hands a garbage inactive pointer to the driver's
181 * cred slot (bearer PASSTHROUGH would leave x509_proxy dangling). */
182 ngx_memzero(&cred, sizeof(cred));
183
184 if (brix_vfs_cred_gate_active(ctx)) {
185 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
186 != NGX_OK)
187 {
188 return brix_vfs_copy_fail(ctx, src, cred_err ? cred_err : EACCES,
189 start);
190 }
191 }
192
193 if (vfs_copy_driver_dst_gate(ctx, leaf, drv, src, d, opts, &store,
194 use_cred ? &cred : NULL, start) != NGX_OK)
195 {
196 return NGX_ERROR;
197 }
198
199 rc = (drv->server_copy != NULL)
200 ? brix_sd_server_copy_maybe_cred(leaf, s, d, &copied,
201 use_cred ? &cred : NULL)
202 : (errno = ENOTSUP, NGX_ERROR);
203 brix_sd_ucred_wipe(&store); /* secret consumed by copy; erase (A-4/T4) */
204 if (rc == NGX_OK) {
205 /* The leaf dispatch above skipped the cache decorator, so nothing has
206 * invalidated the DESTINATION — whose bytes this copy just replaced at
207 * the origin. Without this, a read of `d` keeps serving the pre-copy
208 * object from the store. (Same compensation vfs_unlink/vfs_rename make;
209 * no-op when ctx->sd is not a cache.) The source is unchanged. */
210 brix_metric_cache_evicted(brix_vfs_metrics_proto(ctx),
211 brix_sd_cache_evict(ctx->sd, d));
212 }
213 saved_errno = (rc == NGX_OK) ? 0 : errno;
214 brix_vfs_observe_ctx_op(ctx, src, BRIX_METRIC_OP_COPY, NULL,
215 rc == NGX_OK ? (size_t) copied : 0, rc,
216 saved_errno, start);
217 return rc;
218}

220/*
221 * brix_vfs_copy_ns — POSIX namespace copy path (copy_file_range + fallback).
222 *
223 * WHAT: Translate the public copy opts into brix_ns_copy_opts_t and run the
224 * confined brix_ns_local_copy (copy_file_range with a pread/pwrite
225 * fallback) from `src` to `dst_resolved`, booking OP_COPY on completion.
226 * WHY: This is the default POSIX-backend data mover; it must preserve the
227 * exact namespace status→errno mapping and the best-effort byte count.
228 * HOW: Zero-init and populate ns_opts, call brix_ns_local_copy, map a non-OK
229 * status to errno (sys_errno when set, else the status mapping) via the
230 * shared failure helper, and on success observe with res.bytes (the size
231 * the engine actually copied — no post-copy re-stat).
232 */
233static ngx_int_t
234brix_vfs_copy_ns(brix_vfs_ctx_t *ctx, const char *src,
235 const char *dst_resolved, const brix_vfs_copy_opts_t *opts,
236 uint64_t start)
237{
238 brix_ns_copy_opts_t ns_opts;
239 brix_ns_result_t res;
240
241 ngx_memzero(&ns_opts, sizeof(ns_opts));
242 if (opts != NULL) {
243 ns_opts.recursive = opts->recursive ? 1 : 0;
244 ns_opts.overwrite = opts->overwrite ? 1 : 0;
245 ns_opts.overwrite_dirs = opts->overwrite_dirs ? 1 : 0;
246 ns_opts.preserve_xattrs = opts->preserve_xattrs ? 1 : 0;
247 ns_opts.staged_commit = opts->staged_commit ? 1 : 0;
248 ns_opts.precond = opts->precond; /* C6: decided at the commit */
249 }
250
251 res = brix_ns_local_copy(ctx->log, ctx->root_canon, src, dst_resolved,
252 &ns_opts);
253 if (res.status != BRIX_NS_OK) {
254 int err = res.sys_errno != 0 ? res.sys_errno
255 : brix_vfs_ns_status_errno(res.status);
256
257 if (ns_opts.precond != NULL) {
258 /* C6 refusal telemetry. MATCH_* on this arm is a stat-compare
259 * (advisory); an ABSENT refusal came from O_EXCL or
260 * RENAME_NOREPLACE at the filesystem itself — atomic unless this
261 * host ever degraded (beneath.c latch). The carried precondition
262 * is const on the copy plane, so the verdict is restated on a
263 * local copy rather than written through it. */
264 brix_sd_precond_t seen = *ns_opts.precond;
265
266 seen.atomic = seen.kind == BRIX_SD_PRECOND_ABSENT
267 && !brix_renameat_noreplace_degraded();
268 brix_vfs_precond_refused_observe(&seen, err, "posix");
269 }
270 return brix_vfs_copy_fail(ctx, src, err, start);
271 }
272
273 brix_vfs_observe_ctx_op(ctx, src, BRIX_METRIC_OP_COPY, NULL,
274 (size_t) res.bytes, NGX_OK, 0, start);
275 return NGX_OK;
276}

278/* Copy the resolved ctx source file to dst_resolved (both under ctx->root_canon).
279 * Returns NGX_OK, or NGX_ERROR with errno set (mapped from the namespace status:
280 * EEXIST when the destination exists and overwrite is unset, EXDEV on a confine-
281 * ment escape, EISDIR/ECONFLICT on a directory source). Metered as OP_COPY. */
282ngx_int_t
283brix_vfs_copy(brix_vfs_ctx_t *ctx, const char *dst_resolved,
284 const brix_vfs_copy_opts_t *opts)
285{
286 const char *src = brix_vfs_ctx_path(ctx);
287 uint64_t start = brix_vfs_now_ns();
288 const brix_sd_driver_t *drv;
289
290 if (brix_vfs_confined_mutation_checked(ctx,
291 BRIX_VFS_MUTATE_COPY) != NGX_OK)
292 {
293 return brix_vfs_copy_fail(ctx, src, errno, start);
294 }
295
296 if (ctx->root_canon == NULL || dst_resolved == NULL) {
297 return brix_vfs_copy_fail(ctx, src, EINVAL, start);
298 }
299 if (brix_vfs_require_authorized_target(ctx, dst_resolved,
300 BRIX_VFS_MUTATE_COPY) != NGX_OK)
301 {
302 return brix_vfs_copy_fail(ctx, src, errno, start);
303 }
304
305 /* phase-107 C7: lock gate after the mutation gate (EROFS precedes EBUSY),
306 * on the DESTINATION — a copy reads the source and replaces the
307 * destination's bytes, so only the destination's lock coverage matters. */
308 if (brix_vfs_require_unlocked_at(ctx, dst_resolved,
309 BRIX_VFS_MUTATE_COPY) != NGX_OK)
310 {
311 return brix_vfs_copy_fail(ctx, src, errno, start);
312 }
313
314 /* Non-POSIX backend: copy through the driver's server-copy slot. */
315 drv = brix_vfs_ctx_driver(ctx);
316 if (drv != NULL) {
317 return brix_vfs_copy_driver(ctx, src, dst_resolved, opts, start);
318 }
319
320 return brix_vfs_copy_ns(ctx, src, dst_resolved, opts, start);
321}
