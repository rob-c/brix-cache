1/*
2 * sd_pblock_staged.c — staged atomic-publish vtable slots for the pblock driver.
3 *
4 * WHAT: Implements the staged-write path of brix_sd_pblock_driver:
5 * staged_open/write/commit/abort. Body bytes are written to a fresh blob id
6 * (no visible catalog row yet); commit publishes them atomically by
7 * inserting the final row that points at that blob (the staged blocks simply
8 * become the object — no copy or rename), while abort removes them.
9 *
10 * WHY: Split out of sd_pblock.c (phase-79) to keep every pblock file under the
11 * ~500-line, one-concept cap. Atomic publish is its own concern, distinct
12 * from the object lifecycle (sd_pblock.c), the hot byte path (sd_pblock_io.c)
13 * and the namespace ops (sd_pblock_namespace.c). The functions are
14 * non-static because the driver descriptor names them; commit reuses
15 * sd_pblock_drop_dst from the namespace file. Declarations live in
16 * sd_pblock_internal.h.
17 *
18 * HOW: staged_open gates the parent collection up front (POSIX parity with the
19 * POSIX driver's O_EXCL temp) before any blob is allocated, then reserves a
20 * blob id + object dir. staged_write appends through the packed-block engine
21 * with no persistent block-0 fd (blk0_fd = -1). commit does the replace/
22 * insert against the catalog and consumes the handle on success. ngx-free;
23 * gated by BRIX_HAVE_SQLITE like the rest of the backend.
24 */

78/*
79 * WHAT: Allocate a staged handle and its private pblock state together.
80 * WHY: A single failure path keeps staged-open admission easy to audit.
81 * HOW: Allocate both objects and report ENOMEM after releasing partial state.
82 */
83static brix_sd_staged_t *
84pblock_staged_alloc(pblock_staged_t **state, int *err_out)
85{
86 brix_sd_staged_t *handle = calloc(1, sizeof(*handle));
87
88 *state = calloc(1, sizeof(**state));
89 if (handle != NULL && *state != NULL)
90 return handle;
91 free(handle);
92 free(*state);
93 *state = NULL;
94 if (err_out != NULL)
95 *err_out = ENOMEM;
96 return NULL;
97}

101/* F5: refuse a quota-busting PUT before any body byte is accepted. The
102 * inode delta is known here (an overwrite of an existing row adds none);
103 * bytes are the declared final size when the client sent one (phase-107 C5:
104 * minus the row being replaced - commit re-admits the ACTUAL delta, so this
105 * is admission, not a charge), else 0 and byte admission waits for the
106 * commit as before. errno carries the refusal (EDQUOT). */
107static ngx_int_t
108pblock_staged_admit(pblock_state_t *st, const char *final_path,
109 off_t declared_size, uint32_t uid)
110{
111 pblock_meta prev;
112 int existed;
113 int64_t add_bytes = 0;
114
115 if (!st->quota) {
116 return NGX_OK;
117 }
118 existed = pblock_catalog_lookup(st->cat, final_path, &prev) == 0;
119 if (declared_size > 0) {
120 add_bytes = (int64_t) declared_size - (existed ? prev.size : 0);
121 }
122 if (pblock_quota_admit(st, uid, add_bytes, existed ? 0 : 1) != 0) {
123 return NGX_ERROR;
124 }
125 return NGX_OK;
126}

128/* sd_pblock_staged_open_as — staged open whose eventual committed row is owned
129 * by (uid, gid). The plain slot passes 0/0 (service); staged_open_cred
130 * (sd_pblock_cred.c) passes the requester's resolved catalog ids. */
131brix_sd_staged_t *
132sd_pblock_staged_open_as(brix_sd_instance_t *inst, const char *final_path,
133 mode_t mode, off_t declared_size, uint32_t uid, uint32_t gid, int *err_out)
134{
135 pblock_state_t *st = inst->state;
136 brix_sd_staged_t *handle;
137 pblock_staged_t *ps;
138
139 /* POSIX parity with the posix driver's staged temp (O_EXCL in the final
140 * directory): a missing parent collection fails HERE — before any blob
141 * is allocated or a single body byte is accepted — not at commit. */
142 if (pblock_catalog_parent_ok(st->cat, final_path) != 0) {
143 if (err_out != NULL) { *err_out = errno; }
144 return NULL;
145 }
146
147 if (pblock_staged_admit(st, final_path, declared_size, uid) != NGX_OK) {
148 if (err_out != NULL) { *err_out = errno; }
149 return NULL;
150 }
151
152 handle = pblock_staged_alloc(&ps, err_out);
153 if (handle == NULL) {
154 return NULL;
155 }
156
157 if (pblock_gen_blob_id(ps->blob_id) != 0
158 || pblock_ensure_obj_dir(st, ps->blob_id) != 0)
159 {
160 if (err_out != NULL) { *err_out = errno; }
161 free(handle);
162 free(ps);
163 return NULL;
164 }
165
166 ps->st = st;
167 ps->block_size = st->block_size;
168 ps->size = 0;
169 ps->mode = mode;
170 ps->uid = uid;
171 ps->gid = gid;
172 snprintf(ps->final_path, sizeof(ps->final_path), "%s", final_path);
173 if (st->refs) { /* F10 */
174 ps->wv = brix_wverify_begin();
175 }
176 if (st->snap) { /* F6: block restore */
177 __atomic_add_fetch(&st->open_files, 1, __ATOMIC_RELEASE);
178 }
179 handle->inst = inst;
180 handle->state = ps;
181 if (st->audit) { /* F17 */
182 pblock_audit_log(st->cat, "staged_open", final_path, "", uid, gid, 0, 0);
183 }
184 return handle;
185}

215/* The typed publish precondition against the CURRENT catalog row (C6):
216 * evaluated inside the pblock commit transaction, so the verdict is atomic
217 * with the publish. 0 = passes; -1 with errno = EEXIST (ABSENT vs an existing
218 * row), ECANCELED (MATCH_* mismatch or missing target) or ENOTSUP. */
219static int
220pblock_staged_precond(const brix_sd_precond_t *pre,
221 const pblock_meta *previous, int existed)
222{
223 if (pre == NULL || pre->kind == BRIX_SD_PRECOND_NONE)
224 return 0;
225 if (pre->kind == BRIX_SD_PRECOND_ABSENT) {
226 if (existed) {
227 errno = EEXIST;
228 return -1;
229 }
230 return 0;
231 }
232 if (!existed) {
233 errno = ECANCELED; /* MATCH_* against a missing target */
234 return -1;
235 }
236 return brix_sd_precond_eval_stat(pre, (off_t) previous->size,
237 (time_t) previous->mtime);
238}

240/*
241 * WHAT: Validate and, when necessary, remove a staged commit destination.
242 * WHY: Lease, quota, precondition, and history rules form one namespace gate.
243 * HOW: Lookup once, admit the delta, evaluate the typed publish precondition
244 * (phase-107 C6) against the looked-up row, preserve history, then drop
245 * an overwrite. The caller runs this inside BEGIN IMMEDIATE, so the
246 * compare and the publish commit or roll back together — pblock's
247 * preconditions are ATOMIC at the catalog.
248 */
249static int
250pblock_staged_prepare_destination(pblock_state_t *pst, pblock_staged_t *ps,
251 const brix_sd_precond_t *pre, pblock_meta *previous, int *existed)
252{
253 int rc = pblock_catalog_lookup(pst->cat, ps->final_path, previous);
254
255 if (rc < 0)
256 return -1;
257 if (pst->locks
258 && pblock_locks_ns_check(pst, ps->final_path, ps->uid) != 0)
259 return -1;
260 if (pblock_quota_admit(pst, ps->uid,
261 ps->size - (rc == 0 ? previous->size : 0), rc == 0 ? 0 : 1) != 0)
262 return -1;
263 *existed = rc == 0;
264 if (pblock_staged_precond(pre, previous, *existed) != 0)
265 return -1;
266 if (!*existed)
267 return 0;
268 if (pst->versions > 0)
269 (void) pblock_hist_version_push(pst, ps->final_path, previous);
270 return sd_pblock_drop_dst(pst, ps->final_path, previous) == NGX_OK ? 0 : -1;
271}

276 * HOW: Copy staged ownership and size, stamp time, mode, and transform kind.

280{
281 memset(meta, 0, sizeof(*meta));
282 meta->is_dir = 0;

284 meta->size = ps->size;

294/*
295 * WHAT: Record integrity, anomaly, and audit state after catalog publication.
296 * WHY: These observers must run only after the new row becomes authoritative.
297 * HOW: Update enabled facilities and consume the staged checksum accumulator.
298 */
299static void
300pblock_staged_record_commit(pblock_state_t *pst, pblock_staged_t *ps,
301 pblock_meta *meta, const pblock_meta *previous, int existed)
302{
303 if (pst->lab != NULL) {
304 if (existed)
305 pblock_anomaly_updated(pst, ps->final_path, previous->size,
306 previous->mtime);
307 else
308 pblock_anomaly_created(pst, ps->final_path);
309 }
310 if (pst->csi) /* F3: tag the blob */
311 (void) pblock_csi_flush(pst, ps->blob_id, ps->size, ps->block_size,
312 0, INT64_MAX);
313 {
314 int folded = 0; /* F10: dedup fold */
315
316 if (ps->wv != NULL) {
317 if (pst->refs) {
318 char hash[PBLOCK_REFS_HASH_CAP]; /* W3: sha256-first */
319
320 pblock_refs_wv_hash(ps->wv, ps->size, hash, sizeof(hash));
321 folded = pblock_refs_dedup_publish(pst, ps->final_path, meta,
322 hash) == 1;
323 }
324 brix_wverify_free(ps->wv);
325 }
326
327 /* phase-88 W2: a kept (not folded) small blob comes to rest in the
328 * packed arena — one shared-segment record instead of a per-object
329 * dir + block file. Single-block + untransformed only (the record is
330 * the raw logical bytes); best-effort: a refused admission simply
331 * keeps the striped layout. */
332 if (pst->pack && !folded
333 && ps->size > 0 && ps->size <= pst->pack_max
334 && ps->size <= ps->block_size
335 && pst->xform.kind == PBLOCK_XFORM_NONE)
336 {
337 (void) pblock_pack_admit(pst, meta->blob_id, ps->size,
338 ps->block_size);
339 }
340 }
341 if (pst->audit) {
342 char aux[32];
343
344 snprintf(aux, sizeof(aux), "w=%lld", (long long) ps->size);
345 pblock_audit_log(pst->cat, "commit", ps->final_path, aux,
346 ps->uid, ps->gid, 0, 0);
347 }
348}

350/* sd_pblock_staged_commit — publish the staged blocks atomically by inserting
351 * the final catalog row pointing at the staged blob id (the blocks simply become
352 * the final object — no copy or rename). On success the handle is consumed; on
353 * failure it stays valid and the caller must staged_abort to release it. */
354ngx_int_t
355sd_pblock_staged_commit(brix_sd_staged_t *st, brix_sd_precond_t *pre)
356{
357 pblock_staged_t *ps = st->state;
358 pblock_state_t *pst = ps->st;
359 pblock_meta meta, dmeta;
360 int existed;
361 int err;
362
363 /* C6: one BEGIN IMMEDIATE spans the destination gate (lookup, lease,
364 * quota, PRECONDITION, history, overwrite drop) and the publishing
365 * catalog_put, so compare-and-publish is atomic at the catalog — no
366 * concurrent commit can slip a row in between the check and the put
367 * (the per-driver verdict in §C6's table). The post-publish observers
368 * (anomaly/csi/dedup/pack/audit) stay OUTSIDE: some run their own
369 * transactions, and they must only see an authoritative row. */
370 if (cat_exec(pst->cat, "BEGIN IMMEDIATE;") != 0) {
371 errno = EIO;
372 return NGX_ERROR;
373 }
374 if (pblock_staged_prepare_destination(pst, ps, pre, &dmeta,
375 &existed) != 0) {
376 err = errno;
377 (void) cat_exec(pst->cat, "ROLLBACK;");
378 if (pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE
379 && (err == EEXIST || err == ECANCELED))
380 {
381 pre->atomic = 1; /* refused inside the transaction — a
382 * storage-decided verdict (C6 advisory) */
383 }
384 errno = err;
385 return NGX_ERROR;
386 }
387 pblock_staged_build_meta(ps, &meta);
388
389 /* F7: a crash here leaves the staged blocks on disk with no catalog row
390 * (the open transaction rolls back in WAL recovery) — the canonical
391 * orphan-blob residue pblock-fsck must detect and --gc. */
392 pblock_lab_crash(pst->lab, "mid_staged_commit");
393
394 if (pblock_catalog_put(pst->cat, ps->final_path, &meta) != 0) {
395 err = errno;
396 (void) cat_exec(pst->cat, "ROLLBACK;");
397 errno = err;
398 return NGX_ERROR;
399 }
400 if (cat_exec(pst->cat, "COMMIT;") != 0) {
401 (void) cat_exec(pst->cat, "ROLLBACK;");
402 errno = EIO;
403 return NGX_ERROR;
404 }
405 if (pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE) {
406 pre->atomic = 1; /* decided inside the transaction */
407 }
408
409 pblock_staged_record_commit(pst, ps, &meta, &dmeta, existed);
410
411 if (pst->snap) { /* F6: released — no longer blocks restore */
412 __atomic_sub_fetch(&pst->open_files, 1, __ATOMIC_RELEASE);
413 }
414 free(ps);
415 free(st);
416 return NGX_OK;
417}

419/* sd_pblock_staged_path — driver->staged_path for pblock (phase-88 W1).
420 *
421 * WHAT: The physical path of the staged bytes, for the cache tier's
422 * verify-before-commit (phase-68 cvmfs-cas / manifest signature).
423 * Returns the staged blob's block-0 file when that single file IS the
424 * whole plaintext object — the staged writes fit one block AND no
425 * per-block transform is armed — else NULL ("no path available", the
426 * verify fails closed rather than checking partial/encoded bytes).
427 *
428 * WHY: pblock stripes objects across block files, so a general staged blob
429 * has no single verifiable path. But a cvmfs cache store's objects are
430 * bounded (the publisher's chunk ceiling), so with block_size sized
431 * above that bound every fill is single-block and pblock can serve the
432 * same verify contract as the posix .part file.
433 *
434 * HOW: 1. Gate on size <= block_size and xform NONE (raw bytes on disk).
435 * 2. Resolve the block-0 path into the handle's lazily-filled buffer.
436 */
437const char *
438sd_pblock_staged_path(const brix_sd_staged_t *st)
439{
440 pblock_staged_t *ps = st->state;
441
442 if (ps == NULL
443 || ps->size > ps->block_size
444 || ps->st->xform.kind != PBLOCK_XFORM_NONE)
445 {
446 return NULL;
447 }
448 if (ps->part_path[0] == '\0'
449 && pblock_block_path(ps->st, ps->blob_id, 0, ps->part_path,
450 sizeof(ps->part_path)) != 0)
451 {
452 ps->part_path[0] = '\0';
453 return NULL;
454 }
455 if (ps->size == 0) {
456 /* A zero-byte stage has no block file yet (blocks materialise on the
457 * first write); the verify contract expects an openable path — give it
458 * the empty block 0, exactly what the posix .part is at this point. */
459 int fd = open(ps->part_path, O_RDWR | O_CREAT, 0600);
460
461 if (fd < 0) {
462 return NULL;
463 }
464 close(fd);
465 }
466 return ps->part_path;
467}
