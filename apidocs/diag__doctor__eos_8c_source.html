1/*
2 * diag_doctor_eos.c — EOS-dialect topology enrichment for `xrddiag --map`.
3 *
4 * The XRootD mesh (kXR_locate / CMSD) is structurally blind to an EOS FST farm:
5 * an EOS MGM answers kXR_locate for ANY path with *itself* plus the aggregate
6 * space report, revealing the storage nodes only at open/redirect time. To draw
7 * the real cluster we speak EOS's own out-of-band command channel: the client
8 * opens a magic path `/proc/{user,admin}/` with the command in the CGI opaque
9 * (e.g. `?mgm.cmd=version`) and reads the reply envelope
10 * mgm.proc.stdout=<data>&mgm.proc.stderr=<err>&mgm.proc.retc=<N>
11 *
12 * - `/proc/user/?mgm.cmd=version` → detect EOS + banner
13 * - `/proc/admin/?mgm.cmd=fs&mgm.subcmd=ls&mgm.outformat=m` → enumerate the FSTs
14 *
15 * `fs ls` is admin-gated. An identity without admin rights (e.g. one nginx maps
16 * to `nobody`) fails the /proc/admin/ open with NotAuthorized — recognised, not
17 * unknown — so we mark the MGM `gated` and degrade gracefully rather than error.
18 *
19 * The transport (doctor_eos_proc / doctor_eos_map) is the only wire code; every
20 * parser below (stdout/kv/retc/version/fs) is pure over a caller-supplied buffer
21 * and unit-tested off recorded EOS output. Renderers are pure over doctor_ep.
22 */

29/* 1 when `p` (a strstr hit for a klen-byte key inside `rec`) sits on a token
30 * boundary and is immediately followed by '=' — i.e. it is a whole key. */
31static int
32eos_kv_at(const char *rec, const char *p, size_t klen)
33{
34 char before = (p == rec) ? ' ' : p[-1];
35
36 return (before == ' ' || before == '\t' || before == '\n' || before == '\r')
37 && p[klen] == '=';
38}

41/* Copy one value, which runs from `v` to the next space/tab/newline/&/nul.
42 * `v[j] != '\0'` is tested first, so the separator set is never searched for a
43 * NUL (strchr would report a spurious hit on the terminator). */
44static void
45eos_kv_copy(const char *v, char *out, size_t osz)
46{
47 size_t j = 0;
48
49 while (v[j] != '\0' && strchr(" \t\n\r&", v[j]) == NULL && j < osz - 1) {
50 out[j] = v[j];
51 j++;
52 }
53 out[j] = '\0';
54}

56/* Extract the value of an exact key from an EOS record. The key must sit on a
57 * token boundary (start-of-record or preceding whitespace) and be immediately
58 * followed by '='; the value runs to the next space/&/newline/nul. This exact-
59 * boundary match is what keeps `host` from matching inside `hostport`/`stat.host`
60 * and makes the parser tolerant of unknown extra keys and key-order changes.
61 * Returns 0 with `out` filled (NUL-terminated) on a hit, -1 otherwise. */
62int
63doctor_eos_kv(const char *rec, const char *key, char *out, size_t osz)
64{
65 size_t klen;
66 const char *p;
67
68 if (out == NULL || osz == 0) { return -1; }
69 out[0] = '\0';
70 if (rec == NULL || key == NULL) { return -1; }
71 klen = strlen(key);
72 if (klen == 0) { return -1; }
73
74 for (p = rec; (p = strstr(p, key)) != NULL; p += klen) {
75 if (eos_kv_at(rec, p, klen)) {
76 eos_kv_copy(p + klen + 1, out, osz);
77 return 0;
78 }
79 }
80 return -1;
81}

83/* Locate the `mgm.proc.stdout=` payload span inside a proc reply envelope. The
84 * value ends at the next envelope key (`&mgm.proc.stderr=` / `&mgm.proc.retc=`)
85 * or end-of-buffer. Returns 0 with [*start,*start+*len) set, -1 if absent. */
86int
87doctor_eos_stdout(const char *body, const char **start, int *len)
88{
89 const char *s, *e;
90
91 if (body == NULL || start == NULL || len == NULL) { return -1; }
92 s = strstr(body, "mgm.proc.stdout=");
93 if (s == NULL) { return -1; }
94 s += strlen("mgm.proc.stdout=");
95 e = strstr(s, "&mgm.proc.stderr=");
96 if (e == NULL) { e = strstr(s, "&mgm.proc.retc="); }
97 if (e == NULL) { e = s + strlen(s); }
98 *start = s;
99 *len = (int) (e - s);
100 return 0;
101}

103/* The command return code from a proc reply. Handles both the line envelope
104 * (`mgm.proc.retc=N`) and the JSON envelope (`"retc" : N`). -1 = not present. */
105int
106doctor_eos_retc(const char *body)
107{
108 const char *p;
109
110 if (body == NULL) { return -1; }
111 p = strstr(body, "mgm.proc.retc=");
112 if (p != NULL) { return atoi(p + strlen("mgm.proc.retc=")); }
113 p = strstr(body, "\"retc\"");
114 if (p != NULL) {
115 p = strchr(p, ':');
116 if (p != NULL) {
117 p++;
118 while (*p == ' ' || *p == '"') { p++; }
119 return atoi(p);
120 }
121 }
122 return -1;
123}

125/* Parse `mgm.cmd=version` output into the MGM banner. The stdout payload is
126 * `EOS_INSTANCE=... EOS_SERVER_VERSION=... ...`; because the first token abuts
127 * the `stdout=` separator we copy the span into a local buffer so the boundary
128 * rule in doctor_eos_kv sees it at start-of-record. Returns 1 if this endpoint
129 * is an EOS MGM (EOS_INSTANCE present), 0 otherwise. */
130int
131doctor_eos_parse_version(const char *body, doctor_eos *eos)
132{
133 const char *s;
134 int len;
135 char buf[512];
136 char v[64];
137
138 if (eos == NULL) { return 0; }
139 if (doctor_eos_stdout(body, &s, &len) != 0 || len <= 0) { return 0; }
140 if (len > (int) sizeof(buf) - 1) { len = (int) sizeof(buf) - 1; }
141 memcpy(buf, s, (size_t) len);
142 buf[len] = '\0';
143
144 if (doctor_eos_kv(buf, "EOS_INSTANCE", v, sizeof(v)) != 0) { return 0; }
145 eos->kind = DOC_EOS_MGM;
146 snprintf(eos->instance, sizeof(eos->instance), "%.*s",
147 (int) sizeof(eos->instance) - 1, v);
148 if (doctor_eos_kv(buf, "EOS_SERVER_VERSION", v, sizeof(v)) == 0) {
149 snprintf(eos->version, sizeof(eos->version), "%.*s",
150 (int) sizeof(eos->version) - 1, v);
151 }
152 return 1;
153}

155/* Copy the record starting at sout[*i] (up to the next newline, which is
156 * consumed) into `rec`, truncating at the buffer. Returns the record length. */
157static int
158eos_next_record(const char *sout, int len, int *i, char *rec, size_t rsz)
159{
160 int j = 0;
161
162 while (*i < len && sout[*i] != '\n') {
163 if (j < (int) rsz - 1) {
164 rec[j++] = sout[*i];
165 }
166 (*i)++;
167 }
168 rec[j] = '\0';
169 if (*i < len) {
170 (*i)++; /* consume the newline */
171 }
172 return j;
173}

176/* Project one `fs ls -m` record onto an endpoint slot: address, EOS facts, and
177 * the derived cfg/CMS views. `host` is the already-extracted host key and may
178 * carry an embedded :port, which is used when no `port=` key is present. */
179static void
180eos_fill_fst(const char *rec, char *host, doctor_ep *e)
181{
182 char port[16], val[64];
183 char *colon;
184
185 memset(e, 0, sizeof(*e));
186 e->proto = DXP_ROOT;
187
188 colon = strrchr(host, ':');
189 if (doctor_eos_kv(rec, "port", port, sizeof(port)) == 0) {
190 e->port = atoi(port);
191 } else if (colon != NULL) {
192 *colon = '\0';
193 e->port = atoi(colon + 1);
194 }
195 if (e->port == 0) { e->port = 1095; } /* the FST default port */
196 snprintf(e->host, sizeof(e->host), "%s", host);
197
198 e->eos.kind = DOC_EOS_FST;
199 doctor_eos_kv(rec, "stat.geotag", e->eos.geotag, sizeof(e->eos.geotag));
200 doctor_eos_kv(rec, "configstatus", e->eos.cfgstatus,
201 sizeof(e->eos.cfgstatus));
202 if (doctor_eos_kv(rec, "stat.boot", val, sizeof(val)) == 0) {
203 e->eos.booted = (strcmp(val, "booted") == 0);
204 }
205 if (doctor_eos_kv(rec, "stat.active", val, sizeof(val)) == 0) {
206 e->eos.active = (strcmp(val, "online") == 0);
207 }
208 if (doctor_eos_kv(rec, "stat.statfs.capacity", val, sizeof(val)) == 0) {
209 e->eos.cap_bytes = strtoll(val, NULL, 10);
210 }
211 if (doctor_eos_kv(rec, "stat.statfs.freebytes", val, sizeof(val)) == 0) {
212 e->eos.free_bytes = strtoll(val, NULL, 10);
213 }
214
215 /* Mirror capacity into cfg so the shared "NN% free" renderer works, and
216 * type the FST on the CMS plane (a data server; rw/ro from configstatus)
217 * so the mesh map colours it consistently with located holders. */
218 e->cfg.space_total = e->eos.cap_bytes;
219 e->cfg.space_free = e->eos.free_bytes;
220 if (e->eos.cap_bytes > 0) { e->cfg.scraped = 1; }
221 e->cms.reported = 1;
222 e->cms.role = DOC_CMS_SERVER;
223 e->cms.write = (strstr(e->eos.cfgstatus, "rw") != NULL);
224}

227/* Parse `fs ls -m` (monitoring) output — one filesystem per line — appending an
228 * FST endpoint per record into arr[start..cap). A record is a filesystem iff it
229 * carries a `host=` (fs ls) or `hostport=` (node ls) key. Returns the count of
230 * endpoints appended. Pure: no allocation, no wire. */
231int
232doctor_eos_parse_fs(const char *sout, int len, doctor_ep *arr, int cap, int start)
233{
234 int i = 0, n = start, added = 0;
235 char rec[4096], host[256];
236
237 if (sout == NULL || arr == NULL) { return 0; }
238
239 while (i < len && n < cap) {
240 if (eos_next_record(sout, len, &i, rec, sizeof(rec)) == 0) {
241 continue; /* blank line */
242 }
243 if (doctor_eos_kv(rec, "host", host, sizeof(host)) != 0
244 || host[0] == '\0') {
245 if (doctor_eos_kv(rec, "hostport", host, sizeof(host)) != 0) {
246 continue; /* not a filesystem/node record */
247 }
248 }
249
250 eos_fill_fst(rec, host, &arr[n]);
251 n++;
252 added++;
253 }
254 return added;
255}

259/* Run one EOS /proc command: open the magic path with the command as the CGI
260 * opaque, read the whole reply (proc replies stat as size 0, so grow-read until
261 * EOF), close. Returns a malloc'd NUL-terminated body via *out (caller frees),
262 * or -1 — an open failure (NotAuthorized on an admin-gated command) included. */
263int
264doctor_eos_proc(brix_conn *c, const char *dir, const char *cmd,
265 char **out, brix_status *st)
266{
267 brix_file f;
268 char *buf = NULL;
269 size_t cap = 0, len = 0;
270 int64_t off = 0;
271
272 memset(&f, 0, sizeof(f));
273 if (brix_file_open_opaque(c, dir, cmd, 0, 0, 0, &f, st) != 0) {
274 return -1;
275 }
276 for (;;) {
277 ssize_t r;
278 if (len + 65536 + 1 > cap) {
279 size_t ncap = cap ? cap * 2 : 131072;
280 char *nb = realloc(buf, ncap);
281 if (nb == NULL) {
282 free(buf);
283 brix_file_close(c, &f, st);
284 return -1;
285 }
286 buf = nb;
287 cap = ncap;
288 }
289 r = brix_file_read(c, &f, off, buf + len, 65536, st);
290 if (r < 0) {
291 free(buf);
292 brix_file_close(c, &f, st);
293 return -1;
294 }
295 if (r == 0) { break; }
296 len += (size_t) r;
297 off += r;
298 }
299 brix_file_close(c, &f, st);
300
301 if (buf == NULL) {
302 buf = malloc(1);
303 if (buf == NULL) { return -1; }
304 }
305 buf[len] = '\0';
306 *out = buf;
307 return 0;
308}

310/* Admin plane: an `fs ls -m` answer enumerates the whole FST farm. Replaces the
311 * located self-node with arr[1..]. Returns 1 when the farm hit the array cap. */
312static int
313eos_map_admin(const char *body, doctor_ep *arr, int cap, int *n)
314{
315 const char *sout;
316 int slen, added;
317
318 if (doctor_eos_stdout(body, &sout, &slen) != 0) {
319 return 0;
320 }
321 added = doctor_eos_parse_fs(sout, slen, arr, cap, 1);
322 *n = 1 + added; /* drop the CMS self-node */
323 arr[0].eos.fst_count = added;
324 return added >= cap - 1;
325}

351/* Enrich the fan-out array with the EOS view of urls[0]: detect the MGM via the
352 * version banner and, if enumerable, replace the CMS self-node with the real FST
353 * farm (arr[1..]). Only runs for `--map`. Returns 1 if FSTs were truncated. */
354int
355doctor_eos_map(const diag_args *a, doctor_ep *arr, int cap, int *n)
356{
357 brix_url u;
358 brix_conn c;
359 brix_status st;
360 char *body = NULL;
361 int trunc = 0;
362
363 if (a == NULL || arr == NULL || n == NULL) { return 0; }
364 if (!a->map || *n < 1 || cap < 2) { return 0; }
365
366 brix_status_clear(&st);
367 if (brix_endpoint_parse(a->urls[0], &u, &st) != 0) { return 0; }
368 if (brix_connect(&c, &u, &a->conn, &st) != 0) { return 0; }
369 c.io.timeout_ms = a->probe_timeout_ms > 0 ? a->probe_timeout_ms : 8000;
370
371 brix_status_clear(&st);
372 if (doctor_eos_proc(&c, "/proc/user/", "mgm.cmd=version", &body, &st) != 0
373 || !doctor_eos_parse_version(body, &arr[0].eos)) {
374 free(body);
375 brix_close(&c);
376 return 0; /* not an EOS MGM — mesh untouched */
377 }
378 free(body);
379 body = NULL;
380 brix_status_clear(&st);
381 if (doctor_eos_proc(&c, "/proc/admin/",
382 "mgm.cmd=fs&mgm.subcmd=ls&mgm.outformat=m",
383 &body, &st) == 0
384 && doctor_eos_retc(body) == 0) {
385 trunc = eos_map_admin(body, arr, cap, n);
386 } else {
387 trunc = eos_map_fileinfo(a, &c, arr, cap, n, &st);
388 }
389 free(body);
390 brix_close(&c);
391 return trunc;
392}

406/* Text-report line for an enumerated FST. Returns 1 if it printed (this ep is an
407 * FST), 0 otherwise so the caller can fall through to the generic renderer. */
408int
409doctor_eos_report_fst(const doctor_ep *e)
410{
411 int pct;
412
413 if (e->eos.kind != DOC_EOS_FST) { return 0; }
414 pct = doctor_cfg_capacity_pct(e->cfg.space_total, e->cfg.space_free);
415
416 printf("\n[%s] EOS FST %s:%d\n", doctor_eos_fst_health(&e->eos),
417 e->host, e->port);
418 printf(" eos: geo=%s %s %s cfg=%s",
419 e->eos.geotag[0] ? e->eos.geotag : "?",
420 e->eos.booted ? "booted" : "not-booted",
421 e->eos.active ? "online" : "offline",
422 e->eos.cfgstatus[0] ? e->eos.cfgstatus : "?");
423 if (pct >= 0) { printf(" %d%% free", pct); }
424 printf(" (%s)\n", e->eos.sampled ? "via fileinfo replica sampling"
425 : "from MGM fs ls");
426 return 1;
427}
