1/*
2 * sd_ram_staged.c — the RAM store's write and listing surface.
3 *
4 * WHAT: the staged fill spine (the store's only writer) and prefix directory
5 * iteration. The in-memory user.* xattrs that carried the cache's cinfo
6 * record alongside them are now in sd_ram_xattr.c; the cut was the size
7 * cap, and it was free — that plane touches nothing here but the entry it
8 * looks up.
9 * WHY: sd_ram.c owns the read path and the vtable; keeping the mutating half
10 * beside it — not inside it — holds both files inside the size cap and
11 * puts the whole "how does a byte get in" story in one place.
12 * HOW: a fill builds a DETACHED entry and publishes it under one mutex hold,
13 * so a reader never observes a half-filled object under its final name.
14 * The declared size is RESERVED at open, which is what makes the cap a
15 * guarantee rather than a race: two concurrent fills cannot both be told
16 * there is room for the same bytes.
17 */

30brix_sd_staged_t *
31brix_sd_ram_staged_open(brix_sd_instance_t *inst, const char *final_path,
32 mode_t mode, off_t declared_size, int *err_out)
33{
34 sd_ram_state_t *st = inst->state;
35 brix_sd_staged_t *sh;
36 sd_ram_staged_state_t *ss;
37 int err = 0;
38
39 (void) mode;
40 if (!brix_sd_ram_path_ok(final_path) || declared_size < 0) {
41 err = EINVAL;
42 } else if (st->capacity < (uint64_t) declared_size) {
43 /* The object can never fit, whatever is evicted first — say so now
44 * rather than after pulling it across the network. */
45 err = ENOSPC;
46 }
47 if (err != 0) {
48 if (err_out != NULL) { *err_out = err; }
49 errno = err;
50 return NULL;
51 }
52
53 sh = calloc(1, sizeof(*sh) + sizeof(*ss));
54 if (sh == NULL) {
55 if (err_out != NULL) { *err_out = ENOMEM; }
56 errno = ENOMEM;
57 return NULL;
58 }
59 ss = (sd_ram_staged_state_t *) (sh + 1);
60
61 (void) pthread_mutex_lock(&st->mtx);
62 if (brix_sd_ram_make_room(st, (uint64_t) declared_size) != NGX_OK) {
63 err = ENOSPC; /* full of live objects: degrade, §16 */
64 } else {
65 ss->ent = brix_sd_ram_ent_new(st, final_path, 0);
66 if (ss->ent == NULL) {
67 err = errno;
68 } else {
69 ss->reserved = (uint64_t) declared_size;
70 st->reserved += ss->reserved;
71 }
72 }
73 (void) pthread_mutex_unlock(&st->mtx);
74
75 if (err != 0) {
76 free(sh);
77 if (err_out != NULL) { *err_out = err; }
78 errno = err;
79 return NULL;
80 }
81 sh->inst = inst;
82 sh->state = ss;
83 return sh;
84}

86ssize_t
87brix_sd_ram_staged_write(brix_sd_staged_t *sh, const void *buf, size_t len,
88 off_t off)
89{
90 sd_ram_state_t *st = sh->inst->state;
91 sd_ram_staged_state_t *ss = sh->state;
92 uint64_t end;
93 ngx_int_t rc;
94
95 if (off < 0) {
96 errno = EINVAL;
97 return -1;
98 }
99 end = (uint64_t) off + (uint64_t) len;
100 if (end > (uint64_t) SSIZE_MAX) {
101 errno = EFBIG;
102 return -1;
103 }
104
105 (void) pthread_mutex_lock(&st->mtx);
106 rc = brix_sd_ram_ent_grow(st, ss->ent, (size_t) end, &ss->reserved);
107 if (rc == NGX_OK) {
108 memcpy(ss->ent->data + off, buf, len);
109 if (end > (uint64_t) ss->ent->size) {
110 ss->ent->size = (size_t) end;
111 }
112 }
113 (void) pthread_mutex_unlock(&st->mtx);
114
115 return (rc == NGX_OK) ? (ssize_t) len : -1; /* errno set by grow */
116}

119static void
120sd_ram_staged_unreserve(sd_ram_state_t *st, sd_ram_staged_state_t *ss)
121{
122 st->reserved -= (st->reserved >= ss->reserved) ? ss->reserved : st->reserved;

126/* Evaluate `pre` against whatever currently holds the fill's final name.
127 * Caller holds st->mtx. Returns 0 when the publish may proceed, else the errno
128 * to refuse with. Every refusal is storage-decided (the compare and the publish
129 * happen under one hold), so pre->atomic is set on refusals too. */
130static int
131sd_ram_staged_precond(sd_ram_state_t *st, const char *path,
132 brix_sd_precond_t *pre)
133{
134 sd_ram_ent_t *cur;
135
136 if (pre == NULL || pre->kind == BRIX_SD_PRECOND_NONE) {
137 return 0;
138 }
139 pre->atomic = 1;
140 cur = brix_sd_ram_find(st, path);
141
142 if (pre->kind == BRIX_SD_PRECOND_ABSENT) {
143 return (cur != NULL) ? EEXIST : 0;
144 }
145 if (cur == NULL) {
146 return ENOENT;
147 }
148 if (brix_sd_precond_eval_stat(pre, (off_t) cur->size, cur->mtime) != 0) {
149 return errno; /* ECANCELED, or ENOTSUP for a kind
150 * this evaluator does not know */
151 }
152 return 0;
153}

155ngx_int_t
156brix_sd_ram_staged_commit(brix_sd_staged_t *sh, brix_sd_precond_t *pre)
157{
158 sd_ram_state_t *st = sh->inst->state;
159 sd_ram_staged_state_t *ss = sh->state;
160 int err;
161
162 (void) pthread_mutex_lock(&st->mtx);
163 err = sd_ram_staged_precond(st, ss->ent->path, pre);
164 if (err == 0) {
165 ss->ent->mtime = time(NULL);
166 brix_sd_ram_insert(st, ss->ent); /* charges ent->cap to st->used */
167 ss->ent = NULL;
168 }
169 sd_ram_staged_unreserve(st, ss);
170 if (err != 0) {
171 brix_sd_ram_unref(st, ss->ent); /* never charged: drop the bytes */
172 ss->ent = NULL;
173 }
174 (void) pthread_mutex_unlock(&st->mtx);
175
176 free(sh);
177 if (err != 0) {
178 errno = err;
179 return NGX_ERROR;
180 }
181 return NGX_OK;
182}

207/*
208 * A listing is a SNAPSHOT taken under the lock: the names are copied out at
209 * opendir and readdir walks the copy. The alternative — holding a cursor into
210 * live hash chains across readdir calls — would either hold the mutex for the
211 * whole walk (blocking every fill) or follow a chain a concurrent evict had
212 * already freed. The cache's reaper scans and evicts in the same pass, so that
213 * is not a hypothetical race.
214 */
215typedef struct {
216 char **names; /* owned, each strdup'd */
217 u_char *types; /* DT_REG / DT_DIR */
218 size_t n;
219 size_t i;
220} sd_ram_dir_t;
221
222/* Does `path` name a direct child of directory `dir` (which has no trailing
223 * '/')? Writes the child component into *out when so. */
224static int
225sd_ram_child_of(const char *path, const char *dir, size_t dirlen,
226 const char **out)
227{
228 const char *rest;
229
230 if (strncmp(path, dir, dirlen) != 0 || path[dirlen] != '/') {
231 return 0;
232 }
233 rest = path + dirlen + 1;
234 return (*rest != '\0') ? ((*out = rest), 1) : 0;
235}
236
237/* Append one child name (up to its next '/') to the snapshot. Returns NGX_OK,
238 * or NGX_ERROR with errno ENOMEM. */
239static ngx_int_t
240sd_ram_dir_add(sd_ram_dir_t *d, const char *child, size_t cap)
241{
242 const char *slash = strchr(child, '/');
243 size_t len = (slash != NULL) ? (size_t) (slash - child)
244 : strlen(child);
245 size_t i;
246 char *name;
247
248 if (len == 0 || len >= 256 || d->n >= cap) {
249 return NGX_OK; /* unnameable or snapshot full */
250 }
251 for (i = 0; i < d->n; i++) { /* one entry per intermediate dir */
252 if (strncmp(d->names[i], child, len) == 0
253 && d->names[i][len] == '\0')
254 {
255 return NGX_OK;
256 }
257 }
258 name = strndup(child, len);
259 if (name == NULL) {
260 errno = ENOMEM;
261 return NGX_ERROR;
262 }
263 /* Ownership transfers INTO the snapshot; sd_ram_dir_free releases every
264 * entry.
265 *
266 * gcc 11's -fanalyzer cannot see a symbolic-index store through a
267 * parameter as an escape, so it reports `name` leaking at the return
268 * below. Suppress exactly that diagnostic, at exactly its emission point. */
269#pragma GCC diagnostic push
270#if defined(__GNUC__) && !defined(__clang__)
271#pragma GCC diagnostic ignored "-Wanalyzer-malloc-leak"
272#endif
273 d->names[d->n] = name;
274 d->types[d->n] = (slash != NULL) ? DT_DIR : DT_REG;
275 d->n++;
276 return NGX_OK;
277#pragma GCC diagnostic pop
278}

280/* Fill the snapshot with every direct child of `path`. Caller holds st->mtx. */
281static ngx_int_t
282sd_ram_dir_snapshot(sd_ram_state_t *st, const char *path, sd_ram_dir_t *d,
283 size_t cap)
284{
285 size_t dirlen = strlen(path);
286 size_t b;
287
288 while (dirlen > 1 && path[dirlen - 1] == '/') {
289 dirlen--; /* "/a/" and "/a" list the same */
290 }
291 if (dirlen == 1) {
292 dirlen = 0; /* the root: every path is a child */
293 }
294 for (b = 0; b < st->nbuckets; b++) {
295 sd_ram_ent_t *e;
296
297 for (e = st->buckets[b]; e != NULL; e = e->hnext) {
298 const char *child = NULL;
299
300 if (!sd_ram_child_of(e->path, path, dirlen, &child)) {
301 continue;
302 }
303 if (sd_ram_dir_add(d, child, cap) != NGX_OK) {
304 return NGX_ERROR;
305 }
306 }
307 }
308 return NGX_OK;
309}

311static void
312sd_ram_dir_free(sd_ram_dir_t *d)
313{
314 size_t i;
315
316 for (i = 0; i < d->n; i++) {
317 free(d->names[i]);
318 }
319 free(d->names);
320 free(d->types);
321 free(d);
322}

324/* The finished listing snapshot for `path`, or NULL when out of memory.
325 * Caller holds st->mtx. All-or-nothing: every partial allocation is released
326 * here, so the caller owns either a whole snapshot or nothing — which is also
327 * the shape gcc 11's -fanalyzer can follow, the arrays being allocated and
328 * released inside one function rather than across the opendir/error-arm pair. */
329static sd_ram_dir_t *
330sd_ram_dir_build(sd_ram_state_t *st, const char *path, size_t cap)
331{
332 sd_ram_dir_t *d = calloc(1, sizeof(*d));
333
334 if (d == NULL) {
335 return NULL;
336 }
337 /* Guards kept as separate ifs: the analyzer loses track of the first
338 * allocation across a compound `a == NULL || b == NULL` test. */
339 d->names = calloc(cap + 1, sizeof(*d->names));
340 if (d->names == NULL) {
341 sd_ram_dir_free(d);
342 return NULL;
343 }
344 d->types = calloc(cap + 1, sizeof(*d->types));
345 if (d->types == NULL) {
346 sd_ram_dir_free(d);
347 return NULL;
348 }
349 if (sd_ram_dir_snapshot(st, path, d, cap) != NGX_OK) {
350 sd_ram_dir_free(d);
351 return NULL;
352 }
353 return d;
354}
