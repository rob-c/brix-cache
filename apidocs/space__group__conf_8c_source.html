1/*
2 * space_group_conf.c — parser + lookups for `brix_oss_space` (phase-115 W3.3).
3 *
4 * WHAT: validates one `brix_oss_space <group> <prefix> [quota=...]` line into
5 * the server's brix_oss_space_t table and answers the two lookups the
6 * write gate, kXR_Qspace and kXR_open use (by path, by name).
7 *
8 * WHY: a group name reaches the wire verbatim and a prefix is compared to
9 * confined logical paths, so both are pinned at parse time: a name that
10 * could smuggle a second oss.* key, a prefix with dot segments (it could
11 * never match a canonical path, so the quota would silently never
12 * apply) or a duplicate name/prefix fails `nginx -t` instead of
13 * mis-accounting at runtime.
14 *
15 * HOW: pure functions over ngx_str_t; the setter appends to an ngx_array_t
16 * created lazily in the config pool. Lookups are linear (tables are a
17 * handful of VOs); longest prefix wins so nested groups nest naturally.
18 */
19#include "space_group_conf.h"
20#include "core/ngx_brix_module.h" /* ngx_stream_brix_srv_conf_t.oss_spaces */

22int
23brix_oss_space_name_ok(const ngx_str_t *name)
24{
25 size_t i;
26
27 if (name->len == 0) {
28 return 0;
29 }
30 for (i = 0; i < name->len; i++) {
31 u_char ch = name->data[i];
32
33 if (ch == '&' || ch == '=' || ch == ' ' || ch < 0x20 || ch == 0x7f) {
34 return 0;
35 }
36 }
37 return 1;
38}

40/* One export-relative path segment [seg, seg+n): non-empty, not "." / "..". */
41static int
42space_prefix_segment_ok(const u_char *seg, size_t n)
43{
44 if (n == 0) {
45 return 0; /* "//" */
46 }
47 if (seg[0] != '.') {
48 return 1;
49 }
50 return !(n == 1 || (n == 2 && seg[1] == '.'));
51}
52
53/* An absolute, normalised prefix: leading '/', every segment well formed, no
54 * trailing '/'. The bare root "/" is refused on purpose: that is the
55 * export-wide default group, spelled brix_oss_cgroup + brix_oss_quota. */
56static int
57space_prefix_ok(const ngx_str_t *p)
58{
59 size_t i, seg;
60
61 if (p->len < 2 || p->data[0] != '/' || p->data[p->len - 1] == '/') {
62 return 0;
63 }
64 for (i = 1, seg = 1; i <= p->len; i++) {
65 if (i < p->len && p->data[i] != '/') {
66 continue;
67 }
68 if (!space_prefix_segment_ok(p->data + seg, i - seg)) {
69 return 0;
70 }
71 seg = i + 1;
72 }
73 return 1;
74}

76static char *
77space_parse_quota(ngx_conf_t *cf, const ngx_str_t *arg, off_t *quota)
78{
79 ngx_str_t v;
80
81 if (arg->len < 6 || ngx_strncmp(arg->data, "quota=", 6) != 0) {
82 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
83 "brix_oss_space: \"%V\": expected quota=<size> or quota=-1", arg);
84 return NGX_CONF_ERROR;
85 }
86 v.data = arg->data + 6;
87 v.len = arg->len - 6;
88 if (v.len == 2 && v.data[0] == '-' && v.data[1] == '1') {
89 *quota = -1;
90 return NGX_CONF_OK;
91 }
92 *quota = ngx_parse_offset(&v);
93 if (*quota == (off_t) NGX_ERROR || *quota < 0) {
94 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
95 "brix_oss_space: quota \"%V\" is not a non-negative size or -1", &v);
96 return NGX_CONF_ERROR;
97 }
98 return NGX_CONF_OK;
99}

101static char *
102space_check_unique(ngx_conf_t *cf, const ngx_array_t *spaces,
103 const ngx_str_t *name, const ngx_str_t *prefix)
104{
105 const brix_oss_space_t *dup;
106
107 if (brix_oss_space_by_name(spaces, name->data, name->len) != NULL) {
108 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
109 "brix_oss_space: group \"%V\" is declared twice", name);
110 return NGX_CONF_ERROR;
111 }
112 dup = brix_oss_space_for_path(spaces, (const char *) prefix->data,
113 prefix->len);
114 if (dup != NULL && dup->prefix.len == prefix->len) {
115 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
116 "brix_oss_space: prefix \"%V\" already belongs to group \"%V\"",
117 prefix, &dup->name);
118 return NGX_CONF_ERROR;
119 }
120 return NGX_CONF_OK;
121}

123char *
124brix_conf_set_oss_space(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
125{
126 ngx_stream_brix_srv_conf_t *xcf = conf;
127 ngx_str_t *value = cf->args->elts;
128 brix_oss_space_t *g;
129 off_t quota = -1;
130
131 (void) cmd;
132
133 if (!brix_oss_space_name_ok(&value[1])) {
134 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
135 "brix_oss_space: group name \"%V\" is empty or carries a byte that "
136 "would break the oss.* CGI report grammar (no & = space or "
137 "control chars)", &value[1]);
138 return NGX_CONF_ERROR;
139 }
140 if (!space_prefix_ok(&value[2])) {
141 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
142 "brix_oss_space: prefix \"%V\" must be an absolute export-relative "
143 "path with no empty or dot segments and no trailing slash (the "
144 "export root itself is the default group: brix_oss_cgroup / "
145 "brix_oss_quota)", &value[2]);
146 return NGX_CONF_ERROR;
147 }
148 if (cf->args->nelts == 4
149 && space_parse_quota(cf, &value[3], &quota) != NGX_CONF_OK)
150 {
151 return NGX_CONF_ERROR;
152 }
153 if (xcf->oss_spaces == NULL) {
154 xcf->oss_spaces = ngx_array_create(cf->pool, 4,
155 sizeof(brix_oss_space_t));
156 if (xcf->oss_spaces == NULL) {
157 return NGX_CONF_ERROR;
158 }
159 }
160 if (space_check_unique(cf, xcf->oss_spaces, &value[1], &value[2])
161 != NGX_CONF_OK)
162 {
163 return NGX_CONF_ERROR;
164 }
165 g = ngx_array_push(xcf->oss_spaces);
166 if (g == NULL) {
167 return NGX_CONF_ERROR;
168 }
169 ngx_memzero(g, sizeof(*g));
170 g->name = value[1];
171 g->prefix = value[2];
172 g->quota = quota;
173 return NGX_CONF_OK;
174}

176brix_oss_space_t *
177brix_oss_space_for_path(const ngx_array_t *spaces, const char *logical,
178 size_t len)
179{
180 brix_oss_space_t *g, *best = NULL;
181 ngx_uint_t i;
182
183 if (spaces == NULL || logical == NULL) {
184 return NULL;
185 }
186 g = spaces->elts;
187 for (i = 0; i < spaces->nelts; i++) {
188 size_t pl = g[i].prefix.len;
189
190 if (len < pl || ngx_memcmp(logical, g[i].prefix.data, pl) != 0) {
191 continue;
192 }
193 if (len > pl && logical[pl] != '/') {
194 continue; /* "/atlasdata" is not "/atlas" */
195 }
196 if (best == NULL || pl > best->prefix.len) {
197 best = &g[i];
198 }
199 }
200 return best;
201}
