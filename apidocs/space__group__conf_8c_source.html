1/*
2 * space_group_conf.c — parser + lookups for `brix_oss_space` (phase-115 W3.3).
3 *
4 * WHAT: validates one `brix_oss_space <group> <prefix> [quota=...]` line into
5 * the server's brix_oss_space_t table and answers the two lookups the
6 * write gate, kXR_Qspace and kXR_open use (by path, by name).
7 *
8 * WHY: a group name reaches the wire verbatim and a prefix is compared to
9 * confined logical paths, so both are pinned at parse time: a name that
10 * could smuggle a second oss.* key, a prefix with dot segments (it could
11 * never match a canonical path, so the quota would silently never
12 * apply) or a duplicate name/prefix fails `nginx -t` instead of
13 * mis-accounting at runtime.
14 *
15 * HOW: pure functions over ngx_str_t; the setter appends to an ngx_array_t
16 * created lazily in the config pool. Lookups are linear (tables are a
17 * handful of VOs); longest prefix wins so nested groups nest naturally.
18 */

23int
24brix_oss_space_name_ok(const ngx_str_t *name)
25{
26 size_t i;
27
28 if (name->len == 0) {
29 return 0;
30 }
31 for (i = 0; i < name->len; i++) {
32 u_char ch = name->data[i];
33
34 if (ch == '&' || ch == '=' || ch == ' ' || ch < 0x20 || ch == 0x7f) {
35 return 0;
36 }
37 }
38 return 1;
39}

41/* One export-relative path segment [seg, seg+n): non-empty, not "." / "..". */
42static int
43space_prefix_segment_ok(const u_char *seg, size_t n)
44{
45 if (n == 0) {
46 return 0; /* "//" */
47 }
48 if (seg[0] != '.') {
49 return 1;
50 }
51 return !(n == 1 || (n == 2 && seg[1] == '.'));
52}
53
54/* An absolute, normalised prefix: leading '/', every segment well formed, no
55 * trailing '/'. The bare root "/" is refused on purpose: that is the
56 * export-wide default group, spelled brix_oss_cgroup + brix_oss_quota. */
57static int
58space_prefix_ok(const ngx_str_t *p)
59{
60 size_t i, seg;
61
62 if (p->len < 2 || p->data[0] != '/' || p->data[p->len - 1] == '/') {
63 return 0;
64 }
65 for (i = 1, seg = 1; i <= p->len; i++) {
66 if (i < p->len && p->data[i] != '/') {
67 continue;
68 }
69 if (!space_prefix_segment_ok(p->data + seg, i - seg)) {
70 return 0;
71 }
72 seg = i + 1;
73 }
74 return 1;
75}

77static char *
78space_parse_quota(ngx_conf_t *cf, const ngx_str_t *arg, off_t *quota)
79{
80 ngx_str_t v;
81
82 if (arg->len < 6 || ngx_strncmp(arg->data, "quota=", 6) != 0) {
83 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
84 "brix_oss_space: \"%V\": expected quota=<size> or quota=-1", arg);
85 return NGX_CONF_ERROR;
86 }
87 v.data = arg->data + 6;
88 v.len = arg->len - 6;
89 if (v.len == 2 && v.data[0] == '-' && v.data[1] == '1') {
90 *quota = -1;
91 return NGX_CONF_OK;
92 }
93 *quota = ngx_parse_offset(&v);
94 if (*quota < 0) {
95 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
96 "brix_oss_space: quota \"%V\" is not a non-negative size or -1", &v);
97 return NGX_CONF_ERROR;
98 }
99 return NGX_CONF_OK;
100}

102static char *
103space_check_unique(ngx_conf_t *cf, const ngx_array_t *spaces,
104 const ngx_str_t *name, const ngx_str_t *prefix)
105{
106 const brix_oss_space_t *dup;
107
108 if (brix_oss_space_by_name(spaces, name->data, name->len) != NULL) {
109 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
110 "brix_oss_space: group \"%V\" is declared twice", name);
111 return NGX_CONF_ERROR;
112 }
113 dup = brix_oss_space_for_path(spaces, (const char *) prefix->data,
114 prefix->len);
115 if (dup != NULL && dup->prefix.len == prefix->len) {
116 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
117 "brix_oss_space: prefix \"%V\" already belongs to group \"%V\"",
118 prefix, &dup->name);
119 return NGX_CONF_ERROR;
120 }
121 return NGX_CONF_OK;
122}

124char *
125brix_conf_set_oss_space(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
126{
127 ngx_stream_brix_srv_conf_t *xcf = conf;
128 ngx_str_t *value = cf->args->elts;
129 brix_oss_space_t *g;
130 off_t quota = -1;
131
132 (void) cmd;
133
134 if (!brix_oss_space_name_ok(&value[1])) {
135 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
136 "brix_oss_space: group name \"%V\" is empty or carries a byte that "
137 "would break the oss.* CGI report grammar (no & = space or "
138 "control chars)", &value[1]);
139 return NGX_CONF_ERROR;
140 }
141 if (!space_prefix_ok(&value[2])) {
142 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
143 "brix_oss_space: prefix \"%V\" must be an absolute export-relative "
144 "path with no empty or dot segments and no trailing slash (the "
145 "export root itself is the default group: brix_oss_cgroup / "
146 "brix_oss_quota)", &value[2]);
147 return NGX_CONF_ERROR;
148 }
149 if (cf->args->nelts == 4
150 && space_parse_quota(cf, &value[3], &quota) != NGX_CONF_OK)
151 {
152 return NGX_CONF_ERROR;
153 }
154 if (xcf->oss_spaces == NULL) {
155 xcf->oss_spaces = ngx_array_create(cf->pool, 4,
156 sizeof(brix_oss_space_t));
157 if (xcf->oss_spaces == NULL) {
158 return NGX_CONF_ERROR;
159 }
160 }
161 if (space_check_unique(cf, xcf->oss_spaces, &value[1], &value[2])
162 != NGX_CONF_OK)
163 {
164 return NGX_CONF_ERROR;
165 }
166 g = ngx_array_push(xcf->oss_spaces);
167 if (g == NULL) {
168 return NGX_CONF_ERROR;
169 }
170 ngx_memzero(g, sizeof(*g));
171 g->name = value[1];
172 g->prefix = value[2];
173 g->quota = quota;
174 return NGX_CONF_OK;
175}

177brix_oss_space_t *
178brix_oss_space_for_path(const ngx_array_t *spaces, const char *logical,
179 size_t len)
180{
181 brix_oss_space_t *g, *best = NULL;
182 ngx_uint_t i;
183
184 if (spaces == NULL || logical == NULL) {
185 return NULL;
186 }
187 g = spaces->elts;
188 for (i = 0; i < spaces->nelts; i++) {
189 size_t pl = g[i].prefix.len;
190
191 if (len < pl || ngx_memcmp(logical, g[i].prefix.data, pl) != 0) {
192 continue;
193 }
194 if (len > pl && logical[pl] != '/') {
195 continue; /* "/atlasdata" is not "/atlas" */
196 }
197 if (best == NULL || pl > best->prefix.len) {
198 best = &g[i];
199 }
200 }
201 return best;
202}
