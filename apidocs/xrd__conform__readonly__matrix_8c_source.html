16/* WHAT: Attempt one HTTP-plane mutation and inspect its native denial spelling.
17 * WHY: A shared status alone cannot prove the S3 and OCI response vocabularies.
18 * HOW: PUT one byte through the normal bounded HTTP client and compare status
19 * plus the plane-specific response marker. */
20static int
21http_readonly_matches(xrd_conform_execution *execution, const char *target,
22 const char *plane)
23{
24 brix_weburl url;
25 brix_http_resp response;
26 brix_status status;
27 int rc;
28 int matched;
29
30 if (brix_weburl_parse(target, &url) != 0) return 0;
31 memset(&response, 0, sizeof(response));
32 brix_status_clear(&status);
33 rc = xrd_conform_http_request_body(execution, &url, "PUT", NULL, NULL,
34 "R", 1, 4096, &response, &status);
35 if (rc != 0) return 0;
36 matched = response.status == 403;
37 if (strcmp(plane, "s3") == 0)
38 matched = matched && response.body != NULL
39 && strstr(response.body, "<Code>AccessDenied</Code>") != NULL;
40 else if (strcmp(plane, "oci") == 0)
41 matched = matched && response.body != NULL
42 && strstr(response.body, "DENIED") != NULL;
43 brix_http_resp_free(&response);
44 return matched;
45}

47/* WHAT: Attempt an FTP STOR without opening a data channel.
48 * WHY: A read-only GridFTP endpoint must reject at the control policy gate.
49 * HOW: Use the production FTP parser/session, issue STOR, and require the exact
50 * 550 read-only phrase rather than accepting an unrelated missing error. */
51static int
52gridftp_readonly_matches(xrd_conform_execution *execution, const char *target)
53{
54 brix_ftpurl url;
55 brix_ftp_sess *session;
56 brix_status status;
57 char reason[64];
58 int matched = 0;
59
60 if (brix_ftpurl_parse(target, &url) != 0
61 || !xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return 0;
62 session = calloc(1, sizeof(*session));
63 if (session == NULL) return 0;
64 brix_status_clear(&status);
65 if (brix_ftp_session_open(session, &url, &execution->run->opts,
66 &status) == 0
67 && brix_ftp_cmd(session, &status, "STOR %s", url.path) == 0)
68 matched = session->code == 550
69 && strstr(session->text, "Permission denied (read-only)") != NULL;
70 brix_ftp_close(session);
71 free(session);
72 return matched;
73}
