3/*
4 * access_log.c — structured JSON access-log line emitter for VFS operations.
5 *
6 * WHAT: Emits one machine-parsable JSON access record per completed VFS I/O
7 * operation via brix_access_log_emit(). Each line carries timestamp,
8 * protocol, remote peer, op name, request path, bytes_served, offset,
9 * backend_time_us, status, cache_status, auth_method, and the
10 * authenticated subject (key "sub"). Every fact appears exactly once:
11 * phase 112 removed the pre-110 compatibility spellings ("bytes",
12 * "latency_us", "from_cache", "subject") that duplicated four of them.
13 * WHY: Prometheus counters are aggregate and low-cardinality (no paths or DNs
14 * as labels — INVARIANT #8), but operators still need per-request audit
15 * detail. A JSON log line carries the high-cardinality fields (path,
16 * subject) that must never become metric labels, while reusing the same
17 * unified vocabulary (brix_metric_proto_name / _op_name / _err_name /
18 * _auth_method_name) so logs and metrics line up.
19 * HOW: The line is written with ngx_log_error at NGX_LOG_INFO and prefixed
20 * "brix_access_json: " so a log pipeline can grep and strip the prefix.
21 * Free-text fields are run through brix_access_json_escape() into fixed
22 * stack buffers before being interpolated, so untrusted wire bytes can
23 * never break out of the JSON string. Subject is taken from the identity
24 * (subject, falling back to DN); path/offset/cache come from the
25 * brix_vfs_io_result_t when present.
26 */
27
28/*
29 * brix_access_json_escape — copy src into dst as a JSON-string-safe sequence.
30 *
31 * Escapes '"' and '\\' with a backslash and renders any byte outside printable
32 * ASCII (< 0x20 or >= 0x7f) as a \u00NN escape, so control bytes and non-ASCII
33 * from the wire cannot terminate the surrounding JSON string. Always writes a
34 * NUL terminator (unless dstsz == 0) and never exceeds dstsz, truncating cleanly
35 * if an escape sequence would not fit.
36 */
37static void
38brix_access_json_escape(const char *src, char *dst, size_t dstsz)
39{
40 static const char hex[] = "0123456789abcdef";
41 size_t used;
42
43 if (dstsz == 0) {
44 return;
45 }
46
47 used = 0;
48 if (src == NULL) {
49 src = "";
50 }
51
52 while (*src != '\0' && used + 1 < dstsz) {
53 unsigned char ch = (unsigned char) *src++;
54
55 if (ch == '"' || ch == '\\') {
56 if (used + 2 >= dstsz) {
57 break;
58 }
59 dst[used++] = '\\';
60 dst[used++] = (char) ch;
61 continue;
62 }
63
64 if (ch >= 0x20 && ch < 0x7f) {
65 dst[used++] = (char) ch;
66 continue;
67 }
68
69 if (used + 6 >= dstsz) {
70 break;
71 }
72 dst[used++] = '\\';
73 dst[used++] = 'u';
74 dst[used++] = '0';
75 dst[used++] = '0';
76 dst[used++] = hex[ch >> 4];
77 dst[used++] = hex[ch & 0x0f];
78 }
79
80 dst[used] = '\0';
81}
