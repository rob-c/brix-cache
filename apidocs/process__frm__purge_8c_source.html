1/*
2 * process_frm_purge.c — phase-115 W3.2: arm + drive the tape-buffer purge
3 * timer for a server whose export chain carries a tape:// tier.
4 *
5 * WHAT: brix_init_server_frm_purge_timer() (called from the per-server init
6 * ladder in process_server_init.c) resolves the export's driver chain,
7 * finds the frm tier, and arms a worker-0 timer that runs
8 * brix_sd_frm_purge() every brix_frm_purge_interval with the policy the
9 * brix_frm_purge_watermark / brix_frm_purge_max_bytes directives set.
10 *
11 * WHY: Those two directives were accepted-only since the FRM dissolution
12 * (§13b): parsed, merged, never read. The engine itself is pure libc in
13 * src/fs/backend/frm/sd_frm_purge.c; this TU is the only place that
14 * knows about nginx timers, the server conf and the stage registry.
15 *
16 * HOW: 1. Skip unless a local export root resolves, at least one arm is
17 * configured (0 < hi_ppm < 1e6, or max_bytes > 0), and this is
18 * worker 0 (one engine per host, not N racing walks).
19 * 2. Walk the chain head -> decorator sources for a CAP_NEARLINE tier
20 * whose driver is "frm"; WARN + return when there is none (the
21 * directives are set but cannot act).
22 * 3. pcalloc the timer, first tick = brix_cache_reap_delay(5000 ms).
23 * 4. Each tick: statvfs the online root (arm 1 input), run one pass
24 * with the stage registry as the pin oracle, re-arm unless exiting.
25 * 2.0 F4: the brix_frm_purge_policy table becomes the engine's rule
26 * array once at arm time (percent thresholds scaled by the
27 * group's brix_oss_space quota); rule_of() answers the rule of a
28 * buffer key by longest-prefix group, "*" for the rest.
29 */

59/* The tape tier of an export's chain: the first CAP_NEARLINE instance whose
60 * driver is the frm backend, walking head -> decorator sources. */
61static brix_sd_instance_t *
62brix_frm_purge_find_tier(brix_sd_instance_t *head)
63{
64 brix_sd_instance_t *inst;
65
66 for (inst = head; inst != NULL; inst = brix_vfs_decorator_source(inst)) {
67 if ((brix_sd_caps(inst) & BRIX_SD_CAP_NEARLINE)
68 && inst->driver != NULL
69 && ngx_strcmp(inst->driver->name, "frm") == 0)
70 {
71 return inst;
72 }
73 }
74 return NULL;
75}

77/* Pin oracle: a key is pinned while a live stage request names it. The
78 * registry keys requests by the confined ABSOLUTE lfn (<root_canon><key>). */
79static int
80brix_frm_purge_is_pinned(void *ud, const char *key)
81{
82 brix_frm_purge_timer_t *t = ud;
83 ngx_stream_brix_srv_conf_t *xcf = t->xcf;
84 brix_stage_registry_t *reg = brix_stage_registry_singleton();
85 char lfn[PATH_MAX];
86 char reqid[40];
87
88 if (reg == NULL) {
89 return 0;
90 }
91 if (snprintf(lfn, sizeof(lfn), "%s%s", xcf->common.root_canon, key)
92 >= (int) sizeof(lfn))
93 {
94 return 1; /* unrepresentable: keep it */
95 }
96 return brix_stage_request_find_by_path(reg, lfn, reqid, sizeof(reqid),
97 t->ev.log) == NGX_OK;
98}

100/* 2.0 F4 — the rule a buffer key falls under: its longest-prefix
101 * brix_oss_space group's own rule, else the "*" rule (which also covers
102 * ungrouped keys), else none. A nested group is its own space, so a key of a
103 * group without a rule never inherits the parent group's rule. */
104static int
105brix_frm_purge_rule_of(void *ud, const char *key)
106{
107 brix_frm_purge_timer_t *t = ud;
108 const brix_oss_space_t *sp = brix_oss_space_for_path(t->xcf->oss_spaces,
109 key, ngx_strlen(key));
110 size_t i;
111
112 if (sp == NULL) {
113 return t->default_rule;
114 }
115 for (i = 0; i < t->nrules; i++) {
116 if (sp->name.len == ngx_strlen(t->rules[i].name)
117 && ngx_strncmp(sp->name.data, t->rules[i].name, sp->name.len) == 0)
118 {
119 return (int) i;
120 }
121 }
122 return t->default_rule;
123}

125/* A size threshold is used as is; a percentage is scaled by the group's
126 * brix_oss_space quota (the merge refused a percentage without one). */
127static uint64_t
128brix_frm_purge_rule_bytes(const ngx_stream_brix_srv_conf_t *xcf,
129 const brix_frm_purge_policy_t *p, off_t bytes, ngx_uint_t ppm)
130{
131 const brix_oss_space_t *sp;
132
133 if (!p->percent) {
134 return (uint64_t) bytes;
135 }
136 sp = brix_oss_space_by_name(xcf->oss_spaces, p->group.data, p->group.len);
137 if (sp == NULL || sp->quota <= 0) {
138 return 0;
139 }
140 return (uint64_t) ((double) sp->quota * (double) ppm / 1000000.0);
141}

143static ngx_int_t
144brix_frm_purge_build_rules(ngx_cycle_t *cycle, brix_frm_purge_timer_t *t)
145{
146 const ngx_array_t *pols = t->xcf->frm.purge_policies;
147 const brix_frm_purge_policy_t *p;
148 brix_sd_frm_purge_rule_t *r;
149 ngx_uint_t i;
150
151 t->default_rule = -1;
152 t->polprog = (t->xcf->frm.purge_polprog.len > 0)
153 ? (const char *) t->xcf->frm.purge_polprog.data : NULL;
154 if (pols == NULL || pols->nelts == 0) {
155 return NGX_OK;
156 }
157 r = ngx_pcalloc(cycle->pool, pols->nelts * sizeof(*r));
158 if (r == NULL) {
159 return NGX_ERROR;
160 }
161 p = pols->elts;
162 for (i = 0; i < pols->nelts; i++) {
163 u_char *name = ngx_pnalloc(cycle->pool, p[i].group.len + 1);
164
165 if (name == NULL) {
166 return NGX_ERROR;
167 }
168 ngx_cpystrn(name, p[i].group.data, p[i].group.len + 1);
169 r[i].name = (const char *) name;
170 r[i].hi_bytes = brix_frm_purge_rule_bytes(t->xcf, &p[i], p[i].hi,
171 p[i].hi_ppm);
172 r[i].lo_bytes = brix_frm_purge_rule_bytes(t->xcf, &p[i], p[i].lo,
173 p[i].lo_ppm);
174 r[i].hold_s = p[i].hold_s;
175 r[i].polprog = p[i].polprog ? 1 : 0;
176 if (p[i].group.len == 1 && p[i].group.data[0] == '*') {
177 t->default_rule = (int) i;
178 }
179 }
180 t->rules = r;
181 t->nrules = pols->nelts;
182 return NGX_OK;
183}

185static int
186brix_frm_purge_armed(const brix_frm_conf_t *frm)
187{
188 return (frm->purge_hi_ppm > 0 && frm->purge_hi_ppm < 1000000)
189 || frm->purge_max_bytes > 0
190 || (frm->purge_policies != NULL && frm->purge_policies->nelts > 0);

193static void
194brix_frm_purge_tick(ngx_event_t *ev)
195{
196 brix_frm_purge_timer_t *t = ev->data;
197 ngx_stream_brix_srv_conf_t *xcf = t->xcf;
198 brix_sd_frm_purge_policy_t pol;
199 brix_sd_frm_purge_report_t rep;
200 brix_cache_fs_usage_t usage;
201 char online[PATH_MAX];
202 ngx_msec_t interval;
203
204 ngx_memzero(&pol, sizeof(pol));
205 pol.hi_ppm = xcf->frm.purge_hi_ppm;
206 pol.lo_ppm = xcf->frm.purge_lo_ppm;
207 pol.max_bytes = xcf->frm.purge_max_bytes;
208 pol.min_age_s = BRIX_FRM_PURGE_MIN_AGE_S;
209 pol.is_pinned = brix_frm_purge_is_pinned;
210 pol.ud = t;
211 pol.rules = t->rules; /* 2.0 F4 */
212 pol.nrules = t->nrules;
213 pol.rule_of = brix_frm_purge_rule_of;
214 pol.polprog = t->polprog;
215 pol.polprog_timeout_ms = xcf->frm.copy_timeout > 0
216 ? xcf->frm.copy_timeout
217 : BRIX_FRM_POLPROG_TIMEOUT_MS;
218 if (brix_sd_frm_online_root(t->frm, online, sizeof(online)) == NGX_OK
219 && brix_cache_fs_usage(online, &usage) == NGX_OK)
220 {
221 pol.fs_total = usage.total;
222 pol.fs_used = usage.used;
223 }
224
225 if (brix_sd_frm_purge(t->frm, &pol, &rep, ev->log) == NGX_DECLINED) {
226 ngx_log_error(NGX_LOG_INFO, ev->log, 0,
227 "brix: tape purge \"%s\" skipped: another pass holds the lock",
228 online);
229 }
230
231 if (!ngx_exiting) {
232 interval = xcf->frm.purge_interval_ms;
233 if (interval < BRIX_FRM_PURGE_MIN_INTERVAL_MS) {
234 interval = BRIX_FRM_PURGE_MIN_INTERVAL_MS;
235 }
236 ngx_add_timer(ev, interval);
237 }
238}

240ngx_int_t
241brix_init_server_frm_purge_timer(ngx_cycle_t *cycle,
242 ngx_stream_brix_srv_conf_t *xcf)
243{
244 brix_frm_purge_timer_t *t;
245 brix_sd_instance_t *head, *frm;
246
247 if (xcf->common.root_canon[0] == '\0' || !brix_frm_purge_armed(&xcf->frm)
248 || ngx_worker != 0)
249 {
250 return NGX_OK;
251 }
252 head = brix_vfs_backend_resolve(xcf->common.root_canon, cycle->log);
253 frm = brix_frm_purge_find_tier(head);
254 if (frm == NULL) {
255 ngx_log_error(NGX_LOG_WARN, cycle->log, 0,
256 "brix: brix_frm_purge_* configured but export \"%s\" has no "
257 "tape:// tier; purge engine not armed",
258 xcf->common.root_canon);
259 return NGX_OK;
260 }
261
262 t = ngx_pcalloc(cycle->pool, sizeof(*t));
263 if (t == NULL) {
264 return NGX_ERROR;
265 }
266 t->xcf = xcf;
267 t->frm = frm;
268 if (brix_frm_purge_build_rules(cycle, t) != NGX_OK) {
269 return NGX_ERROR;
270 }
271 t->ev.handler = brix_frm_purge_tick;
272 t->ev.data = t;
273 t->ev.log = cycle->log;
274 t->ev.cancelable = 1; /* don't delay graceful shutdown */
275 ngx_add_timer(&t->ev, brix_cache_reap_delay(BRIX_FRM_PURGE_FIRST_MS));
276
277 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
278 "brix: tape purge engine armed for export \"%s\" (hi=%ui lo=%ui ppm, "
279 "max_bytes=%O, interval=%M ms)",
280 xcf->common.root_canon, xcf->frm.purge_hi_ppm, xcf->frm.purge_lo_ppm,
281 xcf->frm.purge_max_bytes, xcf->frm.purge_interval_ms);
282 if (t->nrules > 0) {
283 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
284 "brix: tape purge policy for export \"%s\": %uz rule(s), "
285 "program \"%s\"", xcf->common.root_canon, t->nrules,
286 t->polprog ? t->polprog : "(none)");
287 }
288 return NGX_OK;
289}
