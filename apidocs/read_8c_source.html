24/*
25 * brix_fsoverload_backoff — the shared memory-budget-overload response (§1.10,
26 * xrootd.fsoverload). A read/readv the process-wide memory budget cannot admit
27 * gets one of two backoffs: a kXR_redirect to brix_fsoverload_redirect's host
28 * (offload the read to a sibling server) when that is configured, else a
29 * kXR_wait(brix_fsoverload_stall) telling the client to retry here later. The
30 * redirect host is NUL-terminated at config time (brix_pstrdupz), so it is passed
31 * straight to brix_send_redirect; kXR_wait stays clamped by brix_max_delay at its
32 * own emission choke point.
33 *
34 * §1.3 kXR_readrdok: a redirect issued in response to a kXR_read/readv (NOT at
35 * open time) can only be followed by a client that advertised the readrdok login
36 * ability. A client that did not — an older client, or one that cleared the bit —
37 * would mishandle a mid-read redirect. For such a client we degrade to the
38 * kXR_wait backoff (retry-here) even when a redirect host is configured, which is
39 * always safe: the read is simply deferred until this node's budget frees.
40 */
41ngx_int_t
42brix_fsoverload_backoff(brix_ctx_t *ctx, ngx_connection_t *c,
43 ngx_stream_brix_srv_conf_t *rconf)
44{
45 if (rconf->fsoverload_redirect_host.len > 0
46 && (ctx->login.ability & 0x04 /* kXR_readrdok */))
47 {
48 return brix_send_redirect(ctx, c,
49 (const char *) rconf->fsoverload_redirect_host.data,
50 (uint16_t) rconf->fsoverload_redirect_port);
51 }
52 return brix_send_wait(ctx, c, (uint32_t) rconf->fsoverload_stall);
53}

55/* Codec-vs-protocol drift guard: the wire codec (shared libxrdproto, deliberately
56 * XProtocol-free) hard-codes the request body as XRDW_BODY_LEN bytes. This is the
57 * one translation unit that sees both that constant and the real XProtocol
58 * ClientRequestHdr, so it ties them together at compile time — if XRootD ever
59 * resized the body region, every xrdw_*_unpack() call here would read the wrong
60 * offsets, and this assert fails the build instead of corrupting requests. */
61_Static_assert(sizeof(((ClientRequestHdr *) 0)->body) == XRDW_BODY_LEN,
62 "wire codec body length must match XProtocol ClientRequestHdr.body");

64/*
65 * read_validate_req — decode the wire request and run the early-return checks.
66 *
67 * WHAT: unpacks the kXR_read body into *io, validates the file handle, serves
68 * the trivial rlen==0 case, clamps oversized requests and rejects negative
69 * offsets.
70 * WHY: every serve strategy needs the same validated (idx, fd, offset, rlen)
71 * tuple; hoisting the checks keeps the dispatcher a pure strategy selector.
72 * HOW: returns 1 when the request is valid and *io is filled (databuf NULL);
73 * returns 0 when the request was fully handled here (ok/error response already
74 * queued) with *rc set to the value the opcode handler must return.
75 */
76static ngx_flag_t
77read_validate_req(brix_ctx_t *ctx, ngx_connection_t *c,
78 brix_read_io_t *io, ngx_int_t *rc)
79{
80 xrdw_read_req_t req;
81
82 /*
83 * The shared codec decodes the big-endian wire body into host order; the file
84 * handle is a 4-byte blob but only byte 0 indexes our slot table
85 * (BRIX_MAX_FILES <= 256); the (unsigned char) cast prevents sign-extension
86 * of a high-bit handle byte into a negative idx.
87 */
88 xrdw_read_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
89 io->idx = (int) (unsigned char) req.fhandle[0];
90 io->offset = req.offset;
91 io->rlen = (size_t) (uint32_t) req.rlen;
92 io->databuf = NULL;
93 io->pathid = 0;
94
95 /*
96 * §1.1: a kXR_read's optional read_args ride the payload — pathid at byte 0
97 * when dlen >= 1 (the response-offload channel selector). Mirror pgread/§1.2:
98 * a NONZERO pathid MUST name one of this session's live kXR_bind paths, else
99 * kXR_ArgInvalid "invalid path ID", exactly as stock refuses an unbound path
100 * ID (read previously ignored the read_args entirely — an inconsistency vs
101 * pgread). Routing the response over that bound path (offloading) is a later
102 * slice; the validated pathid is captured in io for it.
103 */
104 if (ctx->recv.cur_dlen >= 1 && ctx->recv.payload != NULL) {
105 io->pathid = (unsigned) ((u_char *) ctx->recv.payload)[0];
106 if (io->pathid != 0
107 && !brix_session_pathid_bound(ctx->is_bound ? ctx->bound_sessid
108 : ctx->login.sessid,
109 io->pathid))
110 {
111 BRIX_OP_ERR(ctx, BRIX_OP_READ);
112 *rc = brix_send_error(ctx, c, kXR_ArgInvalid, "invalid path ID");
113 return 0;
114 }
115 }
116
117 if (!brix_validate_read_handle(ctx, c, io->idx, "READ",
118 BRIX_OP_READ, rc)) {
119 return 0;
120 }
121
122 if (io->rlen == 0) {
123 BRIX_OP_OK(ctx, BRIX_OP_READ);
124 *rc = brix_send_ok(ctx, c, NULL, 0);
125 return 0;
126 }
127
128 if (io->rlen > BRIX_READ_REQUEST_MAX) {
129 io->rlen = BRIX_READ_REQUEST_MAX;
130 }
131
132 io->fd = ctx->files[io->idx].fd;
133
134 if (io->offset < 0) {
135 brix_log_access(ctx, c, "READ", ctx->files[io->idx].path, "-",
136 0, kXR_IOError, "negative read offset", 0);
137 BRIX_OP_ERR(ctx, BRIX_OP_READ);
138 *rc = brix_send_error(ctx, c, kXR_IOError, "negative read offset");
139 return 0;
140 }
141
142 return 1;
143}

145/*
146 * brix_read_try_offload — §1.1 response offloading (do_Offload/do_OffloadIO
147 * parity): when a validated read carries a NONZERO read_args pathid AND that
148 * bound secondary data channel lives on THIS worker AND is quiescent, serve the
149 * read's response over the SECONDARY's socket instead of the primary control
150 * stream. Returns 1 when the response was routed to the secondary (*rc = the
151 * value brix_handle_read must return), 0 when the read must fall through to the
152 * normal primary-stream serve strategies (byte-identical to before).
153 *
154 * WHAT (eligible case): borrow a buffer from the SECONDARY's read pool, fill its
155 * data region [8 .. 8+n) with the file bytes via one synchronous VFS read, stamp
156 * an 8-byte kXR_ok header carrying the PRIMARY request's streamid at [0 .. 8),
157 * and queue the contiguous [hdr|data] frame on the secondary's out-ring.
158 * WHY it is safe: buffers are acquired AND released on the secondary's ctx, so
159 * the ordinary per-connection drain/release discipline applies with no cross-
160 * connection lifetime tangle — the frame simply rides the machinery the secondary
161 * already runs for its own bound reads. The client correlates the response by
162 * streamid regardless of which stream carries it.
163 * HOW it is gated (this first routing slice deliberately handles only the safe,
164 * common case): same-worker secondary that is idle (no queued response, no async
165 * ack, no in-flight AIO, live fd), and a single-frame (<= one window) read.
166 * Anything else — large/windowed, busy secondary, cross-worker pathid, pool
167 * exhaustion — returns 0 and is served the old way.
168 */
169static ngx_flag_t
170brix_read_try_offload(brix_ctx_t *ctx, ngx_connection_t *c,
171 ngx_stream_brix_srv_conf_t *rconf, brix_read_io_t *io, ngx_int_t *rc)
172{
173 ngx_connection_t *sec_c;
174 brix_ctx_t *sec_ctx;
175 size_t total;
176 u_char *buf;
177 brix_vfs_job_t job;
178 ssize_t nread;
179
180 sec_ctx = brix_read_offload_secondary(ctx, c, io->pathid, &sec_c);
181 if (sec_ctx == NULL) {
182 return 0;
183 }
184
185 /* Single-frame only: clamp to what a read-only handle actually holds; a read
186 * larger than one streaming window belongs on the windowed primary path. */
187 total = read_clamped_total(ctx, io);
188 if (total > (size_t) BRIX_READ_WINDOW) {
189 return 0;
190 }
191
192 /* Borrow the frame buffer from the SECONDARY's pool (acquire+release both on
193 * its ctx) — header (8) + up to `total` data bytes, laid out contiguously. */
194 buf = brix_acquire_read_buffer(sec_ctx, sec_c, XRD_RESPONSE_HDR_LEN + total);
195 if (buf == NULL) {
196 return 0; /* secondary pool exhausted — fall back rather than fail */
197 }
198
199 /* One synchronous fill straight into the frame's data region [8 .. 8+total). */
200 brix_vfs_job_read_init(&job, io->fd, (off_t) io->offset, total,
201 buf + XRD_RESPONSE_HDR_LEN, total, 0);
202 job.csi = ctx->files[io->idx].csi; /* phase-59 W2: verify on read */
203 brix_vfs_job_set_obj(&job, &ctx->files[io->idx].sd_obj);
204 brix_vfs_io_execute(&job);
205 nread = job.nio;
206
207 if (nread < 0) {
208 /* Read/CSI failure: nothing has touched the secondary wire yet, so the
209 * error rides the PRIMARY control stream exactly like the normal path. */
210 brix_release_read_buffer(sec_ctx, sec_c, buf);
211 if (job.io_errno != 0) {
212 errno = job.io_errno;
213 }
214 brix_read_io_failure_log(c->log, "offload", io->fd,
215 (off_t) io->offset, total, errno);
216 BRIX_OP_ERR(ctx, BRIX_OP_READ);
217 *rc = brix_send_error(ctx, c, kXR_IOError, strerror(errno));
218 return 1;
219 }
220
221 /* Stamp the kXR_ok header with the PRIMARY request's streamid. */
222 brix_build_resp_hdr(ctx->recv.cur_streamid, kXR_ok, (uint32_t) nread,
223 (ServerResponseHdr *) buf);
224
225 /* Byte accounting + access log stay on the PRIMARY ctx (the request's owner). */
226 ctx->files[io->idx].bytes_read += (size_t) nread;
227 ctx->totals.bytes += (size_t) nread;
228 if (rconf->access_log_fd != NGX_INVALID_FILE) {
229 char read_detail[64];
230 snprintf(read_detail, sizeof(read_detail), "%lld+%zu",
231 (long long) io->offset, io->rlen);
232 brix_log_access(ctx, c, "READ", ctx->files[io->idx].path,
233 read_detail, 1, 0, NULL, (size_t) nread);
234 }
235 BRIX_OP_OK(ctx, BRIX_OP_READ);
236
237 /* Queue the contiguous [hdr|data] frame on the SECONDARY's out-ring; the
238 * secondary owns `buf` and releases it when the slot drains. */
239 *rc = brix_queue_response_base(sec_ctx, sec_c, buf,
240 XRD_RESPONSE_HDR_LEN + (size_t) nread, buf);
241 if (*rc == NGX_OK) {
242 brix_metric_offload(BRIX_PROTO_ROOT); /* §1.1 observability */
243 }
244
245 /*
246 * Release here on error OR on a full inline send (secondary ring stayed
247 * empty) — only a parked-and-draining response keeps the buffer, whose slot
248 * drain then releases it. Mirrors read_finish_buffered's post-queue release;
249 * brix_release_read_buffer is idempotent so the error/parked overlap is safe.
250 */
251 if (*rc != NGX_OK || sec_ctx->out.count == 0) {
252 brix_release_read_buffer(sec_ctx, sec_c, buf);
253 }
254 return 1;
255}

257/*
258 * brix_handle_read — kXR_read dispatcher: validate, then pick a serve strategy.
259 *
260 * WHAT: routes a validated read to one of the serve paths — SSI/ZIP/codec
261 * early dispatch, zero-copy sendfile, windowed streaming, or the buffered
262 * memory path (warm-probe → AIO → synchronous fallback).
263 * WHY: each strategy has its own invariants (TLS => memory-backed buffers,
264 * cleartext/kTLS => sendfile; heap bounded by the streaming window); keeping
265 * the handler a flat early-return ladder makes the strategy choice auditable.
266 * HOW: read_validate_req() supplies the decoded request; every branch
267 * tail-calls its serve helper.
268 */
269ngx_int_t
270brix_handle_read(brix_ctx_t *ctx, ngx_connection_t *c)
271{
272 brix_read_io_t io;
273 ngx_stream_brix_srv_conf_t *rconf;
274 ngx_int_t rc;
275
276 if (!read_validate_req(ctx, c, &io, &rc)) {
277 return rc;
278 }
279
280 rconf = ngx_stream_get_module_srv_conf(
281 (ngx_stream_session_t *) c->data, ngx_stream_brix_module);
282
283 /* §7 XrdSsi: an SSI handle has no backing file — the first read dispatches the
284 * accumulated request to the service and serves the response. Early dispatch
285 * off the normal fd read path, like zip/slice below. */
286 if (ctx->files[io.idx].ssi != NULL) {
287 BRIX_OP_OK(ctx, BRIX_OP_READ);
288 return brix_ssi_read(ctx, c, io.idx, (uint64_t) io.offset,
289 (uint32_t) io.rlen);
290 }
291
292 /* Phase-57 W2: ZIP member handles translate the read into the archive's
293 * byte range (stored = offset add; deflate = stream inflate) — an early
294 * dispatch off the normal fd read path. */
295 if (ctx->files[io.idx].zip_mode) {
296 return brix_zip_read(ctx, c, io.idx, io.offset, io.rlen);
297 }
298
299 /*
300 * Phase-42 W4: inline read compression (opt-in, off by default). Routed to
301 * its own isolated synchronous handler so EVERYTHING below — the sendfile
302 * fast path, windowed streaming and AIO pipeline — stays byte-identical for
303 * the default (read_codec == 0 / BRIX_CODEC_IDENTITY) case. pgread/readv
304 * have their own handlers and never reach here, so their plaintext + CRC32c
305 * invariant is preserved.
306 */
307 if (ctx->files[io.idx].read_codec != 0) {
308 return brix_read_compressed(ctx, c, rconf, io.idx, (off_t) io.offset,
309 io.rlen);
310 }
311
312 /*
313 * §1.1 response offloading: a read tagged with a live, same-worker bound
314 * pathid is served over that secondary data channel; every ineligible read
315 * (pathid 0, cross-worker, busy secondary, large) returns 0 here and falls
316 * through to the normal primary-stream strategies below, unchanged.
317 */
318 {
319 ngx_int_t orc;
320 if (brix_read_try_offload(ctx, c, rconf, &io, &orc)) {
321 return orc;
322 }
323 }
324
325 /*
326 * Zero-copy sendfile fast path (gate in read_sendfile_serve_fd — the
327 * TLS-vs-cleartext INVARIANT lives there, and the BACKEND elects the fd
328 * for driver-backed handles). Anything that fails the gate drops to the
329 * memory/window path below.
330 */
331 {
332 ngx_fd_t sfd = read_sendfile_serve_fd(ctx, c, &io);
333
334 if (sfd != NGX_INVALID_FILE) {
335 return brix_read_serve_sendfile(ctx, c, rconf, &io, sfd);
336 }
337 }
338
339 /*
340 * Phase 31 W2.1: bound resident heap for large memory-backed reads. This
341 * is the memory path (TLS / non-regular file) — unlike the cleartext
342 * sendfile branch above it must buffer data in heap. Clamp the request to
343 * what the file actually holds (read-only handles have a cached size); if
344 * that exceeds one streaming window, serve the read as a sequence of
345 * window-sized kXR_oksofar chunks ending in kXR_ok, holding only ~one window
346 * in read_scratch at a time instead of the whole request. Writable handles
347 * (size unknown) use rlen and let a short read at EOF terminate early.
348 */
349 {
350 size_t total = read_clamped_total(ctx, &io);
351
352 if (total > (size_t) BRIX_READ_WINDOW) {
353 return read_serve_windowed(ctx, c, rconf, &io, total);
354 }
355 }
356
357 /*
358 * Small memory read (<= one window): single-shot. Admit the full rlen and
359 * buffer it in read_scratch — bounded by the window, so no streaming needed.
360 */
361 return read_serve_buffered(ctx, c, rconf, &io);
362}
