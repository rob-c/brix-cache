1/*
2 * cache_urlcgi_conf.c — the brix_cache_urlcgi directive setter (2.0 F5,
3 * upstream pfc.urlcgi).
4 *
5 * WHAT: brix_cache_urlcgi [blocksize {ignore|<min> <max>}]
6 * [prefetch {ignore|<min> <max>}]
7 * arms (or explicitly disarms) the two per-open client hints a root:// open
8 * may carry as CGI: pfc.blocksize=<bytes> and pfc.prefetch=<blocks>. A
9 * clause not named on the line is ignored, the upstream default.
10 *
11 * WHY: Upstream's pfc.urlcgi lets a client tune the slice granule and the
12 * prefetch runway per open, with the operator bounding what may be asked
13 * for. Same grammar; the bounds follow this cache's own rules — blocksize
14 * bounds are positive multiples of 1m like brix_cache_slice_size, prefetch
15 * bounds are block counts with max >= 1 (a real clamp never has max 0,
16 * which is the "ignored" encoding of brix_cache_urlcgi_conf_t).
17 *
18 * HOW: A word loop over cf->args: each clause keyword takes either `ignore`
19 * or exactly two bounds; a keyword repeated or unknown, a missing bound, or
20 * min > max is a configuration error, reported the nginx way (a string
21 * the core prefixes with the directive name and location).
22 */

38/* `blocksize <min> <max>`: sizes, positive multiples of the slice granule. */
39static char *
40urlcgi_parse_size_bounds(const ngx_str_t *lo, const ngx_str_t *hi,
41 brix_cache_urlcgi_conf_t *u)
42{
43 ssize_t a = ngx_parse_size((ngx_str_t *) lo);
44 ssize_t b = ngx_parse_size((ngx_str_t *) hi);
45
46 if (a <= 0 || b <= 0) {
47 return "blocksize: bad size";
48 }
49 if (a % BRIX_CACHE_SLICE_GRANULE != 0 || b % BRIX_CACHE_SLICE_GRANULE != 0) {
50 return "blocksize: bounds must be positive multiples of 1m";
51 }
52 if (a > b) {
53 return "blocksize: min exceeds max";
54 }
55 u->bs_min = (size_t) a;
56 u->bs_max = (size_t) b;
57 return NGX_CONF_OK;
58}

60/* `prefetch <min> <max>`: block counts, max >= 1 (min 0 lets a client switch
61 * speculation off for its own handle). */
62static char *
63urlcgi_parse_count_bounds(const ngx_str_t *lo, const ngx_str_t *hi,
64 brix_cache_urlcgi_conf_t *u)
65{
66 ngx_int_t a = ngx_atoi(lo->data, lo->len);
67 ngx_int_t b = ngx_atoi(hi->data, hi->len);
68
69 if (a == NGX_ERROR || b == NGX_ERROR) {
70 return "prefetch: bad block count";
71 }
72 if (b < 1) {
73 return "prefetch: max must be at least 1 block";
74 }
75 if (a > b) {
76 return "prefetch: min exceeds max";
77 }
78 u->pf_min = (ngx_uint_t) a;
79 u->pf_max = (ngx_uint_t) b;
80 return NGX_CONF_OK;
81}

83/* One clause at value[i]: returns the words consumed via *step, or an error. */
84static char *
85urlcgi_parse_clause(const ngx_str_t *value, ngx_uint_t i, ngx_uint_t n,
86 brix_cache_urlcgi_conf_t *u, unsigned *seen, ngx_uint_t *step)
87{
88 unsigned bit;
89
90 if (urlcgi_word_is(&value[i], "blocksize")) {
91 bit = URLCGI_CLAUSE_BLOCKSIZE;
92 } else if (urlcgi_word_is(&value[i], "prefetch")) {
93 bit = URLCGI_CLAUSE_PREFETCH;
94 } else {
95 return "unknown clause (expected blocksize or prefetch)";
96 }
97 if (*seen & bit) {
98 return "clause repeated";
99 }
100 *seen |= bit;
101
102 if (i + 1 < n && urlcgi_word_is(&value[i + 1], "ignore")) {
103 *step = 2; /* the pair stays (0, 0) */
104 return NGX_CONF_OK;
105 }
106 if (i + 2 >= n) {
107 return "clause needs \"ignore\" or <min> <max>";
108 }
109 *step = 3;
110 return (bit == URLCGI_CLAUSE_BLOCKSIZE)
111 ? urlcgi_parse_size_bounds(&value[i + 1], &value[i + 2], u)
112 : urlcgi_parse_count_bounds(&value[i + 1], &value[i + 2], u);
113}

115char *
116brix_conf_set_cache_urlcgi(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
117{
118 brix_cache_urlcgi_conf_t *u = (brix_cache_urlcgi_conf_t *)
119 ((char *) conf + cmd->offset);
120 const ngx_str_t *value = cf->args->elts;
121 ngx_uint_t n = cf->args->nelts, i = 1, step;
122 unsigned seen = 0;
123 char *err;
124
125 if (u->bs_max != NGX_CONF_UNSET_SIZE || u->pf_max != NGX_CONF_UNSET_UINT) {
126 return "is duplicate";
127 }
128 /* Naming the directive decides BOTH hints for this block: a clause left
129 * unnamed is explicitly ignored (upstream default), not inherited. */
130 u->bs_min = 0;
131 u->bs_max = 0;
132 u->pf_min = 0;
133 u->pf_max = 0;
134
135 while (i < n) {
136 err = urlcgi_parse_clause(value, i, n, u, &seen, &step);
137 if (err != NGX_CONF_OK) {
138 return err;
139 }
140 i += step;
141 }
142 return NGX_CONF_OK;
143}
