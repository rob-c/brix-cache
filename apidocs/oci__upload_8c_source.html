1/*
2 * oci_upload.c — the blob upload-session state machine (§D4.2, App. J.7).
3 *
4 * WHAT: POST/PATCH/PUT/GET/DELETE routing on `/v2/<name>/blobs/uploads[/<id>]`,
5 * plus the idle-session reaper. The bytes themselves land next door, in
6 * oci_upload_seal.c.
7 * WHY: this is the whole of `podman push`'s data plane, and its contract is
8 * resumability: a client that loses a connection halfway through a
9 * 600 MiB layer must be able to ask "how much did you get?" and carry
10 * on. That is why the session's state is the STAGED FILE ITSELF rather
11 * than a record beside it — the bytes on disk are the only answer that
12 * cannot disagree with reality, and a crashed worker leaves a session
13 * that is still exactly as resumable as it was a moment before.
14 * The J.7 states follow from that directly: OPEN is a zero-length part
15 * file, ACTIVE is a non-empty one, and SEALED / ABORTED / REAPED are
16 * all "the directory is gone" — which is why every one of them answers
17 * 404 and none of them needs a flag to say so.
18 * HOW: each body-carrying method records what it decided in ctx->reg and
19 * hands off to the shared async body reader; the callback finds the
20 * decision waiting for it.
21 *
22 * The raw namespace calls carry per-line vfs-seam-allow markers for the
23 * reason oci_store.c documents at its head: this is the registry's own
24 * staging area, not a VFS export.
25 */

112/* Does this PATCH continue where the staged file ends? A request with no
113 * Content-Range is "append here" by definition and always continues; one that
114 * carries a range only continues when its first byte is our next byte.
115 * Returns NGX_OK to append, NGX_DECLINED to answer 416.
116 *
117 * Real clients send layer PATCHes chunked with NO Content-Range at all
118 * (App. Y-2), so this is the resume path, not the common path. */
119static ngx_int_t
120oci_check_range(ngx_http_request_t *r, off_t have)
121{
122 ngx_table_elt_t *h = r->headers_in.content_range;
123 const u_char *p, *end;
124 off_t first = 0;
125
126 if (h == NULL || h->value.len == 0) {
127 return NGX_OK;
128 }
129 p = h->value.data;
130 end = p + h->value.len;
131
132 /* Accept both the bare "start-end" the registry spec uses and the full
133 * "bytes start-end/total" of RFC 7233 — clients send both. */
134 if ((size_t) (end - p) > 6 && ngx_strncasecmp((u_char *) p,
135 (u_char *) "bytes ", 6) == 0)
136 {
137 p += 6;
138 }
139 while (p < end && *p == ' ') { p++; }
140
141 if (p >= end || *p < '0' || *p > '9') {
142 return NGX_DECLINED;
143 }
144 while (p < end && *p >= '0' && *p <= '9') {
145 first = first * 10 + (*p++ - '0');
146 }
147 return (first == have) ? NGX_OK : NGX_DECLINED;
148}

188/* `?digest=` arrives percent-encoded from real clients (App. Y-1:
189 * `sha256%3A…`), so decode before the grammar sees it — comparing the raw

191static ngx_int_t
192oci_arg_digest(ngx_http_request_t *r, const char *name, size_t name_len,
193 brix_oci_digest_t *out)

196 size_t n;

208/* Append the request body onto the part-file at `base`, under the session's
209 * advisory lock. Returns the new length, or -1 with *status set. */
210/* ---- POST /v2/<name>/blobs/uploads/ ------------------------------------- */
211
212/* `?mount=<digest>&from=<repo>`: the blob is already in the global CAS, so
213 * "uploading" it again would move gigabytes to reach a byte-identical file.
214 * A mark and a 201 is the whole operation. */
215static ngx_int_t
216oci_upload_try_mount(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
217 const brix_oci_store_t *st)
218{
219 brix_oci_digest_t d;
220 char blob[PATH_MAX];
221
222 if (oci_arg_digest(r, "mount", 5, &d) != NGX_OK) {
223 return NGX_DECLINED;
224 }
225 if (brix_oci_store_blob_path(st, &d, blob, sizeof(blob)) != 0
226 || !brix_oci_store_exists(blob, NULL))
227 {
228 return NGX_DECLINED; /* fall back to a real upload */
229 }
230 if (brix_oci_store_mark_layer(st, ctx->req.name, ctx->req.name_len, &d,
231 r->connection->log) != NGX_OK)
232 {
233 return NGX_HTTP_INTERNAL_SERVER_ERROR;
234 }
235 return brix_oci_upload_created(r, &ctx->req, &d);
236}

239static ngx_int_t
240oci_upload_create(ngx_http_request_t *r, const brix_oci_store_t *st,
241 oci_upload_ctx_t *u)
242{
243 char dir[PATH_MAX];
244 char part[PATH_MAX];
245
246 oci_session_id(u->session, sizeof(u->session));
247 u->session_len = ngx_strlen(u->session);
248 u->st = *st;
249 u->base = 0;
250
251 if (brix_oci_store_upload_path(st, u->session, u->session_len, NULL,
252 dir, sizeof(dir)) != 0
253 || brix_oci_store_upload_path(st, u->session, u->session_len, "part",
254 part, sizeof(part)) != 0)
255 {
256 return NGX_ERROR;
257 }
258 if (brix_oci_store_mkparent(part, r->connection->log) != NGX_OK) {
259 return NGX_ERROR;
260 }
261 (void) chmod(dir, OCI_UPLOAD_SESSION_MODE); /* vfs-seam-allow: DOMAIN_REGISTRY — registry staging area, private to the worker until seal */
262
263 /* The empty part-file IS the session: creating it is what makes the
264 * session exist, and its absence is what makes every later request on a
265 * finished id answer 404 without a lookup table to consult. STAGE, not
266 * REGISTRY: a lost session marker is a retried upload, not a broken promise,
267 * so it renames without the fsync a durable domain would pay for. */
268 return brix_oci_store_publish_bytes(st, part, "", 0,
269 BRIX_VFS_DOMAIN_STAGE,
270 r->connection->log);
271}

274ngx_int_t
275brix_oci_upload_start(ngx_http_request_t *r,
276 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx,
277 brix_oci_store_t *st)
278{
279 oci_upload_ctx_t *u;
280 ngx_int_t rc;
281
282 if (r->method != NGX_HTTP_POST) {
283 return brix_oci_error(r, NGX_HTTP_NOT_ALLOWED,
284 BRIX_OCI_ERR_UNSUPPORTED,
285 "start an upload with POST");
286 }
287
288 /* Reaping here rather than on a timer keeps the sweep proportional to the
289 * traffic that creates sessions: an idle registry has nothing to reap and
290 * pays nothing to discover that. */
291 (void) brix_oci_upload_reap(st, lcf->upload_grace, r->connection->log);
292
293 rc = oci_upload_try_mount(r, ctx, st);
294 if (rc != NGX_DECLINED) {
295 return rc;
296 }
297
298 u = oci_upload_ctx(r, ctx);
299 if (u == NULL || oci_upload_create(r, st, u) != NGX_OK) {
300 return NGX_HTTP_INTERNAL_SERVER_ERROR;
301 }
302
303 rc = oci_arg_digest(r, "digest", 6, &u->want);
304 if (rc == NGX_ERROR) {
305 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
306 BRIX_OCI_ERR_DIGEST_INVALID, NULL);
307 }
308 if (rc == NGX_OK) {
309 /* The monolithic shortcut: body, hash and seal in one request. */
310 u->act = OCI_UP_MONOLITHIC;
311 return brix_http_read_body(r, brix_oci_upload_body_handler);
312 }
313
314 if (brix_oci_upload_headers(r, &ctx->req, u->session, 0) != NGX_OK) {
315 return NGX_HTTP_INTERNAL_SERVER_ERROR;
316 }
317 return brix_oci_reply_empty(r, NGX_HTTP_ACCEPTED);
318}

321/* ---- the /<session> methods --------------------------------------------- */
322
323ngx_int_t
324brix_oci_upload_session(ngx_http_request_t *r,
325 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx,
326 brix_oci_store_t *st)
327{
328 oci_upload_ctx_t *u;
329 char dir[PATH_MAX];
330 off_t size;
331 ngx_int_t rc;
332
333 size = oci_session_size(st, ctx->req.session, ctx->req.session_len);
334 if (size < 0) {
335 /* SEALED, ABORTED or REAPED — indistinguishable to a client, and
336 * deliberately so: all three mean "this id is finished". */
337 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
338 BRIX_OCI_ERR_BLOB_UPLOAD_UNKNOWN, NULL);
339 }
340
341 u = oci_upload_ctx(r, ctx);
342 if (u == NULL) {
343 return NGX_HTTP_INTERNAL_SERVER_ERROR;
344 }
345 u->st = *st;
346 u->base = size;
347 u->session_len = ctx->req.session_len;
348 ngx_memcpy(u->session, ctx->req.session, ctx->req.session_len);
349 u->session[ctx->req.session_len] = '\0';
350
351 if (r->method == NGX_HTTP_GET || r->method == NGX_HTTP_HEAD) {
352 if (brix_oci_upload_headers(r, &ctx->req, u->session, size) != NGX_OK) {
353 return NGX_HTTP_INTERNAL_SERVER_ERROR;
354 }
355 return brix_oci_reply_empty(r, NGX_HTTP_NO_CONTENT);
356 }
357
358 if (r->method == NGX_HTTP_DELETE) {
359 if (brix_oci_store_upload_path(st, u->session, u->session_len, NULL,
360 dir, sizeof(dir)) != 0)
361 {
362 return NGX_HTTP_INTERNAL_SERVER_ERROR;
363 }
364 brix_oci_store_drop_dir(dir, r->connection->log);
365 return brix_oci_reply_empty(r, NGX_HTTP_NO_CONTENT);
366 }
367
368 if (r->method == NGX_HTTP_PUT) {
369 rc = oci_arg_digest(r, "digest", 6, &u->want);
370 if (rc != NGX_OK) {
371 return brix_oci_error(r, NGX_HTTP_BAD_REQUEST,
372 BRIX_OCI_ERR_DIGEST_INVALID,
373 "a seal must name the digest it claims");
374 }
375 u->act = OCI_UP_SEAL;
376 return brix_http_read_body(r, brix_oci_upload_body_handler);
377 }
378
379 if (r->method != NGX_HTTP_PATCH) {
380 return brix_oci_error(r, NGX_HTTP_NOT_ALLOWED,
381 BRIX_OCI_ERR_UNSUPPORTED, NULL);
382 }
383
384 /* A Content-Range that does not continue where we are is the resume case,
385 * and the spec's answer is to state the truth and let the client seek:
386 * accepting it at the wrong offset would produce a blob that hashes to
387 * nothing anybody asked for. */
388 rc = oci_check_range(r, size);
389 if (rc != NGX_OK) {
390 if (brix_oci_upload_headers(r, &ctx->req, u->session, size) != NGX_OK) {
391 return NGX_HTTP_INTERNAL_SERVER_ERROR;
392 }
393 return brix_oci_error(r, NGX_HTTP_RANGE_NOT_SATISFIABLE,
394 BRIX_OCI_ERR_BLOB_UPLOAD_INVALID,
395 "resume from the offset in the Range header");
396 }
397
398 u->act = OCI_UP_PATCH;
399 return brix_http_read_body(r, brix_oci_upload_body_handler);
400}

405ngx_uint_t
406brix_oci_upload_reap(const brix_oci_store_t *st, time_t grace, ngx_log_t *log)
407{
408 char root[PATH_MAX];
409 char part[PATH_MAX];
410 char dir[PATH_MAX];
411 struct dirent *ent;
412 struct stat sb;
413 time_t now = ngx_time();
414 ngx_uint_t reaped = 0;
415 DIR *dh;
416
417 if (grace <= 0
418 || (size_t) snprintf(root, sizeof(root), "%s/_uploads",
419 st->root) >= sizeof(root))
420 {
421 return 0;
422 }
423 dh = opendir(root); /* vfs-seam-allow: DOMAIN_REGISTRY — registry staging area sweep, not a VFS export listing */
424 if (dh == NULL) {
425 return 0;
426 }
427
428 while ((ent = readdir(dh)) != NULL) { /* vfs-seam-allow: DOMAIN_REGISTRY — registry staging area sweep, not a VFS export listing */
429 if (ent->d_name[0] == '.') {
430 continue;
431 }
432 if ((size_t) snprintf(part, sizeof(part), "%s/%s/part",
433 root, ent->d_name) >= sizeof(part))
434 {
435 continue;
436 }
437 /* The part-file's mtime is the last time the client sent anything —
438 * which is exactly the idleness the grace window is about. */
439 if (stat(part, &sb) != 0 /* vfs-seam-allow: DOMAIN_REGISTRY — staging idleness probe */
440 || now - sb.st_mtime <= grace)
441 {
442 continue;
443 }
444 if ((size_t) snprintf(dir, sizeof(dir), "%s/%s",
445 root, ent->d_name) < sizeof(dir))
446 {
447 brix_oci_store_drop_dir(dir, log);
448 reaped++;
449 }
450 }
451 (void) closedir(dh);
452
453 return reaped;
454}
