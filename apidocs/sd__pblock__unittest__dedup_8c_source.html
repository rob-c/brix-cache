28/* ---- F10 refcounted blobs + dedup ----------------------------------------- *
29 * Introspect the catalog directly (a second SQLite connection, as a pytest
30 * would) to prove sharing: objects.blob_id tells us which physical blob backs a
31 * path, blobs.refcount tells us how many rows share it. */
32
33/* q_blob_id — the blob_id backing `path` (or "" if the row is missing). */
34static void
35q_blob_id(const char *root, const char *path, char *out, size_t cap)
36{
37 pbut_query_text(root, "SELECT blob_id FROM objects WHERE path = ?1;",
38 path, out, cap);
39}

41/* q_refcount — a blob's tracked refcount, or -1 if there is no row (which the
42 * driver reads as the implicit single reference). */
43static int
44q_refcount(const char *root, const char *blob_id)
45{
46 return pbut_query_int(root,
47 "SELECT refcount FROM blobs WHERE blob_id = ?1;", blob_id);
48}

50/* q_count — evaluate a literal single-value COUNT query against the catalog.
51 * -1 on any error. The SQL is a test constant, never attacker-derived. */
52static int
53q_count(const char *root, const char *sql)
54{
55 return pbut_query_int(root, sql, NULL);

58/* staged_put — publish `data` at `path` through the atomic staged path (the same
59 * path a wire PUT takes), so an overwrite fires F11 version capture / F10 dedup.
60 * 0 or -1. */
61static int
62staged_put(brix_sd_instance_t *inst, const char *path, const char *data,
63 size_t len)
64{
65 int err = 0;
66 brix_sd_staged_t *s = D->staged_open(inst, path, 0644, 0, &err);
67
68 if (s == NULL) {
69 errno = err;
70 return -1;
71 }
72 if (D->staged_write(s, data, len, 0) != (ssize_t) len) {
73 D->staged_abort(s);
74 return -1;
75 }
76 return D->staged_commit(s, NULL) == NGX_OK ? 0 : -1;
77}

79/* trunc_write — O_TRUNC-create `path` (breaks any share), write `data`, close. */
80static int
81trunc_write(brix_sd_instance_t *inst, const char *path, const char *data,
82 size_t len)
83{
84 int err = 0;
85 brix_sd_obj_t *o;
86 ssize_t n;
87
88 o = D->open(inst, path,
89 BRIX_SD_O_WRITE | BRIX_SD_O_READ | BRIX_SD_O_CREATE
90 | BRIX_SD_O_TRUNC,
91 0644, &err);
92 if (o == NULL) {
93 return -1;
94 }
95 n = D->pwrite(o, data, len, 0);
96 pb_close(o);
97 return (n == (ssize_t) len) ? 0 : -1;
98}

117void
118test_dedup_refs(void)
119{
120 char root[] = "/tmp/pb_dedup.XXXXXX";
121 brix_sd_instance_t inst = {0};
122 brix_sd_pblock_conf_t conf = {0};
123 char ba[PBLOCK_BLOB_ID_CAP], bb[PBLOCK_BLOB_ID_CAP];
124 char buf[64];
125 /* 10 bytes over a 4-byte stripe ⇒ 3 blocks: byte-verify walks real blocks. */
126 const char *DATA = "abcdefghij";
127 off_t bytes = 0;
128
129 CHECK(mkdtemp(root) != NULL, "mkdtemp");
130 lab_write_sidecar(root, "dedup=1");
131 conf.root = root;
132 conf.busy_timeout_ms = 2000;
133 conf.block_size = 4;
134 inst.driver = D;
135 inst.caps = D->caps;
136 CHECK(D->init(&inst, &conf) == NGX_OK, "dedup init");
137
138 /* SUCCESS: two identical PUTs fold onto one blob (refcount 2). */
139 CHECK(write_file(&inst, "/d1", DATA, 10) == 0, "seed d1");
140 CHECK(write_file(&inst, "/d2", DATA, 10) == 0, "seed d2");
141 q_blob_id(root, "/d1", ba, sizeof(ba));
142 q_blob_id(root, "/d2", bb, sizeof(bb));
143 CHECK(ba[0] && strcmp(ba, bb) == 0, "identical content shares blob");
144 CHECK(q_refcount(root, ba) == 2, "shared blob refcount 2 (got %d)",
145 q_refcount(root, ba));
146 CHECK(read_file(&inst, "/d1", buf, sizeof(buf)) == 10
147 && memcmp(buf, DATA, 10) == 0, "d1 reads back");
148 CHECK(read_file(&inst, "/d2", buf, sizeof(buf)) == 10
149 && memcmp(buf, DATA, 10) == 0, "d2 reads back");
150
151 /* Unlinking one sharer decrements, never removes the shared blocks. */
152 CHECK(D->unlink(&inst, "/d1", 0) == NGX_OK, "unlink d1");
153 CHECK(q_refcount(root, bb) == 1, "refcount 1 after one unlink (got %d)",
154 q_refcount(root, bb));
155 CHECK(read_file(&inst, "/d2", buf, sizeof(buf)) == 10
156 && memcmp(buf, DATA, 10) == 0, "survivor d2 intact");
157
158 /* server_copy = O(metadata) CoW: dst shares the src blob, refcount 2. */
159 CHECK(D->server_copy(&inst, "/d2", "/d3", &bytes) == NGX_OK, "cow copy");
160 q_blob_id(root, "/d3", ba, sizeof(ba));
161 CHECK(strcmp(ba, bb) == 0, "cow copy shares src blob");
162 CHECK(q_refcount(root, bb) == 2, "cow refcount 2 (got %d)",
163 q_refcount(root, bb));
164
165 /* ERROR/CoW break: overwriting a shared path forks a private blob; the
166 * sibling is untouched and the two blob_ids diverge. */
167 CHECK(trunc_write(&inst, "/d3", "ZZZ", 3) == 0, "overwrite d3");
168 q_blob_id(root, "/d3", ba, sizeof(ba));
169 CHECK(strcmp(ba, bb) != 0, "overwrite broke the share");
170 CHECK(read_file(&inst, "/d2", buf, sizeof(buf)) == 10
171 && memcmp(buf, DATA, 10) == 0, "d2 unchanged after d3 overwrite");
172 CHECK(read_file(&inst, "/d3", buf, sizeof(buf)) == 3
173 && memcmp(buf, "ZZZ", 3) == 0, "d3 has new content");
174 CHECK(q_refcount(root, bb) == 1, "src refcount back to 1 (got %d)",
175 q_refcount(root, bb));
176
177 /* CONCURRENCY (the plan's critical-correctness item): two threads break the
178 * share on the same source blob at once — each must end with its own private
179 * content, and neither may see the other's bytes. */
180 CHECK(write_file(&inst, "/c1", DATA, 10) == 0, "seed c1");
181 CHECK(D->server_copy(&inst, "/c1", "/c2", &bytes) == NGX_OK, "share c1→c2");
182 {
183 pthread_t t1, t2;
184 struct break_arg a1 = { &inst, "/c1", "11111", 5, -1 };
185 struct break_arg a2 = { &inst, "/c2", "222222", 6, -1 };
186
187 CHECK(pthread_create(&t1, NULL, break_body, &a1) == 0, "spawn t1");
188 CHECK(pthread_create(&t2, NULL, break_body, &a2) == 0, "spawn t2");
189 pthread_join(t1, NULL);
190 pthread_join(t2, NULL);
191 CHECK(a1.rc == 0 && a2.rc == 0, "both concurrent breaks succeeded");
192 }
193 CHECK(read_file(&inst, "/c1", buf, sizeof(buf)) == 5
194 && memcmp(buf, "11111", 5) == 0, "c1 has its own content");
195 CHECK(read_file(&inst, "/c2", buf, sizeof(buf)) == 6
196 && memcmp(buf, "222222", 6) == 0, "c2 has its own content");
197 q_blob_id(root, "/c1", ba, sizeof(ba));
198 q_blob_id(root, "/c2", bb, sizeof(bb));
199 CHECK(strcmp(ba, bb) != 0, "concurrent breaks produced distinct blobs");
200
201 D->cleanup(&inst);
202}

204/* SECURITY-NEG (W3 legacy tier): a colliding/forged "crc32:" content_hash
205 * must NOT let differing content alias — a non-sha256 hash only NOMINATES and
206 * dedup byte-verifies the candidate, rejecting the mismatch. ("sha256:" rows
207 * are the engine's own cryptographic identity and fold on trust — forging one
208 * requires writing catalog.db, which is already total store compromise.) */
209void
210test_dedup_forged_hash(void)
211{
212 char root[] = "/tmp/pb_forge.XXXXXX";
213 brix_sd_instance_t inst = {0};
214 brix_sd_pblock_conf_t conf = {0};
215 char bvictim[PBLOCK_BLOB_ID_CAP];
216 char battack[PBLOCK_BLOB_ID_CAP];
217 char hash[64];
218 char db[PATH_MAX], buf[64];
219 sqlite3 *h = NULL;
220 sqlite3_stmt *q = NULL;
221
222 CHECK(mkdtemp(root) != NULL, "mkdtemp");
223 lab_write_sidecar(root, "dedup=1");
224 conf.root = root;
225 conf.busy_timeout_ms = 2000;
226 conf.block_size = 4;
227 inst.driver = D;
228 inst.caps = D->caps;
229 CHECK(D->init(&inst, &conf) == NGX_OK, "forge init");
230
231 /* Victim and attacker: SAME size, DIFFERENT bytes, distinct blobs. */
232 CHECK(write_file(&inst, "/victim", "AAAAAAAA", 8) == 0, "seed victim");
233 CHECK(write_file(&inst, "/attack", "BBBBBBBB", 8) == 0, "seed attack");
234 q_blob_id(root, "/victim", bvictim, sizeof(bvictim));
235 q_blob_id(root, "/attack", battack, sizeof(battack));
236 CHECK(strcmp(bvictim, battack) != 0, "different content keeps blobs apart");
237
238 /* Force BOTH rows onto one legacy "crc32:" hash — a simulated CRC
239 * collision (or a forged legacy row): same-size, same-block-size
240 * candidates the SELECT will now cross-return. */
241 snprintf(hash, sizeof(hash), "crc32:00c0ffee");
242 snprintf(db, sizeof(db), "%s/catalog.db", root);
243 CHECK(sqlite3_open(db, &h) == SQLITE_OK, "forge db open");
244 if (sqlite3_prepare_v2(h,
245 "UPDATE blobs SET content_hash = ?2 WHERE blob_id IN (?1, ?3);",
246 -1, &q, NULL) == SQLITE_OK)
247 {
248 sqlite3_bind_text(q, 1, battack, -1, SQLITE_STATIC);
249 sqlite3_bind_text(q, 2, hash, -1, SQLITE_STATIC);
250 sqlite3_bind_text(q, 3, bvictim, -1, SQLITE_STATIC);
251 CHECK(sqlite3_step(q) == SQLITE_DONE, "forge update");
252 }
253 sqlite3_finalize(q);
254 sqlite3_close(h);
255
256 /* Re-nominate the victim through the dedup slot: its stored hash is the
257 * colliding legacy string, so the attacker blob is its only candidate —
258 * the byte-verify must reject the fold and the blobs stay apart. */
259 CHECK(D->dedup_publish != NULL
260 && D->dedup_publish(&inst, "/victim", "/.gcas/00/c0ffee") == NGX_OK,
261 "legacy re-publish completes");
262 {
263 char bnow[PBLOCK_BLOB_ID_CAP];
264
265 q_blob_id(root, "/victim", bnow, sizeof(bnow));
266 CHECK(strcmp(bnow, battack) != 0,
267 "colliding crc32 hash did NOT alias content");
268 }
269 CHECK(read_file(&inst, "/victim", buf, sizeof(buf)) == 8
270 && memcmp(buf, "AAAAAAAA", 8) == 0, "victim content honest");
271 CHECK(read_file(&inst, "/attack", buf, sizeof(buf)) == 8
272 && memcmp(buf, "BBBBBBBB", 8) == 0, "attacker content untouched");
273
274 D->cleanup(&inst);
275}

308/* F6 snapshots — take/restore via the reserved-namespace control paths, exactly
309 * as the wire reaches them: mkdir /.pblock/snap/<n> takes, mkdir
310 * /.pblock/restore/<n> restores, rmdir /.pblock/snap/<n> drops. snap=1 auto-arms
311 * F10 refs (a snapshot pins shared blobs so a delete-between-take-and-restore
312 * decrements rather than physically removes). Three legs:
313 * SUCCESS — snapshot, delete everything, restore, byte-identical reads.
314 * ERROR — restore refused (EBUSY, not corruption) while a handle is open.
315 * SECURITY-NEG — a hostile snapshot name (SQL/traversal metachars) is rejected;
316 * the name is only ever a bound column, so injection is out. */
317void
318test_snapshot(void)
319{
320 char root[] = "/tmp/pb_snap.XXXXXX";
321 brix_sd_instance_t inst = {0};
322 brix_sd_pblock_conf_t conf = {0};
323 char buf[64];
324
325 CHECK(mkdtemp(root) != NULL, "mkdtemp");
326 lab_write_sidecar(root, "snap=1");
327 conf.root = root;
328 conf.busy_timeout_ms = 2000;
329 conf.block_size = 4;
330 inst.driver = D;
331 inst.caps = D->caps;
332 CHECK(D->init(&inst, &conf) == NGX_OK, "snap init");
333
334 /* SUCCESS: seed two files (b spans >1 block), snapshot, delete everything,
335 * restore, and read back byte-identical content. */
336 CHECK(write_file(&inst, "/a", "alpha", 5) == 0, "seed a");
337 CHECK(write_file(&inst, "/b", "bravodata", 9) == 0, "seed b");
338 CHECK(D->mkdir(&inst, "/.pblock/snap/fix", 0755) == NGX_OK, "take snapshot");
339

347 CHECK(read_file(&inst, "/a", buf, sizeof(buf)) == 5

384/* F11 versioning + trash/undelete — versions=N holds the prior blob on each
385 * overwrite-publish (trimmed to N generations); trash=1 moves an unlink into the
386 * trash (blob held) and mkdir /.pblock/undelete/<path> pops it back. Both build
387 * ON F10 refs (a held blob is a live reference, decremented not freed). Legs:
388 * SUCCESS — three overwrites keep exactly N=2 versions (the oldest is
389 * trimmed and its blob freed); a trashed file undeletes
390 * byte-identical through the reserved control path.
391 * ERROR — undelete of a never-trashed name is ENOENT; undelete over a
392 * live object is EEXIST.
393 * SECURITY-NEG — a hostile undelete path (SQL metachars) is a bound column, so
394 * it can only miss (ENOENT); the trash table is unharmed and a
395 * legitimate undelete still works. */
