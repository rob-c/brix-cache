81/*
82 * root:// leg of the oracle: connect, read the object, hand back its MD5.
83 * Returns 0 on success; on failure records the probe (or prints the connect
84 * diagnostic), stores the process exit code in *exit_rc and returns -1.

86static int
87compare_root_md5(const diag_args *a, brix_url *ua, char *hex, size_t hexsz,

89{

96 *exit_rc = brix_shellcode(&st);

110/*
111 * Cleartext-WebDAV leg: GET the same logical path (binary-safe, so the body is
112 * buffered and checksummed through a temp fd rather than treated as a string).
113 * Same 0 / -1 + *exit_rc contract as compare_root_md5().
114 */
115static int
116compare_davs_md5(const char *dhost, int dport, const char *path,
117 char *hex, size_t hexsz, int *exit_rc)
118{
119 brix_status st;
120 char tmpl[] = "/tmp/xrddiag-davs.XXXXXX";
121 char *body;
122 size_t blen = 0;
123 int http = 0, fd;
124
125 *exit_rc = 1;
126 body = (char *) malloc(1u << 20);
127 if (body == NULL) {
128 fprintf(stderr, "xrddiag: out of memory\n");
129 *exit_rc = 51;
130 return -1;
131 }
132 brix_status_clear(&st);
133 if (brix_http_get(dhost, dport, path, 5000, &http, body, 1u << 20, &blen,
134 &st) != 0) {
135 probe("davs-http", 0, "GET %s:%d%s: %s", dhost, dport, path, st.msg);
136 free(body);
137 return -1;
138 }
139 probe("davs-http", http == 200, "HTTP %d for %s", http, path);
140 if (http != 200) {
141 free(body);
142 printf("Result: %d difference(s)\n", g_fails);
143 return -1;
144 }
145 fd = mkstemp(tmpl);
146 if (fd < 0 || (size_t) write(fd, body, blen) != blen
147 || brix_cksum_fd(fd, XRDC_CK_MD5, hex, hexsz, &st) != 0) {
148 probe("davs-md5", 0, "local md5 failed");
149 if (fd >= 0) { close(fd); unlink(tmpl); }
150 free(body);
151 return -1;
152 }
153 close(fd);
154 unlink(tmpl);
155 free(body);
156 return 0;
157}

162 * compared against the root:// MD5. Verify follows --no-verify-tls; the system
163 * trust store is used (ca_dir NULL). Advisory — a failure here records a probe
164 * but never changes the exit path.

166static void
167compare_davs_tls(const diag_args *a, const char *path, const char *root_md5)

191/*
192 * §15.6 cross-protocol consistency oracle: read the SAME object via root:// and
193 * WebDAV and assert size + MD5 agree. The capability no upstream client has —
194 * this project unifies the planes over one VFS, so a divergence here is a real
195 * cross-protocol bug. The cleartext WebDAV plane (--davs) is always compared;
196 * the HTTPS WebDAV plane (--davs-tls, TLS + chunked) is compared when supplied.
197 * The S3 (SigV4) plane remains deferred — noted, not implemented.
198 */
199int
200do_compare_davs(const diag_args *a)
201{
202 brix_url ua;
203 brix_status st;
204 char dhost[256];
205 int dport, rc = 0;
206 char root_md5[64], davs_md5[64];
207
208 brix_status_clear(&st);
209 if (brix_endpoint_parse(a->url, &ua, &st) != 0) {
210 fprintf(stderr, "xrddiag: %s\n", st.msg);
211 return 50;
212 }
213 if (ua.path[0] == '\0' || strcmp(ua.path, "/") == 0) {
214 fprintf(stderr, "xrddiag: compare --davs needs a file path in the URL\n");
215 return 50;
216 }
217 parse_http_hostport(a->davs, dhost, sizeof(dhost), &dport);
218 printf("Cross-protocol compare %s\n root:// %s:%d davs(http) %s:%d path %s\n",
219 a->url, ua.host, ua.port, dhost, dport, ua.path);
220
221 if (compare_root_md5(a, &ua, root_md5, sizeof(root_md5), &rc) != 0) {
222 return rc;
223 }
224 /* WebDAV plane: cleartext HTTP GET of the same logical path (binary-safe). */
225 if (compare_davs_md5(dhost, dport, ua.path, davs_md5, sizeof(davs_md5),
226 &rc) != 0) {
227 return rc;
228 }
229 probe("davs-md5", strcmp(root_md5, davs_md5) == 0,
230 "root=%s davs=%s", root_md5, davs_md5);
231
232 if (a->davs_tls != NULL && a->davs_tls[0] != '\0') {
233 compare_davs_tls(a, ua.path, root_md5);
234 }
235
236 note("s3", "deferred (needs SigV4 endpoint + credentials)");
237 printf("Result: %d difference(s)\n", g_fails);
238 return g_fails ? 1 : 0;
239}

242/*
243 * WHAT: Parse both compare URLs, validate the A-side path, and open both
244 * connections (`ca`, `cb`).
245 * WHY: Isolates the argument/URL validation + dual-connect prologue (with its
246 * exact error strings and exit codes) from the actual comparison so the
247 * orchestrator reads as a flat setup → compare → teardown.
248 * HOW: Reject a missing reference, a parse failure, or a bare/empty A path; then
249 * connect A and connect B (closing A if B fails). On success fills `ua`/`ub`
250 * and the open `ca`/`cb`. Returns 0 on success; on failure returns the exact
251 * exit code the inline code used (50 / brix_shellcode). No behavior change.
252 */
253static int
254compare_setup(const diag_args *a, brix_url *ua, brix_url *ub,
255 brix_conn *ca, brix_conn *cb)
256{
257 brix_status st;
258
259 if (a->ref_url == NULL) {
260 fprintf(stderr, "xrddiag: compare needs --vs-reference <url>\n");
261 return 50;
262 }
263 brix_status_clear(&st);
264 if (brix_endpoint_parse(a->url, ua, &st) != 0 ||
265 brix_endpoint_parse(a->ref_url, ub, &st) != 0) {
266 fprintf(stderr, "xrddiag: %s\n", st.msg);
267 return 50;
268 }
269 if (ua->path[0] == '\0' || strcmp(ua->path, "/") == 0) {
270 fprintf(stderr, "xrddiag: compare needs a file/dir path in the URL\n");
271 return 50;
272 }
273 if (brix_connect(ca, ua, &a->conn, &st) != 0) {
274 fprintf(stderr, "xrddiag: connect A %s:%d: %s\n", ua->host, ua->port, st.msg);
275 return brix_shellcode(&st);
276 }
277 if (brix_connect(cb, ub, &a->conn, &st) != 0) {
278 fprintf(stderr, "xrddiag: connect B %s:%d: %s\n", ub->host, ub->port, st.msg);
279 brix_close(ca);
280 return brix_shellcode(&st);
281 }
282 return 0;
283}

285/*
286 * WHAT: Compare two directories by their dirlist name-set (order-independent).
287 * WHY: Keeps the O(n*m) set-equality probe out of the orchestrator's compare
288 * block so the file-vs-dir branch stays a one-line dispatch.
289 * HOW: List both dirs; if either fails, emit a "dirlist-set" fail probe. Else
290 * equal iff counts match and every A name appears in B; emit the probe with
291 * the entry counts. Side effect: probe() output only. No behavior change.
292 */
293static void
294compare_dirs(brix_conn *ca, brix_conn *cb, const char *pa, const char *pb)
295{
296 brix_dirent *ea = NULL, *eb = NULL;
297 size_t na = 0, nb = 0;
298 int eq = 1;
299 brix_status sta, stb;
300
301 brix_status_clear(&sta);
302 brix_status_clear(&stb);
303 if (brix_dirlist(ca, pa, 0, &ea, &na, &sta) == 0 &&
304 brix_dirlist(cb, pb, 0, &eb, &nb, &stb) == 0) {
305 if (na != nb) {
306 eq = 0;
307 } else {
308 for (size_t i = 0; i < na && eq; i++) {
309 int hit = 0;
310 for (size_t j = 0; j < nb; j++) {
311 if (strcmp(ea[i].name, eb[j].name) == 0) { hit = 1; break; }
312 }
313 if (!hit) { eq = 0; }
314 }
315 }
316 probe("dirlist-set", eq, "A=%zu B=%zu entries", na, nb);
317 } else {
318 probe("dirlist-set", 0, "list failed");
319 }
320 free(ea);
321 free(eb);
322}

324/*
325 * WHAT: Compare two files by their remote MD5 digests.
326 * WHY: Confines the dual remote_md5 fetch + compare (and its failure-status
327 * formatting) to one helper mirroring compare_dirs.
328 * HOW: Fetch MD5 of both paths; on both-OK emit an "md5" probe of the string
329 * compare, else a fail probe carrying whichever side reported an error.
330 * Side effect: probe() output only. No behavior change.
331 */
332static void
333compare_files(brix_conn *ca, brix_conn *cb, const char *pa, const char *pb)
334{
335 char ha[64], hb[64];
336 brix_status ma, mb;
337
338 brix_status_clear(&ma);
339 brix_status_clear(&mb);
340 if (remote_md5(ca, pa, ha, sizeof(ha), &ma) == 0 &&
341 remote_md5(cb, pb, hb, sizeof(hb), &mb) == 0) {
342 probe("md5", strcmp(ha, hb) == 0, "A=%s B=%s", ha, hb);
343 } else {
344 probe("md5", 0, "A:%s B:%s",
345 ma.kxr ? ma.msg : "ok", mb.kxr ? mb.msg : "ok");
346 }

349/*
350 * WHAT: Stat both paths, probe size equality, then dispatch to the dir- or
351 * file-comparison helper.
352 * WHY: Splits the comparison body from setup/teardown; returns whether any early
353 * exit (stat failure) occurred so the orchestrator can short-circuit.
354 * HOW: Stat A and B; on either failure emit a "stat" probe and return 1 (caller
355 * returns 1). Else probe "size", then compare_dirs / compare_files per
356 * kXR_isDir. Returns 0 to continue. No behavior change.
357 */
358static int
359compare_probe(brix_conn *ca, brix_conn *cb, const brix_url *ua, const brix_url *ub)
360{
361 brix_statinfo sa, sb;
362 brix_status sta, stb;
363 const char *pa = ua->path;
364 const char *pb = (ub->path[0] && strcmp(ub->path, "/")) ? ub->path : ua->path;
365 int oka, okb;
366
367 brix_status_clear(&sta);
368 brix_status_clear(&stb);
369 oka = brix_stat(ca, pa, &sa, &sta) == 0;
370 okb = brix_stat(cb, pb, &sb, &stb) == 0;
371 if (!oka || !okb) {
372 probe("stat", 0, "A:%s B:%s", oka ? "ok" : sta.msg, okb ? "ok" : stb.msg);
373 return 1;
374 }
375 probe("size", sa.size == sb.size, "A=%lld B=%lld",
376 (long long) sa.size, (long long) sb.size);
377
378 if (sa.flags & kXR_isDir) {
379 compare_dirs(ca, cb, pa, pb);
380 } else {
381 compare_files(ca, cb, pa, pb);
382 }
383 return 0;
384}

386int
387do_compare(const diag_args *a)
388{
389 brix_url ua = {0}, ub = {0};
390 brix_conn ca, cb;
391 int rc;
392
393 if (a->davs != NULL) {
394 return do_compare_davs(a);
395 }
396
397 rc = compare_setup(a, &ua, &ub, &ca, &cb);
398 if (rc != 0) {
399 return rc;
400 }
401
402 printf("Compare %s vs %s\n", a->url, a->ref_url);
403
404 if (compare_probe(&ca, &cb, &ua, &ub) != 0) {
405 brix_close(&ca);
406 brix_close(&cb);
407 return 1;
408 }
409
410 brix_close(&ca);
411 brix_close(&cb);
412 printf("Result: %d difference(s)\n", g_fails);
413 return g_fails ? 1 : 0;
