23/* A sidecar / slice file the scrub drives off its data file, never as data of
24 * its own (mirrors cache_reap.c::reap_is_sidecar). The .cinfo/.meta records are
25 * the checksum carriers themselves. */
26static int
27scrub_is_sidecar(const char *name)
28{
29 const char *dot = strrchr(name, '.');
30
31 if (dot != NULL && (strcmp(dot, ".cinfo") == 0 || strcmp(dot, ".meta") == 0
32 || strcmp(dot, ".part") == 0 || strcmp(dot, ".lock") == 0))
33 {
34 return 1;
35 }
36 return strstr(name, ".__xrds") != NULL; /* slice files + slice meta */
37}

39/* Read exactly len bytes at off (EINTR-retried). 1 = full, 0 = short/error. */
40static int
41scrub_pread_full(int fd, unsigned char *buf, size_t len, off_t off)
42{
43 size_t got = 0;
44
45 while (got < len) {
46 ssize_t n = pread(fd, buf + got, len - got, off + (off_t) got);
47
48 if (n < 0 && errno == EINTR) {
49 continue;
50 }
51 if (n <= 0) {
52 return 0;
53 }
54 got += (size_t) n;
55 }
56 return 1;
57}

59/* Verify every set block CRC in *xm against fd's bytes. Returns the number of
60 * mismatches found (bumping the per-block stat counters); a short on-disk block
61 * (record longer than the file) is skipped, never failed. */
62static uint64_t
63scrub_verify_fd(const char *path, int fd, off_t fsize, const brix_xmeta_t *xm,
64 brix_csi_scrub_stats_t *st, brix_csi_scrub_report_fn report, void *u)
65{
66 int64_t g = xm->buffer_size;
67 uint64_t b, mism = 0;
68 unsigned char *blockbuf = malloc((size_t) g);
69
70 if (blockbuf == NULL) {
71 st->errors++;
72 return 0;
73 }
74 for (b = 0; b < xm->nblocks; b++) {
75 int64_t bstart = (int64_t) b * g;
76 int64_t bend = bstart + g;
77 uint32_t got;
78
79 if (xm->blockcrc[b] == BRIX_XMETA_CRC_UNSET) {
80 st->blocks_unset++;
81 continue;
82 }
83 if (bend > xm->file_size) {
84 bend = xm->file_size;
85 }
86 if (bend > fsize || bend <= bstart
87 || !scrub_pread_full(fd, blockbuf, (size_t) (bend - bstart), bstart))
88 {
89 st->blocks_short++; /* cannot vouch for it — skip */
90 continue;
91 }
92 got = brix_crc32c_value(blockbuf, (size_t) (bend - bstart));
93 if (got != xm->blockcrc[b]) {
94 st->mismatches++;
95 mism++;
96 if (report != NULL) {
97 report(u, path, b, xm->blockcrc[b], got);
98 }
99 } else {
100 st->blocks_verified++;
101 }
102 }
103 free(blockbuf);
104 return mism;
105}

107int
108brix_csi_scrub_file(const char *path, brix_csi_scrub_stats_t *st,
109 brix_csi_scrub_report_fn report, void *u)
110{
111 brix_xmeta_t xm;
112 struct stat fst;
113 int fd, rc;
114 uint64_t mism;
115
116 if (path == NULL || st == NULL) {
117 errno = EINVAL;
118 return BRIX_CSI_ERR;
119 }
120 st->files_scanned++;
121
122 switch (brix_xmeta_path_load(path, &xm)) {
123 case BRIX_XMETA_OK:
124 break;
125 case BRIX_XMETA_FOREIGN:
126 return BRIX_CSI_NOTAGS;
127 default:
128 st->errors++;
129 return BRIX_CSI_ERR;
130 }
131 if (!xm.have_blockcrc || xm.blockcrc == NULL || xm.buffer_size <= 0
132 || xm.nblocks == 0)
133 {
134 brix_xmeta_free(&xm);
135 return BRIX_CSI_NOTAGS;
136 }
137 st->files_tagged++;
138
139 fd = open(path, O_RDONLY | O_NOCTTY | O_CLOEXEC | O_NOFOLLOW);
140 if (fd < 0 || fstat(fd, &fst) != 0 || !S_ISREG(fst.st_mode)) {
141 if (fd >= 0) {
142 close(fd);
143 }
144 brix_xmeta_free(&xm);
145 st->errors++;
146 return BRIX_CSI_ERR;
147 }
148
149 mism = scrub_verify_fd(path, fd, fst.st_size, &xm, st, report, u);
150 close(fd);
151 brix_xmeta_free(&xm);
152 rc = (mism > 0) ? BRIX_CSI_MISMATCH : BRIX_CSI_OK;
153 return rc;
154}

163/* Recurse dir (same device as root), scrubbing regular data files, stopping
164 * once *remaining files have been scanned (SIZE_MAX = unlimited). Returns the
165 * count scanned in this subtree. */
166static long
167scrub_dir(const char *dir, const scrub_ctx_t *sc, brix_csi_scrub_stats_t *st,
168 long *remaining)
169{
170 DIR *dp;
171 struct dirent *de;
172 long n = 0;
173 char child[PATH_MAX];
174
175 dp = opendir(dir);
176 if (dp == NULL) {
177 return 0;
178 }
179 while (*remaining != 0 && (de = readdir(dp)) != NULL) {
180 struct stat st_c;
181
182 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0
183 || scrub_is_sidecar(de->d_name))
184 {
185 continue;
186 }
187 if (snprintf(child, sizeof(child), "%s/%s", dir, de->d_name)
188 >= (int) sizeof(child))
189 {
190 continue;
191 }
192 if (lstat(child, &st_c) != 0 || st_c.st_dev != sc->dev) {
193 continue;
194 }
195 if (S_ISDIR(st_c.st_mode)) {
196 n += scrub_dir(child, sc, st, remaining);
197 continue;
198 }
199 if (!S_ISREG(st_c.st_mode)) {
200 continue;
201 }
202 (void) brix_csi_scrub_file(child, st, sc->report, sc->u);
203 n++;
204 if (*remaining > 0) {
205 (*remaining)--;
206 }
207 }
208 closedir(dp);
209 return n;
210}
