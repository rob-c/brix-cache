1/*
2 * streams.c — M8 parallel data streams (kXR_bind substreams).
3 *
4 * WHAT: Attach up to N-1 secondary TCP connections to a primary session via
5 * kXR_bind, and tear them down again.
6 * WHY: `xrdcp --streams N` opens extra bound channels — the parallel-transfer
7 * affordance of the protocol.
8 * HOW: Each secondary re-runs handshake + kXR_protocol [+ TLS] then sends
9 * kXR_bind{primary sessid} (brix_bind in conn.c). The Phase-94 pumps
10 * then fan self-addressed kXR_read/kXR_write REQUEST FRAMES across the
11 * bound secondaries — a BriX extension: stock servers treat a bound path
12 * as a pathid-directed DATA channel (ClientWriteRequest.pathid) and
13 * never answer a request frame arriving there, which would hang the
14 * transfer. So after binding, the primary probes kXR_Qconfig
15 * "brix.substreams": without the "=rw" marker (BriX answers it, stock
16 * echoes the unknown key) the secondaries are torn down again and the
17 * whole transfer stays on the primary. Best-effort throughout: a
18 * secondary that won't bind is skipped, never failing the copy.
19 */

25int
26brix_streams_open(brix_streamset *ss, brix_conn *primary, int streams,
27 brix_status *st)
28{
29 int want, i;
30
31 memset(ss, 0, sizeof(*ss));
32 if (streams <= 1) {
33 return 0;
34 }
35 want = streams - 1;
36 if (want > XRDC_MAX_STREAMS - 1) {
37 want = XRDC_MAX_STREAMS - 1;
38 }
39
40 for (i = 0; i < want; i++) {
41 brix_status bst;
42 brix_status_clear(&bst);
43 if (brix_bind(&ss->sec[ss->n], primary, &bst) != 0) {
44 /* Best-effort: stop at the first failure, keep what bound. */
45 break;
46 }
47 ss->n++;
48 }
49
50 if (ss->n > 0) {
51 char reply[64];
52 brix_status qst;
53
54 brix_status_clear(&qst);
55 if (brix_query(primary, kXR_Qconfig, "brix.substreams",
56 reply, sizeof(reply), &qst) != 0
57 || strstr(reply, "=rw") == NULL)
58 {
59 /* Peer does not serve request frames on bound paths (stock
60 * semantics) — a fanned write would wait forever. Run primary-only. */
61 brix_streams_close(ss);
62 }
63 }
64 (void) st;
65 return ss->n;
66}

68void
69brix_streams_close(brix_streamset *ss)
70{
71 int i;
72 for (i = 0; i < ss->n; i++) {
73 /* A bound stream owns no session of its own — close it quietly, no
74 * kXR_endsess (that belongs to the primary). */
75 brix_tls_free(&ss->sec[i]);
76 if (ss->sec[i].io.fd >= 0) {
77 close(ss->sec[i].io.fd);
78 ss->sec[i].io.fd = -1;
79 }
80 }
81 ss->n = 0;
82}
