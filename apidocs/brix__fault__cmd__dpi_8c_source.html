26/* Render "a.b.c.d:port" as the FTP PASV "a,b,c,d,p1,p2" byte form into `out`.
27 * Returns the length, or 0 if the endpoint does not parse as IPv4:port. */
28static int
29ftp_comma_form(const char *ep, unsigned char *out, size_t cap)
30{
31 unsigned a, b, c, d, port;
32 if (sscanf(ep, "%u.%u.%u.%u:%u", &a, &b, &c, &d, &port) != 5 ||
33 a > 255 || b > 255 || c > 255 || d > 255 || port > 65535) {
34 return 0;
35 }
36 int n = snprintf((char *) out, cap, "%u,%u,%u,%u,%u,%u",
37 a, b, c, d, port >> 8, port & 0xff);
38 return (n > 0 && (size_t) n < cap) ? n : 0;
39}

41/* alg-rewrite <src ip:port> <dst ip:port> [up|down] — model an FTP/GridFTP ALG
42 * that rewrites the endpoint embedded in a PASV reply, installed as a find/
43 * replace pair on the selected direction (default down: the 227 response). */
44static int
45dpi_alg_rewrite(char *args, char *reply, size_t rsz)
46{
47 char *src = strtok(args, " ");
48 char *dst = strtok(NULL, " ");
49 char *dir = strtok(NULL, " ");
50 if (!src || !dst) {
51 snprintf(reply, rsz, "err: alg-rewrite <src ip:port> <dst ip:port> "
52 "[up|down]\n");
53 return 1;
54 }
55 unsigned char fbuf[64], rbuf[64];
56 int fl = ftp_comma_form(src, fbuf, sizeof(fbuf));
57 int rl = ftp_comma_form(dst, rbuf, sizeof(rbuf));
58 if (fl == 0 || rl == 0) {
59 snprintf(reply, rsz, "err: alg-rewrite endpoints must be IPv4:port\n");
60 return 1;
61 }
62 struct fp_mutbuf *M = (dir && strcmp(dir, "up") == 0) ? &g_up_mut : &g_down_mut;
63 pthread_mutex_lock(&g_ext_lock);
64 memcpy(M->find, fbuf, (size_t) fl); M->find_len = fl;
65 memcpy(M->repl, rbuf, (size_t) rl); M->repl_len = rl;
66 pthread_mutex_unlock(&g_ext_lock);
67 return 1;
68}

70/* drop-fin [up|down|both|off] — swallow a direction's EOF (asymmetric teardown).*/
71static void
72dpi_drop_fin(char *args)
73{
74 if (strstr(args, "off")) {
75 g_drop_fin_up = 0; g_drop_fin_down = 0;
76 return;
77 }
78 int d = dir_of(args); /* 0 both / 1 up / 2 down */
79 if (d != 2) { g_drop_fin_up = 1; }
80 if (d != 1) { g_drop_fin_down = 1; }
81}
