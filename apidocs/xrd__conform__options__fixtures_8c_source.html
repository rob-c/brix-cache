4static int
5named_target(const char *value, const char *const *names, size_t name_count,
6 char *name_out, size_t name_capacity, const char **target_out)
7{
8 const char *equals = strchr(value, '=');
9 size_t name_length;
10 size_t index;
11
12 if (equals == NULL || equals == value || equals[1] == '\0') return -1;
13 name_length = (size_t) (equals - value);
14 if (name_length >= name_capacity) return -1;
15 for (index = 0; index < name_count; index++)
16 if (strlen(names[index]) == name_length
17 && memcmp(value, names[index], name_length) == 0) {
18 memcpy(name_out, value, name_length);
19 name_out[name_length] = '\0';
20 *target_out = equals + 1;
21 return 0;
22 }
23 return -1;
24}

171/* WHAT: Parse one denied/mutation plane fixture into a typed destination.
172 * WHY: Both option families have identical vocabulary and duplicate rules.
173 * HOW: Parse through named_target, reject duplicate planes, then publish once.
174 */
175static int
176parse_plane_fixture(int argc, char **argv, int *index, const char *option,
177 xrd_conform_denied_fixture *fixtures, size_t *count)
178{
179 static const char *const planes[] = { "root", "http", "s3" };
180 xrd_conform_denied_fixture candidate;
181 size_t fixture_index;
182
183 if (strcmp(argv[*index], option) != 0) {
184 return 0;
185 }
186 if (*index + 1 >= argc || *count >= XRDC_CONFORM_DENIED_FIXTURE_MAX) {
187 return -1;
188 }
189 if (named_target(argv[++(*index)], planes, sizeof(planes) / sizeof(planes[0]),
190 candidate.plane, sizeof(candidate.plane), &candidate.target) != 0) {
191 return -1;
192 }
193 for (fixture_index = 0; fixture_index < *count; fixture_index++) {
194 if (strcmp(fixtures[fixture_index].plane, candidate.plane) == 0) {
195 return -1;
196 }
197 }
198 fixtures[(*count)++] = candidate;
199 return 1;
200}
