4/*
5 * cns_inventory.h — the CNS path→metadata inventory table (pure, nginx-free).
6 *
7 * WHAT: a fixed-capacity, pointer-free (POD) table of file/dir entries the CNS
8 * manager keeps for the whole federation namespace. The layout is
9 * deliberately plain-old-data — no embedded pointers, no per-entry heap —
10 * so the SAME struct can live either in a per-worker heap block (the v1
11 * single-worker path) or in an nginx shared-memory slab region shared
12 * across every manager worker (the v2 multi-worker path). cns.c owns the
13 * backing store and the mutex; these ops assume the caller holds it.
14 * WHY: a multi-worker redirector had a per-worker inventory, so a mutation that
15 * landed on worker A was invisible to a stat that landed on worker B. A
16 * shared table fixes that; keeping the table logic pure (this file) from
17 * the SHM plumbing (cns.c) lets the slot/upsert/delete semantics be unit-
18 * tested deterministically without an nginx runtime.
19 * HOW: linear-probe over `capacity` fixed slots (a small federation namespace);
20 * apply upserts on ADD/MKDIR and clears on DEL/RMDIR; stat is a lookup.
21 */
22

64/*
65 * Apply one rename. Moves the entry at `oldp` to `newp` AND re-prefixes every
66 * entry that lives *under* `oldp` (i.e. whose path starts with `oldp` followed
67 * by '/'), so renaming a directory carries its whole recorded subtree instead of
68 * stranding the children at a path the cluster no longer has. `size`/`mtime`/
69 * `is_dir` describe the destination as the data server observed it after the
70 * rename and are authoritative for the moved entry; a source the manager never
71 * recorded is inserted rather than dropped, so an inventory that missed the
72 * original ADD still converges.
73 *
74 * A child whose re-prefixed path would exceed BRIX_CNS_PATH_MAX is *removed*
75 * rather than truncated — a truncated path would fabricate an entry naming a
76 * file that does not exist, and a miss is always safer than a wrong hit.
77 *
78 * Returns 0 on success, -1 on a bad argument or a full table. Caller holds the
79 * lock.
80 */
81int brix_cns_inv_rename(brix_cns_inv_t *inv, const char *oldp, const char *newp,
82 uint64_t size, uint64_t mtime, int is_dir, uint32_t server_id);
83
84/* Look `path` up. On a hit fills size/mtime/is_dir (any may be NULL) and returns
