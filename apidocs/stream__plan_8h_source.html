1/*
2 * stream_plan.h — pure planning helpers for the multi-stream native TPC pull.
3 *
4 * WHAT: the three decisions the parallel-stream read loop (source_stream_multi.c)
5 * makes per round, kept free of nginx and sockets so they are unit-tested
6 * directly (stream_plan_unittest.c) instead of only through a live pull:
7 * - tpc_stream_plan_clamp: how many streams a transfer may use, from
8 * the client's tpc.str hint and the server cap
9 * - tpc_stream_plan_round: how far a round of N windowed reads advanced
10 * the file, and whether it reached EOF
11 * - tpc_stream_plan_read_args: the 8-byte kXR_read read_args payload that
12 * steers a reply out of a bound sub-stream
13 * WHY: a windowed pull that mis-accounts a short window silently commits a file
14 * with a hole in it; a clamp that trusts the client hint lets one transfer
15 * open an unbounded fan of sockets to a source. Both are pure arithmetic and
16 * belong under a deterministic test, not behind a rendezvous.
17 * HOW: header-only prototypes; the implementation (stream_plan.c) uses only the
18 * C library. Wire facts: XRootD read_args = { pathid[1], reserved[7] }
19 * (XProtocol.hh), pathid 0 = the requesting (primary) stream.
20 */

26/* Absolute ceiling on parallel source streams for one native TPC pull: the
27 * primary plus up to 14 bound sub-streams. Matches the largest value the
28 * brix_tpc_streams directive accepts (its range check names this number). */
29#define TPC_STREAMS_MAX 15
30#define TPC_SUBSTREAMS_MAX (TPC_STREAMS_MAX - 1)

32/* Length of the kXR_read read_args payload (pathid + 7 reserved bytes). */

35/*
36 * Resolve the stream count for a transfer. `hint` is the client's tpc.str value
37 * (NULL/empty/non-numeric/<=0 → 1: a single stream, the pre-F7 behaviour); `cap`
38 * is the server's brix_tpc_streams (<1 → 1). Returns a value in [1, min(cap,
39 * TPC_STREAMS_MAX)] — the client can lower the server's ceiling, never raise it.
40 */
41int tpc_stream_plan_clamp(const char *hint, int cap);
42
43/*
44 * Judge one round of `nslots` windowed reads of `window` bytes each, issued at
45 * consecutive offsets (slot i covers [base + i*window, base + (i+1)*window)).
46 * got[i] = bytes the source delivered for slot i. On success sets *advance to
47 * the contiguous byte count the round proved (the sum up to and including the
48 * first short slot) and *eof to 1 when slot 0 delivered nothing (the source is
49 * exhausted; nothing after it can hold data). Returns 0, or -1 when a slot AFTER
50 * a short one still delivered bytes — a hole that must fail the pull rather than
51 * be committed. `got` may not be NULL; nslots <= 0 is an error.
52 */
53int tpc_stream_plan_round(const size_t *got, int nslots, size_t window,
54 size_t *advance, int *eof);
55
56/*
57 * Fill the kXR_read read_args payload that asks the source to answer on the
58 * sub-stream bound as `pathid` (0 = answer on the requesting stream).
59 */
60void tpc_stream_plan_read_args(unsigned char out[TPC_READ_ARGS_LEN],
