1/*
2 * sd_pblock_batch.c - bulk namespace delete for the packed-block store
3 * (phase-107 C4).
4 *
5 * WHAT: sd_pblock_unlink_many and the gated core its _cred twin shares. The
6 * whole window runs inside ONE SQLite transaction: N unlinks cost one
7 * journal commit (one fsync in WAL) instead of N autocommits.
8 * WHY: This driver has no wire batch to exploit - the win is transactional.
9 * The per-key semantics are IDENTICAL to sd_pblock_unlink called in a
10 * loop (lease gate F15, trash push F11, refcount release F10, audit F17
11 * all run per key); only the commit boundary moves.
12 * HOW: BEGIN IMMEDIATE -> per-key [gate ->] sd_pblock_unlink -> COMMIT. A
13 * per-key failure is RECORDED (errs[i]) and the batch continues - a
14 * failed statement does not poison the transaction. Only a failed
15 * BEGIN/COMMIT fails the batch itself (done = 0: the rollback un-did
16 * every key, so none may be reported deleted). No slot in the unlink
17 * call graph opens its own transaction (rename and the snapshot ctl
18 * mutators do, but directories never enter a batch - contract in
19 * sd_batch_types.h), so the wrap cannot nest.
20 */

37/*
38 * The shared core. `gate` is the _cred twin's per-key POSIX authority check
39 * (W+X on the parent + sticky rule, sd_pblock_cred.c); NULL for the plain
40 * slot and for a service identity. A key the gate refuses records the gate's
41 * errno and is NEVER attempted - the authority verdict must precede the
42 * mutation, per key, exactly as the single-key slot orders it.
43 */
44ngx_int_t
45sd_pblock_unlink_many_core(brix_sd_instance_t *inst, brix_sd_unlink_batch_t *b,
46 sd_pblock_key_gate_fn gate, void *gctx)
47{
48 pblock_state_t *st = inst->state;
49 size_t i;
50
51 if (b->n > BRIX_SD_BULK_DELETE_WINDOW) {
52 errno = EINVAL;
53 return NGX_ERROR;
54 }
55 if (b->n == 0) {
56 b->done = 0;
57 return NGX_OK;
58 }
59 if (cat_exec(st->cat, "BEGIN IMMEDIATE;") != 0) {
60 errno = EIO;
61 return NGX_ERROR;
62 }
63 for (i = 0; i < b->n; i++) {
64 errno = 0;
65 if (gate != NULL && gate(gctx, b->paths[i]) != NGX_OK) {
66 b->errs[i] = (errno != 0) ? errno : EACCES;
67 continue;
68 }
69 if (sd_pblock_unlink(inst, b->paths[i], 0) == NGX_OK) {
70 b->errs[i] = 0;
71 } else {
72 b->errs[i] = (errno != 0) ? errno : EIO;
73 }
74 }
75 b->done = b->n;
76 if (cat_exec(st->cat, "COMMIT;") != 0) {
77 /* The rollback un-does every key above: report NONE attempted, or a
78 * caller would answer <Deleted> for rows that still exist. */
79 (void) cat_exec(st->cat, "ROLLBACK;");
80 b->done = 0;
81 errno = EIO;
82 return NGX_ERROR;
83 }
84 return NGX_OK;
85}
