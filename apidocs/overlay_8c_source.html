1/*
2 * overlay.c — writable union overlay core for brixMount cvmfs-rw.
3 *
4 * WHAT: implements the upper-tree primitives declared in overlay.h: the
5 * O_NOFOLLOW component walk, union classification, whiteout/opaque
6 * markers, mutations, atomic copy-up, readdir nameset, and CLI cores.
7 * WHY: one small, FUSE/CVMFS-free translation unit owns every union corner
8 * case so they are provable on plain tmp directories.
9 * HOW: ov_split() peels validated path components (".", "..", empty and
10 * oversized components are refused); ov_walk_parent() descends dir by
11 * dir with O_NOFOLLOW|O_DIRECTORY so planted symlinks dead-end instead
12 * of escaping; everything else composes those two.
13 */

29/* ---- path-component parsing --------------------------------------------- */

31/* Peel the leading component of *relp into name[]; advances *relp past the
32 * '/' (or leaves it NULL on the last component). Returns 1 = more components
33 * follow, 0 = this was the leaf, -EINVAL = empty/"."/".."/oversized. */
34static int ov_split(const char **relp, char *name, size_t cap) {
35 const char *rel = *relp;
36 const char *slash = strchr(rel, '/');
37 size_t n = slash ? (size_t) (slash - rel) : strlen(rel);
38
39 if (n == 0 || n >= cap) return -EINVAL;
40 memcpy(name, rel, n);
41 name[n] = '\0';
42 if (strcmp(name, ".") == 0 || strcmp(name, "..") == 0) return -EINVAL;
43 *relp = slash ? slash + 1 : NULL;
44 return slash ? 1 : 0;
45}

65/* ---- the O_NOFOLLOW walk ------------------------------------------------- */
66
67/* ---- Open (optionally create) one directory component of the walk ----
68 *
69 * WHAT: Opens child `name` under `dirfd` with O_NOFOLLOW|O_DIRECTORY. When
70 * `mk` is set and the child is missing, it is created — mode_fn(ud,
71 * prefix) supplies the mode, or 0755 when mode_fn is NULL — then
72 * reopened. Returns the child dirfd (caller closes) or -errno. `dirfd`
73 * is left open for the caller to close in every case.
74 *
75 * WHY: Factoring the create-if-missing-then-reopen step keeps ov_walk_parent_mk
76 * a flat descent loop and confines the mkdir race handling (a racing
77 * creator yielding EEXIST is benign) to one place.
78 *
79 * HOW: 1. Try an O_NOFOLLOW openat of the child directory.
80 * 2. On ENOENT with `mk`, mkdirat with the computed mode (EEXIST is
81 * tolerated) and reopen.
82 * 3. Map any residual failure to -errno; errno reflects the last syscall.
83 */
84static int ov_descend_dir(int dirfd, const char *name, int mk,
85 mode_t (*mode_fn)(void *ud, const char *rel_dir),
86 void *ud, const char *prefix) {
87 int next = openat(dirfd, name, O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC);
88 if (next < 0 && mk && errno == ENOENT) {
89 mode_t m = mode_fn ? mode_fn(ud, prefix) : 0755;
90 if (mkdirat(dirfd, name, m) != 0 && errno != EEXIST) return -errno;
91 next = openat(dirfd, name, O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC);
92 }
93 return next < 0 ? -errno : next;
94}

96/* Open the parent directory of `rel`'s leaf, descending from upper_fd one
97 * component at a time with O_NOFOLLOW|O_DIRECTORY (symlinks dead-end). With
98 * `mk`, missing intermediate dirs are created (mode_fn(ud, prefix) or 0755).
99 * The leaf component is stored in leaf[]. Returns the parent dirfd (caller
100 * closes) or -errno. rel must be non-empty. */
101int ov_walk_parent_mk(const brix_overlay *ov, const char *rel,
102 char *leaf, size_t leafcap, int mk,
103 mode_t (*mode_fn)(void *ud, const char *rel_dir),
104 void *ud) {
105 if (rel == NULL || rel[0] == '\0') return -EINVAL;
106
107 int cur = openat(ov->upper_fd, ".", O_RDONLY | O_DIRECTORY | O_CLOEXEC);
108 if (cur < 0) return -errno;
109
110 char prefix[4096]; /* walked prefix, for mode_fn */
111 size_t plen = 0;
112 const char *p = rel;
113 prefix[0] = '\0';
114
115 for (;;) {
116 int more = ov_split(&p, leaf, leafcap);
117 if (more < 0) { close(cur); return more; }
118 if (more == 0) return cur; /* cur = the leaf's parent */
119
120 size_t ll = strlen(leaf);
121 if (plen + ll + 2 < sizeof(prefix)) {
122 if (plen > 0) prefix[plen++] = '/';
123 memcpy(prefix + plen, leaf, ll + 1);
124 plen += ll;
125 }
126
127 int next = ov_descend_dir(cur, leaf, mk, mode_fn, ud, prefix);
128 close(cur);
129 if (next < 0) return next;
130 cur = next;
131 }
132}

134static int ov_walk_parent(const brix_overlay *ov, const char *rel,
135 char *leaf, size_t leafcap) {
136 return ov_walk_parent_mk(ov, rel, leaf, leafcap, 0, NULL, NULL);
137}

156/* ---- classification ------------------------------------------------------ */

158/* ---- Classify the repo root (empty rel) as the upper directory ----
159 *
160 * WHAT: Reports the union root as BRIX_OV_UPPER, filling *st from the upper
161 * dirfd when st is non-NULL. Returns 0, or -errno if the fstat fails.
162 *
163 * WHY: The empty relative path names the union root, which is always the
164 * upper directory itself; handling it separately keeps the component
165 * walk in brix_overlay_classify free of the root special case.
166 *
167 * HOW: 1. fstat the upper dirfd into *st when a stat buffer was supplied.
168 * 2. Set *state to BRIX_OV_UPPER and return success.
169 */
170static int ov_classify_root(const brix_overlay *ov, struct stat *st,
171 brix_ov_state *state) {
172 if (st != NULL && fstat(ov->upper_fd, st) != 0) return -errno;
173 *state = BRIX_OV_UPPER;
174 return 0;
175}

177/* ---- Descend into an upper subdirectory during classification ----
178 *
179 * WHAT: Opens directory component `name` under `cur` with O_NOFOLLOW and
180 * closes `cur` unconditionally. Sets *opaque when the child carries the
181 * opaque marker. Returns the child dirfd (caller closes) or -errno.
182 *
183 * WHY: The classify walk must note an opaque directory the moment it enters
184 * one and must never follow a planted symlink; isolating the descend
185 * keeps that ordering explicit and the walk loop flat.
186 *
187 * HOW: 1. O_NOFOLLOW openat of the child directory, then close the parent.
188 * 2. On failure return -errno.
189 * 3. Probe for the opaque marker in the child and record it in *opaque.
190 */
191static int ov_descend_child(int cur, const char *name, int *opaque) {
192 int next = openat(cur, name, O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC);
193 close(cur);
194 if (next < 0) return -errno;
195 if (ov_marker_at(next, BRIX_OV_OPQ_NAME)) *opaque = 1;
196 return next;
197}

199/* Walk `rel` from the upper root and decide how the union sees it: a whiteout
200 * on any component, an opaque ancestor, or a non-dir upper component shadowing
201 * the remainder all mask the lower layer; a leaf found in upper wins; anything
202 * else falls through to lower. */
203int brix_overlay_classify(const brix_overlay *ov, const char *rel,
204 struct stat *st, brix_ov_state *state) {
205 *state = BRIX_OV_NONE;
206 if (rel == NULL) return -EINVAL;
207
208 if (rel[0] == '\0') /* the repo root is upper/ */
209 return ov_classify_root(ov, st, state);
210
211 int cur = openat(ov->upper_fd, ".", O_RDONLY | O_DIRECTORY | O_CLOEXEC);
212 if (cur < 0) return -errno;
213
214 int opaque = 0;
215 const char *p = rel;
216 char name[OV_NAME_MAX], wh[OV_NAME_MAX + sizeof(BRIX_OV_WH_PREFIX)];
217
218 for (;;) {
219 int more = ov_split(&p, name, sizeof(name));
220 if (more < 0) { close(cur); return more; }
221
222 ov_wh_name(name, wh, sizeof(wh));
223 if (ov_marker_at(cur, wh)) { /* whiteouted at this level */
224 close(cur);
225 *state = BRIX_OV_MASKED;
226 return 0;
227 }
228
229 struct stat cst;
230 if (fstatat(cur, name, &cst, AT_SYMLINK_NOFOLLOW) != 0) {
231 close(cur); /* not in upper from here on */
232 *state = opaque ? BRIX_OV_MASKED : BRIX_OV_NONE;
233 return 0;
234 }
235
236 if (more == 0) { /* leaf present in upper */
237 close(cur);
238 if (st != NULL) *st = cst;
239 *state = BRIX_OV_UPPER;
240 return 0;
241 }
242
243 if (!S_ISDIR(cst.st_mode)) { /* upper non-dir shadows all below */
244 close(cur);
245 *state = BRIX_OV_MASKED;
246 return 0;
247 }
248
249 cur = ov_descend_child(cur, name, &opaque);
250 if (cur < 0) return cur;
251 }
252}

256int brix_overlay_whiteout(const brix_overlay *ov, const char *rel) {
257 char leaf[OV_NAME_MAX], wh[OV_NAME_MAX + sizeof(BRIX_OV_WH_PREFIX)];
258 int parent = ov_walk_parent(ov, rel, leaf, sizeof(leaf));
259 if (parent == -ENOENT) return 0; /* no upper parent → no marker */
260 if (parent < 0) return parent;
261 ov_wh_name(leaf, wh, sizeof(wh));
262 int present = ov_marker_at(parent, wh);
263 close(parent);
264 return present;

293int brix_overlay_mkdirs(const brix_overlay *ov, const char *rel_dir,
294 mode_t (*mode_fn)(void *ud, const char *rel_dir),
295 void *ud) {
296 if (rel_dir[0] == '\0') return 0; /* the upper root exists */
297
298 char leaf[OV_NAME_MAX];
299 int parent = ov_walk_parent_mk(ov, rel_dir, leaf, sizeof(leaf), 1, mode_fn, ud);
300 if (parent < 0) return parent;
301
302 mode_t m = mode_fn ? mode_fn(ud, rel_dir) : 0755;
303 int rc = 0;
304 if (mkdirat(parent, leaf, m) != 0 && errno != EEXIST) rc = -errno;
305
306 struct stat st; /* an existing non-dir leaf is an error */
307 if (rc == 0 && fstatat(parent, leaf, &st, AT_SYMLINK_NOFOLLOW) == 0
308 && !S_ISDIR(st.st_mode))
309 rc = -ENOTDIR;
310 close(parent);
311 return rc;
312}

370/* Unlink every overlay marker inside dirfd; a real entry → -ENOTEMPTY. */
371static int ov_clear_markers(int dirfd) {
372 int lfd = openat(dirfd, ".", O_RDONLY | O_DIRECTORY | O_CLOEXEC);
373 if (lfd < 0) return -errno;
374 DIR *d = fdopendir(lfd); /* owns lfd from here */
375 if (d == NULL) { int e = errno; close(lfd); return -e; }
376
377 int rc = 0;
378 struct dirent *e;
379 while (rc == 0 && (e = readdir(d)) != NULL) {
380 if (strcmp(e->d_name, ".") == 0 || strcmp(e->d_name, "..") == 0) continue;
381 if (!brix_ov_name_reserved(e->d_name)) { rc = -ENOTEMPTY; break; }
382 if (unlinkat(dirfd, e->d_name, 0) != 0) { rc = -errno; break; }
383 }
384 closedir(d);
385 return rc;
386}
