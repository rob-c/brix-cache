13/* pgwrite_retry_spans_multiple_pages()
14 * A kXR_pgRetry resend must correct exactly one page. Mirrors stock
15 * do_PgWIORetry: an unaligned offset may carry at most (bytes-to-boundary + one
16 * CRC); an aligned offset at most one page unit (kXR_pgUnitSZ = 4100). Returns
17 * 1 if the wire payload `dlen` is too large to be a single-page retry. */
18static int
19pgwrite_retry_spans_multiple_pages(int64_t offset, size_t dlen)
20{
21 size_t in_page = (size_t) (offset & (int64_t) (kXR_pgPageSZ - 1));
22
23 if (in_page != 0) {
24 size_t to_boundary = (size_t) kXR_pgPageSZ - in_page;
25 return dlen > to_boundary + XRD_PGWRITE_CKSZ;
26 }
27 return dlen > (size_t) kXR_pgUnitSZ;
28}

30/* Payload decoding helper
31 * Decodes a kXR_pgwrite payload into a flat pwrite() buffer.
32 *
33 * Payload layout is CRC first:
34 * [XRD_PGWRITE_CKSZ=4 bytes CRC32c][up to XRD_PGWRITE_PAGESZ=4096 bytes data]
35 * repeated for each page fragment. The first and last fragments may be shorter
36 * when the write offset is unaligned or the request ends mid-page.
37 *
38 * Returns NGX_OK on success, NGX_DECLINED on checksum mismatch, and NGX_ERROR
39 * for malformed framing.
40 *
41 * Uses brix_crc32c_copy() to fuse the CRC verification and data copy into a
42 * single pass over each page, avoiding the double-read that separate
43 * brix_crc32c() + ngx_memcpy() calls would incur.
44 */
45ngx_int_t
46brix_pgwrite_decode_payload(const u_char *payload, size_t payload_len,
47 int64_t offset, u_char *flat, size_t *flat_len, int64_t *bad_offset)
48{
49 ssize_t n;
50
51 if (flat_len == NULL || bad_offset == NULL || offset < 0) {
52 return NGX_ERROR;
53 }
54
55 *flat_len = 0;
56 *bad_offset = offset;
57
58 if (payload == NULL || flat == NULL || payload_len <= XRD_PGWRITE_CKSZ) {
59 return NGX_ERROR;
60 }
61
62 /* Shared single-pass decode (libxrdproto): verify each page's CRC32c while
63 * copying into the flat pwrite buffer — byte-identical to what the native
64 * client encodes. -1 = a page's CRC mismatched (→ kXR_ChkSumErr); -2 =
65 * malformed framing / offset overflow. dstcap = payload_len is a safe upper
66 * bound (decoded data is always < payload by at least one CRC). */
67 n = xrdp_pg_decode(payload, payload_len, offset, flat, payload_len,
68 bad_offset);
69 if (n == -1) {
70 return NGX_DECLINED;
71 }
72 if (n < 0) {
73 return NGX_ERROR;
74 }
75 *flat_len = (size_t) n;
76 return NGX_OK;
77}

79/* pgwrite_parse_validate()
80 * WHAT: Unpack the kXR_pgwrite request header, validate the file handle and
81 * payload bounds, and classify a kXR_pgRetry correction.
82 * WHY: Rejects malformed/oversized requests up front (kXR_ArgInvalid) and
83 * downgrades a stray/forged/recovery retry to a normal write so it can
84 * never corrupt the per-handle Fob (matches stock do_PgWIORetry).
85 * HOW: Fills *st from ctx->recv, then: (1) handle check via the shared
86 * brix_validate_write_handle helper; (2) offset/dlen bounds; (3) if the
87 * retry flag is set, reject a multi-page span and clear the flag when the
88 * target page is not registered as in-error. Returns 1 with *rc set when
89 * the caller must return immediately; 0 to continue. */
90static ngx_flag_t
91pgwrite_parse_validate(brix_ctx_t *ctx, ngx_connection_t *c, pgw_state_t *st,
92 ngx_int_t *rc)
93{
94 char write_detail[64];
95
96 xrdw_pgwrite_req_unpack(
97 ((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &st->req);
98 st->idx = (int)(unsigned char) st->req.fhandle[0];
99 st->offset = st->req.offset;
100 st->dlen = ctx->recv.cur_dlen;
101 st->payload = ctx->recv.payload;
102 st->is_retry = (st->req.reqflags & kXR_pgRetry) != 0;
103
104 if (!brix_validate_write_handle(ctx, c, st->idx, "WRITE",
105 BRIX_OP_WRITE, rc)) {
106 return 1;
107 }
108
109 if (st->offset < 0 || st->dlen <= XRD_PGWRITE_CKSZ) {
110 pgw_fmt_detail(write_detail, sizeof(write_detail), st->offset,
111 st->dlen);
112 brix_log_access(ctx, c, "WRITE", ctx->files[st->idx].path,
113 write_detail, 0, kXR_ArgInvalid,
114 "invalid pgwrite payload", 0);
115 BRIX_OP_ERR(ctx, BRIX_OP_WRITE);
116 *rc = brix_send_error(ctx, c, kXR_ArgInvalid, "invalid pgwrite payload");
117 return 1;
118 }
119
120 /* kXR_pgRetry correction request: a retry must correct exactly one
121 * registered page. Too large → reject. Not registered (stray/forged
122 * retry, or a resend during write recovery) → drop the retry flag and
123 * treat as a normal write so it cannot corrupt the Fob. */
124 if (st->is_retry) {
125 size_t page_dlen;
126
127 if (pgwrite_retry_spans_multiple_pages(st->offset, st->dlen)) {
128 pgw_fmt_detail(write_detail, sizeof(write_detail), st->offset,
129 st->dlen);
130 BRIX_RETURN_ERR(ctx, c, BRIX_OP_WRITE, "WRITE",
131 ctx->files[st->idx].path, write_detail,
132 kXR_ArgInvalid,
133 "pgwrite retry of more than one page not allowed");
134 }
135 page_dlen = st->dlen - XRD_PGWRITE_CKSZ;
136 if (!brix_pgw_fob_has(&ctx->files[st->idx], st->offset,
137 (uint32_t) page_dlen)) {
138 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
139 "brix: pgwrite retry %zu@%lld not in error — normal write",
140 page_dlen, (long long) st->offset);
141 st->is_retry = 0;
142 }
143 }
144
145 return 0;
146}

148/* pgwrite_register_bad_pages()
149 * WHAT: Add every freshly-detected corrupt page to the per-handle Fob.
150 * WHY: The bad bytes are still written (accept-then-correct); the Fob is what
151 * blocks a clean kXR_close until every page is corrected. Per-file
152 * capacity overflow → kXR_TooManyErrs (matches stock addOffs).
153 * HOW: Opens the Fob then adds each collected offset; on overflow sets *rc to
154 * the error reply and returns 1 (caller returns immediately). Returns 0
155 * when all pages registered. */
156static ngx_flag_t
157pgwrite_register_bad_pages(brix_ctx_t *ctx, ngx_connection_t *c,
158 pgw_state_t *st, ngx_int_t *rc)
159{
160 char write_detail[64];
161 size_t bi;
162
163 brix_pgw_fob_open(&ctx->files[st->idx]);
164 for (bi = 0; bi < st->bad_count; bi++) {
165 if (!brix_pgw_fob_add(&ctx->files[st->idx], st->bad_pages[bi].off,
166 st->bad_pages[bi].dlen)) {
167 pgw_fmt_detail(write_detail, sizeof(write_detail), st->offset,
168 st->dlen);
169 brix_log_access(ctx, c, "WRITE", ctx->files[st->idx].path,
170 write_detail, 0, kXR_TooManyErrs,
171 "too many uncorrected checksum errors in file", 0);
172 BRIX_OP_ERR(ctx, BRIX_OP_WRITE);
173 *rc = brix_send_error(ctx, c, kXR_TooManyErrs,
174 "too many uncorrected checksum errors in file");
175 return 1;
176 }
177 }
178 return 0;
179}

181/* pgwrite_decode_collect()
182 * WHAT: Decode+CRC-verify the whole payload into the reusable flat buffer,
183 * collect any CRC failures, short-circuit a clean recovery replay, and
184 * register newly-corrupt pages in the Fob.
185 * WHY: Implements the stock CSE (checksum-error) machine: verify EVERY page and
186 * copy them all (good and bad) into flat, so a single corrupt page can be
187 * re-sent without restarting the transfer. INVARIANT 1 (per-page CRC32c)
188 * is byte-frozen inside xrdp_pg_decode_collect.
189 * HOW: Borrows the per-session write_scratch buffer, runs the shared collect
190 * decode (dn = -3 → kXR_TooManyErrs, dn < 0 → kXR_ArgInvalid), then only
191 * for a non-retry with zero bad pages checks the recovery journal for a
192 * replay (→ SUCCESS pgwrite_status short-circuit). Bad pages are added to
193 * the Fob. Returns 1 with *rc set when the caller must return; 0 to
194 * proceed to the write. */
195static ngx_flag_t
196pgwrite_decode_collect(brix_ctx_t *ctx, ngx_connection_t *c, pgw_state_t *st,
197 ngx_int_t *rc)
198{
199 char write_detail[64];
200 ssize_t dn;
201
202 /* Reusable per-session buffer: avoids a malloc/free per request. */
203 st->flat = BRIX_GET_SCRATCH(ctx, c, rd.write_scratch,
204 rd.write_scratch_size, st->dlen);
205 st->flat_sz = 0;
206 st->bad_count = 0;
207
208 if (st->flat == NULL) {
209 *rc = NGX_ERROR;
210 return 1;
211 }
212
213 /* CSE retransmit: verify ALL pages, copy them all into flat
214 * (accept-then-correct), and collect the failures. A non-empty list yields
215 * a SUCCESS kXR_status reply carrying the bad page offsets — NOT an error. */
216 dn = xrdp_pg_decode_collect(st->payload, st->dlen, st->offset, st->flat,
217 st->dlen, st->bad_pages, kXR_pgMaxEpr,
218 &st->bad_count);
219 if (dn == -3) {
220 pgw_fmt_detail(write_detail, sizeof(write_detail), st->offset, st->dlen);
221 BRIX_RETURN_ERR(ctx, c, BRIX_OP_WRITE, "WRITE",
222 ctx->files[st->idx].path, write_detail,
223 kXR_TooManyErrs,
224 "too many checksum errors in pgwrite request");
225 }
226 if (dn < 0) {
227 pgw_fmt_detail(write_detail, sizeof(write_detail), st->offset, st->dlen);
228 BRIX_RETURN_ERR(ctx, c, BRIX_OP_WRITE, "WRITE",
229 ctx->files[st->idx].path, write_detail,
230 kXR_ArgInvalid, "invalid pgwrite payload");
231 }
232 st->flat_sz = (size_t) dn;
233
234 /* kXR_recoverWrts replay detection (post-decode): only short-circuit a clean
235 * replay; a request with bad pages must take the normal accept-then-correct
236 * path so the CSE list is emitted. A kXR_pgRetry correction deliberately
237 * re-writes a range already in the journal, so it must NOT be mistaken for a
238 * replay (that would skip the Fob correction and leave the close gate stuck). */
239 if (!st->is_retry && st->bad_count == 0 &&
240 ctx->files[st->idx].wrts_enabled &&
241 brix_wrts_is_replay(&ctx->files[st->idx], st->offset,
242 (uint32_t) st->flat_sz))
243 {
244 ngx_log_debug2(NGX_LOG_DEBUG_STREAM, c->log, 0,
245 "brix: pgwrite recovery replay skip offset=%lld len=%zu",
246 (long long) st->offset, st->flat_sz);
247 BRIX_OP_OK(ctx, BRIX_OP_WRITE);
248 /* info offset = request offset (stock parity), not offset+len. */
249 *rc = brix_send_pgwrite_status(ctx, c, st->offset);
250 return 1;
251 }
252
253 /* Register newly-corrupt pages in the Fob (close-time gate). */
254 if (st->bad_count > 0) {
255 if (pgwrite_register_bad_pages(ctx, c, st, rc)) {
256 return 1;
257 }
258 }
259
260 return 0;
261}

263/* pgwrite_execute_sync()
264 * WHAT: Write the decoded flat buffer in one syscall, update accounting and
265 * dirty/journal state, clear a corrected retry from the Fob, and send the
266 * mandatory kXR_status reply (CSE list when pages failed CRC32c).
267 * WHY: This is the synchronous path taken when no AIO task was posted (or for a
268 * single-page retry, which commits its Fob correction in-line right after
269 * the write rather than threading retry state through the AIO callback).
270 * HOW: Runs the write through the VFS core, maps a short/failed write to
271 * kXR_IOError, charges bandwidth + write-through dirty tracking, records
272 * the recovery-journal entry, and replies with brix_send_pgwrite_cse when
273 * bad_count > 0 else brix_send_pgwrite_status (info = request offset, stock
274 * parity). Always sets *rc and returns 1 (terminal stage). */
275static ngx_flag_t
276pgwrite_execute_sync(brix_ctx_t *ctx, ngx_connection_t *c, pgw_state_t *st,
277 ngx_int_t *rc)
278{
279 size_t total_written;
280 ssize_t nw;
281 brix_vfs_job_t job;
282
283 /* Synchronous path: write the entire flat buffer in one syscall. */
284 brix_vfs_job_write_init(&job, ctx->files[st->idx].fd,
285 (off_t) st->offset, st->flat, st->flat_sz);
286 job.csi = ctx->files[st->idx].csi; /* phase-59 W2: update page tags */
287 brix_vfs_job_set_obj(&job, &ctx->files[st->idx].sd_obj);
288 brix_vfs_io_execute(&job);
289 nw = job.nio;
290 if (job.io_errno != 0) {
291 errno = job.io_errno;
292 }
293
294 /* Error handling: I/O error or short write */
295 if (nw < 0) {
296 const char *ioerr = strerror(errno);
297 return pgwrite_handle_write_error(ctx, c, st, ioerr, rc);
298 }
299 if ((size_t) nw < st->flat_sz) {
300 return pgwrite_handle_short_write(ctx, c, st, (size_t) nw, rc);
301 }
302
303 total_written = (size_t) nw;
304
305 /* Success path: delegate to focused helpers */
306 pgwrite_update_metrics(ctx, st->idx, total_written);
307 pgwrite_mark_dirty_if_needed(ctx, st->idx, st->offset,
308 total_written, ctx->files[st->idx].wt_enabled);
309 pgwrite_log_success(ctx, c, st, total_written);
310 BRIX_OP_OK(ctx, BRIX_OP_WRITE);
311 pgwrite_cleanup_retry_fob(ctx, st->idx, st->is_retry, st->bad_count,
312 st->offset, st->flat_sz);
313 pgwrite_record_journal(ctx, st->idx, st->offset, total_written,
314 ctx->files[st->idx].wrts_enabled);
315
316 *rc = pgwrite_send_reply(ctx, c, st);
317 return 1;
318}

320/* pgwrite handler section
321 * WHAT: Handle kXR_pgwrite, the page-mode write with a 4-byte CRC32c before
322 * each ≤4 KiB page. Implements the stock CSE (checksum-error) retransmit
323 * machine: verify EVERY page (xrdp_pg_decode_collect), write ALL pages —
324 * good and bad — to disk, and on any failure reply with a SUCCESS
325 * kXR_status frame carrying the pgWrCSE retransmit list
326 * (brix_send_pgwrite_cse) rather than a hard error. Each bad page is
327 * registered in the per-handle Fob (pgw_fob.c).
328 *
329 * WHY: Accept-then-correct lets a single corrupt page be re-sent without
330 * restarting a multi-GB transfer. The kXR_close gate (read/close.c)
331 * returns kXR_ChkSumErr while the Fob is non-empty, so a committed file
332 * never retains known-corrupt bytes despite the bad bytes being written.
333 *
334 * HOW: A flat, early-return orchestrator over the parse-validate / decode-
335 * collect / execute-sync helper sequence. A kXR_pgRetry resend is
336 * single-page-bounded, must target a registered offset (else downgraded to
337 * a normal write), takes the synchronous path, and clears its Fob entry on
338 * a clean re-verify. Non-retry writes try the AIO thread pool first (the
339 * async completion sends the reply); on no-pool fallback they take the
340 * synchronous path.
341 */
342ngx_int_t
343brix_handle_pgwrite(brix_ctx_t *ctx, ngx_connection_t *c)
344{
345 pgw_state_t st = { 0 };
346 ngx_int_t rc = NGX_OK;
347
348 st.rconf = ngx_stream_get_module_srv_conf(
349 (ngx_stream_session_t *) c->data, ngx_stream_brix_module);
350
351 if (pgwrite_parse_validate(ctx, c, &st, &rc)) {
352 return rc;
353 }
354
355 if (pgwrite_decode_collect(ctx, c, &st, &rc)) {
356 return rc;
357 }
358
359 /* oss.maxsize (§3.9): cap on the decoded plaintext's end offset. Applied
360 * after decode so the length is the real byte count, before either the
361 * staged-append or random-write path commits anything. */
362 if (brix_write_within_maxsize(ctx, c, st.rconf, st.idx, st.offset,
363 st.flat_sz, BRIX_OP_WRITE, "WRITE", &rc)) {
364 return rc;
365 }
366
367 /* Whole-object staged-commit adapter (phase-70): a staged write handle has no
368 * random-write fd — append the decoded plaintext to the staged handle (which
369 * enforces sequential offsets) and reply with the mandatory kXR_status frame.
370 *
371 * A staged handle is sequential append-only, so the kXR_pgwrite accept-then-
372 * correct (CSE) protocol is impossible here: a page that fails CRC32c can
373 * never be re-sent to its (now-passed) offset. pgwrite_decode_collect has
374 * already registered any bad page in the Fob (so close/sync stay fail-closed
375 * — INVARIANT 1), but we must NOT append the corrupt bytes and answer a clean
376 * status: the client would never learn to resend and would only discover the
377 * corruption as a mystifying kXR_ChkSumErr at close. Fail the write now with
378 * an actionable checksum error so the client restarts the upload. */
379 if (ctx->files[st.idx].writer != NULL) {
380 char write_detail[64];
381
382 if (st.bad_count > 0) {
383 pgw_fmt_detail(write_detail, sizeof(write_detail), st.offset,
384 st.dlen);
385 BRIX_RETURN_ERR(ctx, c, BRIX_OP_WRITE, "WRITE",
386 ctx->files[st.idx].path, write_detail,
387 kXR_ChkSumErr,
388 "pgwrite checksum error on staged handle");
389 }
390 if (brix_staged_append(ctx, c, st.idx, st.offset, st.flat,
391 st.flat_sz, &rc) != NGX_OK) {
392 return rc; /* error already sent (sequential-append / IO) */
393 }
394 BRIX_OP_OK(ctx, BRIX_OP_WRITE);
395 return brix_send_pgwrite_status(ctx, c, st.offset);
396 }
397
398 /* Retries are single-page (≤ 4 KiB): take the synchronous path so the Fob
399 * correction (brix_pgw_fob_del) commits in-line right after the write,
400 * rather than threading retry state through the AIO callback. */
401 if (!st.is_retry) {
402 ngx_flag_t posted = 0;
403
404 /* Pass NULL as payload_to_free: flat is rd.write_scratch (pool-managed,
405 * not heap-allocated) and must not be freed by the done handler. The CSE
406 * bad-page list (if any) rides along so the async completion sends the
407 * retransmit frame. */
408 rc = brix_try_post_write_aio(ctx, c, st.idx, (off_t) st.offset, st.flat,
409 st.flat_sz, st.offset, 1, NULL,
410 st.bad_count ? st.bad_pages : NULL,
411 st.bad_count,
412 "brix: thread_task_post failed, falling back to sync pgwrite",
413 &posted);
414 if (rc != NGX_OK) {
415 return rc;
416 }
417 if (posted) {
418 /* Async completion sends the mandatory kXR_status pgwrite reply. */
419 return NGX_OK;
420 }
421 }
422
423 (void) pgwrite_execute_sync(ctx, c, &st, &rc);
424 return rc;
425}
