1/* File: origin_id.c — the tpc.org identity of a TPC destination's client
2 *
3 * WHAT: Builds "user.pid@host" — the string the XRootD source pairs against
4 * its rendezvous grant (XrdOfsTPC::genOrg → XrdNetAddr::Name, a raw strcmp in
5 * XrdOfsTPCInfo::Match). The host is the client's PTR name, else the
6 * bracketed numeric literal XrdNetAddr prints for a name-less peer. Two
7 * halves: brix_tpc_origin_build() on the event loop at kXR_open (cached PTR
8 * or the numeric fallback, with a PTR fill started), and
9 * brix_tpc_origin_thread_fill() on the pull thread (blocking PTR through the
10 * DNS driver when the loop-side build was still pending).
11 *
12 * WHY: I-DNS-1 — the old build called getnameinfo(NI_NAMEREQD) on the event
13 * loop for every TPC open: a worker-wide stall for the resolver's full timeout
14 * budget whenever the client's reverse zone was slow or absent. The name still
15 * has to be right, or the source never redeems the grant and the pull-open
16 * hangs on kXR_waitresp until it TTL-expires, so the thread finishes what the
17 * loop could not wait for.
18 *
19 * HOW: Every XRootD server is IPv6 dual-stack and sees an IPv4 client as the
20 * IPv4-MAPPED address ::ffff:a.b.c.d, reverse-resolves THAT, and falls back to
21 * its bracketed literal ("[::ffff:127.0.0.1]" for a non-DNS loopback client).
22 * So both halves of this file key on the mapped form (tpc_origin_lookup_key):
23 * the numeric fallback prints it with inet_ntop, and the PTR lookup asks about
24 * it. The two spellings are NOT interchangeable, which is the whole reason
25 * this file exists — measured on one ordinary host:
26 *
27 * getnameinfo(NI_NAMEREQD) 127.0.0.1 -> "localhost"
28 * ::ffff:127.0.0.1 -> EAI_NONAME
29 * 129.215.213.101 -> "ce4.gridpp.ecdf.ed.ac.uk"
30 * ::ffff:129.215… -> "xrd1.edi.scotgrid.ac.uk"
31 *
32 * and a stock source, asked for the same two peers, wrote exactly the mapped
33 * column into its grant. Asking the plain question therefore yields a name
34 * that is perfectly valid, never equal, and impossible to spot in a log: the
35 * source pairs tpc.org with a raw strcmp, so the grant is simply never
36 * redeemed and the pull-open hangs on kXR_waitresp until the TTL expires.
37 * */

60/* "user.pid@" then the host, truncated rather than dropped (tpc_org is 256
61 * bytes; a 255-byte PTR name would not fit behind the prefix). */
62static void
63tpc_origin_format(const char *user, uint32_t pid, const char *host,
64 char *dst, size_t dst_size)
65{
66 int prefix_len;
67
68 prefix_len = snprintf(dst, dst_size, "%s.%u@", user, (unsigned) pid);
69 if (prefix_len < 0 || (size_t) prefix_len >= dst_size) {
70 dst[0] = '\0';
71 return;
72 }
73 ngx_cpystrn((u_char *) dst + prefix_len, (u_char *) host,
74 dst_size - (size_t) prefix_len);
75}

78/* The address XrdNetAddr holds for this peer, and therefore the one both the
79 * PTR question and the numeric literal must be asked about: an IPv4 peer in
80 * IPv4-MAPPED form, anything else verbatim. NGX_DECLINED when there is no
81 * peer to key on. */
82static ngx_int_t
83tpc_origin_lookup_key(const struct sockaddr *sa, socklen_t len,
84 struct sockaddr_storage *out, socklen_t *out_len)
85{
86 struct sockaddr_in6 *v6;
87
88 if (sa == NULL) {
89 return NGX_DECLINED;
90 }
91 if (sa->sa_family != AF_INET) {
92 if (len == 0 || (size_t) len > sizeof(*out)) {
93 return NGX_DECLINED;
94 }
95 ngx_memcpy(out, sa, len);
96 *out_len = len;
97 return NGX_OK;
98 }
99
100 v6 = (struct sockaddr_in6 *) out;
101 ngx_memzero(v6, sizeof(*v6));
102 v6->sin6_family = AF_INET6;
103 v6->sin6_addr.s6_addr[10] = 0xff;
104 v6->sin6_addr.s6_addr[11] = 0xff;
105 ngx_memcpy(&v6->sin6_addr.s6_addr[12],
106 &((const struct sockaddr_in *) sa)->sin_addr, 4);
107 *out_len = (socklen_t) sizeof(*v6);
108 return NGX_OK;
109}

112/* The host XrdNetAddr::Name prints for a peer without a PTR: the bracketed
113 * literal of the IPv6 (or IPv4-mapped) address, else the accept-time address
114 * text, else "unknown". */
115static void
116tpc_origin_numeric_host(const struct sockaddr *sa, socklen_t len,
117 const ngx_str_t *addr_text, char *host, size_t sz)
118{
119 struct sockaddr_storage key;
120 socklen_t key_len = 0;
121 char numeric[INET6_ADDRSTRLEN];
122
123 host[0] = '\0';
124
125 if (tpc_origin_lookup_key(sa, len, &key, &key_len) == NGX_OK
126 && key.ss_family == AF_INET6)
127 {
128 const struct sockaddr_in6 *s6 = (const struct sockaddr_in6 *) &key;
129
130 if (inet_ntop(AF_INET6, &s6->sin6_addr, numeric, sizeof(numeric))
131 != NULL)
132 {
133 (void) snprintf(host, sz, "[%s]", numeric);
134 }
135 }
136 if (host[0] == '\0' && addr_text != NULL && addr_text->len > 0) {
137 ngx_cpystrn((u_char *) host, addr_text->data,
138 ngx_min(addr_text->len + 1, sz));
139 }
140 if (host[0] == '\0') {
141 ngx_cpystrn((u_char *) host, (u_char *) "unknown", sz);
142 }
143}

146unsigned
147brix_tpc_origin_build(brix_ctx_t *ctx, ngx_connection_t *c,
148 const brix_dns_policy_t *policy, char *dst, size_t dst_size)
149{
150 struct sockaddr_storage key;
151 socklen_t key_len = 0;
152 char host[BRIX_DNS_REVERSE_NAME_LEN];
153 char user[sizeof(ctx->login.user)];
154 uint32_t pid;
155 ngx_int_t rc = NGX_DECLINED;
156 unsigned pending = 0;
157
158 tpc_origin_identity(ctx, user, sizeof(user), &pid);
159
160 if (tpc_origin_lookup_key(c->sockaddr, c->socklen, &key, &key_len)
161 == NGX_OK)
162 {
163 rc = brix_dns_reverse_cached((const struct sockaddr *) &key, key_len,
164 host, sizeof(host));
165 if (rc == NGX_AGAIN) {
166 brix_dns_reverse_prefetch(policy, (const struct sockaddr *) &key,
167 key_len);
168 pending = 1;
169 }
170 }
171 if (rc != NGX_OK) {
172 tpc_origin_numeric_host(c->sockaddr, c->socklen, &c->addr_text, host,
173 sizeof(host));
174 }
175 tpc_origin_format(user, pid, host, dst, dst_size);
176 return pending;
177}

195{

198 char name[BRIX_DNS_REVERSE_NAME_LEN];
199

209 {
210 return; /* keep the numeric literal the loop built */

212 if (brix_dns_reverse_sync(t->conf->common.dns.policy,
213 (const struct sockaddr *) &key, key_len,

215 {
216 return; /* keep the numeric literal the loop built */
217 }
218 tpc_origin_format(t->org_user, t->org_pid, name, t->tpc_org,
