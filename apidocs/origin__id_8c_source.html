1/* File: origin_id.c — the tpc.org identity of a TPC destination's client
2 *
3 * WHAT: Builds "user.pid@host" — the string the XRootD source pairs against
4 * its rendezvous grant (XrdOfsTPC::genOrg → XrdNetAddr::Name, a raw strcmp in
5 * XrdOfsTPCInfo::Match). The host is the client's PTR name, else the
6 * bracketed numeric literal XrdNetAddr prints for a name-less peer. Two
7 * halves: brix_tpc_origin_build() on the event loop at kXR_open (cached PTR
8 * or the numeric fallback, with a PTR fill started), and
9 * brix_tpc_origin_thread_fill() on the pull thread (blocking PTR through the
10 * DNS driver when the loop-side build was still pending).
11 *
12 * WHY: I-DNS-1 — the old build called getnameinfo(NI_NAMEREQD) on the event
13 * loop for every TPC open: a worker-wide stall for the resolver's full timeout
14 * budget whenever the client's reverse zone was slow or absent. The name still
15 * has to be right, or the source never redeems the grant and the pull-open
16 * hangs on kXR_waitresp until it TTL-expires, so the thread finishes what the
17 * loop could not wait for.
18 *
19 * HOW: Every XRootD server is IPv6 dual-stack and sees an IPv4 client as the
20 * IPv4-MAPPED address ::ffff:a.b.c.d, reverse-resolves that, and falls back to
21 * its bracketed literal ("[::ffff:127.0.0.1]" for a non-DNS loopback client).
22 * The numeric fallback therefore maps IPv4 into that form and prints it with
23 * inet_ntop (byte-identical to XrdNetAddr), while the PTR lookup keys on the
24 * peer address as-is: glibc asks the same in-addr.arpa question for the mapped
25 * form, and the plain key shares the cache entry with the host-rule and
26 * authorization lookups.
27 * */

50/* "user.pid@" then the host, truncated rather than dropped (tpc_org is 256
51 * bytes; a 255-byte PTR name would not fit behind the prefix). */
52static void
53tpc_origin_format(const char *user, uint32_t pid, const char *host,
54 char *dst, size_t dst_size)
55{
56 int prefix_len;
57
58 prefix_len = snprintf(dst, dst_size, "%s.%u@", user, (unsigned) pid);
59 if (prefix_len < 0 || (size_t) prefix_len >= dst_size) {
60 dst[0] = '\0';
61 return;
62 }
63 ngx_cpystrn((u_char *) dst + prefix_len, (u_char *) host,
64 dst_size - (size_t) prefix_len);
65}

68/* The host XrdNetAddr::Name prints for a peer without a PTR: the bracketed
69 * literal of the IPv6 (or IPv4-mapped) address, else the accept-time address
70 * text, else "unknown". */
71static void
72tpc_origin_numeric_host(const struct sockaddr *sa, const ngx_str_t *addr_text,
73 char *host, size_t sz)
74{
75 struct sockaddr_in6 v6;
76 char numeric[INET6_ADDRSTRLEN];
77
78 host[0] = '\0';
79
80 if (sa != NULL && sa->sa_family == AF_INET) {
81 const struct sockaddr_in *s4 = (const struct sockaddr_in *) sa;
82
83 ngx_memzero(&v6, sizeof(v6));
84 v6.sin6_family = AF_INET6;
85 v6.sin6_addr.s6_addr[10] = 0xff;
86 v6.sin6_addr.s6_addr[11] = 0xff;
87 ngx_memcpy(&v6.sin6_addr.s6_addr[12], &s4->sin_addr, 4);
88 sa = (const struct sockaddr *) &v6;
89 }
90 if (sa != NULL && sa->sa_family == AF_INET6) {
91 const struct sockaddr_in6 *s6 = (const struct sockaddr_in6 *) sa;
92
93 if (inet_ntop(AF_INET6, &s6->sin6_addr, numeric, sizeof(numeric))
94 != NULL)
95 {
96 (void) snprintf(host, sz, "[%s]", numeric);
97 }
98 }
99 if (host[0] == '\0' && addr_text != NULL && addr_text->len > 0) {
100 ngx_cpystrn((u_char *) host, addr_text->data,
101 ngx_min(addr_text->len + 1, sz));
102 }
103 if (host[0] == '\0') {
104 ngx_cpystrn((u_char *) host, (u_char *) "unknown", sz);
105 }
106}

109unsigned
110brix_tpc_origin_build(brix_ctx_t *ctx, ngx_connection_t *c,
111 const brix_dns_policy_t *policy, char *dst, size_t dst_size)
112{
113 char host[BRIX_DNS_REVERSE_NAME_LEN];
114 char user[sizeof(ctx->login.user)];
115 uint32_t pid;
116 ngx_int_t rc = NGX_DECLINED;
117 unsigned pending = 0;
118
119 tpc_origin_identity(ctx, user, sizeof(user), &pid);
120
121 if (c->sockaddr != NULL) {
122 rc = brix_dns_reverse_cached(c->sockaddr, c->socklen, host,
123 sizeof(host));
124 if (rc == NGX_AGAIN) {
125 brix_dns_reverse_prefetch(policy, c->sockaddr, c->socklen);
126 pending = 1;
127 }
128 }
129 if (rc != NGX_OK) {
130 tpc_origin_numeric_host(c->sockaddr, &c->addr_text, host,
131 sizeof(host));
132 }
133 tpc_origin_format(user, pid, host, dst, dst_size);
134 return pending;
135}
