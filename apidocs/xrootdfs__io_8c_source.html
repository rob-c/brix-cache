54/* Open helper shared by open()/create(): writable + force tri-state. */
55int
56afh_open(uid_t uid, const char *path, int writable, int force, struct fuse_file_info *fi)
57{
58 brix_status st;
59 afh *h = calloc(1, sizeof(*h));
60 if (h == NULL) {
61 return -ENOMEM;
62 }
63 pthread_mutex_init(&h->lock, NULL);
64 h->writable = writable;
65 brix_iobuf_init(&h->io, h, afh_io_pread, afh_io_pwrite, writable,
66 g_readahead, g_writeback);
67 brix_status_clear(&st);
68
69 /* HTTP(S)/WebDAV mounts are READ-ONLY (Range GET); reject write opens. */
70 if (g_web) {
71 if (writable) {
72 afh_free(h);
73 return -EROFS;
74 }
75 char pbuf[XRDC_PATH_MAX];
76 h->wf = brix_webfile_open(&g_weburl, srv_path(path, pbuf, sizeof(pbuf)),
77 g_bearer, g_web_verify, g_web_ca, g_web_proxy,
78 g_max_stall, NULL, &st);
79 if (h->wf == NULL) {
80 afh_free(h);
81 return xfs_err(&st);
82 }
83 fi->fh = (uint64_t) (uintptr_t) h;
84 return 0;
85 }
86
87 /* phase-42 W4/W5: request inline compression when configured — on read opens
88 * the server compresses responses (W4), on write opens it decompresses our
89 * payloads (W5). A server that doesn't support it returns plaintext (the
90 * mfile learns read_codec/write_codec == 0). */
91 char opq[48];
92 const char *opaque = NULL;
93 if (g_compress[0] != '\0') {
94 snprintf(opq, sizeof(opq), "xrootd.compress=%s", g_compress);
95 opaque = opq;
96 }
97 /* W7.2: the DATA plane carries the identity too. Mapping only metadata
98 * would be theatre — the bytes are what the caller is being kept away
99 * from, and kXR_open is where the server decides. */
100 brix_mgr *mgr;
101 int identrc = xfs_ident_get(uid, NULL, &mgr);
102 if (identrc != 0) {
103 afh_free(h);
104 return identrc;
105 }
106 char pbuf[XRDC_PATH_MAX];
107 h->mf = brix_mfile_open(brix_mgr_pick(mgr), srv_path(path, pbuf, sizeof(pbuf)),
108 writable, force, 0, opaque,
109 g_max_stall, g_max_retries, &st);
110 if (h->mf == NULL) {
111 afh_free(h);
112 return xfs_err(&st);
113 }
114 fi->fh = (uint64_t) (uintptr_t) h;
115 return 0;
116}

140/*
141 * xfs_mknod — create an empty file at `path` (FUSE mknod).
142 *
143 * The kernel routes the mknod(2)/mknodat(2) syscall here (and an open(O_CREAT)
144 * only when .create is absent, which it is not). A remote xrootd/WebDAV store
145 * holds regular files only, so a FIFO/device/socket type is refused with EPERM
146 * rather than silently created as a plain file. For a regular file we open a
147 * create-new write handle (force=0 → EEXIST if the path already exists, POSIX
148 * mknod semantics) and immediately flush-and-close it, so an empty file lands on
149 * the server with no lingering handle — mknod returns no open descriptor.
150 * Permission bits are ignored (as xfs_create does); a later chmod sets the mode.
151 */
152int
153xfs_mknod(uid_t uid, const char *path, mode_t mode, dev_t rdev)
154{
155 struct fuse_file_info fi;
156 int rc;
157
158 (void) rdev;
159
160 if ((mode & S_IFMT) != 0 && (mode & S_IFMT) != S_IFREG) {
161 return -EPERM;
162 }
163 if (g_web) {
164 return -EROFS; /* HTTP(S)/WebDAV mounts are read-only */
165 }
166
167 memset(&fi, 0, sizeof(fi));
168 rc = afh_open(uid, path, 1 /* writable */, 0 /* create-new: fail if exists */,
169 &fi);
170 if (rc != 0) {
171 return rc;
172 }
173 return xfs_release(path, &fi); /* flush (empty) + close = commit empty */
174}

177/* WHAT: Validate a short root read against its current open-handle size.
178 * WHY: Cached FUSE reads can pad a premature EOF with zeros, concealing broken
179 * peers as successful corrupt reads. Path stat could name a replacement inode.
180 * HOW: 1. Leave full reads/errors/web handles untouched. 2. Stat only the same
181 * authorized handle. 3. Admit genuine EOF; fail closed on contradictory size. */
182static ssize_t
183afh_check_short_read(afh *handle, ssize_t received, off_t offset,
184 size_t requested, brix_status *status)
185{
186 if (received < 0 || (size_t) received == requested || !handle->mf) {
187 return received;
188 }
189 brix_statinfo info;
190 if (brix_mfile_stat(handle->mf, &info, status) != 0) {
191 return -1;
192 }
193 if (info.size < 0 || (offset < info.size && received < info.size - offset)) {
194 brix_status_set(status, XRDC_EIO, EIO,
195 "premature EOF from remote file handle");
196 return -1;
197 }
198 return received;
199}

202/* WHAT: Return exact available bytes or negative errno for one FUSE read.
203 * WHY: Remote EOF must not become silent zero-filled data before the real EOF.
204 * HOW: 1. Validate handle/range. 2. Read under its existing I/O lock.
205 * 3. Verify short replies by handle before the kernel can cache them. */
206int
207xfs_read(const char *path, char *buf, size_t size, off_t offset,
208 struct fuse_file_info *fi)
209{
210 afh *h = (afh *) (uintptr_t) fi->fh;
211 brix_status st;
212 ssize_t r;
213
214 (void) path;
215 if (h == NULL) {
216 return -EBADF;
217 }
218 if (offset < 0 || size > INT32_MAX) {
219 return -EINVAL;
220 }
221 brix_status_clear(&st);
222 pthread_mutex_lock(&h->lock);
223 r = brix_iobuf_read(&h->io, offset, buf, size, &st);
224 r = afh_check_short_read(h, r, offset, size, &st);
225 pthread_mutex_unlock(&h->lock);
226 return r < 0 ? xfs_err(&st) : (int) r;
227}

230int
231xfs_write(const char *path, const char *buf, size_t size, off_t offset,
232 struct fuse_file_info *fi)
233{
234 if (g_web) return -EROFS;
235 afh *h = (afh *) (uintptr_t) fi->fh;
236 brix_status st;
237 int rc;
238
239 (void) path;
240 if (h == NULL || !h->writable) {
241 return -EBADF;
242 }
243 brix_status_clear(&st);
244 pthread_mutex_lock(&h->lock);
245 rc = brix_iobuf_write(&h->io, offset, buf, size, &st);
246 if (rc == 0 && (int64_t) offset + (int64_t) size > h->wsize) {
247 h->wsize = (int64_t) offset + (int64_t) size; /* track logical EOF */
248 }
249 pthread_mutex_unlock(&h->lock);
250 return rc != 0 ? xfs_err(&st) : (int) size;
251}

297{
