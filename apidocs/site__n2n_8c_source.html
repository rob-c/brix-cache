11/* One path segment carved out of an LFN: a pointer into the source string and
12 * its length (not NUL-terminated). */
13typedef struct {
14 const char *start;
15 size_t len;
16} n2n_seg_t;
17
18/* Advance *cursor to the next non-empty segment: skip the run of '/' separators,
19 * then capture the next run of non-'/' bytes. 1 when a segment was found (cursor
20 * advanced past it), 0 at end of string. This is the tokenizer that lets the
21 * canonicalizer read as a flat loop over already-carved segments. */
22static int
23n2n_next_seg(const char **cursor, n2n_seg_t *seg)
24{
25 const char *p = *cursor;
26
27 while (*p == '/') {
28 p++;
29 }
30 if (*p == '\0') {
31 *cursor = p;
32 return 0;
33 }
34 seg->start = p;
35 while (*p != '\0' && *p != '/') {
36 p++;
37 }
38 seg->len = (size_t) (p - seg->start);
39 *cursor = p;
40 return 1;
41}
42
43/* Append one ordinary segment as "/<seg>" to the canonical buffer at cursor `w`.
44 * Returns the new cursor, or (size_t)-1 when it would not fit within `cap`. */
45static size_t
46n2n_append(char *out, size_t cap, size_t w, const n2n_seg_t *seg)
47{
48 if (w + 1 + seg->len + 1 > cap) {
49 return (size_t) -1;
50 }
51 out[w++] = '/';
52 memcpy(out + w, seg->start, seg->len);
53 w += seg->len;
54 out[w] = '\0';
55 return w;
56}

70int
71brix_n2n_canonicalize(const char *lfn, char *out, size_t cap)
72{
73 const char *cursor = lfn;
74 n2n_seg_t seg = {0};
75 size_t w = 0;
76
77 if (lfn == NULL || out == NULL || cap < 2) {
78 errno = EINVAL;
79 return -1;
80 }
81 out[0] = '\0';
82
83 while (n2n_next_seg(&cursor, &seg)) {
84 if (n2n_seg_is_dot(&seg)) {
85 continue; /* "." — no-op */
86 }
87 if (n2n_seg_is_dotdot(&seg)) {
88 errno = EINVAL; /* ".." — rejected, never resolved (C13) */
89 return -1;
90 }
91 w = n2n_append(out, cap, w, &seg);
92 if (w == (size_t) -1) {
93 errno = ENAMETOOLONG;
94 return -1;
95 }
96 }
97
98 if (w == 0) { /* everything collapsed -> bare root */
99 out[0] = '/';
100 out[1] = '\0';
101 }
102 return 0;
103}

107n2n_copy(char *dst, size_t cap, const char *src)
108{
109 size_t n = strlen(src);
110
111 if (n + 1 > cap) {
112 return -1;
113 }
114 memcpy(dst, src, n + 1);
115 return 0;

118int
119brix_n2n_lfn2pfn(const brix_n2n_cfg_t *cfg, const char *lfn,
120 char *pfn, size_t cap)
121{
122 char canon[1024];
123 int r;
124
125 if (cfg == NULL || lfn == NULL || pfn == NULL || cap == 0) {
126 errno = EINVAL;
127 return -1;
128 }
129 /* Canonicalize before composing: folds "." and "//" so "/a/./b", "/a//b"
130 * and "/a/b" yield one physical name, and rejects ".." traversal. errno is
131 * set by the canonicalizer (EINVAL on "..", ENAMETOOLONG on overflow). */
132 if (brix_n2n_canonicalize(lfn, canon, sizeof(canon)) != 0) {
133 return -1;
134 }
135
136 switch (cfg->scheme) {
137 case BRIX_N2N_IDENTITY:
138 if (n2n_copy(pfn, cap, canon) != 0) {
139 errno = ENAMETOOLONG;
140 return -1;
141 }
142 return 0;
143
144 case BRIX_N2N_RAL:
145 r = snprintf(pfn, cap, "%s:%s%s", cfg->pool, cfg->prefix, canon);
146 break;
147
148 case BRIX_N2N_CEPHFS_PATH:
149 r = snprintf(pfn, cap, "%s%s", cfg->prefix, canon);
150 break;
151
152 default:
153 errno = EINVAL;
154 return -1;
155 }
156 if (r < 0 || (size_t) r >= cap) {
157 errno = ENAMETOOLONG;
158 return -1;
159 }
160 return 0;
161}

163int
164brix_n2n_pfn2lfn(const brix_n2n_cfg_t *cfg, const char *pfn,
165 char *lfn, size_t cap)
166{
167 const char *p;
168 size_t plen;
169
170 if (cfg == NULL || pfn == NULL || lfn == NULL || cap == 0) {
171 return -1;
172 }
173
174 switch (cfg->scheme) {
175 case BRIX_N2N_IDENTITY:
176 return n2n_copy(lfn, cap, pfn);
177
178 case BRIX_N2N_RAL:
179 p = strchr(pfn, ':');
180 if (p == NULL) {
181 return -1; /* not a "<pool>:…" name */
182 }
183 p++; /* past the colon */
184 plen = strlen(cfg->prefix);
185 if (plen > 0) {
186 if (strncmp(p, cfg->prefix, plen) != 0) {
187 return -1;
188 }
189 p += plen;
190 }
191 return n2n_copy(lfn, cap, p);
192
193 case BRIX_N2N_CEPHFS_PATH:
194 p = pfn;
195 plen = strlen(cfg->prefix);
196 if (plen > 0) {
197 if (strncmp(p, cfg->prefix, plen) != 0) {
198 return -1; /* not under the localroot */
199 }
200 p += plen;
201 }
202 return n2n_copy(lfn, cap, p);
203
204 default:
205 return -1;
206 }
207}

209int
210brix_n2n_extract_pool(const char *objname, char *pool, size_t cap,
211 const char **rest)
212{
213 const char *colon;
214 size_t n;
215
216 if (objname == NULL || pool == NULL || cap == 0) {
217 return -1;
218 }
219 colon = strchr(objname, ':');
220 if (colon == NULL) {
221 /* stock XrdCephOss::extractPool: no colon → whole string is the pool. */
222 if (n2n_copy(pool, cap, objname) != 0) {
223 return -1;
224 }
225 if (rest != NULL) {
226 *rest = objname + strlen(objname); /* "" */
227 }
228 return 0;
229 }
230 n = (size_t) (colon - objname);
231 if (n + 1 > cap) {
232 return -1;
233 }
234 memcpy(pool, objname, n);
235 pool[n] = '\0';
236 if (rest != NULL) {
237 *rest = colon + 1;
238 }
239 return 0;
240}
