1/*
2 * sd.h — Storage Driver (SD) interface: the pluggable layer below the VFS.
3 *
4 * WHAT: Declares the capability bitmap, the opaque driver/instance/object/dir/
5 * staged handle types, the POD stat/dirent descriptors, the driver vtable
6 * (brix_sd_driver_s), the small capability-gated accessor helpers, and
7 * the registry API that turns a backend name into a bound per-export
8 * instance. POSIX is the default driver (sd_posix.c); block/object drivers
9 * (phases 55.D/E) register the same way.
10 *
11 * WHY: The VFS (src/fs/) is the protocol-agnostic data plane, but it is still
12 * hard-wired to POSIX syscalls. This header is the seam that lets the VFS
13 * call "move these bytes / mutate this name" against a driver it selected
14 * at config time, while keeping all policy (confinement re-check, metrics,
15 * access log, cache, buffer shaping) above the seam. See
16 * docs/refactor/phase-55-storage-backend-abstraction.md.
17 *
18 * HOW: A driver is a static const brix_sd_driver_t with a caps bitmap and a
19 * flat table of function pointers. The registry (sd_registry.c) builds an
20 * brix_sd_instance_t per export by name; the VFS opens objects on the
21 * instance and runs the worker-safe raw ops (pread/pwrite/...) on the
22 * returned object handle from any dispatch tier. The VFS reaches ALL raw
23 * storage I/O through this seam — raw data syscalls live only in
24 * src/fs/backend/ (invariant 12, enforced by tools/ci/check_vfs_seam.py).
25 */

40/* ---- capability bitmap ----------------------------------------------------
41 * A driver advertises what it can do; the VFS consults this to shape behaviour
42 * (e.g. only emit a sendfile buffer when CAP_SENDFILE). Absences are honest:
43 * the VFS degrades or rejects rather than emulating a missing primitive. */
44typedef enum {
45 BRIX_SD_CAP_FD = 1u << 0, /* exposes a real kernel fd */
46 /* CAP_SENDFILE implies CAP_FD: brix_sd_fd(obj) is a real seekable kernel
47 * fd valid as the source of sendfile(2) and of an nginx file-backed
48 * (b->in_file) buffer for any byte range. A backend without it MUST be
49 * served memory-backed; the VFS read path enforces that fallback. */
50 BRIX_SD_CAP_SENDFILE = 1u << 1,
51 BRIX_SD_CAP_RANDOM_WRITE = 1u << 2, /* pwrite at arbitrary offset */
52 BRIX_SD_CAP_RANGE_READ = 1u << 3, /* pread at arbitrary offset */
53 BRIX_SD_CAP_TRUNCATE = 1u << 4, /* ftruncate */
54 BRIX_SD_CAP_SERVER_COPY = 1u << 5, /* native copy (copy_file_range/COPY)*/
55 BRIX_SD_CAP_XATTR = 1u << 6, /* user.* xattrs / object metadata */
56 BRIX_SD_CAP_HARD_RENAME = 1u << 7, /* atomic rename (else copy+delete) */
57 BRIX_SD_CAP_DIRS = 1u << 8, /* real directories (else key-prefix)*/
58 BRIX_SD_CAP_APPEND = 1u << 9, /* O_APPEND semantics */
59 BRIX_SD_CAP_IOURING = 1u << 10, /* fd is io_uring-submittable */
60 BRIX_SD_CAP_FSCS = 1u << 11, /* filesystem page checksums (CSI) */
61 /* The backend is NEARLINE (tape/MSS): an object may be offline, so a read can
62 * fault a slow async recall instead of returning bytes. A nearline backend
63 * advertises this AND implements the recall slot below; the composing registry
64 * then REQUIRES a cache tier (the recall target, phase-64 P4/§9.4) in front of
65 * it. Drivers that always serve online leave this 0 (the common case). */
66 BRIX_SD_CAP_NEARLINE = 1u << 12, /* tape/MSS: reads may recall (§9) */
67 BRIX_SD_CAP_CATALOG = 1u << 13, /* native object-catalog enumeration */
68 /* phase-71 capability-uniformity: split implicit read-only/writable and
69 * memory-serve assumptions out of VFS backend-identity branches into caps. */
70 BRIX_SD_CAP_DIRS_WRITE = 1u << 14, /* mutable catalog: mkdir/rmdir/rename */
71 BRIX_SD_CAP_XATTR_WRITE = 1u << 15, /* set/remove xattr (read = CAP_XATTR) */
72 BRIX_SD_CAP_MEMFILE = 1u << 16, /* serve bytes memory-backed w/o CAP_FD */
73 BRIX_SD_CAP_BULK_DELETE = 1u << 17, /* unlink_many is a real batch (C4) */
74 /* staged_commit evaluates its brix_sd_precond_t ATOMICALLY at the storage
75 * (phase-107 C6): the compare and the publish cannot interleave with a
76 * concurrent writer. A driver without the bit may still evaluate a
77 * precondition — it then reports the result as ADVISORY by leaving
78 * pre->atomic clear, and the protocol layer must not claim RFC 7232
79 * semantics for it. The bit is the static claim; pre->atomic is the
80 * per-call truth (on `http` the property is a runtime fact about the
81 * origin, probed once and recorded, not a compile-time fact). */
82 BRIX_SD_CAP_PRECOND = 1u << 18
83} brix_sd_cap_t;

398 /* evict (phase-107 C2) — drop the ONLINE copy of `path` while the logical
399 * object survives elsewhere (a cache-store entry, a clean stage-buffer
400 * copy, an MSS-backed online buffer, a simulated tape row, an upstream
401 * kXR_prepare(kXR_evict)). Idempotent: NGX_OK even when the copy is already
402 * absent, with *bytes_out (optional) = bytes actually freed (0 when the
403 * driver cannot know). Gated on the SLOT, not a capability bit — a cache
404 * decorator's eviction and a leaf's nearline release are different
405 * questions and both are legitimate. NULL where the online copy is the
