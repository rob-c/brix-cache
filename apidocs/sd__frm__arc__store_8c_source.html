1/*
2 * sd_frm_arc_store.c — dataset archiver: sidecar index, seal, extract
3 * (phase-115 W3.1; the vtable lives in sd_frm_arc.c)
4 *
5 * WHAT: the three I/O jobs behind the archiver adapter — the per-dataset index
6 * sidecar (`<base>/.arcidx<dataset>.idx`, one "name\tsize\toffset\tcrc"
7 * line per member), sealing a completed dataset into ONE store-only ZIP
8 * under the online buffer and migrating it, and extracting one member
9 * out of a recalled archive into the online buffer.
10 *
11 * WHY: raw storage syscalls belong under src/fs/backend/ (invariant 12); the
12 * vtable file stays declarative and this one owns the filesystem walk,
13 * the tmp+rename publishes and the zip streaming.
14 *
15 * HOW: seal = recursive lstat walk of `<online><dataset>` (regular files
16 * only, the marker excluded, symlinks never followed) → sorted names →
17 * frm_zip writer into `<archive>.tmp` → fsync+rename → inner migrate →
18 * sidecar tmp+rename → inner migrate(marker). Extract = brix_zip_index
19 * (unsafe names skipped and counted) → brix_zip_extract into
20 * inner->create_online (CRC verified; a failed copy is purged).
21 */

40/* "name\tsize\toffset\tcrc\n" → entry; 0 ok / -1 malformed. */
41static int
42arc_parse_line(char *line, brix_zip_entry_t *e)
43{
44 char *tab = strchr(line, '\t');
45 unsigned long long size, off;
46 unsigned crc;
47
48 if (tab == NULL || tab == line || (size_t) (tab - line) >= sizeof(e->name)) {
49 return -1;
50 }
51 if (sscanf(tab + 1, "%llu\t%llu\t%x", &size, &off, &crc) != 3) {
52 return -1;
53 }
54 memset(e, 0, sizeof(*e));
55 memcpy(e->name, line, (size_t) (tab - line));
56 e->size = size;
57 e->lfh_off = off;
58 e->crc = crc;
59 return brix_zip_name_ok(e->name) ? 0 : -1;
60}

80int
81arc_sidecar_lookup(const arc_ctx_t *c, const char *ds, const char *member,
82 brix_zip_entry_t *e, time_t *sealed_at)
83{
84 FILE *fp = arc_sidecar_open(c, ds, sealed_at);
85 char line[ARC_LINE_MAX];
86 int found = 0;
87
88 if (fp == NULL) {
89 return -1;
90 }
91 while (!found && fgets(line, sizeof(line), fp) != NULL) {
92 brix_zip_entry_t cur;
93
94 line[strcspn(line, "\n")] = '\0';
95 if (arc_parse_line(line, &cur) == 0 && strcmp(cur.name, member) == 0) {
96 *e = cur;
97 found = 1;
98 }
99 }
100 fclose(fp);
101 return found;
102}

133/* Publish the sidecar (sorted so arc_list's dedupe works): tmp + rename. */
134int
135arc_sidecar_write(const arc_ctx_t *c, const char *ds, brix_zip_entry_t *ents, unsigned n)
136{
137 char path[PATH_MAX], tmp[PATH_MAX];
138 FILE *fp;
139 unsigned i;
140 int rc = 0;
141
142 if (arc_sidecar_path(c, ds, path, sizeof(path)) != 0
143 || snprintf(tmp, sizeof(tmp), "%s.tmp", path) >= (int) sizeof(tmp))
144 {
145 errno = ENAMETOOLONG;
146 return -1;
147 }
148 qsort(ents, n, sizeof(*ents), arc_entry_cmp);
149 frm_mkparents(tmp);
150 fp = fopen(tmp, "we");
151 if (fp == NULL) {
152 return -1;
153 }
154 for (i = 0; i < n && rc == 0; i++) {
155 if (fprintf(fp, "%s\t%" PRIu64 "\t%" PRIu64 "\t%08x\n", ents[i].name, ents[i].size,
156 ents[i].lfh_off, ents[i].crc) < 0)
157 {
158 rc = -1;
159 }
160 }
161 if (rc == 0 && (fflush(fp) != 0 || fsync(fileno(fp)) != 0)) {
162 rc = -1;
163 }
164 if (fclose(fp) != 0 || rc != 0 || rename(tmp, path) != 0) {
165 (void) unlink(tmp);
166 return -1;
167 }
168 return frm_dirsync_parent(path);
169}

180static int
181arc_scan_push(arc_scan_t *s, const char *rel)
182{
183 char *copy;
184 unsigned index = s->n;
185
186 if (s->n >= BRIX_ZIP_MAX_ENTRIES - 1) {
187 errno = EFBIG;
188 return -1;
189 }
190 if (s->n == s->cap) {
191 unsigned ncap = s->cap ? s->cap * 2 : 64;
192 char **grown = realloc(s->names, ncap * sizeof(*grown));
193
194 if (grown == NULL) {
195 return -1;
196 }
197 s->names = grown;
198 s->cap = ncap;
199 }
200 copy = strdup(rel);
201 if (copy == NULL) {
202 return -1;
203 }
204 /* Ownership transfers INTO the name vector; arc_scan_free releases every
205 * entry.
206 *
207 * gcc 11's -fanalyzer cannot see a symbolic-index store through a
208 * parameter as an escape, so it reports `copy` leaking at the return
209 * below. Suppress exactly that diagnostic, at exactly its emission point. */
210#pragma GCC diagnostic push
211#if defined(__GNUC__) && !defined(__clang__)
212#pragma GCC diagnostic ignored "-Wanalyzer-malloc-leak"
213#endif
214 s->names[index] = copy;
215 s->n = index + 1;
216 return 0;
217#pragma GCC diagnostic pop
218}

222static int
223arc_scan_entry(arc_scan_t *s, const struct dirent *de, unsigned depth)
224{
225 struct stat sb;
226 size_t len = strlen(s->path);
227 int rc = 0;
228
229 if (len >= sizeof(s->path)) {
230 errno = ENAMETOOLONG;
231 return -1;
232 }
233 if (snprintf(s->path + len, sizeof(s->path) - len, "/%s", de->d_name)
234 >= (int) (sizeof(s->path) - len))
235 {
236 errno = ENAMETOOLONG;
237 return -1;
238 }
239 if (lstat(s->path, &sb) != 0) {
240 rc = -1;
241 } else if (S_ISDIR(sb.st_mode)) {
242 rc = arc_scan_dir(s, depth + 1);
243 } else if (S_ISREG(sb.st_mode)) {
244 rc = arc_scan_push(s, s->path + s->root_len + 1);
245 }
246 s->path[len] = '\0'; /* symlinks and specials are not members */
247 return rc;
248}

250static int
251arc_scan_dir(arc_scan_t *s, unsigned depth)
252{
253 DIR *d;
254 struct dirent *de;
255 int rc = 0;
256
257 if (depth > ARC_WALK_MAX_DEPTH) {
258 errno = ELOOP;
259 return -1;
260 }
261 d = opendir(s->path);
262 if (d == NULL) {
263 return -1;
264 }
265 while (rc == 0 && (de = readdir(d)) != NULL) {
266 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0
267 || (depth == 0 && strcmp(de->d_name, BRIX_ARC_MARKER) == 0))
268 {
269 continue;
270 }
271 rc = arc_scan_entry(s, de, depth);
272 }
273 closedir(d);
274 return rc;

294/* Stream every scanned member into the writer. */
295static int
296arc_pack(const arc_scan_t *s, brix_zip_writer_t *w)
297{
298 char src[PATH_MAX];
299 unsigned i;
300
301 for (i = 0; i < s->n; i++) {
302 struct stat sb;
303 int fd, rc;
304
305 if (snprintf(src, sizeof(src), "%.*s/%s", (int) s->root_len, s->path, s->names[i])
306 >= (int) sizeof(src))
307 {
308 errno = ENAMETOOLONG;
309 return -1;
310 }
311 fd = open(src, O_RDONLY | O_CLOEXEC | O_NOFOLLOW);
312 if (fd < 0 || fstat(fd, &sb) != 0) {
313 if (fd >= 0) {
314 close(fd);
315 }
316 return -1;
317 }
318 rc = brix_zip_writer_add(w, s->names[i], fd, (uint64_t) sb.st_size, NULL);
319 close(fd);
320 if (rc != 0) {
321 return -1;
322 }
323 }
324 return 0;
325}

327/* Write `<archive>.tmp`, fsync it and rename it into place. Returns the
328 * finished writer (entries for the sidecar) or NULL with errno. */
329static brix_zip_writer_t *
330arc_write_archive(const arc_scan_t *s, const char *archive)
331{
332 char tmp[PATH_MAX];
333 int fd, rc;
334 brix_zip_writer_t *w;
335
336 if (snprintf(tmp, sizeof(tmp), "%s.tmp", archive) >= (int) sizeof(tmp)) {
337 errno = ENAMETOOLONG;
338 return NULL;
339 }
340 fd = open(tmp, O_RDWR | O_CREAT | O_TRUNC | O_CLOEXEC | O_NOFOLLOW, 0644);
341 if (fd < 0) {
342 return NULL;
343 }
344 w = brix_zip_writer_open(fd);
345 rc = (w != NULL) ? arc_pack(s, w) : -1;
346 rc = (rc == 0) ? brix_zip_writer_finish(w) : rc;
347 rc = (rc == 0 && fsync(fd) != 0) ? -1 : rc;
348 close(fd);
349 if (rc != 0 || rename(tmp, archive) != 0) {
350 int e = errno;
351
352 (void) unlink(tmp);
353 brix_zip_writer_close(w);
354 errno = e;
355 return NULL;
356 }
357 return w;
358}

377static int
378arc_publish(arc_ctx_t *c, const arc_key_t *k, const char *akey, const char *archive,
379 brix_zip_writer_t *w)
380{
381 unsigned n = 0;
382 uint64_t bytes = 0, i;
383 brix_zip_entry_t *ents;
384
385 if (c->inner->migrate(c->ictx, akey) != 0) {
386 int e = errno;
387
388 (void) unlink(archive);
389 ARC_LOG(c, NGX_LOG_ERR, e, "brix: tape archive \"%s\": migrate failed; dataset left unsealed", k->ds);
390 errno = e ? e : EIO;
391 return -1;
392 }
393 ents = arc_writer_entries(w, &n);
394 if (ents == NULL || arc_sidecar_write(c, k->ds, ents, n) != 0) {
395 int e = errno;
396
397 free(ents);
398 ARC_LOG(c, NGX_LOG_ERR, e, "brix: tape archive \"%s\": index sidecar write failed", k->ds);
399 errno = e ? e : EIO;
400 return -1;
401 }
402 for (i = 0; i < n; i++) {
403 bytes += ents[i].size;
404 }
405 free(ents);
406 if (c->inner->migrate(c->ictx, k->key) != 0) {
407 ARC_LOG(c, NGX_LOG_WARN, errno, "brix: tape archive \"%s\": sealed, but the completion "
408 "marker did not migrate", k->ds);
409 return -1;
410 }
411 ARC_LOG(c, NGX_LOG_NOTICE, 0, "brix: tape archive \"%s\": sealed %ud member(s), %uL bytes "
412 "into \"%s\"", k->ds, n, bytes, akey);
413 return 0;
414}

417arc_compose(arc_ctx_t *c, const arc_key_t *k)
418{
419 arc_scan_t s;
420 char akey[PATH_MAX], archive[PATH_MAX];
421 brix_zip_writer_t *w;
422 int rc;
423
424 if (arc_sealed(c, k->ds)) {
425 errno = EEXIST;
426 return -1;
427 }
428 memset(&s, 0, sizeof(s));
429 if (arc_archive_key(k, akey, sizeof(akey)) != 0
430 || frm_online_path(c->base, akey, archive, sizeof(archive)) != 0

432 {
433 errno = ENAMETOOLONG;
434 return -1;
435 }
436 s.root_len = strlen(s.path);
437 if (arc_scan_dir(&s, 0) != 0) {
438 int e = errno;
439

441 ARC_LOG(c, NGX_LOG_ERR, e, "brix: tape archive \"%s\": member walk failed", k->ds);
442 errno = e;
443 return -1;

445 qsort(s.names, s.n, sizeof(*s.names), arc_name_cmp);
446 w = arc_write_archive(&s, archive);
447 arc_scan_free(&s);

451 }

459typedef struct {
460 const char *want;
461 brix_zip_entry_t e;
462 int found;
463 int rebuild; /* no sidecar: collect every entry */
464 brix_zip_entry_t *all;
465 unsigned n, cap;
466} arc_find_t;

468static int
469arc_find_cb(void *ud, const brix_zip_entry_t *e)
470{
471 arc_find_t *f = ud;
472
473 if (!f->found && strcmp(e->name, f->want) == 0) {
474 f->e = *e;
475 f->found = 1;
476 }
477 if (!f->rebuild) {
478 return f->found; /* stop as soon as the member is known */
479 }
480 if (f->n == f->cap) {
481 unsigned ncap = f->cap ? f->cap * 2 : 64;
482 brix_zip_entry_t *grown = realloc(f->all, ncap * sizeof(*grown));
483
484 if (grown == NULL) {
485 f->rebuild = 0; /* keep serving; just do not rebuild */
486 return f->found;
487 }
488 f->all = grown;
489 f->cap = ncap;
490 }
491 f->all[f->n++] = *e;
492 return 0;
493}

519int
520arc_extract(arc_ctx_t *c, const arc_key_t *k)
521{
522 char akey[PATH_MAX], archive[PATH_MAX];
523 arc_find_t f;
524 unsigned skipped = 0;
525 int afd, rc;
526
527 if (arc_archive_key(k, akey, sizeof(akey)) != 0
528 || frm_online_path(c->base, akey, archive, sizeof(archive)) != 0)
529 {
530 errno = ENAMETOOLONG;
531 return -1;
532 }
533 afd = open(archive, O_RDONLY | O_CLOEXEC | O_NOFOLLOW);
534 if (afd < 0) {
535 /* The inner adapter said the archive is online, yet its buffer copy is
536 * not openable: name it — a silent -1 surfaces as a bare "recall failed". */
537 ARC_LOG(c, NGX_LOG_ERR, errno, "brix: tape archive \"%s\": online copy unreadable", k->ds);
538 return -1;
539 }
540 memset(&f, 0, sizeof(f));
541 f.want = k->member;
542 f.rebuild = !arc_sealed(c, k->ds);
543 rc = brix_zip_index(afd, arc_find_cb, &f, &skipped);
544 if (rc != 0) {
545 ARC_LOG(c, NGX_LOG_ERR, errno, "brix: tape archive \"%s\": not a readable zip container", k->ds);
546 }
547 if (skipped != 0) {
548 ARC_LOG(c, NGX_LOG_WARN, 0, "brix: tape archive \"%s\": skipped %ud unsafe member name(s)",
549 k->ds, skipped);
550 }
551 if (rc == 0 && f.rebuild && f.n > 0 && arc_sidecar_write(c, k->ds, f.all, f.n) != 0) {
552 ARC_LOG(c, NGX_LOG_WARN, errno, "brix: tape archive \"%s\": index sidecar rebuild failed", k->ds);
553 }
554 free(f.all);
555 if (rc == 0 && !f.found) {
556 errno = ENOENT;
557 rc = -1;
558 }
559 if (rc == 0) {
560 rc = arc_copy_member(c, k, afd, &f.e);
561 }
562 close(afd);
563 return rc;
564}

566/* Durable-publish barrier. A dataset key has no tape-side entry under its own
567 * name: a MEMBER's migrate is deferred to the seal and the MARKER composes the
568 * archive object — so the inner barrier, which fsyncs the key's TAPE parent as
569 * well as its online parent, answered ENOENT for every member and failed the
570 * commit that had just succeeded (phase-115 W3.1). Lives here, not in sd_frm_arc.c, for the
571 * 600-line cap. Flush what exists: the
572 * online-buffer entry for a member; the archive object's entries for a marker. */
573int
574arc_sync_publish(void *mss, const char *key)
575{
576 arc_ctx_t *c = mss;
577 arc_key_t k;
578 char path[PATH_MAX];
579
580 if (c->inner->sync_publish == NULL) {
581 return 0;
582 }
583 switch (arc_classify(c, key, &k)) {
584 case ARC_KEY_MEMBER:
585 if (frm_online_path(c->base, key, path, sizeof(path)) != 0) {
586 errno = ENAMETOOLONG;
587 return -1;
588 }
589 return frm_dirsync_parent(path);
590 case ARC_KEY_MARKER:
591 if (arc_archive_key(&k, path, sizeof(path)) != 0) {
592 errno = ENAMETOOLONG;
593 return -1;
594 }
595 return c->inner->sync_publish(c->ictx, path);
596 default:
597 return c->inner->sync_publish(c->ictx, key);
598 }
599}
