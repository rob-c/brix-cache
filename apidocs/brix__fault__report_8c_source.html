35void
36res_set(char *slot, const char *fmt, ...)
37{
38 va_list ap;
39 pthread_mutex_lock(&g_res_lock);
40 va_start(ap, fmt);
41 vsnprintf(slot, 192, fmt, ap);
42 va_end(ap);
43 pthread_mutex_unlock(&g_res_lock);
44}

46/* Binary-search the smallest integer value of `lever` in [lo,hi] for which the
47 * oracle command FAILS (non-zero exit = the bug reproduces), assuming severity
48 * is monotonic in the lever. Publishes progress + the boundary to g_bisect_result. */
49void *
50bisect_thread(void *arg)
51{
52 struct bisect_arg *b = arg;
53 long lo = b->lo, hi = b->hi, found = b->hi + 1;
54 int probes = 0;
55 while (lo <= hi) {
56 long mid = lo + (hi - lo) / 2;
57 char cmd[48];
58 snprintf(cmd, sizeof(cmd), "%s %ld", b->lever, mid);
59 apply_command(cmd, NULL, 0);
60 int rc = fp_oracle_run(b->cmd, b->timeout_ms);
61 probes++;
62 res_set(g_bisect_result, "running: %s probe#%d val=%ld rc=%d",
63 b->lever, probes, mid, rc);
64 if (rc == -2) {
65 res_set(g_bisect_result, "error: oracle disabled (--enable-exec)");
66 break;
67 }
68 if (rc != 0) { /* reproduced (fail or inconclusive) → go smaller */
69 found = mid;
70 hi = mid - 1;
71 } else { /* survived → need a harsher value */
72 lo = mid + 1;
73 }
74 }
75 apply_command((char[]){"clear"}, NULL, 0);
76 if (found <= b->hi) {
77 res_set(g_bisect_result, "done: minimal %s=%ld reproduces (%d probes)",
78 b->lever, found, probes);
79 } else {
80 res_set(g_bisect_result, "done: no %s in [%ld,%ld] reproduced (%d probes)",
81 b->lever, b->lo, b->hi, probes);
82 }
83 g_oracle_busy = 0;
84 free(b);
85 return NULL;
86}

88/* Apply a fault for hold_ms, clear it, then poll a health probe until it passes
89 * or times out — asserting the service recovers. Publishes to g_recovery_result. */
90void *
91recovery_thread(void *arg)
92{
93 struct recov_arg *r = arg;
94 char fcmd[256];
95 snprintf(fcmd, sizeof(fcmd), "%s", r->fault);
96 apply_command(fcmd, NULL, 0);
97 res_set(g_recovery_result, "running: fault held %dms", r->hold_ms);
98 usleep((useconds_t) r->hold_ms * 1000);
99 apply_command((char[]){"clear"}, NULL, 0);
100 struct timespec t0;
101 clock_gettime(CLOCK_MONOTONIC, &t0);
102 int elapsed = 0;
103 for (;;) {
104 int rc = fp_oracle_run(r->probe, 2000);
105 if (rc == -2) {
106 res_set(g_recovery_result, "error: oracle disabled (--enable-exec)");
107 break;
108 }
109 elapsed = (int) now_ms_since(t0);
110 if (rc == 0) {
111 res_set(g_recovery_result, "done: recovered in %dms", elapsed);
112 break;
113 }
114 if (elapsed >= r->timeout_ms) {
115 res_set(g_recovery_result, "done: STUCK — no recovery in %dms", elapsed);
116 break;
117 }
118 usleep(250 * 1000);
119 }
120 g_oracle_busy = 0;
121 free(r);
122 return NULL;
123}

134/* The two preconditions every probe-spawning oracle job shares. Returns 1 (with
135 * `reply` filled) when `job` must not start. */

137oracle_refused(const char *job, char *reply, size_t rsz)
138{
139 if (!fp_oracle_enabled()) {

150/* Hand `arg` to `fn` on a detached thread, taking the busy flag with it. On any
151 * failure the flag is released and `arg` freed, so the slot never wedges. */
152static void
153oracle_spawn(void *(*fn)(void *), void *arg, const char *job,
154 char *reply, size_t rsz, const char *okmsg)
155{
156 pthread_t th;
157 g_oracle_busy = 1;
158 if (pthread_create(&th, NULL, fn, arg) != 0) {
159 g_oracle_busy = 0;
160 free(arg);
161 snprintf(reply, rsz, "err: cannot start %s thread\n", job);
162 return;
163 }
164 pthread_detach(th);
165 snprintf(reply, rsz, "%s", okmsg);
166}

168/* bisect <lever> <lo> <hi> <timeout_ms> <oracle-cmd> */
169static void
170oracle_bisect(char *args, char *reply, size_t rsz)
171{
172 struct bisect_arg *b = calloc(1, sizeof(*b));
173 char *lever = strtok(args, " ");
174 char *lo = strtok(NULL, " ");
175 char *hi = strtok(NULL, " ");
176 char *to = strtok(NULL, " ");
177 char *cmd = strtok(NULL, "");
178
179 if (!b || !lever || !lo || !hi || !to || !cmd) {
180 free(b);
181 snprintf(reply, rsz, "err: bisect <lever> <lo> <hi> <timeout_ms> <oracle-cmd>\n");
182 return;
183 }
184 snprintf(b->lever, sizeof(b->lever), "%s", lever);
185 b->lo = atol(lo); b->hi = atol(hi); b->timeout_ms = atoi(to);
186 snprintf(b->cmd, sizeof(b->cmd), "%s", cmd);
187
188 char ok[128];
189 snprintf(ok, sizeof(ok), "ok (bisecting %s in [%ld,%ld]; poll bisect-result)\n",
190 b->lever, b->lo, b->hi);
191 oracle_spawn(bisect_thread, b, "bisect", reply, rsz, ok);
192}

418/* Parse and apply one control command. `line` is mutated. `reply` (may be NULL)
419 * receives a human-readable response of up to `rsz` bytes. */
420void
421apply_command(char *line, char *reply, size_t rsz)
422{
423 const char *j = line;
424 while (*j == ' ' || *j == '\t') {
425 j++;
426 }
427 if (*j == '{') {
428 apply_json_command(j, reply, rsz);
429 return;
430 }
431
432 char verb[32] = "", args[2000] = "";
433 sscanf(line, "%31s %1999[^\n]", verb, args);
434 fp_reply(reply, rsz, "ok\n");
435
436 if (strcmp(verb, "priv") == 0) {
437 fp_priv_command(args, reply, rsz);
438 } else if (strcmp(verb, "event-log") == 0) {
439 cmd_event_log(args, reply, rsz);
440 } else if (strcmp(verb, "toxic") == 0) {
441 fp_toxic_cmd(args, reply, rsz);
442 } else if (strcmp(verb, "route") == 0) {
443 fp_route_cmd(args, reply, rsz);
444 } else if (strcmp(verb, "metrics") == 0) {
445 cmd_metrics_report(reply, rsz);
446 } else if (cmd_set_dispatch(verb, args, reply, rsz)) {
447 return;
448 } else if (strcmp(verb, "status") == 0) {
449 cmd_status(args, reply, rsz);
450 } else if (verb[0] != '\0') {
451 fp_reply(reply, rsz, "err: unknown command\n");
452 }
453}

455void *
456control_thread(void *arg)
457{
458 int lfd = *(int *) arg;
459 free(arg);
460
461 for (;;) {
462 int cfd = accept(lfd, NULL, NULL);
463 if (cfd < 0) {
464 if (errno == EINTR) {
465 continue;
466 }
467 break;
468 }
469 /* A1 persistent session: accumulate bytes and process every complete
470 * newline-delimited command over the ONE connection, replying to each.
471 * `quit`/EOF closes; a line that fills the buffer with no newline is
472 * refused with a diagnostic rather than overrunning the parser. */
473 char acc[2048];
474 size_t used = 0;
475 int done = 0;
476 while (!done) {
477 ssize_t n = read(cfd, acc + used, sizeof(acc) - 1 - used);
478 if (n <= 0) {
479 break;
480 }
481 used += (size_t) n;
482 acc[used] = '\0';
483 size_t start = 0;
484 char *nl;
485 while ((nl = memchr(acc + start, '\n', used - start)) != NULL) {
486 *nl = '\0';
487 char line[2048];
488 snprintf(line, sizeof(line), "%s", acc + start);
489 size_t ll = strlen(line);
490 if (ll > 0 && line[ll - 1] == '\r') {
491 line[ll - 1] = '\0';
492 }
493 start = (size_t) (nl - acc) + 1;
494 if (strcmp(line, "quit") == 0) {
495 done = 1;
496 break;
497 }
498 if (line[0] != '\0') {
499 char reply[2048] = "";
500 apply_command(line, reply, sizeof(reply));
501 (void) write_all(cfd, reply, (ssize_t) strlen(reply));
502 }
503 }
504 if (start > 0) {
505 memmove(acc, acc + start, used - start);
506 used -= start;
507 }
508 if (!done && used >= sizeof(acc) - 1) {
509 static const char msg[] = "err: line too long\n";
510 (void) write_all(cfd, msg, (ssize_t) (sizeof(msg) - 1));
511 /* Half-close our side so the diagnostic is delivered, then drain
512 * the rest of the oversized line so close() sends a clean FIN
513 * rather than an RST that would race the client's read. */
514 shutdown(cfd, SHUT_WR);
515 char drain[2048];
516 while (read(cfd, drain, sizeof(drain)) > 0) {
517 /* discard */
518 }
519 done = 1;
520 }
521 }
522 close(cfd);
523 }
524 return NULL;
525}

527/* Replay a '<seconds> <command>' timeline file (relative to start). */
528void *
529script_thread(void *arg)
530{
531 FILE *fp = fopen((const char *) arg, "r");
532 if (fp == NULL) {
533 fprintf(stderr, "brix-fault-proxy: cannot open --script '%s'\n",
534 (const char *) arg);
535 return NULL;
536 }
537 char line[256];
538 double prev = 0.0;
