47static int read_whole(xrd_conform_execution *execution,
48 brix_conn *connection, const char *path, uint64_t size, uint8_t **bytes,
49 brix_status *status)
50{
51 brix_file file;
52 uint64_t offset = 0;
53 int rc = -1;
54
55 if (size > SIZE_MAX || size > execution->run->config.byte_limit) {
56 brix_status_set(status, XRDC_EUSAGE, 0,
57 "paired object exceeds byte ceiling");
58 return -1;
59 }
60 *bytes = malloc(size == 0 ? 1 : (size_t) size);
61 if (*bytes == NULL) return -1;
62 if (!xrd_conform_reserve_status(execution, 0, status)
63 || brix_file_open_read(connection, path, &file, status) != 0) {
64 free(*bytes);
65 *bytes = NULL;
66 return -1;
67 }
68 while (offset < size) {
69 size_t length = size - offset < CAC_READ_CHUNK
70 ? (size_t) (size - offset) : CAC_READ_CHUNK;
71 ssize_t got;
72
73 if (!xrd_conform_reserve_status(execution, length, status)) break;
74 got = brix_file_read(connection, &file, (int64_t) offset,
75 *bytes + offset, length, status);
76 if (got != (ssize_t) length) break;
77 offset += length;
78 }
79 if (offset == size) rc = 0;
80 {
81 brix_status close_status;
82
83 brix_status_clear(&close_status);
84 if (!xrd_conform_reserve_status(execution, 0, &close_status)
85 || brix_file_close(connection, &file, &close_status) != 0) rc = -1;
86 }
87 if (rc != 0) {
88 free(*bytes);
89 *bytes = NULL;
90 }
91 return rc;
92}

94static int extent_equal(xrd_conform_execution *execution,
95 brix_conn *cache, brix_conn *origin, const brix_file *cache_file,
96 const brix_file *origin_file, uint64_t size, int64_t offset,
97 int32_t length, brix_status *status)
98{
99 uint8_t *cache_body = NULL;
100 uint8_t *origin_body = NULL;
101 uint32_t cache_length = 0;
102 uint32_t origin_length = 0;
103 uint16_t cache_status = 0;
104 uint16_t origin_status = 0;
105 uint64_t remaining = (uint64_t) offset < size
106 ? size - (uint64_t) offset : 0;
107 size_t expected = remaining < (uint32_t) length
108 ? (size_t) remaining : (size_t) length;
109 int cache_rc;
110 int origin_rc;
111 int equal;
112
113 cache_rc = xrd_conform_root_raw_read(execution, cache,
114 cache_file->fhandle, offset, length, NULL, 0, expected,
115 &cache_status, &cache_body, &cache_length, status);
116 origin_rc = xrd_conform_root_raw_read(execution, origin,
117 origin_file->fhandle, offset, length, NULL, 0, expected,
118 &origin_status, &origin_body, &origin_length, status);
119 equal = cache_rc == 0 && origin_rc == 0
120 && cache_status == kXR_ok && origin_status == kXR_ok
121 && cache_length == origin_length
122 && cache_length == expected
123 && (expected == 0 || memcmp(cache_body, origin_body, expected) == 0);
124 free(cache_body);
125 free(origin_body);
126 return equal;
127}

129static int
130fractional_grid_compare(xrd_conform_execution *execution, brix_conn *cache,
131 brix_conn *origin, const brix_file *cache_file,
132 const brix_file *origin_file, uint64_t size, brix_status *status)
133{
134 int64_t offsets[BRIX_CONFORM_AXIS_MAX];
135 int64_t lengths[BRIX_CONFORM_AXIS_MAX];
136 size_t offset_count;
137 size_t offset_index;
138 int pass = 1;
139
140 offset_count = brix_conform_offsets(size, offsets, BRIX_CONFORM_AXIS_MAX);
141 for (offset_index = 0; pass > 0 && offset_index < offset_count;
142 offset_index++) {
143 size_t length_count = brix_conform_lengths(size,
144 offsets[offset_index], lengths, BRIX_CONFORM_AXIS_MAX);
145 size_t length_index;
146
147 for (length_index = 0; pass > 0 && length_index < length_count;
148 length_index++) {
149 int selected = brix_conform_case_selected(
150 execution->run->config.budget, 0, 1, offset_index,
151 offset_count, length_index, length_count,
152 offsets[offset_index], lengths[length_index], size);
153
154 if (!selected || offsets[offset_index] < 0
155 || lengths[length_index] < 0
156 || lengths[length_index] > INT32_MAX) continue;
157 if (!extent_equal(execution, cache, origin, cache_file,
158 origin_file, size, offsets[offset_index],
159 (int32_t) lengths[length_index], status))
160 pass = status->kxr == XRDC_EUSAGE ? -1 : 0;
161 }
162 }
163 return pass;
164}

218grade_cache_population(xrd_conform_execution *execution, brix_conn *cache,
219 const brix_url *cache_url, uint64_t size, brix_status *status)
220{
221 uint8_t *first = NULL;
222 uint8_t *second = NULL;
223 int read_ok;
224 int same;
225
226 if (!xrd_conform_row_active(execution, "CAC-001")) return;
227 read_ok = read_whole(execution, cache, cache_url->path, size, &first,
228 status) == 0
229 && read_whole(execution, cache, cache_url->path, size, &second,
230 status) == 0;
231 same = read_ok && memcmp(first, second, (size_t) size) == 0;
232 xrd_conform_result_set(execution, "CAC-001",
233 same ? BRIX_CONFORM_VERDICT_CONFORMANT
234 : read_ok ? BRIX_CONFORM_VERDICT_DIVERGENT
235 : BRIX_CONFORM_VERDICT_INCONCLUSIVE,
236 same ? "first-and-second-cache-reads-identical"
237 : read_ok ? "cache-population-changed-bytes"
238 : "cache-population-read-failed",
239 NULL, read_ok ? NULL : status->msg);
240 free(first);
241 free(second);

323/* WHAT: Compare one explicit cache endpoint with its explicit origin pair.
324 * WHY: Population and cache correctness cannot be inferred from one endpoint.
325 * HOW: Read the cache twice before the ordinary run, then compare the selected
326 * fractional grid, stat sizes, and server-calculated SHA-256 checksums. */
327void xrd_conform_probe_cache_pair(xrd_conform_execution *execution)
328{
329 brix_url cache_url;
330 brix_url origin_url;
331 brix_conn cache;
332 brix_conn origin;
333 brix_statinfo cache_info;
334 brix_statinfo origin_info;
335 brix_status status;
336 int connect_rc;
337
338 if (!xrd_conform_family_active(execution, "CAC")
339 && !xrd_conform_row_active(execution, "PRP-003")
340 && !xrd_conform_row_active(execution, "PRP-004")) return;
341 brix_status_clear(&status);
342 connect_rc = connect_cache_pair(execution, &cache_url, &origin_url,
343 &cache, &origin, &status);
344 if (connect_rc != 0) {
345 pair_inconclusive(execution,
346 connect_rc == -1 ? "fixture-required:paired-cache-origin"
347 : "fixture-unavailable:paired-cache-origin",
348 &status);
349 return;
350 }
351 xrd_conform_probe_cache_evict(execution, &cache, &origin);
352 xrd_conform_probe_cache_fixture(execution, &cache, &origin, &cache_url,
353 &origin_url);
354 if (!pair_read_active(execution)) {
355 brix_close(&origin);
356 brix_close(&cache);
357 return;
358 }
359 if (load_pair_stats(execution, &cache, &origin, &cache_url, &origin_url,
360 &cache_info, &origin_info, &status) != 0) {
361 pair_inconclusive(execution, "fixture-unavailable:paired-cache-origin",
362 &status);
363 brix_close(&origin);
364 brix_close(&cache);
365 return;
366 }
367 grade_pair_size(execution, cache_info.size, origin_info.size);
368 if (cache_info.size == origin_info.size) {
369 grade_cache_population(execution, &cache, &cache_url,
370 (uint64_t) cache_info.size, &status);
371 grade_cache_grid(execution, &cache, &origin, &cache_url, &origin_url,
372 (uint64_t) cache_info.size, &status);
373 }
374 grade_checksums(execution, &cache, &origin, &cache_url, &origin_url);
375 brix_close(&origin);
376 brix_close(&cache);
377}
