34/* HEAD `objpath` under the (already cred-applied) params: 1 = object exists,
35 * 0 = absent (or HEAD failed). Best-effort existence probe for the stat/mkdir/
36 * rename gates below. */
37static int
38sd_remote_head_exists(const sd_s3_open_params *p)
39{
40 char errbuf[256];
41 sd_s3_file *s3;

43 int exists;
44

54/* ---- xattr surface: x-amz-meta-* as the `user.` namespace --------------
55 *
56 * getxattr("user.<name>") reads x-amz-meta-<name> via a signed HEAD;
57 * listxattr enumerates every x-amz-meta-* header (needs a transport with the
58 * optional resp_headers_raw slot — without it sd_s3_list_meta reports

66 char objpath[768];
67 char errbuf[256];
68
69 sd_remote_s3_key(cfg, path, objpath, sizeof(objpath));
70 sd_remote_s3_params(cfg, objpath, &p);
71 return sd_s3_open_read(&p, errbuf, sizeof(errbuf));
72}

74ssize_t
75sd_remote_getxattr(brix_sd_instance_t *inst, const char *path,
76 const char *name, void *buf, size_t cap)
77{
78 char val[2048];
79 char errbuf[256];
80 sd_s3_meta_buf dst = { val, sizeof(val) };
81 sd_s3_file *s3;
82 ssize_t n;
83
84 if (strncmp(name, "user.", 5) != 0 || name[5] == '\0') {
85 errno = ENODATA; /* only the user. namespace maps to x-amz-meta- */
86 return -1;
87 }
88 s3 = sd_remote_meta_open(inst, path);
89 if (s3 == NULL) {
90 errno = ENOMEM;
91 return -1;
92 }
93 errno = 0;
94 n = sd_s3_get_meta(s3, name + 5, &dst, errbuf, sizeof(errbuf));
95 sd_s3_close(s3);
96 if (n < 0) {
97 if (errno == 0) { errno = EIO; }
98 return -1;
99 }
100 if (n == 0) {
101 errno = ENODATA; /* HEAD ok, attribute absent */
102 return -1;
103 }
104 if (buf == NULL || cap == 0) {
105 return n; /* getxattr(2) size probe */
106 }
107 if ((size_t) n > cap) {
108 errno = ERANGE;
109 return -1;
110 }
111 memcpy(buf, val, (size_t) n);
112 return n;
113}

137/* Shared stat body: HEAD the object, optionally signing with a per-user
138 * ak/sk/region override (NULL = the instance's static service credential).
139 *
140 * S3 has no directories, so classification is by object shape (#4): the export
141 * root is always a directory; a "path" object is a regular file; failing that a
142 * "path/" marker object is a directory. Directory recognition relies on the
143 * marker this driver's mkdir (and put_inner parent-prefix creation) writes — an
144 * externally-created prefix with children but no marker stats as ENOENT here,
145 * though opendir still lists it. */
146static ngx_int_t
147sd_remote_stat_impl(brix_sd_instance_t *inst, const char *path,
148 brix_sd_stat_t *out, const char *ak, const char *sk, const char *region,
149 const char *session)
150{
151 const brix_sd_remote_cfg_t *cfg = inst->state;
152 sd_s3_open_params p;
153 char objpath[768];
154 char errbuf[256];
155 sd_s3_file *s3;
156 int64_t size = 0;
157
158 /* The export root is a directory with no backing marker object. */
159 if (path == NULL || path[0] == '\0'
160 || (path[0] == '/' && path[1] == '\0'))
161 {
162 memset(out, 0, sizeof(*out));
163 out->mode = S_IFDIR | 0755;
164 out->is_dir = 1;
165 return NGX_OK;
166 }
167
168 /* (1) regular file: HEAD "path" for the byte count. */
169 sd_remote_s3_key(cfg, path, objpath, sizeof(objpath));
170 sd_remote_s3_params(cfg, objpath, &p);
171 sd_remote_params_cred(&p, ak, sk, region, session);
172 s3 = sd_s3_open_read(&p, errbuf, sizeof(errbuf));
173 if (s3 != NULL) {
174 int ok = sd_s3_size(s3, &size, errbuf, sizeof(errbuf)) == 0;
175
176 sd_s3_close(s3);
177 if (ok) {
178 memset(out, 0, sizeof(*out));
179 out->size = (off_t) size;
180 out->mode = S_IFREG | 0444;
181 out->is_reg = 1;
182 return NGX_OK;
183 }
184 }
185
186 /* (2) directory: HEAD the "path/" folder marker. */
187 sd_remote_s3_dirkey(cfg, path, objpath, sizeof(objpath));
188 sd_remote_s3_params(cfg, objpath, &p);
189 sd_remote_params_cred(&p, ak, sk, region, session);
190 if (sd_remote_head_exists(&p)) {
191 memset(out, 0, sizeof(*out));
192 out->mode = S_IFDIR | 0755;
193 out->is_dir = 1;
194 return NGX_OK;
195 }
196
197 errno = ENOENT;
198 return NGX_ERROR;
199}

208/* ---- server-side copy (S3 CopyObject, x-amz-copy-source) ----------------
209 *
210 * WHAT: Copy the export-relative `src` object to `dst` entirely within the S3
211 * origin — the bytes never traverse this host. Runs on the instance's
212 * static service credential (Phase-1 namespace policy; a fallback_deny
213 * per-user request is refused upstream by the cred forwarder, which has no
214 * server_copy_cred slot to route to).
215 * WHY: finding #4 — WebDAV COPY / xrdcp server-side copy over an s3:// backend
216 * previously fell through to ENOSYS. S3 exposes CopyObject natively, so
217 * this is a single signed PUT rather than a read-back/re-upload.
218 * HOW: compose the "/bucket/src" copy-source and the "/bucket/dst" request
219 * target, call the shared sd_s3_copy primitive, then best-effort HEAD the
220 * destination for the copied byte count (mirrors the POSIX server_copy;
221 * 0 when the follow-up stat cannot confirm the size).
222 */
223ngx_int_t
224sd_remote_server_copy(brix_sd_instance_t *inst, const char *src,
225 const char *dst, off_t *bytes_out)
226{
227 const brix_sd_remote_cfg_t *cfg = inst->state;
228 sd_s3_open_params p;
229 char srcpath[768];
230 char dstpath[768];
231 char errbuf[256];
232
233 sd_remote_s3_key(cfg, src, srcpath, sizeof(srcpath)); /* /bucket/src */
234 sd_remote_s3_key(cfg, dst, dstpath, sizeof(dstpath)); /* /bucket/dst */
235 sd_remote_s3_params(cfg, dstpath, &p); /* target = dst */
236
237 errno = 0;
238 if (sd_s3_copy(&p, srcpath, errbuf, sizeof(errbuf)) != 0) {
239 if (errno == 0) { errno = EIO; }
240 return NGX_ERROR;
241 }
242 if (bytes_out != NULL) {
243 brix_sd_stat_t st;
244 *bytes_out = (sd_remote_stat(inst, dst, &st) == NGX_OK) ? st.size : 0;
245 }
246 return NGX_OK;
247}

270/* ---- mkdir (zero-byte "path/" folder marker) -----------------------------
271 *
272 * WHAT: Create the directory `path` by PUTting a zero-byte object whose key ends
273 * in '/'. Returns NGX_OK, or NGX_ERROR with errno=EEXIST when the marker
274 * already exists (POSIX mkdir; brix_vfs_backend_mkpath tolerates it per
275 * component) / EIO on a transport failure.
276 * WHY: finding #4 — WebDAV MKCOL / xrdfs mkdir over an s3:// backend hit the NULL
277 * mkdir slot (ENOSYS). Advertising CAP_DIRS_WRITE also turns on put_inner's
278 * S3-PUT parent-prefix creation, which drives this same slot; a working
279 * marker mkdir keeps that cap self-consistent.
280 * HOW: HEAD the marker for idempotency, then a single zero-byte PUT.
281 */
282static ngx_int_t
283sd_remote_mkdir_impl(brix_sd_instance_t *inst, const char *path,
284 const char *ak, const char *sk, const char *region, const char *session)
285{
286 const brix_sd_remote_cfg_t *cfg = inst->state;
287 sd_s3_open_params p;
288 char dirkey[768];
289 char errbuf[256];
290 sd_s3_file *s3;
291
292 sd_remote_s3_dirkey(cfg, path, dirkey, sizeof(dirkey));
293 sd_remote_s3_params(cfg, dirkey, &p);
294 sd_remote_params_cred(&p, ak, sk, region, session);
295
296 if (sd_remote_head_exists(&p)) {
297 errno = EEXIST;
298 return NGX_ERROR;
299 }
300
301 s3 = sd_s3_open_write(&p, 0, SD_REMOTE_MARKER_PART, errbuf, sizeof(errbuf));
302 if (s3 == NULL) {
303 errno = EIO;
304 return NGX_ERROR;
305 }
306 if (sd_s3_commit(s3, errbuf, sizeof(errbuf)) != 0) {
307 sd_s3_abort(s3);
308 sd_s3_close(s3);
309 errno = EIO;
310 return NGX_ERROR;
311 }
312 sd_s3_close(s3);
313 return NGX_OK;
314}

354/* Does the directory prefix for `path` contain at least one child (beyond its
355 * own marker, which sd_s3_list_page skips)? 1 = has a child, 0 = empty,
356 * -1 = probe error (errno set). */
357static int
358sd_remote_prefix_has_child(const brix_sd_remote_cfg_t *cfg, const char *path,
359 const char *ak, const char *sk, const char *region, const char *session)
360{
361 sd_s3_open_params p;
362 const char *rel = (path != NULL) ? path : "/";
363 char root[300];
364 char prefix[768];
365 char cont_out[2048];
366 char errbuf[256];
367 size_t n;
368 int truncated = 0;
369 int found = 0;
370
371 while (*rel == '/') { rel++; }
372 n = strlen(rel);
373 if (n + 1 >= sizeof(prefix)) {
374 errno = ENAMETOOLONG;
375 return -1;
376 }
377 memcpy(prefix, rel, n);
378 if (n > 0 && prefix[n - 1] != '/') { prefix[n++] = '/'; }
379 prefix[n] = '\0';
380
381 snprintf(root, sizeof(root), "/%s/", cfg->bucket); /* bucket-root canon URI */
382 sd_remote_s3_params(cfg, root, &p);
383 sd_remote_params_cred(&p, ak, sk, region, session);
384
385 errno = 0;
386 if (sd_s3_list_page(&p, prefix, "", sd_remote_rename_child_cb, &found,
387 &truncated, cont_out, sizeof(cont_out), errbuf, sizeof(errbuf)) != 0)
388 {
389 if (errno == 0) { errno = EIO; }
390 return -1;
391 }
392 return found;
393}

420/* ---- rename (copy + delete; file or empty directory) ---------------------
421 *
422 * WHAT: Move `src` to `dst`. Returns NGX_OK; NGX_ERROR with errno EEXIST (dst
423 * present under noreplace), ENOENT (src absent), ENOTSUP (src is a
424 * NON-empty directory — no atomic prefix move), EIO otherwise.
425 * WHY: finding #4 — WebDAV MOVE / xrdfs mv over an s3:// backend hit the NULL
426 * rename slot (ENOSYS).
427 * HOW: Classify src (file HEAD, else empty-dir marker after a child probe) and
428 * copy+delete the corresponding object; S3 has no atomic move.
429 */
430static ngx_int_t
431sd_remote_rename_impl(brix_sd_instance_t *inst, const char *src,
432 const char *dst, int noreplace,
433 const char *ak, const char *sk, const char *region, const char *session)
434{
435 const brix_sd_remote_cfg_t *cfg = inst->state;
436 sd_s3_open_params p;
437 char srckey[768];
438 char dstkey[768];
439 int has;
440
441 /* noreplace (POSC/O_EXCL): refuse if the destination file already exists. */
442 if (noreplace) {
443 sd_remote_s3_key(cfg, dst, dstkey, sizeof(dstkey));
444 sd_remote_s3_params(cfg, dstkey, &p);
445 sd_remote_params_cred(&p, ak, sk, region, session);
446 if (sd_remote_head_exists(&p)) {
447 errno = EEXIST;
448 return NGX_ERROR;
449 }
450 }
451
452 /* (1) regular file: HEAD src, then copy "src"->"dst" + delete "src". */
453 sd_remote_s3_key(cfg, src, srckey, sizeof(srckey));
454 sd_remote_s3_params(cfg, srckey, &p);
455 sd_remote_params_cred(&p, ak, sk, region, session);
456 if (sd_remote_head_exists(&p)) {
457 sd_remote_s3_key(cfg, dst, dstkey, sizeof(dstkey));
458 sd_remote_s3_params(cfg, dstkey, &p);
459 sd_remote_params_cred(&p, ak, sk, region, session);
460 return sd_remote_copy_then_delete(cfg, &p, srckey,
461 ak, sk, region, session);
462 }
463
464 /* (2) directory: a non-empty prefix has no atomic move. */
465 has = sd_remote_prefix_has_child(cfg, src, ak, sk, region, session);
466 if (has < 0) {
467 return NGX_ERROR;
468 }
469 if (has > 0) {
470 errno = ENOTSUP;
471 return NGX_ERROR;
472 }
473
474 /* Empty directory: move its marker "src/"->"dst/" if it exists. */
475 sd_remote_s3_dirkey(cfg, src, srckey, sizeof(srckey));
476 sd_remote_s3_params(cfg, srckey, &p);
477 sd_remote_params_cred(&p, ak, sk, region, session);
478 if (!sd_remote_head_exists(&p)) {
479 errno = ENOENT; /* neither file, nor children, nor a marker */
480 return NGX_ERROR;
481 }
482 sd_remote_s3_dirkey(cfg, dst, dstkey, sizeof(dstkey));
483 sd_remote_s3_params(cfg, dstkey, &p);
484 sd_remote_params_cred(&p, ak, sk, region, session);
485 return sd_remote_copy_then_delete(cfg, &p, srckey, ak, sk, region, session);
486}
