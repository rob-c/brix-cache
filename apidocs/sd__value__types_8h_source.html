1/*
2 * sd_value_types.h — the plain-value types a storage driver reads and fills:
3 * residency, space, catalog entries, stat, dirent, setattr. Split out of sd.h
4 * (which stays the slot table + instance/object model) to keep sd.h under the
5 * 600-line cap, the same cut that produced sd_accessors.h and sd_batch_types.h.
6 *
7 * Everything here is standard-C only (off_t/time_t/uint64_t/struct timespec) —
8 * no ngx types — so the header serves the XRDPROTO_NO_NGX libxrdproto plane
9 * unshimmed. Contract text lives with each type; sd.h's slot comments refer
10 * back here by type name.
11 */

20/* Residency of a nearline (tape/MSS) object — the online/offline model the VFS
21 * residency seam (brix_vfs_residency) exposes to protocol handlers so they can
22 * advertise tape state (the HTTP Tape REST API, S3 InvalidObjectState /
23 * x-amz-storage-class, root:// stat's nearline flag) WITHOUT forcing a recall. A
24 * non-nearline driver has no residency slot; the seam reports ONLINE for it (a
25 * plain disk/object export is always resident). */
26typedef enum {
27 BRIX_SD_RES_ONLINE = 0, /* resident in the online buffer, readable now */
28 BRIX_SD_RES_NEARLINE = 1, /* on the backend, stageable (a recall faults it in) */
29 BRIX_SD_RES_OFFLINE = 2, /* on the backend, not retrievable right now */
30 BRIX_SD_RES_LOST = 3 /* the object is gone */
31} brix_sd_residency_t;

33/* Driver space report (optional `space` slot, phase-83 F5): the backend's own
34 * view of total/used/free bytes for the export — quota-aware logical space for
35 * catalog backends (pblock), rather than the raw statvfs(2) of the filesystem
36 * under it. Consumers: kXR_statvfs, SRR reporting. */
37typedef struct {
38 uint64_t total_bytes;
39 uint64_t used_bytes;
40 uint64_t free_bytes;
41} brix_sd_space_t;

43/* One entry the catalog-enumeration verb (driver->enumerate) reports per stored
44 * backend object — independent of the namespace. `key` is the backend object key
45 * (always present). `path` is the logical path the driver recovered for it, or
46 * NULL when it cannot (⇒ an orphan-object candidate). size/mtime are valid only
47 * when have_stat (the enumerator was asked for stats and the per-object stat
48 * succeeded). All pointers are owned by the enumerator and valid only for the
49 * duration of the callback. */
50typedef struct {
51 const char *key;
52 const char *path;
53 int have_stat;
54 off_t size;
55 time_t mtime;
56} brix_sd_catalog_ent_t;

58/* Per-object callback fired by driver->enumerate. Return 0 to continue the
59 * enumeration, non-zero to abort it (that code is returned to the caller). */
60typedef int (*brix_sd_catalog_cb)(void *ctx, const brix_sd_catalog_ent_t *ent);

62/* Protocol-neutral stat the driver fills; the VFS maps it to brix_vfs_stat_t.
63 * uid/gid are the owner ids in the driver's own namespace (POSIX: kernel ids;
64 * pblock: catalog-internal synthetic ids); 0 for backends with no owner model. */
65typedef struct {
66 off_t size;
67 time_t mtime;
68 time_t ctime;
69 mode_t mode;
70 ino_t ino;
71 uid_t uid;
72 gid_t gid;
73 unsigned is_dir:1;
74 unsigned is_reg:1;
75} brix_sd_stat_t;

77/* One directory entry name (NUL-terminated). POSIX = a dirent name; object =
78 * the final path component synthesized from a key under the listing prefix.
79 * d_type is the entry's DT_* kind (dirent.h) when the backend can classify it
80 * cheaply, else DT_UNKNOWN (= 0, what a memzero'd entry reads) — consumers
81 * fall back to a stat; a backend must never guess. NEVER an authorization or
82 * confinement input (a spoofed d_type may only cost a fallback stat). */
83typedef struct {
84 char name[256];
85 unsigned char d_type;
86} brix_sd_dirent_t;

88/* Metadata-mutation request for the driver's setattr slot — the storage-neutral
89 * union of kXR_chmod (mode) and kXR_setattr (times + owner). Each set_* flag gates
90 * its field group; an unset group is left untouched. atime/mtime carry per-field
91 * UTIME_OMIT / UTIME_NOW in tv_nsec (utimensat(2) semantics). uid/gid of
92 * (uid_t)-1 / (gid_t)-1 leave that id unchanged. A driver applies what its
93 * namespace can represent (e.g. a catalog backend may not track owner/atime). */
94typedef struct {
95 unsigned set_mode:1;
96 unsigned set_times:1;
97 unsigned set_owner:1;
98 mode_t mode;
99 struct timespec atime;
100 struct timespec mtime;
101 uid_t uid;
102 gid_t gid;
103} brix_sd_setattr_t;
