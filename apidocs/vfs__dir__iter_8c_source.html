1/*
2 * vfs_dir_iter.c — VFS directory iteration (the readdir family).
3 *
4 * WHAT: Implements brix_vfs_readdir() (pooled-copy name), the zero-copy
5 * brix_vfs_readdir_borrow() (name borrowed from the handle, valid only
6 * until the next readdir/closedir on it), and brix_vfs_readdir_kind()
7 * (d_type classification, no per-entry stat) over the brix_vfs_dir_t
8 * handle that vfs_dir.c opens and closes.
9 *
10 * WHY: Every protocol listing (kXR_dirlist, WebDAV PROPFIND, S3 LIST) shares
11 * the "." / ".." filter, the errno-cleared end-of-stream signal, the
12 * skip-on-vanished-child scan, and the driver-plane fallback; and the
13 * hot single-pass consumer (kXR_dirlist streams each name into the wire
14 * chunk within the same iteration) must not pay a pooled copy per entry
15 * — brix_vfs_readdir was ngx_pnalloc+memcpy'ing every name into the
16 * CONNECTION pool, which both burned CPU and grew the pool per request.
17 *
18 * HOW: One borrowed-name core (vfs_readdir_next) yields the entry name in
19 * place — the POSIX dirent's own d_name, or the handle's de_scratch for
20 * a driver iterator — plus the optional folded child lstat; the classic
21 * entry point wraps it with the pooled copy, the borrow entry point
22 * hands the core's pointer straight through.
23 */
24#include "vfs_internal.h"
25#include "auth/impersonate/impersonate.h"

27/* vfs_readdir_fill_entry — copy an entry name into the pooled name_out.
28 *
29 * WHAT: Fills *name_out with a pooled, NUL-terminated copy of `name`.
30 * WHY: Both readdir variants, on both planes, end by handing the caller a
31 * pool-owned copy of the entry name; one helper kills the copy-paste.
32 * HOW: ngx_pnalloc(len+1) + memcpy + explicit NUL; NGX_ERROR with
33 * errno=ENOMEM on allocation failure. */
34static ngx_int_t
35vfs_readdir_fill_entry(ngx_pool_t *pool, const char *name,
36 ngx_str_t *name_out)
37{
38 name_out->len = ngx_strlen(name);
39 name_out->data = ngx_pnalloc(pool, name_out->len + 1);
40 if (name_out->data == NULL) {
41 errno = ENOMEM;
42 return NGX_ERROR;
43 }
44 ngx_memcpy(name_out->data, name, name_out->len);
45 name_out->data[name_out->len] = '\0';
46 return NGX_OK;
47}

49/* vfs_readdir_next_posix — next POSIX dirent, "." / ".." filtered out.
50 *
51 * WHAT: Yields the next non-dot entry of dh->dir into *de_out.
52 * WHY: Both readdir variants share the errno-cleared readdir(3) idiom that
53 * distinguishes end-of-stream from a real error, and the dot filter.
54 * HOW: errno=0 before each readdir(3): NULL with errno still 0 is NGX_DONE,
55 * NULL with errno set is NGX_ERROR; "." and ".." are skipped in-loop. */
56static ngx_int_t
57vfs_readdir_next_posix(brix_vfs_dir_t *dh, struct dirent **de_out)
58{
59 struct dirent *de;
60
61 for ( ;; ) {
62 errno = 0;
63 de = readdir(dh->dir);
64 if (de == NULL) {
65 return errno == 0 ? NGX_DONE : NGX_ERROR;
66 }
67
68 if (de->d_name[0] == '.'
69 && (de->d_name[1] == '\0'
70 || (de->d_name[1] == '.' && de->d_name[2] == '\0')))
71 {
72 continue;
73 }
74
75 *de_out = de;
76 return NGX_OK;
77 }
78}

80/* vfs_readdir_stat_child — confined lstat of one POSIX entry.
81 *
82 * WHAT: Joins "<dir>/<name>", lstats it under export confinement, and fills
83 * *stat_out on success.
84 * WHY: Per-entry stat must SKIP a bad entry rather than truncate the listing;
85 * returning NGX_ERROR here lets the scan loop drop the entry and go on.
86 * HOW: Off impersonation, a dirfd-relative fstatat of the bare entry name —
87 * O(1) per child, no join, no re-walk: dh->dir was opened through the
88 * RESOLVE_IN_ROOT-confined opendir, and readdir names contain no '/',
89 * so the child cannot escape the already-confined directory. Under
90 * impersonation, the snprintf join + brix_lstat_confined_canon AS THE
91 * MAPPED USER (broker-routed) is mandatory — mapped-user DAC — and is
92 * kept unchanged. lstat/AT_SYMLINK_NOFOLLOW keeps outward symlinks
93 * unfollowed on both paths. The caller skips a failed entry; only the
94 * benign unlink race (ENOENT) is silent — EIO/EACCES etc. are logged so
95 * a shrinking listing is diagnosable. */
96static ngx_int_t
97vfs_readdir_stat_child(brix_vfs_dir_t *dh, const char *name,
98 brix_vfs_stat_t *stat_out)
99{
100 char child[PATH_MAX];
101 struct stat st;
102 int n, rc;
103
104 if (!brix_imp_client_active()) {
105 int dfd = dirfd(dh->dir);
106
107 if (dfd < 0) {
108 /* Only reachable with a torn-down handle; fstatat would answer
109 * EBADF and the entry would vanish with a misleading errno. */
110 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
111 "xrootd[disk]: dirlist of \"%s\" has no directory "
112 "descriptor; entry \"%s\" omitted from the listing",
113 dh->path, name);
114 return NGX_ERROR;
115 }
116 rc = fstatat(dfd, name, &st, AT_SYMLINK_NOFOLLOW);
117 if (rc != 0) {
118 if (errno != ENOENT) {
119 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
120 "xrootd[disk]: dirlist stat of entry \"%s\" "
121 "under \"%s\" failed; entry omitted from the "
122 "listing", name, dh->path);
123 }
124 return NGX_ERROR;
125 }
126 brix_vfs_fill_stat(&st, stat_out);
127 return NGX_OK;
128 }
129
130 n = snprintf(child, sizeof(child), "%s/%s", dh->path, name);
131 if (n < 0 || (size_t) n >= sizeof(child)) {
132 errno = ENAMETOOLONG;
133 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
134 "xrootd[disk]: dirlist entry \"%s\" under \"%s\" joins "
135 "to an unrepresentable path; entry omitted",
136 name, dh->path);
137 return NGX_ERROR;
138 }
139 if (brix_lstat_confined_canon(dh->log, dh->root_canon, child,
140 &st, 1) != 0) {
141 if (errno != ENOENT) {
142 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
143 "xrootd[disk]: dirlist stat of \"%s\" failed; "
144 "entry omitted from the listing", child);
145 }
146 return NGX_ERROR;
147 }
148 brix_vfs_fill_stat(&st, stat_out);
149 return NGX_OK;
150}

152/* vfs_sd_stat_child — stat one driver-plane entry through the driver.
153 *
154 * WHAT: Joins the handle's logical directory with `name` and stats the child
155 * via dh->drv->stat into *sd_st.
156 * WHY: Both readdir variants build the same "<logical>/<name>" join (with
157 * the "/"-root special case) before the driver stat.
158 * HOW: Fixed PATH_MAX join via ngx_snprintf, then the driver's stat verb;
159 * the caller must have checked dh->drv->stat != NULL. */
160static ngx_int_t
161vfs_sd_join_child(const char *directory, const char *name, char *child,
162 size_t cap)
163{
164 int written;
165
166 written = snprintf(child, cap, "%s%s%s", directory,
167 directory[0] != '\0'
168 && directory[ngx_strlen(directory) - 1] == '/'
169 ? "" : "/", name);
170 if (written < 0 || (size_t) written >= cap) {
171 errno = ENAMETOOLONG;
172 return NGX_ERROR;
173 }
174 return NGX_OK;
175}

177/* vfs_sd_stat_child -- stat a driver entry by its physical child name.
178 *
179 * WHAT: Joins the handle's translated directory PFN with a raw driver entry
180 * name and stats that physical key. Returns the driver result.
181 * WHY: Once N2N is active, joining the visible LFN would address a different
182 * key. Driver work must remain physical until the listing is rendered.
183 * HOW: Use the checked fixed-buffer join, then invoke the driver's stat slot. */
184static ngx_int_t
185vfs_sd_stat_child(brix_vfs_dir_t *dh, const char *name, brix_sd_stat_t *sd_st)
186{
187 char child[PATH_MAX];
188
189 if (vfs_sd_join_child(dh->sd_physical, name, child, sizeof(child))
190 != NGX_OK)
191 {
192 return NGX_ERROR;
193 }
194 return dh->drv->stat(dh->sd, child, sd_st);
195}

197/* vfs_sd_render_entry -- turn one raw driver child into its visible LFN name.
198 *
199 * WHAT: Reconstructs the entry's full PFN, reverses the ctx-bound N2N mapping,
200 * validates that the result is an immediate child of the opened logical
201 * directory, and overwrites `entry->name` with that basename.
202 * WHY: Protocols must never expose configured pools/prefixes, and a malformed
203 * backend listing must not escape or change directory level while being
204 * reversed.
205 * HOW: Checked PFN join, shared PFN->LFN translator, exact parent-prefix
206 * check, slash-free basename check, then a capacity-checked copy. */
207static ngx_int_t
208vfs_sd_render_entry(brix_vfs_dir_t *dh, brix_sd_dirent_t *entry)
209{
210 char physical[PATH_MAX];
211 char logical[PATH_MAX];
212 const char *basename;
213 size_t parent_len;
214 size_t name_len;
215
216 if (vfs_sd_join_child(dh->sd_physical, entry->name, physical,
217 sizeof(physical)) != NGX_OK
218 || brix_path_pfn_to_lfn(dh->ctx, physical, logical,
219 sizeof(logical)) != NGX_OK)
220 {
221 return NGX_ERROR;
222 }
223
224 parent_len = ngx_strlen(dh->sd_logical);
225 if (parent_len == 1 && dh->sd_logical[0] == '/') {
226 basename = logical[0] == '/' ? logical + 1 : NULL;
227 } else if (ngx_strncmp(logical, dh->sd_logical, parent_len) == 0
228 && logical[parent_len] == '/')
229 {
230 basename = logical + parent_len + 1;
231 } else {
232 basename = NULL;
233 }
234 if (basename == NULL || basename[0] == '\0' || ngx_strchr(basename, '/')) {
235 errno = EIO;
236 return NGX_ERROR;
237 }
238 name_len = ngx_strlen(basename);
239 if (name_len >= sizeof(entry->name)) {
240 errno = ENAMETOOLONG;
241 return NGX_ERROR;
242 }
243 ngx_memcpy(entry->name, basename, name_len + 1);
244 return NGX_OK;
245}

247/* vfs_readdir_sd — driver-plane body of the readdir core.
248 *
249 * WHAT: Pulls the next entry from the driver iterator into the HANDLE's
250 * dirent scratch, stats the child through the same driver when the
251 * caller wants stats, and yields the scratch name borrowed.
252 * WHY: Keeps the driver iterator's skip-on-vanished-child scan out of the
253 * core; the scratch lives on the handle (not the stack) so the borrow
254 * contract — name valid until the next readdir/closedir — holds on the
255 * driver plane exactly as it does for a POSIX dirent.
256 * HOW: Loops (instead of the former tail-recursion) so a child that vanished
257 * mid-scan is skipped; only NGX_DONE/NGX_ERROR from the iterator stop
258 * the caller's loop. */
259static ngx_int_t
260vfs_readdir_sd(brix_vfs_dir_t *dh, const char **name_out,
261 brix_vfs_stat_t *stat_out)
262{
263 brix_sd_dirent_t *de_sd = &dh->de_scratch;
264 brix_sd_stat_t sd_st;
265 ngx_int_t rc;
266
267 for ( ;; ) {
268 /* Zeroed so a driver that fills only the name yields d_type ==
269 * DT_UNKNOWN (= 0), never scratch garbage from the previous entry. */
270 ngx_memzero(de_sd, sizeof(*de_sd));
271 rc = dh->drv->readdir(dh->sd_dir, de_sd);
272 if (rc != NGX_OK) {
273 return rc; /* NGX_DONE / NGX_ERROR */
274 }
275 if (stat_out == NULL || dh->drv->stat == NULL) {
276 if (vfs_sd_render_entry(dh, de_sd) == NGX_OK) {
277 break;
278 }
279 continue;
280 }
281 if (vfs_sd_stat_child(dh, de_sd->name, &sd_st) == NGX_OK) {
282 if (vfs_sd_render_entry(dh, de_sd) == NGX_OK) {
283 brix_vfs_sd_stat_fill(&sd_st, stat_out);
284 break;
285 }
286 continue;
287 }
288 /* Child vanished mid-scan (ENOENT): skip it and keep enumerating.
289 * Any other driver stat failure is surfaced before the skip. */
290 if (errno != ENOENT) {
291 ngx_log_error(NGX_LOG_ERR, dh->log, errno,
292 "xrootd[disk]: dirlist driver stat of entry \"%s\" "
293 "under \"%s\" failed; entry omitted from the "
294 "listing", de_sd->name, dh->sd_logical);
295 }
296 }
297
298 *name_out = de_sd->name;
299 return NGX_OK;
300}

302/* vfs_readdir_next — the borrowed-name core both public variants share.
303 *
304 * WHAT: Yields the next visible entry's name IN PLACE (no copy) plus the
305 * optional folded child lstat.
306 * WHY: The pooled-copy and zero-copy entry points differ ONLY in how the
307 * name leaves the function; everything else — handle guard, plane
308 * dispatch, dot filter, skip-on-bad-entry stat fold — is this core.
309 * HOW: POSIX: the dirent's own d_name (valid until the next readdir(3) on
310 * dh->dir). Driver plane: the handle's de_scratch (valid until the next
311 * core call). Per-entry stat is folded into the scan so a single bad
312 * entry SKIPS rather than truncating the listing: a child that races an
313 * unlink (ENOENT), or whose joined path is unrepresentable, is dropped
314 * and the scan continues; only NGX_DONE/NGX_ERROR stop the caller. */
315static ngx_int_t
316vfs_readdir_next(brix_vfs_dir_t *dh, const char **name_out,
317 brix_vfs_stat_t *stat_out)
318{
319 struct dirent *de;
320 ngx_int_t rc;
321
322 if (dh == NULL || (dh->dir == NULL && dh->sd_dir == NULL)) {
323 errno = EINVAL;
324 return NGX_ERROR;
325 }
326
327 if (dh->sd_dir != NULL) {
328 return vfs_readdir_sd(dh, name_out, stat_out);
329 }
330
331 for ( ;; ) {
332 rc = vfs_readdir_next_posix(dh, &de);
333 if (rc != NGX_OK) {
334 return rc;
335 }
336 if (stat_out == NULL
337 || vfs_readdir_stat_child(dh, de->d_name, stat_out) == NGX_OK)
338 {
339 break;
340 }
341 }
342
343 *name_out = de->d_name;
344 return NGX_OK;
345}

392/* Yield the next entry's name plus its KIND from the readdir d_type, with no
393 * per-entry stat — for callers that classify dir-vs-file on the fast path and
394 * only stat (via brix_vfs_probe) on a DT_UNKNOWN filesystem. Skips "."/"..";
395 * NGX_DONE at end-of-stream, NGX_ERROR (errno set) on failure. */
396/* vfs_posix_dtype_kind — map a readdir d_type onto the VFS entry kind.
397 *
398 * WHAT: DT_DIR/DT_REG/DT_UNKNOWN map onto their VFS kinds; anything else
399 * (symlink, fifo, socket, ...) is DT_OTHER.
400 * WHY: Callers classify dir-vs-file on the fast path and only stat on a
401 * DT_UNKNOWN filesystem; the mapping is a pure table.
402 * HOW: A single switch — no side effects. */
403static brix_vfs_dirent_kind_t
404vfs_posix_dtype_kind(unsigned char d_type)
405{
406 switch (d_type) {
407 case DT_DIR: return BRIX_VFS_DT_DIR;
408 case DT_REG: return BRIX_VFS_DT_REG;
409 case DT_UNKNOWN: return BRIX_VFS_DT_UNKNOWN;
410 default: return BRIX_VFS_DT_OTHER;
411 }
412}

414/* vfs_sd_entry_kind — classify one driver-plane entry as dir/file.
415 *
416 * WHAT: Returns the entry's kind, preferring the dirent's own d_type and
417 * falling back to a driver stat of the joined child.
418 * WHY: A backend that classifies during enumeration (POSIX readdir) makes
419 * the per-entry stat redundant; one that cannot leaves d_type ==
420 * DT_UNKNOWN (never guessed) and the stat verb decides. Kind is
421 * display/routing metadata only — NEVER an authorization input.
422 * HOW: Non-UNKNOWN d_type maps through the same pure table as the POSIX
423 * plane; on UNKNOWN, guard dh->drv->stat, join via vfs_sd_stat_child,
424 * and map is_dir → DT_DIR / DT_REG. */
425static brix_vfs_dirent_kind_t
426vfs_sd_entry_kind(brix_vfs_dir_t *dh, const brix_sd_dirent_t *de)
427{
428 brix_sd_stat_t sd_st;
429
430 if (de->d_type != DT_UNKNOWN) {
431 return vfs_posix_dtype_kind(de->d_type);
432 }
433
434 if (dh->drv->stat == NULL
435 || vfs_sd_stat_child(dh, de->name, &sd_st) != NGX_OK)
436 {
437 return BRIX_VFS_DT_UNKNOWN;
438 }
439 return sd_st.is_dir ? BRIX_VFS_DT_DIR : BRIX_VFS_DT_REG;
440}
