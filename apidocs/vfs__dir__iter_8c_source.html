1/*
2 * vfs_dir_iter.c — VFS directory iteration (the readdir family).
3 *
4 * WHAT: Implements brix_vfs_readdir() (pooled-copy name), the zero-copy
5 * brix_vfs_readdir_borrow() (name borrowed from the handle, valid only
6 * until the next readdir/closedir on it), and brix_vfs_readdir_kind()
7 * (d_type classification, no per-entry stat) over the brix_vfs_dir_t
8 * handle that vfs_dir.c opens and closes.
9 *
10 * WHY: Every protocol listing (kXR_dirlist, WebDAV PROPFIND, S3 LIST) shares
11 * the "." / ".." filter, the errno-cleared end-of-stream signal, the
12 * skip-on-vanished-child scan, and the driver-plane fallback; and the
13 * hot single-pass consumer (kXR_dirlist streams each name into the wire
14 * chunk within the same iteration) must not pay a pooled copy per entry
15 * — brix_vfs_readdir was ngx_pnalloc+memcpy'ing every name into the
16 * CONNECTION pool, which both burned CPU and grew the pool per request.
17 *
18 * HOW: One borrowed-name core (vfs_readdir_next) yields the entry name in
19 * place — the POSIX dirent's own d_name, or the handle's de_scratch for
20 * a driver iterator — plus the optional folded child lstat; the classic
21 * entry point wraps it with the pooled copy, the borrow entry point
22 * hands the core's pointer straight through.
23 */

28/* vfs_readdir_fill_entry — copy an entry name into the pooled name_out.
29 *
30 * WHAT: Fills *name_out with a pooled, NUL-terminated copy of `name`.
31 * WHY: Both readdir variants, on both planes, end by handing the caller a
32 * pool-owned copy of the entry name; one helper kills the copy-paste.
33 * HOW: ngx_pnalloc(len+1) + memcpy + explicit NUL; NGX_ERROR with
34 * errno=ENOMEM on allocation failure. */
35static ngx_int_t
36vfs_readdir_fill_entry(ngx_pool_t *pool, const char *name,
37 ngx_str_t *name_out)
38{
39 name_out->len = ngx_strlen(name);
40 name_out->data = ngx_pnalloc(pool, name_out->len + 1);
41 if (name_out->data == NULL) {
42 errno = ENOMEM;
43 return NGX_ERROR;
44 }
45 ngx_memcpy(name_out->data, name, name_out->len);
46 name_out->data[name_out->len] = '\0';
47 return NGX_OK;
48}

50/* vfs_readdir_next_posix — next POSIX dirent, "." / ".." filtered out.
51 *
52 * WHAT: Yields the next non-dot entry of dh->dir into *de_out.
53 * WHY: Both readdir variants share the errno-cleared readdir(3) idiom that
54 * distinguishes end-of-stream from a real error, and the dot filter.
55 * HOW: errno=0 before each readdir(3): NULL with errno still 0 is NGX_DONE,
56 * NULL with errno set is NGX_ERROR; "." and ".." are skipped in-loop. */
57static ngx_int_t
58vfs_readdir_next_posix(brix_vfs_dir_t *dh, struct dirent **de_out)
59{
60 struct dirent *de;
61
62 for ( ;; ) {
63 errno = 0;
64 de = readdir(dh->dir);
65 if (de == NULL) {
66 return errno == 0 ? NGX_DONE : NGX_ERROR;
67 }
68
69 if (de->d_name[0] == '.'
70 && (de->d_name[1] == '\0'
71 || (de->d_name[1] == '.' && de->d_name[2] == '\0')))
72 {
73 continue;
74 }
75
76 *de_out = de;
77 return NGX_OK;
78 }
79}

81/* vfs_readdir_stat_child — confined lstat of one POSIX entry.
82 *
83 * WHAT: Joins "<dir>/<name>", lstats it under export confinement, and fills
84 * *stat_out on success.
85 * WHY: Per-entry stat must SKIP a bad entry rather than truncate the listing;
86 * returning NGX_ERROR here lets the scan loop drop the entry and go on.
87 * HOW: Off impersonation, a dirfd-relative fstatat of the bare entry name —
88 * O(1) per child, no join, no re-walk: dh->dir was opened through the
89 * RESOLVE_IN_ROOT-confined opendir, and readdir names contain no '/',
90 * so the child cannot escape the already-confined directory. Under
91 * impersonation, the snprintf join + brix_lstat_confined_canon AS THE
92 * MAPPED USER (broker-routed) is mandatory — mapped-user DAC — and is
93 * kept unchanged. lstat/AT_SYMLINK_NOFOLLOW keeps outward symlinks
94 * unfollowed on both paths. The caller skips a failed entry; only the
95 * benign unlink race (ENOENT) is silent — EIO/EACCES etc. are logged so
96 * a shrinking listing is diagnosable. */
97static ngx_int_t
98vfs_readdir_stat_child(brix_vfs_dir_t *dh, const char *name,
99 brix_vfs_stat_t *stat_out)
100{
101 char child[PATH_MAX];
102 struct stat st;
103 int n, rc;
104
105 if (!brix_imp_client_active()) {
106 int dfd = dirfd(dh->dir);
107
108 if (dfd < 0) {
109 /* Only reachable with a torn-down handle; fstatat would answer
110 * EBADF and the entry would vanish with a misleading errno. */
111 BRIX_LOG(NGX_LOG_ERR, dh->log, errno,
112 "xrootd[disk]: dirlist of \"%s\" has no directory "
113 "descriptor; entry \"%s\" omitted from the listing",
114 dh->path, name);
115 return NGX_ERROR;
116 }
117 rc = fstatat(dfd, name, &st, AT_SYMLINK_NOFOLLOW);
118 if (rc != 0) {
119 if (errno != ENOENT) {
120 BRIX_LOG(NGX_LOG_ERR, dh->log, errno,
121 "xrootd[disk]: dirlist stat of entry \"%s\" "
122 "under \"%s\" failed; entry omitted from the "
123 "listing", name, dh->path);
124 }
125 return NGX_ERROR;
126 }
127 brix_vfs_fill_stat(&st, stat_out);
128 return NGX_OK;
129 }
130
131 n = snprintf(child, sizeof(child), "%s/%s", dh->path, name);
132 if (n < 0 || (size_t) n >= sizeof(child)) {
133 errno = ENAMETOOLONG;
134 BRIX_LOG(NGX_LOG_ERR, dh->log, errno,
135 "xrootd[disk]: dirlist entry \"%s\" under \"%s\" joins "
136 "to an unrepresentable path; entry omitted",
137 name, dh->path);
138 return NGX_ERROR;
139 }
140 if (brix_lstat_confined_canon(dh->log, dh->root_canon, child,
141 &st, 1) != 0) {
142 if (errno != ENOENT) {
143 BRIX_LOG(NGX_LOG_ERR, dh->log, errno,
144 "xrootd[disk]: dirlist stat of \"%s\" failed; "
145 "entry omitted from the listing", child);
146 }
147 return NGX_ERROR;
148 }
149 brix_vfs_fill_stat(&st, stat_out);
150 return NGX_OK;
151}

153/* vfs_sd_stat_child — stat one driver-plane entry through the driver.
154 *
155 * WHAT: Joins the handle's logical directory with `name` and stats the child
156 * via dh->drv->stat into *sd_st.
157 * WHY: Both readdir variants build the same "<logical>/<name>" join (with
158 * the "/"-root special case) before the driver stat.
159 * HOW: Fixed PATH_MAX join via ngx_snprintf, then the driver's stat verb;
160 * the caller must have checked dh->drv->stat != NULL. */
161static ngx_int_t
162vfs_sd_join_child(const char *directory, const char *name, char *child,
163 size_t cap)
164{
165 int written;
166
167 written = snprintf(child, cap, "%s%s%s", directory,
168 directory[0] != '\0'
169 && directory[ngx_strlen(directory) - 1] == '/'
170 ? "" : "/", name);
171 if (written < 0 || (size_t) written >= cap) {
172 errno = ENAMETOOLONG;
173 return NGX_ERROR;
174 }
175 return NGX_OK;
176}

178/* vfs_sd_stat_child -- stat a driver entry by its physical child name.
179 *
180 * WHAT: Joins the handle's translated directory PFN with a raw driver entry
181 * name and stats that physical key. Returns the driver result.
182 * WHY: Once N2N is active, joining the visible LFN would address a different
183 * key. Driver work must remain physical until the listing is rendered.
184 * HOW: Use the checked fixed-buffer join, then invoke the driver's stat slot. */
185static ngx_int_t
186vfs_sd_stat_child(brix_vfs_dir_t *dh, const char *name, brix_sd_stat_t *sd_st)
187{
188 char child[PATH_MAX];
189
190 if (vfs_sd_join_child(dh->sd_physical, name, child, sizeof(child))
191 != NGX_OK)
192 {
193 return NGX_ERROR;
194 }
195 return dh->drv->stat(dh->sd, child, sd_st);
196}

198/* vfs_sd_render_entry -- turn one raw driver child into its visible LFN name.
199 *
200 * WHAT: Reconstructs the entry's full PFN, reverses the handle's N2N mapping,
201 * validates that the result is an immediate child of the opened logical
202 * directory, and overwrites `entry->name` with that basename.
203 * WHY: Protocols must never expose configured pools/prefixes, and a malformed
204 * backend listing must not escape or change directory level while being
205 * reversed.
206 * HOW: Checked PFN join, shared PFN->LFN translator, exact parent-prefix
207 * check, slash-free basename check, then a capacity-checked copy. */
208static ngx_int_t
209vfs_sd_render_entry(brix_vfs_dir_t *dh, brix_sd_dirent_t *entry)
210{
211 char physical[PATH_MAX];
212 char logical[PATH_MAX];
213 const char *basename;
214 size_t parent_len;
215 size_t name_len;
216
217 if (vfs_sd_join_child(dh->sd_physical, entry->name, physical,
218 sizeof(physical)) != NGX_OK
219 || brix_path_cfg_pfn_to_lfn(dh->n2n, physical, logical,
220 sizeof(logical)) != NGX_OK)
221 {
222 return NGX_ERROR;
223 }
224
225 parent_len = ngx_strlen(dh->sd_logical);
226 if (parent_len == 1 && dh->sd_logical[0] == '/') {
227 basename = logical[0] == '/' ? logical + 1 : NULL;
228 } else if (ngx_strncmp(logical, dh->sd_logical, parent_len) == 0
229 && logical[parent_len] == '/')
230 {
231 basename = logical + parent_len + 1;
232 } else {
233 basename = NULL;
234 }
235 if (basename == NULL || basename[0] == '\0' || ngx_strchr(basename, '/')) {
236 errno = EIO;
237 return NGX_ERROR;
238 }
239 name_len = ngx_strlen(basename);
240 if (name_len >= sizeof(entry->name)) {
241 errno = ENAMETOOLONG;
242 return NGX_ERROR;
243 }
244 ngx_memcpy(entry->name, basename, name_len + 1);
245 return NGX_OK;
246}

248/* vfs_readdir_sd — driver-plane body of the readdir core.
249 *
250 * WHAT: Pulls the next entry from the driver iterator into the HANDLE's
251 * dirent scratch, stats the child through the same driver when the
252 * caller wants stats, and yields the scratch name borrowed.
253 * WHY: Keeps the driver iterator's skip-on-vanished-child scan out of the
254 * core; the scratch lives on the handle (not the stack) so the borrow
255 * contract — name valid until the next readdir/closedir — holds on the
256 * driver plane exactly as it does for a POSIX dirent.
257 * HOW: Loops (instead of the former tail-recursion) so a child that vanished
258 * mid-scan is skipped; only NGX_DONE/NGX_ERROR from the iterator stop
259 * the caller's loop. */
260static ngx_int_t
261vfs_readdir_sd(brix_vfs_dir_t *dh, const char **name_out,
262 brix_vfs_stat_t *stat_out)
263{
264 brix_sd_dirent_t *de_sd = &dh->de_scratch;
265 brix_sd_stat_t sd_st;
266 ngx_int_t rc;
267
268 for ( ;; ) {
269 /* Zeroed so a driver that fills only the name yields d_type ==
270 * DT_UNKNOWN (= 0), never scratch garbage from the previous entry. */
271 ngx_memzero(de_sd, sizeof(*de_sd));
272 rc = dh->drv->readdir(dh->sd_dir, de_sd);
273 if (rc != NGX_OK) {
274 return rc; /* NGX_DONE / NGX_ERROR */
275 }
276 if (stat_out == NULL || dh->drv->stat == NULL) {
277 if (vfs_sd_render_entry(dh, de_sd) == NGX_OK) {
278 break;
279 }
280 continue;
281 }
282 if (vfs_sd_stat_child(dh, de_sd->name, &sd_st) == NGX_OK) {
283 if (vfs_sd_render_entry(dh, de_sd) == NGX_OK) {
284 brix_vfs_sd_stat_fill(&sd_st, stat_out);
285 break;
286 }
287 continue;
288 }
289 /* Child vanished mid-scan (ENOENT): skip it and keep enumerating.
290 * Any other driver stat failure is surfaced before the skip. */
291 if (errno != ENOENT) {
292 BRIX_LOG(NGX_LOG_ERR, dh->log, errno,
293 "xrootd[disk]: dirlist driver stat of entry \"%s\" "
294 "under \"%s\" failed; entry omitted from the "
295 "listing", de_sd->name, dh->sd_logical);
296 }
297 }
298
299 *name_out = de_sd->name;
300 return NGX_OK;
301}

303/* vfs_readdir_next — the borrowed-name core both public variants share.
304 *
305 * WHAT: Yields the next visible entry's name IN PLACE (no copy) plus the
306 * optional folded child lstat.
307 * WHY: The pooled-copy and zero-copy entry points differ ONLY in how the
308 * name leaves the function; everything else — handle guard, plane
309 * dispatch, dot filter, skip-on-bad-entry stat fold — is this core.
310 * HOW: POSIX: the dirent's own d_name (valid until the next readdir(3) on
311 * dh->dir). Driver plane: the handle's de_scratch (valid until the next
312 * core call). Per-entry stat is folded into the scan so a single bad
313 * entry SKIPS rather than truncating the listing: a child that races an
314 * unlink (ENOENT), or whose joined path is unrepresentable, is dropped
315 * and the scan continues; only NGX_DONE/NGX_ERROR stop the caller. */
316static ngx_int_t
317vfs_readdir_next(brix_vfs_dir_t *dh, const char **name_out,
318 brix_vfs_stat_t *stat_out)
319{
320 struct dirent *de;
321 ngx_int_t rc;
322
323 if (dh == NULL || (dh->dir == NULL && dh->sd_dir == NULL)) {
324 errno = EINVAL;
325 return NGX_ERROR;
326 }
327
328 if (dh->sd_dir != NULL) {
329 return vfs_readdir_sd(dh, name_out, stat_out);
330 }
331
332 for ( ;; ) {
333 rc = vfs_readdir_next_posix(dh, &de);
334 if (rc != NGX_OK) {
335 return rc;
336 }
337 if (stat_out == NULL
338 || vfs_readdir_stat_child(dh, de->d_name, stat_out) == NGX_OK)
339 {
340 break;
341 }
342 }
343
344 *name_out = de->d_name;
345 return NGX_OK;
346}

393/* Yield the next entry's name plus its KIND from the readdir d_type, with no
394 * per-entry stat — for callers that classify dir-vs-file on the fast path and
395 * only stat (via brix_vfs_probe) on a DT_UNKNOWN filesystem. Skips "."/"..";

397/* vfs_posix_dtype_kind — map a readdir d_type onto the VFS entry kind.
398 *
399 * WHAT: DT_DIR/DT_REG/DT_UNKNOWN map onto their VFS kinds; anything else
400 * (symlink, fifo, socket, ...) is DT_OTHER.
401 * WHY: Callers classify dir-vs-file on the fast path and only stat on a
402 * DT_UNKNOWN filesystem; the mapping is a pure table.
403 * HOW: A single switch — no side effects. */
404static brix_vfs_dirent_kind_t
405vfs_posix_dtype_kind(unsigned char d_type)
406{
407 switch (d_type) {
408 case DT_DIR: return BRIX_VFS_DT_DIR;
409 case DT_REG: return BRIX_VFS_DT_REG;
410 case DT_UNKNOWN: return BRIX_VFS_DT_UNKNOWN;
411 default: return BRIX_VFS_DT_OTHER;
412 }
413}

415/* vfs_sd_entry_kind — classify one driver-plane entry as dir/file.
416 *
417 * WHAT: Returns the entry's kind, preferring the dirent's own d_type and
418 * falling back to a driver stat of the joined child.
419 * WHY: A backend that classifies during enumeration (POSIX readdir) makes
420 * the per-entry stat redundant; one that cannot leaves d_type ==
421 * DT_UNKNOWN (never guessed) and the stat verb decides. Kind is
422 * display/routing metadata only — NEVER an authorization input.
423 * HOW: Non-UNKNOWN d_type maps through the same pure table as the POSIX
424 * plane; on UNKNOWN, guard dh->drv->stat, join via vfs_sd_stat_child,
425 * and map is_dir → DT_DIR / DT_REG. */
426static brix_vfs_dirent_kind_t
427vfs_sd_entry_kind(brix_vfs_dir_t *dh, const brix_sd_dirent_t *de)
428{
429 brix_sd_stat_t sd_st;
430
431 if (de->d_type != DT_UNKNOWN) {
432 return vfs_posix_dtype_kind(de->d_type);
433 }
434
435 if (dh->drv->stat == NULL
436 || vfs_sd_stat_child(dh, de->name, &sd_st) != NGX_OK)
437 {
438 return BRIX_VFS_DT_UNKNOWN;
439 }
440 return sd_st.is_dir ? BRIX_VFS_DT_DIR : BRIX_VFS_DT_REG;
441}
