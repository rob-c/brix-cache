1/* dict.c — the phase-87 G3 trained shared-dictionary endpoint + wire coding.
2 *
3 * WHAT: GET/HEAD /cvmfs/<repo>/.cvmfs-dict/(current|<40-hex id>) (gated
4 * brix_cvmfs_dict, default off): lazily train a zstd dictionary per
5 * worker from a bounded sample of this repo's CACHE-RESIDENT CAS
6 * objects and serve it with its self-certifying id (X-Brix-Dict-Id =
7 * sha1 of the dict bytes). CAS GETs that carry a matching X-Brix-Dict
8 * request header may then be answered dict-coded (Content-Encoding:
9 * zstd-dict) when that is strictly smaller than identity.
10 * WHY: CVMFS repos are millions of SMALL similar objects (catalogs,
11 * manifests, headers) — per-object compression has nothing to seed
12 * from, but a dictionary trained on the corpus does. The coding is a
13 * reversible transform of the STORED bytes, so the client's CAS
14 * verify runs on exactly what it always ran on: a wrong or hostile
15 * dictionary can only FAIL decode (zstd embeds/checks the dictID),
16 * never emit wrong bytes — the client falls back to identity.
17 * HOW: training samples CACHE-RESIDENT objects only, enumerated through the
18 * cache tier's own adapter (brix_cstore_scan over the decorator's
19 * cstore — the sd chain's opendir forwards to the ORIGIN, which cannot
20 * list): COMPLETE objects under this repo's data/ of (0, 64KiB] are
21 * read into a bounded arena, then ZDICT-trained (shared/cvmfs/dict).
22 * The trained dict is cached per worker on lcf (COW after fork — no
23 * SHM, no new globals); a failed training is memoized with a
24 * retry-after so a cold or tiny repo cannot be used to hammer the
25 * sampler. Every failure path is fail-open to identity serving.
26 */

41#define CVMFS_DICT_REPO_SLOTS 4 /* per-worker trained repos (LRU-free:
42 first-come; extra repos serve
43 identity — a cache serves O(few)
44 hot repos in practice) */

72/* Find (or, when `create`, claim) the worker-private slot for this repo.
73 * NULL when the table is full or the name is oversize — callers treat that
74 * as "no dictionary" and serve identity. */
75static cvmfs_dict_entry_t *
76cvmfs_dict_slot(ngx_http_brix_cvmfs_loc_conf_t *lcf, const char *repo,
77 size_t repo_len, int create)
78{
79 cvmfs_dict_state_t *st = lcf->cvmfs.dict_state;
80 ngx_uint_t i;
81
82 if (repo_len == 0 || repo_len >= CVMFS_DICT_REPO_NAME_MAX) {
83 return NULL;
84 }
85 if (st == NULL) {
86 if (!create) {
87 return NULL;
88 }
89 st = calloc(1, sizeof(*st));
90 if (st == NULL) {
91 return NULL;
92 }
93 lcf->cvmfs.dict_state = st;
94 }
95 for (i = 0; i < CVMFS_DICT_REPO_SLOTS; i++) {
96 if (st->repos[i].used
97 && strncmp(st->repos[i].repo, repo, repo_len) == 0
98 && st->repos[i].repo[repo_len] == '\0')
99 {
100 return &st->repos[i];
101 }
102 }
103 if (!create) {
104 return NULL;
105 }
106 for (i = 0; i < CVMFS_DICT_REPO_SLOTS; i++) {
107 if (!st->repos[i].used) {
108 memcpy(st->repos[i].repo, repo, repo_len);
109 st->repos[i].repo[repo_len] = '\0';
110 st->repos[i].used = 1;
111 return &st->repos[i];
112 }
113 }
114 return NULL;
115}

128/* cstore visitor: read one COMPLETE resident CAS object of this repo into
129 * the arena. Every skip is silent (never fatal); early-stops the scan once
130 * the sample cap is reached. */
131static ngx_int_t
132cvmfs_dict_sample_visit(const char *key, const brix_cache_cinfo_t *ci,
133 const brix_sd_stat_t *stx, void *ud)
134{
135 cvmfs_dict_sample_ctx_t *sc = ud;
136 const char *p = key;
137 brix_sd_obj_t *obj;
138 size_t want, off;
139 int verr = 0;
140
141 if (ci == NULL || !(ci->flags & BRIX_CINFO_F_COMPLETE)) {
142 return NGX_OK; /* orphan/partial — never sample */
143 }
144 if (*p == '/') {
145 p++;
146 }
147 if (strncmp(p, "cvmfs/", sizeof("cvmfs/") - 1) != 0) {
148 return NGX_OK;
149 }
150 p += sizeof("cvmfs/") - 1;
151 if (strncmp(p, sc->repo, sc->repo_len) != 0
152 || strncmp(p + sc->repo_len, "/data/", sizeof("/data/") - 1) != 0)
153 {
154 return NGX_OK; /* other repo / non-CAS entry */
155 }
156 if (stx == NULL
157 || stx->size <= 0
158 || (size_t) stx->size > CVMFS_DICT_SAMPLE_MAX
159 || (size_t) stx->size > CVMFS_DICT_ARENA_MAX - sc->arena_used)
160 {
161 return NGX_OK;
162 }
163 want = (size_t) stx->size;
164
165 obj = brix_cstore_serve_open(sc->cs, key, &verr);
166 if (obj == NULL) {
167 return NGX_OK; /* vanished under us — skip */
168 }
169 for (off = 0; off < want; ) {
170 ssize_t rd = obj->driver->pread(obj, sc->arena + sc->arena_used + off,
171 want - off, (off_t) off);
172 if (rd <= 0) {
173 brix_sd_obj_release(obj);
174 return NGX_OK; /* short read — discard sample */
175 }
176 off += (size_t) rd;
177 }
178 brix_sd_obj_release(obj);
179
180 sc->sizes[sc->n++] = want;
181 sc->arena_used += want;
182 return (sc->n >= CVMFS_DICT_TRAIN_SAMPLES) ? NGX_DONE : NGX_OK;
183}

185/* Train this repo's dictionary from resident CAS samples. On success the
186 * entry holds malloc'd dict bytes + id; on any failure the entry stays
187 * untrained with a retry-after memo. All scratch is request-pool, so an
188 * aborted scan leaks nothing past the request. */
189static void
190cvmfs_dict_train_repo(ngx_http_request_t *r,
191 ngx_http_brix_cvmfs_loc_conf_t *lcf, const char *repo, size_t repo_len,
192 cvmfs_dict_entry_t *e)
193{
194 brix_sd_instance_t *sd;
195 brix_cstore_t *cs;
196 unsigned char *arena, *dict_tmp;
197 size_t sizes[CVMFS_DICT_TRAIN_SAMPLES];
198 size_t dict_len = 0;
199 size_t arena_used;
200 unsigned n;
201 cvmfs_dict_sample_ctx_t sc;
202
203 e->retry_at = ngx_time() + CVMFS_DICT_RETRY_SEC; /* assume failure */
204
205 sd = cvmfs_resolve_sd(r, lcf);
206 cs = (sd != NULL) ? brix_sd_cache_cstore(sd) : NULL;
207 if (cs == NULL) {
208 BRIX_LOG(NGX_LOG_INFO, r->connection->log, 0,
209 "cvmfs-dict: repo=%*s not trained (no cache tier to sample)",
210 repo_len, repo);
211 return;
212 }
213
214 arena = ngx_pnalloc(r->pool, CVMFS_DICT_ARENA_MAX);
215 dict_tmp = ngx_pnalloc(r->pool, CVMFS_DICT_TARGET_BYTES);
216 if (arena == NULL || dict_tmp == NULL) {
217 return;
218 }
219
220 ngx_memzero(&sc, sizeof(sc));
221 sc.cs = cs;
222 sc.repo = repo;
223 sc.repo_len = repo_len;
224 sc.arena = arena;
225 sc.sizes = sizes;
226 (void) brix_cstore_scan(cs, cvmfs_dict_sample_visit, &sc);
227 n = sc.n;
228 arena_used = sc.arena_used;
229
230 if (n < CVMFS_DICT_TRAIN_MIN) {
231 BRIX_LOG(NGX_LOG_INFO, r->connection->log, 0,
232 "cvmfs-dict: repo=%*s not trained (resident samples=%ui < %ui)",
233 repo_len, repo, (ngx_uint_t) n, (ngx_uint_t) CVMFS_DICT_TRAIN_MIN);
234 return;
235 }
236
237 if (cvmfs_dict_train(arena, sizes, n, dict_tmp,
238 CVMFS_DICT_TARGET_BYTES, &dict_len) != 0
239 || cvmfs_dict_id(dict_tmp, dict_len, e->id) != 0)
240 {
241 BRIX_LOG(NGX_LOG_INFO, r->connection->log, 0,
242 "cvmfs-dict: repo=%*s training failed (samples=%ui bytes=%uz)",
243 repo_len, repo, (ngx_uint_t) n, arena_used);
244 return;
245 }
246
247 e->bytes = malloc(dict_len);
248 if (e->bytes == NULL) {
249 return;
250 }
251 memcpy(e->bytes, dict_tmp, dict_len);
252 e->len = dict_len;
253 e->retry_at = 0;
254
255 BRIX_LOG(NGX_LOG_INFO, r->connection->log, 0,
256 "cvmfs-dict: repo=%*s trained id=%s bytes=%uz samples=%ui "
257 "corpus=%uz", repo_len, repo, e->id, dict_len, (ngx_uint_t) n,
258 arena_used);
259}

261/* Send `body` (worker-lifetime memory) as a 200 with the dict headers. */
262static ngx_int_t
263cvmfs_dict_send(ngx_http_request_t *r, cvmfs_dict_entry_t *e)
264{
265 ngx_chain_t out;
266 ngx_buf_t *b;
267 ngx_int_t rc;
268
269 r->headers_out.status = NGX_HTTP_OK;
270 r->headers_out.content_length_n = (off_t) e->len;
271 ngx_str_set(&r->headers_out.content_type, "application/octet-stream");
272 r->headers_out.content_type_len = r->headers_out.content_type.len;
273 if (brix_http_set_header(r, "X-Brix-Dict-Id", e->id, NULL) != NGX_OK) {
274 return NGX_HTTP_INTERNAL_SERVER_ERROR;
275 }
276
277 b = ngx_calloc_buf(r->pool);
278 if (b == NULL) {
279 return NGX_HTTP_INTERNAL_SERVER_ERROR;
280 }
281 b->memory = 1;
282 b->pos = e->bytes;
283 b->last = e->bytes + e->len;
284 b->start = b->pos;
285 b->end = b->last;
286 b->last_buf = 1;
287 b->last_in_chain = 1;
288 out.buf = b;
289 out.next = NULL;
290
291 rc = ngx_http_send_header(r);
292 if (rc == NGX_ERROR || rc > NGX_OK || r->header_only) {
293 return rc;
294 }
295 return ngx_http_output_filter(r, &out);
296}

298ngx_int_t
299brix_cvmfs_dict_handle(ngx_http_request_t *r,
300 ngx_http_brix_cvmfs_loc_conf_t *lcf)
301{
302 ngx_http_brix_cvmfs_ctx_t *ctx =
303 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
304 cvmfs_dict_entry_t *e;
305 const char *id;
306 size_t id_len, i;
307 int want_current;
308
309 if (ctx == NULL) {
310 return NGX_HTTP_INTERNAL_SERVER_ERROR;
311 }
312
313 /* rel = ".cvmfs-dict/<id>"; <id> is "current" or a 40-hex dict id */
314 id = ctx->url.rel + sizeof(".cvmfs-dict/") - 1;
315 id_len = ctx->url.rel_len - (sizeof(".cvmfs-dict/") - 1);
316 want_current = (id_len == sizeof("current") - 1
317 && memcmp(id, "current", id_len) == 0);
318 if (!want_current) {
319 if (id_len != CVMFS_DICT_ID_HEXLEN) {
320 return NGX_HTTP_BAD_REQUEST;
321 }
322 for (i = 0; i < id_len; i++) {
323 if (!cvmfs_dict_hexlc(id[i])) {
324 return NGX_HTTP_BAD_REQUEST;
325 }
326 }
327 }
328
329 e = cvmfs_dict_slot(lcf, ctx->url.repo, ctx->url.repo_len, 1);
330 if (e == NULL) {
331 return NGX_HTTP_NOT_FOUND;
332 }
333
334 if (e->len == 0) {
335 if (ngx_time() < e->retry_at) {
336 return NGX_HTTP_NOT_FOUND; /* memoized failure — no retrain */
337 }
338 cvmfs_dict_train_repo(r, lcf, ctx->url.repo, ctx->url.repo_len, e);
339 if (e->len == 0) {
340 return NGX_HTTP_NOT_FOUND;
341 }
342 }
343
344 if (!want_current
345 && ngx_memcmp(id, e->id, CVMFS_DICT_ID_HEXLEN) != 0)
346 {
347 return NGX_HTTP_NOT_FOUND; /* a superseded/foreign dict id */
348 }
349
350 return cvmfs_dict_send(r, e);
351}

370brix_cvmfs_dict_try_serve(ngx_http_request_t *r,
371 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
372 brix_vfs_file_t *fh)
373{
374 cvmfs_dict_entry_t *e;
375 brix_vfs_stat_t vst;
376 ngx_str_t want;
377 unsigned char *src, *out;
378 size_t srclen, outlen = 0;
379
380 if (!lcf->cvmfs.dict
381 || r->method != NGX_HTTP_GET
382 || ctx == NULL
383 || ctx->url.cls != CVMFS_URL_CAS
384 || r->headers_in.range != NULL)

391 return NGX_DECLINED;
392 }

394 e = cvmfs_dict_slot(lcf, ctx->url.repo, ctx->url.repo_len, 0);
395 if (e == NULL || e->len == 0
396 || ngx_memcmp(want.data, e->id, CVMFS_DICT_ID_HEXLEN) != 0)
397 {
398 return NGX_DECLINED; /* unknown/superseded id → identity serve */
399 }
400
401 if (brix_cvmfs_coding_load(r, fh, &vst, CVMFS_DICT_MAX_OBJ, &srclen, &src, &out) != 0) {
402 return NGX_DECLINED;
403 }
404
405 if (cvmfs_dict_compress(e->bytes, e->len, src, srclen, out, srclen,
406 &outlen) != 0)
407 {
408 return NGX_DECLINED; /* no gain (or error) → identity serve */
409 }
410
411 /* commit: from here the response is ours and fh is consumed */
412 brix_vfs_close(fh, r->connection->log);
413
414 return cvmfs_dict_emit(r, vst.mtime, e->id, out, srclen, outlen);
415}
