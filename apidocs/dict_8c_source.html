1/* dict.c — the phase-87 G3 trained shared-dictionary endpoint + wire coding.
2 *
3 * WHAT: GET/HEAD /cvmfs/<repo>/.cvmfs-dict/(current|<40-hex id>) (gated
4 * brix_cvmfs_dict, default off): lazily train a zstd dictionary per
5 * worker from a bounded sample of this repo's CACHE-RESIDENT CAS
6 * objects and serve it with its self-certifying id (X-Brix-Dict-Id =
7 * sha1 of the dict bytes). CAS GETs that carry a matching X-Brix-Dict
8 * request header may then be answered dict-coded (Content-Encoding:
9 * zstd-dict) when that is strictly smaller than identity.
10 * WHY: CVMFS repos are millions of SMALL similar objects (catalogs,
11 * manifests, headers) — per-object compression has nothing to seed
12 * from, but a dictionary trained on the corpus does. The coding is a
13 * reversible transform of the STORED bytes, so the client's CAS
14 * verify runs on exactly what it always ran on: a wrong or hostile
15 * dictionary can only FAIL decode (zstd embeds/checks the dictID),
16 * never emit wrong bytes — the client falls back to identity.
17 * HOW: training samples CACHE-RESIDENT objects only, enumerated through the
18 * cache tier's own adapter (brix_cstore_scan over the decorator's
19 * cstore — the sd chain's opendir forwards to the ORIGIN, which cannot
20 * list): COMPLETE objects under this repo's data/ of (0, 64KiB] are
21 * read into a bounded arena, then ZDICT-trained (shared/cvmfs/dict).
22 * The trained dict is cached per worker on lcf (COW after fork — no
23 * SHM, no new globals); a failed training is memoized with a
24 * retry-after so a cold or tiny repo cannot be used to hammer the
25 * sampler. Every failure path is fail-open to identity serving.
26 */

70/* Find (or, when `create`, claim) the worker-private slot for this repo.
71 * NULL when the table is full or the name is oversize — callers treat that
72 * as "no dictionary" and serve identity. */
73static cvmfs_dict_entry_t *
74cvmfs_dict_slot(ngx_http_brix_cvmfs_loc_conf_t *lcf, const char *repo,
75 size_t repo_len, int create)
76{
77 cvmfs_dict_state_t *st = lcf->cvmfs.dict_state;
78 ngx_uint_t i;
79
80 if (repo_len == 0 || repo_len >= CVMFS_DICT_REPO_NAME_MAX) {
81 return NULL;
82 }
83 if (st == NULL) {
84 if (!create) {
85 return NULL;
86 }
87 st = calloc(1, sizeof(*st));
88 if (st == NULL) {
89 return NULL;
90 }
91 lcf->cvmfs.dict_state = st;
92 }
93 for (i = 0; i < CVMFS_DICT_REPO_SLOTS; i++) {
94 if (st->repos[i].used
95 && strncmp(st->repos[i].repo, repo, repo_len) == 0
96 && st->repos[i].repo[repo_len] == '\0')
97 {
98 return &st->repos[i];
99 }
100 }
101 if (!create) {
102 return NULL;
103 }
104 for (i = 0; i < CVMFS_DICT_REPO_SLOTS; i++) {
105 if (!st->repos[i].used) {
106 memcpy(st->repos[i].repo, repo, repo_len);
107 st->repos[i].repo[repo_len] = '\0';
108 st->repos[i].used = 1;
109 return &st->repos[i];
110 }
111 }
112 return NULL;
113}

126/* cstore visitor: read one COMPLETE resident CAS object of this repo into
127 * the arena. Every skip is silent (never fatal); early-stops the scan once
128 * the sample cap is reached. */
129static ngx_int_t
130cvmfs_dict_sample_visit(const char *key, const brix_cache_cinfo_t *ci,
131 const brix_sd_stat_t *stx, void *ud)
132{
133 cvmfs_dict_sample_ctx_t *sc = ud;
134 const char *p = key;
135 brix_sd_obj_t *obj;
136 size_t want, off;
137 int verr = 0;
138
139 if (ci == NULL || !(ci->flags & BRIX_CINFO_F_COMPLETE)) {
140 return NGX_OK; /* orphan/partial — never sample */
141 }
142 if (*p == '/') {
143 p++;
144 }
145 if (strncmp(p, "cvmfs/", sizeof("cvmfs/") - 1) != 0) {
146 return NGX_OK;
147 }
148 p += sizeof("cvmfs/") - 1;
149 if (strncmp(p, sc->repo, sc->repo_len) != 0
150 || strncmp(p + sc->repo_len, "/data/", sizeof("/data/") - 1) != 0)
151 {
152 return NGX_OK; /* other repo / non-CAS entry */
153 }
154 if (stx == NULL
155 || stx->size <= 0
156 || (size_t) stx->size > CVMFS_DICT_SAMPLE_MAX
157 || (size_t) stx->size > CVMFS_DICT_ARENA_MAX - sc->arena_used)
158 {
159 return NGX_OK;
160 }
161 want = (size_t) stx->size;
162
163 obj = brix_cstore_serve_open(sc->cs, key, &verr);
164 if (obj == NULL) {
165 return NGX_OK; /* vanished under us — skip */
166 }
167 for (off = 0; off < want; ) {
168 ssize_t rd = obj->driver->pread(obj, sc->arena + sc->arena_used + off,
169 want - off, (off_t) off);
170 if (rd <= 0) {
171 brix_sd_obj_release(obj);
172 return NGX_OK; /* short read — discard sample */
173 }
174 off += (size_t) rd;
175 }
176 brix_sd_obj_release(obj);
177
178 sc->sizes[sc->n++] = want;
179 sc->arena_used += want;
180 return (sc->n >= CVMFS_DICT_TRAIN_SAMPLES) ? NGX_DONE : NGX_OK;
181}

183/* Train this repo's dictionary from resident CAS samples. On success the
184 * entry holds malloc'd dict bytes + id; on any failure the entry stays
185 * untrained with a retry-after memo. All scratch is request-pool, so an
186 * aborted scan leaks nothing past the request. */
187static void
188cvmfs_dict_train_repo(ngx_http_request_t *r,
189 ngx_http_brix_cvmfs_loc_conf_t *lcf, const char *repo, size_t repo_len,
190 cvmfs_dict_entry_t *e)
191{
192 brix_sd_instance_t *sd;
193 brix_cstore_t *cs;
194 unsigned char *arena, *dict_tmp;
195 size_t sizes[CVMFS_DICT_TRAIN_SAMPLES];
196 size_t dict_len = 0;
197 size_t arena_used;
198 unsigned n;
199 cvmfs_dict_sample_ctx_t sc;
200
201 e->retry_at = ngx_time() + CVMFS_DICT_RETRY_SEC; /* assume failure */
202
203 sd = cvmfs_resolve_sd(r, lcf);
204 cs = (sd != NULL) ? brix_sd_cache_cstore(sd) : NULL;
205 if (cs == NULL) {
206 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
207 "cvmfs-dict: repo=%*s not trained (no cache tier to sample)",
208 repo_len, repo);
209 return;
210 }
211
212 arena = ngx_pnalloc(r->pool, CVMFS_DICT_ARENA_MAX);
213 dict_tmp = ngx_pnalloc(r->pool, CVMFS_DICT_TARGET_BYTES);
214 if (arena == NULL || dict_tmp == NULL) {
215 return;
216 }
217
218 ngx_memzero(&sc, sizeof(sc));
219 sc.cs = cs;
220 sc.repo = repo;
221 sc.repo_len = repo_len;
222 sc.arena = arena;
223 sc.sizes = sizes;
224 (void) brix_cstore_scan(cs, cvmfs_dict_sample_visit, &sc);
225 n = sc.n;
226 arena_used = sc.arena_used;
227
228 if (n < CVMFS_DICT_TRAIN_MIN) {
229 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
230 "cvmfs-dict: repo=%*s not trained (resident samples=%ui < %ui)",
231 repo_len, repo, (ngx_uint_t) n, (ngx_uint_t) CVMFS_DICT_TRAIN_MIN);
232 return;
233 }
234
235 if (cvmfs_dict_train(arena, sizes, n, dict_tmp,
236 CVMFS_DICT_TARGET_BYTES, &dict_len) != 0
237 || cvmfs_dict_id(dict_tmp, dict_len, e->id) != 0)
238 {
239 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
240 "cvmfs-dict: repo=%*s training failed (samples=%ui bytes=%uz)",
241 repo_len, repo, (ngx_uint_t) n, arena_used);
242 return;
243 }
244
245 e->bytes = malloc(dict_len);
246 if (e->bytes == NULL) {
247 return;
248 }
249 memcpy(e->bytes, dict_tmp, dict_len);
250 e->len = dict_len;
251 e->retry_at = 0;
252
253 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
254 "cvmfs-dict: repo=%*s trained id=%s bytes=%uz samples=%ui "
255 "corpus=%uz", repo_len, repo, e->id, dict_len, (ngx_uint_t) n,
256 arena_used);
257}

296ngx_int_t
297brix_cvmfs_dict_handle(ngx_http_request_t *r,
298 ngx_http_brix_cvmfs_loc_conf_t *lcf)
299{
300 ngx_http_brix_cvmfs_ctx_t *ctx =
301 ngx_http_get_module_ctx(r, ngx_http_brix_cvmfs_module);
302 cvmfs_dict_entry_t *e;
303 const char *id;
304 size_t id_len, i;
305 int want_current;
306
307 if (ctx == NULL) {
308 return NGX_HTTP_INTERNAL_SERVER_ERROR;
309 }
310
311 /* rel = ".cvmfs-dict/<id>"; <id> is "current" or a 40-hex dict id */
312 id = ctx->url.rel + sizeof(".cvmfs-dict/") - 1;
313 id_len = ctx->url.rel_len - (sizeof(".cvmfs-dict/") - 1);
314 want_current = (id_len == sizeof("current") - 1
315 && memcmp(id, "current", id_len) == 0);
316 if (!want_current) {
317 if (id_len != CVMFS_DICT_ID_HEXLEN) {
318 return NGX_HTTP_BAD_REQUEST;
319 }
320 for (i = 0; i < id_len; i++) {
321 if (!cvmfs_dict_hexlc(id[i])) {
322 return NGX_HTTP_BAD_REQUEST;
323 }
324 }
325 }
326
327 e = cvmfs_dict_slot(lcf, ctx->url.repo, ctx->url.repo_len, 1);
328 if (e == NULL) {
329 return NGX_HTTP_NOT_FOUND;
330 }
331
332 if (e->len == 0) {
333 if (ngx_time() < e->retry_at) {
334 return NGX_HTTP_NOT_FOUND; /* memoized failure — no retrain */
335 }
336 cvmfs_dict_train_repo(r, lcf, ctx->url.repo, ctx->url.repo_len, e);
337 if (e->len == 0) {
338 return NGX_HTTP_NOT_FOUND;
339 }
340 }
341
342 if (!want_current
343 && ngx_memcmp(id, e->id, CVMFS_DICT_ID_HEXLEN) != 0)
344 {
345 return NGX_HTTP_NOT_FOUND; /* a superseded/foreign dict id */
346 }
347
348 return cvmfs_dict_send(r, e);
349}

351/* Stat the target and read it into fresh request-pool buffers (`*out` is
352 * capped at srclen: the coded serve must WIN or compress fails).
353 * -1 = identity serve. */
354static int
355cvmfs_dict_load(ngx_http_request_t *r, brix_vfs_file_t *fh,
356 brix_vfs_stat_t *vst, size_t *srclen, unsigned char **src,
357 unsigned char **out)
358{
359 size_t off, n;
360
361 if (brix_vfs_file_stat(fh, vst) != NGX_OK
362 || vst->is_directory
363 || vst->size <= 0
364 || (size_t) vst->size > CVMFS_DICT_MAX_OBJ)
365 {
366 return -1;
367 }
368 n = (size_t) vst->size;
369
370 *src = ngx_pnalloc(r->pool, n);
371 *out = ngx_pnalloc(r->pool, n);
372 if (*src == NULL || *out == NULL) {
373 return -1;
374 }
375 for (off = 0; off < n; ) {
376 ssize_t rd = brix_vfs_file_pread(fh, *src + off, n - off, (off_t) off);
377 if (rd <= 0) {
378 return -1;
379 }
380 off += (size_t) rd;
381 }
382 *srclen = n;
383 return 0;
384}
