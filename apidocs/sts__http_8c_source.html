1/*
2 * sts_http.c — STS transport + response parsing (phase-79 split).
3 *
4 * WHAT: fetch the signed STS URL over http/https with libcurl and parse the
5 * AssumeRole/GetSessionToken XML reply into AccessKeyId / SecretAccessKey
6 * / SessionToken. Returns the HTTP status to the caller and the extracted
7 * credentials in the caller's fixed buffers.
8 * WHY: split out of sts.c (formerly 815 lines) as the side-effecting edge —
9 * network I/O and XML parsing — kept apart from the pure request builder
10 * (sts_sign.c) and the orchestrator (sts.c). Behaviour is byte-for-byte
11 * the pre-split code: same curl hardening, same bounded body, same
12 * name-directed XML descent, same log strings (secrets never logged).
13 * HOW: sts_http_get mirrors webdav/tpc_curl.c — TLS verified, protocols pinned
14 * to http/https, body streamed into a BRIX_STS_RESP_MAX-bounded buffer.
15 * sts_parse_response walks the libxml2 tree for the credential elements;
16 * a no-libxml2 build fails closed with an explicit message.
17 *
18 * The reused building blocks:
19 * - outbound HTTP: libcurl, mirroring src/protocols/webdav/tpc_curl.c
20 * - XML parsing: libxml2 (guarded by BRIX_HAVE_LIBXML2)
21 */

36/*
37 * libcurl write callback: append the response body into a growable pool buffer,
38 * bounded by BRIX_STS_RESP_MAX so a hostile/huge response cannot exhaust memory.
39 * Returns 0 (short write → curl aborts) on overflow or OOM.
40 */
41static size_t
42sts_write_cb(void *ptr, size_t size, size_t nmemb, void *userdata)
43{
44 sts_resp_t *r = userdata;
45 size_t n = size * nmemb;
46
47 if (n == 0) {
48 return 0;
49 }
50 if (r->len + n > BRIX_STS_RESP_MAX) {
51 return 0;
52 }
53 ngx_memcpy(r->buf + r->len, ptr, n);
54 r->len += n;
55 return n;
56}

59/*
60 * sts_http_get — GET the signed STS URL and capture the response body.
61 *
62 * Restricts the transfer to http/https, enforces TLS verification, and streams
63 * the body into `resp` (a pre-sized pool buffer). Returns NGX_OK with the HTTP
64 * status in *http_status, or NGX_ERROR on a transport error (logged, no
65 * secrets). Mirrors the curl setup in webdav_tpc_run_curl_core().
66 */
67ngx_int_t
68sts_http_get(const char *url, sts_resp_t *resp, long *http_status,
69 ngx_log_t *log)
70{
71 CURL *curl;
72 CURLcode res;
73 char errbuf[CURL_ERROR_SIZE];
74 ngx_int_t rc = NGX_ERROR;
75
76 curl = curl_easy_init();
77 if (curl == NULL) {
78 ngx_log_error(NGX_LOG_ERR, log, 0, "brix_sts: curl_easy_init failed");
79 return NGX_ERROR;
80 }
81
82 errbuf[0] = '\0';
83 curl_easy_setopt(curl, CURLOPT_ERRORBUFFER, errbuf);
84 curl_easy_setopt(curl, CURLOPT_NOSIGNAL, 1L);
85 curl_easy_setopt(curl, CURLOPT_URL, url);
86#if CURL_AT_LEAST_VERSION(7, 85, 0)
87 curl_easy_setopt(curl, CURLOPT_PROTOCOLS_STR, "http,https");
88#else
89 curl_easy_setopt(curl, CURLOPT_PROTOCOLS,
90 (long) (CURLPROTO_HTTP | CURLPROTO_HTTPS));
91#endif
92 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);
93 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYHOST, 2L);
94 curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, 10L);
95 curl_easy_setopt(curl, CURLOPT_TIMEOUT, 30L);
96 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, sts_write_cb);
97 curl_easy_setopt(curl, CURLOPT_WRITEDATA, resp);
98
99 res = curl_easy_perform(curl);
100 if (res == CURLE_OK) {
101 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, http_status);
102 rc = NGX_OK;
103 } else {
104 ngx_log_error(NGX_LOG_WARN, log, 0, "brix_sts: STS request failed: %s",
105 errbuf[0] ? errbuf : curl_easy_strerror(res));
106 }
107
108 curl_easy_cleanup(curl);
109 return rc;
110}

113/*
114 * sts_http_post — POST the MinIO-dialect AssumeRole request with header-auth.
115 *
116 * Same curl hardening as sts_http_get (http/https only, TLS verified, bounded
117 * body), but sends `pd->body` with the four signed headers plus Authorization.
118 * The Host header is set explicitly so it matches the signed authority exactly
119 * (curl's default omits a non-default port from Host, which would break SigV4).
120 * Returns NGX_OK with the status in *http_status, or NGX_ERROR (logged, no
121 * secrets — the Authorization signature is a derived MAC, not a secret).
122 */
123ngx_int_t
124sts_http_post(const char *url, const char *host, const sts_post_t *pd,
125 sts_resp_t *resp, long *http_status, ngx_log_t *log)
126{
127 CURL *curl;
128 CURLcode res;
129 struct curl_slist *hdrs = NULL;
130 char errbuf[CURL_ERROR_SIZE];
131 char line[1200];
132 ngx_int_t rc = NGX_ERROR;
133
134 curl = curl_easy_init();
135 if (curl == NULL) {
136 ngx_log_error(NGX_LOG_ERR, log, 0, "brix_sts: curl_easy_init failed");
137 return NGX_ERROR;
138 }
139
140 /* Every one of the four SignedHeaders must ride on the wire with the exact
141 * signed value, or the origin recomputes a different signature and rejects. */
142 ngx_snprintf((u_char *) line, sizeof(line), "Host: %s%Z", host);
143 hdrs = curl_slist_append(hdrs, line);
144 hdrs = curl_slist_append(hdrs, "Content-Type: " BRIX_STS_POST_CTYPE);
145 ngx_snprintf((u_char *) line, sizeof(line),
146 "x-amz-content-sha256: %s%Z", pd->content_sha256);
147 hdrs = curl_slist_append(hdrs, line);
148 ngx_snprintf((u_char *) line, sizeof(line), "x-amz-date: %s%Z",
149 pd->amzdate);
150 hdrs = curl_slist_append(hdrs, line);
151 ngx_snprintf((u_char *) line, sizeof(line), "Authorization: %s%Z",
152 pd->authorization);
153 hdrs = curl_slist_append(hdrs, line);
154
155 if (hdrs == NULL) {
156 curl_easy_cleanup(curl);
157 return NGX_ERROR;
158 }
159
160 errbuf[0] = '\0';
161 curl_easy_setopt(curl, CURLOPT_ERRORBUFFER, errbuf);
162 curl_easy_setopt(curl, CURLOPT_NOSIGNAL, 1L);
163 curl_easy_setopt(curl, CURLOPT_URL, url);
164 curl_easy_setopt(curl, CURLOPT_POST, 1L);
165 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, pd->body);
166 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE, (long) ngx_strlen(pd->body));
167 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, hdrs);
168#if CURL_AT_LEAST_VERSION(7, 85, 0)
169 curl_easy_setopt(curl, CURLOPT_PROTOCOLS_STR, "http,https");
170#else
171 curl_easy_setopt(curl, CURLOPT_PROTOCOLS,
172 (long) (CURLPROTO_HTTP | CURLPROTO_HTTPS));
173#endif
174 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);
175 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYHOST, 2L);
176 curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, 10L);
177 curl_easy_setopt(curl, CURLOPT_TIMEOUT, 30L);
178 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, sts_write_cb);
179 curl_easy_setopt(curl, CURLOPT_WRITEDATA, resp);
180
181 res = curl_easy_perform(curl);
182 if (res == CURLE_OK) {
183 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, http_status);
184 rc = NGX_OK;
185 } else {
186 ngx_log_error(NGX_LOG_WARN, log, 0, "brix_sts: STS request failed: %s",
187 errbuf[0] ? errbuf : curl_easy_strerror(res));
188 }
189
190 curl_slist_free_all(hdrs);
191 curl_easy_cleanup(curl);
192 return rc;
193}

202/*
203 * Recursively search the parse tree for the first element named `name` and copy
204 * its text content into out[outsz]. Returns 1 if found, 0 otherwise. The STS
205 * AssumeRole/GetSessionToken responses nest <Credentials> under result and
206 * response wrappers, so a name-directed descent is simpler than a fixed path.
207 */
208static int
209sts_xml_find(xmlNodePtr node, const char *name, char *out, size_t outsz)
210{
211 xmlNodePtr n;
212
213 for (n = node; n != NULL; n = n->next) {
214 if (n->type == XML_ELEMENT_NODE
215 && xmlStrcmp(n->name, (const xmlChar *) name) == 0)
216 {
217 xmlChar *txt = xmlNodeGetContent(n);
218 if (txt != NULL) {
219 ngx_snprintf((u_char *) out, outsz, "%s%Z", (char *) txt);
220 xmlFree(txt);
221 return 1;
222 }
223 }
224 if (n->children != NULL
225 && sts_xml_find(n->children, name, out, outsz))
226 {
227 return 1;
228 }
229 }
230 return 0;
231}

234/*
235 * sts_parse_response — extract AccessKeyId / SecretAccessKey / SessionToken
236 * from the STS XML body into the caller's fixed buffers. Returns NGX_OK when at
237 * least AccessKeyId and SecretAccessKey are present; NGX_ERROR (with the fault
238 * logged, secrets excluded) otherwise. SessionToken is optional.
239 */
240ngx_int_t
241sts_parse_response(const u_char *body, size_t len,
242 const sts_creds_buf_t *creds, ngx_log_t *log)
243{
244 xmlDocPtr doc;
245 xmlNodePtr root;
246 ngx_int_t rc = NGX_ERROR;
247
248 creds->ak[0] = creds->sk[0] = creds->session[0] = '\0';
249
250 doc = xmlReadMemory((const char *) body, (int) len, "sts.xml", NULL,
251 XML_PARSE_NONET | XML_PARSE_NOERROR | XML_PARSE_NOWARNING);
252 if (doc == NULL) {
253 ngx_log_error(NGX_LOG_ERR, log, 0,
254 "brix_sts: unparseable STS response");
255 return NGX_ERROR;
256 }
257
258 root = xmlDocGetRootElement(doc);
259 if (root != NULL
260 && sts_xml_find(root, "AccessKeyId", creds->ak, creds->aksz)
261 && sts_xml_find(root, "SecretAccessKey", creds->sk, creds->sksz)
262 && creds->ak[0] != '\0' && creds->sk[0] != '\0')
263 {
264 (void) sts_xml_find(root, "SessionToken", creds->session,
265 creds->sesssz);
266 rc = NGX_OK;
267 } else {
268 ngx_log_error(NGX_LOG_ERR, log, 0,
269 "brix_sts: STS response missing credential fields");
270 }
271
272 xmlFreeDoc(doc);
273 return rc;
274}
