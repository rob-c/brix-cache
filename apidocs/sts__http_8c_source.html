1/*
2 * sts_http.c — STS transport + response parsing (phase-79 split).
3 *
4 * WHAT: fetch the signed STS URL over http/https with libcurl and parse the
5 * AssumeRole/GetSessionToken XML reply into AccessKeyId / SecretAccessKey
6 * / SessionToken. Returns the HTTP status to the caller and the extracted
7 * credentials in the caller's fixed buffers.
8 * WHY: split out of sts.c (formerly 815 lines) as the side-effecting edge —
9 * network I/O and XML parsing — kept apart from the pure request builder
10 * (sts_sign.c) and the orchestrator (sts.c). Behaviour is byte-for-byte
11 * the pre-split code: same curl hardening, same bounded body, same
12 * name-directed XML descent, same log strings (secrets never logged).
13 * HOW: sts_http_get mirrors webdav/tpc_curl.c — TLS verified, protocols pinned
14 * to http/https, body streamed into a BRIX_STS_RESP_MAX-bounded buffer.
15 * sts_parse_response walks the libxml2 tree for the credential elements;
16 * a no-libxml2 build fails closed with an explicit message.
17 *
18 * The reused building blocks:
19 * - outbound HTTP: libcurl, mirroring src/protocols/webdav/tpc_curl.c
20 * - XML parsing: libxml2 (guarded by BRIX_HAVE_LIBXML2)
21 */

39/*
40 * libcurl write callback: append the response body into a growable pool buffer,
41 * bounded by BRIX_STS_RESP_MAX so a hostile/huge response cannot exhaust memory.
42 * Returns 0 (short write → curl aborts) on overflow or OOM.
43 */
44static size_t
45sts_write_cb(void *ptr, size_t size, size_t nmemb, void *userdata)
46{
47 sts_resp_t *r = userdata;
48 size_t n = size * nmemb;
49
50 if (n == 0) {
51 return 0;
52 }
53 if (r->len + n > BRIX_STS_RESP_MAX) {
54 return 0;
55 }
56 ngx_memcpy(r->buf + r->len, ptr, n);
57 r->len += n;
58 return n;
59}

62/*
63 * sts_http_get — GET the signed STS URL and capture the response body.
64 *
65 * Restricts the transfer to http/https, enforces TLS verification, and streams
66 * the body into `resp` (a pre-sized pool buffer). Returns NGX_OK with the HTTP
67 * status in *http_status, or NGX_ERROR on a transport error (logged, no
68 * secrets). Mirrors the curl setup in webdav_tpc_run_curl_core().
69 */
70ngx_int_t
71sts_http_get(const struct brix_dns_policy_s *dns, const char *url,
72 sts_resp_t *resp, long *http_status, ngx_log_t *log)
73{
74 CURL *curl;
75 CURLcode res;
76 struct curl_slist *resolve = NULL;
77 char errbuf[CURL_ERROR_SIZE];
78 char reason[BRIX_DNS_ERROR_LEN];
79 ngx_int_t rc = NGX_ERROR;
80
81 curl = curl_easy_init();
82 if (curl == NULL) {
83 ngx_log_error(NGX_LOG_ERR, log, 0, "brix_sts: curl_easy_init failed");
84 return NGX_ERROR;
85 }
86
87 errbuf[0] = '\0';
88 curl_easy_setopt(curl, CURLOPT_ERRORBUFFER, errbuf);
89 curl_easy_setopt(curl, CURLOPT_NOSIGNAL, 1L);
90 curl_easy_setopt(curl, CURLOPT_URL, url);
91 if (brix_dns_curl_pin(curl, dns, url, &resolve, reason, sizeof(reason))
92 != NGX_OK)
93 {
94 ngx_log_error(NGX_LOG_WARN, log, 0,
95 "brix_sts: cannot resolve STS endpoint: %s", reason);
96 curl_easy_cleanup(curl);
97 return NGX_ERROR;
98 }
99#if CURL_AT_LEAST_VERSION(7, 85, 0)
100 curl_easy_setopt(curl, CURLOPT_PROTOCOLS_STR, "http,https");
101#else
102 curl_easy_setopt(curl, CURLOPT_PROTOCOLS,
103 (long) (CURLPROTO_HTTP | CURLPROTO_HTTPS));
104#endif
105 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);
106 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYHOST, 2L);
107 curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, 10L);
108 curl_easy_setopt(curl, CURLOPT_TIMEOUT, 30L);
109 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, sts_write_cb);
110 curl_easy_setopt(curl, CURLOPT_WRITEDATA, resp);
111
112 res = curl_easy_perform(curl);
113 if (res == CURLE_OK) {
114 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, http_status);
115 rc = NGX_OK;
116 } else {
117 ngx_log_error(NGX_LOG_WARN, log, 0, "brix_sts: STS request failed: %s",
118 errbuf[0] ? errbuf : curl_easy_strerror(res));
119 }
120
121 curl_slist_free_all(resolve);
122 curl_easy_cleanup(curl);
123 return rc;
124}

127/*
128 * sts_http_post — POST the MinIO-dialect AssumeRole request with header-auth.
129 *
130 * Same curl hardening as sts_http_get (http/https only, TLS verified, bounded
131 * body), but sends `pd->body` with the four signed headers plus Authorization.
132 * The Host header is set explicitly so it matches the signed authority exactly
133 * (curl's default omits a non-default port from Host, which would break SigV4).
134 * Returns NGX_OK with the status in *http_status, or NGX_ERROR (logged, no
135 * secrets — the Authorization signature is a derived MAC, not a secret).
136 */
137ngx_int_t
138sts_http_post(const struct brix_dns_policy_s *dns, const char *url,
139 const char *host, const sts_post_t *pd, sts_resp_t *resp,
140 long *http_status, ngx_log_t *log)
141{
142 CURL *curl;
143 CURLcode res;
144 struct curl_slist *hdrs = NULL;
145 struct curl_slist *resolve = NULL;
146 char errbuf[CURL_ERROR_SIZE];
147 char reason[BRIX_DNS_ERROR_LEN];
148 char line[BRIX_STS_HTTP_LINE_BUF_SIZE];
149 ngx_int_t rc = NGX_ERROR;
150
151 curl = curl_easy_init();
152 if (curl == NULL) {
153 ngx_log_error(NGX_LOG_ERR, log, 0, "brix_sts: curl_easy_init failed");
154 return NGX_ERROR;
155 }
156
157 /* Every one of the four SignedHeaders must ride on the wire with the exact
158 * signed value, or the origin recomputes a different signature and rejects. */
159 ngx_snprintf((u_char *) line, sizeof(line), "Host: %s%Z", host);
160 hdrs = curl_slist_append(hdrs, line);
161 hdrs = curl_slist_append(hdrs, "Content-Type: " BRIX_STS_POST_CTYPE);
162 ngx_snprintf((u_char *) line, sizeof(line),
163 "x-amz-content-sha256: %s%Z", pd->content_sha256);
164 hdrs = curl_slist_append(hdrs, line);
165 ngx_snprintf((u_char *) line, sizeof(line), "x-amz-date: %s%Z",
166 pd->amzdate);
167 hdrs = curl_slist_append(hdrs, line);
168 ngx_snprintf((u_char *) line, sizeof(line), "Authorization: %s%Z",
169 pd->authorization);
170 hdrs = curl_slist_append(hdrs, line);
171
172 if (hdrs == NULL) {
173 curl_easy_cleanup(curl);
174 return NGX_ERROR;
175 }
176
177 errbuf[0] = '\0';
178 curl_easy_setopt(curl, CURLOPT_ERRORBUFFER, errbuf);
179 curl_easy_setopt(curl, CURLOPT_NOSIGNAL, 1L);
180 curl_easy_setopt(curl, CURLOPT_URL, url);
181 if (brix_dns_curl_pin(curl, dns, url, &resolve, reason, sizeof(reason))
182 != NGX_OK)
183 {
184 ngx_log_error(NGX_LOG_WARN, log, 0,
185 "brix_sts: cannot resolve STS endpoint: %s", reason);
186 curl_slist_free_all(hdrs);
187 curl_easy_cleanup(curl);
188 return NGX_ERROR;
189 }
190 curl_easy_setopt(curl, CURLOPT_POST, 1L);
191 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, pd->body);
192 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE, (long) ngx_strlen(pd->body));
193 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, hdrs);
194#if CURL_AT_LEAST_VERSION(7, 85, 0)
195 curl_easy_setopt(curl, CURLOPT_PROTOCOLS_STR, "http,https");
196#else
197 curl_easy_setopt(curl, CURLOPT_PROTOCOLS,
198 (long) (CURLPROTO_HTTP | CURLPROTO_HTTPS));
199#endif
200 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);
201 curl_easy_setopt(curl, CURLOPT_SSL_VERIFYHOST, 2L);
202 curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, 10L);
203 curl_easy_setopt(curl, CURLOPT_TIMEOUT, 30L);
204 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, sts_write_cb);
205 curl_easy_setopt(curl, CURLOPT_WRITEDATA, resp);
206
207 res = curl_easy_perform(curl);
208 if (res == CURLE_OK) {
209 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, http_status);
210 rc = NGX_OK;
211 } else {
212 ngx_log_error(NGX_LOG_WARN, log, 0, "brix_sts: STS request failed: %s",
213 errbuf[0] ? errbuf : curl_easy_strerror(res));
214 }
215
216 curl_slist_free_all(resolve);
217 curl_slist_free_all(hdrs);
218 curl_easy_cleanup(curl);
219 return rc;
220}

229/*
230 * Recursively search the parse tree for the first element named `name` and copy
231 * its text content into out[outsz]. Returns 1 if found, 0 otherwise. The STS
232 * AssumeRole/GetSessionToken responses nest <Credentials> under result and
233 * response wrappers, so a name-directed descent is simpler than a fixed path.
234 */
235static int
236sts_xml_find(xmlNodePtr node, const char *name, char *out, size_t outsz)
237{
238 xmlNodePtr n;
239
240 for (n = node; n != NULL; n = n->next) {
241 if (n->type == XML_ELEMENT_NODE
242 && xmlStrcmp(n->name, (const xmlChar *) name) == 0)
243 {
244 xmlChar *txt = xmlNodeGetContent(n);
245 if (txt != NULL) {
246 ngx_snprintf((u_char *) out, outsz, "%s%Z", (char *) txt);
247 xmlFree(txt);
248 return 1;
249 }
250 }
251 if (n->children != NULL
252 && sts_xml_find(n->children, name, out, outsz))
253 {
254 return 1;
255 }
256 }
257 return 0;
258}

261/*
262 * sts_parse_response — extract AccessKeyId / SecretAccessKey / SessionToken
263 * from the STS XML body into the caller's fixed buffers. Returns NGX_OK when at
264 * least AccessKeyId and SecretAccessKey are present; NGX_ERROR (with the fault
265 * logged, secrets excluded) otherwise. SessionToken is optional.
266 */
267ngx_int_t
268sts_parse_response(const u_char *body, size_t len,
269 const sts_creds_buf_t *creds, ngx_log_t *log)
270{
271 xmlDocPtr doc;
272 xmlNodePtr root;
273 ngx_int_t rc = NGX_ERROR;
274
275 creds->ak[0] = creds->sk[0] = creds->session[0] = '\0';
276
277 doc = xmlReadMemory((const char *) body, (int) len, "sts.xml", NULL,
278 XML_PARSE_NONET | XML_PARSE_NOERROR | XML_PARSE_NOWARNING);
279 if (doc == NULL) {
280 ngx_log_error(NGX_LOG_ERR, log, 0,
281 "brix_sts: unparseable STS response");
282 return NGX_ERROR;
283 }
284
285 root = xmlDocGetRootElement(doc);
286 if (root != NULL
287 && sts_xml_find(root, "AccessKeyId", creds->ak, creds->aksz)
288 && sts_xml_find(root, "SecretAccessKey", creds->sk, creds->sksz)
289 && creds->ak[0] != '\0' && creds->sk[0] != '\0')
290 {
291 (void) sts_xml_find(root, "SessionToken", creds->session,
292 creds->sesssz);
293 rc = NGX_OK;
294 } else {
295 ngx_log_error(NGX_LOG_ERR, log, 0,
296 "brix_sts: STS response missing credential fields");
297 }
298
299 xmlFreeDoc(doc);
300 return rc;
301}
