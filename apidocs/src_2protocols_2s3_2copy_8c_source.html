1/*
2 * copy.c — S3 CopyObject handler.
3 *
4 * PUT /bucket/dest-key + x-amz-copy-source: /bucket/src-key
5 *
6 * Performs a server-side copy using copy_file_range (with read/write fallback),
7 * then returns a CopyObjectResult XML body containing the new object's ETag and
8 * LastModified timestamp. Source and destination are both confined to the
9 * configured root_canon to prevent path-traversal.
10 */
11/* WHY: S3 CopyObject enables zero-copy server-side duplication — the source and
12 * destination are both confined to root_canon, avoiding client download/upload
13 * round-trip. copy_file_range provides kernel-level zero-copy when filesystems
14 * support it; read/write fallback ensures portability across filesystem types.
15 * Staged commit (write to temp file then rename) prevents partial writes from
16 * being visible to concurrent readers. */

34/* Look up a request header and return its value as a NUL-terminated pool
35 * copy (NULL when absent or on alloc failure) — find + strdupz adapter. */
36static const char *
37s3_copy_find_header(ngx_http_request_t *r, const char *name, size_t nlen)
38{
39 ngx_table_elt_t *h;
40 ngx_str_t value;
41
42 h = brix_http_find_header(r, name, nlen);
43 if (h == NULL
44 || brix_pstrdupz(r->pool, &value, h->value.data, h->value.len)
45 != NGX_OK)
46 {
47 return NULL;
48 }
49 return (const char *) value.data;
50}

52/* Build a transient VFS ctx for a confined S3 op on `fs_path` (mirrors the
53 * canonical s3_build_vfs_ctx in util.c, which this cannot call directly since
54 * s3_build_vfs_ctx's signature is specific to the PUT/POST-object write
55 * path's own loc-conf typedef usage — kept as a distinct, byte-identical-
56 * bind-wise helper here to keep the diff local to copy.c's two call sites).
57 * Binds the same per-user backend credential (+ opt-in mint) policy as
58 * s3_build_vfs_ctx so a remote-backed export's CopyObject (both src and dst)
59 * presents the REQUESTING USER's credential, not the shared service one. */
60static void
61s3_copy_vfs_ctx(ngx_http_request_t *r, const char *fs_path,
62 ngx_http_s3_loc_conf_t *cf, brix_vfs_ctx_t *vctx)
63{
64 ngx_http_s3_req_ctx_t *s3ctx =
65 ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
66 int is_tls = brix_http_request_is_tls(r);
67
68 brix_vfs_ctx_init(vctx, r->pool, r->connection->log, BRIX_PROTO_S3,
69 cf->common.root_canon, cf->common.cache_root_canon,
70 brix_vfs_policy_from_write_enable(cf->common.allow_write),
71 is_tls, (s3ctx != NULL) ? s3ctx->identity : NULL, fs_path);
72 brix_vfs_ctx_bind_backend_cred(vctx,
73 &cf->common.storage_credential_dir,
74 cf->common.storage_credential_fallback);
75 brix_vfs_ctx_bind_backend_mint(vctx,
76 &cf->common.storage_credential_mint_ca_cert,
77 &cf->common.storage_credential_mint_ca_key,
78 cf->common.storage_credential_mint_ttl);
79 s3_vfs_bind_deleg(r, cf, vctx);
80}

82/*
83 * Resolve the COPY source beneath root_canon.
84 *
85 * NGX_DECLINED means fs_path is filled and the caller carries on; anything else
86 * is a finished response. A RESERVED source name leaves by the same door an
87 * absent one does further down this file (404 NoSuchKey, event no_such_key,
88 * byte-identical message), so a client cannot learn from a COPY that a source
89 * name is reserved rather than missing; only a genuine escape is AccessDenied.
90 */
91static ngx_int_t
92s3_copy_resolve_source(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
93 const char *src_key, char *fs_path, size_t fs_path_cap)
94{
95 s3_key_error_t err;
96 int rc;
97
98 rc = s3_resolve_key_ex(cf->common.root_canon, src_key, fs_path,
99 fs_path_cap, cf->common.cache_store_endpoint);
100 if (rc == 0) {
101 return NGX_DECLINED;
102 }
103
104 s3_resolve_key_error(rc, &err);
105 if (rc == 404) {
106 err.message = "The copy source does not exist.";
107 } else if (rc == 403) {
108 err.message = "Copy source path is not accessible.";
109 }
110 return s3_fail(r, err.status, err.code, err.message, err.event);
111}

113ngx_int_t
114s3_handle_copy_object(ngx_http_request_t *r,
115 const char *dst_fs_path,
116 ngx_http_s3_loc_conf_t *cf)
117{
118 const char *copy_src_hdr;
119 const char *src_key;
120 char src_fs_path[PATH_MAX];
121 struct stat dst_sb;
122 brix_vfs_ctx_t vctx;
123 brix_vfs_copy_opts_t copy_opts;
124 char etag_buf[48];
125 char iso_buf[32];
126 char xml_buf[512];
127 size_t xml_len;
128 ngx_buf_t *b;
129 ngx_int_t rc;
130
131 copy_src_hdr = s3_copy_find_header(r, "x-amz-copy-source",
132 sizeof("x-amz-copy-source") - 1);
133 if (copy_src_hdr == NULL) {
134 return s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
135 "InvalidArgument",
136 "Missing x-amz-copy-source header.");
137 }
138
139 /* Strip optional leading '/' then skip the bucket prefix component */
140 src_key = copy_src_hdr;
141 if (*src_key == '/') {
142 src_key++;
143 }
144 {
145 const char *slash = strchr(src_key, '/');
146 if (slash != NULL) {
147 src_key = slash + 1;
148 }
149 }
150 if (*src_key == '\0') {
151 return s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST,
152 "InvalidArgument",
153 "Invalid copy source key.");
154 }
155
156 /* Resolve source path — must stay within root_canon */
157 rc = s3_copy_resolve_source(r, cf, src_key, src_fs_path,
158 sizeof(src_fs_path));
159 if (rc != NGX_DECLINED) {
160 return rc;
161 }
162
163 /*
164 * x-amz-metadata-directive (default COPY): REPLACE swaps the destination's
165 * user metadata for the request's x-amz-meta-*; COPY carries the source's
166 * across. A copy-onto-self with REPLACE is the metadata-only update SDKs (and
167 * our own sd_s3 set-meta) use — it must NOT rewrite the bytes.
168 */
169 {
170 const char *directive = s3_copy_find_header(r, "x-amz-metadata-directive",
171 sizeof("x-amz-metadata-directive") - 1);
172 int replace = (directive != NULL
173 && ngx_strcasecmp((u_char *) directive,
174 (u_char *) "REPLACE") == 0);
175 int is_self = (ngx_strcmp(src_fs_path, dst_fs_path) == 0);
176
177 if (!(is_self && replace)) {
178 /* Route the single-file copy through the metered VFS surface
179 * (OP_COPY). src = ctx path, dst passed explicitly; both confined. */
180 s3_copy_vfs_ctx(r, src_fs_path, cf, &vctx);
181 ngx_memzero(&copy_opts, sizeof(copy_opts));
182 copy_opts.overwrite = 1;
183 copy_opts.staged_commit = 1;
184
185 if (brix_vfs_copy(&vctx, dst_fs_path, &copy_opts) != NGX_OK) {
186 if (errno == ENOENT) {
187 return s3_fail(r, NGX_HTTP_NOT_FOUND, "NoSuchKey",
188 "The copy source does not exist.",
189 BRIX_S3_EVENT_NO_SUCH_KEY);
190 }
191 BRIX_S3_METRIC_INC(
192 events_total[BRIX_S3_EVENT_INTERNAL_ERROR]);
193 return NGX_HTTP_INTERNAL_SERVER_ERROR;
194 }
195 }
196
197 /* Propagate user metadata per the directive (after the bytes land). */
198 if (replace) {
199 (void) s3_apply_put_user_metadata(r, dst_fs_path,
200 cf->common.root_canon);
201 } else {
202 (void) s3_user_meta_copy(r, src_fs_path, dst_fs_path);
203 }
204 }
205
206 /* Confined no-follow probe of the freshly-copied destination for the ETag,
207 * through the VFS seam (non-metered — the CopyObject op already accounts for
208 * this via OP_COPY; a phantom OP_STAT would inflate the stat counter). */
209 {
210 brix_vfs_ctx_t dctx;
211 brix_vfs_stat_t dvst;
212
213 s3_copy_vfs_ctx(r, dst_fs_path, cf, &dctx);
214 if (brix_vfs_probe(&dctx, 1 /* no-follow */, &dvst) != NGX_OK) {
215 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_INTERNAL_ERROR]);
216 return NGX_HTTP_INTERNAL_SERVER_ERROR;
217 }
218 ngx_memzero(&dst_sb, sizeof(dst_sb));
219 dst_sb.st_size = dvst.size;
220 dst_sb.st_mtime = dvst.mtime;
221
222 /* phase-97 §5: a CopyObject publishes a NEW object at the destination,
223 * so the manager needs an ADD for it; the source is untouched and keeps
224 * its entry. Reuses the ETag probe — no second syscall. */
225 brix_cns_emit_at(cf->common.root_canon, BRIX_CNS_ADD, dst_fs_path,
226 (uint64_t) dvst.size, (uint64_t) dvst.mtime);
227 }
228
229 s3_etag(&dst_sb, etag_buf, sizeof(etag_buf));
230 brix_format_iso8601(dst_sb.st_mtime, iso_buf, sizeof(iso_buf));
231
232 xml_len = (size_t) snprintf(xml_buf, sizeof(xml_buf),
233 "<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
234 "<CopyObjectResult "
235 "xmlns=\"http://s3.amazonaws.com/doc/2006-03-01/\">"
236 "<ETag>%s</ETag>"
237 "<LastModified>%s</LastModified>"
238 "</CopyObjectResult>",
239 etag_buf, iso_buf);
240
241 b = ngx_create_temp_buf(r->pool, xml_len);
242 if (b == NULL) {
243 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_INTERNAL_ERROR]);
244 return NGX_HTTP_INTERNAL_SERVER_ERROR;
245 }
246
247 b->last = ngx_cpymem(b->pos, xml_buf, xml_len);
248 b->last_buf = 1;
249
250 return brix_http_send_xml_buffer(r, NGX_HTTP_OK,
251 (ngx_str_t) ngx_string("application/xml"), b);
252}
