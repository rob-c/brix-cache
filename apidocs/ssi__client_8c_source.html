1/*
2 * ssi_client.c — native SSI request/response client. See ssi_client.h for the
3 * two facts that shape this file: one control word with TWO carriers, and two
4 * reply paths that a caller must not be able to tell apart.
5 *
6 * Everything wire-facing is delegated: the control word to the SHARED codec in
7 * libxrdproto (src/protocols/ssi/ssi_rrinfo.c — the same object the server
8 * links), the request bodies to the shared per-opcode packers (wire_codec.h),
9 * and the asynresp envelope to brix_recv_next_asynresp (frame.c). Nothing here
10 * lays out a byte by hand.
11 */
12

21/* ServerOpenBody: fhandle(4) cpsize(4) cptype(4). A retstat open appends a
22 * NUL-terminated StatInfo string after it. */
23#define SSI_OPENBODY_LEN 12

25/*
26 * WHAT: turn a control word into the int64 the kXR offset field carries.
27 * WHY: the read/write bodies are packed by the shared codec, whose offset is a
28 * signed 64-bit field — so the 8 RRInfo bytes have to become an int64
29 * somewhere. Doing it here, once, keeps the byte order owned by
30 * frame_hdr.h instead of being re-derived at each call site.
31 * HOW: encode, then load big-endian, which is exactly how the server reverses
32 * it (brix_ssi_read re-serializes the offset back to 8 bytes before
33 * decoding). The cast cannot lose or misrepresent a bit: the RRInfo's
34 * leading byte is the command (Rxq/Rwt/Can = 0/1/2), so it is the offset's
35 * most-significant byte and the value is always a small positive int64 —
36 * never the negative offset a naive reader would fear.
37 */
38static int64_t
39ssi_ctl_offset(int cmd, uint32_t id, uint32_t size)
40{
41 unsigned char off8[BRIX_SSI_RRINFO_LEN];
42
43 brix_ssi_rrinfo_encode(cmd, id, size, off8);
44 return (int64_t) xrd_get_u64_be(off8);
45}

47/*
48 * WHAT: map an XrdSsiRRInfoAttn tag byte to the event the caller sees.
49 * WHY: this IS the discriminator — the transport builds an identical envelope
50 * for an alert and for the terminal response, so a wrong or unknown tag
51 * must fail loudly here rather than be guessed at. Both reply paths funnel
52 * through this one function so the decision exists exactly once.
53 */
54static int
55ssi_tag_to_ev(char tag, brix_ssi_ev *ev, brix_status *st)
56{
57 if (tag == BRIX_SSI_ATTN_ALRT) { *ev = BRIX_SSI_EV_ALERT; return 0; }
58 if (tag == BRIX_SSI_ATTN_FULL) { *ev = BRIX_SSI_EV_RESPONSE; return 0; }
59 if (tag == BRIX_SSI_ATTN_PEND) { *ev = BRIX_SSI_EV_PENDING; return 0; }
60
61 brix_status_set(st, XRDC_EPROTO, 0, "unknown SSI attn tag 0x%02x",
62 (unsigned) (unsigned char) tag);
63 return -1;
64}

66/*
67 * WHAT: split a delivered payload into prefix / metadata / data.
68 * WHY: the data length is stated NOWHERE on the wire — it is only recoverable
69 * as total - pfx_len - md_len — so both server-declared lengths must be
70 * validated against the frame we actually received before either is used
71 * as an offset. A server that overstates md_len would otherwise walk the
72 * data pointer past the allocation.
73 * HOW: `raw` is adopted on success (the reply owns it) and left untouched on
74 * failure, so the caller's free is unambiguous either way.
75 */
76static int
77ssi_reply_split(uint8_t *raw, uint32_t rawlen, brix_ssi_reply *r,
78 brix_status *st)
79{
80 char tag;
81 uint16_t pfx;
82 uint32_t md;
83
84 if (raw == NULL || rawlen < BRIX_SSI_ATTN_LEN) {
85 brix_status_set(st, XRDC_EPROTO, 0,
86 "SSI reply shorter than its attn prefix (%u < %d)",
87 rawlen, (int) BRIX_SSI_ATTN_LEN);
88 return -1;
89 }
90 brix_ssi_attn_decode(raw, &tag, NULL, &pfx, &md);
91
92 if (pfx < BRIX_SSI_ATTN_LEN || pfx > rawlen || md > rawlen - pfx) {
93 brix_status_set(st, XRDC_EPROTO, 0,
94 "SSI reply lengths do not fit the frame "
95 "(pfx %u + md %u of %u)", (unsigned) pfx, md, rawlen);
96 return -1;
97 }
98 if (ssi_tag_to_ev(tag, &r->ev, st) != 0) {
99 return -1;
100 }
101
102 r->raw = raw;
103 r->meta_len = md;
104 r->meta = (md > 0) ? raw + pfx : NULL;
105 r->data_len = rawlen - pfx - md;
106 r->data = (r->data_len > 0) ? raw + pfx + md : NULL;
107 return 0;
108}

110/*
111 * WHAT: send one control word on the QUERY carrier and hand back the reply body.
112 * WHY: response-wait (Rwt) and cancel (Can) are the same request shape and
113 * differ only in the command byte; the server routes both through
114 * brix_ssi_query, which reads the RRInfo from the query PAYLOAD — not from
115 * an offset field, which this opcode does not have.
116 * HOW: brix_roundtrip (not brix_send/brix_recv) so a clustered endpoint's
117 * redirect is followed, matching every other path-addressed op.
118 */
119static int
120ssi_query_ctl(brix_ssi_sess *s, int cmd, uint8_t **body, uint32_t *blen,
121 brix_status *st)
122{
123 ClientQueryRequest req;
124 unsigned char off8[BRIX_SSI_RRINFO_LEN];
125 uint16_t status;
126
127 memset(&req, 0, sizeof(req));
128 req.requestid = htons(kXR_query);
129 {
130 xrdw_query_req_t q = { .infotype = (uint16_t) kXR_Qopaqug };
131 memcpy(q.fhandle, s->fhandle, XRDW_FHANDLE_LEN);
132 xrdw_query_req_pack(&q, ((ClientRequestHdr *) &req)->body);
133 }
134 brix_ssi_rrinfo_encode(cmd, s->req_id, 0, off8);
135
136 {
137 brix_payload pl = { off8, (uint32_t) BRIX_SSI_RRINFO_LEN };
138 brix_resp_out out = { &status, body, blen };
139 return brix_roundtrip(s->conn, &req, &pl, &out, st);
140 }
141}

153int
154brix_ssi_sess_open(brix_ssi_sess *s, brix_conn *c, const char *service,
155 brix_status *st)
156{
157 ClientOpenRequest req;
158 char *path;
159 size_t need;
160 uint16_t status;
161 uint8_t *body = NULL;
162 uint32_t blen = 0;
163
164 memset(s, 0, sizeof(*s));
165 s->conn = c;
166 s->req_id = 1; /* any nonzero 24-bit id; one request in flight per session */
167
168 need = BRIX_SSI_PREFIX_LEN + strlen(service) + 1;
169 path = (char *) malloc(need);
170 if (path == NULL) {
171 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (%zu)", need);
172 return -1;
173 }
174 snprintf(path, need, "%s%s", BRIX_SSI_PREFIX, service);
175
176 memset(&req, 0, sizeof(req));
177 req.requestid = htons(kXR_open);
178 {
179 /* Read-only + retstat. brix_open_options_build() cannot produce
180 * kXR_retstat — nothing else in this client sets it — so it is OR-ed on
181 * here rather than added to that helper, whose four callers all mean
182 * "ordinary file open". */
183 xrdw_open_req_t o = {
184 .options = (uint16_t) (brix_open_options_build(0, 0, 0, 0)
185 | (uint16_t) kXR_retstat)
186 };
187 xrdw_open_req_pack(&o, ((ClientRequestHdr *) &req)->body);
188 }
189 {
190 brix_payload pl = { path, (uint32_t) (need - 1) };
191 brix_resp_out out = { &status, &body, &blen };
192 if (brix_roundtrip(c, &req, &pl, &out, st) != 0) {
193 free(path);
194 return -1;
195 }
196 }
197 free(path);
198
199 if (blen < XRDC_FHANDLE_LEN) {
200 brix_status_set(st, XRDC_EPROTO, 0,
201 "SSI open reply too short (%u bytes)", blen);
202 free(body);
203 return -1;
204 }
205 /* We asked for retstat, so a StatInfo string MUST follow the 12-byte
206 * ServerOpenBody. Enforcing it is the point of setting the bit: this is the
207 * requirement that makes stock's libXrdSsi accept an SSI open at all, and a
208 * server that answers without one would fail that client while quietly
209 * satisfying a lenient one. */
210 if (blen <= SSI_OPENBODY_LEN) {
211 brix_status_set(st, XRDC_EPROTO, 0,
212 "SSI open reply carries no StatInfo despite kXR_retstat "
213 "(%u bytes, need > %d)", blen, SSI_OPENBODY_LEN);
214 free(body);
215 return -1;
216 }
217 memcpy(s->fhandle, body, XRDC_FHANDLE_LEN);
218 free(body);
219 return 0;
220}

222int
223brix_ssi_submit(brix_ssi_sess *s, const void *req_bytes, size_t len,
224 brix_status *st)
225{
226 ClientWriteRequest req;
227 uint16_t sid = 0, status = 0;
228 uint8_t *body = NULL;
229 uint32_t blen = 0;
230 brix_resp_out out = { &status, &body, &blen };
231 int saved, rc;
232
233 if (len > BRIX_SSI_ID_MAX) {
234 /* reqSize is a u32 on the wire, but a request that large is a caller
235 * bug long before it is a wire limit; refuse rather than truncate. */
236 brix_status_set(st, XRDC_EPROTO, 0,
237 "SSI request too large (%zu bytes)", len);
238 return -1;
239 }
240
241 memset(&req, 0, sizeof(req));
242 req.requestid = htons(kXR_write);
243 {
244 xrdw_write_req_t w = {
245 .offset = ssi_ctl_offset(BRIX_SSI_CMD_RXQ, s->req_id,
246 (uint32_t) len),
247 .pathid = 0
248 };
249 memcpy(w.fhandle, s->fhandle, XRDW_FHANDLE_LEN);
250 xrdw_write_req_pack(&w, ((ClientRequestHdr *) &req)->body);
251 }
252
253 /* A deferring service answers kXR_waitresp here and then PUSHES its reply
254 * frames. Without defer_surfaces, brix_recv would block and swallow the
255 * first of them — an alert, whose tag the caller would never see. Scoped to
256 * this one exchange: the conn is borrowed, and await() pulls the pushed
257 * frames itself, so the flag has no business outliving the submit. */
258 saved = s->conn->defer_surfaces;
259 s->conn->defer_surfaces = 1;
260 {
261 brix_payload pl = { req_bytes, (uint32_t) len };
262 rc = brix_send(s->conn, &req, &pl, &sid, st);
263 if (rc == 0) {
264 rc = brix_recv(s->conn, sid, &out, st);
265 }
266 }
267 s->conn->defer_surfaces = saved;
268
269 if (rc != 0) {
270 return -1;
271 }
272 free(body);
273 s->sid = sid;
274 s->deferred = (status == kXR_waitresp) ? 1 : 0;
275 return 0;
276}

278int
279brix_ssi_await(brix_ssi_sess *s, brix_ssi_reply *r, brix_status *st)
280{
281 uint8_t *body = NULL;
282 uint32_t blen = 0;
283
284 memset(r, 0, sizeof(*r));
285
286 if (s->deferred) {
287 uint16_t status;
288 brix_resp_out out = { &status, &body, &blen };
289 if (brix_recv_next_asynresp(s->conn, s->sid, &out, st) != 0) {
290 return -1;
291 }
292 } else if (ssi_query_ctl(s, BRIX_SSI_CMD_RWT, &body, &blen, st) != 0) {
293 return -1;
294 }
295
296 if (ssi_reply_split(body, blen, r, st) != 0) {
297 free(body);
298 return -1;
299 }
300 if (r->ev == BRIX_SSI_EV_ALERT && ++s->alerts_seen > BRIX_SSI_ALERT_MAX) {
301 brix_ssi_reply_free(r);
302 brix_status_set(st, XRDC_EPROTO, 0,
303 "SSI service sent more than %d alerts without a response",
304 BRIX_SSI_ALERT_MAX);
305 return -1;
306 }
307 return 0;
308}

310int
311brix_ssi_read(brix_ssi_sess *s, void *buf, size_t len, brix_status *st)
312{
313 ClientReadRequest req;
314 uint16_t sid = 0, status = 0;
315 uint8_t *body = NULL;
316 uint32_t blen = 0;
317 brix_resp_out out = { &status, &body, &blen };
318
319 if (len > 0x7fffffffu) {
320 brix_status_set(st, XRDC_EPROTO, 0, "SSI read too large (%zu)", len);
321 return -1;
322 }
323
324 memset(&req, 0, sizeof(req));
325 req.requestid = htons(kXR_read);
326 {
327 xrdw_read_req_t rr = {
328 .offset = ssi_ctl_offset(BRIX_SSI_CMD_RXQ, s->req_id, 0),
329 .rlen = (int32_t) len
330 };
331 memcpy(rr.fhandle, s->fhandle, XRDW_FHANDLE_LEN);
332 xrdw_read_req_pack(&rr, ((ClientRequestHdr *) &req)->body);
333 }
334
335 if (brix_send(s->conn, &req, NULL, &sid, st) != 0) {
336 return -1;
337 }
338 if (brix_recv(s->conn, sid, &out, st) != 0) {
339 return -1;
340 }
341 /* The server clamps to what it has; clamp again so a buggy or hostile
342 * server's over-long body cannot overrun the caller's buffer. */
343 if (blen > len) {
344 blen = (uint32_t) len;
345 }
346 if (blen > 0) {
347 memcpy(buf, body, blen);
348 }
349 free(body);
350 return (int) blen;
351}
