26/* WHAT: mount an HTTP(S)/WebDAV endpoint (read-only, ranged GET).
27 * WHY: isolates the whole web transport bring-up (URL parse, bearer/TLS
28 * policy, reachability probe) from the root:// path in main.
29 * HOW: parse + validate the web URL, derive the export base, stat the
30 * export root up front (fail the mount early if unreachable/denied),
31 * then hand off to fuse_main. Returns the process exit code. */
32static int
33aio_web_mount(int fuse_argc, char **fuse_argv, size_t fuse_argv_cap,
34 const char *endpoint)
35{
36 brix_status st;
37 brix_statinfo si;
38 int drc;
39
40 /* Same rule as the root:// path: fork before the connection pool exists.
41 * The web pool is synchronous today, but keeping both transports on one
42 * ordering means a future background thread here cannot silently
43 * reintroduce the hang. */
44 drc = xfs_daemon_setup(&fuse_argc, fuse_argv, fuse_argv_cap);
45 if (drc != 0) return drc;
46
47 brix_status_clear(&st);
48 if (brix_weburl_parse(endpoint, &g_weburl) != 0) {
49 fprintf(stderr, "xrootdfs: bad web URL: %s\n", endpoint);
50 xfs_daemon_ready(2);
51 return 2;
52 }
53 if (g_weburl.is_s3) {
54 fprintf(stderr, "xrootdfs: s3:// is not supported as a FUSE mount "
55 "(use http/https/dav/davs)\n");
56 xfs_daemon_ready(2);
57 return 2;
58 }
59 g_web = 1;
60 if (g_bearer == NULL) {
61 g_bearer = getenv("BEARER_TOKEN");
62 }
63 g_web_verify = g_opts.verify_host;
64 g_web_ca = brix_resolve_ca_dir(g_opts.ca_dir);
65 g_web_proxy = brix_web_proxy_pem(g_web_proxy_buf, sizeof(g_web_proxy_buf));
66 /* export base = the URL path, trailing '/' trimmed; "/" → "" (verbatim). */
67 aio_set_base(g_weburl.path);
68 /* fail the mount up front if the export root is unreachable/denied. */
69 if (brix_web_stat(&g_weburl, g_base[0] ? g_base : "/", g_bearer,
70 g_web_verify, g_web_ca, g_web_proxy, &si, &st) != 0) {
71 fprintf(stderr, "xrootdfs: %s://%s:%d%s: %s\n",
72 g_weburl.tls ? "https" : "http", g_weburl.host, g_weburl.port,
73 g_weburl.path, st.msg);
74 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
75 return brix_shellcode(&st);
76 }
77 fprintf(stderr,
78 "xrootdfs: mounted %s:%d via %s%s (read-only WebDAV; "
79 "verify=%d, auth=%s, meta-pool=%d)\n",
80 g_weburl.host, g_weburl.port, g_weburl.tls ? "HTTPS" : "HTTP",
81 g_base, g_web_verify, g_bearer ? "bearer" : "anon", g_max_conns);
82
83 /* Pool the metadata path: the probe above validated endpoint/auth/TLS, so
84 * slot-0's eager connect will not be the first failure point. */
85 brix_webmeta_init(&g_web_tmpl, g_weburl.host, g_weburl.port, g_weburl.tls,
86 g_web_verify, g_web_ca, g_web_proxy, g_bearer,
87 0 /* → default 30 s */);
88 brix_status_clear(&st);
89 g_web_pool = brix_cpool_create(&WEB_VT, &g_web_tmpl, g_max_conns, &st);
90 if (g_web_pool == NULL) {
91 fprintf(stderr, "xrootdfs: web pool: %s\n", st.msg);
92 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
93 return brix_shellcode(&st);
94 }
95 /* Success is signalled from xfs_init() once the mount is live. */
96 int rc = fuse_main(fuse_argc, fuse_argv, &xfs_ops, NULL);
97 brix_cpool_destroy(g_web_pool);
98 g_web_pool = NULL;
99 return rc;
100}

103/* WHAT: probe the server's vendor POSIX extensions (kXR_Qconfig "xrdfs.ext")
104 * once at mount.
105 * WHY: utimens/chown/symlink/readlink/link adapt to what is advertised;
106 * absent capabilities keep the honest fallbacks.
107 * HOW: checkout a pooled metadata connection, probe, and check it back in
108 * (unhealthy on a probe failure that severed the link). */
109static void
110aio_probe_ext(brix_status *st)
111{
112 brix_conn *pc = brix_pool_checkout(g_pool, st);
113
114 if (pc != NULL) {
115 int ok = (brix_ext_probe(pc, &g_ext_setattr, &g_ext_symlink,
116 &g_ext_readlink, &g_ext_link, st) == 0);
117 brix_pool_checkin(g_pool, pc, ok ? 1 : xfs_conn_healthy(st));
118 }
119}

122/* WHAT: mount a root[s]:// endpoint (binary XRootD; read-write, resilient).
123 * WHY: isolates the async-driver bring-up (metadata pool + data-stream
124 * manager + extension probe) and its teardown ordering from main.
125 * HOW: daemonize FIRST (so every thread lands in the daemon), then parse the
126 * endpoint, derive the export base, create the pool then the manager
127 * (destroyed in reverse order after fuse_main returns), probe extensions,
128 * run fuse. Returns the process exit code. */
129static int
130aio_root_mount(int fuse_argc, char **fuse_argv, size_t fuse_argv_cap,
131 const char *endpoint)
132{
133 brix_status st;
134 int rc;
135
136 /* Before g_pool/g_mgr — the manager's event-loop thread must be created on
137 * the daemon side of the fork or every read() hangs. */
138 rc = xfs_daemon_setup(&fuse_argc, fuse_argv, fuse_argv_cap);
139 if (rc != 0) return rc;
140
141 brix_status_clear(&st);
142 if (brix_endpoint_parse(endpoint, &g_url, &st) != 0) {
143 fprintf(stderr, "xrootdfs: %s\n", st.msg);
144 xfs_daemon_ready(2);
145 return 2;
146 }
147
148 /* Export base = the URL path component (root://host/data → "/data"), so the
149 * mount roots at that subtree. Trailing '/' trimmed; a bare host (path "/" or
150 * empty) → verbatim FUSE paths. Shared with the web transport via srv_path(). */
151 aio_set_base((g_url.path[0] == '/') ? g_url.path : "");
152
153 g_pool = brix_pool_create(&g_url, &g_opts, g_max_conns, &st);
154 if (g_pool == NULL) {
155 fprintf(stderr, "xrootdfs: connect %s:%d: %s\n",
156 g_url.host, g_url.port, st.msg);
157 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
158 return brix_shellcode(&st);
159 }
160 /* Default: connect all data streams up front (in parallel — ~1×RTT mount).
161 * --lazy-streams trades first-read warm-up for the lowest possible mount
162 * latency by bringing up just one stream now and the rest on demand. */
163 g_mgr = brix_mgr_create(&g_url, &g_opts, g_streams,
164 g_lazy_streams ? 1 : g_streams,
165 g_max_stall, g_keepalive, g_max_retries, &st);
166 if (g_mgr == NULL) {
167 fprintf(stderr, "xrootdfs: async manager: %s\n", st.msg);
168 brix_pool_destroy(g_pool);
169 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
170 return brix_shellcode(&st);
171 }
172
173 aio_probe_ext(&st);
174
175 fprintf(stderr,
176 "xrootdfs: mounted %s:%d (meta-pool=%d, data-streams=%d, "
177 "max-stall=%dms; network-resilient; ext: setattr=%d symlink=%d "
178 "readlink=%d link=%d)\n",
179 g_url.host, g_url.port, g_max_conns, g_streams, g_max_stall,
180 g_ext_setattr, g_ext_symlink, g_ext_readlink, g_ext_link);
181
182 /* Success is signalled from xfs_init() once the mount is live. */
183 rc = fuse_main(fuse_argc, fuse_argv, &xfs_ops, NULL);
184
185 brix_mgr_destroy(g_mgr);
186 brix_pool_destroy(g_pool);
187 return rc;
188}

191/* Entry point for the default (async/resilient) driver. Invoked by the unified
192 * xrootdfs front-end (apps/xrootdfs_main.c); see xrootdfs_drivers.h. */
193int
194xrootdfs_aio_main(int argc, char **argv)
195{
196 const char *endpoint = NULL;
197 char *fuse_argv[64];
198 int fuse_argc = 0;
199 int rc;
200
201 /* Check --version / --help before the argc < 3 guard so they work standalone. */
202 if (argc >= 2) {
203 if (strcmp(argv[1], "--version") == 0) {
204 printf("xrootdfs (BriX-Cache client) %s\n", brix_client_version());
205 return 0;
206 }
207 if (strcmp(argv[1], "--help") == 0) { usage_fp(stdout); return 0; }
208 if (strcmp(argv[1], "-h") == 0) { usage(); return 0; }
209 }
210
211 if (argc < 3) {
212 usage();
213 return 2;
214 }
215 signal(SIGPIPE, SIG_IGN); /* a dropped peer must never kill the mount */
216 memset(&g_opts, 0, sizeof(g_opts));
217 g_opts.verify_host = 1;
218 brix_crypto_init();
219
220 fuse_argv[fuse_argc++] = argv[0];
221
222 rc = aio_parse_args(argc, argv, fuse_argv, &fuse_argc, &endpoint);
223 if (rc >= 0) {
224 return rc;
225 }
226
227 if (endpoint == NULL || fuse_argc < 2) {
228 usage();
229 return 2;
230 }
231 fuse_argv[fuse_argc] = NULL;
232
233 /* HTTP(S)/WebDAV read-only mount when the endpoint is a web URL. */
234 if (brix_is_web_url(endpoint)) {
235 return aio_web_mount(fuse_argc, fuse_argv,
236 sizeof(fuse_argv) / sizeof(fuse_argv[0]), endpoint);
237 }
238
239 return aio_root_mount(fuse_argc, fuse_argv,
240 sizeof(fuse_argv) / sizeof(fuse_argv[0]), endpoint);
241}
