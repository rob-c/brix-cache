27/* WHAT: mount an HTTP(S)/WebDAV endpoint (read-only, ranged GET).
28 * WHY: isolates the whole web transport bring-up (URL parse, bearer/TLS
29 * policy, reachability probe) from the root:// path in main.
30 * HOW: parse + validate the web URL, derive the export base, stat the
31 * export root up front (fail the mount early if unreachable/denied),
32 * then hand off to fuse_main. Returns the process exit code. */
33static int
34aio_web_mount(int fuse_argc, char **fuse_argv, size_t fuse_argv_cap,
35 const char *endpoint)
36{
37 brix_status st;
38 brix_statinfo si;
39 int drc;
40
41 /* Same rule as the root:// path: fork before the connection pool exists.
42 * The web pool is synchronous today, but keeping both transports on one
43 * ordering means a future background thread here cannot silently
44 * reintroduce the hang. */
45 drc = xfs_daemon_setup(&fuse_argc, fuse_argv, fuse_argv_cap);
46 if (drc != 0) return drc;
47
48 brix_status_clear(&st);
49 if (brix_weburl_parse(endpoint, &g_weburl) != 0) {
50 BRIX_CLIENT_LOG("ERROR", "xrootdfs: bad web URL: %s\n", endpoint);
51 xfs_daemon_ready(2);
52 return 2;
53 }
54 if (g_weburl.is_s3) {
55 BRIX_CLIENT_LOG("ERROR", "xrootdfs: s3:// is not supported as a FUSE mount "
56 "(use http/https/dav/davs)\n");
57 xfs_daemon_ready(2);
58 return 2;
59 }
60 g_web = 1;
61 if (g_bearer == NULL) {
62 g_bearer = getenv("BEARER_TOKEN");
63 }
64 g_web_verify = g_opts.verify_host;
65 g_web_ca = brix_resolve_ca_dir(g_opts.ca_dir);
66 g_web_proxy = brix_web_proxy_pem(g_web_proxy_buf, sizeof(g_web_proxy_buf));
67 /* export base = the URL path, trailing '/' trimmed; "/" → "" (verbatim). */
68 aio_set_base(g_weburl.path);
69 /* fail the mount up front if the export root is unreachable/denied. */
70 if (brix_web_stat(&g_weburl, g_base[0] ? g_base : "/", g_bearer,
71 g_web_verify, g_web_ca, g_web_proxy, &si, &st) != 0) {
72 BRIX_CLIENT_LOG("ERROR", "xrootdfs: %s://%s:%d%s: %s\n",
73 g_weburl.tls ? "https" : "http", g_weburl.host, g_weburl.port,
74 g_weburl.path, st.msg);
75 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
76 return brix_shellcode(&st);
77 }
78 BRIX_CLIENT_LOG("INFO",
79 "xrootdfs: mounted %s:%d via %s%s (read-only WebDAV; "
80 "verify=%d, auth=%s, meta-pool=%d)\n",
81 g_weburl.host, g_weburl.port, g_weburl.tls ? "HTTPS" : "HTTP",
82 g_base, g_web_verify, g_bearer ? "bearer" : "anon", g_max_conns);
83
84 /* Pool the metadata path: the probe above validated endpoint/auth/TLS, so
85 * slot-0's eager connect will not be the first failure point. */
86 brix_webmeta_init(&g_web_tmpl, g_weburl.host, g_weburl.port, g_weburl.tls,
87 g_web_verify, g_web_ca, g_web_proxy, g_bearer,
88 0 /* → default 30 s */);
89 brix_status_clear(&st);
90 g_web_pool = brix_cpool_create(&WEB_VT, &g_web_tmpl, g_max_conns, &st);
91 if (g_web_pool == NULL) {
92 BRIX_CLIENT_LOG("ERROR", "xrootdfs: web pool: %s\n", st.msg);
93 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
94 return brix_shellcode(&st);
95 }
96 /* Success is signalled from xfs_ll_init() once the mount is live. */
97 int rc = xfs_session_main(fuse_argc, fuse_argv);
98 brix_cpool_destroy(g_web_pool);
99 g_web_pool = NULL;
100 return rc;
101}

104/* WHAT: probe the server's vendor POSIX extensions (kXR_Qconfig "xrdfs.ext")
105 * once at mount.
106 * WHY: utimens/chown/symlink/readlink/link adapt to what is advertised;
107 * absent capabilities keep the honest fallbacks.
108 * HOW: checkout a pooled metadata connection, probe, and check it back in
109 * (unhealthy on a probe failure that severed the link). */
110static void
111aio_probe_ext(brix_status *st)
112{
113 brix_conn *pc = brix_pool_checkout(g_pool, st);
114
115 if (pc != NULL) {
116 int ok = (brix_ext_probe(pc, &g_ext_setattr, &g_ext_symlink,
117 &g_ext_readlink, &g_ext_link, st) == 0);
118 brix_pool_checkin(g_pool, pc, ok ? 1 : xfs_conn_healthy(st));
119 }
120}

123/* WHAT: mount a root[s]:// endpoint (binary XRootD; read-write, resilient).
124 * WHY: isolates the async-driver bring-up (metadata pool + data-stream
125 * manager + extension probe) and its teardown ordering from main.
126 * HOW: daemonize FIRST (so every thread lands in the daemon), then parse the
127 * endpoint, derive the export base, create the pool then the manager
128 * (destroyed in reverse order after fuse_main returns), probe extensions,
129 * run fuse. Returns the process exit code. */
130static int
131aio_root_mount(int fuse_argc, char **fuse_argv, size_t fuse_argv_cap,
132 const char *endpoint)
133{
134 brix_status st;
135 int rc;
136
137 /* Before g_pool/g_mgr — the manager's event-loop thread must be created on
138 * the daemon side of the fork or every read() hangs. */
139 rc = xfs_daemon_setup(&fuse_argc, fuse_argv, fuse_argv_cap);
140 if (rc != 0) return rc;
141
142 brix_status_clear(&st);
143 if (brix_endpoint_parse(endpoint, &g_url, &st) != 0) {
144 BRIX_CLIENT_LOG("ERROR", "xrootdfs: %s\n", st.msg);
145 xfs_daemon_ready(2);
146 return 2;
147 }
148
149 /* Export base = the URL path component (root://host/data → "/data"), so the
150 * mount roots at that subtree. Trailing '/' trimmed; a bare host (path "/" or
151 * empty) → verbatim FUSE paths. Shared with the web transport via srv_path(). */
152 aio_set_base((g_url.path[0] == '/') ? g_url.path : "");
153
154 g_pool = brix_pool_create(&g_url, &g_opts, g_max_conns, &st);
155 if (g_pool == NULL) {
156 BRIX_CLIENT_LOG("ERROR", "xrootdfs: connect %s:%d: %s\n",
157 g_url.host, g_url.port, st.msg);
158 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
159 return brix_shellcode(&st);
160 }
161 /* Default: connect all data streams up front (in parallel — ~1×RTT mount).
162 * --lazy-streams trades first-read warm-up for the lowest possible mount
163 * latency by bringing up just one stream now and the rest on demand. */
164 g_mgr = brix_mgr_create(&g_url, &g_opts, g_streams,
165 g_lazy_streams ? 1 : g_streams,
166 g_max_stall, g_keepalive, g_max_retries, &st);
167 if (g_mgr == NULL) {
168 BRIX_CLIENT_LOG("ERROR", "xrootdfs: async manager: %s\n", st.msg);
169 brix_pool_destroy(g_pool);
170 xfs_daemon_ready((unsigned char) brix_shellcode(&st));
171 return brix_shellcode(&st);
172 }
173
174 if (xfs_ident_init() != 0) {
175 brix_mgr_destroy(g_mgr);
176 brix_pool_destroy(g_pool);
177 xfs_daemon_ready(2);
178 return 2;
179 }
180
181 aio_probe_ext(&st);
182
183 BRIX_CLIENT_LOG("INFO",
184 "xrootdfs: mounted %s:%d (meta-pool=%d, data-streams=%d, "
185 "max-stall=%dms; network-resilient; ext: setattr=%d symlink=%d "
186 "readlink=%d link=%d)\n",
187 g_url.host, g_url.port, g_max_conns, g_streams, g_max_stall,
188 g_ext_setattr, g_ext_symlink, g_ext_readlink, g_ext_link);
189
190 /* Success is signalled from xfs_ll_init() once the mount is live. */
191 rc = xfs_session_main(fuse_argc, fuse_argv);
192
193 xfs_ident_shutdown();
194 brix_mgr_destroy(g_mgr);
195 brix_pool_destroy(g_pool);
196 return rc;
197}

200/* Entry point for the default (async/resilient) driver. Invoked by the unified
201 * xrootdfs front-end (apps/xrootdfs_main.c); see xrootdfs_drivers.h. */
202int
203xrootdfs_aio_main(int argc, char **argv)
204{
205 const char *endpoint = NULL;
206 char *fuse_argv[64];
207 int fuse_argc = 0;
208 int rc;
209
210 /* Check --version / --help before the argc < 3 guard so they work standalone. */
211 if (argc >= 2) {
212 if (strcmp(argv[1], "--version") == 0) {
213 printf("xrootdfs (BriX-Cache client) %s\n", brix_client_version());
214 return 0;
215 }
216 if (strcmp(argv[1], "--help") == 0) { usage_fp(stdout); return 0; }
217 if (strcmp(argv[1], "-h") == 0) { usage(); return 0; }
218 }
219
220 if (argc < 3) {
221 usage();
222 return 2;
223 }
224 signal(SIGPIPE, SIG_IGN); /* a dropped peer must never kill the mount */
225 memset(&g_opts, 0, sizeof(g_opts));
226 g_opts.verify_host = 1;
227 brix_crypto_init();
228
229 fuse_argv[fuse_argc++] = argv[0];
230 xfs_add_host_opts(fuse_argv, &fuse_argc); /* e.g. macFUSE noappledouble */
231
232 rc = aio_parse_args(argc, argv, fuse_argv, &fuse_argc, &endpoint);
233 if (rc >= 0) {
234 return rc;
235 }
236
237 if (endpoint == NULL || fuse_argc < 2) {
238 usage();
239 return 2;
240 }
241 fuse_argv[fuse_argc] = NULL;
242
243 /* HTTP(S)/WebDAV read-only mount when the endpoint is a web URL. */
244 if (brix_is_web_url(endpoint)) {
245 return aio_web_mount(fuse_argc, fuse_argv,
246 sizeof(fuse_argv) / sizeof(fuse_argv[0]), endpoint);
247 }
248
249 return aio_root_mount(fuse_argc, fuse_argv,
250 sizeof(fuse_argv) / sizeof(fuse_argv[0]), endpoint);
251}
