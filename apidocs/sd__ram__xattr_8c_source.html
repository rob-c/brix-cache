1/*
2 * sd_ram_xattr.c — the RAM store's in-memory user.* xattrs.
3 *
4 * WHAT: getxattr / listxattr / setxattr / removexattr over the per-entry xattr
5 * list, and the two allocators behind them.
6 * WHY: Split out of sd_ram_staged.c, which had grown past the size cap
7 * carrying three planes. These four verbs share no state with the staged
8 * fill or the directory walk beyond the entry they look up, so the cut
9 * costs nothing: sd_ram_internal.h already declared every one of them.
10 * In XATTR meta mode this list IS the cache's cinfo record, which is why
11 * the plane is worth reading on its own.
12 * HOW: Every verb takes st->mtx for the whole lookup-and-mutate, and a node is
13 * built DETACHED (name copied, or nothing allocated at all) before it is
14 * linked, so a failed allocation never leaves a half-built xattr on an
15 * entry a reader can already see.
16 */

114/* A DETACHED xattr node owning a copy of `name`, or NULL. All-or-nothing: a
115 * half-built node is never returned, so the caller's only job is to link it. */
116static sd_ram_xattr_t *
117sd_ram_xattr_new(const char *name)
118{
119 sd_ram_xattr_t *x = calloc(1, sizeof(*x));
120
121 if (x == NULL) {
122 return NULL;
123 }
124 x->name = strdup(name);
125 if (x->name == NULL) {
126 free(x);
127 return NULL;
128 }

132/* Replace (or create) `name` on `ent` with `len` bytes of `val`. Caller holds
133 * st->mtx. Returns 0, or the errno to report. */
134static int
135sd_ram_xattr_put(sd_ram_ent_t *ent, const char *name, const void *val,
136 size_t len)
137{
138 sd_ram_xattr_t *x = sd_ram_xattr_find(ent, name);
139 u_char *nv = malloc(len > 0 ? len : 1);
140
141 if (nv == NULL) {
142 return ENOMEM;
143 }
144 memcpy(nv, val, len);
145
146 /* Ownership of the node and of nv transfers INTO ent->xattrs;
147 * brix_sd_ram_ent_free releases the whole chain.
148 *
149 * gcc 11's -fanalyzer cannot see a store through a parameter into a linked
150 * list as an escape, so it reports both leaking at the return below.
151 * Suppress exactly that diagnostic, at exactly its emission point. */
152#pragma GCC diagnostic push
153#if defined(__GNUC__) && !defined(__clang__)
154#pragma GCC diagnostic ignored "-Wanalyzer-malloc-leak"
155#endif
156 if (x == NULL) {
157 x = sd_ram_xattr_new(name);
158 if (x == NULL) {
159 free(nv);
160 return ENOMEM;
161 }
162 x->next = ent->xattrs;
163 ent->xattrs = x;
164 } else {
165 free(x->val);
166 }
167 x->val = nv;
168 x->len = len;
169 return 0;
170#pragma GCC diagnostic pop
171}
