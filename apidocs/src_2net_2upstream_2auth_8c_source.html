1/* File: auth.c — outbound kXR_auth framing for the transparent upstream connector
2 * WHAT: brix_upstream_send_auth_frame() serialises one ClientAuthRequest (24-byte
3 * header echoing the client's stream ID, `credtype` in the 4-byte slot, dlen =
4 * payload length) followed by the caller's payload into a pool buffer, sets
5 * bs_phase = XRD_UP_BS_AUTH, resets the response accumulator and flushes.
6 * brix_upstream_send_token_auth() is the ztn (WLCG JWT) producer on top of it:
7 * it reads brix_upstream_token_file synchronously via brix_token_read_file()
8 * (small local file < 64 KiB, refreshed externally) and sends "ztn\0" + JWT.
9 * The gsi producers live in auth_gsi.c and use the same framer.
10 *
11 * WHY: an upstream redirector may require its own authentication separate from
12 * the client's. ztn is one round; GSI is two (kXGC_certreq, kXGC_cert) and both
13 * must frame identically — one framer keeps the stream-ID echo, the credtype
14 * slot and the accumulator reset in a single place. Echoing the client's
15 * stream ID keeps request correlation end-to-end; XRD_UP_BS_AUTH tells the
16 * bootstrap dispatcher which reply phase to expect.
17 *
18 * HOW: BRIX_PALLOC_OR_RETURN(frame) → header fill (xrdw_auth_req_pack for the
19 * credtype slot) → payload copy → wbuf/bs_phase/accumulator → brix_upstream_flush;
20 * a partial write arms the write event (the write handler arms the read when
21 * drained), a full write arms the read event directly.
22 */

30/*
31 * brix_upstream_send_auth_frame — frame `payload` as a kXR_auth request and flush.
32 *
33 * Wire layout:
34 * [ClientAuthRequest header (24 B)]
35 * streamid[2] — echo client's stream ID
36 * requestid[2] — kXR_auth
37 * reserved[12] — zeroes
38 * credtype[4] — e.g. "ztn\0" / "gsi\0"
39 * dlen[4] — plen (big-endian)
40 * [Payload (plen bytes)] — protocol-specific (ztn: "ztn\0" + JWT; gsi: bucket
41 * stream from the XrdSecgsi kernel)
42 *
43 * Returns NGX_OK (sent, or partial with events armed) or NGX_ERROR (caller aborts).
44 */
45ngx_int_t
46brix_upstream_send_auth_frame(brix_upstream_t *up, const char credtype[4],
47 const u_char *payload, size_t plen)
48{
49 size_t frame_len;
50 u_char *frame;
51 ClientAuthRequest *hdr;
52 xrdw_auth_req_t b;
53
54 frame_len = sizeof(ClientAuthRequest) + plen;
55 BRIX_PALLOC_OR_RETURN(frame, up->conn->pool, frame_len, NGX_ERROR);
56
57 hdr = (ClientAuthRequest *)(void *) frame;
58 ngx_memzero(hdr, sizeof(*hdr));
59 hdr->streamid[0] = up->req_streamid[0];
60 hdr->streamid[1] = up->req_streamid[1];
61 hdr->requestid = htons(kXR_auth);
62 ngx_memcpy(b.credtype, credtype, 4);
63 xrdw_auth_req_pack(&b, ((ClientRequestHdr *) (void *) frame)->body);
64 hdr->dlen = htonl((kXR_int32) plen);
65
66 if (plen > 0) {
67 ngx_memcpy(frame + sizeof(ClientAuthRequest), payload, plen);
68 }
69
70 up->wbuf = frame;
71 up->wbuf_len = frame_len;
72 up->wbuf_pos = 0;
73 up->bs_phase = XRD_UP_BS_AUTH;
74
75 /* Reset the response accumulator for the kXR_auth reply. */
76 up->rhdr_pos = 0;
77 up->resp_dlen = 0;
78 up->resp_body = NULL;
79 up->resp_body_pos = 0;
80
81 ngx_log_debug3(NGX_LOG_DEBUG_STREAM, up->conn->log, 0,
82 "brix: upstream sending kXR_auth %*s (%uz payload bytes)",
83 (size_t) 3, credtype, plen);
84
85 if (brix_upstream_flush(up) == NGX_ERROR) {
86 return NGX_ERROR;
87 }
88
89 if (up->wbuf_pos < up->wbuf_len) {
90 /* Partial write — arm write event; write handler arms read when done. */
91 if (ngx_handle_write_event(up->conn->write, 0) != NGX_OK) {
92 return NGX_ERROR;
93 }
94 return NGX_OK;
95 }
96
97 /* All bytes written; arm read event to wait for the kXR_auth response. */
98 return ngx_handle_read_event(up->conn->read, 0);
99}

101/*
102 * brix_upstream_send_token_auth — read the configured token file and send it
103 * as a kXR_auth "ztn" frame ("ztn\0" repeated at the payload start per the
104 * XRootD convention, then the raw JWT bytes).
105 *
106 * Returns NGX_ERROR on token-read failure (already logged) or the framer's result.

118 {
119 return NGX_ERROR;
