27/* ---- Generate an RSA keypair through the common EVP interface ----
28 * WHAT: Return an owned key or NULL, using the requested modulus size.
29 * WHY: EVP_RSA_gen is a 3-only convenience API, not a protocol requirement.
30 * HOW: Initialize RSA keygen, set its size, generate, and release temporaries.
31 */
32static inline EVP_PKEY *
33brix_crypto_rsa_keygen(unsigned bits)
34{
35 EVP_PKEY_CTX *context = EVP_PKEY_CTX_new_id(EVP_PKEY_RSA, NULL);
36 EVP_PKEY *key = NULL;
37
38 if (bits <= 0x7fffffffU && context != NULL && EVP_PKEY_keygen_init(context) == 1
39 && EVP_PKEY_CTX_set_rsa_keygen_bits(context, (int) bits) == 1) {
40 if (EVP_PKEY_keygen(context, &key) != 1) {
41 EVP_PKEY_free(key);
42 key = NULL;
43 }
44 }
45 EVP_PKEY_CTX_free(context);
46 return key;
47}

49/* ---- Set an RSA key-generation exponent without consuming the caller's BN ----
50 * WHAT: Return 1 on success and 0 on failure, retaining input ownership.
51 * WHY: OpenSSL 1.1.1's setter consumes its BN; the 3+ set1 setter copies it.
52 * HOW: 1. Use set1 on 3+. 2. Otherwise duplicate and transfer on success only.
53 */
54static inline int
55brix_crypto_rsa_keygen_exponent(EVP_PKEY_CTX *context, BIGNUM *exponent)
56{
57 if (context == NULL || exponent == NULL) {
58 return 0;
59 }
60#if OPENSSL_VERSION_NUMBER >= 0x30000000L
61 return EVP_PKEY_CTX_set1_rsa_keygen_pubexp(context, exponent) > 0;
62#else
63 BIGNUM *copy = BN_dup(exponent);
64
65 if (copy == NULL) {
66 return 0;
67 }
68 if (EVP_PKEY_CTX_set_rsa_keygen_pubexp(context, copy) <= 0) {
69 BN_free(copy);
70 return 0;
71 }
72 return 1;
73#endif
74}

77 * WHAT: Return an owned keypair or NULL after any native key-generation error.
78 * WHY: Credential minting must not depend on OpenSSL 3's convenience Q API.

98/* ---- Compare public keys without accepting unsupported comparisons ----
99 * WHAT: Return 1 only for equal, non-NULL public keys; otherwise return 0.
100 * WHY: Negative/unsupported comparison results must never authorize delegation.
101 * HOW: 1. Reject NULL. 2. Require the version's comparison to return exactly 1.
102 */
103static inline int
104brix_crypto_pkey_equal(EVP_PKEY *left, EVP_PKEY *right)
105{
106 if (left == NULL || right == NULL) {
107 return 0;
108 }
109#if OPENSSL_VERSION_NUMBER >= 0x30000000L
110 return EVP_PKEY_eq(left, right) == 1;
111#else
112 return EVP_PKEY_cmp(left, right) == 1;
113#endif
114}

116/* ---- Copy a DH public number into caller-owned storage ----
117 * WHAT: Return 1 and an owned BIGNUM on success, otherwise 0 and NULL.
118 * WHY: Provider getters allocate while 1.1.1 getters borrow; callers need one
119 * invariant so cleanup cannot free the source key's internal number.
120 * HOW: 1. Clear output. 2. Fetch or duplicate the native public number.
121 */
122static inline int
123brix_crypto_dh_public(EVP_PKEY *key, BIGNUM **public_number)
124{
125 if (public_number == NULL) {
126 return 0;
127 }
128 *public_number = NULL;
129 if (key == NULL || EVP_PKEY_base_id(key) != EVP_PKEY_DH) {
130 return 0;
131 }
132#if OPENSSL_VERSION_NUMBER >= 0x30000000L
133 return EVP_PKEY_get_bn_param(key, OSSL_PKEY_PARAM_PUB_KEY, public_number);
134#else
135 const DH *dh = EVP_PKEY_get0_DH(key);
136 const BIGNUM *borrowed = NULL;
137
138 if (dh == NULL) {
139 return 0;
140 }
141 DH_get0_key(dh, &borrowed, NULL);
142 if (borrowed != NULL) {
143 *public_number = BN_dup(borrowed);
144 }
145 return *public_number != NULL;
146#endif
147}

149#if OPENSSL_VERSION_NUMBER < 0x30000000L
150/* ---- Copy DH parameters without dropping the subgroup order ----
151 * WHAT: Return an owned parameter-only DH or NULL; input remains borrowed.
152 * WHY: A PKCS#3 encode/decode copy can lose q; peer validation needs p/q/g.
153 * HOW: 1. Duplicate each present number. 2. Transfer only on complete success.
154 */
155static inline DH *
156brix_crypto_dh_copy_parameters(const DH *source)
157{
158 const BIGNUM *p = NULL, *q = NULL, *g = NULL;
159 BIGNUM *owned_p, *owned_q, *owned_g;
160 DH *copy;
161
162 DH_get0_pqg(source, &p, &q, &g);
163 if (p == NULL || g == NULL) {
164 return NULL;
165 }
166 owned_p = BN_dup(p);
167 owned_q = q != NULL ? BN_dup(q) : NULL;
168 owned_g = BN_dup(g);
169 copy = DH_new();
170 if (owned_p == NULL || owned_g == NULL || (q != NULL && owned_q == NULL)
171 || copy == NULL || DH_set0_pqg(copy, owned_p, owned_q, owned_g) != 1) {
172 BN_free(owned_p);
173 BN_free(owned_q);
174 BN_free(owned_g);
175 DH_free(copy);
176 return NULL;
177 }
178 return copy;
179}

181/* ---- Wrap a legacy DH peer using owned parameter/public copies ----
182 * WHAT: Return an owned key or NULL; both input handles remain borrowed.
183 * WHY: Isolate the legacy set0/set1 ownership boundary from provider imports.
184 * HOW: 1. Copy the group. 2. Transfer a public-number copy. 3. Wrap and free.
185 */
186static inline EVP_PKEY *
187brix_crypto_dh_peer_legacy(EVP_PKEY *parameters, const BIGNUM *public_number)
188{
189 const DH *borrowed = EVP_PKEY_get0_DH(parameters);
190 DH *copy = borrowed != NULL ? brix_crypto_dh_copy_parameters(borrowed) : NULL;
191 BIGNUM *public_copy = BN_dup(public_number);
192 EVP_PKEY *peer = NULL;
193
194 if (copy != NULL && public_copy != NULL
195 && DH_set0_key(copy, public_copy, NULL) == 1) {
196 public_copy = NULL;
197 peer = EVP_PKEY_new();
198 if (peer != NULL && EVP_PKEY_set1_DH(peer, copy) != 1) {
199 EVP_PKEY_free(peer);
200 peer = NULL;
201 }
202 }
203 BN_free(public_copy);
204 DH_free(copy);
205 return peer;
206}

209/* ---- Attach a peer's DH public number to the negotiated parameters ----
210 * WHAT: Return an owned public-only EVP_PKEY or NULL, retaining both inputs.
211 * WHY: GSI must derive using the peer's actual group, not silently substitute it.
212 * HOW: 1. Copy parameter material. 2. Add the public number. 3. Wrap and clean up.
213 */
214static inline EVP_PKEY *
215brix_crypto_dh_peer(EVP_PKEY *parameters, const BIGNUM *public_number)
216{
217 if (parameters == NULL || public_number == NULL
218 || EVP_PKEY_base_id(parameters) != EVP_PKEY_DH) {
219 return NULL;
220 }
221#if OPENSSL_VERSION_NUMBER >= 0x30000000L
222 EVP_PKEY *peer = NULL;
223 OSSL_PARAM *original = NULL, *addition = NULL, *merged = NULL;
224 OSSL_PARAM_BLD *builder = OSSL_PARAM_BLD_new();
225 EVP_PKEY_CTX *context = EVP_PKEY_CTX_new_id(EVP_PKEY_DH, NULL);
226
227 if (builder != NULL && context != NULL
228 && EVP_PKEY_todata(parameters, EVP_PKEY_KEY_PARAMETERS, &original) == 1
229 && OSSL_PARAM_BLD_push_BN(builder, OSSL_PKEY_PARAM_PUB_KEY, public_number) == 1
230 && (addition = OSSL_PARAM_BLD_to_param(builder)) != NULL
231 && (merged = OSSL_PARAM_merge(original, addition)) != NULL
232 && EVP_PKEY_fromdata_init(context) == 1) {
233 if (EVP_PKEY_fromdata(context, &peer, EVP_PKEY_PUBLIC_KEY, merged) != 1) {
234 EVP_PKEY_free(peer);
235 peer = NULL;
236 }
237 }
238 EVP_PKEY_CTX_free(context);
239 OSSL_PARAM_free(merged);
240 OSSL_PARAM_free(addition);
241 OSSL_PARAM_free(original);
242 OSSL_PARAM_BLD_free(builder);
243 return peer;
244#else
245 return brix_crypto_dh_peer_legacy(parameters, public_number);
246#endif
247}
