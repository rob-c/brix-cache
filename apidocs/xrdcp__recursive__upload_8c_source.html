8/* Shared state for one recursive web-upload walk (local tree → web collection).
9 * `base` is the dst URL's path with trailing slashes trimmed ("" for the root);
10 * the bucket/collection root, into which the source directory's CONTENTS go
11 * (symmetric with recursive download). */
12
13/* Build "<base>/<rel>" (or "/<rel>" when base is the root "") into out.
14 * 0 on success, -1 if it would not fit. */
15int
16web_join(const char *base, const char *rel, char *out, size_t outsz)
17{
18 int w = snprintf(out, outsz, "%s/%s", base, rel);
19 return (w < 0 || (size_t) w >= outsz) ? -1 : 0;
20}

23/* WHAT: one directory entry's local path + upload-root-relative path.
24 * WHY: every walk step derives this pair once and threads it through the
25 * dir/file handlers; a struct keeps those helpers under the parameter cap.
26 * HOW: filled by walk_child_paths on the walk frame's stack (same footprint
27 * as the two buffers the walk loop previously held). */
28typedef struct {
29 char local[XRDC_PATH_MAX]; /* localdir/<name> */
30 char rel[XRDC_PATH_MAX]; /* path relative to the upload root */
31} walk_child_t;
32

34/* WHAT: build a directory entry's local and root-relative paths.
35 * WHY: isolates the overflow-guarded path joins (with the "" top-of-tree rel
36 * special case) from the walk's classify/dispatch loop.
37 * HOW: snprintf both joins with truncation checks. 0 on success, -1 if either
38 * would not fit (caller reports and skips the entry). */
39static int
40walk_child_paths(const char *localdir, const char *rel, const char *name,
41 walk_child_t *out)
42{
43 if ((size_t) snprintf(out->local, sizeof(out->local), "%s/%s",
44 localdir, name) >= sizeof(out->local)) {
45 return -1;
46 }
47 if (rel[0] == '\0') {
48 if ((size_t) snprintf(out->rel, sizeof(out->rel), "%s", name)
49 >= sizeof(out->rel)) {
50 return -1;
51 }
52 return 0;
53 }
54 if ((size_t) snprintf(out->rel, sizeof(out->rel), "%s/%s", rel, name)
55 >= sizeof(out->rel)) {
56 return -1;
57 }
58 return 0;
59}

62/* WHAT: handle one subdirectory during the upload walk: MKCOL it, then recurse.
63 * WHY: creating the remote collection top-down before descending keeps the
64 * child PUTs/MKCOLs from hitting 409 Conflict; S3 has no real dirs.
65 * HOW: MKCOL <base>/<rel> (davs/http only; skipped under --dry-run, which
66 * still recurses so files get printed); on MKCOL failure the subtree is
67 * skipped — its files would 409 — and `fail` is bumped. */
68static void
69walk_handle_dir(web_upload_ctx *c, const walk_child_t *ch)
70{
71 if (!c->u->is_s3 && (c->fo == NULL || !c->fo->dry_run)) {
72 char rpath[XRDC_PATH_MAX * 2];
73 brix_status mst;
74 char proxybuf[512];
75 const char *pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
76
77 brix_status_clear(&mst);
78 if (web_join(c->base, ch->rel, rpath, sizeof(rpath)) != 0
79 || brix_webdav_mkcol(c->u, rpath, c->bearer,
80 c->co ? c->co->verify_host : 1,
81 c->co ? c->co->ca_dir : NULL, pcert, &mst) != 0) {
82 fprintf(stderr, "xrdcp: mkcol %s: %s\n", ch->rel, mst.msg);
83 c->fail++;
84 return; /* skip this subtree — its files would 409 */
85 }
86 }
87 web_upload_walk(c, ch->local, ch->rel);
88}

91/* WHAT: handle one regular file during the upload walk: filter, then PUT.
92 * WHY: keeps the per-file filter/dry-run/URL-build/transfer/accounting chain
93 * out of the walk's directory-scan loop.
94 * HOW: apply --exclude/--include and --dry-run before any I/O; build the
95 * remote URL (overflow-guarded); PUT via copy_one_with_retry; bump `ok`
96 * (progress line unless silent) or `fail` (error line). */
97static void
98walk_handle_file(web_upload_ctx *c, const walk_child_t *ch)
99{
100 char rurl[XRDC_PATH_MAX * 2 + 320];
101 char rpath[XRDC_PATH_MAX * 2];
102 brix_status cst;
103
104 if (c->fo != NULL && !brix_copy_filter_match(c->fo, ch->rel)) {
105 return; /* filtered — skip, not a failure */
106 }
107 if (web_join(c->base, ch->rel, rpath, sizeof(rpath)) != 0
108 || (size_t) snprintf(rurl, sizeof(rurl), "%s://%s:%d%s",
109 c->scheme, c->u->host, c->u->port, rpath)
110 >= sizeof(rurl)) {
111 fprintf(stderr, "xrdcp: remote path too long for %s\n", ch->rel);
112 c->fail++;
113 return;
114 }
115 if (c->fo != NULL && c->fo->dry_run) {
116 printf("[dry-run] copy %s -> %s\n", ch->local, rurl);
117 c->ok++;
118 return;
119 }
120 brix_status_clear(&cst);
121 if (copy_one_with_retry(ch->local, rurl, c->fo, c->co, c->retries,
122 &cst) == 0) {
123 c->ok++;
124 if (c->fo == NULL || !c->fo->silent) {
125 fprintf(stderr, "[%zu] %s -> %s\n", c->ok + c->fail,
126 ch->local, rurl);
127 }
128 } else {
129 c->fail++;
130 fprintf(stderr, "xrdcp: %s: %s\n", rurl, cst.msg);
131 }
132}

135/* Recursively walk a local directory, MKCOL'ing each WebDAV collection (davs/http
136 * only — S3 keys are flat) and PUT'ing each regular file. `rel` is the path of the
137 * current directory relative to the upload root ("" at the top). Symlinks and
138 * special files are skipped (only real dirs + regular files are uploaded). */
139void
140web_upload_walk(web_upload_ctx *c, const char *localdir, const char *rel)
141{
142 DIR *d = opendir(localdir);
143 struct dirent *de;
144
145 if (d == NULL) {
146 fprintf(stderr, "xrdcp: cannot open %s: %s\n", localdir, strerror(errno));
147 c->fail++;
148 return;
149 }
150 while ((de = readdir(d)) != NULL) {
151 walk_child_t ch;
152 struct stat sb;
153
154 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0) {
155 continue;
156 }
157 if (walk_child_paths(localdir, rel, de->d_name, &ch) != 0) {
158 fprintf(stderr, "xrdcp: path too long under %s\n", localdir);
159 c->fail++;
160 continue;
161 }
162 if (lstat(ch.local, &sb) != 0) {
163 fprintf(stderr, "xrdcp: stat %s: %s\n", ch.local, strerror(errno));
164 c->fail++;
165 continue;
166 }
167 if (S_ISDIR(sb.st_mode)) {
168 walk_handle_dir(c, &ch);
169 } else if (S_ISREG(sb.st_mode)) {
170 walk_handle_file(c, &ch);
171 }
172 /* else: symlink / fifo / device — skip (not uploaded) */
173 }
174 closedir(d);
175}

178/* Recursively upload a local directory's CONTENTS into a web (davs/http/s3)
179 * collection: `xrdcp -r ./dir davs://h/coll/` → coll/<files-under-dir>. The wire
180 * has no recursive transfer op, so walk locally + MKCOL + per-file PUT. Returns
181 * 0 if every file uploaded, 1 otherwise. */
182int
183recursive_web_upload(const char *localdir, const char *dst, const brix_copy_opts *o,
184 const brix_opts *co, int retries)
185{
186 brix_weburl u;
187 brix_copy_opts fo;
188 char base[XRDC_PATH_MAX];
189 web_upload_ctx c;
190 size_t blen;
191
192 if (brix_weburl_parse(dst, &u) != 0) {
193 fprintf(stderr, "xrdcp: bad web URL %s\n", dst);
194 return 1;
195 }
196 /* Each file is a plain (non-recursive) copy so brix_copy's "no recursive web"
197 * guard doesn't trip. */
198 fo = *o;
199 fo.recursive = 0;
200
201 snprintf(base, sizeof(base), "%s", u.path);
202 blen = strlen(base);
203 while (blen > 0 && base[blen - 1] == '/') { base[--blen] = '\0'; }
204
205 c.u = &u;
206 c.base = base;
207 c.scheme = web_scheme_str(u.proto);
208 c.bearer = (o != NULL && o->bearer != NULL) ? o->bearer : getenv("BEARER_TOKEN");
209 c.fo = &fo;
210 c.co = co;
211 c.retries = retries;
212 c.ok = 0;
213 c.fail = 0;
214
215 /* Ensure the destination collection itself exists (idempotent). Root ("")
216 * and S3 buckets need no MKCOL. Under --dry-run, skip to avoid creating
217 * the base collection on the remote. */
218 if (!u.is_s3 && base[0] != '\0' && !fo.dry_run) {
219 brix_status mst;
220 char proxybuf[512];
221 const char *pcert = brix_web_proxy_pem(proxybuf, sizeof(proxybuf));
222 brix_status_clear(&mst);
223 if (brix_webdav_mkcol(&u, base, c.bearer, co ? co->verify_host : 1,
224 co ? co->ca_dir : NULL, pcert, &mst) != 0) {
225 fprintf(stderr, "xrdcp: mkcol %s: %s\n", base, mst.msg);
226 /* proceed anyway — PUTs will surface any real problem */
227 }
228 }
229
230 web_upload_walk(&c, localdir, "");
231 if (o == NULL || !o->silent) {
232 fprintf(stderr, "xrdcp: %zu copied, %zu failed (recursive web upload)\n",
233 c.ok, c.fail);
234 }
235 return (c.fail == 0) ? 0 : 1;
236}
