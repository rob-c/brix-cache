1/*
2 * pblock_refs.h — Phase-83 F10: content-addressed dedup + refcounted blobs.
3 *
4 * WHAT: The foundation of wave D: a driver-owned `blobs(blob_id, refcount,
5 * size, block_size, content_hash)` table that lets several `objects` rows
6 * share one physical blob. server_copy becomes a refcount bump
7 * (O(metadata) copy-on-write), publish-time dedup folds identical content
8 * onto an existing blob, and any write to a shared blob first breaks the
9 * share by copying the blocks to a fresh blob (block granularity makes
10 * that cheap).
11 *
12 * WHY: Exercises every CAP_SERVER_COPY caller (kXR clone, WebDAV COPY) against
13 * real copy-on-write semantics, and gives F6 snapshots / F11 versioning
14 * their O(metadata) foundation.
15 *
16 * HOW: Armed by the `dedup=1` static opt (its own gate, like csi/locks — this
17 * is a capability, not a lab toy); an init failure leaves the
18 * byte-for-byte production path. A row's absence means "refcount 1"
19 * (legacy blobs from before the gate was set), so enabling dedup on a
20 * populated export is safe. The content hash (phase-88 W3) is a
21 * PREFIXED STRING: "sha256:<hex>" from the write path's in-order SHA
22 * accumulator (or computed on demand at the dedup_publish slot) — a
23 * cryptographic identity a fold TRUSTS outright, making pblock a real
24 * CAS; "crc32:<hex>" for legacy rows and out-of-order writers — only a
25 * CANDIDATE, always byte-verified before linking so a CRC collision can
26 * never alias content. (A forged sha256 row is out of the threat model:
27 * writing catalog.db at all means owning the whole store. csi=1 remains
28 * the at-rest bit-rot detector either way.) Share-break happens at OPEN
29 * for write-intent opens (a metadata boundary), keeping the byte hot
30 * path free of refcount reads. ngx-free (libc + sqlite3 + libcrypto);
31 * BRIX_HAVE_SQLITE-gated.
32 *
33 * Requires: pblock_store.h (pblock_state_t), sd_pblock_catalog.h (pblock_meta)
34 * before inclusion.
35 */

49/* Build the preferred content-hash string for a completely-written object out
50 * of its write accumulator: "sha256:<hex>" when the in-order SHA covered
51 * exactly `size` bytes, else "crc32:<hex>" when the (order-independent) CRC

53 * PBLOCK_REFS_HASH_CAP. NULL-safe on wv. */

74/* Release one reference to a blob — the single unlink/overwrite entry for BOTH
75 * gate states: with refs off (or on the last reference) it removes the block
76 * files and the blob's csi rows exactly like the pre-F10 code; with a live
77 * shared reference it only decrements. A DB error keeps the blocks (an orphan
78 * for `pblock-fsck --gc`, never a shared-block removal). */
79void pblock_refs_release(const pblock_state_t *st, const char *blob_id,
80 int64_t size, int64_t bs);

82/* Publish-time dedup: with the object's row already pointing at meta->blob_id,
83 * look for an identical existing blob (hash+size+block_size candidates; a
84 * "sha256:" identity folds outright, a "crc32:" candidate is byte-verified
85 * first) and fold onto it — bump the survivor, repoint the row, release our
86 * now-redundant blob. Skips (tracking the blob as-is) when `hash` is empty,
87 * when our blob is already shared, or when no candidate survives. Returns 1
88 * (folded), 0 (kept), -1/errno. */
89int pblock_refs_dedup_publish(pblock_state_t *st, const char *path,
90 pblock_meta *meta, const char *hash);
91
92/* Explicit dedup of an ALREADY-COMMITTED object (phase-88 W1: the
93 * driver->dedup_publish slot): re-nominate `path`'s blob for a fold using the
94 * content hash recorded in its blobs row at commit time — computing (and
95 * recording) a fresh "sha256:" identity from the stored bytes when none was
96 * (a legacy blob, or an out-of-order write history forfeited it). Skips (0)
97 * when the object is a dir/empty or its blob is already shared. Returns 1
98 * (folded), 0 (kept), -1/errno. */
99int pblock_refs_dedup_existing(pblock_state_t *st, const char *path);
100
101/* Copy-on-write share-break for a write-intent open: when meta->blob_id is
102 * shared, copy its blocks to a fresh private blob (or start empty when `trunc`
103 * — the content is being replaced anyway), repoint the object's row, carry the
104 * csi rows over, and release the old reference. Updates meta->blob_id in
105 * place. 0 (private — possibly a no-op) or -1/errno. */
106int pblock_refs_break_share(pblock_state_t *st, const char *path,
107 pblock_meta *meta, int trunc);
