22/* Append a result. status: >0 pass, 0 fail, <0 skipped. */
23void
24bat_add(xrd_battery *b, const char *name, int status, const char *fmt, ...)
25{
26 xrd_check *c;
27 va_list ap;
28 if (b->n >= XRD_MAX_CHECKS) { return; }
29 c = &b->checks[b->n++];
30 snprintf(c->name, sizeof(c->name), "%s", name);
31 c->ok = (status > 0);
32 c->skipped = (status < 0);
33 va_start(ap, fmt);
34 vsnprintf(c->detail, sizeof(c->detail), fmt, ap);
35 va_end(ap);
36 if (status < 0) { b->nskip++; }
37 else if (status > 0) { b->npass++; }
38 else { b->nfail++; }
39}

51/* Write `buf` to an anonymous tmpfile and return its fd (rewound), or -1. */
52int
53tmpfile_with(const uint8_t *buf, size_t n)
54{
55 FILE *f = tmpfile();
56 int fd;
57 if (f == NULL) { return -1; }
58 if (n > 0 && fwrite(buf, 1, n, f) != n) { fclose(f); return -1; }
59 fflush(f);
60 fd = dup(fileno(f));
61 fclose(f);
62 if (fd >= 0) { lseek(fd, 0, SEEK_SET); }
63 return fd;
64}

264probe_write_suite(brix_conn *c, int ext_sa, int ext_sl, int ext_rl, xrd_battery *b)
265{

307 write_xattr_ops(c, file, b);
308 write_symlink_ops(c, dir, file, ext_sl, ext_rl, &sym_left, b);
309 write_rename_truncate_rm(c, dir, file, file2, sym_left, b);

313/* The native root:// functional battery: always-safe reads, then (do_write) a full
314 * write/read/verify/checksum/metadata cycle under a temp dir that is cleaned up. */
315void
316battery_root(const brix_url *u, const brix_opts *o, int do_write, xrd_battery *b)
317{
318 brix_conn c;
319 brix_status st;
320 int ext_sa = 0, ext_sl = 0, ext_rl = 0, ext_ln = 0;
321
322 snprintf(b->protocol, sizeof(b->protocol), "root");
323 brix_status_clear(&st);
324 if (brix_connect(&c, u, o, &st) != 0) {
325 snprintf(b->err, sizeof(b->err), "%s", st.msg);
326 return;
327 }
328 b->reachable = 1;
329
330 probe_root_read_suite(&c, b);
331 probe_path_confinement(&c, b);
332
333 (void) brix_ext_probe(&c, &ext_sa, &ext_sl, &ext_rl, &ext_ln, &st);
334
335 if (!do_write) {
336 bat_add(b, "write-suite", -1, "skipped (pass --rw to run write tests)");
337 brix_close(&c);
338 return;
339 }
340
341 probe_write_suite(&c, ext_sa, ext_sl, ext_rl, b);
342 brix_close(&c);
343}

348void
349xrd_run_battery(const char *endpoint, int do_write, int verify, xrd_battery *b)
350{
351 memset(b, 0, sizeof(*b));
352 snprintf(b->endpoint, sizeof(b->endpoint), "%s", endpoint);
353
354 if (brix_is_web_url(endpoint)) {
355 brix_weburl w;
356 if (brix_weburl_parse(endpoint, &w) != 0) {
357 snprintf(b->protocol, sizeof(b->protocol), "web");
358 snprintf(b->err, sizeof(b->err), "unparseable web URL");
359 return;
360 }
361 if (w.is_s3) {
362 const char *region = getenv("AWS_DEFAULT_REGION");
363 battery_s3(&w, do_write, getenv("AWS_ACCESS_KEY_ID"),
364 getenv("AWS_SECRET_ACCESS_KEY"),
365 region ? region : "us-east-1", verify, b);
366 } else {
367 char *tok = brix_token_discover();
368 battery_web(&w, do_write, tok, verify, b);
369 if (tok != NULL) { free(tok); }
370 }
371 return;
372 }
373 {
374 brix_url u;
375 brix_opts o;
376 brix_status st;
377 memset(&o, 0, sizeof(o));
378 o.verify_host = verify;
379 brix_status_clear(&st);
380 snprintf(b->protocol, sizeof(b->protocol), "root");
381 if (brix_endpoint_parse(endpoint, &u, &st) != 0) {
382 snprintf(b->err, sizeof(b->err), "%s", st.msg);
383 return;
384 }
385 battery_root(&u, &o, do_write, b);
386 }
