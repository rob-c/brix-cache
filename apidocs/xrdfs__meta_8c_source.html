8/* WHAT: Evaluates one stock `stat -q` query string ("IsDir|Offline",
9 * "IsReadable&IsDir", …) against a stat flags word. Returns 1 (holds),
10 * 0 (does not hold), or -1 (unknown flag name — usage error).
11 * WHY: stock xrdfs contract, pinned live against xrdfs 5.6.9: '&' requires
12 * every named flag, '|' is satisfied by any; evaluation is a linear
13 * left-to-right fold with no precedence. Exit codes at the caller:
14 * 0 holds, 55 fails, 50 unknown flag.
15 * HOW: walk '&'/'|'-separated tokens; map each name through the stock
16 * vocabulary table to its kXR bit; fold presence with the separator
17 * that PRECEDED the token. */
18static int
19stat_query_eval(const char *query, int flags)
20{
21 static const struct { const char *name; int bit; } vocab[] = {
22 { "XBitSet", kXR_xset },
23 { "IsDir", kXR_isDir },
24 { "Other", kXR_other },
25 { "Offline", kXR_offline },
26 { "POSCPending", kXR_poscpend },
27 { "IsReadable", kXR_readable },
28 { "IsWriteable", kXR_writable },
29 { NULL, 0 },
30 };
31 const char *cursor = query;
32 int holds = -1; /* -1 = first token pending */
33 char sep = 0;
34
35 while (*cursor != '\0') {
36 char token[24];
37 size_t token_len = 0;
38 int bit = -1, present, k;
39
40 while (*cursor != '\0' && *cursor != '&' && *cursor != '|') {
41 if (token_len + 1 < sizeof(token)) {
42 token[token_len++] = *cursor;
43 }
44 cursor++;
45 }
46 token[token_len] = '\0';
47
48 for (k = 0; vocab[k].name != NULL; k++) {
49 if (strcmp(token, vocab[k].name) == 0) {
50 bit = vocab[k].bit;
51 break;
52 }
53 }
54 if (bit < 0) {
55 return -1;
56 }
57
58 present = (flags & bit) != 0;
59 if (holds < 0) {
60 holds = present;
61 } else if (sep == '&') {
62 holds = holds && present;
63 } else {
64 holds = holds || present;
65 }
66
67 if (*cursor != '\0') {
68 sep = *cursor;
69 cursor++;
70 }
71 }
72
73 return holds < 0 ? -1 : holds;
74}

92/* Stat one operand: build_path → brix_stat → print → evaluate -q. Returns 0
93 * when the path stats clean (and satisfies -q when given), 55 when the -q
94 * query does not hold, -1 on an unknown -q flag (caller aborts with 50), else
95 * the per-path report code. On error nothing goes to stdout for the path so
96 * partial JSON is never emitted. */
97static int
98stat_one_path(brix_conn *c, const char *cwd, const char *arg, int json,
99 const char *query)
100{
101 brix_status st;
102 brix_statinfo si;
103 char path[XRDC_PATH_MAX];
104
105 build_path(cwd, arg, path, sizeof(path));
106
107 brix_status_clear(&st);
108 if (brix_stat(c, path, &si, &st) != 0) {
109 return xrdfs_report_err("stat", path, &st, 0, c);
110 }
111 if (json) { json_statinfo(path, &si); } else { print_statinfo(path, &si); }
112
113 if (query != NULL) {
114 int q = stat_query_eval(query, si.flags);
115
116 if (q < 0) {
117 fprintf(stderr, "xrdfs: stat: unknown -q flag in '%s' "
118 "(XBitSet, IsDir, Other, Offline, POSCPending, "
119 "IsReadable, IsWriteable)\n", query);
120 return -1;
121 }
122 if (q == 0) {
123 return 55; /* stock: query not satisfied */
124 }
125 }
126 return 0;
127}

129/* WHAT: stat [-j] [-q query] <path> [path ...] — print metadata for every
130 * named path in human or JSON format (one JSON object per path).
131 * Returns 0 when all paths stat cleanly (and, with -q, every path
132 * satisfies the flag query), 55 when a path fails the query, else the
133 * first failure's exit code.
134 * WHY: -j enables machine-readable output for scripting and pipeline use.
135 * -q is the stock flag-query contract (parity audit §7.12): shell
136 * scripts branch on "is this a directory / offline / writable" without
137 * parsing output. Multiple operands used to silently act on the LAST
138 * path only (feature-parity audit §9.2); every operand now gets its own
139 * stat, and a failure is reported without hiding the remaining paths'
140 * output.
141 * HOW: stat_parse_flags gathers the flags; the operand pass runs
142 * stat_one_path per path, remembering the first failing exit code
143 * (unknown -q flag aborts the whole command with 50). */
144int
145do_stat(brix_conn *c, const char *cwd, int argc, char **argv)
146{
147 const char *query = NULL;
148 int json = 0, worst = 0, npaths = 0, i;
149
150 stat_parse_flags(argc, argv, &json, &query);
151 for (i = 1; i < argc; i++) {
152 int rc;
153
154 if (strcmp(argv[i], "-j") == 0 || strcmp(argv[i], "--json") == 0) {
155 continue;
156 }
157 if (strcmp(argv[i], "-q") == 0 && i + 1 < argc) {
158 i++; /* skip the query value operand */
159 continue;
160 }
161 npaths++;
162 rc = stat_one_path(c, cwd, argv[i], json, query);
163 if (rc < 0) {
164 return 50; /* unknown -q flag: usage error */
165 }
166 if (rc != 0 && worst == 0) {
167 worst = rc;
168 }
169 }
170 if (npaths == 0) {
171 fprintf(stderr, "usage: stat [-j] [-q query] <path> [path ...]\n");
172 return 50;
173 }
174 return worst;
175}

186/* WHAT: walk_dir visitor for locate -d — locates every non-directory entry
187 * and prints "<path>: <reply>"; failures are reported and counted, the
188 * walk continues. Always returns 0 (never aborts the walk).
189 * WHY: stock deep-locate reports the locations of every FILE under the tree;
190 * one dead entry must not hide the rest.
191 * HOW: skip directories (their children are visited anyway); locate with the
192 * caller's wire options; print or report per entry. */
193static int
194locate_deep_visit(const char *full, const brix_dirent *e, int depth, void *u)
195{
196 locate_deep_ctx *dc = (locate_deep_ctx *) u;
197 brix_status st;
198 char reply[1024];
199
200 (void) depth;
201 if (e->have_stat && (e->st.flags & kXR_isDir)) {
202 return 0;
203 }
204 brix_status_clear(&st);
205 if (brix_locate_opts(dc->conn, full, dc->options, reply, sizeof(reply),
206 &st) != 0) {
207 (void) xrdfs_report_err("locate", full, &st, 0, dc->conn);
208 dc->failures++;
209 return 0;
210 }
211 printf("%s: %s\n", full, reply);
212 return 0;
213}

215/* Fold the locate flag surface into wire option bits. Returns 0 with *arg
216 * set, or 50 after printing usage (unknown flag / missing path). */
217static int
218locate_parse_opts(int argc, char **argv, unsigned *options, int *deep,
219 const char **arg)
220{
221 int i;
222
223 for (i = 1; i < argc; i++) {
224 if (strcmp(argv[i], "-r") == 0) {
225 *options |= kXR_refresh;
226 } else if (strcmp(argv[i], "-n") == 0) {
227 *options |= kXR_nowait;
228 } else if (strcmp(argv[i], "-m") == 0 || strcmp(argv[i], "-h") == 0) {
229 *options |= kXR_prefname;
230 } else if (strcmp(argv[i], "-d") == 0) {
231 *deep = 1;
232 } else if (strcmp(argv[i], "-i") == 0 || strcmp(argv[i], "-p") == 0) {
233 /* accepted for stock CLI compatibility; see WHY above */
234 } else if (argv[i][0] == '-') {
235 break; /* unknown flag: usage below */
236 } else {
237 *arg = argv[i];
238 }
239 }
240 if (i < argc || *arg == NULL) {
241 fprintf(stderr,
242 "usage: locate [-n] [-r] [-d] [-m|-h] [-i] [-p] <path>\n");
243 return 50;
244 }
245 return 0;
246}

248/* locate -d on a directory: walk the tree, locating every file. Returns the
249 * command exit code, or -1 when the target is a plain file (caller falls
250 * through to the single-path locate — stock behaviour). */
251static int
252locate_deep(brix_conn *c, const char *path, unsigned options)
253{
254 brix_status st;
255 brix_statinfo si;
256
257 brix_status_clear(&st);
258 if (brix_stat(c, path, &si, &st) != 0) {
259 return xrdfs_report_err("locate", path, &st, 0, c);
260 }
261 if (!(si.flags & kXR_isDir)) {
262 return -1; /* plain file: single locate */
263 }
264 {
265 locate_deep_ctx dc = { c, options, 0 };
266
267 brix_status_clear(&st);
268 if (walk_dir(c, path, 0, locate_deep_visit, &dc, &st) < 0) {
269 return xrdfs_report_err("locate -d", path, &st, 0, c);
270 }
271 return dc.failures ? 54 : 0;
272 }
273}

275/* WHAT: locate [-n] [-r] [-d] [-m|-h] [-i] [-p] <path> — resolve a path to
276 * its serving endpoint(s); -d recurses, locating every file under a
277 * directory. Returns 0 on success, else the first failure's exit code.
278 * WHY: stock xrdfs flag surface (parity audit §7.12 — options were hardcoded
279 * to 0): -r sets kXR_refresh (BriX flushes + bypasses the loc/redirect
280 * caches, §2.7), -n sets kXR_nowait (immediate possibly-incomplete
281 * answer; stock servers honor it), -m/-h set kXR_prefname (DNS names
282 * over IPs — already this server's default). -i (no name resolution)
283 * and -p (no tried= opaque) are accepted for stock CLI compatibility:
284 * this client already never resolves names in locate replies and never
285 * sends tried= on a first attempt, so both are satisfied by default.
286 * HOW: locate_parse_opts folds flags into wire option bits; -d routes
287 * through locate_deep (a directory walks via walk_dir with
288 * locate_deep_visit; anything else falls through to the single
289 * locate). */
290int
291do_locate(brix_conn *c, const char *cwd, int argc, char **argv)
292{
293 brix_status st;
294 char path[XRDC_PATH_MAX], reply[1024];
295 const char *arg = NULL;
296 unsigned options = 0;
297 int deep = 0, rc;
298
299 rc = locate_parse_opts(argc, argv, &options, &deep, &arg);
300 if (rc != 0) {
301 return rc;
302 }
303 build_path(cwd, arg, path, sizeof(path));
304
305 if (deep) {
306 rc = locate_deep(c, path, options);
307 if (rc >= 0) {
308 return rc;
309 }
310 /* -d on a plain file falls through to the single locate */
311 }
312
313 brix_status_clear(&st);
314 if (brix_locate_opts(c, path, options, reply, sizeof(reply), &st) != 0) {
315 return xrdfs_report_err("locate", path, &st, 0, c);
316 }
317 printf("%s\n", reply);
318 return 0;
319}

323/* WHAT: cache {evict|fevict} <path> — the stock operator cache-eviction

357/* WHAT: spaceinfo [path] — the stock xrdfs report: Path/Total/Free/Used/
358 * Largest free chunk, one per line, labels padded to column 21 (byte
359 * shape captured live from stock 5.6.9). Returns 0 / mapped code.
360 * WHY: §7.12 — BriX had no spaceinfo verb at all; stock scripts parse this
361 * exact layout. The numbers come from kXR_Qspace's oss.* keys
362 * (space/free/used/maxf), the same source stock reads.
363 * HOW: query Qspace on the (cwd-resolved) path, pull the four keys out of
364 * the &-separated reply, print stock's five lines. */
365int
366do_spaceinfo(brix_conn *c, const char *cwd, int argc, char **argv)
367{
368 brix_status st;
369 char path[XRDC_PATH_MAX], reply[1024];
370 const char *keys[4] = { "oss.space=", "oss.free=", "oss.used=",
371 "oss.maxf=" };
372 const char *labels[4] = { "Total:", "Free:", "Used:",
373 "Largest free chunk:" };
374 long long vals[4];
375 int k;
376
377 build_path(cwd, argc >= 2 ? argv[1] : "/", path, sizeof(path));
378 brix_status_clear(&st);
379 if (brix_query(c, kXR_Qspace, path, reply, sizeof(reply), &st) != 0) {
380 return xrdfs_report_err("spaceinfo", path, &st, 0, c);
381 }
382 for (k = 0; k < 4; k++) {
383 const char *hit = strstr(reply, keys[k]);
384
385 if (hit == NULL) {
386 fprintf(stderr, "xrdfs: spaceinfo: server reply lacks %s\n",
387 keys[k]);
388 return 54;
389 }
390 vals[k] = atoll(hit + strlen(keys[k]));
391 }
392 printf("%-20s%s\n", "Path:", path);
393 for (k = 0; k < 4; k++) {
394 printf("%-20s%lld\n", labels[k], vals[k]);
395 }
396 return 0;
397}

416/* df [-h] [path] — friendly disk-space report over kXR_Qspace (the server's oss.*
417 * capacity record). Default path "/". -h humanizes the byte columns. Falls back to
418 * printing the raw reply verbatim when the shape is unrecognized (never crashes).

421do_df(brix_conn *c, const char *cwd, int argc, char **argv)

445 fmt_size(total >= 0 ? total : 0, ts, sizeof(ts), human);
446 fmt_size(used >= 0 ? used : 0, us, sizeof(us), human);

451 } else {
452 snprintf(pctbuf, sizeof(pctbuf), "-");

455 "Size", "Used", "Avail", "Use%", "Largest", "Path");
456 printf("%-12s %-12s %-12s %-6s %-12s %s\n", ts, us, as, pctbuf, ls, path);
457 }
458 return 0;
459}
