14/*
15 * ftp_ev_data_setup.c — the four data-channel setup verbs (PASV/EPSV/PORT/EPRT).
16 *
17 * WHAT: parse and answer the verbs that *nominate* a data endpoint, before any
18 * transfer verb runs: PASV/EPSV open a non-blocking listener and publish its
19 * address; PORT/EPRT pin and screen a client-nominated target.
20 *
21 * WHY: split out of ftp_ev_data.c (coding-standards §1) because nomination and
22 * bring-up are separate concerns with separate risk profiles — this half is
23 * argument parsing and the anti-bounce/SSRF boundary, the other half is event
24 * plumbing and teardown. Keeping the screen beside its two parsers means the
25 * security rule and the strings it validates are read together.
26 *
27 * HOW: state is deposited on the session (fc->pasv_fd for passive,
28 * fc->active_sa + fc->active for active) and consumed later by
29 * brix_ftp_ev_data_open() in ftp_ev_data.c. The confinement rules are ported
30 * verbatim from the sync engine (ftp_do_pasv/ftp_do_port) so both engines expose
31 * the same data-channel security boundary.
32 */

35/* Create the non-blocking listen socket and report the control connection's own
36 * local address in `local` — the listener must bind the same host the client is
37 * already talking to. Returns the fd, or -1 (nothing left open) on any failure. */
38static int
39ev_pasv_open_socket(ftp_ev_t *fc, struct sockaddr_in *local)
40{
41 socklen_t llen = sizeof(*local);
42 int fd;
43
44 ngx_memzero(local, sizeof(*local)); /* getsockname may leave it */
45
46 if (getsockname(fc->c->fd, (struct sockaddr *) local, &llen) != 0
47 || local->sin_family != AF_INET)
48 {
49 return -1;
50 }
51
52 fd = socket(AF_INET, SOCK_STREAM, 0);
53 if (fd < 0) {
54 return -1;
55 }
56 if (ngx_nonblocking(fd) == -1) {
57 (void) close(fd);
58 return -1;
59 }
60 return fd;
61}

64/* Bind `fd` to the first free port of the configured passive range, updating
65 * `addr` in place. 0 on success; -1 when the window is exhausted (or a
66 * non-contention error ended the walk, which the next port would not clear). */
67static int
68ev_pasv_bind_range(ftp_ev_t *fc, int fd, struct sockaddr_in *addr)
69{
70 ngx_int_t p;
71
72 for (p = fc->conf->pasv_port_lo; p <= fc->conf->pasv_port_hi; p++) {
73 addr->sin_port = htons((unsigned short) p);
74 if (bind(fd, (struct sockaddr *) addr, sizeof(*addr)) == 0) {
75 return 0;
76 }
77 if (errno != EADDRINUSE && errno != EADDRNOTAVAIL) {
78 break;
79 }
80 }
81 return -1;
82}

85/* Start listening and read back the address actually bound (the port is only
86 * known here in the ephemeral case). 0 on success, -1 otherwise. */
87static int
88ev_pasv_listen(int fd, struct sockaddr_in *addr)
89{
90 socklen_t blen = sizeof(*addr);
91
92 /* Backlog spans a full MODE E stream fan-out (globus opens all parallel data
93 * connections at once), matching the sync engine. */
94 if (listen(fd, BRIX_FTP_EV_DATA_BACKLOG) != 0
95 || getsockname(fd, (struct sockaddr *) addr, &blen) != 0)
96 {
97 return -1;
98 }
99 return 0;
100}

103/* Publish the listener: RFC 2428 (229, port only) or RFC 959 (227, h1..h4,p1,p2). */
104static ngx_int_t
105ev_pasv_reply(ftp_ev_t *fc, const struct sockaddr_in *addr, int extended)
106{
107 unsigned char ip[4];
108 unsigned port = ntohs(addr->sin_port);
109
110 if (extended) {
111 return brix_ftp_ev_reply(fc,
112 "229 Entering Extended Passive Mode (|||%ud|)\r\n",
113 (ngx_uint_t) port);
114 }
115
116 ngx_memcpy(ip, &addr->sin_addr.s_addr, 4);
117 return brix_ftp_ev_reply(fc,
118 "227 Entering Passive Mode (%ud,%ud,%ud,%ud,%ud,%ud)\r\n",
119 (ngx_uint_t) ip[0], (ngx_uint_t) ip[1],
120 (ngx_uint_t) ip[2], (ngx_uint_t) ip[3],
121 (ngx_uint_t) (port >> 8), (ngx_uint_t) (port & 0xff));
122}

125/* PASV/EPSV: open a non-blocking listener bound to the control connection's local
126 * IP and an ephemeral port; `extended` selects the RFC 2428 (229) vs RFC 959
127 * (227) reply. Mirrors the sync ftp_do_pasv, but the socket is non-blocking so
128 * the later accept runs under the event loop. */
129static ngx_int_t
130ev_do_pasv(ftp_ev_t *fc, int extended)
131{
132 struct sockaddr_in local;
133 struct sockaddr_in bindaddr;
134 int fd;
135
136 fc->active = 0; /* PASV/EPSV overrides PORT */
137
138 if (fc->pasv_fd >= 0) {
139 (void) close(fc->pasv_fd);
140 fc->pasv_fd = -1;
141 }
142
143 fd = ev_pasv_open_socket(fc, &local);
144 if (fd < 0) {
145 return brix_ftp_ev_reply(fc, "425 Cannot open passive connection\r\n");
146 }
147
148 ngx_memzero(&bindaddr, sizeof(bindaddr));
149 bindaddr.sin_family = AF_INET;
150 bindaddr.sin_addr = local.sin_addr; /* same host as control */
151
152 /* Bind the listener. With a configured passive range (brix_gridftp_pasv_port_range)
153 * we walk it and take the first free port so the peer only ever has to reach a
154 * firewall-opened window; unset (lo == 0) keeps the kernel-ephemeral bind. A
155 * fully-occupied range fails the transfer (rather than falling back to a
156 * random, un-firewalled port) so a locked-down deployment stays predictable. */
157 if (fc->conf->pasv_port_lo > 0) {
158 if (ev_pasv_bind_range(fc, fd, &bindaddr) != 0) {
159 (void) close(fd);
160 return brix_ftp_ev_reply(fc,
161 "425 No free passive port in configured range\r\n");
162 }
163 } else {
164 bindaddr.sin_port = 0; /* ephemeral */
165 if (bind(fd, (struct sockaddr *) &bindaddr, sizeof(bindaddr)) != 0) {
166 (void) close(fd);
167 return brix_ftp_ev_reply(fc, "425 Cannot open passive connection\r\n");
168 }
169 }
170
171 if (ev_pasv_listen(fd, &bindaddr) != 0) {
172 (void) close(fd);
173 return brix_ftp_ev_reply(fc, "425 Cannot open passive connection\r\n");
174 }
175
176 fc->pasv_fd = fd;
177 return ev_pasv_reply(fc, &bindaddr, extended);
178}

181/* Parse a classic "h1,h2,h3,h4,p1,p2" PORT argument into (addr, port). Returns
182 * NGX_DECLINED on success (fields filled); on a malformed argument it emits the
183 * 501 reply and returns that reply's status for the caller to propagate. */
184static ngx_int_t
185ev_port_parse_classic(ftp_ev_t *fc, const char *arg, in_addr_t *addr,
186 unsigned *port)
187{
188 unsigned h[4], p[2];
189 unsigned char b[4];
190
191 if (sscanf(arg, "%u,%u,%u,%u,%u,%u",
192 &h[0], &h[1], &h[2], &h[3], &p[0], &p[1]) != 6
193 || (h[0] | h[1] | h[2] | h[3]) > 255 || (p[0] | p[1]) > 255)
194 {
195 return brix_ftp_ev_reply(fc, "501 Bad PORT argument\r\n");
196 }
197 b[0] = (unsigned char) h[0]; b[1] = (unsigned char) h[1];
198 b[2] = (unsigned char) h[2]; b[3] = (unsigned char) h[3];
199 ngx_memcpy(addr, b, 4);
200 *port = (p[0] << 8) | p[1];
201 return NGX_DECLINED;
202}

204/* Parse an "|1|ip|port|" EPRT argument (IPv4 only) into (addr, port). Same
205 * NGX_DECLINED-on-success / reply-status-on-error contract as the classic form. */
206static ngx_int_t
207ev_port_parse_extended(ftp_ev_t *fc, const char *arg, in_addr_t *addr,
208 unsigned *port)
209{
210 char d = arg[0];
211 const char *fam, *ip, *pt, *end;
212 char ipbuf[64];
213 size_t iplen;
214
215 if (d == '\0') { return brix_ftp_ev_reply(fc, "501 Bad EPRT argument\r\n"); }
216 fam = arg + 1;
217 ip = strchr(fam, d);
218 if (ip == NULL) { return brix_ftp_ev_reply(fc, "501 Bad EPRT argument\r\n"); }
219 ip++;
220 pt = strchr(ip, d);
221 if (pt == NULL) { return brix_ftp_ev_reply(fc, "501 Bad EPRT argument\r\n"); }
222 end = strchr(pt + 1, d);
223 if (end == NULL) { return brix_ftp_ev_reply(fc, "501 Bad EPRT argument\r\n"); }
224 if (fam[0] != '1') {
225 return brix_ftp_ev_reply(fc, "522 Only IPv4 (|1|) supported\r\n");
226 }
227 iplen = (size_t) (pt - ip);
228 if (iplen == 0 || iplen >= sizeof(ipbuf)) {
229 return brix_ftp_ev_reply(fc, "501 Bad EPRT address\r\n");
230 }
231 ngx_memcpy(ipbuf, ip, iplen);
232 ipbuf[iplen] = '\0';
233 if (inet_pton(AF_INET, ipbuf, addr) != 1) {
234 return brix_ftp_ev_reply(fc, "501 Bad EPRT address\r\n");
235 }
236 *port = (unsigned) strtoul(pt + 1, NULL, 10);
237 return NGX_DECLINED;
238}

240/* Anti-bounce pin + SSRF screen for an active-mode target. A plain transfer must
241 * target the control peer; an off-peer target is only allowed as a GSI (DCAU A)
242 * TPC leg. Returns NGX_DECLINED when the target is permitted (fc->active_offpeer
243 * recorded), else the emitted reply's status. */
244static ngx_int_t
245ev_port_screen_target(ftp_ev_t *fc, const struct sockaddr_in *tgt)
246{
247 struct sockaddr_in peer;
248 socklen_t plen = sizeof(peer);
249 brix_net_target_policy_t pol;
250 char err[128];
251
252 ngx_memzero(&peer, sizeof(peer)); /* getpeername may leave it */
253 if (getpeername(fc->c->fd, (struct sockaddr *) &peer, &plen) != 0
254 || peer.sin_family != AF_INET)
255 {
256 return brix_ftp_ev_reply(fc, "500 Cannot determine control peer\r\n");
257 }
258 fc->active_offpeer = (peer.sin_addr.s_addr != tgt->sin_addr.s_addr);
259 if (fc->active_offpeer && !(fc->sec_active && fc->dcau_a)) {
260 ngx_log_error(NGX_LOG_WARN, fc->c->log, 0,
261 "brix: gsiftp(ev) rejected active-mode target != control "
262 "peer (no DCAU A; possible FTP-bounce)");
263 return brix_ftp_ev_reply(fc, "500 Data address must match control peer\r\n");
264 }
265
266 ngx_memzero(&pol, sizeof(pol));
267 pol.allow_local = 1;
268 pol.allow_private = 1;
269 if (brix_net_target_check_addr((struct sockaddr *) tgt, &pol,
270 err, sizeof(err)) != NGX_OK)
271 {
272 ngx_log_error(NGX_LOG_WARN, fc->c->log, 0,
273 "brix: gsiftp(ev) active-mode target blocked: %s", err);
274 return brix_ftp_ev_reply(fc, "500 Data address not permitted\r\n");
275 }
276 return NGX_DECLINED;
277}

279/* PORT/EPRT: arm an active-mode target. The nominated address is pinned to the
280 * control peer (anti FTP-bounce; relaxed only for a DCAU-A TPC leg) and screened
281 * through the SSRF policy — identical to the sync ftp_do_port. */
282static ngx_int_t
283ev_do_port(ftp_ev_t *fc, const char *arg, int extended)
284{
285 struct sockaddr_in tgt;
286 in_addr_t addr = 0;
287 unsigned port = 0;
288 ngx_int_t rc;
289
290 /* Seeded, not merely declared: the parsers write (addr, port) only on their
291 * success path, so a helper that ever signalled an error as NGX_DECLINED
292 * would leave the caller dialling an uninitialised address. Zero makes that
293 * degrade into the "501 Bad data port" arm below instead. */
294 ngx_memzero(&tgt, sizeof(tgt));
295 tgt.sin_family = AF_INET;
296
297 rc = extended ? ev_port_parse_extended(fc, arg, &addr, &port)
298 : ev_port_parse_classic(fc, arg, &addr, &port);
299 if (rc != NGX_DECLINED) {
300 return rc;
301 }
302
303 if (port == 0 || port > 65535) {
304 return brix_ftp_ev_reply(fc, "501 Bad data port\r\n");
305 }
306 tgt.sin_addr.s_addr = addr;
307 tgt.sin_port = htons((unsigned short) port);
308
309 rc = ev_port_screen_target(fc, &tgt);
310 if (rc != NGX_DECLINED) {
311 return rc;
312 }
313
314 if (fc->pasv_fd >= 0) { (void) close(fc->pasv_fd); fc->pasv_fd = -1; }
315 fc->active_sa = tgt;
316 fc->active = 1;
317 return brix_ftp_ev_reply(fc, "200 %s command successful\r\n",
318 extended ? "EPRT" : "PORT");
319}
