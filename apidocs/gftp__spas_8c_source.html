10/* The stripe ports kept from one SPAS reply. Only the PORTS survive the parse:
11 * every stripe's address has to be the control channel's peer to be kept at
12 * all, so carrying the addresses forward would just be carrying that same
13 * string N times — and would give a later reader the impression there was a
14 * choice of address here. There is not (see gftp_spas.h). */
15typedef struct {
16 unsigned ports[GFTP_STREAMS_MAX];
17 unsigned n;
18} gftp_spas_list_t;

21/* Is this stripe's address the control channel's own peer? */
22static int
23gftp_spas_same_peer(const gftp_session_t *session, const unsigned char ip[4])
24{
25 char text[16];
26 int n;
27
28 n = snprintf(text, sizeof(text), "%u.%u.%u.%u", ip[0], ip[1], ip[2], ip[3]);
29 if (n <= 0 || n >= (int) sizeof(text)) {
30 return 0;
31 }
32 return strcmp(text, session->peer_ip) == 0;
33}

36/* Take one continuation line of the 229 reply. Returns 0 when the line was a
37 * usable stripe or an empty separator, -1 when the whole reply must be
38 * abandoned — an unreadable line, a stripe pointing somewhere other than the
39 * control peer, or more stripes than one transfer may hold. */
40static int
41gftp_spas_line(const gftp_session_t *session, gftp_spas_list_t *out,
42 const char *line, size_t len)
43{
44 unsigned char ip[4];
45 unsigned port;
46 size_t i;
47
48 for (i = 0; i < len; i++) {
49 if (line[i] != ' ' && line[i] != '\t' && line[i] != '\r') {
50 break;
51 }
52 }
53 if (i == len) {
54 return 0; /* blank separator line */
55 }
56 if (gftp_reply_parse_pasv(line, len, ip, &port) != 0
57 || !gftp_spas_same_peer(session, ip)
58 || out->n == GFTP_STREAMS_MAX)
59 {
60 return -1;
61 }
62 out->ports[out->n++] = port;
63 return 0;
64}

67/* Parse every stripe out of the session's continuation buffer. Returns 0 with
68 * out->n >= 1, or -1 when the reply cannot be used as a whole. */
69static int
70gftp_spas_parse(const gftp_session_t *session, gftp_spas_list_t *out)
71{
72 const char *line = session->cont;
73
74 /* A truncated continuation buffer would silently DROP stripes, and a
75 * transfer connected to a subset of them waits for EOD blocks that arrive
76 * on sockets nobody opened. Refuse the reply rather than the transfer. */
77 if (session->cont_truncated) {
78 return -1;
79 }
80 out->n = 0;
81 while (line != NULL && *line != '\0') {
82 const char *end = strchr(line, '\n');
83 size_t len = (end != NULL) ? (size_t) (end - line) : strlen(line);
84
85 if (gftp_spas_line(session, out, line, len) != 0) {
86 return -1;
87 }
88 line = (end != NULL) ? end + 1 : NULL;
89 }
90 return (out->n >= 1) ? 0 : -1;
91}

94/* Would a striped channel be used at all? Three independent reasons not to
95 * ask, none of them a failure: the operator did not budget for it, the wire is
96 * not carrying offsets (only MODE E blocks can be reassembled out of order
97 * across connections at all), or the origin never advertised SPAS. */
98static int
99gftp_spas_wanted(gftp_session_t *session)
100{
101 if (session->want_streams < 2 || session->mode != GFTP_DMODE_E) {
102 return 0;
103 }
104 return (gftp_feat(session) & GFTP_FEAT_SPAS) != 0;
105}

108/* Ask for the stripe list. Returns 0 with `out` filled, or -1 to fall back. */
109static int
110gftp_spas_request(gftp_session_t *session, gftp_spas_list_t *out)
111{
112 if (gftp_command(session, "SPAS") != 0) {
113 return -1;
114 }
115 if (session->code != 229 || gftp_spas_parse(session, out) != 0
116 || out->n > session->want_streams)
117 {
118 /* Advertised then unusable. Clear the bit so the rest of the session
119 * spends no more round trips discovering the same thing — the same
120 * ask-once rule the ERET fallback follows. */
121 session->feat &= ~GFTP_FEAT_SPAS;
122 return -1;
123 }
124 return 0;
125}
