16/* ---- Create a private raw-capture staging path ----
17 *
18 * WHAT: Makes an owner-only temporary directory and names its session capture.
19 * WHY: Root auth frames may exist briefly but must never enter published output.
20 * HOW: mkdtemp creates mode 0700; brix capture creates the only child on connect.
21 */
22static int
23capture_staging(char directory[64], char path[96])
24{
25 snprintf(directory, 64, "/tmp/brix-conform-capture.XXXXXX");
26 if (mkdtemp(directory) == NULL) return -1;
27 if ((size_t) snprintf(path, 96, "%s/session.raw.xrdcap", directory) >= 96) {
28 rmdir(directory);
29 return -1;
30 }
31 return 0;
32}

136/* ---- Enforce run resource ceilings before an operation ----
137 *
138 * WHAT: Reserves one request and bytes, or returns zero with a stable reason.
139 * WHY: Remote assessment must remain bounded even against hostile endpoints.
140 * HOW: Check wall time and overflow-safe request/byte ceilings before charging.
141 */
142int
143xrd_conform_reserve(xrd_conform_execution *execution, size_t bytes,
144 char *reason, size_t reason_size)
145{
146 time_t now;
147
148 if (execution == NULL) return 0;
149 now = time(NULL);
150 if (now == (time_t) -1 || now - execution->started
151 >= (time_t) execution->run->config.wall_seconds) {
152 snprintf(reason, reason_size, "ceiling:wall-seconds");
153 return 0;
154 }
155 if (execution->requests >= execution->run->config.request_limit) {
156 snprintf(reason, reason_size, "ceiling:request-limit");
157 return 0;
158 }
159 if ((unsigned long long) bytes > execution->run->config.byte_limit
160 || execution->bytes > execution->run->config.byte_limit
161 - (unsigned long long) bytes) {
162 snprintf(reason, reason_size, "ceiling:byte-limit");
163 return 0;
164 }
165 execution->requests++;
166 execution->bytes += (unsigned long long) bytes;
167 return 1;
168}

170/* ---- Seed the explicit result ledger ----
171 *
172 * WHAT: Allocates one result for every immutable clause and applies scope gates.
173 * WHY: Filtered and unauthorized clauses must appear as skips, never disappear.
174 * HOW: Copy catalogue pointers and ask the common scope resolver for each row.

176static int
177results_init(xrd_conform_execution *execution)
178{
179 const brix_conform_clause *clauses;
180 size_t count;
181 size_t index;
182
183 clauses = brix_conform_clauses(&count);
184 execution->results.results = calloc(count, sizeof(*execution->results.results));
185 if (execution->results.results == NULL) return -1;
186 execution->results.count = count;
187 for (index = 0; index < count; index++) {
188 brix_conform_result *result = &execution->results.results[index];
189
190 result->clause = &clauses[index];
191 result->changed = 1;
192 if (brix_conform_clause_selected(&clauses[index],
193 &execution->run->config, result->reason,
194 sizeof(result->reason))) {
195 result->verdict = BRIX_CONFORM_VERDICT_INCONCLUSIVE;
196 snprintf(result->reason, sizeof(result->reason),
197 "probe-not-executed");
198 } else {
199 result->verdict = BRIX_CONFORM_VERDICT_SKIPPED;
200 }
201 }
202 return 0;

433/* ---- Execute and report one remote conformance run ----
434 *
435 * WHAT: Returns the frozen Phase-118 exit code for a complete bounded run.
436 * WHY: Network, ledger, artifact and verdict failures have distinct semantics.
437 * HOW: Initialize, dispatch by URL plane, count, render, write bundle, decide.
438 */
439static int
440execute_once(xrd_conform_run *run)
441{
442 xrd_conform_execution execution;
443 brix_conform_report_meta meta;
444 char timestamp[32];
445 char report_error[256];
446 char capture_directory[64] = "";
447 char raw_capture[96] = "";
448 int capture_enabled = 0;
449 int negative_verified;
450 int runner_rc = 0;
451 int rc;
452
453 memset(&execution, 0, sizeof(execution));
454 execution.run = run;
455 infer_web_profile(run);
456 execution.started = time(NULL);
457 if (execution.started == (time_t) -1 || results_init(&execution) != 0
458 || timestamp_now(timestamp) != 0) {
459 BRIX_CLIENT_LOG("ERROR",
460 "xrd conform: cannot initialize run state\n");
461 free(execution.results.results);
462 return 3;
463 }
464 /* A conformance observation is one wire answer, not a resilient client
465 * transaction. Retrying a hostile kXR_ServerError can both hide the answer
466 * being assessed and exceed the run-level wall ceiling inside one reserved
467 * request, because library resilience has its own stall window. */
468 run->opts.no_retry = 1;
469 grade_scope_safety(&execution);
470 if (capture_staging(capture_directory, raw_capture) != 0) {
471 BRIX_CLIENT_LOG("ERROR",
472 "xrd conform: cannot create private capture staging\n");
473 free(execution.results.results);
474 return 3;
475 }
476 run->opts.capture = raw_capture;
477 xrd_conform_probe_cache_pair(&execution);
478 xrd_conform_probe_tpc_pair(&execution);
479 if (run_selected_targets(&execution, &capture_enabled) != 0) runner_rc = 3;
480 run_fixture_probes(&execution);
481 run->opts.capture = NULL;
482 finalize_cross_plane(&execution);
483 brix_conform_apply_reference_mode(&run->config, &execution.results);
484 if (run->config.baseline != NULL
485 && brix_conform_apply_baseline(run->config.baseline,
486 &execution.results, report_error, sizeof(report_error)) != 0) {
487 BRIX_CLIENT_LOG("ERROR",
488 "xrd conform: baseline rejected: %s\n", report_error);
489 runner_rc = 3;
490 }
491 negative_verified = negative_verification_passes(&execution,
492 runner_rc == 0);
493 if (!negative_verified) runner_rc = 3;
494 brix_conform_run_results_count(&execution.results);
495 report_meta_init(&meta, &execution, timestamp, negative_verified);
496 brix_conform_report_terminal(stdout, &meta, &execution.results);
497 if (publish_reports(&execution, &meta, capture_enabled, raw_capture,
498 report_error, sizeof(report_error)) != 0) {
499 BRIX_CLIENT_LOG("ERROR", "xrd conform: %s\n", report_error);
500 runner_rc = 3;
501 }
502 /* Recount at the exit boundary so renderer activity can never leave the
503 * process status out of step with the verdicts in the durable bundle. */
504 brix_conform_run_results_count(&execution.results);
505 rc = runner_rc == 0
506 ? brix_conform_run_exit_code(&run->config, &execution.results)
507 : runner_rc;
508 free(execution.results.results);
509 capture_staging_remove(capture_directory, raw_capture);
510 return rc;
511}

513/* ---- Keep one hostile network wait inside the run wall budget ----
514 *
515 * WHAT: Clamps connect/handshake and steady-state idle waits to half the
516 * configured conformance wall ceiling.
517 * WHY: The run ceiling is checked between probes, so a peer that accepts TCP
518 * and then black-holes TLS could otherwise spend the default 15-second
519 * bring-up timeout inside a ten-second assessment.
520 * HOW: Preserve stricter operator/library settings and lower only longer waits;
521 * reserve the other half for result classification and report emission.
522 */
523static void
524bound_network_waits(const xrd_conform_run *run)
525{
526 unsigned long long half_ms = (unsigned long long) run->config.wall_seconds
527 * 1000ULL / 2ULL;
528 int limit;
529
530 if (half_ms == 0) half_ms = 1;
531 limit = half_ms > (unsigned long long) INT_MAX ? INT_MAX : (int) half_ms;
532 if (brix_tmo_connect_ms() > limit) brix_tmo_set_connect_ms(limit);
533 if (brix_tmo_io_ms() > limit) brix_tmo_set_io_ms(limit);
534}

536/* ---- Sleep between watch duty-cycle samples ----
537 *
538 * WHAT: Waits the configured interval, resuming after signal interruptions.
539 * WHY: Watch must state and enforce its steady-state duty cycle.
540 * HOW: Repeat nanosleep with the returned remainder until fully elapsed.
541 */
542static void
543watch_sleep(unsigned seconds)
544{
545 struct timespec delay;
546 struct timespec remaining;
547
548 delay.tv_sec = (time_t) seconds;
549 delay.tv_nsec = 0;
550 while (nanosleep(&delay, &remaining) != 0 && errno == EINTR)
551 delay = remaining;
552}

554/* ---- Execute one-shot or continuous conformance shapes ----
555 *
556 * WHAT: Runs accept/full once and watch on a fixed sampling duty cycle.
557 * WHY: A one-shot command labelled watch is operationally dishonest.
558 * HOW: Reinitialize each sample, publish current artifacts, sleep, repeat;
559 * finite --watch-cycles exists for supervision and deterministic tests.
560 */
561static int
562execute_watch(xrd_conform_run *run)
563{
564 unsigned cycle = 0;
565 int aggregate = 0;
566
567 if (run->config.shape != BRIX_CONFORM_RUN_WATCH)
568 return execute_once(run);
569 for (;;) {
570 int rc = execute_once(run);
571
572 if (rc == 3 || rc == 4) return rc;
573 if (rc == 1) aggregate = 1;
574 else if (rc == 2 && aggregate == 0) aggregate = 2;
575 cycle++;
576 if (run->config.watch_cycles != 0
577 && cycle >= run->config.watch_cycles) return aggregate;
578 fflush(stdout);
579 watch_sleep(run->config.watch_interval_seconds);
580 }
581}
