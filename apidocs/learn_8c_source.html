1/* learn.c — phase-87 G11: workload-learned predictive prewarm.
2 *
3 * WHAT: a passive per-worker Markov model of CAS access sequences ("a GET of
4 * X on this connection is followed by Y"), and an advisory prewarm
5 * that fills the predicted successors through the cache-fill seam
6 * before they are requested.
7 * WHY: batch workloads open the same working set in the same order on every
8 * job (loader → catalog → the same .so chain); after one warm pass the
9 * proxy can hide the origin RTT of the NEXT job's cold misses entirely.
10 * This is the *policy* layer over the fill machinery's *mechanism* —
11 * it never invents a transfer path of its own.
12 * HOW: registration mirrors the G17 scrub (config-time per-process statics,
13 * per-worker context at init_process — but EVERY worker, the model is
14 * request-stream-local and workers see disjoint connections). The
15 * request path calls brix_cvmfs_learn_note() on each classified CAS
16 * GET: it records the connection's previous-key → this-key transition
17 * into a fixed-size successor table, then looks this key up and posts
18 * one bounded batch of confident (count >= 2), non-resident successors
19 * to the thread pool, where brix_sd_cache_fill_key() runs the ordinary
20 * verified fill (source → store + cinfo). One task in flight per
21 * export, a per-second rate cap, and a full-table overwrite policy
22 * keep it O(1) memory and immune to mispredict storms: an
23 * unrecognized access matches no node and prewarms nothing.
24 *
25 * PRIVACY / CARDINALITY (INVARIANT #8): the model holds CAS keys,
26 * connection numbers and small counts — never an Authorization value,
27 * token, DN or any per-user content. Predictions learned from one
28 * client's connection fire for every later client; the profile is a
29 * property of the WORKLOAD, not the user. No metrics are emitted —
30 * observability is two log lines (DEBUG on post, INFO on completion).
31 */

61void
62brix_cvmfs_learn_register(const char *root_canon, const ngx_str_t *pool_name)
63{
64 ngx_uint_t i;
65 cvmfs_learn_reg_t *reg = NULL;
66
67 for (i = 0; i < cvmfs_learn_regs_n; i++) {
68 if (ngx_strcmp(cvmfs_learn_regs[i].root, root_canon) == 0) {
69 reg = &cvmfs_learn_regs[i]; /* reload: update in place */
70 break;
71 }
72 }
73 if (reg == NULL) {
74 if (cvmfs_learn_regs_n >= CVMFS_LEARN_MAX_EXPORTS
75 || ngx_strlen(root_canon) >= sizeof(reg->root))
76 {
77 return;
78 }
79 reg = &cvmfs_learn_regs[cvmfs_learn_regs_n++];
80 }
81 ngx_cpystrn((u_char *) reg->root, (u_char *) root_canon,
82 sizeof(reg->root));
83 reg->pool[0] = '\0';
84 if (pool_name != NULL && pool_name->len > 0
85 && pool_name->len < sizeof(reg->pool))
86 {
87 ngx_memcpy(reg->pool, pool_name->data, pool_name->len);
88 reg->pool[pool_name->len] = '\0';
89 }
90}

133static uint64_t
134cvmfs_learn_hash(const char *s)
135{
136 uint64_t h = BRIX_FNV1A64_OFFSET_BASIS;
137
138 for (; *s != '\0'; s++) {
139 h = (h ^ (u_char) *s) * BRIX_FNV1A64_PRIME;
140 }
141 return h;
142}

144/* Thread-pool side: run the ordinary verified whole-file fill for each
145 * predicted key (source → cache store + cinfo). A DECLINED/ERROR fill is
146 * simply not counted — the prewarm is advisory and the on-demand path
147 * remains the correctness authority. */
148static void
149cvmfs_learn_thread(void *data, ngx_log_t *log)
150{
151 cvmfs_learn_ctx_t *lc = data;
152 ngx_uint_t i;
153
154 (void) log;
155 for (i = 0; i < lc->n_keys; i++) {
156 /* Background prewarm: service credential only (no per-user cred). */
157 if (brix_sd_cache_fill_key(lc->inst, lc->keys[i], NULL) == NGX_OK) {
158 lc->filled++;
159 }
160 }
161}

175/* Record one observed prev → cur transition. A hash-slot collision with a
176 * DIFFERENT predecessor overwrites the node (fixed memory beats a perfect
177 * model — the demoted pattern just re-learns); a full successor row evicts
178 * its weakest entry. */
179static void
180cvmfs_learn_record(cvmfs_learn_ctx_t *lc, const char *prev, const char *cur)
181{
182 uint64_t h = cvmfs_learn_hash(prev);
183 cvmfs_learn_node_t *node = &lc->nodes[h % CVMFS_LEARN_NODES];
184 cvmfs_learn_succ_t *weakest;
185 ngx_uint_t i;
186
187 if (!node->used || node->hash != h) {
188 ngx_memzero(node, sizeof(*node));
189 node->used = 1;
190 node->hash = h;
191 }
192 weakest = &node->succ[0];
193 for (i = 0; i < CVMFS_LEARN_SUCC; i++) {
194 cvmfs_learn_succ_t *s = &node->succ[i];
195
196 if (s->key[0] != '\0' && strcmp(s->key, cur) == 0) {
197 if (s->count < 0xffff) {
198 s->count++;
199 }
200 return;
201 }
202 if (s->key[0] == '\0') { /* free slot: install fresh */
203 ngx_cpystrn((u_char *) s->key, (u_char *) cur,
204 CVMFS_LEARN_KEY_MAX);
205 s->count = 1;
206 return;
207 }
208 if (s->count < weakest->count) {
209 weakest = s;
210 }
211 }
212 ngx_cpystrn((u_char *) weakest->key, (u_char *) cur,
213 CVMFS_LEARN_KEY_MAX);
214 weakest->count = 1;
215}

217/* Collect up to one batch of confident, non-resident successor keys into
218 * lc->keys, honoring the per-second rate cap. Returns the batch size. */
219static ngx_uint_t
220cvmfs_learn_collect(cvmfs_learn_ctx_t *lc, cvmfs_learn_node_t *node,
221 brix_cstore_t *cs)
222{
223 brix_cache_cinfo_t ci;
224 ngx_uint_t i, n = 0;
225
226 for (i = 0; i < CVMFS_LEARN_SUCC && n < CVMFS_LEARN_BATCH; i++) {
227 cvmfs_learn_succ_t *s = &node->succ[i];
228
229 if (s->key[0] == '\0' || s->count < CVMFS_LEARN_MIN_COUNT) {
230 continue;
231 }
232 if (brix_cstore_cinfo_load(cs, s->key, &ci) == NGX_OK
233 && (ci.flags & BRIX_CINFO_F_COMPLETE))
234 {
235 continue; /* already resident */
236 }
237 if (lc->rate_n >= CVMFS_LEARN_RATE) {
238 break;
239 }
240 lc->rate_n++;
241 ngx_cpystrn((u_char *) lc->keys[n], (u_char *) s->key,
242 CVMFS_LEARN_KEY_MAX);
243 n++;
244 }
245 return n;
246}

248/* Resolve the prewarm thread pool (configured name or "default"). */
249static ngx_thread_pool_t *
250cvmfs_learn_pool(cvmfs_learn_ctx_t *lc)
251{
252 ngx_str_t pname;
253
254 if (lc->reg->pool[0] != '\0') {
255 pname.data = (u_char *) lc->reg->pool;
256 pname.len = ngx_strlen(lc->reg->pool);
257 } else {
258 ngx_str_set(&pname, "default");
259 }
260 return ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname);
261}

263/* Predict successors of `key` and post one bounded prewarm batch: confident
264 * (count >= 2), not already cache-resident, inside the per-second rate cap,
265 * one task in flight. Every miss condition is a silent no-op — this path
266 * must never affect the serve that triggered it. */
267static void
268cvmfs_learn_predict(cvmfs_learn_ctx_t *lc, ngx_http_request_t *r,
269 brix_sd_instance_t *sd, const char *key)
270{
271 uint64_t h = cvmfs_learn_hash(key);
272 cvmfs_learn_node_t *node = &lc->nodes[h % CVMFS_LEARN_NODES];
273 brix_cstore_t *cs;
274 ngx_thread_pool_t *pool;
275 time_t now;
276 ngx_uint_t n;
277
278 if (lc->busy || !node->used || node->hash != h) {
279 return;
280 }
281 cs = brix_sd_cache_cstore(sd);
282 if (cs == NULL) {
283 return;
284 }
285 now = ngx_time();
286 if (lc->rate_sec != now) {
287 lc->rate_sec = now;
288 lc->rate_n = 0;
289 }
290 n = cvmfs_learn_collect(lc, node, cs);
291 if (n == 0) {
292 return;
293 }
294
295 pool = cvmfs_learn_pool(lc);
296 if (pool == NULL) {
297 return;
298 }
299
300 lc->inst = sd; /* registry-owned, worker-lifetime */
301 lc->n_keys = n;
302 lc->filled = 0;
303 lc->busy = 1;
304 if (ngx_thread_task_post(pool, lc->task) != NGX_OK) {
305 lc->busy = 0;
306 return;
307 }
308 ngx_log_debug2(NGX_LOG_DEBUG_HTTP, r->connection->log, 0,
309 "cvmfs-learn: %ui predicted successor(s) of \"%s\" posted "
310 "for prewarm", n, key);
311}

313/* Request-path hook (event loop, classified CAS tier entry): learn this
314 * connection's transition, then prewarm what usually follows `key`. */
315void
316brix_cvmfs_learn_note(ngx_http_request_t *r,
317 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
318 brix_sd_instance_t *sd, const char *key)
319{
320 cvmfs_learn_ctx_t *lc = NULL;
321 cvmfs_learn_conn_t *cslot;
322 uint64_t conn;
323 ngx_uint_t i;
324
325 if (lcf->cvmfs.learn != 1
326 || r->method != NGX_HTTP_GET
327 || ctx == NULL
328 || ctx->url.cls != CVMFS_URL_CAS
329 || sd == NULL
330 || key == NULL
331 || strlen(key) >= CVMFS_LEARN_KEY_MAX)
332 {
333 return;
334 }
335 for (i = 0; i < cvmfs_learn_regs_n; i++) {
336 if (ngx_strcmp(cvmfs_learn_regs[i].root,
337 lcf->common.root_canon) == 0)
338 {
339 lc = cvmfs_learn_ctxs[i];
340 break;
341 }
342 }
343 if (lc == NULL) {
344 return;
345 }
346
347 conn = (uint64_t) r->connection->number;
348 cslot = &lc->conns[conn % CVMFS_LEARN_CONN_SLOTS];
349 if (cslot->conn == conn && cslot->key[0] != '\0'
350 && strcmp(cslot->key, key) != 0)
351 {
352 cvmfs_learn_record(lc, cslot->key, key);
353 }
354 cslot->conn = conn;
355 ngx_cpystrn((u_char *) cslot->key, (u_char *) key,
356 CVMFS_LEARN_KEY_MAX);
357
358 cvmfs_learn_predict(lc, r, sd, key);
359}

361ngx_int_t
362brix_cvmfs_learn_init_worker(ngx_cycle_t *cycle)
363{
364 ngx_uint_t i;
365
366 /* EVERY worker builds its own model (unlike the worker-0 scrub): the
367 * table learns from the connections THIS worker accepts, and a fill the
368 * siblings race on is serialized by the fill spine itself. */
369 for (i = 0; i < cvmfs_learn_regs_n; i++) {
370 ngx_thread_task_t *task;
371 cvmfs_learn_ctx_t *lc;
372
373 task = ngx_thread_task_alloc(cycle->pool, sizeof(cvmfs_learn_ctx_t));
374 if (task == NULL) {
375 return NGX_ERROR;
376 }
377 lc = task->ctx;
378 lc->task = task;
379 lc->reg = &cvmfs_learn_regs[i];
380
381 brix_task_bind(task, cvmfs_learn_thread, cvmfs_learn_done);
382 task->event.log = cycle->log;
383
384 cvmfs_learn_ctxs[i] = lc;
385 }
386 return NGX_OK;
387}
