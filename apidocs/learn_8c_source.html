1/* learn.c — phase-87 G11: workload-learned predictive prewarm.
2 *
3 * WHAT: a passive per-worker Markov model of CAS access sequences ("a GET of
4 * X on this connection is followed by Y"), and an advisory prewarm
5 * that fills the predicted successors through the cache-fill seam
6 * before they are requested.
7 * WHY: batch workloads open the same working set in the same order on every
8 * job (loader → catalog → the same .so chain); after one warm pass the
9 * proxy can hide the origin RTT of the NEXT job's cold misses entirely.
10 * This is the *policy* layer over the fill machinery's *mechanism* —
11 * it never invents a transfer path of its own.
12 * HOW: registration mirrors the G17 scrub (config-time per-process statics,
13 * per-worker context at init_process — but EVERY worker, the model is
14 * request-stream-local and workers see disjoint connections). The
15 * request path calls brix_cvmfs_learn_note() on each classified CAS
16 * GET: it records the connection's previous-key → this-key transition
17 * into a fixed-size successor table, then looks this key up and posts
18 * one bounded batch of confident (count >= 2), non-resident successors
19 * to the thread pool, where brix_sd_cache_fill_key() runs the ordinary
20 * verified fill (source → store + cinfo). One task in flight per
21 * export, a per-second rate cap, and a full-table overwrite policy
22 * keep it O(1) memory and immune to mispredict storms: an
23 * unrecognized access matches no node and prewarms nothing.
24 *
25 * PRIVACY / CARDINALITY (INVARIANT #8): the model holds CAS keys,
26 * connection numbers and small counts — never an Authorization value,
27 * token, DN or any per-user content. Predictions learned from one
28 * client's connection fire for every later client; the profile is a
29 * property of the WORKLOAD, not the user. No metrics are emitted —
30 * observability is two log lines (DEBUG on post, INFO on completion).
31 */

62void
63brix_cvmfs_learn_register(const char *root_canon, const ngx_str_t *pool_name)
64{
65 ngx_uint_t i;
66 cvmfs_learn_reg_t *reg = NULL;
67
68 for (i = 0; i < cvmfs_learn_regs_n; i++) {
69 if (ngx_strcmp(cvmfs_learn_regs[i].root, root_canon) == 0) {
70 reg = &cvmfs_learn_regs[i]; /* reload: update in place */
71 break;
72 }
73 }
74 if (reg == NULL) {
75 if (cvmfs_learn_regs_n >= CVMFS_LEARN_MAX_EXPORTS
76 || ngx_strlen(root_canon) >= sizeof(reg->root))
77 {
78 return;
79 }
80 reg = &cvmfs_learn_regs[cvmfs_learn_regs_n++];
81 }
82 ngx_cpystrn((u_char *) reg->root, (u_char *) root_canon,
83 sizeof(reg->root));
84 reg->pool[0] = '\0';
85 if (pool_name != NULL && pool_name->len > 0
86 && pool_name->len < sizeof(reg->pool))
87 {
88 ngx_memcpy(reg->pool, pool_name->data, pool_name->len);
89 reg->pool[pool_name->len] = '\0';
90 }
91}

134static uint64_t
135cvmfs_learn_hash(const char *s)
136{
137 uint64_t h = BRIX_FNV1A64_OFFSET_BASIS;
138
139 for (; *s != '\0'; s++) {
140 h = (h ^ (u_char) *s) * BRIX_FNV1A64_PRIME;
141 }
142 return h;
143}

145/* Thread-pool side: run the ordinary verified whole-file fill for each
146 * predicted key (source → cache store + cinfo). A DECLINED/ERROR fill is
147 * simply not counted — the prewarm is advisory and the on-demand path
148 * remains the correctness authority. */
149static void
150cvmfs_learn_thread(void *data, ngx_log_t *log)
151{
152 cvmfs_learn_ctx_t *lc = data;
153 ngx_uint_t i;
154
155 (void) log;
156 for (i = 0; i < lc->n_keys; i++) {
157 /* Background prewarm: service credential only (no per-user cred). */
158 if (brix_sd_cache_fill_key(lc->inst, lc->keys[i], NULL) == NGX_OK) {
159 lc->filled++;
160 }
161 }
162}

176/* Record one observed prev → cur transition. A hash-slot collision with a
177 * DIFFERENT predecessor overwrites the node (fixed memory beats a perfect
178 * model — the demoted pattern just re-learns); a full successor row evicts
179 * its weakest entry. */
180static void
181cvmfs_learn_record(cvmfs_learn_ctx_t *lc, const char *prev, const char *cur)
182{
183 uint64_t h = cvmfs_learn_hash(prev);
184 cvmfs_learn_node_t *node = &lc->nodes[h % CVMFS_LEARN_NODES];
185 cvmfs_learn_succ_t *weakest;
186 ngx_uint_t i;
187
188 if (!node->used || node->hash != h) {
189 ngx_memzero(node, sizeof(*node));
190 node->used = 1;
191 node->hash = h;
192 }
193 weakest = &node->succ[0];
194 for (i = 0; i < CVMFS_LEARN_SUCC; i++) {
195 cvmfs_learn_succ_t *s = &node->succ[i];
196
197 if (s->key[0] != '\0' && strcmp(s->key, cur) == 0) {
198 if (s->count < 0xffff) {
199 s->count++;
200 }
201 return;
202 }
203 if (s->key[0] == '\0') { /* free slot: install fresh */
204 ngx_cpystrn((u_char *) s->key, (u_char *) cur,
205 CVMFS_LEARN_KEY_MAX);
206 s->count = 1;
207 return;
208 }
209 if (s->count < weakest->count) {
210 weakest = s;
211 }
212 }
213 ngx_cpystrn((u_char *) weakest->key, (u_char *) cur,
214 CVMFS_LEARN_KEY_MAX);
215 weakest->count = 1;
216}

218/* Collect up to one batch of confident, non-resident successor keys into
219 * lc->keys, honoring the per-second rate cap. Returns the batch size. */
220static ngx_uint_t
221cvmfs_learn_collect(cvmfs_learn_ctx_t *lc, cvmfs_learn_node_t *node,
222 brix_cstore_t *cs)
223{
224 brix_cache_cinfo_t ci;
225 ngx_uint_t i, n = 0;
226
227 for (i = 0; i < CVMFS_LEARN_SUCC && n < CVMFS_LEARN_BATCH; i++) {
228 cvmfs_learn_succ_t *s = &node->succ[i];
229
230 if (s->key[0] == '\0' || s->count < CVMFS_LEARN_MIN_COUNT) {
231 continue;
232 }
233 if (brix_cstore_cinfo_load(cs, s->key, &ci) == NGX_OK
234 && (ci.flags & BRIX_CINFO_F_COMPLETE))
235 {
236 continue; /* already resident */
237 }
238 if (lc->rate_n >= CVMFS_LEARN_RATE) {
239 break;
240 }
241 lc->rate_n++;
242 ngx_cpystrn((u_char *) lc->keys[n], (u_char *) s->key,
243 CVMFS_LEARN_KEY_MAX);
244 n++;
245 }
246 return n;
247}

249/* Resolve the prewarm thread pool (configured name or "default"). */
250static ngx_thread_pool_t *
251cvmfs_learn_pool(cvmfs_learn_ctx_t *lc)
252{
253 ngx_str_t pname;
254
255 if (lc->reg->pool[0] != '\0') {
256 pname.data = (u_char *) lc->reg->pool;
257 pname.len = ngx_strlen(lc->reg->pool);
258 } else {
259 ngx_str_set(&pname, "default");
260 }
261 return ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname);
262}

264/* Predict successors of `key` and post one bounded prewarm batch: confident
265 * (count >= 2), not already cache-resident, inside the per-second rate cap,
266 * one task in flight. Every miss condition is a silent no-op — this path
267 * must never affect the serve that triggered it. */
268static void
269cvmfs_learn_predict(cvmfs_learn_ctx_t *lc, ngx_http_request_t *r,
270 brix_sd_instance_t *sd, const char *key)
271{
272 uint64_t h = cvmfs_learn_hash(key);
273 cvmfs_learn_node_t *node = &lc->nodes[h % CVMFS_LEARN_NODES];
274 brix_cstore_t *cs;
275 ngx_thread_pool_t *pool;
276 time_t now;
277 ngx_uint_t n;
278
279 if (lc->busy || !node->used || node->hash != h) {
280 return;
281 }
282 cs = brix_sd_cache_cstore(sd);
283 if (cs == NULL) {
284 return;
285 }
286 now = ngx_time();
287 if (lc->rate_sec != now) {
288 lc->rate_sec = now;
289 lc->rate_n = 0;
290 }
291 n = cvmfs_learn_collect(lc, node, cs);
292 if (n == 0) {
293 return;
294 }
295
296 pool = cvmfs_learn_pool(lc);
297 if (pool == NULL) {
298 return;
299 }
300
301 lc->inst = sd; /* registry-owned, worker-lifetime */
302 lc->n_keys = n;
303 lc->filled = 0;
304 lc->busy = 1;
305 if (ngx_thread_task_post(pool, lc->task) != NGX_OK) {
306 lc->busy = 0;
307 return;
308 }
309 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_HTTP, r->connection->log, 0,
310 "cvmfs-learn: %ui predicted successor(s) of \"%s\" posted "
311 "for prewarm", n, key);
312}

314/* Request-path hook (event loop, classified CAS tier entry): learn this
315 * connection's transition, then prewarm what usually follows `key`. */
316void
317brix_cvmfs_learn_note(ngx_http_request_t *r,
318 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
319 brix_sd_instance_t *sd, const char *key)
320{
321 cvmfs_learn_ctx_t *lc = NULL;
322 cvmfs_learn_conn_t *cslot;
323 uint64_t conn;
324 ngx_uint_t i;
325
326 if (lcf->cvmfs.learn != 1
327 || r->method != NGX_HTTP_GET
328 || ctx == NULL
329 || ctx->url.cls != CVMFS_URL_CAS
330 || sd == NULL
331 || key == NULL
332 || strlen(key) >= CVMFS_LEARN_KEY_MAX)
333 {
334 return;
335 }
336 for (i = 0; i < cvmfs_learn_regs_n; i++) {
337 if (ngx_strcmp(cvmfs_learn_regs[i].root,
338 lcf->common.root_canon) == 0)
339 {
340 lc = cvmfs_learn_ctxs[i];
341 break;
342 }
343 }
344 if (lc == NULL) {
345 return;
346 }
347
348 conn = (uint64_t) r->connection->number;
349 cslot = &lc->conns[conn % CVMFS_LEARN_CONN_SLOTS];
350 if (cslot->conn == conn && cslot->key[0] != '\0'
351 && strcmp(cslot->key, key) != 0)
352 {
353 cvmfs_learn_record(lc, cslot->key, key);
354 }
355 cslot->conn = conn;
356 ngx_cpystrn((u_char *) cslot->key, (u_char *) key,
357 CVMFS_LEARN_KEY_MAX);
358
359 cvmfs_learn_predict(lc, r, sd, key);
360}

362ngx_int_t
363brix_cvmfs_learn_init_worker(ngx_cycle_t *cycle)
364{
365 ngx_uint_t i;
366
367 /* EVERY worker builds its own model (unlike the worker-0 scrub): the
368 * table learns from the connections THIS worker accepts, and a fill the
369 * siblings race on is serialized by the fill spine itself. */
370 for (i = 0; i < cvmfs_learn_regs_n; i++) {
371 ngx_thread_task_t *task;
372 cvmfs_learn_ctx_t *lc;
373
374 task = ngx_thread_task_alloc(cycle->pool, sizeof(cvmfs_learn_ctx_t));
375 if (task == NULL) {
376 return NGX_ERROR;
377 }
378 lc = task->ctx;
379 lc->task = task;
380 lc->reg = &cvmfs_learn_regs[i];
381
382 brix_task_bind(task, cvmfs_learn_thread, cvmfs_learn_done);
383 task->event.log = cycle->log;
384
385 cvmfs_learn_ctxs[i] = lc;
386 }
387 return NGX_OK;
388}
