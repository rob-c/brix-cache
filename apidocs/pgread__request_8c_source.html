19/*
20 * brix_pgread_parse_validate - decode the request and run all early checks.
21 *
22 * WHAT: Unpacks the kXR_pgread request into `run`, validates the handle,
23 * rejects negative offset/length, answers a zero-length read with an
24 * empty kXR_status frame, caps rlen, and resolves the fd. Returns 1 to
25 * continue; 0 when the request was fully handled (*rc holds the
26 * handler's return value — error sent, empty response queued, or
27 * NGX_ERROR).
28 *
29 * WHY: All reject/short-circuit paths in one early-return helper keeps the
30 * handler a flat orchestrator (coding-standards §8).
31 *
32 * HOW: Mirrors brix_validate_read_handle's continue-flag + *rc convention
33 * so the caller propagates the exact wire response codes unchanged.
34 */
35ngx_int_t
36brix_pgread_parse_validate(brix_ctx_t *ctx, ngx_connection_t *c,
37 brix_pgread_run_t *run, ngx_int_t *rc)
38{
39 xrdw_pgread_req_t req;
40 ServerStatusResponse_pgRead *hdr_buf;
41
42 xrdw_pgread_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
43 run->idx = (int) (unsigned char) req.fhandle[0];
44 run->offset = req.offset;
45 run->rlen = (size_t) (uint32_t) req.rlen;
46
47 /* §1.2 optional request args ride the payload: pathid at byte 0 when
48 * dlen >= 1, reqflags at byte 1 when dlen >= 2 (extra bytes tolerated) —
49 * stock 5.6.9 parses exactly this shape at any dlen, verified live. A
50 * nonzero pathid must name one of THIS session's live kXR_bind paths;
51 * stock refuses anything else with kXR_ArgInvalid "invalid path ID".
52 * kXR_pgRetry (and unknown flag bits) change nothing server-side: the
53 * pages are re-read and re-checksummed fresh, which is stock behavior.
54 * The response itself still travels the control stream — response
55 * offloading over the bound path is the audit's §1.1 gap, all opcodes
56 * alike. */
57 if (ctx->recv.cur_dlen >= 1 && ctx->recv.payload != NULL) {
58 run->pathid = (unsigned) ((u_char *) ctx->recv.payload)[0];
59 run->reqflags = (ctx->recv.cur_dlen >= 2)
60 ? (unsigned) ((u_char *) ctx->recv.payload)[1] : 0;
61 if (run->pathid != 0
62 && !brix_session_pathid_bound(ctx->is_bound ? ctx->bound_sessid
63 : ctx->login.sessid,
64 run->pathid))
65 {
66 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
67 *rc = brix_send_error(ctx, c, kXR_ArgInvalid,
68 "invalid path ID");
69 return 0;
70 }
71 }
72
73 if (!brix_validate_read_handle(ctx, c, run->idx, "PGREAD",
74 BRIX_OP_PGREAD, rc)) {
75 return 0;
76 }
77
78 if (run->offset < 0) {
79 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
80 *rc = brix_send_error(ctx, c, kXR_IOError,
81 "negative read offset");
82 return 0;
83 }
84
85 /* The wire rlen is a signed 32-bit field; a negative request length is
86 * invalid. Read unsigned it would turn -1 into ~4 GiB (then capped),
87 * silently succeeding where the reference rejects with kXR_ArgInvalid. */
88 if (req.rlen < 0) {
89 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
90 *rc = brix_send_error(ctx, c, kXR_ArgInvalid,
91 "negative read length");
92 return 0;
93 }
94
95 if (run->rlen == 0) {
96 hdr_buf = ngx_palloc(c->pool, sizeof(*hdr_buf));
97 if (hdr_buf == NULL) {
98 *rc = NGX_ERROR;
99 return 0;
100 }
101 brix_build_pgread_status(ctx, run->offset, 0, hdr_buf);
102 BRIX_OP_OK(ctx, BRIX_OP_PGREAD);
103 *rc = brix_queue_response(ctx, c, (u_char *) hdr_buf,
104 sizeof(*hdr_buf));
105 return 0;
106 }
107
108 if (run->rlen > BRIX_READ_REQUEST_MAX) {
109 run->rlen = BRIX_READ_REQUEST_MAX;
110 }
111
112 run->fd = ctx->files[run->idx].fd;
113 return 1;
114}

116/*
117 * brix_pgread_scratch_size - worst-case gapped wire-buffer size.
118 *
119 * WHAT: The byte count the interleaved [CRC32c(4)][data] wire output can
120 * reach for this request — data bytes plus one 4-byte CRC per page
121 * (data is read straight into the gaps; no separate flat copy region).
122 *
123 * WHY: The buffer-size math is pure and self-contained; isolating it keeps
124 * the page-count subtlety (alignment split) documented in one place.
125 *
126 * HOW: File-offset alignment can split an otherwise single-page read across
127 * two pages (short first fragment + remainder), so the page count is
128 * derived from the in-page offset, not just rlen — otherwise the
129 * scratch/out region would be one CRC short.
130 */
131size_t
132brix_pgread_scratch_size(const brix_pgread_run_t *run)
133{
134 size_t n_pages_max;
135
136 n_pages_max = ((size_t) (run->offset & (kXR_pgPageSZ - 1)) + run->rlen
137 + kXR_pgPageSZ - 1) / kXR_pgPageSZ;
138 if (n_pages_max == 0) {
139 n_pages_max = 1;
140 }
141 return run->rlen + n_pages_max * BRIX_PG_CKSZ;
142}

144/*
145 * brix_pgread_try_warm - inline warm-cache fast path.
146 *
147 * WHAT: Attempts the whole read + in-place CRC on the event loop via
148 * preadv2(RWF_NOWAIT). On a full hit fills run->{out_buf, out_size,
149 * flat_buf}, charges the backend byte metric, and returns 1; on any
150 * miss returns 0 with the run output untouched.
151 *
152 * WHY: When the whole range is already page-cache resident, reading + CRCing
153 * inline skips the thread-pool handoff entirely — the handoff latency,
154 * not the copy, is the single-stream (n=1) cost. A miss (not resident /
155 * EOF / error) falls through to the blocking offload, which re-reads the
156 * full range. Only attempted with a pool configured (else the blocking
157 * path runs inline anyway) and for a regular file (RWF_NOWAIT is
158 * meaningful against the page cache). Mirrors the kXR_read Phase-32 probe.
159 *
160 * HOW: Delegates to brix_pgread_read_encode_inplace with nowait=1 against
161 * the handle's effective storage object; a hit means errno stayed 0 and
162 * every requested byte was delivered.
163 */
164ngx_flag_t
165brix_pgread_try_warm(brix_ctx_t *ctx, ngx_stream_brix_srv_conf_t *rconf,
166 brix_pgread_run_t *run)
167{
168 brix_pgread_io_t warm_io = { .nowait = 1, .nread = 0, .io_errno = 0 };
169 size_t warm_osz;
170 brix_sd_obj_t warm_scratch;
171 brix_sd_obj_t *warm_obj;
172
173 if (rconf->common.thread_pool == NULL
174 || !ctx->files[run->idx].is_regular)
175 {
176 return 0;
177 }
178
179 warm_obj = brix_vfs_effective_obj(&ctx->files[run->idx].sd_obj, run->fd,
180 &warm_scratch);
181
182 /* The RWF_NOWAIT probe needs the driver's native preadv2; drivers without
183 * one (remote/object backends) have no page cache to probe — treat as a
184 * miss so the read offloads to the blocking path. */
185 if (warm_obj->driver->preadv2 == NULL) {
186 return 0;
187 }
188
189 warm_osz = brix_pgread_read_encode_inplace(warm_obj, (off_t) run->offset,
190 run->rlen, run->scratch,
191 &warm_io);
192 if (warm_io.io_errno != 0 || warm_io.nread != (ssize_t) run->rlen) {
193 return 0;
194 }
195
196 run->out_size = warm_osz;
197 run->flat_buf = run->scratch;
198 run->out_buf = run->scratch; /* rlen already == bytes encoded */
199
200 /* The warm fast path bypasses brix_vfs_io_execute (where the
201 * cold pgread paths attribute), so charge the per-backend read
202 * total here for the file bytes just read. */
203 brix_metric_backend_bytes(
204 ctx->files[run->idx].sd_obj.driver != NULL
205 ? ctx->files[run->idx].sd_obj.driver->name : "posix",
206 BRIX_METRIC_OP_READ, (size_t) warm_io.nread);
207
208 return 1;
209}
