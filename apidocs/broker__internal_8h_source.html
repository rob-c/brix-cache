113/*
114 * imp_op_ctx_t — one privileged fs-op invocation, bundled. WHAT: the full
115 * argument set of a broker fs op plus the once-computed root-relative path,
116 * so imp_do_op() and every per-op helper take a single explicit parameter.
117 * WHY: explicit data flow with no globals while keeping each imp_op_<name>()
118 * under the complexity gate. HOW: the caller (imp_serve_one) fills every
119 * field except `rel`, which imp_do_op() derives from req->path.
120 */
121typedef struct {
122 int rootfd; /* export-root O_PATH fd (confinement anchor) */
123 const imp_req_t *req; /* decoded request frame */
124 imp_rep_t *rep; /* reply frame being built */
125 int *out_fd; /* OPEN: fd handed back via SCM_RIGHTS */
126 char *data_out; /* trailing reply payload buffer */
127 size_t data_max; /* capacity of data_out */
128 const char *data_in; /* SETXATTR inbound value payload */
129 size_t data_in_len; /* byte count of data_in */
130 const char *rel; /* imp_rel(req->path): root-relative path */
131} imp_op_ctx_t;
