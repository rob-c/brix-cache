56uint64_t
57brix_plat_available_memory(void)
58{
59 vm_size_t page_size;
60 vm_statistics64_data_t vm_stat;
61 mach_msg_type_number_t count = HOST_VM_INFO64_COUNT;
62
63 if (host_page_size(mach_host_self(), &page_size) == KERN_SUCCESS
64 && host_statistics64(mach_host_self(), HOST_VM_INFO64,
65 (host_info64_t) &vm_stat, &count) == KERN_SUCCESS)
66 {
67 uint64_t free_pages = vm_stat.free_count + vm_stat.inactive_count
68 + vm_stat.purgeable_count;
69 return free_pages * page_size;
70 }
71 return 0;
72}

74static void
75sysload_net(brix_plat_sysload_t *sample)
76{
77 int mib[] = { CTL_NET, PF_ROUTE, 0, 0, NET_RT_IFLIST2, 0 };
78 size_t len = 0;
79 char *table;
80 char *at;
81 char *end;
82
83 if (sysctl(mib, sizeof(mib) / sizeof(mib[0]), NULL, &len, NULL, 0) != 0
84 || len == 0) return;
85 table = malloc(len);
86 if (table == NULL) return;
87 if (sysctl(mib, sizeof(mib) / sizeof(mib[0]), table, &len, NULL, 0) != 0) {
88 free(table);
89 return;
90 }
91 end = table + len;
92 for (at = table; at + sizeof(struct if_msghdr) <= end; ) {
93 struct if_msghdr *base = (struct if_msghdr *) at;
94
95 if (base->ifm_msglen == 0 || at + base->ifm_msglen > end) break;
96 if (base->ifm_type == RTM_IFINFO2
97 && base->ifm_msglen >= sizeof(struct if_msghdr2)) {
98 struct if_msghdr2 *info = (struct if_msghdr2 *) at;
99
100 if ((info->ifm_flags & IFF_LOOPBACK) == 0
101 && info->ifm_data.ifi_type != IFT_LOOP) {
102 sample->net_bytes += info->ifm_data.ifi_ibytes
103 + info->ifm_data.ifi_obytes;
104 sample->have |= BRIX_SYSLOAD_HAVE_NET;
105 }
106 }
107 at += base->ifm_msglen;
108 }
109 free(table);
110}

112int
113brix_plat_sysload_sample(brix_plat_sysload_t *sample)
114{
115 vm_statistics64_data_t vm_stat;
116 mach_msg_type_number_t count = HOST_VM_INFO64_COUNT;
117
118 if (sample == NULL) {
119 errno = EINVAL;
120 return -1;
121 }
122 memset(sample, 0, sizeof(*sample));
123 if (getloadavg(&sample->load1, 1) == 1) {
124 sample->have |= BRIX_SYSLOAD_HAVE_LOAD;
125 }
126 sample->total_memory = brix_plat_total_memory();
127 sample->available_memory = brix_plat_available_memory();
128 if (sample->total_memory != 0) sample->have |= BRIX_SYSLOAD_HAVE_MEMORY;
129 if (host_statistics64(mach_host_self(), HOST_VM_INFO64,
130 (host_info64_t) &vm_stat, &count) == KERN_SUCCESS) {
131 sample->paging_events = (uint64_t) vm_stat.pageins
132 + (uint64_t) vm_stat.pageouts;
133 sample->have |= BRIX_SYSLOAD_HAVE_PAGING;
134 }
135 sysload_net(sample);
136 return sample->have != 0 ? 0 : -1;
137}
