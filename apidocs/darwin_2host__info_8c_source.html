49 mach_msg_type_number_t count = HOST_VM_INFO64_COUNT;
50
51 if (host_page_size(mach_host_self(), &page_size) == KERN_SUCCESS

53 (host_info64_t) &vm_stat, &count) == KERN_SUCCESS)
54 {
55 uint64_t free_pages = vm_stat.free_count + vm_stat.inactive_count
56 + vm_stat.purgeable_count;
