19/* ---- Adapter state for the pipelined download sink ----
20 *
21 * WHAT: What download_fast_sink needs to turn a brix_rfile chunk callback into
22 * the caller's pump_sink_fn call plus the per-chunk progress tick, the
23 * cancel check and the --xrate-threshold floor.
24 * WHY: The pipelined reader speaks the brix_rfile sink contract while every
25 * download destination (stdout fd, VFS temp file) speaks pump_sink_fn;
26 * one small adapter bridges them without either side learning about the
27 * other, and keeps the bookkeeping transfer_pump does per chunk intact.
28 * HOW: Filled by download_stream_fast, read (and `moved` advanced) by
29 * download_fast_sink.
30 */
31typedef struct {
32 pump_sink_fn sink;
33 void *sinkctx;
34 const brix_copy_opts *o;
35 int64_t total; /* progress denominator (source size) */
36 int64_t moved; /* bytes drained so far */
37 uint64_t t0_ns; /* pacing epoch */
38} dl_fast_sink_t;
39

41/* ---- Drain one pipelined chunk into the download's destination ----
42 *
43 * WHAT: Writes `len` bytes at absolute offset `off` through the adapter's
44 * pump_sink_fn, ticks progress and applies the rate floor. Returns 0 to
45 * continue or -1 (st set) to abort the transfer.
46 *
47 * WHY: transfer_pump does cancel + progress + pacing around every chunk it
48 * drains; the pipelined path must keep doing all three or --xrate-threshold
49 * and SIGINT would silently stop working on fast copies.
50 *
51 * HOW: 1. An operator cancel aborts with EINTR, exactly as the serial pump does.
52 * 2. Hand the bytes to the destination sink.
53 * 3. Advance `moved`, emit progress, then apply brix_pump_pace's floor.
54 */
55static int
56download_fast_sink(const uint8_t *data, size_t len, int64_t off, void *arg,
57 brix_status *st)
58{
59 dl_fast_sink_t *a = (dl_fast_sink_t *) arg;
60
61 if (brix_copy_quit_requested()) {
62 brix_status_set(st, XRDC_ESOCK, EINTR, "transfer cancelled (signal)");
63 return -1;
64 }
65 if (a->sink(a->sinkctx, data, off, len, st) != 0) {
66 return -1;
67 }
68 a->moved = off + (int64_t) len;
69 pump_emit_progress(a->o, a->moved, a->total);
70 if (brix_pump_pace(a->o, a->t0_ns, a->moved, st) != 0) {
71 return -1;
72 }
73 return 0;
74}

77/* ---- May this download use the pipelined reader? ----
78 *
79 * WHAT: Returns 1 when the transfer is a plain sized SINGLE-STREAM read that
80 * the pipelined path can carry, 0 when it must use the serial pump.
81 *
82 * WHY: Paged I/O (--pgrw) and inline read compression (--compress) have their
83 * own reply framing, and --xrate wants the small, evenly-spaced reads
84 * brix_pump_pace_cap produces — a 4 MiB pipelined chunk would make a capped
85 * transfer lurch.
86 *
87 * -S/--streams is the fourth, and the one that is easy to miss: this path
88 * runs BEFORE download_stream_body binds its secondaries and pipelines on
89 * the primary connection alone, so letting it take a fan-out download does
90 * not slow the transfer down — it silently deletes the feature. `-S 4` is
91 * the DEFAULT (xrdcp.c `opts.streams = 4`), so without this clause no
92 * ordinary download has bound a secondary since the pipelined reader
93 * landed, against a documented contract ("the pumps fan full request
94 * frames across the bound connections", native-client-tools.md) and with
95 * nothing to see: the bytes are right and `-S` still parses.
96 *
97 * Each of the two is worth having on its own — several reads in flight on
98 * one socket, or one read at a time across four — and the caller chooses
99 * by asking for streams or not asking. (`--parallel` is the third shape:
100 * true concurrent striping, which owns its own path.)
101 *
102 * HOW: Require a known size, no pgrw, no compress opaque, no rate cap, and no
103 * bound secondaries requested.
104 */
105int
106download_fast_eligible(const download_body_ctx *j)
107{
108 const brix_copy_opts *o = j->o;
109
110 return j->si->size >= 0 && !o->pgrw && o->xrate_bps <= 0 && o->streams <= 1
111 && (o->compress == NULL || o->compress[0] == '\0');
112}

115/* ---- Stream the body with several reads in flight ----
116 *
117 * WHAT: The pipelined half of download_stream_body: opens the source as a
118 * resilient handle, streams si->size bytes through the caller's sink with
119 * brix_rfile_stream, and closes it. 0 / -1 (st set).
120 *
121 * WHY: The serial pump idles a full round trip per chunk, which on a fast link
122 * is most of the transfer time. brix_rfile_stream keeps several kXR_reads
123 * outstanding and falls back to the serial pump for anything it cannot
124 * finish, so this is a speed change only — reopen-on-sever, resume at
125 * offset and the adaptive request shrink are all still in force.
126 *
127 * HOW: 1. Open resiliently (brix_rfile_open_read owns the open retry loop).
128 * 2. Wire the cancel predicate so a SIGINT stops the stream.
129 * 3. Stream into the adapter; a short stream is the same short-read error
130 * transfer_pump would have reported.
131 * 4. Close the handle, preserving a failure status over the close's.
132 */
133int
134download_stream_fast(const download_body_ctx *j, pump_sink_fn sink,
135 void *sinkctx, brix_status *st)
136{
137 brix_rfile rf;
138 dl_fast_sink_t adapter;
139 int64_t moved = 0;
140 int rc;
141
142 if (brix_rfile_open_read(j->c, j->su->path, NULL, 0,
143 copy_stall_ms(j->o, 60000), &rf, st) != 0) {
144 return -1;
145 }
146 rf.cancel = brix_copy_quit_requested;
147
148 adapter.sink = sink;
149 adapter.sinkctx = sinkctx;
150 adapter.o = j->o;
151 adapter.total = j->si->size;
152 adapter.moved = 0;
153 adapter.t0_ns = brix_mono_ns();
154
155 rc = brix_rfile_stream(&rf, 0, j->si->size, 0, download_fast_sink,
156 &adapter, &moved, st);
157 if (rc == 0 && moved < j->si->size) {
158 brix_status_set(st, XRDC_EPROTO, 0,
159 "short read: got %lld of %lld bytes",
160 (long long) moved, (long long) j->si->size);
161 rc = -1;
162 }
163 {
164 brix_status throwaway;
165 brix_status_clear(&throwaway);
166 brix_rfile_close(&rf, rc == 0 ? st : &throwaway);
167 }
168 return rc;
169}
