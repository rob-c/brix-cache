14/* ---- xattr surface: x-amz-meta-* as the `user.` namespace --------------
15 *
16 * getxattr("user.<name>") reads x-amz-meta-<name> via a signed HEAD;
17 * listxattr enumerates every x-amz-meta-* header (needs a transport with the
18 * optional resp_headers_raw slot — without it sd_s3_list_meta reports
19 * ENOTSUP). Both open the object read-only just for the HEAD.
20 *
21 * The HEAD signs with the per-user ak/sk/region/session when the caller passes
22 * them (NULL = the instance's static service credential), exactly as the stat
23 * and mutation bodies do. Reading metadata is a read of the user's data, and
24 * while these two slots had no *_cred sibling brix_sd_{get,list}xattr_maybe_cred
25 * fell through to the plain slot for every caller it could not refuse: a user
26 * presenting perfectly good S3 keys had the read signed as the EXPORT, so it
27 * returned attributes that identity's own keys would have been denied. (The
28 * forwarder's fallback_deny backstop already refused a credential it could not
29 * route — that arm was never the hole; the silent service-key signing was.) */
30static sd_s3_file *
31sd_remote_meta_open(brix_sd_instance_t *inst, const char *path,
32 const char *ak, const char *sk, const char *region, const char *session)
33{
34 const brix_sd_remote_cfg_t *cfg = inst->state;
35 sd_s3_open_params p;
36 char objpath[768];
37 char errbuf[256];
38
39 sd_remote_s3_key(cfg, path, objpath, sizeof(objpath));
40 sd_remote_s3_params(cfg, objpath, &p);
41 sd_remote_params_cred(&p, ak, sk, region, session);
42 return sd_s3_open_read(&p, errbuf, sizeof(errbuf));
43}

45static ssize_t
46sd_remote_getxattr_impl(brix_sd_instance_t *inst, const char *path,
47 const char *name, void *buf, size_t cap,
48 const char *ak, const char *sk, const char *region, const char *session)
49{
50 char val[2048];
51 char errbuf[256];
52 sd_s3_meta_buf dst = { val, sizeof(val) };
53 sd_s3_file *s3;
54 ssize_t n;
55
56 if (strncmp(name, "user.", 5) != 0 || name[5] == '\0') {
57 errno = ENODATA; /* only the user. namespace maps to x-amz-meta- */
58 return -1;
59 }
60 s3 = sd_remote_meta_open(inst, path, ak, sk, region, session);
61 if (s3 == NULL) {
62 errno = ENOMEM;
63 return -1;
64 }
65 errno = 0;
66 n = sd_s3_get_meta(s3, name + 5, &dst, errbuf, sizeof(errbuf));
67 sd_s3_close(s3);
68 if (n < 0) {
69 if (errno == 0) { errno = EIO; }
70 return -1;
71 }
72 if (n == 0) {
73 errno = ENODATA; /* HEAD ok, attribute absent */
74 return -1;
75 }
76 if (buf == NULL || cap == 0) {
77 return n; /* getxattr(2) size probe */
78 }
79 if ((size_t) n > cap) {
80 errno = ERANGE;
81 return -1;
82 }
83 memcpy(buf, val, (size_t) n);
84 return n;
85}

95/* Cred-scoped getxattr: the HEAD runs as the requesting user. Gate semantics
96 * identical to sd_remote_stat_cred — a cred this S3-only backend cannot use is
97 * refused under fallback_deny rather than signed with the service credential. */
98ssize_t
99sd_remote_getxattr_cred(brix_sd_instance_t *inst, const char *path,
100 const char *name, void *buf, size_t cap, const brix_sd_cred_t *cred)
101{
102 int gate = sd_remote_cred_gate(cred);
103
104 if (gate > 0) {
105 return sd_remote_getxattr_impl(inst, path, name, buf, cap,
106 cred->s3_ak, cred->s3_sk, cred->s3_region, cred->s3_session);
107 }
108 if (gate < 0) {
109 errno = EACCES;
110 return -1;
111 }
112 return sd_remote_getxattr_impl(inst, path, name, buf, cap,
113 NULL, NULL, NULL, NULL);
