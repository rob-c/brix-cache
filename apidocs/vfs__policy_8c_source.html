1/*
2 * vfs_policy.c — the one VFS mutation-policy kernel (phase-105).
3 *
4 * WHAT: Implements the decision every VFS mutation entry point asks before it
5 * touches anything: may THIS request modify exported storage? Owns the
6 * config-derivation helper (brix_vfs_policy_from_write_enable), the
7 * bounded operation-name table, the three kernel forms, the
8 * policy-bearing operation-context helpers, and the raw-open-flag
9 * classifier.
10 *
11 * WHY: The decision used to be spelled out at each call site as
12 * `!ctx->allow_write` -> EACCES, which meant a new mutator inherited the
13 * gate only if its author remembered to write it, and which reported the
14 * same errno as an authorization failure. One kernel makes the gate
15 * inheritable, makes the result distinguishable (EROFS, never EACCES),
16 * and — because it is pure — makes it cheap enough to run BEFORE leaf
17 * resolution, credential selection, cache invalidation, and capability
18 * probing, which is the ordering the security property depends on.
19 *
20 * HOW: Every form funnels into brix_vfs_require_mutation_policy(), which
21 * range-checks the operation and returns NGX_ERROR/EROFS for a read-only
22 * policy. The context forms add the object validation (and, for the path
23 * form, confinement) that must precede it, then attribute one denial
24 * observation — only here, so an edge that already refused never
25 * double-counts. No allocation, no I/O, no backend or credential lookup.
26 */
27#include "vfs_internal.h"
28#include "vfs_policy.h"
29#include "vfs_backend_registry.h"

38/* ---- Map a merged write-enable flag onto the typed policy ----
39 *
40 * WHAT: Returns BRIX_VFS_MUTATION_ALLOWED for exactly 1 and
41 * BRIX_VFS_MUTATION_READ_ONLY for every other value.
42 *
43 * WHY: ngx_flag_t is a signed integer that is NGX_CONF_UNSET (-1) before a
44 * merge completes, so "non-zero means writable" would read an unmerged
45 * or corrupt flag as permission. Only the exact enabled value opens the
46 * endpoint.
47 *
48 * HOW: 1. Compare against 1; 2. return the matching enum value.
49 */
50brix_vfs_mutation_policy_t
51brix_vfs_policy_from_write_enable(ngx_flag_t allow_write)
52{
53 return (allow_write == 1)
54 ? BRIX_VFS_MUTATION_ALLOWED : BRIX_VFS_MUTATION_READ_ONLY;
55}

57/* ---- The stable label for a bounded mutation operation ----
58 *
59 * WHAT: Returns the lowercase name of `op`, or "unknown" out of range.
60 *
61 * WHY: Metrics and structured logs must agree on one closed vocabulary; a
62 * second table would let the two drift and would risk an unbounded label
63 * (INVARIANT #8).
64 *
65 * HOW: 1. Range-check; 2. index the static table.
66 */
67const char *
68brix_vfs_mutation_op_name(brix_vfs_mutation_op_t op)
69{
70 static const char *names[BRIX_VFS_MUTATE_OP_COUNT] = {
71 "open", "write", "truncate", "sync", "mkdir", "remove",
72 "rename", "copy", "setattr", "xattr", "publish", "stage",
73 "evict", "lock", "dedup", "credential",
74 };
75
76 return ((ngx_uint_t) op < BRIX_VFS_MUTATE_OP_COUNT)
77 ? names[op] : "unknown";
78}

80/* ---- Decide a mutation from a policy value alone ----
81 *
82 * WHAT: NGX_OK when `policy` allows mutation; NGX_ERROR with errno EINVAL for
83 * an out-of-range `op` and EROFS for a read-only endpoint.
84 *
85 * WHY: Objects that outlive their request context — file handles, staged
86 * sessions, writer sessions, queued jobs — carry the policy by value and
87 * must be able to re-decide without reaching for a context that may be
88 * gone or for a configuration that may have been reloaded.
89 *
90 * HOW: 1. Reject an operation outside the closed vocabulary; 2. allow only the
91 * exact ALLOWED value; 3. otherwise fail with EROFS.
92 */
93ngx_int_t
94brix_vfs_require_mutation_policy(brix_vfs_mutation_policy_t policy,
95 brix_vfs_mutation_op_t op)
96{
97 if ((ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT) {
98 errno = EINVAL;
99 return NGX_ERROR;
100 }
101
102 if (policy != BRIX_VFS_MUTATION_ALLOWED) {
103 errno = EROFS;
104 return NGX_ERROR;
105 }
106
107 return NGX_OK;
108}

110/* ---- Attribute one read-only denial ----
111 *
112 * WHAT: Bumps the bounded vfs_mutation_denied counter for (proto, op). No-op
113 * for an out-of-range operation.
114 *
115 * WHY: An operator needs to see that an endpoint is refusing writes, and which
116 * family of write, without a path/user/object label ever entering a
117 * metric (INVARIANT #8). Recording it in the kernel and nowhere else is
118 * what keeps an edge rejection and a VFS rejection from both counting.
119 *
120 * HOW: 1. Range-check the op; 2. hand (proto, op) to the metrics recorder.
121 */
122void
123brix_vfs_mutation_denied_observe(brix_proto_t proto, brix_vfs_mutation_op_t op)
124{
125 if ((ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT) {
126 return;
127 }
128
129 brix_metric_vfs_mutation_denied(proto, (ngx_uint_t) op);
130}

132/* ---- Decide a mutation from a request context ----
133 *
134 * WHAT: NGX_OK when `ctx` may mutate; NGX_ERROR with EINVAL for a NULL context
135 * or bad operation, EROFS for a read-only endpoint (observed once).
136 *
137 * WHY: This is the form every handle-less mutation entry point uses. Keeping
138 * the NULL check here rather than at each call site is what makes a
139 * hand-built or forgotten context fail closed instead of dereferencing.
140 *
141 * HOW: 1. Reject a NULL context; 2. run the pure kernel on
142 * ctx->mutation_policy; 3. on EROFS record one denial and preserve errno.
143 */
144ngx_int_t
145brix_vfs_require_mutation(const brix_vfs_ctx_t *ctx, brix_vfs_mutation_op_t op)
146{
147 if (ctx == NULL) {
148 errno = EINVAL;
149 return NGX_ERROR;
150 }
151
152 if (brix_vfs_require_mutation_policy(ctx->mutation_policy, op) != NGX_OK) {
153 if (errno == EROFS) {
154 brix_vfs_mutation_denied_observe(brix_vfs_metrics_proto(ctx), op);
155 /* phase-110 W4: the refusal happens BEFORE any VFS op runs, so no
156 * observer will record it — stamp the outcome class on the
157 * request's/session's monitor here so $brix_status says
158 * "forbidden" on every plane (the same class EROFS maps to in
159 * brix_metric_err_from_errno). */
160 brix_io_monitor_record_err(ctx->io_monitor, BRIX_ERR_FORBIDDEN);
161 errno = EROFS;
162 }
163 return NGX_ERROR;
164 }
165
166 return NGX_OK;
167}

169/* ---- Decide a path-based mutation from a request context ----
170 *
171 * WHAT: Confinement first (NGX_ERROR/EINVAL for an unresolved or escaped
172 * path), then the mutation policy (NGX_ERROR/EROFS). NGX_OK otherwise.
173 *
174 * WHY: Every path mutator owes both checks, and the order matters: an
175 * unconfined path is a malformed request, not a policy question, so it
176 * must not be answered with the endpoint's write posture.
177 *
178 * HOW: 1. brix_vfs_require_confined; 2. brix_vfs_require_mutation.
179 */
180ngx_int_t
181brix_vfs_require_confined_mutation(const brix_vfs_ctx_t *ctx,
182 brix_vfs_mutation_op_t op)
183{
184 if (brix_vfs_require_confined(ctx) != NGX_OK) {
185 return NGX_ERROR;
186 }
187
188 return brix_vfs_require_mutation(ctx, op);
189}

191/* ---- Decide a mutation from a policy an object carries ----
192 *
193 * WHAT: NGX_OK when the carried policy permits `op`; NGX_ERROR with EROFS
194 * (observed once against `proto`) or EINVAL for a bad operation.
195 *
196 * WHY: A handle, staged session, writer or queued job outlives — and may be
197 * used without — the context that created it. Re-reading that context
198 * would be a use-after-free at worst and a policy the request no longer
199 * owns at best, so the object decides from its own copy. Routing that
200 * decision through here rather than an open-coded comparison is what
201 * keeps the counter complete.
202 *
203 * HOW: 1. run the pure kernel; 2. on EROFS record one denial and preserve
204 * errno.
205 */
206ngx_int_t
207brix_vfs_require_carried_mutation(brix_vfs_mutation_policy_t policy,
208 brix_proto_t proto, brix_vfs_mutation_op_t op)
209{
210 if (brix_vfs_require_mutation_policy(policy, op) != NGX_OK) {
211 if (errno == EROFS) {
212 brix_vfs_mutation_denied_observe(proto, op);
213 errno = EROFS;
214 }
215 return NGX_ERROR;
216 }
217
218 return NGX_OK;
219}

221/* ---- Build a policy-bearing operation context ----
222 *
223 * WHAT: Zero-fills *opctx and stores log, root_canon, policy, and proto. No-op
224 * on NULL.
225 *
226 * WHY: The raw export helpers run off the event loop and have no request
227 * context; this bundle is how the request's authority travels with them.
228 * Zeroing first means a partially-filled bundle is READ_ONLY.
229 *
230 * HOW: 1. NULL-guard; 2. memzero; 3. assign the four borrowed fields.
231 */
232void
233brix_vfs_export_op_ctx_init(brix_vfs_export_op_ctx_t *opctx, ngx_log_t *log,
234 const char *root_canon, brix_vfs_mutation_policy_t policy,
235 brix_proto_t proto)
236{
237 if (opctx == NULL) {
238 return;
239 }
240
241 ngx_memzero(opctx, sizeof(*opctx));
242 opctx->log = log;
243 opctx->root_canon = root_canon;
244 opctx->n2n = brix_vfs_backend_n2n(root_canon);
245 opctx->mutation_policy = (policy == BRIX_VFS_MUTATION_ALLOWED)
246 ? BRIX_VFS_MUTATION_ALLOWED : BRIX_VFS_MUTATION_READ_ONLY;
247 opctx->proto = proto;
248}

250/* ---- Derive an operation context from a request VFS context ----
251 *
252 * WHAT: Copies `ctx`'s log, export root, mutation policy, and metrics protocol
253 * into *opctx. A NULL `ctx` yields a READ_ONLY bundle with no root.
254 *
255 * WHY: This is the only sanctioned bridge from a request to an off-thread
256 * helper, and it cannot widen authority: there is no argument with which
257 * a caller could ask for more than the request already had.
258 *
259 * HOW: 1. NULL-guard opctx; 2. for a NULL ctx build the closed bundle;
260 * 3. otherwise forward the four fields through the init helper.
261 */
262void
263brix_vfs_export_op_ctx_from(brix_vfs_export_op_ctx_t *opctx,
264 const brix_vfs_ctx_t *ctx)
265{
266 if (opctx == NULL) {
267 return;
268 }
269
270 if (ctx == NULL) {
271 brix_vfs_export_op_ctx_init(opctx, NULL, NULL,
272 BRIX_VFS_MUTATION_READ_ONLY, BRIX_PROTO_ROOT);
273 return;
274 }
275
276 brix_vfs_export_op_ctx_init(opctx, ctx->log, ctx->root_canon,
277 ctx->mutation_policy, brix_vfs_metrics_proto(ctx));
278 opctx->n2n = ctx->n2n;
279}

281/* ---- Decide a mutation from an operation context ----
282 *
283 * WHAT: NGX_OK when *opctx may mutate; NGX_ERROR with EINVAL for a NULL bundle
284 * or bad operation, EROFS for a read-only endpoint (observed once).
285 *
286 * WHY: The off-thread twin of brix_vfs_require_mutation, so a helper reached
287 * from a thread-pool job is gated by exactly the same kernel rather than
288 * by a second, weaker spelling of the rule.
289 *
290 * HOW: 1. Reject a NULL bundle; 2. run the pure kernel; 3. observe an EROFS
291 * denial against the bundle's protocol.

305 errno = EROFS;

308 }

313/* ---- Classify raw open flags as mutating or not ----
314 *
315 * WHAT: Returns 1 when `flags` can modify the object or the namespace, 0 for a
316 * provably read-only open.
317 *
318 * WHY: brix_vfs_open_fd() takes raw O_* flags, so its policy-bearing wrapper
319 * has to read intent out of the flag word rather than a VFS flag set. A
320 * read-only open must stay free on a read-only export, or the raw helpers
321 * could no longer serve reads.
322 *
323 * HOW: 1. Mask the access mode and compare against O_RDONLY; 2. test the three
324 * creation/size-changing flags; 3. return the disjunction.
325 */
326int
327brix_vfs_open_flags_mutate(int flags)
328{
329 if ((flags & O_ACCMODE) != O_RDONLY) {
330 return 1;
331 }
332
333 return (flags & (O_CREAT | O_TRUNC | O_APPEND)) != 0 ? 1 : 0;
334}
