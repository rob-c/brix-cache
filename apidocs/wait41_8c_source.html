1/*
2 * wait41.c — block until an XRootD server accepts connections.
3 *
4 * WHAT: `wait41-brix [--timeout S] [--full] host[:port]` — poll until the server is
5 * reachable (TCP connect; with --full, a complete handshake+login). Exits 0
6 * when ready, non-zero on timeout. The readiness helper the harness wants.
7 * WHY: A tiny front-end over the client transport/session layer. libXrdCl-free.
8 * HOW: Loop brix_tcp_connect (or brix_connect with --full) until a deadline,
9 * sleeping 1s between attempts.

35/* WHAT: Parse wait41 options and positional endpoint.
36 * WHY: Separate command-line policy from readiness polling.
37 * HOW: Update timeout/full/endpoint and signal immediate help or version exit. */
38static int
39wait41_options(int argc, char **argv, const char **endpoint, int *timeout_s,
40 int *full, int *exit_now)
41{
42 int i;
43
44 *exit_now = 0;
45 for (i = 1; i < argc; i++) {
46 if (strcmp(argv[i], "--version") == 0) {
47 printf("%s (BriX-Cache client) %s\n", brix_prog_base(argv[0]),
48 brix_client_version());
49 *exit_now = 1;
50 return 0;
51 }
52 if (strcmp(argv[i], "--help") == 0 || strcmp(argv[i], "-h") == 0) {
53 *exit_now = 1;
54 return wait41_usage(argv[0]);
55 }
56 if (strcmp(argv[i], "--timeout") == 0 && i + 1 < argc) {
57 *timeout_s = atoi(argv[++i]);
58 } else if (strcmp(argv[i], "--full") == 0) {
59 *full = 1;
60 } else {
61 *endpoint = argv[i];
62 }
63 }
64 return 0;
65}

68/* WHAT: Attempt one readiness probe.
69 * WHY: Keep full-session and TCP-only resource handling out of the retry loop.
70 * HOW: Connect using the selected mode, close successful resources, and report
71 * a boolean ready result.
72 */
73static int
74wait41_probe(const brix_url *url, int full, brix_status *status)
75{
76 if (full) {
77 brix_conn connection;
78
79 if (brix_connect(&connection, url, NULL, status) != 0) {
80 return 0;
81 }
82 brix_close(&connection);
83 return 1;
84 }
85 {
86 int fd = brix_tcp_connect(url->host, url->port, 1000, status);
87
88 if (fd < 0) {
89 return 0;
90 }
91 close(fd);
92 return 1;
93 }
94}

97/* Real main; dispatched from xrddiag (multi-call, see xrddiag.c). */
98int
99brix_wait41_main(int argc, char **argv)
100{
101 brix_url u;
102 brix_status st;
103 const char *endpoint = NULL;
104 int timeout_s = 60, full = 0;
105 int exit_now;
106 time_t deadline;
107
108 wait41_options(argc, argv, &endpoint, &timeout_s, &full, &exit_now);
109 if (exit_now) {
110 return 0;
111 }
112 if (endpoint == NULL) {
113 fprintf(stderr, "usage: %s [--timeout S] [--full] host[:port]\n", brix_prog_base(argv[0]));
114 return 50;
115 }
116
117 brix_crypto_init();
118 brix_status_clear(&st);
119 if (brix_endpoint_parse(endpoint, &u, &st) != 0) {
120 fprintf(stderr, "%s: %s\n", argv[0], st.msg);
121 return 50;
122 }
123
124 deadline = time(NULL) + timeout_s;
125 for (;;) {
126 brix_status_clear(&st);
127 if (wait41_probe(&u, full, &st)) {
128 printf("%s:%d ready\n", u.host, u.port);
129 return 0;
130 }
131 if (time(NULL) >= deadline) {
132 fprintf(stderr, "%s: %s:%d not ready after %ds\n",
133 argv[0], u.host, u.port, timeout_s);
134 return 1;
135 }
136 sleep(1);
137 }
138}
