4/*
5 * verify.h — checksum-on-fill integrity for the read-through cache.
6 *
7 * WHAT: After a fill has streamed a complete file into its ".part" staging file
8 * (but BEFORE the atomic rename that publishes it), recompute the file's
9 * content checksum locally and compare it to the digest the origin
10 * advertised. A mismatch discards the part so a corrupted transfer never
11 * becomes a served cache entry; a match records the verified digest in the
12 * file's .cinfo (BRIX_CINFO_F_VERIFIED) for durable provenance.
13 *
14 * WHY: The cache previously trusted whatever bytes arrived from the origin. A
15 * truncated, bit-rotted, or man-in-the-middled transfer would be cached
16 * and served to every subsequent client. Verifying against the origin's
17 * own checksum closes that gap and is the integrity half of XrdPfc that
18 * the module lacked. The policy is FAIL-CLOSED, BEST-EFFORT (the default):
19 * verify whenever the origin can supply a digest we can compute; commit
20 * (flagged unverified) when it cannot; never serve a proven-bad file.
21 *
22 * HOW: The transport (origin/transport.h) reports the origin's advertised
23 * algorithm+hex via its checksum() op. This module opens the part file
24 * (O_RDONLY|O_NOFOLLOW), drives the shared checksum kernel
25 * (brix_checksum_hex_name_fd) for the SAME algorithm name, and compares
26 * the two hex strings case-insensitively. It is transport-agnostic: the
27 * xroot:// driver feeds it a kXR_Qcksum reply, the HTTP driver a Digest
28 * header — the verify logic is identical.
29 */

35/* Verification policy (config: brix_cache_verify off|best-effort|require,
36 * plus the phase-68 self-verifying mode cvmfs-cas). */
37typedef enum {
38 BRIX_CACHE_VERIFY_OFF = 0, /* never verify (legacy behaviour) */
39 BRIX_CACHE_VERIFY_BESTEFFORT, /* verify if a digest is available (default)*/
40 BRIX_CACHE_VERIFY_REQUIRE, /* a usable digest is mandatory; else fail */
41 BRIX_CACHE_VERIFY_CVMFS_CAS, /* phase-68: the object NAME is the digest

53/* 1 iff `mode` is a SELF-ADDRESSING scheme: the cache key itself names the
54 * expected digest, so the fill needs no origin-advertised checksum at all.
55 * The three such modes share one dispatcher, one fail-closed policy and one
56 * local-posix-store requirement; asking here is what keeps the three call
57 * sites that care from each enumerating the list and drifting apart. */
58#define brix_cache_verify_is_selfaddr(mode) \
59 ((mode) == BRIX_CACHE_VERIFY_CVMFS_CAS \
60 || (mode) == BRIX_CACHE_VERIFY_OCI_DIGEST \
61 || (mode) == BRIX_CACHE_VERIFY_RPM_REPODATA)

69typedef enum {
70 BRIX_CACHE_VERIFY_VERIFIED = 0, /* computed == origin digest */

76/*
77 * Verify the staged part file at `part_path` against `origin` under `mode`.
78 *
79 * Returns:
80 * VERIFIED — origin digest present and matched; out_alg / out_hex hold the
81 * canonical algorithm name and computed hex (for the sidecar).
82 * UNVERIFIED — origin offered no usable digest AND mode==BESTEFFORT: the
83 * caller commits the file but records it unverified. (In
84 * mode==REQUIRE this case returns ERROR instead.)
85 * MISMATCH — origin digest present but differs: caller MUST discard the
86 * part. t error is set (kXR_ChkSumErr).
87 * ERROR — could not compute (open/read failure) or REQUIRE with no
88 * digest. t error is set.
89 *
90 * mode==OFF short-circuits to UNVERIFIED without touching the file.
91 * out_alg (>=16 bytes) / out_hex (>=129 bytes) may be NULL if the caller does
92 * not need the computed values. On VERIFIED the caller persists out_alg/out_hex
93 * into the .meta sidecar (brix_cache_meta_t.cks_alg/cks_hex) it already writes.
94 */
95brix_cache_verify_result_e brix_cache_verify_part(brix_cache_fill_t *t,
96 const char *part_path, const brix_cache_digest_t *origin,
97 brix_cache_verify_mode_e mode, char *out_alg, char *out_hex);
