10/*
11 * WHAT: kXR_QStats, kXR_Qxattr, kXR_QFinfo, kXR_QFSinfo, kXR_Qvisa, kXR_Qopaque, kXR_Qopaquf, kXR_Qopaqug — metadata and plugin-style query handlers.
12 * QStats returns XML-formatted server statistics (connections, bytes, uptime); Qxattr lists extended attributes on a file path;
13 * QFinfo returns "0" as placeholder; QFSinfo delegates to space handler; Qvisa/opaques return FSctl/fctl unsupported responses
14 * matching reference XRootD behavior since nginx-xrootd does not embed the XrdOfs plugin layer.
15 *
16 * WHY: These queries provide server observability (stats), filesystem metadata (xattr, finfo), and extension hooks (visa/opaques) that clients use
17 * to discover server capabilities and file properties. QStats XML matches reference format for monitoring dashboards; xattr returns oss.* key-value
18 * pairs including file type, size, timestamps, and user.U.* extended attributes for HEP data provenance tracking. Opaque queries return unsupported
19 * to maintain protocol compatibility with clients that send FSctl/fctl requests expecting a consistent response shape.
20 *
21 * HOW: Three shared static helpers handle common patterns: arg_missing() logs + sends kXR_ArgMissing error; fsctl_unsupported()/fctl_unsupported()
22 * log + send kXR_Unsupported for plugin operations. payload_equals() compares wire payload against expected text with null-termination handling.
23 * Public APIs follow security chain (extract_path → resolve_path → authdb → vo_acl) where applicable, then delegate to stat/listxattr/getxattr syscalls
24 * or return placeholder/unsupported responses. QStats reads metrics struct + local socket address for port extraction; xattr builds oss.* key-value string
25 * with listxattr iteration filtering user.U.* prefixes.
26 */

52static ngx_flag_t
53brix_query_payload_equals(brix_ctx_t *ctx, const char *text)
54{
55 size_t len, text_len;
56
57 if (ctx->recv.payload == NULL) {
58 return 0;
59 }
60

64 }
65

72static void
73stats_append(char *resp, size_t cap, size_t *pos, const char *fmt, ...)
74{
75 va_list ap;
76 int n;
77
78 if (*pos >= cap) {
79 return;
80 }
81 va_start(ap, fmt);
82 n = vsnprintf(resp + *pos, cap - *pos, fmt, ap);
83 va_end(ap);
84 if (n > 0) {
85 *pos += ((size_t) n < cap - *pos) ? (size_t) n : cap - *pos - 1;
86 }

89/* ---- Which sections did the QStats arg select? ----
90 *
91 * WHAT: Returns the stock selector bitmask for the request payload: letters
92 * a=all b=buff d=poll i=info l=link p=protocol s=sched u=proc (verified
93 * live against 5.6.9); unknown letters contribute nothing; an absent /
94 * empty payload behaves like 'a' (the pre-existing BriX behavior, which
95 * the suites pin).
96 *
97 * WHY: §1.13 — stock clients ask for subsets (mpxstats polls 'p' etc.); BriX
98 * used to ignore the argument entirely and always answer everything.
99 *
100 * HOW: One pass over the payload letters setting bits; 'a' sets all.
101 */
102#define BRIX_QSTATS_INFO 0x01
103#define BRIX_QSTATS_LINK 0x02
104#define BRIX_QSTATS_PROTO 0x04
105#define BRIX_QSTATS_ALL 0x07
106
107static unsigned
108stats_selector(const brix_ctx_t *ctx)
109{
110 unsigned mask = 0;
111 uint32_t i;
112 const char *p = (const char *) ctx->recv.payload;
113
114 if (p == NULL || ctx->recv.cur_dlen == 0) {
115 return BRIX_QSTATS_ALL;
116 }
117 for (i = 0; i < ctx->recv.cur_dlen && p[i] != '\0'; i++) {
118 switch (p[i]) {
119 case 'a': mask |= BRIX_QSTATS_ALL; break;
120 case 'i': mask |= BRIX_QSTATS_INFO; break;
121 case 'l': mask |= BRIX_QSTATS_LINK; break;
122 case 'p': mask |= BRIX_QSTATS_PROTO; break;
123 default: break; /* stock: unknown letters contribute nothing */
124 }
125 }
126 return mask;
127}

129/* brix_query_stats — kXR_QStats: the stock-shaped XML <statistics> document.
130 *
131 * WHAT: Emits the reference wrapper (tod/ver/src/tos/pgm/ins/pid/site
132 * attributes, byte-shape verified live against stock 5.6.9) and the
133 * sections BriX can fill honestly — info, link, xrootd(ops from the
134 * per-op metric slots), oss(v=2, statvfs of the export), sgen — gated
135 * by the stock selector letters. Sections whose letters BriX has no
136 * data for (buff/poll/sched/proc) are simply absent, exactly like an
137 * unknown letter on stock.
138 *
139 * WHY: §1.13 — the old abbreviated document broke XML-parsing consumers:
140 * wrong root attributes (spurious id=, missing tod/src/pid) and no ops
141 * section. Counters that do not exist are emitted as 0 rather than
142 * invented; ver/pgm keep the honest BriX identity.
143 *
144 * HOW: 1. Resolve selector mask + identity fields (hostname from ngx_cycle,
145 * port from metrics or the local sockaddr, tos latched on first use).
146 * 2. Append the wrapper, then each selected section from the metric
147 * slots (misc/err aggregate the unmapped op slots so totals stay
148 * truthful). 3. oss + sgen ride the 'a' selection only, like stock.
149 */
150
151/* Resolve the local listen port for the stats src= field: the metrics slot
152 * first, else the connection's local sockaddr (v4/v6). 0 when unknown. */
153static int
154stats_resolve_port(brix_ctx_t *ctx, ngx_connection_t *c)
155{
156 int port = ctx->metrics ? (int) ctx->metrics->port : 0;
157
158 if (port == 0 && c->local_sockaddr != NULL) {
159 if (c->local_sockaddr->sa_family == AF_INET) {
160 port = (int) ntohs(((struct sockaddr_in *)
161 c->local_sockaddr)->sin_port);
162 } else if (c->local_sockaddr->sa_family == AF_INET6) {
163 port = (int) ntohs(((struct sockaddr_in6 *)
164 c->local_sockaddr)->sin6_port);
165 }
166 }
167 return port;
168}

170ngx_int_t
171brix_query_stats(brix_ctx_t *ctx, ngx_connection_t *c,
172 ngx_stream_brix_srv_conf_t *conf)
173{
174 static time_t tos_latch = 0; /* worker start proxy: first-stats time */
175 char resp[4096];
176 size_t pos = 0;
177 unsigned sel = stats_selector(ctx);
178 int port = stats_resolve_port(ctx, c);
179 time_t now = time(NULL);
180 long ok_open = 0, err_sum = 0, misc = 0;
181 int i;
182
183 if (tos_latch == 0) {
184 tos_latch = now;
185 }
186
187 /* site="" is the summary-monitoring site label federation dashboards read;
188 * populate it from brix_sitename (the advertise.sitename slot), matching
189 * stock's all.sitename. Emitted raw like the adjacent hostname — an operator
190 * sitename is a plain identifier, not attacker input. Empty when unset. */
191 stats_append(resp, sizeof(resp), &pos,
192 "<statistics tod=\"%ld\" ver=\"" BRIX_SERVER_VERSION "\""
193 " src=\"%.*s:%d\" tos=\"%ld\" pgm=\"" BRIX_SERVER_NAME "\""
194 " ins=\"brix\" pid=\"%d\" site=\"%.*s\">",
195 (long) now, (int) ngx_cycle->hostname.len, ngx_cycle->hostname.data,
196 port, (long) tos_latch, (int) ngx_pid,
197 (int) conf->advertise.sitename.len, conf->advertise.sitename.data);
198
199 if (sel & BRIX_QSTATS_INFO) {
200 stats_append(resp, sizeof(resp), &pos,
201 "<stats id=\"info\"><host>%.*s</host><port>%d</port>"
202 "<name>" BRIX_SERVER_NAME "</name></stats>",
203 (int) ngx_cycle->hostname.len, ngx_cycle->hostname.data, port);
204 }
205
206 if ((sel & BRIX_QSTATS_LINK) && ctx->metrics) {
207 /* Stock tag order: num maxn tot in out ctime tmo stall sfps. maxn
208 * (max ever concurrent) is not tracked — the live count is the best
209 * lower bound we can attest. */
210 stats_append(resp, sizeof(resp), &pos,
211 "<stats id=\"link\"><num>%ld</num><maxn>%ld</maxn>"
212 "<tot>%ld</tot><in>%ld</in><out>%ld</out>"
213 "<ctime>0</ctime><tmo>0</tmo><stall>0</stall>"
214 "<sfps>0</sfps></stats>",
215 (long) ctx->metrics->connections_active,
216 (long) ctx->metrics->connections_active,
217 (long) ctx->metrics->connections_total,
218 (long) ctx->metrics->bytes_rx_total,
219 (long) ctx->metrics->bytes_tx_total);
220 }
221
222 if ((sel & BRIX_QSTATS_PROTO) && ctx->metrics) {
223 for (i = 0; i < BRIX_NOPS; i++) {
224 err_sum += (long) ctx->metrics->op_err[i];
225 misc += (long) ctx->metrics->op_ok[i];
226 }
227 ok_open = (long) ctx->metrics->op_ok[BRIX_OP_OPEN_RD]
228 + (long) ctx->metrics->op_ok[BRIX_OP_OPEN_WR];
229 /* misc = every completed op not named in the ops block, so the block
230 * totals stay truthful. */
231 misc -= ok_open
232 + (long) ctx->metrics->op_ok[BRIX_OP_READ]
233 + (long) ctx->metrics->op_ok[BRIX_OP_PGREAD]
234 + (long) ctx->metrics->op_ok[BRIX_OP_READV]
235 + (long) ctx->metrics->op_ok[BRIX_OP_WRITEV]
236 + (long) ctx->metrics->op_ok[BRIX_OP_WRITE]
237 + (long) ctx->metrics->op_ok[BRIX_OP_SYNC]
238 + (long) ctx->metrics->op_ok[BRIX_OP_LOGIN];
239 if (misc < 0) {
240 misc = 0;
241 }
242 stats_append(resp, sizeof(resp), &pos,
243 "<stats id=\"xrootd\"><num>%ld</num>"
244 "<ops><open>%ld</open><rf>0</rf>"
245 "<rd>%ld</rd><pr>%ld</pr><rv>%ld</rv><rs>0</rs>"
246 "<wv>%ld</wv><ws>0</ws><wr>%ld</wr><sync>%ld</sync>"
247 "<getf>0</getf><putf>0</putf><misc>%ld</misc></ops>"
248 "<sig><ok>0</ok><bad>0</bad><ign>0</ign></sig>"
249 "<aio><num>0</num><max>0</max><rej>0</rej></aio>"
250 "<err>%ld</err><rdr>0</rdr><dly>0</dly>"
251 "<lgn><num>%ld</num><af>0</af><au>0</au><ua>0</ua></lgn>"
252 "</stats>",
253 (long) ctx->metrics->connections_total, ok_open,
254 (long) ctx->metrics->op_ok[BRIX_OP_READ],
255 (long) ctx->metrics->op_ok[BRIX_OP_PGREAD],
256 (long) ctx->metrics->op_ok[BRIX_OP_READV],
257 (long) ctx->metrics->op_ok[BRIX_OP_WRITEV],
258 (long) ctx->metrics->op_ok[BRIX_OP_WRITE],
259 (long) ctx->metrics->op_ok[BRIX_OP_SYNC],
260 misc, err_sum,
261 (long) ctx->metrics->op_ok[BRIX_OP_LOGIN]);
262 }
263
264 /* oss + sgen ride the full document only, mirroring stock's 'a'. */
265 if ((sel & BRIX_QSTATS_ALL) == BRIX_QSTATS_ALL) {
266 struct statvfs vfs;
267
268 stats_append(resp, sizeof(resp), &pos,
269 "<stats id=\"ofs\"><role>%s</role></stats>",
270 conf->manager_mode ? "manager" : "server");
271 if (conf->common.root_canon[0] != '\0'
272 && statvfs(conf->common.root_canon, &vfs) == 0)
273 {
274 stats_append(resp, sizeof(resp), &pos,
275 "<stats id=\"oss\" v=\"2\"><paths>1"
276 "<stats id=\"0\"><lp>\"/\"</lp><rp>\"%s\"</rp>"
277 "<tot>%llu</tot><free>%llu</free>"
278 "<ino>%llu</ino><ifr>%llu</ifr></stats></paths>"
279 "<space>0</space></stats>",
280 conf->common.root_canon,
281 (unsigned long long) (vfs.f_blocks * (vfs.f_frsize / 1024)),
282 (unsigned long long) (vfs.f_bavail * (vfs.f_frsize / 1024)),
283 (unsigned long long) vfs.f_files,
284 (unsigned long long) vfs.f_favail);
285 }
286 stats_append(resp, sizeof(resp), &pos,
287 "<stats id=\"sgen\"><as>1</as><et>0</et><toe>%ld</toe></stats>",
288 (long) now);
289 }
290
291 stats_append(resp, sizeof(resp), &pos, "</statistics>");
292
293 brix_log_access(ctx, c, "QUERY", "-", "stats", 1, 0, NULL, 0);
294 BRIX_OP_OK(ctx, BRIX_OP_QUERY_STATS);
295 return brix_send_ok(ctx, c, resp, (uint32_t) (pos + 1));
296}

298/* Prologue for kXR_Qxattr: extract the request path, resolve it beneath the
299 * export root, run the read auth gate, and VFS-probe the target (following
300 * symlinks). Fills pathbuf, full_path and the vctx/vst it probes into. Returns NGX_OK to proceed, or
301 * the send rc / ctx->write_rc that the caller must return on any failure. */
302static ngx_int_t
303xattr_resolve_and_probe(brix_ctx_t *ctx, ngx_connection_t *c,
304 ngx_stream_brix_srv_conf_t *conf, char *pathbuf, char *full_path,
305 brix_vfs_ctx_t *vctx, brix_vfs_stat_t *vst)
306{
307 if (ctx->recv.cur_dlen == 0 || ctx->recv.payload == NULL) {
308 BRIX_OP_ERR(ctx, BRIX_OP_QUERY_XATTR);
309 return brix_send_error(ctx, c, kXR_ArgMissing, "xattr: path required");
310 }
311 if (!brix_extract_path(c->log, ctx->recv.payload, ctx->recv.cur_dlen,
312 pathbuf, BRIX_MAX_PATH + 1, 1)) {
313 BRIX_OP_ERR(ctx, BRIX_OP_QUERY_XATTR);
314 return brix_send_error(ctx, c, kXR_ArgInvalid, "invalid path");
315 }
316 /* phase74-fp: pathbuf is the request path, full_path the output buf. */
317 brix_beneath_full_path(conf->common.root_canon, pathbuf, /* NOLINT(readability-suspicious-call-argument) */
318 full_path, PATH_MAX);
319 if (brix_auth_gate(ctx, c, BRIX_OP_QUERY_XATTR, "QUERY",
320 pathbuf, full_path, conf,
321 BRIX_AUTH_READ, 0) != NGX_OK) {
322 return ctx->write_rc;
323 }
324 /* Stat + xattr list/get all flow through the VFS (one ctx, confined to the
325 * export root). probe (follow) replaces the raw stat; OP_STAT is suppressed
326 * (probe) so only the enclosing QUERY op is accounted. */
327 brix_root_vfs_ctx_init(ctx, c, conf, vctx, full_path);
328 if (brix_vfs_probe(vctx, 0 /* follow */, vst) != NGX_OK) {
329 BRIX_OP_ERR(ctx, BRIX_OP_QUERY_XATTR);
330 return brix_send_error(ctx, c, brix_kxr_from_errno(errno),
331 strerror(errno));
332 }
333 return NGX_OK;
334}

336/* brix_query_xattr — kXR_Qxattr: list a path's extended attributes through the
337 * full security chain (extract → resolve → authdb → VO ACL → stat), returning the
338 * oss.* key-values plus any user.U.*-prefixed xattrs. */
339ngx_int_t
340brix_query_xattr(brix_ctx_t *ctx, ngx_connection_t *c,
341 ngx_stream_brix_srv_conf_t *conf)
342{
343 char pathbuf[BRIX_MAX_PATH + 1];
344 char full_path[PATH_MAX];
345 char resp[4096];
346 int pos = 0;
347 char raw_list[4096];
348 ssize_t list_sz;
349 brix_vfs_ctx_t vctx;
350 brix_vfs_stat_t vst;
351 char ftype;
352 char facc;
353
354 {
355 ngx_int_t prc = xattr_resolve_and_probe(ctx, c, conf, pathbuf,
356 full_path, &vctx, &vst);
357 if (prc != NGX_OK) {
358 return prc;
359 }
360 }
361
362 /* SECURITY: a stat/probe succeeds for anyone who can merely TRAVERSE to the
363 * file (POSIX), so the probe alone would expose oss.used (size) + the user.U.*
364 * xattrs of a peer's UNREADABLE 0600 file to a cross-tenant caller who can
365 * reach its 0755 parent — a metadata oracle. UNDER IMPERSONATION only (the
366 * cross-tenant case), require actual READ access as the mapped user via the
367 * broker, the same gate query-checksum enforces. Gating on
368 * brix_imp_client_active() leaves the single-identity (non-map) path byte-for-
369 * byte unchanged — no new open() for a server that has one filesystem view. */
370 if (vst.is_regular && brix_imp_client_active()) {
371 int rfd = brix_vfs_open_fd(c->log, conf->common.root_canon, full_path,
372 O_RDONLY, 0);
373 if (rfd < 0) {
374 BRIX_OP_ERR(ctx, BRIX_OP_QUERY_XATTR);
375 return brix_send_error(ctx, c, brix_kxr_from_errno(errno),
376 strerror(errno));
377 }
378 close(rfd);
379 }
380
381 if (vst.is_regular) {
382 ftype = 'f';
383 } else if (vst.is_directory) {
384 ftype = 'd';
385 } else {
386 ftype = 'o';
387 }
388
389 facc = (vst.mode & S_IWUSR) ? 'w' : 'r';
390
391 pos = snprintf(resp, sizeof(resp) - 1,
392 "oss.cgroup=default&oss.type=%c&oss.used=%lld"
393 "&oss.mt=%ld&oss.ct=%ld&oss.at=%ld"
394 "&oss.u=*&oss.g=*&oss.fs=%c",
395 ftype, (long long) vst.size,
396 (long) vst.mtime, (long) vst.ctime, (long) vst.atime,
397 facc);
398
399 list_sz = brix_vfs_listxattr(&vctx, raw_list, sizeof(raw_list));
400 if (list_sz > 0) {
401 char *lp = raw_list;
402 char *lend = raw_list + list_sz;
403
404 while (lp < lend && pos < (int) sizeof(resp) - 256) {
405 size_t nlen = strlen(lp);
406
407 if (strncmp(lp, "user.U.", 7) == 0 && nlen > 7) {
408 char val[1024];
409 ssize_t vlen;
410
411 vlen = brix_vfs_getxattr(&vctx, lp, val, sizeof(val) - 1);
412 if (vlen >= 0) {
413 val[vlen] = '\0';
414 pos += snprintf(resp + pos, sizeof(resp) - pos - 1,
415 "&%s=%.*s", lp + 5, (int) vlen, val);
416 }
417 }
418
419 lp += nlen + 1;
420 }
421 }
422
423 brix_log_access(ctx, c, "QUERY", pathbuf, "xattr", 1, 0, NULL, 0);
424 BRIX_OP_OK(ctx, BRIX_OP_QUERY_XATTR);
425
426 return brix_send_ok(ctx, c, resp, (uint32_t) (pos + 1));
427}
