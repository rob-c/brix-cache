1/*
2 * vfs_unlink_many.c — VFS bulk delete (phase-107 C4): the per-level rmtree
3 * chunker and the flat client-batch entry brix_vfs_delete_many().
4 *
5 * WHAT: Two callers of the storage drivers' unlink_many/_cred batch slots:
6 * brix_vfs_rmtree_dispatch() routes a recursive driver delete either to
7 * the classic per-key walk (vfs_unlink.c) or, when the LEAF advertises
8 * BRIX_SD_CAP_BULK_DELETE, to a windowed walk that accumulates up to
9 * BRIX_SD_BULK_DELETE_WINDOW file keys and flushes them in one driver
10 * round trip; brix_vfs_delete_many() serves the S3 DeleteObjects
11 * handler's flat, client-supplied key list under ONE policy check and
12 * ONE metric observation for the whole batch.
13 *
14 * WHY: A DeleteObjects of 1,000 keys over a remote backend was 1,000 signed
15 * HTTPS round trips whose entire purpose was to avoid exactly that. The
16 * tree walk may only batch WITHIN one directory level — a prefix cannot
17 * be removed before its children, and every registered driver except
18 * mirage advertises CAP_DIRS, so the per-level rule IS the rule. The one
19 * place a flat window is legitimate is brix_vfs_delete_many(), where the
20 * client itself supplied a flat key list and no tree walk is involved.
21 *
22 * HOW: The window pre-fills every per-key errno slot with ECANCELED before a
23 * flush so a key the driver never reached can never be reported as
24 * deleted; the batch contract (sd_batch_types.h) has the slot overwrite
25 * exactly the leading `done` entries it attempted. The chunker gates on
26 * the LEAF's caps and dispatches through brix_sd_unlink_many_maybe_cred
27 * on the LEAF (the R-wave truncate_path lesson: gate and dispatch on the
28 * same instance, or a cache-fronted export loses the slot). Each flush
29 * books one brix_metric_vfs_bulk_delete() observation carrying the key
30 * count in the VALUE, never in a label (INVARIANT #8).
31 */

39/* The rmtree accumulation window: up to BRIX_SD_BULK_DELETE_WINDOW owned
40 * (strdup'd) file keys plus their per-key result vector, bound to the leaf
41 * instance + resolved credential so a flush needs no extra arguments. Heap-
42 * allocated by the dispatcher (~12 KiB of pointers/ints — too large for the
43 * worker stack next to the recursion it feeds). */
44typedef struct {
45 brix_sd_instance_t *leaf;
46 const brix_sd_cred_t *cred;
47 const char *paths[BRIX_SD_BULK_DELETE_WINDOW];
48 int errs[BRIX_SD_BULK_DELETE_WINDOW];
49 size_t n;
50} brix_vfs_unlink_window_t;

53static void
54brix_vfs_unlink_window_reset(brix_vfs_unlink_window_t *w)
55{
56 size_t i;

64/* Flush the accumulated window as ONE unlink_many batch on the leaf.
65 *
66 * WHAT: Pre-fills every result slot with ECANCELED, runs the batch through
67 * brix_sd_unlink_many_maybe_cred, books the per-batch metric, and turns
68 * the per-key results back into rmtree semantics: any per-key failure
69 * fails the walk with that key's errno (the classic walk aborts on the
70 * first failed unlink; a batch must not weaken that).
71 * WHY: ECANCELED pre-fill is the "never report an untried key as deleted"
72 * clause of the batch contract made concrete — a transport failure at
73 * key k leaves errs[k..n) exactly as we set them here.
74 * HOW: One metric observation per flush, key count (successes) as the value,
75 * leaf driver as the label. The window is emptied on every exit so the
76 * caller can keep accumulating or abort without leaking key copies. */
77static ngx_int_t
78brix_vfs_unlink_window_flush(brix_vfs_unlink_window_t *w)
79{
80 brix_sd_unlink_batch_t b;
81 size_t i, attempted, removed = 0;
82 ngx_int_t rc;
83 int saved;
84
85 if (w->n == 0) {
86 return NGX_OK;
87 }
88
89 for (i = 0; i < w->n; i++) {
90 w->errs[i] = ECANCELED;
91 }
92 b.paths = w->paths;
93 b.n = w->n;
94 b.errs = w->errs;
95 b.done = 0;
96
97 errno = 0;
98 rc = brix_sd_unlink_many_maybe_cred(w->leaf, &b, w->cred);
99 saved = errno;
100
101 attempted = (rc == NGX_OK) ? w->n : b.done;
102 for (i = 0; i < attempted; i++) {
103 if (w->errs[i] == 0) {
104 removed++;
105 }
106 }
107 brix_metric_vfs_bulk_delete(brix_sd_backend_name(w->leaf), removed);
108
109 if (rc == NGX_OK) {
110 for (i = 0; i < w->n; i++) {
111 if (w->errs[i] != 0) {
112 saved = w->errs[i];
113 rc = NGX_ERROR;
114 break;
115 }
116 }
117 }
118
119 brix_vfs_unlink_window_reset(w);
120 if (rc != NGX_OK) {
121 errno = saved != 0 ? saved : EIO;
122 }
123 return rc;
124}

126/* Add one file key to the window, flushing first when it is full. The copy is
127 * owned by the window (the walk's stack path buffer is reused per level). */
128static ngx_int_t
129brix_vfs_unlink_window_add(brix_vfs_unlink_window_t *w, const char *logical)
130{
131 char *copy;
132

134 && brix_vfs_unlink_window_flush(w) != NGX_OK)
135 {
136 return NGX_ERROR;
137 }

148/* brix_vfs_rmtree_bulk — the windowed twin of brix_vfs_driver_rmtree.
149 *
150 * WHAT: Depth-first walk of `logical`: a file joins the window (deleted at the
151 * next flush); a directory recurses over its children, then FLUSHES the
152 * window, then removes itself. Same stat/opendir/readdir dispatch and
153 * ELOOP depth bound as the classic walk.
154 * WHY: The flush-before-rmdir is the whole correctness content of C4's trap:
155 * a directory's accumulated child files must be gone before the
156 * directory itself, or an S3-appearing success leaves a half-removed
157 * tree on a backend with real collections. Sibling files of an inner
158 * directory may flush early inside its recursion — files are deletable
159 * any time before their OWN parent's boundary, so early is safe.
160 * HOW: stat decides file/dir per child (d_type is a hint, never authority —
161 * sd_remote_dir.c says so in as many words), matching what the classic
162 * walk pays; the saving is unlink round trips, not stats. */
163static ngx_int_t
164brix_vfs_rmtree_bulk(brix_sd_instance_t *leaf, const brix_sd_driver_t *drv,
165 const char *logical, const brix_sd_cred_t *cred, ngx_uint_t depth,
166 brix_vfs_unlink_window_t *w)
167{
168 brix_sd_stat_t st;
169
170 if (depth > BRIX_FS_TREE_MAX_DEPTH) {
171 errno = ELOOP;
172 return NGX_ERROR;
173 }
174 if (brix_sd_stat_maybe_cred(leaf, logical, &st, cred) != NGX_OK) {
175 return NGX_ERROR; /* ENOENT etc. — errno set by the driver */
176 }
177
178 if (!st.is_dir) {
179 return brix_vfs_unlink_window_add(w, logical);
180 }
181
182 if (drv->opendir != NULL) {
183 brix_sd_dir_t *dir;
184 int err = 0;
185
186 dir = brix_sd_opendir_maybe_cred(leaf, logical, &err, cred);
187 if (dir != NULL) {
188 brix_sd_dirent_t de;
189 ngx_int_t drc;
190
191 while ((drc = drv->readdir(dir, &de)) == NGX_OK) {
192 char child[PATH_MAX];
193
194 ngx_snprintf((u_char *) child, sizeof(child), "%s/%s%Z",
195 (logical[0] == '/' && logical[1] == '\0')
196 ? "" : logical,
197 de.name);
198 if (brix_vfs_rmtree_bulk(leaf, drv, child, cred, depth + 1, w)
199 != NGX_OK)
200 {
201 drv->closedir(dir);
202 return NGX_ERROR;
203 }
204 }
205 drv->closedir(dir);
206 if (drc == NGX_ERROR) {
207 return NGX_ERROR;
208 }
209 }
210 }
211
212 /* Directory boundary: every accumulated child file must be gone before the
213 * directory itself is removed. */
214 if (brix_vfs_unlink_window_flush(w) != NGX_OK) {
215 return NGX_ERROR;
216 }
217 return brix_sd_unlink_maybe_cred(leaf, logical, 1, cred);
218}

220/* brix_vfs_rmtree_dispatch — route a recursive driver delete to the windowed
221 * walk (leaf advertises CAP_BULK_DELETE + a real unlink_many) or the classic
222 * per-key walk. The gate probes the LEAF, exactly where the dispatch lands. */
223ngx_int_t
224brix_vfs_rmtree_dispatch(brix_sd_instance_t *leaf,
225 const brix_sd_driver_t *drv, const char *logical,
226 const brix_sd_cred_t *cred)
227{
228 brix_vfs_unlink_window_t *w;
229 ngx_int_t rc;
230 int saved;
231
232 if (brix_sd_supports(leaf, BRIX_SD_CAP_BULK_DELETE) != NGX_OK
233 || leaf->driver->unlink_many == NULL)
234 {
235 return brix_vfs_driver_rmtree(leaf, drv, logical, cred, 0);
236 }
237
238 w = calloc(1, sizeof(*w));
239 if (w == NULL) {
240 errno = ENOMEM;
241 return NGX_ERROR;
242 }
243 w->leaf = leaf;
244 w->cred = cred;
245
246 rc = brix_vfs_rmtree_bulk(leaf, drv, logical, cred, 0, w);
247 if (rc == NGX_OK) {
248 rc = brix_vfs_unlink_window_flush(w); /* top-level file case */
249 }
250 saved = errno;
251 brix_vfs_unlink_window_reset(w);
252 free(w);
253 errno = saved;
254 return rc;
255}

258 * unlink semantics (the same opts brix_vfs_unlink uses), per-key errno into
259 * errs. The batch itself cannot fail here — rc is per key. */

269

291/* The dispatch half of the driver arm: hand the whole logical key list to
292 * the leaf's unlink_many slot when it has one (window-capped by the caller),
293 * else run the per-key unlink loop. Per-key verdicts land in errs[]; *done
294 * counts the keys actually attempted; NGX_OK = every key attempted. */
295static ngx_int_t
296delete_many_dispatch(brix_sd_instance_t *leaf, const char *const *logical,
297 size_t n, int *errs, size_t *done, const brix_sd_cred_t *cred)
298{
299 size_t i;
300 ngx_int_t rc;
301
302 if (leaf->driver->unlink_many != NULL
303 || leaf->driver->unlink_many_cred != NULL)
304 {
305 brix_sd_unlink_batch_t b;
306
307 b.paths = logical;
308 b.n = n;
309 b.errs = errs;
310 b.done = 0;
311 errno = 0;
312 rc = brix_sd_unlink_many_maybe_cred(leaf, &b, cred);
313 *done = (rc == NGX_OK) ? n : b.done;
314 return rc;
315 }
316
317 for (i = 0; i < n; i++) {
318 errno = 0;

320 errs[i] = 0;
321 } else {
322 errs[i] = errno != 0 ? errno : EIO;
323 }
324 (*done)++;
325 }
326 return NGX_OK;

329/* Driver arm of brix_vfs_delete_many: resolve the credential ONCE for the
330 * whole batch, hand the full flat key list to the leaf's unlink_many slot
331 * (window-capped by the caller), or fall back to the per-key unlink loop when
332 * the leaf has no batch slot. Evicts each successfully removed key from a
333 * fronting cache decorator — the leaf dispatch skipped the decorator's own
334 * unlink and with it its cstore evict (the N–P wave's stale-bytes class). */
335static ngx_int_t
336brix_vfs_delete_many_via_driver(brix_vfs_ctx_t *ctx,
337 const char *const *paths, size_t n, int *errs, size_t *done)
338{
339 brix_sd_instance_t *leaf = brix_vfs_ns_leaf(ctx->sd);
340 brix_sd_ucred_t store;
341 brix_sd_cred_t cred;
342 const char **logical;
343 char (*physical)[PATH_MAX];
344 size_t i, removed = 0;
345 ngx_int_t rc;
346 int saved, use_cred = 0, cred_err = 0;
347
348 /* Zero before the gate: an unzeroed cred hands a garbage inactive pointer
349 * to the driver's cred slot (same rule as brix_vfs_delete_via_driver). */
350 ngx_memzero(&cred, sizeof(cred));
351 if (brix_vfs_cred_gate_active(ctx)) {
352 if (brix_vfs_ns_cred(ctx, &store, &cred, &use_cred, &cred_err)
353 != NGX_OK)
354 {
355 errno = cred_err ? cred_err : EACCES;
356 return NGX_ERROR;
357 }
358 }
359
360 logical = malloc(n * sizeof(*logical));
361 physical = malloc(n * sizeof(*physical));
362 if (logical == NULL || physical == NULL) {
363 brix_sd_ucred_wipe(&store);
364 free(logical);
365 free(physical);
366 errno = ENOMEM;
367 return NGX_ERROR;
368 }
369 for (i = 0; i < n; i++) {
370 if (brix_path_resolved_to_pfn(ctx, paths[i], physical[i],
371 sizeof(physical[i])) != NGX_OK)
372 {
373 saved = errno;
374 brix_sd_ucred_wipe(&store);
375 free(logical);
376 free(physical);
377 errno = saved;
378 return NGX_ERROR;
379 }
380 logical[i] = physical[i];
381 }
382
383 rc = delete_many_dispatch(leaf, logical, n, errs, done,
384 use_cred ? &cred : NULL);
385 saved = errno;
386 brix_sd_ucred_wipe(&store); /* secret consumed by the batch; erase */
387
388 for (i = 0; i < *done; i++) {
389 if (errs[i] == 0) {
390 removed++;
391 brix_metric_cache_evicted(brix_vfs_metrics_proto(ctx),
392 brix_sd_cache_evict(ctx->sd,
393 logical[i]));
394 }
395 }
396 brix_metric_vfs_bulk_delete(brix_sd_backend_name(leaf), removed);
397
398 free(logical);
399 free(physical);
400 errno = saved;
401 return rc;
402}

404/* brix_vfs_delete_many — delete a flat, client-supplied batch of confined
405 * paths under ONE policy check and ONE metric observation.
406 *
407 * WHAT: The S3 DeleteObjects batch entry: every element of `paths` is an
408 * already-confined absolute path under ctx->root_canon (the handler's
409 * per-key confinement ran first); errs[i] receives 0 or a positive errno
410 * per key (ENOENT stays ENOENT — the CALLER decides idempotency);
411 * *done counts the leading keys actually attempted.
412 * WHY: One brix_vfs_unlink per key meant one policy check, one metric line
413 * and — over a remote backend — one signed round trip per key. The batch
414 * runs the EROFS gate once BEFORE any key is examined (a read-only
415 * endpoint discloses nothing about which keys exist) and books one
416 * OP_DELETE observation carrying the key count.
417 * HOW: errs is pre-filled with ECANCELED so an untried key can never read as
418 * deleted; NGX_ERROR means the batch itself failed (policy, credential,
419 * or transport — errno says which), NGX_OK means every key was attempted
420 * and errs holds the per-key verdicts. n is capped at the batch window
421 * (the S3 handler's own 1,000-key cap is the same constant). */
422ngx_int_t
423brix_vfs_delete_many(brix_vfs_ctx_t *ctx, const char *const *paths, size_t n,
424 int *errs, size_t *done)
425{
426 const char *anchor;
427 uint64_t start;
428 ngx_int_t rc;
429 int saved;
430 size_t i;
431
432 *done = 0;
433 for (i = 0; i < n; i++) {
434 errs[i] = ECANCELED;
435 }
436 if (n > BRIX_SD_BULK_DELETE_WINDOW) {
437 errno = EINVAL;
438 return NGX_ERROR;
439 }
440
441 start = brix_vfs_now_ns();
442 anchor = brix_vfs_ctx_path(ctx);
443
444 if (brix_vfs_gate_confined(ctx,

446 {
447 saved = errno;

465 /* phase-107 C7: lock gate for every key, BEFORE any arm touches storage —
466 * the batch must not slip past the gate its single-key twin takes
467 * (vfs_unlink.c). The batch form sweeps once with a parent-chain memo (a
468 * flat batch costs n + 2 quiet reads, not 3n over a remote leaf). Refusal
469 * is atomic: no key has been attempted yet, so the whole batch refuses
470 * (errs stays ECANCELED, *done stays 0) rather than leaving a partial
471 * delete behind a lock conflict. Same coverage as the single-key gate
472 * (exact match + depth-infinity ancestors); strict refuses, advisory books
473 * and admits, off reads nothing. */
474 if (brix_vfs_require_unlocked_many(ctx, paths, n,
475 BRIX_VFS_MUTATE_REMOVE) != NGX_OK)
476 {
477 saved = errno;
478 brix_vfs_observe_ctx_op(ctx, anchor, BRIX_METRIC_OP_DELETE, NULL,
479 n, NGX_ERROR, saved, start);
480 return NGX_ERROR;
481 }
482
483 if (brix_vfs_ctx_driver(ctx) != NULL) {
484 rc = brix_vfs_delete_many_via_driver(ctx, paths, n, errs, done);
485 } else {
486 rc = brix_vfs_delete_many_via_namespace(ctx, paths, n, errs, done);
487 }
488 saved = errno;
489 brix_vfs_observe_ctx_op(ctx, anchor, BRIX_METRIC_OP_DELETE, NULL, n,
490 rc, rc == NGX_OK ? 0 : saved, start);
491 return rc;
492}
