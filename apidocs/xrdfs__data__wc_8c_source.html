74/* Parse wc's [-c] [-l] [-w] flags plus the <path> operand into `a` (each want_* set
75 * when its flag is present, a->arg = the bare path or NULL). Never fails. */
76static void
77wc_parse_args(int argc, char **argv, wc_args_t *a)
78{
79 int i;

81 for (i = 1; i < argc; i++) {

84 else if (strcmp(argv[i], "-w") == 0) { a->want_w = 1; }
85 else { a->arg = argv[i]; }

90/* wc [-c] [-l] [-w] <path> — count bytes/lines/words. With no flag, prints all three
91 * (lines words bytes), like wc(1). -c alone is answered from stat (no read); -l/-w
92 * stream the file once. Output columns match the selected counters, then the path. */
93int
94do_wc(brix_conn *c, const char *cwd, int argc, char **argv)
95{
96 brix_status st;
97 brix_statinfo si;
98 char path[XRDC_PATH_MAX];
99 wc_args_t a = {0};
100 long long lines = 0, words = 0, bytes = 0;
101
102 wc_parse_args(argc, argv, &a);
103 if (a.arg == NULL) { fprintf(stderr, "usage: wc [-c] [-l] [-w] <path>\n"); return 50; }
104 if (!a.want_c && !a.want_l && !a.want_w) { a.want_l = a.want_w = a.want_c = 1; }
105 build_path(cwd, a.arg, path, sizeof(path));
106
107 brix_status_clear(&st);
108 if (brix_stat(c, path, &si, &st) != 0) {
109 return xrdfs_report_err("wc", path, &st, 0, c);
110 }
111 bytes = (long long) si.size;
112
113 if (a.want_l || a.want_w) { /* a single streaming pass counts lines + words */
114 int wrc = wc_count_stream(c, path, &lines, &words, &st);
115 if (wrc != 0) { return wrc; }
116 }
117
118 if (a.want_l) { printf(" %lld", lines); }
119 if (a.want_w) { printf(" %lld", words); }
120 if (a.want_c) { printf(" %lld", bytes); }
121 printf(" %s\n", path);
122 return 0;
123}
