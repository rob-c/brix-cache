1/*
2 * rpm_prefetch.c — warming the two files a dnf client always asks for next
3 * (phase-104 D15.10).
4 *
5 * WHAT: after a fresh repomd.xml has been pulled through this mirror, read
6 * the warm set out of it (primary + filelists) and fill those objects
7 * into the cache store on the thread pool, before any client asks.
8 * WHY: Appendix X finding X-3 measured what a stock EL9 dnf does with a
9 * repository index: it fetches primary AND filelists, unconditionally,
10 * every time its metadata window expires. On a cold mirror those two
11 * fills are the client's entire wait, and they are the one speculation
12 * a repository mirror can make that is never wrong — the index the
13 * mirror is holding names them. The alternative is what every mirror
14 * does today: three serial WAN round trips per client, of which the
15 * last two were knowable the moment the first returned.
16 * HOW: the trigger is narrow on purpose — a REPOMD-class request that was a
17 * FILL, i.e. this worker has just pulled a *new* index. The index is
18 * read from the handle the serve is about to use (a positional pread,
19 * so it disturbs nothing), the hrefs are re-checked by the request
20 * grammar (brix_rpm_classify) exactly as if a client had asked for
21 * them, and only the fills themselves are posted to the thread pool.
22 * Nothing here ever runs an origin fetch on the event loop, nothing
23 * parks the client's request on the speculation, and every failure is
24 * silent to the client by construction: a warm fill that does not
25 * happen is the cache miss the client would have had anyway.
26 */

38/* Warm fills in flight on THIS worker. The set is at most two objects per
39 * index, so the cap is about a reload storm (every location re-pulling its
40 * index at once), not about steady state. */
41#define RPM_PREFETCH_MAX_INFLIGHT 4
42
43static ngx_uint_t rpm_prefetch_inflight;
44
45typedef struct {
46 ngx_pool_t *pool;
47 ngx_log_t *log;
48 brix_sd_instance_t *inst;
49 ngx_uint_t count;
50 ngx_uint_t warmed; /* thread out */
51 ngx_uint_t failed; /* thread out */
52 char key[BRIX_RPM_REPOMD_WARM_MAX][BRIX_RPM_KEY_MAX];
53} rpm_prefetch_job_t;

56/* The thread half: the same whole-file fill the miss path runs, one object at
57 * a time. `needs_offload` answering 0 means the object is already COMPLETE in
58 * the store (nothing to warm) or the source is local (nothing worth warming),
59 * so it is the presence check as well as the worth-it check. */
60static void
61rpm_prefetch_thread(void *data, ngx_log_t *log)
62{
63 rpm_prefetch_job_t *t = data;
64 ngx_uint_t i;
65
66 (void) log;
67
68 for (i = 0; i < t->count; i++) {
69 if (!brix_sd_cache_fill_needs_offload(t->inst, t->key[i])) {
70 continue;
71 }
72 if (brix_sd_cache_fill_key(t->inst, t->key[i], NULL) == NGX_OK) {
73 t->warmed++;
74 BRIX_RPM_METRIC_INC(prefetch_total);
75 continue;
76 }
77 t->failed++;
78 BRIX_RPM_METRIC_INC(prefetch_fail_total);
79 }
80}

83/* The completion half (event loop): drop the in-flight count and say what
84 * happened once, at debug level. A warm pass that fetched nothing is the
85 * normal case on a mirror whose index did not change, and an operator who
86 * wants to know how much it absorbed reads the two counters. */
87static void
88rpm_prefetch_done(ngx_event_t *ev)
89{
90 ngx_thread_task_t *task = ev->data;
91 rpm_prefetch_job_t *t = task->ctx;
92 ngx_pool_t *pool = t->pool;
93
94 if (rpm_prefetch_inflight > 0) {
95 rpm_prefetch_inflight--;
96 }
97 ngx_log_debug3(NGX_LOG_DEBUG_HTTP, t->log, 0,
98 "rpm: repodata prefetch done: %ui warmed, %ui failed, "
99 "%ui in flight", t->warmed, t->failed,
100 rpm_prefetch_inflight);
101 ngx_destroy_pool(pool);
102}

105/* Post the composed key set. Every failure is a silent return: this is
106 * speculation, and a mirror that fails a request because it could not
107 * speculate would be worse than one that never speculated. */
108static void
109rpm_prefetch_post(ngx_http_request_t *r, brix_sd_instance_t *sd,
110 char keys[][BRIX_RPM_KEY_MAX], ngx_uint_t count)
111{
112 static ngx_str_t pname = ngx_string("default");
113 ngx_thread_pool_t *tp;
114 ngx_thread_task_t *task;
115 rpm_prefetch_job_t *t;
116 ngx_pool_t *pool;
117 ngx_uint_t i;
118
119 if (rpm_prefetch_inflight >= RPM_PREFETCH_MAX_INFLIGHT) {
120 return;
121 }
122 tp = (ngx_cycle != NULL)
123 ? ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname) : NULL;
124 if (tp == NULL) {
125 return;
126 }
127 /* The job owns a pool of its own: it outlives the request that noticed
128 * the new index, and r->pool is gone the moment the response finalizes. */
129 pool = ngx_create_pool(8192, ngx_cycle->log);
130 if (pool == NULL) {
131 return;
132 }
133 task = ngx_thread_task_alloc(pool, sizeof(rpm_prefetch_job_t));
134 if (task == NULL) {
135 ngx_destroy_pool(pool);
136 return;
137 }
138 t = task->ctx;
139 t->pool = pool;
140 t->log = ngx_cycle->log;
141 t->inst = sd;
142 t->count = count;
143 for (i = 0; i < count; i++) {
144 ngx_cpystrn((u_char *) t->key[i], (u_char *) keys[i],
145 BRIX_RPM_KEY_MAX);
146 }
147
148 brix_task_bind(task, rpm_prefetch_thread, rpm_prefetch_done);
149 task->event.log = t->log;
150 if (ngx_thread_task_post(tp, task) != NGX_OK) {
151 ngx_destroy_pool(pool);
152 return;
153 }
154 rpm_prefetch_inflight++;
155
156 ngx_log_debug2(NGX_LOG_DEBUG_HTTP, r->connection->log, 0,
157 "rpm: repodata prefetch posted: %ui object(s) named by "
158 "\"%V\"", count, &r->uri);
159}

162/* The composed, re-classified key set. An href that does not classify as a
163 * digest-named metadata file is dropped rather than fetched: the index is
164 * upstream data, and the only paths this mirror fetches unasked are the ones
165 * its own request grammar would have accepted from a client. */
166static ngx_uint_t
167rpm_prefetch_keys(ngx_http_request_t *r, ngx_http_brix_rpm_ctx_t *ctx,
168 const char *xml, size_t len, char keys[][BRIX_RPM_KEY_MAX])
169{
170 brix_rpm_repomd_ref_t refs[BRIX_RPM_REPOMD_WARM_MAX];
171 brix_rpm_req_t req;
172 size_t n, i;
173 ngx_uint_t count = 0;
174
175 n = brix_rpm_repomd_warm_set(xml, len, refs, BRIX_RPM_REPOMD_WARM_MAX);
176
177 for (i = 0; i < n; i++) {
178 if (brix_rpm_repomd_sibling_key(ctx->key, ctx->key_len, refs[i].href,
179 refs[i].href_len, keys[count],
180 BRIX_RPM_KEY_MAX) != 0)
181 {
182 continue;
183 }
184 if (brix_rpm_classify(keys[count], ngx_strlen(keys[count]), &req) != 0
185 || req.cls != BRIX_RPM_REQ_METADATA)
186 {
187 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
188 "rpm: ignoring repodata location \"%*s\" named by \"%V\" - "
189 "not a digest-named metadata file",
190 (int) refs[i].href_len, refs[i].href, &r->uri);
191 continue;
192 }
193 count++;
194 }
195 return count;
196}
