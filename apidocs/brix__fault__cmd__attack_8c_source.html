80cmd_trigger(int once, char *args, char *reply, size_t rsz)
81{

186/* `flap [<up_ms> <down_ms>|off]` — bounce the listener on a duty cycle. Every
187 * arm bumps g_flap_gen so the previously-detached thread retires itself. */
188static void
189attack_flap(const char *args)
190{
191 if (strcmp(args, "off") == 0 || args[0] == '\0') {
192 __atomic_add_fetch(&g_flap_gen, 1, __ATOMIC_SEQ_CST);
193 g_flap_on = 0;
194 return;
195 }
196 int up = 0, down = 0;
197 sscanf(args, "%d %d", &up, &down);
198 g_flap_up_ms = up; g_flap_down_ms = down;
199 unsigned gen = __atomic_add_fetch(&g_flap_gen, 1, __ATOMIC_SEQ_CST);
200 pthread_t th;
201 if (pthread_create(&th, NULL, flap_thread, (void *) (uintptr_t) gen) == 0) {
202 pthread_detach(th);
203 g_flap_on = 1;
204 }
205}

207/* `ramp <lever> <start> <end> <ms>` — sweep one lever over a window in a
208 * detached thread. 0 on a usage error (the arg is owned by the thread on 1). */
209static int
210attack_ramp(const char *args)
211{
212 struct ramp_arg *r = calloc(1, sizeof(*r));
213 if (!r) {
214 return 0;
215 }
216 if (sscanf(args, "%23s %lf %lf %d", r->lever, &r->start, &r->end, &r->ms) != 4) {
217 free(r);
218 return 0;
219 }
220 r->gen = __atomic_load_n(&g_ramp_gen, __ATOMIC_SEQ_CST);
221 pthread_t th;
222 if (pthread_create(&th, NULL, ramp_thread, r) != 0) {
223 free(r);
224 } else {
225 pthread_detach(th);
226 }
227 return 1;
228}

317 * different error for each — so they are named rather than left as bare ints. */

322/* The scalar smuggling levers: each is "set fields from a number", identical for
323 * every selected direction, so one direction's config is all this needs.
324 * HTTP_OK if `sub` named one of them, HTTP_UNKNOWN otherwise. */
325static int
326http_lever_scalar(fp_http_cfg *h, const char *sub, const char *rbuf)
327{
328 if (strcmp(sub, "cl-te") == 0) {
329 h->add_cl = 1; h->cl_val = atol(rbuf); h->add_te = 1;
330 } else if (strcmp(sub, "te-cl") == 0) {
331 h->add_te = 1; h->add_cl = 1; h->cl_val = atol(rbuf);
332 } else if (strcmp(sub, "dup-cl") == 0) {
333 h->dup_cl = 1; h->dup_cl_val = atol(rbuf);
334 } else if (strcmp(sub, "obfuscate-te") == 0) {
335 int m = atoi(rbuf);
336 h->obfuscate_te = (m < 1 || m > 3) ? 1 : m;
337 } else if (strcmp(sub, "naked-lf") == 0) {
338 h->naked_lf = 1;
339 } else {
340 return HTTP_UNKNOWN;
341 }
342 return HTTP_OK;
343}

474static int

476{
