2 * oci_token_cache.c — the D1.3 SHM token cache and its key discipline.
3 *
4 * WHAT: the (upstream, scope[, credential]) → bearer mapping the token dance
5 * reads and writes, the pull-scope derivation both it and the D16 proof
6 * gate key their records by, and the credential-blind "share" write a
7 * granted proof uses to lend its bearer to the coalesced fill.
8 * WHY: split from oci_upstream_auth.c when D16 pushed that file past the

10 * owns HOW cache keys are derived (the security property that one
11 * principal's token can never satisfy another's request lives entirely

13 * HOW: every key is a raw sha256 over NUL-separated components, so no wire

28/* Renew this far before the upstream's own expiry, so a token minted at the
29 * edge of its life is never handed to a fill that then out-lives it. */
30#define OCI_TOKEN_SKEW_S 30
31/* An upstream that reports an implausibly short life still gets a cache entry
32 * — otherwise every blob of a large image re-runs the whole dance. */
33#define OCI_TOKEN_MIN_TTL_S 5

35/* ---- token cache -------------------------------------------------------- */
36
37/* sha256 of [data, data+len) as 32 RAW bytes — the form every SHM key in
38 * this plane takes. Exported (internal header) because the D16 proof gate and
39 * the credential digest derive their keys with the same primitive; a second
40 * hex→raw walk would be a second chance to get the nibble math wrong.
41 * 0 = key written / -1 = EVP failure. */
42int
43brix_oci_sha256_key(const void *data, size_t len, u_char key[32])
44{
45 brix_oci_digest_t d;
46 int i;
47
48 if (brix_oci_sha256(data, len, &d) != 0) {
49 return -1;
50 }
51 for (i = 0; i < 32; i++) {
52 /* d.hex is plain `char`, which is signed on x86: promote through
53 * unsigned char so the arithmetic below cannot see a negative. */
54 unsigned hi = (unsigned char) d.hex[i * 2];
55 unsigned lo = (unsigned char) d.hex[i * 2 + 1];
56
57 hi = (hi <= '9') ? hi - '0' : hi - 'a' + 10;
58 lo = (lo <= '9') ? lo - '0' : lo - 'a' + 10;
59 key[i] = (u_char) ((hi << 4) | lo);
60 }
61 return 0;
62}

64/* The cache key is sha256(base_url ‖ 0x00 ‖ scope [‖ 0x00 ‖ cred]), 32 raw
65 * bytes. The NUL separator is what stops ("https://a.io/x", "y") and
66 * ("https://a.io/", "xy") from colliding onto one entry — a collision here
67 * would hand one repository's token to another repository's fill. `cred`
68 * (32 raw bytes, or NULL for the credential-blind entry) extends the same
69 * reasoning to principals: without it, one user's bearer would satisfy every
70 * other user's request for the same scope (D16). */
71static int
72oci_token_cache_key(const brix_oci_upstream_t *up, const char *scope,
73 const u_char *cred, u_char key[32])
74{
75 char buf[1024 + 33];
76 size_t n, s;
77
78 n = strlen(up->base_url);
79 s = strlen(scope);
80 if (n + 1 + s + 33 >= sizeof(buf)) {
81 return -1;
82 }
83 memcpy(buf, up->base_url, n);
84 buf[n] = '\0';
85 memcpy(buf + n + 1, scope, s);
86 n += 1 + s;
87 if (cred != NULL) {
88 buf[n] = '\0';
89 memcpy(buf + n + 1, cred, 32);
90 n += 33;
91 }
92
93 return brix_oci_sha256_key(buf, n, key);
94}

137/* ---- scope derivation --------------------------------------------------- */

139/* When a challenge omits `scope=`, derive the pull scope from the route being
140 * fetched: "repository:<name>:pull". `path` is the canonical key, so the name
141 * is everything between "/v2/" and the terminal — no parsing risk, but the
142 * bound still matters because the result becomes a query parameter. Public
143 * because the D16 proof gate keys its SHM records by this same string — two
144 * derivations would be two chances to disagree about what one repository is. */
145int
146brix_oci_pull_scope(const char *path, char *out, size_t outsz)
147{
148 const char *name, *term;
149 size_t len;
150
151 if (strncmp(path, "/v2/", 4) != 0) {
152 return -1;
153 }
154 name = path + 4;
155
156 term = strstr(name, "/manifests/");
157 if (term == NULL) {
158 term = strstr(name, "/blobs/");
159 }
160 /* The D16 gate also proves the listing routes — the tags/referrers
161 * surfaces are exactly where a private repository's METADATA would
162 * otherwise leak — and they authorize on the same repository:pull. */
163 if (term == NULL) {
164 term = strstr(name, "/tags/");
165 }
166 if (term == NULL) {
167 term = strstr(name, "/referrers/");
168 }
169 if (term == NULL || term == name) {
170 return -1;
171 }
172 len = (size_t) (term - name);
173
174 return (snprintf(out, outsz, "repository:%.*s:pull", (int) len, name)
175 < (int) outsz) ? 0 : -1;
176}
