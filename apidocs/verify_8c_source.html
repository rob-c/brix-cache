1/*
2 * verify.c — checksum-on-fill integrity (see verify.h for the design contract).
3 *
4 * Transport-agnostic: it is handed the origin's advertised digest (from a
5 * kXR_Qcksum reply, an HTTP Digest header, or a Pelican namespace record) and a
6 * staged ".part" file, and decides commit-or-discard under the configured
7 * policy. All work runs in a fill thread-pool worker (blocking pread of the
8 * part file); no event-loop calls.
9 */

41/* brix_cache_hex_ieq — case-insensitive equality of two hex digests * Origins vary in hex case (XRootD lowercases; some HTTP Digest headers upper).
42 * Returns 1 when equal ignoring case, 0 otherwise. Lengths must match too. */
43static int
44brix_cache_hex_ieq(const char *a, const char *b)
45{
46 size_t la = strlen(a);
47 size_t lb = strlen(b);
48
49 if (la != lb) {
50 return 0;
51 }
52 return strncasecmp(a, b, la) == 0;
53}

56/* brix_cache_verify_no_origin — policy verdict when the origin gave no digest.
57 *
58 * WHAT: Returns BRIX_CACHE_VERIFY_ERROR (after recording a kXR_ChkSumErr on t)
59 * under REQUIRE mode, otherwise BRIX_CACHE_VERIFY_UNVERIFIED. Security-neutral:
60 * no bytes are hashed here — this is the "nothing to compare against" branch.
61 *
62 * WHY: Isolates the missing-origin-digest policy fork so the orchestrator stays
63 * a flat sequence; the REQUIRE-vs-permissive decision lives in one place.
64 *
65 * HOW:
66 * 1. REQUIRE mode → set the error on t and return ERROR.
67 * 2. Any other mode → return UNVERIFIED (commit without a checksum).
68 */
69static brix_cache_verify_result_e
70brix_cache_verify_no_origin(brix_cache_fill_t *t, brix_cache_verify_mode_e mode)
71{
72 if (mode == BRIX_CACHE_VERIFY_REQUIRE) {
73 brix_cache_set_error(t, kXR_ChkSumErr, 0,
74 "cache verify: origin provided no checksum (require mode)");
75 return BRIX_CACHE_VERIFY_ERROR;
76 }
77 return BRIX_CACHE_VERIFY_UNVERIFIED;
78}

81/* brix_cache_verify_unsupported — policy verdict for an unimplemented algorithm.
82 *
83 * WHAT: Called when brix_checksum_hex_name_fd returned NGX_DECLINED (we do not
84 * implement the origin's algorithm). Returns BRIX_CACHE_VERIFY_ERROR (recording
85 * a kXR_ChkSumErr) under REQUIRE mode, else logs an INFO note and returns
86 * BRIX_CACHE_VERIFY_UNVERIFIED. No comparison is performed — verification was
87 * impossible, never bypassed.
88 *
89 * WHY: Keeps the "cannot verify this algorithm" fork out of the orchestrator so
90 * its control flow (and thus its complexity) is dominated by the real verify path.
91 *
92 * HOW:
93 * 1. REQUIRE mode → format the unsupported-algorithm message, set it on t,
94 * return ERROR.
95 * 2. Otherwise → if a log is present, emit the "committing unverified" INFO
96 * line, then return UNVERIFIED.
97 */
98static brix_cache_verify_result_e
99brix_cache_verify_unsupported(brix_cache_fill_t *t,
100 const brix_cache_digest_t *origin, brix_cache_verify_mode_e mode,
101 ngx_log_t *log, const char *part_path)
102{
103 if (mode == BRIX_CACHE_VERIFY_REQUIRE) {
104 char msg[96];
105 (void) snprintf(msg, sizeof(msg),
106 "cache verify: unsupported origin checksum algorithm \"%.15s\"",
107 origin->alg);
108 brix_cache_set_error(t, kXR_ChkSumErr, 0, msg);
109 return BRIX_CACHE_VERIFY_ERROR;
110 }
111 if (log != NULL) {
112 ngx_log_error(NGX_LOG_INFO, log, 0,
113 "brix: cache verify: origin checksum \"%s\" not supported "
114 "locally; committing \"%s\" unverified",
115 origin->alg, part_path);
116 }
117 return BRIX_CACHE_VERIFY_UNVERIFIED;
118}

121/* brix_cache_verify_report_mismatch — record and log a digest mismatch.
122 *
123 * WHAT: Formats the "<alg> mismatch (origin=… computed=…)" message, sets it as
124 * a kXR_ChkSumErr on t, and (when a log is present) emits the ERR "discarding
125 * fill" line. Side-effect only; the caller returns BRIX_CACHE_VERIFY_MISMATCH.
126 *
127 * WHY: The mismatch branch is the security-critical failure path; concentrating
128 * its exact error text and logging here keeps that wording byte-for-byte stable
129 * and out of the orchestrator's branch budget.
130 *
131 * HOW:
132 * 1. snprintf the mismatch detail using alg_used (already resolved to the
133 * normalized name or the origin's advertised name) and both digests.
134 * 2. brix_cache_set_error on t with kXR_ChkSumErr.
135 * 3. If log is non-NULL, log the failure at ERR level.
136 */
137static void
138brix_cache_verify_report_mismatch(brix_cache_fill_t *t, ngx_log_t *log,
139 const char *part_path, const char *alg_used,
140 const brix_cache_digest_t *origin, const char *hex)
141{
142 char msg[256];
143
144 (void) snprintf(msg, sizeof(msg),
145 "cache verify: %s mismatch (origin=%.64s computed=%.64s)",
146 alg_used, origin->hex, hex);
147 brix_cache_set_error(t, kXR_ChkSumErr, 0, msg);
148 if (log != NULL) {
149 ngx_log_error(NGX_LOG_ERR, log, 0,
150 "brix: cache verify FAILED for \"%s\": %s — discarding fill",
151 part_path, msg);
152 }
153}

156/* brix_cache_verify_emit — copy the verified algorithm and digest to outputs.
157 *
158 * WHAT: When out_alg / out_hex are non-NULL, writes the truncated, NUL-terminated
159 * algorithm name (<=15 chars) and hex digest (<=128 chars) into them. No-op for
160 * any output pointer that is NULL.
161 *
162 * WHY: Only reached on the verified-match path; extracting it keeps the two
163 * optional-output branches out of the orchestrator and mirrors the exact
164 * truncation limits the callers of brix_cache_verify_part rely on.
165 *
166 * HOW:
167 * 1. If out_alg is set, memcpy min(strlen(alg_used),15) bytes and NUL-terminate.
168 * 2. If out_hex is set, memcpy min(strlen(hex),128) bytes and NUL-terminate.
169 */
170static void
171brix_cache_verify_emit(char *out_alg, char *out_hex, const char *alg_used,
172 const char *hex)
173{
174 if (out_alg != NULL) {
175 size_t n = ngx_min(strlen(alg_used), 15u);
176 ngx_memcpy(out_alg, alg_used, n);
177 out_alg[n] = '\0';
178 }
179 if (out_hex != NULL) {
180 size_t n = ngx_min(strlen(hex), 128u);
181 ngx_memcpy(out_hex, hex, n);
182 out_hex[n] = '\0';
183 }
184}

187brix_cache_verify_result_e
188brix_cache_verify_part(brix_cache_fill_t *t, const char *part_path,
189 const brix_cache_digest_t *origin, brix_cache_verify_mode_e mode,
190 char *out_alg, char *out_hex)
191{
192 ngx_log_t *log = (t->c != NULL) ? t->c->log : NULL;
193 const char *alg_used;
194 char hex[129];
195 char norm[32];
196 ngx_int_t rc;
197 int fd;
198
199 if (mode == BRIX_CACHE_VERIFY_OFF) {
200 return BRIX_CACHE_VERIFY_UNVERIFIED;
201 }
202
203 /* No origin digest available. */
204 if (origin == NULL || origin->alg[0] == '\0' || origin->hex[0] == '\0') {
205 return brix_cache_verify_no_origin(t, mode);
206 }
207
208 fd = open(part_path, O_RDONLY | O_NOFOLLOW | O_CLOEXEC | O_NOCTTY);
209 if (fd < 0) {
210 brix_cache_set_syserror(t, kXR_IOError,
211 "cache verify: part file open failed");
212 return BRIX_CACHE_VERIFY_ERROR;
213 }
214
215 /* Compute the SAME algorithm the origin advertised over the staged bytes. */
216 rc = brix_checksum_hex_name_fd(origin->alg, fd, part_path, log,
217 hex, sizeof(hex), norm, sizeof(norm));
218 close(fd);
219
220 if (rc == NGX_DECLINED) {
221 /* We do not implement the origin's algorithm — cannot verify. */
222 return brix_cache_verify_unsupported(t, origin, mode, log, part_path);
223 }
224
225 if (rc != NGX_OK) {
226 brix_cache_set_syserror(t, kXR_IOError,
227 "cache verify: local checksum computation failed");
228 return BRIX_CACHE_VERIFY_ERROR;
229 }
230
231 alg_used = norm[0] ? norm : origin->alg;
232
233 if (!brix_cache_hex_ieq(hex, origin->hex)) {
234 brix_cache_verify_report_mismatch(t, log, part_path, alg_used,
235 origin, hex);
236 return BRIX_CACHE_VERIFY_MISMATCH;
237 }
238
239 brix_cache_verify_emit(out_alg, out_hex, alg_used, hex);
240 return BRIX_CACHE_VERIFY_VERIFIED;
241}

244/* Phase-68 CVMFS-CAS self-verification — see verify.h. The expected digest is
245 * the CAS hex embedded in the key itself (raw served bytes = sha1(name)). */
246brix_cache_verify_result_e
247brix_cache_verify_cvmfs_cas(const char *part_path, const char *key,
248 ngx_log_t *log, char *out_alg, char *out_hex)
249{
250 cvmfs_url_info_t info;
251 char hex[129];
252 char norm[32];
253 ngx_int_t rc;
254 int fd;
255
256 if (cvmfs_classify_url(key, strlen(key), &info) != 0
257 || info.cls != CVMFS_URL_CAS)
258 {
259 return BRIX_CACHE_VERIFY_UNVERIFIED; /* not content-addressed */
260 }
261 if (info.cas_hex_len != 40) {
262 /* Only the sha1 (40-hex) convention is verifiable today; longer
263 * hashes would need the repo's hash-algorithm advertisement. */
264 return BRIX_CACHE_VERIFY_UNVERIFIED;
265 }
266
267 fd = open(part_path, O_RDONLY | O_NOFOLLOW | O_CLOEXEC | O_NOCTTY); /* vfs-seam-allow: DOMAIN_CACHE — cache-store staging file, svc-owned domain */
268 if (fd < 0) {
269 return BRIX_CACHE_VERIFY_ERROR;
270 }
271 rc = brix_checksum_hex_name_fd("sha1", fd, part_path, log,
272 hex, sizeof(hex), norm, sizeof(norm));
273 close(fd); /* vfs-seam-allow: DOMAIN_CACHE — cache-store staging file, svc-owned domain */
274 if (rc != NGX_OK) {
275 return BRIX_CACHE_VERIFY_ERROR;
276 }
277
278 if (!brix_cache_hex_ieq(hex, info.cas_hex)) {
279 if (log != NULL) {
280 ngx_log_error(NGX_LOG_ERR, log, 0,
281 "brix: cvmfs-cas verify FAILED for \"%s\" "
282 "(name=%.40s computed=%.40s)\n"
283 " cause: origin transfer failed CAS verification — network "
284 "corruption between cache and Stratum-1\n"
285 " fix: check WAN path / middleboxes; object was "
286 "quarantined, client will retry",
287 key, info.cas_hex, hex);
288 }
289 return BRIX_CACHE_VERIFY_MISMATCH;
290 }
291
292 if (out_alg != NULL) {
293 ngx_memcpy(out_alg, "sha1", sizeof("sha1"));
294 }
295 if (out_hex != NULL) {
296 ngx_memcpy(out_hex, hex, 41);
297 }
298 return BRIX_CACHE_VERIFY_VERIFIED;
299}

301/* Phase-104 D2.3 OCI digest self-verification — see verify.h. The expected
302 * digest is the hex the KEY names, so a blob or a digest-pinned manifest
303 * certifies itself with no origin digest, exactly like cvmfs-cas. The key
304 * also names the ALGORITHM, and that is the one we hash under: verifying a
305 * sha512 blob with sha256 would compare two different functions' output. */
306brix_cache_verify_result_e
307brix_cache_verify_oci_digest(const char *part_path, const char *key,
308 ngx_log_t *log, char *out_alg, char *out_hex)
309{
310 brix_oci_req_t req;
311 const char *alg;
312 char hex[BRIX_OCI_HEXLEN_MAX + 1];
313 char norm[32];
314 ngx_int_t rc;
315 int fd;
316
317 if (brix_oci_classify(key, strlen(key), &req) != 0) {
318 return BRIX_CACHE_VERIFY_UNVERIFIED;
319 }
320 /* Only the two digest-addressed routes bind the key to the bytes. A tag
321 * manifest is mutable by definition and the tags list is generated, so
322 * neither is verifiable here — the TTL policy governs those instead. */
323 if (!(req.cls == BRIX_OCI_REQ_BLOB
324 || (req.cls == BRIX_OCI_REQ_MANIFEST && req.ref_is_digest)))
325 {
326 return BRIX_CACHE_VERIFY_UNVERIFIED;
327 }
328
329 fd = open(part_path, O_RDONLY | O_NOFOLLOW | O_CLOEXEC | O_NOCTTY); /* vfs-seam-allow: DOMAIN_CACHE — cache-store staging file, svc-owned domain */
330 if (fd < 0) {
331 return BRIX_CACHE_VERIFY_ERROR;
332 }
333 alg = brix_oci_alg_name(req.digest.alg);
334 rc = brix_checksum_hex_name_fd(alg, fd, part_path, log,
335 hex, sizeof(hex), norm, sizeof(norm));
336 close(fd); /* vfs-seam-allow: DOMAIN_CACHE — cache-store staging file, svc-owned domain */
337 if (rc != NGX_OK) {
338 return BRIX_CACHE_VERIFY_ERROR;
339 }
340
341 if (!brix_cache_hex_ieq(hex, req.digest.hex)) {
342 if (log != NULL) {
343 ngx_log_error(NGX_LOG_ERR, log, 0,
344 "brix: oci-digest verify FAILED for \"%s\" "
345 "(named=%.64s computed=%.64s)\n"
346 " cause: the registry (or its CDN) served bytes that are not "
347 "the object the digest names — corruption or substitution\n"
348 " fix: check the upstream registry and any intercepting "
349 "middlebox; the object was quarantined, the client will retry",
350 key, req.digest.hex, hex);
351 }
352 return BRIX_CACHE_VERIFY_MISMATCH;
353 }
354
355 if (out_alg != NULL) {
356 ngx_memcpy(out_alg, alg, ngx_strlen(alg) + 1);
357 }
358 if (out_hex != NULL) {
359 ngx_memcpy(out_hex, hex,
360 brix_oci_alg_hexlen(req.digest.alg) + 1);
361 }
362 return BRIX_CACHE_VERIFY_VERIFIED;
363}

365/* Phase-104 D15.9 RPM repodata self-verification — see verify.h. The expected
366 * digest is the hex createrepo put in front of the file name, and its LENGTH
367 * is what names the algorithm: a repository built with sha512 metadata must
368 * not be checked with sha256, which would compare two different functions'

378 char norm[32];
379 ngx_int_t rc;
380 int fd;
381
382 if (brix_rpm_classify(key, strlen(key), &req) != 0
383 || req.cls != BRIX_RPM_REQ_METADATA)
