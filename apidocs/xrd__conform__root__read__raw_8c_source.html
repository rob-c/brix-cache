7/* ---- Issue one deliberately shaped kXR_read ----
8 *
9 * WHAT: Returns the raw operation result and optional owned response body.
10 * WHY: Typed reads cannot express negative lengths, bad handles or read args.
11 * HOW: Pack the in-repo codec, reserve the bounded exchange, send and receive.
12 */
13int
14xrd_conform_root_raw_read(xrd_conform_execution *execution,
15 brix_conn *connection,
16 const uint8_t handle[XRDC_FHANDLE_LEN], int64_t offset, int32_t length,
17 const void *args, uint32_t args_length, size_t expected_bytes,
18 uint16_t *wire_status,
19 uint8_t **body, uint32_t *body_length, brix_status *status)
20{
21 ClientReadRequest request;
22 xrdw_read_req_t fields = { .offset = offset, .rlen = length };
23 brix_payload payload = { args, args_length };
24 brix_resp_out response = { wire_status, body, body_length };
25 uint16_t stream_id;
26 char reason[64];
27
28 if (!xrd_conform_reserve(execution, expected_bytes,
29 reason, sizeof(reason))) {
30 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
31 return -1;
32 }
33 memcpy(fields.fhandle, handle, sizeof(fields.fhandle));
34 memset(&request, 0, sizeof(request));
35 request.requestid = htons(kXR_read);
36 xrdw_read_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
37 brix_status_clear(status);
38 if (brix_send(connection, &request,
39 args_length == 0 ? NULL : &payload, &stream_id, status) != 0)
40 return -1;
41 return brix_recv(connection, stream_id, &response, status);
42}

44/* ---- Grade one expected read refusal ----
45 *
46 * WHAT: Records whether a shaped read failed with the required kXR error.
47 * WHY: Security-negative read rows must distinguish safe refusal from faults.
48 * HOW: Execute, free any body, and compare the typed status code exactly.
49 */
50static void
51grade_refusal(xrd_conform_execution *execution, brix_conn *connection,
52 const char *id, const uint8_t handle[XRDC_FHANDLE_LEN], int64_t offset,
53 int32_t length, const void *args, uint32_t args_length, int expected_kxr,
54 const char *pass_reason)
55{
56 uint8_t *body = NULL;
57 uint32_t body_length = 0;
58 uint16_t wire_status = 0;
59 brix_status status;
60 int rc;
61
62 rc = xrd_conform_root_raw_read(execution, connection, handle, offset,
63 length, args,
64 args_length, 0, &wire_status, &body, &body_length, &status);
65 free(body);
66 xrd_conform_result_set(execution, id,
67 rc != 0 && status.kxr == expected_kxr
68 ? BRIX_CONFORM_VERDICT_CONFORMANT
69 : BRIX_CONFORM_VERDICT_DIVERGENT,
70 rc != 0 && status.kxr == expected_kxr
71 ? pass_reason : "raw-read-wrong-answer",
72 brix_kxr_name(expected_kxr), status.msg);
73}

75/* ---- Grade primary and invalid path-id request arguments ----
76 *
77 * WHAT: Exercises RD-020 and RD-021 with one-byte reads carrying read_args.
78 * WHY: A path id must select only a bound transport, never arbitrary state.
79 * HOW: Compare pathid zero bytes, then require refusal of an unbound id.
80 */
81static void
82probe_path_ids(xrd_conform_execution *execution, brix_conn *connection,
83 brix_file *file, const uint8_t *baseline, uint64_t size)
84{
85 uint8_t primary[BRIX_READ_ARGS_LEN] = { 0 };
86 uint8_t invalid[BRIX_READ_ARGS_LEN] = { 0xfe };
87 uint8_t *body = NULL;
88 uint32_t body_length = 0;
89 uint16_t wire_status = 0;
90 brix_status status;
91 int rc;
92 int ok;
93
94 if (size == 0) return;
95 rc = xrd_conform_root_raw_read(execution, connection, file->fhandle, 0, 1,
96 primary, sizeof(primary), 1,
97 &wire_status, &body, &body_length, &status);
98 ok = rc == 0 && wire_status == kXR_ok && body_length == 1
99 && body != NULL && body[0] == baseline[0];
100 xrd_conform_result_set(execution, "RD-020",
101 ok ? BRIX_CONFORM_VERDICT_CONFORMANT
102 : BRIX_CONFORM_VERDICT_DIVERGENT,
103 ok ? "primary-pathid-returned-primary-bytes"
104 : "primary-pathid-wrong-answer",
105 "one byte on primary stream", ok ? NULL : status.msg);
106 free(body);
107 grade_refusal(execution, connection, "RD-021", file->fhandle, 0, 1,
108 invalid, sizeof(invalid), kXR_ArgInvalid, "unbound-pathid-refused");
109}
