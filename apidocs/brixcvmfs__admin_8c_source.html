39static int ad_gc_flags(int argc, char **argv, int idx, cvmfs_gc_opts_t *o) {
40 while (idx < argc) {
41 const char *flag = argv[idx];
42 const char *val = idx + 1 < argc ? argv[idx + 1] : NULL;
43 if (val == NULL) return -1;
44 if (strcmp(flag, "--keep") == 0) o->keep_n = atol(val);
45 else if (strcmp(flag, "--keep-since") == 0) o->keep_since = atol(val);
46 else if (strcmp(flag, "--grace") == 0) o->grace_seconds = atol(val);
47 else return -1;
48 idx += 2;
49 }
50 return 0;
51}

54 /* argv: repo gc <repo_dir> [keys_dir] flags... */
55 if (argc < 3) return ad_usage();
56 cvmfs_gc_opts_t o;
57 memset(&o, 0, sizeof(o));
58 o.repo_dir = argv[2];

60 int idx = 0;
