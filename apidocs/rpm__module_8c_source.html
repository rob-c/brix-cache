1/*
2 * rpm_module.c — nginx wiring for the RPM/dnf mirror plane.
3 *
4 * WHAT: the config lifecycle head (create_loc_conf + the UNSET sentinels the
5 * merge reads), the one directive setter that cannot be a stock slot
6 * writer, the `$rpm_class` log variable, the SHM-zone
7 * postconfiguration, the directive table and the module record.
8 * WHY: the shape is deliberately the OCI module's, because the lifecycle
9 * question is the same one: a location becomes a mirror endpoint the
10 * moment its enabling directive is parsed, and everything else — the
11 * metrics zone, the dashboard zones, the fill thread pool — must exist
12 * for that endpoint whether or not a stream{} block was ever configured.
13 * HOW: create sets sentinels and nothing else (the merge owns every default),
14 * the merge lives in rpm_merge.c, and $rpm_class reads the request ctx
15 * the handler fills. The disposition ($rpm_cache) moved to the shared
16 * $brix_cache_status in phase 112; its five-value vocabulary survives as
17 * the brix_rpm_requests_total{outcome} labels, so the access log and the
18 * metric family still cannot disagree.
19 */

45/* "brix_rpm_mirror <base-url>" — record the upstream AND make this location
46 * the mirror endpoint. The URL is only stored here; parsing (and every refusal
47 * that depends on it) happens at merge time, when inheritance has settled. */
48static char *
49rpm_conf_mirror(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
50{
51 ngx_http_brix_rpm_loc_conf_t *lcf = conf;
52 ngx_http_core_loc_conf_t *clcf;
53 ngx_str_t *value = cf->args->elts;
54
55 (void) cmd;
56 if (lcf->mirror != NGX_CONF_UNSET) {
57 return "is duplicate";
58 }
59
60 lcf->mirror = 1;
61 lcf->mirror_url = value[1];
62
63 clcf = ngx_http_conf_get_module_loc_conf(cf, ngx_http_core_module);
64 clcf->handler = ngx_http_brix_rpm_handler;
65
66 return NGX_CONF_OK;
67}

72static ngx_int_t
73rpm_var_set(ngx_http_request_t *r, ngx_http_variable_value_t *v,
74 const char *val)
75{
76 size_t n = ngx_strlen(val);
77 u_char *p = ngx_pnalloc(r->pool, n);
78
79 if (p == NULL) {
80 return NGX_ERROR;
81 }
82 ngx_memcpy(p, val, n);
83 v->data = p;
84 v->len = n;
85 v->valid = 1;
86 v->no_cacheable = 0;
87 v->not_found = 0;
88 return NGX_OK;
89}

93rpm_var_class(ngx_http_request_t *r, ngx_http_variable_value_t *v,
94 uintptr_t data)
95{
96 ngx_http_brix_rpm_ctx_t *ctx =

123static ngx_int_t
124ngx_http_brix_rpm_preconfiguration(ngx_conf_t *cf)
125{
126 ngx_http_variable_t *v, *nv;
127
128 for (v = ngx_http_brix_rpm_vars; v->name.len; v++) {
129 nv = ngx_http_add_variable(cf, &v->name, v->flags);
130 if (nv == NULL) {
131 return NGX_ERROR;
132 }
133 nv->get_handler = v->get_handler;
134 nv->data = v->data;
135 }
136 return NGX_OK;
137}
