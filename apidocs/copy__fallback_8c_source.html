10/* ---- Seek an explicitly nonzero copy offset ----
11 * WHAT: Return zero or the CRT seek failure.
12 * WHY: Both sides retain the existing optional-offset contract.
13 * HOW: 1. Leave absent/zero offsets alone. 2. Seek an explicit offset.
14 */
15static int
16brix_win32_seek_copy_offset(int fd, const off_t *offset)
17{
18 if (offset == NULL || *offset == 0) {
19 return 0;
20 }
21 return _lseeki64(fd, *offset, SEEK_SET) < 0 ? -1 : 0;
22}

24/* ---- Write one copy buffer with the existing interrupt retry ----
25 * WHAT: Return the CRT write result, retrying one interrupted write.
26 * WHY: Keep write-error handling separate from read/offset accounting.
27 * HOW: 1. Write the buffer. 2. Retry once only when interrupted.
28 */
29static ssize_t
30brix_win32_write_copy_buffer(int fd, const char *buffer, size_t length)
31{
32 ssize_t written = _write(fd, buffer, (unsigned int)length);
33 if (written < 0 && errno == EINTR) {
34 return _write(fd, buffer, (unsigned int)length);
35 }
36 return written;
37}

39 * brix_win32_buffered_copy - Fallback buffered copy implementation
40 *
41 * @in_fd: Input file descriptor
42 * @in_off: Input offset (or NULL)
43 * @out_fd: Output file descriptor
44 * @out_off: Output offset (or NULL)
45 * @len: Bytes to copy
46 * @return: Bytes copied on success, -1 on error
47 *
48 * This is the universal fallback when CopyFile2 and FSCTL_COPY_FILE_RANGE
49 * are not available. Uses a 64KB buffer for reasonable performance.
50 */
51ssize_t
52brix_win32_buffered_copy(int in_fd, off_t *in_off,
53 int out_fd, off_t *out_off,
54 size_t len)
55{
56 char buffer[65536]; /* 64KB buffer */
57 size_t remaining = len;
58 ssize_t total_copied = 0;
59
60 /* Seek to input offset if provided */
61 if (brix_win32_seek_copy_offset(in_fd, in_off) < 0) {
62 return -1;
63 }
64
65 /* Seek to output offset if provided */
66 if (brix_win32_seek_copy_offset(out_fd, out_off) < 0) {
67 return -1;
68 }
69
70 while (remaining > 0) {
71 size_t to_read = (remaining > sizeof(buffer)) ? sizeof(buffer) : remaining;
72 ssize_t bytes_read = _read(in_fd, buffer, (unsigned int)to_read);
73
74 if (bytes_read < 0) {
75 if (errno == EINTR) {
76 continue; /* Retry on interrupt */
77 }
78 return (total_copied > 0) ? total_copied : -1;
79 }
80
81 if (bytes_read == 0) {
82 /* EOF reached */
83 break;
84 }
85
86 ssize_t bytes_written = brix_win32_write_copy_buffer(
87 out_fd, buffer, (size_t)bytes_read);
88 if (bytes_written < 0) {
89 return (total_copied > 0) ? total_copied : -1;
90 }
91
92 total_copied += bytes_written;
93 remaining -= bytes_read;
94 }
95
96 /* Update offsets if provided */
97 if (in_off != NULL) {
98 *in_off += total_copied;
99 }
100 if (out_off != NULL) {
101 *out_off += total_copied;
102 }
103
104 return total_copied;
105}
