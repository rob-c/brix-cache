407 * brix_cache_dirlist_t — heap-owned result of a kXR_dirlist fetch: a growable
408 * array of heap-allocated (malloc/strdup) NUL-terminated entry names.
409 *
410 * WHAT: `names` is a malloc'd array of `count` char* (each entry itself
411 * malloc'd), with `cap` tracking the current array capacity for
412 * amortised-doubling growth.
413 * WHY: sd_xroot's opendir fetches the WHOLE listing once (fetch-all-then-
414 * iterate, matching the rest of the origin-session ops in this file,
415 * which never hold a session open across VFS calls) and readdir then
416 * yields one name per call from this buffer; closedir frees it. A
417 * plain array of owned strings (rather than one flat buffer + offsets)
418 * keeps the free path trivial and bounds each name independently.
419 * HOW: brix_cache_dirlist_init zeroes the struct; brix_cache_dirlist_add
420 * appends one strdup'd name (growing the array as needed); the caller
421 * (sd_xroot_ns.c) reads names[i] directly and calls
422 * brix_cache_dirlist_free exactly once when done.
423 */
424typedef struct {
425 char **names; /* malloc'd array of malloc'd NUL-terminated names */
426 size_t count; /* number of valid entries in names[] */
427 size_t cap; /* allocated capacity of the names[] array */
428} brix_cache_dirlist_t;

431void brix_cache_dirlist_init(brix_cache_dirlist_t *dl);
432/* Free every entry name plus the array itself; resets *dl to empty so a
433 * double-free is harmless (idempotent). NULL-safe. */
434void brix_cache_dirlist_free(brix_cache_dirlist_t *dl);

436/* brix_cache_origin_dirlist — kXR_dirlist <path> against the origin (plain

441 * (VFS readdir contract never yields them). Returns 0 on success (dl
