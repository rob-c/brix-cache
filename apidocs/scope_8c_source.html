13/* ---- Test an exact token inside a comma-separated selector ----
14 *
15 * WHAT: Returns one when list contains value as a whole token.
16 * WHY: Family selection must not confuse RD with RDV or a substring.
17 * HOW: Walk comma boundaries and compare equal-length spans.
18 */
19static int
20selector_has(const char *list, const char *value)
21{
22 const char *cursor;
23 size_t value_len;
24
25 if (list == NULL || list[0] == '\0') return 0;
26 value_len = strlen(value);
27 cursor = list;
28 while (*cursor != '\0') {
29 const char *end = strchr(cursor, ',');
30 size_t token_len = end == NULL ? strlen(cursor) : (size_t) (end - cursor);
31
32 if (token_len == value_len && strncmp(cursor, value, value_len) == 0) {
33 return 1;
34 }
35 if (end == NULL) return 0;
36 cursor = end + 1;
37 }
38 return 0;
39}

41/* ---- Determine whether a role profile includes a family ----
42 *
43 * WHAT: Returns one for families belonging to profile, zero otherwise.
44 * WHY: An operator's declared endpoint role is an independent scope boundary.
45 * HOW: Use explicit comma-separated family sets mirroring Phase 118 section 31.
46 */
47static int
48profile_has_family(const char *profile, const char *family)
49{
50 static const char data[] =
51 "FRM,HSK,PROT,LOG,SIG,OPN,RD,PGR,RDV,WRT,PGW,WVC,SYN,TRN,CKP,STA,DIR,"
52 "MKD,MV,RM,CHM,LOC,FAT,PRP,QRY,ERR,TLS,CLM,CHU,ENV";
53 static const char manager[] =
54 "FRM,HSK,PROT,LOG,SIG,LOC,QRY,RDR,WAI,ERR,TLS,CLM,CMS";
55 static const char http[] = "HTR,HTC,HTE,DAV,XPR,ERR,TLS";
56 static const char s3[] = "HTR,HTC,HTE,S3,XPR,ERR,TLS";
57 static const char tls[] = "TLS,HSK,PROT,SIG";
58 static const char tpc[] = "TPC,LOC,ERR,TLS";
59
60 struct role_set { const char *name; const char *families; };
61 const struct role_set roles[] = {
62 { "data", data }, { "tape", data },
63 { "manager", manager }, { "http", http }, { "s3", s3 },
64 { "tls", tls }, { "tpc", tpc }
65 };
66 size_t index;
67
68 if (profile == NULL || selector_has(profile, "full")) return 1;
69 for (index = 0; index < sizeof(roles) / sizeof(roles[0]); index++) {
70 if (selector_has(profile, roles[index].name)
71 && selector_has(roles[index].families, family)) return 1;
72 }
73 if (selector_has(profile, "cache")
74 && (selector_has(data, family) || strcmp(family, "CAC") == 0
75 || strcmp(family, "RDR") == 0)) return 1;
76 if (selector_has(profile, "proxy")
77 && (selector_has(manager, family) || strcmp(family, "RD") == 0
78 || strcmp(family, "PGR") == 0)) return 1;
79 return 0;
80}

85/* ---- Enforce mutation authority for direct and adversarial probes ----
86 *
87 * WHAT: Returns one when no mutation gate applies or all mutation gates pass.
88 * WHY: ADV is disruption-classed but its corpus can also mutate remote state.
89 * HOW: Treat the ADV family like a mutate-class row and emit the common reason.
90 */
91static int
92mutation_authorized(const brix_conform_clause *clause,
93 const brix_conform_run_config *config, char *reason, size_t reason_size)
94{
95 int required = clause->probe_class == BRIX_CONFORM_PROBE_MUTATE
96 || strcmp(clause->family, "ADV") == 0;
97
98 if (!required || (config->allow_mutate && config->scratch != NULL)) return 1;
99 return scope_skip(reason, reason_size, "%s",
100 config->scratch == NULL ? "no-scratch-prefix"
101 : "probe-class-disabled:mutate");
102}

104/* ---- Apply selectors and role profile ----
105 *
106 * WHAT: Returns one when a row survives all non-authority filters.
107 * WHY: Separating filters keeps the public selection gate below CCN limits.
108 * HOW: Apply exact id, family include/exclude and profile in frozen order.
109 */
110static int
111scope_filters(const brix_conform_clause *clause,
112 const brix_conform_run_config *config, char *reason, size_t reason_size)
113{
114 if (config->only != NULL && strcmp(config->only, clause->id) != 0)
115 return scope_skip(reason, reason_size, "filtered:not-%s", config->only);
116 if (config->families != NULL
117 && !selector_has(config->families, clause->family))
118 return scope_skip(reason, reason_size, "family-excluded:%s",
119 clause->family);
120 if (config->exclude != NULL
121 && (selector_has(config->exclude, clause->family)
122 || selector_has(config->exclude, clause->id)))
123 return scope_skip(reason, reason_size, "explicitly-excluded:%s",
124 clause->id);
125 if (!profile_has_family(config->profile, clause->family))
126 return scope_skip(reason, reason_size, "profile-excluded:%s",
127 config->profile);
128 return 1;
129}

147/* ---- Decide whether a clause is in scope and authorized ----
148 *
149 * WHAT: Returns one to run or zero with an explicit skip reason.
150 * WHY: Profiles and safety flags may narrow a run but must never silently omit.
151 * HOW: Apply exact-id, include/exclude family, profile and probe-class gates.
152 */
153int
154brix_conform_clause_selected(const brix_conform_clause *clause,
155 const brix_conform_run_config *config, char *reason, size_t reason_size)
156{
157 if (clause == NULL || config == NULL) {
158 return scope_skip(reason, reason_size, "%s", "invalid-clause");
159 }
160 if (!scope_filters(clause, config, reason, reason_size)) return 0;
161 if (!mutation_authorized(clause, config, reason, reason_size)) return 0;
162 if (clause->probe_class == BRIX_CONFORM_PROBE_DISRUPT
163 && (!config->allow_disrupt || config->confirm_target == NULL
164 || strcmp(config->confirm_target, config->target) != 0)) {
165 return scope_skip(reason, reason_size, "%s",
166 "probe-class-disabled:disrupt");
167 }
168 if (reason != NULL && reason_size > 0) reason[0] = '\0';
169 return 1;
170}
