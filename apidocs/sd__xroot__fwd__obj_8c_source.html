1/*
2 * sd_xroot_fwd_obj.c — object / dir / staged relays of the forwarding driver.
3 *
4 * WHAT: The handle-keyed slots. A handle opened through the forwarding driver
5 * belongs to an sd_xroot CHILD (obj->inst / obj->driver, d->inst,
6 * st->inst are the child's), so these slots are reached only by the
7 * seams that dispatch through the EXPORT's instance rather than the
8 * handle's own driver (vfs_staged.c's ctx->sd->driver->staged_write,
9 * the close/pread fallbacks in vfs_open_handle.c, ...).
10 * WHY: Those seams must work identically whether the export is a fixed
11 * root:// origin or a forwarding one — declaring the slots keeps every
12 * `inst->driver->X != NULL` capability probe true exactly where
13 * sd_xroot's is.
14 * HOW: Forward to the handle's own driver through sd_xroot_fwd_relay_driver,
15 * which refuses (ENOSYS) a NULL or self-referential driver so nothing can
16 * recurse; a child without the slot answers ENOSYS too.
17 */
18#include "sd_xroot_fwd_internal.h"

20#include <errno.h>
