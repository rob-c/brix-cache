18/* File: push_stream.c — F16 push, the byte mover.
19 *
20 * WHAT: tpc_push_stream() streams the whole local file (t->dst_fd / t->dst_obj,
21 * open READ-ONLY) to an already-opened remote handle at the destination, in
22 * rounds of `nslots` TPC_CHUNK_SIZE windows at consecutive offsets, where
23 * nslots = 1 primary + t->nsub bound sub-streams.
24 *
25 * WHY: a push cannot reuse the pull's loop even though the shape rhymes. The
26 * pull's slot arithmetic (tpc_stream_plan_round) exists because a READ reply is
27 * ambiguous — a short frame may mean EOF, a partial window, or a stopped source,
28 * and only the source knows which. A WRITE has no such ambiguity: the peer
29 * answers kXR_ok or it fails. What IS ambiguous on this side is the local read,
30 * so the completion signal is t->push_size (the size captured on the event loop
31 * from the handle that authorized this transfer) and a local short read before
32 * that offset is a hard error, not an EOF.
33 *
34 * HOW (wire): ClientWriteRequest is 24 bytes — streamid[2], requestid,
35 * fhandle[4], int64 offset, pathid, reserved[3], int32 dlen — header then the
36 * payload bytes. We tag streamid[1]=3 (the data tag, as the pull's reads do) and
37 * send pathid 0 on every frame: a server that accepted our kXR_bind treats a
38 * whole kXR_write arriving on a BOUND socket as belonging to that socket's
39 * session and never inspects the pathid byte, whereas a non-zero pathid on the
40 * PRIMARY connection is refused ("send data inline"). So the sub-stream carries
41 * the frame; the pathid stays 0.
42 *
43 * Each round: read nslots local chunks at consecutive offsets → send one write
44 * frame per slot on that slot's socket → drain one ack per slot → advance. Any
45 * failed ack fails the whole push: a hole in a destination file is never
46 * acceptable, and there is no partial-success outcome to report.
47 */
48
49/* The local read+send buffer is one allocation of TPC_CHUNK_SIZE * nslots. At
50 * the TPC_STREAMS_MAX ceiling that is 15 MiB, which is an operator-chosen
51 * maximum (brix_tpc_streams defaults to 1 = 1 MiB). */
52typedef struct {
53 u_char *buf; /* nslots * TPC_CHUNK_SIZE */
54 size_t got[TPC_STREAMS_MAX]; /* bytes read for each slot this round */
55 int nslots;
56 uint64_t offset; /* file offset of slot 0 this round */
57} tpc_push_round_t;

67/*
68 * WHAT: fill this round's buffer from the local file through the VFS seam.
69 * WHY: INVARIANT 12 — the local file may be a block-striped or object-store
70 * export, not a POSIX fd, so the bytes must come through brix_vfs_* and the
71 * handle's copied sd_obj, never a bare pread. A short read BEFORE push_size is
72 * a truncated source (someone raced us) and must fail rather than silently
73 * publish a short file at the destination.
74 * HOW: one read job per slot at offset + slot*TPC_CHUNK_SIZE, clamped to the
75 * bytes remaining before push_size. Returns 0 with r->got[] filled, or -1 with
76 * t->err_msg / t->xrd_error set.
77 */
78static int
79tpc_push_fill_round(brix_tpc_pull_t *t, tpc_push_round_t *r)
80{
81 int slot;
82
83 for (slot = 0; slot < r->nslots; slot++) {
84 brix_vfs_job_t job;
85 uint64_t at = r->offset + (uint64_t) slot * TPC_CHUNK_SIZE;
86 uint64_t left;
87 size_t want;
88
89 r->got[slot] = 0;
90 if (at >= t->push_size) {
91 continue; /* past EOF: this slot idles */
92 }
93
94 left = t->push_size - at;
95 want = (left < (uint64_t) TPC_CHUNK_SIZE) ? (size_t) left
96 : (size_t) TPC_CHUNK_SIZE;
97
98 brix_vfs_job_read_init(&job, t->dst_fd, (off_t) at, want,
99 r->buf + (size_t) slot * TPC_CHUNK_SIZE,
100 want, 0);
101 brix_vfs_job_set_obj(&job, &t->dst_obj);
102 brix_vfs_io_execute(&job);
103
104 if (job.io_errno != 0 || job.nio < 0) {
105 int err = job.io_errno != 0 ? job.io_errno : EIO;
106
107 snprintf(t->err_msg, sizeof(t->err_msg),
108 "TPC push source read failed: %s", strerror(err));
109 t->xrd_error = kXR_IOError;
110 return -1;
111 }
112 if ((size_t) job.nio != want) {
113 snprintf(t->err_msg, sizeof(t->err_msg),
114 "TPC push source shrank at offset %llu",
115 (unsigned long long) at);
116 t->xrd_error = kXR_IOError;
117 return -1;
118 }
119 r->got[slot] = want;
120 }
121
122 return 0;
123}

125/*
126 * WHAT: send one kXR_write frame (header + payload) for one slot.
127 * WHY: kept separate so the header layout is stated once and the send loop
128 * stays flat.
129 * HOW: 24-byte ClientWriteRequest, offset in network byte order via brix_plat_htobe64,
130 * dlen counting only the payload, pathid 0 (see the file docblock).
131 */
132static int
133tpc_push_send_slot(brix_tpc_pull_t *t, int fd, const u_char *fhandle,
134 uint64_t at, const u_char *data, size_t len)
135{
136 ClientWriteRequest wreq;
137
138 xrd_creq_begin(&wreq, sizeof(wreq), 3, kXR_write);
139 ngx_memcpy(wreq.fhandle, fhandle, XRD_FHANDLE_LEN);
140 wreq.offset = (kXR_int64) brix_plat_htobe64(at);
141 wreq.pathid = 0;

143

145 || tpc_send_all(t, fd, data, len) != 0)

148 "TPC push kXR_write send failed at offset %llu",
149 (unsigned long long) at);
150 t->xrd_error = kXR_ServerError;
151 return -1;
152 }

154 return 0;

157/*
158 * WHAT: read one kXR_write acknowledgement off a slot's socket.
159 * WHY: a write is only delivered once the destination says so; treating an
160 * unread ack as success would let a refused write (quota, EROFS, a revoked
161 * handle) pass as a completed copy.
162 * HOW: one response frame; kXR_ok is the only accepted status, and a kXR_error
163 * body is decoded so the client sees the destination's real reason.
164 */
165static int
166tpc_push_recv_ack(brix_tpc_pull_t *t, int fd, uint64_t at)
167{
168 uint16_t status = 0;
169 uint32_t dlen = 0;
170 u_char *body = NULL;
171 int code = kXR_ServerError;
172 const char *msg = "";
173 size_t msglen = 0;
174
175 if (tpc_recv_response(t, fd, &status, &body, &dlen) != 0) {
176 snprintf(t->err_msg, sizeof(t->err_msg),
177 "TPC push write reply failed at offset %llu",
178 (unsigned long long) at);
179 t->xrd_error = kXR_ServerError;
180 return -1;
181 }
182
183 if (status == kXR_ok) {
184 free(body);
185 return 0;
186 }
187
188 if (status == kXR_error) {
189 (void) xrd_error_body_decode(body, dlen, &code, &msg, &msglen);
190 }
191
192 /* `msg` is a bounded slice INTO body — format before the free. */
193 snprintf(t->err_msg, sizeof(t->err_msg),
194 "TPC push destination refused write at offset %llu: %.*s",
195 (unsigned long long) at, (int) msglen, msg);
196 free(body);
197
198 t->xrd_error = code;
199 return -1;
200}

202/*
203 * WHAT: send every filled slot of one round, then drain every ack.
204 * WHY: sending all slots before reading any ack is what makes the extra streams
205 * worth having — nslots windows are in flight at once. Draining in the same slot
206 * order keeps each ack matched to the socket that carries it.
207 * HOW: two passes over the slots; the first failure in either wins and the
208 * round fails. A failed send still drains the acks for slots already sent, so
209 * the sockets are not left holding an unread frame before the close ladder.
210 */
211static int
212tpc_push_exchange_round(brix_tpc_pull_t *t, int primary_fd,
213 const u_char *fhandle, tpc_push_round_t *r)
214{
215 int rc = 0;
216 int sent = 0;
217 int slot;
218
219 for (slot = 0; slot < r->nslots && r->got[slot] > 0; slot++) {
220 uint64_t at = r->offset + (uint64_t) slot * TPC_CHUNK_SIZE;
221
222 if (tpc_push_send_slot(t, tpc_push_slot_fd(t, primary_fd, slot),
223 fhandle, at,
224 r->buf + (size_t) slot * TPC_CHUNK_SIZE,
225 r->got[slot]) != 0)
226 {
227 rc = -1;
228 break;
229 }
230 sent++;
231 }
232
233 for (slot = 0; slot < sent; slot++) {
234 uint64_t at = r->offset + (uint64_t) slot * TPC_CHUNK_SIZE;
235
236 if (tpc_push_recv_ack(t, tpc_push_slot_fd(t, primary_fd, slot), at)
237 != 0)
238 {
239 rc = -1; /* keep draining: never leave a stray frame */
240 } else if (rc == 0) {
241 t->bytes_written += r->got[slot];
242 }
243 }
244
245 return rc;
246}

248/*
249 * WHAT: how many bytes this round proved, and whether the file is exhausted.
250 * WHY: with writes there is no short-frame ambiguity to judge, so this is plain
251 * arithmetic over what we chose to read — but it stays a named step so the
252 * "advance" and "done" decisions are visible rather than inlined in the loop.
253 */
254static size_t
255tpc_push_round_advance(const tpc_push_round_t *r)
256{
257 size_t total = 0;
258 int slot;
259
260 for (slot = 0; slot < r->nslots; slot++) {
261 total += r->got[slot];
262 }
263 return total;
264}

266/*
267 * tpc_push_stream — stream the local file out to the remote handle.
268 * Returns 0 with t->result = NGX_OK once every byte through t->push_size has
269 * been acknowledged, or -1 with t->err_msg / t->xrd_error set.
270 */
271int
272tpc_push_stream(brix_tpc_pull_t *t, int fd, const u_char *fhandle)
273{
274 tpc_push_round_t r;
275 time_t started = time(NULL);
276 int rc = 0;
277
278 ngx_memzero(&r, sizeof(r));
279 r.nslots = 1 + t->nsub;
280 if (r.nslots < 1) {
281 r.nslots = 1;
282 }
283 if (r.nslots > TPC_STREAMS_MAX) {
284 r.nslots = TPC_STREAMS_MAX;
285 }
286
287 r.buf = malloc((size_t) r.nslots * TPC_CHUNK_SIZE);
288 if (r.buf == NULL) {
289 snprintf(t->err_msg, sizeof(t->err_msg),
290 "TPC push buffer allocation failed");
291 t->xrd_error = kXR_NoMemory;
292 return -1;
293 }
294
295 while (r.offset < t->push_size) {
296 size_t advance;
297
298 if (tpc_stream_check_deadline(t, started, r.offset) != 0) {
299 rc = -1;
300 break;
301 }
302 if (tpc_push_fill_round(t, &r) != 0) {
303 rc = -1;
304 break;
305 }
306 if (tpc_push_exchange_round(t, fd, fhandle, &r) != 0) {
307 rc = -1;
308 break;
309 }
310
311 advance = tpc_push_round_advance(&r);
312 if (advance == 0) {
313 /* Unreachable while offset < push_size (fill_round always reads at
314 * least one byte for slot 0), but a zero advance would spin
315 * forever — fail closed instead. */
316 snprintf(t->err_msg, sizeof(t->err_msg),
317 "TPC push made no progress at offset %llu",
318 (unsigned long long) r.offset);
319 t->xrd_error = kXR_IOError;
320 rc = -1;
321 break;
322 }
323 r.offset += advance;
324 }
325
326 free(r.buf);
327
328 if (rc != 0) {
329 t->result = NGX_ERROR;
330 return -1;
331 }
332
333 /*
334 * Completion gate: every byte the handle claimed must have been
335 * acknowledged. bytes_written counts only ACKED payload, so this rejects a
336 * transfer whose acks went missing as firmly as one that stopped early.
337 */
338 if ((uint64_t) t->bytes_written != t->push_size) {
339 snprintf(t->err_msg, sizeof(t->err_msg),
340 "TPC push incomplete: %llu of %llu bytes acknowledged",
341 (unsigned long long) t->bytes_written,
342 (unsigned long long) t->push_size);
343 t->xrd_error = kXR_IOError;
344 t->result = NGX_ERROR;
345 return -1;
346 }
347
348 t->result = NGX_OK;
349 t->xrd_error = 0;
350 return 0;
351}
