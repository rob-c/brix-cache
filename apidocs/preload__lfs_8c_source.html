1/*
2 * client/lib/platform/linux/preload_lfs.c - the glibc-only entry points
3 *
4 * WHAT: The LFS *64 spellings (open64, stat64, readdir64, fopen64, ...) and
5 * statx, forwarded to the shim's portable wrappers.
6 * WHY: Tools built with _FILE_OFFSET_BITS=64 (coreutils, python) call the
7 * *64 names, and modern coreutils (ls, stat, find, du) call statx; left
8 * uninterposed they would stat the real (absent) local path and
9 * ENOENT. None of these names exists on Darwin, which is why this
10 * is a host file rather than part of the portable shim.
11 * HOW: Each forwarder calls the wrapper by its libc name, which under
12 * LD_PRELOAD is the shim's own definition. On LP64 glibc the *64
13 * structures are layout-identical to the plain ones (asserted), so the
14 * forward is a cast, not a copy.
15 */

138/*
139 * statx() answers the common fields (type/mode/nlink/size/times/ino) for a
140 * remote path from the shim's own stat(), so statx and stat can never
141 * disagree; the caller's `mask` is satisfied for what XRootD can report.
142 */
143int
144statx(int dirfd, const char *path, int flags, unsigned int mask,
145 struct statx *stxbuf)
146{
147 char remote[XRDC_PATH_MAX];
148 struct stat mapped;
149 REAL(statx);
150
151 if (path[0] != '/' || !map_path(path, remote, sizeof(remote))) {
152 return real_statx(dirfd, path, flags, mask, stxbuf);
153 }
154 if (stat(path, &mapped) != 0) {
155 return -1;
156 }
157 memset(stxbuf, 0, sizeof(*stxbuf));
158 stxbuf->stx_mask = STATX_TYPE | STATX_MODE | STATX_NLINK | STATX_SIZE
159 | STATX_MTIME | STATX_INO;
160 stxbuf->stx_mode = (uint16_t) mapped.st_mode;
161 stxbuf->stx_nlink = (uint32_t) mapped.st_nlink;
162 stxbuf->stx_size = (uint64_t) mapped.st_size;
163 stxbuf->stx_ino = (uint64_t) mapped.st_ino;
164 stxbuf->stx_blksize = (uint32_t) mapped.st_blksize;
165 stxbuf->stx_blocks = (uint64_t) mapped.st_blocks;
166 stxbuf->stx_mtime.tv_sec = mapped.st_mtime;
167 stxbuf->stx_atime.tv_sec = mapped.st_atime;
168 stxbuf->stx_ctime.tv_sec = mapped.st_ctime;
169 return 0;
170}
