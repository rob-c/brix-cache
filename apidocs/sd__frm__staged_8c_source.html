29brix_sd_staged_t *
30sd_frm_staged_open(brix_sd_instance_t *inst, const char *final_path,
31 mode_t mode, off_t declared_size, int *err_out)
32{
33 sd_frm_state *st = SD_FRM_ST(inst);
34 sd_frm_staged_state *ss;
35 brix_sd_staged_t *h;
36 int fd;
37
38 fd = st->mss->create_online(st->mss_ctx, final_path, mode);
39 if (fd < 0) {
40 if (err_out) { *err_out = errno ? errno : EIO; }
41 return NULL;
42 }
43
44 /* Phase-107 C5: the ONLINE BUFFER is the disk-resident half this driver
45 * owns — preallocate the declared final size there so a buffer that cannot
46 * hold the object refuses the OPEN (ENOSPC/EDQUOT), not the migrate after
47 * hours of streaming. Tape-side reservation stays the MSS tier's own
48 * scheduling concern. Anything short of "no space" is advisory. */
49 if (declared_size > 0) {
50 brix_sd_obj_t shell;
51
52 ngx_memzero(&shell, sizeof(shell));
53 shell.fd = fd;
54 if (sd_posix_reserve(&shell, declared_size) != NGX_OK
55 && (errno == ENOSPC || errno == EDQUOT))
56 {
57 int err = errno;
58
59 (void) close(fd);
60 (void) st->mss->purge(st->mss_ctx, final_path);
61 if (err_out) { *err_out = err; }
62 errno = err;
63 return NULL;
64 }
65 }
66
67 ss = calloc(1, sizeof(*ss));
68 h = calloc(1, sizeof(*h));
69 if (ss == NULL || h == NULL) {
70 (void) close(fd);
71 free(ss);
72 free(h);
73 if (err_out) { *err_out = ENOMEM; }
74 return NULL;
75 }
76 ss->fst = st;
77 ss->fd = fd;
78 ngx_cpystrn((u_char *) ss->key, (u_char *) final_path, sizeof(ss->key));
79 h->inst = inst;
80 h->state = ss;
81 return h;
82}

92ngx_int_t
93sd_frm_staged_commit(brix_sd_staged_t *st, brix_sd_precond_t *pre)
94{
95 sd_frm_staged_state *ss = st->state;
96 int rc;
97
98 /* Typed publish precondition (phase-107 C6): the MSS adapter's residency
99 * probe is this driver's only view of the tape-side target, so every kind
100 * is ADVISORY here — check-then-migrate, pre->atomic stays 0 (the W7
101 * checklist's verdict for frm; §3.5 forbids claiming atomicity the MSS
102 * never gave). Evaluated BEFORE the migrate so a refused precondition
103 * leaves the tape copy untouched (the caller's staged_abort then purges
104 * the online-buffer temp). */
105 if (pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE) {
106 off_t tsz = 0;
107 time_t tmt = 0;
108 int res = ss->fst->mss->residency(ss->fst->mss_ctx, ss->key,
109 &tsz, &tmt);
110
111 if (pre->kind == BRIX_SD_PRECOND_ABSENT) {
112 if (res != BRIX_RESIDENCY_ABSENT) {
113 errno = EEXIST;
114 return NGX_ERROR;
115 }
116 } else if (res == BRIX_RESIDENCY_ABSENT) {
117 errno = ECANCELED; /* MATCH_* against a missing target */
118 return NGX_ERROR;
119 } else if (brix_sd_precond_eval_stat(pre, tsz, tmt) != 0) {
120 return NGX_ERROR; /* errno = ECANCELED / ENOTSUP */
121 }
122 }
123
124 if (ss->fd >= 0) {
125 (void) close(ss->fd);
126 ss->fd = -1;
127 }
128 /* Publish: migrate the online-buffer object to tape. */
129 rc = ss->fst->mss->migrate(ss->fst->mss_ctx, ss->key);
130 if (rc != 0) {
131 /* Ownership contract: only a SUCCESSFUL commit consumes the handle. A
132 * failed migrate must leave st+ss valid — every caller aborts a failed
133 * commit (stage_engine, cstb_pump_and_commit, cache fetch), and abort
134 * frees them. Freeing here made that mandatory abort a use-after-free,
135 * a double free, and a second purge of the online buffer. */
136 return NGX_ERROR;
137 }
138 free(ss);
139 free(st);
140 return NGX_OK;
141}

158/* Atomic two-name exchange (phase-107 C6): swap the ONLINE-BUFFER copies via
159 * the adapter's exchange verb, then MIGRATE both keys so tape truth catches
160 * up — an un-migrated swap would serve OLD content after a purge + recall.
161 * Both names must be online (the verb answers ENOENT otherwise; the caller
162 * recalls first); an adapter without the verb (stub) refuses ENOTSUP, never
163 * a two-rename emulation (§3.5). Failure discipline mirrors the C3 barrier:
164 * a first-migrate failure un-exchanges (nothing shipped, state fully
165 * restored); a second-migrate failure cannot be unwound — `a` already
166 * shipped — so it logs at crit and FAILS rather than claim tape consistency
167 * the MSS does not have (the vfs_rename.c durable-publish doctrine). */
168ngx_int_t
169sd_frm_exchange(brix_sd_instance_t *inst, const char *a, const char *b)
170{
171 sd_frm_state *st = SD_FRM_ST(inst);
172 int err;
173
174 if (st->mss->exchange == NULL) {
175 errno = ENOTSUP;
176 return NGX_ERROR;
177 }
178 if (st->mss->exchange(st->mss_ctx, a, b) != 0) {
179 return NGX_ERROR; /* errno from the verb */
180 }
181 if (st->mss->migrate(st->mss_ctx, a) != 0) {
182 err = errno ? errno : EIO;
183 (void) st->mss->exchange(st->mss_ctx, a, b);
184 errno = err;
185 return NGX_ERROR;
186 }
187 if (st->mss->migrate(st->mss_ctx, b) != 0) {
188 err = errno ? errno : EIO;
189 ngx_log_error(NGX_LOG_CRIT, st->log, err,
190 "brix: frm exchange: tape catch-up migrate failed for "
191 "\"%s\" after \"%s\" shipped", b, a);
192 errno = err;
193 return NGX_ERROR;
194 }
195 return NGX_OK;
196}

199/* Durable-publish barrier (phase-107 C3): the frm publish lands in the LOCAL
200 * POSIX online buffer before migrate ships it to the MSS — flush that entry's
201 * parent directory through the adapter's sync_publish verb so the published
202 * name survives a crash of this host. Tape-side durability is the MSS's own
203 * contract; an adapter without the verb has nothing local to flush. */
204ngx_int_t
205sd_frm_sync_publish(brix_sd_instance_t *inst, const char *path)
206{
207 sd_frm_state *st = SD_FRM_ST(inst);
208
209 if (st->mss->sync_publish == NULL) {
210 return NGX_OK;
211 }
212 return (st->mss->sync_publish(st->mss_ctx, path) == 0) ? NGX_OK : NGX_ERROR;
213}
