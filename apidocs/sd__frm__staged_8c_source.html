32brix_sd_frm_seal(brix_sd_instance_t *inst, const char *key)
33{
34 sd_frm_state *st;

36
37 if (!brix_sd_frm_instance_is(inst) || key == NULL) {
38 errno = EINVAL;

56/* migrate() answered BRIX_MSS_MIGRATE_DEFERRED: `key` is a dataset completion
57 * marker and sealing it (compose + ship the archive) is minutes of tape work
58 * that must not run on the client's close. Hand it to the durable stage
59 * engine (kind archive: journaled, retried on the brix_frm_fail_backoff
60 * sweep, replayed after a restart, dead-lettered at brix_frm_fail_retries)
61 * when this is the event loop and a journal is configured. Without a
62 * journal, or from the engine's own mover thread (a FLUSH whose destination
63 * is this tier -- the submit FIFO is the loop's), seal inline right here. */
64static ngx_int_t
65frm_seal_defer(sd_frm_state *fst, brix_sd_instance_t *inst, const char *key)
66{
67 brix_stage_opts_t opts;
68
69 if (brix_stage_engine_journal_dir()[0] == '\0' || !brix_stage_on_loop()) {
70 return brix_sd_frm_seal(inst, key);
71 }
72 ngx_memzero(&opts, sizeof(opts));
73 opts.async = 1;
74 opts.export_root = fst->export_root;
75 if (brix_stage_submit(BRIX_STAGE_ARCHIVE, inst, key, inst, key, &opts)
76 == NULL)
77 {
78 return brix_sd_frm_seal(inst, key); /* refused the handle: inline */
79 }
80 return NGX_OK; /* queued (or ran inline) */
81}

83/* Publish `key` from the online buffer to tape: the adapter's migrate ships
84 * it now (0), fails (-1, errno; the frm migrate-failed event carries it), or
85 * -- a dataset completion marker under the archiver -- defers the seal. */
86static ngx_int_t
87frm_publish(brix_sd_instance_t *inst, const char *key)
88{
89 sd_frm_state *fst = SD_FRM_ST(inst);
90 int rc;
91
92 rc = fst->mss->migrate(fst->mss_ctx, key);
93 if (rc < 0) {
94 char num[24];
95
96 brix_stage_events_emit("frm", "migrate-failed", NULL, key, "errno",
97 brix_stage_events_num(num, sizeof(num), errno),
98 NULL);
99 return NGX_ERROR;
100 }
101 if (rc == BRIX_MSS_MIGRATE_DEFERRED) {
102 return frm_seal_defer(fst, inst, key);
103 }
104 brix_stage_events_emit("frm", "migrate-done", NULL, key, NULL);
105 return NGX_OK;
106}

110brix_sd_staged_t *
111sd_frm_staged_open(brix_sd_instance_t *inst, const char *final_path,
112 mode_t mode, off_t declared_size, int *err_out)
113{
114 sd_frm_state *st = SD_FRM_ST(inst);
115 sd_frm_staged_state *ss;
116 brix_sd_staged_t *h;
117 int fd;
118
119 fd = st->mss->create_online(st->mss_ctx, final_path, mode);
120 if (fd < 0) {
121 if (err_out) { *err_out = errno ? errno : EIO; }
122 return NULL;
123 }
124
125 /* Phase-107 C5: the ONLINE BUFFER is the disk-resident half this driver
126 * owns — preallocate the declared final size there so a buffer that cannot
127 * hold the object refuses the OPEN (ENOSPC/EDQUOT), not the migrate after
128 * hours of streaming. Tape-side reservation stays the MSS tier's own
129 * scheduling concern. Anything short of "no space" is advisory. */
130 if (declared_size > 0) {
131 brix_sd_obj_t shell;
132
133 ngx_memzero(&shell, sizeof(shell));
134 shell.fd = fd;
135 if (sd_posix_reserve(&shell, declared_size) != NGX_OK
136 && (errno == ENOSPC || errno == EDQUOT))
137 {
138 int err = errno;
139
140 (void) close(fd);
141 (void) st->mss->purge(st->mss_ctx, final_path);
142 if (err_out) { *err_out = err; }
143 errno = err;
144 return NULL;
145 }
146 }
147
148 ss = calloc(1, sizeof(*ss));
149 h = calloc(1, sizeof(*h));
150 if (ss == NULL || h == NULL) {
151 (void) close(fd);
152 free(ss);
153 free(h);
154 if (err_out) { *err_out = ENOMEM; }
155 return NULL;
156 }
157 ss->fst = st;
158 ss->fd = fd;
159 ngx_cpystrn((u_char *) ss->key, (u_char *) final_path, sizeof(ss->key));
160 h->inst = inst;
161 h->state = ss;
162 return h;
163}

173/*
174 * WHAT: Evaluate a typed publish precondition (phase-107 C6) for a staged
175 * commit, BEFORE the migrate, so a refusal leaves the tape copy
176 * untouched (the caller's staged_abort then purges the buffer temp).
177 * WHY: The online buffer under `key` at commit time is this handle's OWN —
178 * staged_open created it — so the adapter's residency probe answers
179 * ONLINE for every staged create. Judged by residency, ABSENT (a
180 * kXR_new open, an If-None-Match PUT) refused EVERY create with EEXIST:
181 * no root:// or HTTP create could ever commit to a tape export
182 * (phase-115 W3.1).
183 * HOW: ABSENT asks the durable copy (`on_tape`, phase-115 W3.2). An
184 * adapter without the verb cannot see past the buffer, so the open-time
185 * existence check stands alone — every kind is ADVISORY here anyway
186 * (pre->atomic stays 0, the W7 checklist's verdict for frm; §3.5
187 * forbids claiming atomicity the MSS never gave). MATCH_* keep the
188 * residency stat: with the buffer shadowing the target they compare
189 * against what is online, a limitation recorded with the phase.
190 */
191static ngx_int_t
192frm_commit_precond(const sd_frm_staged_state *ss, const brix_sd_precond_t *pre)
193{
194 const brix_mss_adapter_t *mss = ss->fst->mss;
195 off_t tsz = 0;
196 time_t tmt = 0;
197 int res;
198
199 if (pre == NULL || pre->kind == BRIX_SD_PRECOND_NONE) {
200 return NGX_OK;
201 }
202 if (pre->kind == BRIX_SD_PRECOND_ABSENT) {
203 if (mss->on_tape != NULL
204 && mss->on_tape(ss->fst->mss_ctx, ss->key) == 1)
205 {
206 errno = EEXIST;
207 return NGX_ERROR;
208 }
209 return NGX_OK;
210 }
211 res = mss->residency(ss->fst->mss_ctx, ss->key, &tsz, &tmt);
212 if (res == BRIX_RESIDENCY_ABSENT) {
213 errno = ECANCELED; /* MATCH_* against a missing target */
214 return NGX_ERROR;
215 }
216 if (brix_sd_precond_eval_stat(pre, tsz, tmt) != 0) {
217 return NGX_ERROR; /* errno = ECANCELED / ENOTSUP */
218 }
219 return NGX_OK;
220}

222ngx_int_t
223sd_frm_staged_commit(brix_sd_staged_t *st, brix_sd_precond_t *pre)
224{
225 sd_frm_staged_state *ss = st->state;
226
227 if (frm_commit_precond(ss, pre) != NGX_OK) {
228 return NGX_ERROR;
229 }
230
231 if (ss->fd >= 0) {
232 (void) close(ss->fd);
233 ss->fd = -1;
234 }
235 /* Publish: migrate the online-buffer object to tape (a dataset completion
236 * marker queues its archive seal instead -- 2.0 F3, frm_publish). */
237 if (frm_publish(st->inst, ss->key) != NGX_OK) {
238 /* Ownership contract: only a SUCCESSFUL commit consumes the handle. A
239 * failed migrate must leave st+ss valid — every caller aborts a failed
240 * commit (stage_engine, cstb_pump_and_commit, cache fetch), and abort
241 * frees them. Freeing here made that mandatory abort a use-after-free,
242 * a double free, and a second purge of the online buffer. */
243 return NGX_ERROR;
244 }
245 free(ss);
246 free(st);
247 return NGX_OK;
248}

265/* Atomic two-name exchange (phase-107 C6): swap the ONLINE-BUFFER copies via
266 * the adapter's exchange verb, then MIGRATE both keys so tape truth catches
267 * up — an un-migrated swap would serve OLD content after a purge + recall.
268 * Both names must be online (the verb answers ENOENT otherwise; the caller
269 * recalls first); an adapter without the verb (stub) refuses ENOTSUP, never
270 * a two-rename emulation (§3.5). Failure discipline mirrors the C3 barrier:
271 * a first-migrate failure un-exchanges (nothing shipped, state fully
272 * restored); a second-migrate failure cannot be unwound — `a` already
273 * shipped — so it logs at crit and FAILS rather than claim tape consistency
274 * the MSS does not have (the vfs_rename.c durable-publish doctrine). */
275ngx_int_t
276sd_frm_exchange(brix_sd_instance_t *inst, const char *a, const char *b)
277{
278 sd_frm_state *st = SD_FRM_ST(inst);
279 int err;
280
281 if (st->mss->exchange == NULL) {
282 errno = ENOTSUP;
283 return NGX_ERROR;
284 }
285 if (st->mss->exchange(st->mss_ctx, a, b) != 0) {
286 return NGX_ERROR; /* errno from the verb */
287 }
288 if (frm_publish(inst, a) != NGX_OK) {
289 err = errno ? errno : EIO;
290 (void) st->mss->exchange(st->mss_ctx, a, b);
291 errno = err;
292 return NGX_ERROR;
293 }
294 if (frm_publish(inst, b) != NGX_OK) {
295 err = errno ? errno : EIO;
296 ngx_log_error(NGX_LOG_CRIT, st->log, err,
297 "brix: frm exchange: tape catch-up migrate failed for "
298 "\"%s\" after \"%s\" shipped", b, a);
299 errno = err;
300 return NGX_ERROR;
301 }
302 return NGX_OK;
303}
