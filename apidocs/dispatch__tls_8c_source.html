4 * WHAT: Classifies the session/data/login capabilities exercised by a request

16/* ---- Enforce WebDAV TLS capability requirements ----
17 *
18 * WHAT: Return NGX_DECLINED when the request may proceed, or a completed 403
19 * response when any exercised capability requires TLS.
20 * WHY: Session, data, and login policies must agree with the root protocol and
21 * protect both header and query credential transports.
22 * HOW: 1. Classify method and credential capabilities. 2. Ask the shared gate.
23 * 3. On refusal scrub query tokens, log the capability, and count the 403.
24 */
25ngx_int_t
26webdav_tls_require_gate(ngx_http_request_t *r,
27 ngx_http_brix_webdav_loc_conf_t *conf)
28{
29 ngx_uint_t caps;
30 ngx_uint_t refused;
31
32 if (conf->common.tls_require == 0) {
33 return NGX_DECLINED;
34 }
35
36 caps = BRIX_TLSREQ_SESSION;
37 if (r->method == NGX_HTTP_GET || r->method == NGX_HTTP_PUT) {
38 caps |= BRIX_TLSREQ_DATA;
39 }
40 if (r->headers_in.authorization != NULL
41 || brix_http_query_has_decoded_name(r->args, "authz",
42 BRIX_HTTP_QUERY_CASE_INSENSITIVE)
43 || brix_http_query_has_decoded_name(r->args, "access_token",
44 BRIX_HTTP_QUERY_CASE_INSENSITIVE))
45 {
46 caps |= BRIX_TLSREQ_LOGIN;
47 }
48
49 refused = brix_tls_gate_refused(conf->common.tls_require, caps,
50 r->connection->ssl != NULL);
51 if (refused == 0) {
52 return NGX_DECLINED;
53 }
54
55 brix_http_redact_request_query_tokens(r);
56 BRIX_LOG(NGX_LOG_WARN, r->connection->log, 0,
57 "brix: cleartext request refused by brix_tls_require %s (method=%V)",
58 brix_tls_cap_name(refused), &r->method_name);
59 return webdav_metrics_return(r, NGX_HTTP_FORBIDDEN);
60}
