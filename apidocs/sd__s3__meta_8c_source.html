28/*
29 * WHAT: Run an already-signed HEAD and hand back the LIVE response. 0 with
30 * *resp filled (the caller must resp_free it), or -1 with errbuf set.
31 * WHY: Three readers want the same wire leg and the same status verdict — a
32 * user-metadata attribute, a stored checksum, and the archive-state pair
33 * (sd_s3_archive.c) — and they differ only in what they sign and which
34 * headers they then pull out. Handing back the live response rather than
35 * one extracted value is what lets the archive reader take THREE headers
36 * off ONE round trip instead of HEADing the object three times.
37 * HOW: The caller signs (the signer differs: a plain HEAD vs one carrying an
38 * extra x-amz-* header, which AWS requires in the signed set) and passes
39 * the finished header block. Non-200 is mapped and the response released
40 * here, so a failing caller never has to remember to free.
41 */
42int
43sd_s3_head_send(sd_s3_file *f, const char *hdrs, brix_s3_resp_t *resp,
44 char *errbuf, size_t errcap)
45{
46 if (f->transport->request(f->tctx, f->host, f->port, f->tls, "HEAD",
47 f->key, hdrs, NULL, 0, f->timeout_ms, resp,
48 errbuf, errcap) != 0)
49 {
50 return -1;
51 }
52 if (resp->status != 200) {
53 int rc = sd_s3_status_err(resp->status, "HEAD", f->key, errbuf, errcap);
54 f->transport->resp_free(resp);
55 return rc; /* -1 */
56 }
57 return 0;
58}

60/*
61 * WHAT: One named response header off an already-signed HEAD. >0 = the value's
62 * length, 0 = the header is absent, -1 = error (errbuf).
63 * WHY: An absent header is NOT an error: on S3 it means "this object carries
64 * no such value", which both the metadata and the checksum reader report
65 * as a decline rather than a failure.
66 */
67static ssize_t
68sd_s3_head_header(sd_s3_file *f, const char *hdrs, const char *want,
69 const sd_s3_meta_buf *out, char *errbuf, size_t errcap)
70{
71 brix_s3_resp_t resp;
72
73 if (sd_s3_head_send(f, hdrs, &resp, errbuf, errcap) != 0) {
74 return -1;
75 }
76 if (f->transport->resp_header(&resp, want, out->buf, out->cap) != 0) {
77 f->transport->resp_free(&resp);
78 out->buf[0] = '\0';
79 return 0; /* header absent */
80 }
81 f->transport->resp_free(&resp);
82 return (ssize_t) strlen(out->buf);
83}

135/* One "user.<name>" listxattr entry from a raw "x-amz-meta-<name>: v" header
136 * line. Appends to buf[*used..cap) when buf is non-NULL and always advances
137 * *need by the entry size; the caller compares need vs cap at the end. */
138static void
139sd_s3_list_meta_emit(const char *name, size_t nlen, char *buf, size_t cap,
140 size_t *need)
141{
142 size_t entry = sizeof("user.") - 1 + nlen + 1;
143
144 if (buf != NULL && *need + entry <= cap) {
145 memcpy(buf + *need, "user.", sizeof("user.") - 1);
146 memcpy(buf + *need + sizeof("user.") - 1, name, nlen);
147 buf[*need + entry - 1] = '\0';
148 }
149 *need += entry;
150}
151
152/* Scan the raw response header block for "x-amz-meta-<name>:" lines and emit
153 * each as a "user.<name>" listxattr entry (the advisory blob is skipped — it
154 * surfaces as POSIX attrs, not as a user xattr). Returns the total byte need;
155 * the caller compares it against cap. */
156static size_t
157sd_s3_list_meta_scan(const char *hdrs, char *buf, size_t cap)
158{
159 static const char pfx[] = "x-amz-meta-";
160 const size_t pfxlen = sizeof(pfx) - 1;
161 const char *p;
162 size_t need = 0;
163
164 for (p = hdrs; *p != '\0'; ) {
165 const char *eol = p + strcspn(p, "\r\n");
166 const char *colon;
167
168 if ((size_t) (eol - p) > pfxlen
169 && strncasecmp(p, pfx, pfxlen) == 0
170 && (colon = memchr(p, ':', (size_t) (eol - p))) != NULL
171 && colon > p + pfxlen)
172 {
173 const char *name = p + pfxlen;
174 size_t nlen = (size_t) (colon - name);
175
176 /* The advisory blob surfaces as POSIX attrs, not as a user xattr. */
177 if (nlen != sizeof(BRIX_META_ADVISORY_S3META) - 1
178 || strncasecmp(name, BRIX_META_ADVISORY_S3META, nlen) != 0)
179 {
180 sd_s3_list_meta_emit(name, nlen, buf, cap, &need);
181 }
182 }
183 p = eol + strspn(eol, "\r\n");
184 }
185 return need;
186}

188ssize_t
189sd_s3_list_meta(sd_s3_file *f, char *buf, size_t cap,
190 char *errbuf, size_t errcap)
191{
192 char auth[SD_S3_AUTH_HDRS_CAP];
193 brix_s3_resp_t resp;
194 const char *hdrs;
195 size_t need;
196
197 if (f == NULL) {
198 sd_s3_set_err(errbuf, errcap, "s3 list-meta: bad parameters");
199 errno = EINVAL;
200 return -1;
201 }
202 if (f->transport->resp_headers_raw == NULL) {
203 sd_s3_set_err(errbuf, errcap,
204 "s3 list-meta: transport cannot enumerate headers");
205 errno = ENOTSUP;
206 return -1;
207 }
208 if (sd_s3_sign(f, "HEAD", "", auth, sizeof(auth)) != 0) {
209 sd_s3_set_err(errbuf, errcap, "s3 HEAD: SigV4 sign failed on %s", f->key);
210 errno = EIO;
211 return -1;
212 }
213 if (f->transport->request(f->tctx, f->host, f->port, f->tls, "HEAD",
214 f->key, auth, NULL, 0, f->timeout_ms, &resp,
215 errbuf, errcap) != 0)
216 {
217 errno = EIO;
218 return -1;
219 }
220 if (resp.status != 200) {
221 int rc = sd_s3_status_err(resp.status, "HEAD", f->key, errbuf, errcap);
222 f->transport->resp_free(&resp);
223 return rc; /* -1, errno mapped from the HTTP status */
224 }
225 hdrs = f->transport->resp_headers_raw(&resp);
226 if (hdrs == NULL) {
227 f->transport->resp_free(&resp);
228 sd_s3_set_err(errbuf, errcap, "s3 list-meta: no raw header block");
229 errno = ENOTSUP;
230 return -1;
231 }
232
233 need = sd_s3_list_meta_scan(hdrs, buf, cap);
234 f->transport->resp_free(&resp);
235
236 if (buf == NULL || cap == 0) {
237 return (ssize_t) need; /* listxattr(2) size probe */
238 }
239 if (need > cap) {
240 sd_s3_set_err(errbuf, errcap, "s3 list-meta: buffer too small");
241 errno = ERANGE;
242 return -1;
243 }
244 return (ssize_t) need;
245}

273static int
274sd_s3_set_meta_f(sd_s3_file *f, const sd_s3_meta_kv *kv, size_t nkv,
275 char *errbuf, size_t errcap)
276{
277 char auth[SD_S3_AUTH_HDRS_CAP];
278 sd_s3_sign_hdr_t extra[2 + 32];
279 sd_s3_sign_req_t req;
280 char names[32][160];
281 char lname[160];
282 size_t n_extra = 0, i, j, nl;
283 brix_s3_resp_t resp;
284
285 if (nkv > 32) {
286 sd_s3_set_err(errbuf, errcap, "s3 set-meta: too many attributes");
287 return -1;
288 }
289 /* Self-copy with REPLACE: update only metadata, never re-upload the bytes. */
290 extra[n_extra].name = "x-amz-copy-source";
291 extra[n_extra++].value = f->key;
292 extra[n_extra].name = "x-amz-metadata-directive";
293 extra[n_extra++].value = "REPLACE";
294
295 for (i = 0; i < nkv; i++) {
296 if (kv[i].name == NULL || kv[i].value == NULL) {
297 sd_s3_set_err(errbuf, errcap, "s3 set-meta: null attribute");
298 return -1;
299 }
300 nl = strlen(kv[i].name);
301 if (nl >= sizeof(lname)) {
302 sd_s3_set_err(errbuf, errcap, "s3 set-meta: attribute name too long");
303 return -1;
304 }
305 for (j = 0; j < nl; j++) { /* AWS lowercases user-metadata names */
306 char c = kv[i].name[j];
307 lname[j] = (c >= 'A' && c <= 'Z') ? (char) (c - 'A' + 'a') : c;
308 }
309 lname[nl] = '\0';
310 if (snprintf(names[i], sizeof(names[i]), "x-amz-meta-%s", lname)
311 >= (int) sizeof(names[i]))
312 {
313 sd_s3_set_err(errbuf, errcap, "s3 set-meta: attribute name too long");
314 return -1;
315 }
316 extra[n_extra].name = names[i];
317 extra[n_extra++].value = kv[i].value;
318 }
319
320 req.method = "PUT";
321 req.canon_qs = "";
322 req.extra = extra;
323 req.n_extra = n_extra;
324 if (sd_s3_sign_ext(f, &req, auth, sizeof(auth)) != 0) {
325 sd_s3_set_err(errbuf, errcap, "s3 set-meta: SigV4 sign failed on %s",
326 f->key);
327 return -1;
328 }
329 if (f->transport->request(f->tctx, f->host, f->port, f->tls, "PUT",
330 f->key, auth, NULL, 0, f->timeout_ms, &resp,
331 errbuf, errcap) != 0)
332 {
333 return -1;
334 }
335 if (resp.status != 200) {
336 int rc = sd_s3_status_err(resp.status, "CopyObject(REPLACE)", f->key,
337 errbuf, errcap);
338 f->transport->resp_free(&resp);
339 return rc;
340 }
341 f->transport->resp_free(&resp);
342 return 0;
343}
