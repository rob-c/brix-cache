25/*
26 * Extract the raw token string from a JSON response.
27 *
28 * The JSON must contain a "access_token": "<token>" field.
29 * Returns NGX_OK and fills `token_out` (allocated in r->pool) on success.
30 */

33 const char *json,
34 ngx_str_t *token_out)
35{
36 char token[TPC_CRED_MAX_TOKEN_LEN + 1];

43 BRIX_LOG(NGX_LOG_ERR, r->connection->log, 0,
44 "tpc_cred: %s", err);
45 return NGX_ERROR;
46 }
47
48 tok_len = strlen(token);
49 token_out->len = tok_len;
50 token_out->data = ngx_pnalloc(r->pool, tok_len + 1);
51 if (token_out->data == NULL) {
52 BRIX_LOG(NGX_LOG_ERR, r->connection->log, 0,
53 "tpc_cred: pnalloc failed for token");
54 return NGX_ERROR;
55 }
56 memcpy(token_out->data, token, tok_len);
57 token_out->data[tok_len] = '\0';
58
59 return NGX_OK;
60}
