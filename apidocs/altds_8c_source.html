1/*
2 * cms/altds.c — §2.12: alternate-data-server liveness monitor (cms.altds).
3 *
4 * WHAT: A periodic nonblocking TCP connect probe of the co-located foreign
5 * data server (127.0.0.1:<brix_cms_altds port>). On DOWN, every logged-in
6 * manager link gets kYR_status(suspend) so the mesh stops selecting this
7 * node; on recovery, kYR_status(resume). The advertisement half of §2.12
8 * (dPort = altds port in the login) lives in send.c.
9 *
10 * WHY: The CMS half of this node fronts a foreign data server it does not
11 * host in-process; a dead foreign DS with a live heartbeat would keep
12 * attracting redirects into connection refusals. Stock cmsd monitors its
13 * altds the same way.
14 *
15 * HOW: One timer per CMS-client worker (worker 0). Each tick opens a
16 * nonblocking socket, connects to loopback:<port>, and classifies: immediate
17 * success / EINPROGRESS-with-writable-within-timeout = UP, refusal/timeout =
18 * DOWN. The wait is event-driven (ngx_get_connection on the probe fd); a
19 * 2s cap bounds a black-holed probe. Only state TRANSITIONS emit status
20 * frames — steady state is silent.
21 */

47/*
48 * altds_broadcast — send kYR_status to every logged-in manager link.
49 *
50 * WHAT: suspend=1 sends CMS_ST_SUSPEND, else CMS_ST_RESUME|CMS_ST_NOSTAGE
51 * (the same resume shape login uses). Send failures are left to the
52 * heartbeat machinery — a broken link reconnects and re-announces.
53 * WHY: The verdict must reach ALL redundant managers, not the rotation pick.
54 * HOW: Walk conf->cms.ctxs like the CNS fan-out does.
55 */
56static void
57altds_broadcast(brix_cms_altds_t *ad, int suspend)
58{
59 ngx_uint_t i;
60 ngx_brix_cms_ctx_t *ctx;
61 u_char mod = suspend ? CMS_ST_SUSPEND
62 : (CMS_ST_RESUME | CMS_ST_NOSTAGE);
63
64 for (i = 0; i < ad->conf->cms.nctxs; i++) {
65 ctx = ad->conf->cms.ctxs[i];
66 if (ctx == NULL || ctx->connection == NULL || !ctx->logged_in) {
67 continue;
68 }
69 (void) brix_cms_send_frame(ctx->connection, 0, CMS_RR_STATUS, mod,
70 NULL, 0);
71 }
72}

74/*
75 * altds_verdict — fold one probe outcome into the up/down state machine.
76 *
77 * WHAT: Emits the suspend/resume broadcast ONLY on a transition and logs it.
78 * WHY: Steady-state chatter would spam every manager each interval.
79 * HOW: Compare with ad->down; update; broadcast on change.
80 */
81static void
82altds_verdict(brix_cms_altds_t *ad, int alive)
83{
84 if (alive == !ad->down) {
85 return; /* no transition */
86 }
87 ad->down = !alive;
88
89 ngx_log_error(NGX_LOG_NOTICE, ad->cycle->log, 0,
90 "brix: cms altds 127.0.0.1:%i is %s — %s",
91 (int) ad->conf->cms.altds_port,
92 alive ? "back" : "down",
93 alive ? "resuming this node in the cluster"
94 : "suspending this node in the cluster");
95 brix_cms_log_action(ad->cycle->log, "altds",
96 (const char *) ad->conf->cms.manager.data, "out",
97 NULL, alive,
98 alive ? "foreign data server recovered -> resume"
99 : "foreign data server unreachable -> suspend");
100 altds_broadcast(ad, !alive);
101}

113/*
114 * altds_probe_event — the pending connect resolved (writable) or timed out.
115 *
116 * WHAT: Reads SO_ERROR for the verdict on writability; ev->timedout is DOWN.
117 * WHY: A nonblocking connect signals completion via writability; SO_ERROR
118 * distinguishes success from a delayed refusal.
119 * HOW: Classify, close the probe, fold the verdict.
120 */
121static void
122altds_probe_event(ngx_event_t *ev)
123{
124 ngx_connection_t *c = ev->data;
125 brix_cms_altds_t *ad = c->data;
126 int err = 0;
127 socklen_t errlen = sizeof(err);
128 int alive = 0;
129
130 if (!ev->timedout) {
131 if (getsockopt(c->fd, SOL_SOCKET, SO_ERROR, &err, &errlen) == 0
132 && err == 0)
133 {
134 alive = 1;
135 }
136 }
137 if (c->write->timer_set) {
138 ngx_del_timer(c->write);
139 }
140 altds_probe_close(ad);
141 altds_verdict(ad, alive);
142}

144/*
145 * altds_tick — periodic probe: nonblocking TCP connect to the foreign DS.
146 *
147 * WHAT: Immediate success/refusal classifies synchronously; EINPROGRESS
148 * parks the fd in the event loop with a bounded wait.
149 * WHY: The probe must never block the worker — the CMS heartbeat shares it.
150 * HOW: socket + O_NONBLOCK + connect to loopback:<port>; wrap pending fds
151 * in ngx_get_connection with altds_probe_event on write-ready.
152 */
153static void
154altds_tick(ngx_event_t *ev)
155{
156 brix_cms_altds_t *ad = ev->data;
157 struct sockaddr_in sin;
158 ngx_connection_t *c;
159 int fd, rc;
160
161 if (!ngx_exiting) {
162 ngx_add_timer(&ad->tick, ad->conf->cms.altds_interval);
163 }
164 if (ad->probe != NULL) {
165 return; /* previous probe still pending — its timeout will verdict */
166 }
167
168 fd = socket(AF_INET, SOCK_STREAM | SOCK_NONBLOCK | SOCK_CLOEXEC, 0);
169 if (fd < 0) {
170 return; /* fd pressure: skip this tick, keep the last verdict */
171 }
172
173 ngx_memzero(&sin, sizeof(sin));
174 sin.sin_family = AF_INET;
175 sin.sin_port = htons((uint16_t) ad->conf->cms.altds_port);
176 sin.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
177
178 rc = connect(fd, (struct sockaddr *) &sin, sizeof(sin));
179 if (rc == 0) {
180 close(fd);
181 altds_verdict(ad, 1);
182 return;
183 }
184 if (errno != EINPROGRESS) {
185 close(fd);
186 altds_verdict(ad, 0);
187 return;
188 }
189
190 c = ngx_get_connection(fd, ad->cycle->log);
191 if (c == NULL) {
192 close(fd);
193 return;
194 }
195 c->data = ad;
196 c->read->handler = altds_probe_event;
197 c->write->handler = altds_probe_event;
198 if (ngx_handle_write_event(c->write, 0) != NGX_OK) {
199 ngx_close_connection(c);
200 return;
201 }
202 ad->probe = c;
203 ngx_add_timer(c->write, BRIX_ALTDS_PROBE_TIMEOUT_MS);
204}

207brix_cms_altds_start(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *conf)
208{
209 brix_cms_altds_t *ad = &brix_cms_altds;

211 if (conf->cms.altds_port <= 0 || !conf->cms.altds_monitor

215 }
216
217 ad->cycle = cycle;
218 ad->conf = conf;

220 ad->tick.handler = altds_tick;
221 ad->tick.data = ad;
222 ad->tick.log = cycle->log;
223 ad->tick.cancelable = 1;
224

226
227 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
