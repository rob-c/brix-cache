14/* Release a driver object obtained from driver->open() by a caller that holds it
15 * by POINTER (not the VFS, which adopts the object by value and frees the shell
16 * itself in vfs_open.c): close it via its own vtable, then free a heap-allocated
17 * shell (heap_shell=1, e.g. POSIX's malloc'd obj — allocated off inst->pool so a
18 * cache-fill thread never touches the thread-unsafe ngx_cycle->pool). NULL-safe;
19 * a pool-allocated shell (heap_shell=0) is just closed. */
20static inline void
21brix_sd_obj_release(brix_sd_obj_t *o)
22{
23 if (o == NULL) {
24 return;
25 }
26 if (o->driver != NULL && o->driver->close != NULL) {
27 o->driver->close(o);
28 }
29 if (o->heap_shell) {
30 free(o);
31 }
32}
