1/*
2 * sd_remote_checksum.c — checksum offload for the remote-origin (s3://) driver.
3 *
4 * WHAT: The `query_checksum` vtable slot. Answers a checksum request from the
5 * digest the S3 origin ALREADY stores for the object, read off one signed
6 * HEAD, instead of reading the object's bytes back to hash them.
7 *
8 * WHY: Without it, a checksum against an s3:// export is a full-object Range-GET
9 * walk across the network — for a WLCG dataset that is the whole transfer,
10 * paid a second time for a value the store computed at upload. This is the
11 * s3 sibling of the root:// kXR_Qcksum and the http RFC-3230 offloads; the
12 * byte-reading compute in core/compat/integrity_info.c stays the fallback.
13 *
14 * HOW: S3 exposes two independent surfaces, and they are not interchangeable:
15 * - the additional-checksum headers (`x-amz-checksum-<algo>`), present
16 * only when the uploader supplied one, base64-encoded, and returned
17 * only when the request asks (`x-amz-checksum-mode: ENABLED`);
18 * - the `ETag`, which is an md5 of the object — but ONLY for a
19 * single-part upload. A multipart ETag is an md5 OF THE PART DIGESTS
20 * with a `-<parts>` suffix and is not the object's md5 at all, so it is
21 * refused rather than handed on.
22 * The value is normalised to lowercase hex through the shared digest
23 * grammar (core/compat/digest_header.c) — the same normalisation the http
24 * offload and the WebDAV PUT verifier use, so an S3-sourced digest and an
25 * origin-sourced one cannot be spelled differently.
26 *
27 * A decline is never a failure: per the slot contract the caller falls back to
28 * reading the bytes, so an object uploaded without a checksum, an algorithm S3
29 * does not compute, or a multipart ETag simply costs one HEAD.
30 */

39/* Longest value read off the wire: a base64 sha-256 is 44 bytes, a quoted
40 * multipart ETag with a large part count is under 60. 128 leaves room for an
41 * origin that pads or quotes generously; anything longer is not a digest this
42 * driver can use and the grammar refuses it. */

62/*
63 * WHAT: Reduce a raw `ETag` value to the object's md5 hex, or refuse it.
64 * WHY: An ETag is an md5 of the OBJECT only when the object was uploaded in one
65 * part. AWS spells a multipart ETag "<hex>-<nparts>" — an md5 of the
66 * concatenated part digests, which matches nothing a client can compute —
67 * and server-side-encrypted objects (SSE-KMS/SSE-C) have an ETag that is
68 * not an md5 at all. Handing either back as "the md5" is worse than
69 * declining, because the caller presents it as authoritative.
70 * HOW: Strip the surrounding quotes, then refuse anything but a bare 32-char
71 * run — the '-' of a multipart tag and the width of anything else both
72 * fail that test, so no separate multipart special-case is needed.
73 */
74static int
75sd_remote_etag_md5(char *val)
76{
77 size_t len = strlen(val);
78 size_t i;
79
80 if (len >= 2 && val[0] == '"' && val[len - 1] == '"') {
81 memmove(val, val + 1, len - 2);
82 val[len - 2] = '\0';
83 len -= 2;
84 }
85 if (len != 32) {
86 return -1; /* multipart ("-<n>"), SSE, or not an md5 at all */
87 }
88 for (i = 0; i < len; i++) {
89 if (!((val[i] >= '0' && val[i] <= '9')
90 || (val[i] >= 'a' && val[i] <= 'f')
91 || (val[i] >= 'A' && val[i] <= 'F')))
92 {
93 return -1;
94 }
95 }
96 return 0;
97}

99/* Index of the table row for `algo`, or -1 when S3 computes no such digest. */
100static int
101sd_remote_ck_slot(const char *algo)
102{
103 size_t i;
104
105 for (i = 0; i < sizeof(sd_remote_ck_map) / sizeof(sd_remote_ck_map[0]); i++) {
106 if (strcmp(algo, sd_remote_ck_map[i].algo) == 0) {
107 return (int) i;
108 }
109 }
110 return -1;
111}

113ngx_int_t
114sd_remote_query_checksum(brix_sd_obj_t *obj, const char *algo, char *hex_out,
115 size_t hex_sz)
116{
117 sd_remote_obj_state *st;
118 sd_s3_meta_buf mb;
119 char raw[SD_REMOTE_CK_RAW_MAX];
120 char hex[BRIX_DIGEST_HEX_MAX];
121 char errbuf[256];
122 ssize_t n;
123 int slot;
124
125 if (obj == NULL || obj->state == NULL || algo == NULL || hex_out == NULL
126 || hex_sz == 0)
127 {
128 return NGX_DECLINED;
129 }
130 slot = sd_remote_ck_slot(algo);
131 if (slot < 0) {
132 return NGX_DECLINED; /* S3 stores no such digest — never ask */
133 }
134 st = obj->state;
135 if (st->s3 == NULL) {
136 return NGX_DECLINED;
137 }
138
139 mb.buf = raw;
140 mb.cap = sizeof(raw);
141 n = sd_s3_get_checksum(st->s3, sd_remote_ck_map[slot].hdr, &mb, errbuf,
142 sizeof(errbuf));
143 if (n < 0) {
144 return NGX_ERROR; /* transport/HTTP fault → compute the bytes */
145 }
146 if (n == 0) {
147 return NGX_DECLINED; /* uploaded without this checksum */
148 }
149 if (sd_remote_ck_map[slot].b64 == 0 && sd_remote_etag_md5(raw) != 0) {
150 return NGX_DECLINED;
151 }
152 if (brix_digest_value_hex((u_char *) raw, ngx_strlen(raw),
153 sd_remote_ck_map[slot].b64, hex, sizeof(hex))
154 != NGX_OK)
155 {
156 return NGX_DECLINED; /* unusable value: refuse it whole */
157 }
158 brix_digest_hex_pad(algo, hex, sizeof(hex));
159 if (ngx_strlen(hex) + 1 > hex_sz) {
160 return NGX_DECLINED;
161 }
162 ngx_cpystrn((u_char *) hex_out, (u_char *) hex, hex_sz);
163 return NGX_OK;
164}
