18static int
19gftp_data_port(gftp_session_t *session, unsigned *port)
20{
21 unsigned char ignored[4];
22
23 if (gftp_command(session, "EPSV") != 0) {
24 return -1;

43 return 0;

46static int
47gftp_data_connect(gftp_session_t *session)
48{
49 struct addrinfo hints;
50 struct addrinfo *result = NULL;
51 char service[16];
52 unsigned port;
53 int fd;
54
55 if (gftp_data_port(session, &port) != 0) {
56 return -1;
57 }
58 if (port < 1024 || port > 65535 || session->peer_ip[0] == '\0') {
59 gftp_set_error(session, EACCES,
60 "refusing unsafe GridFTP passive data target");
61 return -1;
62 }
63 memset(&hints, 0, sizeof(hints));
64 hints.ai_socktype = SOCK_STREAM;
65 hints.ai_family = AF_UNSPEC;
66 hints.ai_flags = AI_NUMERICHOST;
67 (void) snprintf(service, sizeof(service), "%u", port);
68 if (getaddrinfo(session->peer_ip, service, &hints, &result) != 0) {
69 gftp_set_error(session, EHOSTUNREACH,
70 "cannot resolve pinned GridFTP data peer");
71 return -1;
72 }
73 fd = socket(result->ai_family, result->ai_socktype, result->ai_protocol);
74 if (fd >= 0 && brix_connect_fd_deadline(fd, result->ai_addr,
75 result->ai_addrlen, session->timeout_ms) != 0) {
76 close(fd);
77 fd = -1;
78 }
79 freeaddrinfo(result);
80 if (fd < 0) {
81 gftp_set_error(session, ECONNREFUSED,
82 "cannot connect to pinned GridFTP data peer");
83 }
84 return fd;
85}

200gftp_slurp_append(gftp_session_t *session, char **data, size_t *used,
201 const uint8_t *part, size_t len)
202{
203 char *larger;
204
205 if (len > GFTP_SLURP_MAX - *used) {

207 return -1;

209 larger = realloc(*data, *used + len + 1);
210 if (larger == NULL) {

212 return -1;
213 }
214 memcpy(larger + *used, part, len);

216 larger[*used] = '\0';
