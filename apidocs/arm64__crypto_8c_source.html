73/*
74 * Hardware CRC32C using ARMv8-A CRC extensions
75 *
76 * Performance: ~1 cycle per byte (vs ~10 cycles/byte for software)
77 * Instructions used:
78 * - __crc32cb: CRC32 of byte
79 * - __crc32ch: CRC32 of halfword
80 * - __crc32cw: CRC32 of word
81 * - __crc32cd: CRC32 of doubleword
82 */
83
84uint32_t
85brix_crc32c_arm64_hw(const uint8_t *buf, size_t len, uint32_t crc)
86{
87 /* Process 8 bytes at a time using 64-bit CRC */
88 while (len >= 8) {
89 uint64_t val;
90 memcpy(&val, buf, sizeof(val));
91 crc = __crc32cd(crc, val);
92 buf += 8;
93 len -= 8;
94 }
95
96 /* Process 4 bytes */
97 if (len >= 4) {
98 uint32_t val;
99 memcpy(&val, buf, sizeof(val));
100 crc = __crc32cw(crc, val);
101 buf += 4;
102 len -= 4;
103 }
104
105 /* Process 2 bytes */
106 if (len >= 2) {
107 uint16_t val;
108 memcpy(&val, buf, sizeof(val));
109 crc = __crc32ch(crc, val);
110 buf += 2;
111 len -= 2;
112 }
113
114 /* Process 1 byte */
115 if (len >= 1) {
116 crc = __crc32cb(crc, *buf);
117 }
118
119 return crc;
120}

167/*
168 * NEON-accelerated checksum using SIMD
169 *
170 * Processes 16 bytes per iteration using 128-bit NEON registers
171 * Can be extended to 32 bytes with SVE (Scalable Vector Extension)
172 *
173 * Algorithm: Sum all 64-bit words modulo 2^64
174 *
175 * Performance: ~4x faster than scalar implementation
176 */
177
178uint64_t
179brix_checksum_neon(const void *buf, size_t len)
180{
181 const uint64_t *data = (const uint64_t *)buf;
182 uint64x2_t sum0 = vdupq_n_u64(0);
183 uint64x2_t sum1 = vdupq_n_u64(0);
184 size_t i;
185
186 /* Process 32 bytes (4 x 64-bit words) per iteration */
187 for (i = 0; i < len / 32; i++) {
188 uint64x2_t v0 = vld1q_u64(data + i * 4);
189 uint64x2_t v1 = vld1q_u64(data + i * 4 + 2);
190
191 sum0 = vaddq_u64(sum0, v0);
192 sum1 = vaddq_u64(sum1, v1);
193 }
194
195 /* Accumulate sums */
196 sum0 = vaddq_u64(sum0, sum1);
197
198 /* Horizontal add */
199 uint64_t result = vgetq_lane_u64(sum0, 0) + vgetq_lane_u64(sum0, 1);
200
201 /* Handle remaining bytes */
202 size_t remaining = len % 32;
203 const uint8_t *remaining_buf = (const uint8_t *)(data + i * 4);
204
205 for (size_t j = 0; j < remaining; j++) {
206 result += remaining_buf[j];
207 }
208
209 return result;
210}

212#endif /* __ARM_NEON */

222/*
223 * SVE (Scalable Vector Extension) checksum
224 *
225 * SVE provides variable-length vector registers (128-2048 bits)
226 * This implementation adapts to the hardware's vector length
227 *
228 * Performance: Scales with vector length (2x NEON on 256-bit SVE)
229 *
230 * Note: Requires ARMv8.2-A or later with SVE extension
231 */
232
233uint64_t
234brix_checksum_sve(const void *buf, size_t len)
235{
236 const uint64_t *data = (const uint64_t *)buf;
237 size_t vl; /* Vector length in elements */
238 uint64_t sum = 0;
239 size_t i = 0;
240
241 /* Get vector length */
242 vl = svcntd(); /* Count of 64-bit elements in vector */
243
244 /* Process vectors */
245 while (i + vl <= len / 8) {
246 svuint64_t v = svld1_u64(svptrue_b64(), data + i);
247 sum += svaddv_u64(svptrue_b64(), v);
248 i += vl;
249 }
250
251 /* Handle scalar tail */
252 for (size_t j = i * 8; j < len; j++) {
253 sum += ((const uint8_t *)buf)[j];
254 }
255
256 return sum;
257}

259#endif /* __ARM_FEATURE_SVE */
