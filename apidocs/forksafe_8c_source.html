1/*
2 * forksafe.c — §7.7 fork safety: neuter inherited connections in the child.
3 *
4 * WHAT: A process-wide registry of live brix_conns plus pthread_atfork
5 * handlers. After fork(), the CHILD's copies of every registered
6 * connection are neutered: the fd is closed (per-process descriptor
7 * table — the parent's socket is untouched and NO bytes move), the
8 * shared-buffer diagnostics are abandoned unflushed, and the conn is
9 * marked `forked` so every later operation fails with a clean
10 * non-retryable error instead of writing into the parent's stream.
11 * WHY: HEP frameworks fork with sessions open. An inherited fd is shared
12 * with the parent: any child write — including brix_close's
13 * fire-and-forget kXR_endsess, or an atexit teardown — interleaves
14 * frames into the parent's session (stream corruption, sid collisions,
15 * and a child endsess KILLS the parent's session server-side).
16 * Flushing inherited stdio buffers (the --capture file) would
17 * duplicate the parent's buffered bytes; abandoning them leaks a
18 * little memory in a process that is about to exec or exit, which is
19 * the correct trade.
20 * HOW: brix_connect registers on success, brix_close unregisters first
21 * thing. The atfork trio locks the registry across the fork (prepare/
22 * parent) so the child never sees a half-updated table, and the child
23 * handler re-inits the mutex and walks the slots. Fixed table:
24 * overflow conns simply stay unregistered and un-neutered — counted,
25 * and far above any real tool's concurrent connection count.
26 */

51/* ---- Child-side neuter of every registered connection ----
52 *
53 * WHAT: Closes each registered conn's fd (no protocol goodbye, no TLS
54 * close_notify), abandons the SSL/capture handles unfreed, and marks
55 * the conn forked.
56 *
57 * WHY: close() in the child only drops the child's descriptor reference —
58 * the parent's socket lives on and, critically, nothing is transmitted.
59 * SSL_free/fclose would emit bytes or flush shared buffers; leaking
60 * them in the child is deliberate.
61 *
62 * HOW: The mutex was taken pre-fork, so the table is consistent; re-init it
63 * for the child (the parent's lock state is meaningless here), then
64 * walk and neuter.
65 */
66static void
67forksafe_child(void)
68{
69 int i;
70
71 pthread_mutex_init(&g_fs_mx, NULL);
72 for (i = 0; i < BRIX_FORKSAFE_MAX; i++) {
73 brix_conn *c = g_fs_reg[i];
74
75 if (c == NULL) {
76 continue;
77 }
78 if (c->io.fd >= 0) {
79 close(c->io.fd);
80 c->io.fd = -1;
81 }
82 c->io.ssl = NULL; /* abandoned: SSL_free would send close_notify */
83 c->diag.cap = NULL; /* abandoned: fclose would flush shared buffers */
84 c->forked = 1;
85 }
86}

90void
91brix_forksafe_register(brix_conn *c)
92{
93 static int installed = 0;
94 int i;
95
96 pthread_mutex_lock(&g_fs_mx);
97 if (!installed) { /* install the atfork trio exactly once per process */
98 (void) pthread_atfork(forksafe_prepare, forksafe_parent,
99 forksafe_child);
100 installed = 1;
101 }
102 for (i = 0; i < BRIX_FORKSAFE_MAX; i++) {
103 if (g_fs_reg[i] == NULL) {
104 g_fs_reg[i] = c;
105 pthread_mutex_unlock(&g_fs_mx);
106 return;
107 }
108 }
109 g_fs_overflow++; /* unregistered ⇒ un-neutered; capacity is generous */
110 pthread_mutex_unlock(&g_fs_mx);
111}

113void
114brix_forksafe_unregister(brix_conn *c)
115{
116 int i;
117
118 pthread_mutex_lock(&g_fs_mx);
119 for (i = 0; i < BRIX_FORKSAFE_MAX; i++) {
120 if (g_fs_reg[i] == c) {
121 g_fs_reg[i] = NULL;
122 break;
123 }
124 }
125 pthread_mutex_unlock(&g_fs_mx);
126}
