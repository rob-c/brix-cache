15ngx_int_t
16brix_imp_init_worker(ngx_cycle_t *cycle)
17{
18 char sockbuf[256];
19
20 /* Load the grid-mapfile before the map-mode early return: the worker-side
21 * authorization gate applies in off, single and map modes. */
22 if (imp_settings.gridmap.len > 0) {
23 char mapbuf[PATH_MAX];
24
25 ngx_snprintf((u_char *) mapbuf, sizeof(mapbuf), "%V%Z",
26 &imp_settings.gridmap);
27 if (brix_idmap_gate_init(mapbuf, cycle->log) != NGX_OK) {
28 BRIX_LOG(NGX_LOG_EMERG, cycle->log, 0,
29 "impersonate: authz-gate grid-mapfile load failed (\"%s\")",
30 mapbuf);
31 return NGX_ERROR;
32 }
33 }
34
35 if (imp_settings.mode != BRIX_IMP_MAP) {
36 return NGX_OK;
37 }
38
39 ngx_snprintf((u_char *) sockbuf, sizeof(sockbuf), "%V%Z",
40 &imp_settings.socket);
41 if (brix_imp_client_connect(sockbuf, cycle->log) != NGX_OK) {
42 /* Broker may still be starting; the client retries lazily per op. */
43 BRIX_LOG(NGX_LOG_WARN, cycle->log, 0,
44 "impersonate: broker not reachable yet at \"%s\" "
45 "(will reconnect on first op)", sockbuf);
46 }
47 return NGX_OK;
48}

63void
64brix_imp_request_begin(const brix_identity_t *id)
65{
66 if (imp_settings.mode != BRIX_IMP_MAP) {
67 return;
68 }
69 /* Mark the request active before setting the principal: an empty identity
70 * must route to the broker and deny, never fall back to the worker. */
71 brix_imp_mark_in_request(1);
72 brix_imp_set_principal(id != NULL ? imp_principal_of(id) : NULL);
73}
