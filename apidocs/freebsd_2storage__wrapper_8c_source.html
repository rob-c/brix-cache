11/* WHAT: Permit sendfile only for a known non-ZFS filesystem on this fd.
12 * WHY: Native OpenZFS 2.4.2 and 2.4.4 deadlock page-busy/range locks when
13 * sendfile races a write; a read-only handle cannot exclude other writers.
14 * HOW: 1. Query the open fd, never its pathname. 2. Refuse ZFS or an unknown
15 * filesystem. 3. Restore errno: this is a transport choice, not an error. */
16int
17brix_plat_file_sendfile_safe(int fd)
18{
19 struct statfs filesystem;
20 int saved = errno;
21 int safe = fstatfs(fd, &filesystem) == 0
22 && filesystem.f_fstypename[0] != '\0'
23 && strcmp(filesystem.f_fstypename, "zfs") != 0;
24
25 errno = saved;
26 return safe;
27}

90/* WHAT: Return a close-on-exec readable alias of an already-open file.
91 * WHY: FreeBSD has no Linux procfs reopen operation for unlinked write-only fds.
92 * HOW: Reopen F_KINFO's name and check identity; otherwise duplicate only a
93 * readable original, matching the existing BSD alias fallback contract. */
94int
95brix_plat_fd_reopen_readonly(int descriptor)
96{
97 struct kinfo_file info;
98 int reopened, flags;
99
100 memset(&info, 0, sizeof(info));
101 info.kf_structsize = sizeof(info);
102 if (fcntl(descriptor, F_KINFO, &info) == 0 && info.kf_path[0] == '/') {
103 reopened = open(info.kf_path, O_RDONLY | O_CLOEXEC | O_NOFOLLOW); /* vfs-seam-allow: NOT_STORAGE - native alias of an already-open file */
104 if (reopened >= 0) {
105 if (brix_freebsd_same_file(descriptor, reopened)) { return reopened; }
106 close(reopened);
107 }
108 }
109 flags = fcntl(descriptor, F_GETFL);
110 if (flags >= 0 && (flags & O_ACCMODE) != O_WRONLY) {
111 return fcntl(descriptor, F_DUPFD_CLOEXEC, 0);
112 }
113 if (flags >= 0) { errno = EACCES; }
114 return -1;
115}
