7/* 1 = the upload of local FILE lc (stat SB) to remote rc may be skipped
8 * because the remote destination is up to date under --sync; 0 = copy.
9 *
10 * WHY: a failed remote stat (missing file, error) or a directory in the way
11 * falls through to the copy — never skip on an undeterminable compare. Runs
12 * before the dry-run print so -r --sync --dry-run only lists files that
13 * would actually be transferred. */
14static int
15ul_sync_skip(const copy_walk_ctx *w, const char *rc, const char *lc,
16 const struct stat *sb)
17{
18 brix_statinfo si;
19 brix_status sst;
20
21 if (!w->o->sync) {
22 return 0;
23 }
24 brix_status_clear(&sst);
25 return brix_stat(w->c, rc, &si, &sst) == 0 && !(si.flags & kXR_isDir)
26 && brix_sync_should_skip(w->o->sync_cmp,
27 (long long) sb->st_size,
28 (long long) sb->st_mtime,
29 (long long) si.size,
30 (long long) si.mtime)
31 && (w->o->sync_cmp != XRDC_SYNC_CKSUM
32 || sync_cksum_match(w->c, rc, lc, w->o));
33}

36/* Transfer one local regular FILE lc (stat SB) to the remote path rc
37 * (upload walk).
38 *
39 * HOW: filter → sync-skip → dry-run guard → copy_one_l2r → --remove-source.
40 * Returns 0 (copied or skipped) or -1 (fatal, w->st set by copy_one_l2r). */
41static int
42tree_ul_file(const copy_walk_ctx *w, const char *relc,
43 const char *lc, const char *rc, const struct stat *sb)
44{
45 if (!brix_copy_filter_match(w->o, relc)) {
46 return 0;
47 }
48 if (ul_sync_skip(w, rc, lc, sb)) {
49 return 0; /* up-to-date — skip */
50 }
51 if (w->o->dry_run) {
52 printf("[dry-run] copy %s -> %s\n", lc, rc);
53 return 0;
54 }
55 if (copy_one_l2r(w->c, lc, rc, w->o, w->st) != 0) {
56 return -1;
57 }
58 if (w->o->remove_source && !w->o->dry_run) {
59 (void) unlink(lc);
60 }
61 return 0;
62}

65/* Handle one local directory entry NAME of the upload walk.
66 *
67 * HOW: skip dots; build the copy-root-relative path (relc) plus the local
68 * and remote child paths; lstat (not stat) so symlinks are detected, not
69 * followed — a link to a parent directory would otherwise drive unbounded
70 * recursion; symlinks are skipped (loop-safe; mirrors official -r default).
71 * Directories recurse via copy_tree_upload (then best-effort rmdir under
72 * --remove-source); regular files go through tree_ul_file. Returns 0 to
73 * continue the walk, -1 fatal (w->st set). */
74static int
75tree_ul_entry(const copy_walk_ctx *w, const char *name)
76{
77 char relc[XRDC_PATH_MAX];
78 char lc[XRDC_PATH_MAX], rc[XRDC_PATH_MAX];
79 struct stat sb;
80
81 if (dirent_is_dot(name)) {
82 return 0;
83 }
84 /* Build the path of this entry relative to the copy root so that
85 * brix_copy_filter_match can match at both full-rel and basename. */
86 if (rel_join(w->rel, name, relc, sizeof(relc)) != 0
87 || path_join(w->lpath, name, lc, sizeof(lc)) != 0
88 || path_join(w->rpath, name, rc, sizeof(rc)) != 0) {
89 brix_status_set(w->st, XRDC_EUSAGE, 0,
90 "recursive copy: path too long under %s", w->lpath);
91 return -1;
92 }
93 if (lstat(lc, &sb) != 0) {
94 return 0; /* vanished between readdir and stat — skip */
95 }
96 if (S_ISLNK(sb.st_mode)) {
97 return 0; /* skip symlinks (loop-safe; mirrors official -r default) */
98 }
99 if (S_ISDIR(sb.st_mode)) {
100 copy_walk_ctx cw = { w->c, rc, lc, relc, w->o, w->st };
101 if (copy_tree_upload(&cw) != 0) {
102 return -1;
103 }
104 if (w->o->remove_source && !w->o->dry_run) {
105 (void) rmdir(lc);
106 }
107 return 0;
108 }
109 if (S_ISREG(sb.st_mode)) {
110 return tree_ul_file(w, relc, lc, rc, &sb);
111 }
112 return 0; /* sockets/fifos/devices — not copied */
113}

116/* Recurse a local tree (lpath) into a remote tree (rpath) under conn c.
117 *
118 * WHAT: uploads every regular file under lpath to the mirrored remote path,
119 * applying --exclude/--include filters and honoring --dry-run and --sync
120 * (size/mtime/cksum up-to-date skip per o->sync_cmp).
121 * WHY: same rel-threading rationale as copy_tree_download — filter patterns
122 * must see the full relative path so they behave consistently at depth.
123 * HOW: skip brix_mkdir in dry-run mode; delegate each readdir entry to
124 * tree_ul_entry (filter → sync-skip → dry-run → copy_one_l2r). Sync
125 * runs before dry-run so -r --sync --dry-run only lists files that
126 * would actually be transferred. */
127int
128copy_tree_upload(const copy_walk_ctx *w)
129{
130 DIR *d;
131 struct dirent *de;
132 brix_status mst;
133
134 brix_status_clear(&mst);
135 if (!w->o->dry_run) {
136 (void) brix_mkdir(w->c, w->rpath, 0755, 1 /*parents*/, &mst); /* may already exist */
137 }
138 d = opendir(w->lpath);
139 if (d == NULL) {
140 brix_status_set(w->st, XRDC_ESOCK, errno, "opendir %s: %s",
141 w->lpath, strerror(errno));
142 return -1;
143 }
144 while ((de = readdir(d)) != NULL) {
145 if (tree_ul_entry(w, de->d_name) != 0) {
146 closedir(d);
147 return -1;
148 }
149 }
150 closedir(d);
151 /* Mirror-delete pass: remove remote entries that the local tree no longer
152 * has. Runs only when --sync --delete is active; lists the remote
153 * directory here (post-upload) so the remote snapshot is fresh. */
154 if (w->o->sync_delete) {
155 mirror_delete_remote(w->c, w->rpath, w->lpath, w->rel, w->o);
156 }
157 return 0;
158}

161/* Build the recursive copy's destination root from the source path, using
162 * rsync-style trailing-slash semantics:
163 *
164 * src WITHOUT a trailing slash ("dir") -> NEST the tree under the source's
165 * last path component: <dst>/<basename(dir)>/... This matches stock
166 * `xrdcp -r <dir> <dst>` and avoids silently merging two differently-named
167 * source trees into one flattened destination.
168 * src WITH a trailing slash ("dir/") -> FLAT mirror: copy the CONTENTS of
169 * the source straight into <dst>/... (no extra basename level), exactly as
170 * `rsync dir/ dst/` does.
171 *
172 * A degenerate basename ('.', '/', or empty — e.g. the whole-export `//.`/`//`
173 * forms) has no meaningful name to nest under, so the destination is likewise
174 * used verbatim. Returns 0 on success, -1 if the composed path would overflow. */
175int
176recursive_dest_root(const char *dstdir, const char *srcpath,
177 char *out, size_t outsz)
178{
179 size_t len = strlen(srcpath);
180 const char *base;
181 size_t blen, dl, i;
182 const char *sep;
183
184 /* Trailing slash on the source => flat mirror (copy contents into <dst>). */
185 if (len > 0 && srcpath[len - 1] == '/') {
186 return ((size_t) snprintf(out, outsz, "%s", dstdir) >= outsz) ? -1 : 0;
187 }
188
189 base = srcpath;
190 for (i = len; i > 0; i--) {
191 if (srcpath[i - 1] == '/') { base = srcpath + i; break; }
192 }
193 blen = (size_t) (srcpath + len - base);
194
195 if (blen == 0 || (blen == 1 && base[0] == '.')) { /* nothing to nest */
196 return ((size_t) snprintf(out, outsz, "%s", dstdir) >= outsz) ? -1 : 0;
197 }
198 dl = strlen(dstdir);
199 sep = (dl > 0 && dstdir[dl - 1] == '/') ? "" : "/";
200 return ((size_t) snprintf(out, outsz, "%s%s%.*s", dstdir, sep,
201 (int) blen, base) >= outsz) ? -1 : 0;
202}

205/* Recursive copy entry: connect once, walk the source tree. Direction-aware. */
206int
207copy_recursive(const copy_recurse_req *rq, brix_status *st)
208{
209 const brix_url *su = rq->su;
210 const brix_copy_opts *o = rq->o;
211 brix_conn c;
212 int rc;
213 char destroot[XRDC_PATH_MAX];
214
215 /* Nest under the source basename (stock parity); see recursive_dest_root. */
216 if (recursive_dest_root(rq->du->path, su->path, destroot,
217 sizeof(destroot)) != 0) {
218 brix_status_set(st, XRDC_EUSAGE, 0,
219 "recursive copy: destination path too long");
220 return -1;
221 }
222
223 if (rq->download) {
224 copy_walk_ctx w = { &c, su->path, destroot, "", o, st };
225 if (brix_connect(&c, su, rq->co, st) != 0) { return -1; }
226 rc = copy_tree_download(&w);
227 /* Best-effort: remove the source root dir when the whole tree succeeded.
228 * The walker already removed each file and subdir, so the root will only
229 * succeed when nothing was filtered; failure is silently ignored. */
230 if (rc == 0 && o->remove_source && !o->dry_run) {
231 brix_status rst;
232 brix_status_clear(&rst);
233 (void) brix_rmdir(&c, su->path, &rst);
234 }
235 } else {
236 copy_walk_ctx w = { &c, destroot, su->path, "", o, st };
237 if (brix_connect(&c, rq->du, rq->co, st) != 0) { return -1; }
238 rc = copy_tree_upload(&w);
239 /* Best-effort: remove the local source root after a fully-successful
240 * upload walk (only succeeds if the directory is now empty). */
241 if (rc == 0 && o->remove_source && !o->dry_run) {
242 (void) rmdir(su->path);
243 }
244 }
245 brix_close(&c);
246 return rc;
247}
