1/*
2 * sd_ceph_meta.c — the two object-metadata slots of the flat RADOS driver:
3 * `setattr` (advisory POSIX mode/owner/mtime) and `query_checksum` (a CRC32C the
4 * OSDs compute over the object's own bytes, without shipping them).
5 *
6 * WHY THIS FILE EXISTS
7 * Both slots were NULL on this driver, and both had a visible consequence.
8 * A NULL `setattr` makes chmod/kXR_setattr over a ceph:// export a silent
9 * no-op — the VFS treats the missing slot as "nothing to do", so the client is
10 * told the change succeeded and the mode reads back unchanged. A NULL
11 * `query_checksum` makes every checksum request a full-object read back across
12 * the cluster network to hash bytes the OSDs can hash in place.
13 *
14 * They live together because they are the same shape: one object, addressed by
15 * its oid, interrogated or amended WITHOUT an open handle in the namespace
16 * case. They are in a file of their own rather than appended to sd_ceph_io.c
17 * or sd_ceph_object.c because both of those are already close to the 600-line
18 * cap.
19 *
20 * ADVISORY METADATA
21 * RADOS has no POSIX metadata: an object has bytes, an mtime and xattrs, and
22 * nothing else. The approved model (see meta_advisory.h) is to persist mode /
23 * uid+gid / mtime as one reserved-xattr string and overlay it on stat, exactly
24 * as the S3 backend persists the same string in x-amz-meta-xrd-unixattr. The
25 * translation from the setattr request to that string is shared with the S3
26 * backend (meta_advisory_sd.h) so the two cannot disagree about what
27 * UTIME_OMIT, UTIME_NOW or a lone (uid_t)-1 mean.
28 *
29 * CHECKSUM CONDITIONING — the part that is easy to get silently wrong
30 * The OSD's CRC32C op is a table-driven CRC32C *update* seeded with the
31 * caller's init value, with no pre- or post-conditioning of its own. The
32 * canonical "crc32c" this project speaks (XRootD's XrdCksCalccrc32C, and the
33 * S3 x-amz-checksum-crc32c value) is the standard Castagnoli CRC32C, which is
34 * that same update seeded with 0xFFFFFFFF and finished by XOR with 0xFFFFFFFF.
35 * So the seed and the final XOR below are not decoration; drop either and the
36 * driver hands back a confident, authoritative, wrong digest. Every path that
37 * cannot establish the value with certainty returns NGX_DECLINED instead —
38 * per the slot contract a decline costs one byte-reading recompute, while a
39 * wrong digest is presented to the client as fact.
40 */

47/*
48 * WHAT: Turn one OSD CRC32C reply buffer into lowercase hex.
49 * WHY: Split out of the slot, and left OUTSIDE the BRIX_HAVE_CEPH gate, so the
50 * conditioning arithmetic above is exercised by the cluster-free unit
51 * suite (sd_ceph_unittest.c) rather than only by a lab with a live pool.
52 * This is the half that decides whether the digest is right.
53 * HOW: The reply is a little-endian uint32 count followed by that many
54 * little-endian CRC32C values. Chunking is not requested, so the only
55 * shape this driver can interpret is a count of exactly one; anything
56 * else — a short buffer, a chunked reply from a future OSD, a count of
57 * zero — is reported as uninterpretable (-1) and declined by the caller.
58 * Returns 0 on success with `hex` NUL-terminated, -1 otherwise.
59 */
60int
61sd_ceph_ck_crc32c_hex(const unsigned char *reply, size_t len, char *hex,
62 size_t cap)
63{
64 uint32_t count, crc;
65
66 if (reply == NULL || hex == NULL || cap < 9 || len < 8) {
67 return -1;
68 }
69 count = (uint32_t) reply[0] | ((uint32_t) reply[1] << 8)
70 | ((uint32_t) reply[2] << 16) | ((uint32_t) reply[3] << 24);
71 if (count != 1 || len < 4 + 4 * (size_t) count) {
72 return -1;
73 }
74 crc = (uint32_t) reply[4] | ((uint32_t) reply[5] << 8)
75 | ((uint32_t) reply[6] << 16) | ((uint32_t) reply[7] << 24);
76
77 /* Post-conditioning: see the file header. The OSD returns the raw running
78 * value; the standard CRC32C is that value XOR 0xFFFFFFFF. */
79 crc ^= 0xFFFFFFFFu;
80
81 (void) snprintf(hex, cap, "%08x", (unsigned) crc);
82 return 0;
83}

97/*
98 * WHAT: Apply a setattr request to `path` by patching its advisory blob.
99 * WHY: The ioctx-explicit core (see sd_ceph_internal.h): in RADOS the ioctx IS
100 * the identity asserted at the OSDs, so the credential-scoped twin must be
101 * able to run this exact body on the CALLER's connection. A body that
102 * reached st->ioctx would write the caller's metadata under the export's
103 * authority.
104 * HOW: Read the current blob, patch in the representable fields, write it back.
105 * The read is where the object's existence is established: a missing xattr
106 * and a missing object both come back as -ENODATA/-ENOENT depending on the
107 * OSD, so an absent blob is disambiguated with a rados_stat before it is
108 * treated as "empty" — otherwise setattr on a path that does not exist
109 * would CREATE an object carrying nothing but a mode, and stat would
110 * thereafter report a file that was never written.
111 */
112ngx_int_t
113sd_ceph_setattr_io(sd_ceph_state_t *st, rados_ioctx_t io, const char *path,
114 const brix_sd_setattr_t *attr)
115{
116 brix_meta_advisory_t delta;
117 char oid[1024];
118 char blob[SD_CEPH_ADVISORY_MAX];
119 uint64_t size = 0;
120 time_t mtime = 0;
121 int n;
122
123 if (attr == NULL) {
124 errno = EINVAL;
125 return NGX_ERROR;
126 }
127 /* Nothing the advisory model can represent (an atime-only request, say) is
128 * success without a write — and it is decided before any cluster round
129 * trip. */
130 if (!brix_meta_advisory_from_setattr(attr, &delta)) {
131 return NGX_OK;
132 }
133 if (sd_ceph_key(st->key_prefix, path, oid, sizeof(oid)) != 0) {
134 return NGX_ERROR;
135 }
136
137 n = rados_getxattr(io, oid, BRIX_META_ADVISORY_XATTR, blob,
138 sizeof(blob) - 1);
139 if (n < 0) {
140 if (n != -ENODATA && n != -ENOENT) {
141 errno = -n;
142 return NGX_ERROR;
143 }
144 /* No blob yet — but is there an OBJECT? Never fabricate one. */
145 if (sd_ceph_set_errno(rados_stat(io, oid, &size, &mtime))) {
146 return NGX_ERROR;
147 }
148 n = 0;
149 }
150 blob[n] = '\0';
151
152 if (brix_meta_advisory_patch(blob, sizeof(blob), &delta) < 0) {
153 errno = EIO;
154 return NGX_ERROR;
155 }
156 if (sd_ceph_set_errno(rados_setxattr(io, oid, BRIX_META_ADVISORY_XATTR,
157 blob, strlen(blob))))
158 {
159 return NGX_ERROR;
160 }
161 return NGX_OK;
162}

179/*
180 * WHAT: The `query_checksum` vtable slot — answer a crc32c request from the
181 * OSDs instead of reading the object back to hash it.
182 * WHY: A checksum over a ceph:// export otherwise pulls the whole object
183 * across the cluster network for a value the OSDs can compute where the
184 * bytes already are. For a WLCG dataset that is the entire transfer, paid
185 * again.
186 * HOW: A rados_stat first, then one read op carrying the checksum action over
187 * [0, size). The stat is not redundant: the length to checksum has to be
188 * fixed when the op is BUILT, and obj->state's cached size was taken at
189 * open — checksumming a stale length would digest a prefix of the object
190 * and present it as the whole. It also establishes existence, so a missing
191 * object is ENOENT rather than a checksum of nothing.
192 *
193 * Only "crc32c" is offered: the OSD also computes xxhash32/xxhash64, and
194 * neither is in this project's canonical algorithm set. A STRIPED object
195 * declines — its bytes live in sibling stripe objects, and a checksum of
196 * the head object alone would be the digest of a fraction of the file
197 * presented as the digest of the whole. Worker-safe: it touches only the
198 * object state and its own stack, exactly like sd_ceph_pread.
199 */
200ngx_int_t
201sd_ceph_query_checksum(brix_sd_obj_t *obj, const char *algo, char *hex_out,
202 size_t hex_sz)
203{
204 sd_ceph_obj_state_t *os;
205 rados_read_op_t op;
206 unsigned char reply[64];
207 /* Seed: the standard CRC32C pre-conditioning — see the file header. */
208 const unsigned char init[4] = { 0xFF, 0xFF, 0xFF, 0xFF };
209 uint64_t size = 0;
210 time_t mtime = 0;
211 int prval = 0;
212 int rc;
213
214 if (obj == NULL || obj->state == NULL || algo == NULL || hex_out == NULL
215 || hex_sz < 9)
216 {
217 return NGX_DECLINED;
218 }
219 if (strcmp(algo, "crc32c") != 0) {
220 return NGX_DECLINED; /* the OSD holds no other canonical digest */
221 }
222 os = obj->state;
223 if (os->striped) {
224 return NGX_DECLINED; /* head object is not the file */
225 }
226 if (sd_ceph_set_errno(rados_stat(os->ioctx, os->oid, &size, &mtime))) {
227 return NGX_ERROR;
228 }
229 if (size == 0) {
230 /* CRC32C of no bytes is 0xFFFFFFFF ^ 0xFFFFFFFF; asking the OSD for a
231 * zero-length checksum returns an empty reply we would have to
232 * decline. */
233 (void) snprintf(hex_out, hex_sz, "%08x", 0u);
234 return NGX_OK;
235 }
236
237 op = rados_create_read_op();
238 if (op == NULL) {
239 return NGX_ERROR;
240 }
241 /* Zeroed first: the OSD fills only as much of `reply` as it produced, and a
242 * short or absent reply must parse as uninterpretable rather than as
243 * whatever the stack happened to hold. */
244 memset(reply, 0, sizeof(reply));
245 rados_read_op_checksum(op, LIBRADOS_CHECKSUM_TYPE_CRC32C,
246 (const char *) init, sizeof(init),
247 0, (size_t) size, 0,
248 (char *) reply, sizeof(reply), &prval);
249 rc = rados_read_op_operate(op, os->ioctx, os->oid, 0);
250 rados_release_read_op(op);
251
252 if (rc < 0 || prval < 0) {
253 errno = -(rc < 0 ? rc : prval);
254 return NGX_ERROR; /* the caller recomputes from the bytes */
255 }
256 if (sd_ceph_ck_crc32c_hex(reply, sizeof(reply), hex_out, hex_sz) != 0) {
257 return NGX_DECLINED; /* uninterpretable: never guess a digest */
258 }
259 return NGX_OK;
260}
