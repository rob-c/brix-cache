93/* brix_srv_count_servers — §2.2: occupied data-serving slots. Managers
94 * ("M"), supervisors ("R") and peers ("P") are cluster control/overflow
95 * capacity, not data servers, so they do not count toward the floor. */
96ngx_uint_t
97brix_srv_count_servers(void)
98{
99 brix_srv_table_t *tbl;
100 brix_srv_entry_t *e;
101 ngx_uint_t i, n = 0;
102
103 tbl = srv_table();
104 if (tbl == NULL) {
105 return 0;
106 }
107
108 ngx_shmtx_lock(&brix_srv_mutex);
109 for (i = 0; i < tbl->capacity; i++) {
110 e = &tbl->slots[i];
111 if (e->in_use && (e->role[0] == 'S'
112 || (e->role[0] == 'P' && e->role[1] == 'S')))
113 {
114 n++;
115 }
116 }
117 ngx_shmtx_unlock(&brix_srv_mutex);
118 return n;
119}

129/*
130 * brix_srv_find_supervisor — §2.9: pick the least-utilised live supervisor.
131 *
132 * WHAT: Scans in-use, non-blacklisted "R"-role entries and returns the one
133 * with the lowest util_pct (first-seen wins ties). 1 = found.
134 * WHY: A manager at its brix_cms_server_max_direct cap redirects a new
135 * server login to a supervisor (kYR_try at login) — tree formation
136 * needs a target, and the coolest supervisor spreads the subtree load.
137 * HOW: Same locked scan shape as the other selectors, filtered to role "R".
138 */
139int
140brix_srv_find_supervisor(char *host_out, size_t host_size,
141 uint16_t *port_out)
142{
143 brix_srv_table_t *tbl;
144 brix_srv_entry_t *e;
145 ngx_uint_t i;
146 int best = -1;
147 uint32_t best_util = 0;
148
149 tbl = srv_table();
150 if (tbl == NULL) {
151 return 0;
152 }
153
154 ngx_shmtx_lock(&brix_srv_mutex);
155
156 for (i = 0; i < tbl->capacity; i++) {
157 e = &tbl->slots[i];
158 if (!e->in_use || e->role[0] != 'R') {
159 continue;
160 }
161 if (e->blacklisted_until != 0
162 && e->blacklisted_until > ngx_current_msec)
163 {
164 continue;
165 }
166 if (best == -1 || e->util_pct < best_util) {
167 best = (int) i;
168 best_util = e->util_pct;
169 }
170 }
171
172 if (best >= 0) {
173 e = &tbl->slots[best];
174 ngx_cpystrn((u_char *) host_out, (u_char *) e->host, host_size);
175 *port_out = e->port;
176 }
177
178 ngx_shmtx_unlock(&brix_srv_mutex);
179 return best >= 0;
180}
