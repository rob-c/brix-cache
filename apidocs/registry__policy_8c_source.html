172/* brix_srv_count_servers — §2.2: occupied data-serving slots. Managers
173 * ("M"), supervisors ("R") and peers ("P") are cluster control/overflow
174 * capacity, not data servers, so they do not count toward the floor. */
175ngx_uint_t
176brix_srv_count_servers(void)
177{
178 brix_srv_table_t *tbl;
179 brix_srv_entry_t *e;
180 ngx_uint_t i, n = 0;
181
182 tbl = srv_table();
183 if (tbl == NULL) {
184 return 0;
185 }
186
187 ngx_shmtx_lock(brix_srv_get_mutex());
188 for (i = 0; i < tbl->capacity; i++) {
189 e = &tbl->slots[i];
190 if (e->in_use && (e->role[0] == 'S'
191 || (e->role[0] == 'P' && e->role[1] == 'S')))
192 {
193 n++;
194 }
195 }
196 ngx_shmtx_unlock(brix_srv_get_mutex());
197 return n;
198}

209/*
210 * brix_srv_find_supervisor — §2.9: pick the least-utilised live supervisor.
211 *
212 * WHAT: Scans in-use, non-blacklisted "R"-role entries and returns the one
213 * with the lowest util_pct (first-seen wins ties). 1 = found.
214 * WHY: A manager at its brix_cms_server_max_direct cap redirects a new
215 * server login to a supervisor (kYR_try at login) — tree formation
216 * needs a target, and the coolest supervisor spreads the subtree load.
217 * HOW: Same locked scan shape as the other selectors, filtered to role "R".
218 */
219int
220brix_srv_find_supervisor(char *host_out, size_t host_size,
221 uint16_t *port_out)
222{
223 brix_srv_table_t *tbl;
224 brix_srv_entry_t *e;
225 ngx_uint_t i;
226 int best = -1;
227 uint32_t best_util = 0;
228
229 tbl = srv_table();
230 if (tbl == NULL) {
231 return 0;
232 }
233
234 ngx_shmtx_lock(brix_srv_get_mutex());
235
236 for (i = 0; i < tbl->capacity; i++) {
237 e = &tbl->slots[i];
238 if (!e->in_use || e->role[0] != 'R') {
239 continue;
240 }
241 if (e->blacklisted_until != 0
242 && e->blacklisted_until > ngx_current_msec)
243 {
244 continue;
245 }
246 if (best == -1 || e->util_pct < best_util) {
247 best = (int) i;
248 best_util = e->util_pct;
249 }
250 }
251
252 if (best >= 0) {
253 e = &tbl->slots[best];
254 ngx_cpystrn((u_char *) host_out, (u_char *) e->host, host_size);
255 *port_out = e->port;
256 }
257
258 ngx_shmtx_unlock(brix_srv_get_mutex());
259 return best >= 0;
260}
