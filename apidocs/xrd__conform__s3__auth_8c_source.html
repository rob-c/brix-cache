83 const char *region, time_t signed_at, unsigned expires,
84 brix_weburl *presigned)
85{

159static void
160probe_presigned_urls(xrd_conform_execution *execution,
161 const brix_weburl *url)
162{
163 const char *access;
164 const char *secret;
165 const char *region;
166 brix_weburl valid, expired, replay;
167 brix_status status;
168 unsigned valid_status = 0, expired_status = 0, replay_status = 0;
169 time_t now = time(NULL);
170 int credential_state;
171 int valid_rc, expired_rc, replay_rc;
172 int written;
173
174 if (!xrd_conform_row_active(execution, "S3-006")
175 && !xrd_conform_row_active(execution, "S3-007")) return;
176 credential_state = xrd_conform_s3_credentials(&access, &secret, &region);
177 if (credential_state <= 0 || now == (time_t) -1
178 || presigned_url(url, access, secret, region, now, 300, &valid) != 0
179 || presigned_url(url, access, secret, region, now - 3600, 1,
180 &expired) != 0) {
181 presigned_unavailable(execution);
182 return;
183 }
184 replay = *url;
185 written = snprintf(replay.path, sizeof(replay.path),
186 "%s.brix-conform-presign-replay%s", url->path,
187 strchr(valid.path, '?'));
188 if (written < 0 || (size_t) written >= sizeof(replay.path)) return;
189 valid_rc = presigned_get(execution, &valid, &valid_status, &status);
190 expired_rc = presigned_get(execution, &expired, &expired_status, &status);
191 replay_rc = presigned_get(execution, &replay, &replay_status, &status);
192 grade_presigned_expiry(execution, valid_rc, valid_status, expired_rc,
193 expired_status);
194 grade_presigned_replay(execution, valid_rc, valid_status, replay_rc,
195 replay_status);
196}

198static void
199probe_auth_confusion(xrd_conform_execution *execution,
200 const brix_weburl *url)
201{
202 char headers[2048];
203 char *token = NULL;
204 brix_weburl signed_url = *url;
205 brix_status status;
206 unsigned http_status = 0;
207 int rc;
208 int written;
209
210 if (!xrd_conform_row_active(execution, "S3-008")) return;
211 if (url->is_s3) {
212 token = brix_token_discover();
213 written = token == NULL ? -1 : snprintf(headers, sizeof(headers),
214 "Authorization: Bearer %s\r\n", token);
215 if (!xrd_conform_credential_safe(token) || written < 0
216 || (size_t) written >= sizeof(headers)) {
217 free(token);
218 return;
219 }
220 rc = request_with_headers(execution, url, "HEAD",
221 headers, 0, &http_status, &status);
222 free(token);
223 if (rc == 0)
224 execution->s3_bearer_state = auth_rejected(http_status) ? -1 : 1;
225 return;
226 }
227 signed_url.is_s3 = 1;
228 rc = xrd_conform_s3_auth_headers(&signed_url, "HEAD", headers);
229 if (rc != 0 || headers[0] == '\0') return;
230 rc = request_with_headers(execution, url, "HEAD", headers, 0,
231 &http_status, &status);
232 if (rc == 0)
233 execution->http_sigv4_state = auth_rejected(http_status) ? -1 : 1;
234}

269/* WHAT: Return one for AccessDenied, zero for wrong, or minus one for transport.
270 * WHY: The caller must distinguish a divergent answer from missing evidence.
271 * HOW: Send one bounded PUT and inspect its status and exact XML code. */
272static int
273readonly_put_refused(xrd_conform_execution *execution,
274 const brix_weburl *scratch)
275{
276 static const char body[] = "R";
277 brix_http_resp response;
278 brix_status status;
279 int rc;
280 int refused;
281
282 memset(&response, 0, sizeof(response));
283 brix_status_clear(&status);
284 rc = xrd_conform_http_request_body(execution, scratch, "PUT", NULL, NULL,
285 body, sizeof(body) - 1, 4096, &response, &status);
286 refused = rc == 0 && response.status == 403 && response.body != NULL
287 && strstr(response.body, "<Code>AccessDenied</Code>") != NULL;
288 if (rc == 0) brix_http_resp_free(&response);
289 return rc == 0 ? refused : -1;
290}

313/* ---- Observe the S3 half of the paired read-only assertion ----
314 *
315 * WHAT: Records exact AccessDenied refusal and post-refusal object absence.
316 * WHY: A cosmetic 403 is insufficient if the rejected PUT still mutates data.
317 * HOW: PUT one byte to a builder-confined key, require the AWS XML spelling,
318 * then GET the key and require it to remain absent.
319 */
320void
321xrd_conform_probe_s3_readonly(xrd_conform_execution *execution,
322 const brix_weburl *url)
323{
324 brix_weburl scratch = *url;
325 int refused;
326 int absent;
327
328 if (!url->is_s3 || !readonly_clause_selected(execution)
329 || (!xrd_conform_row_active(execution, "S3-009")
330 && !xrd_conform_row_active(execution, "XPR-006"))) return;
331 if (brix_conform_scratch_path(execution->run->config.scratch,
332 "brix-conform-s3-readonly", scratch.path,
333 sizeof(scratch.path)) != 0) return;
334 refused = readonly_put_refused(execution, &scratch);
335 absent = readonly_key_absent(execution, &scratch);
336 if (refused >= 0 && absent >= 0)
337 execution->s3_readonly_state = refused && absent ? -1 : 1;
338}

341 *
342 * WHAT: Resolves S3-009 after every primary and --also target has run.

348xrd_conform_finalize_s3_readonly(xrd_conform_execution *execution)
