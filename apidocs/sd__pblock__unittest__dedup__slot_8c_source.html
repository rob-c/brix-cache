1/*
2 * sd_pblock_unittest_dedup_slot.c — phase-88 W1 driver->dedup_publish slot
3 * (the pblock leg of the G13 commit-time-dedup seam), split from
4 * sd_pblock_unittest_dedup.c for the file-size cap. Three legs:
5 * SUCCESS — an object whose overlapping-write history cleared its
6 * commit-time hash is folded by the slot (on-demand CRC +
7 * byte-verified candidate) onto its byte-identical sibling.
8 * ERROR — a missing path is a clean ENOENT; a refs-off export is a
9 * clean ENOTSUP (never a silent wrong success).
10 * SECURITY-NEG — a forged blobs.content_hash cannot make the slot alias
11 * differing content (byte-verify rejects the candidate).
12 * Shared harness + q_blob_id/q_refcount-style catalog introspection are local
13 * (a second SQLite connection, as a pytest would).
14 */

65/* write_file_overlap — create `path` and write `data` with a deliberate
66 * OVERLAP (whole body, then the first byte again): final content == data, but
67 * the wverify accumulator degrades, so the close-time publish records the blob
68 * with a CLEARED hash — exactly the "commit-time fold missed" state the slot
69 * exists to repair. Returns 0 or -1. */
70static int
71write_file_overlap(brix_sd_instance_t *inst, const char *path,
72 const char *data, size_t len)
73{
74 int err = 0;
75 brix_sd_obj_t *o;
76 int ok;
77
78 o = D->open(inst, path,
79 BRIX_SD_O_WRITE | BRIX_SD_O_CREATE | BRIX_SD_O_TRUNC, 0644,
80 &err);
81 if (o == NULL) {
82 return -1;
83 }
84 ok = D->pwrite(o, data, len, 0) == (ssize_t) len
85 && D->pwrite(o, data, 1, 0) == 1; /* the overlap */
86 return pb_close(o) == NGX_OK && ok ? 0 : -1;
87}

89void
90test_dedup_slot(void)
91{
92 char root[] = "/tmp/pb_dslot.XXXXXX";
93 brix_sd_instance_t inst = {0};
94 brix_sd_pblock_conf_t conf = {0};
95 char ba[PBLOCK_BLOB_ID_CAP], bb[PBLOCK_BLOB_ID_CAP];
96 char hash[80], buf[64];
97 /* 10 bytes over a 4-byte stripe ⇒ 3 blocks: the on-demand CRC and the
98 * byte-verify both walk real striped blocks. */
99 const char *DATA = "abcdefghij";
100
101 CHECK(D->dedup_publish != NULL, "pblock advertises dedup_publish");
102 CHECK(D->dedup_gc == NULL, "pblock needs no alias GC (refcount-driven)");
103
104 CHECK(mkdtemp(root) != NULL, "mkdtemp");
105 lab_write_sidecar(root, "dedup=1");
106 conf.root = root;
107 conf.busy_timeout_ms = 2000;
108 conf.block_size = 4;
109 inst.driver = D;
110 inst.caps = D->caps;
111 CHECK(D->init(&inst, &conf) == NGX_OK, "slot init");
112
113 /* SUCCESS: /s1 committed cleanly (hash recorded); /s2 written with an
114 * overlap — same bytes, cleared hash, so commit-time dedup missed it and
115 * the blobs stay distinct. The slot computes the hash on demand,
116 * byte-verifies, and folds. */
117 CHECK(write_file(&inst, "/s1", DATA, 10) == 0, "seed s1");
118 CHECK(write_file_overlap(&inst, "/s2", DATA, 10) == 0, "seed s2 (overlap)");
119 slot_blob_id(root, "/s1", ba, sizeof(ba));
120 slot_blob_id(root, "/s2", bb, sizeof(bb));
121 CHECK(ba[0] && bb[0] && strcmp(ba, bb) != 0,
122 "cleared-hash commit left the blobs distinct");
123 slot_get_hash(root, bb, hash, sizeof(hash));
124 CHECK(hash[0] == '\0', "overlap history cleared s2's hash (got \"%s\")",
125 hash);
126
127 CHECK(D->dedup_publish(&inst, "/s2", "/.gcas/ab/cdef") == NGX_OK,
128 "slot publish folds");
129 slot_blob_id(root, "/s2", bb, sizeof(bb));
130 CHECK(strcmp(ba, bb) == 0, "slot folded s2 onto s1's blob");
131 CHECK(slot_refcount(root, ba) == 2, "folded blob refcount 2 (got %d)",
132 slot_refcount(root, ba));
133 CHECK(read_file(&inst, "/s1", buf, sizeof(buf)) == 10
134 && memcmp(buf, DATA, 10) == 0, "s1 reads back");
135 CHECK(read_file(&inst, "/s2", buf, sizeof(buf)) == 10
136 && memcmp(buf, DATA, 10) == 0, "s2 reads back");
137
138 /* Idempotence: publishing an already-shared object is a benign no-op. */
139 CHECK(D->dedup_publish(&inst, "/s2", "/.gcas/ab/cdef") == NGX_OK,
140 "re-publish is benign");
141 CHECK(slot_refcount(root, ba) == 2, "refcount unchanged (got %d)",
142 slot_refcount(root, ba));
143
144 /* ERROR: a missing path is a clean ENOENT. */
145 errno = 0;
146 CHECK(D->dedup_publish(&inst, "/absent", "/.gcas/ab/cdef") == NGX_ERROR
147 && errno == ENOENT, "missing path is ENOENT (errno %d)", errno);
148
149 /* SECURITY-NEG (legacy tier): pin both same-size different-content blobs
150 * onto ONE colliding "crc32:" hash — a non-sha256 candidate must be
151 * byte-verified, so the slot refuses the alias. */
152 CHECK(write_file(&inst, "/v1", "AAAAAAAA", 8) == 0, "seed v1");
153 CHECK(write_file_overlap(&inst, "/v2", "BBBBBBBB", 8) == 0, "seed v2");
154 slot_blob_id(root, "/v1", ba, sizeof(ba));
155 slot_blob_id(root, "/v2", bb, sizeof(bb));
156 slot_set_hash(root, ba, "crc32:00c0ffee");
157 slot_set_hash(root, bb, "crc32:00c0ffee");
158 CHECK(D->dedup_publish(&inst, "/v2", "/.gcas/ff/0000") == NGX_OK,
159 "legacy publish completes");
160 {
161 char b2[PBLOCK_BLOB_ID_CAP];
162
163 slot_blob_id(root, "/v2", b2, sizeof(b2));
164 CHECK(strcmp(b2, ba) != 0,
165 "colliding crc32 hash did NOT alias content");
166 }
167 CHECK(read_file(&inst, "/v2", buf, sizeof(buf)) == 8
168 && memcmp(buf, "BBBBBBBB", 8) == 0, "v2 content honest");
169 CHECK(read_file(&inst, "/v1", buf, sizeof(buf)) == 8
170 && memcmp(buf, "AAAAAAAA", 8) == 0, "v1 content honest");
171
172 /* W3 (sha tier): an in-order commit records a "sha256:" identity; two
173 * identical commits share it; the OUT-OF-ORDER twin records the legacy
174 * "crc32:" tier instead (no combine exists for SHA), so the tiers are
175 * exactly the write-order split the design names. */
176 CHECK(write_file(&inst, "/h1", DATA, 10) == 0, "seed h1");
177 slot_blob_id(root, "/h1", ba, sizeof(ba));
178 slot_get_hash(root, ba, hash, sizeof(hash));
179 CHECK(strncmp(hash, "sha256:", 7) == 0 && strlen(hash) == 7 + 64,
180 "in-order commit recorded a sha256 identity (got \"%s\")", hash);
181 {
182 /* Out-of-order but complete: tail then head — CRC combines, SHA
183 * cannot, so the recorded hash falls back to the legacy tier. */
184 int err = 0;
185 brix_sd_obj_t *o = D->open(&inst, "/h2",
186 BRIX_SD_O_WRITE | BRIX_SD_O_CREATE
187 | BRIX_SD_O_TRUNC, 0644, &err);
188
189 CHECK(o != NULL, "open h2 (err %d)", err);
190 if (o != NULL) {
191 CHECK(D->pwrite(o, "FGHIJ", 5, 5) == 5, "h2 tail first");
192 CHECK(D->pwrite(o, "ABCDE", 5, 0) == 5, "h2 head second");
193 CHECK(pb_close(o) == NGX_OK, "close h2");
194 }
195 slot_blob_id(root, "/h2", bb, sizeof(bb));
196 slot_get_hash(root, bb, hash, sizeof(hash));
197 CHECK(strncmp(hash, "crc32:", 6) == 0,
198 "out-of-order commit fell back to the crc tier (got \"%s\")",
199 hash);
200 }
201
202 D->cleanup(&inst);
203
204 /* ERROR (gate off): a refs-off export answers ENOTSUP, never a silent
205 * wrong success — the config layer keys brix_cache_global_cas off this. */
206 {
207 char root2[] = "/tmp/pb_dslotoff.XXXXXX";
208 brix_sd_instance_t inst2 = {0};
209 brix_sd_pblock_conf_t conf2 = {0};
210
211 CHECK(mkdtemp(root2) != NULL, "mkdtemp off");
212 conf2.root = root2; /* deliberately NO dedup opt */
213 conf2.busy_timeout_ms = 2000;
214 conf2.block_size = 4;
215 inst2.driver = D;
216 inst2.caps = D->caps;
217 CHECK(D->init(&inst2, &conf2) == NGX_OK, "off init");
218 CHECK(write_file(&inst2, "/x", DATA, 10) == 0, "seed x");
219 errno = 0;
220 CHECK(D->dedup_publish(&inst2, "/x", "/.gcas/ab/cdef") == NGX_ERROR
221 && errno == ENOTSUP, "refs-off is ENOTSUP (errno %d)", errno);
222 D->cleanup(&inst2);
223 }
224}
