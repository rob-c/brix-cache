1/*
2 * access_query.c - WebDAV access-phase query boundary.
3 *
4 * WHAT: Parses the request query exactly once, rejects ambiguous credential
5 * names, and prepares the XrdHttp compatibility context.
6 * WHY: Authentication must consume one canonical request-owned parse, while
7 * native nginx log sources are scrubbed before any later error can expose
8 * a query credential.
9 * HOW: Validate unique decoded names, reject encoded or repeated credential
10 * keys, then delegate XrdHttp parsing and TPC header injection.

16/* ---- Parse and validate the query authentication boundary ----
17 *
18 * WHAT: Return NGX_OK with xctx->query populated, NGX_HTTP_BAD_REQUEST for an
19 * ambiguous credential query, or NGX_HTTP_INTERNAL_SERVER_ERROR when the
20 * request context cannot be allocated.
21 * WHY: Alternate encodings and duplicate credential keys must not let two
22 * consumers authenticate different interpretations of one request.
23 * HOW: 1. Allocate the request context. 2. Parse and require unique names.
24 * 3. Require literal credential-key spellings and at most one token key.
25 * 4. Scrub query tokens on every refusal.
26 */
27ngx_int_t
28access_parse_query_boundary(ngx_http_request_t *r)
29{
30 xrdhttp_req_ctx_t *xctx = xrdhttp_get_ctx(r);
31 ngx_uint_t i;
32 ngx_uint_t token_params = 0;
33
34 if (xctx == NULL) {
35 brix_http_redact_request_query_tokens(r);
36 return NGX_HTTP_INTERNAL_SERVER_ERROR;
37 }
38 if (brix_http_query_parse(r->pool, r->args, &xctx->query) < 0
39 || brix_http_query_unique(&xctx->query,
40 BRIX_HTTP_QUERY_CASE_INSENSITIVE) < 0)
41 {
42 brix_http_redact_request_query_tokens(r);
43 return NGX_HTTP_BAD_REQUEST;
44 }
45 for (i = 0; i < xctx->query.count; i++) {
46 const brix_http_query_param_t *param = &xctx->query.params[i];
47 int is_authz = param->name.len == sizeof("authz") - 1
48 && ngx_strncasecmp(param->name.data, (u_char *) "authz",
49 param->name.len) == 0;
50 int is_access_token =
51 param->name.len == sizeof("access_token") - 1
52 && ngx_strncasecmp(param->name.data, (u_char *) "access_token",
53 param->name.len) == 0;
54
55 token_params += is_authz || is_access_token;
56 if ((is_authz || is_access_token)
57 && (param->raw_name.len != param->name.len
58 || ngx_strncasecmp(param->raw_name.data, param->name.data,
59 param->name.len) != 0))
60 {
61 brix_http_redact_request_query_tokens(r);
62 return NGX_HTTP_BAD_REQUEST;
63 }
64 }
65 if (token_params > 1) {
66 brix_http_redact_request_query_tokens(r);
67 return NGX_HTTP_BAD_REQUEST;
68 }
69 return NGX_OK;
70}

72/* ---- Prepare the compatibility request after authentication ----
73 *
74 * WHAT: Return NGX_OK after parsing XrdHttp metadata and injecting TPC
75 * headers, or the corresponding HTTP/internal error.
76 * WHY: The access orchestrator needs one checked operation instead of ignoring
77 * either parser's return value.
78 * HOW: Parse XrdHttp state first; only on success inject the TPC headers.
79 */
80ngx_int_t
81access_prepare_xrdhttp(ngx_http_request_t *r)
82{
83 ngx_int_t rc;
84
85 rc = xrdhttp_parse_request(r);
86 if (rc != NGX_OK) {
87 return rc;
88 }
89 return xrdhttp_inject_tpc_headers(r) == NGX_OK
90 ? NGX_OK : NGX_HTTP_INTERNAL_SERVER_ERROR;
91}
