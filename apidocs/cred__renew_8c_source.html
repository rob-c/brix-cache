22/*
23 * WHAT: Decide whether a credential expiring at `expires_at` needs renewing at
24 * `now`, given the `lead_secs` window.
25 * WHY: Called once per streamed chunk, so it must be cheap, total and free of
26 * side effects; every "unknown" input resolves to NOT_NEEDED so that
27 * enabling the feature can never turn a previously-working pull into a
28 * refusal on its own.
29 * HOW: Three early-return guards for the inputs that carry no decision, then
30 * the expiry test, then the remaining-lifetime comparison. The
31 * subtraction runs only after `expires_at > now` is established, so it
32 * cannot underflow however far apart the two clocks are.
33 */
34brix_tpc_renew_verdict_t
35brix_tpc_renew_verdict(time_t expires_at, time_t now, time_t lead_secs)
36{
37 time_t remaining;
38
39 if (expires_at <= 0) {
40 return BRIX_TPC_RENEW_NOT_NEEDED; /* no expiry is known */
41 }
42 if (lead_secs <= 0) {
43 return BRIX_TPC_RENEW_NOT_NEEDED; /* renewal disabled */
44 }
45 if (now <= 0) {
46 return BRIX_TPC_RENEW_NOT_NEEDED; /* no usable clock */
47 }
48
49 if (now >= expires_at) {
50 return BRIX_TPC_RENEW_EXPIRED;
51 }
52
53 remaining = expires_at - now; /* > 0 by the test above */
54 if (remaining <= lead_secs) {
55 return BRIX_TPC_RENEW_DUE;
56 }
57
58 return BRIX_TPC_RENEW_NOT_NEEDED;
59}

61/*
62 * WHAT: Report whether `token_mode` can mint a fresh credential.
63 * WHY: The renewal leg must fail closed rather than re-present a stale token
64 * when the mode is a forwarding one; making the authority an explicit
65 * allow-list means a new mode is un-mintable until someone adds it here
66 * deliberately.
67 * HOW: Exact string match against renew_minting_modes; NULL and the empty
68 * string short-circuit to 0.
69 */
70int
71brix_tpc_renew_mode_can_mint(const char *token_mode)
72{
73 size_t i;
74
75 if (token_mode == NULL || token_mode[0] == '\0') {
76 return 0;
77 }
78
79 for (i = 0; i < sizeof(renew_minting_modes) / sizeof(renew_minting_modes[0]);
80 i++)
81 {
82 if (strcmp(token_mode, renew_minting_modes[i]) == 0) {
83 return 1;
84 }
85 }
86
87 return 0;
88}

90/*
91 * WHAT: Map a verdict to a stable log string.
92 * WHY: Operator-facing log lines and the tests that read them need one spelling
93 * per verdict that does not drift with the enum.
94 * HOW: Switch with an explicit default so an unmapped future value is visible
95 * as "unknown" instead of falling off the end.
96 */
97const char *
98brix_tpc_renew_verdict_name(brix_tpc_renew_verdict_t verdict)
99{
100 switch (verdict) {
101 case BRIX_TPC_RENEW_NOT_NEEDED:
102 return "not-needed";
103 case BRIX_TPC_RENEW_DUE:
104 return "due";
105 case BRIX_TPC_RENEW_EXPIRED:
106 return "expired";
107 default:
108 return "unknown";
109 }
110}
