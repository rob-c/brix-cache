1/*
2 * api_snapshot_dns.c - the "dns" panel of the dashboard snapshot (phase-116).
3 *
4 * WHAT: dashboard_fill_dns() — one row per runtime-DNS target (directive,
5 * host, port, state, last error, next retry, counters) plus the
6 * per-state summary, read from the worker-local registry (net/dns).
7 * WHY: "why is this node not in the cluster" is answered here in one look:
8 * a brix_cms_manager / brix_upstream / mirror name that never resolved
9 * shows its state and the resolver's error text; the Prometheus gauge
10 * (net/dns/metrics.c) deliberately carries no error text.
11 * HOW: Mirrors the other dashboard_fill_* panels: populate keys on a
12 * caller-supplied object, tolerate jansson OOM (NULL children are
13 * dropped by json_object_set_new / json_array_append_new). Hostnames
14 * come from the operator's own configuration, so they are not redacted.
15 */

20static json_t *
21dashboard_dns_target_row(const brix_dns_target_t *t)
22{
23 json_t *o = json_object();
24 ngx_msec_t now = ngx_current_msec;
25
26 if (o == NULL) {
27 return NULL;
28 }
29 json_object_set_new(o, "directive", json_string(t->directive));
30 json_object_set_new(o, "host", json_string((const char *) t->host.data));
31 json_object_set_new(o, "port", json_integer((json_int_t) t->port));
32 json_object_set_new(o, "state", json_string(brix_dns_state_name(t->state)));
33 json_object_set_new(o, "literal", t->literal ? json_true() : json_false());
34 json_object_set_new(o, "addresses", json_integer((json_int_t) t->naddrs));
35 json_object_set_new(o, "last_error", json_string(t->last_error));
36 json_object_set_new(o, "next_retry_ms",
37 json_integer(t->next_retry > now ? (json_int_t) (t->next_retry - now) : 0));
38 json_object_set_new(o, "resolutions", json_integer((json_int_t) t->resolutions));
39 json_object_set_new(o, "failures", json_integer((json_int_t) t->failures));
40 json_object_set_new(o, "zone",
41 json_string(t->dns && t->dns->status_zone.data
42 ? (const char *) t->dns->status_zone.data : ""));
43 return o;
44}

47void
48dashboard_fill_dns(json_t *target)
49{
50 ngx_uint_t counts[BRIX_DNS_STATE_COUNT];
51 ngx_uint_t i, n = brix_dns_targets_n();
52 json_t *rows = json_array();
53
54 brix_dns_targets_count(counts);
55 json_object_set_new(target, "targets_total", json_integer((json_int_t) n));
56 for (i = 0; i < BRIX_DNS_STATE_COUNT; i++) {
57 json_object_set_new(target, brix_dns_state_name(i),
58 json_integer((json_int_t) counts[i]));
59 }
60 if (rows == NULL) {
61 return;
62 }
63 for (i = 0; i < n; i++) {
64 json_array_append_new(rows, dashboard_dns_target_row(brix_dns_target_at(i)));
65 }
66 json_object_set_new(target, "targets", rows);
67}
