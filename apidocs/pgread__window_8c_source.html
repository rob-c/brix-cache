1/*
2 * pgread_window.c — primary-path windowed kXR_pgread streaming.
3 *
4 * WHAT: Serves a pgread larger than one streaming window as a train of
5 * kXR_status frames (kXR_PartialResult ... kXR_FinalResult), each
6 * produced window-by-window into the single hot rd.read_scratch slot
7 * as [32-byte pgRead status header][gapped [CRC32c(4)][page] data].
8 *
9 * WHY: The monolithic path fills a request-sized rd_pool slot (8 MiB for an
10 * xrdcp-shaped pgread) before the first byte leaves; at streaming rates
11 * that destination never stays cache-resident and the page-cache-to-user
12 * copy dominates worker CPU. A ~2 MiB window destination stays LLC-hot
13 * (30-40% cheaper per byte on this class of host), the stream holds one
14 * window of memory budget instead of the whole request, and the first
15 * frame reaches the wire after one window instead of the full request.
16 *
17 * HOW: Reuses the Phase 31 windowed-read state machine (ctx->rd.win_*,
18 * brix_read_window_pump in reads.c) with rd.win_pgread set. The pump
19 * cuts every window on the absolute 4 KiB page grid (each partial frame
20 * carries whole [CRC][page] units — the same wire shape the §1.3
21 * chunked pool streamer sends), the worker thread runs the in-place
22 * encode+CRC (BRIX_VFS_IO_PGREAD), and brix_pgread_window_emit stamps
23 * the status header ahead of the encoded bytes and queues the
24 * contiguous frame. The stream self-serializes: recv stays suspended
25 * in XRD_ST_AIO for the whole train and send.c re-enters the pump when
26 * a parked frame drains, so read_scratch is never overwritten while a
27 * frame still references it.
28 */

41/*
42 * brix_pgread_window_want — data bytes for the next window.
43 *
44 * Cut on the absolute 4 KiB page grid (parity with brix_pgread_chunk_len in
45 * the §1.3 streamer, at window scale): every window after the first then
46 * starts page-aligned, so each frame carries whole [CRC][page] units and the
47 * per-page CRC framing never splits a page across frames.
48 */
49size_t
50brix_pgread_window_want(off_t cur, size_t left)
51{
52 off_t grid_end;
53 size_t wlen;
54
55 grid_end = (off_t) (((uint64_t) cur + (uint64_t) BRIX_READ_WINDOW)
56 & ~((uint64_t) kXR_pgPageSZ - 1));
57 wlen = (size_t) (grid_end - cur);
58 return (wlen > left) ? left : wlen;
59}

61/*
62 * brix_pgread_window_scratch — frame-buffer bytes for a window of `want` data
63 * at offset `cur`: the 32-byte status header plus the gapped wire encoding
64 * (one 4-byte CRC per page; offset alignment can split the first page — same
65 * page-count math as brix_pgread_scratch_size).
66 */
67size_t
68brix_pgread_window_scratch(off_t cur, size_t want)
69{
70 size_t n_pages;
71
72 n_pages = ((size_t) (cur & (kXR_pgPageSZ - 1)) + want + kXR_pgPageSZ - 1)
73 / kXR_pgPageSZ;
74 if (n_pages == 0) {
75 n_pages = 1;
76 }
77 return sizeof(ServerStatusResponse_pgRead) + want + n_pages * BRIX_PG_CKSZ;
78}

80/*
81 * brix_pgread_window_try_warm — inline warm-cache probe for ONE window.
82 *
83 * WHAT: Attempts the window's read + in-place gapped encode + per-page CRC
84 * on the event loop via preadv2(RWF_NOWAIT) into `datap` (the byte
85 * after the frame headroom). On a full hit sets *nread and *out_size,
86 * charges the backend byte metric, and returns 1; any miss returns 0
87 * with the outputs untouched.
88 *
89 * WHY: The windowed train self-serializes (recv suspended for its whole
90 * duration), so the pool buys no read/send overlap here — each window's
91 * thread-pool round-trip is pure handoff latency plus two scheduler
92 * wakeups (4 per xrdcp-shaped 8 MiB request; thousands per transfer),
93 * which is the drag when every core is busy. A resident window served
94 * inline deletes that entirely. Inline work stays bounded to one
95 * window (≤ BRIX_READ_WINDOW — the same inline budget the single-shot
96 * brix_pgread_try_warm accepts); a cold window posts to the pool as
97 * before, keeping the event loop off blocking disk reads.
98 *
99 * HOW: Mirrors brix_pgread_try_warm: pool configured + regular file +
100 * native preadv2 on the effective storage object; a hit means errno
101 * stayed 0 and every requested byte was delivered (a short window —
102 * EOF or partial residency — is a miss, and the blocking path re-reads
103 * it and owns the EOF framing).
104 */
105ngx_flag_t
106brix_pgread_window_try_warm(brix_ctx_t *ctx, ngx_stream_brix_srv_conf_t *rconf,
107 u_char *datap, size_t want, ssize_t *nread, size_t *out_size)
108{
109 brix_pgread_io_t warm_io = { .nowait = 1, .nread = 0, .io_errno = 0 };
110 size_t warm_osz;
111 brix_sd_obj_t warm_scratch;
112 brix_sd_obj_t *warm_obj;
113
114 if (rconf->common.thread_pool == NULL
115 || !ctx->files[ctx->rd.win_idx].is_regular)
116 {
117 return 0;
118 }
119
120 warm_obj = brix_vfs_effective_obj(&ctx->files[ctx->rd.win_idx].sd_obj,
121 ctx->rd.win_fd, &warm_scratch);
122 if (warm_obj->driver->preadv2 == NULL) {
123 return 0;
124 }
125
126 warm_osz = brix_pgread_read_encode_inplace(warm_obj, ctx->rd.win_offset,
127 want, datap, &warm_io);
128 if (warm_io.io_errno != 0 || warm_io.nread != (ssize_t) want) {
129 return 0;
130 }
131
132 /* The warm path bypasses brix_vfs_io_execute (where the posted and
133 * inline-fallback windows attribute), so charge the per-backend read
134 * total here. */
135 brix_metric_backend_bytes(
136 ctx->files[ctx->rd.win_idx].sd_obj.driver != NULL
137 ? ctx->files[ctx->rd.win_idx].sd_obj.driver->name : "posix",
138 BRIX_METRIC_OP_READ, (size_t) warm_io.nread);
139
140 *nread = warm_io.nread;
141 *out_size = warm_osz;
142 return 1;
143}

145/*
146 * brix_pgread_window_emit — frame + queue one completed window, advancing the
147 * continuation state (the pgread twin of brix_read_window_emit; called via
148 * the pump's dispatch). The worker (or inline fallback) already encoded the
149 * window into read_scratch + 32; this stamps the kXR_status header in the
150 * headroom and queues the contiguous frame. resptype is kXR_PartialResult
151 * for every window except the last planned one or EOF, which sends
152 * kXR_FinalResult (a zero-byte final frame is the wire EOF shape, matching
153 * brix_pgread_done_eof). Returns NGX_ERROR if the read or the queue failed
154 * (the terminating error frame has been sent); otherwise NGX_OK with
155 * ctx->state == XRD_ST_SENDING while the frame drains and rd.win_active
156 * cleared when this was the final frame.
157 */
158ngx_int_t
159brix_pgread_window_emit(brix_ctx_t *ctx, ngx_connection_t *c,
160 ssize_t nread, size_t out_size, int io_errno)
161{
162 u_char *frame = ctx->rd.read_scratch;
163 int64_t cur = (int64_t) ctx->rd.win_offset;
164 size_t got;
165 u_char resptype;
166
167 if (nread < 0) {
168 brix_read_io_failure_log(c->log, "pgread-windowed", ctx->rd.win_fd,
169 ctx->rd.win_offset, ctx->rd.win_remaining,
170 io_errno);
171 ctx->rd.win_active = 0;
172 ctx->state = XRD_ST_REQ_HEADER;
173 ctx->recv.hdr_pos = 0;
174 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
175 /* Partial frames may already be on the wire, so the terminating
176 * kXR_error must carry the ORIGINAL request's streamid (§1.3
177 * parity) — never cur_streamid, which the next inbound header
178 * overwrites. */
179 (void) brix_send_error_sid(ctx, c, ctx->rd.win_streamid, kXR_IOError,
180 io_errno ? strerror(io_errno) : "async pgread error");
181 return NGX_ERROR;
182 }
183
184 got = (size_t) nread;
185 ctx->files[ctx->rd.win_idx].bytes_read += got;
186 ctx->totals.bytes += got;
187
188 /* A short window is EOF and MUST end the train here: pgread framing
189 * allows a partial page only in the FINAL frame, so emitting the short
190 * window as kXR_PartialResult (and EOF as a later empty final) makes
191 * the client reject the tail page as corrupt. Only a full window with
192 * bytes still owed continues the train. */
193 if (got == brix_pgread_window_want((off_t) cur, ctx->rd.win_remaining)
194 && got < ctx->rd.win_remaining)
195 {
196 ctx->rd.win_remaining -= got;
197 ctx->rd.win_offset += (off_t) got;
198 } else {
199 ctx->rd.win_remaining = 0;
200 }
201
202 resptype = ctx->rd.win_remaining == 0
203 ? (u_char) kXR_FinalResult : (u_char) kXR_PartialResult;
204
205 brix_build_pgread_status_sid(ctx->rd.win_streamid, cur,
206 (uint32_t) out_size, resptype,
207 (ServerStatusResponse_pgRead *) frame);
208
209 if (resptype == (u_char) kXR_FinalResult) {
210 ctx->rd.win_active = 0;
211 BRIX_OP_OK(ctx, BRIX_OP_PGREAD);
212 }
213
214 ctx->state = XRD_ST_REQ_HEADER;
215 ctx->recv.hdr_pos = 0;
216
217 if (brix_queue_response_base(ctx, c, frame,
218 sizeof(ServerStatusResponse_pgRead)
219 + out_size, frame) != NGX_OK)
220 {
221 /* Hard send error: the socket is gone — stop pumping windows into
222 * it (the caller resumes the event loop, which will observe the
223 * dead connection). */
224 ctx->rd.win_active = 0;
225 return NGX_ERROR;
226 }
227
228 if (ctx->state != XRD_ST_SENDING) {
229 brix_release_read_buffer(ctx, c, frame); /* no-op scratch slot */
230 }
231 return NGX_OK;
232}

234/*
235 * brix_pgread_serve_windowed — windowed-streaming entry for the pgread
236 * handler (the pgread twin of read_serve_windowed): admits one window's
237 * worth of budget, arms the shared windowed-read state machine with
238 * win_pgread set, and kicks the pump. cur_streamid is snapshotted into
239 * win_streamid because every frame of the train must echo the originating
240 * request's stream id.
241 */
242ngx_int_t
243brix_pgread_serve_windowed(brix_ctx_t *ctx, ngx_connection_t *c,
244 ngx_stream_brix_srv_conf_t *rconf, const brix_pgread_run_t *run)
245{
246 /* Admit one window's worth — the stream holds ~one encoded window in
247 * read_scratch, not the full request, so many more fit under the
248 * budget (parity with read_serve_windowed). */
249 if (!brix_budget_admit(ctx, rconf->memory_budget,
250 brix_pgread_window_scratch(0, (size_t) BRIX_READ_WINDOW))) {
251 return brix_fsoverload_backoff(ctx, c, rconf);
252 }
253
254 ctx->rd.win_active = 1;
255 ctx->rd.win_pgread = 1;
256 ctx->rd.win_readv = 0;
257 ctx->rd.win_prefetch = 0; /* round-12: a fresh train starts with no
258 * read-ahead in flight or stashed (the recv
259 * defer barrier guarantees quiescence) */
260 ctx->rd.win_ready = 0;
261 ctx->rd.win_fd = run->fd;
262 ctx->rd.win_idx = run->idx;
263 ctx->rd.win_offset = (off_t) run->offset;
264 ctx->rd.win_remaining = run->rlen;
265 ctx->rd.win_streamid[0] = ctx->recv.cur_streamid[0];
266 ctx->rd.win_streamid[1] = ctx->recv.cur_streamid[1];
267
268 brix_prefetch_read_file(c->log, &ctx->files[run->idx],
269 (off_t) run->offset, run->rlen,
270 ctx->files[run->idx].writable
271 ? 0 : ctx->files[run->idx].cached_size);
272
273 if (rconf->access_log_fd != NGX_INVALID_FILE) {
274 char detail[64];
275
276 snprintf(detail, sizeof(detail), "%lld+%zu",
277 (long long) run->offset, run->rlen);
278 brix_log_access(ctx, c, "PGREAD", ctx->files[run->idx].path,
279 detail, 1, 0, NULL, run->rlen);
280 }
281
282 brix_read_window_pump(ctx, c, rconf);
283 return NGX_OK;
284}
