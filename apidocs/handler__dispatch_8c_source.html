1/*
2 * handler_dispatch.c — S3 post-auth dispatch: URI parsing, token-scope gate,
3 * and the bucket/empty-key/list routing orchestrator.
4 *
5 * WHAT: Owns the S3 content handler's post-authentication routing. Parses the
6 * path-style URI (/<bucket>/<key>) into bucket + key, enforces the WLCG
7 * bearer-token scope over the requested path, and dispatches bucket-level
8 * (empty-key) and list operations. Object-key method routing lives in the
9 * sibling handler_object_route.c; the entry handler and auth gate live in
10 * handler.c.
11 *
12 * WHY: Split out of handler.c (976 lines) under the phase-79 500-line file-size
13 * cap. The URI parser and the after-auth dispatch tree form one cohesive
14 * concern — turning a raw request into (bucket, key) and steering it to the
15 * right family of handlers — distinct from the entry/auth concern (handler.c)
16 * and the object-method concern (handler_object_route.c).
17 *
18 * HOW: s3_dispatch_after_auth() is the orchestrator: parse URI, apply the
19 * token-scope gate, then branch to list / empty-key(bucket) / object routing.
20 * Each gate returns NGX_DECLINED to continue dispatch; any other value means a
21 * response was already sent and must be returned as-is (see the per-function
22 * doc blocks for that sentinel convention). s3_reject_write_disabled() and
23 * s3_read_body_metric() are shared with handler_object_route.c and so are
24 * non-static (declared in s3_handler_internal.h).
25 */

47/*
48 *
49 * WHAT: Parses the S3 request URI into bucket name and object key components. Supports
50 * path-style addressing (/<bucket>/<key>) and list requests (/<bucket>/?list-type=2).
51 * Performs URL decoding on the object key portion to handle encoded characters in paths.
52 * Returns 1 for success, 0 for malformed URI, -1 if bucket name doesn't match configured value.
53 *
54 * WHY: S3 clients use path-style URIs that must be parsed into bucket + key before any filesystem
55 * operations can proceed. The parser handles both object access (GET/PUT) and list requests
56 * uniformly — distinguishing between them happens later via s3_is_list_request(). */
57/*
58 * HOW: Advances past the leading '/' in the URI, then checks if the remaining prefix matches conf->bucket.len bytes. If bucket match succeeds, advances past '/' to extract the key portion which is URL-decoded via s3_urldecode(). Returns -1 on bucket mismatch (NoSuchBucket), 0 on malformed URI (InvalidURI), 1 on success.
59 */
60/*
61 * Parse the request URI into bucket + key.
62 *
63 * Path-style: /<bucket>/<key>
64 * List: /<bucket>/?list-type=2[&prefix=...&delimiter=...&...]
65 *
66 * Returns:
67 * 1 - success
68 * 0 - malformed URI
69 * -1 - bucket name mismatch
70 * */
71
72static int
73s3_parse_uri(ngx_http_request_t *r,
74 ngx_http_s3_loc_conf_t *cf,
75 u_char *key_out, size_t key_sz)
76{
77 const u_char *uri = r->uri.data;
78 size_t ulen = r->uri.len;
79 size_t blen = cf->bucket.len;
80
81 if (ulen == 0 || uri[0] != '/') {
82 return 0;
83 }
84
85 uri++;
86 ulen--;
87
88 if (blen > 0) {
89 if (ulen < blen || ngx_strncmp(uri, cf->bucket.data, blen) != 0) {
90 return -1;
91 }
92 uri += blen;
93 ulen -= blen;
94
95 if (ulen == 0) {
96 /* /bucket with no trailing slash — valid, empty key (e.g. ListObjects) */
97 key_out[0] = '\0';
98 return 1;
99 }
100 if (uri[0] != '/') {
101 return 0;
102 }
103 uri++;
104 ulen--;
105 }
106
107 if (ulen == 0) {
108 key_out[0] = '\0';
109 return 1;
110 }
111
112 if (brix_http_urldecode(uri, ulen, (char *) key_out, key_sz,
113 BRIX_URLDECODE_PLUS_TO_SPACE |
114 BRIX_URLDECODE_REJECT_NUL) != BRIX_URLDECODE_OK)
115 {
116 return 0;
117 }
118
119 return 1;
120}

122/*
123 * Parse the URI into key. Returns NGX_DECLINED when key is valid and dispatch
124 * may continue. On a rejected URI the XML error response has already been
125 * sent and the value the content handler must return is passed back. The
126 * continue outcome is a distinct sentinel (NGX_DECLINED, which no send path
127 * produces) rather than NGX_OK because a fully-sent error body makes
128 * ngx_http_output_filter() return NGX_OK — comparing the response rc against
129 * NGX_OK would let dispatch fall through with an unwritten key after the
130 * request was already answered.
131 */
132static ngx_int_t
133s3_dispatch_parse_uri(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
134 ngx_uint_t method_slot, u_char *key, size_t key_cap)
135{
136 int parse_rc = s3_parse_uri(r, cf, key, key_cap);
137
138 if (parse_rc == -1) {
139 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_INVALID_URI]);
140 return s3_metrics_return_method(
141 r, method_slot,
142 s3_send_xml_error(r, NGX_HTTP_NOT_FOUND, "NoSuchBucket",
143 "The specified bucket does not exist."));
144 }
145 if (parse_rc == 0) {
146 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_INVALID_URI]);
147 return s3_metrics_return_method(
148 r, method_slot,
149 s3_send_xml_error(r, NGX_HTTP_BAD_REQUEST, "InvalidURI",
150 "Couldn't parse the specified URI."));
151 }
152 return NGX_DECLINED;
153}

155/*
156 * Token-scope gate. Returns NGX_DECLINED when the operation may proceed. On
157 * a scope deny the 403 response has already been sent and the value the
158 * content handler must return is passed back (same sentinel convention as
159 * s3_dispatch_parse_uri — the response rc itself is NGX_OK once the body is
160 * flushed, so it must never gate dispatch).
161 */
162static ngx_int_t
163s3_check_token_scope(ngx_http_request_t *r, ngx_http_s3_req_ctx_t *s3ctx,
164 ngx_uint_t method_slot, const u_char *key)
165{
166 brix_acc_op_t aop;
167 int need_write;
168 char logical[PATH_MAX];
169
170 if (s3ctx->identity == NULL
171 || !(s3ctx->identity->auth_method & BRIX_AUTHN_TOKEN))
172 {
173 return NGX_DECLINED;
174 }
175
176 aop = s3_method_aop(r);
177 need_write = (aop == BRIX_AOP_CREATE || aop == BRIX_AOP_DELETE);
178 (void) ngx_snprintf((u_char *) logical, sizeof(logical), "/%s%Z",
179 (const char *) key);
180 if (brix_identity_check_token_scope(s3ctx->identity, logical, need_write)
181 == NGX_OK)
182 {
183 return NGX_DECLINED;
184 }
185 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_ACCESS_DENIED]);
186 return s3_metrics_return_method(r, method_slot,
187 s3_send_xml_error(r, NGX_HTTP_FORBIDDEN, "AccessDenied",
188 "token scope does not cover this object"));
189}

283/*
284 * Resolve key -> confined fs_path. Returns NGX_DECLINED when dispatch may
285 * continue. On a refusal the response has already been sent and the value the
286 * content handler must return is passed back (same sentinel convention as
287 * s3_dispatch_parse_uri).
288 *
289 * The refusal is the resolver's own — an escape is AccessDenied, a RESERVED
290 * internal name is NoSuchKey/404 and books no_such_key, which is what a key that
291 * simply is not there answers. Anything else here would let an unauthenticated
292 * probe read the reserved-name policy off the status line.
293 */
294static ngx_int_t
295s3_resolve_object_key(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
296 ngx_http_s3_req_ctx_t *s3ctx, ngx_uint_t method_slot, const u_char *key,
297 char *fs_path, size_t fs_path_cap)
298{
299 s3_key_error_t err;
300 int rc;
301
302 rc = s3_resolve_key_ex(cf->common.root_canon, (const char *) key, fs_path,
303 fs_path_cap, cf->common.cache_store_endpoint);
304 if (rc == 0) {
305 ngx_cpystrn((u_char *) s3ctx->fs_path, (u_char *) fs_path,
306 sizeof(s3ctx->fs_path));
307 return NGX_DECLINED;
308 }
309
310 s3_resolve_key_error(rc, &err);
311 if (rc == 404) {
312 /* Reserved name: hand the whole answer to the absent-key router so the
313 * per-method shape matches too, not only the status. */
314 return s3_dispatch_object_absent(r, cf, method_slot, &err);
315 }
316 BRIX_S3_METRIC_INC(events_total[err.event]);
317 return s3_metrics_return_method(
318 r, method_slot,
319 s3_send_xml_error(r, err.status, err.code, err.message));
320}

322/*
323 * s3_apply_native_authdb — native u/g/p/h READ ACL gate (phase-101 W5.2c).
324 *
325 * Enforces the bare brix_authdb rules on S3 exactly as webdav/access.c does
326 * (and root:// on stream), closing the W5 gap where a bare brix_authdb off
327 * webdav was accepted-but-inert. Reads THIS export's finalized copy
328 * (cf->authdb_final, realpath'd against its own root — never the shared source),
329 * matching the confined fs_path in the same realpath space. READ methods only
330 * (writes gate on allow_write + their own per-op checks, mirroring webdav).
331 * Returns NGX_DECLINED to continue, or the already-sent 403 rc on deny.
332 */
333static ngx_int_t
334s3_apply_native_authdb(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
335 ngx_http_s3_req_ctx_t *s3ctx, ngx_uint_t method_slot, const char *fs_path)
336{
337 char peer[NGX_SOCKADDR_STRLEN];
338 size_t pn;
339
340 if (cf->authdb_final == NULL
341 || (r->method != NGX_HTTP_GET && r->method != NGX_HTTP_HEAD))
342 {
343 return NGX_DECLINED;
344 }
345
346 pn = ngx_min(r->connection->addr_text.len, sizeof(peer) - 1);
347 ngx_memcpy(peer, r->connection->addr_text.data, pn);
348 peer[pn] = '\0';
349
350 brix_authdb_query_t query = {
351 .rules = cf->authdb_final,
352 .identity = s3ctx->identity,
353 .peer_ip = peer,
354 .resolved_path = fs_path,
355 .needed_privs = BRIX_AUTH_READ,
356 };
357
358 if (brix_check_authdb_identity(r->connection->log, &query) != NGX_OK) {
359 BRIX_S3_METRIC_INC(events_total[BRIX_S3_EVENT_ACCESS_DENIED]);
360 return s3_metrics_return_method(r, method_slot,
361 s3_send_xml_error(r, NGX_HTTP_FORBIDDEN, "AccessDenied",
362 "authdb: access denied"));
363 }
364 return NGX_DECLINED;
365}

367/*
368 * Post-auth S3 op dispatch (parse URI -> route by method). Runs inside the
369 * caller's impersonation bracket; see ngx_http_s3_handler.
370 */
371ngx_int_t
372s3_dispatch_after_auth(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
373 ngx_http_s3_req_ctx_t *s3ctx, ngx_uint_t method_slot,
374 int is_list_request, ngx_flag_t is_post_object_form)
375{
376 u_char key[S3_MAX_KEY] = {0};
377 char fs_path[PATH_MAX];
378 ngx_int_t rc;
379
380 /* Each gate below returns NGX_DECLINED to continue dispatch; any other
381 * value (including NGX_OK) means the response was already sent and must
382 * be returned as-is — see s3_dispatch_parse_uri's doc block. */
383 rc = s3_dispatch_parse_uri(r, cf, method_slot, key, sizeof(key));
384 if (rc != NGX_DECLINED) {
385 return rc;
386 }
387
388 /*
389 * WLCG bearer-token scope enforcement.
390 *
391 * When a request was authenticated via a bearer token, verify that the
392 * token's scope covers the requested path and operation BEFORE dispatching
393 * to any sub-handler. Non-token requests (SigV4, anonymous) are allowed
394 * unconditionally by brix_identity_check_token_scope (scopes only apply to
395 * token auth). Empty keys (bucket-level ops, list) map to logical path "/".
396 */
397 rc = s3_check_token_scope(r, s3ctx, method_slot, key);
398 if (rc != NGX_DECLINED) {
399 return rc;
400 }
401
402 if (is_list_request) {
403 return s3_metrics_return_method(r, method_slot, s3_handle_list(r, cf));
404 }
405
406 if (key[0] == '\0') {
407 return s3_dispatch_empty_key(r, cf, method_slot, is_post_object_form);
408 }
409
410 rc = s3_resolve_object_key(r, cf, s3ctx, method_slot, key, fs_path,
411 sizeof(fs_path));
412 if (rc != NGX_DECLINED) {
413 return rc;
414 }
415 /* phase-101 W5.2c: native brix_authdb READ ACL on the confined fs_path. */
416 rc = s3_apply_native_authdb(r, cf, s3ctx, method_slot, fs_path);
417 if (rc != NGX_DECLINED) {
418 return rc;
419 }
420 return s3_dispatch_object_method(r, cf, s3ctx, method_slot, key, fs_path);
421}
