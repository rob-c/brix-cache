1/* File: connect.c — source-host resolution and TCP connect for a native TPC pull
2 * WHAT: tpc_connect() resolves the remote root:// origin through the brix DNS
3 * driver (brix_dns_resolve_sync: IP literal → per-worker cache → the event
4 * loop's nginx resolver via the bridge → libc, all off the event loop),
5 * re-checks every candidate address against the SSRF policy (I-DNS-3), and
6 * dials the first that connects within BRIX_TPC_CONNECT_TIMEOUT_SEC with the I/O
7 * timeouts and SciTags flow label applied. brix_tpc_check_src_policy() is the
8 * event-loop preflight the kXR_open destination gate runs: a verdict from an IP
9 * literal or the cache, or "unknown" so the open can park on an async resolve.
10 *
11 * WHY: Native TPC pull requires nginx to establish a TCP connection to the remote
12 * origin before it can send handshake frames and read the file. The resolution
13 * must iterate over every answered address (IPv4/IPv6) because some may be
14 * unreachable; the per-address policy check prevents connecting to loopback or
15 * private ranges when configured to reject them, and repeating it on the thread
16 * closes the rebinding window between preflight and connect. Nothing here may
17 * block the worker: the preflight never resolves, the connect runs in the pool.
18 *
19 * HOW: preflight = brix_net_target_check_cached (0 permitted / -1 refused /
20 * 1 no cached answer). Connect = brix_dns_resolve_sync → for each address:
21 * brix_net_target_check_addr → socket(family, SOCK_STREAM) → SO_RCVTIMEO/
22 * SO_SNDTIMEO → flow label → brix_connect_fd_deadline. Returns the connected
23 * fd or -1 with t->err_msg set.
24 * */

38/* Dial one candidate: socket → timeouts → flow label → deadline connect.
39 * Returns the connected fd, or -1 with the socket closed. */
40static int
41tpc_connect_candidate(const brix_tpc_pull_t *t, struct sockaddr *sa,
42 socklen_t salen)
43{
44 int fd;
45
46 fd = socket(sa->sa_family, SOCK_STREAM, 0);
47 if (fd < 0) {
48 return -1;
49 }
50
51 brix_apply_socket_io_timeouts(fd, BRIX_TPC_IO_TIMEOUT_SEC);
52
53 /*
54 * SciTags (phase-34): stamp the IPv6 flow label on the OUTBOUND pull
55 * socket before connecting (codes resolved on the event loop in
56 * start_pull). No-op on IPv4 / when not marked; fail-open.
57 */
58 if (t->pmark_exp != 0) {
59 (void) brix_pmark_flowlabel_apply_addr(fd, sa, salen, t->pmark_exp,
60 t->pmark_act, ngx_cycle->log);
61 }
62
63 if (brix_connect_fd_deadline(fd, sa, salen,
64 BRIX_TPC_CONNECT_TIMEOUT_SEC * 1000) == 0)
65 {
66 return fd;
67 }
68
69 close(fd);
70 return -1;
71}

74/* WHAT: resolve + policy-check + TCP connect to the TPC origin (thread-pool
75 * task). Returns the connected fd or -1 with t->err_msg set. */
76int
77tpc_connect(brix_tpc_pull_t *t)
78{
79 brix_dns_addr_t addrs[BRIX_DNS_MAX_ADDRS];
80 brix_net_target_policy_t policy;
81 char reason[BRIX_DNS_ERROR_LEN];
82 ngx_uint_t n, i;
83 uint16_t src_port;
84
85 src_port = t->src_port ? t->src_port : TPC_DEFAULT_PORT;
86
87 ngx_memzero(&policy, sizeof(policy));
88 policy.allow_local = t->conf->common.tpc_allow_local;
89 policy.allow_private = t->conf->common.tpc_allow_private;
90 policy.dns = t->conf->common.dns.policy;
91
92 n = brix_dns_resolve_sync(policy.dns, t->src_host, src_port, BRIX_AF_AUTO,
93 SOCK_STREAM, addrs, BRIX_DNS_MAX_ADDRS, reason,
94 sizeof(reason));
95 if (n == 0) {
96 snprintf(t->err_msg, sizeof(t->err_msg),
97 "TPC DNS resolution failed for %s: %s", t->src_host, reason);
98 return -1;
99 }
100
101 for (i = 0; i < n; i++) {
102 struct sockaddr *sa = (struct sockaddr *) &addrs[i].ss;
103 char ssrf_err[128];
104 int fd;
105
106 /* I-DNS-3: the address actually dialled is the one checked. */
107 if (brix_net_target_check_addr(sa, &policy, ssrf_err, sizeof(ssrf_err))
108 != NGX_OK)
109 {
110 snprintf(t->err_msg, sizeof(t->err_msg),
111 "TPC source host %s: %s", t->src_host, ssrf_err);
112 return -1;
113 }
114
115 fd = tpc_connect_candidate(t, sa, addrs[i].len);
116 if (fd >= 0) {
117 return fd;
118 }
119 }
120
121 snprintf(t->err_msg, sizeof(t->err_msg), "TPC connect to %s failed",
122 t->src_host);
123 return -1;
124}

127/*
128 * brix_tpc_check_src_policy — event-loop SSRF preflight for native root:// TPC.
129 *
130 * Bare host+port form of brix_net_target_check_cached(): the verdict comes
131 * from an IP literal or the per-worker DNS cache and nothing is resolved
132 * here. Returns 0 when every cached address is permitted, -1 with err_msg
133 * when the host is missing, definitively unknown or resolves to a prohibited
134 * range, and 1 when no answer is cached yet (the caller parks the open on an
135 * async resolve — launch_dns.c).
136 */
137int
138brix_tpc_check_src_policy(const ngx_stream_brix_srv_conf_t *conf,
139 const char *src_host, uint16_t src_port, char *err_msg, size_t err_msg_sz)
140{
141 brix_net_target_t target;
142 brix_net_target_policy_t policy;
143 ngx_int_t rc;
144
145 if (src_host == NULL || src_host[0] == '\0') {
146 snprintf(err_msg, err_msg_sz, "TPC source host missing");
147 return -1;
148 }
149
150 ngx_memzero(&target, sizeof(target));
151 target.host.data = (u_char *) src_host;
152 target.host.len = ngx_strlen(src_host);
153 target.port = src_port ? src_port : TPC_DEFAULT_PORT;
154 target.has_port = 1;
155
156 ngx_memzero(&policy, sizeof(policy));
157 policy.allow_local = conf->common.tpc_allow_local;
158 policy.allow_private = conf->common.tpc_allow_private;
159 policy.dns = conf->common.dns.policy;
160
161 rc = brix_net_target_check_cached(&target, &policy, err_msg, err_msg_sz);
162 if (rc == NGX_OK) {
163 return 0;
164 }
165 return rc == NGX_DECLINED ? 1 : -1;
166}
