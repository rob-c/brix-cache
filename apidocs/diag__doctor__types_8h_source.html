1/*
2 * diag_doctor_types.h — the doctor subsystem's endpoint model.
3 *
4 * WHAT: doctor_ep and every sub-record it aggregates — the advertised-config
5 * scrape (doctor_cfg), the mesh round-trip probe (doctor_lat), the deep
6 * read-only reconnaissance (doctor_recon), the CMS locate-plane
7 * classification (doctor_cmsloc) and the EOS /proc dialect
8 * (doctor_eos / doctor_eos_rep).
9 * WHY: split out of diag_internal.h (600-line cap, coding-standards §1) as one
10 * coherent group: these types are written by the doctor probes and read by
11 * the doctor renderers, and by nothing else in the tool.
12 * HOW: types only — no prototypes, no storage. Included from diag_internal.h
13 * after dx_finding/DOC_* so it is never included directly.
14 */

18/*
19 * WHAT: the advertised-configuration + capacity face of one endpoint, scraped
20 * over the wire by doctor_scrape_config (kXR_Qconfig + kXR_Qspace).
21 * WHY: the config/perf advisor (phase-93) classifies *values*, not error codes;
22 * this block holds the scraped scalars the computed rules run over.
23 * HOW: pointer-free and PII-free by construction — only advertised scalars, no
24 * path ever. Absent/unsupported keys keep their sentinel (strings "",
25 * ints -1 for numeric caps; booleans 0). space_* are -1 until Qspace runs.
26 */
27typedef struct {
28 int scraped; /* 1 = the Qconfig/Qspace scrape ran */
29 char version[48]; /* kXR_Qconfig "version" ("" = absent) */
30 char role[24]; /* "manager"/"server"/... ("" = absent) */
31 char sitename[64]; /* "" if unset/unsupported */
32 int tpc, tpcdlg; /* 0/1 advertised */
33 int have_adler32, have_crc32c; /* parsed from the "chksum" CSV */
34 int bind_max, pio_max; /* parallelism caps (-1 = absent) */
35 int readv_iov_max, readv_ior_max; /* readv caps (-1 = absent) */
36 int pgread; /* per-page CRC read supported (0/1) */
37 int64_t space_total, space_free; /* kXR_Qspace bytes (-1 = not pulled) */
38} doctor_cfg;

40/*
41 * WHAT: round-trip latency of one mesh node, sampled over the two XRootD control
42 * planes: the data-server plane (a kXR_stat "/" round-trip) and the CMS
43 * redirect plane (a kXR_locate "/" round-trip, the CMSD-served query).
44 * WHY: --latency answers "how far is each server, and is the redirect plane as
45 * responsive as the data plane?" Every sample is a full request→reply, so
46 * the figure is bi-directional (out and back) by construction.
47 * HOW: filled by doctor_latency_probe; probed=0 until it runs. Times are ms;
48 * *_ok is the successful-sample count per plane (0 = plane unreachable).
49 */
50typedef struct {
51 int probed; /* 1 = doctor_latency_probe ran */
52 int samples; /* samples attempted per plane */
53 int xr_ok, cms_ok; /* successful samples, data / cms plane */
54 double xr_min, xr_avg, xr_max; /* data-plane stat RTT, ms */
55 double cms_min, cms_avg, cms_max; /* cms-plane locate RTT, ms */
56} doctor_lat;

60/*
61 * WHAT: the deep read-only reconnaissance face of one endpoint (--deep-recon):
62 * operational counters parsed from `query stats a`, a full Qconfig key
63 * sweep (how many advertised keys the server actually answers), the
64 * decoded kXR_protocol capability bits, and a bounded list of top-level
65 * namespace roots our identity is allowed to see.
66 * WHY: --config-audit classifies a handful of values into advice; deep-recon
67 * instead surfaces the whole read-only picture an operator might want to
68 * eyeball on one server — traffic, logins, TPC accounting, capability
69 * flags, authorized roots — in a single pass. Diagnostic, not advisory.
70 * HOW: filled by doctor_recon_probe over the already-open connection. PII-free:
71 * only server-advertised scalars and the top-level path *names* the server
72 * itself returns from a dirlist of "/"; never a token or a user path. Every
73 * numeric field is int64 with -1 = "the server did not report it". No goto.
74 */
75typedef struct {
76 int probed; /* 1 = doctor_recon_probe ran */
77 /* --- link plane (stats id="link") --- */
78 int64_t conns_total; /* <tot> cumulative connections */
79 int64_t bytes_in, bytes_out; /* <in>/<out> bytes over all links */
80 /* --- xrootd op plane (stats id="xrootd") --- */
81 int64_t ops_open, ops_rd, ops_wr; /* <ops><open>/<rd>/<wr> */
82 int64_t ops_err, ops_rdr, ops_dly; /* <err>/<rdr>/<dly> */
83 int64_t lgn_num, lgn_au, lgn_af; /* <lgn> total / authed / auth-failed */
84 /* --- ofs TPC accounting (stats id="ofs" -> <tpc>) --- */
85 int have_tpc; /* 1 = a <tpc> block was present */
86 int64_t tpc_grant, tpc_deny, tpc_err;
87 /* --- oss capacity (stats id="oss", first path; best-effort) --- */
88 int64_t oss_total, oss_free; /* bytes */
89 int64_t ino_total, ino_free; /* inodes */
90 /* --- http plane (stats id="http"; best-effort) --- */
91 int have_http;
92 int64_t http_reqs, http_in, http_out, http_tpc_pull, http_tpc_push;
93 /* --- Qconfig key sweep --- */
94 int cfg_probed, cfg_supported; /* keys tried / keys the server answers */
95 char cid[64], cms[96]; /* cluster id / cms manager (if advertised) */
96 /* --- capability bits, decoded from server_flags --- */
97 unsigned caps;
98 /* --- bounded authorized-root discovery (top-level names only) --- */
99 int roots_listed; /* 1 = a dirlist of "/" succeeded */
100 int nroots; /* entries captured (<= RECON_MAX_ROOTS) */
101 int roots_more; /* 1 = "/" held more than we captured */
102 char roots[RECON_MAX_ROOTS][64];
103} doctor_recon;

105/*
106 * CMS-plane classification of a node, read straight from the manager's
107 * kXR_locate answer (each token is "<type><access>host:port"). This is the
108 * authority for what a node IS even when we cannot open an inbound connection
109 * to it — so a firewalled/IPv6-only holder is still typed, not a blank DOWN.
110 * type : 'S'/'s' data server · 'M'/'m' subordinate manager (redirector)
111 * lowercase = pending (queued/staging, not yet serving)
112 * acc : 'r' read-only · 'w' read/write
113 */
114typedef enum {
115 DOC_CMS_NONE = 0, /* not learned from a locate (e.g. a directly-probed root) */
116 DOC_CMS_SERVER, /* 'S'/'s' — data server (a file holder) */
117 DOC_CMS_MANAGER /* 'M'/'m' — subordinate manager / redirector */
118} doctor_cms_role;
119
120typedef struct {
121 doctor_cms_role role; /* server vs redirector, from the locate type byte */
122 int pending; /* lowercase type => queued/staging, not yet online */
123 int write; /* 'w' access => read/write; else read-only */
124 int reported;/* 1 => this classification came from a CMS locate */
125} doctor_cmsloc;

127/*
128 * EOS dialect classification (diag_doctor_eos.c). A stock kXR_locate against an
129 * EOS MGM only ever returns the MGM itself, so the CMS plane cannot see the FST
130 * farm behind it. EOS instead overloads the XRootD protocol with a /proc command
131 * channel (open+read "/proc/{user,admin}/?mgm.cmd=..."): `mgm.cmd=version` types
132 * the MGM, and `mgm.cmd=fs&mgm.subcmd=ls&mgm.outformat=m` (admin) enumerates the
133 * FST inventory. `kind` records what was learned about THIS endpoint.
134 */
135typedef enum {
136 DOC_EOS_NONE = 0, /* not an EOS node (no version banner / not from fs ls) */
137 DOC_EOS_MGM, /* endpoint is an EOS MGM (version banner seen) */
138 DOC_EOS_FST /* endpoint is an FST enumerated from the MGM `fs ls` */
139} doctor_eos_kind;
140
141typedef struct {
142 doctor_eos_kind kind;
143 /* --- MGM (kind == DOC_EOS_MGM) --- */
144 char instance[64]; /* EOS_INSTANCE */
145 char version[32]; /* EOS_SERVER_VERSION */
146 int gated; /* FST enumeration attempted but admin-gated (NotAuthorized) */
147 int sampled; /* FSTs came from unprivileged fileinfo replica sampling */
148 int fst_count; /* FSTs enumerated (0 when gated with no sampling) */
149 /* --- FST (kind == DOC_EOS_FST) --- */
150 char geotag[40]; /* stat.geotag */
151 char cfgstatus[16]; /* configstatus: rw|ro|drain|off|empty... */
152 int booted; /* stat.boot == "booted" */
153 int active; /* stat.active == "online" */
154 int64_t cap_bytes; /* stat.statfs.capacity */
155 int64_t free_bytes; /* stat.statfs.freebytes */
156} doctor_eos;

158/*
159 * One replica record parsed from an EOS `fileinfo` reply — the unprivileged
160 * (user-plane) FST-discovery path. When admin `fs ls` is gated for our identity,
161 * `fileinfo` on real files still names the FSTs holding their replicas; walking a
162 * sample of files and unioning these records surfaces the storage farm without
163 * admin rights (partial coverage: only FSTs that hold a sampled file appear).
164 */
165typedef struct {
166 char host[256]; /* replica FST host (bare hostname column) */
167 int port; /* FST port (default 1095 — fileinfo omits it) */
168 char geotag[40]; /* geotag column */
169 char cfgstatus[16]; /* configstatus column: rw|ro|drain|off... */
170 int booted; /* boot column == "booted" */
171 int active; /* active column == "online" */
172} doctor_eos_rep;
