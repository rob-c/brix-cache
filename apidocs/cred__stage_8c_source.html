21/* Per-uid staging root on tmpfs. /dev/shm is 1777 (sticky, world-writable), so
22 * every uid can create its OWN 0700 subdirectory here that no other uid can enter
23 * or delete; the security boundary is that dir's mode + ownership, checked below.
24 * Per-uid naming keeps distinct workers/users from tripping over each other's dir
25 * ownership on a shared host. */
26#define BRIX_CRED_STAGE_BASE "/dev/shm/brix-creds"
27
28/* REQUIRE a real directory, owned by us, with no group/other access. A path
29 * that fails any of these (a foreign squatter, a loosened mode, a symlink) is
30 * unsafe for a secret — fail closed rather than trust it. Shared by BOTH
31 * credential arms (phase-108 C11.2): the persistent destination dir gets
32 * exactly the scrutiny the volatile staging dir always had, re-checked on
33 * every call. */
34static int
35cred_dir_check(const char *dir)
36{
37 struct stat st;
38
39 if (lstat(dir, &st) != 0) {
40 return -1;
41 }
42 if (!S_ISDIR(st.st_mode)
43 || st.st_uid != geteuid()
44 || (st.st_mode & 0077) != 0)
45 {
46 errno = EPERM;
47 return -1;
48 }
49 return 0;
50}

52int
53brix_cred_stage_dir(char *out, size_t outsz)
54{
55 char dir[64];
56 int n;
57
58 n = snprintf(dir, sizeof(dir), "%s.%u",
59 BRIX_CRED_STAGE_BASE, (unsigned) geteuid());
60 if (n < 0 || (size_t) n >= sizeof(dir)) {
61 errno = ENAMETOOLONG;
62 return -1;
63 }
64
65 if (mkdir(dir, 0700) != 0 && errno != EEXIST) {
66 return -1;
67 }
68
69 if (cred_dir_check(dir) != 0) {
70 return -1;
71 }
72
73 if ((size_t) n + 1 > outsz) {
74 errno = ENAMETOOLONG;
75 return -1;
76 }
77 memcpy(out, dir, (size_t) n + 1);
78 return 0;

83/* Create a fresh 0600 file "<base><8 hex chars>" with the full C11.2 flag
84 * set. O_CREAT|O_EXCL never follows a symlink and refuses a squatter (even a
85 * dangling link answers EEXIST); a fresh random suffix per attempt makes the
86 * EEXIST retry safe. Returns the fd with the path in `path`, or -1. */
87static int
88cred_create_excl(const char *base, char *path, size_t pathsz)
89{
90 unsigned char rnd[4];
91 int attempt, fd, n;
92
93 for (attempt = 0; attempt < 16; attempt++) {
94 if (getentropy(rnd, sizeof(rnd)) != 0) {
95 return -1;
96 }
97 n = snprintf(path, pathsz, "%s%02x%02x%02x%02x",
98 base, rnd[0], rnd[1], rnd[2], rnd[3]);
99 if (n < 0 || (size_t) n >= pathsz) {
100 errno = ENAMETOOLONG;
101 return -1;
102 }
103 fd = open(path,
104 O_CREAT | O_EXCL | O_WRONLY | O_NOFOLLOW | O_CLOEXEC,
105 S_IRUSR | S_IWUSR);
106 if (fd >= 0) {
107 return fd;
108 }
109 if (errno != EEXIST) {
110 return -1;
111 }
112 }
113 errno = EEXIST;
114 return -1;
115}

117/* EINTR-safe full-length write; a short write that never completes is an
118 * error, never a silent truncation of a secret. */
119static int
120cred_write_full(int fd, const void *bytes, size_t len)
121{
122 const unsigned char *p = bytes;
123 size_t off = 0;
124
125 while (off < len) {
126 ssize_t w = write(fd, p + off, len - off);
127 if (w < 0) {
128 if (errno == EINTR) {
129 continue;
130 }
131 return -1;
132 }
133 off += (size_t) w;
134 }
135 return 0;
136}

150/* The publish durability barrier: a rename is not durable until its parent
151 * directory is. Open the dir itself and fsync it; close is checked — an
152 * unreported barrier failure would be a silent durability downgrade. */
153static int
154cred_dir_flush(const char *dir)
155{
156 int fd, saved;
157
158 fd = open(dir, O_RDONLY | O_DIRECTORY | O_CLOEXEC);
159 if (fd < 0) {
160 return -1;
161 }
162 if (fsync(fd) != 0) {
163 saved = errno;
164 (void) close(fd);
165 errno = saved;
166 return -1;
167 }
168 return close(fd) == 0 ? 0 : -1;
169}

171/* A basename component for a credential file: non-empty, no '/', not a dot
172 * dir — anything else could walk the write out of the claimed directory. */
173static int
174cred_name_ok(const char *s)
175{
176 if (s == NULL || s[0] == '\0' || strchr(s, '/') != NULL) {
177 return 0;
178 }
179 if (strcmp(s, ".") == 0 || strcmp(s, "..") == 0) {
180 return 0;
181 }
182 return 1;
183}

185/* VOLATILE arm: the temp file under the private tmpfs staging dir IS the
186 * product; the caller consumes and unlinks it. Deliberately NO fsync — the
187 * §3.3 carve-out: durability for a secret that must not survive reboot is an
188 * anti-goal, and tmpfs has nothing to sync to anyway. */
189static int
190cred_write_volatile(const brix_cred_write_req_t *req, const void *bytes,
191 size_t len, char *path_out, size_t path_outsz)
192{
193 char dir[64];
194 char base[192];
195 int fd, n, saved;
196
197 if (brix_cred_stage_dir(dir, sizeof(dir)) != 0) {
198 return -1; /* fail closed — never /tmp */
199 }
200
201 n = snprintf(base, sizeof(base), "%s/%s", dir, req->prefix);
202 if (n < 0 || (size_t) n >= sizeof(base)) {
203 errno = ENAMETOOLONG;
204 return -1;
205 }
206
207 fd = cred_create_excl(base, path_out, path_outsz);
208 if (fd < 0) {
209 return -1;
210 }
211
212 /* Defensive: pin 0600 regardless of umask/platform open() behaviour. */
213 if (fchmod(fd, S_IRUSR | S_IWUSR) != 0
214 || cred_write_full(fd, bytes, len) != 0)
215 {
216 return cred_fail_fd(fd, path_out);
217 }
218
219 if (close(fd) != 0) { /* close IS a write error here */
220 saved = errno;
221 (void) unlink(path_out);
222 errno = saved;
223 return -1;
224 }
225 return 0;
226}

228/* PERSISTENT arm: stage "<dir>/.<name>.<random>", fsync the data, publish by
229 * rename to "<dir>/<name>", then flush the parent. The destination dir is
230 * held to the SAME standard as the staging dir (cred_dir_check): a
231 * group-readable credential dir is refused with EPERM, not tolerated. */
232static int
233cred_write_persistent(const brix_cred_write_req_t *req, const void *bytes,
234 size_t len, char *path_out, size_t path_outsz)
235{
236 char base[PATH_MAX];
237 char tmp[PATH_MAX];
238 int fd, n, saved;
239
240 if (mkdir(req->dir, 0700) != 0 && errno != EEXIST) {
241 return -1;
242 }
243 if (cred_dir_check(req->dir) != 0) {
244 return -1;
245 }
246
247 /* Compose the final path first so a too-small caller buffer is refused
248 * before any file exists. */
249 n = snprintf(path_out, path_outsz, "%s/%s", req->dir, req->name);
250 if (n < 0 || (size_t) n >= path_outsz) {
251 errno = ENAMETOOLONG;
252 return -1;
253 }
254
255 n = snprintf(base, sizeof(base), "%s/.%s.", req->dir, req->name);
256 if (n < 0 || (size_t) n >= sizeof(base)) {
257 errno = ENAMETOOLONG;
258 return -1;
259 }
260
261 fd = cred_create_excl(base, tmp, sizeof(tmp));
262 if (fd < 0) {
263 return -1;
264 }
265
266 if (fchmod(fd, S_IRUSR | S_IWUSR) != 0
267 || cred_write_full(fd, bytes, len) != 0
268 || fsync(fd) != 0)
269 {
270 return cred_fail_fd(fd, tmp);
271 }
272
273 if (close(fd) != 0) { /* close IS a write error here */
274 saved = errno;
275 (void) unlink(tmp);
276 errno = saved;
277 return -1;
278 }
279
280 if (rename(tmp, path_out) != 0) {
281 saved = errno;
282 (void) unlink(tmp);
283 errno = saved;
284 return -1;
285 }
286
287 /* A barrier failure after rename is reported but the published file is
288 * NOT unlinked: the rename may have replaced a live credential, and
289 * destroying it would be worse than the lost durability barrier. */
290 return cred_dir_flush(req->dir);
291}

293int
294brix_cred_write_engine(const brix_cred_write_req_t *req, const void *bytes,
295 size_t len, char *path_out, size_t path_outsz)
296{
297 if (req == NULL || path_out == NULL || path_outsz == 0
298 || (bytes == NULL && len > 0)
299 || (unsigned) req->arm >= BRIX_CRED_ARM_COUNT
300 || (unsigned) req->kind >= BRIX_CRED_KIND_COUNT)
301 {
302 errno = EINVAL;
303 return -1;
304 }
305
306 if (req->arm == BRIX_CRED_ARM_VOLATILE) {
307 if (!cred_name_ok(req->prefix)) {
308 errno = EINVAL;
309 return -1;
310 }
311 return cred_write_volatile(req, bytes, len, path_out, path_outsz);
312 }
313
314 if (req->dir == NULL || req->dir[0] == '\0' || !cred_name_ok(req->name)) {
315 errno = EINVAL;
316 return -1;
317 }
318 return cred_write_persistent(req, bytes, len, path_out, path_outsz);
319}
