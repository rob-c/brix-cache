38/*
39 * Phase 51 (A3): per-source-IP connection accounting (per-worker), so one peer
40 * cannot consume every slot of the global cap. A small open-addressing table
41 * keyed by the peer IP string; linear-probe. If the table is full (an abusive
42 * fan-out of distinct IPs — itself bounded by the global cap) we fail OPEN: the
43 * global cap still bounds the total, we just stop enforcing per-IP for the
44 * overflow. Lock-free: per-worker, event-loop only.
45 */
46#define BRIX_CMS_IP_SLOTS 4096
47
48typedef struct {
49 char ip[BRIX_CMS_HANDLER_IP_BUF]; /* peer IP string; '\0' = free slot */
50 ngx_uint_t count; /* live connections from this IP in this worker */
51} brix_cms_ip_slot_t;
52
53static brix_cms_ip_slot_t brix_cms_ip_table[BRIX_CMS_IP_SLOTS];
54
55static ngx_uint_t
56brix_cms_ip_hash(const char *ip)
57{
58 ngx_uint_t h = BRIX_CMS_IP_HASH_INIT;
59 const u_char *p = (const u_char *) ip;
60
61 while (*p) {
62 h = ((h << 5) + h) ^ (ngx_uint_t) *p++; /* djb2-xor */
63 }
64 return h % BRIX_CMS_IP_SLOTS;
65}

67/* Find the slot for ip; if create, claim a free slot when absent. Returns NULL
68 * when absent and (not create OR table full). Probes a bounded window. */
69static brix_cms_ip_slot_t *
70brix_cms_ip_find(const char *ip, int create)
71{
72 ngx_uint_t start = brix_cms_ip_hash(ip);
73 ngx_uint_t i;
74
75 for (i = 0; i < BRIX_CMS_IP_SLOTS; i++) {
76 brix_cms_ip_slot_t *slot =
77 &brix_cms_ip_table[(start + i) % BRIX_CMS_IP_SLOTS];
78
79 if (slot->ip[0] == '\0') {
80 if (!create) {
81 return NULL; /* absent */
82 }
83 ngx_cpystrn((u_char *) slot->ip, (u_char *) ip, sizeof(slot->ip));
84 slot->count = 0;
85 return slot;
86 }
87 if (ngx_strcmp(slot->ip, ip) == 0) {
88 return slot; /* found */
89 }
90 }
91 return NULL; /* table full */
92}

183/*
184 * server_handler.c — CMS server connection handler
185 *
186 * WHAT: Accepts TCP connections from the XRootD CMS manager and maintains a
187 * persistent heartbeat session. Each accepted connection is assigned a
188 * read/write handler pair that exchanges periodic load reports.
189 *
190 * WHY: The CMS manager needs to know which data servers are alive, how much
191 * free space they have, and their utilisation percentage so it can route
192 * client requests (kXR_locate / kXR_redirect) to the best server.
193 *
194 * HOW: On accept we allocate a cms_srv_ctx_t on the connection pool, set up
195 * ping_timer for periodic heartbeats, assign read/write handlers,
196 * and immediately arm the first read event. The read handler (recv.c)
197 * dispatches incoming frames; the write handler (send.c) fires load
198 * reports at conf->interval_ms intervals.
199 */
200
201void
202brix_cms_srv_handler(ngx_stream_session_t *s)
203/*
204 *
205 * WHAT: Entry point for CMS server connections accepted by the stream module.
206 * Allocates context, sets up handlers and timer, arms first read.
207 *
208 * WHY: The CMS manager connects to this port as a data-server client. We need
209 * a per-connection state object (ctx) that tracks the connection pointer,
210 * how many bytes remain in the current frame header, the peer host string,
211 * and the heartbeat timer.
212 *
213 * HOW: 1. pcalloc ctx on c->pool (auto-cleaned on pool destruction).
214 * 2. Initialise ctx fields: c pointer, hdr-in_need = CMS_HDR_LEN,
215 * host from ngx_sock_ntop, interval_ms from conf.
216 * 3. Set ping_timer log/data, assign c->data = ctx.
217 * 4. Replace default read/write handlers with cms_srv_read/cms_srv_write.
218 * 5. Log debug line and arm first read.
219 */
220
221{
222 ngx_connection_t *c;
223 brix_cms_srv_ctx_t *ctx;
224 ngx_stream_brix_cms_srv_conf_t *conf;
225 ngx_stream_brix_srv_conf_t *root_conf;
226 size_t len;
227
228 c = s->connection;
229
230 ctx = ngx_pcalloc(c->pool, sizeof(brix_cms_srv_ctx_t));
231 if (ctx == NULL) {
232 ngx_stream_finalize_session(s, NGX_STREAM_INTERNAL_SERVER_ERROR);
233 return;
234 }
235
236 ctx->c = c;
237 ctx->in_need = NGX_BRIX_CMS_HDR_LEN;
238
239 len = ngx_sock_ntop(c->sockaddr, c->socklen,
240 (u_char *) ctx->host, sizeof(ctx->host) - 1, 0);
241 ctx->host[len] = '\0';
242
243 conf = ngx_stream_get_module_srv_conf(s, ngx_stream_brix_cms_srv_module);
244 ctx->conf = conf;
245 ctx->interval_ms = (ngx_msec_t) conf->interval * BRIX_CMS_SEC_TO_MS_MULTIPLIER;
246 if (ctx->interval_ms < BRIX_CMS_MIN_INTERVAL_MS) {
247 /* Never arm a 0/sub-1s self-rearming ping timer: interval 0 → 0ms timer
248 * → epoll_wait(timeout=0) busy-loop pinning the worker. Floor at 1s. */
249 ctx->interval_ms = BRIX_CMS_MIN_INTERVAL_MS;
250 }
251 ctx->login_timeout_ms = conf->login_timeout; /* WS3 */
252 ctx->idle_timeout_ms = conf->idle_timeout; /* WS3 */
253
254 /*
255 * WS4 — admission cap. Reject before installing any frame handler so an
256 * abusive admitted peer cannot exhaust memory/fds with idle connections.
257 * 0 = unlimited (back-compat).
258 */
259 if (conf->max_connections > 0
260 && brix_cms_srv_conn_count() >= (ngx_uint_t) conf->max_connections)
261 {
262 BRIX_LOG(NGX_LOG_WARN, c->log, 0,
263 "brix: CMS server: connection from %s refused — "
264 "max_connections (%i) reached",
265 ctx->host, conf->max_connections);
266 BRIX_RESIL_METRIC_INC(cms_cap_rejections_total);
267 ngx_stream_finalize_session(s, NGX_STREAM_FORBIDDEN);
268 return;
269 }
270
271 /*
272 * A3 — per-source-IP admission cap, so one peer cannot consume every slot of
273 * the global cap above. 0 = disabled (back-compat).
274 */
275 if (conf->max_connections_per_ip > 0
276 && brix_cms_srv_ip_count(ctx->host)
277 >= (ngx_uint_t) conf->max_connections_per_ip)
278 {
279 BRIX_LOG(NGX_LOG_WARN, c->log, 0,
280 "brix: CMS server: connection from %s refused — "
281 "max_connections_per_ip (%i) reached",
282 ctx->host, conf->max_connections_per_ip);
283 BRIX_RESIL_METRIC_INC(cms_cap_rejections_total);
284 ngx_stream_finalize_session(s, NGX_STREAM_FORBIDDEN);
285 return;
286 }
287
288 /*
289 * W1b — accept-time CIDR allowlist gate. Reject before installing any
290 * frame handler so an unauthorised peer never reaches the LOGIN/registry
291 * path. An SSS-only listener passes this network-layer check and is gated
292 * cryptographically before registration; a listener with neither control
293 * is refused here as a defense-in-depth tripwire.
294 */
295 if (brix_cms_srv_check_peer(c, conf) != NGX_OK) {
296 BRIX_DIAG(NGX_LOG_NOTICE, c->log, 0,
297 "xrootd[cms]: rejected data-server registration from %s",
298 "the peer's IP is not covered by the brix_cms_server_allow "
299 "allowlist (or the allowlist is wrong)",
300 "if this is a legitimate data server, add its IP/CIDR to "
301 "brix_cms_server_allow; otherwise this is an unauthorised peer "
302 "being correctly refused",
303 ctx->host);
304 ngx_stream_finalize_session(s, NGX_STREAM_FORBIDDEN);
305 return;
306 }
307
308 /* W1a — require the sss handshake before registration iff a keytab is set. */
309 ctx->auth_state = (conf->sss_keys != NULL) ? CMS_AUTH_REQUESTED
310 : CMS_AUTH_NONE;
311
312 root_conf = ngx_stream_get_module_srv_conf(s, ngx_stream_brix_module);
313 ctx->role_type = (root_conf != NULL && root_conf->caps.supervisor)
314 ? "R" : "M";
315 /* Pure manager (no local export): stock cmsd stamps kYR_metaman on its
316 * downward state queries so nodes know the sender holds nothing itself. */
317 ctx->metaman = (root_conf != NULL && !root_conf->caps.supervisor
318 && root_conf->rootfd < 0);
319 if (root_conf != NULL) {
320 ctx->sess = brix_sess_begin(root_conf->session_log,
321 root_conf->access_log_fd,
322 BRIX_SESS_PROTO_CMS,
323 BRIX_SESS_DIR_IN,
324 ctx->host, ngx_strlen(ctx->host),
325 BRIX_SESS_AM_HOST, NULL);
326 }
327
328 ctx->ping_timer.log = c->log;
329 ctx->ping_timer.data = ctx;
330
331 /* WS4 + A3: admitted — count it globally and per-IP (decremented in
332 * brix_cms_srv_close). */
333 brix_cms_srv_conn_inc();
334 brix_cms_srv_ip_inc(ctx->host);
335 ctx->counted = 1;
336
337 c->data = ctx;
338 c->read->handler = brix_cms_srv_read;
339 c->write->handler = brix_cms_srv_write;
340
341 /*
342 * WS5: OS-level dead-peer reaping on the accepted socket, so a silently-
343 * dropped data node is torn down by the kernel. Best-effort, non-fatal.
344 */
345 brix_apply_tcp_deadpeer_opts(c->fd, conf->tcp_keepalive,
346 conf->tcp_user_timeout);
347
348 /*
349 * WS3: arm the LOGIN handshake deadline (absolute, from accept). A peer that
350 * never completes LOGIN (+sss xauth) — or trickles a partial header — is
351 * closed by the read handler's ev->timedout path instead of squatting a ctx +
352 * fd forever (slowloris). Replaced by the post-login idle watchdog in
353 * cms_srv_complete_login. 0 = disabled.
354 */
355 if (ctx->login_timeout_ms > 0) {
356 ngx_add_timer(c->read, ctx->login_timeout_ms);
357 }
358
359 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, c->log, 0,
360 "brix: CMS server accepted from %s", ctx->host);
361
362 brix_cms_srv_read(c->read);
363}
