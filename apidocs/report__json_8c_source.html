12/* ---- Emit one JSON string safely ----
13 *
14 * WHAT: Writes a quoted JSON string, escaping controls and metacharacters.
15 * WHY: Targets, server evidence and clause text are untrusted report content.
16 * HOW: Escape quote/slash controls and encode other controls as \u00XX.
17 */
18void
19brix_conform_json_string(FILE *out, const char *value)
20{
21 const unsigned char *cursor = (const unsigned char *) (value == NULL ? "" : value);
22
23 fputc('"', out);
24 while (*cursor != '\0') {
25 if (*cursor == '"' || *cursor == '\\') {
26 fputc('\\', out);
27 fputc(*cursor, out);
28 } else if (*cursor == '\n') {
29 fputs("\\n", out);
30 } else if (*cursor == '\r') {
31 fputs("\\r", out);
32 } else if (*cursor == '\t') {
33 fputs("\\t", out);
34 } else if (iscntrl(*cursor)) {
35 fprintf(out, "\\u%04x", (unsigned) *cursor);
36 } else {
37 fputc(*cursor, out);
38 }
39 cursor++;
40 }
41 fputc('"', out);
42}

78/* ---- Emit the baseline changed-only result set ----
79 *
80 * WHAT: Writes a deterministic JSON artifact containing only changed clauses.
81 * WHY: Automation using --baseline asked for transitions, not full-ledger noise.
82 * HOW: Reuse canonical result objects and filter on the comparison bit.
83 */
84void
85brix_conform_report_diff_json(FILE *out,
86 const brix_conform_report_meta *meta,
87 const brix_conform_run_results *results)
88{
89 size_t index;
90 int first = 1;
91
92 fputs("{\"schema\":\"brix.conform.diff.v1\",\"target\":", out);
93 brix_conform_json_string(out, meta->target);
94 fprintf(out, ",\"baseline_applied\":%s,\"changed\":%zu,\"results\":[",
95 results->baseline_applied ? "true" : "false", results->changed);
96 for (index = 0; index < results->count; index++) {
97 if (!results->results[index].changed) continue;
98 if (!first) fputc(',', out);
99 first = 0;
100 json_result(out, &results->results[index]);
101 }
102 fputs("]}\n", out);
103}

156/* ---- Emit SARIF findings for divergent and inconclusive clauses ----
157 *
158 * WHAT: Writes SARIF 2.1.0 results with stable rule ids and fingerprints.
159 * WHY: CI code-scanning views need actionable non-green rows, not pass noise.
160 * HOW: Filter result vector and render one result per actionable clause.
161 */
162void
163brix_conform_report_sarif(FILE *out, const brix_conform_report_meta *meta,
164 const brix_conform_run_results *results)
165{
166 size_t index;
167 int first = 1;
168
169 (void) meta;
170 fputs("{\"version\":\"2.1.0\",\"$schema\":\"https://json.schemastore.org/sarif-2.1.0.json\","
171 "\"runs\":[{\"tool\":{\"driver\":{\"name\":\"xrd conform\"}},\"results\":[", out);
172 for (index = 0; index < results->count; index++) {
173 const brix_conform_result *result = &results->results[index];
174
175 if (result->verdict != BRIX_CONFORM_VERDICT_DIVERGENT
176 && result->verdict != BRIX_CONFORM_VERDICT_INCONCLUSIVE) continue;
177 if (!first) fputc(',', out);
178 first = 0;
179 fputs("{\"ruleId\":\"brix.conform.", out);
180 fputs(result->clause->id, out);
181 fputs("\",\"level\":\"", out);
182 fputs(result->verdict == BRIX_CONFORM_VERDICT_DIVERGENT ? "error" : "warning", out);
183 fputs("\",\"message\":{\"text\":", out);
184 brix_conform_json_string(out, result->reason[0] ? result->reason : result->clause->text);
185 fputs("},\"partialFingerprints\":{\"clause\":", out);
186 brix_conform_json_string(out, result->clause->id);
187 fputs("}}", out);
188 }
189 fputs("]}]}\n", out);
190}
