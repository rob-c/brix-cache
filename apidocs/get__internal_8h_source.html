13/*
14 * get_serve_state_t — the serve-ready handle + derived metadata produced by
15 * the resolve/open phase and consumed by the serving phases. It bundles the
16 * open VFS handle with the POSIX-shaped stat (`sb`, used by the multipart and
17 * pre-header paths), the VFS stat (`vst`, used by the ranged serve), and the
18 * fd-cache bookkeeping (`from_cache`/`cache_path`) so the phase helpers take a
19 * single explicit state object instead of a long parameter list. It is filled
20 * only when the resolve phase returns NGX_OK.
21 */
22typedef struct {
23 brix_vfs_file_t *fh; /* open handle (owned by the caller flow) */
24 struct stat sb; /* POSIX stat for multipart/pre-header use */
25 brix_vfs_stat_t vst; /* VFS stat for the ranged serve */
26 ngx_uint_t from_cache; /* fd came from the read-through cache tier */
27 const char *cache_path; /* cache path for access accounting */
28} get_serve_state_t;
