1/*
2 * io_xfer_unittest.c — standalone unit test for the TPC full-transfer loop.
3 *
4 * gcc -Wall -Wextra -Werror -I src/tpc/outbound -o /tmp/tpc_xfer_ut \
5 * src/tpc/outbound/io_xfer_unittest.c src/tpc/outbound/io_xfer.c \
6 * -lssl -lcrypto && /tmp/tpc_xfer_ut
7 *
8 * Exit 0 = all checks pass. No nginx, no TPC context, no TLS handshake: the
9 * loop is driven over a socketpair with ssl == NULL, which is the cleartext
10 * path every TPC pull uses before (and unless) it upgrades on kXR_gotoTLS.
11 *
12 * What matters here is that a PARTIAL transfer never reports success. TPC wire
13 * framing reads a fixed-size ServerResponseHdr and then a length-prefixed body,
14 * so a loop that returns 0 having moved fewer bytes than asked leaves the
15 * stream desynchronised — the next header is parsed out of body bytes, and an
16 * attacker-controlled source could steer that. Hence the truncation cases.
17 */

42static void
43test_send_then_receive_is_byte_exact(void)
44{
45 int sv[2];
46 char payload[4096];
47 char got[4096];
48
49 mk_pair(sv);
50 memset(payload, 'A', sizeof(payload));
51 memset(got, 0, sizeof(got));
52
53 CHECK(brix_tpc_xfer_all(NULL, sv[0], payload, sizeof(payload),

55 CHECK(brix_tpc_xfer_all(NULL, sv[1], got, sizeof(got),
56 BRIX_TPC_XFER_RECV) == 0);
57 CHECK(memcmp(payload, got, sizeof(payload)) == 0);

63static void
64test_receive_reassembles_a_dribbled_stream(void)
65{
66 int sv[2];
67 char got[64];
68
69 mk_pair(sv);
70 memset(got, 0, sizeof(got));
71
72 /* Three separate writes: the receive loop must return only once all 64
73 * bytes have arrived, not after the first recv() returns 10. */
74 CHECK(write(sv[0], "0123456789", 10) == 10);
75 CHECK(write(sv[0], "abcdefghijklmnopqrstuvwxyz0123", 30) == 30);
76 CHECK(write(sv[0], "ABCDEFGHIJKLMNOPQRSTUVWX", 24) == 24);

79 BRIX_TPC_XFER_RECV) == 0);

162static void
163test_a_short_frame_cannot_be_replayed_as_a_full_one(void)
164{
165 int sv[2];
166 char got[16];
167
168 mk_pair(sv);
169 memset(got, 0xEE, sizeof(got));
170
171 CHECK(write(sv[0], "hdr", 3) == 3);
172 close(sv[0]);
173
174 /* The loop consumed the 3 real bytes and then failed. The tail must still
175 * be the caller's fill, never something the loop invented to reach len. */
176 CHECK(brix_tpc_xfer_all(NULL, sv[1], got, sizeof(got),
177 BRIX_TPC_XFER_RECV) == -1);
178 CHECK(memcmp(got, "hdr", 3) == 0);
179 CHECK((unsigned char) got[15] == 0xEE);
180
181 close(sv[1]);
182}
