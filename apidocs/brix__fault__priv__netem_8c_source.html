1/*
2 * brix_fault_priv_netem.c — the `tc qdisc ... netem` plane of the privileged levers.
3 *
4 * WHAT: the per-feature netem fragment table (delay, loss, corrupt, duplicate,
5 * reorder, rate, limit), the qdisc (re)install, and the `priv netem <sub>`
6 * command surface.
7 *
8 * WHY: split out of brix_fault_priv.c, which owns three independent host-global
9 * subsystems (netem / nft cut / MTU) and had grown past the 600-line cap
10 * (coding-standards §1). netem is the largest and the most self-contained
11 * of the three: it touches only the fragment table and the NIC's egress
12 * qdisc, so it lifts out whole.
13 *
14 * HOW: the whole qdisc is re-emitted on every change (`tc qdisc replace`), so
15 * features compose and any one can be cleared independently. Every operand
16 * is validated before it reaches argv; there is no shell. State stays owned
17 * by brix_fault_priv.c and is reached through brix_fault_priv_internal.h —
18 * the caller holds g_lock for every entry point here.
19 */

28/* Rebuild and install the whole netem qdisc from the current fragment table, or
29 * delete it when every slot is empty. Caller holds g_lock. */
30int
31netem_apply(void)
32{
33 if (g_iface[0] == '\0') {
34 return -2; /* no interface configured */
35 }
36
37 int empty = 1;
38 for (int i = 0; i < NE_N; i++) {
39 if (g_ne[i][0]) {
40 empty = 0;
41 break;
42 }
43 }
44 if (empty) {
45 if (!g_netem_on) {
46 return 0; /* nothing was ever installed */
47 }
48 char *argv[] = { "tc", "qdisc", "del", "dev", g_iface, "root", NULL };
49 int rc = priv_run(argv);
50 g_netem_on = 0;
51 /* rc 2 == "no such qdisc" (already clean); treat as success. */
52 return (rc == 0 || rc == 2) ? 0 : -1;
53 }
54
55 /* tc qdisc replace dev IFACE root netem <tokens...> */
56 char *argv[8 + NE_N * 8];
57 int ac = 0;
58 argv[ac++] = "tc";
59 argv[ac++] = "qdisc";
60 argv[ac++] = "replace";
61 argv[ac++] = "dev";
62 argv[ac++] = g_iface;
63 argv[ac++] = "root";
64 argv[ac++] = "netem";
65
66 static char frag[NE_N][96]; /* mutable copies; strtok writes into these */
67 for (int i = 0; i < NE_N; i++) {
68 if (!g_ne[i][0]) {
69 continue;
70 }
71 memcpy(frag[i], g_ne[i], sizeof(frag[i]));
72 for (char *t = strtok(frag[i], " "); t; t = strtok(NULL, " ")) {
73 if (ac < (int) (sizeof(argv) / sizeof(argv[0])) - 1) {
74 argv[ac++] = t;
75 }
76 }
77 }
78 argv[ac] = NULL;
79 if (priv_run(argv) != 0) {
80 return -1;
81 }
82 g_netem_on = 1;
83 return 0;
84}

86/* Set the delay slot: "delay <ms>ms [<jit>ms [<corr>% [distribution <d>]]]". */
87static int
88netem_set_delay(char *args, char *reply, size_t rsz)
89{
90 char *ms = strtok(args, " ");
91 char *jit = strtok(NULL, " ");
92 char *corr = strtok(NULL, " ");
93 char *dist = strtok(NULL, " ");
94 if (!ms || strcmp(ms, "off") == 0) {
95 g_ne[NE_DELAY][0] = '\0';
96 return netem_apply();
97 }
98 char mt[24], jt[24], ct[16];
99 if (fmt_uint(ms, mt, sizeof(mt), 0) != 0) {
100 snprintf(reply, rsz, "err: bad delay ms\n");
101 return -1;
102 }
103 int n = snprintf(g_ne[NE_DELAY], sizeof(g_ne[NE_DELAY]), "delay %sms", mt);
104 if (jit && fmt_uint(jit, jt, sizeof(jt), 0) == 0) {
105 n += snprintf(g_ne[NE_DELAY] + n, sizeof(g_ne[NE_DELAY]) - n, " %sms", jt);
106 if (corr && fmt_pct(corr, ct, sizeof(ct)) == 0) {
107 n += snprintf(g_ne[NE_DELAY] + n, sizeof(g_ne[NE_DELAY]) - n, " %s", ct);
108 }
109 }
110 if (dist && (!strcmp(dist, "normal") || !strcmp(dist, "pareto") ||
111 !strcmp(dist, "paretonormal") || !strcmp(dist, "uniform"))) {
112 snprintf(g_ne[NE_DELAY] + n, sizeof(g_ne[NE_DELAY]) - n,
113 " distribution %s", dist);
114 }
115 return netem_apply();
116}

118/* Set one single-percentage netem feature (corrupt/duplicate/reorder/loss). */
119static int
120netem_set_pct(int slot, const char *verb, char *args, char *reply, size_t rsz)
121{
122 char *p = strtok(args, " ");
123 char *corr = strtok(NULL, " ");
124 if (!p || strcmp(p, "off") == 0) {
125 g_ne[slot][0] = '\0';
126 return netem_apply();
127 }
128 char pt[16], ct[16];
129 if (fmt_pct(p, pt, sizeof(pt)) != 0) {
130 snprintf(reply, rsz, "err: bad %s percentage\n", verb);
131 return -1;
132 }
133 int n = snprintf(g_ne[slot], sizeof(g_ne[slot]), "%s %s", verb, pt);
134 if (corr && fmt_pct(corr, ct, sizeof(ct)) == 0) {
135 snprintf(g_ne[slot] + n, sizeof(g_ne[slot]) - n, " %s", ct);
136 }
137 return netem_apply();
138}

140/* Set bursty Gilbert-Elliott loss: "loss gemodel <p>% [<r>% [<1-h>% [<1-k>%]]]".
141 * p = good->bad, r = bad->good, 1-h = loss-in-bad, 1-k = loss-in-good. */
142static int
143netem_set_gemodel(char *args, char *reply, size_t rsz)
144{
145 char *tok[4] = { 0 };
146 int nt = 0;
147 for (char *t = strtok(args, " "); t && nt < 4; t = strtok(NULL, " ")) {
148 tok[nt++] = t;
149 }
150 if (nt == 0 || strcmp(tok[0], "off") == 0) {
151 g_ne[NE_LOSS][0] = '\0';
152 return netem_apply();
153 }
154 char buf[96];
155 int n = snprintf(buf, sizeof(buf), "loss gemodel");
156 for (int i = 0; i < nt; i++) {
157 char pt[16];
158 if (fmt_pct(tok[i], pt, sizeof(pt)) != 0) {
159 snprintf(reply, rsz, "err: bad gemodel percentage\n");
160 return -1;
161 }
162 n += snprintf(buf + n, sizeof(buf) - n, " %s", pt);
163 }
164 memcpy(g_ne[NE_LOSS], buf, sizeof(g_ne[NE_LOSS]));
165 return netem_apply();
166}

168/* `netem show` — the armed clauses, in qdisc order. */
169static void
170netem_show(char *reply, size_t rsz)
171{
172 char *w = reply;
173 int left = (int) rsz;
174 int k = snprintf(w, left, "netem[%s]:", g_iface[0] ? g_iface : "?");
175 w += k; left -= k;
176 for (int i = 0; i < NE_N && left > 1; i++) {
177 if (g_ne[i][0]) {
178 int m = snprintf(w, left, " %s", g_ne[i]);
179 w += m; left -= m;
180 }
181 }
182 snprintf(w, left > 0 ? left : 0, "\n");
183}

229#define NETEM_NO_SUB (-3)

231static int
232netem_arm(const char *sub, char *rest, char *reply, size_t rsz)
233{
234 if (strcmp(sub, "delay") == 0) {
235 return netem_set_delay(rest, reply, rsz);
236 }
237 if (strcmp(sub, "loss") == 0) {
238 return netem_set_pct(NE_LOSS, "loss random", rest, reply, rsz);
239 }
240 if (strcmp(sub, "loss-gemodel") == 0) {
241 return netem_set_gemodel(rest, reply, rsz);
242 }
243 if (strcmp(sub, "corrupt") == 0) {
244 return netem_set_pct(NE_CORRUPT, "corrupt", rest, reply, rsz);
245 }
246 if (strcmp(sub, "duplicate") == 0) {
247 return netem_set_pct(NE_DUP, "duplicate", rest, reply, rsz);
248 }
249 if (strcmp(sub, "reorder") == 0) {
250 return netem_set_pct(NE_REORDER, "reorder", rest, reply, rsz);
251 }
252 if (strcmp(sub, "rate") == 0) {
253 return netem_set_rate(rest, reply, rsz);
254 }
255 if (strcmp(sub, "limit") == 0) {
256 return netem_set_limit(rest, reply, rsz);
257 }
258 return NETEM_NO_SUB;
259}

261/* Turn an arming result into the operator-facing reply. */
262static void
263netem_report(int rc, char *reply, size_t rsz)
264{
265 if (rc == -2) {
266 snprintf(reply, rsz, "err: no --priv-iface configured for netem\n");
267 } else if (rc != 0 && strncmp(reply, "err:", 4) != 0) {
268 /* A validation handler already left a specific "err: ..." message; only
269 * the qdisc install itself falls through with reply still == "ok". */
270 snprintf(reply, rsz, "err: tc failed (see proxy stderr)\n");
271 } else if (rc == 0 && g_ne[NE_REORDER][0] && !g_ne[NE_DELAY][0]) {
272 snprintf(reply, rsz, "ok (note: reorder needs a delay to take effect)\n");
273 }
274}

276int
277netem_command(char *args, char *reply, size_t rsz)
278{
279 char *sub = strtok(args, " ");
280 char *rest = strtok(NULL, ""); /* remainder after the sub-verb */
281 char empty[1] = "";
282 if (!rest) {
283 rest = empty;
284 }
285 if (!sub || strcmp(sub, "show") == 0) {
286 netem_show(reply, rsz);
287 return 0;
288 }
289 if (strcmp(sub, "clear") == 0 || strcmp(sub, "off") == 0) {
290 return netem_clear();
291 }
292 int rc = netem_arm(sub, rest, reply, rsz);
293 if (rc == NETEM_NO_SUB) {
294 snprintf(reply, rsz, "err: unknown netem sub-command '%s'\n", sub);
295 return -1;
296 }
297 netem_report(rc, reply, rsz);
298 return rc == 0 ? 0 : -1;
299}
