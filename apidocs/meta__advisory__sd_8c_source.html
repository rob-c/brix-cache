12/*
13 * WHAT: Translate the present groups of *attr into *delta.
14 * WHY: See the header — the three sentinels (UTIME_OMIT, UTIME_NOW, and the
15 * (id)-1 "leave unchanged" pair) are the part that must not be re-derived
16 * per backend.
17 * HOW: atime is dropped: no object store tracks it, and the advisory blob has
18 * no field for it, so a request that carries ONLY an atime is reported as
19 * "nothing to persist" (return 0) rather than provoking a write that would
20 * store nothing.
21 */
22int
23brix_meta_advisory_from_setattr(const brix_sd_setattr_t *attr,
24 brix_meta_advisory_t *delta)
25{
26 if (attr == NULL || delta == NULL) {
27 return 0;
28 }
29 memset(delta, 0, sizeof(*delta));
30
31 if (attr->set_mode) {
32 delta->have_mode = 1;
33 delta->mode = attr->mode;
34 }
35
36 /* Owner: the advisory model carries uid AND gid together, so a lone (id)-1
37 * ("leave this one unchanged") makes the change unrepresentable — persist
38 * neither rather than inventing a value for the omitted half. */
39 if (attr->set_owner && attr->uid != (uid_t) -1 && attr->gid != (gid_t) -1) {
40 delta->have_owner = 1;
41 delta->uid = attr->uid;
42 delta->gid = attr->gid;
43 }
44
45 if (attr->set_times && attr->mtime.tv_nsec != UTIME_OMIT) {
46 delta->have_mtime = 1;
47 if (attr->mtime.tv_nsec == UTIME_NOW) {
48 delta->mtime = time(NULL);
49 delta->mtime_ns = 0;
50 } else {
51 delta->mtime = attr->mtime.tv_sec;
52 delta->mtime_ns = attr->mtime.tv_nsec;
53 }
54 }
55
56 return (delta->have_mode || delta->have_owner || delta->have_mtime) ? 1 : 0;
57}
