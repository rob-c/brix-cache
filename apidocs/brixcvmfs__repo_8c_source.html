1/* brixcvmfs_repo.c — `brixcvmfs repo` family: Stratum-0 repository lifecycle
2 * (phase-96 S3). Unprivileged, tool-surface only (G14): no FUSE, no root.
3 *
4 * brixcvmfs repo mkfs <fqrn> <repo_dir> [keys_dir] — mint a repository
5 * brixcvmfs repo info <repo_dir> [keys_dir] — parse + verify
6 * brixcvmfs repo resign <repo_dir> [keys_dir] — re-sign trust chain
7 *
8 * mkfs creates the full on-disk shape the read stack verifies: RSA master +
9 * certificate keys (generated once, reused when present), an empty root
10 * catalog (revision 1) CAS-stored under data/, the certificate object, the
11 * reflog, and the signed .cvmfspublished / .cvmfswhitelist pair. `info` is the
12 * local fsck-lite: it re-runs the whole trust chain (whitelist sig → expiry →
13 * fingerprint membership → manifest sig → body binding) and exits non-zero on
14 * any break. `resign` refreshes both signatures + whitelist expiry in place.
15 */

48static unsigned char *rp_read_file(const char *path, size_t *len) {
49 FILE *f = fopen(path, "rb");
50 if (f == NULL) return NULL;
51 if (fseek(f, 0, SEEK_END) != 0) { fclose(f); return NULL; }
52 long n = ftell(f);
53 if (n < 0) { fclose(f); return NULL; }
54 rewind(f);
55 unsigned char *buf = malloc(n > 0 ? (size_t) n : 1);
56 if (buf == NULL) { fclose(f); return NULL; }
57 int ok = n == 0 || fread(buf, 1, (size_t) n, f) == (size_t) n;
58 fclose(f);
59 if (!ok) { free(buf); return NULL; }
60 *len = (size_t) n;
61 return buf;
62}

82/* Self-signed X.509 for the repo signing key, CN=<fqrn>. */
83static int rp_write_cert(const char *path, EVP_PKEY *pk, const char *fqrn) {
84 X509 *x = X509_new();
85 if (x == NULL) return -1;
86 ASN1_INTEGER_set(X509_get_serialNumber(x), (long) getpid());
87 X509_gmtime_adj(X509_getm_notBefore(x), 0);
88 X509_gmtime_adj(X509_getm_notAfter(x), 3650L * 86400);
89 X509_set_pubkey(x, pk);
90 X509_NAME *name = X509_get_subject_name(x);
91 X509_NAME_add_entry_by_txt(name, "CN", MBSTRING_ASC,
92 (const unsigned char *) fqrn, -1, -1, 0);
93 X509_set_issuer_name(x, name);
94 int ok = X509_sign(x, pk, EVP_sha256()) != 0;
95 FILE *f = ok ? fopen(path, "w") : NULL;
96 if (f != NULL) {
97 ok = PEM_write_X509(f, x) == 1;
98 ok = fclose(f) == 0 && ok;
99 } else {
100 ok = 0;
101 }
102 X509_free(x);
103 return ok ? 0 : -1;
104}

117/* Load the key pair from keys_dir, generating any missing piece (mkfs), or
118 * failing when generation is not allowed (resign). */
119static int rp_load_keys(const char *keys_dir, const char *fqrn, int generate,
120 EVP_PKEY **master, EVP_PKEY **certpk) {
121 rp_keypaths_t p;
122 rp_key_paths(keys_dir, fqrn, &p);
123
124 *master = cvmfs_sign_load_key(p.masterkey);
125 *certpk = cvmfs_sign_load_key(p.certkey);
126 if (*master != NULL && *certpk != NULL) return 0;
127 if (!generate) {
128 EVP_PKEY_free(*master);
129 EVP_PKEY_free(*certpk);
130 return rp_err("signing keys missing under", keys_dir);
131 }
132
133 if (mkdir(keys_dir, 0700) != 0 && errno != EEXIST)
134 return rp_err("cannot create keys dir", keys_dir);
135 if (*master == NULL) {
136 *master = EVP_RSA_gen(2048);
137 if (*master == NULL || rp_write_pem(p.masterkey, *master, 1) != 0
138 || rp_write_pem(p.masterpub, *master, 0) != 0)
139 return rp_err("master key generation failed", NULL);
140 }
141 if (*certpk == NULL) {
142 *certpk = EVP_RSA_gen(2048);
143 if (*certpk == NULL || rp_write_pem(p.certkey, *certpk, 1) != 0
144 || rp_write_cert(p.cert, *certpk, fqrn) != 0)
145 return rp_err("certificate generation failed", NULL);
146 }
147 return 0;
148}

171static int rp_write_whitelist(const char *repo_dir, const char *fqrn,
172 const unsigned char *cert_pem, size_t cert_len,
173 EVP_PKEY *master) {
174 char fp[1][60];
175 if (cvmfs_cert_fingerprint(cert_pem, cert_len, fp[0], sizeof(fp[0])) != 0)
176 return -1;
177
178 time_t exp = time(NULL) + WHITELIST_DAYS * 86400L;
179 struct tm tm_utc;
180 char expiry[16];
181 gmtime_r(&exp, &tm_utc);
182 strftime(expiry, sizeof(expiry), "%Y%m%d%H%M%S", &tm_utc);
183
184 char body[2048], path[600];
185 int blen = cvmfs_whitelist_body(NULL, expiry, fqrn, (const char (*)[60]) fp, 1,
186 body, sizeof(body));
187 snprintf(path, sizeof(path), "%s/.cvmfswhitelist", repo_dir);
188 return rp_write_signed(path, body, blen, master, 0); /* whitelist: raw */
189}

193/* Mint the empty revision-1 root catalog and CAS-store it. */
194static int rp_store_root_catalog(const char *repo_dir, cvmfs_objstore_t *store,
195 cvmfs_hash_t *out, size_t *stored_len) {
196 char tmp[600];
197 snprintf(tmp, sizeof(tmp), "%s/.brix.catalog.tmp", repo_dir);
198 unlink(tmp);
199
200 cvmfs_catwriter_t *w = cvmfs_catwriter_create(tmp);
201 if (w == NULL) return -1;
202 cvmfs_catrow_t root;
203 memset(&root, 0, sizeof(root));
204 root.path = "";
205 root.flags = CVMFS_FLAG_DIR;
206 root.mode = 040755;
207 root.size = 4096;
208 root.mtime = (int64_t) time(NULL);
209 char now[32];
210 snprintf(now, sizeof(now), "%ld", (long) time(NULL));
211 if (cvmfs_catwriter_insert(w, &root) != 0
212 || cvmfs_catwriter_set_property(w, "revision", "1") != 0
213 || cvmfs_catwriter_set_property(w, "schema", "2.5") != 0
214 || cvmfs_catwriter_set_property(w, "schema_revision", "2") != 0
215 || cvmfs_catwriter_set_property(w, "last_modified", now) != 0
216 || cvmfs_catwriter_update_counters(w) != 0) {
217 cvmfs_catwriter_abort(w);
218 unlink(tmp);
219 return -1;
220 }
221 if (cvmfs_catwriter_commit(w) != 0) { unlink(tmp); return -1; }
222
223 size_t plain_len = 0;
224 unsigned char *plain = rp_read_file(tmp, &plain_len);
225 unlink(tmp);
226 if (plain == NULL) return -1;
227 int rc = cvmfs_object_store(store, plain, plain_len, 'C', 1, out, stored_len);
228 free(plain);
229 return rc;
230}

232/* Seed .cvmfsreflog with the root anchors and compute the 'Y' checksum. */
233static int rp_seed_reflog(const char *repo_dir, const cvmfs_hash_t *cat,
234 const cvmfs_hash_t *cert, cvmfs_hash_t *checksum) {
235 char path[600];
236 snprintf(path, sizeof(path), "%s/.cvmfsreflog", repo_dir);
237 cvmfs_reflog_t *r = cvmfs_reflog_open(path);
238 if (r == NULL) return -1;
239 int64_t now = (int64_t) time(NULL);
240 int rc = cvmfs_reflog_add(r, cat, CVMFS_REFLOG_CATALOG, now);
241 if (rc == 0) rc = cvmfs_reflog_add(r, cert, CVMFS_REFLOG_CERTIFICATE, now);
242 if (cvmfs_reflog_close(r) != 0) rc = -1;
243 if (rc == 0) rc = cvmfs_reflog_checksum(path, checksum);
244 return rc;
245}

247/* Emit the revision-1 objects + signed trust chain into an opened store. */
248static int rp_mkfs_emit(const char *fqrn, const char *repo_dir,
249 cvmfs_objstore_t *store, const unsigned char *cert_pem,
250 size_t cert_len, EVP_PKEY *master, EVP_PKEY *certpk,
251 cvmfs_hash_t *root_out) {
252 cvmfs_manifest_wr_t m;
253 memset(&m, 0, sizeof(m));
254 size_t csize = 0;
255 if (rp_store_root_catalog(repo_dir, store, &m.root_catalog, &csize) != 0)
256 return rp_err("root catalog emission failed", NULL);
257 if (cvmfs_object_store(store, cert_pem, cert_len, 'X', 1,
258 &m.certificate, NULL) != 0)
259 return rp_err("certificate object store failed", NULL);
260 if (rp_seed_reflog(repo_dir, &m.root_catalog, &m.certificate,
261 &m.reflog_checksum) != 0)
262 return rp_err("reflog seed failed", NULL);
263 m.catalog_size = (long) csize;
264 m.revision = 1;
265 m.fqrn = fqrn;
266 m.timestamp = (long) time(NULL);
267 m.ttl = REPO_TTL_S;
268 if (rp_write_manifest(repo_dir, &m, certpk) != 0
269 || rp_write_whitelist(repo_dir, fqrn, cert_pem, cert_len, master) != 0)
270 return rp_err("trust-chain signing failed", NULL);
271 *root_out = m.root_catalog;
272 return 0;
273}

275static int rp_mkfs(const char *fqrn, const char *repo_dir, const char *keys_dir) {
276 char manifest_path[600];
277 snprintf(manifest_path, sizeof(manifest_path), "%s/.cvmfspublished", repo_dir);
278 if (access(manifest_path, F_OK) == 0)
279 return rp_err("repository already published at", repo_dir);
280 if (mkdir(repo_dir, 0755) != 0 && errno != EEXIST)
281 return rp_err("cannot create repo dir", repo_dir);
282
283 EVP_PKEY *master = NULL, *certpk = NULL;
284 if (rp_load_keys(keys_dir, fqrn, 1, &master, &certpk) != 0) return 1;
285
286 rp_keypaths_t kp;
287 rp_key_paths(keys_dir, fqrn, &kp);
288 size_t cert_len = 0;
289 unsigned char *cert_pem = rp_read_file(kp.cert, &cert_len);
290
291 cvmfs_objstore_t store;
292 int rc = cert_pem != NULL && cvmfs_objstore_open(&store, repo_dir) == 0 ? 0 : 1;
293
294 cvmfs_hash_t root;
295 if (rc == 0)
296 rc = rp_mkfs_emit(fqrn, repo_dir, &store, cert_pem, cert_len,
297 master, certpk, &root);
298 if (rc == 0) {
299 char hex[64];
300 cvmfs_hash_to_hex(&root, 0, hex, sizeof(hex));
301 printf("repository %s created at %s\n revision 1, root catalog %s\n"
302 " keys in %s (public key: %s.pub)\n", fqrn, repo_dir, hex,
303 keys_dir, fqrn);
304 }
305
306 if (cert_pem != NULL) cvmfs_objstore_close(&store);
307 free(cert_pem);
308 EVP_PKEY_free(master);
309 EVP_PKEY_free(certpk);
310 return rc;
311}

326/* Pull + inflate the certificate object named by the manifest. */
327static unsigned char *rp_load_cert_object(const char *repo_dir, const cvmfs_manifest_t *m,
328 size_t *cert_len) {
329 cvmfs_objstore_t store;
330 if (cvmfs_objstore_open(&store, repo_dir) != 0) return NULL;
331 unsigned char stored[65536];
332 unsigned char *pem = NULL;
333 long n = cvmfs_object_read_stored(&store, &m->certificate, 'X',
334 stored, sizeof(stored));
335 if (n > 0) {
336 pem = malloc(65536);
337 if (pem != NULL
338 && cvmfs_object_inflate(stored, (size_t) n, pem, 65536, cert_len) != 0) {
339 free(pem);
340 pem = NULL;
341 }
342 }
343 cvmfs_objstore_close(&store);
344 return pem;
345}

347/* Verify whitelist sig + expiry + cert-fingerprint membership. 0 = trusted. */
348static int rp_check_whitelist(const char *repo_dir, const char *keys_dir,
349 const cvmfs_manifest_t *m,
350 const unsigned char *cert_pem, size_t cert_len) {
351 char path[600];
352 snprintf(path, sizeof(path), "%s/.cvmfswhitelist", repo_dir);
353 size_t wlen = 0;
354 unsigned char *wbuf = rp_read_file(path, &wlen);
355 if (wbuf == NULL) return rp_err("whitelist missing", path);
356
357 rp_keypaths_t kp;
358 rp_key_paths(keys_dir, m->repo_name, &kp);
359 size_t plen = 0;
360 unsigned char *pub = rp_read_file(kp.masterpub, &plen);
361
362 cvmfs_whitelist_t w;
363 char fp[60];
364 int rc = pub == NULL ? rp_err("master public key missing", kp.masterpub)
365 : cvmfs_whitelist_parse(wbuf, wlen, &w) != 0 ? rp_err("whitelist unparsable", NULL)
366 : cvmfs_verify_whitelist(&w, pub, plen) != 0 ? rp_err("whitelist signature INVALID", NULL)
367 : cvmfs_whitelist_expired(&w, (long) time(NULL)) ? rp_err("whitelist EXPIRED", NULL)
368 : cvmfs_cert_fingerprint(cert_pem, cert_len, fp, sizeof(fp)) != 0
369 || !cvmfs_whitelist_lists_fp(&w, fp) ? rp_err("certificate not whitelisted", NULL)
370 : 0;
371 free(pub);
372 free(wbuf);
373 return rc;
374}

376static int rp_info(const char *repo_dir, const char *keys_dir) {
377 cvmfs_manifest_t m;
378 unsigned char *mbuf = rp_load_manifest(repo_dir, &m);
379 if (mbuf == NULL) return rp_err("no parsable .cvmfspublished in", repo_dir);
380
381 char root_hex[64], cert_hex[64];
382 cvmfs_hash_to_hex(&m.root_catalog, 0, root_hex, sizeof(root_hex));
383 cvmfs_hash_to_hex(&m.certificate, 0, cert_hex, sizeof(cert_hex));
384 printf("repository ...... %s\n", m.repo_name);
385 printf("revision ........ %ld\n", m.revision);
386 printf("root catalog .... %s (%ld bytes stored)\n", root_hex, m.catalog_size);
387 printf("certificate ..... %s\n", cert_hex);
388 printf("published ....... %ld (ttl %lds)\n", m.timestamp, m.ttl);
389
390 size_t cert_len = 0;
391 unsigned char *cert_pem = rp_load_cert_object(repo_dir, &m, &cert_len);
392 int rc;
393 if (cert_pem == NULL) {
394 rc = rp_err("certificate object unreadable", cert_hex);
395 } else if (cvmfs_verify_manifest(&m, cert_pem, cert_len) != 0) {
396 rc = rp_err("manifest signature INVALID", NULL);
397 } else {
398 rc = rp_check_whitelist(repo_dir, keys_dir, &m, cert_pem, cert_len);
399 }
400 if (rc == 0) {
401 cvmfs_hash_t y;
402 char reflog[600];
403 snprintf(reflog, sizeof(reflog), "%s/.cvmfsreflog", repo_dir);
404 if (m.reflog_checksum.len > 0
405 && (cvmfs_reflog_checksum(reflog, &y) != 0 || !cvmfs_hash_eq(&y, &m.reflog_checksum)))
406 rc = rp_err("reflog checksum mismatch (tamper or partial publish)", NULL);
407 }
408 printf("trust chain ..... %s\n", rc == 0 ? "OK" : "BROKEN");
409 free(cert_pem);
410 free(mbuf);
411 return rc;
412}

416static int rp_resign(const char *repo_dir, const char *keys_dir) {
417 cvmfs_manifest_t m;
418 unsigned char *mbuf = rp_load_manifest(repo_dir, &m);
419 if (mbuf == NULL) return rp_err("no parsable .cvmfspublished in", repo_dir);
420
421 EVP_PKEY *master = NULL, *certpk = NULL;
422 if (rp_load_keys(keys_dir, m.repo_name, 0, &master, &certpk) != 0) {
423 free(mbuf);
424 return 1;
425 }
426
427 cvmfs_manifest_wr_t wr;
428 memset(&wr, 0, sizeof(wr));
429 wr.root_catalog = m.root_catalog;
430 wr.catalog_size = m.catalog_size;
431 wr.certificate = m.certificate;
432 wr.revision = m.revision;
433 wr.fqrn = m.repo_name;
434 wr.timestamp = (long) time(NULL);
435 wr.ttl = m.ttl;
436 wr.history = m.history;
437 char reflog[600];
438 snprintf(reflog, sizeof(reflog), "%s/.cvmfsreflog", repo_dir);
439 if (access(reflog, F_OK) == 0
440 && cvmfs_reflog_checksum(reflog, &wr.reflog_checksum) != 0) {
441 free(mbuf);
442 EVP_PKEY_free(master);
443 EVP_PKEY_free(certpk);
444 return rp_err("reflog checksum failed", reflog);
445 }
446
447 rp_keypaths_t kp;
448 rp_key_paths(keys_dir, m.repo_name, &kp);
449 size_t cert_len = 0;
450 unsigned char *cert_pem = rp_read_file(kp.cert, &cert_len);
451 int rc = cert_pem == NULL ? rp_err("certificate missing", kp.cert)
452 : rp_write_manifest(repo_dir, &wr, certpk) != 0 ? rp_err("manifest resign failed", NULL)
453 : rp_write_whitelist(repo_dir, m.repo_name, cert_pem, cert_len, master) != 0
454 ? rp_err("whitelist resign failed", NULL)
455 : 0;
456 if (rc == 0)
457 printf("re-signed %s (revision %ld, whitelist +%dd)\n",
458 m.repo_name, m.revision, WHITELIST_DAYS);
459 free(cert_pem);
460 free(mbuf);
461 EVP_PKEY_free(master);
462 EVP_PKEY_free(certpk);
463 return rc;
464}

486int brixcvmfs_repo_main(int argc, char **argv) {
487 /* argv[0] = "repo" after the front-end shift. */
488 static const char *const txn_cmds[] =
489 { "transaction", "abort", "publish", "fsck", NULL };
490 static const char *const admin_cmds[] = { "gc", "tag", NULL };
491 const char *cmd = argc >= 2 ? argv[1] : "";
492 char kb[600];
493
494 if (rp_cmd_in(cmd, txn_cmds))
495 return brixcvmfs_txn_main(argc, argv);
496 if (rp_cmd_in(cmd, admin_cmds))
497 return brixcvmfs_admin_main(argc, argv);
498 if (strcmp(cmd, "mkfs") == 0 && (argc == 4 || argc == 5))
499 return rp_mkfs(argv[2], argv[3],
500 rp_keys_arg(argc, argv, 4, argv[3], kb, sizeof(kb)));
501 if (strcmp(cmd, "info") == 0 && (argc == 3 || argc == 4))
502 return rp_info(argv[2], rp_keys_arg(argc, argv, 3, argv[2], kb, sizeof(kb)));
503 if (strcmp(cmd, "resign") == 0 && (argc == 3 || argc == 4))
