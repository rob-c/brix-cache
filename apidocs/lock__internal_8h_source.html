42/*
43 * webdav_lock_expired_cleanup — opportunistic removal of an EXPIRED lock's
44 * stored state (defined in lock.c).
45 *
46 * WHAT: On a WRITABLE export, drop the stale lock xattr and — when `reap_null`
47 * is set — the lock-null placeholder it reserved. On a read-only export,
48 * do nothing at all.
49 * WHY: Phase-105 Appendix H.2. Discovering an expired lock happens on read
50 * paths (LOCK refresh, the If-header lock check, UNLOCK of a lock that
51 * already lapsed), and a read-only endpoint must not mutate the export as
52 * a side effect of reading it. The VFS would refuse the removexattr with
53 * EROFS anyway; declining to attempt it keeps every ordinary request off
54 * the mutation-denied counter, so that counter keeps meaning "something
55 * tried to write to a read-only export".
56 * HOW: Callers keep treating the expired lock as ABSENT regardless — request
57 * semantics never depend on whether the cleanup ran. Pass reap_null = 0
58 * on a path that is about to be re-locked (see webdav_lock_reap_null).
59 */
60void
61webdav_lock_expired_cleanup(ngx_http_request_t *r, const char *path,
62 const webdav_lock_xattr_t *e, int reap_null);
