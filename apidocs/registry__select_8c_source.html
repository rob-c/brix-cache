9/* Return 1 if path starts with any colon-delimited token in paths. */
10int
11srv_path_matches(const char *paths, const char *path)
12{
13 const char *p, *end;
14 size_t tok_len, path_len;
15
16 path_len = strlen(path);
17 p = paths;
18
19 while (*p) {
20 end = strchr(p, ':');
21 if (end == NULL) {
22 end = p + strlen(p);
23 }
24 tok_len = (size_t)(end - p);
25
26 if (tok_len > 0) {
27 /* "/" root token matches every path regardless of leading slash. */
28 if (tok_len == 1 && p[0] == '/') {
29 return 1;
30 }
31 /* Longest-prefix match: the token must align with a directory
32 * boundary so that token "/data" never matches "/database".
33 * After the literal prefix compares equal, accept on any of three
34 * mutually-exclusive boundary conditions:
35 * (1) token itself ends with '/' (e.g. "/data/" in the registry)
36 * -- the slash is already the boundary;
37 * (2) the path has a '/' immediately past the prefix (e.g. path
38 * "/data/file" against token "/data") -- next char is the
39 * directory separator;
40 * (3) exact match -- the path ends right at the token boundary
41 * (path[tok_len] is the NUL terminator). */
42 if (path_len >= tok_len
43 && ngx_strncmp(path, p, tok_len) == 0
44 && (p[tok_len - 1] == '/'
45 || path[tok_len] == '/' || path[tok_len] == '\0'))
46 {
47 return 1;
48 }
49 }
50
51 p = *end ? end + 1 : end;
52 }
53 return 0;
54}

58/*
59 * srv_sel_tier_t - one candidate tier in the three-tier selection ladder.
60 *
61 * Each tier tracks the single best slot seen so far by metric: `idx` is the slot
62 * index (-1 = none yet) and `val` is that slot's winning metric. The three tiers
63 * are ranked fresh > any > black; the ladder resolves to the highest non-empty
64 * tier (see srv_sel_state_winner).
65 */
66typedef struct {
67 int idx; /* best slot index in this tier, or -1 if none */
68 uint32_t val; /* metric of the best slot */
69} srv_sel_tier_t;

72/*
73 * srv_sel_state_t - full selection-ladder accumulator for one scan.
74 * fresh - live: not stale AND not blacklisted (the preferred tier)
75 * any - not blacklisted, any staleness (stale-but-live fallback)
76 * over - §2.3: live but over the cms.sched maxload ceiling — used only
77 * when every cooler node is gone (graceful degradation to the
78 * least-loaded overloaded node instead of a false NotFound)
79 * peer - §2.17: a peer cluster's entry ("P" role) — consulted only when
80 * no local server (fresh, stale, or overloaded) can serve
81 * black - blacklisted last-resort candidate (only when allow_blacklisted)
82 * `for_write` fixes the metric direction for the whole scan: writes maximise
83 * free_mb, reads minimise util_pct.
84 */
85/* Phase-89 W5: cap on the fresh-tier candidate list collected for the affinity
86 * stick. Slots past the cap still compete on metric via the normal tiers; the
87 * stick just cannot land on them — with 64 co-exporting fresh servers the
88 * spread loss is negligible and the array stays a cheap stack object. */
89#define SRV_SEL_AFFINITY_MAX 64
90
91typedef struct {
92 srv_sel_tier_t fresh;
93 srv_sel_tier_t any;
94 srv_sel_tier_t over;
95 srv_sel_tier_t peer;
96 srv_sel_tier_t black;
97 int for_write;
98 /* Phase-89 W5: every fresh-tier candidate seen this scan (not just the
99 * metric winner), so the affinity hash can pick a stable member. §2.3:
100 * the metric is kept alongside so the fuzz band can compare members. */
101 int fresh_cands[SRV_SEL_AFFINITY_MAX];
102 uint32_t fresh_vals[SRV_SEL_AFFINITY_MAX];
103 ngx_uint_t n_fresh;
104} srv_sel_state_t;

107/* WHAT: Seed one tier to "empty" with the sentinel metric for the scan direction.
108 * WHY: A write scan improves upward from 0; a read scan improves downward from
109 * UINT32_MAX. Seeding the val to the worst possible value lets the first
110 * real candidate always win without a separate "first" special-case.
111 * HOW: idx = -1 (no candidate); val = 0 for writes, (uint32_t)-1 for reads. */
112static void
113srv_sel_tier_init(srv_sel_tier_t *tier, int for_write)
114{
115 tier->idx = -1;
116 tier->val = for_write ? 0 : (uint32_t) -1;
117}

120/* WHAT: Initialise all three selection tiers and record the scan direction.
121 * WHY: Centralises the fresh/any/black seeding so the scan loop starts clean.
122 * HOW: Store for_write, then seed each tier via srv_sel_tier_init. */
123static void
124srv_sel_state_init(srv_sel_state_t *st, int for_write)
125{
126 st->for_write = for_write;
127 srv_sel_tier_init(&st->fresh, for_write);
128 srv_sel_tier_init(&st->any, for_write);
129 srv_sel_tier_init(&st->over, for_write);
130 srv_sel_tier_init(&st->peer, for_write);
131 srv_sel_tier_init(&st->black, for_write);
132 st->n_fresh = 0;
133}

136/* WHAT: Report whether `metric` beats a tier's current best for this direction.
137 * WHY: Encapsulates the single tie-break rule shared by every tier: writes
138 * prefer a strictly greater free_mb, reads a strictly lesser util_pct.
139 * Using strict comparison preserves first-seen determinism on ties —
140 * identical to the original inlined tests.
141 * HOW: Empty tier (idx == -1) always loses; otherwise compare by direction. */
142static int
143srv_sel_metric_better(const srv_sel_tier_t *tier, uint32_t metric, int for_write)
144{
145 if (tier->idx == -1) {
146 return 1;
147 }
148 return for_write ? (metric > tier->val) : (metric < tier->val);
149}

152/* WHAT: Offer slot `idx` (metric `metric`) to a single tier, updating if better.
153 * WHY: One place owns the "is this a new tier winner?" decision, so fresh/any/
154 * black all apply exactly the same tie-break.
155 * HOW: If srv_sel_metric_better says yes, record idx+metric as the new best. */
156static void
157srv_sel_tier_offer(srv_sel_tier_t *tier, int idx, uint32_t metric, int for_write)
158{
159 if (srv_sel_metric_better(tier, metric, for_write)) {
160 tier->idx = idx;
161 tier->val = metric;
162 }
163}

166/* WHAT: Classify one matched slot and fold it into the right selection tier(s).
167 * WHY: Isolates the per-slot policy — blacklisted → black tier only; peers
168 * (§2.17) → their own last-resort-before-black tier; maxload-exceeded
169 * nodes (§2.3) → the `over` tier; live → always the `any` tier, and
170 * additionally the `fresh` tier when not stale. Staleness uses a signed
171 * msec diff so ngx_current_msec wrap is tolerated.
172 * HOW: Compute the metric (srv_sel_metric — legacy load-weight blend or the
173 * §2.3 cms.sched component blend, registry_select_sched.c), then route
174 * the slot to its tier. */
175static void
176srv_sel_state_consider(srv_sel_state_t *st, int idx, const brix_srv_entry_t *e,
177 int is_black)
178{
179 uint32_t metric;
180 ngx_uint_t is_stale;
181
182 metric = srv_sel_metric(e, st->for_write);
183
184 if (is_black) {
185 /* allow_blacklisted only — a last-resort tier below live servers. */
186 srv_sel_tier_offer(&st->black, idx, metric, st->for_write);
187 return;
188 }
189
190 /* §2.17: a peer cluster ("P") is consulted only when NO local server can
191 * serve — it never competes with local capacity, however stale/loaded. */
192 if (e->role[0] == 'P' && e->role[1] == '\0') {
193 srv_sel_tier_offer(&st->peer, idx, metric, st->for_write);
194 return;
195 }
196
197 /* §2.3: a node over the cms.sched maxload ceiling drops below every
198 * cooler node (incl. stale ones) but still beats shipping to a peer or a
199 * blacklisted node — graceful degradation, not a refusal. */
200 if (srv_sel_over_maxload(e)) {
201 srv_sel_tier_offer(&st->over, idx, metric, st->for_write);
202 return;
203 }
204
205 srv_sel_tier_offer(&st->any, idx, metric, st->for_write);
206
207 /*
208 * Phase 39 (WS7): a server that has not heartbeated within stale_after_ms is
209 * de-preferred but still tracked as a fallback, so an all-stale storm
210 * degrades to the freshest stale server rather than a false NotFound. The
211 * signed diff tolerates ngx_current_msec wrap.
212 */
213 is_stale = (brix_srv_stale_after_ms > 0
214 && (ngx_msec_int_t) (ngx_current_msec - e->last_seen)
215 > (ngx_msec_int_t) brix_srv_stale_after_ms);
216 if (!is_stale) {
217 srv_sel_tier_offer(&st->fresh, idx, metric, st->for_write);
218 if (st->n_fresh < SRV_SEL_AFFINITY_MAX) {
219 st->fresh_vals[st->n_fresh] = metric;
220 st->fresh_cands[st->n_fresh++] = idx;
221 }
222 }
223}

226/* WHAT: fnv1a over the NUL-terminated path (Phase-89 W5 affinity stick).
227 * WHY: The stick must be a pure function of the path so every worker on every
228 * manager picks the same member of the same candidate set; fnv1a is the
229 * design-of-record hash (loc_cache.c) for path-shaped keys.
230 * HOW: Standard XOR-multiply loop over the bytes. */
231static uint32_t
232srv_sel_path_hash(const char *path)
233{
234 uint32_t h = BRIX_FNV1A32_OFFSET_BASIS;
235
236 while (*path != '\0') {
237 h ^= (uint32_t) (u_char) *path++;
238 h *= BRIX_FNV1A32_PRIME;
239 }
240 return h;
241}

244/* WHAT: Resolve the accumulated tiers to the winning slot index (or -1).
245 * WHY: Fixes the tier priority in one place: a live-fresh server always beats
246 * a stale-live one, which beats an overloaded one (§2.3 maxload), which
247 * beats a peer cluster (§2.17), which beats a blacklisted last resort.
248 * HOW: Walk the ladder top-down, returning the first non-empty tier. */
249static int
250srv_sel_state_winner(const srv_sel_state_t *st)
251{
252 if (st->fresh.idx >= 0) {
253 return st->fresh.idx;
254 }
255 if (st->any.idx >= 0) {
256 return st->any.idx;
257 }
258 if (st->over.idx >= 0) {
259 return st->over.idx;
260 }
261 if (st->peer.idx >= 0) {
262 return st->peer.idx;
263 }
264 return st->black.idx;
265}

267/*
268 * srv_sel_fuzz_pick — §2.3: cms.sched fuzz round-robin band.
269 *
270 * WHAT: When >1 fresh candidate's metric sits within fuzz percent of the best
271 * metric, treat them as equal and rotate a per-worker cursor over the
272 * band, returning the picked slot index. Returns `best` unchanged when
273 * fuzz is off, the band is a singleton, or the ladder winner is not from
274 * the fresh tier.
275 * WHY: Strict metric comparison pins all traffic to one node when several
276 * are effectively equally loaded; stock cms.sched fuzz spreads it. The
277 * rotation is per-worker (a plain static) — deterministic alternation,
278 * no shared state, matching the balancing-hint (not SLA) posture.
279 * HOW: Compute the band bound from the best fresh metric (reads: best +
280 * fuzz%; writes: best - fuzz%), collect band members in scan order, and
281 * index them with an advancing counter.
282 */
283static int
284srv_sel_fuzz_pick(const srv_sel_state_t *st, int best)
285{
286 static ngx_uint_t rotate;
287 int band[SRV_SEL_AFFINITY_MAX];
288 ngx_uint_t i, n_band = 0;
289 uint64_t bound;
290
291 if (brix_srv_sched.fuzz == 0 || st->n_fresh < 2
292 || best != st->fresh.idx)
293 {
294 return best;
295 }
296
297 if (st->for_write) {
298 bound = (uint64_t) st->fresh.val
299 * (100 - brix_srv_sched.fuzz) / 100;
300 } else {
301 bound = (uint64_t) st->fresh.val
302 * (100 + brix_srv_sched.fuzz) / 100;
303 }
304
305 for (i = 0; i < st->n_fresh; i++) {
306 if (st->for_write ? ((uint64_t) st->fresh_vals[i] >= bound)
307 : ((uint64_t) st->fresh_vals[i] <= bound))
308 {
309 band[n_band++] = st->fresh_cands[i];
310 }
311 }
312
313 if (n_band < 2) {
314 return best;
315 }
316 return band[rotate++ % n_band];
317}

320/* WHAT
321 * Selects the best data server for a given path from the registry table.
322 * Used by kXR_locate and kXR_open to redirect clients to optimal servers.
323
324 * WHY
325 * Selection policy: reads → lowest util_pct (least loaded); writes → highest
326 * free_mb (most available space). Path matching uses longest-prefix over colon-
327 * delimited tokens in each entry's paths field.
328 *
329 * allow_blacklisted gives the open/stat handlers a LAST-RESORT tier: when no
330 * live server matches, fall back to a currently-blacklisted one rather than a
331 * false NotFound. A CMS heartbeat drop blacklists a server for 30 s even though
332 * its data plane is almost always still serving, so under load a transient blip
333 * should still redirect to the (live) node. kXR_locate passes 0 — it must
334 * report only live servers, so a genuinely dead node is still "not found" there.
335
336 * HOW
337 * Locks mutex → scans all occupied slots → filters by srv_path_matches() →
338 * folds each match into the three-tier accumulator (srv_sel_state_consider) →
339 * resolves the winner (srv_sel_state_winner), preferring fresh→stale→
340 * (blacklisted, only if allow_blacklisted). Writes host+port to output
341 * buffers. Unlocks and returns 1/0.
342 */
343int
344srv_select_core(const char *path, int for_write, int allow_blacklisted,
345 char *host_out, size_t host_size, uint16_t *port_out)
346{
347 brix_srv_table_t *tbl;
348 brix_srv_entry_t *e;
349 ngx_uint_t i;
350 srv_sel_state_t st;
351 int best;
352
353 tbl = srv_table();
354 if (tbl == NULL) {
355 return 0;
356 }
357
358 srv_sel_state_init(&st, for_write);
359
360 ngx_shmtx_lock(&brix_srv_mutex);
361
362 for (i = 0; i < tbl->capacity; i++) {
363 int is_black;
364
365 e = &tbl->slots[i];
366 if (!e->in_use) {
367 continue;
368 }
369 if (!srv_path_matches(e->paths, path)) {
370 continue;
371 }
372
373 is_black = (e->blacklisted_until != 0
374 && e->blacklisted_until > ngx_current_msec);
375
376 if (is_black && !allow_blacklisted) {
377 continue; /* strict callers (e.g. locate) skip blacklisted */
378 }
379
380 srv_sel_state_consider(&st, (int) i, e, is_black);
381 }
382
383 best = srv_sel_state_winner(&st);
384
385 /* Phase-89 W5: with affinity on and >1 eligible FRESH candidate, the path
386 * hash — not the metric — picks the member, so repeated requests for one
387 * path stick to one server (cache locality). LOCKED precedence (phase-61
388 * note 2): the filter above already excluded blacklisted/stale slots, so a
389 * drained host is never sticky; an empty/singleton fresh tier keeps the
390 * ladder winner unchanged. §2.3: affinity wins over the fuzz band (a
391 * sticky path must not rotate); fuzz applies only without affinity. */
392 if (brix_srv_affinity && st.n_fresh > 1) {
393 best = st.fresh_cands[srv_sel_path_hash(path) % st.n_fresh];
394 } else {
395 best = srv_sel_fuzz_pick(&st, best);
396 }
397
398 if (best >= 0) {
399 e = &tbl->slots[best];
400 ngx_cpystrn((u_char *) host_out, (u_char *) e->host, host_size);
401 *port_out = e->port;
402 }
403
404 ngx_shmtx_unlock(&brix_srv_mutex);
405 return best >= 0;
406}

433/* WHAT: Count occupied servers that export a prefix covering path — the number
434 * of distinct data servers a client could be redirected to for this path.
435 * WHY: The tried/triedrc retry protocol (see brix_manager_tried_exhausted)
436 * needs to know how many candidates exist so it can tell when the client
437 * has exhausted them all and the answer is definitively "not found".
438 * HOW: Same path scan as srv_select_core (in_use, prefix match) under the
439 * registry spinlock, returning the count. Blacklisted slots ARE counted:
440 * the open/stat path (brix_srv_select_or_blacklisted) can redirect to a
441 * blacklisted server as a last resort, so it IS a candidate the client may
442 * be sent to. Counting it keeps tried-exhausted consistent — a client
443 * that already tried that one server then converges to NotFound instead of
444 * being bounced back to it. (A server that left the cluster is
445 * unregistered → in_use == 0 → not counted.) */
446int
447brix_srv_count_matching(const char *path)
448{
449 brix_srv_table_t *tbl;
450 brix_srv_entry_t *e;
451 ngx_uint_t i;
452 int n = 0;
453
454 tbl = srv_table();
455 if (tbl == NULL) {
456 return 0;
457 }
458
459 ngx_shmtx_lock(&brix_srv_mutex);
460 for (i = 0; i < tbl->capacity; i++) {
461 e = &tbl->slots[i];
462 if (!e->in_use) {
463 continue;
464 }

466 continue;
467 }

469 }

476/* WHAT: Honour the XRootD client's tried/triedrc retry protocol at a manager.
477 * When a data server returns an error the client re-issues the request to
478 * the manager with ?tried=h1,h2&triedrc=rc1,rc2 listing servers it already
479 * attempted. Returns 1 when every server exporting this path has already
480 * been tried — the caller must then answer kXR_NotFound instead of
481 * redirecting, otherwise the client bounces manager<->data-server until it
482 * hits its redirect limit (the divergence conformance testing caught).
483 * WHY: A conformant redirector must converge to "not found" once the client has
484 * visited every candidate; reference cmsd does this via the tried list.
485 * HOW: Extract the opaque (everything after '?') from the raw request payload,
486 * locate "tried=", count its comma-separated hosts, and compare against
487 * brix_srv_count_matching(clean_path). Conservative: a zero match count
488 * falls through to the normal CMS-locate path so this does not prematurely
489 * short-circuit hierarchical (parent-locate) clusters. */
490int
491brix_manager_tried_exhausted(const u_char *payload, size_t payload_len,
492 const char *clean_path)
493{
494 char opaque[1024];
495 const u_char *q;
496 size_t olen;
497 const char *t, *p;
498 int n_tried, n_match;
499
500 if (payload == NULL || payload_len == 0) {
501 return 0;
502 }
503 q = memchr(payload, '?', payload_len);
504 if (q == NULL) {
505 return 0; /* no opaque -> client's first attempt */
506 }
507 q++;
508 olen = (size_t) (payload + payload_len - q);
509 if (olen > 0 && q[olen - 1] == '\0') {
510 olen--; /* trim trailing NUL some payloads carry */
511 }
512 if (olen == 0 || olen >= sizeof(opaque)) {
513 return 0; /* empty or too long to inspect — be safe */
514 }
515 ngx_memcpy(opaque, q, olen);
516 opaque[olen] = '\0';
517
518 t = strstr(opaque, "tried=");
519 if (t == NULL) {
520 return 0;
521 }
522 t += 6; /* skip "tried=" */
523 if (*t == '\0' || *t == '&') {
524 return 0; /* present but empty */
525 }
526
527 n_tried = 1;
528 for (p = t; *p && *p != '&'; p++) {
529 if (*p == ',') {
530 n_tried++;
531 }
532 }
533
534 n_match = brix_srv_count_matching(clean_path);
535 return (n_match > 0 && n_tried >= n_match);
536}
