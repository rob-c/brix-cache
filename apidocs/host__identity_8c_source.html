23int
24brix_host_identity_valid(const char *name, size_t len)
25{
26 size_t i;
27
28 if (name == NULL || len == 0 || len > BRIX_HOST_IDENTITY_MAX) {
29 return 0;
30 }
31 for (i = 0; i < len; i++) {
32 unsigned char ch = (unsigned char) name[i];
33 int ok = (ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z')
34 || (ch >= '0' && ch <= '9') || ch == '.' || ch == '-'
35 || ch == ':';
36 if (!ok) {
37 return 0;
38 }
39 }
40 return 1;
41}
