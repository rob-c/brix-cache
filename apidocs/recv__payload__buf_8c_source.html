3/*
4 * recv_payload_buf.c — the reusable heap payload buffer for the recv framing
5 * loop (split from recv_process.c to keep it under the size cap). Two ways
6 * to size it: _ensure at request start (the buffer is empty, so a resize may
7 * free-then-alloc) and _grow mid-request (the kXR_writev / kXR_chkpoint body
8 * extension raises the expected length after bytes have landed, so a resize
9 * must copy the payload_pos bytes already received). Both NUL-terminate at
10 * dlen so string-parsing handlers can treat the payload as a C string.
11 */
12
13static ngx_int_t

15 uint32_t dlen, ngx_flag_t preserve)
16{
17 u_char *buf;
18 size_t need;
19
20 if (dlen > (uint32_t) (SIZE_MAX - 1)) {
21 return NGX_ERROR;
22 }
23 need = (size_t) dlen + 1;
24
25 if (ctx->recv.payload_buf != NULL && ctx->recv.payload_buf_size >= need) {
26 ctx->recv.payload = ctx->recv.payload_buf;
27 ctx->recv.payload[dlen] = '\0';
28 return NGX_OK;
29 }
30
31 buf = ngx_alloc(need, c->log);

33 return NGX_ERROR;
34 }

39 }

41 }
42
43 ctx->recv.payload_buf = buf;
44 ctx->recv.payload_buf_size = need;
45 ctx->recv.payload = buf;

49}
