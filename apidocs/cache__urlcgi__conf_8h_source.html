1/*
2 * cache_urlcgi_conf.h — brix_cache_urlcgi (2.0 F5, upstream pfc.urlcgi).
3 *
4 * WHAT: The per-server clamp pair for each of the two per-open client cache
5 * hints a root:// open may carry as CGI — pfc.blocksize=<bytes> (the slice
6 * granule a NEW partial object is created with) and pfc.prefetch=<blocks>
7 * (that handle's background-prefetch runway) — plus its init/merge/adopt
8 * helpers, so the http and stream conf planes and the cache policy share
9 * one shape.
10 *
11 * WHY: Upstream's `pfc.urlcgi [blocksize {ignore|min max}] [prefetch
12 * {ignore|min max}]` lets a client tune both per open while the operator
13 * bounds what may be asked for; the default is both ignored. Same grammar,
14 * same default. Encoding: *_max == 0 means "the hint is ignored" (a real
15 * clamp always has max >= 1), so the merge/adopt sentinels
16 * (NGX_CONF_UNSET_SIZE / NGX_CONF_UNSET_UINT = "not set in this block")
17 * never collide with a configured value.
18 */
19#ifndef BRIX_CACHE_URLCGI_CONF_H
20#define BRIX_CACHE_URLCGI_CONF_H

26/* The slice-engine granule: brix_cache_slice_size and every urlcgi blocksize
27 * bound must be a positive multiple of it, and a clamped pfc.blocksize hint
28 * is rounded down to it. */
29#define BRIX_CACHE_SLICE_GRANULE (1024 * 1024)

31typedef struct {
32 size_t bs_min; /* blocksize clamp floor, bytes (1m multiple) */
33 size_t bs_max; /* ...ceiling; 0 = pfc.blocksize ignored;
34 * NGX_CONF_UNSET_SIZE = not set in this block */
35 ngx_uint_t pf_min; /* prefetch clamp floor, blocks (0 = client may
36 * switch speculation off for its handle) */
37 ngx_uint_t pf_max; /* ...ceiling; 0 = pfc.prefetch ignored;
38 * NGX_CONF_UNSET_UINT = not set in this block */
39} brix_cache_urlcgi_conf_t;

41static ngx_inline void
42brix_cache_urlcgi_conf_init(brix_cache_urlcgi_conf_t *u)

50/* Merge: each (min, max) pair inherits from the enclosing block as a UNIT
51 * (the max sentinel marks "not set here"); the outermost default is ignored. */
52static ngx_inline void
53brix_cache_urlcgi_conf_merge(brix_cache_urlcgi_conf_t *conf,
54 const brix_cache_urlcgi_conf_t *prev)
55{
56 if (conf->bs_max == NGX_CONF_UNSET_SIZE) {
57 conf->bs_min = prev->bs_min;
58 conf->bs_max = (prev->bs_max == NGX_CONF_UNSET_SIZE) ? 0 : prev->bs_max;
59 }
60 if (conf->pf_max == NGX_CONF_UNSET_UINT) {
61 conf->pf_min = prev->pf_min;
62 conf->pf_max = (prev->pf_max == NGX_CONF_UNSET_UINT) ? 0 : prev->pf_max;
63 }
64}

66/* Adopt (cross-plane unification, brix_shared_adopt_unified): a pair the
67 * destination never set takes the source's, again as a unit. */
68static ngx_inline void
69brix_cache_urlcgi_conf_adopt(brix_cache_urlcgi_conf_t *dst,
70 const brix_cache_urlcgi_conf_t *src)
71{
72 if (dst->bs_max == NGX_CONF_UNSET_SIZE && src->bs_max != NGX_CONF_UNSET_SIZE) {
73 dst->bs_min = src->bs_min;
74 dst->bs_max = src->bs_max;
75 }
76 if (dst->pf_max == NGX_CONF_UNSET_UINT && src->pf_max != NGX_CONF_UNSET_UINT) {
77 dst->pf_min = src->pf_min;
78 dst->pf_max = src->pf_max;
79 }
80}

82/* The policy value: a conf that reaches tier registration unmerged (sentinel
83 * still present) behaves like the merged default — ignored. */
84static ngx_inline brix_cache_urlcgi_conf_t
85brix_cache_urlcgi_conf_effective(const brix_cache_urlcgi_conf_t *c)
86{
87 brix_cache_urlcgi_conf_t e = *c;
88
89 if (e.bs_max == NGX_CONF_UNSET_SIZE) {
90 e.bs_max = 0;
91 }
92 if (e.pf_max == NGX_CONF_UNSET_UINT) {
93 e.pf_max = 0;
94 }
95 return e;
96}
