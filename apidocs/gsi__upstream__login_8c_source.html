1/*
2 * gsi_upstream_login.c — Phase-4b GSI delegation: threaded blocking login to the
3 * upstream AS THE USER + fd handoff into the async relay (Tasks 3+4).
4 *
5 * WHY: the proven in-process GSI client (the cache origin client) is BLOCKING —
6 * it runs the certreq/cert handshake on a thread-pool worker. The terminating
7 * proxy is event-driven. So for `brix_tap_proxy_auth gsi` we offload the whole
8 * connect+GSI-login (presenting the client's DELEGATED proxy) to a thread, then
9 * promote the authenticated fd to the proxy's async forward/relay.
10 *
11 * HOW: thread body builds a synthetic origin conf (host/port + the delegated proxy
12 * path + the proxy server's CA store) and reuses brix_cache_origin_connect +
13 * _bootstrap; on success it transfers the authenticated fd. The completion
14 * handler (event loop) wraps that fd in an ngx_connection_t already at IDLE
15 * (bootstrap done), unlinks the temp credential, and dispatches the saved
16 * request. Lifecycle guarded by client_ctx->destroyed (the worker may finish
17 * after the client vanished). GSI auth is cleartext-data here (tls=0), which is
18 * what the tap observes and what the fd handoff supports.
19 */

49/*
50 * WHAT: Thread-pool worker that performs blocking connect + GSI login with delegated proxy.
51 *
52 * WHY: The GSI client (cache origin client) is blocking and runs on a thread-pool worker,
53 * while the proxy is event-driven. This offloads the entire connect+GSI-login handshake
54 * (presenting the client's delegated proxy) to a thread, then promotes the authenticated
55 * fd to the proxy's async relay. Uses a synthetic origin config to reuse existing
56 * brix_cache_origin_connect/bootstrap code paths.
57 *
58 * HOW:
59 * - Build synthetic origin config (host/port, delegated proxy path, CA store)
60 * - Call brix_cache_origin_connect() and brix_cache_origin_bootstrap()
61 * - On success: transfer authenticated fd from oc to g->result_fd (oc.fd = -1)
62 * - Call brix_cache_origin_close() (frees SSL, but fd already transferred)
63 * - Completion handler (proxy_gsi_login_done) wraps fd in ngx_connection_t and dispatches
64 */
65static void
66proxy_gsi_login_thread(void *data, ngx_log_t *log)
67{
68 proxy_gsi_login_t *g = data;
69 ngx_stream_brix_srv_conf_t *synth = calloc(1, sizeof(*synth));
70 brix_cache_fill_t *t = calloc(1, sizeof(*t));
71 brix_cache_origin_conn_t oc;
72
73 (void) log;
74 g->result_fd = -1;
75
76 if (synth == NULL || t == NULL) {
77 free(synth);
78 free(t);
79 return;
80 }
81
82 oc.fd = -1;
83 oc.ssl = NULL;
84 oc.ssl_ctx = NULL;
85
86 synth->cache_origin_host.data = (u_char *) g->host;
87 synth->cache_origin_host.len = ngx_strlen(g->host);
88 synth->cache_origin_port = g->port;
89 synth->cache_origin_tls = 0; /* cleartext GSI auth */
90 synth->cache_origin_family = (ngx_uint_t) g->family;
91 synth->cache_origin_x509_proxy.data = (u_char *) g->deleg_path;
92 synth->cache_origin_x509_proxy.len = ngx_strlen(g->deleg_path);
93 synth->gsi_store = g->gsi_store; /* borrowed; do not free */
94 synth->common.dns.policy = g->dns; /* phase-116: one DNS path */
95 t->conf = synth;
96
97 if (brix_cache_origin_connect(t, &oc) == 0
98 && brix_cache_origin_bootstrap(t, &oc) == 0)
99 {
100 g->result_fd = oc.fd; /* transfer ownership */
101 oc.fd = -1;
102 }
103
104 brix_cache_origin_close(&oc); /* frees SSL (if any); fd already taken */
105 free(synth);
106 free(t);
107}

110/*
111 * WHAT: Wrap an authenticated fd in an ngx_connection_t and wire it to the proxy relay.
112 *
113 * WHY: After the GSI login thread completes, the authenticated fd must be integrated into
114 * the event-driven proxy. This function creates the ngx_connection_t, sets up handlers,
115 * and marks the connection at IDLE (bootstrap already done in the thread).
116 *
117 * HOW:
118 * - Call ngx_get_connection() to wrap the fd
119 * - Create connection pool and set to non-blocking mode
120 * - Set data pointer to proxy context, assign recv/send handlers
121 * - Set state to XRD_PX_IDLE (bootstrap complete), mark from_pool=1
122 * - Register for read events via ngx_handle_read_event()
123 * - Return NGX_OK on success, NGX_ERROR on any allocation/handler failure
124 */
125static ngx_int_t
126proxy_gsi_promote_fd(brix_proxy_ctx_t *proxy, int fd)
127{
128 ngx_connection_t *client_conn = proxy->client_conn;
129 ngx_connection_t *uconn;
130
131 uconn = ngx_get_connection(fd, client_conn->log);
132 if (uconn == NULL) {
133 return NGX_ERROR;
134 }
135 uconn->pool = ngx_create_pool(BRIX_PROXY_POOL_SIZE, client_conn->log);
136 if (uconn->pool == NULL) {
137 ngx_free_connection(uconn);
138 return NGX_ERROR;
139 }
140 if (ngx_nonblocking(fd) == -1) {
141 ngx_destroy_pool(uconn->pool);
142 ngx_free_connection(uconn);
143 return NGX_ERROR;
144 }
145
146 uconn->data = proxy;
147 uconn->recv = ngx_recv;
148 uconn->send = ngx_send;
149 uconn->recv_chain = ngx_recv_chain;
150 uconn->send_chain = ngx_send_chain;
151 uconn->log = client_conn->log;
152 uconn->read->log = client_conn->log;
153 uconn->write->log = client_conn->log;
154 uconn->read->handler = brix_proxy_read_handler;
155 uconn->write->handler = brix_proxy_write_handler;
156
157 proxy->conn = uconn;
158 proxy->state = XRD_PX_IDLE; /* bootstrap already done in the thread */
159 proxy->from_pool = 1; /* => skip the bootstrap path */
160
161 if (ngx_handle_read_event(uconn->read, 0) != NGX_OK) {
162 return NGX_ERROR;
163 }
164 return NGX_OK;
165}

168/*
169 * WHAT: Event-loop completion handler for GSI login thread.
170 *
171 * WHY: The thread runs asynchronously; this handler runs in the event loop when the thread
172 * completes. It cleans up the temp credential file, checks if the client is still alive,
173 * promotes the authenticated fd, and dispatches the saved request (or resumes the read loop).
174 *
175 * HOW:
176 * - Unlink the temp delegated proxy credential file (vfs-seam-allow: DOMAIN_CREDENTIAL)
177 * - Check if client_ctx exists and !destroyed (client may have vanished during login)
178 * - If result_fd < 0: abort with "GSI upstream login failed"
179 * - Call proxy_gsi_promote_fd() to integrate the fd; abort on failure
180 * - If saved_req exists: dispatch it via brix_proxy_dispatch_pending()
181 * - Otherwise: set state to XRD_ST_REQ and resume read via brix_schedule_read_resume()
182 */
183static void
184proxy_gsi_login_done(ngx_event_t *ev)
185{
186 ngx_thread_task_t *task = ev->data;
187 proxy_gsi_login_t *g = task->ctx;
188 brix_proxy_ctx_t *proxy = g->proxy;
189
190 if (g->deleg_path[0] != '\0') {
191 (void) unlink(g->deleg_path); /* vfs-seam-allow: DOMAIN_CREDENTIAL — config-domain delegated GSI proxy credential temp (not export storage) */
192 }
193
194 /* Client vanished while we were logging in: drop the authed fd and stop. */
195 if (proxy->client_ctx == NULL || proxy->client_ctx->destroyed) {
196 if (g->result_fd >= 0) {
197 ngx_close_socket(g->result_fd);
198 }
199 return;
200 }
201
202 if (g->result_fd < 0) {
203 brix_proxy_abort(proxy, "proxy: GSI upstream login failed");
204 return;
205 }
206
207 if (proxy_gsi_promote_fd(proxy, g->result_fd) != NGX_OK) {
208 ngx_close_socket(g->result_fd);
209 brix_proxy_abort(proxy, "proxy: GSI upstream handoff failed");
210 return;
211 }
212
213 if (proxy->saved_req != NULL) {
214 brix_proxy_dispatch_pending(proxy);
215 } else {
216 proxy->client_ctx->state = XRD_ST_REQ_HEADER;
217 (void) brix_schedule_read_resume(proxy->client_conn);
218 }
219}

222/*
223 * WHAT: Initiate async GSI connection to upstream with delegated proxy authentication.
224 *
225 * WHY: For `brix_tap_proxy_auth gsi`, the client's delegated X.509 proxy must be presented
226 * to the upstream during connection. This spawns a thread for the blocking GSI login,
227 * parks the client read loop, and resumes when authentication completes.
228 *
229 * HOW:
230 * - Validate thread_pool is configured (required for async operation)
231 * - Validate client has delegated proxy (ctx->gsi.deleg_proxy_pem/len)
232 * - Allocate thread task and write delegated proxy to temp file (0600)
233 * - Populate proxy_gsi_login_t with host, port, family, deleg_path, gsi_store, dns policy
234 * - Bind task to proxy_gsi_login_thread (worker) and proxy_gsi_login_done (completion)
235 * - Post task to thread_pool; unlink temp file on failure
236 * - Set client state to XRD_ST_PROXY (park read loop) and return NGX_OK
237 */
238ngx_int_t
239brix_proxy_gsi_connect_async(brix_proxy_ctx_t *proxy,
240 ngx_stream_brix_srv_conf_t *conf, ngx_str_t *host, uint16_t port)
241{
242 ngx_connection_t *c = proxy->client_conn;
243 brix_ctx_t *ctx = proxy->client_ctx;
244 ngx_thread_task_t *task;
245 proxy_gsi_login_t *g;
246
247 if (conf->common.thread_pool == NULL) {
248 ngx_log_error(NGX_LOG_ERR, c->log, 0,
249 "brix_tap_proxy_auth gsi: a thread_pool is required");
250 return NGX_ERROR;
251 }
252 if (ctx->gsi.deleg_proxy_pem == NULL || ctx->gsi.deleg_proxy_len == 0) {
253 ngx_log_error(NGX_LOG_ERR, c->log, 0,
254 "xrootd tap proxy: no delegated X.509 proxy from the client "
255 "(client must delegate; is the client GSI with delegation?)");
256 return NGX_ERROR;
257 }
258
259 task = ngx_thread_task_alloc(c->pool, sizeof(proxy_gsi_login_t));
260 if (task == NULL) {
261 return NGX_ERROR;
262 }
263 g = task->ctx;
264
265 if (brix_proxy_gsi_write_pem_temp(ctx->gsi.deleg_proxy_pem,
266 ctx->gsi.deleg_proxy_len,
267 g->deleg_path, sizeof(g->deleg_path)) != 0)
268 {
269 ngx_log_error(NGX_LOG_ERR, c->log, ngx_errno,
270 "xrootd tap proxy: cannot persist delegated proxy");
271 return NGX_ERROR;
272 }
273
274 ngx_cpystrn((u_char *) g->host, host->data, sizeof(g->host));
275 g->port = port;
276 g->family = BRIX_AF_AUTO;
277 g->gsi_store = conf->gsi_store; /* the proxy server's CA store (borrowed) */
278 g->dns = conf->common.dns.policy;
279 g->log = c->log;
280 g->proxy = proxy;
281 g->client_conn = c;
282 g->result_fd = -1;
283
284 brix_task_bind(task, proxy_gsi_login_thread, proxy_gsi_login_done);
285
286 if (ngx_thread_task_post(conf->common.thread_pool, task) != NGX_OK) {
287 (void) unlink(g->deleg_path); /* vfs-seam-allow: DOMAIN_CREDENTIAL — config-domain delegated GSI proxy credential temp (not export storage) */
288 ngx_log_error(NGX_LOG_ERR, c->log, 0,
289 "xrootd tap proxy: GSI login thread post failed");
290 return NGX_ERROR;
291 }
292
293 /* Park the client read loop until the login completes. */
294 proxy->client_ctx->state = XRD_ST_PROXY;
295 return NGX_OK;
296}
