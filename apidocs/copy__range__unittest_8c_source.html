151/**
152 * verify_files_equal - Verify two files are identical
153 *
154 * @file1: First file path
155 * @file2: Second file path
156 * @return: 0 if equal, -1 if different
157 */
158static int verify_files_equal(const char *file1, const char *file2)
159{
160 HANDLE h1, h2;
161 BYTE buffer1[65536], buffer2[65536];
162 DWORD bytes_read1, bytes_read2;
163
164 h1 = CreateFileA(file1, GENERIC_READ, 0, NULL, OPEN_EXISTING,
165 FILE_ATTRIBUTE_NORMAL, NULL);
166 if (h1 == INVALID_HANDLE_VALUE) {
167 return -1;
168 }
169
170 h2 = CreateFileA(file2, GENERIC_READ, 0, NULL, OPEN_EXISTING,
171 FILE_ATTRIBUTE_NORMAL, NULL);
172 if (h2 == INVALID_HANDLE_VALUE) {
173 CloseHandle(h1);
174 return -1;
175 }
176
177 while (1) {
178 if (!ReadFile(h1, buffer1, sizeof(buffer1), &bytes_read1, NULL) ||
179 !ReadFile(h2, buffer2, sizeof(buffer2), &bytes_read2, NULL)) {
180 CloseHandle(h1);
181 CloseHandle(h2);
182 return -1;
183 }
184
185 if (bytes_read1 != bytes_read2) {
186 CloseHandle(h1);
187 CloseHandle(h2);
188 return -1;
189 }
190
191 if (bytes_read1 == 0) {
192 break; /* EOF */
193 }
194
195 if (memcmp(buffer1, buffer2, bytes_read1) != 0) {
196 CloseHandle(h1);
197 CloseHandle(h2);
198 return -1;
199 }
200 }
201
202 CloseHandle(h1);
203 CloseHandle(h2);
204 return 0;
205}

207/**
208 * get_file_size - Get file size in bytes
209 */
210static off_t get_file_size(const char *filename)
211{
212 HANDLE hFile;
213 LARGE_INTEGER size;
214 DWORD high_part;
215
216 hFile = CreateFileA(filename, GENERIC_READ, 0, NULL, OPEN_EXISTING,
217 FILE_ATTRIBUTE_NORMAL, NULL);
218 if (hFile == INVALID_HANDLE_VALUE) {
219 return -1;
220 }
221
222 size.LowPart = GetFileSize(hFile, &high_part);
223 size.HighPart = (LONG) high_part;
224 CloseHandle(hFile);
225
226 if (size.LowPart == INVALID_FILE_SIZE) {
227 return -1;
228 }
229
230 return (off_t)size.QuadPart;
231}

315static void test_copy_range(void)
316{
317 ssize_t result;
318 off_t in_off = 1024; /* Start at offset 1KB */
319 off_t out_off = 0;
320 const size_t range_size = 4096; /* Copy 4KB */
321 const size_t file_size = 8192; /* 8KB total file */
322
323 test_begin("Range copy (FSCTL_COPY_FILE_RANGE)");
324
325 if (brix_test_copy_file(file_size, &in_off, &out_off, range_size, &result) < 0) {
326 return;
327 }
328
329 if (result != (ssize_t)range_size) {
330 test_fail("Wrong number of bytes copied");
331 cleanup_test_directory();
332 return;
333 }
334
335 /* Verify offsets were updated */
336 if (in_off != 1024 + (off_t)range_size || out_off != (off_t)range_size) {
337 test_fail("Offsets not updated correctly");
338 cleanup_test_directory();
339 return;
340 }
341
342 test_pass();
343 cleanup_test_directory();
