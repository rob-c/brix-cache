12/* ---- Parse touch's argument vector ----
13 *
14 * WHAT: Scans argv for -c/-a/-m/-t|--timestamp flags and the single path
15 * operand, writing each into the caller's out-params. Returns 0 on
16 * success, or 50 (after printing a diagnostic) on a malformed
17 * -t/--timestamp value.
18 *
19 * WHY: The flag scan — especially the -t branch that consumes the next argv
20 * slot and parses a timestamp — is what pushes do_touch over the
21 * complexity cap; isolating it keeps the command body linear.
22 *
23 * HOW: 1. Walk argv setting the boolean out-param for each recognised option.
24 * 2. For -t/--timestamp with a following argument, parse it into *tspec
25 * via touch_parse_time; on failure print the usage hint and return 50.
26 * 3. Treat any other token (including a bare -t with no following value,
27 * matching the original loop) as the path operand, last one winning.
28 */
29static int
30touch_parse_argv(int argc, char **argv, int *no_create, int *do_atime,
31 int *do_mtime, int *have_t, struct timespec *tspec,
32 const char **arg)
33{
34 int i;
35
36 for (i = 1; i < argc; i++) {
37 if (strcmp(argv[i], "-c") == 0) { *no_create = 1; }
38 else if (strcmp(argv[i], "-a") == 0) { *do_atime = 1; }
39 else if (strcmp(argv[i], "-m") == 0) { *do_mtime = 1; }
40 else if ((strcmp(argv[i], "-t") == 0 || strcmp(argv[i], "--timestamp") == 0)
41 && i + 1 < argc) {
42 if (touch_parse_time(argv[++i], tspec) != 0) {
43 fprintf(stderr, "xrdfs: touch: bad -t/--timestamp value '%s' "
44 "(want [[CC]YY]MMDDhhmm[.ss])\n", argv[i]);
45 return 50;
46 }
47 *have_t = 1;
48 } else { *arg = argv[i]; }
49 }
50 return 0;
51}

54/* ---- Populate the utimensat times[2] pair for touch ----
55 *
56 * WHAT: Fills times[0] (atime) and times[1] (mtime) with UTIME_NOW, an explicit
57 * parsed timestamp, or UTIME_OMIT according to which fields touch selected.
58 *
59 * WHY: Encapsulates the per-field UTIME_NOW / OMIT / explicit decision so
60 * do_touch reads as a flat sequence and stays under the complexity cap.
61 *
62 * HOW: 1. Zero tv_sec for both slots; set tv_nsec to UTIME_NOW when the field
63 * is selected without an explicit -t, UTIME_OMIT when unselected, and 0
64 * (a real seconds value follows) when an explicit -t was given.
65 * 2. When have_t is set, copy tspec into each selected slot.
66 */
67static void
68touch_fill_times(struct timespec times[2], int do_atime, int do_mtime,
69 int have_t, const struct timespec *tspec)
70{
71 times[0].tv_sec = times[1].tv_sec = 0;
72 times[0].tv_nsec = do_atime ? (have_t ? 0 : UTIME_NOW) : UTIME_OMIT;
73 times[1].tv_nsec = do_mtime ? (have_t ? 0 : UTIME_NOW) : UTIME_OMIT;
74 if (have_t) {
75 if (do_atime) { times[0] = *tspec; }
76 if (do_mtime) { times[1] = *tspec; }
77 }
78}

81/* ---- Ensure the touch target exists before stamping ----
82 *
83 * WHAT: With no_create set, reports whether the file is absent; otherwise
84 * creates the file if it does not exist. Returns 1 when do_touch should
85 * stop early with success (-c on an absent file), 0 to proceed to setattr.
86 *
87 * WHY: POSIX touch treats -c on a missing file as a silent no-op and otherwise
88 * creates the file before stamping; isolating that keeps do_touch's error
89 * handling focused on the setattr call that actually reports failures.
90 *
91 * HOW: 1. Under -c, stat the path; an absent file returns 1 (no-op), an
92 * existing file returns 0 to continue to the stamp.
93 * 2. Otherwise open write with force=0 (kXR_new); on success close it. An
94 * already-existing file makes the open fail, which is ignored — the
95 * file is present and the caller's setattr still runs. Returns 0.
96 */
97static int
98touch_ensure_file(brix_conn *c, const char *path, int no_create)
99{
100 brix_status cs;
101
102 if (no_create) {
103 brix_statinfo si;
104 brix_status_clear(&cs);
105 if (brix_stat(c, path, &si, &cs) != 0) { return 1; }
106 return 0;
107 }
108 /* Create-if-absent: force=0 ⇒ kXR_new (fails if it already exists, which we
109 * ignore — the file is there and the setattr below still runs). */
110 {
111 brix_file f;
112 brix_status_clear(&cs);
113 if (brix_file_open_write(c, path, 0 /*force=new*/, 0 /*posc*/, &f, &cs) == 0) {
114 brix_file_close(c, &f, &cs);
115 }
116 }
117 return 0;
118}

121/* touch [-c] [-a] [-m] [-t STAMP] <path> — create the file if absent (unless -c) and
122 * set its access/modification times (default: both to now). -a/-m restrict to
123 * atime/mtime; -t sets an explicit [[CC]YY]MMDDhhmm[.ss] time. NEVER changes
124 * ownership: brix_setattr is always called with set_owner = 0. */
125int
126do_touch(brix_conn *c, const char *cwd, int argc, char **argv)
127{
128 brix_status st;
129 char path[XRDC_PATH_MAX];
130 struct timespec times[2], tspec;
131 const char *arg = NULL;
132 int no_create = 0, do_atime = 0, do_mtime = 0, have_t = 0, prc;
133
134 tspec.tv_sec = 0; /* unused unless have_t; init to silence -Wmaybe-uninit */
135 tspec.tv_nsec = 0;
136 prc = touch_parse_argv(argc, argv, &no_create, &do_atime, &do_mtime,
137 &have_t, &tspec, &arg);
138 if (prc != 0) {
139 return prc;
140 }
141 if (arg == NULL) {
142 fprintf(stderr, "usage: touch [-c] [-a] [-m] [-t/--timestamp STAMP] <path>\n");
143 return 50;
144 }
145 if (!do_atime && !do_mtime) { do_atime = do_mtime = 1; } /* default: both */
146 build_path(cwd, arg, path, sizeof(path));
147
148 /* atime = slot 0, mtime = slot 1; per-field UTIME_OMIT when not selected. */
149 touch_fill_times(times, do_atime, do_mtime, have_t, &tspec);
150
151 if (touch_ensure_file(c, path, no_create)) {
152 return 0;
153 }
154
155 brix_status_clear(&st);
156 if (brix_setattr(c, path, 1 /*set_times*/, times, 0 /*set_owner*/,
157 (uint32_t) -1, (uint32_t) -1, &st) != 0) {
158 return xrdfs_report_err("touch", path, &st, 1, c);
159 }
160 return 0;
161}

164/* ln [-s] [-f] <target> <linkpath> — create a hard link (default) or a symbolic link
165 * (-s), GNU arg order (target first). -f removes an existing linkpath first
166 * (best-effort, non-atomic). For -s the target is stored verbatim (link content, not
167 * path-resolved); only linkpath is confined. Hard links confine both paths. */
168int
169do_ln(brix_conn *c, const char *cwd, int argc, char **argv)
170{
171 brix_status st;
172 char linkpath[XRDC_PATH_MAX], oldpath[XRDC_PATH_MAX];
173 const char *target = NULL, *link = NULL;
174 int symbolic = 0, force = 0, i;
175
176 for (i = 1; i < argc; i++) {
177 if (strcmp(argv[i], "-s") == 0) { symbolic = 1; }
178 else if (strcmp(argv[i], "-f") == 0) { force = 1; }
179 else if (target == NULL) { target = argv[i]; }
180 else if (link == NULL) { link = argv[i]; }
181 }
182 if (target == NULL || link == NULL) {
183 fprintf(stderr, "usage: ln [-s] [-f] <target> <linkpath>\n");
184 return 50;
185 }
186 build_path(cwd, link, linkpath, sizeof(linkpath));
187
188 brix_status_clear(&st);
189 if (force) {
190 brix_status rmst;
191 brix_status_clear(&rmst);
192 (void) brix_rm(c, linkpath, &rmst); /* best-effort; ignore "not found" */
193 }
194 if (symbolic) {
195 if (brix_symlink(c, target, linkpath, &st) != 0) { /* target verbatim */
196 fprintf(stderr, "xrdfs: ln -s %s %s: %s\n", target, linkpath, st.msg);
197 xrdfs_op_hints(&st, 1, c); /* WS-3/WS-7 */
198 return brix_shellcode(&st);
199 }
200 return 0;
201 }
202 build_path(cwd, target, oldpath, sizeof(oldpath)); /* hard link: both confined */
203 if (brix_link(c, oldpath, linkpath, &st) != 0) {
204 fprintf(stderr, "xrdfs: ln %s %s: %s\n", oldpath, linkpath, st.msg);
205 xrdfs_op_hints(&st, 1, c); /* WS-3/WS-7 */
206 return brix_shellcode(&st);
207 }
208 return 0;
209}

263/* xattr ls|get|set|rm — extended attributes via kXR_fattr (client/lib/fattr.c).
264 * xattr ls <path> list attribute names
265 * xattr get <path> <name> print one value
266 * xattr set <path> <name> <value> set/replace a value
267 * xattr rm <path> <name> delete an attribute
268 * `xattr <path>` with no subcommand is treated as `ls`. */
269int
270xattr_ls(brix_conn *c, const char *path)
271{
272 brix_status st;
273 char names[8192];
274 size_t total = 0, off;
275
276 brix_status_clear(&st);
277 if (brix_fattr_list(c, path, names, sizeof(names), &total, &st) != 0) {
278 return xrdfs_report_err("xattr ls", path, &st, 0, c);
279 }
280 /* The server returns a NUL-separated list of managed names tagged with a
281 * one-letter namespace prefix ("U.<name>" for the user namespace). Strip the
282 * "<X>." tag so the printed names round-trip directly through xattr get/set. */
283 for (off = 0; off < total && names[off] != '\0'; ) {
284 const char *name = names + off;
285 if (name[0] >= 'A' && name[0] <= 'Z' && name[1] == '.') { name += 2; }
286 printf("%s\n", name);
287 off += strlen(names + off) + 1;
288 }
289 return 0;
290}

398{
399 char path[XRDC_PATH_MAX];
400
401 if (argc < 2) {

498 * with --wait, poll each path's residency until online or the timeout (default 300 s). */

501{

521 fprintf(stderr, "xrdfs: stage: %s\n", st.msg);
522 return brix_shellcode(&st);
523 }
524 if (reply[0] != '\0') { printf("%s\n", reply); }
525 if (wait) {
526 for (i = 0; i < np; i++) {
527 int w = wait_online(c, paths[i], timeout, &st);
528 if (w < 0) {
529 rc = xrdfs_report_err("stage --wait", paths[i], &st, 1, c);
530 } else if (w == 1) {
531 fprintf(stderr, "xrdfs: stage --wait %s: still offline after %ds\n",
532 paths[i], timeout);
533 rc = 1;
534 } else {
535 printf("online: %s\n", paths[i]);
536 }
537 }
538 }
539 return rc;
540}
