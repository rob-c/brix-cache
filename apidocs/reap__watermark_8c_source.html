11/*
12 * brix_cache_purge_to_max_bytes — evict oldest-first until the cache's OWN bytes
13 * fall to `max_bytes` (audit §4.7, upstream `pfc.diskusage files`).
14 *
15 * WHAT: Sums the size of every cached object this node holds and, if that exceeds
16 * max_bytes, evicts oldest-first until it is back within the cap. Unlike the
17 * ppm engine this targets cache-OWNED bytes, not filesystem occupancy — so a
18 * cache that SHARES a filesystem with other data is bounded by what IT holds
19 * rather than by the whole mount's fullness. `max_bytes == 0` = off.
20 * WHY: On a shared filesystem statvfs reflects everyone's data: the FS watermark
21 * either never fires (huge mount) or thrashes (a noisy neighbour), so a cache
22 * needs a cap on its own footprint too. Same candidate set + same
23 * brix_cache_evict_one primitive as the ppm engine (shared via
24 * evict_internal.h), so a victim is demoted / unregistered / sidecar-cleaned
25 * identically.
26 * HOW: Collect + sort oldest-first (shared collector), sum owned bytes, and if
27 * over the cap evict oldest-first tracking the remaining owned bytes as
28 * initial_owned - evicted_bytes (no per-unlink statvfs: skip_usage_remeasure
29 * is set). The CALLER owns the cross-worker eviction lock. Writes the evicted
30 * counts (out-pointers may be NULL); NGX_OK, or NGX_ERROR if the candidate
31 * collection could not start (e.g. the cache root vanished).
32 */
33ngx_int_t
34brix_cache_purge_to_max_bytes(ngx_stream_brix_srv_conf_t *conf, ngx_log_t *log,
35 uint64_t max_bytes, ngx_uint_t *evicted_files_out,
36 uint64_t *evicted_bytes_out)
37{
38 brix_cache_evict_list_t list;
39 brix_cache_fs_usage_t usage; /* unused target — evict_one skips it */
40 brix_cache_evict_ctx_t ec;
41 const char *phys_root;
42 uint64_t owned = 0;
43 size_t i;
44
45 if (evicted_files_out != NULL) { *evicted_files_out = 0; }
46 if (evicted_bytes_out != NULL) { *evicted_bytes_out = 0; }
47
48 if (conf == NULL || max_bytes == 0) {
49 return NGX_OK;
50 }
51 phys_root = brix_cache_state_root(conf);
52 if (phys_root == NULL) {
53 return NGX_ERROR;
54 }
55
56 ngx_memzero(&ec, sizeof(ec));
57 ec.conf = conf;
58 ec.log = log;
59 ec.list = &list;
60 ec.usage = &usage;
61 ec.skip_usage_remeasure = 1; /* bytes target, not FS occupancy */
62
63 if (brix_cache_collect_and_sort(&ec, phys_root, NULL) != NGX_OK) {
64 return NGX_ERROR;
65 }
66
67 for (i = 0; i < list.nelts; i++) {
68 if (list.elts[i].size > 0) {
69 owned += (uint64_t) list.elts[i].size;
70 }
71 }
72
73 /* Evict oldest-first while the still-owned bytes (initial minus what we have
74 * already reclaimed) exceed the cap. evict_one accumulates ec.evicted_bytes. */
75 if (owned > max_bytes) {
76 for (i = 0; i < list.nelts
77 && (owned - ec.evicted_bytes) > max_bytes; i++)
78 {
79 if (list.evicted[i]) {
80 continue;
81 }
82 (void) brix_cache_evict_one(&ec, i); /* skip_usage → always NGX_OK */
83 }
84 }
85
86 brix_cache_free_candidates(&list);
87
88 if (evicted_files_out != NULL) { *evicted_files_out = ec.evicted_files; }
89 if (evicted_bytes_out != NULL) { *evicted_bytes_out = ec.evicted_bytes; }
90 return NGX_OK;
91}

93/* Cheap FS-occupancy pre-check off the 1s TTL-cached sampler: publish the usage
94 * ratio metric and return 1 when over the high watermark. A statvfs failure is
95 * a monitoring fault (log + treat as not-over), never a cache fault. */
96static int
97watermark_fs_over(ngx_stream_brix_srv_conf_t *conf, const char *phys_root,
98 brix_cache_fs_usage_t *usage, ngx_log_t *log)
99{
100 if (brix_cache_fs_usage_sampled((char *) phys_root, 1000, usage) != NGX_OK) {
101 ngx_log_error(NGX_LOG_WARN, log, errno,
102 "brix: watermark reaper could not stat cache root \"%s\"", phys_root);
103 return 0;
104 }
105 brix_metric_cache_usage_ratio(usage->occupancy_ppm);
106 return usage->occupancy_ppm > conf->reaper.high_watermark;
107}

109ngx_uint_t
110brix_cache_watermark_purge(ngx_stream_brix_srv_conf_t *conf, ngx_log_t *log)
111{
112 brix_cache_fs_usage_t usage;
113 char lock_path[PATH_MAX];
114 ngx_uint_t evicted_files = 0;
115 uint64_t evicted_bytes = 0;
116 int cache_active;
117 const char *phys_root;
118 int fs_wm_on; /* the ppm FS-occupancy watermark armed */
119 int files_wm_on; /* the owned-bytes (max_bytes) cap armed */
120 int fs_over = 0;
121
122 if (conf == NULL) {
123 return 0;
124 }
125 /* §14a unification: a cache is active under EITHER the legacy activation
126 * (conf->cache, set by brix_cache on) OR the composable tier grammar (a
127 * cache_store, conf->cache==0). The physical dir the reaper walks comes from
128 * brix_cache_state_root, now tier-aware (returns the posix cache_store dir). */
129 cache_active = (conf->cache != 0) || (conf->common.cache_store.len > 0);
130 phys_root = brix_cache_state_root(conf);
131 if (!cache_active || phys_root == NULL) {
132 return 0;
133 }
134
135 fs_wm_on = (conf->reaper.high_watermark != 0
136 && conf->reaper.high_watermark < BRIX_CACHE_PPM_FULL_SCALE);
137 files_wm_on = (conf->reaper.max_bytes > 0);
138 if (!fs_wm_on && !files_wm_on) {
139 return 0; /* neither watermark armed */
140 }
141
142 /*
143 * FS-occupancy arm: cheap pre-check off the TTL-cached sampler (1s) — avoid
144 * taking the cross-worker lock while the cache is calm. A statvfs failure is a
145 * monitoring fault, not a cache fault: log-and-skip the FS arm, but still run
146 * the owned-bytes arm below (which needs no statvfs).
147 */
148 if (fs_wm_on) {
149 fs_over = watermark_fs_over(conf, phys_root, &usage, log);
150 }
151
152 /* Nothing to do: FS below its mark (or unarmed) and no owned-bytes cap. The
153 * owned-bytes arm has no cheap statvfs proxy, so when it is armed we always
154 * proceed to take the lock and let the purge measure + decide. */
155 if (!fs_over && !files_wm_on) {
156 return 0;
157 }
158
159 ngx_memzero(lock_path, sizeof(lock_path));
160 if (brix_cache_try_evict_lock(conf, lock_path, sizeof(lock_path), log)
161 != NGX_OK)
162 {
163 return 0; /* another worker is purging */
164 }
165
166 /* FS arm: reap down to the LOW watermark (hysteresis), oldest-first. */
167 if (fs_over) {
168 ngx_uint_t ef = 0;
169 uint64_t eb = 0;
170 (void) brix_cache_purge_to_target(conf, NULL, NULL, NULL,
171 conf->reaper.low_watermark, log, &ef, &eb);
172 evicted_files += ef;
173 evicted_bytes += eb;
174 }
175 /* Owned-bytes arm: reap down to the byte cap, oldest-first (§4.7). Reuses the
176 * same candidate set + evict primitive; measures the cache's own footprint. */
177 if (files_wm_on) {
178 ngx_uint_t ef = 0;
179 uint64_t eb = 0;
180 (void) brix_cache_purge_to_max_bytes(conf, log,
181 (uint64_t) conf->reaper.max_bytes, &ef, &eb);
182 evicted_files += ef;
183 evicted_bytes += eb;
184 }
185
186 brix_cache_evict_unlock(lock_path);
187
188 if (evicted_files > 0) {
189 brix_metric_cache_watermark_purge(evicted_files, evicted_bytes);
190 ngx_log_error(NGX_LOG_NOTICE, log, 0,
191 "brix: watermark reaper purged %ui file(s), %uL bytes "
192 "from \"%s\"",
193 evicted_files, (uint64_t) evicted_bytes, phys_root);
194 }
195 return evicted_files;
196}
