32/* brix_conf_set_wt_mode — parse brix_wt_mode sync|async. sync flushes dirty
33 * close data before kXR_close returns; async posts close flushes to the thread
34 * pool. An explicit kXR_sync always flushes synchronously. */
35
36char *
37brix_conf_set_wt_mode(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
38{
39 ngx_stream_brix_srv_conf_t *xcf = conf;
40 ngx_str_t *value;
41
42 value = cf->args->elts;
43 (void) cmd;
44
45 if (ngx_strcmp(value[1].data, "sync") == 0) {
46 xcf->wt.mode = BRIX_WT_MODE_SYNC;
47 } else if (ngx_strcmp(value[1].data, "async") == 0) {
48 xcf->wt.mode = BRIX_WT_MODE_ASYNC;
49 } else {
50 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
51 "brix_wt_mode: invalid value \"%V\", must be sync or async",
52 &value[1]);
53 return NGX_CONF_ERROR;
54 }
55
56 BRIX_CONF_LOG(NGX_LOG_NOTICE, cf, 0,
57 "brix: write-through mode: %s", (xcf->wt.mode == BRIX_WT_MODE_SYNC) ? "sync" : "async");
58 return NGX_CONF_OK;
59}

61/* brix_conf_set_wt_origin — parse brix_wt_origin (same format as cache_origin:
62 * plain host:port or root://host:port, IPv6 brackets supported), the destination
63 * for write-through propagation when wt_enable=on. */
64
65char *
66brix_conf_set_wt_origin(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
67{
68 ngx_stream_brix_srv_conf_t *xcf = conf;
69 ngx_str_t *value;
70 char *addr_copy, *colon, *endp;
71 const u_char *addr_data;
72 size_t addr_len;
73 long pnum;
74
75 value = cf->args->elts;
76 (void) cmd;
77
78 if (xcf->wt.origin_host.len > 0) {
79 return "is duplicate";
80 }
81
82 /* Accept plain host:port or root://host:port (same as cache_origin). */
83 addr_data = value[1].data;
84 addr_len = value[1].len;
85
86 if (addr_len > sizeof("root://") - 1
87 && ngx_strncmp(addr_data, "root://", sizeof("root://") - 1) == 0)
88 {
89 addr_data += sizeof("root://") - 1;
90 addr_len -= sizeof("root://") - 1;
91 }
92
93 BRIX_PNALLOC_OR_RETURN(addr_copy, cf->pool, addr_len + 1, NGX_CONF_ERROR);
94 ngx_memcpy(addr_copy, addr_data, addr_len);
95 addr_copy[addr_len] = '\0';
96
97 /* Parse host:port — supports IPv6 bracket notation. */
98 if (addr_copy[0] == '[') {
99 char *rb = strchr(addr_copy, ']');
100 if (rb == NULL || *(rb + 1) != ':') {
101 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
102 "brix_wt_origin: invalid address \"%V\"", &value[1]);
103 return NGX_CONF_ERROR;
104 }
105 size_t hostlen = (size_t)(rb - addr_copy - 1);
106 if (brix_pstrdupz(cf->pool, &xcf->wt.origin_host,
107 (u_char *) addr_copy + 1, hostlen) != NGX_OK) {
108 return NGX_CONF_ERROR;
109 }
110 pnum = strtol(rb + 2, &endp, 10);
111 } else {
112 colon = strrchr(addr_copy, ':');
113 if (colon == NULL) {
114 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
115 "brix_wt_origin: missing port in \"%V\"", &value[1]);
116 return NGX_CONF_ERROR;
117 }
118 size_t hostlen = (size_t)(colon - addr_copy);
119 if (brix_pstrdupz(cf->pool, &xcf->wt.origin_host,
120 (u_char *) addr_copy, hostlen) != NGX_OK) {
121 return NGX_CONF_ERROR;
122 }
123 pnum = strtol(colon + 1, &endp, 10);
124 }
125
126 if (*endp != '\0' || pnum < BRIX_VFS_PORT_MIN || pnum > BRIX_VFS_PORT_MAX) {
127 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
128 "brix_wt_origin: invalid port in \"%V\" (1-65535)", &value[1]);
129 return NGX_CONF_ERROR;
130 }
131 xcf->wt.origin_port = (uint16_t) pnum;
132
133 BRIX_CONF_LOG(NGX_LOG_NOTICE, cf, 0,
134 "brix: write-through origin: %s:%d",
135 (char *) xcf->wt.origin_host.data, (int) xcf->wt.origin_port);
136
137 return NGX_CONF_OK;
138}

150{
151 ngx_stream_brix_srv_conf_t *xcf = conf;
