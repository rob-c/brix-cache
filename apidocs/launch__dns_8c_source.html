1/* File: launch_dns.c — the kXR_open TPC-destination leg parked on DNS (phase-116)
2 *
3 * WHAT: When the source host of a native TPC pull has no cached answer, the
4 * SSRF gate cannot decide on the event loop without blocking. This file parks
5 * the kXR_open in XRD_ST_AIO on an async brix_dns_resolve() of the source host,
6 * applies the loopback/private policy to every answered address, and either
7 * refuses ("DNS resolution failed for ..." / "... resolves to a prohibited
8 * address") or resumes the prepare pipeline at brix_tpc_prepare_pull_resolved().
9 *
10 * WHY: I-DNS-1 — no getaddrinfo on the event loop: the previous gate blocked
11 * the worker for the resolver's full timeout budget on an unresolvable source.
12 * Deferring the verdict to the pull thread instead would let an unresolvable
13 * source open its destination successfully and fail only at kXR_sync, changing
14 * the wire contract clients and tests pin (the open itself reports the DNS
15 * failure).
16 *
17 * HOW: A self-owned wait (ngx_calloc) holds the request, a copy of the parsed
18 * tpc.* params and destination path, and the streamid. A c->pool cleanup
19 * cancels the request if the client goes away first; the completion disarms
20 * that cleanup, restores the request context (brix_aio_restore_request — 0
21 * means the ctx is gone), answers, and re-arms the loop with brix_aio_resume().
22 * An inline completion (the resolver answered from its own cache) is detected
23 * through the `parked` flag and finished on the starter's stack without ever
24 * entering XRD_ST_AIO. The request logs to the cycle log: c->log lives in the
25 * pool the cleanup guards, and a thread completion must never read it late.
26 * */

63/* 0 = every answered address passes the source policy; -1 = refuse (err). */
64static int
65tpc_dns_wait_verdict(const tpc_dns_wait_t *w, char *err, size_t errsz)
66{
67 brix_net_target_policy_t policy;
68
69 if (w->dns.rc != NGX_OK || w->dns.naddrs == 0) {
70 (void) snprintf(err, errsz, "DNS resolution failed for %s: %s",
71 w->tpc.src_host,
72 w->dns.error ? w->dns.error : "no usable address");
73 return -1;
74 }
75
76 ngx_memzero(&policy, sizeof(policy));
77 policy.allow_local = w->conf->common.tpc_allow_local;
78 policy.allow_private = w->conf->common.tpc_allow_private;
79 policy.dns = w->conf->common.dns.policy;
80
81 return brix_net_target_check_addrs(w->dns.addrs, w->dns.naddrs, &policy,
82 w->tpc.src_host, err, errsz) == NGX_OK
83 ? 0 : -1;
84}

87/* Answer the open from the verdict; frees w. TPC_ANSWERED or NGX_ERROR. */
88static ngx_int_t
89tpc_dns_wait_answer(tpc_dns_wait_t *w)
90{
91 char err[512];
92 ngx_int_t rc;
93
94 if (tpc_dns_wait_verdict(w, err, sizeof(err)) != 0) {
95 rc = brix_tpc_refuse(w->ctx, w->c, w->dst_path, kXR_NotAuthorized,
96 err);
97 } else {
98 rc = brix_tpc_prepare_pull_resolved(w->ctx, w->c, w->conf, &w->tpc,
99 w->dst_path, w->options,
100 w->mode_bits);
101 rc = (rc == NGX_OK) ? TPC_ANSWERED : NGX_ERROR;
102 }
103 ngx_free(w);
104 return rc;
105}

108static void
109tpc_dns_wait_handler(brix_dns_req_t *req)
110{
111 tpc_dns_wait_t *w = req->data;
112 ngx_connection_t *c = w->c;
113
114 if (!w->parked) {
115 w->done = 1; /* inline: the starter answers */
116 return;
117 }
118
119 w->cln->handler = NULL; /* the pool no longer owns the wait */
120 if (!brix_aio_restore_request(w->ctx, w->streamid)) {
121 ngx_free(w); /* client gone: nobody to answer */
122 return;
123 }
124 (void) tpc_dns_wait_answer(w);
125 brix_aio_resume(c);
126}

129ngx_int_t
130brix_tpc_prepare_park_dns(brix_ctx_t *ctx, ngx_connection_t *c,
131 ngx_stream_brix_srv_conf_t *conf, const brix_tpc_params_t *tpc,
132 const char *dst_path, uint16_t options, uint16_t mode_bits)
133{
134 tpc_dns_wait_t *w;
135
136 w = ngx_calloc(sizeof(tpc_dns_wait_t), c->log);
137 if (w == NULL) {
138 return brix_tpc_refuse(ctx, c, dst_path, kXR_ServerError,
139 "TPC source lookup: out of memory");
140 }
141 w->cln = ngx_pool_cleanup_add(c->pool, 0);
142 if (w->cln == NULL) {
143 ngx_free(w);
144 return brix_tpc_refuse(ctx, c, dst_path, kXR_ServerError,
145 "TPC source lookup: out of memory");
146 }
147 w->cln->handler = tpc_dns_wait_sever;
148 w->cln->data = w;
149
150 w->ctx = ctx;
151 w->c = c;
152 w->conf = conf;
153 w->tpc = *tpc;
154 ngx_cpystrn((u_char *) w->dst_path, (u_char *) dst_path,
155 sizeof(w->dst_path));
156 w->streamid[0] = ctx->recv.cur_streamid[0];
157 w->streamid[1] = ctx->recv.cur_streamid[1];
158 w->options = options;
159 w->mode_bits = mode_bits;
160
161 w->dns.name.data = (u_char *) w->tpc.src_host;
162 w->dns.name.len = ngx_strlen(w->tpc.src_host);
163 w->dns.port = tpc->src_port ? tpc->src_port : TPC_DEFAULT_PORT;
164 w->dns.af = BRIX_AF_AUTO;
165 w->dns.socktype = SOCK_STREAM;
166 w->dns.policy = conf->common.dns.policy;
167 w->dns.log = ngx_cycle->log;
168 w->dns.handler = tpc_dns_wait_handler;
169 w->dns.data = w;
170
171 if (brix_dns_resolve(&w->dns) != NGX_OK) {
172 w->cln->handler = NULL;
173 ngx_free(w);
174 return brix_tpc_refuse(ctx, c, dst_path, kXR_ServerError,
175 "TPC source lookup could not be started");
176 }
177
178 if (w->done) {
179 /* answered inline: finish on this stack, never XRD_ST_AIO */
180 w->cln->handler = NULL;
181 return tpc_dns_wait_answer(w);
182 }
183
184 w->parked = 1;
185 ctx->state = XRD_ST_AIO;
186 return TPC_ANSWERED;
187}
