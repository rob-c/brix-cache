28int
29brix_plat_pipe2(int pipefd[2], int flags)
30{
31 if (pipe(pipefd) < 0) {
32 return -1;
33 }
34
35 /* Apply flags via fcntl */
36 if (flags & BRIX_PIPE_CLOEXEC) {
37 fcntl(pipefd[0], F_SETFD, FD_CLOEXEC);
38 fcntl(pipefd[1], F_SETFD, FD_CLOEXEC);
39 }
40
41 if (flags & BRIX_PIPE_NONBLOCK) {
42 int fflags;
43 fflags = fcntl(pipefd[0], F_GETFL, 0);
44 fcntl(pipefd[0], F_SETFL, fflags | O_NONBLOCK);
45 fflags = fcntl(pipefd[1], F_GETFL, 0);
46 fcntl(pipefd[1], F_SETFL, fflags | O_NONBLOCK);
47 }
48
49 return 0;
50}

113/* Same inode (and device) behind both descriptors: guards the F_GETPATH
114 * re-open against a rename/replace between the name lookup and the open. */
115static int
116darwin_same_file(int a, int b)
117{
118 struct stat sa, sb;
119
120 if (fstat(a, &sa) != 0 || fstat(b, &sb) != 0) { /* vfs-seam-allow: NOT_STORAGE - PAL identity check of an already-open file */
121 return 0;
122 }
123 return sa.st_dev == sb.st_dev && sa.st_ino == sb.st_ino;
124}

126int
127brix_plat_fd_reopen_readonly(int fd)
128{
129 char path[MAXPATHLEN];
130 int rfd, flags;
131
132 if (fcntl(fd, F_GETPATH, path) == 0) {
133 rfd = open(path, O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: NOT_STORAGE - PAL descriptor alias of an already-open file */
134 if (rfd >= 0) {
135 if (darwin_same_file(fd, rfd)) {
136 return rfd;
137 }
138 close(rfd);
139 }
140 }
141 /* Unlinked (no live name): only a readable original can be aliased. */
142 flags = fcntl(fd, F_GETFL);
143 if (flags >= 0 && (flags & O_ACCMODE) != O_WRONLY) {
144 rfd = fcntl(fd, F_DUPFD_CLOEXEC, 0);
145 return rfd;
146 }
147 errno = EACCES;
148 return -1;
149}
