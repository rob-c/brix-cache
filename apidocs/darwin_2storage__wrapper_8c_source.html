61static int
62darwin_copy_write_all(int fd, const unsigned char *buffer, size_t len,
63 off_t *off, size_t *written)
64{
65 *written = 0;
66 while (*written < len) {
67 ssize_t nw = darwin_copy_write(fd, buffer + *written,
68 len - *written, off);
69
70 if (nw < 0 && errno == EINTR) {
71 continue;
72 }
73 if (nw <= 0) {
74 return -1;
75 }
76 *written += (size_t) nw;
77 if (off != NULL) {
78 *off += nw;
79 }
80 }
81 return 0;
82}

84static ssize_t
85darwin_copy_write_failure(size_t copied, size_t written, off_t *in_off)
86{
87 if (written != 0) {
88 copied += written;

90 *in_off += (off_t) written;

92 } else if (copied == 0) {
93 return -1;
94 }
95 return (ssize_t) copied;

98int
99brix_plat_pipe2(int pipefd[2], int flags)
100{
101 if (pipe(pipefd) < 0) {
102 return -1;
103 }
104
105 /* Apply flags via fcntl */
106 if (flags & BRIX_PIPE_CLOEXEC) {
107 fcntl(pipefd[0], F_SETFD, FD_CLOEXEC);
108 fcntl(pipefd[1], F_SETFD, FD_CLOEXEC);
109 }
110
111 if (flags & BRIX_PIPE_NONBLOCK) {
112 int fflags;
113 fflags = fcntl(pipefd[0], F_GETFL, 0);
114 fcntl(pipefd[0], F_SETFL, fflags | O_NONBLOCK);
115 fflags = fcntl(pipefd[1], F_GETFL, 0);
116 fcntl(pipefd[1], F_SETFL, fflags | O_NONBLOCK);
117 }
118
119 return 0;
120}

122ssize_t
123brix_plat_copy_range(int in_fd, off_t *in_off,
124 int out_fd, off_t *out_off,
125 size_t len, unsigned int flags)
126{
127 unsigned char *buffer;
128 size_t copied = 0;
129
130 if (flags != 0) {
131 errno = EINVAL;
132 return -1;
133 }
134 if (len == 0) return 0;
135 buffer = malloc(len < BRIX_DARWIN_COPY_BUFFER
136 ? len : BRIX_DARWIN_COPY_BUFFER);
137 if (buffer == NULL) return -1;
138 while (copied < len) {
139 size_t wanted = len - copied;
140 ssize_t nr;
141 size_t written = 0;
142
143 if (wanted > BRIX_DARWIN_COPY_BUFFER) wanted = BRIX_DARWIN_COPY_BUFFER;
144 nr = darwin_copy_read_retry(in_fd, buffer, wanted, in_off);
145 if (nr <= 0) {
146 if (nr < 0 && copied == 0) copied = (size_t) -1;
147 break;
148 }
149 if (darwin_copy_write_all(out_fd, buffer, (size_t) nr, out_off,
150 &written) != 0)
151 {
152 ssize_t result = darwin_copy_write_failure(copied, written,
153 in_off);
154 free(buffer);
155 return result;
156 }
157 copied += (size_t) nr;
158 if (in_off != NULL) *in_off += nr;
159 if ((size_t) nr < wanted) break;
160 }
161 free(buffer);
162 return copied == (size_t) -1 ? -1 : (ssize_t) copied;
163}

187int
188brix_plat_reserve(int fd, off_t size)
189{
190 fstore_t store;
191
192 if (size < 0) {
193 errno = EINVAL;
194 return -1;
195 }
196 memset(&store, 0, sizeof(store));
197 store.fst_flags = F_ALLOCATECONTIG;
198 store.fst_posmode = F_PEOFPOSMODE;
199 store.fst_length = size;
200 if (fcntl(fd, F_PREALLOCATE, &store) == 0) return 0;
201 store.fst_flags = F_ALLOCATEALL;
202 return fcntl(fd, F_PREALLOCATE, &store);
203}

224/* Same inode (and device) behind both descriptors: guards the F_GETPATH
225 * re-open against a rename/replace between the name lookup and the open. */
226static int
227darwin_same_file(int a, int b)
228{
229 struct stat sa, sb;
230
231 if (fstat(a, &sa) != 0 || fstat(b, &sb) != 0) { /* vfs-seam-allow: NOT_STORAGE - PAL identity check of an already-open file */
232 return 0;
233 }
234 return sa.st_dev == sb.st_dev && sa.st_ino == sb.st_ino;
235}

237int
238brix_plat_fd_reopen_readonly(int fd)
239{
240 char path[MAXPATHLEN];
241 int rfd, flags;
242
243 if (fcntl(fd, F_GETPATH, path) == 0) {
244 rfd = open(path, O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: NOT_STORAGE - PAL descriptor alias of an already-open file */
245 if (rfd >= 0) {
246 if (darwin_same_file(fd, rfd)) {
247 return rfd;
248 }
249 close(rfd);
250 }
251 }
252 /* Unlinked (no live name): only a readable original can be aliased. */
253 flags = fcntl(fd, F_GETFL);
254 if (flags >= 0 && (flags & O_ACCMODE) != O_WRONLY) {
255 rfd = fcntl(fd, F_DUPFD_CLOEXEC, 0);
256 return rfd;
257 }
258 errno = EACCES;
259 return -1;
260}
