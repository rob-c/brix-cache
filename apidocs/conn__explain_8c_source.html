16/* ---- Print the server-role summary line decoded from server_flags ----
17 *
18 * WHAT: writes the "Roles:" line (server / manager / proxy, or "(unknown)"
19 * when neither server nor manager bit is set) to out.
20 *
21 * WHY: the role decode is one nameable concern; splitting it keeps
22 * brix_explain_conn a flat sequence and each decode under complexity cap.
23 *
24 * HOW: 1. test each role capability bit in the negotiated flags word; 2. emit
25 * its label or an empty string; 3. append "(unknown)" only when neither
26 * the server nor the manager bit is present.
27 */
28static void
29explain_roles(unsigned f, FILE *out)
30{
31 fprintf(out, "Roles: %s%s%s%s\n",
32 (f & kXR_isServer) ? "server " : "",
33 (f & kXR_isManager) ? "manager " : "",
34 (f & kXR_attrProxy) ? "proxy " : "",
35 (f & (kXR_isServer | kXR_isManager)) ? "" : "(unknown)");
36}

38/* ---- Print the negotiated-capabilities summary line ----
39 *
40 * WHAT: writes the "Caps:" line (TLS availability + gotoTLS requirement,
41 * pgread/pgwrite support, POSC support) decoded from server_flags to out.
42 *
43 * WHY: capability decode is a distinct concern from role decode; isolating it
44 * keeps the orchestrator flat and each helper independently reviewable.
45 *
46 * HOW: 1. map kXR_haveTLS to available/no and annotate with gotoTLS when the
47 * server mandates TLS; 2. map kXR_suppgrw and kXR_supposc to yes/no.
48 */
49static void
50explain_caps(unsigned f, FILE *out)
51{
52 fprintf(out, "Caps: TLS=%s%s pgread/pgwrite=%s POSC=%s\n",
53 (f & kXR_haveTLS) ? "available" : "no",
54 (f & kXR_gotoTLS) ? " (gotoTLS: required)" : "",
55 (f & kXR_suppgrw) ? "yes" : "no",
56 (f & kXR_supposc) ? "yes" : "no");
57}

59/* ---- Print the live TLS transport state, flagging a silent downgrade ----
60 *
61 * WHAT: writes the "TLS:" line — active with version/cipher when a session is
62 * up, or an explicit downgrade WARNING when the server advertised gotoTLS
63 * yet the transport is cleartext, or plain "inactive (cleartext)".
64 *
65 * WHY: a gotoTLS→cleartext downgrade is a security-relevant condition operators
66 * must see; keeping the three-way decision in one helper preserves it
67 * verbatim while capping the orchestrator's complexity.
68 *
69 * HOW: 1. query brix_tls_info for the live cipher; 2. if active, print
70 * version/cipher (defaulting either to "?"); 3. else if the gotoTLS bit is
71 * set, print the downgrade warning; 4. else report plain cleartext.
72 */
73static void
74explain_tls(brix_conn *c, unsigned f, FILE *out)
75{
76 const char *ver = NULL, *cipher = NULL;
77
78 if (brix_tls_info(c, &ver, &cipher)) {
79 fprintf(out, "TLS: active — %s / %s\n",
80 ver ? ver : "?", cipher ? cipher : "?");
81 } else if (f & kXR_gotoTLS) {
82 fprintf(out, "TLS: INACTIVE — WARNING: server advertised gotoTLS but "
83 "the session is cleartext (downgrade)\n");
84 } else {
85 fprintf(out, "TLS: inactive (cleartext)\n");
86 }
87}

89/* ---- Print any credentials discoverable in the process environment ----
90 *
91 * WHAT: writes a "Credentials (in environment):" block describing a discovered
92 * bearer token and/or an X509_USER_PROXY, when either is present; prints
93 * nothing when neither exists.
94 *
95 * WHY: §15.2 — surfacing environment credentials even when they were NOT the
96 * chosen protocol lets operators see "you have a token but the server
97 * didn't offer ztn" and "your proxy is expired". Best-effort/read-only.
98 *
99 * HOW: 1. discover a token and read $X509_USER_PROXY; 2. if either is present,
100 * print the header; 3. explain the token (freeing the discovered copy);
101 * 4. explain the proxy cert when the path is non-empty.
102 */
103static void
104explain_env_credentials(FILE *out)
105{
106 char *tok = brix_token_discover();
107 const char *proxy = getenv("X509_USER_PROXY");
108
109 if (tok != NULL || (proxy != NULL && proxy[0] != '\0')) {
110 fprintf(out, "Credentials (in environment):\n");
111 if (tok != NULL) {
112 brix_token_explain(tok, out);
113 free(tok);
114 }
115 if (proxy != NULL && proxy[0] != '\0') {
116 brix_gsi_cert_explain(proxy, out);
117 }
118 }
119}

121/* ---- Print the 16-byte session id as lowercase hex on the "Session:" line ----
122 *
123 * WHAT: writes "Session: " followed by the hex encoding of c->sessid to out.
124 *
125 * WHY: the byte-loop encode is a separate concern from the flag/cap decodes;
126 * isolating it keeps the orchestrator a flat call sequence.
127 *
128 * HOW: 1. print the label; 2. walk every session-id byte emitting two hex
129 * digits; 3. terminate the line.

131static void
132explain_session_id(brix_conn *c, FILE *out)

143/*
144 * brix_explain_conn — narrate an established session: endpoint, server roles,
145 * negotiated caps, signing, the auth choice, the TLS state, and the session id.
146 *
147 * WHAT: a read-only report over the conn fields conn.c/auth.c already populated.
148 * WHY: both `xrdfs explain` and `xrddiag check` want the same human-readable
149 * picture of what the handshake actually negotiated — single-source it here.
150 * HOW: decode server_flags into roles/caps, defer the per-module auth detail to
151 * brix_auth_explain, and call brix_tls_info for the live cipher (flagging a
152 * gotoTLS→cleartext downgrade as a warning).
153 */
154void
155brix_explain_conn(brix_conn *c, const brix_opts *opts, FILE *out)
156{
157 unsigned f;
158
159 if (c == NULL || out == NULL) {
160 return;
161 }
162 f = (unsigned) c->server_flags;
163
164 fprintf(out, "Endpoint: %s:%d\n", c->host, c->port);
165 explain_roles(f, out);
166 explain_caps(f, out);
167 fprintf(out, "Signing: sec_level=%d%s\n", c->sec_level,
168 c->signing_active ? " (kXR_sigver active)" : "");
169
170 fprintf(out, "Auth:\n");
171 brix_auth_explain(c, opts != NULL ? opts : &c->opts, out);
172
173 explain_tls(c, f, out);
174 explain_env_credentials(out);
175 explain_session_id(c, out);
176}
