1/*
2 * sec_sss.c — SSS (Simple Shared Secret) auth module.
3 *
4 * WHAT: Build the kXR_auth payload for the "sss" protocol: a self-contained
5 * credential the server decrypts with the shared key from its keytab.
6 * WHY: SSS lets trusted hosts authenticate with a pre-shared symmetric key (no
7 * PKI, no token issuer) — common for intra-site XRootD/CMS traffic.
8 * HOW: Discover the keytab ($XrdSecSSSKT / $XrdSecsssKT / ~/.xrd/sss.keytab),
9 * pick the first live key, and assemble exactly what the server's encoder
10 * produces (src/auth/sss/auth_proxy_credential.c): a 16-byte outer header
11 * ("sss\0" ver spare kn enc key-id-BE) followed by BF32( 40-byte data
12 * header [32 random + gen_time-BE + USEDATA] + identity TLVs + IEEE-CRC32 ).
13 *
14 * Three shapes, in precedence order (release-2.0 F9):
15 * * a registry is attached (opts.sss_id) ⇒ mint the entity registered
16 * for opts.sss_lid, and FAIL when that login id is not registered;
17 * * --sss-sndlid ⇒ send the identity-less SNDLID form, then mint the
18 * login id the server names in its kXR_authmore reply (sss_more);
19 * * any of --sss-vorg/--sss-role/--sss-endorse/--sss-creds-file ⇒ a v2
20 * entity credential carrying those fields.
21 * With none of them the wire is byte-identical to the historical v1
22 * single-round NAME-only credential.
23 *
24 * wire: kXR_auth credtype "sss"; blob per src/auth/sss/sss_internal.h.
25 */

42/* Fallback login name for the NAME TLV: this process's own user. Callers that
43 * serve several users from one process override it via opts.sss_user. */
44static const char *
45local_user(void)
46{
47 struct passwd *pw = getpwuid(geteuid());
48 if (pw != NULL && pw->pw_name != NULL && pw->pw_name[0] != '\0') {
49 return pw->pw_name;
50 }
51 return "xrd";
52}

54static int
55sss_load_first_key(brix_sss_key *out, brix_status *st)
56{
57 char path[XRDC_PATH_MAX];
58 brix_sss_key keys[XRDC_SSS_KEYS_MAX];
59 int n = 0;
60
61 brix_sss_keytab_default(path, sizeof(path));
62 if (brix_sss_keytab_read(path, keys, XRDC_SSS_KEYS_MAX, &n, st) != 0) {
63 return -1;
64 }
65 if (n == 0) {
66 brix_status_set(st, XRDC_EAUTH, 0, "keytab %s has no usable keys", path);
67 return -1;
68 }
69 *out = keys[0];
70 return 0;
71}

73static int
74sss_have(brix_conn *c)
75{
76 brix_sss_key key;
77 brix_status st;
78 /* Store is a fast "yes"; a store miss falls through to keytab discovery so a
79 * tool whose store lacks the sss handler still finds a local key. */
80 if (c != NULL && c->opts.cred != NULL
81 && brix_cred_available(c->opts.cred, XRDC_CRED_SSS)) {
82 return 1;
83 }
84 brix_status_clear(&st);
85 return sss_load_first_key(&key, &st) == 0;
86}

88static int
89sss_pick_key(brix_conn *c, brix_sss_key *key, brix_status *st)
90{
91 /* Acquire the keytab from the credential store when present; fall back to
92 * env/default discovery on failure so env-sourced keytabs work as today. */
93 if (c != NULL && c->opts.cred != NULL) {
94 brix_cred_view v;
95 if (brix_cred_acquire(c->opts.cred, XRDC_CRED_SSS, 0, &v, st) == 0
96 && v.path != NULL) {
97 brix_sss_key keys[XRDC_SSS_KEYS_MAX];
98 int n = 0;
99 if (brix_sss_keytab_read(v.path, keys, XRDC_SSS_KEYS_MAX, &n, st) != 0) {
100 return -1;
101 }
102 if (n == 0) {
103 brix_status_set(st, XRDC_EAUTH, 0,
104 "keytab %s has no usable keys", v.path);
105 return -1;
106 }
107 *key = keys[0];
108 return 0;
109 }
110 /* acquire failed or path missing — fall back to env discovery */
111 brix_status_clear(st);
112 }
113 return sss_load_first_key(key, st);
114}

138/*
139 * WHAT: Fill *id from the connection's entity options, under `name`.
140 * WHY: The bounded setters refuse an over-long field rather than truncating
141 * it, because a truncated VO or role is a DIFFERENT authorization claim
142 * — the server's parser refuses the same way, so both ends agree.
143 * HOW: brix_sss_ident_set() for the strings; the proxied credential is read
144 * from its file HERE, at mint time, so a rotated credential is picked up
145 * without reconnecting and the bytes are never cached in brix_opts.
146 */
147static int
148sss_ident_from_opts(brix_conn *c, const char *name, brix_sss_ident *id,
149 brix_status *st)
150{
151 uint8_t creds[BRIX_SSS_ENT_CREDS_MAX];
152 size_t creds_len = 0;
153 int rc;
154
155 if (brix_sss_ident_set(id, name, c->opts.sss_vorg, c->opts.sss_role,
156 NULL, c->opts.sss_endorse) != 0) {
157 brix_status_set(st, XRDC_EAUTH, 0,
158 "sss: identity field too long for the wire format");
159 return -1;
160 }
161 if (c->opts.sss_creds_file == NULL || c->opts.sss_creds_file[0] == '\0') {
162 return 0;
163 }
164 if (brix_sss_creds_read_file(c->opts.sss_creds_file, creds, sizeof(creds),
165 &creds_len) != 0) {
166 brix_status_set(st, XRDC_EAUTH, 0,
167 "sss: cannot read proxied credential %s",
168 c->opts.sss_creds_file);
169 return -1;
170 }
171 rc = brix_sss_ident_set_creds(id, creds, creds_len);
172 if (rc != 0) {
173 brix_status_set(st, XRDC_EAUTH, 0,
174 "sss: proxied credential %s is too large",
175 c->opts.sss_creds_file);
176 }
177 return rc;
178}

181/*
182 * WHAT: Resolve the identity to mint: the registry entry for `lid` when a
183 * registry is attached, else the connection's own entity options under
184 * `name`.
185 * WHY: A registry means this process speaks for several users. A lookup miss
186 * must therefore be an ERROR — substituting the process identity would
187 * present one user's request under another user's name, which is exactly
188 * the confused-deputy bug the registry exists to prevent.
189 * HOW: brix_sss_id_find() is an exact match with no fallback; its miss becomes
190 * an EAUTH status naming the login id that was not registered.
191 */
192static int
193sss_resolve_ident(brix_conn *c, const char *lid, const char *name,
194 brix_sss_ident *id, brix_status *st)
195{
196 if (c == NULL) {
197 brix_status_set(st, XRDC_EAUTH, 0, "sss: no connection context");
198 return -1;
199 }
200 if (c->opts.sss_id != NULL) {
201 if (brix_sss_id_find(c->opts.sss_id, lid, id) != 0) {
202 brix_status_set(st, XRDC_EAUTH, 0,
203 "sss: no identity registered for login id \"%s\"",
204 (lid != NULL) ? lid : "");
205 return -1;
206 }
207 return 0;
208 }
209 return sss_ident_from_opts(c, name, id, st);
210}

213/*
214 * WHAT: The prologue both minters share: a fresh nonce, the SSS-epoch gen_time
215 * and a heap buffer of the caller's size.
216 * WHY: v1 and the entity form differ only in which kernel they call and how big
217 * the buffer is; one prologue means the RNG-failure and out-of-memory
218 * paths are written, and audited, exactly once.
219 * HOW: RAND_bytes into `nonce`, clock into `*gen_time`, malloc into `*blob`. On
220 * any failure `*blob` is left NULL and `st` carries the reason.
221 */
222static int
223sss_mint_prepare(uint8_t nonce[32], uint32_t *gen_time, size_t size,
224 uint8_t **blob, brix_status *st)
225{
226 *blob = NULL;
227
228 if (RAND_bytes(nonce, 32) != 1) {
229 brix_status_set(st, XRDC_EAUTH, 0, "sss: RAND_bytes failed");
230 return -1;
231 }
232 *gen_time = (uint32_t) (time(NULL) - XRDC_SSS_BASE_TIME);
233
234 *blob = (uint8_t *) malloc(size);
235 if (*blob == NULL) {
236 brix_status_set(st, XRDC_EAUTH, 0, "sss: out of memory");
237 return -1;
238 }
239 return 0;
240}

243/*
244 * WHAT: Mint one SSS credential for `ent` (NULL ⇒ the SNDLID form) into a fresh
245 * heap buffer handed back as the kXR_auth payload.
246 * WHY: RNG and clock belong at the edge; the byte assembly lives in the shared
247 * kernel (brix_sss_build_entity_credential, libxrdproto) so client and
248 * server mint the identical wire format from one audited implementation.
249 * HOW: 32 random bytes + the SSS-epoch gen_time, then the kernel; the buffer is
250 * BRIX_SSS_ENTITY_BLOB_MAX, which bounds every field at its cap.
251 */
252static int
253sss_mint(const brix_sss_key *key, const brix_sss_entity_t *ent,
254 uint8_t **payload, uint32_t *plen, brix_status *st)
255{
256 uint8_t *blob;
257 uint8_t nonce[32];
258 uint32_t gen_time;
259 size_t blob_len = 0;
260
261 if (sss_mint_prepare(nonce, &gen_time, BRIX_SSS_ENTITY_BLOB_MAX, &blob, st)
262 != 0)
263 {
264 return -1;
265 }
266 if (brix_sss_build_entity_credential(key->key, key->key_len,
267 (uint64_t) key->id, ent, nonce,
268 gen_time, blob,
269 BRIX_SSS_ENTITY_BLOB_MAX,
270 &blob_len) != 0) {
271 free(blob);
272 brix_status_set(st, XRDC_EAUTH, 0, "sss: credential build failed");
273 return -1;
274 }
275
276 *payload = blob;
277 *plen = (uint32_t) blob_len;
278 return 0;
279}

282/*
283 * WHAT: Mint the historical v1 credential (a bare NAME TLV, no entity).
284 * WHY: This is the byte-for-byte path every deployed server already accepts,
285 * and it stays the default: a client that configures no entity field
286 * sends exactly what it sent before this feature existed.
287 * HOW: The v1 kernel, unchanged, into the 256-byte buffer it has always used
288 * (HDR + 40-byte data header + TLV + CRC).
289 */
290static int
291sss_mint_v1(brix_conn *c, const brix_sss_key *key, uint8_t **payload,
292 uint32_t *plen, brix_status *st)
293{
294 uint8_t *blob;
295 uint8_t nonce[32];
296 uint32_t gen_time;
297 size_t blob_len = 0;
298
299 if (sss_mint_prepare(nonce, &gen_time, 256, &blob, st) != 0) {
300 return -1;
301 }
302 if (brix_sss_build_credential(key->key, key->key_len, (uint64_t) key->id,
303 sss_proposed_user(c), nonce, gen_time, blob,
304 256, &blob_len) != 0) {
305 free(blob);
306 brix_status_set(st, XRDC_EAUTH, 0,
307 "sss: credential build failed (legacy provider?)");
308 return -1;
309 }
310
311 *payload = blob;
312 *plen = (uint32_t) blob_len;
313 return 0;
314}

317static int
318sss_first(brix_conn *c, const char *parms, uint8_t **payload, uint32_t *plen,
319 brix_status *st)
320{
321 brix_sss_key key;
322 brix_sss_ident id;
323 brix_sss_entity_t ent;
324 int rc;
325
326 (void) parms;
327
328 if (sss_pick_key(c, &key, st) != 0) {
329 return -1;
330 }
331
332 /* --sss-sndlid: send the identity-less form and let the server name the
333 * login id in its kXR_authmore reply; sss_more() then mints the entity. */
334 if (c != NULL && c->opts.sss_sndlid) {
335 return sss_mint(&key, NULL, payload, plen, st);
336 }
337
338 /* No registry and no entity field configured ⇒ the historical v1 wire. */
339 if (c == NULL || (c->opts.sss_id == NULL && !sss_entity_wanted(c))) {
340 return sss_mint_v1(c, &key, payload, plen, st);
341 }
342
343 if (sss_resolve_ident(c, c->opts.sss_lid, sss_proposed_user(c), &id, st) != 0) {
344 return -1;
345 }
346 brix_sss_ident_to_entity(&id, &ent);
347 rc = sss_mint(&key, &ent, payload, plen, st);
348 brix_sss_ident_set_creds(&id, NULL, 0); /* wipe any proxied credential */
349 return rc;
350}

353/*
354 * WHAT: Second round: the server answered the SNDLID form with an encrypted
355 * challenge naming the login id it wants; mint that identity.
356 * WHY: Without this the SNDLID form is unusable — the auth loop reports
357 * "server asked for more rounds (unsupported)" and the session dies.
358 * HOW: Decrypt the challenge with the same key (which also proves the peer
359 * holds the shared secret — a wrong key cannot produce a readable LGID),
360 * resolve that login id through the registry when one is attached, and
361 * mint a USEDATA credential for it.
362 */
363static int
364sss_more(brix_conn *c, const uint8_t *sbody, uint32_t slen, uint8_t **payload,
365 uint32_t *plen, brix_status *st)
366{
367 brix_sss_key key;
368 brix_sss_ident id;
369 brix_sss_entity_t ent;
370 char lgid[XRDC_SSS_LID_MAX];
371 int rc;
372
373 if (sss_pick_key(c, &key, st) != 0) {
374 return -1;
375 }
376 if (brix_sss_challenge_lgid(key.key, key.key_len, sbody, (size_t) slen,
377 lgid, sizeof(lgid)) != 0) {
378 brix_status_set(st, XRDC_EAUTH, 0,
379 "sss: unreadable authmore challenge (wrong key?)");
380 return -1;
381 }
382 if (sss_resolve_ident(c, lgid, lgid, &id, st) != 0) {
383 return -1;
384 }
385 brix_sss_ident_to_entity(&id, &ent);
386 rc = sss_mint(&key, &ent, payload, plen, st);
387 brix_sss_ident_set_creds(&id, NULL, 0); /* wipe any proxied credential */
388 return rc;
389}
