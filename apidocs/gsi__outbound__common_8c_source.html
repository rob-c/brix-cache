2 * WHAT: Completes auth handshake after kXR_login returns kXR_authmore.
3 * - Uses WLCG JWT (ztn) from brix_tpc_outbound_bearer_file OR
4 * - Uses GSI cert chain when server advertises &P=ztn / &P=gsi

35/* WHAT: Big-endian uint32 wire encoding helper — htonl(v) → ngx_memcpy to output buffer. */
36
37void
38tpc_put_u32(u_char *p, uint32_t v)
39{
40 uint32_t be;
41
42 /*
43 * XRootD security buckets carry length/tag fields as network-order
44 * (big-endian) uint32. htonl() normalises from host order, then we copy the
45 * raw 4 bytes — ngx_memcpy (not a *(uint32_t *)p store) so callers can write
46 * to unaligned cursor positions inside a packed wire buffer.
47 */
48 be = htonl(v);
49 ngx_memcpy(p, &be, sizeof(be));
50}

52/* WHAT: Construct kXR_auth ClientRequestHdr and send.
53 * - Builds streamid + requestid + ctype from cred_payload
54 * - Sets dlen via htonl
55 * - Sends header then cred_payload
56 * - Returns 0 or -1 with error code */
57
58int
59tpc_send_kxr_auth(brix_tpc_pull_t *t, int fd, u_char stream_seq,
60 const u_char *cred_payload, uint32_t cred_len)
61{
62 ClientRequestHdr hdr;
63 u_char ctype[4];
64
65 /*
66 * Build the fixed 24-byte kXR_auth request header. ClientRequestHdr is the
67 * generic overlay: streamid[2] | requestid (u16) | body[16] | dlen (u32).
68 * For kXR_auth that body region maps to ClientAuthRequest as
69 * reserved[12] | credtype[4], so the credtype lands at body offset 12
70 * (= absolute byte offset 16 in the header).
71 */
72 ngx_memzero(&hdr, sizeof(hdr));
73 /* streamid[1] is the low byte; we use it as the handshake sequence number
74 * (streamid[0] stays 0). Matches the seq the server echoes in replies. */
75 hdr.streamid[1] = stream_seq;
76 hdr.requestid = htons(kXR_auth); /* network order on the wire */
77 /*
78 * The credtype field advertises the auth protocol name as the first 4 bytes
79 * of the payload, NUL-padded — "gsi\0", "ztn\0", etc. Mirror those bytes
80 * verbatim into the header's credtype slot; the full name (and its trailing
81 * data) is also sent in the payload below. cred_payload must be >= 4 bytes.
82 */
83 ngx_memcpy(ctype, cred_payload, sizeof(ctype));
84 ngx_memcpy(hdr.body + 12, ctype, 4); /* credtype field (offset 16) */
85 hdr.dlen = htonl((kXR_int32) cred_len); /* payload length, net order */
86
87 /*
88 * Wire order: header first, then the cred_len-byte payload. Both go through
89 * tpc_send_all which loops over partial writes; a short/failed send here
90 * leaves the socket mid-message, so the caller must treat -1 as fatal for
91 * this connection (it does — every caller aborts the pull).
92 */
93 if (tpc_send_all(t, fd, &hdr, sizeof(hdr)) != 0) {
94 snprintf(t->err_msg, sizeof(t->err_msg), "TPC kXR_auth send hdr failed");
95 t->xrd_error = kXR_ServerError;
96 return -1;
97 }
98 if (tpc_send_all(t, fd, cred_payload, cred_len) != 0) {
99 snprintf(t->err_msg, sizeof(t->err_msg), "TPC kXR_auth send body failed");
100 t->xrd_error = kXR_ServerError;
101 return -1;
102 }
103
104 return 0;
105}

142/* WHAT: JWT bearer auth path.
143 * - Checks delegated_token[0] != '\0' (OAuth2/OIDC result)
144 * - Allocates 4+token_len, copies "ztn\x00" + token
145 * - Sends kXR_auth via tpc_send_kxr_auth(seq=3)
146 * - Receives response checking status == kXR_ok
147 * - Frees cred/body
148 * - Returns 0 or -1 with error code
149 * - Caller: tpc/thread.c (auth dispatch for &P=ztn) */
150
151int
152tpc_outbound_ztn_seq(brix_tpc_pull_t *t, int fd, int seq)
153{
154 u_char *cred;
155 uint32_t cred_len;
156 uint16_t status;
157 uint32_t dlen;
158 u_char *body;
159 size_t token_len;
160
161 /*
162 * Use the delegated token from OAuth2/OIDC token exchange when available;
163 * otherwise fall back to reading from the bearer file.
164 */
165 if (t->delegated_token[0] != '\0') {
166 /* Already populated by an earlier OAuth2/OIDC exchange for this pull. */
167 token_len = strlen(t->delegated_token);
168 } else {
169 u_char token_buf[BRIX_TPC_BEARER_MAX];
170 if (tpc_read_bearer_token(t, token_buf, sizeof(token_buf), &token_len)
171 != 0)
172 {
173 return -1;
174 }
175 /*
176 * Cache the file-sourced token in t->delegated_token so any later auth
177 * round reuses it. token_len + 1 copies the trailing NUL that
178 * tpc_read_bearer_token guarantees; re-derive token_len from the cached
179 * copy so a NUL embedded in the file truncates here, not on the wire.
180 */
181 ngx_memcpy(t->delegated_token, token_buf, token_len + 1);
182 token_len = strlen(t->delegated_token);
183 }
184
185 /*
186 * ztn credential layout: 4-byte protocol tag "ztn\0" followed by the raw
187 * token bytes (no NUL, no length prefix — dlen in the header bounds it).
188 */
189 cred_len = (uint32_t) (4 + token_len);
190 cred = malloc((size_t) cred_len);
191 if (cred == NULL) {
192 snprintf(t->err_msg, sizeof(t->err_msg), "TPC ztn cred OOM");
193 t->xrd_error = kXR_NoMemory;
194 return -1;
195 }
196
197 ngx_memcpy(cred, "ztn\x00", 4); /* protocol tag, slots 0..3 */
198 ngx_memcpy(cred + 4, t->delegated_token, token_len);
199
200 /*
201 * `seq` is the handshake sequence: 3 for the first authenticated request on
202 * a freshly bootstrapped socket (protocol=1, login=2), 4 for a mid-transfer
203 * renewal. cred is a malloc()'d temporary owned here — free it on every exit
204 * path (the send helper copies what it needs into the header and writes the
205 * body inline).
206 */
207 if (tpc_send_kxr_auth(t, fd, seq, cred, cred_len) != 0) {
208 free(cred);
209 return -1;
210 }
211 free(cred);
212
213 /*
214 * tpc_recv_response allocates *body for the server reply; we own it from
215 * here. The reply payload is unused for ztn (success is signalled by status
216 * alone), so free it unconditionally before inspecting status.
217 */
218 body = NULL;
219 if (tpc_recv_response(t, fd, &status, &body, &dlen) != 0) {
220 snprintf(t->err_msg, sizeof(t->err_msg), "TPC ztn auth recv failed");
221 t->xrd_error = kXR_ServerError;
222 return -1;
223 }
224 free(body);
225
226 if (status != kXR_ok) {
227 snprintf(t->err_msg, sizeof(t->err_msg),
228 "TPC ztn auth rejected (status=%u)", (unsigned) status);
229 t->xrd_error = kXR_AuthFailed;
230 return -1;
231 }
232
233 /* W8.2: the source accepted this credential, so it is the one thing a
234 * mid-transfer renewal may replace. A pull that authenticated some other
235 * way — anonymously, or with a GSI proxy — has no ztn credential in play
236 * and must never be sent a renewal kXR_auth. */
237 t->cred_presented = 1;
238 return 0;
239}

241/* WHAT: The original two-argument entry point — a ztn kXR_auth at handshake
242 * sequence 3. WHY: every existing caller (gsi_outbound_finish.c) means exactly
243 * that, the first authenticated request on a freshly bootstrapped outbound
244 * socket; only mid-transfer renewal (tpc/outbound/tpc_token_renew.c) needs a
245 * different sequence, and it says so explicitly. */
246int
247tpc_outbound_ztn(brix_tpc_pull_t *t, int fd)
248{
249 return tpc_outbound_ztn_seq(t, fd, 3);
250}
