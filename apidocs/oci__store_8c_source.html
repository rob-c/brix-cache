1/*
2 * oci_store.c — the local registry's on-disk store (Appendix B.3, §D4.1).
3 *
4 * WHAT: build every path the push surface writes, and perform the five
5 * primitive operations over them — exists, verify, publish-atomically,
6 * put/get a small text record, remove.
7 * WHY: a registry store is a content-addressed tree whose entire safety
8 * argument is "no wire byte ever reaches a path component unvalidated".
9 * Concentrating the path construction in one file is what makes that
10 * argument checkable: every builder below takes an ALREADY-CLASSIFIED
11 * name, digest hex or session id (§0.7.2 — a name that classifies has
12 * no '..' and no leading '/', a digest is 64 lowercase hex characters
13 * by grammar), so the traversal defense is the classifier's, proven
14 * once, rather than a sanitizer repeated per call site and eventually
15 * forgotten in one of them.
16 * HOW: fixed-size buffers and snprintf with an overflow check on every
17 * build; writes go through the phase-108 C10 service-publish verb
18 * (brix_service_publish_*), which stages to a random confined sibling,
19 * fsyncs the data on the durable REGISTRY domain, then renames and flushes
20 * the parent — so a reader never observes a partial object AND an object
21 * the client was told was 201 Created survives a crash.
22 *
23 * The raw namespace calls below carry per-line vfs-seam-allow markers
24 * (invariant #12), for the reason oci_meta.c gives at its own head: this is
25 * the module's OWN store, not a VFS export. Routing it through brix_vfs_*
26 * would hand a miss to the cache decorator, which would try to FILL it from
27 * an upstream registry — and on the registry surface there is none. Object
28 * BYTES leaving the store still go out through the VFS + file_serve pipeline
29 * (oci_registry.c), which is where invariant #2 and the metering live; what
30 * stays here is store bookkeeping and the staging of received uploads.
31 */

51ngx_int_t
52brix_oci_store_init(brix_oci_store_t *st, ngx_http_brix_oci_loc_conf_t *lcf)
53{
54 const char *root;
55 size_t n;
56
57 /* The merge has already folded brix_oci_registry_root into common.root
58 * and canonicalised it, so the canonical form is the ONLY form worth
59 * reading here: taking the raw directive string back would reintroduce
60 * the symlinks and "." components brix_prepare_export_root removed. */
61 root = lcf->common.root_canon;
62
63 if (root == NULL || root[0] != '/') {
64 return NGX_ERROR;
65 }
66 n = ngx_strlen(root);
67 if (n == 0 || n >= sizeof(st->root)) {
68 return NGX_ERROR;
69 }
70
71 /* A trailing slash would double up in every builder below; normalise it
72 * away here so the builders can concatenate unconditionally. */
73 while (n > 1 && root[n - 1] == '/') {
74 n--;
75 }
76 ngx_memcpy(st->root, root, n);
77 st->root[n] = '\0';
78 st->root_len = n;
79
80 return NGX_OK;
81}

84/* Every builder funnels through this so the overflow check exists once. The
85 * printf format attribute earns its keep twice: it type-checks fmt against the
86 * arguments at every builder callsite, and it is what tells the compiler the
87 * vsnprintf below is forwarding a checked format rather than an arbitrary
88 * runtime string (-Wformat-nonliteral). */

90oci_store_fmt(char *out, size_t outsz, const char *fmt, ...)
91 __attribute__((format(printf, 3, 4)));

93static int
94oci_store_fmt(char *out, size_t outsz, const char *fmt, ...)
95{
96 va_list ap;
97 int n;
98
99 va_start(ap, fmt);
100 n = vsnprintf(out, outsz, fmt, ap);
101 va_end(ap);
102
103 return (n < 0 || (size_t) n >= outsz) ? -1 : 0;
104}

146ngx_int_t
147brix_oci_store_mkparent(const char *path, ngx_log_t *log)
148{
149 char buf[PATH_MAX];
150 size_t n = ngx_strlen(path);
151 char *slash;
152
153 if (n == 0 || n >= sizeof(buf)) {
154 return NGX_ERROR;
155 }
156 ngx_memcpy(buf, path, n + 1);
157
158 slash = strrchr(buf, '/');
159 if (slash == NULL || slash == buf) {
160 return NGX_OK; /* parent is "/" — already there */
161 }
162 *slash = '\0';
163
164 /* Walk left to right creating each component. EEXIST is the common case
165 * (the tree is shallow and shared), so it is not an error — anything
166 * else is, and stops the walk rather than pressing on into a store whose
167 * shape we no longer know. */
168 for (char *p = buf + 1; *p != '\0'; p++) {
169 if (*p != '/') {
170 continue;
171 }
172 *p = '\0';
173 if (mkdir(buf, OCI_STORE_DIR_MODE) != 0 /* vfs-seam-allow: DOMAIN_REGISTRY — registry store tree, not a VFS export object */
174 && ngx_errno != NGX_EEXIST)
175 {
176 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
177 "oci: cannot create store directory \"%s\"", buf);
178 return NGX_ERROR;
179 }
180 *p = '/';
181 }
182 if (mkdir(buf, OCI_STORE_DIR_MODE) != 0 /* vfs-seam-allow: DOMAIN_REGISTRY — registry store tree, not a VFS export object */
183 && ngx_errno != NGX_EEXIST)
184 {
185 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
186 "oci: cannot create store directory \"%s\"", buf);
187 return NGX_ERROR;
188 }
189
190 return NGX_OK;
191}

209ngx_int_t
210brix_oci_store_verify(const char *path, const brix_oci_digest_t *want,
211 ngx_log_t *log)
212{
213 brix_oci_hash_ctx_t sha;
214 brix_oci_digest_t got;
215 u_char buf[OCI_STORE_IO_CHUNK];
216 ngx_int_t rc = NGX_ERROR;
217 ssize_t n;
218 int fd;
219
220 fd = open(path, O_RDONLY | O_CLOEXEC); /* vfs-seam-allow: DOMAIN_REGISTRY — seal-time hash of the staged upload, before it becomes an object */
221 if (fd < 0) {
222 return NGX_ERROR;
223 }
224 /* Hash under the algorithm the CALLER is verifying against, never a
225 * fixed one: verifying a sha512 blob with sha256 would compare two
226 * different functions' output and reject every honest upload. */
227 if (brix_oci_hash_init(&sha, want->alg) != 0) {
228 (void) close(fd);
229 return NGX_ERROR;
230 }
231
232 /* Sequential whole-file read: the seal is the one moment the registry
233 * must know what it just received, and hashing here costs one pass over
234 * bytes that are still in page cache from the PATCH that wrote them. */
235 for ( ;; ) {
236 n = read(fd, buf, sizeof(buf));
237 if (n < 0) {
238 if (ngx_errno == NGX_EINTR) {
239 continue;
240 }
241 ngx_log_error(NGX_LOG_ERR, log, ngx_errno,
242 "oci: unreadable staged upload \"%s\"", path);
243 brix_oci_hash_abort(&sha);
244 (void) close(fd);
245 return NGX_ERROR;
246 }
247 if (n == 0) {
248 break;
249 }
250 if (brix_oci_hash_update(&sha, buf, (size_t) n) != 0) {
251 brix_oci_hash_abort(&sha);
252 (void) close(fd);
253 return NGX_ERROR;
254 }
255 }
256 (void) close(fd);
257
258 if (brix_oci_hash_final(&sha, &got) != 0) {
259 return NGX_ERROR;
260 }
261 rc = brix_oci_digest_eq(&got, want) ? NGX_OK : NGX_DECLINED;
262
263 return rc;
264}

267/* Fill the service-publish request the two adapters share: same store root and
268 * log, the caller's domain and exclusivity; the mode is left 0 so the published
269 * object inherits the staged temp's private 0600 — the mode every store object
270 * has always carried, now without a redundant fchmod. */
271static void
272oci_store_publish_req(brix_service_publish_req_t *req,
273 const brix_oci_store_t *st, const char *final_path, brix_vfs_domain_t domain,
274 unsigned excl, ngx_log_t *log)
275{
276 ngx_memzero(req, sizeof(*req));
277 req->log = log;
278 req->domain = domain;
279 req->root_canon = st->root;
280 req->final_path = final_path;
281 req->excl = excl;
282}

409int
410brix_oci_store_tag_list(const brix_oci_store_t *st, const char *name,
411 size_t name_len, char *out, size_t outsz)
412{
413 char dir[PATH_MAX];
414 struct dirent *ent;
415 size_t used = 0;
416 int count = 0;
417 DIR *dh;
418
419 out[0] = '\0';
420
421 if (brix_oci_store_repo_path(st, name, name_len, "tags",
422 dir, sizeof(dir)) != 0)
423 {
424 return -1;
425 }
426 dh = opendir(dir); /* vfs-seam-allow: DOMAIN_REGISTRY — registry's own store index, not a VFS export listing */
427 if (dh == NULL) {
428 return 0; /* a repo with no tags is not an error */
429 }
430
431 while ((ent = readdir(dh)) != NULL) { /* vfs-seam-allow: DOMAIN_REGISTRY — registry's own store index, not a VFS export listing */
432 size_t len = ngx_strlen(ent->d_name);
433
434 /* Skip dotfiles, over-long entries, and a crash-orphaned publish temp
435 * (<tag>.xrd-tmp.<pid>.<rand>) that the startup reaper has not yet swept
436 * — a stale temp is not a tag. */
437 if (ent->d_name[0] == '.' || used + len + 2 > outsz
438 || brix_tmp_is_temp_name(ent->d_name)) {
439 continue;
440 }
441 ngx_memcpy(out + used, ent->d_name, len);
442 used += len;
443 out[used++] = '\n';
444 out[used] = '\0';
445 count++;
446 }
447 (void) closedir(dh);
448
449 return count;
450}

485void
486brix_oci_store_drop_dir(const char *dir, ngx_log_t *log)
487{
488 static const char *const children[] = { "part", "meta" };
489 char path[PATH_MAX];
490 size_t i;
491
492 /* A session directory holds exactly the two children this module puts
493 * there, so the teardown enumerates them rather than walking: a recursive
494 * remove over an attacker-influenced path is a bigger primitive than this
495 * job needs. */
496 for (i = 0; i < sizeof(children) / sizeof(children[0]); i++) {
497 if (oci_store_fmt(path, sizeof(path), "%s/%s", dir, children[i]) == 0) {
498 (void) brix_oci_store_remove(path, log);
499 }
500 }
501 if (rmdir(dir) != 0 && ngx_errno != NGX_ENOENT) { /* vfs-seam-allow: DOMAIN_REGISTRY — session directory teardown (abort / reap / seal) */
502 ngx_log_error(NGX_LOG_WARN, log, ngx_errno,
503 "oci: cannot remove upload session \"%s\"", dir);
504 }
505}
