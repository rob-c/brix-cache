1/*
2 * prefetch.c — never-blocking forward probe + background fill (phase-116 W7).
3 *
4 * WHAT: brix_dns_lookup_cached() answers from the IP-literal parser or the
5 * per-worker cache and otherwise starts one background fill and reports
6 * NGX_AGAIN; brix_dns_prefetch() is that fill on its own.
7 * WHY: Some event-loop decisions want a hostname's addresses but must not
8 * wait for them: the loop-side TPC source preflight answers from the
9 * cache and parks the kXR_open on its own async request otherwise
10 * (brix_dns_cache_probe: no fill), while warm-up sites want the fill
11 * started for a later probe (brix_dns_lookup_cached). Without a probe
12 * those sites would either block (I-DNS-1) or resolve twice.
13 * HOW: The fill is a self-owned brix_dns_req_t (ngx_calloc, hostname copied
14 * behind it) whose handler frees it; the driver stores the answer in
15 * brix_dns_req_finish(). brix_dns_cache_mark_pending() dedups fills
16 * for the same (name, af) while one is in flight.
17 */
18#include "net/dns/dns.h"

22dns_prefetch_done(brix_dns_req_t *req)

33void
34brix_dns_prefetch(const brix_dns_policy_t *policy, const char *host,
35 in_port_t port, brix_af_policy_t af)
36{
37 brix_dns_req_t *req;
38 ngx_str_t name;
39 ngx_log_t *log = ngx_cycle->log; /* the fill may outlive its caller */
40 time_t hold = BRIX_DNS_PENDING_HOLD;
41
42 if (host == NULL || host[0] == '\0') {
43 return;
44 }
45 name.data = (u_char *) host;
46 name.len = ngx_strlen(host);
47 if (name.len >= BRIX_RESOLV_DOMAIN_LEN) {
48 return;
49 }
50 if (policy != NULL) {
51 time_t budget = (time_t) policy->rc.timeout * policy->rc.attempts
52 * (1 + (policy->search ? policy->rc.nsearch : 0)) + 1;
53
54 hold = ngx_max(hold, budget);
55 }
56 if (!brix_dns_cache_mark_pending(&name, af, hold)) {
57 return; /* cached, negative or in flight */
58 }
59 req = ngx_calloc(sizeof(brix_dns_req_t) + name.len + 1, log);
60 if (req == NULL) {
61 return;
62 }
63 req->name.data = (u_char *) (req + 1);
64 ngx_memcpy(req->name.data, name.data, name.len);
65 req->name.data[name.len] = '\0';
66 req->name.len = name.len;
67 req->port = port;
68 req->af = af;
69 req->socktype = SOCK_STREAM;
70 req->policy = policy;
71 req->log = log;
72 req->handler = dns_prefetch_done;
73 if (brix_dns_resolve(req) != NGX_OK) {
74 ngx_free(req); /* handler did not run */
75 }
76}

79static ngx_int_t
80dns_lookup_literal(brix_dns_req_t *req, brix_dns_addr_t *addrs,
81 ngx_uint_t *naddrs, char *err, size_t errsz)
82{
83 brix_dns_addr_t lit;
84
85 if (brix_dns_parse_literal(&req->name, &lit) != NGX_OK) {
86 return NGX_AGAIN; /* not a literal */
87 }
88 if ((req->af == BRIX_AF_INET && lit.ss.ss_family != AF_INET)
89 || (req->af == BRIX_AF_INET6 && lit.ss.ss_family != AF_INET6))
90 {
91 if (err != NULL && errsz > 0) {
92 (void) ngx_snprintf((u_char *) err, errsz,
93 "address family excluded by policy%Z");
94 }
95 return NGX_DECLINED;
96 }
97 ngx_inet_set_port((struct sockaddr *) &lit.ss, req->port);
98 addrs[0] = lit;
99 *naddrs = 1;
100 return NGX_OK;
101}

104/* Literal -> cache, no fill. NGX_OK / NGX_DECLINED / NGX_AGAIN (unknown). */
105ngx_int_t
106brix_dns_cache_probe(const brix_dns_policy_t *policy, const char *host,
107 in_port_t port, brix_af_policy_t af, brix_dns_addr_t *addrs,
108 ngx_uint_t max, ngx_uint_t *naddrs, char *err, size_t errsz)
109{
110 brix_dns_req_t req;
111 ngx_uint_t i;
112 ngx_int_t rc;
113
114 *naddrs = 0;
115 brix_dns_set_err(err, errsz, "");
116 if (host == NULL || host[0] == '\0' || max == 0) {
117 brix_dns_set_err(err, errsz, "empty hostname");
118 return NGX_DECLINED;
119 }
120 ngx_memzero(&req, sizeof(req));
121 req.name.data = (u_char *) host;
122 req.name.len = ngx_strlen(host);
123 req.port = port;
124 req.af = af;
125 req.socktype = SOCK_STREAM;
126 req.policy = policy;
127 req.log = ngx_cycle->log;
128
129 rc = dns_lookup_literal(&req, addrs, naddrs, err, errsz);
130 if (rc != NGX_AGAIN) {
131 return rc;
132 }
133 if (brix_dns_cache_lookup(&req) != NGX_OK) {
134 brix_dns_set_err(err, errsz, "no cached answer");
135 return NGX_AGAIN;
136 }
137 if (req.rc != NGX_OK || req.naddrs == 0) {
138 brix_dns_set_err(err, errsz,
139 req.error ? req.error : "no usable address");
140 return NGX_DECLINED;
141 }
142 *naddrs = ngx_min(req.naddrs, max);
143 for (i = 0; i < *naddrs; i++) {
144 addrs[i] = req.addrs[i];
145 }
146 return NGX_OK;
147}
