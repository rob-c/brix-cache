3 *
4 * WHAT: The nginx-coupled vtable slots of brix_sd_posix_driver — stat/unlink/
5 * mkdir/rename/server_copy, directory iteration and xattr metadata —

63/* sd_posix_ns_result — collapse a namespace result to NGX_OK, or set errno (prefer
64 * sys_errno, else the status mapping for derived states) and return NGX_ERROR. */
65static ngx_int_t
66sd_posix_ns_result(brix_ns_result_t res)
67{
68 if (res.status == BRIX_NS_OK) {
69 return NGX_OK;
70 }
71 errno = res.sys_errno != 0 ? res.sys_errno
72 : brix_vfs_ns_status_errno(res.status);
73 return NGX_ERROR;

76/* sd_posix_abs_key — materialise the vtable's root-RELATIVE key (leading
77 * slash, matching sd_posix_open/stat and the non-POSIX drivers) as the
78 * ABSOLUTE path under root_canon that the brix_ns_* / *_confined_canon
79 * helpers take (they strip root_canon; the relative form silently failed).
80 * Returns root_canon for the helper call that follows, or NULL with errno
81 * ENAMETOOLONG when the key does not fit. */
82static const char *
83sd_posix_abs_key(brix_sd_instance_t *inst, const char *path, char *abs,
84 size_t cap)
85{
86 sd_posix_state_t *st = inst->state;
87
88 if ((size_t) snprintf(abs, cap, "%s%s", st->root_canon, path) >= cap) {
89 errno = ENAMETOOLONG;
90 return NULL;
91 }
92 return st->root_canon;
93}

97{
98 brix_ns_delete_opts_t opts;
99 sd_posix_state_t *st;

102

104 if (root == NULL) {

113 * semantics, minus a root open/close per delete. */
114 return sd_posix_ns_result(
115 brix_ns_delete_at(inst->log, st->rootfd, root, abspath, &opts));
116 }
117 return sd_posix_ns_result(
118 brix_ns_delete(inst->log, root, abspath, &opts));

121/* WHAT: Apply a metadata mutation (mode / times / owner) to `path`, kernel-
122 * confined under the export root.
123 * WHY: The setattr slot was unimplemented for the default POSIX driver, so a
124 * tier decorator's metadata fixup (sd_stage hydration forcing the write
125 * spool owner-rw, sd_cache's forward) silently no-opped on a posix
126 * store; kXR_setattr parity for driver-backed exports needs it too.
127 * HOW: Root-relative key -> absolute path (the unlink/mkdir convention), then
128 * the existing confined helpers: brix_chmod_confined_canon for set_mode,
129 * brix_setattr_confined_canon for set_times/set_owner. First failure
130 * returns NGX_ERROR with errno left set. */
131ngx_int_t
132sd_posix_setattr(brix_sd_instance_t *inst, const char *path,
133 const brix_sd_setattr_t *attr)
134{
135 char abspath[PATH_MAX];
136 const char *root;
137
138 root = sd_posix_abs_key(inst, path, abspath, sizeof(abspath));
139 if (root == NULL) {
140 return NGX_ERROR;
141 }
142 if (attr->set_mode
143 && brix_chmod_confined_canon(inst->log, root, abspath,
144 attr->mode & 07777) != 0)
145 {
146 return NGX_ERROR;
147 }
148 if (attr->set_times || attr->set_owner) {
149 struct timespec times[2];
150
151 times[0] = attr->atime;
152 times[1] = attr->mtime;
153 if (brix_setattr_confined_canon(inst->log, root, abspath,
154 attr->set_times ? 1 : 0, times,
155 attr->set_owner ? 1 : 0, attr->uid, attr->gid) != 0)
156 {
157 return NGX_ERROR;
158 }
159 }
160 return NGX_OK;
161}

219ngx_int_t
220sd_posix_server_copy(brix_sd_instance_t *inst, const char *src,
221 const char *dst, off_t *bytes_out)
222{
223 sd_posix_state_t *st = inst->state;
224 brix_ns_copy_opts_t opts;
225 char abssrc[PATH_MAX];
226 char absdst[PATH_MAX];
227 ngx_int_t rc;
228
229 /* Same contract as sd_posix_rename: the vtable key is root-RELATIVE, but
230 * brix_ns_local_copy strips root_canon off ABSOLUTE paths and treats a
231 * non-match as a cross-root copy (EXDEV). Handing it the relative form
232 * failed every server-side COPY on a driver-backed export — WebDAV COPY
233 * answered 403 and S3 CopyObject 500 — while a plain export (NULL driver,
234 * VFS namespace path) worked, which is why it went unnoticed. */
235 if (sd_posix_abs_key(inst, src, abssrc, sizeof(abssrc)) == NULL
236 || sd_posix_abs_key(inst, dst, absdst, sizeof(absdst)) == NULL)
237 {
238 return NGX_ERROR;
239 }
240
241 ngx_memzero(&opts, sizeof(opts));
242 opts.overwrite = 1;
243 rc = sd_posix_ns_result(
244 brix_ns_local_copy(inst->log, st->root_canon, abssrc, absdst, &opts));
245
246 if (rc == NGX_OK && bytes_out != NULL) {
247 struct stat sb;
248 *bytes_out = (sd_posix_stat(inst, dst, &(brix_sd_stat_t){0}) == NGX_OK
249 && brix_lstat_beneath(st->rootfd, dst, &sb) == 0)
250 ? sb.st_size : 0;
251 }
252 return rc;
253}

262brix_sd_dir_t *
263sd_posix_opendir(brix_sd_instance_t *inst, const char *path, int *err_out)
264{
265 sd_posix_state_t *st = inst->state;
266 brix_sd_dir_t *dir;
267 sd_posix_dir_t *pd;
268 int fd;
269 DIR *dp;
270
271 fd = brix_open_beneath(st->rootfd, path, O_RDONLY | O_DIRECTORY, 0);
272 if (fd < 0) {
273 if (err_out != NULL) { *err_out = errno; }
274 return NULL;
275 }
276
277 dp = fdopendir(fd);
278 if (dp == NULL) {
279 if (err_out != NULL) { *err_out = errno; }
280 close(fd);
281 return NULL;
282 }
283
284 dir = ngx_pcalloc(inst->pool, sizeof(*dir));
285 pd = ngx_pcalloc(inst->pool, sizeof(*pd));
286 if (dir == NULL || pd == NULL) {
287 closedir(dp);
288 if (err_out != NULL) { *err_out = ENOMEM; }
289 return NULL;
290 }
291
292 pd->dp = dp;
293 dir->inst = inst;
294 dir->state = pd;
295 return dir;
296}

298ngx_int_t
299sd_posix_readdir(brix_sd_dir_t *d, brix_sd_dirent_t *out)
300{
301 sd_posix_dir_t *pd = d->state;
302 struct dirent *de;
303
304 for ( ;; ) {
305 errno = 0;
306 de = readdir(pd->dp);
307 if (de == NULL) {
308 return errno != 0 ? NGX_ERROR : NGX_DONE;
309 }
310 if (de->d_name[0] == '.'
311 && (de->d_name[1] == '\0'
312 || (de->d_name[1] == '.' && de->d_name[2] == '\0')))
313 {
314 continue; /* skip "." and ".." */
315 }
316 ngx_cpystrn((u_char *) out->name, (u_char *) de->d_name,
317 sizeof(out->name));
318 /* The kernel's own classification, DT_UNKNOWN on filesystems that
319 * don't fill it — never guessed here (consumers stat on UNKNOWN). */
320 out->d_type = de->d_type;
321 return NGX_OK;
322 }
