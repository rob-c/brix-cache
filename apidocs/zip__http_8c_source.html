25static int
26zip_http_name_ok(const char *out, size_t n)
27{
28 if (n == 0 || out[0] == '/' || ngx_strcmp(out, "..") == 0) {
29 return 0;
30 }
31 if (n >= 3 && out[0] == '.' && out[1] == '.' && out[2] == '/') {
32 return 0;
33 }
34 if (strstr(out, "/../") != NULL) {
35 return 0;
36 }
37 if (n >= 3 && ngx_strcmp(out + n - 3, "/..") == 0) {
38 return 0;
39 }
40 return 1;
41}

88/*
89 * WHAT: open the archive through the confined VFS read surface and verify it
90 * is a regular file.
91 * WHY: the member fd is used transiently to read the central directory and
92 * extract/dup the member; every failure maps to a distinct HTTP status.
93 * HOW: brix_vfs_open() under a WEBDAV read-only ctx, then fstat/S_ISREG.
94 * Returns NGX_OK with *fhp and *sizep set, or an HTTP status (fh
95 * already closed).
96 */
97static ngx_int_t
98zip_http_open_archive(ngx_http_request_t *r,
99 const brix_vfs_ctx_t *vfs_scope,
100 const char *archive_full, brix_vfs_file_t **fhp, off_t *sizep)
101{
102 struct stat ast;
103 brix_vfs_ctx_t vctx;
104 brix_vfs_file_t *fh;
105 int vfs_err = 0;
106 if (vfs_scope == NULL
107 || brix_vfs_ctx_derive_path(&vctx, vfs_scope, archive_full) != NGX_OK)
108 {
109 return NGX_HTTP_INTERNAL_SERVER_ERROR;
110 }
111 fh = brix_vfs_open(&vctx, BRIX_VFS_O_READ, &vfs_err);
112 if (fh == NULL) {
113 errno = vfs_err;
114 if (errno == ENOENT || errno == ENOTDIR) return NGX_HTTP_NOT_FOUND;
115 if (errno == EACCES || errno == EPERM) return NGX_HTTP_FORBIDDEN;
116 return NGX_HTTP_INTERNAL_SERVER_ERROR;
117 }
118 if (fstat(brix_vfs_file_fd(fh), &ast) != 0 || !S_ISREG(ast.st_mode)) {
119 (void) brix_vfs_close(fh, r->connection->log);
120 return NGX_HTTP_INTERNAL_SERVER_ERROR;
121 }
122
123 *fhp = fh;
124 *sizep = (off_t) ast.st_size;
125 return NGX_OK;
126}

128/*

131 * HOW: brix_zip_find_member() over the open archive fd; NOMEMBER → 404,

140

151/*
152 * WHAT: apply a single client byte Range against the member's UNCOMPRESSED
153 * length.
154 * WHY: ZIP members are ranged on their logical (inflated) content, not the
155 * archive bytes, so the standard range filter cannot be used.
156 * HOW: parses "bytes=a-[b]" / "bytes=-suffix" from the Range header into
157 * sp->start/sp->len (pre-set to the whole span). Returns NGX_OK, or
158 * NGX_ERROR when the range is unsatisfiable (start beyond total).
159 */
160static ngx_int_t
161zip_http_parse_range(ngx_http_request_t *r, zip_http_span_t *sp)
162{
163 ngx_table_elt_t *range = r->headers_in.range;
164 u_char *p, *eov, *dash;
165
166 if (range == NULL || range->value.len <= 7
167 || ngx_strncasecmp(range->value.data, (u_char *) "bytes=", 6) != 0)
168 {
169 return NGX_OK;
170 }
171
172 p = range->value.data + 6;
173 eov = range->value.data + range->value.len;
174 dash = (u_char *) ngx_strlchr(p, eov, '-');
175 if (dash == NULL) {
176 return NGX_OK;
177 }
178
179 {
180 off_t a = (dash > p) ? ngx_atoof(p, dash - p) : -1;
181 off_t b2 = (dash + 1 < eov) ? ngx_atoof(dash + 1, eov - dash - 1)
182 : -1;
183 if (a >= 0) { /* bytes=a-[b] */
184 sp->start = a;
185 sp->len = (b2 >= 0 && b2 >= a) ? (b2 - a + 1) : (sp->total - a);
186 } else if (b2 > 0) { /* bytes=-suffix */
187 sp->start = (b2 < sp->total) ? sp->total - b2 : 0;
188 sp->len = sp->total - sp->start;
189 }
190 }
191 if (sp->start >= sp->total) {
192 return NGX_ERROR;
193 }
194 if (sp->start + sp->len > sp->total) {
195 sp->len = sp->total - sp->start;
196 }
197 return NGX_OK;
198}

200/*
201 * WHAT: stage a STORE member for zero-copy sendfile over the archive range.
202 * WHY: stored members need no inflation — the response body IS a byte range
203 * of the archive file, so a dup'd fd + file-backed buf is cheapest.
204 * HOW: dup the archive fd (kept alive across the async send by a pool
205 * cleanup), close the VFS handle, point b at data_off + span. Always
206 * consumes fh; returns NGX_OK or 500.
207 */
208static ngx_int_t
209zip_http_stage_stored(ngx_http_request_t *r, brix_vfs_file_t *fh,
210 const brix_zip_member_t *m, const zip_http_span_t *sp, ngx_buf_t *b)
211{
212 ngx_file_t *f = ngx_pcalloc(r->pool, sizeof(ngx_file_t));
213 ngx_pool_cleanup_t *cln = ngx_pool_cleanup_add(r->pool, sizeof(ngx_fd_t));
214 int send_fd;
215
216 send_fd = dup(brix_vfs_file_fd(fh));
217 (void) brix_vfs_close(fh, r->connection->log);
218 if (f == NULL || cln == NULL || send_fd == (ngx_fd_t) -1) {
219 if (send_fd != (ngx_fd_t) -1) (void) close(send_fd);
220 return NGX_HTTP_INTERNAL_SERVER_ERROR;
221 }
222 cln->handler = zip_http_close_fd;
223 *(ngx_fd_t *) cln->data = send_fd;
224 f->fd = send_fd;
225 f->log = r->connection->log;
226 b->in_file = 1;
227 b->file = f;
228 b->file_pos = (off_t) m->data_off + sp->start;
229 b->file_last = (off_t) m->data_off + sp->start + sp->len;
230 return NGX_OK;
231}

233/*
234 * WHAT: stage a DEFLATE member by inflating it into an r->pool buffer.
235 * WHY: compressed members cannot be ranged/sent from the archive bytes;
236 * they are fully inflated (capped at ZIP_HTTP_MEM_MAX → 413) and served
237 * from memory.
238 * HOW: ngx_palloc + brix_zip_extract_full(), then point b at the requested
239 * span of the inflated data. Always consumes fh; returns NGX_OK,
240 * 413 (too large) or 500.
241 */
242static ngx_int_t
243zip_http_stage_deflate(ngx_http_request_t *r, brix_vfs_file_t *fh,
244 const brix_zip_member_t *m, const zip_http_span_t *sp, ngx_buf_t *b)
245{
246 u_char *mem;
247
248 if (m->uncomp_size > ZIP_HTTP_MEM_MAX) {
249 (void) brix_vfs_close(fh, r->connection->log);
250 return NGX_HTTP_REQUEST_ENTITY_TOO_LARGE;
251 }
252 mem = ngx_palloc(r->pool, m->uncomp_size ? m->uncomp_size : 1);
253 if (mem == NULL
254 || brix_zip_extract_full(brix_vfs_file_fd(fh), m, mem,
255 m->uncomp_size) < 0)
256 {
257 (void) brix_vfs_close(fh, r->connection->log);
258 return NGX_HTTP_INTERNAL_SERVER_ERROR;
259 }
260 (void) brix_vfs_close(fh, r->connection->log);
261 b->memory = 1;
262 b->pos = mem + sp->start;
263 b->last = mem + sp->start + sp->len;
264 return NGX_OK;
265}

267/*
268 * WHAT: fill the response headers for the (possibly partial) member body.
269 * WHY: 200 vs 206 + Content-Range depend only on the served span, so the
270 * header block is a pure function of (r, span).
271 * HOW: sets status/content_length/content_type; on a partial span appends a
272 * "Content-Range: bytes a-b/total" header (allocation failures leave
273 * the header out, matching the original best-effort behaviour).
274 */
275static void
276zip_http_set_headers(ngx_http_request_t *r, const zip_http_span_t *sp)
277{
278 r->headers_out.status = (sp->len == sp->total)
279 ? NGX_HTTP_OK
280 : NGX_HTTP_PARTIAL_CONTENT;
281 r->headers_out.content_length_n = sp->len;
282 r->allow_ranges = 1;
283 ngx_str_set(&r->headers_out.content_type, "application/octet-stream");
284 r->headers_out.content_type_len = r->headers_out.content_type.len;
285
286 if (sp->len != sp->total) {
287 ngx_table_elt_t *cr = ngx_list_push(&r->headers_out.headers);
288 if (cr != NULL) {
289 cr->hash = 1;
290 ngx_str_set(&cr->key, "Content-Range");
291 cr->value.data = ngx_pnalloc(r->pool,
292 sizeof("bytes -/-") + 3 * NGX_OFF_T_LEN);
293 if (cr->value.data != NULL) {
294 cr->value.len = ngx_sprintf(cr->value.data, "bytes %O-%O/%O",
295 sp->start,
296 sp->start + sp->len - 1,
297 sp->total)
298 - cr->value.data;
299 }
300 }
301 }

304ngx_int_t
305brix_zip_http_serve(ngx_http_request_t *r,
306 const brix_vfs_ctx_t *vfs_scope, size_t cd_max,
307 const char *archive_full, const char *member)
308{
309 brix_zip_member_t m;
310 zip_http_span_t sp;
311 ngx_buf_t *b;
312 ngx_int_t rc;
313 brix_vfs_file_t *fh = NULL;
314 off_t archive_size = 0;
315
316 rc = zip_http_open_archive(r, vfs_scope, archive_full, &fh,
317 &archive_size);
318 if (rc != NGX_OK) {
319 return rc;
320 }
321
322 if (cd_max == 0) {
323 cd_max = 16 * 1024 * 1024;
324 }
325 rc = zip_http_locate_member(fh, archive_size, member, cd_max, &m);
326 if (rc != NGX_OK) {
327 (void) brix_vfs_close(fh, r->connection->log);
328 return rc;
329 }
330
331 /* Single byte-range against the UNCOMPRESSED length. */
332 sp.total = (off_t) m.uncomp_size;
333 sp.start = 0;
334 sp.len = sp.total;
335 if (zip_http_parse_range(r, &sp) != NGX_OK) {
336 (void) brix_vfs_close(fh, r->connection->log);
337 r->headers_out.status = NGX_HTTP_RANGE_NOT_SATISFIABLE;
338 r->headers_out.content_length_n = 0;
339 ngx_http_send_header(r);
340 return ngx_http_send_special(r, NGX_HTTP_LAST);
341 }
342
343 b = ngx_pcalloc(r->pool, sizeof(ngx_buf_t));
344 if (b == NULL) {
345 (void) brix_vfs_close(fh, r->connection->log);
346 return NGX_HTTP_INTERNAL_SERVER_ERROR;
347 }
348
349 rc = (m.method == BRIX_ZIP_METHOD_STORE)
350 ? zip_http_stage_stored(r, fh, &m, &sp, b)
351 : zip_http_stage_deflate(r, fh, &m, &sp, b);
352 if (rc != NGX_OK) {
353 return rc;
354 }
355 b->last_buf = 1;
356 b->last_in_chain = 1;
357
358 zip_http_set_headers(r, &sp);
359
360 return brix_http_send_single_buf(r, b);
361}
