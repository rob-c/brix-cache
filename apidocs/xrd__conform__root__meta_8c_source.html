7/* WHAT: Grade a boolean assertion with stable evidence. WHY: Metadata rows use

10grade(xrd_conform_execution *execution, const char *id, int success,
11 const char *pass_reason, const char *fail_reason)
12{
13 xrd_conform_result_set(execution, id,

19/* WHAT: Reserve one metadata exchange or mark its rows inconclusive. WHY:
20 * Metadata operations are real requests and must obey the same hard ceiling.
21 * HOW: Charge zero body bytes and copy the stable ceiling reason to each row. */
22static int
23reserve_stat(xrd_conform_execution *execution, const char *const *ids,
24 size_t count)
25{
26 char reason[64];
27 size_t index;
28
29 if (xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return 1;
30 for (index = 0; index < count; index++) {
31 xrd_conform_result_set(execution, ids[index],
32 BRIX_CONFORM_VERDICT_INCONCLUSIVE, reason, NULL, NULL);
33 }
34 return 0;
35}

37static int
38parse_stat_body(const uint8_t *body, uint32_t body_len, brix_statinfo *out)
39{
40 brix_statline_ext ext;
41 char text[256];
42
43 if (body == NULL || body_len == 0 || body_len >= sizeof(text)) return -1;
44 memcpy(text, body, body_len);
45 text[body_len] = '\0';
46 if (brix_statline_parse(text, &out->id, &out->size, &out->flags,
47 &out->mtime, &ext) != 0) return -1;
48 return 0;
49}

51static int
52stat_handle(brix_conn *connection, const brix_file *file,
53 brix_statinfo *out, brix_status *status)
54{
55 ClientStatRequest request;
56 uint8_t *body = NULL;
57 uint32_t body_len = 0;
58 uint16_t wire_status = 0;
59 int rc;
60
61 memset(&request, 0, sizeof(request));
62 request.requestid = htons(kXR_stat);
63 memcpy(request.fhandle, file->fhandle, sizeof(request.fhandle));
64 rc = brix_roundtrip_resilient(connection, &request, NULL, 0,
65 XRDC_OP_READONLY, 0, &wire_status, &body, &body_len, status);
66 if (rc == 0) rc = parse_stat_body(body, body_len, out);
67 free(body);
68 return rc;
69}

71static int
72statx_pair(brix_conn *connection, const char *path, uint8_t flags[2],
73 brix_status *status)
74{
75 ClientStatxRequest request;
76 uint8_t *body = NULL;
77 uint32_t body_len = 0;
78 uint16_t wire_status = 0;
79 char payload[XRDC_PATH_MAX + 3];
80 int length;
81 int rc;
82
83 length = snprintf(payload, sizeof(payload), "%s\n/", path);
84 if (length < 0 || (size_t) length >= sizeof(payload)) return -1;
85 memset(&request, 0, sizeof(request));
86 request.requestid = htons(kXR_statx);
87 rc = brix_roundtrip_resilient(connection, &request, payload,
88 (uint32_t) length, XRDC_OP_READONLY, 0, &wire_status, &body,
89 &body_len, status);
90 if (rc == 0 && body_len == 2) memcpy(flags, body, 2);
91 else rc = -1;
92 free(body);
93 return rc;
94}

96static void
97probe_stat_boundaries(xrd_conform_execution *execution,
98 brix_conn *connection, const char *path, const brix_statinfo *baseline)
99{
100 static const unsigned char controls[] = { '\n', '\t', 0x7f };
101 char test_path[XRDC_PATH_MAX];
102 char long_path[4097];
103 brix_statinfo info;
104 brix_status status;
105 char reason[64];
106 size_t index;
107 int safe = 1;
108 int exercised = 1;
109
110 for (index = 0; index < sizeof(controls); index++) {
111 size_t path_len = strlen(path);
112
113 if (path_len + 16 >= sizeof(test_path)) {
114 exercised = 0;
115 break;
116 }
117 memcpy(test_path, path, path_len);
118 test_path[path_len] = (char) controls[index];
119 memcpy(test_path + path_len + 1, "brix-control", 13);
120 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) {
121 exercised = 0;
122 break;
123 }
124 brix_status_clear(&status);
125 if (brix_stat(connection, test_path, &info, &status) == 0
126 && info.id == baseline->id) safe = 0;
127 }
128 grade(execution, "STA-017", exercised && safe,
129 "control-path-answered-or-refused-without-truncation",
130 exercised ? "control-path-silently-truncated" : "probe-budget-exhausted");
131
132 memset(long_path, 'a', sizeof(long_path) - 1);
133 long_path[0] = '/';
134 long_path[sizeof(long_path) - 1] = '\0';
135 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return;
136 brix_status_clear(&status);
137 {
138 int refused = brix_stat(connection, long_path, &info, &status) != 0;
139 int pass = refused && status.kxr == kXR_ArgTooLong;
140
141 xrd_conform_result_set(execution, "STA-018",
142 pass ? BRIX_CONFORM_VERDICT_CONFORMANT
143 : BRIX_CONFORM_VERDICT_DIVERGENT,
144 pass ? "overlong-stat-kXR_ArgTooLong"
145 : "overlong-stat-wrong-answer",
146 "kXR_ArgTooLong", status.msg);
147 }
148}

150static void
151probe_stat_forms(xrd_conform_execution *execution, brix_conn *connection,
152 const brix_file *file, const char *path, const brix_statinfo *path_stat)
153{
154 static const char *const statx_ids[] = { "STA-015", "STA-016" };
155 brix_statinfo handle_stat;
156 brix_status status;
157 char vfs[256];
158 char trailing;
159 char reason[64];
160 long long nodes_rw, free_rw, util_rw, nodes_stage, free_stage, util_stage;
161 uint8_t statx_flags[2];
162 int handle_ok;
163 int vfs_fields;
164
165 if (xrd_conform_reserve(execution, 0, reason, sizeof(reason))) {
166 brix_status_clear(&status);
167 handle_ok = stat_handle(connection, file, &handle_stat, &status) == 0;
168 grade(execution, "STA-008", handle_ok
169 && handle_stat.id == path_stat->id
170 && handle_stat.size == path_stat->size
171 && handle_stat.flags == path_stat->flags
172 && handle_stat.mtime == path_stat->mtime,
173 "handle-stat-matches-path-stat", "handle-stat-differs-from-path-stat");
174 }
175
176 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return;
177 brix_status_clear(&status);
178 memset(vfs, 0, sizeof(vfs));
179 if (brix_statvfs(connection, "/", vfs, sizeof(vfs), &status) != 0) {
180 grade(execution, "STA-013", 0, "vfs-stat-six-field-form",
181 "vfs-stat-request-failed");
182 return;
183 }
184 vfs_fields = sscanf(vfs, "%lld %lld %lld %lld %lld %lld %c",
185 &nodes_rw, &free_rw, &util_rw, &nodes_stage, &free_stage,
186 &util_stage, &trailing);
187 grade(execution, "STA-013", vfs_fields == 6,
188 "vfs-stat-six-field-form", "vfs-stat-file-form-or-invalid");
189
190 if (!reserve_stat(execution, statx_ids,
191 sizeof(statx_ids) / sizeof(statx_ids[0]))) return;
192 brix_status_clear(&status);
193 handle_ok = statx_pair(connection, path, statx_flags, &status) == 0;
194 grade(execution, "STA-015", handle_ok
195 && statx_flags[0] == (path_stat->flags & (kXR_isDir | kXR_offline))
196 && (statx_flags[1] & kXR_isDir) != 0,
197 "statx-one-ordered-flag-byte-per-path",
198 "statx-shape-or-order-mismatch");
199 grade(execution, "STA-016", handle_ok
200 && statx_flags[0] == (path_stat->flags & (kXR_isDir | kXR_offline)),
201 "statx-flags-agree-with-stat", "statx-flags-disagree-with-stat");
202}

245/* WHAT: Exercise stat size/type/time/stability and missing-path mapping.
246 * WHY: These assertions are fully observable through the typed metadata API.
247 * HOW: Stat twice, compare learned body size, then stat a keyed absent sibling. */
248void
249xrd_conform_probe_stat(xrd_conform_execution *execution,
250 brix_conn *connection, const brix_file *file, const char *path,
251 uint64_t size)
252{
253 static const char *const first_ids[] = {
254 "STA-001", "STA-002", "STA-003", "STA-009", "STA-010", "STA-014"
255 };
256 static const char *const second_ids[] = { "STA-014" };
257 brix_statinfo first, second, root;
258 brix_status status;
259 time_t now = time(NULL);
260 int first_ok;
261 int second_ok;
262 int second_reserved;
263 int root_reserved;
264 int root_ok = 0;
265
266 brix_status_clear(&status);
267 if (!reserve_stat(execution, first_ids,
268 sizeof(first_ids) / sizeof(first_ids[0]))) return;
269 first_ok = brix_stat(connection, path, &first, &status) == 0;
270 brix_status_clear(&status);
271 second_reserved = reserve_stat(execution, second_ids,
272 sizeof(second_ids) / sizeof(second_ids[0]));
273 second_ok = second_reserved
274 && brix_stat(connection, path, &second, &status) == 0;
275 grade_primary_stat(execution, &first, first_ok, size, now);
276 root_reserved = reserve_stat(execution, second_ids,
277 sizeof(second_ids) / sizeof(second_ids[0]));
278 if (root_reserved) {
279 brix_status_clear(&status);
280 root_ok = brix_stat(connection, "/", &root, &status) == 0;
281 }
282 if (second_reserved && root_reserved) {
283 grade(execution, "STA-014", first_ok && second_ok
284 && first.id == second.id
285 && root_ok && root.id != first.id,
286 "stat-id-stable-and-distinct", "stat-id-not-stable-or-distinct");
287 }
288 if (first_ok) {
289 probe_stat_forms(execution, connection, file, path, &first);
290 probe_stat_boundaries(execution, connection, path, &first);
291 }
292 probe_missing_stat(execution, connection, path, &first);
293}
