20int
21xvfs_pread_full(brix_sd_obj_t *obj, void *buf, size_t len, off_t off,
22 size_t *nread)
23{
24 size_t done = 0;
25 u_char *p = buf;
26
27 while (done < len) {
28 ssize_t n = obj->driver->pread(obj, p + done, len - done,
29 off + (off_t) done);
30 if (n < 0) {
31 if (errno == EINTR) {
32 continue;
33 }
34 if (nread != NULL) {
35 *nread = done;
36 }
37 return -1;
38 }
39 if (n == 0) {
40 break; /* EOF — short read is success */
41 }
42 done += (size_t) n;
43 }
44
45 if (nread != NULL) {
46 *nread = done;
47 }
48 return 0;
49}

63int
64xvfs_pwrite_full(brix_sd_obj_t *obj, const void *buf, size_t len, off_t off,
65 size_t *written, int *short_io)
66{
67 size_t done = 0;
68 const u_char *p = buf;
69
70 if (short_io != NULL) {
71 *short_io = 0;
72 }
73
74 while (done < len) {
75 ssize_t n = obj->driver->pwrite(obj, p + done, len - done,
76 off + (off_t) done);
77 if (n < 0) {
78 if (errno == EINTR) {
79 continue;
80 }
81 if (done > 0 && short_io != NULL) {
82 *short_io = 1; /* partial-then-error */
83 }
84 if (written != NULL) {
85 *written = done;
86 }
87 return -1;
88 }
89 if (n == 0) {
90 if (short_io != NULL) {
91 *short_io = 1;
92 }
93 if (written != NULL) {
94 *written = done;
95 }
96 errno = EIO;
97 return -1;
98 }
99 done += (size_t) n;
100 }
101
102 if (written != NULL) {
103 *written = done;
104 }
105 return 0;
106}

127/* How much of `buf` the next step of `win` may use: the rest of a bounded
128 * window, or the whole buffer when the window runs to EOF. */
129static size_t
130xvfs_window_step(const xvfs_window_t *win, size_t bufsz, off_t off)
131{
132 off_t left;
133
134 if (win->len < 0) {
135 return bufsz;
136 }
137 left = win->off + win->len - off;
138 return (left < (off_t) bufsz) ? (size_t) left : bufsz;
139}

141int
142xvfs_drain_window(brix_sd_obj_t *src, brix_sd_obj_t *dst, void *buf,
143 size_t bufsz, xvfs_window_t *win)
144{
145 off_t off;
146
147 if (buf == NULL || bufsz == 0 || win == NULL || win->off < 0) {
148 errno = EINVAL;
149 return -1;
150 }
151
152 for (off = win->off; win->len < 0 || off < win->off + win->len; ) {
153 size_t want = xvfs_window_step(win, bufsz, off);
154 ssize_t r = xvfs_pread_once(src, buf, want, off);
155
156 if (r < 0) {
157 return -1; /* read error (errno set by driver) */
158 }
159 if (r == 0) {
160 if (win->len >= 0) {
161 errno = EIO; /* the window was not all there */
162 return -1;
163 }
164 break; /* EOF — whole object copied */
165 }
166 if (xvfs_pwrite_full(dst, buf, (size_t) r, off, NULL, NULL) != 0) {
167 return -1; /* write error (errno set) */
168 }
169 off += r;
170 }
171
172 win->end = off;
173 return 0;
174}

192xvfs_stage_fd(int src_fd, const char *stage_dir)
193{
194 char tmpl[PATH_MAX];
195 char *buf;
196 brix_sd_obj_t s, d;
197 int dst_fd, rd_fd, e, n;
198
199 if (stage_dir == NULL || stage_dir[0] == '\0') {
200 errno = EINVAL;
201 return -1;
202 }

204 if (n < 0 || (size_t) n >= sizeof(tmpl)) {
205 errno = ENAMETOOLONG;
206 return -1;
207 }
208 dst_fd = mkstemp(tmpl);
209 if (dst_fd < 0) {
210 return -1;
211 }
212 (void) unlink(tmpl); /* anonymous: bytes live behind the fd(s) */
213
214 buf = malloc(BRIX_VFS_CORE_BUF_SIZE);
215 if (buf == NULL) {
216 e = ENOMEM;
217 close(dst_fd);
218 errno = e;
219 return -1;
220 }
221 brix_sd_posix_wrap(&s, src_fd);
222 brix_sd_posix_wrap(&d, dst_fd);
223 if (xvfs_drain(&s, &d, buf, BRIX_VFS_CORE_BUF_SIZE, NULL) != 0) {
224 e = errno;
225 free(buf);
226 close(dst_fd);
227 errno = e;
228 return -1;
229 }

231

233 * aliases the open file (/proc/self/fd on Linux, F_GETPATH or dup on
234 * Darwin). */
235 rd_fd = brix_plat_fd_reopen_readonly(dst_fd);
236 e = errno;

240 return -1;
241 }
