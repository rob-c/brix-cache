48/* brix_path_resolved_to_pfn -- translate one confined VFS path for a driver.
49 *
50 * WHAT: Strips the export root from `resolved_path`, then translates that LFN
51 * through the ctx-bound N2N configuration into `pfn`. Returns NGX_OK or
52 * NGX_ERROR with errno set by the translator.
53 * WHY: Confinement is defined over the logical export path, while storage
54 * drivers address physical keys. One adapter fixes the security-relevant
55 * ordering and prevents callers from accidentally prefixing root_canon.
56 * HOW: Validate the inputs, derive the export-relative LFN with the existing
57 * VFS helper, then delegate all composition to brix_path_lfn_to_pfn(). */
58ngx_int_t
59brix_path_resolved_to_pfn(const brix_vfs_ctx_t *ctx,
60 const char *resolved_path, char *pfn, size_t cap)
61{
62 const char *lfn;
63
64 if (ctx == NULL || resolved_path == NULL || pfn == NULL || cap == 0) {
65 errno = EINVAL;
66 return NGX_ERROR;
67 }
68 lfn = brix_vfs_export_relative(ctx, resolved_path);
69 return brix_path_lfn_to_pfn(ctx, lfn, pfn, cap);
70}
