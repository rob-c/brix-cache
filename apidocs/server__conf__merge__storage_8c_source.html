21 * WHAT: merge the compression + ZIP-access group and the write-through staging
22 * roots/backends + backpressure watermarks + dirty-reaper age + cache
23 * allow/deny prefix inheritance.
24 * WHY: the staging LOW watermark auto-derives from the just-merged HIGH; keeping

26 * HOW: inherit the compression/ZIP scalars, then the staging roots and the

95/*
96 * WHAT: merge the read-through cache origin + eviction group — origin address/
97 * TLS/family, lock timeout, the on-fill eviction threshold, the watermark
98 * reaper, the file-size/memory budgets, and the readv segment size.
99 * WHY: the reaper HIGH/LOW watermarks derive from the just-merged eviction
100 * threshold; grouping keeps that dependency chain visible.
101 * HOW: inherit the origin scalars, then the reaper watermarks (HIGH ← eviction
102 * threshold, LOW ← HIGH − 5%) and the budget/segment defaults.
103 */
104static void
105brix_merge_srv_cache_origin(ngx_stream_brix_srv_conf_t *conf,
106 ngx_stream_brix_srv_conf_t *prev)
107{
108 ngx_conf_merge_str_value(conf->cache_origin, prev->cache_origin, "");
109 ngx_conf_merge_value(conf->cache_origin_tls, prev->cache_origin_tls, 0);
110 ngx_conf_merge_uint_value(conf->cache_origin_family,
111 prev->cache_origin_family, BRIX_AF_AUTO);
112 ngx_conf_merge_value(conf->cache_lock_timeout,
113 prev->cache_lock_timeout, 300);
114 ngx_conf_merge_uint_value(conf->cache_eviction_threshold,
115 prev->cache_eviction_threshold, 900000);
116
117 /* Phase-88 loose end: brix_cache_evict_at/_to (tier grammar, PERCENT) are
118 * an alternate spelling of the watermark pair — an explicitly-set pair
119 * seeds the reaper (percent -> ppm); the ppm-native brix_cache_high/
120 * low_watermark directives win when both are given. A lone evict_at takes
121 * the documented 80% target, a lone evict_to the 90% trigger. Percent
122 * range/order is validated in runtime_server.c BEFORE the ppm check so a
123 * unit mistake gets the percent-worded error. */
124 if (conf->common.cache_evict_at != NGX_CONF_UNSET_UINT
125 || conf->common.cache_evict_to != NGX_CONF_UNSET_UINT)
126 {
127 ngx_conf_merge_uint_value(conf->reaper.high_watermark,
128 prev->reaper.high_watermark,
129 (conf->common.cache_evict_at == NGX_CONF_UNSET_UINT
130 ? 90 : conf->common.cache_evict_at) * 10000);
131 ngx_conf_merge_uint_value(conf->reaper.low_watermark,
132 prev->reaper.low_watermark,
133 (conf->common.cache_evict_to == NGX_CONF_UNSET_UINT
134 ? 80 : conf->common.cache_evict_to) * 10000);
135 }
136
137 /* Watermark reaper: HIGH defaults to the on-fill eviction threshold so an
138 * existing config keeps its bound; LOW defaults 50000 ppm (5%) below HIGH for
139 * hysteresis; the timer runs every 60s by default. The ordering invariant
140 * (0 < low < high < 1e6) is enforced in runtime_server.c. */
141 ngx_conf_merge_uint_value(conf->reaper.high_watermark,
142 prev->reaper.high_watermark,
143 conf->cache_eviction_threshold);
144 ngx_conf_merge_uint_value(conf->reaper.low_watermark,
145 prev->reaper.low_watermark,
146 conf->reaper.high_watermark > 50000
147 ? conf->reaper.high_watermark - 50000
148 : conf->reaper.high_watermark / 2);
149 ngx_conf_merge_sec_value(conf->reaper.reap_interval,
150 prev->reaper.reap_interval, 60);
151 ngx_conf_merge_off_value(conf->reaper.max_bytes, prev->reaper.max_bytes, 0);
152 ngx_conf_merge_off_value(conf->cache_max_file_size,
153 prev->cache_max_file_size, 0);
154 ngx_conf_merge_off_value(conf->memory_budget,
155 prev->memory_budget, 768 * 1024 * 1024);
156 /* Default = stock XRootD maxReadv_ior = maxBuffsz(2 MiB) - sizeof(readahead_list). */
157 ngx_conf_merge_size_value(conf->readv_segment_size,
158 prev->readv_segment_size,
159 (size_t) (2 * 1024 * 1024) - BRIX_READV_SEGSIZE);
160}

162/*
163 * WHAT: merge the io_uring backend group, the checksum-on-fill verify mode, the
164 * Pelican advertisement group, and inherit the compiled include-regex.
165 * WHY: the advertisement interval is clamped to the federation minimum and the
166 * regex inheritance is a struct-copy neither expressible via
167 * ngx_conf_merge_*; grouping keeps both special cases together.
168 * HOW: inherit the io_uring/verify/advertise scalars, floor the advertise
169 * interval at 60s, and copy the parent's compiled regex when unset.
170 */
171static void
172brix_merge_srv_iouring_advertise(ngx_stream_brix_srv_conf_t *conf,
173 ngx_stream_brix_srv_conf_t *prev)
174{
175 /* Phase 44: optional io_uring backend — default OFF (strictly opt-in).
176 *
177 * WHY not AUTO: the startup probe can only prove the ring accepts opcodes
178 * and (SB-W hardening) that the registered eventfd delivers a NOP
179 * completion — neither of which proves real buffered file WRITES complete
180 * on THIS kernel + filesystem. On at least one production host (EL9 elrepo
181 * 6.15, plain local fs) NOPs drained fine yet io_uring writes never
182 * completed: transfers wedged after exactly queue_depth in-flight ops
183 * (queue_depth x 32 KiB = 8 MiB) with a worker spinning, and a torn-down
184 * connection's still-in-flight ops became a late-CQE use-after-free. The
185 * thread pool is both correct there and FASTER (50 vs stall). io_uring is
186 * a performance option, not a correctness feature, so it must be an
187 * explicit, operator-verified `brix_io_uring on` — never silently engaged.
188 * `on` still fail-fasts if the backend is unavailable; `auto` remains for
189 * anyone who wants best-effort enable. */
190 ngx_conf_merge_uint_value(conf->io_uring,
191 prev->io_uring, BRIX_IO_URING_OFF);
192 /* D-3: seccomp filter default OFF — strictly opt-in (audit-first rollout).
193 * An empirical flip to ENFORCE (2026-07) confirmed it must stay opt-in: the
194 * worker deny-set KILLs execve, and a few EXTERNAL-helper features fork+exec:
195 * the FRM "exec" MSS adapter ($BRIX_FRM_STAGECMD, a real HSM) / OIDC token
196 * fetch / native-TPC token-exchange / the kXR_prepare hook, so a blanket
197 * enforce default would break them. (The DEFAULT tape/nearline backend uses
198 * the built-in POSIX stub adapter — recall/migrate/purge are plain file copies,
199 * no exec — so it is fine under strict enforce.) Sites that use those now
200 * set
201 * `brix_seccomp_allow_exec on` to allowlist execve under enforce (ptrace/
202 * process_vm_* stay killed); the xattr allowlist gap is fixed; and HTTP-only
203 * (WebDAV/S3) workers are filtered too (the install runs from the WebDAV
204 * init_process when there is no stream{} block). So opt-in `brix_seccomp
205 * enforce` (+ allow_exec when needed) is viable for every deployment. */
206 ngx_conf_merge_uint_value(conf->seccomp,
207 prev->seccomp, BRIX_SECCOMP_OFF);
208 /* E-4: negative-path backoff default OFF (threshold 0) — availability-first,
209 * strictly opt-in. window_ms/backoff_s only matter when threshold > 0. */
210 ngx_conf_merge_uint_value(conf->negcache.threshold,
211 prev->negcache.threshold, 0);
212 ngx_conf_merge_uint_value(conf->negcache.window_ms,
213 prev->negcache.window_ms, 0);
214 ngx_conf_merge_uint_value(conf->negcache.backoff_s,
215 prev->negcache.backoff_s, 0);
216 ngx_conf_merge_value(conf->io_uring_queue_depth,
217 prev->io_uring_queue_depth,
218 BRIX_IO_URING_QUEUE_DEPTH);
219 ngx_conf_merge_str_value(conf->io_uring_panic_file,
220 prev->io_uring_panic_file, "");
221 ngx_conf_merge_value(conf->io_uring_admin, prev->io_uring_admin, 0);
222 ngx_conf_merge_value(conf->io_uring_restrict, prev->io_uring_restrict, 1);
223
224
225 /* Checksum-on-fill is merged with the rest of the shared preamble
226 * (common.cache_verify_mode / common.cache_verify_digest); the standalone
227 * (brix_cache) spine reads it through brix_cache_verify_effective(). */
228
229 /* Pelican cache advertisement (default off; interval clamped to the
230 * federation minimum of 60s = MinFedTokenTickerRate). */
231 ngx_conf_merge_value(conf->advertise.enable, prev->advertise.enable, 0);
232 ngx_conf_merge_msec_value(conf->advertise.interval,
233 prev->advertise.interval, BRIX_ADVERTISE_INTERVAL_DEFAULT_MS);
234 if (conf->advertise.interval < BRIX_ADVERTISE_INTERVAL_DEFAULT_MS) {
235 conf->advertise.interval = BRIX_ADVERTISE_INTERVAL_DEFAULT_MS;
236 }
237 ngx_conf_merge_str_value(conf->advertise.key,
238 prev->advertise.key, "");
239 ngx_conf_merge_str_value(conf->advertise.data_url, prev->advertise.data_url, "");
240 ngx_conf_merge_str_value(conf->advertise.web_url, prev->advertise.web_url, "");
241 ngx_conf_merge_str_value(conf->advertise.sitename, prev->advertise.sitename, "");
242 ngx_conf_merge_str_value(conf->advertise.issuer_url, prev->advertise.issuer_url, "");
243 /* The federation discovery authority: no default host — an unset federation
244 * leaves the advertiser disarmed; the port defaults to HTTPS. */
245 ngx_conf_merge_str_value(conf->advertise.federation,
246 prev->advertise.federation, "");
247 ngx_conf_merge_uint_value(conf->advertise.federation_port,
248 prev->advertise.federation_port, 443);
249 if (conf->advertise.ns == NULL) {
250 conf->advertise.ns = prev->advertise.ns;
251 }
252
253 /* Durable async backend-op queue: default off; batch 64 ops; coalesce cap
254 * 200ms. When enabled, mutations (unlink/rmdir/rename/mkdir/write-commit) are
255 * journalled + parked until the batch flushes (size OR time, whichever first).
256 * A batch of 0 or a 0ms wait would defeat coalescing, so clamp both up. */
257 ngx_conf_merge_value(conf->backend_async, prev->backend_async, 0);
258 ngx_conf_merge_uint_value(conf->backend_async_batch,
259 prev->backend_async_batch, 64);
260 if (conf->backend_async_batch < 1) {
261 conf->backend_async_batch = 1;
262 }
263 ngx_conf_merge_msec_value(conf->backend_async_wait,
264 prev->backend_async_wait, 200);
265
266 /* Inherit compiled regex from parent if the child didn't set one */
267 if (!conf->include_regex.set && prev->include_regex.set) {
268 conf->include_regex.str = prev->include_regex.str;
269 conf->include_regex.re = prev->include_regex.re;
270 conf->include_regex.set = 1;
271 }
272}

274/* Storage: read/write compression, ZIP access, the read-through cache (origin,
275 * sizing, eviction, slice validation, include-regex inheritance), the memory
276 * budget, readv segment sizing, and the io_uring backend. */
277char *
278brix_merge_srv_storage(ngx_conf_t *cf, ngx_stream_brix_srv_conf_t *conf,
279 ngx_stream_brix_srv_conf_t *prev)
280{
281 /* common.* (storage backend, pblock stripe, tier grammar) is merged by
282 * ngx_http_brix_shared_merge() in brix_merge_srv_security — only the
283 * stream-specific validation below stays here. */
284
285 /* §6.5: the tier slice size must be 0 (off) or a positive multiple of the
286 * 1 MiB cinfo block granule (so a partial fill never records a mis-aligned
287 * block) — the same rule the legacy brix_cache_slice enforced. */
288 if (conf->common.cache_slice_size != 0
289 && (conf->common.cache_slice_size < (1024 * 1024)
290 || (conf->common.cache_slice_size % (1024 * 1024)) != 0))
291 {
292 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
293 "brix_cache_slice_size must be a positive multiple of 1m");
294 return NGX_CONF_ERROR;
295 }
296
297 /* Background block prefetch: the in-flight cap bounds detached thread-pool
298 * jobs per worker — an unbounded value would let one client queue arbitrary
299 * speculative origin traffic. */
300 if (conf->common.cache_prefetch != NGX_CONF_UNSET
301 && (conf->common.cache_prefetch < 0
302 || conf->common.cache_prefetch > 64))
303 {
304 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
305 "brix_cache_prefetch must be between 0 and 64");
306 return NGX_CONF_ERROR;
307 }
308 if (conf->common.cache_prefetch_window != NGX_CONF_UNSET_SIZE
309 && conf->common.cache_prefetch_window != 0
310 && conf->common.cache_prefetch_window < 64 * 1024)
311 {
312 BRIX_CONF_LOG(NGX_LOG_EMERG, cf, 0,
313 "brix_cache_prefetch_window must be 0 or at least 64k");
314 return NGX_CONF_ERROR;
315 }
316
317 brix_merge_srv_zip_stage(conf, prev);
318 brix_merge_srv_cache_origin(conf, prev);
319 brix_merge_srv_iouring_advertise(conf, prev);
320
321 return NGX_CONF_OK;
322}
