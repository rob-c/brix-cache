4/*
5 * brix_check_token_scope — enforce WLCG token scopes on a path operation.
6 *
7 * Only active when ctx->token.auth == 1 (bearer-token session). GSI and
8 * anonymous sessions are not restricted by this function — the brix_allow_write
9 * directive and VO ACL remain their access gate.
10 *
11 * logical_path: the client-facing XRootD path (e.g. "/cms/store/file.root"),
12 * NOT the resolved filesystem path.
13 * need_write: 1 for mutations (write/create/delete/chmod/mv/mkdir/rmdir),
14 * 0 for reads (stat/dirlist/locate/open-read/fattr-get).
15 *
16 * Returns NGX_OK if access is granted, NGX_ERROR if denied (the caller must
17 * send the error response and log the access failure).
18 */
19ngx_int_t
20brix_check_token_scope(brix_ctx_t *ctx, const char *logical_path,
21 int need_write)
22{
23 if (!ctx->token.auth) {
24 return NGX_OK;
25 }
26
27 if (ctx->identity != NULL) {
28 return brix_identity_check_token_scope(ctx->identity, logical_path,
29 need_write);
30 }
31
32 if (need_write) {
33 return brix_token_check_write(ctx->token.scopes,
34 ctx->token.scope_count, logical_path)
35 ? NGX_OK : NGX_ERROR;
36 }
37
38 return brix_token_check_read(ctx->token.scopes,
39 ctx->token.scope_count, logical_path)
40 ? NGX_OK : NGX_ERROR;
41}

44brix_min_sec_enforce(brix_ctx_t *ctx, ngx_connection_t *c,
45 ngx_stream_brix_srv_conf_t *conf)
46{
47 if (conf->min_sec_level == BRIX_MIN_SEC_NONE) {
48 return BRIX_DISPATCH_CONTINUE;
49 }
50
51 /*
52 * compat and intense both require a TLS-encrypted transport. brix_tls only
53 * ADVERTISES kXR_ableTLS; a client is free to finish login/auth in cleartext
54 * and reach here — that is precisely the walked-down session this floor
55 * refuses. We reject every data/metadata opcode (session opcodes were
56 * dispatched before this gate) with kXR_TLSRequired; A-1 pairs the upstream
57 * leg so neither side can be downgraded independently.
58 */
59 if (c->ssl == NULL || c->ssl->connection == NULL) {
60 ngx_log_error(NGX_LOG_WARN, c->log, 0,
61 "brix: cleartext session refused by brix_min_sec_level=%ui (opcode=%d)",
62 conf->min_sec_level, (int) ctx->recv.cur_reqid);
63 return brix_send_error(ctx, c, kXR_TLSRequired,
64 "server security policy requires a TLS-encrypted session");
65 }
66
67 /*
68 * intense additionally requires a non-anonymous identity. An auth=none
69 * listener authenticates nobody (anonymous login sets auth_done=1 with no
70 * credential), so even over TLS it is below the floor.
71 */
72 if (conf->min_sec_level >= BRIX_MIN_SEC_INTENSE
73 && conf->auth == BRIX_AUTH_NONE)
74 {
75 ngx_log_error(NGX_LOG_WARN, c->log, 0,
76 "brix: anonymous session refused by brix_min_sec_level=intense (opcode=%d)",
77 (int) ctx->recv.cur_reqid);
78 return brix_send_error(ctx, c, kXR_NotAuthorized,
79 "server security policy requires an authenticated identity");
80 }
81
82 return BRIX_DISPATCH_CONTINUE;
83}

85/* tls_require_opcode_caps — capability set an opcode exercises (vfs_secgate
86 * mask bits). login/auth → LOGIN; the read/write families → SESSION|DATA;
87 * the handshake/session-control opcodes (protocol/bind/ping/endsess/set/
88 * sigver) → 0 so the in-protocol TLS upgrade path is never blocked by the
89 * very policy that demands it; everything else (stat/dirlist/open/mkdir/…)
90 * → SESSION. TPC is gated at its own choke point (read/open_tpc.c) where
91 * the tpc.* opaque keys are actually parsed. */
92static ngx_uint_t
93tls_require_opcode_caps(ngx_uint_t reqid)
94{
95 switch (reqid) {
96 case kXR_login:
97 case kXR_auth:
98 return BRIX_TLSREQ_LOGIN;
99 case kXR_read:
100 case kXR_readv:
101 case kXR_pgread:
102 case kXR_write:
103 case kXR_writev:
104 case kXR_pgwrite:
105 return BRIX_TLSREQ_SESSION | BRIX_TLSREQ_DATA;
106 case kXR_protocol:
107 case kXR_bind:
108 case kXR_ping:
109 case kXR_endsess:
110 case kXR_set:
111 case kXR_sigver:
112 return 0;
113 default:
114 return BRIX_TLSREQ_SESSION;
115 }
116}

118ngx_int_t
119brix_tls_require_enforce(brix_ctx_t *ctx, ngx_connection_t *c,
120 ngx_stream_brix_srv_conf_t *conf)
121{
122 ngx_uint_t refused, is_tls;
123
124 if (conf->common.tls_require == 0) {
125 return BRIX_DISPATCH_CONTINUE;
126 }
127
128 is_tls = (c->ssl != NULL && c->ssl->connection != NULL);
129 refused = brix_tls_gate_refused(conf->common.tls_require,
130 tls_require_opcode_caps(
131 ctx->recv.cur_reqid),
132 is_tls);
133 if (refused == 0) {
134 return BRIX_DISPATCH_CONTINUE;
135 }
136
137 ngx_log_error(NGX_LOG_WARN, c->log, 0,
138 "brix: cleartext request refused by brix_tls_require %s "
139 "(opcode=%d)",
140 brix_tls_cap_name(refused), (int) ctx->recv.cur_reqid);
141 return brix_send_error(ctx, c, kXR_TLSRequired,
142 "server security policy requires TLS for this capability");
143}

167ngx_int_t
168brix_dispatch_require_write(brix_ctx_t *ctx, ngx_connection_t *c,
169 ngx_stream_brix_srv_conf_t *conf)
170{
171 ngx_int_t rc;
172
173 rc = brix_dispatch_require_auth(ctx, c);
174 if (rc != BRIX_DISPATCH_CONTINUE) {
175 return rc;
176 }
177
178 /* Phase 94: a bound secondary may carry a plain kXR_write data payload for a
179 * primary-published writable handle (parallel upload) — but nothing else
180 * (no open/close/mkdir/mv/rm/writev/pgwrite): those stay primary-only so a
181 * secondary can never mutate the namespace or the handle table. */
182 if (ctx->is_bound && ctx->recv.cur_reqid != kXR_write) {
183 return brix_send_error(ctx, c, kXR_NotAuthorized,
184 "bound streams may only read or write primary "
185 "handles");
186 }
187
188 if (!conf->common.allow_write) {
189 return brix_send_error(ctx, c, kXR_fsReadOnly,
190 "this is a read-only server");
191 }
192
193 return BRIX_DISPATCH_CONTINUE;
