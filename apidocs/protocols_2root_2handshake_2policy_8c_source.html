8/*
9 * brix_check_token_scope — enforce WLCG token scopes on a path operation.
10 *
11 * Only active when ctx->token.auth == 1 (bearer-token session). GSI and
12 * anonymous sessions are not restricted by this function — the brix_allow_write
13 * directive and VO ACL remain their access gate.
14 *
15 * logical_path: the client-facing XRootD path (e.g. "/cms/store/file.root"),
16 * NOT the resolved filesystem path.
17 * need_write: 1 for mutations (write/create/delete/chmod/mv/mkdir/rmdir),
18 * 0 for reads (stat/dirlist/locate/open-read/fattr-get).
19 *
20 * Returns NGX_OK if access is granted, NGX_ERROR if denied (the caller must
21 * send the error response and log the access failure).
22 */
23ngx_int_t
24brix_check_token_scope(brix_ctx_t *ctx, const char *logical_path,
25 int need_write)
26{
27 if (!ctx->token.auth) {
28 return NGX_OK;
29 }
30
31 if (ctx->identity != NULL) {
32 return brix_identity_check_token_scope(ctx->identity, logical_path,
33 need_write);
34 }
35
36 if (need_write) {
37 return brix_token_check_write(ctx->token.scopes,
38 ctx->token.scope_count, logical_path)
39 ? NGX_OK : NGX_ERROR;
40 }
41
42 return brix_token_check_read(ctx->token.scopes,
43 ctx->token.scope_count, logical_path)
44 ? NGX_OK : NGX_ERROR;
45}

48brix_min_sec_enforce(brix_ctx_t *ctx, ngx_connection_t *c,
49 ngx_stream_brix_srv_conf_t *conf)
50{
51 if (conf->min_sec_level == BRIX_MIN_SEC_NONE) {
52 return BRIX_DISPATCH_CONTINUE;
53 }
54
55 /*
56 * compat and intense both require a TLS-encrypted transport. brix_tls only
57 * ADVERTISES kXR_ableTLS; a client is free to finish login/auth in cleartext
58 * and reach here — that is precisely the walked-down session this floor
59 * refuses. We reject every data/metadata opcode (session opcodes were
60 * dispatched before this gate) with kXR_TLSRequired; A-1 pairs the upstream
61 * leg so neither side can be downgraded independently.
62 */
63 if (c->ssl == NULL || c->ssl->connection == NULL) {
64 BRIX_LOG(NGX_LOG_WARN, c->log, 0,
65 "brix: cleartext session refused by brix_min_sec_level=%ui (opcode=%d)",
66 conf->min_sec_level, (int) ctx->recv.cur_reqid);
67 return brix_send_error(ctx, c, kXR_TLSRequired,
68 "server security policy requires a TLS-encrypted session");
69 }
70
71 /*
72 * intense additionally requires a non-anonymous identity. An auth=none
73 * listener authenticates nobody (anonymous login sets auth_done=1 with no
74 * credential), so even over TLS it is below the floor.
75 */
76 if (conf->min_sec_level >= BRIX_MIN_SEC_INTENSE
77 && conf->auth == BRIX_AUTH_NONE)
78 {
79 BRIX_LOG(NGX_LOG_WARN, c->log, 0,
80 "brix: anonymous session refused by brix_min_sec_level=intense (opcode=%d)",
81 (int) ctx->recv.cur_reqid);
82 return brix_send_error(ctx, c, kXR_NotAuthorized,
83 "server security policy requires an authenticated identity");
84 }
85
86 return BRIX_DISPATCH_CONTINUE;
87}

89/* tls_require_opcode_caps — capability set an opcode exercises (vfs_secgate
90 * mask bits). login/auth → LOGIN; the read/write families → SESSION|DATA;
91 * the handshake/session-control opcodes (protocol/bind/ping/endsess/set/
92 * sigver) → 0 so the in-protocol TLS upgrade path is never blocked by the
93 * very policy that demands it; everything else (stat/dirlist/open/mkdir/…)
94 * → SESSION. TPC is gated at its own choke point (read/open_tpc.c) where
95 * the tpc.* opaque keys are actually parsed. */
96static ngx_uint_t
97tls_require_opcode_caps(ngx_uint_t reqid)
98{
99 switch (reqid) {
100 case kXR_login:
101 case kXR_auth:
102 return BRIX_TLSREQ_LOGIN;
103 case kXR_read:
104 case kXR_readv:
105 case kXR_pgread:
106 case kXR_write:
107 case kXR_writev:
108 case kXR_pgwrite:
109 return BRIX_TLSREQ_SESSION | BRIX_TLSREQ_DATA;
110 case kXR_protocol:
111 case kXR_bind:
112 case kXR_ping:
113 case kXR_endsess:
114 case kXR_set:
115 case kXR_sigver:
116 return 0;
117 default:
118 return BRIX_TLSREQ_SESSION;
119 }
120}

123brix_tls_require_enforce(brix_ctx_t *ctx, ngx_connection_t *c,
124 ngx_stream_brix_srv_conf_t *conf)
125{

130 }
131
132 is_tls = (c->ssl != NULL && c->ssl->connection != NULL);

134 tls_require_opcode_caps(

145 return brix_send_error(ctx, c, kXR_TLSRequired,
146 "server security policy requires TLS for this capability");

173 ngx_stream_brix_srv_conf_t *conf)
174{
175 xrdw_open_req_t req;
176
177 xrdw_open_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
178 if (!brix_open_options_is_write(req.options) || conf->manager_mode) {
179 return BRIX_DISPATCH_CONTINUE;

181

183 * Authentication still runs next in dispatch_read, and the TPC handler

192 {
193 return BRIX_DISPATCH_CONTINUE;
194 }

199 brix_io_monitor_record_err(&ctx->io_monitor, BRIX_ERR_FORBIDDEN);
200 return brix_send_error(ctx, c, kXR_fsReadOnly,

204/*
205 * brix_write_gate_tpc_push_sync — is this request the arm-or-fire kXR_sync of an
206 * F16 native TPC PUSH?
207 *
208 * WHAT: 1 only for kXR_sync naming an already-open handle that carries the
209 * tpc_push bit and whose transfer has not finished; 0 for everything else.
210 * WHY: kXR_sync belongs in the write table — a sync normally flushes bytes this
211 * server wrote — but a push SOURCE writes nothing locally. It opens its
212 * own file for READ (the task's dst_fd is O_RDONLY, and done.c never
213 * unlinks it), streams the bytes out, and the REMOTE destination does the
214 * writing behind its own brix_allow_write. Without this exemption the
215 * two syncs that arm and fire the copy would meet "this is a read-only
216 * server", so an operator could only export data by push if they first
217 * made the export writable — handing every client a mutation privilege the
218 * feature does not need. Fixing the gate is the honest fix; widening the
219 * export is not.
220 * HOW: scoped as narrowly as the wire permits. The opcode must be kXR_sync,
221 * the handle must be in bounds and open, and it must already carry
222 * tpc_push — a bit set only by brix_open_handle_tpc(), only on a listener
223 * with `brix_tpc_push on`, and only after the destination passed the
224 * egress allowlist and the cached SSRF verdict. Auth and the bound-stream
225 * refusal above are NOT relaxed; a kXR_sync on any other handle, and every
226 * other write opcode, still meets the read-only refusal.
227 */
228static int
229brix_write_gate_tpc_push_sync(brix_ctx_t *ctx)
230{
231 xrdw_sync_req_t req;
232 int idx;
233
234 if (ctx->recv.cur_reqid != kXR_sync || ctx->files == NULL) {
235 return 0;
236 }
237
238 xrdw_sync_req_unpack(((ClientRequestHdr *) ctx->recv.hdr_buf)->body, &req);
239 idx = (int) (unsigned char) req.fhandle[0];
240
241 if (idx < 0 || idx >= BRIX_MAX_FILES) {
242 return 0;
243 }
244
245 return ctx->files[idx].tpc_push && !ctx->files[idx].tpc_done;
246}

248ngx_int_t
249brix_dispatch_require_write(brix_ctx_t *ctx, ngx_connection_t *c,
250 ngx_stream_brix_srv_conf_t *conf)
251{
252 ngx_int_t rc;
253
254 /* The typed endpoint policy precedes identity and token evaluation. Use
255 * the pure kernel here because this is the protocol-edge tripwire: the VFS
256 * denial counter remains reserved for mutations that actually reach the
257 * authority-of-last-resort. */
258 if (!brix_write_gate_tpc_push_sync(ctx)
259 && brix_vfs_require_mutation_policy(
260 brix_vfs_policy_from_write_enable(conf->common.allow_write),
261 BRIX_VFS_MUTATE_WRITE) != NGX_OK)
262 {
263 brix_io_monitor_record_err(&ctx->io_monitor, BRIX_ERR_FORBIDDEN);
264 return brix_send_error(ctx, c, kXR_fsReadOnly,
265 "this is a read-only server");
266 }
267
268 rc = brix_dispatch_require_auth(ctx, c);
269 if (rc != BRIX_DISPATCH_CONTINUE) {
270 return rc;
271 }
272
273 /* Phase 94: a bound secondary may carry a plain kXR_write data payload for a
274 * primary-published writable handle (parallel upload) — but nothing else
275 * (no open/close/mkdir/mv/rm/writev/pgwrite): those stay primary-only so a
276 * secondary can never mutate the namespace or the handle table. */
277 if (ctx->is_bound && ctx->recv.cur_reqid != kXR_write) {
278 return brix_send_error(ctx, c, kXR_NotAuthorized,
279 "bound streams may only read or write primary "
280 "handles");
281 }
282
283 return BRIX_DISPATCH_CONTINUE;
284}
