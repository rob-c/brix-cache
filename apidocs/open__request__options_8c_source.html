29/*
30 * WHAT: Detect mutually exclusive kXR_open access-mode bits.
31 * WHY: Treating an all-bits or multi-mode request as an ordered set of hints
32 * partially honours attacker-chosen semantics (for example truncate plus
33 * append) instead of rejecting the contradictory request atomically.
34 * HOW: Normalize the stock-compatible wrto|apnd spelling to append, then count
35 * the effective access modes; zero/one is defined and two or more fails
36 * before TPC, path, or VFS work.
37 */
38int
39brix_open_options_conflict(uint16_t options)
40{
41 uint16_t access = options
42 & (kXR_open_read | kXR_open_updt | kXR_open_apnd | kXR_open_wrto);
43 unsigned count = 0;
44
45 /* Stock clients send wrto|apnd for a write-only append open. In that
46 * shape wrto declares the access direction and apnd refines its placement;
47 * they are one effective mode, not two contradictory modes. */
48 if ((access & (kXR_open_wrto | kXR_open_apnd))
49 == (kXR_open_wrto | kXR_open_apnd))
50 {
51 access &= (uint16_t) ~kXR_open_wrto;
52 }
53
54 while (access != 0) {
55 count += access & 1U;
56 access >>= 1;
57 }
58 return count > 1;
59}
