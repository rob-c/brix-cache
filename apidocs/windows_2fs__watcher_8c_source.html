37/*
38 * Map PAL BRIX_FS_EVENT_* to Windows FILE_NOTIFY_FILTER_*
39 *
40 * Note: Windows uses filters for what to watch, not events that occurred
41 */
42DWORD
43brix_win_map_pal_to_notify_filter(uint32_t events)
44{
45 DWORD filter = 0;
46
47 if (events & (BRIX_FS_EVENT_CREATE | BRIX_FS_EVENT_DELETE)) {
48 filter |= FILE_NOTIFY_CHANGE_FILE_NAME;
49 }
50
51 if (events & (BRIX_FS_EVENT_WRITE | BRIX_FS_EVENT_ATTRIB)) {
52 filter |= FILE_NOTIFY_CHANGE_LAST_WRITE;
53 filter |= FILE_NOTIFY_CHANGE_SIZE;
54 filter |= FILE_NOTIFY_CHANGE_ATTRIBUTES;
55 }
56
57 if (events & BRIX_FS_EVENT_RENAME) {
58 filter |= FILE_NOTIFY_CHANGE_FILE_NAME;
59 filter |= FILE_NOTIFY_CHANGE_DIR_NAME;
60 }
61
62 /* Additional filters for completeness */
63 filter |= FILE_NOTIFY_CHANGE_CREATION;
64 filter |= FILE_NOTIFY_CHANGE_LAST_ACCESS;
65 filter |= FILE_NOTIFY_CHANGE_SECURITY;
66
67 return filter;
68}

103/*

105 */
106static brix_win_watch_desc_t *
107brix_win_remove_watch_from_list(brix_plat_fs_watcher_t *watcher, int wd)
108{
109 brix_win_watch_desc_t **prev = &watcher->watches;
110 brix_win_watch_desc_t *current = watcher->watches;
111
112 while (current != NULL) {
113 if (current->wd == wd) {
114 *prev = current->next;
115 watcher->event_count--;
116 return current;
117 }
118 prev = &current->next;
119 current = current->next;
120 }
121
122 return NULL;
123}

158int
159brix_plat_fs_watcher_add(brix_plat_fs_watcher_t *watcher,
160 const char *path, uint32_t events)
161{
162 /*
163 * Add a directory watch
164 *
165 * Windows ReadDirectoryChangesW requirements:
166 * - Must open directory with FILE_LIST_DIRECTORY permission
167 * - Must use FILE_FLAG_BACKUP_SEMANTICS for directory handles
168 * - Supports recursive watching (optional)
169 */
170 brix_win_watch_desc_t *wdesc;
171 HANDLE dir_handle;
172 DWORD notify_filter;
173 char normalized_path[MAX_PATH];
174
175 if (watcher == NULL || path == NULL) {
176 errno = EINVAL;
177 return -1;
178 }
179
180 /* Normalize path (forward slashes to backslashes) */
181 strncpy(normalized_path, path, sizeof(normalized_path) - 1);
182 normalized_path[sizeof(normalized_path) - 1] = '\0';
183 brix_win32_normalize_path(normalized_path);
184
185 /* Open directory handle */
186 dir_handle = CreateFileA(
187 normalized_path,
188 FILE_LIST_DIRECTORY, /* Permission: list directory contents */
189 FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
190 NULL,
191 OPEN_EXISTING,
192 FILE_FLAG_BACKUP_SEMANTICS | /* Required for directory handles */
193 FILE_FLAG_OVERLAPPED, /* Async I/O */
194 NULL
195 );
196
197 if (dir_handle == INVALID_HANDLE_VALUE) {
198 brix_win32_set_errno(GetLastError());
199 return -1;
200 }
201
202 /* Allocate watch descriptor */
203 wdesc = (brix_win_watch_desc_t *)calloc(1, sizeof(brix_win_watch_desc_t));
204 if (wdesc == NULL) {
205 CloseHandle(dir_handle);
206 errno = ENOMEM;
207 return -1;
208 }
209
210 /* Initialize watch descriptor */
211 wdesc->dir_handle = dir_handle;
212 wdesc->events = events;
213 wdesc->wd = watcher->next_wd++;
214 strncpy(wdesc->path, normalized_path, sizeof(wdesc->path) - 1);
215
216 /* Initialize overlapped structure */
217 memset(&wdesc->overlapped, 0, sizeof(OVERLAPPED));
218 wdesc->overlapped.hEvent = CreateEventA(NULL, TRUE, FALSE, NULL);
219
220 if (wdesc->overlapped.hEvent == NULL) {
221 free(wdesc);
222 CloseHandle(dir_handle);
223 errno = ENOMEM;
224 return -1;
225 }
226
227 /* Map PAL events to Windows notify filter */
228 notify_filter = brix_win_map_pal_to_notify_filter(events);
229
230 /* Start monitoring with ReadDirectoryChangesW */
231 /* Note: This will complete asynchronously */
232 BOOL result = ReadDirectoryChangesW(
233 dir_handle,
234 wdesc->overlapped.Pointer, /* Buffer in overlapped structure */
235 BRIX_WIN_EVENT_BUFFER_SIZE,
236 FALSE, /* Not recursive (Phase 2: make configurable) */
237 notify_filter,
238 NULL, /* bytes_returned - NULL for async */
239 &wdesc->overlapped,
240 NULL /* completion routine - NULL for event-based */
241 );
242
243 if (!result) {
244 DWORD error = GetLastError();
245 if (error != ERROR_IO_PENDING) {
246 /* Real error (not just pending) */
247 brix_win32_set_errno(error);
248 CloseHandle(wdesc->overlapped.hEvent);
249 free(wdesc);
250 CloseHandle(dir_handle);
251 return -1;
252 }
253 /* ERROR_IO_PENDING is expected for async operation */
254 }
255
256 /* Add to watch list */
257 brix_win_add_watch_to_list(watcher, wdesc);
258
259 return wdesc->wd;
260}

262int
263brix_plat_fs_watcher_rm(brix_plat_fs_watcher_t *watcher, int wd)
264{
265 /*
266 * Remove a watch
267 *
268 * Cleanup steps:
269 * 1. Cancel pending I/O
270 * 2. Close event handle
271 * 3. Close directory handle
272 * 4. Free watch descriptor
273 */
274 brix_win_watch_desc_t *wdesc;
275
276 if (watcher == NULL || wd < 0) {
277 errno = EINVAL;
278 return -1;
279 }
280
281 /* Find and remove from list */
282 wdesc = brix_win_remove_watch_from_list(watcher, wd);
283 if (wdesc == NULL) {
284 errno = ENOENT;
285 return -1;
286 }
287
288 /* Cancel pending I/O */
289 CancelIo(wdesc->dir_handle);
290
291 /* Wait for cancellation to complete */
292 Sleep(10); /* Brief delay for cancellation */
293
294 /* Close handles */
295 if (wdesc->overlapped.hEvent != NULL) {
296 CloseHandle(wdesc->overlapped.hEvent);
297 }
298
299 CloseHandle(wdesc->dir_handle);
300
301 /* Free watch descriptor */
302 free(wdesc);
303
304 return 0;
305}

307void
308brix_plat_fs_watcher_destroy(brix_plat_fs_watcher_t *watcher)
309{
310 /*
311 * Destroy filesystem watcher
312 *
313 * Cleanup all remaining watches
314 */
315 brix_win_watch_desc_t *wdesc;
316 brix_win_watch_desc_t *next;
317
318 if (watcher == NULL) {
319 return;
320 }
321
322 /* Remove all watches */
323 wdesc = watcher->watches;
324 while (wdesc != NULL) {
325 next = wdesc->next;
326
327 /* Cancel I/O */
328 CancelIo(wdesc->dir_handle);
329
330 /* Close handles */
331 if (wdesc->overlapped.hEvent != NULL) {
332 CloseHandle(wdesc->overlapped.hEvent);
333 }
334 CloseHandle(wdesc->dir_handle);
335
336 /* Free descriptor */
337 free(wdesc);
338
339 wdesc = next;
340 }
341
342 watcher->watches = NULL;
343 watcher->event_count = 0;
344
345 /* Close completion port if created (Phase 2) */
346 if (watcher->completion_port != NULL) {
347 CloseHandle(watcher->completion_port);
348 watcher->completion_port = NULL;
349 }
350}

353 * LIMITATIONS SUMMARY (vs inotify/kqueue)
354 * ========================================================================== */

356/*
357 * ReadDirectoryChangesW Limitations vs inotify (Linux) and kqueue (macOS):
358 *
359 * 1. Event Granularity:
360 * - inotify: Per-file watches with fine-grained events
361 * - kqueue: Per-file descriptor watches with vnode events
362 * - Windows: Directory-based watches, less granular
363 *
364 * 2. Event Types:
365 * - inotify: IN_ACCESS, IN_MODIFY, IN_ATTRIB, IN_CLOSE_*, IN_OPEN, etc.
366 * - kqueue: NOTE_DELETE, NOTE_WRITE, NOTE_EXTEND, NOTE_ATTRIB, etc.
367 * - Windows: FILE_ACTION_* (ADDED, REMOVED, MODIFIED, RENAMED)
368 * - Missing: Access tracking, close events, open events
369 *
370 * 3. Event Reliability:
371 * - inotify: Queue-based, no event loss (unless queue overflows)
372 * - kqueue: Kernel queue, reliable delivery
373 * - Windows: Buffer-based, can overflow and lose events
374 *
375 * 4. Rename Handling:
376 * - inotify: IN_MOVED_FROM/IN_MOVED_TO with cookie for pairing
377 * - kqueue: NOTE_RENAME (single event)
378 * - Windows: FILE_ACTION_RENAMED_OLD_NAME/NEW_NAME (no cookie)
379 *
380 * 5. Interface:
381 * - inotify: File descriptor interface (select/poll/epoll compatible)
382 * - kqueue: File descriptor interface (select/poll/kqueue compatible)
383 * - Windows: HANDLE interface (requires WaitForMultipleObjects/IOCP)
384 *
385 * 6. Recursive Watching:
386 * - inotify: Manual (watch each subdirectory)
387 * - kqueue: Manual (watch each subdirectory)
388 * - Windows: Built-in support (hParameter to ReadDirectoryChangesW)
389 *
390 * 7. Performance:
391 * - inotify: Very efficient, kernel-side filtering
392 * - kqueue: Very efficient, kernel-side filtering
393 * - Windows: Good, but buffer copies and async complexity
394 *
395 * Phase 2 Improvements:
396 * - IOCP integration for scalable event handling
397 * - Better buffer management to prevent overflow
398 * - Recursive watching support
399 * - Proper UTF-16 to UTF-8 filename conversion
400 * - Timestamp extraction via GetFileTime
401 */
