1/*
2 * pblock_quota.c — F5 usage rollup + quota admission for the pblock driver.
3 *
4 * WHAT: Implements pblock_quota.h: trigger-maintained `usage` rollup (see the
5 * header for why triggers — transactional by construction) plus the
6 * admission checks the driver calls at catalog boundaries.
7 *
8 * HOW: The three triggers mirror insert/delete/update of `objects` rows into

36/* One UPSERT per scope; `B` is the byte contribution expression. */
37#define Q_UPS(scope_lit, id_expr, B) \
38 "INSERT INTO usage VALUES(" scope_lit "," id_expr "," B ",1)" \
39 " ON CONFLICT(scope,id) DO UPDATE SET" \
40 " bytes=bytes+excluded.bytes, inodes=inodes+excluded.inodes;"
41#define Q_DEC(scope_lit, id_expr, B) \
42 "UPDATE usage SET bytes=bytes-(" B "), inodes=inodes-1" \
43 " WHERE scope=" scope_lit " AND id=" id_expr ";"
44
45#define NEW_B "CASE WHEN NEW.is_dir=0 THEN NEW.size ELSE 0 END"
46#define OLD_B "CASE WHEN OLD.is_dir=0 THEN OLD.size ELSE 0 END"

48int
49pblock_quota_init(pblock_state_t *st)
50{
51 static const char *ddl[] = {
52 "CREATE TABLE IF NOT EXISTS usage("
53 " scope TEXT NOT NULL, id INTEGER NOT NULL,"
54 " bytes INTEGER NOT NULL, inodes INTEGER NOT NULL,"
55 " PRIMARY KEY(scope, id));",
56
57 "CREATE TRIGGER IF NOT EXISTS usage_ai AFTER INSERT ON objects BEGIN "
58 Q_UPS("'total'", "0", NEW_B)
59 Q_UPS("'uid'", "NEW.uid", NEW_B)
60 Q_UPS("'gid'", "NEW.gid", NEW_B)
61 "END;",
62
63 "CREATE TRIGGER IF NOT EXISTS usage_ad AFTER DELETE ON objects BEGIN "
64 Q_DEC("'total'", "0", OLD_B)
65 Q_DEC("'uid'", "OLD.uid", OLD_B)
66 Q_DEC("'gid'", "OLD.gid", OLD_B)
67 "END;",
68
69 "CREATE TRIGGER IF NOT EXISTS usage_au AFTER UPDATE OF size, uid, gid "
70 "ON objects BEGIN "
71 Q_DEC("'total'", "0", OLD_B)
72 Q_DEC("'uid'", "OLD.uid", OLD_B)
73 Q_DEC("'gid'", "OLD.gid", OLD_B)
74 Q_UPS("'total'", "0", NEW_B)
75 Q_UPS("'uid'", "NEW.uid", NEW_B)
76 Q_UPS("'gid'", "NEW.gid", NEW_B)
77 "END;",
78
79 /* rebuild: honest rollup even when quota is enabled on a populated
80 * export (or after a run without quota). */
81 "DELETE FROM usage;",
82 "INSERT INTO usage SELECT 'total', 0,"
83 " COALESCE(SUM(CASE WHEN is_dir=0 THEN size ELSE 0 END),0), COUNT(*)"
84 " FROM objects;",
85 "INSERT INTO usage SELECT 'uid', uid,"
86 " SUM(CASE WHEN is_dir=0 THEN size ELSE 0 END), COUNT(*)"
87 " FROM objects GROUP BY uid;",
88 "INSERT INTO usage SELECT 'gid', gid,"
89 " SUM(CASE WHEN is_dir=0 THEN size ELSE 0 END), COUNT(*)"
90 " FROM objects GROUP BY gid;",
91 };
92 size_t i;
93
94 if (cat_exec(st->cat, "BEGIN;") != 0) {
95 return -1;
96 }
97 for (i = 0; i < sizeof(ddl) / sizeof(ddl[0]); i++) {
98 if (cat_exec(st->cat, ddl[i]) != 0) {
99 (void) cat_exec(st->cat, "ROLLBACK;");
100 return -1;
101 }
102 }
103 return cat_exec(st->cat, "COMMIT;");
104}

106int
107pblock_quota_usage(const pblock_state_t *st, const char *scope, int64_t id,
108 int64_t *bytes, int64_t *inodes)
109{
110 sqlite3_stmt *q;
111
112 *bytes = 0;
113 *inodes = 0;
114 q = cat_prepare(st->cat,
115 "SELECT bytes, inodes FROM usage WHERE scope = ?1 AND id = ?2;");
116 if (q == NULL) {
117 return -1;
118 }
119 sqlite3_bind_text(q, 1, scope, -1, SQLITE_STATIC);
120 sqlite3_bind_int64(q, 2, id);
121 if (sqlite3_step(q) == SQLITE_ROW) {
122 *bytes = sqlite3_column_int64(q, 0);
123 *inodes = sqlite3_column_int64(q, 1);
124 }
125 sqlite3_finalize(q);
126 return 0;
127}

142int
143pblock_quota_admit(const pblock_state_t *st, uint32_t uid,
144 int64_t add_bytes, int64_t add_inodes)
145{
146 int64_t bytes, inodes, limit;
147
148 if (!st->quota) {
149 return 0;
150 }
151 if (pblock_quota_usage(st, "total", 0, &bytes, &inodes) != 0) {
152 return 0; /* fail-open on a rollup read error */
153 }
154 if ((st->quota_bytes > 0 && add_bytes > 0
155 && bytes + add_bytes > st->quota_bytes)
156 || (st->quota_inodes > 0 && add_inodes > 0
157 && inodes + add_inodes > st->quota_inodes))
158 {
159 errno = EDQUOT;
160 return -1;
161 }
162
163 limit = quota_uid_limit(st, uid);
164 if (limit > 0 && add_bytes > 0) {
165 if (pblock_quota_usage(st, "uid", (int64_t) uid, &bytes, &inodes) != 0) {
166 return 0;
167 }
168 if (bytes + add_bytes > limit) {
169 errno = EDQUOT;
170 return -1;
171 }
172 }
173 return 0;
174}

176int64_t
177pblock_quota_max_size(const pblock_state_t *st, uint32_t uid, int64_t cur_size)
178{
179 int64_t bytes, inodes, room = INT64_MAX, limit;
180
181 if (!st->quota) {
182 return INT64_MAX;
183 }
184 if (st->quota_bytes > 0
185 && pblock_quota_usage(st, "total", 0, &bytes, &inodes) == 0)
186 {
187 room = st->quota_bytes - bytes;
188 }
189 limit = quota_uid_limit(st, uid);
190 if (limit > 0
191 && pblock_quota_usage(st, "uid", (int64_t) uid, &bytes, &inodes) == 0
192 && limit - bytes < room)
193 {
194 room = limit - bytes;
195 }
196 if (room == INT64_MAX) {
197 return INT64_MAX; /* no byte limit in play */
198 }
199 if (room < 0) {
200 room = 0;
201 }
202 return cur_size + room;
203}

221/* sd_pblock_reserve — phase-107 C5: admit a declared final size at open.
222 *
223 * WHAT: Decide, before the first byte is accepted, whether this handle may
224 * grow the object to `size`: the F5 quota ceiling snapshotted at open
225 * (per-uid + export, EDQUOT) and the physical free space of the block
226 * store's filesystem (ENOSPC).
227 * WHY: A declaration the catalog quota or the disk cannot satisfy must fail
228 * the OPEN (kXR_NoSpace / 507), not the write hours of streaming later.
229 * HOW: Pure admission — no block is chained (block files materialise on
230 * write; a physical preallocation would buy nothing on a store whose
231 * blocks are independent files). statvfs failure is ignored: quota is
232 * the deterministic contract, the physical probe is best-effort. */
233ngx_int_t
234sd_pblock_reserve(brix_sd_obj_t *obj, off_t size)
235{
236 pblock_obj_t *os = obj->state;
237 struct statvfs sv;
238
239 if ((int64_t) size > os->quota_max) {
240 errno = EDQUOT;
241 return NGX_ERROR;
242 }
243 if (statvfs(os->st->data_dir, &sv) == 0
244 && (uint64_t) size > (uint64_t) sv.f_bavail * (uint64_t) sv.f_frsize)
245 {
246 errno = ENOSPC;
247 return NGX_ERROR;
248 }
249 return NGX_OK;
250}
