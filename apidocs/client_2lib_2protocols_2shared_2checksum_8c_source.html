1/*
2 * checksum.c — local streaming checksums + the server-side kXR_Qcksum query.
3 *
4 * WHAT: Compute adler32 / crc32c / md5 over a local file descriptor, and ask the
5 * server for a file's checksum (kXR_query, infotype kXR_Qcksum). Both yield
6 * a lowercase hex digest so xrdcp --cksum can compare local vs source.
7 * WHY: --cksum[:source] needs (a) a digest of the bytes we transferred and (b)
8 * the server's digest of the same file, then asserts they agree.
9 * HOW: The fd compute is delegated to the SHARED kernel brix_cksum_*_fd
10 * (src/core/compat/checksum_core.c, linked via libxrdproto) — the exact same
11 * adler32/crc32c/md5 code the nginx module runs, so client and server agree
12 * by construction (single source). This file only maps the client's algo
13 * enum, hex-encodes, and drives the Qcksum wire query (payload "<algo>
14 * <path>" → reply "<algo> <hexdigest>").
15 *
16 * wire: XProtocol.hh kXR_query infotype kXR_Qcksum — body "<algo> <path>" → "<algo> <hex>".
17 */

19#include "core/compat/checksum_core.h" /* shared fd→checksum kernels (libxrdproto) */

46int
47brix_cksum_algo_parse(const char *name, brix_cksum_algo *out)
48{
49 if (name == NULL) {
50 return -1;
51 }
52 for (size_t n = 0; n < sizeof(cksum_names) / sizeof(cksum_names[0]); n++) {
53 if (strcmp(name, cksum_names[n].name) == 0) {
54 *out = cksum_names[n].algo;
55 return 0;
56 }
57 }
58 return -1;
59}

61/* MD5/SHA family: shared EVP digest kernel → lowercase hex. */
62static int
63cksum_digest_family_fd(int fd, brix_cksum_algo algo, char *hex, size_t hexsz,
64 brix_status *st)
65{
66 int kind = (algo == XRDC_CK_MD5) ? BRIX_CK_MD5
67 : (algo == XRDC_CK_SHA1) ? BRIX_CK_SHA1
68 : (algo == XRDC_CK_SHA256) ? BRIX_CK_SHA256
69 : BRIX_CK_SHA512;
70 unsigned char dg[64]; /* EVP_MAX_MD_SIZE */
71 unsigned int dn = 0;
72
73 if (brix_cksum_digest_fd(kind, fd, dg, &dn) != 0) {
74 brix_status_set(st, XRDC_ESOCK, errno, "digest: %s",
75 strerror(errno));
76 return -1;
77 }
78 if (hexsz < (size_t) dn * 2 + 1) {
79 brix_status_set(st, XRDC_EUSAGE, 0, "hex buffer too small");
80 return -1;
81 }
82 brix_hex_encode(dg, dn, hex);
83 return 0;
84}

86/* crc64/crc64nvme: 64-bit kernel → 16 zero-padded hex digits. */
87static int
88cksum_u64_family_fd(int fd, brix_cksum_algo algo, char *hex, size_t hexsz,
89 brix_status *st)
90{
91 int kind = (algo == XRDC_CK_CRC64) ? BRIX_CK_CRC64
92 : BRIX_CK_CRC64NVME;
93 uint64_t value;
94
95 if (brix_cksum_u64_fd(kind, fd, &value) != 0) {
96 brix_status_set(st, XRDC_ESOCK, errno, "checksum read: %s",
97 strerror(errno));
98 return -1;
99 }
100 if (hexsz < 17) { /* 16 hex digits + NUL */
101 brix_status_set(st, XRDC_EUSAGE, 0, "hex buffer too small");
102 return -1;
103 }
104 snprintf(hex, hexsz, "%016llx", (unsigned long long) value);
105 return 0;
106}

131int
132brix_cksum_fd(int fd, brix_cksum_algo algo, char *hex, size_t hexsz,
133 brix_status *st)
134{
135 /* Delegate the compute to the shared (ngx-free) kernel — the same code the
136 * nginx module uses (src/core/compat/checksum_core.c via libxrdproto). The kernel
137 * preads from offset 0; callers pass freshly-opened regular-file fds. */
138 if (algo == XRDC_CK_MD5 || algo == XRDC_CK_SHA1
139 || algo == XRDC_CK_SHA256 || algo == XRDC_CK_SHA512) {
140 return cksum_digest_family_fd(fd, algo, hex, hexsz, st);
141 }
142 if (algo == XRDC_CK_CRC64 || algo == XRDC_CK_CRC64NVME) {
143 return cksum_u64_family_fd(fd, algo, hex, hexsz, st);
144 }
145 return cksum_u32_family_fd(fd, algo, hex, hexsz, st);
146}

148int
149brix_query_cksum(brix_conn *c, const char *path, const char *algo_name,
150 char *hex, size_t hexsz, brix_status *st)
151{
152 ClientQueryRequest req;
153 uint16_t status;
154 uint8_t *body = NULL;
155 uint32_t blen = 0;
156 char payload[XRDC_PATH_MAX + 32];
157 int plen;
158 const char *sp;
159
160 /* kXR_Qcksum wire format is the PATH, with the algorithm requested via the
161 * standard "?cks.type=<algo>" CGI — NOT a leading "<algo> <path>" (which a
162 * standard server, e.g. EOS, treats as a bogus relative path and rejects:
163 * "Check summing relative path 'adler32 /…' is disallowed"). Our own server
164 * accepts a bare path too, so this is correct for both. */
165 if (algo_name != NULL && algo_name[0] != '\0') {
166 plen = snprintf(payload, sizeof(payload), "%s%ccks.type=%s", path,
167 (strchr(path, '?') != NULL) ? '&' : '?', algo_name);
168 } else {
169 plen = snprintf(payload, sizeof(payload), "%s", path);
170 }
171 if (plen < 0 || (size_t) plen >= sizeof(payload)) {
172 brix_status_set(st, XRDC_EUSAGE, 0, "checksum query path too long");
173 return -1;
174 }
175
176 memset(&req, 0, sizeof(req));
177 req.requestid = htons(kXR_query);
178 {
179 xrdw_query_req_t b = { .infotype = kXR_Qcksum };
180 xrdw_query_req_pack(&b, ((ClientRequestHdr *) &req)->body);
181 }
182 /* fhandle stays zero: this is a path-based (not open-handle) query. */
183
184 /* Read-only/idempotent: a re-query after a sever yields the same digest. */
185 if (brix_roundtrip_resilient(c, &req, payload, (uint32_t) plen,
186 XRDC_OP_READONLY, 0,
187 &status, &body, &blen, st) != 0) {
188 return -1;
189 }
190
191 /* Reply: "<algo> <hexdigest>" — take the token after the (last) space. */
192 {
193 char tmp[256];
194 uint32_t n = (blen < sizeof(tmp) - 1) ? blen : (uint32_t) (sizeof(tmp) - 1);
195 memcpy(tmp, body, n);
196 tmp[n] = '\0';
197 free(body);
198
199 sp = strrchr(tmp, ' ');
200 sp = (sp != NULL) ? sp + 1 : tmp; /* tolerate a digest-only reply */
201 /* trim trailing whitespace/newline */
202 {
203 size_t L = strlen(sp);
204 while (L > 0 && (sp[L - 1] == '\n' || sp[L - 1] == '\r'
205 || sp[L - 1] == ' ')) {
206 ((char *) sp)[--L] = '\0';
207 }
208 if (L == 0 || L + 1 > hexsz) {
209 brix_status_set(st, XRDC_EPROTO, 0,
210 "unparseable checksum reply: \"%s\"", tmp);
211 return -1;
212 }
213 }
214 snprintf(hex, hexsz, "%s", sp);
215 }
216 return 0;
217}
