4/*
5 * dashboard/files.c — admin file browser + downloader for the monitoring UI.
6 *
7 * Two admin-auth-only endpoints, both confined to brix_dashboard_browse_root
8 * via the kernel openat2(RESOLVE_BENEATH) primitive (brix_open_beneath), so a
9 * client-supplied ?path= can never escape the configured root:
10 *
11 * GET /xrootd/api/v1/files?path=<rel> JSON listing of a directory:
12 * each entry's name, type, size, local owner (uid -> username), and
13 * modification + creation (statx btime) timestamps.
14 * GET /xrootd/api/v1/download?path=<rel> stream one regular file as an
15 * application/octet-stream attachment.
16 *
17 * SECURITY:
18 * - Always admin-auth (ngx_http_brix_dashboard_check_auth) — never the
19 * anonymous read tier, mirroring the config-download endpoint.
20 * - Disabled unless brix_dashboard_browse_root is configured (endpoints 404).
21 * - Confinement is the kernel's RESOLVE_BENEATH; per-entry statx uses
22 * AT_SYMLINK_NOFOLLOW so a symlink is reported, never followed.
23 */

43/* Map a confined-open failure to an HTTP status. A RESOLVE_BENEATH escape
44 * attempt surfaces as EXDEV/ELOOP — report it as 403 (forbidden), not 500. */
45static ngx_int_t
46dashboard_open_status(int e)
47{
48 if (e == ENOENT || e == ENOTDIR) {
49 return NGX_HTTP_NOT_FOUND;
50 }
51 if (e == EACCES || e == EPERM || e == EXDEV || e == ELOOP) {
52 return NGX_HTTP_FORBIDDEN;
53 }
54 return NGX_HTTP_INTERNAL_SERVER_ERROR;
55}

58/* Resolve uid -> local username; falls back to the decimal uid on lookup miss.
59 * A 1-deep cache short-circuits the common "whole dir owned by one user" case. */
60static void
61dashboard_owner_name(uid_t uid, char *out, size_t outsz)
62{
63 static uid_t s_cached_uid;
64 static char s_cached_name[64];
65 static int s_have_cache;
66
67 struct passwd pw;
68 struct passwd *res = NULL;
69 char buf[BRIX_MEDIUM_BUF_SIZE];
70
71 if (s_have_cache && s_cached_uid == uid) {
72 ngx_snprintf((u_char *) out, outsz, "%s%Z", s_cached_name);
73 return;
74 }
75 if (getpwuid_r(uid, &pw, buf, sizeof(buf), &res) == 0 && res != NULL
76 && res->pw_name != NULL)
77 {
78 ngx_snprintf((u_char *) out, outsz, "%s%Z", res->pw_name);
79 ngx_snprintf((u_char *) s_cached_name, sizeof(s_cached_name), "%s%Z",
80 res->pw_name);
81 s_cached_uid = uid;
82 s_have_cache = 1;
83 return;
84 }
85 ngx_snprintf((u_char *) out, outsz, "%ud%Z", (ngx_uint_t) uid);
86}

88/*
89 * Extract + URL-decode the ?path= query argument into out[] as a path RELATIVE
90 * to the browse root (leading '/' stripped; empty/"/" => "."). Rejects an
91 * embedded NUL (decode artefact / smuggling). Returns NGX_OK / NGX_ERROR.
92 * The kernel RESOLVE_BENEATH at open time is what actually enforces confinement;
93 * this only normalises the input.
94 */
95static ngx_int_t
96dashboard_files_get_path(ngx_http_request_t *r, char *out, size_t outsz)
97{
98 ngx_str_t raw;
99 u_char *dst, *src;
100 size_t n;
101
102 if (ngx_http_arg(r, (u_char *) "path", 4, &raw) != NGX_OK
103 || raw.len == 0)
104 {
105 ngx_memcpy(out, ".", 2);
106 return NGX_OK;
107 }
108 if (raw.len >= outsz) {
109 return NGX_ERROR;
110 }
111
112 /* URL-decode in place into out (decoded length <= encoded length). */
113 dst = (u_char *) out;
114 src = raw.data;
115 ngx_unescape_uri(&dst, &src, raw.len, 0);
116 n = (size_t) (dst - (u_char *) out);
117 out[n] = '\0';
118
119 if (ngx_strlchr((u_char *) out, (u_char *) out + n, '\0') != NULL) {
120 return NGX_ERROR; /* embedded NUL */
121 }
122
123 /* Strip a leading '/' so the value is relative to the browse root; map the
124 * root itself to "." (what openat2 wants for the anchor dir). */
125 {
126 char *p = out;
127 while (*p == '/') {
128 p++;
129 }
130 if (*p == '\0') {
131 ngx_memcpy(out, ".", 2);
132 } else if (p != out) {
133 ngx_memmove(out, p, strlen(p) + 1);
134 }
135 }
136 return NGX_OK;
137}

140/* The per-entry metadata the listing renders. Filled by dashboard_entry_stat
141 * from statx(2) on Linux and fstatat(2) elsewhere, so the JSON schema is the
142 * same on every platform. btime is 0 when the filesystem does not report a
143 * creation time (the client falls back to mtime). */
144typedef struct {
145 mode_t mode;
146 off_t size;
147 uid_t uid;
148 time_t mtime;
149 time_t btime;
150} dashboard_entry_stat_t;
151
152/* Stat one directory entry without following symlinks (a symlink is listed
153 * as "other"). 0 on success, -1 (errno set) when the entry vanished or is
154 * unreadable. The PAL supplies the birth time (Linux statx, BSD
155 * st_birthtimespec) and reports 0 where the host has none. */
156static int
157dashboard_entry_stat(int dirfd, const char *name, dashboard_entry_stat_t *out)
158{
159 struct stat st;
160
161 if (brix_plat_fstatat_btime(dirfd, name, AT_SYMLINK_NOFOLLOW, &st,
162 &out->btime) != 0)
163 {
164 return -1;
165 }
166 out->mode = st.st_mode;
167 out->size = st.st_size;
168 out->uid = st.st_uid;
169 out->mtime = st.st_mtime;
170 return 0;
171}

173/* Build one entry object from a stat result. Returns a json_t (caller owns) or
174 * NULL on OOM. */
175static json_t *
176dashboard_files_entry(const char *name, const dashboard_entry_stat_t *st)
177{
178 json_t *o = json_object();
179 char owner[64];
180
181 if (o == NULL) { return NULL; }
182 dashboard_owner_name(st->uid, owner, sizeof(owner));
183
184 json_object_set_new(o, "name", json_string(name));
185 json_object_set_new(o, "type",
186 S_ISDIR(st->mode) ? json_string("dir")
187 : S_ISREG(st->mode) ? json_string("file")
188 : json_string("other"));
189 json_object_set_new(o, "size", json_integer((json_int_t) st->size));
190 json_object_set_new(o, "owner", json_string(owner));
191 json_object_set_new(o, "uid", json_integer((json_int_t) st->uid));
192 json_object_set_new(o, "mtime", json_integer((json_int_t) st->mtime));
193 /* Creation time (birth) when the filesystem supports it; 0 => client falls
194 * back to mtime. */
195 json_object_set_new(o, "btime", json_integer((json_int_t) st->btime));
196 return o;
197}

199/*
200 * WHAT: Run the admin-auth + method + feature-enabled preamble common to both
201 * file endpoints, then extract the confined relative ?path= into out[].
202 * WHY: files_handler and download_handler share an identical gate; factoring it
203 * keeps each orchestrator flat and the security checks in one place.
204 * HOW: feature-disabled/auth/method map to their HTTP status exactly as before
205 * (auth returns its own rc); on success emit the AGPL source offer and
206 * normalise the path. Returns NGX_OK when the caller may proceed, else the
207 * HTTP status (or auth rc) to return verbatim.
208 */
209static ngx_int_t
210dashboard_files_gate(ngx_http_request_t *r,
211 ngx_http_brix_dashboard_loc_conf_t *conf,
212 char *out, size_t outsz)
213{
214 ngx_int_t rc;
215
216 if (conf->browse_root_canon[0] == '\0') {
217 return NGX_HTTP_NOT_FOUND; /* feature disabled */
218 }
219 rc = ngx_http_brix_dashboard_check_auth(r, conf, 0);
220 if (rc != NGX_OK) { return rc; }
221 if (r->method != NGX_HTTP_GET && r->method != NGX_HTTP_HEAD) {
222 return NGX_HTTP_NOT_ALLOWED;
223 }
224 brix_http_source_offer(r);
225
226 if (dashboard_files_get_path(r, out, outsz) != NGX_OK) {
227 return NGX_HTTP_BAD_REQUEST;
228 }
229 return NGX_OK;
230}

232/*
233 * WHAT: Open the confined directory named by relpath under the browse root.
234 * WHY: Isolates the two-fd RESOLVE_BENEATH open + fdopendir dance (with its
235 * own error mapping) from the listing orchestrator.
236 * HOW: open the root anchor, openat2-beneath the relpath as a directory, hand
237 * the fd to fdopendir. On success *out_dp is set and *out_dirfd holds the
238 * still-open descriptor (owned by the DIR*, closed by closedir). Returns
239 * NGX_OK or the HTTP status to return; *out_dp is NULL on any error.
240 */
241static ngx_int_t
242dashboard_files_opendir(ngx_http_brix_dashboard_loc_conf_t *conf,
243 const char *relpath, DIR **out_dp, int *out_dirfd)
244{
245 int rootfd, dirfd;
246
247 *out_dp = NULL;
248 *out_dirfd = -1;
249
250 rootfd = brix_beneath_open_root(conf->browse_root_canon);
251 if (rootfd < 0) {
252 return NGX_HTTP_INTERNAL_SERVER_ERROR;
253 }
254 dirfd = brix_open_beneath(rootfd, relpath,
255 O_RDONLY | O_DIRECTORY | O_CLOEXEC, 0);
256 close(rootfd);
257 if (dirfd < 0) {
258 return dashboard_open_status(ngx_errno);
259 }
260 *out_dp = fdopendir(dirfd);
261 if (*out_dp == NULL) {
262 close(dirfd);
263 return NGX_HTTP_INTERNAL_SERVER_ERROR;
264 }
265 *out_dirfd = dirfd;
266 return NGX_OK;
267}

269/*
270 * WHAT: Read every visible entry of an open directory into the JSON array arr,
271 * capped at DASHBOARD_FILES_MAX_ENTRIES.
272 * WHY: Pulls the readdir/statx/skip loop out of the orchestrator so the handler
273 * stays a flat sequence; entry emission stays byte-identical.
274 * HOW: skip "."/".."; per-entry AT_SYMLINK_NOFOLLOW statx (a symlink is
275 * reported, never followed); vanished/unreadable entries are skipped;
276 * on hitting the cap set *truncated and stop. dirfd is the statx anchor.
277 */
278static void
279dashboard_files_scan(DIR *dp, int dirfd, json_t *arr, int *truncated)
280{
281 struct dirent *de;
282 ngx_uint_t count = 0;
283
284 while ((de = readdir(dp)) != NULL) {
285 dashboard_entry_stat_t st;
286 json_t *e;
287
288 if (de->d_name[0] == '.'
289 && (de->d_name[1] == '\0'
290 || (de->d_name[1] == '.' && de->d_name[2] == '\0')))
291 {
292 continue; /* skip "." and ".." (UI navigates via breadcrumb) */
293 }
294 if (count >= DASHBOARD_FILES_MAX_ENTRIES) {
295 *truncated = 1;
296 break;
297 }
298 if (dashboard_entry_stat(dirfd, de->d_name, &st) != 0) {
299 continue; /* vanished mid-scan / unreadable — skip */
300 }
301 e = dashboard_files_entry(de->d_name, &st);
302 if (e == NULL || json_array_append_new(arr, e) != 0) {
303 continue;
304 }
305 count++;
306 }
307}

309/*
310 * WHAT: Build the JSON listing document (schema + echoed path + truncated flag +
311 * entries array) and send it as the 200 response.
312 * WHY: Separates document assembly + send from the scan, and localises the two
313 * json_object()/json_array() OOM guards.
314 * HOW: allocate root+arr (503 on OOM); scan the directory into arr; echo the
315 * normalised path (root sentinel "." -> "/"); send. closedir(dp) also
316 * closes the underlying dirfd. Returns the HTTP result of the send.
317 */
318static ngx_int_t
319dashboard_files_render(ngx_http_request_t *r, DIR *dp, int dirfd,
320 const char *relpath)
321{
322 json_t *root, *arr;
323 int truncated = 0;
324
325 root = json_object();
326 arr = json_array();
327 if (root == NULL || arr == NULL) {
328 if (root) { json_decref(root); }
329 if (arr) { json_decref(arr); }
330 closedir(dp);
331 return NGX_HTTP_INTERNAL_SERVER_ERROR;
332 }
333
334 dashboard_files_scan(dp, dirfd, arr, &truncated);
335 closedir(dp); /* also closes dirfd */
336
337 /* Re-derive the normalised path for echo (strip the "." root sentinel). */
338 dashboard_json_set_schema(root);
339 json_object_set_new(root, "path",
340 json_string((relpath[0] == '.' && relpath[1] == '\0') ? "/" : relpath));
341 json_object_set_new(root, "truncated", truncated ? json_true() : json_false());
342 json_object_set_new(root, "entries", arr);
343
344 return dashboard_json_send(r, NGX_HTTP_OK, root);
345}

367/* GET /xrootd/api/v1/download?path=<rel> */
368/* Build a safe Content-Disposition filename from the basename of relpath:
369 * strip the directory, then any byte that could break the header (quote,
370 * control, backslash) becomes '_'. */
371static void
372dashboard_download_filename(const char *relpath, char *out, size_t outsz)
373{
374 const char *base = strrchr(relpath, '/');
375 size_t i;
376
377 base = (base != NULL) ? base + 1 : relpath;
378 if (base[0] == '\0') {
379 base = "download";
380 }
381 for (i = 0; base[i] != '\0' && i + 1 < outsz; i++) {
382 unsigned char c = (unsigned char) base[i];
383 out[i] = (c < 0x20 || c == '"' || c == '\\' || c == 0x7f) ? '_'
384 : (char) c;
385 }
386 out[i] = '\0';
387}

389/*
390 * WHAT: Confined-open the regular file named by relpath and fstat it into *sb.
391 * WHY: Isolates the RESOLVE_BENEATH open + type check (with error mapping) so
392 * the download orchestrator stays a flat sequence.
393 * HOW: open beneath the root with O_NOFOLLOW (a symlink is never followed);
394 * fstat; reject non-regular as 404. On success *out_fd holds the open fd
395 * (caller owns it); on any error *out_fd is -1 and the fd is closed here.
396 * Returns NGX_OK or the HTTP status to return verbatim.
397 */
398static ngx_int_t
399dashboard_download_open(ngx_http_brix_dashboard_loc_conf_t *conf,
400 const char *relpath, int *out_fd, struct stat *sb)
401{
402 int rootfd, fd;
403
404 *out_fd = -1;
405
406 rootfd = brix_beneath_open_root(conf->browse_root_canon);
407 if (rootfd < 0) {
408 return NGX_HTTP_INTERNAL_SERVER_ERROR;
409 }
410 fd = brix_open_beneath(rootfd, relpath,
411 O_RDONLY | O_NOFOLLOW | O_CLOEXEC, 0);
412 close(rootfd);
413 if (fd < 0) {
414 return dashboard_open_status(ngx_errno);
415 }
416 if (fstat(fd, sb) != 0) {
417 close(fd);
418 return NGX_HTTP_INTERNAL_SERVER_ERROR;
419 }
420 if (!S_ISREG(sb->st_mode)) {
421 close(fd);
422 return NGX_HTTP_NOT_FOUND; /* only regular files are downloadable */
423 }
424 *out_fd = fd;
425 return NGX_OK;
426}

428/*
429 * WHAT: Set the octet-stream download response headers (status, length, type,
430 * last-modified, Content-Disposition attachment, Cache-Control no-store).
431 * WHY: Separates header assembly (pure header mutation) from the fd lifecycle
432 * and the send, keeping the orchestrator flat.
433 * HOW: derive a header-safe attachment filename from the basename; push the two
434 * extra headers. A Content-Disposition value OOM is the one failure and
435 * maps to 503; the caller then closes the fd. Returns NGX_OK or the HTTP
436 * status to return.
437 */
438static ngx_int_t
439dashboard_download_set_headers(ngx_http_request_t *r, const struct stat *sb,
440 const char *relpath)
441{
442 char fname[256];
443 char disp[320];
444 ngx_table_elt_t *h;
445 int n;
446
447 dashboard_download_filename(relpath, fname, sizeof(fname));
448
449 r->headers_out.status = NGX_HTTP_OK;
450 r->headers_out.content_length_n = (off_t) sb->st_size;
451 r->headers_out.content_type =
452 (ngx_str_t) ngx_string("application/octet-stream");
453 r->headers_out.content_type_len = r->headers_out.content_type.len;
454 r->headers_out.last_modified_time = sb->st_mtime;
455
456 n = ngx_snprintf((u_char *) disp, sizeof(disp),
457 "attachment; filename=\"%s\"", fname) - (u_char *) disp;
458 h = ngx_list_push(&r->headers_out.headers);
459 if (h != NULL) {
460 h->hash = 1;
461 ngx_str_set(&h->key, "Content-Disposition");
462 h->value.len = (size_t) n;
463 h->value.data = ngx_pnalloc(r->pool, (size_t) n);
464 if (h->value.data == NULL) { return NGX_HTTP_INTERNAL_SERVER_ERROR; }
465 ngx_memcpy(h->value.data, disp, (size_t) n);
466 }
467 h = ngx_list_push(&r->headers_out.headers);
468 if (h != NULL) {
469 h->hash = 1;
470 ngx_str_set(&h->key, "Cache-Control");
471 ngx_str_set(&h->value, "no-store");
472 }
473 return NGX_OK;
474}

476/*
477 * WHAT: Emit the download response body for an open regular file fd.
478 * WHY: Splits the HEAD/empty short-circuit from the streaming path so each is a
479 * single linear branch owning the fd's fate.
480 * HOW: for HEAD or a zero-length file, send headers + a LAST special and close
481 * fd here; otherwise hand fd to brix_http_send_file_range, which sends
482 * headers, builds the file buf, and closes fd via pool cleanup. Returns
483 * the HTTP result.
484 */
485static ngx_int_t
486dashboard_download_body(ngx_http_request_t *r, int fd, const char *relpath,
487 const struct stat *sb)
488{
489 ngx_int_t rc;
490
491 if (r->method == NGX_HTTP_HEAD || sb->st_size == 0) {
492 rc = ngx_http_send_header(r);
493 close(fd);
494 if (rc == NGX_ERROR || rc > NGX_OK) { return rc; }
495 return ngx_http_send_special(r, NGX_HTTP_LAST);
496 }
497
498 /* Stream the file (sendfile on cleartext; nginx reads it for TLS). The
499 * helper sends headers, builds the file buf, and closes fd via pool cleanup. */
500 return brix_http_send_file_range(r, fd, relpath, 0, (off_t) sb->st_size, 1);
501}
