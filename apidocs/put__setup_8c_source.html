29/* Add an "X-Upload-Offset: <off>" response header (the resumable-PUT progress
30 * marker the client reads to know where to continue). Best-effort. */
31static void
32webdav_set_upload_offset(ngx_http_request_t *r, off_t off)
33{
34 ngx_table_elt_t *h = ngx_list_push(&r->headers_out.headers);
35 if (h == NULL) {
36 return;
37 }
38 h->hash = 1;
39 ngx_str_set(&h->key, "X-Upload-Offset");
40 h->value.data = ngx_pnalloc(r->pool, NGX_OFF_T_LEN);
41 if (h->value.data == NULL) {
42 h->hash = 0;
43 return;
44 }
45 h->value.len = ngx_sprintf(h->value.data, "%O", off) - h->value.data;
46}

48/*
49 * Resumable Content-Range PUT (brix_webdav_upload_resume on). Writes this
50 * chunk to a persistent, identity-keyed partial at its absolute offset and
51 * commits (atomic rename) only when the upload is complete. Across separate PUT
52 * requests — and across an nginx restart — the partial survives, so the client
53 * resumes from X-Upload-Offset. Append-only: the chunk must start exactly at the
54 * current partial size, else 409 + X-Upload-Offset tells the client the truth.
55 */
56static void
57webdav_put_ranged_resume(ngx_http_request_t *r,
58 ngx_http_brix_webdav_loc_conf_t *conf, const char *path,
59 const brix_http_content_range_t *cr)
60{
61 brix_staged_file_t staged;
62 off_t cur = 0;
63 brix_http_body_summary_t bs;
64 ngx_http_brix_webdav_req_ctx_t *wctx =
65 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
66 const char *principal = (wctx != NULL && wctx->dn[0] != '\0')
67 ? wctx->dn : NULL;
68 const char *stage_dir = conf->upload_stage_dir_canon[0]
69 ? conf->upload_stage_dir_canon : NULL;
70
71 {
72 brix_staged_open_req_t staged_req = {
73 .root_canon = conf->common.root_canon,
74 .final_path = path,
75 .mode = NGX_FILE_DEFAULT_ACCESS,
76 .principal = principal,
77 .stage_dir = stage_dir,
78 };
79 if (brix_staged_open_resume(r->connection->log, &staged_req, &staged, &cur)
80 != NGX_OK)
81 {
82 ngx_int_t st = brix_http_map_errno(ngx_errno);
83 webdav_metrics_finalize_request(r,
84 st >= 500 ? NGX_HTTP_INTERNAL_SERVER_ERROR : st);
85 return;
86 }
87
88 /* Append-only contiguity: the chunk must begin at the current partial size.
89 * If it doesn't, report the real resume offset and let the client re-issue —
90 * keep the partial (close without unlink). */
91 if (cr->start != cur) {
92 brix_staged_abort(r->connection->log, conf->common.root_canon,
93 &staged, 0 /* keep */);
94 webdav_set_upload_offset(r, cur);
95 webdav_send_status_only(r, NGX_HTTP_CONFLICT);
96 return;
97 }
98
99 if (r->request_body != NULL) {
100 if (brix_http_body_write_to_fd_at(r, staged.fd, path, &bs, cr->start)
101 != NGX_OK)
102 {
103 /* keep the partial; the client can re-send this chunk */
104 brix_staged_abort(r->connection->log, conf->common.root_canon,
105 &staged, 0 /* keep */);
106 webdav_metrics_finalize_request(r, NGX_HTTP_INTERNAL_SERVER_ERROR);
107 return;
108 }
109 BRIX_WEBDAV_METRIC_ADD(bytes_rx_total, bs.bytes);
110 cur = cr->end + 1;
111 /* Flush so the reported resume offset is durable across a restart. */
112 (void) fsync(staged.fd);
113 }
114
115 /* Final chunk (last byte of the declared total) → commit; else keep partial
116 * and report the next expected offset. */
117 if (cr->total >= 0 && cr->end + 1 >= cr->total) {
118 /* Commit the staged partial onto the destination — atomic rename on the
119 * same filesystem, or copy-then-rename when staging on a different device
120 * (brix_webdav_stage_dir). brix_commit_staged owns neither close nor
121 * the active flag, so finalize the staged struct ourselves. On a stage
122 * device, record a durable pending-commit marker first so an interrupted
123 * cross-device move is finished by the reaper across a restart. */
124 ngx_flag_t stage_tracked = (stage_dir != NULL);
125 ngx_int_t crc;
126 if (stage_tracked) {
127 (void) brix_stage_mark_pending(staged.tmp_path, path,
128 r->connection->log);
129 }
130 /* flags 0 = durable: WebDAV keeps phase-51 semantics unconditionally
131 * for now (brix_durable_commit currently gates the root:// close). */
132 crc = brix_commit_staged(staged.fd, staged.tmp_path, path,
133 staged.final_mode, 0, r->connection->log);
134 if (staged.fd != NGX_INVALID_FILE) {
135 ngx_close_file(staged.fd);
136 staged.fd = NGX_INVALID_FILE;
137 }
138 staged.active = 0;
139 if (crc != NGX_OK) {
140 /* Keep the marker + partial — the reaper / a client retry completes it. */
141 webdav_metrics_finalize_request(r, NGX_HTTP_INTERNAL_SERVER_ERROR);
142 return;
143 }
144 if (stage_tracked) {
145 brix_stage_unmark_pending(staged.tmp_path);
146 }
147 /* C-2 (phase-56): the commit rename just materialised the destination
148 * outside brix_vfs_open/vfs_staged — drop any per-worker cached
149 * negative so a stat racing the publish never sees a stale ENOENT. */
150 brix_vfs_neg_stat_forget(conf->common.root_canon, (const char *) path);
151 webdav_put_persist_checksums(r, (const char *) path); /* §8.3 */
152 webdav_send_status_only(r, NGX_HTTP_CREATED);
153 return;
154 }
155
156 brix_staged_abort(r->connection->log, conf->common.root_canon,
157 &staged, 0 /* keep partial for the next chunk */);
158 webdav_set_upload_offset(r, cur);
159 webdav_send_status_only(r, NGX_HTTP_OK);
160 }
161}

163/*
164 * webdav_put_precheck — resolve the target and run pre-write gating.
165 *
166 * WHAT: Resolves the request path into `path`, dispatches a resumable
167 * Content-Range PUT (which fully handles its own response), probes the target
168 * for existence, and evaluates the ETag preconditions. Returns
169 * WEBDAV_PUT_CONTINUE to proceed to the staged-write path with `*created`
170 * set, or WEBDAV_PUT_DONE when the request has already been answered (path
171 * error → 409/4xx, resumable handled, or an ETag precondition failed).
172 *
173 * WHY: Consolidates every gate that must pass before a staged temp is opened,
174 * keeping the orchestrator a flat sequence and the "never open before we know
175 * we should" invariant in one place. Behavior is byte-identical to the prior
176 * inline block (same status codes, same order of checks).
177 *
178 * HOW:
179 * 1. Resolve the path; a missing intermediate collection is RFC 4918 §9.7.1
180 * 409 Conflict, any other resolve error propagates its status.
181 * 2. If resumable uploads are on and a Content-Range header is present, parse
182 * it (malformed → 400) and hand off to webdav_put_ranged_resume; DONE.
183 * 3. Probe the target through the metered VFS to populate a local stat for the
184 * ETag check and decide `*created` (0 = exists, 1 = new).
185 * 4. Run the weak-ETag precondition check; a failure sends its status. DONE.
186 */
187webdav_put_step_t
188webdav_put_precheck(ngx_http_request_t *r,
189 ngx_http_brix_webdav_loc_conf_t *conf, char *path, int *created)
190{
191 struct stat sb;
192 ngx_int_t rc;
193
194 *created = 0;
195 ngx_memzero(&sb, sizeof(sb));
196
197 rc = ngx_http_brix_webdav_resolve_path(r, conf->common.root_canon, path,
198 WEBDAV_MAX_PATH);
199 if (rc != NGX_OK) {
200 /* RFC 4918 §9.7.1: missing intermediate collection → 409 Conflict */
201 if (rc == NGX_HTTP_NOT_FOUND) {
202 webdav_metrics_finalize_request(r, NGX_HTTP_CONFLICT);
203 return WEBDAV_PUT_DONE;
204 }
205 webdav_metrics_finalize_request(r, rc);
206 return WEBDAV_PUT_DONE;
207 }
208
209 /*
210 * Resumable upload: a Content-Range PUT places its chunk at an absolute
211 * offset on a persistent identity-keyed partial and commits only when the
212 * upload completes (brix_webdav_upload_resume on). Handled before the
213 * whole-body staged-write path below. A malformed Content-Range is 400.
214 */
215 if (conf->common.upload_resume) {
216 ngx_table_elt_t *crh = brix_http_find_header(
217 r, "Content-Range", sizeof("Content-Range") - 1);
218 if (crh != NULL && crh->value.len > 0) {
219 brix_http_content_range_t cr;
220 brix_http_parse_content_range(crh->value.data, crh->value.len, &cr);
221 if (!cr.present) {
222 webdav_metrics_finalize_request(r, NGX_HTTP_BAD_REQUEST);
223 return WEBDAV_PUT_DONE;
224 }
225 webdav_put_ranged_resume(r, conf, path, &cr);
226 return WEBDAV_PUT_DONE;
227 }
228 }
229
230 {
231 ngx_http_brix_webdav_req_ctx_t *rx =
232 ngx_http_get_module_ctx(r, ngx_http_brix_webdav_module);
233 brix_vfs_ctx_t pctx;
234 brix_vfs_stat_t pst;
235 int is_tls = brix_http_request_is_tls(r);
236 brix_vfs_ctx_init(&pctx, r->pool, r->connection->log,
237 BRIX_PROTO_WEBDAV, conf->common.root_canon, conf->common.cache_root_canon,
238 brix_vfs_policy_from_write_enable(conf->common.allow_write),
239 is_tls,
240 (rx != NULL) ? rx->identity : NULL, path);
241 brix_vfs_ctx_bind_backend_cred(&pctx,
242 &conf->common.storage_credential_dir,
243 conf->common.storage_credential_fallback);
244 webdav_vfs_bind_deleg(r, conf, &pctx);
245 if (brix_vfs_probe(&pctx, 0 /* follow */, &pst) == NGX_OK) {
246 *created = 0; /* target exists — sb populated for the ETag check */
247 sb.st_mtime = pst.mtime;
248 sb.st_size = pst.size;
249 sb.st_mode = (mode_t) pst.mode;
250 } else {
251 *created = 1; /* new file — sb unused (etag check sees !exists) */
252 }
253 }
254
255 rc = brix_http_check_etag_preconditions(
256 r, !*created, &sb, BRIX_ETAG_WEAK, BRIX_HTTP_COND_WEAK_EQUIV);
257 if (rc != NGX_OK) {
258 webdav_metrics_finalize_request(r, rc);
259 return WEBDAV_PUT_DONE;
260 }
261
262 return WEBDAV_PUT_CONTINUE;
263}

265/*
266 * webdav_put_open_target — open the unified write session for the final path.
267 *
268 * WHAT: Builds the staged-write VFS ctx and opens a brix_vfs_writer session on
269 * it with BRIX_VFS_O_ATOMIC (force the staged temp+publish path so a failed
270 * write never leaves a partial at the final path) and the export's
271 * verify-on-write setting. On success returns WEBDAV_PUT_CONTINUE with
272 * `*out_writer` set; on failure maps the captured errno to a clean 4xx (or 500
273 * for no-4xx-contract errnos), sends it, and returns WEBDAV_PUT_DONE.
274 *
275 * WHY: Keeps the "body is written to a staged temp and atomically renamed only
276 * on success" invariant (WS8/HTTP-4) and its errno→status mapping in one
277 * place, matching S3 PUT's s3_put_finalize_fs_error. Routing the write
278 * through the writer gives WebDAV the same common verified-write call every
279 * backend shares. Status codes unchanged.
280 *
281 * HOW:
282 * 1. Init the staged-write ctx (canonical data-plane build + backend route).
283 * 2. brix_vfs_writer_open(O_ATOMIC, verify_write); a non-NULL result succeeds.
284 * 3. ENOENT/ENOTDIR (missing parent collection) → 409 (RFC 4918 §9.7.1).
285 * 4. Otherwise map errno via brix_http_map_errno, log the path, and finalize
286 * with the mapped 4xx or 500.
287 */
288webdav_put_step_t
289webdav_put_open_target(ngx_http_request_t *r,
290 ngx_http_brix_webdav_loc_conf_t *conf, const char *path,
291 brix_vfs_ctx_t *vctx, brix_vfs_writer_t **out_writer)
292{
293 brix_vfs_writer_t *writer;
294 int staged_err = 0;
295 ngx_int_t status;
296
297 *out_writer = NULL;
298
299 /*
300 * Open a write session that stages a temp beside the final path (O_ATOMIC,
301 * O_EXCL, unique name, up to 16 attempts) through the metered VFS surface.
302 * The body is written here and atomically renamed onto the final path only
303 * on success — never exposing a partial object. verify_write folds a
304 * read-back integrity check on commit when the export opts in.
305 */
306 webdav_vfs_ctx_build_data(r, conf, path, vctx);
307 /* Route through the export's selected storage backend (NULL ⇒ POSIX).
308 * PUT is allow_write-gated at the access phase, so the staged open's
309 * write-gate never fires here. */
310 vctx->sd = brix_webdav_backend_instance(conf, r->connection->log);
311 /* Phase-107 C5: a PUT's Content-Length IS the declared final size — the
312 * staged plane forwards it (remote derives a legal multipart part size,
313 * posix/frm preallocate) and an unsatisfiable declaration fails the open
314 * with ENOSPC → 507 before the body streams. Chunked (-1) declares none. */
315 if (r->headers_in.content_length_n > 0) {
316 vctx->declared_size = (off_t) r->headers_in.content_length_n;
317 }
318
319 writer = brix_vfs_writer_open(vctx, BRIX_VFS_O_ATOMIC,
320 conf->common.verify_write, &staged_err);
321 if (writer != NULL) {
322 *out_writer = writer;
323 return WEBDAV_PUT_CONTINUE;
324 }
325
326 /* Use the errno the VFS open captured (logging may clobber errno). */
327 {
328 int open_errno = staged_err;
329
330 if (open_errno == NGX_ENOENT || open_errno == NGX_ENOTDIR) {
331 /* RFC 4918 §9.7.1 — PUT to non-existent parent collection is 409 */
332 webdav_metrics_finalize_request(r, NGX_HTTP_CONFLICT);
333 return WEBDAV_PUT_DONE;
334 }
335 /*
336 * A DAC-denied staged-temp create (e.g. the accessor lacks write on the
337 * parent collection → EACCES/EPERM, EXDEV/ELOOP on a confinement-blocked
338 * traversal) is a forbidden, bounded request — surface it as a clean 4xx
339 * via the shared errno map (EACCES/EPERM → 403, ENOSPC → 507, …), not a
340 * blanket 500. Only an errno with no clean 4xx contract (e.g. EIO) keeps
341 * the internal-error path. This mirrors S3 PUT's s3_put_finalize_fs_error.
342 */
343 status = brix_http_map_errno(open_errno);
344 brix_log_safe_path(r->connection->log, NGX_LOG_ERR,
345 (ngx_uint_t) open_errno,
346 "brix_webdav: staged open for write failed for: "
347 "\"%s\"", path);
348 webdav_metrics_finalize_request(
349 r, status >= 500 ? NGX_HTTP_INTERNAL_SERVER_ERROR : status);
350 return WEBDAV_PUT_DONE;
351 }
352}
