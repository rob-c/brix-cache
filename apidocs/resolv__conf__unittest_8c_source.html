1/*
2 * resolv_conf_unittest.c — standalone unit test for the resolv.conf parser
3 * (phase-116 W1). Built and run by tests/cmdscripts/c_regression_units.py
4 * (dns_resolv_conf) and by tests/test_phase116_resolv_conf_parser.py.
5 *
6 * WHAT: Exercises defaults, keyword parsing, limits, comments, env overrides
7 * the missing-file path, an unreadable path (a directory) and a
8 * symlinked file against glibc's documented behaviour.
9 * WHY: The parser is the single truth for "what the host resolver would do";
10 * a drift here silently changes which nameserver every brix worker asks.
11 * HOW: Plain assert-style checks with a counter; exit status 0 = all pass.

66 CHECK(rc.nsearch == 2); /* last wins */

69 CHECK(rc.ndots == 3 && rc.timeout == 7 && rc.attempts == 4);
70 CHECK(rc.rotate == 1);

74/* Only address literals reach nginx's resolver: anything glibc would
75 * skip is skipped here too, ":port" and "[v6]:port" are kept, a zone is
76 * dropped whether bare or bracketed. */
77static void
78test_nameserver_literals(void)
79{
80 brix_resolv_conf_t rc;
81 const char *text =
82 "nameserver\n"
83 "nameserver not-an-ip\n"
84 "nameserver 999.1.1.1\n"
85 "nameserver 10.0.0.2:0\n"
86 "nameserver 10.0.0.2:x\n"
87 "nameserver 10.0.0.2:70000\n"
88 "nameserver [fe80::1\n"
89 "nameserver [fe80::1]x\n"
90 "nameserver [not-an-ip]:53\n"
91 "nameserver 10.0.0.1:5353\n"
92 "nameserver [::1]:5354\n"
93 "nameserver [fe80::1%eth0]\n";
94
95 brix_resolv_conf_defaults(&rc);
96 brix_resolv_conf_parse_text(&rc, text, strlen(text));
97 CHECK(rc.nnameservers == 3);
98 CHECK(strcmp(rc.nameservers[0], "10.0.0.1:5353") == 0);
99 CHECK(strcmp(rc.nameservers[1], "[::1]:5354") == 0);
100 CHECK(strcmp(rc.nameservers[2], "fe80::1") == 0);
101
102 /* a bracketed zone with a port is re-assembled without the zone */
103 text = "nameserver [fe80::1%eth0]:53\n";
104 brix_resolv_conf_defaults(&rc);
105 brix_resolv_conf_parse_text(&rc, text, strlen(text));
106 CHECK(rc.nnameservers == 1);
107 CHECK(strcmp(rc.nameservers[0], "[fe80::1]:53") == 0);
108
109 /* every nameserver line unparseable: none left, the default is NOT
110 * silently restored (the builder warns and falls back to libc) */
111 text = "nameserver nope\nsearch lab.example\n";
112 brix_resolv_conf_defaults(&rc);
113 brix_resolv_conf_parse_text(&rc, text, strlen(text));
114 CHECK(rc.nnameservers == 0);
115 CHECK(rc.nsearch == 1);
116
117 /* no nameserver line at all: the 127.0.0.1 default stands (glibc) */
118 text = "search lab.example\noptions ndots:2\n";
119 brix_resolv_conf_defaults(&rc);
120 brix_resolv_conf_parse_text(&rc, text, strlen(text));
121 CHECK(rc.nnameservers == 1);
122 CHECK(strcmp(rc.nameservers[0], "127.0.0.1") == 0);
123}

174test_file(void)

176 brix_resolv_conf_t rc;

193 CHECK(rc.nsearch == 1 && strcmp(rc.search[0], "lab.example") == 0);
194 unlink(path);
195}

198/* A path that opens but yields no readable content (a directory) is the
199 * operator's path being unusable, not an empty resolv.conf: the loader must
200 * fail so the caller warns. A genuinely empty regular file is a valid
201 * resolv.conf and keeps glibc's 127.0.0.1 default silently.
202 */
203static void
204test_unreadable_content(void)
205{
206 brix_resolv_conf_t rc;
207 char dir[] = "/tmp/brix_resolv_ut_dir_XXXXXX";
208 char path[] = "/tmp/brix_resolv_ut_empty_XXXXXX";
209 int fd;
210
211 unsetenv("LOCALDOMAIN");
212 unsetenv("RES_OPTIONS");
213
214 CHECK(mkdtemp(dir) != NULL);
215 CHECK(brix_resolv_conf_load(&rc, dir) == -1);
216 CHECK(rc.defaulted == 1 && rc.nnameservers == 1);
217 CHECK(strcmp(rc.nameservers[0], "127.0.0.1") == 0);
218 rmdir(dir);
219
220 fd = mkstemp(path);
221 CHECK(fd >= 0);
222 if (fd < 0) {
223 return;
224 }
225 close(fd);
226 CHECK(brix_resolv_conf_load(&rc, path) == 0);
227 CHECK(rc.defaulted == 0 && rc.nnameservers == 1);
228 CHECK(strcmp(rc.nameservers[0], "127.0.0.1") == 0);
229 unlink(path);
230}

233/* A symlink is followed to its target's contents: /etc/resolv.conf is one on
234 * every systemd host (phase-116 amendment 1).
235 */
236static void
237test_symlinked_file(void)
238{
239 brix_resolv_conf_t rc;
240 char path[] = "/tmp/brix_resolv_ut_tgt_XXXXXX";
241 char link[sizeof(path) + 5];
242 const char *text = "nameserver 192.0.2.54\n";
243 int fd = mkstemp(path);
244
245 CHECK(fd >= 0);
246 if (fd < 0) {
247 return;
248 }
249 CHECK(write(fd, text, strlen(text)) == (ssize_t) strlen(text));
250 close(fd);
251 snprintf(link, sizeof(link), "%s.link", path);
252 CHECK(symlink(path, link) == 0);
253 unsetenv("LOCALDOMAIN");
254 unsetenv("RES_OPTIONS");
255 CHECK(brix_resolv_conf_load(&rc, link) == 0);
256 CHECK(rc.nnameservers == 1
257 && strcmp(rc.nameservers[0], "192.0.2.54") == 0);
258 unlink(link);
259 unlink(path);
260}
