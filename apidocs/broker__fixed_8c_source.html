13/* WHAT: Compare complete credentials, independent of group enumeration order.
14 * WHY: A uid match alone could retain removed or foreign supplementary grants.
15 * HOW: Require equal uid/gid/count and membership for every bounded group. */
16int
17brix_imp_fixed_creds_equal(const brix_idmap_creds_t *left,
18 const brix_idmap_creds_t *right)
19{
20 uint64_t matched = 0;
21
22 if (left->uid != right->uid || left->gid != right->gid
23 || left->ngroups != right->ngroups || left->ngroups < 0
24 || left->ngroups > BRIX_IDMAP_MAXGROUPS)
25 {
26 return 0;
27 }
28 for (int index = 0; index < left->ngroups; index++) {
29 int present = 0;
30 for (int other = 0; other < right->ngroups; other++) {
31 uint64_t bit = UINT64_C(1) << other;
32 if (!(matched & bit) && left->groups[index] == right->groups[other]) {
33 matched |= bit;
34 present = 1;
35 break;
36 }
37 }
38 if (!present) { return 0; }
39 }
40 return 1;
41}

45/* WHAT: Release only this pool's descriptors and their kernel-owned children.
46 * WHY: Supervisor failure must not strand helpers or kill reused numeric pids.
47 * HOW: Close each channel and process descriptor, then free the pool. */
48void
49brix_imp_fixed_pool_close(brix_imp_fixed_pool_t *pool)
50{
51 if (pool == NULL) { return; }
52 for (size_t index = 0; index < pool->count; index++) {
53 if (pool->entries[index].channel >= 0) {
54 close(pool->entries[index].channel);
55 }
56 close(pool->entries[index].process_fd);
57 }
58 ngx_free(pool);
59}

61/* WHAT: Snapshot potentially eligible account names without nesting NSS scans.
62 * WHY: getpwnam/getgrouplist can invalidate getpwent's shared iteration state.
63 * HOW: Copy bounded names, excluding obvious reserved/service identities;
64 * membership/deny-list resolution happens only after endpwent. */
65static int
66fixed_account_names(char (*names)[IMP_PRINC_MAX], size_t *count)
67{
68 struct passwd *account;
69
70 *count = 0;
71 setpwent();
72 for (;;) {
73 errno = 0;
74 account = getpwent();
75 if (account == NULL) { break; }
76 if (account->pw_uid < brix_idmap_get_min_uid()
77 || account->pw_uid == brix_imp_get_broker_user_uid()
78 || account->pw_uid == brix_imp_get_broker_allow_uid())
79 {
80 continue;
81 }
82 size_t length = strlen(account->pw_name);
83 if (*count == BRIX_IMP_FIXED_MAX || length >= IMP_PRINC_MAX) {
84 endpwent();
85 errno = E2BIG;
86 return -1;
87 }
88 ngx_memcpy(names[(*count)++], account->pw_name, length + 1);
89 }
90 int status = errno == 0 ? 0 : -1;
91 endpwent();
92 return status;
93}

95/* WHAT: Add one permitted complete identity, deduplicating true aliases.
96 * WHY: Both mapping policy and the syscall-edge hard floor remain authoritative.
97 * HOW: Resolve via the existing mapper, reject forbidden credentials, then
98 * spawn a bounded helper only when no exact identity already exists. */
99static int
100fixed_add_account(brix_imp_fixed_pool_t *pool, const char *name, int rootfd,
101 ngx_log_t *log)
102{
103 brix_idmap_creds_t creds;
104 size_t index = pool->count;
105
106 ngx_memzero(&creds, sizeof(creds));
107 if (idmap_resolve_user(name, &creds) != 0 || !idmap_creds_allowed(&creds)
108 || brix_imp_creds_privileged(&creds, BRIX_IMP_HARD_MIN_ID, NULL, NULL))
109 {
110 return 0;
111 }
112 for (size_t prior = 0; prior < index; prior++) {
113 if (brix_imp_fixed_creds_equal(&creds, &pool->entries[prior].creds)) {
114 return 0;
115 }
116 }
117 if (index == BRIX_IMP_FIXED_MAX) { errno = E2BIG; return -1; }
118 if (brix_imp_fixed_spawn(rootfd, &creds, &pool->entries[index].channel,
119 &pool->entries[index].process_fd, log) != 0)
120 {
121 return -1;
122 }
123 pool->entries[index].creds = creds;
124 pool->count++;
125 return 0;
126}

128/* WHAT: Build all helper identities from a bounded, frozen account snapshot.
129 * WHY: No runtime request may require the supervisor to regain root or fork it.
130 * HOW: Own the name scratch allocation, enumerate first, then resolve/spawn. */
131static int
132fixed_populate(brix_imp_fixed_pool_t *pool, int rootfd, ngx_log_t *log)
133{
134 char (*names)[IMP_PRINC_MAX];
135 size_t count = 0;
136 int status;
137
138 names = ngx_alloc(BRIX_IMP_FIXED_MAX * IMP_PRINC_MAX, log);
139 if (names == NULL) { return -1; }
140 status = fixed_account_names(names, &count);
141 for (size_t index = 0; status == 0 && index < count; index++) {
142 status = fixed_add_account(pool, names[index], rootfd, log);
143 }
144 ngx_free(names);
145 return status;
146}

148/* WHAT: Start fixed helpers then irreversibly drop the supervising broker.
149 * WHY: Neither the supervisor nor any serving helper may retain saved root.
150 * HOW: Require a dedicated service account, populate helpers, drop all service
151 * uid/gid slots, close the supervisor's export root and report its identity. */
152brix_imp_fixed_pool_t *
153brix_imp_fixed_pool_start(int listenfd, int rootfd, ngx_log_t *log)
154{
155 brix_imp_fixed_pool_t *pool;
156 brix_idmap_creds_t service;
157
158 (void) listenfd;
159 ngx_memzero(&service, sizeof(service));
160 service.uid = brix_imp_get_broker_user_uid();
161 service.gid = brix_imp_get_broker_user_gid();
162 service.ngroups = 1;
163 service.groups[0] = service.gid;
164 if (getuid() != 0 || service.uid == (uid_t) -1 || service.uid == 0
165 || service.gid == (gid_t) -1 || service.gid == 0)
166 {
167 BRIX_LOG(NGX_LOG_EMERG, log, 0,
168 "impersonate: fixed-identity broker requires a dedicated non-root "
169 "brix_idmap_broker_user; refusing a root-equivalent fallback");
170 errno = EPERM;
171 return NULL;
172 }
173 pool = ngx_alloc(sizeof(*pool), log);
174 if (pool == NULL) { return NULL; }
175 ngx_memzero(pool, sizeof(*pool));
176 if (fixed_populate(pool, rootfd, log) != 0
177 || brix_imp_fixed_drop(&service) != 0)
178 {
179 BRIX_LOG(NGX_LOG_EMERG, log, ngx_errno,
180 "impersonate: fixed-identity helper setup failed; refusing to serve");
181 brix_imp_fixed_pool_close(pool);
182 return NULL;
183 }
184 close(rootfd);
185 BRIX_LOG(NGX_LOG_NOTICE, log, 0,
186 "impersonate: non-root fixed-identity supervisor uid=%d helpers=%uz; "
187 "all serving identities permanently dropped root",
188 (int) service.uid, pool->count);
189 return pool;
190}

192/* WHAT: Forward an authorized operation to its exact fixed-identity helper.
193 * WHY: New/group-changed identities or dead helpers must never fall back to
194 * service/root I/O, and mutations must not be replayed after transport failure.
195 * HOW: Match complete credentials, exchange once through the shared codec;
196 * permanently close a broken channel and return an explicit I/O error. */
197int
198brix_imp_fixed_pool_dispatch(brix_imp_fixed_pool_t *pool,
199 const brix_idmap_creds_t *creds, imp_op_ctx_t *operation)
200{
201 for (size_t index = 0; index < pool->count; index++) {
202 if (!brix_imp_fixed_creds_equal(creds, &pool->entries[index].creds)) {
203 continue;
204 }
205 int channel = pool->entries[index].channel;
206 if (channel < 0) { return -EIO; }
207 if (imp_write_full(channel, operation->req, sizeof(*operation->req)) == 0
208 && (operation->data_in_len == 0
209 || imp_write_full(channel, operation->data_in,
210 operation->data_in_len) == 0)
211 && imp_recv_reply(channel, operation->rep, operation->out_fd,
212 operation->data_out, operation->data_max) == 0)
213 {
214 return operation->rep->status;
215 }
216 close(channel);
217 pool->entries[index].channel = -1;
218 return -EIO;
219 }
220 return -EACCES;
221}
