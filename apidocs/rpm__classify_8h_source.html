1/* rpm_classify.h — the RPM/dnf repository route classifier (phase-104 D15.9).
2 *
3 * WHAT: turn one decoded request URI into a typed repository route — which of
4 * the four object classes a dnf client is asking for, and, for the
5 * digest-named metadata files, the checksum their own name carries.
6 * WHY: an RPM repository is static HTTP with exactly one interesting
7 * property: the file NAMES say which objects are immutable. repomd.xml
8 * is the mutable freshness root; everything createrepo writes beside it
9 * is `<checksum>-<name>`, so its name IS its digest; packages are
10 * immutable in practice and verified client-side. Every cache decision
11 * the mirror makes — TTL vs forever, verify-at-edge vs trust — is that
12 * one distinction, so it is read exactly once, here, at the edge. The
13 * traversal defense rides along: a path that classifies cannot escape
14 * the store, because the grammar rejects every component that could.
15 * HOW: pure C over the decoded URI — no nginx types, no allocation, spans
16 * point into the caller's buffer. That is what lets the fill-side
17 * verify (src/fs/cache/verify.c) and the cache TTL policy
18 * (sd_cache_policy.c) re-read a cache key with the SAME grammar the
19 * gate used, rather than each growing its own idea of what a repodata
20 * path looks like.
21 */
22#ifndef BRIX_PROTOCOLS_RPM_CLASSIFY_H
23#define BRIX_PROTOCOLS_RPM_CLASSIFY_H
24
25#include <stddef.h>
26
27/* The canonical key is the request URI verbatim (§D15.9.3), so the cap is a
28 * path cap: mirror trees nest deep (dist/version/repo/arch/os/Packages/x/…)
29 * but never past this, and an over-long path is refused rather than truncated
30 * into a different object's name. */
31#define BRIX_RPM_PATH_MAX 1024
32
33typedef enum {
34 BRIX_RPM_REQ_REPOMD = 0, /* …/repodata/repomd.xml[.asc|.key] — mutable */
35 BRIX_RPM_REQ_METADATA, /* …/repodata/<hex>-<name> — self-addressing */
36 BRIX_RPM_REQ_PACKAGE, /* …/<pkg>.rpm | .drpm — immutable */
37 BRIX_RPM_REQ_AUX, /* every other legal repo file — mutable */
38 BRIX_RPM_REQ_BAD, /* refused; `err` says why */
39 BRIX_RPM_REQ_COUNT
40} brix_rpm_class_t;
41
42typedef enum {
43 BRIX_RPM_ERR_NONE = 0,
44 BRIX_RPM_ERR_PATH_INVALID, /* traversal, empty component, bad byte */
45 BRIX_RPM_ERR_PATH_TOO_LONG, /* past BRIX_RPM_PATH_MAX */
46 BRIX_RPM_ERR_NOT_A_FILE /* directory-shaped: a mirror serves files */
47} brix_rpm_err_t;
48
49typedef struct {
50 brix_rpm_class_t cls;
51 brix_rpm_err_t err; /* set iff cls == _BAD */
52 const char *file; size_t file_len; /* the last component */
53 const char *hex; size_t hex_len; /* METADATA: its own checksum */
54} brix_rpm_req_t;
55
56/* Classify `uri` (DECODED, NUL-termination not required). Returns 0 with
57 * out->cls naming a route, or -1 with out->cls == BRIX_RPM_REQ_BAD and
58 * out->err naming the refusal. `uri` must outlive `*out`: the spans point
59 * into it. */
60int brix_rpm_classify(const char *uri, size_t len, brix_rpm_req_t *out);
61
62/* The checksum algorithm a digest-named metadata file is named under, decided
63 * by the hex length createrepo wrote — sha1 (40), sha256 (64), sha384 (96),
64 * sha512 (128). NULL for any other length, which is what keeps a file merely
65 * BEGINNING with hex digits from being verified against the wrong function. */
66const char *brix_rpm_alg_for_hexlen(size_t hex_len);
67
68/* Stable lowercase token for the route — metric label and log field. */
69const char *brix_rpm_class_str(brix_rpm_class_t cls);
70
71/* 1 iff this class is MUTABLE and must therefore expire on a TTL rather than
72 * live in the cache forever. The mirror, the cache TTL policy and the docs
73 * all ask this question; asking it in one place is what stops them drifting. */
74int brix_rpm_class_is_mutable(brix_rpm_class_t cls);
75
76#endif /* BRIX_PROTOCOLS_RPM_CLASSIFY_H */
