1/*
2 * sd_xroot_fwd_ns.c — path-keyed relays of the forwarding driver.
3 *
4 * WHAT: Every namespace slot resolves the key to a child origin, then hands
5 * the origin-relative path to that child through the shared
6 * brix_sd_*_maybe_cred forwarders (sd_cred_forward.h).
7 * WHY: The forwarders own the cred/fallback_deny/ENOSYS contract, so a
8 * forwarded call behaves exactly as a call on a fixed root:// export;
9 * the plain slots are their `_cred` twins with no credential.
10 * HOW: Two-key operations (rename, server_copy) resolve both keys and refuse
11 * EXDEV unless they land on the same origin — an origin cannot move a
12 * file it does not hold, and the wire maps EXDEV to kXR_NotAuthorized.
13 */
14#include "sd_xroot_fwd_internal.h"
15
16#include <errno.h>
17
18/* SD_XROOT_FWD_HOP — the whole body of a one-key relay: resolve the slot's
19 * `key` parameter against `inst` (errno set by the resolver on refusal),
20 * return `errval` when it refuses, else RETURN the delegated `call`, which
21 * names the origin child as `child` and the origin-relative path as `t.path`.
22 * Same shape as PBLOCK_CRED_GATED (sd_pblock_cred.c). */
23#define SD_XROOT_FWD_HOP(key, errval, call) \
24 do { \
25 brix_sd_xroot_fwd_target_t t; \
26 brix_sd_instance_t *child = sd_xroot_fwd_resolve(inst, key, \
27 &t); \
28 if (child == NULL) { \
29 return errval; \
30 } \
31 return call; \
32 } while (0)

77/* Resolve a two-key operation: both keys must land on one origin. */
78static brix_sd_instance_t *
79sd_xroot_fwd_resolve_pair(brix_sd_instance_t *inst, const char *src,
80 const char *dst, brix_sd_xroot_fwd_target_t *ts,
81 brix_sd_xroot_fwd_target_t *td)
82{
83 brix_sd_instance_t *child_src = sd_xroot_fwd_resolve(inst, src, ts);
84 brix_sd_instance_t *child_dst;
85
86 if (child_src == NULL) {
87 return NULL;
88 }
89 child_dst = sd_xroot_fwd_resolve(inst, dst, td);
90 if (child_dst == NULL) {
91 return NULL;
92 }
93 if (child_src != child_dst) {
94 errno = EXDEV;
95 return NULL;
96 }
97 return child_src;
98}
