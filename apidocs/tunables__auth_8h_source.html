9/*
10 * seccomp-BPF worker syscall filter (hyper-hardening-plan §D-3).
11 *
12 * A per-worker syscall allowlist installed at the tail of init_process, after
13 * every setup syscall has run, so only the steady-state serving set must be
14 * enumerated. Mode enum (stored in ngx_stream_brix_srv_conf_t.seccomp, set via
15 * an ngx_conf_enum_t slot; the strictest value across enabled server blocks wins
16 * for the process):
17 * OFF = no filter installed (default — strictly opt-in).
18 * AUDIT = filter loaded with a log-only default action: allowlisted syscalls
19 * run silently, everything else is ALLOWED but logged to the kernel
20 * audit log (SECCOMP … audit records). Converges the set risk-free.
21 * ENFORCE = allowlisted syscalls run; the named-dangerous set (execve/execveat/
22 * ptrace/process_vm_*) is KILLED; any other non-allowlisted syscall
23 * fails EPERM (fail-safe: a missed entry degrades one call, it does
24 * not crash the worker).
25 *
26 * Requires a build with libseccomp (config sets -DBRIX_HAVE_SECCOMP); without it
27 * AUDIT/ENFORCE fail closed at init (the worker refuses to run rather than serve
28 * unfiltered while the operator believes it is filtered).
29 */
30#define BRIX_SECCOMP_OFF 0
31#define BRIX_SECCOMP_AUDIT 1
32#define BRIX_SECCOMP_ENFORCE 2

34/*
35 * Phase 33 — GSI ephemeral DH key pool (src/gsi/keypool.c).
36 *
37 * Per-worker warm pool of pre-generated ffdhe2048 keys so kXGC_certreq never runs
38 * keygen on the nginx event thread under a concurrent handshake burst. The target
39 * warm count (SIZE_DEFAULT, runtime-tunable via brix_gsi_keypool_size) is the
40 * refill ceiling, sized to cover a worst-case concurrent-handshake burst; when the
41 * pool falls to half the target an off-thread refill of REFILL_BATCH keys is
42 * scheduled. Each key is ~1-2 KB → target keys ≈ 100-130 KB per worker.
43 *
44 * SEED_DEFAULT keys are generated synchronously at worker start (brix_gsi_
45 * keypool_seed); the remainder up to the target fills OFF the event thread so boot
46 * is not blocked on ~target keygens (was the dominant per-worker startup cost). CAP
47 * bounds the static ring so the target directive cannot be set unboundedly high.
48 */
49#define BRIX_GSI_KEYPOOL_CAP 256 /* static ring capacity (ceiling) */
50#define BRIX_GSI_KEYPOOL_SIZE_DEFAULT 64 /* default warm/refill target */
51#define BRIX_GSI_KEYPOOL_SEED_DEFAULT 4 /* default synchronous boot seed */
52#define BRIX_GSI_KEYPOOL_REFILL_BATCH 32

54/*
55 * Clock-skew tolerance for JWT nbf/exp validation.
56 * Even with NTP, production systems commonly drift 1–5 seconds. The WLCG
57 * Token Profile recommends that servers accept a small grace window so that
58 * freshly-issued tokens are not rejected by a server whose clock lags slightly.
59 * 30 seconds is generous enough for any reasonable NTP configuration.
60 */
