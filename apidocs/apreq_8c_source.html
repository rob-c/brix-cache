3/*
4 * apreq.c — raw-krb5 AP-REQ builder for the outbound origin leg (§5.7).
5 *
6 * Implements the apreq.h contract: turn a delegated user's TGT (carried as a
7 * ccache PATH) plus the origin service principal into the "krb5\0"+AP-REQ payload
8 * a stock XRootD / brix krb5 acceptor validates with krb5_rd_req. This is the
9 * RAW-krb5 dialect the reference implementation actually speaks — the GSSAPI
10 * forwarding engine in forward.c speaks gss_init_sec_context tokens instead and
11 * cannot authenticate to a real "&P=krb5" origin. The step order mirrors the
12 * native client's krb5_acquire() (client/lib/auth/sec/sec_krb5.c) so the two
13 * emit byte-identical credentials. Without BRIX_HAVE_KRB5 the file still compiles:
14 * the entry reports unavailable and returns NGX_ERROR.
15 */

37/*
38 * Acquire the ccache/principals/ticket and render the "krb5\0"+AP-REQ payload.
39 *
40 * Every fallible krb5 step runs here on a created context, writing the acquired
41 * resources back through out-params (locals start NULL/zero) so the caller can
42 * run one linear NULL-safe cleanup — the same shape as the client's krb5_acquire.
43 * Never frees; ownership of every resource stays with the caller.
44 */
45static ngx_int_t
46apreq_acquire(krb5_context ctx, ngx_pool_t *pool, const char *ccache_path,
47 const char *origin_spn, krb5_ccache *cc, krb5_principal *client,
48 krb5_principal *server, krb5_creds *in_creds, krb5_creds **out_creds,
49 krb5_auth_context *auth, krb5_data *apreq, ngx_str_t *out_payload,
50 ngx_log_t *log)
51{
52 krb5_error_code krc;
53 u_char *p;
54
55 krc = krb5_cc_resolve(ctx, ccache_path, cc);
56 if (krc != 0) {
57 brix_krb5_apreq_log(ctx, krc, "resolve delegated ccache", log);
58 return NGX_ERROR;
59 }
60 krc = krb5_cc_get_principal(ctx, *cc, client);
61 if (krc != 0) {
62 brix_krb5_apreq_log(ctx, krc, "delegated ccache has no principal", log);
63 return NGX_ERROR;
64 }
65 krc = krb5_parse_name(ctx, origin_spn, server);
66 if (krc != 0) {
67 brix_krb5_apreq_log(ctx, krc, "bad origin service principal", log);
68 return NGX_ERROR;
69 }
70
71 in_creds->client = *client;
72 in_creds->server = *server;
73 krc = krb5_get_credentials(ctx, 0, *cc, in_creds, out_creds);
74 if (krc != 0) {
75 brix_krb5_apreq_log(ctx, krc, "cannot get origin service ticket", log);
76 return NGX_ERROR;
77 }
78 krc = krb5_mk_req_extended(ctx, auth, 0, NULL, *out_creds, apreq);
79 if (krc != 0) {
80 brix_krb5_apreq_log(ctx, krc, "mk_req failed", log);
81 return NGX_ERROR;
82 }
83
84 /* "krb5\0" (name as a NUL-terminated string, per XrdSecInterface) then the
85 * raw AP-REQ — exactly the native client's wire (sec_krb5.c krb5_acquire). */
86 p = ngx_pnalloc(pool, 5 + apreq->length);
87 if (p == NULL) {
88 return NGX_ERROR;
89 }
90 ngx_memcpy(p, "krb5", 5);
91 ngx_memcpy(p + 5, apreq->data, apreq->length);
92 out_payload->data = p;
93 out_payload->len = 5 + (size_t) apreq->length;
94 return NGX_OK;
95}
