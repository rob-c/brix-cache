1/*
2 * cms/perf_pgm.c — §2.11: external machine-load feed (stock cms.perf pgm).
3 *
4 * WHAT: Spawns the operator's load program once per CMS-client worker with
5 * its stdout on a pipe; each stdout line "cpu net xeq mem pag" (five 0-100
6 * integers, whitespace-separated) becomes the current machine-load override
7 * consumed by the heartbeat (send.c calls brix_cms_perf_get before falling
8 * back to the /proc meter). A dead program is respawned with a 5s backoff;
9 * an override older than 2x brix_cms_perf_interval goes stale and the /proc
10 * meter takes back over — a wedged feed degrades, never lies.
11 *
12 * WHY: Stock cms.perf pgm parity — sites feed cmsd load figures from custom
13 * monitors (batch-system pressure, GPU occupancy, SLA state) that /proc
14 * cannot see. The pgm form is NOT part of the excluded plugin architecture:
15 * it is a config-driven child process, in scope per the audit.
16 *
17 * HOW: pipe2(O_NONBLOCK|O_CLOEXEC) + posix_spawn("/bin/sh -c <pgm>") with
18 * stdout dup2'd onto the pipe. The read end is wrapped in an
19 * ngx_connection_t so the worker's event loop drives line parsing — no
20 * threads, no blocking reads. State is a per-worker static (one feed per
21 * worker running the CMS client, which is worker 0 — cms_start.c).
22 */

52/*
53 * perf_skip_blanks — first non-blank offset at or after a cursor.
54 *
55 * WHAT: Returns the first offset >= cursor whose byte is neither space nor
56 * tab, or len when the remainder of the line is blank.
57 * WHY: Both the field walk and the trailing-junk check need it, and they must
58 * agree on what a column separator is — a monitor may pad columns, but
59 * nothing else counts as blank.
60 * HOW: 1. Advance while in range and the byte is ' ' or '\t'.
61 * 2. Return the offset reached.
62 */
63static size_t
64perf_skip_blanks(const u_char *line, size_t len, size_t cursor)
65{
66 while (cursor < len && (line[cursor] == ' ' || line[cursor] == '\t')) {
67 cursor++;
68 }
69 return cursor;
70}

73/*
74 * perf_parse_field — one blank-prefixed decimal column, clamped to 100.
75 *
76 * WHAT: Reads the digit run starting at the first non-blank byte at or after
77 * *cursor, stores it clamped to 100 in *out and advances *cursor past
78 * it. Returns 0 on success, -1 when no digit is there or the value
79 * exceeds 1000. *cursor and *out are untouched on failure.
80 * WHY: A monitor value is a percentage; anything over 1000 is a garbled line,
81 * not a big number, so bound the accumulator rather than let it wrap.
82 * HOW: 1. Skip blanks; reject when the line ends or the byte is not a digit.
83 * 2. Accumulate decimal digits, rejecting past 1000.
84 * 3. Publish the clamped value and the new cursor.
85 */
86static int
87perf_parse_field(const u_char *line, size_t len, size_t *cursor, uint8_t *out)
88{
89 ngx_uint_t value = 0;
90 size_t at = perf_skip_blanks(line, len, *cursor);
91
92 if (at >= len || line[at] < '0' || line[at] > '9') {
93 return -1;
94 }
95
96 while (at < len && line[at] >= '0' && line[at] <= '9') {
97 value = value * 10 + (ngx_uint_t) (line[at] - '0');
98 if (value > 1000) {
99 return -1;
100 }
101 at++;
102 }
103
104 *cursor = at;
105 *out = (uint8_t) (value > 100 ? 100 : value);
106 return 0;
107}

110/*
111 * perf_parse_line — parse "cpu net xeq mem pag" into vals[5].
112 *
113 * WHAT: Five whitespace-separated decimal integers, each clamped to 100.
114 * Returns 0 on success, -1 on a malformed line (ignored, feed keeps
115 * its previous values).
116 * WHY: A monitor emitting a partial/garbled line must not zero the node's
117 * advertised load — reject, don't guess.
118 * HOW: 1. Parse five fields in turn; any rejection fails the line.
119 * 2. Require the remainder to be blank — trailing junk is malformed.
120 */
121static int
122perf_parse_line(const u_char *line, size_t len, uint8_t out5[5])
123{
124 size_t cursor = 0;
125 ngx_uint_t field;
126
127 for (field = 0; field < 5; field++) {
128 if (perf_parse_field(line, len, &cursor, &out5[field]) != 0) {
129 return -1;
130 }
131 }
132
133 return (perf_skip_blanks(line, len, cursor) == len) ? 0 : -1;
134}

138/* perf_teardown — close the pipe connection and reap the child, then arm the
139 * respawn backoff. Idempotent. */
140static void
141perf_teardown(brix_cms_perf_t *pf)
142{
143 if (pf->conn != NULL) {
144 ngx_close_connection(pf->conn);
145 pf->conn = NULL;
146 }
147 if (pf->child > 0) {
148 (void) kill(pf->child, SIGTERM);
149 (void) waitpid(pf->child, NULL, WNOHANG);
150 pf->child = -1;
151 }
152 if (!ngx_exiting && !pf->respawn.timer_set) {
153 ngx_add_timer(&pf->respawn, 5000);
154 }
155}

157/*
158 * perf_read_handler — drain stdout lines from the feed program.
159 *
160 * WHAT: Accumulates bytes into linebuf, parsing each complete line; a good
161 * line refreshes vals/updated. EOF or a read error tears down and
162 * arms the respawn backoff.
163 * WHY: Event-driven so a chatty or silent monitor costs the worker nothing.
164 * HOW: Nonblocking read loop; over-long lines are discarded to the next
165 * newline (reject, don't truncate-and-parse).
166 */
167static void
168perf_read_handler(ngx_event_t *ev)
169{
170 ngx_connection_t *c = ev->data;
171 brix_cms_perf_t *pf = c->data;
172 u_char chunk[256];
173 ssize_t got;
174 ssize_t i;
175
176 for ( ;; ) {
177 got = read(c->fd, chunk, sizeof(chunk));
178 if (got == 0) {
179 ngx_log_error(NGX_LOG_WARN, pf->cycle->log, 0,
180 "brix: cms perf pgm exited (EOF) — respawning in 5s");
181 perf_teardown(pf);
182 return;
183 }
184 if (got < 0) {
185 if (ngx_errno == NGX_EAGAIN) {
186 return;
187 }
188 ngx_log_error(NGX_LOG_WARN, pf->cycle->log, ngx_errno,
189 "brix: cms perf pgm read failed — respawning in 5s");
190 perf_teardown(pf);
191 return;
192 }
193
194 for (i = 0; i < got; i++) {
195 if (chunk[i] == '\n') {
196 uint8_t v[5];
197
198 if (pf->line_pos < sizeof(pf->linebuf)
199 && perf_parse_line(pf->linebuf, pf->line_pos, v) == 0)
200 {
201 ngx_memcpy(pf->vals, v, sizeof(pf->vals));
202 pf->updated = ngx_current_msec;
203 pf->have_vals = 1;
204 } else if (pf->line_pos > 0) {
205 ngx_log_error(NGX_LOG_INFO, pf->cycle->log, 0,
206 "brix: cms perf pgm: malformed line ignored");
207 }
208 pf->line_pos = 0;
209 continue;
210 }
211 if (pf->line_pos < sizeof(pf->linebuf)) {
212 pf->linebuf[pf->line_pos] = chunk[i];
213 }
214 pf->line_pos++; /* past the cap = line poisoned, drops above */
215 }
216 }
217}

219/*
220 * perf_spawn — start the feed program with stdout on a fresh pipe.
221 *
222 * WHAT: pipe2 + posix_spawn("/bin/sh", "-c", pgm); wraps the read end in an
223 * ngx_connection_t with perf_read_handler. On any failure arms the
224 * respawn backoff (the feed is an enhancement — its absence degrades
225 * to the /proc meter, never blocks the heartbeat).
226 * WHY: posix_spawn (not fork+exec) keeps the ASan/threads story simple and
227 * is the pattern the tree already uses for external programs.
228 * HOW: Child gets the pipe write end as fd 1; both pipe ends are CLOEXEC in
229 * the parent so respawns cannot leak fds into siblings.
230 */
231static void
232perf_spawn(brix_cms_perf_t *pf)
233{
234 int fds[2];
235 posix_spawn_file_actions_t fa;
236 char *argv[4];
237 ngx_connection_t *c;
238 pid_t pid;
239 int rc;
240
241 if (pipe2(fds, O_CLOEXEC) != 0) {
242 ngx_log_error(NGX_LOG_ERR, pf->cycle->log, ngx_errno,
243 "brix: cms perf pgm: pipe2 failed");
244 perf_teardown(pf);
245 return;
246 }
247
248 posix_spawn_file_actions_init(&fa);
249 posix_spawn_file_actions_adddup2(&fa, fds[1], 1);
250
251 /* The nginx worker's listening sockets are intentionally inherited by
252 * workers, but an external perf monitor must not retain them. Keeping a
253 * listener alive after its worker exits makes the next lifecycle instance
254 * fail to bind the same CMS-node ports. closefrom runs after dup2, so the
255 * child keeps stdin/stdout/stderr while every inherited descriptor >= 3
256 * is closed before /bin/sh starts. */
257 rc = posix_spawn_file_actions_addclosefrom_np(&fa, 3);
258 if (rc != 0) {
259 posix_spawn_file_actions_destroy(&fa);
260 close(fds[0]);
261 close(fds[1]);
262 ngx_log_error(NGX_LOG_ERR, pf->cycle->log, rc,
263 "brix: cms perf pgm: close inherited fds failed");
264 perf_teardown(pf);
265 return;
266 }
267
268 argv[0] = "/bin/sh";
269 argv[1] = "-c";
270 argv[2] = (char *) pf->pgm.data;
271 argv[3] = NULL;
272
273 rc = posix_spawn(&pid, "/bin/sh", &fa, NULL, argv, environ);
274 posix_spawn_file_actions_destroy(&fa);
275 close(fds[1]);
276
277 if (rc != 0) {
278 ngx_log_error(NGX_LOG_ERR, pf->cycle->log, rc,
279 "brix: cms perf pgm: spawn of \"%V\" failed",
280 &pf->pgm);
281 close(fds[0]);
282 perf_teardown(pf);
283 return;
284 }
285 pf->child = pid;
286
287 if (fcntl(fds[0], F_SETFL, O_NONBLOCK) != 0) {
288 ngx_log_error(NGX_LOG_ERR, pf->cycle->log, ngx_errno,
289 "brix: cms perf pgm: O_NONBLOCK failed");
290 close(fds[0]);
291 perf_teardown(pf);
292 return;
293 }
294
295 c = ngx_get_connection(fds[0], pf->cycle->log);
296 if (c == NULL) {
297 close(fds[0]);
298 perf_teardown(pf);
299 return;
300 }
301 c->data = pf;
302 c->read->handler = perf_read_handler;
303 c->write->handler = NULL;
304 if (ngx_handle_read_event(c->read, 0) != NGX_OK) {
305 perf_teardown(pf);
306 return;
307 }
308 pf->conn = c;
309 pf->line_pos = 0;
310
311 ngx_log_error(NGX_LOG_NOTICE, pf->cycle->log, 0,
312 "brix: cms perf pgm started: \"%V\" (pid %P)",
313 &pf->pgm, pid);
314}

323void
324brix_cms_perf_start(ngx_cycle_t *cycle, ngx_stream_brix_srv_conf_t *conf)
325{
326 brix_cms_perf_t *pf = &brix_cms_perf;
327 u_char *pgm_z;
328
329 if (conf->cms.perf_pgm.len == 0 || pf->cycle != NULL) {
330 return; /* feature off, or already started on this worker */
331 }
332
333 /* NUL-terminated copy — posix_spawn argv needs a C string. */
334 pgm_z = ngx_pnalloc(cycle->pool, conf->cms.perf_pgm.len + 1);
335 if (pgm_z == NULL) {
336 return;
337 }
338 ngx_memcpy(pgm_z, conf->cms.perf_pgm.data, conf->cms.perf_pgm.len);
339 pgm_z[conf->cms.perf_pgm.len] = '\0';
340
341 pf->cycle = cycle;
342 pf->pgm.data = pgm_z;
343 pf->pgm.len = conf->cms.perf_pgm.len;
344 pf->fresh_ms = conf->cms.perf_int * 2;
345 pf->child = -1;
346 pf->respawn.handler = perf_respawn_timer;
347 pf->respawn.data = pf;
348 pf->respawn.log = cycle->log;
349 pf->respawn.cancelable = 1;
350
351 perf_spawn(pf);
352}

355brix_cms_perf_get(uint8_t out5[5])
356{
357 brix_cms_perf_t *pf = &brix_cms_perf;
358
359 if (!pf->have_vals
360 || (ngx_msec_int_t) (ngx_current_msec - pf->updated)
361 > (ngx_msec_int_t) pf->fresh_ms)
362 {
363 return 0; /* never fed, or stale — caller uses the /proc meter */
364 }
365 ngx_memcpy(out5, pf->vals, 5);
366 return 1;
