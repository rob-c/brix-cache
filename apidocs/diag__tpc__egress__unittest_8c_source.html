49static tpce_verdict
50classify(int kxr, int errno_, const char *msg, double el, double budget)
51{
52 brix_status st;
53 char detail[256];
54 memset(&st, 0, sizeof(st));
55 st.kxr = kxr;
56 st.sys_errno = errno_;
57 if (msg != NULL) {
58 snprintf(st.msg, sizeof(st.msg), "%s", msg);
59 }
60 return tpce_classify_trigger(&st, el, budget, detail, sizeof(detail));
61}

64/* SUCCESS: kxr==0 (the pull completed) is the worst-case ACCEPTED verdict. */
65static void
66test_accepted(void)
67{
68 CHECK(classify(0, 0, NULL, 12.0, 4000.0) == TPCE_ACCEPTED);
69 /* a success status never mis-reads a stale errno/msg. */
70 CHECK(classify(0, ECONNREFUSED, "connection refused", 5.0, 4000.0)
71 == TPCE_ACCEPTED);
72}

75/* PERMITTED variants: the gateway originated and we can tell how far it got. */
76static void
77test_permitted_variants(void)
78{
79 /* conn-refused wins on errno OR on message text. */
80 CHECK(classify(kXR_FSError, ECONNREFUSED, "", 3.0, 4000.0)
81 == TPCE_CONN_REFUSED);
82 CHECK(classify(kXR_FSError, 0, "connection refused by source", 3.0, 4000.0)
83 == TPCE_CONN_REFUSED);
84 /* filtered: explicit timeout errno, timeout text, or budget elapsed. */
85 CHECK(classify(-1, ETIMEDOUT, "", 100.0, 4000.0) == TPCE_FILTERED);
86 CHECK(classify(kXR_FSError, 0, "operation timed out", 50.0, 4000.0)
87 == TPCE_FILTERED);
88 CHECK(classify(kXR_FSError, 0, "", 3700.0, 4000.0) == TPCE_FILTERED);
89 /* the gateway's generic "TPC connect to <h> failed" (no errno): a FAST reply
90 * is an active refusal, a SLOW one is its own connect timeout (filtered). */
91 CHECK(classify(kXR_ServerError, 0, "TPC connect to h failed", 6.0, 4000.0)
92 == TPCE_CONN_REFUSED);
93 CHECK(classify(kXR_ServerError, 0, "TPC connect to h failed", 2500.0, 4000.0)
94 == TPCE_FILTERED);
95 /* reached, lfn absent: NotFound code or "no such"/"not found" text. */
96 CHECK(classify(kXR_NotFound, 0, "", 40.0, 4000.0) == TPCE_REACHED_NOENT);
97 CHECK(classify(kXR_FSError, 0, "no such file or directory", 40.0, 4000.0)
98 == TPCE_REACHED_NOENT);
99 /* reached, some other error the source answered with. */
100 CHECK(classify(kXR_ServerError, 0, "internal source error", 40.0, 4000.0)
101 == TPCE_REACHED_ERROR);
102}

105/* PRECEDENCE: conn-refused must beat the budget-elapsed timeout heuristic when
106 * BOTH could match (a fast RST whose measured time also crossed the threshold
107 * on a tiny budget). refused is the more specific fact. */
108static void
109test_precedence(void)
110{
111 /* errno=ECONNREFUSED but elapsed also >= 0.9*budget → still CONN_REFUSED. */
112 CHECK(classify(kXR_FSError, ECONNREFUSED, "", 3800.0, 4000.0)
113 == TPCE_CONN_REFUSED);
114 /* success beats everything. */
115 CHECK(classify(0, ETIMEDOUT, "timed out", 3900.0, 4000.0) == TPCE_ACCEPTED);
116}

119/* EGRESS-DENIAL discriminator: only source/egress-policy text counts as a
120 * working guard; an unrelated NotAuthorized (read-only, token scope) does not. */
121static void
122test_egress_denial(void)
123{
124 CHECK(tpce_msg_is_egress_denial("TPC source host not permitted: 10.0.0.1"));
125 CHECK(tpce_msg_is_egress_denial("prohibited private address"));
126 CHECK(tpce_msg_is_egress_denial("loopback source refused"));
127 CHECK(tpce_msg_is_egress_denial("SSRF policy: egress blocked"));
128 /* unrelated denials must NOT read as an egress guard. */
129 CHECK(!tpce_msg_is_egress_denial("export is read-only"));
130 CHECK(!tpce_msg_is_egress_denial("token scope does not cover this path"));
131 CHECK(!tpce_msg_is_egress_denial(""));
132 CHECK(!tpce_msg_is_egress_denial(NULL));
133}

136/* AFTER-ARM: the arm result drives egress_permitted + verdict correctly. */
137static void
138test_after_arm(void)
139{
140 brix_status st;
141 tpce_result r;
142 brix_conn c;
143 brix_file f;
144
145 memset(&c, 0, sizeof(c));
146 memset(&f, 0, sizeof(f));
147
148 /* arm refused with an egress-policy NotAuthorized → REFUSED_POLICY, no egress. */
149 memset(&r, 0, sizeof(r));
150 memset(&st, 0, sizeof(st));
151 st.kxr = kXR_NotAuthorized;
152 snprintf(st.msg, sizeof(st.msg), "TPC source host not permitted");
153 tpce_after_arm(&c, &f, -1, &st, 4000.0, &r);
154 CHECK(r.verdict == TPCE_REFUSED_POLICY && r.egress_permitted == 0);
155 CHECK(r.arm_kxr == kXR_NotAuthorized);
156
157 /* arm refused with an UNRELATED NotAuthorized → ARM_ERROR, not policy. */
158 memset(&r, 0, sizeof(r));
159 memset(&st, 0, sizeof(st));
160 st.kxr = kXR_NotAuthorized;
161 snprintf(st.msg, sizeof(st.msg), "export is read-only");
162 tpce_after_arm(&c, &f, -1, &st, 4000.0, &r);
163 CHECK(r.verdict == TPCE_ARM_ERROR && r.egress_permitted == 0);
164}
