27static int
28raw_login_name(xrd_conform_execution *execution, const brix_url *url,
29 const char *username, brix_conn *connection, char security[256])
30{
31 brix_opts options = execution->run->opts;
32 brix_status status;
33 ClientLoginRequest login;
34 const uint8_t zero_sid[2] = { 0, 0 };
35 uint16_t wire_status = kXR_error;
36 uint16_t sid;
37 uint8_t *body = NULL;
38 uint32_t body_length = 0;
39 brix_resp_out output = { &wire_status, &body, &body_length };
40 size_t security_length;
41 int pass;
42
43 options.capture = NULL;
44 options.no_retry = 1;
45 brix_status_clear(&status);
46 if (brix_connect_no_login(connection, url, &options, &status) != 0)
47 return -1;
48 xrd_pack_login_request(&login, zero_sid, 1, username,
49 (uint8_t) (kXR_ver005 | kXR_asyncap));
50 pass = xrd_conform_reserve_probe(execution, sizeof(login))
51 && brix_send(connection, &login, NULL, &sid, &status) == 0
52 && brix_recv(connection, sid, &output, &status) == 0
53 && wire_status == kXR_ok && body_length > BRIX_SESSION_ID_LEN;
54 if (!pass) {
55 free(body);
56 brix_close(connection);
57 return -1;
58 }
59 security_length = body_length - BRIX_SESSION_ID_LEN;
60 if (security_length >= 256) security_length = 255;
61 memcpy(security, body + BRIX_SESSION_ID_LEN, security_length);
62 security[security_length] = '\0';
63 free(body);
64 return 0;
65}

74static int
75first_protocol(const char *security, char protocol[5])
76{
77 const char *name = strstr(security, "&P=");
78 const char *end;
79 size_t length;
80
81 if (name == NULL) return -1;
82 name += 3;
83 end = name;
84 while (*end != '\0' && *end != ',' && *end != '&') end++;
85 length = (size_t) (end - name);
86 if (length == 0 || length > 4) return -1;
87 memcpy(protocol, name, length);
88 protocol[length] = '\0';
89 return 0;

92static int
93send_auth(xrd_conform_execution *execution, brix_conn *connection,
94 const char *protocol, conform_auth_observation *observation)
95{
96 static const uint8_t invalid[] = "brix-conform-garbage";
97 ClientAuthRequest auth;
98 xrdw_auth_req_t body_fields;
99 brix_payload payload = { invalid, sizeof(invalid) - 1 };
100 brix_status status;
101 uint16_t wire_status = kXR_error;
102 uint16_t sid;
103 uint8_t *body = NULL;
104 uint32_t body_length = 0;
105 brix_resp_out output = { &wire_status, &body, &body_length };
106 int round;
107
108 memset(observation, 0, sizeof(*observation));
109 for (round = 0; round < CONFORM_AUTH_ROUND_LIMIT; round++) {
110 memset(&auth, 0, sizeof(auth));
111 auth.requestid = htons(kXR_auth);
112 memset(&body_fields, 0, sizeof(body_fields));
113 memcpy(body_fields.credtype, protocol, strlen(protocol));
114 xrdw_auth_req_pack(&body_fields, ((ClientRequestHdr *) &auth)->body);
115 if (!xrd_conform_reserve_probe(execution, sizeof(auth) + payload.len)) return -1;
116 brix_status_clear(&status);
117 if (brix_send(connection, &auth, &payload, &sid, &status) != 0)
118 return -1;
119 wire_status = kXR_error;
120 body = NULL;
121 body_length = 0;
122 if (brix_recv(connection, sid, &output, &status) != 0) {
123 free(body);
124 observation->classified = status.kxr != 0;
125 observation->terminal = 1;
126 observation->kxr = status.kxr;
127 return 0;
128 }
129 free(body);
130 if (wire_status == kXR_ok) {
131 observation->classified = 1;
132 observation->accepted = 1;
133 observation->terminal = 1;
134 return 0;
135 }
136 if (wire_status != kXR_authmore) return -1;
137 }
138 observation->classified = 1;
139 return 0;
140}

258static int
259redirect_host_is_ipv6(const uint8_t *body, uint32_t length, int *valid)
260{
261 size_t index;
262 size_t colons = 0;
263 size_t host_length = 0;
264
265 *valid = 0;
266 if (length <= 4) return 0;
267 for (index = 4; index < length; index++) {
268 if (body[index] == '\0' || body[index] == '?') break;
269 if (body[index] == ':') colons++;
270 host_length++;
271 }
272 if (host_length == 0 || index == length) return 0;
273 *valid = 1;
274 return body[4] == '[' || colons > 1;
275}

286/* WHAT: Send login with both ability bytes zero and inspect its first answer.
287 * WHY: Following a redirect would hide a server that invented IPv6 ability
288 * the client never advertised and strand IPv4-only callers.
289 * HOW: Stop after brix_recv surfaces the raw status; accept login, a defined
290 * refusal, or a non-IPv6 redirect, and reject malformed/IPv6-only hops. */
291void
292xrd_conform_probe_login_abilities(xrd_conform_execution *execution,
293 const brix_url *url)
294{
295 brix_opts options = execution->run->opts;
296 brix_conn connection;
297 brix_status status;
298 ClientLoginRequest login;
299 const uint8_t zero_sid[2] = { 0, 0 };
300 uint16_t wire_status = kXR_error;
301 uint16_t sid;
302 uint8_t *body = NULL;
303 uint32_t body_length = 0;
304 brix_resp_out output = { &wire_status, &body, &body_length };
305 int rc;
306 int valid = 0;
307 int ipv6 = 0;
308
309 if (!xrd_conform_row_active(execution, "LOG-025")) return;
310 options.capture = NULL;
311 brix_status_clear(&status);
312 if (brix_connect_no_login(&connection, url, &options, &status) != 0) {
313 inconclusive(execution, "LOG-025",
314 "fixture-unavailable:zero-ability-login");
315 return;
316 }
317 xrd_pack_login_request(&login, zero_sid, 1, "conform", kXR_ver005);
318 if (!xrd_conform_reserve_probe(execution, sizeof(login))
319 || brix_send(&connection, &login, NULL, &sid, &status) != 0) {
320 brix_close(&connection);
321 inconclusive(execution, "LOG-025",
322 "fixture-unavailable:zero-ability-login");
323 return;
324 }
325 rc = brix_recv(&connection, sid, &output, &status);
326 if (rc == 0 && wire_status == kXR_redirect)
327 ipv6 = redirect_host_is_ipv6(body, body_length, &valid);
328 free(body);
329 brix_close(&connection);
330 if (rc != 0 && status.kxr <= 0) {
331 inconclusive(execution, "LOG-025",
332 "fixture-unavailable:zero-ability-login");
333 return;
334 }
335 xrd_conform_grade_active(execution, "LOG-025",
336 zero_ability_response_valid(rc, &status, wire_status, valid, ipv6),
337 "zero-ability-login-received-no-ipv6-only-redirect",
338 wire_status == kXR_redirect && ipv6
339 ? "zero-ability-login-received-ipv6-only-redirect"
340 : "zero-ability-login-received-malformed-response");
341}
