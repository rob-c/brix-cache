3/*
4 * carry.c — async-safe carry of a delegated krb5 credential (phase-70 §5.7).
5 *
6 * Implements the carry.h contract: export a captured GSS initiator credential to
7 * a FILE ccache (brix_krb5_cred_to_ccache) and re-acquire it from that path on
8 * the async fill task (brix_krb5_cred_from_ccache / _carry_release). The FILE
9 * ccache is the serialisable artifact that lets a request-scoped gss_cred_id_t
10 * cross into brix_cache_fill_t as a plain path — the same trick the gsi leg uses
11 * for x509 proxy PEMs. Without BRIX_HAVE_KRB5 the file still compiles: every entry
12 * reports unavailable and returns NGX_ERROR.
13 */

45/* Render "FILE:<path>" into buf, bounded. Returns NGX_OK / NGX_ERROR (overflow).
46 * A ccache path longer than the buffer is refused rather than silently truncated
47 * (a truncated path would resolve to the wrong file). Kept free of ngx_snprintf
48 * so the object links into the ngx-core-free krb5 unit harness. */
49static ngx_int_t
50brix_krb5_file_ccname(const char *path, char *buf, size_t buflen)
51{
52 static const char prefix[] = "FILE:";
53 size_t plen = sizeof(prefix) - 1; /* 5, excludes NUL */
54 size_t need;
55
56 need = plen + ngx_strlen(path);
57 if (need + 1 > buflen) { /* +1 for the NUL */
58 return NGX_ERROR;
59 }
60 ngx_memcpy(buf, prefix, plen);
61 ngx_memcpy(buf + plen, path, ngx_strlen(path) + 1); /* copy incl. NUL */
62 return NGX_OK;
63}

65ngx_int_t
66brix_krb5_cred_to_ccache(void *deleg_gss_cred, const char *path, ngx_log_t *log)
67{
68 char ccname[NGX_MAX_PATH + sizeof("FILE:")];
69 OM_uint32 maj, min = 0;
70 gss_key_value_element_desc elem;
71 gss_key_value_set_desc store;
72
73 if (deleg_gss_cred == NULL || path == NULL) {
74 ngx_log_error(NGX_LOG_WARN, log, 0,
75 "brix: krb5 carry: export missing cred/path");
76 return NGX_ERROR;
77 }
78
79 if (brix_krb5_file_ccname(path, ccname, sizeof(ccname)) != NGX_OK) {
80 ngx_log_error(NGX_LOG_WARN, log, 0,
81 "brix: krb5 carry: export ccache path too long");
82 return NGX_ERROR;
83 }
84
85 /* RFC 5588 gss_store_cred_into with an explicit "ccache" store element writes
86 * the initiator cred's forwarded TGT into the named FILE ccache, OVERWRITING
87 * it (overwrite=1) so a pre-created 0-byte temp is a clean target — unlike the
88 * deprecated gss_krb5_copy_ccache, which cannot initialise and rejects an empty
89 * file as "bad format". default_cred=0 keeps it off the process default ccache.
90 * libkrb5 creates the FILE 0600. The ccname/deleg_cred are borrowed. */
91 elem.key = "ccache";
92 elem.value = ccname;
93 store.count = 1;
94 store.elements = &elem;
95
96 maj = gss_store_cred_into(&min, (gss_cred_id_t) deleg_gss_cred,
97 GSS_C_INITIATE, GSS_C_NO_OID,
98 1 /* overwrite */, 0 /* not default */,
99 &store, NULL, NULL);
100 if (GSS_ERROR(maj)) {
101 ngx_log_error(NGX_LOG_WARN, log, 0,
102 "brix: krb5 carry: gss_store_cred_into failed"
103 " (maj=0x%08xui min=0x%08xui)",
104 (ngx_uint_t) maj, (ngx_uint_t) min);
105 return NGX_ERROR;
106 }
107
108 return NGX_OK;
109}

112brix_krb5_cred_from_ccache(const char *path, void **out_gss_cred,
113 void **out_hold, ngx_log_t *log)
114{

116 krb5_ccache cc = NULL;

120 brix_krb5_carry_hold_t *hold;
121 char ccname[NGX_MAX_PATH + sizeof("FILE:")];
122
123 if (path == NULL || out_gss_cred == NULL || out_hold == NULL) {
124 ngx_log_error(NGX_LOG_WARN, log, 0,
125 "brix: krb5 carry: import missing args");
126 return NGX_ERROR;
127 }
128 *out_gss_cred = NULL;
129 *out_hold = NULL;
130
131 if (brix_krb5_file_ccname(path, ccname, sizeof(ccname)) != NGX_OK) {
132 ngx_log_error(NGX_LOG_WARN, log, 0,
133 "brix: krb5 carry: import ccache path too long");
134 return NGX_ERROR;
135 }
136
137 hold = malloc(sizeof(brix_krb5_carry_hold_t));
138 if (hold == NULL) {
139 ngx_log_error(NGX_LOG_WARN, log, 0,
140 "brix: krb5 carry: import hold alloc failed");
141 return NGX_ERROR;
142 }
143
144 krc = krb5_init_context(&kctx);
145 if (krc != 0) {
146 brix_krb5_carry_log(NULL, krc, "krb5_init_context", log);
147 free(hold);
148 return NGX_ERROR;
149 }
150
151 krc = krb5_cc_resolve(kctx, ccname, &cc);
152 if (krc != 0) {
153 brix_krb5_carry_log(kctx, krc, "krb5_cc_resolve", log);
154 krb5_free_context(kctx);
155 free(hold);
156 return NGX_ERROR;
157 }
158
159 maj = gss_krb5_import_cred(&min, cc, NULL, NULL, &gcred);
160 if (GSS_ERROR(maj)) {
161 ngx_log_error(NGX_LOG_WARN, log, 0,
162 "brix: krb5 carry: gss_krb5_import_cred failed"
163 " (maj=0x%08xui min=0x%08xui)",
164 (ngx_uint_t) maj, (ngx_uint_t) min);
165 krb5_cc_close(kctx, cc);
166 krb5_free_context(kctx);
167 free(hold);
168 return NGX_ERROR;
169 }
170
171 hold->ctx = kctx;
172 hold->cc = cc;
173 *out_gss_cred = gcred;
174 *out_hold = hold;
175 return NGX_OK;
176}
