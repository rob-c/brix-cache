7/*
8 * carry.c — async-safe carry of a delegated krb5 credential (phase-70 §5.7).
9 *
10 * Implements the carry.h contract: export a captured GSS initiator credential to
11 * a FILE ccache (brix_krb5_cred_to_ccache) and re-acquire it from that path on
12 * the async fill task (brix_krb5_cred_from_ccache / _carry_release). The FILE
13 * ccache is the serialisable artifact that lets a request-scoped gss_cred_id_t
14 * cross into brix_cache_fill_t as a plain path — the same trick the gsi leg uses
15 * for x509 proxy PEMs. Without BRIX_HAVE_KRB5 the file still compiles: every entry
16 * reports unavailable and returns NGX_ERROR.
17 */

26/* Backing handles kept alive for a re-imported cred's lifetime. The ccache
27 * handle from krb5_cc_resolve() is bound to its krb5_context, and
28 * gss_krb5_import_cred() retains the handle — so both must outlive the cred and
29 * are freed together by brix_krb5_cred_carry_release(). */

31 krb5_context ctx;
32 krb5_ccache cc;

49/* Render "FILE:<path>" into buf, bounded. Returns NGX_OK / NGX_ERROR (overflow).
50 * A ccache path longer than the buffer is refused rather than silently truncated
51 * (a truncated path would resolve to the wrong file). Kept free of ngx_snprintf
52 * so the object links into the ngx-core-free krb5 unit harness. */
53static ngx_int_t
54brix_krb5_file_ccname(const char *path, char *buf, size_t buflen)
55{
56 static const char prefix[] = "FILE:";
57 size_t plen = sizeof(prefix) - 1; /* 5, excludes NUL */
58 size_t need;
59
60 need = plen + ngx_strlen(path);
61 if (need + 1 > buflen) { /* +1 for the NUL */
62 return NGX_ERROR;
63 }
64 ngx_memcpy(buf, prefix, plen);
65 ngx_memcpy(buf + plen, path, ngx_strlen(path) + 1); /* copy incl. NUL */
66 return NGX_OK;
67}

69ngx_int_t
70brix_krb5_cred_to_ccache(void *deleg_gss_cred, const char *path, ngx_log_t *log)
71{
72 char ccname[NGX_MAX_PATH + sizeof("FILE:")];
73 OM_uint32 maj, min = 0;
74 gss_key_value_element_desc elem;
75 gss_key_value_set_desc store;
76
77 if (deleg_gss_cred == NULL || path == NULL) {
78 ngx_log_error(NGX_LOG_WARN, log, 0,
79 "brix: krb5 carry: export missing cred/path");
80 return NGX_ERROR;
81 }
82
83 if (brix_krb5_file_ccname(path, ccname, sizeof(ccname)) != NGX_OK) {
84 ngx_log_error(NGX_LOG_WARN, log, 0,
85 "brix: krb5 carry: export ccache path too long");
86 return NGX_ERROR;
87 }
88
89 /* RFC 5588 gss_store_cred_into with an explicit "ccache" store element writes
90 * the initiator cred's forwarded TGT into the named FILE ccache, OVERWRITING
91 * it (overwrite=1) so a pre-created 0-byte temp is a clean target — unlike the
92 * deprecated gss_krb5_copy_ccache, which cannot initialise and rejects an empty
93 * file as "bad format". default_cred=0 keeps it off the process default ccache.
94 * libkrb5 creates the FILE 0600. The ccname/deleg_cred are borrowed. */
95 elem.key = "ccache";
96 elem.value = ccname;
97 store.count = 1;
98 store.elements = &elem;
99
100 maj = gss_store_cred_into(&min, (gss_cred_id_t) deleg_gss_cred,
101 GSS_C_INITIATE, GSS_C_NO_OID,
102 1 /* overwrite */, 0 /* not default */,
103 &store, NULL, NULL);
104 if (GSS_ERROR(maj)) {
105 ngx_log_error(NGX_LOG_WARN, log, 0,
106 "brix: krb5 carry: gss_store_cred_into failed"
107 " (maj=0x%08xui min=0x%08xui)",
108 (ngx_uint_t) maj, (ngx_uint_t) min);
109 return NGX_ERROR;
110 }
111
112 return NGX_OK;
113}

115ngx_int_t
116brix_krb5_cred_from_ccache(const char *path, void **out_gss_cred,
117 void **out_hold, ngx_log_t *log)
118{
119 krb5_context kctx = NULL;
120 krb5_ccache cc = NULL;
121 krb5_error_code krc;
122 gss_cred_id_t gcred = GSS_C_NO_CREDENTIAL;
123 OM_uint32 maj, min = 0;
124 brix_krb5_carry_hold_t *hold;
125 char ccname[NGX_MAX_PATH + sizeof("FILE:")];
126
127 if (path == NULL || out_gss_cred == NULL || out_hold == NULL) {
128 ngx_log_error(NGX_LOG_WARN, log, 0,
129 "brix: krb5 carry: import missing args");
130 return NGX_ERROR;
131 }
132 *out_gss_cred = NULL;
133 *out_hold = NULL;
134
135 if (brix_krb5_file_ccname(path, ccname, sizeof(ccname)) != NGX_OK) {
136 ngx_log_error(NGX_LOG_WARN, log, 0,
137 "brix: krb5 carry: import ccache path too long");
138 return NGX_ERROR;
139 }
140
141 hold = malloc(sizeof(brix_krb5_carry_hold_t));
142 if (hold == NULL) {
143 ngx_log_error(NGX_LOG_WARN, log, 0,
144 "brix: krb5 carry: import hold alloc failed");
145 return NGX_ERROR;
146 }
147
148 krc = krb5_init_context(&kctx);
149 if (krc != 0) {
150 brix_krb5_carry_log(NULL, krc, "krb5_init_context", log);
151 free(hold);
152 return NGX_ERROR;
153 }
154
155 krc = krb5_cc_resolve(kctx, ccname, &cc);
156 if (krc != 0) {
157 brix_krb5_carry_log(kctx, krc, "krb5_cc_resolve", log);
158 krb5_free_context(kctx);
159 free(hold);
160 return NGX_ERROR;
161 }
162
163 maj = gss_krb5_import_cred(&min, cc, NULL, NULL, &gcred);
164 if (GSS_ERROR(maj)) {
165 ngx_log_error(NGX_LOG_WARN, log, 0,
166 "brix: krb5 carry: gss_krb5_import_cred failed"
167 " (maj=0x%08xui min=0x%08xui)",
168 (ngx_uint_t) maj, (ngx_uint_t) min);
169 krb5_cc_close(kctx, cc);
170 krb5_free_context(kctx);
171 free(hold);
172 return NGX_ERROR;
173 }
174
175 hold->ctx = kctx;
176 hold->cc = cc;
177 *out_gss_cred = gcred;
178 *out_hold = hold;
179 return NGX_OK;
180}

183brix_krb5_cred_carry_release(void *gss_cred, void *hold_v, ngx_log_t *log)
184{
185 brix_krb5_carry_hold_t *hold = hold_v;
186 OM_uint32 min = 0;
187
188 (void) log;
189
190 if (gss_cred != NULL) {
191 gss_cred_id_t g = (gss_cred_id_t) gss_cred;
192 (void) gss_release_cred(&min, &g);
193 }
194 if (hold != NULL) {
195 if (hold->cc != NULL) {
196 krb5_cc_close(hold->ctx, hold->cc);
197 }
198 if (hold->ctx != NULL) {
199 krb5_free_context(hold->ctx);
200 }
201 free(hold);
202 }
203}
