12/*
13 * Journal record grammar sizes. A field's escaped form is at most twice its
14 * source (every byte escapable), so these are derived from the struct widths
15 * rather than guessed: owner[64] -> 127+NUL, req.path[1024] -> 2047+NUL. The
16 * line buffer holds a whole worst-case record — three numeric fields, four
17 * tabs, both escaped fields and the newline — with room to spare, because a
18 * line that does NOT fit is precisely the case the replay path must be able to
19 * recognise and discard whole.
20 */
21#define CTA_JOURNAL_OWNER_ESC 128
22#define CTA_JOURNAL_PATH_ESC 2048
23#define CTA_JOURNAL_LINE_MAX 4096

25/*
26 * Escape one text field into the journal's record grammar.
27 *
28 * WHAT: copies `in` to `out`, replacing the two bytes the grammar reserves —
29 * tab (field separator) and newline (record separator) — and the escape
30 * byte itself with two-character sequences.
31 *
32 * WHY: SECURITY. `req.path` is `Notification.file.lpath` straight off the
33 * wire, up to 1023 bytes chosen by whoever submitted the request. Written
34 * raw, a path containing a newline FORGES A SECOND RECORD, and replay
35 * believes it — including its `owner` field, which is the principal
36 * cta_queue_cancel() checks. A submitter could therefore plant a queue
37 * entry owned by somebody else, or restate a real id under a different
38 * owner, and the forgery only takes effect after a restart, where nobody
39 * is watching the request that planted it. A tab is the same attack one
40 * field narrower: it shifts every later field left. `owner` is escaped
41 * too — it is an identity string rather than a wire field, but a grammar
42 * that holds only for the fields we currently believe are safe is not a
43 * grammar.
44 *
45 * HOW: '\\' -> "\\\\", '\t' -> "\\t", '\n' -> "\\n", '\r' -> "\\r"; every other
46 * byte verbatim. Returns 0, or -1 when the escaped form does not fit —
47 * the caller then writes NO record, because a truncated one re-parses as
48 * a different (and attacker-chosen) record rather than as an error.
49 */
50static int
51journal_escape(const char *in, char *out, size_t cap)
52{
53 size_t o = 0;
54 size_t i;
55
56 for (i = 0; in[i] != '\0'; i++) {
57 char esc = 0;
58
59 switch (in[i]) {
60 case '\\': esc = '\\'; break;
61 case '\t': esc = 't'; break;
62 case '\n': esc = 'n'; break;
63 case '\r': esc = 'r'; break;
64 default: break;
65 }
66 if (esc == 0) {
67 if (o + 1 >= cap) { return -1; }
68 out[o++] = in[i];
69 continue;
70 }
71 if (o + 2 >= cap) { return -1; }
72 out[o++] = '\\';
73 out[o++] = esc;
74 }
75 out[o] = '\0';
76 return 0;
77}

80/*
81 * The exact inverse of journal_escape, bounded by `cap`.
82 *
83 * An UNKNOWN escape ("\\x") yields 'x' and a trailing lone '\\' is dropped:
84 * replay reads a file an operator may have hand-edited and an attacker may have
85 * reached through the hole this pair of functions closes, so a malformed input
86 * must produce a harmless field — never a parse that resynchronises onto
87 * attacker-chosen bytes, and never a write past `out`.
88 */
89static void
90journal_unescape(const char *in, char *out, size_t cap)
91{
92 size_t o = 0;
93 size_t i;
94
95 for (i = 0; in[i] != '\0' && o + 1 < cap; i++) {
96 char c = in[i];
97
98 if (c != '\\') {
99 out[o++] = c;
100 continue;
101 }
102 if (in[i + 1] == '\0') {
103 break; /* trailing lone escape: drop it */
104 }
105 i++;
106 switch (in[i]) {
107 case 't': out[o++] = '\t'; break;
108 case 'n': out[o++] = '\n'; break;
109 case 'r': out[o++] = '\r'; break;
110 default: out[o++] = in[i]; break;
111 }
112 }
113 out[o] = '\0';
114}

117/*
118 * True when `line` (as returned by fgets) is a WHOLE record.
119 *
120 * A record whose escaped form still exceeds the line buffer cannot be a record
121 * this code wrote, so it is either corruption or a hand edit. fgets would hand
122 * back its head and then its tail as if the tail were a record of its own —
123 * letting a long field resynchronise the parser onto bytes chosen by whoever
124 * supplied it. Draining to the next newline denies that: the whole line is
125 * discarded, or nothing is. A final line with no newline is complete (EOF),
126 * not truncated.
127 */
128static int
129journal_line_whole(FILE *f, const char *line)
130{
131 int c;
132
133 if (strchr(line, '\n') != NULL || feof(f)) {
134 return 1;
135 }
136 while ((c = fgetc(f)) != EOF && c != '\n') {
137 /* discard the remainder of the over-long line */
138 }
139 return 0;
140}

143/* Append one entry's current state to the journal (tab-delimited record). */
144/*
145 * Append one entry's current state to the journal, as ONE record.
146 *
147 * The record is built whole and emitted with a SINGLE write(2) on an O_APPEND
148 * descriptor, rather than with fprintf + fflush. That is not a style choice:
149 * every worker inherits this one descriptor from the master, so the journal is
150 * written concurrently by the whole worker set, and only a single append of a
151 * complete record keeps two workers' records from interleaving into a third
152 * record that neither wrote. stdio promises no such thing about where its
153 * buffer boundaries fall.
154 */
155static void
156journal_append(brix_cta_queue_t *q, const cta_req_t *e)
157{
158 char rec[CTA_JOURNAL_LINE_MAX];
159 char owner[CTA_JOURNAL_OWNER_ESC];
160 char path[CTA_JOURNAL_PATH_ESC];
161 ssize_t written;
162 int n;
163
164 if (q->journal_fd < 0) {
165 return;
166 }
167 if (journal_escape(e->owner, owner, sizeof(owner)) != 0
168 || journal_escape(e->req.path, path, sizeof(path)) != 0)
169 {
170 return; /* see journal_escape: never write a truncated record */
171 }
172 n = snprintf(rec, sizeof(rec), "%llu\t%d\t%d\t%s\t%s\n",
173 (unsigned long long) e->id, (int) e->req.op, (int) e->state,
174 owner, path);
175 if (n <= 0 || (size_t) n >= sizeof(rec)) {
176 return; /* same rule as a field that will not fit */
177 }
178 /* Best-effort journal: a failed write only degrades crash-replay fidelity;
179 * the in-memory queue stays authoritative, so there is no error to
180 * propagate from an append. */
181 written = write(q->journal_fd, rec, (size_t) n);
182 (void) written;
183}

185/* Legal transitions, table-driven. A request may advance SUBMITTED→QUEUED→ACTIVE
186 * →COMPLETE, fail from ACTIVE, or be cancelled from any non-terminal state. */
187static int
188transition_ok(cta_state_t from, cta_state_t to)
189{
190 switch (from) {
191 case CTA_ST_SUBMITTED:
192 return to == CTA_ST_QUEUED || to == CTA_ST_CANCELED;
193 case CTA_ST_QUEUED:
194 return to == CTA_ST_ACTIVE || to == CTA_ST_CANCELED ||
195 to == CTA_ST_FAILED;
196 case CTA_ST_ACTIVE:
197 return to == CTA_ST_COMPLETE || to == CTA_ST_FAILED ||
198 to == CTA_ST_CANCELED;
199 case CTA_ST_COMPLETE:
200 case CTA_ST_FAILED:
201 case CTA_ST_CANCELED:
202 return 0; /* terminal */
203 }
204 return 0;
205}

244/* Find an existing slot for id, or claim a free one (replay only). */
245static cta_req_t *
246replay_slot(brix_cta_queue_t *q, uint64_t id)
247{
248 int i, free_slot = -1;
249
250 for (i = 0; i < CTA_QUEUE_MAX; i++) {
251 if (q->slots[i].in_use && q->slots[i].id == id) {
252 return &q->slots[i];
253 }
254 if (!q->slots[i].in_use && free_slot < 0) {
255 free_slot = i;
256 }
257 }
258 if (free_slot < 0) {
259 return NULL;
260 }
261 memset(&q->slots[free_slot], 0, sizeof(q->slots[free_slot]));
262 q->slots[free_slot].in_use = 1;
263 q->slots[free_slot].id = id;
264 return &q->slots[free_slot];
265}

267int
268cta_queue_open_journal(brix_cta_queue_t *q, const char *path)
269{
270 FILE *f = fopen(path, "r"); /* replay cursor, separate from the append fd */
271 char line[CTA_JOURNAL_LINE_MAX];
272 int fd;
273
274 while (f != NULL && fgets(line, sizeof(line), f) != NULL) {
275 unsigned long long id;
276 int op = 0, state = 0;
277 char owner_esc[CTA_JOURNAL_OWNER_ESC] = "";
278 char path_esc[CTA_JOURNAL_PATH_ESC] = "";
279 char owner[64] = "", pathbuf[1024] = "";
280 cta_req_t *e;
281
282 if (!journal_line_whole(f, line)) {
283 continue; /* over-long: see journal_line_whole */
284 }
285 /* "id\top\tstate\towner\tpath" — the text fields are escaped
286 * (journal_escape), so a raw tab or newline can no longer occur INSIDE
287 * one and this split stays exact. Either may be empty, in which case
288 * its conversion fails harmlessly and the "" initialiser stands. */
289 if (sscanf(line, "%llu\t%d\t%d\t%127[^\t]\t%2047[^\n]",
290 &id, &op, &state, owner_esc, path_esc) < 3) {
291 continue; /* need at least id/op/state */
292 }
293 journal_unescape(owner_esc, owner, sizeof(owner));
294 journal_unescape(path_esc, pathbuf, sizeof(pathbuf));
295
296 e = replay_slot(q, (uint64_t) id);
297 if (e == NULL) {
298 break;
299 }
300 e->req.op = (cta_op_t) op;
301 e->state = (cta_state_t) state;
302 snprintf(e->owner, sizeof(e->owner), "%s", owner);
303 snprintf(e->req.path, sizeof(e->req.path), "%s", pathbuf);
304 if ((uint64_t) id >= q->next_id) {
305 q->next_id = (uint64_t) id + 1;
306 }
307 }
308 if (f != NULL) {
309 fclose(f);
310 }
311 /*
312 * The append descriptor. Opened here — which, for the shared queue, means
313 * in the MASTER during zone init, before fork — so every worker inherits
314 * this same open file description and its shared O_APPEND offset. One
315 * journal, one cursor, no per-worker files to reconcile on replay.
316 */
317 fd = open(path, O_WRONLY | O_CREAT | O_APPEND, 0600);
318 if (fd < 0) {
319 return -1;
320 }
321 q->journal_fd = fd;
322 return 0;
323}

325cta_req_t *
326cta_queue_submit(brix_cta_queue_t *q, const cta_request_t *r, const char *owner)
327{
328 int i;
329
330 for (i = 0; i < CTA_QUEUE_MAX; i++) {
331 if (!q->slots[i].in_use) {
332 cta_req_t *e = &q->slots[i];
333 memset(e, 0, sizeof(*e));
334 e->req = *r;
335 e->state = CTA_ST_SUBMITTED;
336 e->id = q->next_id++;
337 e->in_use = 1;
338 if (owner != NULL) {
339 size_t n = strlen(owner);
340 if (n >= sizeof(e->owner)) {
341 n = sizeof(e->owner) - 1;
342 }
343 memcpy(e->owner, owner, n);
344 e->owner[n] = '\0';
345 }
346 journal_append(q, e);
347 return e;
348 }
349 }
350 return NULL; /* full */
351}
