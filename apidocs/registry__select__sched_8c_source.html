1/*
2 * registry_select_sched.c — §2.3/§2.5 selection policy: cms.sched component
3 * blending, the maxload ceiling, and stage-aware selection.
4 *
5 * WHAT: The metric policy layer of the selection ladder. srv_sel_metric()
6 * turns one registry entry into the scalar the tier accumulators compare —
7 * either the legacy load-weight blend (Phase-89 W4, byte-identical when
8 * cms.sched is unset) or the stock cms.sched per-component weighted blend of
9 * the five raw heartbeat theLoad bytes. srv_sel_over_maxload() implements
10 * the maxload ceiling. brix_srv_select_stage() is the second phase of stage-
11 * aware selection (§2.5): pick the roomiest stage-capable node for a file no
12 * one currently holds.
13 *
14 * WHY: Split from registry_select.c so the ladder mechanics (tiers, affinity,
15 * fuzz) and the metric policy are individually reviewable and the file-size
16 * ceiling holds. The weights are process-global set-once config (like
17 * brix_srv_load_weight) — never mutated after fork.
18 *
19 * HOW: All pure computation except brix_srv_select_stage, which takes the
20 * registry spinlock for its scan like every other selector.
21 */
22#include "registry_internal.h"

24/*
25 * sched_active — is the §2.3 component engine configured?
26 *
27 * WHAT: Returns 1 when any component weight is non-zero (fuzz/maxload alone
28 * do not switch the metric — they modulate the ladder, not the scalar).
29 * WHY: An all-zero weight vector must keep scoring byte-identical to the
30 * legacy path so existing deployments see no behavior change.
31 * HOW: OR over the six weight fields.
32 */
33static int
34sched_active(void)
35{
36 return (brix_srv_sched.cpu | brix_srv_sched.io | brix_srv_sched.runq
37 | brix_srv_sched.mem | brix_srv_sched.pag
38 | brix_srv_sched.space) != 0;
39}

41/*
42 * sched_machine_load — the blended machine-load figure for the ceiling test.
43 *
44 * WHAT: Returns the node's machine load 0-100: the weighted mean of the five
45 * theLoad bytes when the component engine is on, else the recorded max
46 * (load_pct) — the bottleneck resource.
47 * WHY: The maxload ceiling must judge the same figure the operator weighted;
48 * with no weights the stock-like "hottest component" is the honest read.
49 * HOW: Sum weight×byte over the five machine components (space is disk
50 * occupancy, not machine load — excluded) and divide by the weight sum;
51 * a zero machine-weight sum falls back to load_pct.
52 */
53static uint32_t
54sched_machine_load(const brix_srv_entry_t *e)
55{
56 uint64_t num, den;
57
58 den = (uint64_t) brix_srv_sched.cpu + brix_srv_sched.io
59 + brix_srv_sched.runq + brix_srv_sched.mem + brix_srv_sched.pag;
60 if (den == 0) {
61 return e->load_pct;
62 }
63
64 num = (uint64_t) brix_srv_sched.cpu * e->load5[0]
65 + (uint64_t) brix_srv_sched.io * e->load5[1]
66 + (uint64_t) brix_srv_sched.runq * e->load5[2]
67 + (uint64_t) brix_srv_sched.mem * e->load5[3]
68 + (uint64_t) brix_srv_sched.pag * e->load5[4];
69 return (uint32_t) (num / den);
70}

72/*
73 * srv_sel_over_maxload — §2.3: is this node over the cms.sched ceiling?
74 *
75 * WHAT: Returns 1 when maxload is configured and the node's blended machine
76 * load exceeds it. 0 when the ceiling is off (the default).
77 * WHY: Stock cms.sched maxload stops sending clients to a saturated node
78 * while cooler nodes exist; the ladder in registry_select.c degrades to
79 * the least-loaded overloaded node when everyone is hot (a divergence
80 * from stock's delay-the-client — BriX prefers serving degraded over
81 * stalling, and the SUPCount hold gate covers the not-ready case).
82 * HOW: Compare sched_machine_load() against the configured ceiling.
83 */
84int
85srv_sel_over_maxload(const brix_srv_entry_t *e)
86{
87 return brix_srv_sched.maxload > 0
88 && sched_machine_load(e) > brix_srv_sched.maxload;
89}

91/*
92 * srv_sel_metric — the scalar one registry entry contributes to the ladder.
93 *
94 * WHAT: Reads minimise the result, writes maximise it. With cms.sched
95 * weights set: reads score the weighted mean of the five theLoad bytes
96 * plus the space (util_pct) component; writes keep free_mb as the base
97 * (capacity places data) scaled down by the machine load in proportion
98 * to the machine-weight share, so a hot node with equal space loses.
99 * Without weights: the legacy Phase-89 W4 load_weight blend, or the raw
100 * util/free metric when that too is off — byte-identical to before.
101 * WHY: One function owns "what does better mean", so locate/open/stat and
102 * the stage selector can never disagree.
103 * HOW: Dispatch on sched_active(); both blends are pure integer arithmetic
104 * widened to 64-bit before the divide.
105 */
106uint32_t
107srv_sel_metric(const brix_srv_entry_t *e, int for_write)
108{
109 uint64_t num, den;
110 uint32_t machine;
111 ngx_uint_t w;
112
113 if (!sched_active()) {
114 /* Legacy: Phase-89 W4 single-weight blend (0 = raw metric). */
115 uint32_t metric = for_write ? e->free_mb : e->util_pct;
116
117 w = brix_srv_load_weight;
118 if (w == 0) {
119 return metric;
120 }
121 if (for_write) {
122 return metric
123 - (uint32_t) ((uint64_t) metric * w * e->load_pct / 10000);
124 }
125 return (uint32_t) (((100 - w) * (uint64_t) metric
126 + w * (uint64_t) e->load_pct) / 100);
127 }
128
129 if (for_write) {
130 /* Capacity dominates; machine load discounts it by up to its weight
131 * share at full load (mirrors the shape of the legacy write blend). */
132 machine = sched_machine_load(e);
133 return e->free_mb
134 - (uint32_t) ((uint64_t) e->free_mb * machine / 200);
135 }
136
137 num = (uint64_t) brix_srv_sched.cpu * e->load5[0]
138 + (uint64_t) brix_srv_sched.io * e->load5[1]
139 + (uint64_t) brix_srv_sched.runq * e->load5[2]
140 + (uint64_t) brix_srv_sched.mem * e->load5[3]
141 + (uint64_t) brix_srv_sched.pag * e->load5[4]
142 + (uint64_t) brix_srv_sched.space * e->util_pct;
143 den = (uint64_t) brix_srv_sched.cpu + brix_srv_sched.io
144 + brix_srv_sched.runq + brix_srv_sched.mem + brix_srv_sched.pag
145 + brix_srv_sched.space;
146 return (uint32_t) (num / den); /* den > 0: sched_active() checked */
147}

149/*
150 * brix_srv_select_stage — §2.5: pick the roomiest stage-capable node.
151 *
152 * WHAT: Scans live (non-blacklisted, non-stale-excluded) entries whose export
153 * prefixes cover path AND that advertised staging (kYR_status stage
154 * bit), picking the one with the most free space. Returns 1 and fills
155 * host/port, 0 when no stage-capable node matches.
156 * WHY: Stock cmsd's two-phase read selection: prefer a node that HAS the
157 * file; else route the client to a node that can RECALL it — by free
158 * space, because the recall is a write. The caller (locate/open) runs
159 * this only after the holder plane (loc cache / kYR_have) came up
160 * empty, so this function needs no holder knowledge.
161 * HOW: Same scan shape as srv_select_core with the write-direction metric,
162 * filtered to e->stage == 1; peers and managers never stage for us.
163 */
164int
165brix_srv_select_stage(const char *path, char *host_out, size_t host_size,
166 uint16_t *port_out)
167{
168 brix_srv_table_t *tbl;
169 brix_srv_entry_t *e;
170 ngx_uint_t i;
171 int best = -1;
172 uint32_t best_free = 0;
173
174 tbl = srv_table();
175 if (tbl == NULL) {
176 return 0;
177 }
178
179 ngx_shmtx_lock(&brix_srv_mutex);
180
181 for (i = 0; i < tbl->capacity; i++) {
182 e = &tbl->slots[i];
183 if (!e->in_use || !e->stage
184 || e->role[0] == 'M' || e->role[0] == 'R'
185 || (e->role[0] == 'P' && e->role[1] == '\0'))
186 {
187 continue;
188 }
189 if (e->blacklisted_until != 0
190 && e->blacklisted_until > ngx_current_msec)
191 {
192 continue;
193 }
194 if (!srv_path_matches(e->paths, path)) {
195 continue;
196 }
197 if (best == -1 || e->free_mb > best_free) {
198 best = (int) i;
199 best_free = e->free_mb;
200 }
201 }
202
203 if (best >= 0) {
204 e = &tbl->slots[best];
205 ngx_cpystrn((u_char *) host_out, (u_char *) e->host, host_size);
206 *port_out = e->port;
207 }
208
209 ngx_shmtx_unlock(&brix_srv_mutex);
210 return best >= 0;
211}
