10/* File: src/core/aio/uring.c — the io_uring per-worker ring singleton and the
11 * init/exit lifecycle orchestrator (phase-79 split from the original 1033-line
12 * uring.c).
13 * WHAT: Owns the file-static per-worker ring singleton + its accessor, and the
14 * two lifecycle entry points: brix_uring_init_worker (drive the ordered
15 * bring-up steps then publish the ring) and brix_uring_exit_worker (tear
16 * it down). The ordered register-phase steps live in uring_bringup.c;
17 * detection/gating in uring_probe.c; the CQE reaper + slot table in
18 * uring_reap.c; submission in uring_submit.c.
19 *
20 * WHY: The backend must be invisible unless explicitly built (pkg-config
21 * liburing) and runtime-available (opcode probe). `auto`/`off` always
22 * start and silently fall back; `on` is a hard requirement — this
23 * orchestrator is the §32.7 backstop that refuses to run a worker that
24 * cannot honour `on` (brix_uring_init_fail returns NGX_ERROR under `on`).
25 *
26 * HOW: All liburing-specific code is under #if (BRIX_HAVE_LIBURING). When the
27 * macro is undefined the lifecycle entry points compile to inert stubs. */

49/*
50 * brix_uring_teardown — release every ring resource that has been brought up,
51 * in reverse order, idempotently (driven by which fields are set). Mirrors
52 * nginx core's eventfd cleanup: ngx_free_connection + close(fd) rather than
53 * ngx_close_connection, to avoid double-closing the eventfd. No goto.
54 */
55static void
56brix_uring_teardown(brix_uring_t *u)
57{
58 if (u->ring_active && u->eventfd >= 0) {
59 io_uring_unregister_eventfd(&u->ring);
60 }
61
62 if (u->evc != NULL) {
63 if (u->evc->read->active) {
64 (void) ngx_del_event(u->evc->read, NGX_READ_EVENT, 0);
65 }
66 ngx_free_connection(u->evc);
67 u->evc->fd = (ngx_socket_t) -1;
68 u->evc = NULL;
69 }
70
71 if (u->eventfd >= 0) {
72 close(u->eventfd);
73 u->eventfd = -1;
74 }
75
76 if (u->ring_active) {
77 io_uring_queue_exit(&u->ring);
78 u->ring_active = 0;
79 }
80
81 u->enabled = 0;
82 /* u->slots is deliberately KEPT: the table is cycle-pool-owned (freed with
83 * the worker, never manually) and is reused verbatim by a P44-A re-enable,
84 * so repeated kill-switch flips cannot grow the cycle pool. */
85}

87/*
88 * brix_uring_init_fail — log the bring-up failure, tear down whatever was set
89 * up, and return the right verdict: NGX_ERROR under `on` (the worker refuses to
90 * run on the thread pool — master respawns; the §32.7 backstop) or NGX_OK under
91 * `auto` (silent degrade to the thread pool).
92 */
93static ngx_int_t
94brix_uring_init_fail(brix_uring_t *u, ngx_cycle_t *cycle, ngx_uint_t mode_on,
95 const char *what)
96{
97 /* Hoist the level into a variable: the ngx_log_error macro does not
98 * parenthesize its level argument, so a raw ternary would bind under >=. */
99 ngx_uint_t level = mode_on ? NGX_LOG_EMERG : NGX_LOG_NOTICE;
100
101 ngx_log_error(level, cycle->log, ngx_errno,
102 "brix: io_uring bring-up failed at %s%s", what,
103 mode_on
104 ? " — \"brix_io_uring on\" requires it; this worker refuses to run"
105 : "; falling back to the thread pool");
106
107 brix_uring_teardown(u);
108 return mode_on ? NGX_ERROR : NGX_OK;
109}

111/*
112 * brix_uring_bring_up — run the ordered bring-up steps and publish the ring.
113 *
114 * Shared by the after-fork init (brix_uring_init_worker) and the P44-A
115 * runtime re-enable (brix_uring_quiesce_tick): queue_init [R_DISABLED if
116 * restricting] -> register restrictions + enable -> NOP self-test -> burst
117 * self-test -> fake-connection epoll bridge -> slot table -> kill-switch
118 * attach. Returns NULL with u->enabled = 1 on success, or the failing step's
119 * name with partial state left for the caller to tear down (the two callers
120 * differ in verdict: init_fail vs retry-next-tick).
121 */
122static const char *
123brix_uring_bring_up(brix_uring_t *u, ngx_cycle_t *cycle,
124 const brix_uring_scan_t *scan)
125{
126 const char *what;
127
128 u->log = cycle->log;
129 u->eventfd = -1;
130 u->evc = NULL;
131 u->inflight = 0;
132 u->ring_active = 0;
133 u->enabled = 0;
134 u->restrict_ops = 0;
135 u->queue_depth = (uint32_t) scan->depth;
136
137 /* 1-3. create + notify-wire + restrict + enable the ring. */
138 what = uring_setup_rings(u, scan->want_restrict);
139 if (what != NULL) {
140 return what;
141 }
142
143 /* 4. NOP self-test: prove submit -> complete works AND that the registered
144 * eventfd actually delivers the completion notification. */
145 what = uring_selftest_nop(u);
146 if (what != NULL) {
147 return what;
148 }
149
150 /* 4b. UNDER-LOAD delivery self-test: fill the ring with queue_depth ops and
151 * require EVERY completion to arrive via the eventfd within the deadline. */
152 what = uring_selftest_burst(u);
153 if (what != NULL) {
154 return what;
155 }
156
157 /* 5. wire the eventfd into the worker's epoll via a fake connection. */
158 what = uring_install_eventfd(u, cycle);
159 if (what != NULL) {
160 return what;
161 }
162
163 /* 6. completion-slot table (pool-owned, reused across quiesce cycles). */
164 what = uring_register_buffers(u, cycle);
165 if (what != NULL) {
166 return what;
167 }
168
169 /* SB-W5b: attach the cross-worker kill-switch flag (NULL if the zone was
170 * not registered — the selector then reads "enabled"). */
171 u->disabled_flag = brix_uring_killswitch_ptr();
172
173 u->enabled = 1;
174 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
175 "brix: io_uring disk-I/O backend active (queue_depth=%ui%s)",
176 (ngx_uint_t) u->queue_depth,
177 u->restrict_ops ? ", restricted" : "");
178
179 return NULL;
180}

182/*
183 * brix_uring_init_worker — create this worker's ring after fork.
184 *
185 * Scans every enabled server block: the ring is created if any block wants
186 * io_uring (mode on/auto); queue depth is the max requested; restrictions are
187 * applied unless any wanting block turned them off. Any bring-up failure
188 * routes through brix_uring_init_fail (NGX_OK under auto, NGX_ERROR under on).
189 * When no block wants io_uring this is a no-op.
190 */
191ngx_int_t
192brix_uring_init_worker(ngx_cycle_t *cycle)
193{
194 brix_uring_t *u = &brix_uring_worker_ring;
195 brix_uring_scan_t scan;
196 const char *what;
197
198 if (uring_probe_features(cycle, &scan) != NGX_OK) {
199 return NGX_OK; /* nobody wants a ring / auto blocks on a bare host */
200 }
201
202 what = brix_uring_bring_up(u, cycle, &scan);
203 if (what != NULL) {
204 return brix_uring_init_fail(u, cycle, scan.mode_on, what);
205 }
206
207 /* P44-A: save the winning conf so a runtime quiesce can re-create the
208 * ring, then start the per-worker maintenance timer (quiesce/re-enable
209 * driver) and the panic-file mirror if a path was configured. */
210 brix_uring_saved_scan = scan;
211 brix_uring_rearm_ok = 1;
212 (void) brix_uring_panicfile_arm(cycle, &scan.panic_file);
213 (void) brix_uring_maint_arm(cycle);
214
215 return NGX_OK;
216}

218/*
219 * brix_uring_quiesce_tick — P44-A ring quiesce-and-teardown (§36 tier).
220 *
221 * Driven by the per-worker maintenance timer (uring_admin.c, 2 s cadence).
222 * While the kill switch merely stops NEW submissions at the selector, the
223 * quiesce goes further: once every in-flight CQE has drained, the ring and
224 * its eventfd are torn down so the kernel attack surface (the ring fds)
225 * disappears from the process, not just from the hot path. When the switch
226 * clears, the ring is re-created from the saved bring-up conf — including the
227 * full self-test ladder, so a re-enable can never resurrect a ring the worker
228 * would not have trusted at boot. A failed re-enable is logged and retried
229 * on the next tick (never fatal at runtime, even under mode `on`: the kill
230 * switch is an incident tool and the pool tier keeps serving).
231 */
232void
233brix_uring_quiesce_tick(void)
234{
235 brix_uring_t *u = &brix_uring_worker_ring;
236 ngx_cycle_t *cycle = (ngx_cycle_t *) ngx_cycle;
237 const char *what;
238
239 if (!brix_uring_rearm_ok) {
240 return; /* this worker never had a ring: nothing to manage */
241 }
242
243 if (brix_uring_killswitch_get()) {
244 /* Disabled: the ring stays alive only while ops are still in flight
245 * (the reaper needs it); the moment the last CQE drains, drop it. */
246 if (u->enabled && u->inflight == 0) {
247 brix_uring_teardown(u);
248 ngx_log_error(NGX_LOG_NOTICE, cycle->log, 0,
249 "brix: io_uring ring quiesced (kill switch): "
250 "ring + eventfd released");
251 }
252 return;
253 }
254
255 if (u->enabled) {
256 return; /* running normally */
257 }
258
259 what = brix_uring_bring_up(u, cycle, &brix_uring_saved_scan);
260 if (what != NULL) {
261 ngx_log_error(NGX_LOG_ERR, cycle->log, ngx_errno,
262 "brix: io_uring re-enable failed at %s; "
263 "retrying on the next maintenance tick", what);
264 brix_uring_teardown(u);
265 }
266}
