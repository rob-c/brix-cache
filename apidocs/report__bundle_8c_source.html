30/* ---- Render the generation-bound bundle completion record ----
31 *
32 * WHAT: Emits the schema, timestamp and tool version that commit one bundle.
33 * WHY: Consumers must reject a JSON report whose later renderers did not finish.
34 * HOW: Bind the marker to immutable metadata already present in report.json.
35 */
36static void
37report_bundle_marker(FILE *out, const brix_conform_report_meta *meta,
38 const brix_conform_run_results *results)
39{
40 (void) results;
41 fputs("{\"schema\":\"brix.conform.bundle.v1\",\"timestamp\":", out);
42 brix_conform_json_string(out, meta->timestamp);
43 fputs(",\"tool_version\":", out);
44 brix_conform_json_string(out, meta->tool_version);
45 fputs("}\n", out);
46}

48/* ---- Remove any prior completion record ----
49 *
50 * WHAT: Returns zero after removing the marker or finding it absent.
51 * WHY: A failed rewrite must never inherit a successful older generation.
52 * HOW: Unlink only the fixed marker name beneath the already-open directory.

61/* ---- Open one private report file beneath a trusted directory fd ----
62 *
63 * WHAT: Returns an fdopen stream or NULL without following a final symlink.
64 * WHY: An attacker must not redirect a report write onto another file.
65 * HOW: openat with create/truncate, nofollow and mode 0600, then fdopen.
66 */
67static FILE *
68report_file_open(int dirfd, const char *name)
69{
70 int fd = openat(dirfd, name,
71 O_WRONLY | O_CREAT | O_TRUNC | O_NOFOLLOW, 0600);
72
73 if (fd < 0) return NULL;
74 return fdopen(fd, "w");
75}

77/* ---- Render and durably close one report file ----
78 *
79 * WHAT: Returns zero only when opening, rendering, flushing and close succeed.
80 * WHY: A partial artifact must make the overall run report failure explicit.
81 * HOW: Open securely, invoke the pure writer, fflush, fsync, fclose.
82 */
83static int
84report_one(int dirfd, const char *name, report_writer writer,
85 const brix_conform_report_meta *meta,
86 const brix_conform_run_results *results)
87{
88 FILE *stream = report_file_open(dirfd, name);
89 int fd;
90 int failed = 0;
91
92 if (stream == NULL) return -1;
93 writer(stream, meta, results);
94 fd = fileno(stream);
95 if (fflush(stream) != 0) failed = 1;
96 if (!failed && fsync(fd) != 0) failed = 1;
97 if (fclose(stream) != 0) failed = 1;
98 return failed ? -1 : 0;
99}

125/* ---- Open/create the private report directory ----
126 *
127 * WHAT: Returns a no-follow directory fd or -1 with a bounded diagnostic.
128 * WHY: Directory setup and argument validation are one reusable failure seam.
129 * HOW: Validate pointers, mkdir 0700, then open as a directory without links.
130 */
131static int
132report_directory_open(const char *out_dir,
133 const brix_conform_report_meta *meta,
134 const brix_conform_run_results *results, char *error, size_t error_size)
135{
136 int dirfd;
137
138 if (out_dir == NULL || meta == NULL || results == NULL) {
139 if (error != NULL && error_size > 0)
140 snprintf(error, error_size, "invalid report arguments");
141 return -1;
142 }
143 if (mkdir(out_dir, 0700) != 0 && errno != EEXIST) {
144 if (error != NULL && error_size > 0)
145 snprintf(error, error_size, "cannot create report directory: %s",
146 strerror(errno));
147 return -1;
148 }
149 dirfd = open(out_dir, O_RDONLY | O_DIRECTORY | O_NOFOLLOW);
150 if (dirfd < 0 && error != NULL && error_size > 0)
151 snprintf(error, error_size, "cannot open report directory: %s",
152 strerror(errno));
153 return dirfd;
154}

156static int
157report_bundle_fail(int dirfd, int remove_marker, const char *message,

173/* ---- Write the complete report artifact bundle ----
174 *
175 * WHAT: Returns zero after six durable artifacts and their commit marker.
176 * WHY: CI and site review consumers require consistent simultaneous formats.
177 * HOW: Create/open private dir without symlink following, render six files.
178 */
179int
180brix_conform_report_bundle(const char *out_dir,
181 const brix_conform_report_meta *meta,
182 const brix_conform_run_results *results, char *error, size_t error_size)
183{
184 int dirfd;
185
186 dirfd = report_directory_open(out_dir, meta, results, error, error_size);
187 if (dirfd < 0) return -1;
188 if (report_bundle_marker_remove(dirfd) != 0) {
189 return report_bundle_fail(dirfd, 0,
190 "cannot clear report completion marker", error, error_size);
191 }
192 if (report_all(dirfd, meta, results) != 0) {
193 return report_bundle_fail(dirfd, 0, "cannot write report bundle",
194 error, error_size);
195 }
196 if (report_one(dirfd, BRIX_CONFORM_BUNDLE_MARKER, report_bundle_marker,
197 meta, results) != 0) {
198 return report_bundle_fail(dirfd, 1, "cannot commit report bundle",
199 error, error_size);
200 }
201 if (fsync(dirfd) != 0) {
202 return report_bundle_fail(dirfd, 1, "cannot sync report directory",
203 error, error_size);
204 }
205 close(dirfd);
206 if (error != NULL && error_size > 0) error[0] = '\0';
207 return 0;
208}
