1/*
2 * oci_registry.c — the local registry surface's router (§D4.1).
3 *
4 * WHAT: the entry point for a `brix_oci_registry on` location: gate →
5 * authorize → dispatch to the read path (serve from the local store) or
6 * to one of the write engines (upload sessions, manifest PUT, DELETE).
7 * WHY: the mirror and the registry share a URL grammar and an error
8 * envelope, but nothing else: one caches somebody else's objects and
9 * may refetch anything it drops, the other IS the source of truth and
10 * may drop nothing. Keeping the two routers apart is what stops that
11 * distinction from being one `if (lcf->mirror)` away from being lost —
12 * most visibly on the read path, where a registry that "filled" a miss
13 * from an upstream would silently launder unpushed content into a
14 * repository somebody trusts.
15 * HOW: reads go through the same serving pipeline as every other protocol in
16 * the tree (brix_vfs_open → brix_http_serve_file_ranged), so ranges,
17 * conditionals and the TLS/sendfile split (INVARIANT #2) are handled
18 * once, correctly, elsewhere. This file opens nothing itself.
19 */

40/* Resolve what the request names to a path in the store. Manifests reached by
41 * tag are resolved through the tag pointer first — that indirection is the
42 * whole reason tags are mutable and digests are not. */
43static ngx_int_t
44oci_registry_resolve(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
45 brix_oci_store_t *st, char *path, size_t path_size)
46{
47 brix_oci_digest_t d;
48 char tagfile[PATH_MAX];
49 char line[BRIX_OCI_DIGEST_STRLEN];
50 ssize_t n;
51
52 if (ctx->req.cls == BRIX_OCI_REQ_BLOB) {
53 if (brix_oci_store_blob_path(st, &ctx->req.digest, path, path_size)
54 != 0)
55 {
56 return NGX_ERROR;
57 }
58 /* The CAS is global, but a repository only serves what it has been
59 * told it holds: without the reference mark, one tenant could read
60 * another tenant's private layer by guessing nothing more than its
61 * digest — which a leaked manifest hands over for free. */
62 if (brix_oci_store_layer_path(st, ctx->req.name, ctx->req.name_len,
63 &ctx->req.digest, tagfile,
64 sizeof(tagfile)) != 0
65 || !brix_oci_store_exists(tagfile, NULL))
66 {
67 return NGX_DECLINED;
68 }
69 return NGX_OK;
70 }
71
72 if (ctx->req.ref_is_digest) {
73 return brix_oci_store_manifest_path(st, ctx->req.name,
74 ctx->req.name_len, &ctx->req.digest, NULL,
75 path, path_size) == 0 ? NGX_OK : NGX_ERROR;
76 }
77
78 if (brix_oci_store_tag_path(st, &ctx->req, tagfile, sizeof(tagfile)) != 0) {
79 return NGX_ERROR;
80 }
81 n = brix_oci_store_get_text(tagfile, line, sizeof(line));
82 if (n <= 0) {
83 return NGX_DECLINED;
84 }
85 while (n > 0 && (line[n - 1] == '\n' || line[n - 1] == '\r')) {
86 line[--n] = '\0';
87 }
88 if (brix_oci_digest_parse(line, (size_t) n, &d) != 0) {
89 return NGX_ERROR; /* a corrupt tag file is our bug */
90 }
91 return brix_oci_store_manifest_path(st, ctx->req.name, ctx->req.name_len,
92 &d, NULL, path, path_size) == 0 ? NGX_OK : NGX_ERROR;
93}

96/* Serve one stored object. No fill, no upstream, no coalescing: on this
97 * surface a miss is a miss, and inventing bytes for it is precisely the
98 * failure mode the surface exists to prevent. */
99static ngx_int_t
100oci_registry_serve(ngx_http_request_t *r, ngx_http_brix_oci_loc_conf_t *lcf,
101 ngx_http_brix_oci_ctx_t *ctx, brix_oci_store_t *st)
102{
103 brix_http_serve_opts_t opts;
104 brix_http_serve_result_t result;
105 brix_oci_present_t *pres;
106 brix_vfs_ctx_t vctx;
107 brix_vfs_file_t *fh;
108 brix_vfs_stat_t vst;
109 char path[PATH_MAX];
110 int is_tls = 0;
111 int vfs_err = 0;
112 ngx_int_t rc;
113
114 rc = oci_registry_resolve(r, ctx, st, path, sizeof(path));
115 if (rc == NGX_ERROR) {
116 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
117 BRIX_OCI_ERR_UNAVAILABLE, NULL);
118 }
119 if (rc == NGX_DECLINED) {
120 ctx->disp = BRIX_OCI_OUT_ERROR;
121 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
122 (ctx->req.cls == BRIX_OCI_REQ_BLOB)
123 ? BRIX_OCI_ERR_BLOB_UNKNOWN
124 : BRIX_OCI_ERR_MANIFEST_UNKNOWN, NULL);
125 }
126
127 is_tls = brix_http_request_is_tls(r);
128 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log, BRIX_PROTO_OCI,
129 st->root, "", BRIX_VFS_MUTATION_READ_ONLY, is_tls, NULL, path);
130 brix_http_vfs_bind_no_rules(&lcf->common, &vctx);
131 vctx.sd = brix_vfs_backend_resolve(st->root, r->connection->log);
132
133 fh = brix_vfs_open(&vctx, BRIX_VFS_O_READ, &vfs_err);
134 if (fh == NULL) {
135 ctx->disp = BRIX_OCI_OUT_ERROR;
136 return brix_oci_error(r, brix_oci_errno_status(r, vfs_err),
137 (ctx->req.cls == BRIX_OCI_REQ_BLOB)
138 ? BRIX_OCI_ERR_BLOB_UNKNOWN
139 : BRIX_OCI_ERR_MANIFEST_UNKNOWN, NULL);
140 }
141 if (brix_vfs_file_stat(fh, &vst) != NGX_OK || vst.is_directory) {
142 brix_vfs_close(fh, r->connection->log);
143 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
144 BRIX_OCI_ERR_MANIFEST_UNKNOWN, NULL);
145 }
146
147 pres = ngx_pcalloc(r->pool, sizeof(*pres));
148 if (pres == NULL
149 || brix_oci_present_prepare(r, lcf, ctx, fh, &vst, path, pres)
150 != NGX_OK)
151 {
152 brix_vfs_close(fh, r->connection->log);
153 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
154 BRIX_OCI_ERR_UNAVAILABLE, NULL);
155 }
156
157 ctx->disp = BRIX_OCI_OUT_LOCAL;
158 r->allow_ranges = 1;
159
160 brix_oci_present_serve_opts(&opts, pres);
161 return brix_http_serve_file_ranged(r, fh, &vst, path, &opts, &result);
162}

165ngx_int_t
166brix_oci_registry_tags(ngx_http_request_t *r, ngx_http_brix_oci_ctx_t *ctx,
167 brix_oci_store_t *st)
168{
169 u_char *body;
170 char *tags;
171 size_t used;
172 char *p;
173 int count;
174
175 tags = ngx_palloc(r->pool, OCI_TAGS_MAX);
176 body = ngx_palloc(r->pool, OCI_TAGS_MAX + BRIX_OCI_KEY_MAX);
177 if (tags == NULL || body == NULL) {
178 return NGX_HTTP_INTERNAL_SERVER_ERROR;
179 }
180
181 count = brix_oci_store_tag_list(st, ctx->req.name, ctx->req.name_len,
182 tags, OCI_TAGS_MAX);
183 if (count < 0) {
184 return brix_oci_error(r, NGX_HTTP_NOT_FOUND,
185 BRIX_OCI_ERR_NAME_UNKNOWN, NULL);
186 }
187
188 used = (size_t) snprintf((char *) body, OCI_TAGS_MAX + BRIX_OCI_KEY_MAX,
189 "{\"name\":\"%.*s\",\"tags\":[",
190 (int) ctx->req.name_len, ctx->req.name);
191
192 /* The store hands back newline-separated names; the wire wants a JSON
193 * array. Every name here has already passed the tag grammar, so it needs
194 * no escaping — the grammar excludes quote and backslash by construction
195 * (§0.7.2), which is why this can be a straight copy. */
196 for (p = tags; *p != '\0'; ) {
197 char *nl = strchr(p, '\n');
198 size_t len = (nl != NULL) ? (size_t) (nl - p) : ngx_strlen(p);
199
200 if (used + len + 4 >= OCI_TAGS_MAX + BRIX_OCI_KEY_MAX) {
201 break;
202 }
203 if (p != tags) {
204 body[used++] = ',';
205 }
206 body[used++] = '"';
207 ngx_memcpy(body + used, p, len);
208 used += len;
209 body[used++] = '"';
210
211 if (nl == NULL) {
212 break;
213 }
214 p = nl + 1;
215 }
216 body[used++] = ']';
217 body[used++] = '}';
218
219 ctx->disp = BRIX_OCI_OUT_LOCAL;
220 return brix_oci_send_body(r, NGX_HTTP_OK, "application/json", body, used);
221}

224/* Route → engine. The read/write split is explicit rather than derived from
225 * the method alone, because "GET on an upload session" and "GET on a blob"
226 * are different surfaces that happen to share a verb. */
227static ngx_int_t
228oci_registry_dispatch(ngx_http_request_t *r,
229 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx,
230 brix_oci_store_t *st)
231{
232 int is_read = (r->method & (NGX_HTTP_GET | NGX_HTTP_HEAD)) != 0;
233
234 switch (ctx->req.cls) {
235
236 case BRIX_OCI_REQ_UPLOAD_START:
237 return brix_oci_upload_start(r, lcf, ctx, st);
238
239 case BRIX_OCI_REQ_UPLOAD_SESSION:
240 return brix_oci_upload_session(r, lcf, ctx, st);
241
242 case BRIX_OCI_REQ_TAGS_LIST:
243 if (!is_read) {
244 break;
245 }
246 return brix_oci_registry_tags(r, ctx, st);
247
248 case BRIX_OCI_REQ_REFERRERS:
249 if (!is_read) {
250 break;
251 }
252 return brix_oci_referrers_get(r, ctx, st);
253
254 case BRIX_OCI_REQ_MANIFEST:
255 if (is_read) {
256 return oci_registry_serve(r, lcf, ctx, st);
257 }
258 if (r->method == NGX_HTTP_PUT) {
259 return brix_oci_manifest_put(r, lcf, ctx, st);
260 }
261 if (r->method == NGX_HTTP_DELETE) {
262 return brix_oci_registry_delete(r, ctx, st);
263 }
264 break;
265
266 case BRIX_OCI_REQ_BLOB:
267 if (is_read) {
268 return oci_registry_serve(r, lcf, ctx, st);
269 }
270 if (r->method == NGX_HTTP_DELETE) {
271 return brix_oci_registry_delete(r, ctx, st);
272 }
273 break;
274
275 default:
276 break;
277 }
278
279 ctx->disp = BRIX_OCI_OUT_REFUSED;
280 return brix_oci_error(r, NGX_HTTP_NOT_ALLOWED, BRIX_OCI_ERR_UNSUPPORTED,
281 "that method is not defined for this endpoint");
282}

285ngx_int_t
286brix_oci_registry_handle(ngx_http_request_t *r,
287 ngx_http_brix_oci_loc_conf_t *lcf, ngx_http_brix_oci_ctx_t *ctx)
288{
289 brix_oci_store_t st;
290 char principal[256];
291 char *uri;
292 ngx_int_t rc;
293
294 /* The classified request holds SPANS into this buffer, and a body-reading
295 * method (PATCH, PUT) is re-entered by nginx long after this frame has
296 * returned — so the URI copy lives in the request pool, never on the
297 * stack. A stack buffer here reads back as garbage in the Location the
298 * second PATCH of a chunked layer emits. */
299 uri = ngx_pnalloc(r->pool, BRIX_OCI_KEY_MAX);
300 if (uri == NULL) {
301 return NGX_HTTP_INTERNAL_SERVER_ERROR;
302 }
303
304 if (brix_oci_store_init(&st, lcf) != NGX_OK) {
305 return brix_oci_error(r, NGX_HTTP_INTERNAL_SERVER_ERROR,
306 BRIX_OCI_ERR_UNAVAILABLE,
307 "this location has no usable registry root");
308 }
309
310 rc = brix_oci_registry_gate(r, ctx, uri, BRIX_OCI_KEY_MAX);
311 if (rc != NGX_DECLINED) {
312 return rc;
313 }
314
315 /* Authorization runs AFTER classification and BEFORE any store access, so
316 * the scope check has a validated repository name to test against and no
317 * unauthorized request has yet learned whether an object exists. */
318 rc = brix_oci_registry_authz(r, lcf, ctx, principal, sizeof(principal));
319 if (rc == NGX_DONE) {
320 return NGX_OK; /* refused; the envelope is written */
321 }
322 if (rc != NGX_OK) {
323 return rc;
324 }
325
326 return oci_registry_dispatch(r, lcf, ctx, &st);
327}
