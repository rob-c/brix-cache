16/*
17 * brix_tpc_metric_book — the ONE place a terminal TPC outcome reaches the
18 * unified counters.
19 *
20 * WHAT: Books the transfer twice over, into two families that answer different
21 * questions: brix_tpc_transfers_total/_bytes_total (TPC-specific, carries
22 * the pull/push direction) and brix_io_ops_total{op="tpc"} (the unified
23 * per-op ledger, so a TPC shows up alongside every other operation the
24 * server performed).
25 * WHY: `op="tpc"` was a declared-but-unreachable slot: the WebDAV op mapping
26 * named it for COPY, but the protocol-level op_done is deliberately
27 * restricted to the data plane (READ/WRITE) to avoid double-booking the
28 * VFS-observed namespace ops, so nothing ever incremented it. Booking it
29 * here — the single call site both transports already funnel through —
30 * gives the row exactly one owner (see the owner table in
31 * docs/08-metrics-monitoring/metrics-bug-patterns.md, Pattern 6).
32 * HOW: Count-only for the unified row: a TPC has no request-scoped duration
33 * to file (see brix_metric_op_count).
34 */
35static void
36brix_tpc_metric_book(ngx_uint_t protocol, ngx_uint_t direction, size_t bytes,
37 brix_err_class_t err)
38{
39 brix_proto_t proto = brix_tpc_metric_proto(protocol);
40
41 brix_metric_tpc(proto, direction == BRIX_TPC_DIR_PUSH, bytes, err);
42 brix_metric_op_count(proto, BRIX_METRIC_OP_TPC, err);
43}
