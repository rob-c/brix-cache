1/*
2 * cta_shm.c — the CTA request queue's shared-memory home. See cta_shm.h.
3 *
4 * Follows the same shape as src/fs/xfer/stage_waiter.c: the table is
5 * slab-allocated via brix_shm_table_alloc() so it never clobbers the slab-pool
6 * header that nginx's SIGCHLD handler walks on every child death, and the
7 * mutex is bound to the pool's own lock word so a worker dying mid-operation
8 * is recovered rather than stranding the queue for everyone.
9 */

44/*
45 * Zone init. Runs in the MASTER, before fork.
46 *
47 * That timing is the whole design: the journal fd opened here is inherited by
48 * every worker, so all of them append to one open file description with one
49 * shared O_APPEND offset and one journal serves the entire set. Replay likewise
50 * happens once, here, rather than once per worker — which is what used to make
51 * every worker start from the same next_id and then hand out colliding ids.
52 */
53static ngx_int_t
54cta_shm_init_zone(ngx_shm_zone_t *shm_zone, void *data)
55{
56 brix_cta_queue_t *q;
57 ngx_flag_t fresh;
58
59 q = brix_shm_table_alloc(shm_zone, data, sizeof(brix_cta_queue_t),
60 &cta_mtx, &fresh);
61 if (q == NULL) {
62 return NGX_ERROR;
63 }
64 if (!fresh) {
65 return NGX_OK; /* reload re-attached: live state must be preserved */
66 }
67 cta_queue_init(q);
68 if (cta_journal_path[0] != '\0'
69 && cta_queue_open_journal(q, cta_journal_path) != 0)
70 {
71 BRIX_LOG(NGX_LOG_ERR, shm_zone->shm.log, ngx_errno,
72 "brix: CTA journal \"%s\" unusable; the queue will not "
73 "survive a restart", cta_journal_path);
74 /* Not fatal: an unwritable journal degrades restart recovery, it does
75 * not make the service wrong. Failing here would take a working
76 * deployment down over a permissions change. */
77 }
78 return NGX_OK;
79}

82ngx_int_t
83brix_cta_shm_configure(ngx_conf_t *cf, ngx_str_t *journal)
84{
85 ngx_str_t zone_name = ngx_string("brix_cta_queue");
86 size_t len;
87
88 if (journal != NULL && journal->len > 0) {
89 len = journal->len;
90 if (len >= sizeof(cta_journal_path)) {
91 len = sizeof(cta_journal_path) - 1;
92 }
93 ngx_memcpy(cta_journal_path, journal->data, len);
94 cta_journal_path[len] = '\0';
95 } else {
96 cta_journal_path[0] = '\0';
97 }
98
99 cta_zone = ngx_shared_memory_add(cf, &zone_name,
100 brix_shm_zone_size(sizeof(brix_cta_queue_t)),
101 &ngx_stream_brix_module);
102 if (cta_zone == NULL) {
103 return NGX_ERROR;
104 }
105 brix_shm_zone_warn_on_resize(cf, cta_zone, "brix_ssi_service cta");
106 cta_zone->init = cta_shm_init_zone;
107 cta_zone->data = (void *) 1;
108 return NGX_OK;
109}

113 *
114 * Each holds cta_mtx for exactly one queue operation. Nothing here calls an
115 * executor: a lock held across an archive/retrieve would serialise every

119cta_req_t *
120brix_cta_shm_submit(const cta_request_t *r, const char *owner)
121{
122 brix_cta_queue_t *q = brix_cta_shm_queue();
123 cta_req_t *e;
124
125 if (q == NULL) {
126 return NULL;
127 }
128 ngx_shmtx_lock(&cta_mtx);
129 e = cta_queue_submit(q, r, owner);
130 ngx_shmtx_unlock(&cta_mtx);
131 return e;
132}
