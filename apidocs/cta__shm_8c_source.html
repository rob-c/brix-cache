1/*
2 * cta_shm.c — the CTA request queue's shared-memory home. See cta_shm.h.
3 *
4 * Follows the same shape as src/fs/xfer/stage_waiter.c: the table is
5 * slab-allocated via brix_shm_table_alloc() so it never clobbers the slab-pool
6 * header that nginx's SIGCHLD handler walks on every child death, and the
7 * mutex is bound to the pool's own lock word so a worker dying mid-operation
8 * is recovered rather than stranding the queue for everyone.
9 */

43/*
44 * Zone init. Runs in the MASTER, before fork.
45 *
46 * That timing is the whole design: the journal fd opened here is inherited by
47 * every worker, so all of them append to one open file description with one
48 * shared O_APPEND offset and one journal serves the entire set. Replay likewise
49 * happens once, here, rather than once per worker — which is what used to make
50 * every worker start from the same next_id and then hand out colliding ids.
51 */
52static ngx_int_t
53cta_shm_init_zone(ngx_shm_zone_t *shm_zone, void *data)
54{
55 brix_cta_queue_t *q;
56 ngx_flag_t fresh;
57
58 q = brix_shm_table_alloc(shm_zone, data, sizeof(brix_cta_queue_t),
59 &cta_mtx, &fresh);
60 if (q == NULL) {
61 return NGX_ERROR;
62 }
63 if (!fresh) {
64 return NGX_OK; /* reload re-attached: live state must be preserved */
65 }
66 cta_queue_init(q);
67 if (cta_journal_path[0] != '\0'
68 && cta_queue_open_journal(q, cta_journal_path) != 0)
69 {
70 ngx_log_error(NGX_LOG_ERR, shm_zone->shm.log, ngx_errno,
71 "brix: CTA journal \"%s\" unusable; the queue will not "
72 "survive a restart", cta_journal_path);
73 /* Not fatal: an unwritable journal degrades restart recovery, it does
74 * not make the service wrong. Failing here would take a working
75 * deployment down over a permissions change. */
76 }
77 return NGX_OK;
78}

81ngx_int_t
82brix_cta_shm_configure(ngx_conf_t *cf, ngx_str_t *journal)
83{
84 ngx_str_t zone_name = ngx_string("brix_cta_queue");
85 size_t len;
86
87 if (journal != NULL && journal->len > 0) {
88 len = journal->len;
89 if (len >= sizeof(cta_journal_path)) {
90 len = sizeof(cta_journal_path) - 1;
91 }
92 ngx_memcpy(cta_journal_path, journal->data, len);
93 cta_journal_path[len] = '\0';
94 } else {
95 cta_journal_path[0] = '\0';
96 }
97
98 cta_zone = ngx_shared_memory_add(cf, &zone_name,
99 brix_shm_zone_size(sizeof(brix_cta_queue_t)),
100 &ngx_stream_brix_module);
101 if (cta_zone == NULL) {
102 return NGX_ERROR;
103 }
104 brix_shm_zone_warn_on_resize(cf, cta_zone, "brix_ssi_service cta");
105 cta_zone->init = cta_shm_init_zone;
106 cta_zone->data = (void *) 1;
107 return NGX_OK;
108}

111/* ---- locked wrappers ---------------------------------------------------
112 *
113 * Each holds cta_mtx for exactly one queue operation. Nothing here calls an
114 * executor: a lock held across an archive/retrieve would serialise every
115 * worker behind one tape operation.

118cta_req_t *
119brix_cta_shm_submit(const cta_request_t *r, const char *owner)
120{
121 brix_cta_queue_t *q = brix_cta_shm_queue();
122 cta_req_t *e;
123
124 if (q == NULL) {
125 return NULL;
126 }
127 ngx_shmtx_lock(&cta_mtx);
128 e = cta_queue_submit(q, r, owner);
129 ngx_shmtx_unlock(&cta_mtx);
130 return e;
131}
