1/*
2 * WHAT: Export stream-side user identity and VO (virtual organisation) tracking
3 * metrics to Prometheus text exposition format.
4 *
5 * WHY: XRootD sessions carry VOMS attributes (VO membership) and user identities
6 * (DN or token sub). Tracking these per-session allows operators to see which
7 * VO groups consume most bandwidth, how many unique users connect over time,
8 * and whether the bounded tracking tables are overflowing. Unlike request-level
9 * metrics this is a gauge-style view of active session identity distribution.
10 *
11 * HOW: Iterates two bounded LRU tables in shared memory (vo_global + user_tracking)
12 * using atomic fetch-add(0) for lock-free reads. Emit four metric families:
13 * 1) brix_vo_bytes_tx/rx_total — per-VO byte counters (counter)
14 * 2) brix_vo_requests_total — per-VO request count (counter)
15 * 3) brix_vo_overflow_total — evicted VO entries since process start (counter)
16 * 4) brix_unique_users_current/gauge + total/counter + evictions/counter
17 * — active users, lifetime users, and LRU eviction count (gauge + counter)
18 * 5) brix_user_sessions_total{hash} — session count per hashed user ID (gauge)
19 * VO names truncated to BRIX_VO_NAME_LEN characters; users identified by
20 * FNV-1a hash of DN or token sub. Empty slots are skipped.
21 */
22
23#include "metrics_internal.h"
24
25/* Emit one per-VO counter row per occupied slot: `row_fmt` carries the
26 * family's row format ("...{vo=\"%s\"} %lu\n") and `counter_off` the
27 * vo-slot-relative offset of its counter, read via the lock-free
28 * ngx_atomic_fetch_add(&x, 0) load idiom. One copy of the fan-out loop for
29 * the three per-VO counter families. */
30static void
31stream_tracking_emit_vo_rows(metrics_writer_t *mw, ngx_brix_metrics_t *shm,
32 const char *row_fmt, size_t counter_off)
33{
34 ngx_uint_t i;
35
36 for (i = 0; i < BRIX_VO_MAX_TRACKED; i++) {
37 ngx_brix_vo_slot_t *vo = &shm->vo_global.slots[i];
38 ngx_atomic_t *counter;
39
40 if (!vo->name[0]) { continue; }
41 counter = (ngx_atomic_t *) ((char *) vo + counter_off);
42 mw_printf(mw, row_fmt, vo->name,
43 (unsigned long) ngx_atomic_fetch_add(counter, 0));
44 }
45}

47void
48brix_export_stream_tracking_metrics(metrics_writer_t *mw,
49 ngx_brix_metrics_t *shm)
50{
51 ngx_uint_t i;
52
53 mw_printf(mw,
54 "# HELP brix_vo_bytes_tx_total "
55 "Bytes sent to clients grouped by virtual organisation. "
56 "VO names are truncated to %d characters; the metric family has one entry per VO.\n"
57 "# TYPE brix_vo_bytes_tx_total counter\n", BRIX_VO_NAME_LEN - 1);
58 stream_tracking_emit_vo_rows(mw, shm,
59 "brix_vo_bytes_tx_total{vo=\"%s\"} %lu\n",
60 offsetof(ngx_brix_vo_slot_t, bytes_tx_total));
61
62 mw_printf(mw,
63 "# HELP brix_vo_bytes_rx_total "
64 "Bytes received from clients grouped by virtual organisation. "
65 "VO names are truncated to %d characters.\n"
66 "# TYPE brix_vo_bytes_rx_total counter\n", BRIX_VO_NAME_LEN - 1);
67 stream_tracking_emit_vo_rows(mw, shm,
68 "brix_vo_bytes_rx_total{vo=\"%s\"} %lu\n",
69 offsetof(ngx_brix_vo_slot_t, bytes_rx_total));
70
71 mw_printf(mw,
72 "# HELP brix_vo_requests_total "
73 "Requests grouped by virtual organisation. VO names are truncated.\n"
74 "# TYPE brix_vo_requests_total counter\n");
75 stream_tracking_emit_vo_rows(mw, shm,
76 "brix_vo_requests_total{vo=\"%s\"} %lu\n",
77 offsetof(ngx_brix_vo_slot_t, requests_total));
78
79 mw_printf(mw,
80 "# HELP brix_vo_overflow_total "
81 "VO entries that exceeded the tracking limit and were evicted.\n"
82 "# TYPE brix_vo_overflow_total counter\n");
83 mw_printf(mw, "brix_vo_overflow_total %lu\n",
84 (unsigned long) ngx_atomic_fetch_add(&shm->vo_global.overflow_total, 0));
85
86 mw_printf(mw,
87 "# HELP brix_unique_users_current "
88 "Currently tracked unique user identities (bounded LRU, max %d). "
89 "Users are identified by DN or token sub via FNV-1a hash.\n"
90 "# TYPE brix_unique_users_current gauge\n", BRIX_USERS_MAX_TRACKED);
91 mw_printf(mw, "brix_unique_users_current %lu\n",
92 (unsigned long) ngx_atomic_fetch_add(&shm->user_tracking.unique_count, 0));
93
94 mw_printf(mw,
95 "# HELP brix_unique_users_total "
96 "Lifetime unique user identities seen since process start. "
97 "Never decremented.\n"
98 "# TYPE brix_unique_users_total counter\n");
99 mw_printf(mw, "brix_unique_users_total %lu\n",
100 (unsigned long) ngx_atomic_fetch_add(&shm->user_tracking.total_unique, 0));
101
102 mw_printf(mw,
103 "# HELP brix_user_evictions_total "
104 "User identity slots evicted from the tracking table.\n"
105 "# TYPE brix_user_evictions_total counter\n");
106 mw_printf(mw, "brix_user_evictions_total %lu\n",
107 (unsigned long) ngx_atomic_fetch_add(&shm->user_tracking.evictions_total, 0));
108
109 mw_printf(mw,
110 "# HELP brix_user_sessions_total "
111 "Sessions per tracked user identity. Sum across all entries equals total authenticated sessions.\n"
112 "# TYPE brix_user_sessions_total gauge\n");
113 for (i = 0; i < BRIX_USERS_MAX_TRACKED; i++) {
114 ngx_brix_user_slot_t *u = &shm->user_tracking.slots[i];
115 if (!u->id_hash) { continue; }
116 mw_printf(mw, "brix_user_sessions_total{hash=\"%08x\"} %lu\n", /* metric-cardinality-allow: 8-hex identity hash, rows bounded by the fixed BRIX_USERS_MAX_TRACKED slot table with eviction */
117 u->id_hash, (unsigned long) ngx_atomic_fetch_add(&u->sessions_total, 0));
118 }
119}
