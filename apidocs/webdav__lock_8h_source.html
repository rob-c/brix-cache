17/* Lock xattr read/write/delete (prop_xattr.c); the record encode/decode moved

19/* setxattr the encoded lock onto `path`. Pass flags=XATTR_CREATE for atomic
20 * cross-worker lock acquisition: NGX_DECLINED means another worker won the race
21 * (EEXIST -> caller maps to 423). NGX_OK / NGX_ERROR otherwise. Takes the
22 * request (not just a log) so the lock xattr is written as the mapped user under
23 * impersonation (root_canon is derived from the request's loc conf). */

26/* getxattr+decode the lock on `path`. NGX_OK; NGX_DECLINED if no lock present
27 * (or path gone); NGX_ERROR on a real getxattr fault. */

29 webdav_lock_xattr_t *e);

40/*
41 * webdav_lock_startup_sweep — recursively remove every persisted lock xattr
42 * (WEBDAV_LOCK_XATTR_KEY) under root_canon. Used at startup when

49/* Walk path -> export root checking each level's lock xattr (O(path_depth)
50 * getxattr calls); expired locks are lazily deleted in passing. NGX_HTTP_LOCKED
51 * if an unexpired lock not matched by the request If header covers `path`; 500
52 * on a getxattr fault; NGX_OK otherwise. need_write is currently advisory. */
53ngx_int_t webdav_check_locks(ngx_http_request_t *r, const char *path,
54 int need_write);
55/* webdav_check_locks() plus, when `path` is a directory, a recursive descendant
56 * scan (opendir walk). Use for DELETE/COPY/MOVE on collections so a lock on any
57 * child blocks the op. Same return codes as webdav_check_locks(). */
58ngx_int_t webdav_check_locks_tree(ngx_http_request_t *r, const char *path);
59/* Append <D:lockdiscovery> for `path` to the head/tail chain (r->pool), reading
60 * the live lock xattr (empty element if none/expired; owner XML-escaped; expired
61 * locks lazily deleted). NGX_OK / NGX_ERROR on chain alloc failure. */
62ngx_int_t webdav_lock_append_discovery(ngx_http_request_t *r, const char *path,
63 ngx_chain_t **head, ngx_chain_t **tail);
64/* Append the static <D:supportedlock> element (exclusive+shared write) to the

66ngx_int_t webdav_lock_append_supported(ngx_http_request_t *r,
67 ngx_chain_t **head, ngx_chain_t **tail);
