100/* Walk clear, every individual documented bit, their legal union, and one
101 * deliberately unknown bit. Per-op clause tables supply the documented mask. */
102size_t
103brix_conform_option_masks(uint64_t known_mask, uint64_t *out, size_t cap)
104{
105 uint64_t bit;
106 uint64_t unknown = 1;
107 size_t count = 0;
108
109#define ADD_MASK(v) do { if (count < cap) out[count] = (v); count++; } while (0)
110 ADD_MASK(0);
111 for (bit = 1; bit != 0; bit <<= 1) {
112 if ((known_mask & bit) != 0) ADD_MASK(bit);
113 }
114 if (known_mask != 0 && (known_mask & (known_mask - 1)) != 0) {
115 ADD_MASK(known_mask);
116 }
117 while (unknown != 0 && (known_mask & unknown) != 0) unknown <<= 1;
118 if (unknown != 0) ADD_MASK(known_mask | unknown);
119#undef ADD_MASK
120 return count;
121}
