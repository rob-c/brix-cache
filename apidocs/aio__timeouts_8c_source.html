9/* WHAT: Fold one request's future deadline into the running "nearest deadline"
10 * accumulator *next (ns; 0 = none seen yet).
11 * WHY: Both the reconnecting-parked scan and the alive-inflight scan must track
12 * the soonest un-expired deadline to size the next epoll_wait; sharing one
13 * helper keeps that min-reduction identical across both paths.
14 * HOW: A zero deadline_ns means "no deadline" and is ignored; otherwise the
15 * value replaces *next when it is the first candidate or strictly sooner —
16 * byte-for-byte the original inline `(next == 0 || dl < next)` test. */
17static void
18timeouts_track_next(uint64_t deadline_ns, uint64_t *next)
19{
20 if (deadline_ns != 0 && (*next == 0 || deadline_ns < *next)) {
21 *next = deadline_ns;
22 }
23}
24
25/* WHAT: Expire every parked request on a RECONNECTING conn whose deadline has
26 * passed; fold survivors into *next.
27 * WHY: While a link is rebuilding, parked requests still owe their caller a
28 * deadline; ones that ran out the reconnect budget must fail with the
29 * "(reconnecting)" status, the rest keep the loop's wake timer tight.
30 * HOW: Walks the pending singly-linked list via a pointer-to-pointer so an
31 * expired node is unlinked in place before areq_complete frees it; live
32 * nodes advance the cursor and feed timeouts_track_next. Preserves the
33 * original unlink-then-complete ordering exactly. */
34static void
35timeouts_expire_parked(brix_aconn *ac, uint64_t now, uint64_t *next)
36{
37 brix_areq **pp = &ac->pending;
38 while (*pp != NULL) {
39 brix_areq *r = *pp;
40 if (r->deadline_ns != 0 && now >= r->deadline_ns) {
41 *pp = r->pend_next;
42 brix_status st;
43 brix_status_set(&st, XRDC_ESOCK, ETIMEDOUT,
44 "request deadline exceeded (reconnecting)");
45 areq_complete(r, -1, XRDC_ESOCK, &st);
46 } else {
47 timeouts_track_next(r->deadline_ns, next);
48 pp = &r->pend_next;
49 }
50 }
51}

53/* WHAT: Expire every in-flight request on an ALIVE conn whose deadline has
54 * passed; fold survivors into *next. Returns 1 if a timed-out request was
55 * a keepalive ping (the link is presumed dead), else 0.
56 * WHY: Isolating the inflight-slot scan keeps loop_process_timeouts flat and
57 * lets the caller run the transport-error recovery only AFTER the whole
58 * slot array is drained — recursing mid-scan would corrupt the walk.
59 * HOW: Tombstones an expired slot (dropping count, bumping tomb) before
60 * areq_complete frees the request, matching the original bookkeeping order;
61 * live slots feed timeouts_track_next. The ping-timeout flag is only
62 * reported, never acted on here. */
63static int
64timeouts_expire_inflight(brix_aconn *ac, uint64_t now, uint64_t *next)
65{
66 int ping_timed_out = 0;
67 for (uint32_t i = 0; i < ac->inflight.cap; i++) {
68 brix_areq *r = ac->inflight.slots[i];
69 if (r == NULL || r == REQMAP_TOMB || r->deadline_ns == 0) {
70 continue;
71 }
72 if (now >= r->deadline_ns) {
73 ac->inflight.slots[i] = REQMAP_TOMB;
74 ac->inflight.count--;
75 ac->inflight.tomb++;
76 if (r->is_ping) {
77 ping_timed_out = 1;
78 }
79 brix_status st;
80 brix_status_set(&st, XRDC_ESOCK, ETIMEDOUT,
81 "request deadline exceeded");
82 areq_complete(r, -1, XRDC_ESOCK, &st);
83 } else {
84 timeouts_track_next(r->deadline_ns, next);
85 }
86 }
87 return ping_timed_out;
88}

90/* WHAT: Run the deadline scan for one connection, folding its soonest survivor
91 * into *next.
92 * WHY: Each conn state routes to a different expiry path; centralizing the
93 * dispatch keeps the top-level sweep a single clean loop body.
94 * HOW: RECONNECTING → parked-request expiry; ALIVE → inflight expiry followed
95 * by a post-scan transport-error kick when a heartbeat ping timed out
96 * (safe now that the slot array is fully drained); any other state is
97 * skipped, matching the original early-`continue` structure. */
98static void
99timeouts_scan_conn(brix_aconn *ac, uint64_t now, uint64_t *next)
100{
101 if (ac->state == ACONN_RECONNECTING) {
102 timeouts_expire_parked(ac, now, next);
103 return;
104 }
105 if (ac->state != ACONN_ALIVE) {
106 return;
107 }
108
109 if (timeouts_expire_inflight(ac, now, next)) {
110 /* inflight scan for this ac is done — safe to recurse */
111 brix_status st;
112 brix_status_set(&st, XRDC_ESOCK, 0, "keepalive heartbeat timed out");
113 aconn_on_transport_error(ac, &st);
114 }
115}

117/* WHAT: Convert the nearest future deadline into the epoll_wait timeout (ms).
118 * WHY: The sweep produces an absolute ns deadline; the event loop needs a
119 * relative, clamped millisecond budget for its next block.
120 * HOW: No deadline → the idle tick; otherwise round the remaining ns up by 1ms,
121 * clamp to [1, AIO_TICK_MS]. Byte-for-byte the original tail arithmetic. */
122static int
123timeouts_next_wait_ms(uint64_t next, uint64_t now)
124{
125 if (next == 0) {
126 return AIO_TICK_MS;
127 }
128 uint64_t dt_ns = (next > now) ? next - now : 0;
129 int ms = (int) (dt_ns / 1000000ULL) + 1;
130 if (ms > AIO_TICK_MS) {
131 ms = AIO_TICK_MS;
132 }
133 if (ms < 1) {
134 ms = 1;
135 }
136 return ms;
137}

139/* Fail any request whose deadline has passed; a timed-out heartbeat ping is
140 * promoted to a transport error (the link is presumed dead -> reconnect). Returns
141 * ms until the next deadline (capped at AIO_TICK_MS) for the next epoll_wait. */
142int
143loop_process_timeouts(brix_loop *l)
144{
145 uint64_t now = brix_mono_ns();
146 uint64_t next = 0; /* nearest future deadline (ns), 0 = none */
147
148 for (brix_aconn *ac = l->aconns; ac != NULL; ac = ac->next) {
149 timeouts_scan_conn(ac, now, &next);
150 }
151
152 return timeouts_next_wait_ms(next, now);
153}
