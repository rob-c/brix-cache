1/* File: sss_entity.h — SSS v2 entity credential builder + LGID challenge decoder
2 * WHAT: The ngx-free kernel that mints a full XrdSecsss v2 *entity* credential
3 * (NAME + VORG + ROLE + GRPS + ENDO + a proxied CRED blob) and decodes the
4 * server's kXR_authmore challenge (the LGID it wants the client to answer
5 * with). Shared by the module's proxy arm (src/net/proxy/, forwarding the
6 * front-side identity upstream) and the native client (client/lib/auth/sec/
7 * sec_sss.c, --sss-vorg/--sss-role/--sss-endorse/--sss-creds-file and the
8 * per-connection ID registry).
9 * WHY: brix_sss_build_credential() (sss_bf.c) is the frozen v1 NAME-only wire —
10 * byte-identical to what every 1.x client sent — and stays untouched. The
11 * 2.0 breadth (release-2.0-readiness F9) needs the remaining entity TLVs,
12 * the SNDLID first round and the challenge decode in ONE place so the server
13 * parser (src/auth/sss/auth_identity_challenge.c), the proxy and the client
14 * cannot drift on field order, caps or the packed-string NUL rule.
15 * HOW: A NULL entity mints the SNDLID form (40-byte data header, option byte
16 * BRIX_SSS_OPT_SNDLID, no TLVs) — the server answers kXR_authmore carrying
17 * an LGID; brix_sss_challenge_lgid() decrypts that reply and returns the
18 * LGID so the caller can look its registry up and mint the USEDATA entity.
19 * Strings are packed NUL-terminated with a 2-byte big-endian length (NUL
20 * included); the CRED TLV carries raw bytes. Every string is capped at the
21 * receiver-side buffer (BRIX_SSS_ENT_*_MAX in protocols/root/protocol/sss.h)
22 * and the builder fails closed (-1) rather than truncating, so a proxied
23 * DN or credential can never be silently shortened into a different identity.
24 */

33/* One entity to encode. NULL / "" strings and a NULL / zero-length creds are
34 * omitted from the wire; a NULL / "" name is sent as "xrd" (the v1 default). */
35typedef struct {
36 const char *name; /* NAME 0x01 — user name / DN */
37 const char *vorg; /* VORG 0x02 — virtual organisation */
38 const char *role; /* ROLE 0x03 — VO role */
39 const char *grps; /* GRPS 0x04 — comma-separated group list */
40 const char *endo; /* ENDO 0x05 — endorsements */
41 const uint8_t *creds; /* CRED 0x06 — proxied credential, raw bytes */
42 size_t creds_len;
43} brix_sss_entity_t;

45/*
46 * Mint an SSS credential for `ent` (USEDATA) or, when ent == NULL, the SNDLID
47 * first-round header. key/key_len/key_id select the keytab entry; nonce32 is
48 * the caller's 32 random bytes; gen_time is seconds since BRIX_SSS_BASE_TIME.
49 * Writes the complete outer-header + BF32 body to out (capacity out_max, use
50 * BRIX_SSS_ENTITY_BLOB_MAX) and its length to *out_len. Returns 0, or -1 on a
51 * NULL argument, a string at or over its cap, a creds blob over its cap, a
52 * too-small out buffer or a cipher failure.
53 */
54int brix_sss_build_entity_credential(const uint8_t *key, size_t key_len,
55 uint64_t key_id, const brix_sss_entity_t *ent, const uint8_t nonce32[32],
56 uint32_t gen_time, uint8_t *out, size_t out_max, size_t *out_len);
