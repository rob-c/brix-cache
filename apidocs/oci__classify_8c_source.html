1/* oci_classify.c — the `/v2/` route classifier (phase-104 D0.2).
2 *
3 * WHAT: parse a decoded request URI into brix_oci_req_t: which §0.7.1 endpoint
4 * it names plus the validated name/reference/digest/session spans.
5 * WHY: see oci_classify.h — one validating parse at the edge is the whole
6 * traversal defense for every path this module later composes.
7 * HOW: the API prefix is located by the literal "/v2/", which is
8 * component-boundary-safe on its own (no other byte sequence contains
9 * it across a boundary). What follows is matched RIGHT-TO-LEFT on the
10 * five reserved terminals, so a repository actually named "blobs" or
11 * "v2" still classifies: only the last two or three components decide
12 * the route, and everything left of them is the name. That needs just the
13 * final three slash positions — no component array, no bound to overflow.
14 */
15

22/* One slash-delimited span of the post-prefix path. */
23typedef struct {
24 const char *p;
25 size_t n;
26} oci_span_t;
27
28/* The last three slash offsets of [s, s+n), newest first; absent ones are -1.
29 * Three is exactly what the terminal grammar needs (`<name>/blobs/uploads/<id>`
30 * is the deepest form), so the scan is bounded by the grammar, not by a cap. */
31static void
32oci_tail_slashes(const char *s, size_t n, long *p1, long *p2, long *p3)
33{
34 long i;
35
36 *p1 = *p2 = *p3 = -1;
37 for (i = (long) n - 1; i >= 0; i--) {
38 if (s[i] != '/') {
39 continue;
40 }
41 if (*p1 < 0) {
42 *p1 = i;
43 } else if (*p2 < 0) {
44 *p2 = i;
45 } else {
46 *p3 = i;
47 return;
48 }
49 }
50}
51
52/* The span between two slash offsets (`from` is the slash BEFORE the span, or
53 * -1 for "start of string"; `to` is the slash after it, or n for "end"). */
54static oci_span_t
55oci_seg(const char *s, long from, long to)
56{
57 oci_span_t sp;
58
59 sp.p = s + from + 1;
60 sp.n = (size_t) (to - from - 1);
61 return sp;
62}

64static int
65oci_seg_is(oci_span_t sp, const char *lit)
66{
67 size_t n = strlen(lit);
68
69 return sp.n == n && memcmp(sp.p, lit, n) == 0;
70}

72/* Session identifiers become a staged-file basename: [A-Za-z0-9_-]{1,128}. */
73static int
74oci_session_valid(const char *s, size_t n)
75{
76 size_t i;
77
78 if (n == 0 || n > BRIX_OCI_SESSION_MAX) {
79 return -1;
80 }
81 for (i = 0; i < n; i++) {
82 char c = s[i];
83
84 if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
85 || (c >= '0' && c <= '9') || c == '_' || c == '-'))
86 {
87 return -1;
88 }
89 }
90 return 0;
91}

114/* `manifests/<ref>`: a reference is a digest or a tag, and which one it is
115 * decides the cache policy downstream (immutable vs revalidated), so the
116 * classifier records it rather than making every consumer re-guess. */
117static int
118oci_route_manifest(brix_oci_req_t *out, const char *rest, long p2,
119 oci_span_t last)
120{
121 brix_oci_digest_t d;
122
123 if (oci_set_name(out, rest, (size_t) p2) != 0) {
124 return -1;
125 }
126 out->cls = BRIX_OCI_REQ_MANIFEST;
127 out->ref = last.p;
128 out->ref_len = last.n;
129
130 if (brix_oci_digest_parse(last.p, last.n, &d) == 0) {
131 out->ref_is_digest = 1;
132 out->digest = d;
133 return 0;
134 }
135 /* A reference that is neither a valid digest nor a valid tag is a grammar
136 * violation, not a missing manifest — 400, so a client cannot probe the
137 * store with malformed references and read 404 vs 400 as an oracle. */
138 if (brix_oci_tag_valid(last.p, last.n) != 0) {
139 return oci_bad(out, BRIX_OCI_ERR_NAME_INVALID);
140 }
141 return 0;
142}

144/* `blobs/<digest>` and the two upload forms that share the `blobs/` prefix. */
145static int
146oci_route_blobs(brix_oci_req_t *out, const char *rest, long p2,
147 oci_span_t last)
148{
149 brix_oci_digest_t d;
150
151 if (oci_seg_is(last, "uploads")) {
152 if (oci_set_name(out, rest, (size_t) p2) != 0) {
153 return -1;
154 }
155 out->cls = BRIX_OCI_REQ_UPLOAD_START;
156 return 0;
157 }
158 if (oci_set_name(out, rest, (size_t) p2) != 0) {
159 return -1;
160 }
161 if (brix_oci_digest_parse(last.p, last.n, &d) != 0) {
162 return oci_bad(out, BRIX_OCI_ERR_DIGEST_INVALID);
163 }
164 out->cls = BRIX_OCI_REQ_BLOB;
165 out->digest = d;
166 return 0;
167}

169/* `referrers/<digest>` — the reference here is the SUBJECT, the manifest that
170 * others point at, and only a digest may name it: a tag would make the answer
171 * depend on where that tag pointed when each referrer was pushed, which is the
172 * mutability the referrers graph exists to escape. */
173static int
174oci_route_referrers(brix_oci_req_t *out, const char *rest, long p2,
175 oci_span_t last)
176{
177 brix_oci_digest_t subject;
178
179 if (oci_set_name(out, rest, (size_t) p2) != 0) {
180 return -1;
181 }
182 if (brix_oci_digest_parse(last.p, last.n, &subject) != 0) {
183 return oci_bad(out, BRIX_OCI_ERR_DIGEST_INVALID);
184 }
185 out->cls = BRIX_OCI_REQ_REFERRERS;
186 out->digest = subject;
187 return 0;
188}

208/* Locate the API prefix and return the offset just past it, or -1.
209 * "/v2/" cannot straddle a component boundary — the leading slash IS the
210 * boundary — so a plain substring search is already boundary-correct, and a
211 * location-prefixed mount ("/local/v2/…") needs no special case. */
212static long
213oci_api_prefix_end(const char *uri, size_t len)
214{
215 size_t i;
216
217 for (i = 0; i + 4 <= len; i++) {
218 if (memcmp(uri + i, "/v2/", 4) == 0) {
219 return (long) (i + 4);
220 }
221 }
222 /* Only when nothing deeper matched: the bare, slashless "/v2" ping. Tried
223 * second so "/v2/library/v2" is a manifest-space path, not an API root. */
224 if (len >= 3 && memcmp(uri + len - 3, "/v2", 3) == 0) {
225 return (long) len;
226 }
227 return -1;
228}

230/* The terminal half of the grammar: after "/v2/" has been stripped and one
231 * trailing slash normalised away, what is left is
232 * "<name>/<terminal>[/<reference>]" and the only remaining question is which
233 * terminal it names. Split out of brix_oci_classify so the normalisation
234 * rules and the route table each read as a single decision.
235 */
236static int
237oci_route_terminal(brix_oci_req_t *out, const char *rest, size_t rest_len)
238{
239 long p1, p2, p3;
240 oci_span_t last, seg2, seg3;
241
242 oci_tail_slashes(rest, rest_len, &p1, &p2, &p3);
243 if (p1 < 0 || p2 < 0) {
244 /* Fewer than two slashes cannot spell any terminal — a name with no
245 * endpoint after it. */
246 return oci_bad(out, BRIX_OCI_ERR_NAME_UNKNOWN);
247 }
248 last = oci_seg(rest, p1, (long) rest_len);
249 seg2 = oci_seg(rest, p2, p1);
250 seg3 = (p3 >= 0) ? oci_seg(rest, p3, p2) : oci_seg(rest, -1, p2);
251
252 if (oci_seg_is(seg2, "manifests")) {
253 return oci_route_manifest(out, rest, p2, last);
254 }
255 if (oci_seg_is(seg2, "blobs")) {
256 return oci_route_blobs(out, rest, p2, last);
257 }
258 if (oci_seg_is(seg2, "uploads") && oci_seg_is(seg3, "blobs")) {
259 /* p3 < 0 means "blobs/uploads/<id>" with no name at all. */
260 if (p3 < 0) {
261 return oci_bad(out, BRIX_OCI_ERR_NAME_INVALID);
262 }
263 return oci_route_upload_session(out, rest, p3, last);
264 }
265 if (oci_seg_is(seg2, "referrers")) {
266 return oci_route_referrers(out, rest, p2, last);
267 }
268 if (oci_seg_is(seg2, "tags") && oci_seg_is(last, "list")) {
269 if (oci_set_name(out, rest, (size_t) p2) != 0) {
270 return -1;
271 }
272 out->cls = BRIX_OCI_REQ_TAGS_LIST;
273 return 0;
274 }
275 return oci_bad(out, BRIX_OCI_ERR_NAME_UNKNOWN);
276}

278int
279brix_oci_classify(const char *uri, size_t len, brix_oci_req_t *out)
280{
281 const char *rest;
282 size_t rest_len;
283 long end;
284
285 memset(out, 0, sizeof(*out));
286
287 end = oci_api_prefix_end(uri, len);
288 if (end < 0) {
289 return oci_bad(out, BRIX_OCI_ERR_NAME_UNKNOWN);
290 }
291 rest = uri + end;
292 rest_len = len - (size_t) end;
293
294 /* One trailing slash is the canonical spelling of the upload-start route
295 * and harmless noise everywhere else; strip it so the terminal match sees
296 * a single shape. A second one is a malformed path, not a spelling. */
297 if (rest_len > 0 && rest[rest_len - 1] == '/') {
298 rest_len--;
299 }
300 if (rest_len == 0) {
301 out->cls = BRIX_OCI_REQ_API_ROOT;
302 return 0;
303 }
304 if (rest[rest_len - 1] == '/') {
305 return oci_bad(out, BRIX_OCI_ERR_NAME_INVALID);
306 }
307
308 return oci_route_terminal(out, rest, rest_len);
309}
