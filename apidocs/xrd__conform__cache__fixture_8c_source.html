7static int read_object(xrd_conform_execution *execution,
8 brix_conn *connection, const char *path, uint64_t size, uint8_t **bytes,
9 brix_status *status)
10{
11 brix_file file;
12 uint64_t offset = 0;
13 int rc = -1;
14
15 *bytes = NULL;
16 if (size > SIZE_MAX || size > execution->run->config.byte_limit) {
17 brix_status_set(status, XRDC_EUSAGE, 0,
18 "cache fixture exceeds byte ceiling");
19 return -1;
20 }
21 *bytes = malloc(size == 0 ? 1 : (size_t) size);
22 if (*bytes == NULL) return -1;
23 if (!xrd_conform_reserve_status(execution, 0, status)
24 || brix_file_open_read(connection, path, &file, status) != 0) {
25 free(*bytes);
26 *bytes = NULL;
27 return -1;
28 }
29 while (offset < size) {
30 size_t length = size - offset < CACHE_FIXTURE_CHUNK
31 ? (size_t) (size - offset) : CACHE_FIXTURE_CHUNK;
32 ssize_t got;
33
34 if (!xrd_conform_reserve_status(execution, length, status)) break;
35 got = brix_file_read(connection, &file, (int64_t) offset,
36 *bytes + offset, length, status);
37 if (got != (ssize_t) length) break;
38 offset += length;
39 }
40 if (offset == size) rc = 0;
41 {
42 brix_status ignored;
43
44 brix_status_clear(&ignored);
45 if (xrd_conform_reserve_status(execution, 0, &ignored))
46 brix_file_close(connection, &file, &ignored);
47 }
48 if (rc != 0) {
49 free(*bytes);
50 *bytes = NULL;
51 }
52 return rc;
53}

55/* ---- Compare two complete fixture reads ----
56 * WHAT: Return one only when both reads succeeded with identical full bytes.
57 * WHY: Capacity, corruption and ordinary cache probes share one byte oracle.
58 * HOW: 1. Require both successful reads. 2. Compare nonempty bounded content.
59 */
60static int
61fixture_bytes_match(int cache_rc, int origin_rc, const uint8_t *cache_bytes,
62 const uint8_t *origin_bytes, size_t size)
63{
64 return cache_rc == 0 && origin_rc == 0
65 && (size == 0 || memcmp(cache_bytes, origin_bytes, size) == 0);
66}

88static void grade_capacity(xrd_conform_execution *execution, int cache_rc,
89 int origin_rc, const uint8_t *cache_bytes, const uint8_t *origin_bytes,
90 size_t size, const brix_status *status)
91{
92 int served = fixture_bytes_match(cache_rc, origin_rc, cache_bytes,

94 int refused = cache_rc != 0 && (status->kxr == kXR_NoSpace

96 int setup_failed = origin_rc != 0;
97
98 xrd_conform_result_set(execution, "CAC-005",
99 setup_failed ? BRIX_CONFORM_VERDICT_INCONCLUSIVE
100 : served || refused ? BRIX_CONFORM_VERDICT_CONFORMANT
101 : BRIX_CONFORM_VERDICT_DIVERGENT,
102 setup_failed ? "over-capacity-origin-unavailable"
103 : served ? "over-capacity-cache-evicted-and-served"
104 : refused ? "over-capacity-cache-refused-defined"
105 : "over-capacity-cache-transient-or-wrong",
106 "origin bytes or kXR_NoSpace/kXR_overQuota/kXR_NoMemory",
107 setup_failed ? status->msg : NULL);
108}

142/* WHAT: Grade one externally prepared cache state against its explicit origin.
143 * WHY: Partial, in-flight, over-capacity, and corrupt state cannot be created
144 * honestly through the public ROOT protocol.
145 * HOW: Require a named fixture kind, independently read the origin, and grade
146 * only the matching row with byte equality or its narrowly defined refusal. */
147void xrd_conform_probe_cache_fixture(xrd_conform_execution *execution,
148 brix_conn *cache, brix_conn *origin, const brix_url *cache_url,
149 const brix_url *origin_url)
150{
151 const char *kind = execution->run->cache_fixture_kind;
152 const char *id = fixture_row(kind);
153 brix_statinfo cache_info;
154 brix_statinfo origin_info;
155 brix_status status;
156 uint8_t *cache_bytes = NULL;
157 uint8_t *origin_bytes = NULL;
158 int cache_rc;
159 int origin_rc;
160 size_t size;
161
162 if (id == NULL || !xrd_conform_row_active(execution, id)) return;
163 brix_status_clear(&status);
164 if (!xrd_conform_reserve_status(execution, 0, &status)
165 || brix_stat(origin, origin_url->path, &origin_info, &status) != 0
166 || origin_info.size < 0) {
167 xrd_conform_result_set(execution, id,
168 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
169 "cache-fixture-origin-stat-unavailable", NULL, status.msg);
170 return;
171 }
172 origin_rc = read_object(execution, origin, origin_url->path,
173 (uint64_t) origin_info.size, &origin_bytes, &status);
174 brix_status_clear(&status);
175 cache_rc = xrd_conform_reserve_status(execution, 0, &status)
176 && brix_stat(cache, cache_url->path, &cache_info, &status) == 0
177 && cache_info.size == origin_info.size
178 ? read_object(execution, cache, cache_url->path,
179 (uint64_t) origin_info.size, &cache_bytes, &status) : -1;
180 size = (size_t) origin_info.size;
181 if (strcmp(kind, "partial") == 0)
182 grade_bytes(execution, id, "partial-cache-filled-and-unfilled-bytes-correct",
183 cache_rc, origin_rc, cache_bytes, origin_bytes, size, &status);
184 else if (strcmp(kind, "filling") == 0)
185 grade_bytes(execution, id, "serve-while-filling-all-offsets-correct",
186 cache_rc, origin_rc, cache_bytes, origin_bytes, size, &status);
187 else if (strcmp(kind, "capacity") == 0)
188 grade_capacity(execution, cache_rc, origin_rc, cache_bytes,
189 origin_bytes, size, &status);
190 else
191 grade_corrupt(execution, cache_rc, origin_rc, cache_bytes,
192 origin_bytes, size, &status);
193 free(cache_bytes);
194 free(origin_bytes);
195}
