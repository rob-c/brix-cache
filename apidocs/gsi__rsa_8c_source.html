17/* ---- Generate one serialized-buffer-safe GSI random challenge tag ----
18 *
19 * WHAT: Fills out[8] with unbiased cryptographically random ASCII letters and
20 * digits. Returns 1 on success and 0 when the RNG fails.
21 *
22 * WHY: XrdSecgsi carries this challenge through its serialized-buffer parser;
23 * ':' is a field delimiter there and makes an otherwise valid handshake
24 * fail intermittently. Raw random bytes also include NUL. Alphanumeric
25 * bytes are opaque to that grammar while retaining 62 choices per byte.
26 *
27 * HOW: Draw bytes in batches, reject values >= 248 (the largest multiple of 62
28 * below 256), and map accepted bytes modulo the 62-character alphabet.
29 */
30int
31brix_gsi_rand_tag(uint8_t out[8])
32{
33 static const uint8_t alphabet[] =
34 "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
35 uint8_t random[8];
36 size_t produced = 0;
37 size_t i;
38
39 while (produced < sizeof(random)) {
40 if (!brix_gsi_rand(random, sizeof(random))) {
41 OPENSSL_cleanse(random, sizeof(random));
42 return 0;
43 }
44 for (i = 0; i < sizeof(random) && produced < sizeof(random); i++) {
45 if (random[i] < 248) {
46 out[produced++] = alphabet[random[i] % 62];
47 }
48 }
49 }
50 OPENSSL_cleanse(random, sizeof(random));
51 return 1;
52}

55/*
56 * RSA private-key signature, byte-exact with XrdCryptosslRSA::EncryptPrivate:
57 * EVP_PKEY_sign over the RAW input (no message digest) with RSA PKCS#1 v1.5
58 * padding. This is the GSI proof-of-possession op — the client signs the
59 * server's random tag with the proxy private key (kXRS_signed_rtag); the server
60 * recovers it with the public key. `out` must hold >= EVP_PKEY_size(key) bytes.
61 * Returns the signature length, or 0 on failure. (Single-chunk: the rtag is far
62 * smaller than the key modulus, so no chunking loop is needed.)
63 */
64size_t
65brix_gsi_rsa_sign_raw(EVP_PKEY *key, const uint8_t *in, size_t inlen,
66 uint8_t *out)
67{
68 return brix_gsi_rsa_encrypt_private(key, in, inlen, out,
69 (size_t) EVP_PKEY_size(key) + inlen);
70}

73/*
74 * XrdCryptosslRSA::EncryptPrivate — RSA private-key "encrypt" (sign), chunked.
75 * Input is split into <key_size - 11>-byte chunks; each is EVP_PKEY_sign'd with
76 * PKCS#1 v1.5 into a key_size-byte output block. Used to sign the DH cipher
77 * blob (v>=DHsigned) and the rtag. out must hold >= ceil(inlen/lcmax)*key_size.
78 * Returns the total signature length, or 0.
79 */
80size_t
81brix_gsi_rsa_encrypt_private(EVP_PKEY *key, const uint8_t *in, size_t inlen,
82 uint8_t *out, size_t outmax)
83{
84 EVP_PKEY_CTX *ctx = EVP_PKEY_CTX_new(key, NULL);
85 size_t ksz = (size_t) EVP_PKEY_size(key);
86 size_t lcmax = ksz > 11 ? ksz - 11 : 1;
87 size_t kk = 0, ke = 0;
88 int ok = 1;
89
90 if (ctx == NULL || EVP_PKEY_sign_init(ctx) != 1
91 || EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PADDING) != 1) {
92 brix_evp_pkey_ctx_free(ctx);
93 return 0;
94 }
95 while (kk < inlen && ok) {
96 size_t lc = (inlen - kk < lcmax) ? (inlen - kk) : lcmax;
97 size_t lout = outmax - ke;
98 if (EVP_PKEY_sign(ctx, out + ke, &lout, in + kk, lc) != 1) {
99 ok = 0;
100 } else {
101 kk += lc;
102 ke += lout;
103 }
104 }
105 brix_evp_pkey_ctx_free(ctx);
106 return ok ? ke : 0;
107}

110/*
111 * XrdCryptosslRSA::DecryptPublic — RSA public-key "decrypt" (verify-recover),
112 * chunked: each key_size-byte input block is EVP_PKEY_verify_recover'd (PKCS#1)
113 * back to plaintext. Recovers a DH cipher blob signed by the peer's cert key.
114 * Returns the recovered length, or 0.
115 */
116size_t
117brix_gsi_rsa_decrypt_public(EVP_PKEY *key, const uint8_t *in, size_t inlen,
118 uint8_t *out, size_t outmax)
119{
120 EVP_PKEY_CTX *ctx = EVP_PKEY_CTX_new(key, NULL);
121 size_t ksz = (size_t) EVP_PKEY_size(key);
122 size_t kk = 0, ke = 0;
123 int ok = 1;
124
125 if (ctx == NULL || ksz == 0 || EVP_PKEY_verify_recover_init(ctx) != 1
126 || EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PADDING) != 1) {
127 brix_evp_pkey_ctx_free(ctx);
128 return 0;
129 }
130 while (kk + ksz <= inlen && ok) {
131 size_t lout = outmax - ke;
132 if (EVP_PKEY_verify_recover(ctx, out + ke, &lout, in + kk, ksz) != 1) {
133 ok = 0;
134 } else {
135 kk += ksz;
136 ke += lout;
137 }
138 }
139 brix_evp_pkey_ctx_free(ctx);
140 return ok ? ke : 0;
141}

144int
145brix_gsi_rsa_verify_raw(EVP_PKEY *key, const uint8_t *signature,
146 size_t signature_len, const uint8_t *expected,
147 size_t expected_len)
148{
149 uint8_t *recovered;
150 size_t key_size;
151 size_t recovered_len;
152 int equal;
153
154 if (key == NULL || signature == NULL || expected == NULL
155 || expected_len == 0 || expected_len > 64) {
156 return 0;
157 }
158 key_size = (size_t) EVP_PKEY_size(key);
159 if (key_size < expected_len + 11 || key_size > 16384
160 || signature_len != key_size) {
161 return 0;
162 }
163 recovered = malloc(key_size);
164 if (recovered == NULL) {
165 return 0;
166 }
167 recovered_len = brix_gsi_rsa_decrypt_public(key, signature, signature_len,
168 recovered, key_size);
169 equal = recovered_len == expected_len
170 && CRYPTO_memcmp(recovered, expected, expected_len) == 0;
171 OPENSSL_cleanse(recovered, key_size);
172 free(recovered);
173 return equal;
174}
