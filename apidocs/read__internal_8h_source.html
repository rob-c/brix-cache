22/* WHAT: Hand an owned read frame to the selected secondary, then retire it
23 * only on failure or an entirely inline send. WHY: The pending ring, not the
24 * primary request, owns parked storage; all read families share that contract.
25 * HOW: Queue, retain the existing offload metric, and release only when the
26 * queue did not retain the buffer. Release remains deliberately idempotent.
27 */
28static ngx_inline ngx_int_t
29brix_read_offload_queue(brix_ctx_t *secondary, ngx_connection_t *connection,
30 u_char *buffer, size_t length)
31{
32 ngx_int_t result = brix_queue_response_base(secondary, connection,
33 buffer, length, buffer);
34 if (result == NGX_OK) {
35 brix_metric_offload(BRIX_PROTO_ROOT);
36 }
37 if (result != NGX_OK || secondary->out.count == 0) {
38 brix_release_read_buffer(secondary, connection, buffer);
39 }
40 return result;
41}

43/*
44 * brix_read_io_t — decoded per-request read parameters, threaded through the
45 * serve helpers below.
46 *
47 * WHAT: the validated (idx, fd, offset, rlen) tuple of one kXR_read plus the
48 * per-in-flight memory buffer once the buffered path acquires it.
49 * WHY: the read handler dispatches across several serve strategies (sendfile,
50 * windowed, warm-probe, AIO, sync); passing one struct keeps every helper at a
51 * small explicit signature instead of re-plumbing five scalars each time.
52 * HOW: filled by read_validate_req(); databuf stays NULL until the buffered
53 * path allocates it. File-local only — never crosses the event loop (the
54 * windowed/AIO state machines snapshot what they need into ctx as before).

63 control stream); validated as a live bound path */
64} brix_read_io_t;

81/* Resolve a data-path `pathid` to a live secondary connection with a free ring
82 * slot, for offloading a single-frame read/readv/pgread reply. Returns the
83 * secondary ctx (with *sec_c_out set) when the reply may pipeline behind its
84 * existing responses, or NULL to fall back to the primary. A free slot needs
85 * every pending response counted — queued (out.count) + write-ack (wr_inflight)
86 * + read AIO (rd.aio_inflight) strictly below pipeline_depth — so the queued
87 * frame can never overrun the ring. resp_async streams stay on the control path. */
88static inline brix_ctx_t *
89brix_read_offload_secondary(brix_ctx_t *ctx, ngx_connection_t *c,
90 uint32_t pathid, ngx_connection_t **sec_c_out)
91{
92 const u_char *sessid;
93 ngx_connection_t *sec_c;
94 brix_ctx_t *sec_ctx;
95
96 if (pathid == 0) {
97 return NULL; /* primary/control stream — no lookup needed */
98 }
99 /* The pathid was validated against this session key at request decode. */
100 sessid = ctx->is_bound ? ctx->bound_sessid : ctx->login.sessid;
101 sec_c = brix_offload_lookup(sessid, pathid);
102 if (sec_c == NULL || sec_c == c) {
103 return NULL; /* bound elsewhere (or self) — use the control stream */
104 }
105 sec_ctx = ngx_stream_get_module_ctx((ngx_stream_session_t *) sec_c->data,
106 ngx_stream_brix_module);
107 if (sec_ctx == NULL || sec_ctx->destroyed
108 || sec_c->fd == (ngx_socket_t) -1 || sec_ctx->out.resp_async
109 || sec_ctx->out.count + sec_ctx->out.wr_inflight
110 + sec_ctx->rd.aio_inflight >= sec_ctx->out.pipeline_depth)
111 {
112 return NULL;
113 }
114 *sec_c_out = sec_c;
115 return sec_ctx;
116}
