20/*
21 * brix_read_io_t — decoded per-request read parameters, threaded through the
22 * serve helpers below.
23 *
24 * WHAT: the validated (idx, fd, offset, rlen) tuple of one kXR_read plus the
25 * per-in-flight memory buffer once the buffered path acquires it.
26 * WHY: the read handler dispatches across several serve strategies (sendfile,
27 * windowed, warm-probe, AIO, sync); passing one struct keeps every helper at a
28 * small explicit signature instead of re-plumbing five scalars each time.
29 * HOW: filled by read_validate_req(); databuf stays NULL until the buffered
30 * path allocates it. File-local only — never crosses the event loop (the
31 * windowed/AIO state machines snapshot what they need into ctx as before).
32 */

34 int idx; /* file-table slot */
35 ngx_fd_t fd; /* backing fd for the slot */

38 u_char *databuf; /* per-in-flight buffer (memory path only) */

40 control stream); validated as a live bound path */

58/* Resolve a data-path `pathid` to a live secondary connection with a free ring
59 * slot, for offloading a single-frame read/readv/pgread reply. Returns the
60 * secondary ctx (with *sec_c_out set) when the reply may pipeline behind its
61 * existing responses, or NULL to fall back to the primary. A free slot needs
62 * every pending response counted — queued (out.count) + write-ack (wr_inflight)
63 * + read AIO (rd.aio_inflight) strictly below pipeline_depth — so the queued
64 * frame can never overrun the ring. resp_async streams stay on the control path. */
65static inline brix_ctx_t *
66brix_read_offload_secondary(brix_ctx_t *ctx, ngx_connection_t *c,
67 uint32_t pathid, ngx_connection_t **sec_c_out)
68{
69 const u_char *sessid;
70 ngx_connection_t *sec_c;
71 brix_ctx_t *sec_ctx;
72
73 if (pathid == 0) {
74 return NULL; /* primary/control stream — no lookup needed */
75 }
76 /* The pathid was validated against this session key at request decode. */
77 sessid = ctx->is_bound ? ctx->bound_sessid : ctx->login.sessid;
78 sec_c = brix_offload_lookup(sessid, pathid);
79 if (sec_c == NULL || sec_c == c) {
80 return NULL; /* bound elsewhere (or self) — use the control stream */
81 }
82 sec_ctx = ngx_stream_get_module_ctx((ngx_stream_session_t *) sec_c->data,
83 ngx_stream_brix_module);
84 if (sec_ctx == NULL || sec_ctx->destroyed
85 || sec_c->fd == (ngx_socket_t) -1 || sec_ctx->out.resp_async
86 || sec_ctx->out.count + sec_ctx->out.wr_inflight
87 + sec_ctx->rd.aio_inflight >= sec_ctx->out.pipeline_depth)
88 {
89 return NULL;
90 }
91 *sec_c_out = sec_c;
92 return sec_ctx;
93}
