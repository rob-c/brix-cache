1/*
2 * sd_block_ns.c — the block driver's SERVER plane: the per-export instance and
3 * the fixed-extent namespace built on top of it.
4 *
5 * WHAT: init (probe the device capacity and derive the extent geometry), the
6 * namespace slots that present the device as N equal-size logical objects
7 * named by their 0-based index ("/0", "/1", ...), and the space report
8 * that answers with the DEVICE capacity rather than the statvfs of
9 * whatever filesystem the mount point happens to sit on.
10 * WHY: A block device has no directory namespace of its own, so the export
11 * synthesizes one. Split out of sd_block.c, which carries the raw byte
12 * plane and had reached the 600-line file cap; this plane needs an nginx
13 * pool and log, so it compiles only in the module build (it is listed in
14 * ./config and deliberately absent from shared/xrdproto/Makefile).
15 * HOW: Opening "/N" returns the device fd WINDOWED to the extent's byte range
16 * [N*extent_size, (N+1)*extent_size); the window travels on obj->state and
17 * every raw op in sd_block.c shifts and clamps against it, so one object
18 * can never read or scribble into its neighbour.
19 */

48/* sd_block_is_root — the empty path or a lone "/" is the namespace root. */
49static int
50sd_block_is_root(const char *path)
51{
52 return path == NULL || path[0] == '\0'
53 || (path[0] == '/' && path[1] == '\0');
54}

56/* sd_block_parse_index — an extent name is a leading-'/'-optional run of
57 * decimal digits ("/0", "12"). Anything else — a non-numeric component, an
58 * embedded '/', a path-escape attempt — is not an extent: return -1 so the
59 * caller reports ENOENT (the namespace exposes ONLY the fixed extents). */
60static int64_t
61sd_block_parse_index(const char *path)
62{
63 const char *p = path;
64 int64_t idx = 0;
65
66 if (p == NULL) {
67 return -1;
68 }
69 if (*p == '/') {
70 p++;
71 }
72 if (*p == '\0') {
73 return -1; /* the root dir, not an extent */
74 }
75 for (; *p != '\0'; p++) {
76 if (*p < '0' || *p > '9') {
77 return -1;
78 }
79 idx = idx * 10 + (*p - '0');
80 if (idx > 0x7fffffff) {
81 return -1;
82 }
83 }
84 return idx;
85}

87/* sd_block_init — probe the device capacity (BLKGETSIZE64 for a real block
88 * device, st_size for a regular file used as one) and derive the fixed extent
89 * geometry. extent_size == 0 makes the whole device one extent. */
90ngx_int_t
91sd_block_init(brix_sd_instance_t *inst, void *driver_conf)
92{
93 const brix_sd_block_conf_t *conf = driver_conf;
94 sd_block_state_t *st;
95 struct stat sb;
96 off_t cap;
97 size_t dlen;
98 int fd;
99
100 if (conf == NULL || conf->device == NULL || conf->device[0] == '\0') {
101 errno = EINVAL;
102 return NGX_ERROR;
103 }
104
105 fd = open(conf->device, O_RDONLY);
106 if (fd < 0) {
107 return NGX_ERROR; /* errno set by open(2) */
108 }
109 if (fstat(fd, &sb) != 0) {
110 int e = errno;
111 close(fd);
112 errno = e;
113 return NGX_ERROR;
114 }
115 cap = sb.st_size;
116#ifdef BLKGETSIZE64
117 if (S_ISBLK(sb.st_mode)) {
118 uint64_t sz = 0;
119 if (ioctl(fd, BLKGETSIZE64, &sz) == 0) {
120 cap = (off_t) sz;
121 }
122 }
123#endif
124 close(fd);
125
126 if (cap <= 0) {
127 errno = ENODEV;
128 return NGX_ERROR; /* an empty/zero-length device */
129 }
130
131 st = ngx_pcalloc(inst->pool, sizeof(*st));
132 if (st == NULL) {
133 errno = ENOMEM;
134 return NGX_ERROR;
135 }
136 dlen = ngx_strlen(conf->device);
137 st->device = ngx_pnalloc(inst->pool, dlen + 1);
138 if (st->device == NULL) {
139 errno = ENOMEM;
140 return NGX_ERROR;
141 }
142 ngx_memcpy(st->device, conf->device, dlen + 1);
143
144 st->capacity = cap;
145 st->extent_size = (conf->extent_size > 0) ? (off_t) conf->extent_size : cap;
146 st->extents = (uint32_t) ((cap + st->extent_size - 1) / st->extent_size);
147 inst->state = st;
148 return NGX_OK;
149}

151/* sd_block_open — resolve an extent name to a windowed handle on the device. A
152 * non-extent name is ENOENT; the root is EISDIR; a valid extent opens the device
153 * (never create/truncate) and binds the extent window into obj->state. */
154brix_sd_obj_t *
155sd_block_open(brix_sd_instance_t *inst, const char *path, int sd_flags,
156 mode_t mode, int *err_out)
157{
158 sd_block_state_t *st = inst->state;
159 brix_sd_obj_t *obj;
160 sd_block_obj_t *os;
161 int64_t idx;
162 int fd;
163
164 if (sd_block_is_root(path)) {
165 if (err_out != NULL) { *err_out = EISDIR; }
166 return NULL;
167 }
168 idx = sd_block_parse_index(path);
169 if (idx < 0 || (uint32_t) idx >= st->extents) {
170 if (err_out != NULL) { *err_out = ENOENT; }
171 return NULL;
172 }
173
174 /* A device is opened in place: create/truncate/exclusive are meaningless. */
175 sd_flags &= ~(BRIX_SD_O_CREATE | BRIX_SD_O_TRUNC | BRIX_SD_O_EXCL);
176 fd = brix_sd_block_open_unconfined(st->device, sd_flags, mode);
177 if (fd < 0) {
178 if (err_out != NULL) { *err_out = errno; }
179 return NULL;
180 }
181
182 /* Heap-allocate the shell + window (ngx_calloc, NOT inst->pool): open can run
183 * on a cache-fill worker thread and inst->pool is the thread-unsafe cycle
184 * pool. heap_shell frees the shell in the adopting layer; sd_block_close
185 * frees the window. */
186 obj = ngx_calloc(sizeof(*obj), inst->log);
187 os = ngx_calloc(sizeof(*os), inst->log);
188 if (obj == NULL || os == NULL) {
189 if (obj != NULL) { ngx_free(obj); }
190 if (os != NULL) { ngx_free(os); }
191 close(fd);
192 if (err_out != NULL) { *err_out = ENOMEM; }
193 return NULL;
194 }
195
196 os->base = (off_t) idx * st->extent_size;
197 os->len = st->extent_size;
198 if (os->base + os->len > st->capacity) {
199 os->len = st->capacity - os->base; /* short device tail */
200 }
201
202 obj->driver = inst->driver;
203 obj->inst = inst;
204 obj->fd = fd;
205 obj->state = os;
206 obj->heap_shell = 1;
207 return obj;
208}

210/* sd_block_close — close the handle fd and free the extent window. */
211ngx_int_t
212sd_block_close(brix_sd_obj_t *obj)
213{
214 ngx_int_t rc = NGX_OK;
215
216 if (obj == NULL) {
217 return NGX_OK;
218 }
219 if (obj->fd != NGX_INVALID_FILE) {
220 if (close(obj->fd) != 0) {
221 rc = NGX_ERROR;
222 }
223 obj->fd = NGX_INVALID_FILE;
224 }
225 if (obj->state != NULL) {
226 ngx_free(obj->state);
227 obj->state = NULL;
228 }
229 return rc;
230}

232/* sd_block_stat — the root is a directory; a valid extent index is a fixed-size
233 * regular object (the last extent may be a short device tail); anything else is
234 * ENOENT. */
235ngx_int_t
236sd_block_stat(brix_sd_instance_t *inst, const char *path, brix_sd_stat_t *out)
237{
238 sd_block_state_t *st = inst->state;
239 int64_t idx;
240
241 ngx_memzero(out, sizeof(*out));
242 if (sd_block_is_root(path)) {
243 out->mode = S_IFDIR | 0755;
244 out->is_dir = 1;
245 return NGX_OK;
246 }
247 idx = sd_block_parse_index(path);
248 if (idx < 0 || (uint32_t) idx >= st->extents) {
249 errno = ENOENT;
250 return NGX_ERROR;
251 }
252 out->mode = S_IFREG | 0644;
253 out->is_reg = 1;
254 out->size = ((uint32_t) idx == st->extents - 1)
255 ? st->capacity - (off_t) idx * st->extent_size
256 : st->extent_size;
257 return NGX_OK;
258}

260/* sd_block_opendir — only the root lists (the extents are a flat namespace); any
261 * other path is ENOTDIR. */
262brix_sd_dir_t *
263sd_block_opendir(brix_sd_instance_t *inst, const char *path, int *err_out)
264{
265 sd_block_state_t *st = inst->state;
266 brix_sd_dir_t *dir;
267 sd_block_dir_t *bd;
268
269 if (!sd_block_is_root(path)) {
270 if (err_out != NULL) { *err_out = ENOTDIR; }
271 return NULL;
272 }
273 dir = ngx_pcalloc(inst->pool, sizeof(*dir));
274 bd = ngx_pcalloc(inst->pool, sizeof(*bd));
275 if (dir == NULL || bd == NULL) {
276 if (err_out != NULL) { *err_out = ENOMEM; }
277 return NULL;
278 }
279 bd->next = 0;
280 bd->extents = st->extents;
281 dir->inst = inst;
282 dir->state = bd;
283 return dir;
284}

309/* sd_block_space — the export's capacity as the DEVICE, not as a filesystem.
310 *
311 * WHAT: total = the probed device capacity; used = the same; free = 0.
312 * WHY: Without this slot kXR_Qspace/kXR_statvfs fall back to statvfs(2) on the
313 * local export root, which for a block export describes an unrelated
314 * filesystem — usually the root fs holding the mount point, whose free
315 * bytes have nothing to do with the raw device the objects live on.
316 * free = 0 is the honest answer rather than a pessimistic one: the extent
317 * set is fixed at init and no operation on this driver creates an object,
318 * so there is genuinely no space in which a NEW object could be made.
319 * Writes into an existing extent are unaffected — they never consult it.
320 * HOW: Report st->capacity, which init probed with BLKGETSIZE64 (or st_size
321 * for a regular file standing in for a device). */
322ngx_int_t
323sd_block_space(brix_sd_instance_t *inst, brix_sd_space_t *out)
324{
325 const sd_block_state_t *st = inst->state;
326
327 if (st == NULL || out == NULL) {
328 errno = EINVAL;
329 return NGX_ERROR;
330 }
331 out->total_bytes = (uint64_t) st->capacity;
332 out->used_bytes = (uint64_t) st->capacity;
333 out->free_bytes = 0;
334 return NGX_OK;
335}
