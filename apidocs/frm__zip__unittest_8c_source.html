1/*
2 * frm_zip_unittest.c — standalone checks for the store-only ZIP32 container
3 *
4 * WHAT: round-trips members through frm_zip.c and probes the reader against
5 * hostile input (path-escaping names, corrupted bytes, non-archives).
6 *
7 * WHY: the archiver trusts this container for byte-exact recall from tape;
8 * a wrong offset or an accepted "../" name would silently corrupt or
9 * escape a dataset. Runs without nginx (see tests/test_phase115_tape_arc.py).
10 *
11 * HOW: cc -Wall -Wextra -Werror -I src -o t frm_zip_unittest.c frm_zip.c \
12 * core/compat/crc32_ieee.c && ./t → "all checks passed"
13 */

36static int
37tmp_fd(void)
38{
39 const char *dir = getenv("TMPDIR");
40 char path[4096];
41 int fd;
42
43 snprintf(path, sizeof(path), "%s/frm_zip_ut.XXXXXX", dir ? dir : "/tmp");
44 fd = mkstemp(path);
45 if (fd >= 0) {

51static int
52fd_with(const char *data, size_t len)
53{
54 int fd = tmp_fd();
55
56 if (fd >= 0 && len > 0 && pwrite(fd, data, len, 0) != (ssize_t) len) {
57 return -1;
58 }
59 return fd;

92static int
93extract_equals(int zfd, const brix_zip_entry_t *e, const char *want, size_t len)
94{
95 char buf[256];
96 int dst = tmp_fd();
97 int ok;
98
99 if (dst < 0 || brix_zip_extract(zfd, e, dst) != 0) {
100 if (dst >= 0) {
101 close(dst);
102 }
103 return 0;
104 }
105 ok = (pread(dst, buf, sizeof(buf), 0) == (ssize_t) len && memcmp(buf, want, len) == 0);
106 close(dst);
107 return ok;
108}

114static int
115build_archive(void)
116{
117 int fd = tmp_fd();
118 brix_zip_writer_t *w = brix_zip_writer_open(fd);
119
120 CHECK(w != NULL, "writer open");
121 CHECK(add_member(w, "a.bin", A, sizeof(A) - 1) == 0, "add a.bin");
122 CHECK(add_member(w, "sub/b.bin", B, sizeof(B) - 1) == 0, "add sub/b.bin");
123 CHECK(add_member(w, "c.txt", C, 0) == 0, "add empty c.txt");
124 CHECK(brix_zip_writer_count(w) == 3, "writer count 3");
125 CHECK(brix_zip_writer_entry(w, 1)->size == sizeof(B) - 1, "entry 1 size");
126 CHECK(brix_zip_writer_finish(w) == 0, "finish");
127 brix_zip_writer_close(w);
128 return fd;
129}

131static void
132test_round_trip(void)
133{
134 int fd = build_archive();
135 index_t ix = { { { { 0 }, 0, 0, 0 } }, 0 };
136 unsigned skipped = 99;
137
138 CHECK(brix_zip_index(fd, collect, &ix, &skipped) == 0, "index ok");
139 CHECK(skipped == 0, "nothing skipped");
140 CHECK(ix.n == 3, "three entries");
141 CHECK(strcmp(ix.e[0].name, "a.bin") == 0 && strcmp(ix.e[1].name, "sub/b.bin") == 0

143 CHECK(ix.e[0].lfh_off == 0 && ix.e[1].lfh_off == 30 + 5 + sizeof(A) - 1, "local header offsets");
144 CHECK(extract_equals(fd, &ix.e[0], A, sizeof(A) - 1), "extract a.bin byte-exact");
145 CHECK(extract_equals(fd, &ix.e[1], B, sizeof(B) - 1), "extract sub/b.bin byte-exact");
146 CHECK(extract_equals(fd, &ix.e[2], C, 0), "extract empty member");
147 close(fd);

150/* A foreign tool rewrites one directory name to "../xy" (same length): the
151 * reader must skip it, count it, and still serve the others. */
152static void
153test_unsafe_name_skipped(void)
154{
155 int fd = build_archive();
156 struct stat sb;
157 uint8_t *img;
158 size_t i;
159 index_t ix = { { { { 0 }, 0, 0, 0 } }, 0 };
160 unsigned skipped = 0;
161 int patched = 0;
162
163 fstat(fd, &sb);
164 img = malloc((size_t) sb.st_size);
165 CHECK(pread(fd, img, (size_t) sb.st_size, 0) == sb.st_size, "read image");
166 /* the central-directory copy of "a.bin" follows the last local header */
167 for (i = (size_t) sb.st_size - 46; i > 0 && !patched; i--) {
168 if (memcmp(img + i, "PK\x01\x02", 4) == 0 && memcmp(img + i + 46, "a.bin", 5) == 0) {
169 memcpy(img + i + 46, "../xy", 5);
170 patched = 1;
171 }
172 }
173 CHECK(patched, "found a.bin in the central directory");
174 CHECK(pwrite(fd, img, (size_t) sb.st_size, 0) == sb.st_size, "write patched image");
175 free(img);
176
177 CHECK(brix_zip_index(fd, collect, &ix, &skipped) == 0, "index of patched archive ok");
178 CHECK(skipped == 1, "one unsafe entry skipped");
179 CHECK(ix.n == 2 && strcmp(ix.e[0].name, "sub/b.bin") == 0, "safe entries still served");
180 CHECK(extract_equals(fd, &ix.e[0], B, sizeof(B) - 1), "extract after skip still byte-exact");
181 close(fd);
182}

184static void
185test_corruption_detected(void)
186{
187 int fd = build_archive();
188 index_t ix = { { { { 0 }, 0, 0, 0 } }, 0 };
189 uint8_t byte;
190 int dst;
191
192 CHECK(brix_zip_index(fd, collect, &ix, NULL) == 0, "index before corruption");
193 /* flip one payload byte of a.bin (data starts after the 30+5-byte header) */
194 CHECK(pread(fd, &byte, 1, 35 + 3) == 1, "read payload byte");
195 byte ^= 0x5a;
196 CHECK(pwrite(fd, &byte, 1, 35 + 3) == 1, "corrupt payload byte");
197 dst = tmp_fd();
198 errno = 0;
199 CHECK(brix_zip_extract(fd, &ix.e[0], dst) == -1 && errno == EIO, "extract of corrupted member -> EIO");
200 CHECK(extract_equals(fd, &ix.e[1], B, sizeof(B) - 1), "other member unaffected");
201 close(dst);
202 close(fd);
203}

205static void
206test_limits_and_names(void)
207{
208 int fd = tmp_fd();
209 brix_zip_writer_t *w = brix_zip_writer_open(fd);
210 char longname[BRIX_ZIP_NAME_MAX + 8];
211
212 errno = 0;

217 brix_zip_writer_close(w);
218 close(fd);
219
220 CHECK(brix_zip_name_ok("a/b/c.txt") == 1, "plain nested name ok");
221 CHECK(brix_zip_name_ok(".hidden") == 1, "dotfile ok");
222 CHECK(brix_zip_name_ok("") == 0, "empty rejected");
223 CHECK(brix_zip_name_ok("/abs") == 0, "leading slash rejected");
224 CHECK(brix_zip_name_ok("dir/") == 0, "trailing slash rejected");

227 CHECK(brix_zip_name_ok("a/../b") == 0, "dotdot segment rejected");
228 CHECK(brix_zip_name_ok("..") == 0, "bare dotdot rejected");
229 CHECK(brix_zip_name_ok("a\\b") == 0, "backslash rejected");
