18/*
19 * WHAT: filesystem verb -> exec `xrdfs <endpoint> <verb> [paths...]`.
20 * WHY: xrdfs separates the connect endpoint from the path, so when the target
21 * is a full root:// URL (or an alias that resolves to one) carrying a
22 * path, split it: `xrd stat root://h//d/f` -> `xrdfs root://h:port stat
23 * /d/f`. A bare host:port (or anything not a root:// URL) is passed
24 * through unchanged.
25 * HOW: fs_find_endpoint() scans for the endpoint URL; split mode maps every
26 * arg via fs_map_split_args(), pass-through mode copies args verbatim.
27 */
28static int
29cmd_fs_verb(int argc, char **argv)
30{
31 const char *cmd = argv[1];
32 const char *pfx = brix_prog_prefix(brix_prog_base(argv[0]));
33 fs_split_t sp;
34 char **nv;
35 int i, k = 0, split;
36
37 if (argc < 3) {
38 fprintf(stderr, "xrd %s: needs an <endpoint>\n", cmd);
39 return 50;
40 }
41 split = fs_find_endpoint(argc, argv, &sp);
42 nv = (char **) malloc((size_t) (argc + 3) * sizeof(char *));
43 if (nv == NULL) { fprintf(stderr, "xrd: out of memory\n"); return 51; }
44 nv[k++] = (char *) "xrdfs";
45 if (split) {
46 nv[k++] = sp.endpoint; /* connect endpoint (host:port) */
47 nv[k++] = (char *) cmd; /* the verb */
48 k = fs_map_split_args(nv, k, argc, argv, &sp);
49 if (k < 0) {
50 free(nv);
51 return 50;
52 }
53 } else {
54 nv[k++] = argv[2]; /* bare endpoint as given */
55 nv[k++] = (char *) cmd;
56 for (i = 3; i < argc; i++) { /* paths/flags verbatim */
57 nv[k++] = argv[i];
58 }
59 }
60 nv[k] = NULL;
61 exec_tool(pfx, "xrdfs", nv);
62 return 127; /* unreachable: exec_tool does not return */
63}

66/*
67 * WHAT: emit a did-you-mean hint when the user typed an unrecognised xrd
68 * command; search both the named xrd dispatch table and FS_VERBS.
69 * WHY: spec WS-7: every unknown-command site must offer a suggestion when
70 * one exists within DL distance ≤ 2 (TTY-gated, C3 compliant).
71 * HOW: build a merged NULL-terminated names array — dispatch-table names in
72 * their historical order, skipping '-'-prefixed flag aliases, then
73 * FS_VERBS — and pass it to brix_suggest().
74 */
75static void
76report_unknown_command(const char *cmd)
77{
78 const char *all_names[80]; /* FS_VERBS(36) + named commands(24) + pad */
79 int n = 0;
80 int i;
81 const char *suggestion;
82
83 for (i = 0; XRD_DISPATCH[i].name != NULL && n < 79; i++) {
84 if (XRD_DISPATCH[i].name[0] != '-') {
85 all_names[n++] = XRD_DISPATCH[i].name;
86 }
87 }
88 for (i = 0; FS_VERBS[i] != NULL && n < 79; i++) {
89 all_names[n++] = FS_VERBS[i];
90 }
91 all_names[n] = NULL;
92
93 fprintf(stderr, "xrd: unknown command '%s'\n\n", cmd);
94 suggestion = brix_suggest(cmd, all_names);
95 if (suggestion != NULL) {
96 brix_cli_hint("hint: did you mean '%s'?\n", suggestion);
97 }
98}
