1/*
2 * vfs_policy_domain.c — the typed storage-domain assert (phase-107 C8/C9).
3 *
4 * WHAT: Implements brix_vfs_domain_mutation() and brix_vfs_service_mutation():
5 * the runtime check that the instance a service-storage mutator is about
6 * to drive really IS the kind of storage the caller claims
7 * (inst->domain, fs/backend/sd.h).
8 *
9 * WHY: The export/service split was reasoning in a document; what enforced it
10 * was nothing. A dedup publish, a cache invalidation, a journal write
11 * pointed at an export root would have mutated client-named storage with
12 * no gate in the way. The assert refuses with EINVAL and logs at crit —
13 * deliberately NOT the export kernel's EROFS, because this is a
14 * programming error a client cannot provoke, and an EROFS here would be
15 * a lie caught by the wrong test.
16 *
17 * HOW: Validate, then compare the claim against the field. EXPORT is the one
18 * special case: it routes to the phase-105 kernel with the fail-closed
19 * READ_ONLY policy, because an instance carries no request policy —
20 * export mutations must arrive through the policy-bearing forms
21 * (vfs_policy.c / vfs_policy_export.c), never through a domain claim.
22 * No allocation, no I/O, no backend call; one crit log line on a
23 * mismatch is the only side effect.
24 */

34/* ---- Assert an instance belongs to a claimed storage domain ----
35 *
36 * WHAT: NGX_OK when inst->domain == domain (service domains) — EROFS via the
37 * phase-105 kernel for an EXPORT claim; NGX_ERROR/EINVAL otherwise.
38 *
39 * WHY: The caller states its claim instead of the assert inferring "service
40 * or not" from the instance, so a mis-composed tier (a CACHE mutator
41 * handed a STAGE store) is caught, not just an export-pointed one.
42 *
43 * HOW: 1. Range-check inst, op, domain; 2. route an EXPORT claim to the
44 * kernel under the fail-closed policy; 3. compare the claim; 4. log the
45 * mismatch at crit and refuse with EINVAL.
46 */
47ngx_int_t
48brix_vfs_domain_mutation(const brix_sd_instance_t *inst,
49 brix_vfs_domain_t domain, brix_vfs_mutation_op_t op)
50{
51 if (inst == NULL
52 || (ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT
53 || (ngx_uint_t) domain >= BRIX_VFS_DOMAIN_COUNT)
54 {
55 errno = EINVAL;
56 return NGX_ERROR;
57 }
58
59 if (domain == BRIX_VFS_DOMAIN_EXPORT) {
60 /* An instance carries no request policy, so the only policy available
61 * to an export claim is the fail-closed default: always EROFS. Export
62 * mutations arrive through the policy-bearing kernel forms. */
63 return brix_vfs_require_mutation_policy(BRIX_VFS_MUTATION_READ_ONLY,
64 op);
65 }
66
67 if (inst->domain != domain) {
68 if (inst->log != NULL) {
69 ngx_log_error(NGX_LOG_CRIT, inst->log, 0,
70 "brix vfs: service mutation \"%s\" claimed domain %d but the "
71 "instance is domain %d — refused (wrong_domain)",
72 brix_vfs_mutation_op_name(op), (int) domain,
73 (int) inst->domain);
74 }
75 errno = EINVAL;
76 return NGX_ERROR;
77 }
78
79 brix_metric_vfs_domain_mutation((ngx_uint_t) domain, (ngx_uint_t) op);
80 return NGX_OK;
81}

83/* ---- Assert a claimed storage domain with no instance in hand ----
84 *
85 * WHAT: The instance-free form of the assert above, for mutators whose
86 * service storage is not a bound driver instance at all — the
87 * credential staging dir (phase-108 C11), the OCI store tree (C10).
88 * NGX_OK for a valid service-domain claim (one metric sample booked);
89 * EROFS via the phase-105 kernel for an EXPORT claim; EINVAL out of
90 * range.
91 *
92 * WHY: brix_vfs_domain_mutation() deliberately refuses a NULL instance —
93 * for instance-bearing paths that is a wiring bug. But a credential
94 * write has no brix_sd_instance_t and never will; without this form
95 * those verbs would either skip the domain gate (untyped, unaccounted)
96 * or fake an instance (worse). With nothing to cross-check, the claim's
97 * enforcement is the seam guard's source-level entitlement table plus
98 * the EXPORT routing here: no caller can launder an export mutation
99 * through a bare claim, because EXPORT routes to the fail-closed
100 * kernel and answers EROFS.
101 *
102 * HOW: 1. Range-check op and domain; 2. route an EXPORT claim to the kernel
103 * under the fail-closed policy (with one crit line — a service caller
104 * claiming EXPORT is a programming error, same as a domain mismatch);
105 * 3. book the accounting sample and allow.
106 */
107ngx_int_t
108brix_vfs_domain_claim(ngx_log_t *log, brix_vfs_domain_t domain,
109 brix_vfs_mutation_op_t op)
110{
111 if ((ngx_uint_t) op >= BRIX_VFS_MUTATE_OP_COUNT
112 || (ngx_uint_t) domain >= BRIX_VFS_DOMAIN_COUNT)
113 {
114 errno = EINVAL;
115 return NGX_ERROR;
116 }
117
118 if (domain == BRIX_VFS_DOMAIN_EXPORT) {
119 if (log != NULL) {
120 ngx_log_error(NGX_LOG_CRIT, log, 0,
121 "brix vfs: service mutation \"%s\" claimed the EXPORT domain "
122 "with no instance — refused (wrong_domain)",
123 brix_vfs_mutation_op_name(op));
124 }
125 /* Only the fail-closed policy is available to a bare claim: export
126 * mutations arrive through the policy-bearing kernel forms. */
127 return brix_vfs_require_mutation_policy(BRIX_VFS_MUTATION_READ_ONLY,
128 op);
129 }
130
131 brix_metric_vfs_domain_mutation((ngx_uint_t) domain, (ngx_uint_t) op);
132 return NGX_OK;
133}

135/* ---- Assert an instance is service storage at all ----
136 *
137 * WHAT: NGX_OK when inst->domain is any service domain; NGX_ERROR/EINVAL when
138 * it is EXPORT (or the arguments are invalid).
139 *
140 * WHY: gcas' dedup slots know only "this must not be the export"; making
141 * them claim CACHE would couple the CAS layer to the tier shape for no
142 * safety gain. The instance's own domain is the claim.
143 *
144 * HOW: Refuse EXPORT with the crit log, then delegate to the general form
145 * with the instance's own domain (which then validates op and matches
146 * trivially).
147 */
148ngx_int_t
149brix_vfs_service_mutation(const brix_sd_instance_t *inst,
150 brix_vfs_mutation_op_t op)
151{
152 if (inst == NULL) {
153 errno = EINVAL;
154 return NGX_ERROR;
155 }
156
157 if (inst->domain == BRIX_VFS_DOMAIN_EXPORT) {
158 if (inst->log != NULL) {
159 ngx_log_error(NGX_LOG_CRIT, inst->log, 0,
160 "brix vfs: service mutation \"%s\" against EXPORT-domain "
161 "storage — refused (wrong_domain)",
162 brix_vfs_mutation_op_name(op));
163 }
164 errno = EINVAL;
165 return NGX_ERROR;
166 }
167
168 return brix_vfs_domain_mutation(inst, inst->domain, op);
169}
