14/* WHAT: Create a node entry and reply once. WHY: Preallocate identity ownership.
15 * HOW: 1. Resolve child. 2. Reserve node/alias. 3. Mutate as caller. 4. Reply. */
16static void
17xfs_ll_new(fuse_req_t req, fuse_ino_t parent, const char *name,
18 xfs_new_fn operation, xfs_new_args *args)
19{
20 xfs_session *session = xfs_req_lock(req);
21 char path[XRDC_PATH_MAX];
22 xfs_node_ticket ticket = {0};
23 int rc = xfs_req_child(session, parent, name, path);
24 if (rc == 0) {
25 rc = xfs_nodes_prepare(&session->nodes, path, &ticket);
26 }
27 if (rc == 0) {
28 rc = operation(xfs_req_uid(req), path, args);
29 }
30 if (rc == 0) {
31 rc = xfs_ll_entry(req, session, path, args->fi, &ticket, NULL);
32 if (rc != 0 && args->fi && args->fi->fh) {
33 xfs_release(path, args->fi);
34 }
35 }
36 xfs_nodes_cancel(&session->nodes, &ticket);
37 pthread_mutex_unlock(&session->mutex);
38 if (rc != 0) {
39 fuse_reply_err(req, -rc);
40 }
41}

146/* WHAT: Create an alias with its source kernel identity. WHY: Fix BSD LINK EIO.
147 * HOW: 1. Resolve source/destination. 2. Reserve alias. 3. Link as caller.
148 * 4. Reply using the original node, never a second path-derived id. */
149void
150xfs_ll_link(fuse_req_t req, fuse_ino_t ino, fuse_ino_t parent, const char *name)
151{
152 xfs_session *session = xfs_req_lock(req);
153 char from[XRDC_PATH_MAX], to[XRDC_PATH_MAX];
154 xfs_node_ticket ticket = {0};
155 int rc = xfs_req_path(session, ino, from);
156 if (rc == 0) { rc = xfs_req_child(session, parent, name, to); }
157 if (rc == 0) { rc = xfs_nodes_prepare(&session->nodes, to, &ticket); }
158 if (rc == 0) { rc = xfs_link(xfs_req_uid(req), from, to); }
159 if (rc == 0) {
160 xfs_node *source = xfs_nodes_find(&session->nodes, ino);
161 rc = xfs_ll_entry(req, session, to, NULL, &ticket, source);
162 }
163 xfs_nodes_cancel(&session->nodes, &ticket);
164 pthread_mutex_unlock(&session->mutex);
165 if (rc != 0) {
166 fuse_reply_err(req, -rc);
167 }
168}

170/* WHAT: Recognize rename between aliases of the same inode.
171 * WHY: POSIX requires a no-op; deleting either alias would corrupt the registry.
172 * HOW: 1. Resolve both keys. 2. Compare actual inode ownership. */
173static int
174xfs_rename_same(xfs_nodes *table, const char *from, const char *to)
175{
176 xfs_alias *source = xfs_nodes_alias(table, from);
177 xfs_alias *target = xfs_nodes_alias(table, to);
178 return source && target && source->node == target->node;
179}

181/* WHAT: Rename a file/subtree without invalidating descendant identities.
182 * WHY: Open directories and files remain addressable after parent rename.
183 * HOW: 1. Resolve both names. 2. Reserve all replacement keys. 3. Mutate as uid.
184 * 4. Commit keys only on success, or release reservations unchanged. */
185void
186xfs_ll_rename(fuse_req_t req, fuse_ino_t parent, const char *name,
187 fuse_ino_t newparent, const char *newname, unsigned flags)
188{
189 xfs_session *session = xfs_req_lock(req);
190 char from[XRDC_PATH_MAX], to[XRDC_PATH_MAX];
191 xfs_rename_change *changes = NULL;
192 int rc = xfs_req_child(session, parent, name, from);
193 if (rc == 0) { rc = xfs_req_child(session, newparent, newname, to); }
194 int same = rc == 0 && xfs_rename_same(&session->nodes, from, to);
195 if (rc == 0 && !same) {
196 rc = xfs_nodes_rename_prepare(&session->nodes, from, to, &changes);
197 }
198 if (rc == 0) {
199 rc = xfs_rename(xfs_req_uid(req), from, to, flags);
200 }
201 xfs_nodes_rename_finish(&session->nodes, changes, to, rc == 0 && !same);
202 xfs_req_finish(req, session, rc);
203}
