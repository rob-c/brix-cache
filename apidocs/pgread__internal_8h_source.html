13/* CRC32c word size per page unit ([CRC32c(4)][data]); == kXR_pgUnitSZ - page. */
14#define BRIX_PG_CKSZ ((size_t) (kXR_pgUnitSZ - kXR_pgPageSZ))

16/*
17 * Warm-probe ceiling: a request at most this large is read+CRCed inline on
18 * the event loop when page-cache resident (the thread-pool handoff, not the
19 * copy, dominates at this size). Anything larger always posts to the pool:
20 * with pgreads pipelined, a pool thread reads+CRCs request N+1 into its own
21 * rd_pool slot WHILE the event loop writev-s response N — overlapping the two
22 * kernel copies that an inline serve would serialize. One streaming window
23 * keeps the inline cost well under a millisecond.
24 */
25#define BRIX_PGREAD_WARM_INLINE_MAX ((size_t) BRIX_READ_WINDOW)

27/*
28 * brix_pgread_run_t - per-request state threaded through the pgread steps.
29 *
30 * WHAT: The decoded request (handle index, fd, offset, capped length), the
31 * per-request wire buffer, and the produced output {out_buf, out_size,
32 * flat_buf} — filled by exactly one producer path (warm hit, AIO
33 * offload, or sync fallback).
34 *
35 * WHY: Makes the phase-72.A invariant structural: out_buf/flat_buf/out_size
36 * start NULL/0 (the handler zeroes the struct) and every producer sets
37 * all three through this one struct, so the pre-framing out_buf==NULL
38 * guard catches any path that failed to produce output.
39 */
40typedef struct {
41 int idx; /* file-handle table index */
42 int fd; /* resolved file descriptor */
43 int64_t offset; /* requested file offset */
44 size_t rlen; /* capped request length; sync path: bytes read */
45 unsigned pathid; /* §1.2 request args: response path (0 = primary) */
46 unsigned reqflags; /* §1.2 request args: kXR_pgRetry or 0 */
47 u_char *scratch; /* gapped wire buffer (this request's rd_pool slot;
48 the offload path substitutes the secondary's
49 frame buffer instead) */
50 u_char *out_buf; /* encoded output start (NULL until produced) */
51 u_char *flat_buf; /* buffer to release after send (NULL until set) */
52 size_t out_size; /* encoded output bytes (0 until produced) */
53} brix_pgread_run_t;
