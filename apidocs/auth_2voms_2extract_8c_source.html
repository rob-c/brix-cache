11/*
12 * WHAT: Public entry point for extracting VOMS virtual organisation membership
13 * from an x509 proxy certificate. Called by the GSI authentication path after
14 * the proxy chain is verified to populate ctx->primary_vo and ctx->vo_list.
15 *
16 * WHY: VO-based ACL enforcement (brix_require_vo) requires knowledge of which
17 * virtual organisations the user belongs to. VOMS extensions embedded in the
18 * certificate provide this information. This function encapsulates the entire
19 * extraction pipeline — chain preparation, API call, and result collection.
20 *
21 * HOW: Four-phase flow:
22 * 1. Pre-checks — return NGX_DECLINED if VOMS library not loaded or parameters
23 * missing (graceful degradation per INVARIANT #8: metric labels low-cardinality)
24 * 2. Buffer preparation — convert ngx_str_t paths to NUL-terminated buffers with
25 * bounds validation against PATH_MAX
26 * 3. VOMS extraction — initialise API, duplicate certificate chain and remove the
27 * leaf (VOMS needs parent certificates for extension lookup), call retrieve()
28 * with VOMS_RECURSE_CHAIN flag; non-critical errors (VOMS_VERR_NOEXT /
29 * VOMS_VERR_NODATA) are silently skipped per GSI auth convention
30 * 4. Result collection — delegate to brix_collect_voms_vos(); cleanup chain and
31 * API state regardless of outcome
32 *
33 * INVARIANT: All wire paths → resolve_path() before open(). This function operates
34 * on certificate data, not filesystem paths, so the invariant does not apply directly.
35 */
36
37/*
38 * WHAT: Validate that VOMS extraction can proceed and that the caller-supplied
39 * directory paths fit their fixed on-stack buffers.
40 *
41 * WHY: Splits the pre-flight guard ladder out of the main entry point so the
42 * latter stays under the complexity cap. Preserves the original two-tier
43 * disposition exactly: unavailable/missing inputs degrade gracefully
44 * (NGX_DECLINED) while over-length paths are a hard error (NGX_ERROR).
45 *
46 * HOW: Mirror the original guard order — VOMS library loaded, mandatory
47 * non-empty inputs present, then path lengths bounded below the destination
48 * buffer size. Returns NGX_OK when extraction may continue.
49 */
50static ngx_int_t
51brix_voms_precheck(X509 *leaf, const ngx_str_t *vomsdir,
52 const ngx_str_t *cert_dir, size_t buf_sz)
53{
54 if (!brix_voms_available()) {
55 return NGX_DECLINED;
56 }
57
58 if (leaf == NULL || vomsdir == NULL || cert_dir == NULL
59 || vomsdir->len == 0 || cert_dir->len == 0)
60 {
61 return NGX_DECLINED;
62 }
63
64 if (vomsdir->len >= buf_sz || cert_dir->len >= buf_sz) {
65 return NGX_ERROR;
66 }
67
68 return NGX_OK;
69}

71/*
72 * WHAT: Copy an ngx_str_t path into a NUL-terminated on-stack buffer.
73 *
74 * WHY: The VOMS init() API takes C strings; ngx_str_t is not NUL-terminated.
75 * Factored out to avoid duplicating the copy-and-terminate idiom per path.
76 *
77 * HOW: Bounds have already been validated by brix_voms_precheck(); copy the
78 * bytes and append the terminator.

80static void
81brix_voms_path_to_buf(const ngx_str_t *src, char *dst)
82{
83 ngx_memcpy(dst, src->data, src->len);
84 dst[src->len] = '\0';
85}

110/*
111 * WHAT: Duplicate the verified certificate chain and drop the leaf, producing
112 * the parent-only stack that the VOMS retrieve() call expects.
113 *
114 * WHY: VOMS looks up the AC extension in the parent certificates, so the leaf
115 * (index 0, matching *leaf) must be removed. Isolating this keeps the pointer
116 * juggling and its NULL/empty edge cases out of the main flow.
117 *
118 * HOW: For a non-empty chain, sk_X509_dup() the stack; on allocation failure
119 * signal via *ok = 0 (caller must clean up the API handle). If the duplicated
120 * head equals the leaf, delete it. A NULL/empty input yields a NULL chain,
121 * which is valid input to retrieve(). Sets *ok = 1 on success.
122 */
123static STACK_OF(X509) *
124brix_voms_build_parent_chain(STACK_OF(X509) *chain, X509 *leaf, int *ok)
125{
126 STACK_OF(X509) *voms_chain = NULL;
127
128 *ok = 1;
129
130 if (chain == NULL || sk_X509_num(chain) <= 0) {
131 return NULL;
132 }
133
134 voms_chain = sk_X509_dup(chain);
135 if (voms_chain == NULL) {
136 *ok = 0;
137 return NULL;
138 }
139
140 if (sk_X509_num(voms_chain) > 0
141 && X509_cmp(sk_X509_value(voms_chain, 0), leaf) == 0)
142 {
143 sk_X509_delete(voms_chain, 0);
144 }
145
146 return voms_chain;
147}

149/*
150 * WHAT: Perform the VOMS retrieve() call and, on success, collect the VO/FQAN
151 * results into the caller's buffers.
152 *
153 * WHY: Concentrates the retrieve/error/collect decision in one helper so the
154 * entry point reads as a linear pipeline. Preserves the original disposition:
155 * VOMS_VERR_NOEXT / VOMS_VERR_NODATA are silently ignored (NGX_DECLINED per
156 * initial rc); other retrieve failures WARN-log and return NGX_ERROR.
157 *
158 * HOW: Call retrieve() with VOMS_RECURSE_CHAIN. On failure, classify the error;
159 * on success, delegate to brix_collect_voms_vos(). Uses a local error buffer for
160 * the human-readable message.
161 */
162static ngx_int_t
163brix_voms_retrieve_and_collect(ngx_log_t *log, X509 *leaf,
164 STACK_OF(X509) *voms_chain, struct voms_data *vd,
165 const brix_voms_out_t *out)
166{
167 brix_voms_api_t *api = brix_voms_get_api_internal();
168 char errbuf[512];
169 int error = 0;
170
171 if (!api->retrieve(leaf, voms_chain, VOMS_RECURSE_CHAIN, vd, &error))
172 {
173 if (error != VOMS_VERR_NOEXT && error != VOMS_VERR_NODATA) {
174 ngx_log_error(NGX_LOG_WARN, log, 0,
175 "brix: VOMS extraction failed: %s",
176 api->error_message(vd, error, errbuf,
177 (int) sizeof(errbuf)));
178 return NGX_ERROR;
179 }
180
181 return NGX_DECLINED;
182 }
183
184 return brix_collect_voms_vos(vd, out);
185}

187/*
188 * WHAT: brix_extract_voms_fqans — the full-fidelity entry point: everything
189 * brix_extract_voms_info() produces, plus the raw-FQAN CSV.
190 * WHY: 2.0 F20 — the VO-name views cannot carry a VOMS role (their safety
191 * predicate rejects '/'), so an identity built from them alone has an
192 * empty acc_role_csv and the authdb `l` selector / XrdAcc `role` template
193 * can never match. Callers that build an identity want this form;
194 * callers that only gate on VO membership can keep the older one.
195 * HOW: identical pipeline; `out` simply carries one more destination. The
196 * chain input is bundled through brix_voms_in_t so the signature stays
197 * within the five-parameter limit.
198 */
199ngx_int_t
200brix_extract_voms_fqans(ngx_log_t *log, const brix_voms_in_t *in,
201 const ngx_str_t *vomsdir, const ngx_str_t *cert_dir,
202 const brix_voms_out_t *out)
203{
204 struct voms_data *vd;
205 X509 *leaf = in->leaf;
206 STACK_OF(X509) *chain = in->chain;
207 STACK_OF(X509) *voms_chain = NULL;
208 char vomsdir_buf[PATH_MAX];
209 char cert_dir_buf[PATH_MAX];
210 int chain_ok = 0;
211 ngx_int_t rc;
212
213 rc = brix_voms_precheck(leaf, vomsdir, cert_dir, sizeof(vomsdir_buf));
214 if (rc != NGX_OK) {
215 return rc;
216 }
217
218 brix_voms_path_to_buf(vomsdir, vomsdir_buf);
219 brix_voms_path_to_buf(cert_dir, cert_dir_buf);
220 brix_voms_reset_outputs(out);
221
222 {
223 brix_voms_api_t *api = brix_voms_get_api_internal();
224
225 vd = api->init(vomsdir_buf, cert_dir_buf);
226 if (vd == NULL) {
227 return NGX_ERROR;
228 }
229
230 voms_chain = brix_voms_build_parent_chain(chain, leaf, &chain_ok);
231 if (!chain_ok) {
232 api->destroy(vd);
233 return NGX_ERROR;
234 }
235
236 rc = brix_voms_retrieve_and_collect(log, leaf, voms_chain, vd, out);
237
238 if (voms_chain != NULL) {
239 sk_X509_free(voms_chain);
240 }
241 api->destroy(vd);
242 }
243 return rc;
244}
