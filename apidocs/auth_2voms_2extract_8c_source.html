4/*
5 * extract.c — VO / FQAN extraction from a verified proxy chain
6 *
7 * WHAT: brix_extract_voms_fqans (the full form) and brix_extract_voms_info
8 * (VO views only), declared in ngx_brix_module.h and voms_http.h.
9 * WHY: Every GSI path (root://, davs://, gsiftp://, cvmfs) fills the
10 * connection identity's VO and FQAN views from here after the proxy
11 * chain has been verified; VO-scoped authorization (brix_require_vo,
12 * the authdb selectors) reads those views.
13 * HOW: brix_voms_retrieve over the shared native engine with the module's
14 * trust: the VOMS server chain against the brix_voms_cert_dir store and
15 * the configured brix_vomsdir LSC files. Only entries whose verdict is
16 * BRIX_VOMS_OK reach the views. A proxy without the extension yields
17 * NGX_DECLINED (a plain grid proxy); an extension that fails
18 * verification is logged at WARN and yields NGX_ERROR with empty views.
19 */

25 if (in == NULL || in->leaf == NULL || vomsdir == NULL || cert_dir == NULL

28 return NGX_DECLINED;

31 return NGX_ERROR;
32 }
33 return NGX_OK;

52static void
53brix_voms_log_rejections(ngx_log_t *log, const brix_voms_result_t *res)
54{
55 int i;
56 char vo[64];
57
58 for (i = 0; i < res->n; i++) {
59 const brix_voms_entry_t *e = &res->entries[i];
60

62 continue;
63 }
64 brix_sanitize_log_string(e->vo, vo, sizeof(vo));
65 ngx_log_error(NGX_LOG_WARN, log, 0,

71ngx_int_t
72brix_extract_voms_fqans(ngx_log_t *log, const brix_voms_in_t *in,
73 const ngx_str_t *vomsdir, const ngx_str_t *cert_dir,
74 const brix_voms_out_t *out)
75{
76 brix_voms_trust_t trust;
77 brix_voms_result_t *res = NULL;
78 brix_voms_status_t status;
79 char vomsdir_buf[PATH_MAX];
80 ngx_int_t rc;
81
82 rc = brix_voms_precheck(in, vomsdir, cert_dir);
83 if (rc != NGX_OK) {
84 return rc;
85 }
86 brix_voms_reset_outputs(out);
87 ngx_memcpy(vomsdir_buf, vomsdir->data, vomsdir->len);
88 vomsdir_buf[vomsdir->len] = '\0';
89
90 trust.store = brix_voms_trust_store(log, cert_dir);
91 if (trust.store == NULL) {
92 return NGX_ERROR; /* fail closed: no CA directory, no VO */
93 }
94 trust.vomsdir = vomsdir_buf;
95 trust.now = 0;
96 trust.skew_seconds = BRIX_VOMS_SKEW_SECONDS;
97
98 status = brix_voms_retrieve(in->leaf, in->chain, &trust, &res);
99 X509_STORE_free(trust.store);
100 if (status == BRIX_VOMS_ERR_NOEXT) {
101 return NGX_DECLINED; /* a plain grid proxy */
102 }
103 if (res == NULL) {
104 ngx_log_error(NGX_LOG_WARN, log, 0,
105 "brix: VOMS extension rejected: %s", brix_voms_strerror(status));
106 return NGX_ERROR;
107 }
108 brix_voms_log_rejections(log, res);
109 rc = (status == BRIX_VOMS_OK) ? brix_collect_voms_vos(res, out) : NGX_ERROR;
110 brix_voms_result_free(res);
111 return rc;
112}
