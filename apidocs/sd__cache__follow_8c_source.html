1/*
2 * sd_cache_follow.c — serve-while-filling for the whole-file fill (audit §4.5).
3 *
4 * WHAT: A cache miss whose whole-file fill is ALREADY in flight no longer waits
5 * for that fill to commit. The arriving reader opens a "follower" object
6 * over the filler's staged temp file and reads the bytes already pumped;
7 * a read at the fill frontier is answered EAGAIN, which the root read
8 * path turns into kXR_wait, so the client streams at the origin's pace.
9 *
10 * WHY: Before this, whole-file mode was strictly foreground: every concurrent
11 * reader of a cold object serialised behind one fill and paid the FULL
12 * object's transfer time before its first byte. Slice mode already served
13 * partial content (sd_cache_partial.c); whole-file mode is the gap the
14 * parity audit records as §4.5 (upstream XrdPfc serve-while-filling).
15 *
16 * HOW: The coordination channel is the filesystem, so it works across workers
17 * and across processes with no SHM and no IPC — the same doctrine as the
18 * cache fill lock (fs/cache/lock.c). The filler publishes
19 * `<local_root><key>.brixfill` naming its staged temp path; a follower
20 * reads the marker, opens that temp file O_RDONLY once, and uses its own
21 * fd's SIZE as the frontier (the pump writes strictly sequentially, so
22 * size == bytes readable). Termination is decided from the follower's own
23 * fd, not from the marker, because only the fd is race-free:
24 *
25 * st_nlink == 0 the staged file was unlinked → the fill ABORTED → EIO
26 * marker gone the staged file was renamed → COMMITTED → real EOF
27 * otherwise still pumping → EAGAIN (kXR_wait)
28 *
29 * so the fill spine must commit (rename) BEFORE withdrawing the marker,
30 * and unlink the staged file BEFORE withdrawing it on an abort — both
31 * enforced in sd_cache_follow_withdraw. A filler that dies without
32 * withdrawing leaves the marker behind; the follower bounds that with its
33 * own no-progress deadline (policy.serve_while_filling seconds since the
34 * frontier last moved) and fails ETIMEDOUT rather than hanging.
35 *
36 * The follower deliberately advertises NO capabilities: caps == 0 means
37 * brix_sd_fd() returns NGX_INVALID_FILE, so no sendfile span is ever
38 * built over a file that is still growing.
39 */

66/* `<local_root><key>.brixfill` into `dst`. Returns 0, or -1 when the store has
67 * no local root (a remote cache store has no marker plane) or the name would
68 * overflow. */
69static int
70follow_marker_path(sd_cache_inst_state *st, const char *key, char *dst,
71 size_t dstsz)
72{
73 const char *root = brix_cstore_local_root(&st->cstore);
74
75 if (root == NULL || key == NULL || key[0] != '/') {
76 return -1;
77 }
78 if ((size_t) snprintf(dst, dstsz, "%s%s%s", root, key,
79 SD_CACHE_FOLLOW_SUFFIX) >= dstsz)
80 {
81 return -1;
82 }
83 return 0;
84}

87/* 1 iff `path` lies under the cache store's local root, at a path-component
88 * boundary. The staged temp is always created beside its final object, so this
89 * is exactly the set of paths a well-formed marker can name. */
90static int
91follow_path_under_root(sd_cache_inst_state *st, const char *path)
92{
93 const char *root = brix_cstore_local_root(&st->cstore);
94 size_t n;
95
96 if (root == NULL) {
97 return 0;
98 }
99 n = strlen(root);
100 while (n > 1 && root[n - 1] == '/') { /* ignore a trailing slash */
101 n--;
102 }
103 return strncmp(path, root, n) == 0 && path[n] == '/';
104}

107/* Read the staged path (and the declared size, when `size_out` is non-NULL) out
108 * of the marker at `marker`. Returns 0 on a well-formed marker, -1 otherwise
109 * (missing, truncated, or a path that is not absolute — a marker is only ever
110 * written by this file, so a malformed one is treated as absent rather than
111 * trusted). */
112static int
113follow_read_marker(const char *marker, char *dst, size_t dstsz, off_t *size_out)
114{
115 char buf[PATH_MAX + 64];
116 ssize_t n;
117 char *sp, *end;
118 long long declared;
119 int fd;
120
121 fd = open(marker, O_RDONLY | O_CLOEXEC);
122 if (fd < 0) {
123 return -1;
124 }
125 n = read(fd, buf, sizeof(buf) - 1);
126 (void) close(fd);
127 if (n <= 0) {
128 return -1;
129 }
130 buf[n] = '\0';
131 buf[strcspn(buf, "\n")] = '\0';
132 sp = strchr(buf, ' '); /* "<pid> <declared size> <path>" */
133 if (sp == NULL) {
134 return -1;
135 }
136 declared = strtoll(sp + 1, &end, 10);
137 if (end == sp + 1 || *end != ' ' || end[1] != '/' || declared < 0) {
138 return -1;
139 }
140 sp = end + 1;
141 if (strlen(sp) >= dstsz) {
142 return -1;
143 }
144 memcpy(dst, sp, strlen(sp) + 1);
145 if (size_out != NULL) {
146 *size_out = (off_t) declared;
147 }
148 return 0;
149}

152void
153sd_cache_follow_publish(sd_cache_inst_state *st, const char *key,
154 const char *staged_path, off_t declared_size)
155{
156 char marker[PATH_MAX];
157 char line[PATH_MAX + 64];
158 int fd, len;
159
160 if (st == NULL || staged_path == NULL || staged_path[0] != '/'
161 || st->policy.serve_while_filling <= 0
162 || follow_marker_path(st, key, marker, sizeof(marker)) != 0)
163 {
164 return;
165 }
166 len = snprintf(line, sizeof(line), "%ld %lld %s\n", (long) getpid(),
167 (long long) (declared_size > 0 ? declared_size : 0),
168 staged_path);
169 if (len <= 0 || (size_t) len >= sizeof(line)) {
170 return;
171 }
172 fd = open(marker, O_WRONLY | O_CREAT | O_TRUNC | O_CLOEXEC, 0600);
173 if (fd < 0) {
174 return;
175 }
176 if (write(fd, line, (size_t) len) != len) {
177 (void) close(fd);
178 (void) unlink(marker);
179 return;
180 }
181 (void) close(fd);
182 ngx_log_debug1(NGX_LOG_DEBUG_CORE, st->log, 0,
183 "sd_cache: serve-while-filling armed for \"%s\"", key);
184}

187void
188sd_cache_follow_arm(sd_cache_inst_state *st, const char *key,
189 brix_sd_staged_t *staged, off_t declared_size)
190{
191 const brix_sd_driver_t *drv;
192 const char *path;
193
194 if (st == NULL || staged == NULL || staged->inst == NULL
195 || st->policy.serve_while_filling <= 0)
196 {
197 return;
198 }
199 /* SECURITY: a verifying fill's staged bytes are provisional — see the
200 * header. Only an unverified fill may be followed. */
201 if (st->policy.verify != BRIX_CACHE_VERIFY_OFF) {
202 return;
203 }
204 drv = staged->inst->driver;
205 if (drv == NULL || drv->staged_path == NULL) {
206 return; /* remote staged plane: no file to follow */
207 }
208 path = drv->staged_path(staged);
209 if (path == NULL) {
210 return;
211 }
212 sd_cache_follow_publish(st, key, path, declared_size);
213}

216void
217sd_cache_follow_withdraw(sd_cache_inst_state *st, const char *key, int aborted)
218{
219 char marker[PATH_MAX];
220 char staged[PATH_MAX];
221
222 if (st == NULL || st->policy.serve_while_filling <= 0
223 || follow_marker_path(st, key, marker, sizeof(marker)) != 0)
224 {
225 return;
226 }
227 /* Fail closed: a follower decides "aborted" from st_nlink == 0 on its own
228 * fd, so the staged file must be gone BEFORE the marker is. A driver whose
229 * staged_abort already unlinked it makes this a harmless ENOENT. */
230 if (aborted
231 && follow_read_marker(marker, staged, sizeof(staged), NULL) == 0)
232 {
233 (void) unlink(staged);
234 }
235 (void) unlink(marker);
236}

239/* The frontier, refreshed from the follower's own fd. Sets *nlink so the caller
240 * can tell an aborted fill (0) from a committed one. Returns -1 on fstat
241 * failure. */
242static off_t
243follow_frontier(sd_cache_follow_t *f, nlink_t *nlink)
244{
245 struct stat sb;
246
247 if (fstat(f->fd, &sb) != 0) {
248 return -1;
249 }
250 *nlink = sb.st_nlink;
251 if (sb.st_size > f->frontier) {
252 f->frontier = sb.st_size;
253 f->moved_at = ngx_current_msec;
254 }
255 return f->frontier;
256}

259/* The read verdict once the reader has caught up with the frontier: 0 = clean
260 * EOF (the fill committed), -1 with errno set otherwise. */
261static ssize_t
262follow_at_frontier(sd_cache_follow_t *f, nlink_t nlink)
263{
264 ngx_msec_int_t idle;
265
266 if (nlink == 0) {
267 errno = EIO; /* the fill aborted under us */
268 return -1;
269 }
270 if (f->declared > 0 && f->frontier >= f->declared) {
271 return 0; /* every declared byte is pumped: EOF without waiting
272 * for the commit to withdraw the marker */
273 }
274 if (access(f->marker, F_OK) != 0) {
275 return 0; /* committed: the frontier is final */
276 }
277 idle = (ngx_msec_int_t) (ngx_current_msec - f->moved_at);
278 if (idle > (ngx_msec_int_t) (f->st->policy.serve_while_filling * 1000)) {
279 errno = ETIMEDOUT; /* the filler died without withdrawing */
280 return -1;
281 }
282 errno = EAGAIN; /* still pumping: retry (kXR_wait) */
283 return -1;
284}

287static ssize_t
288sd_cache_follow_pread(brix_sd_obj_t *obj, void *buf, size_t len, off_t off)
289{
290 sd_cache_follow_t *f = obj->state;
291 nlink_t nlink = 1;
292 off_t frontier;
293
294 if (off < 0) {
295 errno = EINVAL;
296 return -1;
297 }
298 frontier = follow_frontier(f, &nlink);
299 if (frontier < 0) {
300 return -1;
301 }
302 if (off >= frontier) {
303 return follow_at_frontier(f, nlink);
304 }
305 if ((off_t) len > frontier - off) {
306 len = (size_t) (frontier - off); /* short read at the frontier */
307 }
308 return pread(f->fd, buf, len, off);
309}

312static ngx_int_t
313sd_cache_follow_fstat(brix_sd_obj_t *obj, brix_sd_stat_t *out)
314{
315 sd_cache_follow_t *f = obj->state;
316 struct stat sb;
317
318 if (fstat(f->fd, &sb) != 0) {
319 return NGX_ERROR;
320 }
321 memset(out, 0, sizeof(*out));
322 /* The DECLARED size off the marker, NOT st_size. The phase-107 C5 reserve
323 * uses fallocate(FALLOC_FL_KEEP_SIZE), which claims blocks without moving
324 * st_size, so the staged file's st_size IS the frontier — reporting it
325 * would tell the client the object is short and it would stop reading at
326 * the frontier. Only a source that declared no size falls back to it. */
327 out->size = (f->declared > 0) ? f->declared : sb.st_size;
328 out->mtime = sb.st_mtime;
329 out->ctime = sb.st_ctime;
330 out->mode = sb.st_mode;
331 out->ino = sb.st_ino;
332 out->uid = sb.st_uid;
333 out->gid = sb.st_gid;
334 out->is_reg = 1;
335 return NGX_OK;
336}

376brix_sd_obj_t *
377sd_cache_follow_open(sd_cache_inst_state *st, const char *key, int *err_out)
378{
379 sd_cache_follow_t *f;
380 brix_sd_obj_t *obj;
381 char marker[PATH_MAX];
382 char staged[PATH_MAX];
383 off_t declared = 0;
384 int fd;
385
386 if (err_out != NULL) { *err_out = ENOENT; }
387
388 if (st == NULL || st->policy.serve_while_filling <= 0
389 /* SECURITY, defence in depth: sd_cache_follow_arm already refuses to
390 * publish under a verify policy, so a marker seen here is stale or
391 * planted. Following it would stream bytes no digest ever checked to a
392 * client whose export demands verification, so the read side refuses
393 * independently rather than trusting the publisher. */
394 || st->policy.verify != BRIX_CACHE_VERIFY_OFF
395 || follow_marker_path(st, key, marker, sizeof(marker)) != 0
396 || follow_read_marker(marker, staged, sizeof(staged), &declared) != 0)
397 {
398 return NULL; /* no fill in flight for this key */
399 }
400 /* SECURITY: the marker names a path this process will open and stream to a
401 * client, so it is confined to the cache store's own root — the staged temp
402 * always sits beside its final object (brix_staged_open), so a marker that
403 * points anywhere else is malformed and is refused rather than followed. */
404 if (!follow_path_under_root(st, staged)) {
405 ngx_log_error(NGX_LOG_WARN, st->log, 0,
406 "sd_cache: serve-while-filling marker for \"%s\" names a staged "
407 "path outside the cache store - ignored", key);
408 return NULL;
409 }
410 fd = open(staged, O_RDONLY | O_CLOEXEC);
411 if (fd < 0) {
412 return NULL; /* raced the commit/abort: refill */
413 }
414 f = calloc(1, sizeof(*f));
415 obj = calloc(1, sizeof(*obj));
416 if (f == NULL || obj == NULL) {
417 free(f);
418 free(obj);
419 (void) close(fd);
420 if (err_out != NULL) { *err_out = ENOMEM; }
421 return NULL;
422 }
423 f->st = st;
424 f->fd = fd;
425 f->declared = declared;
426 f->moved_at = ngx_current_msec;
427 memcpy(f->marker, marker, strlen(marker) + 1);
428
429 obj->driver = &sd_cache_follow_driver;
430 obj->inst = &sd_cache_follow_inst;
431 obj->fd = NGX_INVALID_FILE;
432 obj->state = f;
433 obj->heap_shell = 1;
434 (void) sd_cache_follow_fstat(obj, &obj->snap);
435
436 ngx_log_debug1(NGX_LOG_DEBUG_CORE, st->log, 0,
437 "sd_cache: following in-flight fill of \"%s\"", key);
438 if (err_out != NULL) { *err_out = 0; }
439 return obj;
440}
