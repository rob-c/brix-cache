38/*
39 * webdav_check_lock_at — evaluate the lock (if any) recorded on one ancestor
40 * path `check` while checking whether the target path is blocked.
41 *
42 * WHAT: Reads the lock xattr on `check`; if active and it covers the target and
43 * the client does not own it, reports the operation as blocked. Expired
44 * locks are cleaned up lazily (delete + lock-null reap) exactly as before.
45 * WHY: Isolating the per-level lock decision keeps the ascent loop in
46 * webdav_check_locks a simple, low-complexity path-walk.
47 * HOW: The ancestor `check`/`check_len` and the immutable target `path`/
48 * `path_len` travel together in a single walk struct. Returns NGX_OK
49 * (not blocked here), NGX_HTTP_LOCKED (an owning-mismatch lock covers the
50 * target), or NGX_HTTP_INTERNAL_SERVER_ERROR (xattr read failure).
51 * Coverage rule is unchanged: exact match, or a depth-infinity ancestor
52 * on a path boundary.
53 */
54/* One level of the ancestor walk: the (mutable) ancestor path being examined
55 * plus the (immutable) target path the whole walk is checking. */
56typedef struct {
57 const char *check;
58 size_t check_len;
59 const char *path;
60 size_t path_len;
61} webdav_lock_walk_t;
62
63static ngx_int_t
64webdav_check_lock_at(ngx_http_request_t *r, const webdav_lock_walk_t *w)
65{
66 webdav_lock_xattr_t e;
67 ngx_int_t rc;
68 int covers;
69
70 rc = webdav_lock_xattr_read(r, w->check, &e);
71 if (rc == NGX_ERROR) {
72 return NGX_HTTP_INTERNAL_SERVER_ERROR;
73 }
74 if (rc != NGX_OK) {
75 return NGX_OK;
76 }
77
78 if (e.expires <= (int64_t) ngx_time()) {
79 webdav_lock_expired_cleanup(r, w->check, &e, 1 /* reap lock-null */);
80 return NGX_OK;
81 }
82
83 /* Lock covers path if: exact match or depth-infinity ancestor. */
84 covers = (w->check_len == w->path_len)
85 || (e.depth_infinity
86 && w->check_len < w->path_len
87 && (w->check[w->check_len - 1] == '/'
88 || w->path[w->check_len] == '/'));
89 if (covers && !webdav_lock_if_header_matches(r, e.token)) {
90 return NGX_HTTP_LOCKED;
91 }
92 return NGX_OK;
93}

95/*
96 * webdav_check_locks — walk from path up to export root, checking for active
97 * xattr locks at each level. O(path_depth) xattr reads.
98 *
99 * Returns NGX_HTTP_LOCKED if an active, client-unowned lock blocks this
100 * operation; NGX_OK otherwise.
101 */
102ngx_int_t
103webdav_check_locks(ngx_http_request_t *r, const char *path, int need_write)
104{
105 ngx_http_brix_webdav_loc_conf_t *conf;
106 char check[PATH_MAX];
107 size_t root_len;
108 webdav_lock_walk_t w;
109 ngx_int_t blocked;
110
111 (void) need_write;
112
113 conf = ngx_http_get_module_loc_conf(r, ngx_http_brix_webdav_module);
114 root_len = strlen(conf->common.root_canon);
115
116 ngx_cpystrn((u_char *) check, (u_char *) path, sizeof(check));
117
118 w.check = check;
119 w.path = path;
120 w.path_len = strlen(path);
121
122 for (;;) {
123 w.check_len = strlen(check);
124
125 blocked = webdav_check_lock_at(r, &w);
126 if (blocked != NGX_OK) {
127 return blocked;
128 }
129
130 /* Stop at or above export root. */
131 if (w.check_len <= root_len) {
132 break;
133 }
134
135 if (!brix_lock_path_ascend(check, w.check_len)) {
136 break;
137 }
138
139 if (strlen(check) < root_len) {
140 break;
141 }
142 }
143
144 return NGX_OK;
145}

147/*
148 * check_locks_descendants — scan direct and recursive children of a directory
149 * for unexpired xattr locks not covered by the client's If header.
150 * Does NOT walk upward; the caller is responsible for checking ancestors.
151 */
152static ngx_int_t
153check_locks_descendants(ngx_http_request_t *r, const char *dir)
154{
155 brix_vfs_ctx_t vctx;
156 brix_vfs_dir_t *dp;
157 char child[PATH_MAX];
158 size_t dir_len;
159 webdav_lock_xattr_t e;
160 ngx_int_t rc;
161
162 /* Confined NON-metered opendir (recursive lock scan; a planted in-export
163 * symlink cannot redirect it out of the export root). */
164 webdav_vfs_ctx_build(r, dir, &vctx);
165 dp = brix_vfs_opendir_quiet(&vctx, NULL);
166 if (dp == NULL) {
167 return NGX_OK;
168 }
169
170 dir_len = strlen(dir);
171 rc = NGX_OK;
172
173 for ( ;; ) {
174 ngx_str_t name;
175 brix_vfs_dirent_kind_t dkind;
176 const char *dname;
177 int is_dir;
178 ngx_int_t xrc;
179
180 /* "."/".." filtered by readdir_kind; kind from d_type. */
181 if (brix_vfs_readdir_kind(dp, &name, &dkind) != NGX_OK) {
182 break; /* NGX_DONE (end) or error → stop */
183 }
184 dname = (const char *) name.data;
185
186 if (dir_len + 1 + name.len + 1 > sizeof(child)) {
187 continue; /* path too long — skip */
188 }
189 snprintf(child, sizeof(child), "%s/%s", dir, dname);
190
191 /* Check lock xattr directly on this child. */
192 xrc = webdav_lock_xattr_read(r, child, &e);
193 if (xrc == NGX_ERROR) {
194 rc = NGX_HTTP_INTERNAL_SERVER_ERROR;
195 break;
196 }
197
198 if (xrc == NGX_OK) {
199 if (e.expires > (int64_t) ngx_time()) {
200 if (!webdav_lock_if_header_matches(r, e.token)) {
201 rc = NGX_HTTP_LOCKED;
202 break;
203 }
204 } else {
205 webdav_lock_expired_cleanup(r, child, &e,
206 1 /* reap lock-null */);
207 }
208 }
209 /* xrc == NGX_DECLINED: no lock xattr on this child — OK */
210
211 /* Recurse into subdirectories. d_type gives the answer directly; on a
212 * DT_UNKNOWN filesystem fall back to a confined no-follow probe so a
213 * trailing symlink is never followed into recursion. */
214 is_dir = (dkind == BRIX_VFS_DT_DIR);
215 if (dkind == BRIX_VFS_DT_UNKNOWN) {
216 brix_vfs_ctx_t cctx;
217 brix_vfs_stat_t cvst;
218
219 webdav_vfs_ctx_build(r, child, &cctx);
220 is_dir = (brix_vfs_probe(&cctx, 1 /* no-follow */, &cvst) == NGX_OK
221 && cvst.is_directory);
222 }
223 if (is_dir) {
224 rc = check_locks_descendants(r, child);
225 if (rc != NGX_OK) {
226 break;
227 }
228 }
229 }
230
231 brix_vfs_closedir(dp, r->connection->log);
232 return rc;
233}

235/*
236 * webdav_check_locks_tree — check locks on a path and all its descendants.
237 * Use this instead of webdav_check_locks() for DELETE/COPY/MOVE on
238 * collections, where a lock on any child must block the operation.
239 */
240ngx_int_t
241webdav_check_locks_tree(ngx_http_request_t *r, const char *path)
242{
243 brix_vfs_ctx_t vctx;
244 brix_vfs_stat_t vst;
245 ngx_int_t rc;
246
247 rc = webdav_check_locks(r, path, 1);
248 if (rc != NGX_OK) {
249 return rc;
250 }
251
252 /* Only recurse for a directory (confined no-follow probe). */
253 webdav_vfs_ctx_build(r, path, &vctx);
254 if (brix_vfs_probe(&vctx, 1 /* no-follow */, &vst) == NGX_OK
255 && vst.is_directory)
256 {
257 rc = check_locks_descendants(r, path);
258 }
259
260 return rc;
261}
