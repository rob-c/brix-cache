1/*
2 * rpm_repomd.c — the warm-set extractor for a repository index.
3 *
4 * See rpm_repomd.h for WHAT/WHY/HOW. The shape this file reads is the one
5 * createrepo writes:
6 *
7 * <repomd …>
8 * <data type="primary">
9 * <checksum type="sha256">…</checksum>
10 * <location href="repodata/<hex>-primary.xml.gz"/>
11 * </data>
12 * <data type="filelists"> … </data>
13 * </repomd>
14 *
15 * It is deliberately NOT a general XML reader. Everything it takes out of the
16 * document is one attribute value, and that value is then re-checked by the
17 * request grammar before it can turn into a fetch — so the parser's job is to
18 * be conservative and finite, not complete. A document it does not recognise
19 * yields an empty warm set, which costs a cold client exactly what it costs
20 * today.
21 */

32static const char *
33rpm_mem_find(const char *hay, size_t n, const char *needle)
34{
35 size_t nl = strlen(needle);
36 size_t i;
37
38 if (nl == 0 || nl > n) {
39 return NULL;
40 }
41 for (i = 0; i + nl <= n; i++) {
42 if (hay[i] == needle[0] && memcmp(hay + i, needle, nl) == 0) {
43 return hay + i;
44 }
45 }
46 return NULL;
47}

50/* The value of attribute `name` inside the span [p, p+n), double-quoted the
51 * way every createrepo release writes it. Single quotes are legal XML and are
52 * not accepted: a repomd this mirror cannot read is one it declines to
53 * speculate on, and that is a cache miss rather than a wrong fetch. */
54static int
55rpm_attr(const char *p, size_t n, const char *name, const char **val,
56 size_t *val_len)
57{
58 char pat[32];
59 const char *at, *end;
60 size_t used;
61
62 used = (size_t) snprintf(pat, sizeof(pat), "%s=\"", name);
63 if (used >= sizeof(pat)) {
64 return -1;
65 }
66 at = rpm_mem_find(p, n, pat);
67 if (at == NULL) {
68 return -1;
69 }
70 at += used;
71 end = memchr(at, '"', (size_t) (p + n - at));
72 if (end == NULL) {
73 return -1;
74 }
75 *val = at;
76 *val_len = (size_t) (end - at);
77 return 0;
78}

81/* An href this mirror is willing to turn into a fetch. The composition below
82 * cannot make an unsafe path safe, so everything that could make one is
83 * refused here: an absolute path or a scheme would leave the repository, a
84 * `..` would leave it upwards, and an XML entity would mean the byte the
85 * client resolves is not the byte this fetched. */
86static int
87rpm_href_ok(const char *p, size_t n)
88{
89 size_t i;
90
91 if (n == 0 || n >= BRIX_RPM_REPOMD_HREF_MAX || p[0] == '/') {
92 return 0;
93 }
94 for (i = 0; i < n; i++) {
95 if (p[i] == ':' || p[i] == '&' || p[i] == '\\' || p[i] == '%'
96 || (unsigned char) p[i] <= 0x20 || (unsigned char) p[i] >= 0x7f)
97 {
98 return 0;
99 }
100 if (p[i] == '.' && i + 1 < n && p[i + 1] == '.') {
101 return 0;
102 }
103 }
104 return 1;
105}

124/* One <data> element: its span ends at the closing tag, or — for a document
125 * that never closes it — at the next element or the end of the buffer, so a
126 * truncated repomd cannot make the reader run past what it has. */
127static size_t
128rpm_data_span(const char *p, size_t n)
129{
130 const char *close = rpm_mem_find(p, n, "</data>");
131 const char *next = rpm_mem_find(p + 1, n - 1, "<data");
132
133 if (close != NULL && (next == NULL || close < next)) {
134 return (size_t) (close - p);
135 }
136 return (next != NULL) ? (size_t) (next - p) : n;
137}

140/* The location href inside one <data> element, or -1. The href is read from
141 * the <location> tag alone rather than from the element as a whole: a
142 * <checksum> carries no href, but a future sibling element might, and the
143 * warm set must name the file this data block publishes and nothing else. */
144static int
145rpm_data_href(const char *p, size_t n, const char **href, size_t *href_len)
146{
147 const char *loc = rpm_mem_find(p, n, "<location");
148
149 if (loc == NULL) {
150 return -1;
151 }
152 return rpm_attr(loc, n - (size_t) (loc - p), "href", href, href_len);
153}

156size_t
157brix_rpm_repomd_warm_set(const char *xml, size_t len,
158 brix_rpm_repomd_ref_t *out, size_t max)
159{
160 const char *p = xml;
161 size_t left = len, found = 0;
162
163 if (xml == NULL || out == NULL || max == 0 || len == 0) {
164 return 0;
165 }
166
167 while (found < max) {
168 const char *data, *type, *href;
169 size_t span, type_len, href_len;
170
171 data = rpm_mem_find(p, left, "<data");
172 if (data == NULL) {
173 break;
174 }
175 left -= (size_t) (data - p);
176 p = data;
177 span = rpm_data_span(p, left);
178
179 if (rpm_attr(p, span, "type", &type, &type_len) == 0
180 && rpm_type_is_warm(type, type_len)
181 && rpm_data_href(p, span, &href, &href_len) == 0
182 && rpm_href_ok(href, href_len))
183 {
184 out[found].href = href;
185 out[found].href_len = href_len;
186 found++;
187 }
188
189 p += span;
190 left -= span;
191 if (left == 0) {
192 break;
193 }
194 }
195 return found;
196}

199int
200brix_rpm_repomd_sibling_key(const char *repomd_key, size_t key_len,
201 const char *href, size_t href_len, char *out, size_t out_size)
202{
203 static const char marker[] = "/repodata/";
204 const char *at;
205 size_t base;
206
207 if (repomd_key == NULL || href == NULL || out == NULL || out_size == 0) {
208 return -1;
209 }
210 /* The href is relative to the repository ROOT, which is the parent of the
211 * repodata/ directory the index lives in — so the base is everything up
212 * to and including the slash before "repodata". A key with more than one
213 * "/repodata/" in it is anchored on the LAST, the one that holds this
214 * index. */
215 at = NULL;
216 for (base = 0; base + sizeof(marker) - 1 <= key_len; base++) {
217 if (memcmp(repomd_key + base, marker, sizeof(marker) - 1) == 0) {
218 at = repomd_key + base;
219 }
220 }
221 if (at == NULL) {
222 return -1;
223 }
224 base = (size_t) (at - repomd_key) + 1; /* keep the leading slash */
225
226 if (base + href_len + 1 > out_size) {
227 return -1;
228 }
229 memcpy(out, repomd_key, base);
230 memcpy(out + base, href, href_len);
231 out[base + href_len] = '\0';
232 return 0;
233}
