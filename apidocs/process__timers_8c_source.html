1/*
2 * process_timers.c — the self-arming per-worker maintenance timers.
3 *
4 * WHAT: Owns every maintenance-timer callback (CRL hot-reload, CMS
5 * pending-locate reaper, upload stage-out reaper, cache stale-dirty
6 * reaper, async stage-flush scheduler), their static ngx_event_t globals,
7 * and the three timer-arming entry points the init_process orchestrator
8 * calls. Each static timer event is co-located with BOTH its handler and
9 * its arming function so the global never crosses a file boundary.
10 * WHY: Split (phase-79 file-size cap) out of the former 925-line process.c.
11 * Grouping the timer machinery keeps the "arm -> fire -> re-arm" lifecycle
12 * of each maintenance timer reviewable in one place. brix_crl_reload_handler
13 * and brix_cache_reap_handler are non-static (declared in
14 * process_internal.h) because a server's init ladder in
15 * process_server_init.c arms those two per-server; the three arming
16 * functions are non-static because init_process (process.c) calls them.
17 * HOW: Every handler re-arms itself with ngx_add_timer unless ngx_exiting, and
18 * marks its timer cancelable so a draining worker is never held past its
19 * real work. No behaviour change from the split.
20 */

34/*
35 * nginx's timer-expiry debug log (ngx_event_expire_timers, compiled in on a
36 * --with-debug nginx and active when a server sets `error_log ... debug`) reads
37 * ngx_event_ident(ev->data) == ((ngx_connection_t *) ev->data)->fd for EVERY
38 * timer it fires. Our worker-init maintenance timers are connection-less, so a
39 * NULL ev->data made that log dereference NULL and SIGSEGV the worker — which
40 * nginx respawns, so it crash-loops (a flood of core dumps that spiked system
41 * load hard under the debug-logging test fleet). Point such timers' ev->data at
42 * this dummy connection (fd = -1) so the debug log has a valid fd to read; the

49/* Timer callback: rebuild the GSI X509_STORE from the configured CRL file at the
50 * reload interval, then re-arm the timer. */
51void
52brix_crl_reload_handler(ngx_event_t *ev)
53{
54 ngx_stream_brix_srv_conf_t *xcf = ev->data;
55 struct stat st;
56 int is_reg_file;
57
58 /*
59 * E5: skip the CRL re-parse + store rebuild when the CRL source is an
60 * unchanged regular file, so a large CRL on slow storage never stalls the
61 * event loop on every interval. A CApath DIRECTORY is always rebuilt — its
62 * mtime does not reflect content changes of the CRL files inside it.
63 */
64 is_reg_file = (xcf->crl.len > 0
65 && stat((const char *) xcf->crl.data, &st) == 0
66 && S_ISREG(st.st_mode));
67
68 if (is_reg_file && xcf->crl_mtime != 0 && st.st_mtime == xcf->crl_mtime) {
69 ngx_log_debug1(NGX_LOG_DEBUG_EVENT, ev->log, 0,
70 "brix: CRL \"%s\" unchanged — skipping reload",
71 xcf->crl.data);
72 if (xcf->crl_reload > 0 && !ngx_exiting) {
73 ngx_add_timer(ev, (ngx_msec_t) xcf->crl_reload * 1000);
74 }
75 return;
76 }
77
78 ngx_log_error(NGX_LOG_INFO, ev->log, 0,
79 "brix: CRL reload timer fired, rebuilding store "
80 "from \"%s\"", xcf->crl.data);
81
82 /* NULL scope: the CRL hot-reload timer MUST rebuild from fresh CRLs on
83 * disk, never reuse the memoised startup store. */
84 if (brix_rebuild_gsi_store(xcf, ev->log, NULL) != NGX_OK) {
85 BRIX_DIAG_CRIT(ev->log, 0,
86 "brix: CRL reload failed for \"%s\" — keeping previous store",
87 "the CRL file/dir is unreadable, malformed, or mid-rewrite",
88 "check the path's permissions and that fetch-crl writes atomically; "
89 "the server keeps serving with the LAST-GOOD CRLs, so a newly "
90 "revoked cert may still be accepted until the next good reload",
91 xcf->crl.data);
92 } else if (is_reg_file) {
93 /* Record mtime only on a successful rebuild so a failed load is retried
94 * (not skipped) on the next interval. */
95 xcf->crl_mtime = st.st_mtime;
96 }
97
98 /* Re-arm the timer (suppressed once the worker is exiting). */
99 if (xcf->crl_reload > 0 && !ngx_exiting) {
100 ngx_add_timer(ev, (ngx_msec_t) xcf->crl_reload * 1000);
101 }
102}

178/*
179 * Phase-59 W2b: per-server at-rest CSI scrub. Recomputes every recorded block
180 * CRC of every tagged data file under the export root and surfaces at-rest rot
181 * (metrics + error.log DIAG) that the hot read path — which only verifies the
182 * blocks a read FULLY spans — would otherwise never notice. ev->data is the
183 * server conf. The pacing IS the operator's brix_csi_scrub_interval: one full
184 * walk per interval, re-armed at the same cadence.
185 *
186 * The engine (fs/backend/csi_scrub.c) is pure libc + xmeta + crc32c and is unit-
187 * tested standalone (tests/c/test_csi_scrub.c); this handler is the thin nginx
188 * caller that resolves the root, drives one walk, and folds the mismatch count
189 * into the shared-memory metrics slot.
190 */
191
192/* The per-server SHM metrics slot for `xcf`, or NULL when metrics are
193 * unconfigured (so the scrub degrades cleanly with no counter). Mirrors the slot
194 * resolution in fs/cache/cache_reap.c::reap_metrics_slot + connection/handler.c. */
195static ngx_brix_srv_metrics_t *
196brix_csi_scrub_metrics_slot(const ngx_stream_brix_srv_conf_t *xcf)
197{
198 ngx_brix_metrics_t *shm;
199
200 if (xcf->metrics_slot < 0
201 || xcf->metrics_slot >= BRIX_METRICS_MAX_SERVERS
202 || ngx_brix_shm_zone == NULL
203 || ngx_brix_shm_zone->data == NULL
204 || ngx_brix_shm_zone->data == (void *) 1)
205 {
206 return NULL;
207 }
208 shm = ngx_brix_shm_zone->data;
209 return &shm->servers[xcf->metrics_slot];
210}

212/* Per-mismatch report sink: one DIAG line per corrupt block so an operator sees
213 * the offending path + block, not just an opaque counter tick. */
214static void
215brix_csi_scrub_report(void *u, const char *path, uint64_t block,
216 uint32_t want, uint32_t got)
217{
218 ngx_log_t *log = u;
219
220 BRIX_DIAG_CRIT(log, 0,
221 "brix: CSI scrub found at-rest corruption in \"%s\" block %uL "
222 "(recorded crc32c %08xD, on-disk %08xD)",
223 "a data block's on-disk bytes no longer match the checksum recorded "
224 "when it was written — silent storage rot, a truncating restore, or an "
225 "out-of-band edit that bypassed the gateway",
226 "restore the file from a good replica; the gateway keeps serving it, so "
227 "a spanning read will now fail closed for the affected client",
228 path, block, want, got);
229}

231void
232brix_csi_scrub_handler(ngx_event_t *ev)
233{
234 ngx_stream_brix_srv_conf_t *xcf = ev->data;
235 brix_csi_scrub_stats_t st;
236 const char *root = (const char *) xcf->common.root_canon;
237
238 ngx_memzero(&st, sizeof(st));
239 (void) brix_csi_scrub_walk(root, &st, 0 /* unlimited: interval is the pace */,
240 brix_csi_scrub_report, ev->log);
241
242 if (st.mismatches > 0) {
243 ngx_brix_srv_metrics_t *m = brix_csi_scrub_metrics_slot(xcf);
244
245 if (m != NULL) {
246 (void) ngx_atomic_fetch_add(&m->csi_scrub_mismatch_total,
247 (ngx_atomic_int_t) st.mismatches);
248 }
249 ngx_log_error(NGX_LOG_ERR, ev->log, 0,
250 "brix: CSI scrub of \"%s\" found %uL corrupt block(s) "
251 "across %uL tagged file(s)",
252 root, st.mismatches, st.files_tagged);
253 } else {
254 ngx_log_debug3(NGX_LOG_DEBUG_EVENT, ev->log, 0,
255 "brix: CSI scrub of \"%s\" clean "
256 "(%uL files, %uL blocks verified)",
257 root, st.files_scanned, st.blocks_verified);
258 }
259
260 if (xcf->csi.scrub_interval > 0 && !ngx_exiting) {
261 ngx_add_timer(ev, (ngx_msec_t) xcf->csi.scrub_interval * 1000);
262 }
263}

266 * in every worker (the queue is per-worker); a no-op when empty. */

271brix_stage_sched_handler(ngx_event_t *ev)

284/* ---- Arm the per-worker async stage-flush scheduler timer ----
285 *
286 * WHAT: Installs and starts the 1s brix_stage_sched_timer that drains this
287 * worker's deferred (async) stage-flush queue. No return value — timer arming
288 * cannot fail.
289 *
290 * WHY: The deferred-flush queue is per-worker and protocol-agnostic, so the
291 * timer must run in HTTP-only (WebDAV/S3) workers too; it is therefore armed
292 * BEFORE the stream-config early-return in init_process. The tick is a no-op
293 * when the queue is empty.
294 *
295 * HOW:
296 * 1. Point the static timer event at brix_stage_sched_handler with the
297 * cycle log.
298 * 2. Mark it cancelable so it never delays graceful shutdown.
299 * 3. Arm it at BRIX_STAGE_SCHED_MS.
300 */
301void
302brix_init_stage_sched_timer(ngx_cycle_t *cycle)
303{
304 brix_stage_sched_timer.handler = brix_stage_sched_handler;
305 brix_stage_sched_timer.data = &brix_maint_timer_conn;
306 brix_stage_sched_timer.log = cycle->log;
307 /* cancelable: a background maintenance timer must NOT keep a gracefully
308 * shutting-down worker alive until worker_shutdown_timeout — nginx's
309 * ngx_event_no_timers_left() ignores cancelable timers, so the worker exits
310 * as soon as its real work is done instead of lingering seconds on restart. */
311 brix_stage_sched_timer.cancelable = 1;
312 ngx_add_timer(&brix_stage_sched_timer, BRIX_STAGE_SCHED_MS);
313}

315/* ---- Arm the worker-0 pending-locate reaper timer (managers only) ----
