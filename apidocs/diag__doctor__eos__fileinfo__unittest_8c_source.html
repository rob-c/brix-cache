63/* doctor_eos_proc: hand back the fixture whose file the command names. The `cmd`
64 * carries `mgm.cmd=fileinfo&mgm.path=<full>`; we key on the basename. */
65int
66doctor_eos_proc(brix_conn *c, const char *dir, const char *cmd,
67 char **out, brix_status *st)
68{
69 const char *body;
70 (void) c; (void) dir; (void) st;
71
72 if (strstr(cmd, "opendata-logo.png") != NULL) { body = FILEINFO_A; }
73 else if (strstr(cmd, "b.root") != NULL) { body = FILEINFO_B; }
74 else { return -1; }
75 *out = strdup(body);
76 return *out ? 0 : -1;
77}

116 *ents = e; *count = 3; return 0;
