4/* sigver.c — request signature verification (kXR_sigver, stock XrdSecProtect
5 * secver 0) and security-level enforcement.
6 * WHAT: Owns the "verify" half of XRootD request signing. Three entry points:
7 * brix_verify_pending_sigver() consumes the sigver state recorded when a
8 * kXR_sigver request arrived and validates that the immediately-following
9 * covered request matches the expected opcode, streamid, and signature;
10 * the static brix_verify_sigver_sig() decrypts the client's signature
11 * blob with the negotiated GSI session cipher and compares it against
12 * SHA-256(seqno || header || payload-unless-nodata) in constant time; and
13 * brix_signing_enforce_level() rejects opcodes that the configured
14 * brix_security_level requires to be signed but were not.
15 *
16 * WHY: XRootD signs requests so a passive observer cannot inject or tamper with
17 * mutating operations once a session is authenticated. This file enforces
18 * that contract on the server side: a mismatched/absent signature must fail
19 * closed (kXR_NotAuthorized) rather than silently passing the request to the
20 * read/write dispatchers. Splitting verify (here) from the kXR_sigver request
21 * handler (src/session/signing.c, which records the pending state) keeps the
22 * "record intent" and "check intent against next request" responsibilities
23 * in separate, single-purpose files.
24 *
25 * HOW: kXR_sigver records pending state (expected reqid, streamid, signature
26 * blob, nodata flag, seqno) elsewhere; dispatch then calls
27 * brix_verify_pending_sigver() on the next request, which gates on
28 * ctx->sigver.signing_active, matches expectrid + streamid against the
29 * covered request, and delegates the cryptographic check to
30 * brix_verify_sigver_sig() using the sigver-owned copy of the session
31 * cipher (ctx->sigver.sig_cipher/sig_key — armed by
32 * gsi_arm_request_signing(), surviving the §F6 delegation cleanse). On
33 * success it sets ctx->sigver.verified; brix_signing_enforce_level()
34 * later consults that flag plus brix_sigver_opcode_requires() (a level
35 * 0-4 policy table) to decide whether an unsigned opcode is permitted. */
36
37/*
38 * Decrypt and compare the signature over the covered request (stock
39 * XrdSecProtect secver 0: the blob is the session-cipher encryption of the
40 * plain SHA-256 over seqno_be(8) || request header(24) || payload — payload
41 * omitted when the nodata flag was set). brix_gsi_sigver_verify() decrypts
42 * with the sigver-owned session key (IV-stripped on the signed-DH path) and
43 * CRYPTO_memcmp's the 32-byte hash. Returns BRIX_DISPATCH_CONTINUE on a match;
44 * otherwise sends kXR_ServerError when the armed cipher cannot be resolved or
45 * kXR_NotAuthorized on a signature mismatch and returns that send's result.
46 */
47static ngx_int_t
48brix_verify_sigver_sig(brix_ctx_t *ctx, ngx_connection_t *c)
49{
50 brix_gsi_cipher_t cipher;
51
52 if (!brix_gsi_cipher_lookup(ctx->sigver.sig_cipher, &cipher)) {
53 ngx_log_error(NGX_LOG_ERR, c->log, 0,
54 "brix: sigver session cipher \"%s\" unavailable for reqid=%d",
55 ctx->sigver.sig_cipher, (int) ctx->recv.cur_reqid);
56 return brix_send_error(ctx, c, kXR_ServerError,
57 "signature verification failed");
58 }
59
60 /* Shared kernel (libxrdproto gsi_core) — decrypts the blob and recomputes
61 * over the SAME covered bytes the client signs. Single source of the
62 * covered-byte layout with the native client's signer. */
63 if (!brix_gsi_sigver_verify(&cipher, ctx->sigver.sig_key,
64 ctx->sigver.sig_use_iv,
65 ctx->sigver.sig, (size_t) ctx->sigver.sig_len,
66 ctx->sigver.seqno, ctx->recv.hdr_buf,
67 ctx->recv.payload, ctx->recv.cur_dlen,
68 ctx->sigver.nodata))
69 {
70 ngx_log_error(NGX_LOG_WARN, c->log, 0,
71 "brix: sigver signature mismatch for reqid=%d",
72 (int) ctx->recv.cur_reqid);
73 return brix_send_error(ctx, c, kXR_NotAuthorized,
74 "signature verification failed");
75 }
76
77 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
78 "brix: sigver verified reqid=%d",
79 (int) ctx->recv.cur_reqid);
80
81 return BRIX_DISPATCH_CONTINUE;
82}

84/*
85 * Verify the pending kXR_sigver signature against the next request.
86 *
87 * kXR_sigver itself lives in src/session/signing.c because that request records
88 * pending state. This file owns the other half: checking that pending state
89 * before the covered request is routed.
90 */
91ngx_int_t
92brix_verify_pending_sigver(brix_ctx_t *ctx, ngx_connection_t *c)
93{
94 ngx_int_t rc;
95
96 ctx->sigver.verified = 0;
97
98 if (ctx->sigver.pending && ctx->recv.cur_reqid != kXR_sigver) {
99 ctx->sigver.pending = 0;
100
101 if (ctx->sigver.signing_active) {
102 if (ctx->sigver.expectrid != ctx->recv.cur_reqid) {
103 ngx_log_error(NGX_LOG_WARN, c->log, 0,
104 "brix: sigver expectrid=%d but got reqid=%d",
105 (int) ctx->sigver.expectrid,
106 (int) ctx->recv.cur_reqid);
107 /* kXR_InvalidRequest: the request is malformed or not allowed now */
108 return brix_send_error(ctx, c, kXR_InvalidRequest,
109 "signed request opcode mismatch");
110 }
111
112 /* Stock parity (XrdSecProtect::Verify): the sigver frame and the
113 * covered request must share a streamid — the hash covers the
114 * header as sent, so a stream mismatch can never verify anyway;
115 * failing early gives a precise diagnostic. */
116 if (ctx->sigver.sid[0] != ctx->recv.hdr_buf[0]
117 || ctx->sigver.sid[1] != ctx->recv.hdr_buf[1])
118 {
119 ngx_log_error(NGX_LOG_WARN, c->log, 0,
120 "brix: sigver streamid mismatch for reqid=%d",
121 (int) ctx->recv.cur_reqid);
122 return brix_send_error(ctx, c, kXR_InvalidRequest,
123 "signed request streamid mismatch");
124 }
125
126 rc = brix_verify_sigver_sig(ctx, c);
127 if (rc != BRIX_DISPATCH_CONTINUE) {
128 return rc;
129 }
130
131 ctx->sigver.verified = 1;
132 }
133 } else if (ctx->recv.cur_reqid == kXR_sigver) {
134 ctx->sigver.pending = 0;
135 }
136
137 return BRIX_DISPATCH_CONTINUE;
138}

140/*
141 * Policy table mapping (opcode, security_level) → whether a signature is required.
142 * Levels mirror XRootD's brix_security_level: 0=none, 1=compatible (nothing
143 * required), 2=standard (mutations + kXR_open), 3=intense (everything post-login),
144 * 4=pedantic (everything). Session/auth state-machine opcodes (login, protocol,
145 * auth, endsess, ping, sigver, bind) are always exempt. Returns non-zero when the
146 * opcode must be signed at the given level.
147 */
148static int
149brix_sigver_opcode_requires(uint16_t opcode, ngx_uint_t level)
150{
151 /* Policy table now lives in the shared gsi_core.c (single source with the
152 * native client's signer). Level 4 (pedantic) folds into "everything". */
153 return brix_gsi_sigver_required(opcode, (int) level);
154}

156/*
157 * brix_signing_enforce_level — enforce the configured brix_security_level.
158 *
159 * Checks whether the current opcode requires a signature at the configured
160 * security level. If it does and the request was not signed (verified_signing=0),
161 * rejects the request with kXR_NotAuthorized.
162 */
163ngx_int_t
164brix_signing_enforce_level(brix_ctx_t *ctx, ngx_connection_t *c,
165 ngx_stream_brix_srv_conf_t *conf)
166{
167 if (!ctx->sigver.signing_active || conf->security_level == 0) {
168 return BRIX_DISPATCH_CONTINUE;
169 }
170
171 if (brix_sigver_opcode_requires(ctx->recv.cur_reqid, conf->security_level)) {
172 if (!ctx->sigver.verified) {
173 ngx_log_error(NGX_LOG_WARN, c->log, 0,
174 "brix: unsigned request %d rejected by security_level=%d",
175 (int) ctx->recv.cur_reqid, (int) conf->security_level);
176 return brix_send_error(ctx, c, kXR_NotAuthorized,
177 "request signing required for this opcode");
178 }
179
180 /* Pedantic mode: also enforce that the signature covered the payload. */
181 if (conf->security_level >= 4 && ctx->sigver.nodata && ctx->recv.cur_dlen > 0) {
182 ngx_log_error(NGX_LOG_WARN, c->log, 0,
183 "brix: pedantic signing rejection: sigver nodata flag set but payload present");
184 return brix_send_error(ctx, c, kXR_NotAuthorized,
185 "payload signing required in pedantic mode");
186 }
187 }
188
189 return BRIX_DISPATCH_CONTINUE;
190}
