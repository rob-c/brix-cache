8/* Reconnect worker (its own thread). Retries brix_reconnect with exponential
9 * backoff + jitter until success or the reconnect deadline, then signals the loop
10 * via rc_finished + the eventfd. Touches only conn (via brix_reconnect) and the
11 * rc_* handoff fields — never the loop-owned per-conn state. */
12void *
13rc_worker_main(void *arg)
14{
15 brix_aconn *ac = arg;
16 uint64_t seed = (uint64_t) (uintptr_t) ac ^ brix_mono_ns();
17 int attempt = 0;
18 brix_status st;
19 char host[256];
20 int port;
21
22 brix_status_set(&st, XRDC_ESOCK, 0, "reconnect not attempted");
23 snprintf(host, sizeof(host), "%s", ac->conn->host);
24 port = ac->conn->port;
25
26 while (brix_mono_ns() < ac->reconnect_deadline_ns) {
27 if (brix_reconnect(ac->conn, host, port, &st) == 0) {
28 ac->rc_result = 0;
29 __atomic_store_n(&ac->rc_finished, 1, __ATOMIC_RELEASE);
30 uint64_t one = 1;
31 ssize_t w = write(ac->loop->evfd, &one, sizeof(one));
32 (void) w;
33 return NULL;
34 }
35 attempt++;
36 /* Fast reconnect cadence: a transport drop is instant, not server
37 * overload, so retry promptly — 100ms..1s — which is what lets a session
38 * recover on a high packet-loss link where each multi-RTT bring-up is
39 * frequently severed. Total reconnect time is still bounded by max_stall
40 * (reconnect_deadline_ns). */
41 uint64_t base_ms = 100ULL << ((attempt < 4) ? attempt : 4); /* 100ms..1.6s */
42 if (base_ms > 1000) {
43 base_ms = 1000;
44 }
45 seed ^= seed << 13; seed ^= seed >> 7; seed ^= seed << 17; /* xorshift */
46 uint64_t sleep_ms = base_ms + (seed % 100); /* + jitter */
47 uint64_t now = brix_mono_ns();
48 if (now >= ac->reconnect_deadline_ns) {
49 break;
50 }
51 uint64_t remain_ns = ac->reconnect_deadline_ns - now;
52 uint64_t sleep_ns = sleep_ms * 1000000ULL;
53 if (sleep_ns > remain_ns) {
54 sleep_ns = remain_ns;
55 }
56 struct timespec ts = { (time_t) (sleep_ns / 1000000000ULL),
57 (long) (sleep_ns % 1000000000ULL) };
58 nanosleep(&ts, NULL);
59 }
60
61 ac->rc_result = -1;
62 ac->rc_st = st;
63 __atomic_store_n(&ac->rc_finished, 1, __ATOMIC_RELEASE);
64 uint64_t one = 1;
65 ssize_t w = write(ac->loop->evfd, &one, sizeof(one));
66 (void) w;
67 return NULL;
68}

72void
73loop_push_cmd(brix_loop *l, cmd *c)
74{
75 pthread_mutex_lock(&l->cq_lock);
76 c->next = NULL;
77 if (l->cq_tail != NULL) {
78 l->cq_tail->next = c;
79 } else {
80 l->cq_head = c;
81 }
82 l->cq_tail = c;
83 pthread_mutex_unlock(&l->cq_lock);
84
85 uint64_t one = 1;
86 ssize_t wr = write(l->evfd, &one, sizeof(one));
87 (void) wr;

91/* Run a synchronous control command (ADD/CLOSE/STOP) and wait for the loop. */
92void
93loop_run_control(brix_loop *l, cmd_type type, brix_aconn *ac)
94{
95 pthread_mutex_t mx = PTHREAD_MUTEX_INITIALIZER;
96 pthread_cond_t cv = PTHREAD_COND_INITIALIZER;
97 int done = 0;
98
99 cmd *c = (cmd *) calloc(1, sizeof(*c));
100 if (c == NULL) {
101 return; /* best-effort; nothing else we can do */
102 }
103 c->type = type;
104 c->ac = ac;
105 c->dmx = &mx;
106 c->dcv = &cv;
107 c->done = &done;
108
109 loop_push_cmd(l, c);
110
111 pthread_mutex_lock(&mx);
112 while (!done) {
113 pthread_cond_wait(&cv, &mx);
114 }
115 pthread_mutex_unlock(&mx);
116}

119void
120loop_drain_commands(brix_loop *l)
121{
122 pthread_mutex_lock(&l->cq_lock);
123 cmd *head = l->cq_head;
124 l->cq_head = l->cq_tail = NULL;
125 pthread_mutex_unlock(&l->cq_lock);
126
127 while (head != NULL) {
128 cmd *c = head;
129 head = head->next;
130
131 switch (c->type) {
132 case CMD_ADD_ACONN: {
133 brix_aconn *ac = c->ac;
134 if (io_engine_arm(l, ac, EPOLLIN) == 0) {
135 ac->next = l->aconns;
136 l->aconns = ac;
137 } else {
138 ac->dead = 1; /* attach will surface failure via submits */
139 }
140 break;
141 }
142 case CMD_SUBMIT:
143 aconn_submit_cmd(c->ac, c);
144 break;
145 case CMD_CLOSE_ACONN:
146 aconn_destroy(c->ac);
147 break;
148 case CMD_STOP:
149 l->stop = 1;
150 break;
151 }
152
153 if (c->done != NULL) {
154 pthread_mutex_lock(c->dmx);
155 *c->done = 1;
156 pthread_cond_signal(c->dcv);
157 pthread_mutex_unlock(c->dmx);
158 }
159 free(c);
160 }
161}

164/* loop */
165void *
166loop_thread(void *arg)
167{
168 brix_loop *l = (brix_loop *) arg;
169 struct epoll_event events[AIO_MAXEV];
170
171 int timeout = AIO_TICK_MS;
172 while (!l->stop) {
173 int n = io_engine_wait(l, events, AIO_MAXEV, timeout);
174
175 loop_drain_commands(l);
176 if (l->stop) {
177 break;
178 }
179
180 for (int i = 0; i < n; i++) {
181 void *ptr = events[i].data.ptr;
182 if (ptr == l) { /* eventfd wakeup */
183 uint64_t v;
184 ssize_t rd = read(l->evfd, &v, sizeof(v));
185 (void) rd;
186 continue;
187 }
188 aconn_handle_io((brix_aconn *) ptr, events[i].events);
189 }
190
191 /* adopt finished reconnects + send heartbeats on idle links */
192 for (brix_aconn *ac = l->aconns; ac != NULL; ac = ac->next) {
193 aconn_poll_reconnect(ac);
194 aconn_maybe_ping(ac);
195 }
196
197 timeout = loop_process_timeouts(l);
198 }
199
200 /* shutdown: tear down every remaining connection (fails their in-flight) */
201 while (l->aconns != NULL) {
202 aconn_destroy(l->aconns);
203 }
204 return NULL;
205}

208/* public */
209/* WHAT: Tear down a partially-initialized loop and return NULL for the caller.
210 * WHY: brix_loop_create acquires epfd, evfd, and an always-initialized cq_lock
211 * in sequence; any step can fail. Centralizing the unwind keeps each error
212 * path a flat `return` (no goto) while freeing exactly what was acquired.
213 * HOW: Both fds start at -1 and are only set once successfully opened, so the
214 * >=0 guards close only real descriptors; cq_lock is initialized before any
215 * failure can reach here, so it is always safe to destroy. Mirrors the
216 * original single-exit `fail:` ladder byte-for-byte. */
217brix_loop *
218brix_loop_create_fail(brix_loop *l)
219{
220 io_engine_teardown(l);
221 pthread_mutex_destroy(&l->cq_lock);
222 free(l);
223 return NULL;
224}

227/* Decide the loop readiness engine: io_uring only when XRDC_IO_URING_LOOP=on
228 * (or =1) AND the runtime probe passes; otherwise epoll. Default OFF — the
229 * loop engine is the experimental, highest-risk tier. Best-effort: an
230 * unavailable ring silently falls back to epoll (not a hard error). */

232brix_loop_want_uring(void)
233{
234#if (BRIX_HAVE_LIBURING)
235 const char *e = getenv("XRDC_IO_URING_LOOP");
236 if (e == NULL || (strcmp(e, "on") != 0 && strcmp(e, "1") != 0
237 && strcmp(e, "rxtx") != 0)) {
238 return 0;
239 }
240 return brix_uring_available();
241#else
242 return 0;
243#endif
244}

264brix_loop *
265brix_loop_create(brix_status *st)
266{
267 brix_loop *l = (brix_loop *) calloc(1, sizeof(*l));
268 if (l == NULL) {
269 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (loop)");
270 return NULL;
271 }
272 pthread_mutex_init(&l->cq_lock, NULL);
273 l->epfd = -1;
274 l->evfd = -1;
275 l->use_uring = brix_loop_want_uring(); /* phase-44 loop engine (default off) */
276 l->use_rxtx = l->use_uring && brix_loop_want_rxtx(); /* ii-b tier (P44-C) */
277
278 if (io_engine_setup(l, st) != 0) {
279 return brix_loop_create_fail(l);
280 }
281 if (pthread_create(&l->thread, NULL, loop_thread, l) != 0) {
282 brix_status_set(st, XRDC_ESOCK, errno, "pthread_create: %s", strerror(errno));
283 return brix_loop_create_fail(l);
284 }
285 l->thread_ok = 1;
286 return l;
287}

290void
291brix_loop_destroy(brix_loop *l)
292{
293 if (l == NULL) {
294 return;
295 }
296 if (l->thread_ok) {
297 cmd *c = (cmd *) calloc(1, sizeof(*c));
298 if (c != NULL) {
299 c->type = CMD_STOP;
300 loop_push_cmd(l, c);
301 } else {
302 l->stop = 1; /* fallback: still wakes via any traffic */
303 uint64_t one = 1;
304 ssize_t wr = write(l->evfd, &one, sizeof(one));
305 (void) wr;
306 }
307 pthread_join(l->thread, NULL);
308 }
309 io_engine_teardown(l);
310 pthread_mutex_destroy(&l->cq_lock);
311 free(l);
312}

315brix_aconn *
316brix_aconn_attach(brix_loop *l, brix_conn *conn, brix_status *st)
317{
318 if (l == NULL || conn == NULL) {
319 brix_status_set(st, XRDC_EUSAGE, 0, "attach: null argument");
320 return NULL;
321 }
322 if (conn->signing_active) {
323 brix_status_set(st, XRDC_EAUTH, 0,
324 "request signing active — async pipelining not supported; "
325 "use the synchronous path");
326 return NULL;
327 }
328
329 int flags = fcntl(conn->io.fd, F_GETFL, 0);
330 if (flags < 0 || fcntl(conn->io.fd, F_SETFL, flags | O_NONBLOCK) < 0) {
331 brix_status_set(st, XRDC_ESOCK, errno, "set non-blocking: %s", strerror(errno));
332 return NULL;
333 }
334
335 brix_aconn *ac = (brix_aconn *) calloc(1, sizeof(*ac));
336 if (ac == NULL) {
337 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (aconn)");
338 return NULL;
339 }
340 ac->loop = l;
341 ac->conn = conn;
342 ac->fd = conn->io.fd;
343 ac->ssl = conn->io.ssl;
344 ac->next_sid = 1;
345 ac->uring_slot = -1; /* phase-44: no io_uring poll slot yet (0 is valid!) */
346 ac->state = ACONN_ALIVE;
347 ac->last_activity_ns = brix_mono_ns();
348
349 /* resilience defaults (override with brix_aconn_set_resilience) */
350 ac->max_stall_ms = 60000; /* 60 s of reconnect patience */
351 ac->keepalive_ms = 15000; /* heartbeat after 15 s idle */
352 ac->def_retries = 5;
353
354 loop_run_control(l, CMD_ADD_ACONN, ac);
355
356 if (ac->dead) {
357 brix_status_set(st, XRDC_ESOCK, 0, "epoll registration failed");
358 free(ac->inflight.slots);
359 free(ac);
360 return NULL;
361 }
362 return ac;
363}

378brix_aconn_set_resilience(brix_aconn *ac, int max_stall_ms,

402 }
403
404 cmd *c = (cmd *) calloc(1, sizeof(*c));
405 if (c == NULL) {
406 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (submit)");

408 }
409 c->type = CMD_SUBMIT;
410 c->ac = ac;
411 memcpy(c->hdr, hdr24, XRD_REQUEST_HDR_LEN);
412 if (plen > 0 && payload != NULL) {
413 c->payload = (uint8_t *) malloc(plen);
414 if (c->payload == NULL) {
415 free(c);
416 brix_status_set(st, XRDC_EPROTO, 0, "out of memory (payload)");
417 return -1;
418 }
419 memcpy(c->payload, payload, plen);
420 c->plen = plen;
421 }
422 c->cb = cb;
423 c->ctx = ctx;
424 c->deadline_ms = opts ? opts->deadline_ms : 0;
425 c->max_retries = opts ? opts->max_retries : -1;
426 c->retry_safe = opts ? opts->retry_safe : 0;
427
428 loop_push_cmd(ac->loop, c);
429 return 0;
430}

465brix_aio_call_ex(brix_aconn *ac, const void *hdr24,
466 const void *payload, uint32_t plen,
467 const brix_aio_opts *opts,
468 uint16_t *kxr, uint8_t **body, uint32_t *blen,
469 brix_status *st)
470{
471 call_wait w;
472 memset(&w, 0, sizeof(w));
473 pthread_mutex_init(&w.mx, NULL);
474 pthread_cond_init(&w.cv, NULL);
475
476 if (brix_aio_submit_ex(ac, hdr24, payload, plen, opts, call_cb, &w, st) != 0) {
477 pthread_mutex_destroy(&w.mx);
478 pthread_cond_destroy(&w.cv);
479 return -1;
480 }
481

483 while (!w.done) {
484 pthread_cond_wait(&w.cv, &w.mx);

488 pthread_cond_destroy(&w.cv);

492 *st = w.st;

496 }
497 if (kxr != NULL) {
498 *kxr = w.kxr;
499 }
500 if (body != NULL) {
501 *body = w.body;
502 } else {
503 free(w.body);
504 }
505 if (blen != NULL) {
506 *blen = w.blen;
507 }
508 return 0;
509}
