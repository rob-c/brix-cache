47sigv4_is_unreserved(u_char c)
48{
49 return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')
50 || (c >= '0' && c <= '9') || c == '-' || c == '_'
51 || c == '.' || c == '~';

54static int
55sigv4_encoded_next(sigv4_encoded_cursor_t *cursor, u_char *out)
56{
57 static const u_char hex[] = "0123456789ABCDEF";
58 u_char c;
59
60 if (cursor->pos >= cursor->len) {
61 return 0;
62 }
63 c = cursor->data[cursor->pos];
64 if (sigv4_is_unreserved(c)) {
65 cursor->pos++;
66 *out = c;
67 return 1;
68 }
69 if (cursor->escape_phase == 0) {
70 cursor->escape_phase = 1;
71 *out = '%';
72 } else if (cursor->escape_phase == 1) {
73 cursor->escape_phase = 2;
74 *out = hex[c >> 4];
75 } else {
76 cursor->escape_phase = 0;
77 cursor->pos++;
78 *out = hex[c & 0x0f];
79 }
80 return 1;
81}

83static int
84sigv4_encoded_cmp(const u_char *a, size_t alen,
85 const u_char *b, size_t blen)
86{
87 sigv4_encoded_cursor_t ac = { a, alen, 0, 0 };
88 sigv4_encoded_cursor_t bc = { b, blen, 0, 0 };
89 u_char av;
90 u_char bv;
91 int ahas;
92 int bhas;
93
94 for (;;) {
95 ahas = sigv4_encoded_next(&ac, &av);
96 bhas = sigv4_encoded_next(&bc, &bv);
97 if (!ahas || !bhas) {
98 return ahas - bhas;
99 }
100 if (av != bv) {
101 return av < bv ? -1 : 1;
102 }
103 }
104}

106/* SigV4 compares the RFC 3986 encoded name bytes, then encoded value bytes. */
107static int
108qparam_cmp(const void *a, const void *b)
109{
110 const qparam_t *pa = (const qparam_t *) a;
111 const qparam_t *pb = (const qparam_t *) b;
112 int r;
113
114 r = sigv4_encoded_cmp(pa->name, pa->name_len, pb->name, pb->name_len);
115 return r != 0 ? r
116 : sigv4_encoded_cmp(pa->value, pa->value_len,
117 pb->value, pb->value_len);
118}

127/*
128 * Parse one "name=value" (or bare-flag "name") query segment spanning
129 * [segment_start, segment_end), with `equals` pointing at the first '=' in the
130 * segment or NULL when there is none, url-decoding the name and value into *p.
131 * Returns 1 if the parameter was stored, 0 if it should be skipped — a decode
132 * failure on either half, or the X-Amz-Signature parameter when skip_signature
133 * is set (it must not appear in the string it signs).
134 */
135static int
136sigv4_store_param(qparam_t *p, const u_char *segment_start,
137 const u_char *segment_end, const u_char *equals, ngx_flag_t skip_signature)
138{
139 size_t raw_name_len;
140 size_t raw_value_len;
141 const u_char *raw_value;
142
143 /* No '=' in the segment -> a bare flag parameter: the whole segment is the
144 * name and the value is empty (""), which SigV4 still canonicalises as
145 * "name=". When '=' is present, name is everything before it and value is
146 * everything after (excluding the '=' itself). */
147 raw_name_len = (size_t) ((equals ? equals : segment_end) - segment_start);
148 raw_value = equals ? equals + 1 : (u_char *) "";
149 raw_value_len = equals ? (size_t) (segment_end - equals - 1) : 0;
150
151 if (brix_http_urldecode(segment_start, raw_name_len,
152 (char *) p->name, sizeof(p->name),
153 BRIX_URLDECODE_PLUS_TO_SPACE |
154 BRIX_URLDECODE_REJECT_NUL) != BRIX_URLDECODE_OK)
155 {
156 return -1;
157 }
158 p->name_len = strlen((char *) p->name);
159
160 if (skip_signature
161 && p->name_len == 15
162 && ngx_strcmp(p->name, (u_char *) "X-Amz-Signature") == 0)
163 {
164 return 0;
165 }
166
167 if (brix_http_urldecode(raw_value, raw_value_len,
168 (char *) p->value, sizeof(p->value),
169 BRIX_URLDECODE_PLUS_TO_SPACE |
170 BRIX_URLDECODE_REJECT_NUL) != BRIX_URLDECODE_OK)
171 {
172 return -1;
173 }
174 p->value_len = strlen((char *) p->value);
175 return 1;
176}

178/* ---- Scan one query-string segment, locating its end and first '=' ----
179 *
180 * WHAT: Given the start of a segment and the end of the query string, returns a
181 * pointer to the segment end (the next '&' or the query-string end, whichever
182 * comes first) and writes the position of the FIRST '=' within the segment to
183 * *equals_out, or NULL when the segment contains no '='.
184 *
185 * WHY: The name/value split in a SigV4 query parameter is defined by the first
186 * '=' only — a value may legitimately contain further '=' bytes, so those later
187 * '=' bytes belong to the value, not the split. Isolating the scan keeps the
188 * parse loop flat and the "first-equals-wins" rule stated in exactly one place,
189 * which is load-bearing for byte-identical canonicalisation.
190 *
191 * HOW:
192 * 1. Walk from segment_start until end or a '&' delimiter is reached.
193 * 2. Record the position of the first '=' seen (leave NULL if none).
194 * 3. Publish the first-'=' position via *equals_out and return the end pointer.
195 */
196static const u_char *
197sigv4_scan_segment(const u_char *segment_start, const u_char *end,
198 const u_char **equals_out)
199{
200 const u_char *segment_end = segment_start;
201 const u_char *equals = NULL;
202
203 while (segment_end < end && *segment_end != '&') {
204 if (*segment_end == '=' && equals == NULL) {
205 equals = segment_end;
206 }
207 segment_end++;
208 }
209
210 *equals_out = equals;
211 return segment_end;
212}

214/* ---- Parse a raw query string into decoded, storable parameters ----
215 *
216 * WHAT: Splits [qs, qs+qslen) on '&' into at most max_params segments, decodes
217 * each into params[], and returns the number of parameters stored. Segments
218 * that are empty or rejected by sigv4_store_param (decode failure, or the
219 * X-Amz-Signature parameter when skip_signature is set) are dropped.
220 *
221 * WHY: The canonical query string is the exact byte sequence a client signs, so
222 * the parse must reproduce SigV4's segmentation precisely. Capping at max_params
223 * bounds the caller's fixed-size params[] stack array. Skipping X-Amz-Signature
224 * keeps the signature field out of the string it signs (self-referential).
225 *
226 * HOW:
227 * 1. Walk the query string one '&'-delimited segment at a time.
228 * 2. For each segment, locate its end and first '=' via sigv4_scan_segment.
229 * 3. Store non-empty segments accepted by sigv4_store_param, counting them.
230 * 4. Step past the '&' delimiter, or stop when the query-string end is hit.
231 */
232static int
233sigv4_parse_qs(const u_char *qs, size_t qslen, ngx_flag_t skip_signature,
234 qparam_t *params, int max_params)
235{
236 int param_count = 0;
237 const u_char *cursor = qs;
238 const u_char *end = qs + qslen;
239
240 while (cursor < end) {
241 const u_char *segment_start = cursor;
242 const u_char *equals = NULL;
243 const u_char *segment_end =
244 sigv4_scan_segment(segment_start, end, &equals);
245 int stored;
246
247 if (segment_start < segment_end) {
248 if (param_count >= max_params) {
249 return -1;
250 }
251 stored = sigv4_store_param(&params[param_count], segment_start,
252 segment_end, equals, skip_signature);
253 if (stored < 0) {
254 return -1;
255 }
256 if (stored > 0) {
257 param_count++;
258 }
259 }
260
261 /* Step past the '&' delimiter; if segment_end already hit `end` there is
262 * no delimiter to skip, so leave cursor at end to terminate the loop. */
263 cursor = (segment_end < end) ? segment_end + 1 : segment_end;
264 }
265
266 return param_count;
267}

269/* ---- Percent-encode one component and append it to the output buffer ----
270 *
271 * WHAT: Percent-encodes [src, src+slen) and appends the result at out[oi],
272 * returning the new write offset. If the encoded bytes would not fit within
273 * outsz (keeping the strict "< outsz" reservation for the caller's trailing
274 * NUL), nothing is written and oi is returned unchanged.
275 *
276 * WHY: Every append into the canonical buffer must be bounds-guarded so a short
277 * buffer truncates cleanly rather than overflowing; centralising the encode-and-
278 * guard step keeps that safety rule identical for both the name and the value.
279 *
280 * HOW:
281 * 1. Percent-encode the source component into a local scratch buffer.
282 * 2. Append it only if oi + encoded_len stays strictly below outsz.
283 * 3. Return the advanced offset (or the original offset when it did not fit).
284 */
285static int
286sigv4_append_encoded(u_char *out, size_t *oi, size_t outsz,
287 const u_char *src, size_t slen)
288{
289 u_char enc[BRIX_S3_QUERY_ENCODE_BUF_SIZE];
290 ssize_t n = uriencode_param(src, slen, enc, sizeof(enc));
291
292 if (n < 0 || *oi + (size_t) n >= outsz) {
293 return -1;
294 }
295 ngx_memcpy(out + *oi, enc, (size_t) n);
296 *oi += (size_t) n;
297 return 1;
298}

300/* ---- Emit sorted parameters as the canonical "name=value&..." string ----
301 *
302 * WHAT: Writes the already-sorted params[] as "enc(name)=enc(value)" joined by
303 * '&' into out[], NUL-terminates it, and returns the byte length written
304 * (excluding the NUL).
305 *
306 * WHY: This is the final canonical query string that both client and server feed
307 * into the signature. The '&' separator, the '=' between name and value, and the
308 * per-append bounds guard must be byte-identical to what clients compute, so the
309 * emission is expressed once here with reserved space for the trailing NUL.
310 *
311 * HOW:
312 * 1. For each parameter, prepend '&' before all but the first (bounds-guarded).
313 * 2. Append the encoded name, then a '=', then the encoded value.
314 * 3. NUL-terminate the buffer and return the written length.
315 */
316static ssize_t
317sigv4_emit_canonical_qs(const qparam_t *params, int param_count,
318 u_char *out, size_t outsz)
319{
320 size_t oi = 0;
321
322 for (int i = 0; i < param_count; i++) {
323 if (i > 0) {
324 if (oi + 1 >= outsz) {
325 return -1;
326 }
327 out[oi++] = '&';
328 }
329 if (sigv4_append_encoded(out, &oi, outsz, params[i].name,
330 params[i].name_len) < 0
331 || oi + 1 >= outsz)
332 {
333 return -1;
334 }
335 out[oi++] = '=';
336 if (sigv4_append_encoded(out, &oi, outsz, params[i].value,
337 params[i].value_len) < 0)
338 {
339 return -1;
340 }
341 }
342
343 out[oi] = '\0';
344 return oi;
345}

347/* ---- Build the SigV4 canonical query string ----
348 *
349 * WHAT: Parses the raw query string [qs, qs+qslen), sorts its parameters, and
350 * writes the canonical "enc(name)=enc(value)&..." form into out[] (NUL-
351 * terminated), returning the byte length written (excluding the NUL).
352 *
353 * WHY: SigV4 requires parameters sorted alphabetically by name then value with
354 * names and values percent-encoded — this is the exact string clients compute
355 * and sign, so it must be reproduced byte-for-byte. Parameter count is capped at
356 * 64 to bound the stack allocation of the params[] array.
357 *
358 * HOW:
359 * 1. Parse the query string into at most 64 decoded parameters.
360 * 2. Sort them by name then value using qparam_cmp (SigV4 ordering).
361 * 3. Emit the sorted parameters as the canonical query string.
362 */
363ssize_t
364build_canonical_qs(const u_char *qs, size_t qslen, ngx_flag_t skip_signature,
365 u_char *out, size_t outsz)
366{
367 qparam_t params[64];
368 int param_count;
369
370 param_count = sigv4_parse_qs(qs, qslen, skip_signature, params, 64);
371 if (param_count < 0) {
372 return -1;
373 }
374 qsort(params, (size_t) param_count, sizeof(params[0]), qparam_cmp);
375 return sigv4_emit_canonical_qs(params, param_count, out, outsz);
376}

378ssize_t
379build_canonical_parsed_qs(const brix_http_query_t *query,
380 ngx_flag_t skip_signature, u_char *out, size_t outsz)
381{
382 qparam_t params[64];
383 int count;
384 ngx_uint_t i;
385
386 if (query == NULL || !query->parsed || out == NULL || outsz == 0) {
387 return -1;
388 }
389
390 count = 0;
391 for (i = 0; i < query->count; i++) {
392 const brix_http_query_param_t *src = &query->params[i];
393
394 if (skip_signature && src->name.len == 15
395 && ngx_strcmp(src->name.data,
396 (u_char *) "X-Amz-Signature") == 0)
397 {
398 continue;
399 }
400 if (count >= 64 || src->name.len >= sizeof(params[count].name)
401 || src->value.len >= sizeof(params[count].value))
402 {
403 return -1;
404 }
405 ngx_memcpy(params[count].name, src->name.data, src->name.len);
406 params[count].name[src->name.len] = '\0';
407 params[count].name_len = src->name.len;
408 ngx_memcpy(params[count].value, src->value.data, src->value.len);
409 params[count].value[src->value.len] = '\0';
410 params[count].value_len = src->value.len;
411 count++;
412 }
413
414 qsort(params, (size_t) count, sizeof(params[0]), qparam_cmp);
415 return sigv4_emit_canonical_qs(params, count, out, outsz);
416}
