44/* Reap dead children; a slot whose child died gets its (possibly wedged)
45 * mount detached and the slot freed so the repo can remount on next access. */
46static void af_reap_children(void) {
47 for (;;) {
48 int st = 0;
49 pid_t pid = waitpid(-1, &st, WNOHANG);
50 if (pid <= 0) return;
51 pthread_mutex_lock(&g_af.tab.mu);
52 int idx = brixautofs_find_pid_locked(&g_af.tab, pid);
53 char fqrn[BRIXAUTOFS_FQRN_MAX] = "";
54 if (idx >= 0) {
55 snprintf(fqrn, sizeof(fqrn), "%s", g_af.tab.slot[idx].fqrn);
56 brixautofs_release_locked(&g_af.tab, idx);
57 }
58 pthread_mutex_unlock(&g_af.tab.mu);
59 if (fqrn[0] == '\0') continue;
60 char mntpath[768];
61 af_child_path(fqrn, mntpath, sizeof(mntpath));
62 if (af_is_mounted(mntpath)) af_umount_path(mntpath);
63 if (!g_af.shutting_down) af_log("child for %s exited", fqrn);
64 }
65}

67/* Unmount every child, then reap. Idempotent — runs from the control thread
68 * on SIGTERM (while the umbrella's worker threads still serve the path walks)
69 * and again after fuse_loop returns. */
70static void af_teardown_children(void) {
71 g_af.shutting_down = 1;
72 for (int i = 0; i < BRIXAUTOFS_MAX_REPOS; i++) {
73 pthread_mutex_lock(&g_af.tab.mu);
74 char fqrn[BRIXAUTOFS_FQRN_MAX] = "";
75 pid_t pid = 0;
76 if (g_af.tab.slot[i].st != BRIXAUTOFS_FREE) {
77 snprintf(fqrn, sizeof(fqrn), "%s", g_af.tab.slot[i].fqrn);
78 pid = g_af.tab.slot[i].pid;
79 brixautofs_release_locked(&g_af.tab, i);
80 }
81 pthread_mutex_unlock(&g_af.tab.mu);
82 if (fqrn[0] == '\0') continue;
83
84 char mntpath[768];
85 af_child_path(fqrn, mntpath, sizeof(mntpath));
86 af_umount_path(mntpath);
87 if (pid > 0) {
88 int st = 0;
89 for (int w = 0; w < 30 && waitpid(pid, &st, WNOHANG) == 0; w++) {
90 struct timespec ts = { 0, 100 * 1000 * 1000 };
91 nanosleep(&ts, NULL);
92 }
93 if (waitpid(pid, &st, WNOHANG) == 0) {
94 kill(pid, SIGTERM);
95 waitpid(pid, &st, 0);
96 }
97 }
98 }
99}

101/* Control thread: owns the self-pipe. 'C' → reap; 'T' → teardown children
102 * (worker threads still live to serve the umount path walks), then end the
103 * umbrella session; unmounting our own mountpoint makes every worker's
104 * /dev/fuse read return ENODEV, so the loop exits deterministically. */
105static void *af_control_thread(void *arg) {
106 (void) arg;
107 for (;;) {
108 char c = 0;
109 ssize_t n = read(g_af.sigpipe[0], &c, 1);
110 if (n <= 0) {
111 if (n < 0 && errno == EINTR) continue;
112 return NULL;
113 }
114 if (c == 'C') { af_reap_children(); continue; }
115 af_teardown_children();
116 fuse_exit(g_af.fuse);
117 af_umount_path(g_af.mnt);
118 return NULL;
119 }
120}

122/* Idle expiry (root only): umount2(MNT_EXPIRE) two-phase — first call marks
123 * (EAGAIN), an untouched repo expires on the next tick (0), any access in
124 * between clears the mark in-kernel. The reaper frees the slot when the
125 * child exits after its session ends. EPERM ⇒ no CAP_SYS_ADMIN ⇒ disable. */
126static void *af_idle_thread(void *arg) {
127 (void) arg;
128 unsigned tick = (unsigned) g_af.o.idle_s / 2;
129 if (tick == 0) tick = 1;
130 for (;;) {
131 sleep(tick);
132 if (g_af.shutting_down) return NULL;
133 char repos[BRIXAUTOFS_MAX_REPOS][BRIXAUTOFS_FQRN_MAX];
134 int n = 0;
135 pthread_mutex_lock(&g_af.tab.mu);
136 for (int i = 0; i < BRIXAUTOFS_MAX_REPOS; i++)
137 if (g_af.tab.slot[i].st == BRIXAUTOFS_MOUNTED)
138 snprintf(repos[n++], BRIXAUTOFS_FQRN_MAX, "%s", g_af.tab.slot[i].fqrn);
139 pthread_mutex_unlock(&g_af.tab.mu);
140 for (int i = 0; i < n; i++) {
141 char mntpath[768];
142 af_child_path(repos[i], mntpath, sizeof(mntpath));
143 if (umount2(mntpath, MNT_EXPIRE) == 0) {
144 af_log("idle-expired %s", repos[i]);
145 } else if (errno == EPERM) {
146 af_log("idle expiry needs CAP_SYS_ADMIN — disabled");
147 return NULL;
148 }
149 /* EAGAIN = marked for next tick; EBUSY = in use: both fine */
150 }
151 }
152}

156static void af_opts_o_list(char *list, autofs_opts_t *o) {
157 char *save = NULL;
158 for (char *t = strtok_r(list, ",", &save); t; t = strtok_r(NULL, ",", &save)) {
159 if (strncmp(t, "idle=", 5) == 0) o->idle_s = atoi(t + 5);
160 else if (strncmp(t, "timeout=", 8) == 0) o->spawn_timeout_s = atoi(t + 8);
161 else if (strncmp(t, "cachebase=", 10) == 0)
162 snprintf(o->cache_base, sizeof(o->cache_base), "%s", t + 10);
163 else if (strncmp(t, "mntbase=", 8) == 0)
164 snprintf(o->mnt_base, sizeof(o->mnt_base), "%s", t + 8);
165 else if (strncmp(t, "repos=", 6) == 0)
166 snprintf(o->repos, sizeof(o->repos), "%s", t + 6);
167 else if (strcmp(t, "allow_other") == 0) o->allow_other = 1;
168 else { /* forward to the umbrella's libfuse */
169 size_t cur = strlen(o->fuse_extra);
170 snprintf(o->fuse_extra + cur, sizeof(o->fuse_extra) - cur,
171 "%s%s", cur ? "," : "", t);
172 }
173 }

176static void af_parse_opts(int argc, char **argv, int start, autofs_opts_t *o) {
177 memset(o, 0, sizeof(*o));
178 o->idle_s = 600;
179 o->spawn_timeout_s = 60;
180 const char *env_cache = getenv("BRIXCVMFS_CACHE");
181 snprintf(o->cache_base, sizeof(o->cache_base), "%s",
182 env_cache ? env_cache : "/var/lib/brixcvmfs");
183 char obuf[512];
184 for (int i = start; i < argc; i++) {
185 if (strcmp(argv[i], "-o") == 0 && i + 1 < argc) {
186 snprintf(obuf, sizeof(obuf), "%s", argv[++i]);
187 af_opts_o_list(obuf, o);
188 } else if (strncmp(argv[i], "-o", 2) == 0 && argv[i][2] != '\0') {
189 snprintf(obuf, sizeof(obuf), "%s", argv[i] + 2);
190 af_opts_o_list(obuf, o);
191 } else if (strcmp(argv[i], "-f") == 0) {
192 o->foreground = 1;
193 } else if (strcmp(argv[i], "-d") == 0) {
194 o->foreground = 1;
195 o->debug = 1;
196 }

200/* Ghost list: config.d .conf basenames are the operator-configured repos. */
201static void af_scan_configured(const char *etc_root) {
202 char dd[512];
203 snprintf(dd, sizeof(dd), "%s/config.d", etc_root[0] ? etc_root : "/etc/cvmfs");
204 DIR *d = opendir(dd);
205 if (d == NULL) return;
206 struct dirent *e;
207 while ((e = readdir(d)) != NULL && g_af.nghost < BRIXAUTOFS_MAX_REPOS) {
208 const char *dot = strrchr(e->d_name, '.');
209 if (dot == NULL || strcmp(dot, ".conf") != 0) continue;
210 char name[BRIXAUTOFS_FQRN_MAX];
211 snprintf(name, sizeof(name), "%.*s", (int)(dot - e->d_name), e->d_name);
212 if (brixautofs_valid_fqrn(name))
213 snprintf(g_af.ghost[g_af.nghost++], BRIXAUTOFS_FQRN_MAX, "%s", name);
214 }
215 closedir(d);
216}

218/* brixautofs entry — dispatched by the brixMount umbrella:
219 * brixMount autofs <etc-root|-> <mountdir> [-f|-d] [-o idle=…,…] */
220/* Resolve + create the child mount farm (an absolute dir the umbrella never
221 * path-walks itself) and reject a farm nested under the umbrella mountpoint.
222 * Fills g_af.farm. Returns 0 on success, 1 on error. */
223static int
224af_setup_mount_farm(void)
225{
226 char farm_raw[512];
227 if (g_af.o.mnt_base[0])
228 snprintf(farm_raw, sizeof(farm_raw), "%s", g_af.o.mnt_base);
229 else
230 snprintf(farm_raw, sizeof(farm_raw), "%s/.mnt", g_af.o.cache_base);
231 if (af_mkdir_p(farm_raw) != 0) {
232 af_log("cannot create mount farm %s: %s", farm_raw, strerror(errno));
233 return 1;
234 }
235 char farm_rp[PATH_MAX];
236 const char *farm_src =
237 realpath(farm_raw, farm_rp) != NULL ? farm_rp : farm_raw;
238 if (strlen(farm_src) >= sizeof(g_af.farm)) {
239 af_log("mount farm path too long: %s", farm_src);
240 return 1;
241 }
242 snprintf(g_af.farm, sizeof(g_af.farm), "%.*s",
243 (int)(sizeof(g_af.farm) - 1), farm_src);
244 size_t mnl = strlen(g_af.mnt);
245 if (strncmp(g_af.farm, g_af.mnt, mnl) == 0
246 && (g_af.farm[mnl] == '/' || g_af.farm[mnl] == '\0')) {
247 af_log("mount farm %s must not live under the umbrella %s", g_af.farm, g_af.mnt);
248 return 1;
249 }
250 return 0;
251}

254/* Load the cascaded CVMFS config for repo gating: the strict-mount flag and the
255 * repo allow/ghost list (option override wins), then scan the configured set.
256 * Fills g_af.{strict,repos}. */
257static void
258af_load_repo_config(void)
259{
260 cvmfs_conf_t cf;
261 cvmfs_conf_init(&cf);
262 cvmfs_conf_load_cascade(&cf, g_af.etc[0] ? g_af.etc : NULL, "");
263 const char *strict = cvmfs_conf_get(&cf, "CVMFS_STRICT_MOUNT");
264 g_af.strict = strict != NULL
265 && (strcmp(strict, "yes") == 0 || strcmp(strict, "on") == 0
266 || strcmp(strict, "1") == 0);
267 const char *repos = cvmfs_conf_get(&cf, "CVMFS_REPOSITORIES");
268 snprintf(g_af.repos, sizeof(g_af.repos), "%s",
269 g_af.o.repos[0] ? g_af.o.repos : (repos ? repos : ""));
270 if (g_af.strict && g_af.repos[0] == '\0')
271 af_log("warning: CVMFS_STRICT_MOUNT set with no CVMFS_REPOSITORIES — "
272 "every mount will be refused");
273 af_scan_configured(g_af.etc);
274}

277/* Build the umbrella libfuse args (mountpoint goes to fuse_mount, not the arg
278 * list), create the session, mount it, and daemonize. Returns 0 with g_af.fuse
279 * live, or 1 after cleaning up any partial state. */
280static int
281af_fuse_bringup(char *argv0)
282{
283 char oarg[600];
284 snprintf(oarg, sizeof(oarg), "fsname=brixautofs,subtype=cvmfs%s%s%s",
285 g_af.o.allow_other ? ",allow_other" : "",
286 g_af.o.fuse_extra[0] ? "," : "", g_af.o.fuse_extra);
287 char *fargv[8];
288 int fargc = 0;
289 fargv[fargc++] = argv0;
290 if (g_af.o.debug) fargv[fargc++] = (char *) "-d";
291 fargv[fargc++] = (char *) "-o";
292 fargv[fargc++] = oarg;
293 struct fuse_args fargs = FUSE_ARGS_INIT(fargc, fargv);
294
295 g_af.fuse = fuse_new(&fargs, &af_ops, sizeof(af_ops), NULL);
296 if (g_af.fuse == NULL) { af_log("fuse_new failed"); return 1; }
297 if (fuse_mount(g_af.fuse, g_af.mnt) != 0) {
298 af_log("cannot mount umbrella on %s", g_af.mnt);
299 fuse_destroy(g_af.fuse);
300 return 1;
301 }
302 if (fuse_daemonize(g_af.o.foreground) != 0) {
303 af_log("daemonize failed");
304 fuse_unmount(g_af.fuse);
305 fuse_destroy(g_af.fuse);
306 return 1;
307 }
308 return 0;
309}

312/* Self-pipe + signal handlers, installed AFTER daemonize (fork drops threads;
313 * the self-pipe carries signals to the control thread). Returns 0, or 1 after
314 * unmount/destroy if the pipe cannot be created. */
315static int
316af_install_signals(void)
317{
318 if (pipe(g_af.sigpipe) != 0) {
319 af_log("pipe: %s", strerror(errno));
320 fuse_unmount(g_af.fuse);
321 fuse_destroy(g_af.fuse);
322 return 1;
323 }
324 struct sigaction sa;
325 memset(&sa, 0, sizeof(sa));
326 sa.sa_handler = af_signal;
327 sa.sa_flags = SA_RESTART | SA_NOCLDSTOP;
328 sigaction(SIGTERM, &sa, NULL);
329 sigaction(SIGINT, &sa, NULL);
330 sigaction(SIGCHLD, &sa, NULL);
331 signal(SIGPIPE, SIG_IGN);
332 return 0;
333}

336/* Spin the control + optional idle threads, run the umbrella event loop, then
337 * (on external unmount) tear the children down and join. Returns the process
338 * exit code. */
339static int
340af_run(void)
341{
342 pthread_t ctl, idle;
343 pthread_create(&ctl, NULL, af_control_thread, NULL);
344 int have_idle = g_af.o.idle_s > 0;
345 if (have_idle) pthread_create(&idle, NULL, af_idle_thread, NULL);
346
347 af_log("serving %s (etc=%s idle=%ds strict=%d)", g_af.mnt,
348 g_af.etc[0] ? g_af.etc : "/etc/cvmfs", g_af.o.idle_s, g_af.strict);
349 int rc = fuse_loop_mt(g_af.fuse, 0);
350
351 af_teardown_children();
352 close(g_af.sigpipe[1]); /* control thread read() returns 0 → exits */
353 pthread_join(ctl, NULL);
354 if (have_idle) { pthread_cancel(idle); pthread_join(idle, NULL); }
355 fuse_unmount(g_af.fuse);
356 fuse_destroy(g_af.fuse);
357 return rc == 0 ? 0 : 1;
358}

361int brixautofs_main(int argc, char **argv) {
362 if (argc < 3) {
363 fprintf(stderr,
364 "usage: brixMount autofs <etc-root|-> <mountdir> [-f|-d] [-o opts]\n"
365 " -o idle=<s> unmount idle repos after <s> seconds (default 600, 0=off)\n"
366 " -o timeout=<s> per-repo mount bring-up timeout (default 60)\n"
367 " -o cachebase=<d> child cache dirs under <d>/<fqrn> (default /var/lib/brixcvmfs)\n"
368 " -o mntbase=<d> child mount farm; repos appear as symlinks to <d>/<fqrn>\n"
369 " (default <cachebase>/.mnt)\n"
370 " -o repos=a:b restrict/ghost-list repos (overrides CVMFS_REPOSITORIES)\n"
371 " -o allow_other let other users read the mounts (needs root or user_allow_other)\n");
372 return 2;
373 }
374
375 memset(&g_af.tab, 0, sizeof(g_af.tab));
376 brixautofs_table_init(&g_af.tab);
377 if (strcmp(argv[1], "-") != 0)
378 snprintf(g_af.etc, sizeof(g_af.etc), "%s", argv[1]);
379 snprintf(g_af.mnt, sizeof(g_af.mnt), "%s", argv[2]);
380 size_t ml = strlen(g_af.mnt);
381 while (ml > 1 && g_af.mnt[ml - 1] == '/') g_af.mnt[--ml] = '\0';
382 af_parse_opts(argc, argv, 3, &g_af.o);
383
384 if (af_setup_mount_farm() != 0)
385 return 1;
386
387 /* children resolve their own repo config from the same etc root */
388 if (g_af.etc[0]) setenv("BRIXCVMFS_ETC", g_af.etc, 1);
389
390 af_load_repo_config();
391
392 mkdir(g_af.mnt, 0755); /* mount(2) will fail loudly if unusable */
393
394 if (af_fuse_bringup(argv[0]) != 0)
395 return 1;
396
397 if (af_install_signals() != 0)
398 return 1;
399
400 return af_run();
401}
