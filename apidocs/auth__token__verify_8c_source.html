29/*
30 * wt_validate_registry — run path+op-scoped registry validation for a token.
31 *
32 * WHAT: Copies the request URI into a bounded NUL-terminated buffer, populates a
33 * brix_token_registry_args_t from the location config + parsed macaroon secret,
34 * and validates the token against the issuer registry, returning the validator
35 * rc (0 = accepted) with verified claims in *claims.
36 * WHY: registry authz (phase-59 W1) is path+op dependent and therefore MUST NOT
37 * consult the token-keyed caches — it is split out so the orchestrator can route
38 * a registry config straight here every request without cache branching.
39 * HOW: bound the URI to sizeof(pathz)-1, build the args struct (secret only when
40 * slen>0), derive the op class from the HTTP method, and call
41 * brix_token_validate_registry() with a throwaway bucket out-param.
42 */
43static int
44wt_validate_registry(const wt_validate_ctx_t *v)
45{
46 ngx_http_request_t *r = v->r;
47 char pathz[2048];
48 size_t plen;
49 int bucket = 0;
50 brix_token_registry_args_t ra;
51
52 plen = (r->uri.len < sizeof(pathz) - 1) ? r->uri.len : sizeof(pathz) - 1;
53 ngx_memcpy(pathz, r->uri.data, plen);
54 pathz[plen] = '\0';
55
56 ra.log = r->connection->log;
57 ra.token = v->token;
58 ra.token_len = v->token_len;
59 ra.reg = v->conf->token_registry;
60 ra.macaroon_secret = v->slen > 0 ? v->secret : NULL;
61 ra.secret_len = (size_t) v->slen;
62 ra.clock_skew = (int) v->conf->common.token_clock_skew;
63 ra.claims = v->claims;
64
65 return brix_token_validate_registry(&ra, pathz,
66 webdav_token_op_class(r), &bucket);
67}

69/*
70 * wt_validate_jwks — validate a token against the JWKS keys + macaroon secret.
71 *
72 * WHAT: Builds a brix_token_validate_args_t from the location config's JWKS keys,
73 * issuer/audience, clock skew, and the supplied secret, then runs the JWT/macaroon
74 * validator, returning its rc (0 = accepted) with verified claims in *claims.
75 * WHY: both the primary validation and the old-secret grace-period retry issue an
76 * identical validate call differing only in the secret bytes; centralising the
77 * args assembly keeps the two callsites byte-identical and off the orchestrator's
78 * complexity budget.
79 * HOW: populate the args struct (secret NULL when slen<=0), call
80 * brix_token_validate(); no side effects beyond writing *claims on success. The
81 * secret/slen are taken as explicit overrides so the grace-period retry can pass
82 * the old secret while reusing the same immutable context.
83 */
84static int
85wt_validate_jwks(const wt_validate_ctx_t *v,
86 const u_char *secret, ssize_t slen)
87{
88 brix_token_validate_args_t va;
89
90 va.log = v->r->connection->log;
91 va.token = v->token;
92 va.token_len = v->token_len;
93 va.keys = v->conf->jwks_keys;
94 va.key_count = v->conf->jwks_key_count;
95 va.expected_issuer = (const char *) v->conf->common.token_issuer.data;
96 va.expected_audience = (const char *) v->conf->common.token_audience.data;
97 va.macaroon_secret = slen > 0 ? secret : NULL;
98 va.secret_len = (size_t) slen;
99 va.clock_skew = (int) v->conf->common.token_clock_skew;
100 va.claims = v->claims;
101
102 return brix_token_validate(&va);
103}

105/*
106 * wt_grace_retry — retry macaroon validation with the rotated-out old secret.
107 *
108 * WHAT: When primary validation has failed on a non-registry request and an old
109 * macaroon secret is configured, parses it and re-runs JWKS/macaroon validation
110 * with the old key, logging an informational note on success. Returns the retry
111 * rc, or the unchanged incoming rc when the retry is not applicable.
112 * WHY: an nginx -s reload that rotates the macaroon secret would otherwise hard-
113 * break every in-flight macaroon; accepting the old secret until tokens expire
114 * gives graceful migration. Registry validation has no shared-secret concept and
115 * is excluded.
116 * HOW: return prev_rc unchanged unless (prev_rc!=0 && !via_registry && old secret
117 * present); parse the old secret, and only if it parses (>0) call wt_validate_jwks
118 * with it, emitting the grace-period NGX_LOG_INFO line when it now accepts.
119 */
120static int
121wt_grace_retry(const wt_validate_ctx_t *v, int via_registry, int prev_rc)
122{
123 ngx_http_brix_webdav_loc_conf_t *conf = v->conf;
124 u_char old_secret[64];
125 ssize_t old_slen;
126 int rc;
127
128 /* Grace-period fallback: if the primary secret rejected a macaroon token
129 * and an old secret is configured, try validating with the old key.
130 * This lets in-flight tokens survive nginx -s reload during key rotation. */
131 if (prev_rc == 0 || via_registry || conf->common.token_macaroon_secret_old.len == 0) {
132 return prev_rc;
133 }
134
135 old_slen = brix_macaroon_secret_parse(
136 (const char *) conf->common.token_macaroon_secret_old.data,
137 conf->common.token_macaroon_secret_old.len,
138 old_secret, sizeof(old_secret));
139 if (old_slen <= 0) {
140 return prev_rc;
141 }
142
143 rc = wt_validate_jwks(v, old_secret, old_slen);
144 if (rc == 0) {
145 ngx_log_error(NGX_LOG_INFO, v->r->connection->log, 0,
146 "brix_webdav: macaroon accepted via old secret "
147 "(grace-period key rotation)");
148 }
149 return rc;
150}

152/*
153 * wt_check_issuer_keys — resolve token validity via caches, registry or keys.
154 *
155 * WHAT: Determines whether the presented token is valid, filling *claims and
156 * flagging *cache_hit / *via_registry for the caller's caching decision. Returns
157 * 0 when the token is accepted (from L1, L2, registry, JWKS, or the grace-period
158 * old-secret retry), non-zero when rejected.
159 * WHY: this is the crypto/lookup hot path — consulting the cheapest source first
160 * (per-worker L1, then cross-worker L2 SHM) avoids re-running signature checks and
161 * JSON parsing on the event loop under load. Registry authz is path+op dependent
162 * so it MUST bypass the token-keyed caches and re-validate every request.
163 * HOW: lazily create L1; on a non-registry request probe L1 then L2 (promoting an
164 * L2 hit into L1); otherwise validate via registry or JWKS; finally apply the
165 * old-secret grace retry. Only successfully validated claims are ever cached, and
166 * caching itself is left to the caller.
167 */
168int
169wt_check_issuer_keys(const wt_validate_ctx_t *v,
170 int *cache_hit, int *via_registry)
171{
172 ngx_http_brix_webdav_loc_conf_t *conf = v->conf;
173 const char *token = v->token;
174 size_t token_len = v->token_len;
175 brix_token_claims_t *claims = v->claims;
176 int rc;
177
178 /*
179 * Token-validation caches, consulted cheapest-first so token auth does not
180 * re-run crypto + JSON parsing on the event loop under load:
181 * L1 — always-on, per-worker, lockless (lazily created here). A hit skips
182 * BOTH the signature verification AND the L2 spinlock.
183 * L2 — the optional cross-worker SHM cache. An L2 hit is promoted into L1
184 * so the next presentation to this worker is an L1 hit.
185 * Only successfully validated claims are ever cached.
186 */
187 *cache_hit = 0;
188
189 /* The token-validity caches are keyed on the token alone. Registry authz
190 * (phase-59 W1) is path+op dependent, so a registry config MUST bypass the
191 * caches and re-run the per-request base_path/strategy check every time. */
192 *via_registry = (conf->token_registry != NULL);
193
194 if (conf->token_l1 == NULL) {
195 conf->token_l1 = brix_token_l1_create(ngx_cycle->pool,
196 BRIX_TOKEN_L1_SLOTS);
197 }
198
199 if (!*via_registry
200 && brix_token_l1_lookup(conf->token_l1, token, token_len, claims))
201 {
202 rc = 0;
203 *cache_hit = 1;
204 } else if (!*via_registry && conf->common.token_cache_kv != NULL
205 && brix_token_cache_lookup(conf->common.token_cache_kv,
206 token, token_len, claims))
207 {
208 rc = 0;
209 *cache_hit = 1;
210 brix_token_l1_store(conf->token_l1, token, token_len, claims);
211 } else if (*via_registry) {
212 rc = wt_validate_registry(v);
213 } else {
214 rc = wt_validate_jwks(v, v->secret, v->slen);
215 }
216
217 return wt_grace_retry(v, *via_registry, rc);
218}
