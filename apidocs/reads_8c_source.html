6/*
7 * reads.c — thread-pool offload for the stream kXR_read / kXR_pgread opcodes.
8 *
9 * WHAT: Two read paths share this file. (1) Windowed memory read
10 * (brix_read_window_pump / _emit) streams a large memory-backed
11 * (TLS / non-regular) read as fill->drain->fill chunks. (2) Plain
12 * kXR_read AIO (brix_read_aio_thread / _done) offloads a single
13 * file reads to a worker thread and returns one chained response.
14 * The third member of the family — pgread AIO, which interleaves a
15 * per-page CRC32C into the wire output — lives in pgreads.c.
16 *
17 * WHY: File I/O (and the pgread CRC32C loop) can block; running them on the
18 * nginx event-loop thread would stall every other connection on this
19 * worker. The thread pool absorbs the blocking syscall and CPU-bound
20 * checksum so the event loop stays responsive.
21 *
22 * HOW: Each path splits into a *_thread half (runs on a worker thread, may
23 * only touch the task struct — never ctx/connection/pool) and a *_done
24 * half (runs back on the event loop, owns all protocol/state mutation
25 * and chain building). The two halves communicate only through the
26 * brix_{read,pgread}_aio_t task struct carried by ngx_thread_task_t.
27 * Every *_done first re-validates the connection via
28 * brix_aio_restore_stream/_request (the stream may have died while the
29 * task ran) and ends by calling brix_aio_resume() to re-arm events.
30 */

55/*
56 * brix_read_window_emit — build + queue one window's wire chunk from
57 * rd.read_scratch[0..nread), advancing the continuation state. Status is
58 * kXR_oksofar for every window except the last (or a short read at EOF), which
59 * is kXR_ok. Returns NGX_ERROR if the read failed or the chain could not be
60 * built (an error response has already been sent); otherwise NGX_OK, with
61 * ctx->state == XRD_ST_SENDING if the chunk is still draining and
62 * ctx->rd.win_active cleared when this was the final window.
63 */
64static ngx_int_t
65brix_read_window_emit(brix_ctx_t *ctx, ngx_connection_t *c,
66 ssize_t nread, int io_errno)
67{
68 ngx_chain_t *chain;
69 uint16_t status;
70 size_t got;
71
72 if (nread < 0) {
73 brix_read_io_failure_log(c->log, "windowed", ctx->rd.win_fd,
74 ctx->rd.win_offset, ctx->rd.win_remaining,
75 io_errno);
76 ctx->rd.win_active = 0;
77 ctx->state = XRD_ST_REQ_HEADER;
78 ctx->recv.hdr_pos = 0;
79 BRIX_OP_ERR(ctx, BRIX_OP_READ);
80 brix_send_error(ctx, c, kXR_IOError,
81 io_errno ? strerror(io_errno) : "async read error");
82 return NGX_ERROR;
83 }
84
85 got = (size_t) nread;
86 ctx->files[ctx->rd.win_idx].bytes_read += got;
87 ctx->totals.bytes += got;
88
89 if (got < ctx->rd.win_remaining) {
90 ctx->rd.win_remaining -= got;
91 ctx->rd.win_offset += (off_t) got;
92 } else {
93 ctx->rd.win_remaining = 0;
94 }
95
96 /* Last planned window, or a short read (EOF), terminates the response. */
97 status = (ctx->rd.win_remaining == 0 || got == 0) ? kXR_ok : kXR_oksofar;
98
99 chain = brix_build_window_chain(ctx, c, ctx->rd.read_scratch, got, status);
100 if (chain == NULL) {
101 ctx->rd.win_active = 0;
102 ctx->state = XRD_ST_REQ_HEADER;
103 return NGX_ERROR;
104 }
105
106 if (status == kXR_ok) {
107 ctx->rd.win_active = 0;
108 BRIX_OP_OK(ctx, BRIX_OP_READ);
109 }
110
111 ctx->state = XRD_ST_REQ_HEADER;
112 ctx->recv.hdr_pos = 0;
113 brix_queue_response_chain(ctx, c, chain, ctx->rd.read_scratch);
114 if (ctx->state != XRD_ST_SENDING) {
115 brix_release_read_buffer(ctx, c, ctx->rd.read_scratch); /* no-op slot */
116 }
117 return NGX_OK;
118}

120/*
121 * brix_read_window_pump — read the next window into rd.read_scratch and emit it,
122 * looping while sends complete synchronously. Posts an AIO task when a thread
123 * pool is available (returns with state XRD_ST_AIO; brix_read_aio_done resumes
124 * the pump); otherwise reads the window inline (bounded to one window). When
125 * the windowed read finishes it resumes the event loop for the next request.
126 */
127void
128brix_read_window_pump(brix_ctx_t *ctx, ngx_connection_t *c,
129 ngx_stream_brix_srv_conf_t *rconf)
130{
131 for ( ;; ) {
132 size_t want;
133 u_char *databuf;
134 ssize_t nread;
135
136 if (!ctx->rd.win_active) {
137 brix_aio_resume(c);
138 return;
139 }
140
141 want = ctx->rd.win_remaining < (size_t) BRIX_READ_WINDOW
142 ? ctx->rd.win_remaining : (size_t) BRIX_READ_WINDOW;
143
144 databuf = BRIX_GET_SCRATCH(ctx, c, rd.read_scratch, rd.read_scratch_size,
145 want);
146 if (databuf == NULL) {
147 ctx->rd.win_active = 0;
148 ctx->state = XRD_ST_REQ_HEADER;
149 brix_send_error(ctx, c, kXR_NoMemory, "read window alloc failed");
150 brix_aio_resume(c);
151 return;
152 }
153 brix_budget_sync(ctx);

155 /* Driver-backed (memory-served, fd<0) handles — e.g. a remote root://
156 * storage backend — must NOT be posted to the pread thread pool: the
157 * driver drives an event-loop-bound origin connection (brix_cache_*),
158 * which returns EIO when driven from a worker thread. Serve such reads
159 * inline on the event loop (the branch below), matching how the ≤window
160 * buffered path and the sendfile gate already exclude driver handles. */
161 if (rconf->common.thread_pool != NULL
162 && ctx->files[ctx->rd.win_idx].sd_obj.driver == NULL) {
163 ngx_thread_task_t *task = ctx->rd.read_aio_task;
164 brix_read_aio_t *t;
165 ngx_flag_t posted = 0;
166
167 /*
168 * One task struct is allocated once per stream and cached on
169 * ctx->rd.read_aio_task, then reused across every window of this read
170 * (and across later reads) to avoid a pool allocation per window.
171 * On reuse the task is dirty from its last trip through the pool:
172 * task->next must be cleared (the pool threads it onto its run
173 * queue) and event.complete reset to 0 (ngx_post_event set it when
174 * the previous completion fired) or the next post would be ignored.
175 */
176 if (task == NULL) {
177 task = ngx_thread_task_alloc(c->pool,
178 sizeof(brix_read_aio_t));
179 if (task != NULL) {
180 ctx->rd.read_aio_task = task;
181 }
182 } else {
183 task->next = NULL;
184 task->event.complete = 0;
185 }
186
187 if (task != NULL) {
188 t = task->ctx;
189 t->c = c;
190 t->ctx = ctx;
191 t->conf = rconf;
192 t->fd = ctx->rd.win_fd;
193 t->handle_idx = ctx->rd.win_idx;
194 t->offset = ctx->rd.win_offset;
195 t->rlen = want;
196 t->databuf = databuf;
197 /*
198 * Snapshot the 2-word streamid into the task so the completion
199 * callback can verify it still matches the live ctx — by the
200 * time the worker finishes, this connection may have been torn
201 * down and the slot reused by an unrelated stream.
202 */
203 t->streamid[0] = ctx->rd.win_streamid[0];
204 t->streamid[1] = ctx->rd.win_streamid[1];
205 t->nread = 0;
206 t->io_errno = 0;
207 /*
208 * The task struct is shared with the single-shot buffered path
209 * (read_post_aio), so csi/obj still hold the LAST read's handle
210 * state. The worker routes the pread through t->obj's driver
211 * (which carries its own fd) whenever obj.driver != NULL, so a
212 * stale obj sends this window to the previous handle's fd —
213 * wrong file if the number was recycled, EBADF if closed or
214 * write-only — and a stale csi is a dangling pointer once that
215 * handle closed. Rebind both to the windowed read's own handle
216 * on every post.
217 */
218 t->csi = ctx->files[ctx->rd.win_idx].csi;
219 t->obj = ctx->files[ctx->rd.win_idx].sd_obj;
220 t->start_ns = brix_phase_now_ns(); /* phase-56 D-2 */
221 /* phase-32 WS3: the windowed read self-serializes on win_active
222 * (recv stays suspended in XRD_ST_AIO across its windows), so it
223 * is NOT counted in rd.aio_inflight — only the pipelined single-
224 * shot path is. Clear explicitly to survive task-struct reuse. */
225 t->counted = 0;
226 brix_task_bind(task, brix_read_aio_thread,
227 brix_read_aio_done);
228 (void) brix_aio_post_task(ctx, c, rconf->common.thread_pool,
229 task, "brix: window task post failed, sync fallback",
230 &posted);
231 if (posted) {
232 return; /* async: done callback resumes the pump */
233 }
234 /* post failed (queue full): fall through to the inline read. */
235 }
236 }
237
238 /*
239 * Inline fallback (no pool configured, or post failed): do the blocking
240 * VFS read on the event-loop thread for this one window only, then let
241 * the for(;;) loop pick up the next window. Bounded to a single window
242 * so a large read can never monopolise the loop for more than
243 * BRIX_READ_WINDOW.
244 */
245 {
246 brix_vfs_job_t job;
247
248 brix_vfs_job_read_init(&job, ctx->rd.win_fd,
249 ctx->rd.win_offset, want, databuf,
250 want, 0);
251 /* Same handle binding as the posted task above: verify CSI pages
252 * and route driver-backed handles through their storage object. */
253 job.csi = ctx->files[ctx->rd.win_idx].csi;
254 brix_vfs_job_set_obj(&job, &ctx->files[ctx->rd.win_idx].sd_obj);
255 brix_vfs_io_execute(&job);
256 nread = job.nio;
257 if (brix_read_window_emit(ctx, c, nread, job.io_errno)
258 == NGX_ERROR)
259 {
260 brix_aio_resume(c);
261 return;
262 }
263 }
264 if (ctx->state == XRD_ST_SENDING) {
265 return; /* async send: send.c resumes the pump on drain */
266 }
267 /* sync send complete → loop reads the next window */
268 }
269}

271/*
272 * brix_read_aio_thread — thread-pool worker for kXR_read.
273 *
274 * Runs on a worker thread; must not touch nginx state, connection pools, or
275 * any field that is not owned by the task struct. Only the blocking VFS core
276 * syscall belongs here; all protocol work happens in the done callback on the
277 * main thread.
278 *
279 * t->nread: set to pread return value (< 0 on error, 0 on EOF, > 0 on success).
280 * t->io_errno: saved errno on failure.
281 */
282void
283brix_read_aio_thread(void *data, ngx_log_t *log)
284{
285 brix_read_aio_t *t = data;
286 brix_vfs_job_t job;
287
288 /*
289 * Worker threads execute the VFS-owned thread-safe core only; all protocol
290 * state updates stay on the event-loop side in the completion callback.
291 */
292 brix_vfs_job_read_init(&job, t->fd, t->offset, t->rlen,
293 t->databuf, t->rlen, 0);
294 job.csi = t->csi; /* phase-59 W2: verify in the worker */
295 brix_vfs_job_set_obj(&job, &t->obj); /* Layer 3: route via driver if bound */
296 brix_vfs_io_execute(&job);
297
298 t->nread = job.nio;
299 t->io_errno = job.io_errno; /* CSI mismatch surfaces as EIO here */
300}

302/*
303 * brix_read_aio_done — main-thread completion callback for kXR_read AIO.
304 *
305 * Called by nginx's event loop after the thread pool posts the result via
306 * ngx_post_event. Responsibilities:
307 * 1. Guard against stale connection (ctx->destroyed check via restore_stream).
308 * 2. On I/O error: send kXR_IOError, release databuf, resume event loop.
309 * 3. On success: build a response chain (chunked if > 16 MiB), update per-
310 * handle and session byte counters, queue the chain.
311 * 4. Call brix_aio_resume() to re-arm the appropriate event (write or read).
312 *
313 * NOTE: if the chain send blocks (XRD_ST_SENDING), databuf is kept alive as
314 * wchain_base and freed by brix_release_pending_buffer after full drain.
315 */
316/*
317 * The stream vanished while this read ran. On the last outstanding op (no
318 * pwrite and no read still in flight) run the held teardown that
319 * brix_defer_teardown_if_writing parked; otherwise touch nothing — another
320 * completion still references ctx.
321 */
322static void
323brix_read_aio_orphaned(brix_read_aio_t *t, brix_ctx_t *ctx,
324 ngx_connection_t *c)
325{
326 if (t->counted && ctx->out.finalize_pending
327 && ctx->out.wr_inflight == 0 && ctx->rd.aio_inflight == 0)
328 {
329 brix_run_deferred_teardown(ctx, c); /* frees ctx — return now */
330 }
331}

334/*
335 * Phase 31 W2.1: this completion belongs to one window of a windowed read.
336 * Emit its chunk, then take exactly one of three paths (no fall-through):
337 * a) chunk still draining (XRD_ST_SENDING): hand off; send.c re-enters the
338 * pump once the socket has flushed this window.
339 * b) chunk sent synchronously and more windows remain (win_active still set):
340 * drive the next window now.
341 * c) windowed read complete (win_active cleared by emit): resume the normal
342 * request event loop.
343 */
344static void
345brix_read_aio_window_done(brix_read_aio_t *t, brix_ctx_t *ctx,
346 ngx_connection_t *c)
347{
348 if (brix_read_window_emit(ctx, c, t->nread, t->io_errno) == NGX_ERROR) {
349 brix_aio_resume(c);
350 return;
351 }
352 if (ctx->state == XRD_ST_SENDING) {
353 return; /* (a) send.c resumes the pump when it drains */
354 }
355 if (ctx->rd.win_active) {
356 brix_read_window_pump(ctx, c, t->conf); /* (b) sync-sent: next window */
357 return;
358 }
359 brix_aio_resume(c); /* (c) finished */
360}

380/* Successful single-chunk read: account the bytes, frame the response and queue
381 * it, then release or park the pool buffer depending on whether it drained. */
382static void
383brix_read_aio_deliver(brix_read_aio_t *t, brix_ctx_t *ctx, ngx_connection_t *c)
384{
385 ngx_chain_t *rsp_chain;
386
387 ctx->files[t->handle_idx].bytes_read += (size_t) t->nread;
388 ctx->totals.bytes += (size_t) t->nread;
389 BRIX_OP_OK(ctx, BRIX_OP_READ);
390
391 rsp_chain = brix_build_chunked_chain(ctx, c,
392 t->databuf, (size_t) t->nread);
393 if (rsp_chain == NULL) {
394 brix_release_read_buffer(ctx, c, t->databuf);
395 ctx->state = XRD_ST_REQ_HEADER;
396 brix_aio_resume(c);
397 return;
398 }
399
400 ctx->state = XRD_ST_REQ_HEADER;
401 ctx->recv.hdr_pos = 0;
402
403 brix_queue_response_chain(ctx, c, rsp_chain, t->databuf);
404 if (ctx->state != XRD_ST_SENDING) {
405 brix_release_read_buffer(ctx, c, t->databuf);
406 } else {
407 /*
408 * Parked and draining: the buffer is a per-in-flight rd_pool slot and the
409 * header is per-slot, so this memory-backed read is SAFE to pipeline.
410 * phase-32 WS3: recv no longer suspends on a cold read's AIO — after the
411 * post it keeps receiving and issues the next read into a different pool
412 * buffer with its own per-slot task (read_post_aio), so cold reads now
413 * pipeline like the warm-cache inline path. Single-chunk only (non-
414 * windowed read <= BRIX_READ_WINDOW < CHUNK_MAX).
415 */
416 ctx->out.resp_pipelinable = 1;
417 }
418 brix_aio_resume(c);
419}

423brix_read_aio_done(ngx_event_t *ev)
424{
425 ngx_thread_task_t *task = ev->data;
426 brix_read_aio_t *t = task->ctx;
427 brix_ctx_t *ctx = t->ctx;
428 ngx_connection_t *c = t->c;
429
430 /*
431 * phase-32 WS3 (concurrent read-AIO): a counted single-shot read is no
432 * longer in flight. Decrement BEFORE the liveness guard so a deferred
433 * teardown (the client vanished while pipelined reads were running) fires on
434 * the LAST completion — once no worker thread references any rd_pool buffer.
435 * Mirrors the wr_inflight discipline in brix_write_aio_done.
436 */
437 if (t->counted && ctx->rd.aio_inflight > 0) {
438 ctx->rd.aio_inflight--;
439 }
440
441 if (!brix_aio_restore_stream(ctx, t->streamid)) {
442 brix_read_aio_orphaned(t, ctx, c);
443 return;
444 }
445
446 /* phase-56 D-2: file this completed read (per window for a windowed read —
447 * each window is one physical I/O op) into the op-latency histogram. */
448 brix_aio_metric_done(t->start_ns, BRIX_METRIC_OP_READ);
449
450 if (ctx->rd.win_active) {
451 brix_read_aio_window_done(t, ctx, c);
452 return;
453 }
454 if (t->nread < 0) {
455 brix_read_aio_failed(t, ctx, c);
