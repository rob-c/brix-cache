6/*
7 * reads.c — thread-pool offload for the stream kXR_read / kXR_pgread opcodes.
8 *
9 * WHAT: The worker-thread half and every completion half of the read-AIO
10 * family: brix_read_aio_thread (the blocking VFS read) and
11 * brix_read_aio_done, which routes a completion to the single-shot
12 * deliver path or to the windowed train (whose forward drive — pump,
13 * per-window post, round-12 double-buffered read-ahead — lives in
14 * reads_window.c). pgread AIO, which interleaves a per-page CRC32C
15 * into the wire output, lives in pgreads.c.
16 *
17 * WHY: File I/O (and the pgread CRC32C loop) can block; running them on the
18 * nginx event-loop thread would stall every other connection on this
19 * worker. The thread pool absorbs the blocking syscall and CPU-bound
20 * checksum so the event loop stays responsive.
21 *
22 * HOW: Each path splits into a *_thread half (runs on a worker thread, may
23 * only touch the task struct — never ctx/connection/pool) and a *_done
24 * half (runs back on the event loop, owns all protocol/state mutation
25 * and chain building). The two halves communicate only through the
26 * brix_{read,pgread}_aio_t task struct carried by ngx_thread_task_t.
27 * Every *_done first re-validates the connection via
28 * brix_aio_restore_stream/_request (the stream may have died while the
29 * task ran) and ends by calling brix_aio_resume() to re-arm events.
30 */

52/*
53 * brix_read_aio_thread — thread-pool worker for kXR_read.
54 *
55 * Runs on a worker thread; must not touch nginx state, connection pools, or
56 * any field that is not owned by the task struct. Only the blocking VFS core
57 * syscall belongs here; all protocol work happens in the done callback on the
58 * main thread.
59 *
60 * t->nread: set to pread return value (< 0 on error, 0 on EOF, > 0 on success).
61 * t->io_errno: saved errno on failure.
62 */
63void
64brix_read_aio_thread(void *data, ngx_log_t *log)
65{
66 brix_read_aio_t *t = data;
67 brix_vfs_job_t job;
68
69 /*
70 * Worker threads execute the VFS-owned thread-safe core only; all protocol
71 * state updates stay on the event-loop side in the completion callback.
72 */
73 brix_vfs_job_read_init(&job, t->fd, t->offset, t->rlen,
74 t->databuf, t->rlen, 0);
75 if (t->pg) {
76 job.op = BRIX_VFS_IO_PGREAD; /* windowed pgread: in-place gapped
77 * encode + per-page CRC32c */
78 }
79 job.csi = t->csi; /* phase-59 W2: verify in the worker */
80 brix_vfs_job_set_obj(&job, &t->obj); /* Layer 3: route via driver if bound */
81 brix_vfs_io_execute(&job);
82
83 t->out_size = job.out_size; /* pg only: encoded wire bytes */
84 t->nread = job.nio;
85 t->io_errno = job.io_errno; /* CSI mismatch surfaces as EIO here */
86}

88/*
89 * brix_read_aio_done — main-thread completion callback for kXR_read AIO.
90 *
91 * Called by nginx's event loop after the thread pool posts the result via
92 * ngx_post_event. Responsibilities:
93 * 1. Guard against stale connection (ctx->destroyed check via restore_stream).
94 * 2. On I/O error: send kXR_IOError, release databuf, resume event loop.
95 * 3. On success: build a response chain (chunked if > 16 MiB), update per-
96 * handle and session byte counters, queue the chain.
97 * 4. Call brix_aio_resume() to re-arm the appropriate event (write or read).
98 *
99 * NOTE: if the chain send blocks (XRD_ST_SENDING), databuf is kept alive as
100 * wchain_base and freed by brix_release_pending_buffer after full drain.
101 */
102/*
103 * The stream vanished while this read ran. On the last outstanding op (no
104 * pwrite and no read still in flight) run the held teardown that
105 * brix_defer_teardown_if_writing parked; otherwise touch nothing — another
106 * completion still references ctx.
107 */
108static void
109brix_read_aio_orphaned(brix_read_aio_t *t, brix_ctx_t *ctx,
110 ngx_connection_t *c)
111{
112 if (t->counted && ctx->out.finalize_pending
113 && ctx->out.wr_inflight == 0 && ctx->rd.aio_inflight == 0)
114 {
115 brix_run_deferred_teardown(ctx, c); /* frees ctx — return now */
116 }
117}

120/*
121 * Phase 31 W2.1: this completion belongs to one window of a windowed read.
122 * Emit its chunk, then take exactly one of three paths (no fall-through):
123 * a) chunk still draining (XRD_ST_SENDING): hand off; send.c re-enters the
124 * pump once the socket has flushed this window.
125 * b) chunk sent synchronously and more windows remain (win_active still set):
126 * drive the next window now.
127 * c) windowed read complete (win_active cleared by emit): resume the normal
128 * request event loop.
129 */
130static void
131brix_read_aio_window_done(brix_read_aio_t *t, brix_ctx_t *ctx,
132 ngx_connection_t *c)
133{
134 if (brix_read_window_emit_step(ctx, c, t->conf, t->nread, t->out_size,
135 t->io_errno) == NGX_ERROR)
136 {
137 brix_read_window_park_or_resume(ctx, c);
138 return;
139 }
140 /* phase-56 D-2, one sample per REQUEST: the emit clears win_active on the
141 * final window, so this is the windowed read's single histogram sample.
142 * It has to happen HERE — before the pump/resume below, either of which
143 * may finish the request and retire ctx. */
144 if (!ctx->rd.win_active) {
145 brix_aio_metric_done(t->start_ns, BRIX_METRIC_OP_READ);
146 }
147 if (ctx->state == XRD_ST_SENDING) {
148 return; /* (a) send.c resumes the pump when it drains */
149 }
150 if (ctx->rd.win_active) {
151 brix_read_window_pump(ctx, c, t->conf); /* (b) sync-sent: next window */
152 return;
153 }
154 brix_aio_resume(c); /* (c) finished */
155}

158/*
159 * Round 12: a read-ahead window finished on the worker. If the train died
160 * while it ran (an emit failed), discard the result — and if the pump parked
161 * the connection on this very completion, restore the request loop. Live
162 * train: stash the result (win_ready) and let the previous frame's drain
163 * (send.c → pump) emit it; if the drain already happened, emit now.
164 */
165static void
166brix_read_window_prefetch_done(brix_read_aio_t *t, brix_ctx_t *ctx,
167 ngx_connection_t *c)
168{
169 ctx->rd.win_prefetch = 0;
170
171 if (!ctx->rd.win_active) {
172 if (ctx->state == XRD_ST_AIO) {
173 ctx->state = XRD_ST_REQ_HEADER;
174 ctx->recv.hdr_pos = 0;
175 brix_aio_resume(c);
176 }
177 return;
178 }
179
180 ctx->rd.win_ready = 1;
181 ctx->rd.win_pf_nread = t->nread;
182 ctx->rd.win_pf_osz = t->out_size;
183 ctx->rd.win_pf_errno = t->io_errno;
184
185 if (ctx->state == XRD_ST_SENDING) {
186 return; /* previous frame still draining; send.c re-enters the pump */
187 }
188 brix_read_window_pump(ctx, c, t->conf);
189}

209/* Successful single-chunk read: account the bytes, frame the response and queue
210 * it, then release or park the pool buffer depending on whether it drained. */
211static void
212brix_read_aio_deliver(brix_read_aio_t *t, brix_ctx_t *ctx, ngx_connection_t *c)
213{
214 ngx_chain_t *rsp_chain;
215
216 ctx->files[t->handle_idx].bytes_read += (size_t) t->nread;
217 ctx->totals.bytes += (size_t) t->nread;
218 BRIX_OP_OK(ctx, BRIX_OP_READ);
219
220 rsp_chain = brix_build_chunked_chain(ctx, c,
221 t->databuf, (size_t) t->nread);
222 if (rsp_chain == NULL) {
223 brix_release_read_buffer(ctx, c, t->databuf);
224 ctx->state = XRD_ST_REQ_HEADER;
225 brix_aio_resume(c);
226 return;
227 }
228
229 ctx->state = XRD_ST_REQ_HEADER;
230 ctx->recv.hdr_pos = 0;
231
232 brix_queue_response_chain(ctx, c, rsp_chain, t->databuf);
233 if (ctx->state != XRD_ST_SENDING) {
234 brix_release_read_buffer(ctx, c, t->databuf);
235 } else {
236 /*
237 * Parked and draining: the buffer is a per-in-flight rd_pool slot and the
238 * header is per-slot, so this memory-backed read is SAFE to pipeline.
239 * phase-32 WS3: recv no longer suspends on a cold read's AIO — after the
240 * post it keeps receiving and issues the next read into a different pool
241 * buffer with its own per-slot task (read_post_aio), so cold reads now
242 * pipeline like the warm-cache inline path. Single-chunk only (non-
243 * windowed read <= BRIX_READ_WINDOW < CHUNK_MAX).
244 */
245 ctx->out.resp_pipelinable = 1;
246 }
247 brix_aio_resume(c);
248}

251void
252brix_read_aio_done(ngx_event_t *ev)
253{
254 ngx_thread_task_t *task = ev->data;
255 brix_read_aio_t *t = task->ctx;
256 brix_ctx_t *ctx = t->ctx;
257 ngx_connection_t *c = t->c;
258
259 /*
260 * phase-32 WS3 (concurrent read-AIO): a counted single-shot read is no
261 * longer in flight. Decrement BEFORE the liveness guard so a deferred
262 * teardown (the client vanished while pipelined reads were running) fires on
263 * the LAST completion — once no worker thread references any rd_pool buffer.
264 * Mirrors the wr_inflight discipline in brix_write_aio_done.
265 */
266 if (t->counted && ctx->rd.aio_inflight > 0) {
267 ctx->rd.aio_inflight--;
268 }
269
270 if (!brix_aio_restore_stream(ctx, t->streamid)) {
271 brix_read_aio_orphaned(t, ctx, c);
272 return;
273 }
274
275 /* phase-56 D-2: file this completed read into the op-latency histogram.
276 * The exporter's io_ops_total books ONE op per kXR_read REQUEST (the
277 * legacy per-port fold), so a windowed read must sample ONCE too — the
278 * windowed path files its own sample on the final window (inside
279 * brix_read_aio_window_done / the pump's win_ready path, where ctx is
280 * still alive), or the histogram count would exceed the op count and
281 * break aio.h's "AIO-sampled subset of ops" contract.
282 *
283 * Round 12: windowed-family completions route on TASK IDENTITY, never on
284 * win_active alone — a pipelined single-shot completion (per-slot task)
285 * can land while a train is active and must take the classic path below.
286 * Within the family, win_prefetch says whether this is a read-ahead. */
287 if (task == ctx->rd.read_aio_task) {
288 if (ctx->rd.win_prefetch) {
289 brix_read_window_prefetch_done(t, ctx, c);
290 } else if (ctx->rd.win_active) {
291 brix_read_aio_window_done(t, ctx, c);
292 } else {
293 brix_aio_resume(c); /* stale windowed completion — train gone */
294 }
295 return;
296 }
297 brix_aio_metric_done(t->start_ns, BRIX_METRIC_OP_READ);
298 if (t->nread < 0) {
299 brix_read_aio_failed(t, ctx, c);
300 return;
301 }
302 brix_read_aio_deliver(t, ctx, c);
303}
