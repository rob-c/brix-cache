18brix_plat_event_init(void)
19{
20 int kq = kqueue();
21
22 if (kq < 0) {
23 return -1; /* errno set by kqueue() */

25
26 /* Set CLOEXEC to avoid fd leak on exec() */

43int
44brix_plat_event_watch(int event_fd, int fd, uint32_t events)
45{
46 struct kevent ev;
47 int filter = 0;
48 int filter_flags = EV_ADD | EV_ENABLE | EV_CLEAR;
49
50 /* Translate event mask to kqueue filter */
51 if (events & BRIX_EVENT_READ) {
52 filter = EVFILT_READ;
53 } else if (events & BRIX_EVENT_WRITE) {
54 filter = EVFILT_WRITE;
55 } else if (events & (BRIX_EVENT_DELETE | BRIX_EVENT_MODIFY | BRIX_EVENT_CREATE)) {
56 /* Filesystem events use EVFILT_VNODE */
57 filter = EVFILT_VNODE;
58
59 if (events & BRIX_EVENT_DELETE) {
60 filter_flags |= NOTE_DELETE;
61 }
62 if (events & BRIX_EVENT_MODIFY) {
63 filter_flags |= NOTE_WRITE;
64 }
65 if (events & BRIX_EVENT_CREATE) {
66 filter_flags |= NOTE_EXTEND;
67 }
68 } else {
69 errno = EINVAL;
70 return -1;
71 }
72
73 EV_SET(&ev, (uintptr_t)fd, filter, filter_flags, 0, 0, NULL);
74
75 if (kevent(event_fd, &ev, 1, NULL, 0, NULL) < 0) {
76 return -1; /* errno set by kevent() */
77 }
78
79 return 0;
80}

83brix_plat_event_wait(int event_fd, void *events, int max_events, int timeout_ms)
84{
85 struct kevent *kev = (struct kevent *)events;
86 struct timespec ts;
87 struct timespec *tsp;
88 int nready;
89
90 if (timeout_ms < 0) {
91 tsp = NULL; /* Infinite timeout */
92 } else {
93 ts.tv_sec = timeout_ms / 1000;
94 ts.tv_nsec = (timeout_ms % 1000) * 1000000;
95 tsp = &ts;
96 }
97
98 nready = kevent(event_fd, NULL, 0, kev, (intptr_t)max_events, tsp);
99
100 if (nready < 0) {
101 if (errno == EINTR) {
102 return 0; /* Interrupted by signal - not an error */
103 }
104 return -1; /* errno set by kevent() */
105 }
106
107 return nready;
