33/*
34 * brix_conf_set_mint_ca — setter for "brix_storage_credential_mint_ca <cert>
35 * <key>" (phase-2 T9). Validates both PEM files load-parse at config time
36 * (nginx -t fails loudly on a bad mint CA instead of every mint request
37 * failing at runtime) and stores their paths into the shared preamble's
38 * storage_credential_mint_ca_cert / _key fields. TRUST NOTE: configuring this
39 * directive means the frontend will sign per-user x509 proxies with this CA
40 * key — the ORIGIN must trust this CA for minted credentials to be usable;
41 * see src/fs/backend/cred_mint.h for the full trust-model note.
42 */
43char *
44brix_conf_set_mint_ca(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
45{
46 ngx_http_brix_common_conf_t *c = conf;
47 ngx_str_t *value = cf->args->elts;
48 FILE *f;
49 X509 *cert;
50 EVP_PKEY *key;
51
52 (void) cmd;
53
54 f = fopen((const char *) value[1].data, "r");
55 cert = (f != NULL) ? PEM_read_X509(f, NULL, NULL, NULL) : NULL;
56 if (f != NULL) {
57 (void) fclose(f); /* read-only stream: close failure cannot lose data */
58 }
59 if (cert == NULL) {
60 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
61 "brix_storage_credential_mint_ca: cannot parse CA cert \"%V\"",
62 &value[1]);
63 return NGX_CONF_ERROR;
64 }
65 X509_free(cert);
66
67 f = fopen((const char *) value[2].data, "r");
68 key = (f != NULL) ? PEM_read_PrivateKey(f, NULL, NULL, NULL) : NULL;
69 if (f != NULL) {
70 (void) fclose(f); /* read-only stream: close failure cannot lose data */
71 }
72 if (key == NULL) {
73 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
74 "brix_storage_credential_mint_ca: cannot parse CA key \"%V\"",
75 &value[2]);
76 return NGX_CONF_ERROR;
77 }
78 brix_evp_pkey_free(key);
79
80 c->common.storage_credential_mint_ca_cert = value[1];
81 c->common.storage_credential_mint_ca_key = value[2];
82 return NGX_CONF_OK;
83}

85/*
86 * brix_conf_set_peers — setter for "brix_cache_peers <member> <member> ..."
87 * (phase-85 F8 sibling mesh). Each member is "host:port", with this node's own
88 * ring slot written "self=host:port". The tokens are only COLLECTED here (into
89 * an ngx_str_t array on the shared preamble); shape validation — exactly one
90 * self=, ≥2 members, well-formed authorities — runs at tier registration where
91 * the [emerg] can name the export.
92 */
93char *
94brix_conf_set_peers(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
95{
96 ngx_http_brix_common_conf_t *c = conf;
97 ngx_str_t *value = cf->args->elts;
98 ngx_str_t *slot;
99 ngx_uint_t i;
100
101 (void) cmd;
102
103 if (c->common.cache_peers != NULL) {
104 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
105 "brix_cache_peers: duplicate directive — list every ring "
106 "member in one declaration");
107 return NGX_CONF_ERROR;
108 }
109 c->common.cache_peers = ngx_array_create(cf->pool, cf->args->nelts - 1,
110 sizeof(ngx_str_t));
111 if (c->common.cache_peers == NULL) {
112 return NGX_CONF_ERROR;
113 }
114 for (i = 1; i < cf->args->nelts; i++) {
115 slot = ngx_array_push(c->common.cache_peers);
116 if (slot == NULL) {
117 return NGX_CONF_ERROR;
118 }
119 *slot = value[i];
120 }
121 return NGX_CONF_OK;
122}

124/*
125 * brix_conf_set_backend_tx_endpoint — setter for
126 * "brix_backend_token_exchange_endpoint <url>" (P90-70.8 slice of the phase-70
127 * §6 invariant: trust config is validated at LOAD, not first use). The
128 * runtime exchange client (auth/token/exchange.c brix_tx_http_post) pins
129 * libcurl to HTTPS-only — a subject token and the client secret ride every
130 * request — so a non-https endpoint can never succeed; without this check it
131 * would only surface as every EXCHANGE delegation failing (fail-closed deny)
132 * at first use. Reject it at nginx -t time instead: https:// scheme, a
133 * non-empty host, and no whitespace/control bytes (the value is handed to
134 * CURLOPT_URL verbatim).
135 */
136char *
137brix_conf_set_backend_tx_endpoint(ngx_conf_t *cf, ngx_command_t *cmd,
138 void *conf)
139{
140 char *rv;
141 ngx_str_t *ep;
142 ngx_uint_t i;
143
144 rv = ngx_conf_set_str_slot(cf, cmd, conf);
145 if (rv != NGX_CONF_OK) {
146 return rv;
147 }
148 ep = (ngx_str_t *) ((char *) conf + cmd->offset);
149
150 if (ep->len <= sizeof("https://") - 1
151 || ngx_strncasecmp(ep->data, (u_char *) "https://", 8) != 0
152 || ep->data[8] == '/') {
153 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
154 "brix_backend_token_exchange_endpoint: \"%V\" is not an https:// "
155 "URL with a host — the exchange client is HTTPS-only (a subject "
156 "token and the client secret ride every request), so this "
157 "endpoint could never be reached", ep);
158 return NGX_CONF_ERROR;
159 }
160 for (i = 0; i < ep->len; i++) {
161 if (ep->data[i] <= ' ' || ep->data[i] == 0x7f) {
162 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
163 "brix_backend_token_exchange_endpoint: whitespace or control "
164 "byte at offset %ui", i);
165 return NGX_CONF_ERROR;
166 }
167 }
168 return NGX_CONF_OK;
169}
