1/* brixoci_gc.c — `brixoci gc` (phase-104 D15.3): reclaim what a delete left.
2 *
3 * brixoci gc <store-dir> [--grace SECS] [--dry-run] [--json]
4 *
5 * WHAT: the offline face of the registry GC pass — options in, one pass over
6 * an on-disk `brix_oci_registry_root`, a report out.
7 * WHY: the request handlers deliberately never delete a blob (see
8 * shared/oci/gc.h for why that is the only correct thing for a handler
9 * to do). Answering the whole-store question offline is what this
10 * subcommand is for, and running it by hand is what an operator wants
11 * when the answer must be inspected before it is acted on — which is
12 * what `--dry-run` and `--json` are for.
13 * HOW: the pass itself is shared/oci/gc.c, because the registry's own
14 * maintenance timer runs exactly the same one (D15.5). Everything in
15 * this file is argument handling and presentation: a divergence
16 * between what the tool sweeps and what the server sweeps would be a
17 * divergence nobody could see from either side.
18 */

43 brix_json_kv_ll(stdout, "layer_marks_dropped", (long long) s->marks, 1);
44 brix_json_kv_ll(stdout, "referrers_dropped", (long long) s->refs, 1);

50static void
51gc_report_text(const brix_oci_gc_t *c)
52{
53 const brix_oci_gc_stats_t *s = &c->st;
54 const char *verb = c->dry_run ? "would sweep" : "swept";
55
56 printf("scanned %lu repositories, %lu manifests\n", s->repos,
57 s->manifests);
58 printf("%s %lu blobs (%llu bytes), kept %lu live, %lu within grace\n",
59 verb, s->blobs_swept, s->bytes, s->blobs_live, s->blobs_young);
60 printf("%s %lu stale layer marks, %lu dangling referrer descriptors\n",
61 verb, s->marks, s->refs);
62}

65int
66brixoci_gc_run(const brixoci_opts_t *o, const char **pos, int npos, char *err,
67 size_t errlen)
68{
69 brix_oci_gc_t c;
70 int rc;
71
72 if (npos != 1) {
73 snprintf(err, errlen, "gc takes exactly one store directory");
74 return BRIXOCI_EUSAGE;
75 }
76 memset(&c, 0, sizeof(c));
77 c.root = pos[0];
78 c.grace = o->grace;
79 c.dry_run = o->dry_run;
80
81 rc = brix_oci_gc_run(&c, err, errlen);
82 if (rc == BRIX_OCI_GC_EROOT) {
83 return BRIXOCI_EUSAGE;
84 }
85 if (rc != BRIX_OCI_GC_OK) {
86 return BRIX_OCI_REG_ETRANSPORT;
87 }
88 if (o->json) {
89 gc_report_json(&c);
90 } else {
91 gc_report_text(&c);
92 }
93 return BRIX_OCI_REG_OK;
94}
