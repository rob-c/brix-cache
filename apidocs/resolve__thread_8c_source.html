1/*
2 * resolve_thread.c — the one libc resolver call, and the blocking entry
3 * point thread-pool code uses (phase-116 W4 + Appendix A).
4 *
5 * WHAT: dns_libc_resolve() is the ONLY getaddrinfo() in src/ (guard
6 * tools/ci/check_dns_seam.py). Two callers use it:
7 * - the thread-pool backend of brix_dns_resolve() (no usable nginx
8 * resolver in scope): a task posted to the "default" pool whose
9 * body resolves and whose completion finishes the request on the
10 * event loop;
11 * - brix_dns_resolve_sync(): the blocking flavour for thread-pool
12 * code (net_target pin, TPC/cache-origin/gsiftp connects, cvmfs
13 * probes, curl pinning), which tries an IP literal, the per-worker
14 * cache, the loop's nginx resolver via the bridge, and only then
15 * libc — so with `brix_resolver auto` even blocking callers follow
16 * the resolv.conf the operator pointed brix at.
17 * WHY: I-DNS-4: no getaddrinfo on the event loop. Where the operator has
18 * not enabled brix_resolver, libc is still the right authority — it
19 * reads resolv.conf, nsswitch and /etc/hosts — it just may not block a
20 * worker. Confining the call to one file makes the seam auditable.
21 * HOW: The thread task owns its allocation (a private pool destroyed by the
22 * completion) so a request needs no pool and can be cancelled: cancel
23 * clears tc->req and the completion finds nothing to finish. The
24 * thread body reads only the copies in its ctx, never the request.
25 * libc answers carry no TTL, so they are cached for BRIX_DNS_THREAD_TTL
26 * seconds; only a definitive "no such name" is negative-cached.
27 */

48 * negative-cache; EAI_AGAIN is the libc spelling of a timed-out query. */
49static ngx_uint_t
50dns_libc_result(int rc)
51{

54 || rc == EAI_NODATA

64/* The libc resolver — the one getaddrinfo() call in src/ (the DNS seam).
65 * Answers carry `port`; *result is the BRIX_DNS_RESULT_* class of the
66 * outcome, and every call is counted once in brix_dns_lookups_total. */
67static ngx_uint_t
68dns_libc_resolve(const char *host, in_port_t port, brix_af_policy_t af,
69 int socktype, brix_dns_addr_t *addrs, ngx_uint_t max, char *err,
70 size_t errsz, ngx_uint_t *result)
71{
72 struct addrinfo hints, *res, *ai;
73 char portstr[8];
74 ngx_uint_t n = 0;
75 int rc;
76
77 ngx_memzero(&hints, sizeof(hints));
78 hints.ai_family = (int) af;
79 hints.ai_socktype = socktype ? socktype : SOCK_STREAM;
80 (void) ngx_snprintf((u_char *) portstr, sizeof(portstr), "%d%Z",
81 (int) port);
82
83 rc = getaddrinfo(host, portstr, &hints, &res);
84 if (rc != 0 || res == NULL) {
85 if (err != NULL && errsz > 0) {
86 (void) ngx_snprintf((u_char *) err, errsz, "%s%Z",
87 rc == EAI_SYSTEM ? strerror(errno)
88 : gai_strerror(rc));
89 }
90 *result = dns_libc_result(rc);
91 brix_dns_lookup_note(*result);
92 return 0;
93 }
94 for (ai = res; ai != NULL && n < max; ai = ai->ai_next) {
95 if (ai->ai_addrlen > sizeof(addrs[n].ss)) {
96 continue;
97 }
98 ngx_memcpy(&addrs[n].ss, ai->ai_addr, ai->ai_addrlen);
99 addrs[n].len = ai->ai_addrlen;
100 n++;
101 }
102 freeaddrinfo(res);
103 if (n == 0 && err != NULL && errsz > 0) {
104 (void) ngx_snprintf((u_char *) err, errsz, "no usable address%Z");
105 }
106 *result = n > 0 ? BRIX_DNS_RESULT_OK : BRIX_DNS_RESULT_ERROR;
107 brix_dns_lookup_note(*result);
108 return n;
109}

129static ngx_uint_t
130dns_sync_copy(const brix_dns_req_t *req, brix_dns_addr_t *addrs,
131 ngx_uint_t max, char *err, size_t errsz)
132{
133 ngx_uint_t i, n = ngx_min(req->naddrs, max);
134
135 if (req->rc != NGX_OK || n == 0) {
136 if (err != NULL && errsz > 0) {
137 (void) ngx_snprintf((u_char *) err, errsz, "%s%Z",
138 req->error ? req->error : "no usable address");
139 }
140 return 0;
141 }
142 for (i = 0; i < n; i++) {
143 addrs[i] = req->addrs[i];
144 }
145 return n;
146}

149static void
150dns_sync_cache_libc(brix_dns_req_t *req, const brix_dns_addr_t *addrs,
151 ngx_uint_t n, unsigned definitive)
152{
153 ngx_uint_t i;
154
155 if (n > 0) {
156 req->rc = NGX_OK;
157 req->naddrs = ngx_min(n, (ngx_uint_t) BRIX_DNS_MAX_ADDRS);
158 for (i = 0; i < req->naddrs; i++) {
159 req->addrs[i] = addrs[i];
160 }
161 req->ttl = BRIX_DNS_THREAD_TTL;
162 if (req->policy != NULL && req->ttl > req->policy->max_ttl) {
163 req->ttl = req->policy->max_ttl;
164 }
165 } else {
166 req->rc = NGX_ERROR;
167 req->naddrs = 0;
168 req->ttl = definitive
169 ? (req->policy ? req->policy->negative_ttl
170 : BRIX_DNS_NEG_TTL_DEFAULT)
171 : 0;
172 }
173 brix_dns_cache_store(req);
174}

177/* IP literal fast path: 1 = one answer stored (carrying `port`), 0 = the
178 * literal's family is excluded by policy (*err set), NGX_DECLINED = not a
179 * literal at all. */
180static ngx_int_t
181dns_sync_literal(const ngx_str_t *name, in_port_t port, brix_af_policy_t af,
182 brix_dns_addr_t *addrs, char *err, size_t errsz)
183{
184 brix_dns_addr_t lit;
185
186 if (brix_dns_parse_literal(name, &lit) != NGX_OK) {
187 return NGX_DECLINED;
188 }
189 if ((af == BRIX_AF_INET && lit.ss.ss_family != AF_INET)
190 || (af == BRIX_AF_INET6 && lit.ss.ss_family != AF_INET6))
191 {
192 brix_dns_set_err(err, errsz, "address family excluded by policy");
193 return 0;
194 }
195 ngx_inet_set_port((struct sockaddr *) &lit.ss, port);
196 addrs[0] = lit;
197 return 1;
198}

201ngx_uint_t
202brix_dns_resolve_sync(const brix_dns_policy_t *policy, const char *host,
203 in_port_t port, brix_af_policy_t af, int socktype, brix_dns_addr_t *addrs,
204 ngx_uint_t max, char *err, size_t errsz)
205{
206 brix_dns_req_t req;
207 ngx_uint_t n;
208 ngx_int_t rc;
209 ngx_uint_t result;
210
211 brix_dns_set_err(err, errsz, "");
212 if (host == NULL || host[0] == '\0' || max == 0) {
213 brix_dns_set_err(err, errsz, "empty hostname");
214 return 0;
215 }
216 dns_sync_req_init(&req, policy, host, port, af, socktype);
217
218 rc = dns_sync_literal(&req.name, port, af, addrs, err, errsz);
219 if (rc != NGX_DECLINED) {
220 return (ngx_uint_t) rc;
221 }
222
223 if (brix_dns_cache_lookup(&req) == NGX_OK) {
224 return dns_sync_copy(&req, addrs, max, err, errsz);
225 }
226
227 rc = brix_dns_bridge_resolve(policy, host, port, af, req.socktype, addrs,
228 max, &n, err, errsz);
229 if (rc == NGX_OK) {
230 return n;
231 }
232
233 n = dns_libc_resolve(host, port, af, req.socktype, addrs, max, err, errsz,
234 &result);
235 dns_sync_cache_libc(&req, addrs, n, result == BRIX_DNS_RESULT_NXDOMAIN);
236 return n;
237}

259static void
260dns_thread_done(ngx_event_t *ev)
261{
262 ngx_thread_task_t *task = ev->data;
263 dns_thread_ctx_t *tc = task->ctx;
264 brix_dns_req_t *req = tc->req;
265 ngx_pool_t *pool = tc->pool;
266 ngx_uint_t i;
267
268 if (req == NULL) {
269 ngx_destroy_pool(pool); /* cancelled while in flight */
270 return;
271 }
272 req->task = NULL;
273 if (tc->naddrs == 0) {
274 req->rc = NGX_ERROR;
275 req->error = "getaddrinfo failed";
276 req->naddrs = 0;
277 ngx_log_error(NGX_LOG_INFO, req->log, 0,
278 "brix dns: \"%V\" via libc: %s", &req->name, tc->err);
279 req->ttl = 0;
280 if (tc->definitive) {
281 req->ttl = req->policy ? req->policy->negative_ttl
282 : BRIX_DNS_NEG_TTL_DEFAULT;
283 }
284 } else {
285 req->naddrs = tc->naddrs;
286 for (i = 0; i < tc->naddrs; i++) {
287 req->addrs[i] = tc->addrs[i];
288 }
289 req->rc = NGX_OK;
290 req->error = NULL;
291 req->ttl = BRIX_DNS_THREAD_TTL;
292 if (req->policy != NULL && req->ttl > req->policy->max_ttl) {
293 req->ttl = req->policy->max_ttl;
294 }
295 }
296 /* the handler may release the request; the ctx is ours until here */
297 brix_dns_req_finish(req);
298 ngx_destroy_pool(pool);
299}
