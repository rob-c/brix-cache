1/* origin_probe.c — per-worker RTT probe for brix_cvmfs_origin_select rtt.
2 *
3 * WHAT: a repeating per-worker timer posts a thread-pool task that measures
4 * TCP connect RTT to every configured origin endpoint; the event-loop
5 * completion folds samples into an EWMA, ranks endpoints, and pushes
6 * ranks into the sd_http driver.
7 * WHY: on a Tier-2 with erratic routing, configured order and geography
8 * both lie; measured connect RTT is what the fills will actually feel.
9 * HOW: measurement is blocking (brix DNS resolve + nonblocking connect + poll)
10 * and therefore lives on a thread-pool worker, mirroring the cache
11 * fill I/O pattern; the event loop only re-arms the timer and writes
12 * ranks (relaxed atomics — see sd_http.h). Exports register at config
13 * time (merge runs in the master; the list survives the fork), and
14 * every worker arms its own timers at init_process — the same
15 * lifecycle the cache reaper uses.
16 */

63void
64brix_cvmfs_rtt_register(const char *root_canon, time_t interval,
65 const ngx_str_t *pool_name, const brix_dns_policy_t *dns)
66{
67 ngx_uint_t i;
68 cvmfs_rtt_reg_t *reg = NULL;
69
70 for (i = 0; i < cvmfs_rtt_regs_n; i++) {
71 if (ngx_strcmp(cvmfs_rtt_regs[i].root, root_canon) == 0) {
72 reg = &cvmfs_rtt_regs[i]; /* reload: update in place */
73 break;
74 }
75 }
76 if (reg == NULL) {
77 if (cvmfs_rtt_regs_n >= CVMFS_PROBE_MAX_EXPORTS
78 || ngx_strlen(root_canon) >= sizeof(reg->root))
79 {
80 return;
81 }
82 reg = &cvmfs_rtt_regs[cvmfs_rtt_regs_n++];
83 }
84 ngx_cpystrn((u_char *) reg->root, (u_char *) root_canon,
85 sizeof(reg->root));
86 reg->interval = (interval > 0) ? interval : 60;
87 reg->dns = dns;
88 reg->pool[0] = '\0';
89 if (pool_name != NULL && pool_name->len > 0
90 && pool_name->len < sizeof(reg->pool))
91 {
92 ngx_memcpy(reg->pool, pool_name->data, pool_name->len);
93 reg->pool[pool_name->len] = '\0';
94 }
95}

116/* One nonblocking connect, timed. Returns RTT µs, or -1 on any failure
117 * (refused, unreachable, timeout, resolution failure). Shared with the
118 * on-demand geo-answer probe (geo_answer.c) — ONE connect-RTT implementation. */
119long
120brix_cvmfs_connect_rtt_us(const brix_dns_policy_t *dns, const char *host,
121 int port, int timeout_ms)
122{
123 brix_dns_addr_t addr;
124 struct pollfd pfd;
125 struct timespec t0, t1;
126 char reason[BRIX_DNS_ERROR_LEN];
127 int fd, soerr = 0;
128 socklen_t slen = sizeof(soerr);
129 long us = -1;
130
131 /* phase-116: the export's resolver policy, first answer only — the probe
132 * times one connect, it does not walk candidates */
133 if (brix_dns_resolve_sync(dns, host, (in_port_t) port, BRIX_AF_AUTO,
134 SOCK_STREAM, &addr, 1, reason, sizeof(reason))
135 == 0)
136 {
137 return -1;
138 }
139 fd = socket(addr.ss.ss_family, SOCK_STREAM | SOCK_NONBLOCK, 0); /* vfs-seam-allow: NOT_STORAGE — probe socket, non-export resource */
140 if (fd < 0) {
141 return -1;
142 }
143 clock_gettime(CLOCK_MONOTONIC, &t0);
144 if (connect(fd, (struct sockaddr *) &addr.ss, addr.len) == 0) {
145 clock_gettime(CLOCK_MONOTONIC, &t1);
146 us = (t1.tv_sec - t0.tv_sec) * 1000000L
147 + (t1.tv_nsec - t0.tv_nsec) / 1000L;
148 } else if (errno == EINPROGRESS) {
149 pfd.fd = fd;
150 pfd.events = POLLOUT;
151 if (poll(&pfd, 1, timeout_ms) == 1
152 && getsockopt(fd, SOL_SOCKET, SO_ERROR, &soerr, &slen) == 0
153 && soerr == 0)
154 {
155 clock_gettime(CLOCK_MONOTONIC, &t1);
156 us = (t1.tv_sec - t0.tv_sec) * 1000000L
157 + (t1.tv_nsec - t0.tv_nsec) / 1000L;
158 }
159 }
160 close(fd); /* vfs-seam-allow: NOT_STORAGE — probe socket, non-export resource */
161 return us;
162}

192/* Render the per-endpoint probe outcome as one bounded "; "-joined line:
193 * "host:port rank=N ewma=Xus rtt=Y" — the full evidence trail behind a
194 * ranking decision, so an operator can see WHY endpoint order changed. */
195static void
196cvmfs_probe_table(const cvmfs_probe_ctx_t *pc, const double *metric,
197 const int *ranks, char *buf, size_t cap)
198{
199 size_t off = 0;
200 char samp[32];
201 int i, w;
202
203 buf[0] = '\0';
204 for (i = 0; i < pc->n && off + 1 < cap; i++) {
205 if (pc->sample_us[i] < 0) {
206 snprintf(samp, sizeof(samp), "UNREACHABLE");
207 } else {
208 snprintf(samp, sizeof(samp), "%ldus", pc->sample_us[i]);
209 }
210 w = snprintf(buf + off, cap - off, "%s%s:%d rank=%d ewma=%.0fus rtt=%s",
211 (i > 0) ? "; " : "", pc->hosts[i], pc->ports[i],
212 ranks[i], metric[i], samp);
213 if (w < 0 || (size_t) w >= cap - off) {
214 break;
215 }
216 off += (size_t) w;
217 }
218}

220/* Log per-endpoint reachability TRANSITIONS (probe ok → fail and back) at
221 * WARN/NOTICE — steady state stays quiet, the flap is what matters. */
222static void
223cvmfs_probe_log_reachability(cvmfs_probe_ctx_t *pc, ngx_log_t *log)
224{
225 int i, failed;
226
227 for (i = 0; i < pc->n; i++) {
228 failed = (pc->sample_us[i] < 0);
229 if (failed == pc->probe_down[i]) {
230 continue;
231 }
232 pc->probe_down[i] = failed;
233 if (failed) {
234 ngx_log_error(NGX_LOG_WARN, log, 0,
235 "cvmfs rtt probe: origin %s:%d UNREACHABLE (connect probe "
236 "failed/timed out within %dms) - penalised in ranking at "
237 "%.0fus ewma until it answers again",
238 pc->hosts[i], pc->ports[i], CVMFS_PROBE_TIMEOUT_MS,
239 pc->ewma_us[i]);
240 } else {
241 ngx_log_error(NGX_LOG_NOTICE, log, 0,
242 "cvmfs rtt probe: origin %s:%d reachable again "
243 "(rtt %ldus, ewma %.0fus - preference recovers as the "
244 "ewma decays)",
245 pc->hosts[i], pc->ports[i], pc->sample_us[i],
246 pc->ewma_us[i]);
247 }
248 }
249}

261/* event-loop side: fold EWMA, rank, push, re-arm. Every rank ORDER change is
262 * logged with the full evidence table (NOTICE — this is the moment the next
263 * fill starts going somewhere else); steady-state ticks stay at INFO. */
264static void
265cvmfs_probe_done(ngx_event_t *ev)
266{
267 ngx_thread_task_t *task = ev->data;
268 cvmfs_probe_ctx_t *pc = task->ctx;
269 brix_sd_instance_t *inst;
270 double metric[SD_HTTP_EP_MAX] = { 0.0 };
271 char tbl[1024];
272 int ranks[SD_HTTP_EP_MAX] = { 0 };
273 int i, best;
274 int first_tick, changed, prev_best;
275
276 for (i = 0; i < pc->n; i++) {
277 double s = (pc->sample_us[i] < 0) ? (double) CVMFS_PROBE_FAIL_US
278 : (double) pc->sample_us[i];
279 pc->ewma_us[i] = (pc->ewma_us[i] == 0.0)
280 ? s : pc->ewma_us[i] * 0.75 + s * 0.25;
281 metric[i] = pc->ewma_us[i];
282 }
283 brix_cvmfs_rank_by_metric(metric, pc->n, ranks);
284
285 cvmfs_probe_log_reachability(pc, ev->log);
286
287 inst = cvmfs_unwrap_http(
288 brix_vfs_backend_resolve(pc->reg->root, ev->log));
289 if (inst != NULL) {
290 sd_http_set_ranks(inst, ranks, pc->n);
291 }
292
293 first_tick = (pc->prev_ranks[0] < 0);
294 changed = 0;
295 for (i = 0; i < pc->n && !first_tick; i++) {
296 if (ranks[i] != pc->prev_ranks[i]) {
297 changed = 1;
298 }
299 }
300 best = cvmfs_rank_zero_index(ranks, pc->n);
301 cvmfs_probe_table(pc, metric, ranks, tbl, sizeof(tbl));
302
303 if (first_tick) {
304 ngx_log_error(NGX_LOG_NOTICE, ev->log, 0,
305 "cvmfs rtt initial ranking (export %s): preferred %s:%d - %s",
306 pc->reg->root, pc->hosts[best], pc->ports[best], tbl);
307 } else if (changed) {
308 prev_best = cvmfs_rank_zero_index(pc->prev_ranks, pc->n);
309 ngx_log_error(NGX_LOG_NOTICE, ev->log, 0,
310 "cvmfs rtt ranking CHANGED (export %s): preferred %s:%d -> %s:%d "
311 "- %s",
312 pc->reg->root, pc->hosts[prev_best], pc->ports[prev_best],
313 pc->hosts[best], pc->ports[best], tbl);
314 } else {
315 ngx_log_error(NGX_LOG_INFO, ev->log, 0,
316 "cvmfs rtt ranks (export %s): preferred %s:%d - %s",
317 pc->reg->root, pc->hosts[best], pc->ports[best], tbl);
318 }
319 for (i = 0; i < pc->n; i++) {
320 pc->prev_ranks[i] = ranks[i];
321 }
322
323 if (!ngx_exiting) {
324 ngx_add_timer(&pc->timer, (ngx_msec_t) pc->reg->interval * 1000
325 + (ngx_msec_t) (ngx_random() % 1000));
326 }
327}

329/* timer side: post the probe task (skip the tick if no pool resolves) */
330static void
331cvmfs_probe_fire(ngx_event_t *ev)
332{
333 cvmfs_probe_ctx_t *pc = ev->data;
334 ngx_thread_pool_t *pool;
335 ngx_str_t pname;
336
337 if (pc->reg->pool[0] != '\0') {
338 pname.data = (u_char *) pc->reg->pool;
339 pname.len = ngx_strlen(pc->reg->pool);
340 } else {
341 ngx_str_set(&pname, "default");
342 }
343 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, &pname);
344 if (pool == NULL || ngx_thread_task_post(pool, pc->task) != NGX_OK) {
345 if (!ngx_exiting) {
346 ngx_add_timer(ev, (ngx_msec_t) pc->reg->interval * 1000);
347 }
348 }
349 /* re-armed by cvmfs_probe_done after the samples land */
350}

353brix_cvmfs_rtt_init_worker(ngx_cycle_t *cycle)
354{
355 ngx_uint_t i;
356
357 for (i = 0; i < cvmfs_rtt_regs_n; i++) {
358 const cvmfs_rtt_reg_t *reg = &cvmfs_rtt_regs[i];
359 ngx_thread_task_t *task;
360 cvmfs_probe_ctx_t *pc;
361 const char *host;
362 int port, idx;
363
364 task = ngx_thread_task_alloc(cycle->pool, sizeof(cvmfs_probe_ctx_t));
365 if (task == NULL) {
366 return NGX_ERROR;
367 }
368 pc = task->ctx;
369 pc->task = task;
370 pc->reg = reg;
371 for (idx = 0; idx < SD_HTTP_EP_MAX; idx++) {
372 if (brix_vfs_backend_http_endpoint_at(reg->root, idx, &host,
373 &port) != 0)
374 {
375 break;
376 }
377 ngx_cpystrn((u_char *) pc->hosts[idx], (u_char *) host,
378 sizeof(pc->hosts[idx]));
379 pc->ports[idx] = port;
380 }
381 pc->n = idx;
382 if (pc->n == 0) {
383 continue; /* not an http backend (misconfig) */
384 }
385 for (idx = 0; idx < SD_HTTP_EP_MAX; idx++) {
386 pc->prev_ranks[idx] = -1; /* -1 = no ranking pushed yet */
387 }
388
389 brix_task_bind(task, cvmfs_probe_thread, cvmfs_probe_done);
390 task->event.log = cycle->log;
391
392 pc->timer.handler = cvmfs_probe_fire;
393 pc->timer.data = pc;
394 pc->timer.log = cycle->log;
395 /* first probe within 500 ms so ranks exist before the first fill */
396 ngx_add_timer(&pc->timer, (ngx_msec_t) (ngx_random() % 500) + 1);
397 }
398 return NGX_OK;
