83/* Build a "<cmd> [operand] [dir]" verb line from a flat JSON command object.
84 * The operand is the first present of args/pct/at/value/ms/bytes/kbps (a number
85 * is copied verbatim so a large byte count never becomes %g scientific
86 * notation). Returns 0 on success, -1 if "cmd" is absent/malformed. */
87static int
88json_build_verb(const char *json, char *out, size_t outsz)
89{
90 char cmd[64] = "", dir[16] = "", operand[1024] = "";
91 if (fp_json_get(json, "cmd", cmd, sizeof cmd) != 1 || cmd[0] == '\0') {
92 return -1;
93 }
94 static const char *const OPKEYS[] = {
95 "args", "pct", "at", "value", "ms", "bytes", "kbps", NULL };
96 for (int i = 0; OPKEYS[i] != NULL; i++) {
97 if (fp_json_get(json, OPKEYS[i], operand, sizeof operand) == 1) {
98 break;
99 }
100 operand[0] = '\0';
101 }
102 (void) fp_json_get(json, "dir", dir, sizeof dir);
103 int n = snprintf(out, outsz, "%s%s%s%s%s", cmd,
104 operand[0] ? " " : "", operand,
105 dir[0] ? " " : "", dir);
106 return (n < 0 || (size_t) n >= outsz) ? -1 : 0;
107}

109/* JSON control input, two coexisting dialects over the ONE verb grammar. The
110 * legacy dialect is marked by an "args" string; everything else is structured:
111 * * legacy ({"cmd":..,"args":".."}): reprojected onto the verb line and
112 * answered with the verb's own plain reply ("ok" / "err: ...").
113 * * structured ({"cmd":..[,"pct"/"at"/"value"/"dir"/"format":..]}): answered
114 * with a JSON result {"ok":true|false,...}; a status query returns the typed
115 * snapshot (which also carries a "text" line for substring readers).
116 * A command is never defined twice — both dialects run through apply_command. */
117void
118apply_json_command(const char *json, char *reply, size_t rsz)
119{
120 char tmp[8];
121 int has_args = fp_json_get(json, "args", tmp, sizeof tmp) == 1;
122 char cmd[64] = "";
123 int have_cmd = fp_json_get(json, "cmd", cmd, sizeof cmd) == 1 && cmd[0] != '\0';
124
125 if (!has_args && have_cmd) {
126 if (strcmp(cmd, "status") == 0) {
127 json_status_typed(reply, rsz);
128 return;
129 }
130 char verbline[2000];
131 if (json_build_verb(json, verbline, sizeof verbline) != 0) {
132 fp_reply(reply, rsz, "{\"ok\":false,\"error\":\"parse\"}\n");
133 return;
134 }
135 char scratch[2048] = "";
136 apply_command(verbline, scratch, sizeof scratch);
137 if (strncmp(scratch, "err", 3) == 0) {
138 fp_reply(reply, rsz, "{\"ok\":false,\"error\":\"unknown command\"}\n");
139 } else {
140 fp_reply(reply, rsz, "{\"ok\":true}\n");
141 }
142 return;
143 }
144 /* Legacy string dialect: reproject to a verb line and run it verbatim. */
145 char verbline[2000];
146 if (fp_json_to_verb(json, verbline, sizeof verbline) == 0) {
147 apply_command(verbline, reply, rsz);
148 return;
149 }
150 /* Reprojection failed. A CLOSED object that names cmd/args (bad value or
151 * missing cmd) is a legacy command error; an UNCLOSED / unreadable object is
152 * a JSON parse error. */
153 if (strchr(json, '}') != NULL &&
154 (strstr(json, "\"cmd\"") != NULL || strstr(json, "\"args\"") != NULL)) {
155 fp_reply(reply, rsz, "err: bad json command\n");
156 } else {
157 fp_reply(reply, rsz, "{\"ok\":false,\"error\":\"parse\"}\n");
158 }
159}
