1/*
2 * sd_s3.h — shared S3 object-store storage driver (read + write/MPU paths).
3 *
4 * WHAT: The S3 storage logic — SigV4 signing (shared kernels), HEAD/Range-GET,
5 * and single-PUT + multipart upload (sd_s3_write.c) — living once in
6 * src/fs/backend/, used by the userland clients (and any future server
7 * consumer) through an injected HTTP transport (sd_s3_transport.h).
8 * WHY: S3 was previously a client-only backend welded to client/lib's HTTP
9 * stack. The transport injection breaks that coupling so the protocol logic
10 * is shared, ngx-free, in libxrdproto.
11 * HOW: Endpoint + credentials + transport are bound at open into an opaque
12 * handle; ops build signed requests and run them through the transport. The
13 * error model is a neutral (int rc, errbuf) pair — no xrdc_status / ngx.
14 */

107/* ---- object metadata (x-amz-meta-*) ----------------------------------- *
108 * S3 has no in-place metadata mutation: get is a HEAD that reads one user-meta
109 * header; set is a copy-onto-self with x-amz-metadata-directive: REPLACE. The
110 * advisory unix-attr blob (mode/uid/gid/mtime) rides in the reserved key
111 * x-amz-meta-<BRIX_META_ADVISORY_S3META>. Attribute names are lowercased to
112 * match the S3 user-metadata contract. */
113
114/* Destination buffer for sd_s3_get_meta: the value lands NUL-terminated in
115 * buf[0..cap). */
116typedef struct { char *buf; size_t cap; } sd_s3_meta_buf;
117
118/* Read user-metadata header x-amz-meta-<name> via a signed HEAD into out
119 * (NUL-terminated). Returns the value length (0 when the attribute is absent),
120 * or -1 with errbuf on a transport/HTTP error. */
121ssize_t sd_s3_get_meta(sd_s3_file *f, const char *name,
122 const sd_s3_meta_buf *out, char *errbuf, size_t errcap);
123
124/* Read ONE checksum-bearing response header off a signed HEAD that asks the
125 * origin to return the checksums it stored at upload time
126 * (`x-amz-checksum-mode: ENABLED`). `hdr_name` is the header to extract — an
127 * `x-amz-checksum-<algo>` (base64, per the S3 additional-checksum feature) or
128 * `ETag` (hex, and an md5 of the object ONLY for a single-part upload). The
129 * value lands NUL-terminated in out->buf; the ENCODING is the caller's to
130 * interpret, so this layer stays free of any digest grammar. Returns the value
131 * length, 0 when the origin sent no such header, or -1 with errbuf. */
132ssize_t sd_s3_get_checksum(sd_s3_file *f, const char *hdr_name,
133 const sd_s3_meta_buf *out, char *errbuf,
134 size_t errcap);

136/* Enumerate every user-metadata name on the object (signed HEAD + raw header
137 * walk) into buf as a listxattr(2)-style NUL-separated block of "user.<name>"
138 * entries; the reserved advisory key (BRIX_META_ADVISORY_S3META) is omitted —
139 * it surfaces as POSIX attrs, not as a user xattr. cap==0 probes the required
140 * size. Returns the byte count, or -1 with errbuf + errno (ERANGE when buf is
141 * too small; ENOTSUP when the transport cannot enumerate headers). */
142ssize_t sd_s3_list_meta(sd_s3_file *f, char *buf, size_t cap,

163/* ---- object listing (ListObjectsV2, delimited) ----------------------------
164 * Enumerate ONE page of the bucket under `prefix` with delimiter '/', so the
165 * result is a single directory level: <Contents> keys directly under the prefix
166 * (files) and <CommonPrefixes> (sub-directories). `p->key` MUST address the
167 * bucket root ("/bucket/") — that is the canonical URI the request is signed
168 * against; `prefix` (may be "") and the continuation token ride in the query
169 * string. `cb` fires once per entry with the entry BASENAME (a sub-directory
170 * carries no trailing slash) and `is_dir` distinguishing the two; the
171 * directory-marker object (Key == prefix) is skipped. Returning non-zero from
172 * `cb` stops enumeration of the current page early (reported as success). On
173 * success *truncated says whether more pages remain and cont_out receives the
174 * NextContinuationToken (empty when not truncated) to feed the next call's
175 * cont_in. 0 / -1 (errbuf + errno mapped from the HTTP status). */
176typedef int (*sd_s3_list_cb)(void *ud, const char *name, int is_dir);
177int sd_s3_list_page(const sd_s3_open_params *p, const char *prefix,
178 const char *cont_in, sd_s3_list_cb cb, void *ud,
179 int *truncated, char *cont_out, size_t cont_cap,
180 char *errbuf, size_t errcap);

182/* ---- flat object listing (ListObjectsV2, NO delimiter) --------------------
183 * The same request without `delimiter`, so one page reports up to 1000 keys
184 * under `prefix` at ANY depth — the shape driver->enumerate wants. `cb` fires
185 * once per <Contents> entry with the FULL key (not a basename; there are no
186 * <CommonPrefixes> in an undelimited listing) plus the size and mtime S3
187 * reports in the listing itself, so a stat-bearing enumeration costs no extra
188 * request. mtime is 0 when <LastModified> is absent or unparsable — never a
189 * guessed epoch. Directory-marker objects (a key ending in '/') ARE reported;
190 * they are real stored objects, and it is the caller's namespace model that
191 * decides whether to skip them. Paging, early stop and the return contract are
192 * identical to sd_s3_list_page. */
193typedef int (*sd_s3_list_flat_cb)(void *ud, const char *key, uint64_t size,
194 time_t mtime);
195int sd_s3_list_flat_page(const sd_s3_open_params *p, const char *prefix,
196 const char *cont_in, sd_s3_list_flat_cb cb, void *ud,
197 int *truncated, char *cont_out, size_t cont_cap,
