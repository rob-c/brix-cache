36/* fork+execvp `argv` with stdout silenced (stderr inherited for diagnostics).
37 * Returns 0 on a clean exit(0), the child's non-zero exit code otherwise, or -1
38 * if the process could not be created. */
39int
40priv_run(char *const argv[])
41{
42 pid_t pid = fork();
43 if (pid < 0) {
44 return -1;
45 }
46 if (pid == 0) {
47 int nul = open("/dev/null", O_WRONLY);
48 if (nul >= 0) {
49 dup2(nul, STDOUT_FILENO);
50 close(nul);
51 }
52 execvp(argv[0], argv);
53 _exit(127);
54 }
55 int st = 0;
56 while (waitpid(pid, &st, 0) < 0 && errno == EINTR) {
57 /* retry */
58 }
59 if (WIFEXITED(st)) {
60 return WEXITSTATUS(st);
61 }
62 return -1;
63}

65/* fork+execvp `argv`, piping `input` to the child's stdin (for `nft -f -`).
66 * Same return convention as priv_run(). */
67int
68priv_run_stdin(char *const argv[], const char *input)
69{
70 int pfd[2];
71 if (pipe(pfd) != 0) {
72 return -1;
73 }
74 pid_t pid = fork();
75 if (pid < 0) {
76 close(pfd[0]);
77 close(pfd[1]);
78 return -1;
79 }
80 if (pid == 0) {
81 dup2(pfd[0], STDIN_FILENO);
82 close(pfd[0]);
83 close(pfd[1]);
84 int nul = open("/dev/null", O_WRONLY);
85 if (nul >= 0) {
86 dup2(nul, STDOUT_FILENO);
87 close(nul);
88 }
89 execvp(argv[0], argv);
90 _exit(127);
91 }
92 close(pfd[0]);
93 size_t len = strlen(input), off = 0;
94 while (off < len) {
95 ssize_t w = write(pfd[1], input + off, len - off);
96 if (w < 0) {
97 if (errno == EINTR) {
98 continue;
99 }
100 break;
101 }
102 off += (size_t) w;
103 }
104 close(pfd[1]);
105 int st = 0;
106 while (waitpid(pid, &st, 0) < 0 && errno == EINTR) {
107 /* retry */
108 }
109 return WIFEXITED(st) ? WEXITSTATUS(st) : -1;
110}

114/* An interface name that both matches a conservative charset and actually exists
115 * under /sys/class/net (so a bad --priv-iface fails loudly, not silently). */
116int
117valid_iface(const char *s)
118{
119 size_t n = strlen(s);
120 if (n == 0 || n >= IFNAMSIZ) {
121 return 0;
122 }
123 for (size_t i = 0; i < n; i++) {
124 char c = s[i];
125 if (!isalnum((unsigned char) c) && c != '.' && c != '_' &&
126 c != '-' && c != '@') {
127 return 0;
128 }
129 }
130 char path[64 + IFNAMSIZ];
131 snprintf(path, sizeof(path), "/sys/class/net/%s", s);
132 struct stat sb;
133 return stat(path, &sb) == 0;
134}
