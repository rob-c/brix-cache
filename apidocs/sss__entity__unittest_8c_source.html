4 * gcc -Wall -Wextra -Werror -I src -o /tmp/sss_entity_ut \

9 * Exit 0 = all checks pass ("all checks passed" on stdout). Pins: the outer

46/* Decrypt out[] back into clear[] and verify the CRC trailer; returns the
47 * cleartext length (without CRC) or 0 on any failure. */
48static size_t
49decrypt(const uint8_t *out, size_t out_len, uint8_t *clear, size_t clear_max)
50{
51 size_t plain_len, clear_len;
52 uint32_t crc;
53
54 if (out_len < BRIX_SSS_HDR_LEN + BRIX_SSS_DATA_HDR_LEN + 4
55 || brix_sss_bf_crypt(0, KEY, sizeof(KEY), out + BRIX_SSS_HDR_LEN,
56 out_len - BRIX_SSS_HDR_LEN, clear, clear_max,
57 &plain_len) != 0
58 || plain_len < BRIX_SSS_DATA_HDR_LEN + 4)
59 {
60 return 0;
61 }
62 clear_len = plain_len - 4;
63 crc = ((uint32_t) clear[clear_len] << 24) | ((uint32_t) clear[clear_len + 1] << 16)
64 | ((uint32_t) clear[clear_len + 2] << 8) | (uint32_t) clear[clear_len + 3];
65 return crc == brix_crc32_ieee(clear, clear_len) ? clear_len : 0;
66}

68/* Find TLV `type` in the cleartext body; returns its value pointer / length. */
69static const uint8_t *
70find_tlv(const uint8_t *clear, size_t clear_len, uint8_t type, size_t *len)
71{
72 const uint8_t *p = clear + BRIX_SSS_DATA_HDR_LEN, *end = clear + clear_len;
73
74 while ((size_t) (end - p) >= 3) {
75 size_t n = ((size_t) p[1] << 8) | (size_t) p[2];
76
77 if ((size_t) (end - p - 3) < n) {
78 return NULL;
79 }
80 if (p[0] == type) {
81 *len = n;
82 return p + 3;
83 }
84 p += 3 + n;
85 }
86 return NULL;
87}

101static void
102test_full_entity(void)
103{
104 static const uint8_t creds[] = { 0x00, 0x01, 0xff, 'X', 0x00, 0x7f };
105 brix_sss_entity_t ent = { "alice", "cms", "production", "/cms,/cms/prod",
106 "endorsed-by=broker", creds, sizeof(creds) };
107 uint8_t out[BRIX_SSS_ENTITY_BLOB_MAX], clear[BRIX_SSS_ENTITY_BLOB_MAX];
108 size_t out_len = 0, clear_len, n = 0;
109 const uint8_t *v;
110
111 check("full: build", brix_sss_build_entity_credential(KEY, sizeof(KEY), KEY_ID,
112 &ent, NONCE, GEN_TIME, out, sizeof(out), &out_len) == 0);
113 check("full: magic", memcmp(out, "sss\0", 4) == 0 && out[4] == 1
114 && out[6] == 0 && out[7] == BRIX_SSS_ENC_BF32);
115 check("full: key id", out[8] == 0x01 && out[11] == 0x04 && out[15] == 0x08);
116 clear_len = decrypt(out, out_len, clear, sizeof(clear));
117 check("full: crc", clear_len > BRIX_SSS_DATA_HDR_LEN);
118 check("full: nonce", memcmp(clear, NONCE, 32) == 0);
119 check("full: gen_time", clear[32] == 0x01 && clear[35] == 0x04);
120 check("full: usedata", clear[39] == BRIX_SSS_OPT_USEDATA);
121 check_str_tlv(clear, clear_len, BRIX_SSS_TYPE_NAME, "alice", "full: name");
122 check_str_tlv(clear, clear_len, BRIX_SSS_TYPE_VORG, "cms", "full: vorg");
123 check_str_tlv(clear, clear_len, BRIX_SSS_TYPE_ROLE, "production", "full: role");
124 check_str_tlv(clear, clear_len, BRIX_SSS_TYPE_GRPS, "/cms,/cms/prod", "full: grps");
125 check_str_tlv(clear, clear_len, BRIX_SSS_TYPE_ENDO, "endorsed-by=broker", "full: endo");
126 v = find_tlv(clear, clear_len, BRIX_SSS_TYPE_CRED, &n);
127 check("full: cred raw", v != NULL && n == sizeof(creds)
128 && memcmp(v, creds, n) == 0);
129 check("full: name first", clear[BRIX_SSS_DATA_HDR_LEN] == BRIX_SSS_TYPE_NAME);
130}

209/* Build a server-style challenge: echoed outer header + BF32(header + TLVs + CRC). */
210static size_t
211make_challenge(const uint8_t *key, size_t key_len, int with_lgid, int corrupt,
212 uint8_t *out, size_t out_max)
213{
214 static const char lgid[] = "nobody";
215 uint8_t plain[128];
216 size_t n = BRIX_SSS_DATA_HDR_LEN, crypt_out = 0;
217 uint32_t crc;
218
219 memset(plain, 0, sizeof(plain));
220 memcpy(plain, NONCE, 32);
221 plain[39] = BRIX_SSS_OPT_USEDATA;
222 if (with_lgid) {
223 plain[n++] = BRIX_SSS_TYPE_LGID;
224 plain[n++] = 0;
225 plain[n++] = (uint8_t) sizeof(lgid);
226 memcpy(plain + n, lgid, sizeof(lgid));
227 n += sizeof(lgid);
228 }
229 crc = brix_crc32_ieee(plain, n) ^ (corrupt ? 1u : 0u);
230 plain[n] = (uint8_t) (crc >> 24); plain[n + 1] = (uint8_t) (crc >> 16);
231 plain[n + 2] = (uint8_t) (crc >> 8); plain[n + 3] = (uint8_t) crc;
232 memset(out, 0, BRIX_SSS_HDR_LEN);
233 memcpy(out, "sss", 4);
234 out[4] = 1;
235 out[7] = BRIX_SSS_ENC_BF32;
236 if (brix_sss_bf_crypt(1, key, key_len, plain, n + 4, out + BRIX_SSS_HDR_LEN,
237 out_max - BRIX_SSS_HDR_LEN, &crypt_out) != 0) {
238 return 0;
239 }
240 return BRIX_SSS_HDR_LEN + crypt_out;
241}

243static void
244test_challenge(void)
245{
246 uint8_t body[256];
247 char lgid[64];
248 size_t n;
249
250 n = make_challenge(KEY, sizeof(KEY), 1, 0, body, sizeof(body));
251 check("chal: good", n > 0 && brix_sss_challenge_lgid(KEY, sizeof(KEY), body, n,
252 lgid, sizeof(lgid)) == 0 && strcmp(lgid, "nobody") == 0);
253 check("chal: wrong key", brix_sss_challenge_lgid(WRONG_KEY, sizeof(WRONG_KEY),
254 body, n, lgid, sizeof(lgid)) != 0);
255 check("chal: truncated", brix_sss_challenge_lgid(KEY, sizeof(KEY), body,
256 n - 8, lgid, sizeof(lgid)) != 0);
257 check("chal: too short", brix_sss_challenge_lgid(KEY, sizeof(KEY), body,
258 BRIX_SSS_HDR_LEN + 8, lgid, sizeof(lgid)) != 0);
259 check("chal: tiny lgid buf", brix_sss_challenge_lgid(KEY, sizeof(KEY), body,
260 n, lgid, 3) == 0 && strcmp(lgid, "no") == 0);
261 body[0] = 'x';
262 check("chal: bad magic", brix_sss_challenge_lgid(KEY, sizeof(KEY), body, n,
263 lgid, sizeof(lgid)) != 0);
264 body[0] = 's';
265 body[7] = 'Z';
266 check("chal: bad enc", brix_sss_challenge_lgid(KEY, sizeof(KEY), body, n,
267 lgid, sizeof(lgid)) != 0);
268
269 n = make_challenge(KEY, sizeof(KEY), 1, 1, body, sizeof(body));
270 check("chal: corrupt crc", n > 0 && brix_sss_challenge_lgid(KEY, sizeof(KEY),
271 body, n, lgid, sizeof(lgid)) != 0);
272 n = make_challenge(KEY, sizeof(KEY), 0, 0, body, sizeof(body));
273 check("chal: no lgid", n > 0 && brix_sss_challenge_lgid(KEY, sizeof(KEY),
274 body, n, lgid, sizeof(lgid)) != 0);
275 check("chal: null args", brix_sss_challenge_lgid(KEY, sizeof(KEY), NULL, n,
276 lgid, sizeof(lgid)) != 0);
277}
