19/* ---- brix_vfs_deleg_sts_cred (call-ready hook, §5.5) -----------------------
20 *
21 * WHAT: Exchange the node's S3 service credential for temporary credentials
22 * scoped to the inbound identity via brix_s3_sts_assume(), and stamp the
23 * result onto cred->s3_ak/s3_sk/s3_region (mode EXCHANGE).
24 *
25 * WHY: An S3 SigV4 secret is never transmitted, so the origin cannot be handed
26 * the caller's key; STS is the EXCHANGE path (§5.5). This helper is the
27 * single seam where the STS result becomes the sd_remote-consumable cred
28 * form, mirroring brix_vfs_deleg_exchange for bearers.
29 *
30 * HOW: Calls brix_s3_sts_assume() with the ctx identity and the caller-supplied
31 * conf; on NGX_OK borrows the pool-copied ak/sk/region onto *cred. On
32 * failure → deny (never the service cred under fallback-deny). Secrets are
33 * never logged.
34 *
35 * DRIVEN (phase-70 §5.5 origin leg closed): both prerequisites are now wired.
36 * (a) The STS conf source exists — brix_backend_s3_sts_access_key/
37 * _secret_key/_region/_ttl (+ the pre-existing _endpoint/_role) are stamped
38 * onto the live-cred bag by brix_proto_deleg_stamp_conf (deleg_wire.c) via
39 * brix_vfs_deleg_set_sts, reachable from every front-door capture site.
40 * (b) sd_remote's cred path threads s3_ak/sk/region/session through to the
41 * origin keys, and sd_s3_sign_ex folds the STS session token into
42 * x-amz-security-token + the SigV4 signature. brix_vfs_deleg_live_cred now
43 * calls this from the no-forwardable-bytes branch when the leaf backend
44 * accepts BRIX_SD_CRED_S3. */
45ngx_int_t
46brix_vfs_deleg_sts_cred(brix_vfs_ctx_t *ctx, const brix_s3_sts_conf_t *cf,
47 brix_sd_cred_t *cred, int *use_cred, int *err_out)
48{
49 ngx_str_t ak = ngx_null_string;
50 ngx_str_t sk = ngx_null_string;
51 ngx_str_t session = ngx_null_string;
52 brix_s3_sts_out_t out = { &ak, &sk, &session };
53
54 *use_cred = 0;
55
56 if (ctx == NULL || cf == NULL) {
57 return brix_vfs_deleg_deny(ctx, use_cred, err_out,
58 BRIX_CRED_FAIL_MISSING);
59 }
60
61 if (brix_s3_sts_assume(ctx->pool, ctx->identity, cf, &out, ctx->log)
62 != NGX_OK) {
63 ngx_log_error(NGX_LOG_ERR, ctx->log, 0,
64 "brix: S3 STS exchange failed - denying (no service-cred fallback "
65 "for EXCHANGE)");
66 return brix_vfs_deleg_deny(ctx, use_cred, err_out,
67 BRIX_CRED_FAIL_EXCHANGE);
68 }
69
70 cred->s3_ak = (const char *) ak.data;
71 cred->s3_sk = (const char *) sk.data;
72 cred->s3_region = (cf->region.len > 0) ? (const char *) cf->region.data
73 : NULL;
74 /* STS temporary credentials are unusable without their session token: it
75 * must ride in x-amz-security-token AND the SigV4 signature on every origin
76 * request (sd_s3_sign_ex folds it in). GetSessionToken/AssumeRole always
77 * return one; an empty session (defensive) leaves it NULL. */
78 cred->s3_session = (session.len > 0) ? (const char *) session.data : NULL;
79 cred->mode = BRIX_CRED_EXCHANGE;
80 *use_cred = 1;
81
82 return NGX_OK;
83}

85/* ---- brix_vfs_deleg_krb5_token (call-ready hook, §5.7) ---------------------
86 *
87 * WHAT: Initiate the first GSSAPI leg to the origin AS the inbound user, using a
88 * forwardable delegated GSS credential, via brix_krb5_deleg_to_origin().
89 *
90 * WHY: krb5 is only backend-usable by GSSAPI forwarding (§5.7); this seam turns
91 * the captured delegated cred + origin service principal into the first-leg
92 * token the origin-auth caller then drives to completion.
93 *
94 * HOW: Guarded by brix_krb5_forward_available() so a build without krb5, or a
95 * request without a forwardable ticket, cleanly reports unavailable (the
96 * caller falls back to SELECT). On success *out_token holds the first-leg
97 * GSS token (pool-copied).
98 *
99 * RETAINED REFERENCE DIALECT — SUPERSEDED, not on the production path. The live
100 * krb5 origin leg is the RAW AP-REQ path (brix_cache_origin_auth_krb5_raw,
101 * driven from origin_protocol_bootstrap.c), because stock XRootD krb5
102 * speaks raw krb5_rd_req, NOT a GSSAPI gss_init_sec_context init-token
103 * exchange (phase-88 UPDATE (iv); phase-92 §5). This GSSAPI-init hook has
104 * zero production callers and is kept only as a reference implementation of
105 * the GSSAPI dialect (compiles, exercised by its own unit). Do NOT read the
106 * absence of a caller as INFRA-BLOCKED — it is deliberately retained-unused. */
107ngx_int_t
108brix_vfs_deleg_krb5_token(brix_vfs_ctx_t *ctx, void *deleg_gss_cred,
109 const char *origin_service_princ, ngx_str_t *out_token)
110{
111 if (ctx == NULL || out_token == NULL) {
112 return NGX_ERROR;
113 }
114
115 if (!brix_krb5_forward_available()) {
116 ngx_log_error(NGX_LOG_INFO, ctx->log, 0,
117 "brix: krb5 credential forwarding unavailable - falling back to "
118 "SELECT");
119 return NGX_ERROR;
120 }
121
122 return brix_krb5_deleg_to_origin(ctx->pool, deleg_gss_cred,
123 origin_service_princ, out_token, ctx->log);
124}
