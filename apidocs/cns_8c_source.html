1/*
2 * cns.c — Composite Cluster Name Space event codec + inventory host (§6). See cns.h.
3 *
4 * The inventory slot logic lives in cns_inventory.c (pure, POD). This file hosts
5 * the backing store: an nginx SHM slab shared across every manager worker when a
6 * `brix_cns_zone` is registered (brix_cns_configure, wired from
7 * postconfiguration), else a lazily-allocated per-worker heap table for the
8 * common single-worker redirector. All apply/stat/count run under the zone's
9 * slab lock (SHM path) so concurrent workers never corrupt a slot.
10 */

23static void
24put_u64(uint8_t *p, uint64_t v)
25{
26 p[0] = (uint8_t) (v >> 56); p[1] = (uint8_t) (v >> 48);
27 p[2] = (uint8_t) (v >> 40); p[3] = (uint8_t) (v >> 32);
28 p[4] = (uint8_t) (v >> 24); p[5] = (uint8_t) (v >> 16);
29 p[6] = (uint8_t) (v >> 8); p[7] = (uint8_t) v;
30}

32static uint64_t
33get_u64(const uint8_t *p)
34{
35 return ((uint64_t) p[0] << 56) | ((uint64_t) p[1] << 48)
36 | ((uint64_t) p[2] << 40) | ((uint64_t) p[3] << 32)
37 | ((uint64_t) p[4] << 24) | ((uint64_t) p[5] << 16)
38 | ((uint64_t) p[6] << 8) | (uint64_t) p[7];
39}

41size_t
42brix_cns_event_encode(uint8_t op, const char *path, uint64_t size,
43 uint64_t mtime, uint8_t *buf, size_t bufsz)
44{
45 size_t plen = path ? strlen(path) : 0;
46
47 if (plen == 0 || plen > BRIX_CNS_PATH_MAX
48 || bufsz < BRIX_CNS_HDR_LEN + plen)
49 {
50 return 0;
51 }
52 buf[0] = op;
53 buf[1] = buf[2] = buf[3] = 0;
54 put_u64(buf + 4, size);
55 put_u64(buf + 12, mtime);
56 buf[20] = (uint8_t) (plen >> 8);
57 buf[21] = (uint8_t) plen;
58 memcpy(buf + BRIX_CNS_HDR_LEN, path, plen);
59 return BRIX_CNS_HDR_LEN + plen;
60}

62ngx_int_t
63brix_cns_event_decode(const uint8_t *buf, size_t len, uint8_t *op,
64 uint64_t *size, uint64_t *mtime, char *path,
65 size_t pathsz)
66{
67 size_t plen;
68
69 if (buf == NULL || len < BRIX_CNS_HDR_LEN) {
70 return NGX_ERROR;
71 }
72 plen = ((size_t) buf[20] << 8) | buf[21];
73 if (plen == 0 || plen > BRIX_CNS_PATH_MAX
74 || len < BRIX_CNS_HDR_LEN + plen || plen >= pathsz)
75 {
76 return NGX_ERROR;
77 }
78 *op = buf[0];
79 *size = get_u64(buf + 4);
80 *mtime = get_u64(buf + 12);
81 memcpy(path, buf + BRIX_CNS_HDR_LEN, plen);
82 path[plen] = '\0';
83 return NGX_OK;
84}

86size_t
87brix_cns_event_encode_mv(const char *oldpath, const char *newpath,
88 uint64_t size, uint64_t mtime, int is_dir,
89 uint8_t *buf, size_t bufsz)
90{
91 size_t olen = oldpath ? strlen(oldpath) : 0;
92 size_t nlen = newpath ? strlen(newpath) : 0;
93 size_t need;
94
95 if (olen == 0 || olen > BRIX_CNS_PATH_MAX
96 || nlen == 0 || nlen > BRIX_CNS_PATH_MAX)
97 {
98 return 0;
99 }
100 need = BRIX_CNS_HDR_LEN + olen + 2 + nlen;
101 if (bufsz < need) {
102 return 0;
103 }
104
105 if (brix_cns_event_encode(BRIX_CNS_MV, oldpath, size, mtime, buf,
106 bufsz) == 0)
107 {
108 return 0;
109 }
110 buf[1] = is_dir ? 1 : 0; /* rsvd[0] carries the dest's dir-ness */
111 buf[BRIX_CNS_HDR_LEN + olen] = (uint8_t) (nlen >> 8);
112 buf[BRIX_CNS_HDR_LEN + olen + 1] = (uint8_t) nlen;
113 memcpy(buf + BRIX_CNS_HDR_LEN + olen + 2, newpath, nlen);
114 return need;
115}

117ngx_int_t
118brix_cns_event_decode_mv(const uint8_t *buf, size_t len, int *is_dir,
119 char *newpath, size_t newsz)
120{
121 size_t olen, nlen, off;
122
123 if (buf == NULL || newpath == NULL || len < BRIX_CNS_HDR_LEN) {
124 return NGX_ERROR;
125 }
126 olen = ((size_t) buf[20] << 8) | buf[21];
127 off = BRIX_CNS_HDR_LEN + olen;
128 if (olen == 0 || olen > BRIX_CNS_PATH_MAX || len < off + 2) {
129 return NGX_ERROR;
130 }
131 nlen = ((size_t) buf[off] << 8) | buf[off + 1];
132 if (nlen == 0 || nlen > BRIX_CNS_PATH_MAX || len < off + 2 + nlen
133 || nlen >= newsz)
134 {
135 return NGX_ERROR;
136 }
137 memcpy(newpath, buf + off + 2, nlen);
138 newpath[nlen] = '\0';
139 if (is_dir != NULL) {
140 *is_dir = (buf[1] != 0);
141 }
142 return NGX_OK;
143}

145/* ===================== inventory backing store ===================== */

147/* Cross-worker SHM zone (registered by brix_cns_configure). When present, the
148 * inventory lives in its slab and cns_mtx (bound to the slab-pool lock) serialises
149 * every worker's access. When absent, s_heap is a per-worker fallback table. */
150static ngx_shm_zone_t *cns_zone;
151static ngx_shmtx_t cns_mtx;
152static ngx_uint_t cns_slots_req;
153
154static brix_cns_inv_t *s_heap; /* lazy per-worker fallback table */
155
156static ngx_int_t
157cns_init_zone(ngx_shm_zone_t *shm_zone, void *data)
158{
159 brix_cns_inv_t *inv;
160 ngx_flag_t fresh;
161 ngx_uint_t cap = cns_slots_req ? cns_slots_req : BRIX_CNS_DEFAULT_SLOTS;
162
163 inv = brix_shm_table_alloc(shm_zone, data,
164 brix_cns_inv_bytes((uint32_t) cap),
165 &cns_mtx, &fresh);
166 if (inv == NULL) {
167 return NGX_ERROR;
168 }
169 if (fresh) {
170 brix_cns_inv_init(inv, (uint32_t) cap); /* helper already zeroed slots */
171 }
172 return NGX_OK;
173}

175ngx_int_t
176brix_cns_configure(ngx_conf_t *cf, ngx_uint_t slots)
177{
178 ngx_str_t zone_name = ngx_string("brix_cns_inventory");
179 size_t zone_size;
180
181 if (slots == 0) { slots = BRIX_CNS_DEFAULT_SLOTS; }
182 cns_slots_req = slots;
183
184 zone_size = brix_shm_zone_size(brix_cns_inv_bytes((uint32_t) slots));
185 cns_zone = ngx_shared_memory_add(cf, &zone_name, zone_size,
186 &ngx_stream_brix_module);
187 if (cns_zone == NULL) {
188 return NGX_ERROR;
189 }
190 cns_zone->init = cns_init_zone;
191 cns_zone->data = (void *) 1;
192 return NGX_OK;
193}

195/* Resolve the live table. SHM when the zone is initialised; else the per-worker
196 * heap fallback (lazily allocated). *shared reports whether cns_mtx must be held. */
197static brix_cns_inv_t *
198cns_active_table(ngx_flag_t *shared)
199{
200 if (cns_zone != NULL
201 && cns_zone->data != NULL
202 && cns_zone->data != (void *) 1)
203 {
204 *shared = 1;
205 return (brix_cns_inv_t *) cns_zone->data;
206 }
207 *shared = 0;
208 if (s_heap == NULL) {
209 s_heap = calloc(1, brix_cns_inv_bytes(BRIX_CNS_DEFAULT_SLOTS));
210 if (s_heap != NULL) {
211 brix_cns_inv_init(s_heap, BRIX_CNS_DEFAULT_SLOTS);
212 }
213 }
214 return s_heap;
215}

217ngx_int_t
218brix_cns_apply(uint8_t op, const char *path, uint64_t size, uint64_t mtime,
219 uint32_t server_id)
220{
221 ngx_flag_t shared;
222 brix_cns_inv_t *inv = cns_active_table(&shared);
223 int rc;
224
225 if (inv == NULL) {
226 return NGX_ERROR;
227 }
228 if (shared) { ngx_shmtx_lock(&cns_mtx); }
229 rc = brix_cns_inv_apply(inv, op, path, size, mtime, server_id);
230 if (shared) { ngx_shmtx_unlock(&cns_mtx); }
231 return (rc == 0) ? NGX_OK : NGX_ERROR;
232}

252ngx_int_t
253brix_cns_stat(const char *path, struct stat *out)
254{
255 ngx_flag_t shared;
256 brix_cns_inv_t *inv = cns_active_table(&shared);
257 uint64_t csize = 0, cmtime = 0;
258 int is_dir = 0, rc;
259
260 if (out == NULL) {
261 return NGX_ERROR;
262 }
263 if (inv == NULL) {
264 return NGX_DECLINED;
265 }
266 if (shared) { ngx_shmtx_lock(&cns_mtx); }
267 rc = brix_cns_inv_stat(inv, path, &csize, &cmtime, &is_dir);
268 if (shared) { ngx_shmtx_unlock(&cns_mtx); }
269
270 if (rc != 0) {
271 return (rc < 0) ? NGX_ERROR : NGX_DECLINED;
272 }
273 ngx_memzero(out, sizeof(*out));
274 out->st_size = (off_t) csize;
275 out->st_mtime = (time_t) cmtime;
276 out->st_mode = is_dir ? (S_IFDIR | 0755) : (S_IFREG | 0644);
277 out->st_nlink = 1;
278 return NGX_OK;
279}
