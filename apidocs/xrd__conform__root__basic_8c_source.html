77/* WHAT: Observe the pre-login protocol response on an isolated connection.
78 * WHY: A malformed first response prevents brix_connect from returning a
79 * connection, so the ordinary post-connect shape audit cannot retain the
80 * owning PROT-001/002 evidence.
81 * HOW: Exchange only the canonical handshake and protocol request, bound every
82 * read, and keep the malformed fixture out of the primary session. */
83static void
84read_bootstrap_observation(const brix_url *url, brix_io *io,
85 brix_status *status, bootstrap_observation *observed)
86{
87 ClientInitHandShake handshake;
88 ClientProtocolRequest protocol;
89 const uint8_t streamid[2] = { 0, 1 };
90 uint8_t handshake_header[XRD_RESPONSE_HDR_LEN];
91 uint8_t handshake_body[sizeof(ServerProtocolBody)];
92 uint8_t response_header[XRD_RESPONSE_HDR_LEN];
93 uint16_t response_sid = 0, response_status = 0;
94 uint32_t response_length = 0;
95 uint16_t handshake_sid = 0, handshake_status = 0;
96 uint32_t handshake_length = 0;
97 memset(observed, 0, sizeof(*observed));
98 xrd_pack_handshake(&handshake);
99 xrd_pack_protocol_request(&protocol, streamid, kXR_secreqs);
100 io->fd = brix_tcp_connect(url->host, url->port, io->timeout_ms, status);
101 if (io->fd >= 0
102 && brix_write_full(io, &handshake, sizeof(handshake), status) == 0
103 && brix_read_full(io, handshake_header, sizeof(handshake_header),
104 status) == 0) {
105 xrd_resp_hdr_unpack(handshake_header, &handshake_sid,
106 &handshake_status, &handshake_length);
107 observed->handshake_answered = 1;
108 observed->handshake_v5 = handshake_sid == 0 && handshake_status == 0
109 && handshake_length == sizeof(handshake_body)
110 && brix_read_full(io, handshake_body, sizeof(handshake_body),
111 status) == 0;
112 observed->handshake_version_ok = observed->handshake_v5
113 && xrd_get_u32_be(handshake_body) >= 0x300
114 && xrd_get_u32_be(handshake_body) <= kXR_PROTOCOLVERSION;
115 if (observed->handshake_v5
116 && brix_write_full(io, &protocol, sizeof(protocol), status) == 0
117 && brix_read_full(io, response_header, sizeof(response_header),
118 status) == 0) {
119 xrd_resp_hdr_unpack(response_header, &response_sid,
120 &response_status, &response_length);
121 observed->protocol_answered = response_sid == 1
122 && response_status == kXR_ok;
123 observed->protocol_body_long_enough = observed->protocol_answered
124 && response_length >= sizeof(ServerProtocolBody);
125 }
126 }
127}

132{

155static void
156probe_protocol_bootstrap(xrd_conform_execution *execution,
157 const brix_url *url)
158{
159 bootstrap_observation observed;
160 brix_io io;
161 brix_status status;
162 char reason[64];
163
164 if ((!xrd_conform_family_active(execution, "PROT")
165 && !xrd_conform_family_active(execution, "HSK"))
166 || url->scheme != XRDC_SCHEME_ROOT
167 || !xrd_conform_reserve(execution,
168 sizeof(ClientInitHandShake) + sizeof(ClientProtocolRequest),
169 reason, sizeof(reason))) return;
170 memset(&io, 0, sizeof(io));
171 io.fd = -1;
172 io.timeout_ms = 3000;
173 io.stall_deadline_ms = 3000;
174 brix_status_clear(&status);
175 read_bootstrap_observation(url, &io, &status, &observed);
176 if (io.fd >= 0) close(io.fd);
177 grade_bootstrap_observation(execution, &observed, &status);
178}

180static void
181probe_bootstrap_shape(xrd_conform_execution *execution, brix_conn *connection)
182{
183 const uint8_t *body = connection->protocol_body;
184 uint32_t length = connection->protocol_body_len;
185 int have_sec = length >= 14 && body[8] == 'S';
186 int sec_level_ok = have_sec ? body[12] <= 4 : 0;
187 int sec_size_ok = have_sec
188 ? length == 14U + (uint32_t) body[13] * 2U : 0;
189 size_t index;
190 int secvec_ok = sec_size_ok;
191
192 for (index = 0; secvec_ok && index < body[13]; index++)
193 if (body[15 + index * 2] > 2) secvec_ok = 0;
194 grade_clause(execution, "HSK-001", connection->handshake_body_len > 0,
195 "canonical-handshake-answered", "handshake-reply-missing", NULL);
196 grade_clause(execution, "HSK-002", connection->handshake_body_len == 8,
197 "v5-eight-byte-handshake-body", "non-v5-handshake-body", NULL);
198 grade_clause(execution, "HSK-003",
199 connection->server_protocol >= 0x300
200 && connection->server_protocol <= kXR_PROTOCOLVERSION,
201 "protocol-version-in-supported-range", "protocol-version-out-of-range",
202 NULL);
203 grade_clause(execution, "PROT-001",
204 length >= sizeof(ServerProtocolBody), "pre-login-protocol-answered",
205 "pre-login-protocol-missing", NULL);
206 grade_clause(execution, "PROT-002",
207 length >= sizeof(ServerProtocolBody), "short-protocol-body-present",
208 "protocol-body-too-short", NULL);
209 grade_clause(execution, "PROT-004", have_sec,
210 "secreqs-returned-S-trailer", "secreqs-trailer-missing", NULL);
211 grade_clause(execution, "PROT-006", sec_level_ok,
212 "security-level-in-domain", "security-level-out-of-domain", NULL);
213 grade_clause(execution, "PROT-007", secvec_ok,
214 "security-vector-requirements-in-domain",
215 "security-vector-requirement-invalid", NULL);
216 grade_clause(execution, "PROT-008", sec_size_ok,
217 "security-vector-size-matches-body", "security-vector-size-mismatch",
218 NULL);

221static int
222protocol_exchange(xrd_conform_execution *execution, brix_conn *connection,
223 int32_t version, uint8_t flags, uint8_t expect, uint8_t **body,
224 uint32_t *body_length, brix_status *status)
225{
226 ClientProtocolRequest request;
227 xrdw_protocol_req_t fields = {
228 .clientpv = version, .flags = flags, .expect = expect
229 };
230 uint16_t sid, wire_status = 0;
231 brix_resp_out response = { &wire_status, body, body_length };
232 char reason[64];
233
234 if (!reserve_request(execution, reason)) return -1;
235 memset(&request, 0, sizeof(request));
236 request.requestid = htons(kXR_protocol);
237 xrdw_protocol_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
238 brix_status_clear(status);
239 if (brix_send(connection, &request, NULL, &sid, status) != 0) return -1;
240 return brix_recv(connection, sid, &response, status);
241}

243static void
244probe_protocol_repeats(xrd_conform_execution *execution,
245 brix_conn *connection)
246{
247 uint8_t *plain = NULL;
248 uint8_t *repeat = NULL;
249 uint8_t *legacy = NULL;
250 uint32_t plain_len = 0, repeat_len = 0, legacy_len = 0;
251 brix_status status;
252 int plain_ok;
253 int repeat_ok;
254 int legacy_ok;
255 uint8_t repeat_flags = kXR_secreqs | kXR_ableTLS;
256
257 if (connection->want_tls) repeat_flags |= kXR_wantTLS;
258
259 plain_ok = protocol_exchange(execution, connection, 0x300, 0,
260 kXR_ExpLogin, &plain, &plain_len, &status) == 0;
261 plain_ok = plain_ok && plain_len == sizeof(ServerProtocolBody)
262 && connection->protocol_body_len >= 14
263 && connection->protocol_body[8] == 'S';
264 grade_clause(execution, "PROT-004", plain_ok,
265 "S-trailer-only-when-requested",
266 "unrequested-security-trailer-present", status.msg);
267 repeat_ok = protocol_exchange(execution, connection, kXR_PROTOCOLVERSION,
268 repeat_flags, kXR_ExpLogin, &repeat, &repeat_len,
269 &status) == 0;
270 repeat_ok = repeat_ok
271 && repeat_len == connection->protocol_body_len
272 && memcmp(repeat, connection->protocol_body, repeat_len) == 0;
273 grade_clause(execution, "PROT-009", repeat_ok,
274 "repeated-protocol-body-identical", "repeated-protocol-body-changed",
275 status.msg);
276 grade_clause(execution, "PROT-010", repeat_ok,
277 "pre-and-post-login-capabilities-identical",
278 "capabilities-depend-on-login", status.msg);
279 legacy_ok = protocol_exchange(execution, connection, 0x100, 0,
280 kXR_ExpLogin, &legacy, &legacy_len, &status) == 0
281 && legacy_len >= sizeof(ServerProtocolBody);
282 grade_clause(execution, "PROT-011", legacy_ok,
283 "below-floor-version-answered", "below-floor-version-refused",
284 status.msg);
285 free(plain);
286 free(repeat);
287 free(legacy);

300 brix_conn *connection)
301{
302 uint8_t *bif = NULL;
303 uint8_t *tpc = NULL;

322 grade_clause(execution, "PROT-012", tpc_ok,

324 "tpc-expect-changed-capability-flags", status.msg);
325 unknown_ok = protocol_exchange(execution, connection, kXR_PROTOCOLVERSION,
326 0, (uint8_t) (kXR_ExpMask + 1), &unknown, &unknown_len, &status) == 0;

351static int
352split_path(const char *path, char parent[XRDC_PATH_MAX], const char **name)
353{
354 const char *slash = strrchr(path, '/');
355 size_t length;
356
357 if (slash == NULL || slash[1] == '\0') return -1;
358 *name = slash + 1;
359 length = slash == path ? 1U : (size_t) (slash - path);
360 if (length >= XRDC_PATH_MAX) return -1;
361 memcpy(parent, path, length);
362 parent[length] = '\0';
363 return 0;
364}
