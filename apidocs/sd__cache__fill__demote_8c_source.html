28/*
29 * WHAT: Copy one hot cache object into an already-open cold staged object.
30 * WHY: Separate the streaming state machine from tier/resource acquisition.
31 * HOW: Retry interrupted reads, write each chunk at its source offset, and
32 * commit only after a clean EOF; the caller retains cleanup ownership.
33 */
34static ngx_int_t
35sd_cache_demote_copy(brix_sd_obj_t *source, brix_sd_staged_t *staged,
36 u_char *buf)
37{
38 off_t off = 0;
39
40 for ( ;; ) {
41 ssize_t n = source->driver->pread(source, buf, SD_CACHE_CHUNK, off);
42
43 if (n < 0) {
44 if (errno == EINTR) {
45 continue;
46 }
47 return NGX_ERROR;
48 }
49 if (n == 0) {
50 return staged->inst->driver->staged_commit(staged, NULL);
51 }
52 if (staged->inst->driver->staged_write(staged, buf, (size_t) n,
53 off) < 0)
54 {
55 return NGX_ERROR;
56 }
57 off += n;
58 }
59}

62/* Demote the HOT cached object `key` into the cold store tier (phase-85 F7):
63 * copy its bytes from the cache store into the cold store via the cold
64 * driver's staged spine (write + commit — never a torn cold object). Called by
65 * the eviction engine on space-pressure victims ONLY, just before the hot copy
66 * is removed; write invalidation never demotes (a written-over object is stale
67 * and must not survive in cold). Defined here (not sd_cache.c) to reuse the
68 * fill spine's move-granule idiom. NGX_OK / NGX_DECLINED (no cold tier — not
69 * an error) / NGX_ERROR with errno set (the caller evicts anyway: space relief
70 * wins and the origin refill preserves correctness). */
71/* Phase-107 C5: a demotion knows the exact final size up front (the hot copy

74 * hot object cannot answer (the cold tier then reserves nothing). */
75static off_t
76sd_cache_demote_declared_size(brix_sd_obj_t *so)
77{
78 brix_sd_stat_t hst;
79
80 if (so->driver->fstat != NULL && so->driver->fstat(so, &hst) == NGX_OK) {
81 return hst.size;
82 }
83 return 0;
84}
