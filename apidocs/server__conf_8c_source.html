1/*
2 * server_conf.c — server-block config lifecycle: allocate, merge, enable.
3 *
4 * create_srv_conf() allocates one srv_conf per server block with every field at
5 * an NGX_CONF_UNSET sentinel (or NULL), so the merge step can tell an omitted
6 * directive from one explicitly set. merge_srv_conf() applies nginx parent→child
7 * inheritance (ngx_conf_merge_* for scalars, brix_merge_arrays() for the
8 * VO/group/manager-map rule sets); it is a flat orchestrator that delegates each
9 * configuration area to a per-area entry point. enable() handles
10 * "xrootd on|off;", swapping in our session handler when on.
11 *
12 * Phase-79 file-size split: the per-area MERGE helpers were moved into three
13 * focused sibling files, keeping this file under the 500-line cap. The five
14 * area entry points the orchestrator calls are declared in
15 * server_conf_internal.h and defined in:
16 * server_conf_merge_security.c — identity/crypto + storage-plane merges
17 * server_conf_merge_cluster.c — TPC + cluster/session merges
18 * server_conf_merge_proxy_net.c — upstream/proxy + network-fault merge
19 * The create-side per-area initialisers stay here, beside the creator they serve.
20 */

269/*
270 * This function creates a new server-level configuration object for nginx.
271 * It allocates memory and initializes every field to an "unset" or NULL state,
272 * allowing the merge step later to distinguish between missing directives and
273 * explicitly configured values.
274 */
275void *
276ngx_stream_brix_create_srv_conf(ngx_conf_t *cf)
277{
278 ngx_stream_brix_srv_conf_t *conf;
279
280 /*
281 * nginx allocates one per-server config object during parsing and then
282 * merges parent/child scopes later. Start everything in an explicit
283 * "unset" or NULL state so the merge step can tell whether a directive
284 * was omitted or configured intentionally.
285 *
286 * Scalar fields use nginx's UNSET sentinels when they participate in merge
287 * logic; runtime-only objects start out NULL/invalid and are created later
288 * during postconfiguration once parsing has finished. The per-field init is
289 * split into one helper per configuration area, mirroring the merge helpers.
290 */
291 conf = ngx_pcalloc(cf->pool, sizeof(ngx_stream_brix_srv_conf_t));
292 if (conf == NULL) {
293 return NULL;
294 }
295
296 ngx_http_brix_shared_init(&conf->common);
297 brix_create_srv_security(conf);
298 brix_create_srv_storage(conf);
299 brix_create_srv_cluster(conf);
300 brix_create_srv_proxy_net(conf);
301
302 return conf;
303}

305/*
306 * Standard nginx parent→child scope inheritance. Delegates to one per-area entry
307 * point (each defined in a sibling server_conf_merge_*.c and declared in
308 * server_conf_internal.h), invoked in the original linear order so cross-area
309 * derivations still observe their already-merged inputs (e.g. writethrough's
310 * origin/decision in proxy_net depends on the cache fields settled in storage;
311 * CMS timeout derivation in cluster depends on the merged cms_interval). Any
312 * entry point returning NGX_CONF_ERROR aborts the merge (the error is already
313 * logged).
314 */
315char *
316ngx_stream_brix_merge_srv_conf(ngx_conf_t *cf, void *parent, void *child)
317{
318 ngx_stream_brix_srv_conf_t *prev = parent;
319 ngx_stream_brix_srv_conf_t *conf = child;
320
321 /* phase-101 W3 (variant A): seed the storage bare names (brix_export,
322 * brix_storage_backend, brix_storage_credential, brix_allow_write,
323 * brix_verify_write) from ngx_stream_brix_common_module — their single
324 * stream-plane owner — into this server's embedded preamble BEFORE the
325 * per-area merges, so brix_merge_srv_storage() derives root_canon and the
326 * backend from the adopted values exactly as it did when root registered
327 * them directly. Only fills still-UNSET common.* slots, so a value set on
328 * THIS server wins and everything else this module owns is untouched. */
329 brix_stream_common_adopt(cf, &conf->common);
330
331 if (brix_merge_srv_security(cf, conf, prev) != NGX_CONF_OK) {
332 return NGX_CONF_ERROR;
333 }
334 if (brix_shared_clone_vo_rules(cf, &conf->common) != NGX_OK) {
335 return NGX_CONF_ERROR;
336 }
337 if (brix_merge_srv_storage(cf, conf, prev) != NGX_CONF_OK) {
338 return NGX_CONF_ERROR;
339 }
340 if (brix_merge_srv_tpc(cf, conf, prev) != NGX_CONF_OK) {
341 return NGX_CONF_ERROR;
342 }
343 if (brix_merge_srv_cluster(cf, conf, prev) != NGX_CONF_OK) {
344 return NGX_CONF_ERROR;
345 }
346 if (brix_merge_srv_proxy_net(cf, conf, prev) != NGX_CONF_OK) {
347 return NGX_CONF_ERROR;
348 }
349
350 return NGX_CONF_OK;
351}

357ngx_stream_brix_enable(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
358{

362

364 if (rv != NGX_CONF_OK) {
365 return rv;
366 }
367
368 /* Explicit `xrootd off;` leaves the server block as a normal stream server. */
369 if (!xcf->common.enable) {
370 return NGX_CONF_OK;
371 }
372

375 * our session handler for this server block.
376 */
377 cscf = ngx_stream_conf_get_module_srv_conf(cf, ngx_stream_core_module);
378 cscf->handler = ngx_stream_brix_handler;
379
380 return NGX_CONF_OK;
381}
