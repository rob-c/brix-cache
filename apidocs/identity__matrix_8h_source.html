4/*
5 * identity_matrix.h — the TPC identity matrix (2.0 F18), BriX's spelling of
6 * XRootD's `ofs.tpc allow|require|restrict|oids`.
7 *
8 * WHAT: a second, INNER authorization plane for third-party copy, expressed
9 * over the *initiating identity* rather than over the host this server would
10 * dial. Four controls:
11 * brix_tpc_allow_identity dn|group|host|vo <pattern> [...] — who may TPC
12 * brix_tpc_require all|client|dest <auth> — how they authed
13 * brix_tpc_restrict <path>... — where
14 * brix_tpc_oids on|off — object-id paths
15 *
16 * WHY: today's TPC confinement is entirely host-plane — brix_tpc_source_guard /
17 * _source_allow name the peers this server may originate to, and
18 * brix_tpc_allow_local / _allow_private bound the address ranges. That
19 * answers "which machines", never "which users". An operator running a
20 * shared gateway wants "only the CMS VO, only over GSI, only under /store"
21 * — the identity matrix stock xrootd has had for years. It layers ON TOP of
22 * the host plane and can only ever NARROW: the host verdict is evaluated
23 * first and a matrix rule is never consulted after a host denial, so no
24 * identity rule can widen an egress decision.
25 *
26 * HOW: the whole verdict is a pure, allocation-free, nginx-free function
27 * (brix_tpc_matrix_evaluate) over borrowed C strings, so every branch is
28 * proven offline by identity_matrix_unittest.c — the same split
29 * egress_guard.c uses. The ngx layer behind XRDPROTO_NO_NGX only adapts an
30 * identity + a conf into that call and renders the refusal text. Rules are
31 * stored in the conf arrays AS the pure structs (conf tokens are already
32 * NUL-terminated and live for the cycle), so there is no per-request
33 * conversion and no second spelling of a rule.
34 *
35 * Evaluation order is a SECURITY order, not a stylistic one:
36 * 1. oids — a `*`-prefixed (object-id) path is refused unless opted in.
37 * 2. allow — the subject must match at least one rule when any is configured.
38 * 3. require— the subject's auth method must be one of those demanded of its
39 * party when any rule names that party.
40 * 4. restrict — the path must sit under one of the configured prefixes.
41 * Each stage is default-permit when it is UNCONFIGURED (so the matrix is a
42 * no-op for deployments that do not opt in) and fail-closed once configured
43 * (an unmatched subject is DENIED, never defaulted). The one exception is
44 * `oids`, which is default-DENY exactly as in stock xrootd: BriX has no
45 * object-id namespace at all, so a `*`-prefixed TPC path is nonsense here and
46 * refusing it is a pure narrowing.
47 */

51/*
52 * The two parties a `require` rule can name. The distinction is the whole
53 * point of the directive and is decided by ONE wire fact: a native TPC leg
54 * carrying tpc.org was opened by the peer SERVER on the client's behalf and
55 * therefore presents the SERVER's credential; a leg without tpc.org was opened
56 * by the initiating CLIENT and presents the client's own. `require dest` is
57 * therefore not satisfiable by a client credential, and vice versa.
58 */
59#define BRIX_TPC_PARTY_CLIENT 0
60#define BRIX_TPC_PARTY_DEST 1
61#define BRIX_TPC_PARTY_ALL 2 /* only ever a RULE's party, never a subject's */

63/*
64 * One `brix_tpc_allow_identity` line. Every non-empty field must match (AND
65 * within a rule); rules OR across lines — stock xrootd's semantics for
66 * `ofs.tpc allow [dn ..] [group ..] [host ..] [vo ..]`. A rule with no field
67 * set matches NOTHING (it cannot be written: the setter demands a pair).
68 * NULL and "" are both "field not set".
69 */
70typedef struct {
71 const char *dn; /* exact subject DN */
72 const char *group; /* one member of the subject's group CSV */
73 const char *host; /* egress_guard host pattern: exact, or '.'-suffix */
74 const char *vo; /* one member of the subject's VO CSV */
75} brix_tpc_allow_rule_t;
76
77/* One `brix_tpc_require <party> <auth>` line. */
78typedef struct {
79 int party; /* BRIX_TPC_PARTY_{CLIENT,DEST,ALL} */
80 const char *auth; /* auth-method label, matched case-insensitively */
81} brix_tpc_require_rule_t;

83/*
84 * The subject a leg presents. Borrowed strings, never owned; NULL is read as
85 * "" everywhere, so an unauthenticated or unresolved field simply fails to
86 * match any rule that names it (fail-closed).
87 */
88typedef struct {
89 const char *dn; /* subject DN ("" when none) */
90 const char *groups; /* comma-separated group paths */
91 const char *vos; /* comma-separated VO names */
92 const char *host; /* reverse-resolved peer FQDN ("" when unknown) */
93 const char *auth; /* auth-method label: GSI / TOKEN / SSS / ... / NONE */
94} brix_tpc_subject_t;

145/*
146 * brix_tpc_restrict_match — component-aware prefix test.
147 *
148 * 1 when `path` IS `prefix` or sits strictly beneath it at a '/' boundary.
149 * Deliberately NARROWER than stock xrootd's plain string prefix: `/data` does
150 * NOT match `/database`. The narrower direction is the safe one — an operator
151 * who wrote `/data` meant the directory — and the difference is documented in
152 * the user guide rather than hidden. A `prefix` of "/" matches every absolute

154 * LOGICAL paths (traversal already removed), so no `..` can dodge the test.
155 */
