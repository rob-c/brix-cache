24/* Stream the whole remote file into a private anonymous temp fd. Returns the fd
25 * (already unlinked, caller closes) or -1 with *st set. */
26static int
27verify_download_tmp(brix_conn *c, const char *path, brix_status *st)
28{
29 char tmpl[] = "/tmp/xrdstorascan.XXXXXX";
30 int fd;
31 brix_file f;
32 int64_t off = 0;
33 char *buf;
34
35 fd = mkstemp(tmpl);
36 if (fd < 0) {
37 brix_status_set(st, XRDC_ESOCK, 0, "mkstemp failed");
38 return -1;
39 }
40 (void) unlink(tmpl); /* anonymous fd — no symlink/planted-file race */
41
42 if (brix_file_open_read(c, path, &f, st) != 0) {
43 close(fd);
44 return -1;
45 }
46 buf = (char *) malloc(1u << 20);
47 if (buf == NULL) {
48 brix_file_close(c, &f, st);
49 close(fd);
50 brix_status_set(st, XRDC_ESOCK, 0, "out of memory");
51 return -1;
52 }
53 for (;;) {
54 ssize_t n = brix_file_read(c, &f, off, buf, 1u << 20, st);
55 if (n < 0) {
56 free(buf);
57 brix_file_close(c, &f, st);
58 close(fd);
59 return -1;
60 }
61 if (n == 0) {
62 break;
63 }
64 if (write(fd, buf, (size_t) n) != n) {
65 free(buf);
66 brix_file_close(c, &f, st);
67 close(fd);
68 brix_status_set(st, XRDC_ESOCK, 0, "temp write failed");
69 return -1;
70 }
71 off += n;
72 }
73 free(buf);
74 (void) brix_file_close(c, &f, st);
75 return fd;
76}

97/*
98 * verify_parse_args — decode `verify` options into url/algo/quiet.
99 * WHY: keeps cmd_verify a linear pipeline (parse → connect → compare).
100 * HOW: --algo takes a value, -q/--quiet sets quiet, exactly one positional
101 * URL; anything else (or a missing URL) prints usage → SX_USAGE.
102 */
103static int
104verify_parse_args(int argc, char **argv, const char **url,
105 const char **algo, int *quiet, const char *prog)
106{
107 int i;
108
109 for (i = 0; i < argc; i++) {
110 const char *a = argv[i];
111 if (opt_take("--algo", argc, argv, &i, algo)) {
112 continue;
113 }
114 if (strcmp(a, "-q") == 0 || strcmp(a, "--quiet") == 0) {
115 *quiet = 1;
116 } else if (a[0] == '-') {
117 return usage(prog, SX_USAGE);
118 } else if (*url == NULL) {
119 *url = a;
120 } else {
121 return usage(prog, SX_USAGE);
122 }
123 }
124 if (*url == NULL) {
125 return usage(prog, SX_USAGE);
126 }
127 return SX_OK;
128}

130/*
131 * verify_compute_wire_hex — recompute the checksum from bytes over the wire.
132 * WHY: the download+rewind+digest block is a self-contained step of the
133 * verify pipeline; isolating it keeps cmd_verify under the gate.
134 * HOW: stream the file into an anonymous temp fd, rewind, digest it with the
135 * requested algorithm; errors are reported to stderr here and mapped to
136 * the shell exit code the caller returns (caller still owns brix_close).
137 */
138static int
139verify_compute_wire_hex(brix_conn *c, const char *path,
140 brix_cksum_algo algo, char *hex, size_t hexsz)
141{
142 brix_status st;
143 int tmpfd;
144
145 brix_status_clear(&st);
146 tmpfd = verify_download_tmp(c, path, &st);
147 if (tmpfd < 0) {
148 fprintf(stderr, "xrdstorascan: download %s: %s\n", path, st.msg);
149 return brix_shellcode(&st);
150 }
151 if (lseek(tmpfd, 0, SEEK_SET) < 0 ||
152 brix_cksum_fd(tmpfd, algo, hex, hexsz, &st) != 0) {
153 fprintf(stderr, "xrdstorascan: checksum %s: %s\n", path, st.msg);
154 close(tmpfd);
155 return SX_ERROR;
156 }
157 close(tmpfd);
158 return SX_OK;
159}

161int
162cmd_verify(int argc, char **argv, const char *prog)
163{
164 const char *url = NULL;
165 const char *algo = "adler32";
166 int quiet = 0;
167 brix_url u;
168 brix_conn c;
169 brix_status st;
170 brix_cksum_algo algo_enum;
171 char server_hex[STORASCAN_HEX_MAX];
172 char computed_hex[STORASCAN_HEX_MAX];
173 int rc;
174 storascan_cks_status verdict;
175
176 if (argc >= 1 && strcmp(argv[0], "--help") == 0) {
177 return verify_help(prog);
178 }
179 rc = verify_parse_args(argc, argv, &url, &algo, &quiet, prog);
180 if (rc != SX_OK) {
181 return rc;
182 }
183 if (brix_cksum_algo_parse(algo, &algo_enum) != 0) {
184 fprintf(stderr, "xrdstorascan: unsupported algorithm '%s'\n", algo);
185 return SX_USAGE;
186 }
187
188 rc = storascan_connect(url, &u, &c, &st);
189 if (rc != SX_OK) {
190 return rc;
191 }
192
193 /* Reference value: the server's recorded checksum. */
194 if (brix_query_cksum(&c, u.path, algo, server_hex, sizeof(server_hex), &st) != 0) {
195 fprintf(stderr, "xrdstorascan: %s %s: %s\n", "query checksum", u.path, st.msg);
196 brix_close(&c);
197 return brix_shellcode(&st);
198 }
199
200 /* Recompute from the bytes pulled over the wire. */
201 rc = verify_compute_wire_hex(&c, u.path, algo_enum,
202 computed_hex, sizeof(computed_hex));
203 brix_close(&c);
204 if (rc != SX_OK) {
205 return rc;
206 }
207
208 verdict = storascan_cks_compare(computed_hex, server_hex);
209 switch (verdict) {
210 case STORASCAN_CKS_MATCH:
211 if (!quiet) {
212 printf("OK %s %s %s\n", u.path, algo, computed_hex);
213 }
214 return SX_OK;
215 case STORASCAN_CKS_MISMATCH:
216 fprintf(stderr, "MISMATCH %s %s: wire=%s recorded=%s\n",
217 u.path, algo, computed_hex, server_hex);
218 return SX_MISMATCH;
219 case STORASCAN_CKS_MISSING:
220 default:
221 fprintf(stderr, "NO-RECORD %s %s: server reported no checksum\n",
222 u.path, algo);
223 return SX_NORECORD;
224 }
225}
