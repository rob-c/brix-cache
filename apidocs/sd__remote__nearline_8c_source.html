1/*
2 * sd_remote_nearline.c — the nearline (archive) pair of the remote-origin
3 * (s3://) driver: residency (are this object's bytes readable now?) and recall
4 * (start bringing them back).
5 *
6 * WHY THIS FILE EXISTS
7 * An S3 bucket with a GLACIER or DEEP_ARCHIVE lifecycle is the cheapest cold
8 * store WLCG sites actually deploy, and until this file existed the driver had
9 * neither nearline slot: brix_vfs_residency reported ONLINE for every key (its
10 * answer for a driver with no residency model), so the first read of an
11 * archived object came back as an opaque 403 InvalidObjectState from the
12 * origin — a hard error, with nothing above able to say "it is on tape, ask
13 * again later", and no way at all to START the restore.
14 *
15 * THE WIRE
16 * S3 splits the two halves exactly as this driver's slots do:
17 * HEAD → x-amz-storage-class / x-amz-restore / x-amz-archive-status
18 * POST ?restore → RestoreObject
19 * The HEAD is a pure read: it never touches the archive tier and never starts
20 * a restore, which is precisely what the residency contract requires.
21 *
22 * OPT-IN, AND WHY IT MUST BE
23 * Neither slot is reachable unless the instance advertises
24 * BRIX_SD_CAP_NEARLINE, which brix_sd_remote_create sets only from
25 * cfg->nearline. CAP_NEARLINE is a CONTRACT, not a hint: tier_build refuses to
26 * compose a nearline backend without a cache tier in front (the recall target,
27 * §9.4), so inferring it from a storage class seen on one object would turn
28 * every working s3:// export into a startup failure the first time somebody
29 * archived a single key.
30 *
31 * IDENTITY
32 * residency has no _cred twin: it signs as the export's service credential,
33 * because a residency probe is the gateway's own housekeeping, driven by the
34 * cache tier on a miss. recall carries one (phase-107 C2): a CLIENT-initiated
35 * prestage (kXR_prepare) is the user's request, so the RestoreObject — an
36 * operation the bucket owner may well have IAM-scoped per user, and one that
37 * is BILLED — signs with the requester's keys when the VFS cred gate resolves
38 * them, and the internal residency probe inside recall_cred signs the same
39 * way so a deny-mode request never reaches the origin under the service
40 * credential at all (the stat_cred rationale). The cache tier's own
41 * recall-at-fill keeps calling the plain slot and stays export-charged.
42 */

50/* A storage class is one token; an x-amz-restore value is a fixed-shape pair
51 * ("ongoing-request=\"false\", expiry-date=\"...\""). 128 fits both with room,
52 * and anything longer is not a value this build can classify. */
53#define SD_REMOTE_ARCH_MAX 128

55/* The storage classes whose objects are NOT directly readable. Everything else
56 * S3 offers — STANDARD, the IA tiers, INTELLIGENT_TIERING, GLACIER_IR — serves a
57 * GET immediately, which is why this is a list of the archival ones rather than
58 * a list of the online ones: see sd_remote_is_archived on why that direction is
59 * the safe one. */
60static const char *sd_remote_archived_class[] = {
61 "GLACIER", "DEEP_ARCHIVE",
62};
63
64/*
65 * WHAT: 1 iff the object's bytes are in an archive tier and so cannot be read
66 * without a restore.
67 * WHY: Two independent things put an object in that state, and missing either
68 * one would make the driver claim an unreadable object is online: an
69 * explicit archival storage class, and INTELLIGENT_TIERING's automatic
70 * demotion, which keeps the storage class as INTELLIGENT_TIERING and
71 * reports the demotion in x-amz-archive-status instead.
72 * HOW: An exact match against the archival class list, or any non-empty
73 * archive status (its two values, ARCHIVE_ACCESS and DEEP_ARCHIVE_ACCESS,
74 * both mean archived; the header is absent otherwise).
75 *
76 * An UNRECOGNISED storage class is treated as online. That is the opposite
77 * of the http driver's locality mapping, which errors on an unknown token,
78 * and the asymmetry is deliberate: the Tape REST API's localities are a
79 * closed vocabulary, so an unknown one means we are not talking to that
80 * API at all, whereas AWS adds storage classes routinely and every one it
81 * has ever added outside this list serves reads directly. Guessing
82 * "archived" for a new class would park every open on a restore that was
83 * never needed; guessing "online" degrades to exactly the behaviour this
84 * driver had before the slot existed.
85 */
86static int
87sd_remote_is_archived(const char *sclass, const char *astatus)
88{
89 size_t i;
90
91 if (astatus[0] != '\0') {
92 return 1;
93 }
94 for (i = 0; i < sizeof(sd_remote_archived_class)
95 / sizeof(sd_remote_archived_class[0]); i++) {
96 if (strcmp(sclass, sd_remote_archived_class[i]) == 0) {
97 return 1;
98 }
99 }
100 return 0;
101}

103/*
104 * WHAT: Classify an archived object from its x-amz-restore header.
105 * WHY: An archived object has three states, not two, and the header is the only
106 * thing that separates them: no restore has been asked for (the header is
107 * absent), one is running, or one has completed and a temporary copy is
108 * readable RIGHT NOW. The last is genuinely ONLINE — reporting it as
109 * NEARLINE would make the cache re-request a restore for bytes it could
110 * already have fetched, and pay for the archive retrieval twice.
111 * HOW: `ongoing-request="false"` is the completed spelling; anything else with
112 * the header present is in flight. Both in-flight and never-requested are
113 * NEARLINE — from the caller's side they differ only in whether recall's
114 * RestoreObject starts the job or joins it, which recall does not need to
115 * know in advance.
116 */
117static brix_sd_residency_t
118sd_remote_restore_state(const char *restore)
119{
120 if (strstr(restore, "ongoing-request=\"false\"") != NULL) {
121 return BRIX_SD_RES_ONLINE;
122 }
123 return BRIX_SD_RES_NEARLINE;
124}

126/* sd_remote_residency — vtable residency slot: classify `key` WITHOUT restoring.
127 *
128 * WHAT: NGX_OK with *out set to ONLINE or NEARLINE, or NGX_ERROR with errno set
129 * (ENOENT for a key the bucket does not hold).
130 * WHY: Every protocol plane advertises tape state from this seam, and all of
131 * them must be able to say "archived" without paying for a retrieval.
132 * HOW: One signed HEAD, three headers, the two classifiers above. OFFLINE and
133 * LOST are never reported: S3 has no notion of "temporarily unretrievable"
134 * (an archived object is always restorable) and none of "destroyed" (a key
135 * that is gone is simply absent, which is ENOENT — inventing LOST from a
136 * 404 would tell every plane above that a file had been DESTROYED because
137 * the bucket merely does not have it).
138 */
139static ngx_int_t
140sd_remote_residency_impl(brix_sd_instance_t *inst, const char *key,
141 brix_sd_residency_t *out, const char *ak, const char *sk,
142 const char *region, const char *session)
143{
144 const brix_sd_remote_cfg_t *cfg;
145 sd_s3_open_params p;
146 sd_s3_file *s3;
147 char objpath[768];
148 char errbuf[256];
149 char sclass[SD_REMOTE_ARCH_MAX];
150 char restore[SD_REMOTE_ARCH_MAX];
151 char astatus[SD_REMOTE_ARCH_MAX];
152 const sd_s3_meta_buf b_class = { sclass, sizeof(sclass) };
153 const sd_s3_meta_buf b_restore = { restore, sizeof(restore) };
154 const sd_s3_meta_buf b_status = { astatus, sizeof(astatus) };
155 const sd_s3_archive_buf_t want = { &b_class, &b_restore, &b_status };
156 int rc;
157
158 if (inst == NULL || inst->state == NULL || key == NULL || out == NULL) {
159 errno = EINVAL;
160 return NGX_ERROR;
161 }
162 cfg = inst->state;
163 sd_remote_s3_key(cfg, key, objpath, sizeof(objpath));
164 sd_remote_s3_params(cfg, objpath, &p);
165 sd_remote_params_cred(&p, ak, sk, region, session);
166
167 s3 = sd_s3_open_read(&p, errbuf, sizeof(errbuf));
168 if (s3 == NULL) {
169 return NGX_ERROR;
170 }
171 rc = sd_s3_archive_state(s3, &want, errbuf, sizeof(errbuf));
172 sd_s3_close(s3);
173 if (rc != 0) {
174 return NGX_ERROR; /* errno mapped from the HTTP status */
175 }
176 if (!sd_remote_is_archived(sclass, astatus)) {
177 *out = BRIX_SD_RES_ONLINE;
178 return NGX_OK;
179 }
180 *out = sd_remote_restore_state(restore);
181 return NGX_OK;
182}

191/* sd_remote_recall — vtable recall slot: bring `key` online, without blocking.
192 *
193 * WHAT: NGX_OK (already readable — a normal cache fill follows), NGX_AGAIN (a
194 * restore is under way; reqid_out is left empty), or NGX_ERROR with errno.
195 * WHY: This is the verb sd_cache drives on every nearline miss. Returning
196 * NGX_AGAIN instead of stalling is the whole point: the open fails soft
197 * with EAGAIN and the plane above answers "staging, retry later" rather
198 * than holding a worker for the hours a DEEP_ARCHIVE retrieval takes.
199 * HOW: Residency first — a RestoreObject against an object that is already
200 * readable is a paid no-op on every cache miss of a hot key. Only a
201 * NEARLINE answer posts the restore, and that is ALWAYS NGX_AGAIN: the
202 * verb is asynchronous by definition, so acceptance means "queued", never
203 * "done". A restore that FAILS is still NGX_AGAIN — the object is
204 * genuinely not readable either way, and "retry later" describes that to a
205 * client far better than a hard error on an advisory hint does.
206 *
207 * reqid_out stays empty: RestoreObject issues no request id at all, and
208 * the slot contract says an empty id means "queued, poll the state" —
209 * which is exactly what a caller must do with S3.
210 */
211static ngx_int_t
212sd_remote_recall_impl(brix_sd_instance_t *inst, const char *key,
213 char reqid_out[40], const char *ak, const char *sk, const char *region,
214 const char *session)
215{
216 const brix_sd_remote_cfg_t *cfg;
217 sd_s3_open_params p;
218 brix_sd_residency_t res;
219 char objpath[768];
220 char errbuf[256];
221
222 if (reqid_out != NULL) {
223 reqid_out[0] = '\0';
224 }
225 if (sd_remote_residency_impl(inst, key, &res, ak, sk, region, session)
226 != NGX_OK)
227 {
228 return NGX_ERROR; /* errno already set (ENOENT/EACCES/…) */
229 }
230 if (res == BRIX_SD_RES_ONLINE) {
231 return NGX_OK; /* readable — a normal fill follows */
232 }
233 cfg = inst->state;
234 sd_remote_s3_key(cfg, key, objpath, sizeof(objpath));
235 sd_remote_s3_params(cfg, objpath, &p);
236 sd_remote_params_cred(&p, ak, sk, region, session);
237 (void) sd_s3_restore(&p, cfg->restore_days, errbuf, sizeof(errbuf));
238 return NGX_AGAIN; /* queued — the caller polls residency */
239}

248/* recall_cred (phase-107 C2) — the SAME probe-then-restore, signed with the
249 * requesting user's keys (sd_remote_cred_gate: 1 = override, 0 = static
250 * fallback, -1 = deny — the stat_cred shape). Both the HEAD and the
251 * RestoreObject POST complete before return: S3's restore is asynchronous at
252 * the SERVICE (the caller polls residency), but the REQUEST itself is one
253 * synchronous signed round trip, so the borrowed cred is consumed inside the
254 * call and never retained — the sd.h copy rule binds only a driver that keeps
255 * issuing requests after it returns (the opendir asymmetry this driver
256 * documented). */
257ngx_int_t
258sd_remote_recall_cred(brix_sd_instance_t *inst, const char *key,
259 const brix_sd_cred_t *cred, char reqid_out[40])
260{
261 int gate = sd_remote_cred_gate(cred);
262
263 if (gate > 0) {
264 return sd_remote_recall_impl(inst, key, reqid_out,
265 cred->s3_ak, cred->s3_sk, cred->s3_region, cred->s3_session);
266 }
267 if (gate < 0) {
268 errno = EACCES;
269 return NGX_ERROR;
270 }
271 return sd_remote_recall_impl(inst, key, reqid_out,
272 NULL, NULL, NULL, NULL);
273}
