38/*
39 * WHAT: Appends managed names from one bounded NUL-separated xattr list.
40 * Returns NGX_OK, or NGX_ERROR for malformed or over-cap input.
41 *
42 * WHY: Backend lengths and terminators are untrusted; accepting a partial list
43 * can hide attributes and an unbounded strlen can read past the buffer.
44 *
45 * HOW: Finds every terminator with memchr, decodes the managed name, computes
46 * the complete entry with checked additions, then appends it atomically. */
47static ngx_int_t
48fattr_recurse_emit_names(fattr_recurse_ctx_t *rctx, brix_vfs_ctx_t *vctx,
49 const char *fpath, const char *relpath, const char *xlist, size_t list_sz)
50{
51 const char *cursor = xlist;
52 const char *end = xlist + list_sz;
53 size_t relpath_len = strlen(relpath);
54
55 while (cursor < end) {
56 const char *nul;
57 brix_fattr_stored_t stored;
58 int decoded;
59
60 nul = memchr(cursor, '\0', (size_t) (end - cursor));
61 if (nul == NULL) {
62 errno = EIO;
63 return NGX_ERROR;
64 }
65 decoded = fattr_storage_decode_list(rctx->pool, vctx, fpath, -1,
66 cursor, &stored);
67 if (decoded < 0) return NGX_ERROR;
68 if (decoded == 1) {
69 size_t entry_len;
70
71 if (brix_size_add(relpath_len, 1U, &entry_len) != NGX_OK
72 || brix_size_add(entry_len, stored.nlen, &entry_len) != NGX_OK
73 || brix_size_add(entry_len, 1U, &entry_len) != NGX_OK
74 || rctx->len > rctx->cap
75 || entry_len > rctx->cap - rctx->len)
76 {
77 errno = ERANGE;
78 return NGX_ERROR;
79 }
80 ngx_memcpy(rctx->buf + rctx->len, relpath, relpath_len);
81 rctx->len += relpath_len;
82 rctx->buf[rctx->len++] = ':';
83 ngx_memcpy(rctx->buf + rctx->len, stored.name, stored.nlen);
84 rctx->len += stored.nlen;
85 rctx->buf[rctx->len++] = '\0';
86 }
87 cursor = nul + 1;
88 }
89 return NGX_OK;
90}

92/*
93 * WHAT: Adds managed names from one regular file to the recursive response.
94 * Returns NGX_ERROR if a backend violates the bounded-list contract.
95 *
96 * WHY: A positive backend byte count must never be trusted past the supplied
97 * stack buffer, even when every built-in backend obeys the VFS contract.
98 *
99 * HOW: Derives the file context, performs one bounded list read, validates the
100 * returned count, and delegates parsing to fattr_recurse_emit_names. */
101static ngx_int_t
102fattr_recurse_file(fattr_recurse_ctx_t *rctx, const char *fpath,
103 const brix_vfs_stat_t *vst)
104{
105 char xlist[FATTR_RECURSE_XLIST_BUF];
106 ssize_t list_sz;
107 const char *relpath;
108 brix_vfs_ctx_t fvctx;
109
110 if (!vst->is_regular) return NGX_OK;
111 relpath = fpath + rctx->root_len;
112 if (*relpath == '/') relpath++;
113
114 fattr_recurse_init_node_ctx(rctx, &fvctx, fpath);
115 list_sz = brix_vfs_listxattr(&fvctx, xlist, sizeof(xlist));
116 if (list_sz <= 0) return NGX_OK;
117 if ((size_t) list_sz > sizeof(xlist)) {
118 errno = EIO;
119 return NGX_ERROR;
120 }
121 return fattr_recurse_emit_names(rctx, &fvctx, fpath, relpath, xlist,
122 (size_t) list_sz);
123}

125/*
126 * WHAT: Walks a confined directory tree and accumulates managed attributes.
127 * Returns NGX_ERROR without sending a partial response on any hard limit.
128 *
129 * WHY: Symlinks, oversized paths and deep trees must not escape confinement or
130 * produce a deceptively incomplete security-relevant enumeration.
131 *
132 * HOW: Uses VFS no-follow metadata, descends real directories to a fixed depth,
133 * and closes the directory before propagating every child failure. */
134static ngx_int_t
135fattr_recurse_dir(fattr_recurse_ctx_t *rctx, const char *dir_path, int depth)
136{
137 brix_vfs_ctx_t dvctx;
138 brix_vfs_dir_t *dir;
139 ngx_str_t name;
140 brix_vfs_stat_t vst;
141 char fpath[BRIX_PATH_MAX];
142 int path_len;
143
144 if (depth > FATTR_RECURSE_MAX_DEPTH) {
145 errno = ELOOP;
146 return NGX_ERROR;
147 }
148 fattr_recurse_init_node_ctx(rctx, &dvctx, dir_path);
149 dir = brix_vfs_opendir_quiet(&dvctx, NULL);
150 if (dir == NULL) return NGX_OK;
151
152 while (brix_vfs_readdir(dir, &name, &vst) == NGX_OK) {
153 ngx_int_t child_rc;
154
155 if (name.len == 0 || name.data[0] == '.') continue;
156 path_len = snprintf(fpath, sizeof(fpath), "%s/%s", dir_path,
157 (char *) name.data);
158 if (path_len < 0 || path_len >= (int) sizeof(fpath)) {
159 errno = ENAMETOOLONG;
160 brix_vfs_closedir(dir, rctx->log);
161 return NGX_ERROR;
162 }
163 child_rc = vst.is_directory
164 ? fattr_recurse_dir(rctx, fpath, depth + 1)
165 : fattr_recurse_file(rctx, fpath, &vst);
166 if (child_rc != NGX_OK) {
167 brix_vfs_closedir(dir, rctx->log);
168 return NGX_ERROR;
169 }
170 }
171 brix_vfs_closedir(dir, rctx->log);
172 return NGX_OK;
173}

175/*
176 * WHAT: Builds and sends one recursive fattr response under a fixed cap.
177 * Returns the nginx send result, mapping size failures to ArgTooLong.
178 *
179 * WHY: Recursive enumeration must be atomic: truncating at the cap would make
180 * a successful response indistinguishable from a complete inventory.
181 *
182 * HOW: Allocates the capped accumulator, walks the tree, maps any failure, and
183 * sends the complete list only after the walk succeeds. */
184static ngx_int_t
185fattr_list_recurse(brix_ctx_t *ctx, ngx_connection_t *c,
186 brix_vfs_ctx_t *vctx, const char *path)
187{
188 fattr_recurse_ctx_t rctx;
189 u_char *buf;
190
191 buf = ngx_palloc(c->pool, FATTR_RECURSE_RESP_CAP);
192 if (buf == NULL) {
193 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
194 return brix_send_error(ctx, c, kXR_NoMemory, "out of memory");
195 }
196 rctx.buf = buf;
197 rctx.cap = FATTR_RECURSE_RESP_CAP;
198 rctx.len = 0;
199 rctx.root_len = strlen(path);
200 rctx.vfs_scope = vctx;
201 rctx.log = vctx->log;
202 rctx.pool = c->pool;
203
204 if (fattr_recurse_dir(&rctx, path, 0) != NGX_OK) {
205 int walk_errno = errno;
206
207 BRIX_OP_ERR(ctx, BRIX_OP_FATTR);
208 if (walk_errno == ERANGE || walk_errno == E2BIG
209 || walk_errno == ENAMETOOLONG)
210 {
211 return brix_send_error(ctx, c, kXR_ArgTooLong,
212 "recursive fattr list too large");
213 }
214 return brix_send_error(ctx, c, kXR_FSError,
215 "recursive fattr list failed");
216 }
217 BRIX_OP_OK(ctx, BRIX_OP_FATTR);
218 if (rctx.len == 0) return brix_send_ok(ctx, c, NULL, 0);
219 return brix_send_ok(ctx, c, rctx.buf, (uint32_t) rctx.len);
220}

222/*
223 * WHAT: Dispatches directory requests carrying kXR_fa_recurse.
224 * Returns one when handled and writes the send result to *out.
225 *
226 * WHY: Keeps the standard flat-list path independent from the local recursive
227 * extension while sharing the already-authorized VFS request context.
228 *
229 * HOW: Probes the target through VFS and invokes the recursive builder only for
230 * directories; all other requests return zero for normal processing. */
231int
232fattr_list_try_recurse(brix_ctx_t *ctx, ngx_connection_t *c,
233 brix_vfs_ctx_t *vctx, const char *path, int options, ngx_int_t *out)
234{
235 brix_vfs_stat_t rvst;
236
237 if ((options & kXR_fa_recurse) && path != NULL
238 && brix_vfs_probe(vctx, 0, &rvst) == NGX_OK && rvst.is_directory)
239 {
240 *out = fattr_list_recurse(ctx, c, vctx, path);
241 return 1;
242 }
243 return 0;
244}
