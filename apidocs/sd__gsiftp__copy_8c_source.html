1/*
2 * sd_gsiftp_copy.c — same-origin server copy for the GridFTP backend.
3 *
4 * WHAT: Fills the driver's `server_copy` slot: copy one object to another path
5 * on the SAME origin without the client in the loop. Until this existed
6 * the slot was NULL, and `brix_vfs_copy_driver` turned every WebDAV COPY
7 * on a gsiftp-backed export into ENOTSUP — the operation was not slow,
8 * it was absent.
9 *
10 * WHY: Without the slot a client that wants /a copied to /b has to GET the
11 * whole object and PUT it back, so the bytes cross the client link twice
12 * and the copy is only as durable as that client's connection. Here they
13 * never leave the gateway<->origin link, and the destination appears
14 * whole or not at all.
15 *
16 * HOW: ONE control session does both legs — FTP is sequential on the control
17 * channel, so after the source's 226 the same session is idle and ready
18 * to store. The bytes land in a local scratch file between the legs
19 * (tmpfile(), the same spill the staged write path already takes for
20 * every PUT), because gftp_retrieve PUSHES to a sink while gftp_store
21 * PULLS from a source: the two loops cannot drive each other without a
22 * thread, and a thread per copy is a worse trade than a scratch file.
23 * Publication is the driver's existing idiom — STOR to a random temp
24 * name, then RNFR/RNTO — so a failed copy leaves the destination's old
25 * bytes untouched rather than half-replaced.
26 *
27 * The size comes from a stat taken BEFORE the transfer and is enforced
28 * after it: gftp_retrieve is a bounded read, so a short origin answer is
29 * a short file, and publishing that would be silent truncation.
30 */

45/* Land one retrieved span in the scratch file at its own offset. */
46static int
47sd_gsiftp_copy_sink(void *ctx, off_t offset, const uint8_t *data, size_t len)
48{
49 sd_gsiftp_copy_sink_t *sink = ctx;
50 size_t done = 0;
51
52 while (done < len) {
53 ssize_t n = pwrite(sink->fd, data + done, len - done, offset + done);
54
55 if (n < 0) {
56 if (errno == EINTR) {
57 continue;
58 }
59 return -1;
60 }
61 if (n == 0) {
62 errno = ENOSPC;
63 return -1;
64 }
65 done += (size_t) n;
66 }
67 sink->written += (off_t) len;
68 return 0;
69}

72/*
73 * Pull the whole source object into `fd`.
74 *
75 * `size` is both the request and the acceptance test. gftp_retrieve reads a
76 * BOUNDED span, so an origin that answers with fewer bytes than it just
77 * reported in its own stat produces a shorter file and no error of its own —
78 * the mismatch has to be caught here or the truncation gets published.
79 */
80static int
81sd_gsiftp_copy_fetch(gftp_session_t *session, const char *remote_src,
82 off_t size, int fd)
83{
84 sd_gsiftp_copy_sink_t sink;
85 size_t received = 0;
86
87 sink.fd = fd;
88 sink.written = 0;
89
90 if (gftp_retrieve(session, remote_src, 0, (size_t) size,
91 sd_gsiftp_copy_sink, &sink, &received) != 0) {
92 return -1;
93 }
94 if ((off_t) received != size || sink.written != size) {
95 errno = EIO;
96 return -1;
97 }
98 return 0;
99}

102/* Read back the scratch file for the store leg. */
103static ssize_t
104sd_gsiftp_copy_source(void *ctx, uint8_t *data, size_t cap)
105{
106 sd_gsiftp_copy_sink_t *source = ctx;
107 ssize_t n;
108
109 do {
110 n = pread(source->fd, data, cap, source->written);
111 } while (n < 0 && errno == EINTR);
112 if (n > 0) {
113 source->written += n;
114 }
115 return n;
116}

119/*
120 * Publish the scratch file at `remote_dst` atomically.
121 *
122 * The temp name is not a nicety: a COPY that overwrites an existing object by
123 * STORing straight onto it replaces a whole file with a partial one for the
124 * length of the transfer, and leaves it that way if the transfer fails. A
125 * failed publish deletes its own temp rather than leaving litter the operator
126 * has to recognise as ours.
127 */
128static int
129sd_gsiftp_copy_publish(gftp_session_t *session, const char *remote_dst, int fd)
130{
131 sd_gsiftp_copy_sink_t source;
132 char temp_path[GSIFTP_PATH_CAP];
133 int rc;
134
135 if (sd_gsiftp_temp_path(remote_dst, temp_path) != 0) {
136 return -1;
137 }
138 source.fd = fd;
139 source.written = 0;
140
141 rc = gftp_store(session, temp_path, sd_gsiftp_copy_source, &source);
142 if (rc == 0) {
143 rc = gftp_expect(session, 300, 399, "RNFR %s", temp_path);
144 }
145 if (rc == 0) {
146 rc = gftp_expect(session, 200, 299, "RNTO %s", remote_dst);
147 }
148 if (rc != 0) {
149 (void) gftp_command(session, "DELE %s", temp_path);
150 return -1;
151 }
152 return 0;
153}

156/*
157 * Resolve both ends and refuse the degenerate case.
158 *
159 * A copy onto its own path is refused rather than performed: the transfer
160 * would work — scratch, temp, rename over the original — but it rewrites a
161 * healthy object for no gain, and any failure in the middle would damage the
162 * only copy of a file the caller never asked to modify.
163 */
164static int
165sd_gsiftp_copy_paths(const sd_gsiftp_state *state, const char *src,
166 const char *dst, char remote_src[GSIFTP_PATH_CAP],
167 char remote_dst[GSIFTP_PATH_CAP])
168{
169 if (sd_gsiftp_path(state, src, remote_src) != 0
170 || sd_gsiftp_path(state, dst, remote_dst) != 0) {
171 return -1;
172 }
173 if (strcmp(remote_src, remote_dst) == 0) {
174 errno = EINVAL;
175 return -1;
176 }
177 return 0;
178}

181static ngx_int_t
182sd_gsiftp_server_copy_impl(brix_sd_instance_t *inst, const char *src,
183 const char *dst, off_t *bytes_out, const brix_sd_cred_t *cred)
184{
185 sd_gsiftp_state *state = inst->state;
186 gftp_session_t session;
187 brix_sd_stat_t source_stat;
188 FILE *scratch;
189 char remote_src[GSIFTP_PATH_CAP];
190 char remote_dst[GSIFTP_PATH_CAP];
191 const char *proxy;
192 int err = 0;
193 int rc;
194
195 if (sd_gsiftp_select_proxy(state, cred, &proxy, &err) != 0
196 || sd_gsiftp_copy_paths(state, src, dst, remote_src, remote_dst) != 0) {
197 return NGX_ERROR;
198 }
199 /* Before the scratch file, not after: a copy of a path that is not there
200 * should cost one round trip and no local disk. */
201 if (sd_gsiftp_stat_impl(inst, src, &source_stat, cred) != NGX_OK) {
202 return NGX_ERROR;
203 }
204 scratch = tmpfile();
205 if (scratch == NULL) {
206 return NGX_ERROR;
207 }
208 if (sd_gsiftp_session(&session, state, proxy) != 0) {
209 fclose(scratch);
210 return NGX_ERROR;
211 }
212 rc = sd_gsiftp_copy_fetch(&session, remote_src, source_stat.size,
213 fileno(scratch));
214 if (rc == 0) {
215 rc = sd_gsiftp_copy_publish(&session, remote_dst, fileno(scratch));
216 }
217 gftp_session_close(&session);
218 fclose(scratch);
219 if (rc != 0) {
220 return NGX_ERROR;
221 }
222 if (bytes_out != NULL) {
223 *bytes_out = source_stat.size;
224 }
225 return NGX_OK;
226}
