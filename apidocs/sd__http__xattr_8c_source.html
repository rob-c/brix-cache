1/*
2 * sd_http_xattr.c — reading extended attributes on an http origin, out of the
3 * WebDAV dead properties that carry them.
4 *
5 * WHAT: The getxattr/listxattr vtable slots and their credential-scoped twins,
6 * over a named-prop PROPFIND Depth:0 (one attribute) and a propname
7 * PROPFIND Depth:0 (the whole set). The write half is sd_http_xattr_write.c.
8 *
9 * WHY: `http` was the only namespace-capable driver in the tree with NO xattr
10 * support at all, and everything the gateway layers on top of a per-object
11 * key/value store went dark behind an http origin: WebDAV LOCK tokens,
12 * WebDAV PROPPATCH dead properties, S3 object tagging, S3 user metadata,
13 * and root:// kXR_fattr. Each of those already has one storage-neutral
14 * spelling — an xattr on the object — so closing the slot re-lights all of
15 * them at once rather than teaching five features about HTTP.
16 *
17 * HOW: RFC 4918 §15 dead properties are exactly the primitive wanted: arbitrary
18 * server-preserved name/value pairs on a resource. One xattr becomes one
19 * element in the BriX xattr XML namespace, named `bxa` + lowercase hex of
20 * the xattr name; the value is lowercase hex too.
21 *
22 * Hex on BOTH halves is a security property, not a style: an xattr name or
23 * value is arbitrary bytes chosen by a remote client, and interpolating
24 * those into a request body is how markup injection happens. Hex has no
25 * XML metacharacters and no NUL problem, so a value of `]]></D:prop>` is
26 * just eighteen more hex digits on the wire. It also makes the mapping
27 * total and reversible, which a "translate to the natural dead property"
28 * scheme cannot be: keys like `user.s3.tagging` have no namespace/local
29 * pair to translate INTO, and the webdav dead-prop codec that owns the
30 * other direction is ngx_pool_t-based, so calling it from ngx-free backend
31 * code would invert the layering. The cost is that these properties are
32 * opaque to a native WebDAV client — recorded, deliberate, and the reason
33 * the element carries a BriX-specific namespace rather than squatting DAV:.
34 *
35 * Every read goes through ONE propfind helper and every write through ONE
36 * proppatch helper (sd_http_xattr_write.c), so the credential gate, the
37 * endpoint pinning and the multistatus verdict have a single spelling
38 * across the four ops. The wire spelling itself — namespace, element
39 * marker, size ceilings — lives in sd_http_xattr_internal.h so the reader
40 * and the writer cannot drift apart.
41 */

53/*
54 * WHAT: Encode one xattr name as the element local name `bxa<hex(name)>`.
55 * WHY: The name is remote-chosen bytes; it must never reach a request body in a
56 * form that could close a tag or open one.
57 * HOW: Reject empty (EINVAL) and over-long (ERANGE, matching Linux setxattr),
58 * then hex the bytes after the fixed `bxa` marker that tells a scan which
59 * properties on a resource are ours.
60 */
61int
62sd_http_xattr_elem(const char *name, char *out, size_t cap)
63{
64 size_t nlen;
65
66 if (name == NULL || name[0] == '\0') {
67 errno = EINVAL;
68 return -1;
69 }
70 nlen = strlen(name);
71 if (nlen > SD_HTTP_XATTR_NAME_MAX || 3 + 2 * nlen + 1 > cap) {
72 errno = ERANGE;
73 return -1;
74 }
75 memcpy(out, "bxa", 3);
76 brix_hex_encode((const uint8_t *) name, nlen, out + 3);
77 return 0;
78}

81/*
82 * WHAT: Decode `n` hex digits at `p` into out[cap], returning the byte count.
83 * WHY: Both halves of the mapping travel as hex, so exactly one decoder is
84 * needed — and it must refuse rather than salvage: a half-decoded value
85 * handed to a caller as if it were the stored one is a silent corruption.
86 * HOW: Odd length or any non-hex digit is -1; an over-long run is -1 too, so a
87 * short destination never sees a partial write.
88 */
89static ssize_t
90sd_http_xattr_unhex(const char *p, size_t n, unsigned char *out, size_t cap)
91{
92 size_t i;
93
94 if ((n & 1u) != 0 || n / 2 > cap) {
95 return -1;
96 }
97 for (i = 0; i < n; i += 2) {
98 int hi = brix_hex_from_char((unsigned char) p[i]);
99 int lo = brix_hex_from_char((unsigned char) p[i + 1]);
100
101 if (hi < 0 || lo < 0) {
102 return -1;
103 }
104 out[i / 2] = (unsigned char) ((hi << 4) | lo);
105 }
106 return (ssize_t) (n / 2);
107}

110/*
111 * WHAT: Resolve the credential for one xattr request into an auth header line
112 * and an optional per-user x509 proxy path.
113 * WHY: All four ops present the requesting user's identity the same way, and
114 * the deny-mode refusal (a proxy-only cred the transport cannot present)
115 * must happen BEFORE any request leaves this host.
116 * HOW: The shared cred gate, then the shared open-cred resolver; the per-user

133/*
134 * WHAT: Issue one Depth:0 named-prop PROPFIND and hand back its body as a
135 * NUL-terminated, bounded, malloc'd copy the caller frees.
136 * WHY: get and list differ only in the request body and what they scan for;
137 * sharing the request leg keeps one credential path and one size ceiling.
138 * HOW: The shared propfind issuer, pinned to the primary endpoint — a property
139 * read must see the same replica the matching write acts on — then a
140 * bounded copy, because the tag scanners walk a C string.
141 */
142static int
143sd_http_xattr_propfind(sd_http_inst_state *is, const char *path,
144 const char *body, const brix_sd_cred_t *cred, char **xml_out,
145 size_t *len_out)
146{
147 char open_auth[SD_HTTP_AUTH_MAX];
148 const char *auth, *cert;
149 brix_s3_resp_t resp;
150 sd_http_pf_t pf;
151 const void *rbody;
152 size_t blen = 0;
153 char *xml = NULL;
154 int err = 0;
155
156 if (sd_http_xattr_auth(is, cred, open_auth, sizeof(open_auth),
157 &auth, &cert) != 0)
158 {
159 return -1;
160 }
161 pf.key = path;
162 pf.auth = auth;
163 pf.cert_pem = (cert != NULL && cert[0] != '\0') ? cert : NULL;
164 pf.depth = 0;
165 pf.force_primary = 1;
166 pf.body = body;
167
168 if (sd_http_propfind_issue(is, &pf, &resp, &err) != 0) {
169 errno = err;
170 return -1;
171 }
172 rbody = is->transport->resp_body(&resp, &blen);
173 if (rbody == NULL || blen == 0 || blen > SD_HTTP_XATTR_XML_MAX) {
174 is->transport->resp_free(&resp);
175 errno = EIO; /* not a multistatus we can read */
176 return -1;
177 }
178 xml = malloc(blen + 1);
179 if (xml == NULL) {
180 is->transport->resp_free(&resp);
181 errno = ENOMEM;
182 return -1;
183 }
184 memcpy(xml, rbody, blen);
185 xml[blen] = '\0';
186 is->transport->resp_free(&resp);
187
188 *xml_out = xml;
189 *len_out = blen;
190 return 0;
191}

194/*
195 * WHAT: 1 iff the first `<…status>` element at or after `p` reads 2xx.
196 * WHY: A named-prop PROPFIND for an ABSENT property still returns the element —
197 * inside a 404 propstat. Reading the element without reading the propstat
198 * status is how "no such attribute" becomes "an empty attribute".
199 * HOW: RFC 4918 orders `propstat = prop, status`, so the first status after our
200 * element is the one that judges it; its text is a status-line, and the
201 * code is the token after the HTTP version.
202 */
203int
204sd_http_xattr_status_ok(const char *p, const char *end)
205{
206 const char *st = sd_http_xml_open(p, end, "status");
207 const char *gt, *te;
208
209 if (st == NULL) {
210 return 0;
211 }
212 gt = memchr(st, '>', (size_t) (end - st));
213 if (gt == NULL || gt[-1] == '/') {
214 return 0;
215 }
216 te = memchr(gt + 1, '<', (size_t) (end - (gt + 1)));
217 if (te == NULL) {
218 te = end;
219 }
220 while (gt + 1 < te && *gt != ' ') {
221 gt++; /* skip the HTTP-version token */
222 }
223 return (te - gt >= 4 && gt[1] == '2');
224}

227/*
228 * WHAT: Read one property's hex text out of a 207 body into `buf`.
229 * WHY: Split from the request leg so the POSIX buffer contract (bufsz 0 = ask
230 * the size, short buffer = ERANGE with nothing written) is stated once.
231 * HOW: Locate our element, refuse the self-closing form and a non-2xx propstat
232 * as ENODATA, decode the hex text, and only then decide whether it fits.
233 */
234static ssize_t
235sd_http_xattr_extract(const char *xml, size_t xlen, const char *elem,
236 void *buf, size_t bufsz)
237{
238 const char *end = xml + xlen;
239 const char *p = sd_http_xml_open(xml, end, elem);
240 const char *gt, *te;
241 unsigned char val[SD_HTTP_XATTR_VALUE_MAX];
242 ssize_t vlen;
243
244 if (p == NULL) {
245 errno = ENODATA;
246 return -1;
247 }
248 gt = memchr(p, '>', (size_t) (end - p));
249 if (gt == NULL || gt[-1] == '/' || !sd_http_xattr_status_ok(gt, end)) {
250 errno = ENODATA; /* empty element or 404 propstat */
251 return -1;
252 }
253 te = memchr(gt + 1, '<', (size_t) (end - (gt + 1)));
254 if (te == NULL) {
255 errno = EIO;
256 return -1;
257 }
258 vlen = sd_http_xattr_unhex(gt + 1, (size_t) (te - (gt + 1)), val,
259 sizeof(val));
260 if (vlen < 0) {
261 errno = EIO; /* not a value this driver wrote */
262 return -1;
263 }
264 if (bufsz == 0) {
265 return vlen; /* size enquiry */
266 }
267 if ((size_t) vlen > bufsz) {
268 errno = ERANGE; /* refuse; never truncate */
269 return -1;
270 }
271 memcpy(buf, val, (size_t) vlen);
272 return vlen;
273}

276ssize_t
277sd_http_getxattr_common(brix_sd_instance_t *inst, const char *path,
278 const char *name, void *buf, size_t bufsz, const brix_sd_cred_t *cred)
279{
280 sd_http_inst_state *is;
281 char elem[SD_HTTP_XATTR_ELEM_MAX];
282 char body[SD_HTTP_XATTR_ELEM_MAX + 256];
283 char *xml = NULL;
284 size_t xlen = 0;
285 ssize_t rc;
286
287 if (inst == NULL || inst->state == NULL || path == NULL
288 || (buf == NULL && bufsz != 0))
289 {
290 errno = EINVAL;
291 return -1;
292 }
293 is = inst->state;
294 if (sd_http_xattr_elem(name, elem, sizeof(elem)) != 0) {
295 return -1; /* errno set by the encoder */
296 }
297 snprintf(body, sizeof(body),
298 "<?xml version=\"1.0\" encoding=\"utf-8\"?>"
299 "<D:propfind xmlns:D=\"DAV:\" xmlns:B=\"%s\">"
300 "<D:prop><B:%s/></D:prop></D:propfind>",
301 SD_HTTP_XATTR_NS, elem);
302
303 if (sd_http_xattr_propfind(is, path, body, cred, &xml, &xlen) != 0) {
304 return -1;
305 }
306 rc = sd_http_xattr_extract(xml, xlen, elem, buf, bufsz);
307 free(xml);
308 return rc;
309}

312/*
313 * WHAT: Length of the element local name starting at `p` (after the '<'), with
314 * any namespace prefix already stripped.
315 * WHY: listxattr enumerates properties by PATTERN (`bxa…`), which the shared
316 * exact-name tag scanner cannot express.
317 * HOW: Skip a `prefix:` if one is present, then run to the first byte that can
318 * end a name; 0 means this was not a name-shaped tag.
319 */
320static size_t
321sd_http_xattr_local(const char **pp, const char *end)
322{
323 const char *p = *pp;
324 const char *colon = NULL;
325 const char *s;
326
327 for (s = p; s < end && *s != '>' && *s != '/' && *s != ' ' && *s != '\t'
328 && *s != '\r' && *s != '\n'; s++)
329 {
330 if (*s == ':') {
331 colon = s;
332 }
333 }
334 if (colon != NULL) {
335 p = colon + 1;
336 }
337 *pp = p;
338 return (size_t) (s - p);
339}

342/*
343 * WHAT: Append every `bxa…` property name found in a propname 207 to `buf` as
344 * NUL-terminated strings, returning the total size the answer needs.
345 * WHY: listxattr's contract is "how many bytes would this take" when bufsz is
346 * 0, and ERANGE — not a truncated list — when the buffer is too small.
347 * HOW: Walk start tags, decode the hex after the `bxa` marker, and account for
348 * the size on EVERY name even after the buffer is known to be short, so
349 * the enquiry and the fill agree.
350 */
351static ssize_t
352sd_http_xattr_collect(const char *xml, size_t xlen, void *dst, size_t bufsz)
353{
354 const char *end = xml + xlen;
355 const char *p = xml;
356 char *buf = dst;
357 size_t need = 0;
358 int over = 0;
359
360 while (p < end && (p = memchr(p, '<', (size_t) (end - p))) != NULL) {
361 unsigned char nm[SD_HTTP_XATTR_NAME_MAX + 1];
362 const char *lp = ++p;
363 size_t llen;
364 ssize_t nlen;
365
366 if (lp >= end || *lp == '/' || *lp == '!' || *lp == '?') {
367 continue;
368 }
369 llen = sd_http_xattr_local(&lp, end);
370 if (llen <= 3 || memcmp(lp, "bxa", 3) != 0) {
371 continue;
372 }
373 nlen = sd_http_xattr_unhex(lp + 3, llen - 3, nm, sizeof(nm) - 1);
374 if (nlen <= 0) {
375 continue; /* not a property this driver wrote */
376 }
377 if (bufsz != 0 && need + (size_t) nlen + 1 <= bufsz) {
378 memcpy(buf + need, nm, (size_t) nlen);
379 buf[need + (size_t) nlen] = '\0';
380 } else if (bufsz != 0) {
381 over = 1;
382 }
383 need += (size_t) nlen + 1;
384 }
385 if (over) {
386 errno = ERANGE;
387 return -1;
388 }
389 return (ssize_t) need;
390}
