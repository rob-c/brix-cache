29static int
30setup_test_file(void)
31{
32 char temp_dir[MAX_PATH];
33 HANDLE handle;
34
35 /* Get temp directory */
36 if (GetTempPathA(sizeof(temp_dir), temp_dir) == 0) {
37 fprintf(stderr, "Failed to get temp directory\n");
38 return -1;
39 }
40
41 /* Build test file path */
42 snprintf(test_file_path, sizeof(test_file_path),
43 "%s\\brix_xattr_test_%lu.txt", temp_dir, (unsigned long) GetCurrentProcessId());
44
45 /* Create test file */
46 handle = CreateFileA(
47 test_file_path,
48 GENERIC_READ | GENERIC_WRITE,
49 0,
50 NULL,
51 CREATE_ALWAYS,
52 FILE_ATTRIBUTE_NORMAL,
53 NULL
54 );
55
56 if (handle == INVALID_HANDLE_VALUE) {
57 fprintf(stderr, "Failed to create test file: %s\n", test_file_path);
58 return -1;
59 }
60
61 CloseHandle(handle);
62 return 0;
63}
