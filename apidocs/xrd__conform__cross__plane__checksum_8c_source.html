17/* WHAT: Normalize one exact SHA-256 hexadecimal value.
18 * WHY: Protocols may vary hex case, but junk and truncated values are not proof.
19 * HOW: Require exactly 64 hex digits and copy their lowercase representation. */
20static int
21normalize_sha256(const char *value, char out[65])
22{
23 size_t index;
24
25 if (value == NULL || strlen(value) != 64) return -1;
26 for (index = 0; index < 64; index++) {
27 unsigned char byte = (unsigned char) value[index];
28
29 if (!isxdigit(byte)) return -1;
30 out[index] = (char) tolower(byte);
31 }
32 out[64] = '\0';
33 return 0;
34}

56/* WHAT: Validate an S3 ChecksumSHA256 header against adopted response bytes.
57 * WHY: Base64 metadata is evidence only when canonical and byte-exact.
58 * HOW: Decode the fixed SHA-256 shape, re-encode canonically, then compare raw. */
59static int
60s3_sha256_matches(const char *value, const brix_http_resp *baseline,
61 char checksum[65])
62{
63 unsigned char decoded[33];
64 unsigned char expected[32];
65 unsigned char canonical[45];
66 int decoded_length;
67 size_t index;
68
69 if (strlen(value) != 44 || value[43] != '=') return 0;
70 decoded_length = EVP_DecodeBlock(decoded,
71 (const unsigned char *) value, 44);
72 if (decoded_length != 33) return 0;
73 if (EVP_EncodeBlock(canonical, decoded, 32) != 44
74 || strcmp((const char *) canonical, value) != 0) return 0;
75 brix_s3_sha256_hex(baseline->body, baseline->body_len, checksum);
76 for (index = 0; index < 32; index++) {
77 unsigned high = (unsigned) (checksum[index * 2] <= '9'
78 ? checksum[index * 2] - '0'
79 : checksum[index * 2] - 'a' + 10);
80 unsigned low = (unsigned) (checksum[index * 2 + 1] <= '9'
81 ? checksum[index * 2 + 1] - '0'
82 : checksum[index * 2 + 1] - 'a' + 10);
83
84 expected[index] = (unsigned char) ((high << 4) | low);
85 }
86 return memcmp(decoded, expected, sizeof(expected)) == 0;
87}
