41/* Pick the next read offset for op index k, given file size and block.
42 * Uses the worker's private rand_r seed so concurrent workers never share
43 * PRNG state. */
44static int64_t
45bench_offset(bench_worker *w, uint64_t k)
46{
47 int64_t span = w->fsize - (int64_t) w->block;
48
49 if (span <= 0) {
50 return 0;
51 }
52 if (w->random) {
53 uint64_t r = ((uint64_t) rand_r(&w->seed) << 16) ^
54 (uint64_t) rand_r(&w->seed);
55 return (int64_t) (r % (uint64_t) (span + 1));
56 }
57 return (int64_t) ((k * (uint64_t) w->block) % (uint64_t) (span + 1));
58}

60static void *
61bench_run(void *arg)
62{
63 bench_worker *w = (bench_worker *) arg;
64 brix_conn c;
65 brix_status st;
66 brix_file f;
67 char *buf;
68 uint64_t k = 0;
69
70 brix_status_clear(&st);
71 if (brix_connect(&c, &w->u, NULL, &st) != 0) {
72 w->err = 1;
73 snprintf(w->errmsg, sizeof(w->errmsg), "connect: %s", st.msg);
74 return NULL;
75 }
76 if (brix_file_open_read(&c, w->u.path, &f, &st) != 0) {
77 w->err = 1;
78 snprintf(w->errmsg, sizeof(w->errmsg), "open: %s", st.msg);
79 brix_close(&c);
80 return NULL;
81 }
82 buf = (char *) malloc(w->block);
83 if (buf == NULL) {
84 w->err = 1;
85 snprintf(w->errmsg, sizeof(w->errmsg), "out of memory");
86 brix_file_close(&c, &f, &st);
87 brix_close(&c);
88 return NULL;
89 }
90
91 for (;;) {
92 uint64_t t0, t1;
93 int64_t off;
94 ssize_t n;
95
96 if (w->deadline_ns != 0) {
97 if (brix_mono_ns() >= w->deadline_ns) {
98 break;
99 }
100 } else if (k >= w->op_budget) {
101 break;
102 }
103
104 off = bench_offset(w, k);
105 t0 = brix_mono_ns();
106 n = brix_file_read(&c, &f, off, buf, w->block, &st);
107 t1 = brix_mono_ns();
108 if (n < 0) {
109 w->err = 1;
110 snprintf(w->errmsg, sizeof(w->errmsg), "read: %s", st.msg);
111 break;
112 }
113 if (w->lat_n < STORASCAN_LAT_CAP) {
114 w->lat_ms[w->lat_n++] = (double) (t1 - t0) / 1.0e6;
115 }
116 w->bytes += (uint64_t) n;
117 k++;
118 if (n == 0) { /* empty file: avoid a tight spin */
119 break;
120 }
121 }
122
123 free(buf);
124 (void) brix_file_close(&c, &f, &st);
125 brix_close(&c);
126 return NULL;
127}

145/*
146 * bench_workers_free — release a cell's worker/thread arrays.
147 * WHY: single owner for the alloc set (lat arrays + w + th) so every

149 * HOW: lat_ms slots are NULL-safe (calloc'd workers), so this is valid on

152static void
153bench_workers_free(bench_worker *w, pthread_t *th, int parallel)
154{
155 int i;
156
157 for (i = 0; i < parallel; i++) {
158 free(w[i].lat_ms);
159 }
160 free(w);
161 free(th);

164/*
165 * bench_workers_init — fill per-worker state + allocate latency buffers.
166 * WHY: splits cell setup out of bench_cell's orchestration.
167 * HOW: seeds each worker's private rand_r state, splits the op budget evenly
168 * and records the raw duration (rebased to a deadline in
169 * bench_workers_run). Any latency-buffer allocation failure reports
170 * "out of memory" and returns -1 (caller frees via bench_workers_free).
171 */
172static int
173bench_workers_init(const bench_cell_cfg_t *cfg, bench_worker *w,
174 char *errmsg, size_t errsz)
175{
176 int i, rc = 0;
177
178 for (i = 0; i < cfg->parallel; i++) {
179 w[i].url = cfg->url;
180 w[i].u = *cfg->u;
181 w[i].block = cfg->block;
182 w[i].random = cfg->random;
183 w[i].fsize = cfg->fsize;
184 w[i].seed = (unsigned) (brix_mono_ns() + (uint64_t) i * 2654435761u);
185 w[i].deadline_ns = cfg->duration_ns; /* rebased for time mode */
186 w[i].op_budget = cfg->total_ops / (uint64_t) cfg->parallel;
187 w[i].lat_ms = (double *) malloc(STORASCAN_LAT_CAP * sizeof(double));
188 if (w[i].lat_ms == NULL) {
189 snprintf(errmsg, errsz, "out of memory");
190 rc = -1;
191 }
192 }
193 return rc;
194}

196/*
197 * bench_workers_run — spawn one thread per worker, join them all, time it.
198 * WHY: isolates the thread lifecycle from setup and aggregation.
199 * HOW: time mode rebases every worker's deadline to now + duration before
200 * the spawn loop; a failed pthread_create is recorded as that worker's
201 * error and its slot skipped at join. t0/t1 bracket the whole cell.
202 */
203static void
204bench_workers_run(bench_worker *w, pthread_t *th, const bench_cell_cfg_t *cfg,
205 uint64_t *t0, uint64_t *t1)
206{
207 int i;
208
209 *t0 = brix_mono_ns();
210 if (cfg->duration_ns != 0) {
211 uint64_t deadline = *t0 + cfg->duration_ns;
212 for (i = 0; i < cfg->parallel; i++) {
213 w[i].deadline_ns = deadline;
214 }
215 }
216 for (i = 0; i < cfg->parallel; i++) {
217 if (pthread_create(&th[i], NULL, bench_run, &w[i]) != 0) {
218 w[i].err = 1;
219 snprintf(w[i].errmsg, sizeof(w[i].errmsg), "pthread_create failed");
220 th[i] = 0;
221 }
222 }
223 for (i = 0; i < cfg->parallel; i++) {
224 if (th[i] != 0) {
225 pthread_join(th[i], NULL);
226 }
227 }
228 *t1 = brix_mono_ns();
229}

252/*
253 * bench_workers_compute — merge per-worker samples into one result.
254 * WHY: percentile math needs a single sorted sample array; the merge is a
255 * self-contained aggregation step.
256 * HOW: sums bytes + sample counts, concatenates the per-worker latency
257 * arrays into one buffer and hands it to the pure statistics core.
258 * Returns -1 only when the merge buffer cannot be allocated.
259 */
260static int
261bench_workers_compute(const bench_worker *w, const bench_cell_cfg_t *cfg,
262 uint64_t elapsed_ns, storascan_bench_result *out)
263{
264 double *all_lat;
265 size_t all_n = 0, off = 0;
266 uint64_t total_bytes = 0;
267 int i;
268
269 for (i = 0; i < cfg->parallel; i++) {
270 all_n += w[i].lat_n;
271 total_bytes += w[i].bytes;
272 }
273 all_lat = (double *) malloc((all_n ? all_n : 1) * sizeof(double));
274 if (all_lat == NULL) {
275 return -1;
276 }
277 for (i = 0; i < cfg->parallel; i++) {
278 memcpy(all_lat + off, w[i].lat_ms, w[i].lat_n * sizeof(double));
279 off += w[i].lat_n;
280 }
281 storascan_bench_compute(all_lat, all_n, total_bytes,
282 (double) elapsed_ns / 1.0e9, out);
283 free(all_lat);
284 return 0;
285}

287/* Run one (block, parallel) cell; fill *out. Returns 0 / -1 (worker error). */
288static int
289bench_cell(const bench_cell_cfg_t *cfg, storascan_bench_result *out,
290 char *errmsg, size_t errsz)
291{
292 bench_worker *w;
293 pthread_t *th;
294 uint64_t t0, t1;
295 int rc;
296
297 w = (bench_worker *) calloc((size_t) cfg->parallel, sizeof(*w));
298 th = (pthread_t *) calloc((size_t) cfg->parallel, sizeof(*th));
299 if (w == NULL || th == NULL) {
300 free(w);
301 free(th);
302 snprintf(errmsg, errsz, "out of memory");
303 return -1;
304 }
305 if (bench_workers_init(cfg, w, errmsg, errsz) != 0) {
306 bench_workers_free(w, th, cfg->parallel);
307 return -1;
308 }
309
310 bench_workers_run(w, th, cfg, &t0, &t1);
311
312 rc = bench_workers_error(w, cfg->parallel, errmsg, errsz);
313 if (bench_workers_compute(w, cfg, t1 - t0, out) != 0) {
314 rc = -1;
315 snprintf(errmsg, errsz, "out of memory");
316 }
317 bench_workers_free(w, th, cfg->parallel);
318 return rc;
319}

321/* Parse a comma-separated list of sizes/ints into out[]; returns count or -1. */
322static int
323parse_list(const char *s, int as_bytes, long *out, int max)
324{
325 char *copy = strdup(s);
326 char *save = NULL;
327 char *tok;
328 int n = 0;
329
330 if (copy == NULL) {
331 return -1;
332 }
333 for (tok = strtok_r(copy, ",", &save); tok != NULL && n < max;
334 tok = strtok_r(NULL, ",", &save)) {
335 long v = as_bytes ? (long) brix_parse_bytes(tok) : atol(tok);
336 if (v <= 0) {
337 free(copy);
338 return -1;
339 }
340 out[n++] = v;
341 }
342 free(copy);
343 return n;
344}

385/*
386 * bench_parse_args — decode `bench` options into *ba.
387 * WHY: keeps cmd_bench a linear pipeline (parse → validate → sweep).
388 * HOW: value-taking flags via opt_take; --duration and --count are mutually
389 * exclusive (last one wins, zeroing the other); exactly one positional
390 * URL; unknown options print usage → SX_USAGE.
391 */
392static int
393bench_parse_args(int argc, char **argv, bench_args_t *ba, const char *prog)
394{
395 const char *v;
396 int i;
397
398 for (i = 0; i < argc; i++) {
399 const char *a = argv[i];
400 if (opt_take("--op", argc, argv, &i, &v)) {
401 if (strcmp(v, "read") != 0) {
402 fprintf(stderr, "xrdstorascan: bench phase 1 supports --op read only\n");
403 return SX_USAGE;
404 }
405 } else if (opt_take("--block", argc, argv, &i, &ba->block_s)) {
406 } else if (opt_take("--parallel", argc, argv, &i, &ba->par_s)) {
407 } else if (opt_take("--duration", argc, argv, &i, &v)) {
408 ba->duration_s = atol(v);
409 ba->count = 0;
410 } else if (opt_take("--count", argc, argv, &i, &v)) {
411 ba->count = atol(v);
412 ba->duration_s = 0;
413 } else if (opt_take("--pattern", argc, argv, &i, &ba->pattern)) {
414 } else if (strcmp(a, "--json") == 0) {
415 ba->json = 1;
416 } else if (a[0] == '-') {
417 return usage(prog, SX_USAGE);
418 } else if (ba->url == NULL) {
419 ba->url = a;
420 } else {
421 return usage(prog, SX_USAGE);
422 }
423 }
424 return SX_OK;
425}

427/*

429 * WHY: separates "is the command line sane" from option decoding.

434bench_validate_args(bench_args_t *ba, const char *prog)
435{
436 if (ba->url == NULL) {

480 * WHY: the nested sweep is the run step of the bench pipeline; isolating it

490

495 bench_cell_cfg_t cfg;
496

498 cfg.u = u;
499 cfg.fsize = fsize;

519int
520cmd_bench(int argc, char **argv, const char *prog)
521{
522 bench_args_t ba;
523 int rc;
524 brix_url u;
525 brix_conn c;
526 brix_status st;
527 brix_statinfo sti;
528
529 /* --help as the first subcommand arg → print bench usage to stdout
530 * and exit cleanly (WS-2). */
531 if (argc >= 1 && strcmp(argv[0], "--help") == 0) {
532 return bench_help(prog);
533 }
534
535 memset(&ba, 0, sizeof(ba));
536 ba.block_s = "1M,4M";
537 ba.par_s = "1,8";
538 ba.pattern = "seq";
539 ba.duration_s = 5;
540
541 rc = bench_parse_args(argc, argv, &ba, prog);
542 if (rc != SX_OK) {
543 return rc;
544 }
545 rc = bench_validate_args(&ba, prog);
546 if (rc != SX_OK) {
547 return rc;
548 }
549
550 rc = storascan_connect(ba.url, &u, &c, &st);
551 if (rc != SX_OK) {
552 return rc;
553 }
554 if (brix_stat(&c, u.path, &sti, &st) != 0) {
555 fprintf(stderr, "xrdstorascan: stat %s: %s\n", u.path, st.msg);
556 brix_close(&c);
557 return brix_shellcode(&st);
558 }
559 brix_close(&c); /* workers open their own connections */
560
561 if (!ba.json) {
562 printf("# bench %s size=%lld bytes pattern=%s %s\n",
563 u.path, (long long) sti.size, ba.pattern,
564 ba.count ? "count" : "duration");
565 printf("%-10s %-8s %14s %10s %9s %9s %9s\n",
566 "block", "parallel", "MiB/s", "IOPS", "p50_ms", "p95_ms", "p99_ms");
567 }
568
569 return bench_run_matrix(&ba, &u, sti.size);
570}
