26/* ---- Parse a positive unsigned ceiling ----
27 *
28 * WHAT: Returns zero and fills output, or -1 for malformed/zero/overflow input.
29 * WHY: Resource ceilings must not silently become unlimited after bad input.
30 * HOW: strtoul with full-string, errno and unsigned-range validation.
31 */
32static int
33parse_unsigned(const char *value, unsigned *out)
34{
35 char *end = NULL;
36 unsigned long parsed;
37
38 if (value == NULL || value[0] == '\0') return -1;
39 errno = 0;
40 parsed = strtoul(value, &end, 10);
41 if (errno != 0 || end == value || *end != '\0' || parsed == 0
42 || parsed > UINT_MAX) return -1;
43 *out = (unsigned) parsed;
44 return 0;
45}

131{

133 size_t needed;

371static int
372parse_error_fixture(int argc, char **argv, int *index, xrd_conform_run *run)
373{
374 const char *value = option_value(argc, argv, index);
375 const char *equals;
376 size_t name_length;
377 size_t fixture_index;
378 int is_http = 0;
379 xrd_conform_error_fixture *fixture;
380
381 if (value == NULL
382 || run->error_fixture_count >= XRDC_CONFORM_ERROR_FIXTURE_MAX)
383 return -1;
384 equals = strchr(value, '=');
385 if (equals == NULL || equals == value || equals[1] == '\0') return -1;
386 name_length = (size_t) (equals - value);
387 if (name_length > 5
388 && memcmp(value + name_length - 5, "@http", 5) == 0) {
389 name_length -= 5;
390 is_http = 1;
391 }
392 if (name_length >= sizeof(run->error_fixtures[0].errno_name)
393 || !xrd_conform_error_fixture_name_valid(value, name_length))
394 return -1;
395 for (fixture_index = 0; fixture_index < run->error_fixture_count;
396 fixture_index++)
397 if (run->error_fixtures[fixture_index].is_http == is_http
398 && strlen(run->error_fixtures[fixture_index].errno_name)
399 == name_length
400 && memcmp(run->error_fixtures[fixture_index].errno_name,
401 value, name_length) == 0) return -1;
402 fixture = &run->error_fixtures[run->error_fixture_count++];
403 memcpy(fixture->errno_name, value, name_length);
404 fixture->errno_name[name_length] = '\0';
405 fixture->target = equals + 1;
406 fixture->is_http = is_http;
407 return 1;
408}
