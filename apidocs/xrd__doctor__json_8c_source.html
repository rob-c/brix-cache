8/* Emit a JSON string literal for `s` (escaped, NULL → ""). */
9void
10xrd_json_str(FILE *f, const char *s)
11{
12 fputc('"', f);
13 for (; s != NULL && *s != '\0'; s++) {
14 unsigned char ch = (unsigned char) *s;
15 if (ch == '"' || ch == '\\') { fputc('\\', f); fputc((int) ch, f); }
16 else if (ch == '\n') { fputs("\\n", f); }
17 else if (ch == '\t') { fputs("\\t", f); }
18 else if (ch == '\r') { fputs("\\r", f); }
19 else if (ch < 0x20) { fprintf(f, "\\u%04x", (unsigned) ch); }
20 else { fputc((int) ch, f); }
21 }
22 fputc('"', f);
23}

26/*
27 * WHAT: JSON emit — the connect/identity head of the doctor report (endpoint,
28 * port, connected, connect_error, role, server_flags, auth, sec_list).
29 * WHY: extracted from xrd_doctor_json (phase-72 H4) so each report section is
30 * one small renderer; the emitted bytes are frozen (CLI UX contract).
31 * HOW: writes directly to stdout in the exact original order/punctuation;
32 * connect_error appears only for a failed connect with a message.
33 */
34static void
35doctor_json_head(const xrd_probe *p)
36{
37 printf("{\n");
38 printf(" \"endpoint\": "); xrd_json_str(stdout, p->host);
39 printf(",\n \"port\": %d,\n", p->port);
40 printf(" \"connected\": %s,\n", p->connected ? "true" : "false");
41 if (!p->connected && p->err[0] != '\0') {
42 printf(" \"connect_error\": "); xrd_json_str(stdout, p->err); printf(",\n");
43 }
44 printf(" \"role\": "); xrd_json_str(stdout, xrd_role_str(p->server_flags));
45 printf(",\n \"server_flags\": %u,\n", (unsigned) p->server_flags);
46 printf(" \"auth\": "); xrd_json_str(stdout, p->connected ? p->auth : "");
47 printf(",\n \"sec_list\": "); xrd_json_str(stdout, p->sec_list); printf(",\n");
48}

68/*
69 * WHAT: JSON emit — the "cert" object (subject/issuer/SAN/validity/verdicts),
70 * or the literal `null` when no server certificate was captured.
71 * WHY: the cert block is the largest single section of the old branch ladder;
72 * isolating it removes most of xrd_doctor_json's complexity.
73 * HOW: formats the validity epochs locally (the buffers were only ever used
74 * for this section) and emits fields in the original fixed order.
75 */
76static void
77doctor_json_cert(const xrd_probe *p)
78{
79 char nb[32], na[32];
80
81 printf(" \"cert\": ");
82 if (!p->cert.have) {
83 printf("null,\n");
84 return;
85 }
86 xrd_fmt_epoch(p->cert.not_before, nb, sizeof(nb));
87 xrd_fmt_epoch(p->cert.not_after, na, sizeof(na));
88 printf("{\n \"subject\": "); xrd_json_str(stdout, p->cert.subject);
89 printf(",\n \"issuer\": "); xrd_json_str(stdout, p->cert.issuer);
90 printf(",\n \"sans\": "); xrd_json_str(stdout, p->cert.sans);
91 printf(",\n \"not_before\": %ld", p->cert.not_before);
92 printf(",\n \"not_before_utc\": "); xrd_json_str(stdout, nb);
93 printf(",\n \"not_after\": %ld", p->cert.not_after);
94 printf(",\n \"not_after_utc\": "); xrd_json_str(stdout, na);
95 printf(",\n \"days_left\": %ld", p->cert.days_left);
96 printf(",\n \"expired\": %s", p->cert.expired ? "true" : "false");
97 printf(",\n \"not_yet_valid\": %s", p->cert.not_yet_valid ? "true" : "false");
98 printf(",\n \"host_match\": %s", p->cert.host_match ? "true" : "false");
99 printf(",\n \"self_signed\": %s\n },\n", p->cert.self_signed ? "true" : "false");
100}

164/*
165 * WHAT: JSON emit — one functional-battery object inside the "tests" array
166 * (endpoint/protocol/reachability, pass counts, per-check results).
167 * WHY: splitting the per-battery body out of the array loop keeps both under
168 * the complexity gate while preserving the exact nesting/punctuation.
169 * HOW: `idx` drives the leading-comma idiom for the enclosing array; the
170 * "error" field appears only for an unreachable battery with a message.
171 */
172static void
173doctor_json_battery(const xrd_battery *bt, int idx)
174{
175 int j;
176
177 printf("%s\n {\n", idx ? "," : "");
178 printf(" \"endpoint\": "); xrd_json_str(stdout, bt->endpoint);
179 printf(",\n \"protocol\": "); xrd_json_str(stdout, bt->protocol);
180 printf(",\n \"reachable\": %s", bt->reachable ? "true" : "false");
181 if (!bt->reachable && bt->err[0] != '\0') {
182 printf(",\n \"error\": "); xrd_json_str(stdout, bt->err);
183 }
184 printf(",\n \"passed\": %d, \"failed\": %d, \"skipped\": %d",
185 bt->npass, bt->nfail, bt->nskip);
186 printf(",\n \"checks\": [");
187 for (j = 0; j < bt->n; j++) {
188 printf("%s\n { \"name\": ", j ? "," : "");
189 xrd_json_str(stdout, bt->checks[j].name);
190 printf(", \"status\": ");
191 xrd_json_str(stdout, bt->checks[j].skipped ? "skip"
192 : (bt->checks[j].ok ? "pass" : "fail"));
193 printf(", \"detail\": ");
194 xrd_json_str(stdout, bt->checks[j].detail);
195 printf(" }");
196 }
197 printf("%s]\n }", bt->n ? "\n " : "");
198}
