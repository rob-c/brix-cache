28/* default port of a scheme (1080 = libcurl's proxy default) */
29static int
30pin_default_port(const char *target, size_t scheme_len, int proxy)
31{
32 if (scheme_len == 5 && strncasecmp(target, "https", 5) == 0) {
33 return 443;
34 }
35 return proxy ? 1080 : 80;

39/* host + port of a URL or "host[:port]" target; 0 / -1 (malformed) */
40static int
41pin_split(const char *target, int proxy, char *host, size_t hostsz, int *port)
42{
43 const char *p = target, *sep = strstr(target, "://");
44 const char *end, *colon, *at;
45 size_t scheme_len = 0, hlen;
46
47 if (sep != NULL) {
48 scheme_len = (size_t) (sep - target);
49 p = sep + 3;
50 }
51 end = p + strcspn(p, "/?#");
52 at = memchr(p, '@', (size_t) (end - p));
53 if (at != NULL) {
54 p = at + 1; /* drop user:pw@ */
55 }
56 *port = pin_default_port(target, scheme_len, proxy);
57 if (*p == '[') { /* [v6]:port */
58 const char *close = memchr(p, ']', (size_t) (end - p));
59 if (close == NULL) {
60 return -1;
61 }
62 hlen = (size_t) (close - p - 1);
63 p++;
64 colon = (close + 1 < end && close[1] == ':') ? close + 1 : NULL;
65 } else {
66 colon = memchr(p, ':', (size_t) (end - p));
67 hlen = colon != NULL ? (size_t) (colon - p) : (size_t) (end - p);
68 }
69 if (hlen == 0 || hlen >= hostsz) {
70 return -1;
71 }
72 if (colon != NULL) {
73 char *e = NULL;
74 long v = strtol(colon + 1, &e, 10);
75 if (e == colon + 1 || e != end || v <= 0 || v > 65535) {
76 return -1;
77 }
78 *port = (int) v;
79 }
80 memcpy(host, p, hlen);
81 host[hlen] = '\0';
82 return 0;
83}

96/* "host:port:ip1,ip2" for one target; 1 = entry written, 0 = literal (no
97 * entry needed), -1 = the name has no address */
98static int
99pin_entry(const char *target, int proxy, char *out, size_t outsz)
100{
101 char host[PIN_HOST_MAX], ip[BRIX_RESOLVE_NTOP_LEN];
102 int port, n, i;
103 size_t off;
104 brix_resolve_addr addrs[BRIX_RESOLVE_MAX];
105
106 if (pin_split(target, proxy, host, sizeof(host), &port) != 0) {
107 return -1;
108 }
109 if (pin_is_literal(host)) {
110 return 0;
111 }
112 n = brix_resolve(host, port, brix_netpref_family(), SOCK_STREAM,
113 BRIX_RESOLVE_ADDRCONFIG, addrs, BRIX_RESOLVE_MAX, NULL);
114 if (n <= 0) {
115 return -1;
116 }
117 off = (size_t) snprintf(out, outsz, "%s:%d:", host, port);
118 for (i = 0; i < n && off < outsz; i++) {
119 brix_resolve_ntop((struct sockaddr *) &addrs[i].ss, addrs[i].len,
120 ip, sizeof(ip));
121 off += (size_t) snprintf(out + off, outsz - off,
122 addrs[i].family == AF_INET6 ? "%s[%s]" : "%s%s",
123 i > 0 ? "," : "", ip);
124 }
125 return off < outsz ? 1 : -1;
126}

129/* resolve list for the hop: the target and, when set, the proxy */
130static CURLcode
131pin_build(const cvmfs_curl_transfer *t, const char *url,
132 struct curl_slist **list)
133{
134 char entry[PIN_ENTRY_MAX];
135 int rc;
136
137 *list = NULL;
138 if (t->proxy != NULL && t->proxy[0] != '\0') {
139 rc = pin_entry(t->proxy, 1, entry, sizeof(entry));
140 if (rc < 0) {
141 return CURLE_COULDNT_RESOLVE_PROXY;
142 }
143 if (rc > 0) {
144 *list = curl_slist_append(*list, entry);
145 }
146 }
147 rc = pin_entry(url, 0, entry, sizeof(entry));
148 if (rc < 0) {
149 curl_slist_free_all(*list);
150 *list = NULL;
151 return CURLE_COULDNT_RESOLVE_HOST;
152 }
153 if (rc > 0) {
154 *list = curl_slist_append(*list, entry);
155 }
156 return CURLE_OK;
157}

160/* one hop: pin, perform, report the redirect target (owned by libcurl until
161 * the next perform — the caller copies it) */
162static CURLcode
163pin_attempt(CURL *c, const cvmfs_curl_transfer *t, const char *url,
164 char *next, size_t nextsz)
165{
166 struct curl_slist *list = NULL;
167 char *redir = NULL;
168 CURLcode rc;
169
170 next[0] = '\0';
171 rc = pin_build(t, url, &list);
172 if (rc != CURLE_OK) {
173 return rc;
174 }
175 curl_easy_setopt(c, CURLOPT_RESOLVE, list);
176 curl_easy_setopt(c, CURLOPT_URL, url);
177 if (t->on_hop != NULL) {
178 t->on_hop(t->ud);
179 }
180 rc = curl_easy_perform(c);
181 if (rc == CURLE_OK
182 && curl_easy_getinfo(c, CURLINFO_REDIRECT_URL, &redir) == CURLE_OK
183 && redir != NULL)
184 {
185 snprintf(next, nextsz, "%s", redir);
186 }
187 curl_easy_setopt(c, CURLOPT_RESOLVE, NULL); /* before the free */
188 curl_slist_free_all(list);
189 return rc;
190}
