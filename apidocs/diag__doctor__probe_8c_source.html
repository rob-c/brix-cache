4 * WHAT: the connection-level diagnosis primitives shared by every endpoint

20/* Stream a remote file through the live connection into fd; returns 0 / -1 and
21 * sets *out_bytes to the number of bytes written. Reuses the authenticated conn
22 * (no reconnect), so it measures the data path itself. */
23int
24download_to_fd(brix_conn *c, const char *path, int fd, int64_t *out_bytes,
25 brix_status *st)
26{
27 brix_file f;
28 int64_t off = 0;
29 char *buf;
30
31 if (brix_file_open_read(c, path, &f, st) != 0) {
32 return -1;
33 }
34 buf = (char *) malloc(1u << 20);
35 if (buf == NULL) {
36 brix_file_close(c, &f, st);
37 brix_status_set(st, XRDC_EPROTO, 0, "download: out of memory");
38 return -1;
39 }
40 for (;;) {
41 ssize_t r = brix_file_read(c, &f, off, buf, 1u << 20, st);
42 ssize_t w = 0;
43 if (r < 0) {
44 free(buf);
45 brix_file_close(c, &f, st);
46 return -1;
47 }
48 if (r == 0) {
49 break;
50 }
51 while (w < r) {
52 ssize_t k = write(fd, buf + w, (size_t) (r - w));
53 if (k < 0) {
54 free(buf);
55 brix_file_close(c, &f, st);
56 brix_status_set(st, XRDC_ESOCK, 0, "download: local write failed");
57 return -1;
58 }
59 w += k;
60 }
61 off += r;
62 }
63 free(buf);
64 if (out_bytes != NULL) {
65 *out_bytes = off;
66 }
67 return brix_file_close(c, &f, st);
68}

88/* Throughput probe over an established conn: TTFB (first read) + MB/s (whole file). */
89int
90doctor_xfer(brix_conn *c, const char *path, double *ttfb_ms, double *mbps,
91 int64_t *bytes)
92{
93 brix_file f;
94 brix_status st;
95 uint8_t *buf;
96 int64_t off = 0;
97 uint64_t t0, tf = 0, t1;
98
99 brix_status_clear(&st);
100 if (brix_file_open_read(c, path, &f, &st) != 0) {
101 return -1;
102 }
103 buf = (uint8_t *) malloc(1u << 20);
104 if (buf == NULL) {
105 brix_file_close(c, &f, &st);
106 return -1;
107 }
108 t0 = brix_mono_ns();
109 for (;;) {
110 ssize_t r = brix_file_read(c, &f, off, buf, 1u << 20, &st);
111 if (r < 0) { free(buf); brix_file_close(c, &f, &st); return -1; }
112 if (tf == 0) { tf = brix_mono_ns(); } /* time-to-first-byte */
113 if (r == 0) { break; }
114 off += r;
115 }
116 t1 = brix_mono_ns();
117 free(buf);
118 brix_file_close(c, &f, &st);
119 *ttfb_ms = (double) (tf - t0) / 1e6;
120 *bytes = off;
121 {
122 double secs = (double) (t1 - t0) / 1e9;
123 if (secs <= 0.0) { secs = 1e-9; }
124 *mbps = (double) off / 1e6 / secs;
125 }
126 return 0;
127}

130/* /metrics signal: reachable? + any kXR_wait/budget shedding gauge nonzero. */
131void
132doctor_metrics(const char *host, int port, doctor_ep *e)
133{
134 char *body;
135 brix_status st;
136 int http = 0;
137 char *line, *save;
138
139 body = (char *) malloc(1u << 20);
140 if (body == NULL) {
141 return;
142 }
143 brix_status_clear(&st);
144 if (brix_http_get(host, port, "/metrics", 4000, &http, body, 1u << 20, NULL,
145 &st) != 0) {
146 free(body);
147 return;
148 }
149 e->metrics_http = http;
150 for (line = strtok_r(body, "\n", &save); line != NULL;
151 line = strtok_r(NULL, "\n", &save)) {
152 if (line[0] == '#') {
153 continue;
154 }
155 if (strstr(line, "kXR_wait") != NULL || strstr(line, "_wait_") != NULL
156 || strstr(line, "budget") != NULL || strstr(line, "shed") != NULL) {
157 /* nonzero counter at the end of the line ⇒ active shedding */
158 char *sp = strrchr(line, ' ');
159 if (sp != NULL && strtod(sp + 1, NULL) > 0.0) {
160 e->shedding = 1;
161 }
162 }
163 }
164 free(body);
165}

168/*
169 * WHAT: forged-credential rejection probes (bad-signature + alg=none JWTs).
170 * WHY: extracted from doctor_auth_suite for the complexity gate; the
171 * forged-token matrix is one self-contained concern.
172 * HOW: build each forged JWT with dx_make_jwt and assert rejection via
173 * dx_authz_forged. Only called when the server advertises ztn.
174 */
175static void
176auth_suite_forged_probes(const diag_args *a, const brix_url *u, doctor_ep *e)
177{
178 char fsig[1024], fnone[1024];
179
180 /* No kid: a kid the server doesn't know would be rejected at key SELECTION,
181 * short-circuiting the signature check — we want the server to reach
182 * signature verification (and reject the garbage sig) so the test actually
183 * exercises signature verification on typical single-key deployments. */
184 if (dx_make_jwt(
185 "{\"alg\":\"RS256\",\"typ\":\"JWT\"}",
186 "{\"iss\":\"https://xrddiag.invalid\",\"sub\":\"xrddiag-probe\","
187 "\"scope\":\"storage.read:/ storage.modify:/\",\"exp\":4102444800}",
188 "ZHVtbXktc2lnbmF0dXJlLW5vdC12YWxpZA", fsig, sizeof(fsig)) == 0) {
189 dx_authz_forged(a, u, "authz-forgesig", fsig, e);
190 }
191 if (dx_make_jwt(
192 "{\"alg\":\"none\",\"typ\":\"JWT\"}",
193 "{\"iss\":\"https://xrddiag.invalid\",\"sub\":\"xrddiag-probe\","
194 "\"scope\":\"storage.read:/ storage.modify:/\",\"exp\":4102444800}",
195 "", fnone, sizeof(fnone)) == 0) {
196 dx_authz_forged(a, u, "authz-algnone", fnone, e);
197 }
198}

201/*
202 * WHAT: auth-suite probes that need the operator's real token (expired-token
203 * rejection + read-only-scope write denial).
204 * WHY: extracted from doctor_auth_suite for the complexity gate.
205 * HOW: discover the ambient token, gate each probe on the token's metadata
206 * (the write-scope assertion is gated like the write probe), free it.
207 */
208static void
209auth_suite_token_probes(const diag_args *a, const brix_url *u, int ztn_adv,
210 doctor_ep *e)
211{
212 char *tok = brix_token_discover();
213 brix_token_meta m;
214
215 if (tok == NULL) {
216 return;
217 }
218 brix_token_meta_get(tok, &m);
219 if (ztn_adv && m.valid && m.expired) {
220 dx_authz_expired(a, u, tok, e);
221 }
222 if (a->allow_write && ztn_adv && m.valid && m.has_scope
223 && m.has_read && !m.has_write
224 && (dx_is_loopback(e->host) || a->authorized)) {
225 dx_authz_scope(a, u, tok, e);
226 }
227 free(tok);
228}

231/*
232 * The full auth/permissions suite. Opens its own scoped connections (the credential
233 * matrix) — does not reuse the primary. Read-only assertions always run; the
234 * write-scope assertion is gated like the write probe. PII-free: only verdicts +
235 * kXR codes are recorded, never token/cert/scope contents.
236 */
237void
238doctor_auth_suite(const diag_args *a, const brix_url *u, const char *target,
239 int have_target, doctor_ep *e)
240{
241 char sec_list[256];
242 int ztn_adv;
243
244 /* 1) anonymous access must be denied on an auth-required server — this also
245 * discovers the server's advertised auth (&P=) from a force_anon session. */
246 dx_authz_anon(a, u, target, have_target, sec_list, sizeof(sec_list), e);
247 ztn_adv = (strstr(sec_list, "ztn") != NULL);
248
249 /* 2,3) forged-credential rejection — only if the server takes bearer tokens. */
250 if (ztn_adv) {
251 auth_suite_forged_probes(a, u, e);
252 } else {
253 dx_record(e, &(dx_note){ "authz-token", DX_OK, 0,
254 "server does not offer bearer-token auth (forged-token tests N/A)", "" });
255 }
256
257 /* 4,5) tests that require the operator's real token. */
258 auth_suite_token_probes(a, u, ztn_adv, e);
259}

262/*
263 * Run the active-diagnosis battery over an already-open connection. Read-only probes
264 * always run; write/stage probes run only when --allow-write is set AND the target is
265 * loopback or the operator passed --i-am-authorized (mutations on a remote server).
266 */
267void
268doctor_diagnose(const diag_args *a, brix_conn *c, const brix_url *u,
269 const dx_target *t, doctor_ep *e)
270{
271 dx_probe_auth(c, e);
272 dx_probe_namespace(c, e);
273 if (t->have) {
274 dx_probe_read(c, t->path, e);
275 dx_probe_checksum(c, t->path, e);
276 }
277 {
278 char loc[2048];
279 brix_status lst;
280 brix_status_clear(&lst);
281 if (brix_locate(c, u->path[0] ? u->path : "/", loc, sizeof(loc), &lst) != 0) {
282 dx_record_status(e, "locate", &lst);
283 } else {
284 char *t, *save;
285 for (t = strtok_r(loc, " \t\r\n", &save); t != NULL;
286 t = strtok_r(NULL, " \t\r\n", &save)) {
287 if (t[0] == 'S') { e->holders++; }
288 }
289 if (e->holders == 0) {
290 dx_record(e, &(dx_note){ "locate", DX_WARN, 0, "no replica located for the path",
291 "check data-server health and the CMS/manager registry" });
292 } else {
293 dx_record(e, &(dx_note){ "locate", DX_OK, 0, "replica(s) located", "" });
294 }
295 }
296 }
297 if (a->allow_write) {
298 int permitted = dx_is_loopback(e->host) || a->authorized;
299 if (!permitted) {
300 dx_record(e, &(dx_note){ "write", DX_WARN, 0,
301 "write probe skipped on a non-loopback host without --i-am-authorized",
302 "re-run with --i-am-authorized to actively probe the write path" });
303 } else {
304 dx_probe_write(c, e);
305 if (t->have && e->offline_seen) {
306 dx_probe_stage(c, t->path, e);
307 }
308 }
309 }
310 if (a->auth_suite) {
311 doctor_auth_suite(a, u, t->path, t->have, e);
312 }
313}
