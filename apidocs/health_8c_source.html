1/*
2 * health.c — content handler for the `brix_health on;` location (phase-47 W2).
3 *
4 * WHAT: ngx_http_brix_health_handler() serves GET/HEAD /healthz as a small
5 * JSON document so an external load balancer or a Kubernetes liveness/
6 * readiness probe has a cheap endpoint to poll. Liveness is implicit: if
7 * this worker can accept the connection and run the handler it returns 200;
8 * the `time`/`time_epoch` fields (built per request from nginx's cached
9 * clock) prove the answer is freshly generated, not replayed by a cache.
10 * `?verbose` adds cheap, non-secret readiness signals (metrics SHM mapped,
11 * worker pid, nginx version) plus the service surface: every bound listen
12 * socket (addr/port/plane/open, with auth + connection counters for root://
13 * listeners) and every registered export (path + backend).
14 *
15 * WHY: The module already exposes /metrics (scrape) and the dashboard, but
16 * neither is a clean probe target — /metrics is a large text body and the
17 * dashboard needs auth. A dedicated 200/JSON endpoint keeps probes light and
18 * lets an operator wire `livenessProbe: GET /healthz` directly. It lives in
19 * the metrics module (not a new .so) and mirrors the read-only SRR/metrics
20 * handlers: method-gated, body discarded, no request input affects output.
21 *
22 * HOW: classify ?verbose → build the JSON in the request pool with ngx_snprintf
23 * (no jansson; the document is tiny and fixed-shape) → set status/headers →
24 * send_header → output the single memory buffer. HEAD stops after the header.
25 * Reads only process globals (ngx_pid) and the metrics SHM zone pointer; it
26 * never touches the request body and never emits a secret.
27 */

52/* ---- Which nginx plane owns a listening socket ----
53 *
54 * WHAT: Returns "http", "stream", or "other" for a bound listening socket.
55 *
56 * WHY: The endpoint list reports every bound socket; the plane tag is what
57 * lets a prober match a row to the protocol tables (root:// listeners are
58 * stream, WebDAV/S3/CVMFS/metrics are http) without knowing port layout.
59 *
60 * HOW: 1. Compare ls->handler against the two public init-connection entry
61 * points nginx binds at listen time.
62 * 2. Anything else (no third family is configured today) is "other".
63 */
64static const char *
65health_listener_layer(const ngx_listening_t *ls)
66{
67 if (ls->handler == ngx_http_init_connection) {
68 return "http";
69 }
70 if (ls->handler == ngx_stream_init_connection) {
71 return "stream";
72 }
73 return "other";
74}

77/* ---- SHM metrics slot for a stream listen port ----
78 *
79 * WHAT: Returns the in-use per-server metrics slot whose listen port equals
80 * `port`, or NULL when no stream server block owns that port.
81 *
82 * WHY: The slot already carries a root:// listener's identity (auth flavour)
83 * and its liveness evidence (connection counters); joining by port lets the
84 * endpoint list surface both without any new bookkeeping on the data path.
85 *
86 * HOW: 1. Linear scan of the fixed slot array (BRIX_METRICS_MAX_SERVERS).
87 * 2. First slot with in_use set and a matching port wins (listen ports
88 * are unique across server blocks by construction).
89 */
90static ngx_brix_srv_metrics_t *
91health_stream_slot(ngx_brix_metrics_t *m, ngx_uint_t port)
92{
93 ngx_uint_t slot;
94
95 if (m == NULL) {
96 return NULL;
97 }
98 for (slot = 0; slot < BRIX_METRICS_MAX_SERVERS; slot++) {
99 if (m->servers[slot].in_use && m->servers[slot].port == port) {
100 return &m->servers[slot];
101 }
102 }
103 return NULL;
104}

107/* ---- Append the ,"endpoints":[...] array (verbose) ----
108 *
109 * WHAT: Appends one JSON row per bound listening socket — address text, port,
110 * owning plane, and whether the fd is open — plus proto/auth/connection
111 * counters for stream listeners that own a metrics slot. Returns the new
112 * write cursor (never past `last`; ngx_snprintf is bounds-clamped).
113 *
114 * WHY: This is the "which endpoints are still alive" half of the document: an
115 * open fd reported by a worker that answered this request is being polled by
116 * a live event loop, and the monotonic connection counters let a prober
117 * distinguish a quiet-but-alive port from a dead one across two polls.
118 *
119 * HOW: 1. Walk ngx_cycle->listening, the ground truth of bound sockets.
120 * 2. Emit addr/port/layer/open for every socket.
121 * 3. For stream sockets, join the SHM slot by port and append the root://
122 * identity and counters.
123 */
124static u_char *
125health_append_endpoints(u_char *p, u_char *last, ngx_brix_metrics_t *m)
126{
127 ngx_cycle_t *cycle = (ngx_cycle_t *) ngx_cycle;
128 ngx_listening_t *ls = cycle->listening.elts;
129 ngx_uint_t i;
130 int first = 1;
131
132 p = ngx_snprintf(p, (size_t) (last - p), ",\"endpoints\":[");
133
134 for (i = 0; i < cycle->listening.nelts; i++) {
135 ngx_uint_t port;
136 ngx_brix_srv_metrics_t *slot;
137
138 port = (ngx_uint_t) ngx_inet_get_port(ls[i].sockaddr);
139 p = ngx_snprintf(p, (size_t) (last - p),
140 "%s{\"addr\":\"%V\",\"port\":%ui,\"layer\":\"%s\","
141 "\"open\":%s",
142 first ? "" : ",", &ls[i].addr_text, port,
143 health_listener_layer(&ls[i]),
144 ls[i].fd != (ngx_socket_t) -1 ? "true" : "false");
145 first = 0;
146
147 slot = health_stream_slot(m, port);
148 if (slot != NULL) {
149 p = ngx_snprintf(p, (size_t) (last - p),
150 ",\"proto\":\"root\",\"auth\":\"%s\","
151 "\"connections_active\":%uL,"
152 "\"connections_total\":%uL",
153 slot->auth,
154 (uint64_t) slot->connections_active,
155 (uint64_t) slot->connections_total);
156 }
157 p = ngx_snprintf(p, (size_t) (last - p), "}");
158 }
159 return ngx_snprintf(p, (size_t) (last - p), "]");
160}

163/* ---- Append the ,"exports":[...] array (verbose) ----
164 *
165 * WHAT: Appends one row per registered export — its canonical root path and
166 * the configured storage backend. Returns the new write cursor.
167 *
168 * WHY: The endpoint rows say which ports answer; the export rows say which
169 * namespace paths those ports serve — together they are the whole service
170 * surface. Only config-time identity is emitted: no credentials, no origin
171 * hosts (those stay in cvmfs_origins) and no live probing.
172 *
173 * HOW: 1. Iterate the registry (config-time count, stable per process).
174 * 2. Emit root_canon + backend for each entry export_info() fills.
175 */
176static u_char *
177health_append_exports(u_char *p, u_char *last)
178{
179 ngx_uint_t i, n_exports = brix_vfs_backend_export_count();
180 int first = 1;
181
182 p = ngx_snprintf(p, (size_t) (last - p), ",\"exports\":[");
183 for (i = 0; i < n_exports; i++) {
184 brix_vfs_backend_info_t info;
185
186 if (brix_vfs_backend_export_info(i, &info) != NGX_OK) {
187 continue;
188 }
189 p = ngx_snprintf(p, (size_t) (last - p),
190 "%s{\"path\":\"%s\",\"backend\":\"%s\"}",
191 first ? "" : ",", info.root_canon, info.backend);
192 first = 0;
193 }
194 return ngx_snprintf(p, (size_t) (last - p), "]");
195}

198/* ---- Verbose buffer headroom for the endpoint/export arrays ----
199 *
200 * WHAT: Returns the extra bytes the verbose document needs on top of the
201 * fixed base capacity.
202 *
203 * WHY: The base document is fixed-shape, but listeners and exports scale with
204 * the deployment; sizing from the live counts keeps ngx_snprintf's clamped
205 * writes from truncating the JSON mid-array on a large fleet.
206 *
207 * HOW: 1. 256 bytes bounds a listener row (NGX_SOCKADDR_STRLEN plus the fixed
208 * keys and two 20-digit counters).
209 * 2. Export rows are path-dominated: root strlen + fixed key overhead.
210 */
211static size_t
212health_verbose_extra(void)
213{
214 ngx_cycle_t *cycle = (ngx_cycle_t *) ngx_cycle;
215 ngx_uint_t i, n_exports = brix_vfs_backend_export_count();
216 size_t extra;
217
218 extra = 256 * cycle->listening.nelts;
219 for (i = 0; i < n_exports; i++) {
220 brix_vfs_backend_info_t info;
221
222 if (brix_vfs_backend_export_info(i, &info) == NGX_OK) {
223 extra += ngx_strlen(info.root_canon) + 64;
224 }
225 }
226 return extra;
227}

230/*
231 * Render the health document into a pool buffer. Non-verbose is a fixed
232 * liveness line; verbose appends the readiness object. Returns the buffer (and
233 * its length via *len), or NULL on allocation failure.
234 */
235static u_char *
236health_build_json(ngx_http_request_t *r, ngx_uint_t verbose, size_t *len)
237{
238 size_t cap = 2048;
239 u_char *buf;
240 u_char *p;
241 const char *shm_state;
242 char now_iso[32];
243 ngx_brix_metrics_t *m;
244 ngx_uint_t generation = 0;
245 uint64_t config_hash = 0;
246
247 if (verbose) {
248 cap += health_verbose_extra();
249 }
250 BRIX_PNALLOC_OR_RETURN(buf, r->pool, cap, NULL);
251
252 /* Built per request (not a compile-time constant), so two polls that
253 * return different times prove the handler — not a cache — answered. */
254 brix_format_iso8601(ngx_time(), now_iso, sizeof(now_iso));
255
256 /* The metrics SHM zone is created at config time (metrics/config.c) and is
257 * the module's shared state — "mapped" once a stream server block exists.
258 * When mapped it also carries the config/reload fingerprint published by the
259 * master in init_module (brix_config_version_publish). */
260 m = (ngx_brix_shm_zone != NULL) ? ngx_brix_shm_zone->data : NULL;
261 if (m != NULL) {
262 generation = (ngx_uint_t) m->config_generation;
263 config_hash = m->config_hash;
264 }
265
266 if (!verbose) {
267 /* config_generation/config_version are cheap and reload-relevant, so
268 * they ride on the default (non-verbose) document: a probe can confirm a
269 * reload took effect without opting into the heavier readiness block. */
270 p = ngx_snprintf(buf, cap,
271 "{\"status\":\"ok\",\"service\":\"" BRIX_SERVER_NAME "\","
272 "\"version\":\"" BRIX_SERVER_VERSION "\","
273 "\"time\":\"%s\",\"time_epoch\":%T,"
274 "\"config_generation\":%ui,"
275 "\"config_version\":\"%016xL\"}\n",
276 now_iso, ngx_time(), generation, config_hash);
277 *len = (size_t) (p - buf);
278 return buf;
279 }
280
281 shm_state = (m != NULL) ? "mapped" : "unmapped";
282
283 p = ngx_snprintf(buf, cap,
284 "{\"status\":\"ok\",\"service\":\"" BRIX_SERVER_NAME "\","
285 "\"version\":\"" BRIX_SERVER_VERSION "\","
286 "\"time\":\"%s\",\"time_epoch\":%T,"
287 "\"config_generation\":%ui,"
288 "\"config_version\":\"%016xL\","
289 "\"checks\":{"
290 "\"metrics_shm\":\"%s\","
291 "\"worker_pid\":%P,"
292 "\"nginx_version\":\"%s\""
293 "}",
294 now_iso, ngx_time(), generation, config_hash,
295 shm_state, ngx_pid, NGINX_VERSION);
296
297 /* The service surface: every bound listen socket, then every export the
298 * VFS registry knows. Both read config-time/process state only. */
299 p = health_append_endpoints(p, buf + cap, m);
300 p = health_append_exports(p, buf + cap);
301
302 /* phase-68: per-origin health of every http-backed export (the CVMFS
303 * Stratum-1 sets). fail_score is the sd_http EWMA (0 = healthy). */
304 {
305 ngx_uint_t i, n_exports = brix_vfs_backend_export_count();
306 int first = 1;
307
308 p = ngx_snprintf(p, (size_t) (buf + cap - p), ",\"cvmfs_origins\":[");
309 for (i = 0; i < n_exports; i++) {
310 brix_vfs_backend_info_t info;
311 brix_sd_instance_t *inst;
312 char hosts[SD_HTTP_EP_MAX][256];
313 int ports[SD_HTTP_EP_MAX];
314 int scores[SD_HTTP_EP_MAX];
315 int j, n;
316
317 if (brix_vfs_backend_export_info(i, &info) != NGX_OK
318 || ngx_strcmp(info.backend, "http") != 0)
319 {
320 continue;
321 }
322 inst = brix_vfs_backend_resolve(info.root_canon,
323 r->connection->log);
324 while (inst != NULL
325 && ngx_strcmp(inst->driver->name, "http") != 0)
326 {
327 inst = brix_sd_cache_source_instance(inst);
328 }
329 n = sd_http_health_snapshot(inst, hosts, ports, scores,
330 SD_HTTP_EP_MAX);
331 for (j = 0; j < n; j++) {
332 p = ngx_snprintf(p, (size_t) (buf + cap - p),
333 "%s{\"host\":\"%s\",\"port\":%d,\"fail_score\":%d}",
334 first ? "" : ",", hosts[j], ports[j], scores[j]);
335 first = 0;
336 }
337 }
338 p = ngx_snprintf(p, (size_t) (buf + cap - p), "]}\n");
339 }
340 *len = (size_t) (p - buf);
341 return buf;
342}

345ngx_int_t
346ngx_http_brix_health_handler(ngx_http_request_t *r)
347{
348 ngx_http_brix_metrics_loc_conf_t *lcf;
349 ngx_int_t rc;
350 u_char *buf;
351 size_t len;
352 ngx_buf_t *b;
353 ngx_chain_t out;
354
355 lcf = ngx_http_get_module_loc_conf(r, ngx_http_brix_metrics_module);
356 if (!lcf->health) {
357 return NGX_DECLINED;
358 }
359
360 /* AGPL-3.0 sec.13: offer remote users the source (X-Source header). */
361 brix_http_source_offer(r);
362
363 if (r->method != NGX_HTTP_GET && r->method != NGX_HTTP_HEAD) {
364 return NGX_HTTP_NOT_ALLOWED;
365 }
366
367 rc = ngx_http_discard_request_body(r);
368 if (rc != NGX_OK) {
369 return rc;
370 }
371
372 buf = health_build_json(r, health_verbose_requested(r), &len);
373 if (buf == NULL) {
374 return NGX_HTTP_INTERNAL_SERVER_ERROR;
375 }
376
377 r->headers_out.status = NGX_HTTP_OK;
378 r->headers_out.content_length_n = (off_t) len;
379
380 {
381 ngx_str_t ct = ngx_string("application/json");
382 r->headers_out.content_type = ct;
383 r->headers_out.content_type_len = ct.len;
384 r->headers_out.content_type_lowcase = NULL;
385 }
386
387 rc = ngx_http_send_header(r);
388 if (rc == NGX_ERROR || rc > NGX_OK || r->header_only) {
389 return rc;
390 }
391
392 BRIX_PCALLOC_OR_RETURN(b, r->pool, sizeof(*b), NGX_HTTP_INTERNAL_SERVER_ERROR);
393 b->pos = b->start = buf;
394 b->last = b->end = buf + len;
395 b->memory = 1;
396 b->last_buf = 1;
397
398 out.buf = b;
399 out.next = NULL;
400 return ngx_http_output_filter(r, &out);
401}
