29/*
30 * WHAT: Validate and copy all configured GSI file-system paths.
31 * WHY: Bound every NUL-terminated path before passing it to libc/OpenSSL.
32 * HOW: Require cert/key/CA, check fixed-buffer limits, then copy each value.
33 */
34static ngx_int_t
35brix_ftp_gsi_paths(ngx_conf_t *cf, ngx_stream_brix_ftp_srv_conf_t *conf,
36 char *cert, size_t cert_cap, char *key, size_t key_cap,
37 char *ca, size_t ca_cap)
38{
39 if (conf->common.certificate.len == 0 || conf->common.certificate_key.len == 0
40 || conf->common.trusted_ca.len == 0)
41 {
42 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
43 "brix_gridftp_gsi requires brix_certificate, "
44 "brix_certificate_key and brix_trusted_ca");
45 return NGX_ERROR;
46 }
47 if (conf->common.certificate.len >= cert_cap
48 || conf->common.certificate_key.len >= key_cap
49 || conf->common.trusted_ca.len >= ca_cap)
50 {
51 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
52 "brix_gridftp_gsi certificate/key/CA path too long");
53 return NGX_ERROR;
54 }
55 ngx_memcpy(cert, conf->common.certificate.data, conf->common.certificate.len);
56 cert[conf->common.certificate.len] = '\0';
57 ngx_memcpy(key, conf->common.certificate_key.data, conf->common.certificate_key.len);
58 key[conf->common.certificate_key.len] = '\0';
59 ngx_memcpy(ca, conf->common.trusted_ca.data, conf->common.trusted_ca.len);
60 ca[conf->common.trusted_ca.len] = '\0';
61 return NGX_OK;
62}

65/*
66 * WHAT: Allocate the callback-free TLS context used by the mem-BIO GSI engine.
67 * WHY: nginx SSL callbacks assume connection ex-data that this engine lacks.
68 * HOW: Create a generic TLS context, register pool cleanup, and pin TLS 1.2.
69 */
70static SSL_CTX *
71brix_ftp_gsi_ssl_ctx(ngx_conf_t *cf)
72{
73 SSL_CTX *ctx;
74 ngx_pool_cleanup_t *cln;
75
76 ctx = SSL_CTX_new(TLS_method());
77 if (ctx == NULL) {
78 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
79 "brix_gridftp_gsi: SSL_CTX_new failed");
80 return NULL;
81 }
82 cln = ngx_pool_cleanup_add(cf->pool, 0);
83 if (cln == NULL) {
84 SSL_CTX_free(ctx);
85 return NULL;
86 }
87 cln->handler = brix_ftp_ssl_ctx_cleanup;
88 cln->data = ctx;
89 SSL_CTX_set_min_proto_version(ctx, TLS1_2_VERSION);
90 SSL_CTX_set_max_proto_version(ctx, TLS1_2_VERSION);
91 return ctx;
92}

95/*
96 * WHAT: Load and validate the GridFTP host certificate and private key.
97 * WHY: Refuse configuration before any session can use mismatched credentials.
98 * HOW: Load the PEM chain and key, then require OpenSSL's key-pair check.
99 */
100static ngx_int_t
101brix_ftp_gsi_host_credentials(ngx_conf_t *cf, SSL_CTX *ctx,
102 const char *cert, const char *key)
103{
104 if (SSL_CTX_use_certificate_chain_file(ctx, cert) == 1
105 && SSL_CTX_use_PrivateKey_file(ctx, key, SSL_FILETYPE_PEM) == 1
106 && SSL_CTX_check_private_key(ctx) == 1)
107 {
108 return NGX_OK;
109 }
110 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
111 "brix_gridftp_gsi: cannot load host cert %s / key %s",
112 cert, key);
113 return NGX_ERROR;
114}

117/* brix_ftp_build_gsi — construct the host TLS context (cert/key) and the client
118 * proxy trust store once the GSI directives are known. Unlike root:// / WebDAV
119 * TLS, the mem-BIO GSSAPI engine (gsi_mech.c) drives handshakes on a bare SSL
120 * object with no nginx connection attached, so we must NOT use ngx_ssl_create():
121 * it installs nginx info/servername callbacks that deref an ngx_connection_t via
122 * SSL ex-data our SSL never has, crashing mid-handshake. A plain OpenSSL
123 * SSL_CTX (as in the phase-82 interop probe) sidesteps every such callback. */
124char *
125brix_ftp_build_gsi(ngx_conf_t *cf, ngx_stream_brix_ftp_srv_conf_t *conf)
126{
127 struct stat stbuf;
128 int ca_is_dir;
129 char ca_raw[PATH_MAX];
130 char cert_raw[PATH_MAX];
131 char key_raw[PATH_MAX];
132 SSL_CTX *ctx;
133
134 if (brix_ftp_gsi_paths(cf, conf, cert_raw, sizeof(cert_raw),
135 key_raw, sizeof(key_raw), ca_raw, sizeof(ca_raw))
136 != NGX_OK)
137 {
138 return NGX_CONF_ERROR;
139 }
140
141 conf->tls_ctx = ngx_pcalloc(cf->pool, sizeof(ngx_ssl_t));
142 if (conf->tls_ctx == NULL) {
143 return NGX_CONF_ERROR;
144 }
145 conf->tls_ctx->log = cf->log;
146
147 /* Version-flexible method (not TLS_server_method): the same context serves
148 * both roles — the control channel and the passive data channel accept
149 * (SSL_accept), while a gsiftp↔gsiftp TPC source leg connects out on the
150 * data channel (SSL_connect). A server-only context makes SSL_connect fail
151 * with "called a function you should not call". */
152 ctx = brix_ftp_gsi_ssl_ctx(cf);
153 if (ctx == NULL) {
154 return NGX_CONF_ERROR;
155 }
156 if (brix_ftp_gsi_host_credentials(cf, ctx, cert_raw, key_raw) != NGX_OK) {
157 return NGX_CONF_ERROR;
158 }
159 conf->tls_ctx->ctx = ctx;
160
161 ca_is_dir = (stat(ca_raw, &stbuf) == 0 && S_ISDIR(stbuf.st_mode)); /* vfs-seam-allow: DOMAIN_CONFIG — trust-anchor path (CApath dir vs CAfile bundle), not export storage */
162
163 conf->ca_store = brix_build_ca_store_cached(cf->cycle, cf->log,
164 ca_is_dir ? ca_raw : NULL, /* CApath (hashed dir) */
165 ca_is_dir ? NULL : ca_raw, /* or CAfile bundle */
166 NULL, /* no CRL for the POC */
167 X509_V_FLAG_ALLOW_PROXY_CERTS, /* RFC 3820 proxies */
168 NULL, BRIX_SP_MODE_OFF, BRIX_CRL_MODE_OFF);
169 if (conf->ca_store == NULL) {
170 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
171 "brix_gridftp_gsi: cannot build CA trust store from %s", ca_raw);
172 return NGX_CONF_ERROR;
173 }
174
175 ngx_conf_log_error(NGX_LOG_NOTICE, cf, 0,
176 "brix: GridFTP gsiftp security enabled (cert=%V ca=%s)",
177 &conf->common.certificate, ca_raw);
178 return NGX_CONF_OK;
179}
