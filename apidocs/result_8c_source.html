52/* ---- Recount every result verdict ----
53 *
54 * WHAT: Resets and fills aggregate counters from the result vector.
55 * WHY: One derived source prevents counter drift after filtering/baseline work.
56 * HOW: Walk once and increment the matching typed counter.
57 */
58void
59brix_conform_run_results_count(brix_conform_run_results *results)
60{
61 size_t index;
62
63 if (results == NULL) return;
64 results->conformant = 0;
65 results->divergent = 0;
66 results->skipped = 0;
67 results->inconclusive = 0;
68 results->changed = 0;
69 for (index = 0; index < results->count; index++) {
70 if (results->results[index].changed) results->changed++;
71 switch (results->results[index].verdict) {
72 case BRIX_CONFORM_VERDICT_CONFORMANT: results->conformant++; break;
73 case BRIX_CONFORM_VERDICT_DIVERGENT: results->divergent++; break;
74 case BRIX_CONFORM_VERDICT_SKIPPED: results->skipped++; break;
75 case BRIX_CONFORM_VERDICT_INCONCLUSIVE: results->inconclusive++; break;
76 }
77 }

80/* ---- Map aggregate outcomes onto Phase-118 exit codes ----
