1/*
2 * gftp_spas.h — GridFTP striped passive data channels (GFD.020 §5.1) for the
3 * outbound gsiftp storage driver.
4 *
5 * WHAT: `SPAS` asks the origin for SEVERAL data-channel listeners at once and
6 * answers with one address per stripe; the driver connects to all of them and
7 * reassembles the single MODE E block stream that arrives across them. This
8 * header owns the connection GROUP a transfer runs over, striped or not, so the
9 * transfer code never has to know which of the two it got.
10 *
11 * WHY: one TCP connection is one congestion window. On the transatlantic
12 * paths this driver exists for, a single stream leaves most of the link unused
13 * no matter how large the socket buffers are, and striping is the accepted
14 * remedy — it is why GridFTP has extended block mode at all. The receiver
15 * (gftp_mode_e.c) was written for the general case from the start, so what was
16 * missing was only the way to ASK.
17 *
18 * WHY IT IS A CEILING, NOT A REQUIREMENT: `mode=` and `prot=` say what the
19 * bytes ARE — how they are framed, whether they are protected — and an origin
20 * that will not honour them fails the transfer, because serving under weaker
21 * terms would give the operator neither the property nor a diagnostic.
22 * `streams=<n>` says how FAST the same, identically verified bytes arrive. An
23 * origin that does not advertise SPAS, answers it with an error, offers more
24 * stripes than the operator budgeted, or names an address that may not be
25 * dialled is therefore served by the single pinned connection instead of
26 * failing — exactly as an origin without `ERET` is (gftp_feat.h).
27 *
28 * WHY THE ADDRESS CHECK IS THE WHOLE SECURITY STORY: a SPAS reply is the one
29 * place in this protocol where an origin hands the client a LIST of addresses
30 * to dial. gftp_dc_open() never trusted even the single PASV address — it
31 * discards it and dials the control channel's own peer — and striping does not
32 * get to relax that: a stripe naming any other address is the classic FTP
33 * bounce, turning the storage driver into a port scanner aimed wherever the
34 * origin (or whoever spoofed its reply) likes. Every stripe must be the
35 * control peer; one that is not abandons the striped attempt entirely, and
36 * nothing is dialled. The cost is that a genuinely MULTI-HOST striped server
37 * is read over one connection rather than several, which is a performance
38 * limit and never a correctness one.
39 *
40 * HOW: gftp_dc_group_open() is the only entry point the transfer code uses. It
41 * decides whether striping is wanted at all (store line, MODE E, FEAT), issues
42 * SPAS, screens every stripe, dials the ones it kept through gftp_dc_open_at(),
43 * and falls back to a single gftp_dc_open() whenever any of that does not hold.
44 */

50/* The data connections of ONE transfer. `n` is 1 for an unstriped transfer, so
51 * a caller never branches on how the group was opened. */
52typedef struct {
53 gftp_dc_t conns[GFTP_STREAMS_MAX];
54 unsigned n;
55} gftp_dc_group_t;

57/* Open the data connections for one retrieve. Returns 0 with group->n >= 1, or
58 * -1 with the session error set. Never returns a partially-open group: on
59 * failure every connection is closed. */
60int gftp_dc_group_open(gftp_session_t *session, gftp_dc_group_t *group);

62/* Raise every connection in the group to PROT P, or do nothing on a clear
63 * session. Returns 0, or -1 with the group closed and the session error set.
64 *
65 * Separate from the open, and called only AFTER the transfer command, for the
66 * reason gftp_dc_secure() states: the origin starts its side of each stripe's
67 * handshake when the transfer begins, not when the socket arrives. */
68int gftp_dc_group_secure(gftp_session_t *session, gftp_dc_group_t *group);

70/* Close every connection in the group. Idempotent. */
