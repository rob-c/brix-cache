4/*
5 * sd_ram_internal.h — shared types for the RAM object store (sd_ram.c and
6 * sd_ram_staged.c). Not a public header: nothing outside src/fs/backend/ram/
7 * includes it (the driver is reached through brix_sd_ram_driver alone).
8 *
9 * WHAT: a per-worker hash table of heap-resident objects, its byte accounting,
10 * and the one mutex that guards both.
11 * WHY: the store is mutated from the event loop (open/stat/unlink) AND from
12 * cache-fill worker threads (staged_write/commit), exactly the mix that
13 * cost cinfo_l1 a TSan-found corruption before it grew a lock — so the
14 * table carries the same pthread mutex, for the same reason.
15 * HOW: entries are refcounted like POSIX links: unlink detaches the name and
16 * drops a reference, and the bytes are freed by whichever holder puts the
17 * last one down, so an open handle keeps reading an evicted object rather
18 * than following a dangling pointer.
19 */

35/* One stored object (or directory). `path` is the store-relative logical name
36 * with a leading '/'; `data`/`size` are the object bytes (NULL/0 for a dir and
37 * for a zero-length object). `cap` is the allocation behind `data` — growth is
38 * amortised, so cap >= size and the accounting charges `cap`, never `size`. */
39typedef struct sd_ram_ent_s {
40 struct sd_ram_ent_s *hnext; /* hash-bucket chain */
41 struct sd_ram_ent_s *lru_prev; /* MRU-first eviction list */
42 struct sd_ram_ent_s *lru_next;
43 char *path;
44 u_char *data;
45 size_t size;
46 size_t cap;
47 sd_ram_xattr_t *xattrs;
48 time_t mtime;
49 time_t ctime;
50 uint64_t ino;
51 unsigned refs; /* names + open handles */
52 unsigned is_dir:1;
53 unsigned unlinked:1; /* name detached; bytes await the last ref */
54 unsigned charged:1; /* ent->cap is counted in st->used — set by
55 * insert, cleared only by the free that
56 * discounts it, so unlink-while-open (which
57 * frees LATER, from close) still discounts
58 * exactly once */
59} sd_ram_ent_t;

61/* Per-instance state: the table, the cap, and the accounting the `space` slot
62 * reports. `reserved` holds bytes claimed by fills that have not committed yet
63 * (a declared_size reservation), so two concurrent fills cannot both be told
64 * there is room for the same bytes. */
65typedef struct {
66 sd_ram_ent_t **buckets;
67 sd_ram_ent_t *lru_head; /* most recently used */
68 sd_ram_ent_t *lru_tail; /* the next eviction victim */
69 size_t nbuckets;
70 size_t count;
71 uint64_t used; /* committed object bytes */
72 uint64_t reserved; /* in-flight fill reservations */
73 uint64_t capacity; /* the ram:<size> hard cap */
74 uint64_t next_ino;
75 pthread_mutex_t mtx;
76 ngx_log_t *log;
77} sd_ram_state_t;
