20/* The podman rule: first path component is a host iff it has '.' or ':' or
21 * is exactly "localhost". */
22static int
23looks_like_host(const char *s, size_t n)
24{
25 return memchr(s, '.', n) != NULL || memchr(s, ':', n) != NULL ||
26 (n == 9 && memcmp(s, "localhost", 9) == 0);
27}

29/* Peel a podman-style registry prefix from *rest when the first component
30 * identifies itself as a host. The authority itself is parsed by the shared
31 * URL grammar — brackets, port and all — so a host this tool will dial is
32 * exactly a host the proxy would have accepted in a base URL, and an IPv6
33 * literal keeps the one canonical (unbracketed) spelling everywhere below. */
34static int
35ref_registry(const char **rest, brix_oci_ref_t *out, char *err,
36 size_t errlen)
37{
38 const char *slash = strchr(*rest, '/');
39 size_t alen;
40
41 if (slash == NULL) {
42 /* A bracketed literal with nothing after it is a host and no
43 * repository — worth saying so, rather than failing later as a
44 * repository name full of colons. */
45 return ((*rest)[0] == '[')
46 ? ref_fail(err, errlen, "registry host without a repository")
47 : 0;
48 }
49 alen = (size_t) (slash - *rest);
50 if (!looks_like_host(*rest, alen)) {
51 return 0;
52 }
53 if (brix_oci_url_authority(*rest, alen, out->host, sizeof(out->host),
54 &out->port) != 0)
55 {
56 return ref_fail(err, errlen, "invalid registry host or port");
57 }
58 *rest = slash + 1;
59 return 0;
60}

62/* Validate and retain the optional digest, before tag parsing sees its colon. */
63static int
64ref_digest(const char *at, brix_oci_ref_t *out, char *err, size_t errlen)
65{
66 brix_oci_digest_t d;
67
68 if (at == NULL) {
69 return 0;
70 }
71 if (brix_oci_digest_parse(at + 1, strlen(at + 1), &d) != 0) {
72 return ref_fail(err, errlen,
73 "invalid digest (want sha256:<64 lowercase hex> "
74 "or sha512:<128 lowercase hex>)");
75 }
76 /* Re-emit through the formatter rather than the parsed hex alone: the
77 * algorithm the ref pinned is part of the identity, and every path and
78 * URL built from this ref is keyed by it. */
79 if (brix_oci_digest_format(&d, out->digest, sizeof(out->digest)) < 0) {
80 return ref_fail(err, errlen, "digest too long");
81 }
82 out->has_digest = 1;
83 return 0;
84}

101/* Pull a terminal tag from a name span and update its remaining name length. */
102static int
103ref_tag(const char *rest, const char *at, brix_oci_ref_t *out,
104 size_t *name_len, char *err, size_t errlen)
105{
106 const char *start;
107 const char *colon;
108 size_t tlen;
109
110 *name_len = at != NULL ? (size_t) (at - rest) : strlen(rest);
111 start = ref_last_slash(rest, *name_len);
112 start = start != NULL ? start : rest;
113 colon = memchr(start, ':', (size_t) (rest + *name_len - start));
114 if (colon == NULL) {
115 memcpy(out->tag, "latest", 7);
116 return 0;
117 }
118 tlen = (size_t) (rest + *name_len - colon - 1);
119 if (tlen == 0 || tlen >= sizeof(out->tag) ||
120 brix_oci_tag_valid(colon + 1, tlen) != 0) {
121 return ref_fail(err, errlen, "invalid tag");
122 }
123 memcpy(out->tag, colon + 1, tlen);
124 out->tag[tlen] = '\0';
125 *name_len = (size_t) (colon - rest);
126 return 0;
127}

146brix_oci_ref_parse(const char *s, brix_oci_ref_t *out, char *err,
147 size_t errlen)
148{
149 const char *at;
150 const char *rest = s;
151 size_t n;
152
153 memset(out, 0, sizeof(*out));
154 if (s == NULL || s[0] == '\0') {
155 return ref_fail(err, errlen, "empty image reference");

159 }
160
161 /* [@digest] first — it may contain ':' and must not confuse the tag. */

164 ref_tag(rest, at, out, &n, err, errlen) != 0) {
165 return -1;
166 }
167 return ref_name(rest, n, out, err, errlen);
