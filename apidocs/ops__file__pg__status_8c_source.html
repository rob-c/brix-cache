8static int
9read_pg_error(brix_conn *c, uint32_t dlen, brix_status *st)
10{
11 uint8_t *body = NULL;
12 int errnum = 0;
13
14 if (dlen > 0 && dlen <= XRDC_DLEN_MAX) {
15 body = (uint8_t *) malloc(dlen);
16 if (body != NULL && brix_read_full(&c->io, body, dlen, st) == 0) {
17 const char *message = "";
18 size_t message_len = 0;
19
20 xrd_error_body_decode(body, dlen, &errnum, &message, &message_len);
21 brix_status_set(st, errnum, 0, "%.*s (%s)", (int) message_len,
22 message ? message : "", brix_kxr_name(errnum));
23 st->wire_message_len = (uint32_t) message_len;
24 st->wire_message_terminated = dlen > 4 && body[dlen - 1] == '\0';
25 }
26 }
27 free(body);
28 if (st->kxr == 0)
29 brix_status_set(st, XRDC_EPROTO, 0, "paged operation error");
30 return -1;
31}

33static int
34validate_status_body(const uint8_t *body, uint16_t sid,
35 uint8_t request_delta, brix_status *st)
36{
37 uint32_t want_crc = xrd_get_u32_be(body);
38 uint32_t got_crc = brix_crc32c_value(body + 4, XRDC_PG_STATUSBODY - 4);
39
40 if (want_crc != got_crc) {
41 brix_status_set(st, XRDC_EINTEGRITY, 0,
42 "kXR_status header CRC mismatch (got %08x want %08x)",
43 got_crc, want_crc);
44 return -1;
45 }
46 if (xrd_get_u16_be(body + 4) != sid) {
47 brix_status_set(st, XRDC_EPROTO, 0,
48 "kXR_status body stream id does not match header");
49 return -1;
50 }
51 if (body[6] != request_delta) {
52 brix_status_set(st, XRDC_EPROTO, 0,
53 "kXR_status request id mismatch (got %u, want %u)",
54 body[6], request_delta);
55 return -1;
56 }
57 if (body[8] != 0 || body[9] != 0 || body[10] != 0 || body[11] != 0) {
58 brix_status_set(st, XRDC_EPROTO, 0,
59 "kXR_status reserved bytes are non-zero");
60 return -1;
61 }
62 if (body[7] > kXR_ProgressInfo) {
63 brix_status_set(st, XRDC_EPROTO, 0,
64 "invalid kXR_status response type %u", body[7]);
65 return -1;
66 }
67 return 0;
68}

70int
71read_status_frame(brix_conn *c, uint16_t want_sid, uint8_t request_delta,
72 uint8_t *resptype, uint32_t *pgdlen, int64_t *foff,
73 brix_status *st)

75 uint8_t hdr[XRD_RESPONSE_HDR_LEN];

78 uint32_t dlen;
79 uint64_t offset_be;
80
81 if (brix_read_full(&c->io, hdr, sizeof(hdr), st) != 0) return -1;
82 xrd_resp_hdr_unpack(hdr, &sid, &status, &dlen);
83 if (status == kXR_error) return read_pg_error(c, dlen, st);
84 if (status != kXR_status) {
85 brix_status_set(st, XRDC_EPROTO, 0,
86 "expected kXR_status, got %u", status);
87 return -1;
88 }
89 if (want_sid != 0xffff && sid != want_sid) {
90 brix_status_set(st, XRDC_EPROTO, 0,
91 "stream id mismatch (got %u, want %u)", sid, want_sid);
92 return -1;
93 }
94 if (dlen != XRDC_PG_STATUSBODY) {
95 brix_status_set(st, XRDC_EPROTO, 0,
96 "bad kXR_status body length %u (want %d)",
97 dlen, XRDC_PG_STATUSBODY);
98 return -1;
99 }
100 if (brix_read_full(&c->io, body, sizeof(body), st) != 0) return -1;
101 if (validate_status_body(body, sid, request_delta, st) != 0) return -1;
102
103 *resptype = body[7];
104 *pgdlen = xrd_get_u32_be(body + 12);
105 memcpy(&offset_be, body + 16, sizeof(offset_be));
106 *foff = (int64_t) be64toh(offset_be);
107 return 0;
108}
