1/*
2 * vfs_writer.c — unified verified write session across every storage backend.
3 *
4 * WHAT: brix_vfs_writer_open/write/commit/abort — one write entry point a
5 * protocol path (GridFTP STOR) uses regardless of backend, with an
6 * optional self-computed read-back integrity check folded in.
7 * WHY: The write mechanics differ per backend — a random-write backend (the
8 * POSIX default export, pblock) is patched in place through an O_WRITE
9 * handle, while an object store (S3) has no seekable file and must be
10 * written as an atomic staged upload — and the verify-on-write logic
11 * (accumulate a CRC, re-read the persisted object, compare) was hand-rolled
12 * in the STOR path. Folding both behind one session gives every filesystem
13 * the same verified-write call and lets a single caller serve S3 too.
14 * HOW: writer_open resolves the backend's CAP_RANDOM_WRITE bit from ctx->sd
15 * (NULL ⇒ the POSIX default export ⇒ random-write): a random backend opens
16 * an in-place handle, an object backend opens a staged upload. Each write is
17 * routed through the driver (brix_vfs_file_pwrite / brix_vfs_staged_write)
18 * and, when verifying, folded into a brix_wverify accumulator. commit closes
19 * or atomically publishes, then — when verifying a non-empty object — drives
20 * brix_vfs_wverify_check over a fresh read-only handle and unlinks on any
21 * mismatch. No goto; early-return per coding-standards.
22 */
23#include "vfs_writer_internal.h" /* session struct + spill (phase-107 C1) */
24#include "core/compat/copy_range.h" /* brix_copy_range — zero-copy fd ingest */

26/* Does the backend behind `ctx` accept an in-place random-offset write? The
27 * POSIX default export has no resolved sd instance (ctx->sd == NULL) and is
28 * always random-write; any other backend must advertise CAP_RANDOM_WRITE. */
29static int
30writer_random_backend(const brix_vfs_ctx_t *ctx)
31{
32 if (ctx->sd == NULL) {
33 return 1;
34 }
35 return (brix_sd_caps(ctx->sd) & BRIX_SD_CAP_RANDOM_WRITE) != 0;
36}

38/* Release the descriptor/temp a partially-constructed writer holds without
39 * publishing anything (used on an open-time failure and by writer_abort). The
40 * random path leaves any created object in place — a mid-STOR failure is
41 * REST-resumable — while the staged path drops its unpublished temp. */
42static void
43writer_release(brix_vfs_writer_t *w)
44{
45 if (w->fh != NULL) {
46 brix_vfs_close(w->fh, w->log);
47 w->fh = NULL;
48 }
49 if (w->st != NULL) {
50 brix_vfs_staged_abort(w->st, 1 /* remove temp */);
51 w->st = NULL;
52 }
53 brix_vfs_writer_spill_discard(w); /* unlink any spill scratch (C1 T3) */
54 if (w->wv != NULL) {
55 brix_wverify_free(w->wv);
56 w->wv = NULL;
57 }
58}

72brix_vfs_writer_t *
73brix_vfs_writer_open(brix_vfs_ctx_t *ctx, unsigned flags, int verify,
74 int *err_out)
75{
76 brix_vfs_writer_t *w;
77 int verr = 0;
78
79 writer_set_error(err_out, 0);
80 if (ctx == NULL) {
81 writer_set_error(err_out, EINVAL);
82 return NULL;
83 }
84
85 w = ngx_pcalloc(ctx->pool, sizeof(*w));
86 if (w == NULL) {
87 writer_set_error(err_out, ENOMEM);
88 return NULL;
89 }
90 /* Self-contain a deep copy of ctx: a write session outlives the request that
91 * opened it (the caller's cred ctx can be a stack frame), and commit re-derives
92 * from w->ctx for the verify read-back + unlink-on-mismatch. */
93 w->ctx = brix_vfs_ctx_pool_clone(ctx, ctx->pool);
94 if (w->ctx == NULL) {
95 writer_set_error(err_out, ENOMEM);
96 return NULL;
97 }
98 w->pool = ctx->pool;
99 w->log = ctx->log;
100 w->verify = verify ? 1 : 0;
101 w->mutation_policy = ctx->mutation_policy;
102 w->spill.fd = NGX_INVALID_FILE; /* pcalloc's 0 is a real fd */
103 /* O_ATOMIC forces the staged temp+publish path even for a random-write backend
104 * so a failed write leaves no partial at the final path (WebDAV/S3 PUT). */
105 w->mode = ((flags & BRIX_VFS_O_ATOMIC) == 0 && writer_random_backend(w->ctx))
106 ? BRIX_VFS_WRITER_RANDOM
107 : BRIX_VFS_WRITER_SEQUENTIAL;
108
109 if (w->mode == BRIX_VFS_WRITER_RANDOM) {
110 unsigned oflags = BRIX_VFS_O_WRITE | BRIX_VFS_O_CREATE
111 | (flags & BRIX_VFS_O_TRUNC);
112 w->fh = brix_vfs_open(w->ctx, oflags, &verr);
113 if (w->fh == NULL) {
114 writer_set_error(err_out, verr ? verr : EIO);
115 return NULL;
116 }
117 } else {
118 w->st = brix_vfs_staged_open(w->ctx, NGX_FILE_DEFAULT_ACCESS,
119 16 /* excl-name attempts */, &verr);
120 if (w->st == NULL) {
121 writer_set_error(err_out, verr ? verr : EIO);
122 return NULL;
123 }
124 /* BRIX_VFS_WRITER_O_UNORDERED (phase-107 C1): the caller declares the
125 * extents may arrive out of order, so provision the spill now rather
126 * than on the first violation. The MUTATE_OPEN gate has already fired
127 * inside brix_vfs_staged_open, so a read-only endpoint never reaches
128 * this line — the scratch is created strictly AFTER the gate. Failure
129 * to provision is not fatal at open: a strictly in-order stream still
130 * succeeds, and a real reorder re-attempts (and then errors). */
131 if (flags & BRIX_VFS_WRITER_O_UNORDERED) {
132 (void) brix_vfs_writer_spill_enter(w, 0, 0);
133 }
134 }
135
136 if (w->verify) {
137 w->wv = brix_wverify_begin();
138 if (w->wv == NULL) {
139 writer_release(w);
140 writer_set_error(err_out, ENOMEM);
141 return NULL;
142 }
143 }
144 return w;
145}

147/* Push one extent onto the chosen backend path (no verify bookkeeping). */
148static ngx_int_t
149writer_put(brix_vfs_writer_t *w, const void *buf, size_t len, off_t off)
150{
151 switch (w->mode) {
152
153 case BRIX_VFS_WRITER_RANDOM: {
154 const u_char *p = buf;
155 size_t left = len;
156
157 while (left > 0) {
158 ssize_t n = brix_vfs_file_pwrite(w->fh, p, left, off);
159 if (n <= 0) {
160 return NGX_ERROR;
161 }
162 p += n;
163 left -= (size_t) n;
164 off += n;
165 }
166 return NGX_OK;
167 }
168
169 case BRIX_VFS_WRITER_SPILL:
170 return brix_vfs_writer_spill_put(w, buf, len, off);
171
172 case BRIX_VFS_WRITER_FAILED:
173 errno = ENOSPC; /* scratch was exhausted; nothing recovers */
174 return NGX_ERROR;
175
176 case BRIX_VFS_WRITER_SEQUENTIAL:
177 break;
178 }
179
180 /* Staged / object store: the upload is built sequentially into the
181 * temp/upload, so a reordered extent cannot be patched in place. Phase-107
182 * C1 replaced the old EINVAL refusal here with the one-way T1 promotion
183 * into spill mode; a rewind below the already-staged prefix (or a missing
184 * spill root) is still refused inside spill_enter. */
185 if (off != w->staged_cursor) {
186 if (brix_vfs_writer_spill_enter(w, off, len) != NGX_OK) {
187 return NGX_ERROR;
188 }
189 return brix_vfs_writer_spill_put(w, buf, len, off);
190 }
191 if (brix_vfs_staged_write(w->st, buf, len, off) != NGX_OK) {
192 return NGX_ERROR;
193 }
194 w->staged_cursor += (off_t) len;
195 return NGX_OK;
196}

198ngx_int_t
199brix_vfs_writer_write(brix_vfs_writer_t *w, const void *buf, size_t len,
200 off_t off)
201{
202 if (w == NULL || w->finished || buf == NULL || off < 0) {
203 errno = EINVAL;
204 return NGX_ERROR;
205 }
206 if (len == 0) {
207 return NGX_OK;
208 }
209 if (brix_vfs_gate_mutation(w->ctx, BRIX_VFS_MUTATE_WRITE) != NGX_OK)
210 {
211 return NGX_ERROR;
212 }
213 if (writer_put(w, buf, len, off) != NGX_OK) {
214 return NGX_ERROR;
215 }
216 if (w->wv != NULL && brix_wverify_update(w->wv, buf, off, len) != 0) {
217 errno = EIO;
218 return NGX_ERROR;
219 }
220 w->written += (off_t) len;
221 return NGX_OK;
222}

224/* Read the source fd in 64 KiB chunks and push each through the normal write
225 * engine, which dispatches random/staged and folds the verify CRC. Used whenever
226 * the bytes must pass through userspace: verify is on, the destination is a
227 * staged/object upload, or the random backend has no single seekable fd (pblock). */
228static ngx_int_t
229writer_ingest_bounce(brix_vfs_writer_t *w, int src_fd, off_t src_off, size_t len,
230 off_t dst_off)
231{
232 off_t s = src_off;
233 off_t d = dst_off;
234 size_t left = len;
235
236 while (left > 0) {
237 u_char chunk[65536];
238 size_t want = left < sizeof(chunk) ? left : sizeof(chunk);
239 ssize_t n = pread(src_fd, chunk, want, s);
240
241 if (n < 0) {
242 if (errno == EINTR) {
243 continue;
244 }
245 return NGX_ERROR;
246 }
247 if (n == 0) { /* source shorter than the caller promised */
248 errno = EIO;
249 return NGX_ERROR;
250 }
251 if (brix_vfs_writer_write(w, chunk, (size_t) n, d) != NGX_OK) {
252 return NGX_ERROR;
253 }
254 s += n;
255 d += (off_t) n;
256 left -= (size_t) n;
257 }
258 return NGX_OK;
259}

261ngx_int_t
262brix_vfs_writer_write_fd(brix_vfs_writer_t *w, int src_fd, off_t src_off,
263 size_t len, off_t dst_off)
264{
265 if (w == NULL || w->finished || src_fd < 0 || src_off < 0 || dst_off < 0) {
266 errno = EINVAL;
267 return NGX_ERROR;
268 }
269 if (len == 0) {
270 return NGX_OK;
271 }
272 if (brix_vfs_gate_mutation(w->ctx, BRIX_VFS_MUTATE_WRITE) != NGX_OK)
273 {
274 return NGX_ERROR;
275 }
276
277 /* Zero-copy fast path: a single-fd, sendfile-capable random backend with no
278 * verify obligation can move the bytes kernel-side without a userspace bounce.
279 * sendfile_fd() returns NGX_INVALID_FILE for an object/block backend (no single
280 * seekable destination), which — like verify and the staged path — falls back
281 * to the bounce so block routing / the CRC accumulator are not bypassed. */
282 if (w->mode == BRIX_VFS_WRITER_RANDOM && !w->verify) {
283 ngx_fd_t dfd = brix_vfs_file_sendfile_fd(w->fh);
284
285 if (dfd != NGX_INVALID_FILE) {
286 if (brix_copy_range(w->log, src_fd, src_off, dfd, dst_off,
287 len, NULL, NULL) != NGX_OK) {
288 return NGX_ERROR;
289 }
290 w->written += (off_t) len;
291 return NGX_OK;
292 }
293 }
294
295 return writer_ingest_bounce(w, src_fd, src_off, len, dst_off);
296}

298off_t
299brix_vfs_writer_expected_off(const brix_vfs_writer_t *w)
300{
301 if (w == NULL) {
302 return -1;
303 }
304 /* Advisory only since phase-107 C1: the writer accepts any offset in RANDOM
305 * and SPILL modes and self-promotes SEQUENTIAL -> SPILL on a reorder, so
306 * callers must NOT pre-refuse on a mismatch — submit the write and let the
307 * writer's errno decide (EINVAL = unservable order, ENOSPC = no scratch). */
308 switch (w->mode) {
309 case BRIX_VFS_WRITER_SEQUENTIAL:
310 return w->staged_cursor;
311 case BRIX_VFS_WRITER_SPILL:
312 return w->spill.high_water;
313 case BRIX_VFS_WRITER_RANDOM:
314 case BRIX_VFS_WRITER_FAILED:
315 break;
316 }
317 return w->written;
318}

320/* Re-open the just-committed object read-only and confirm the driver persisted
321 * exactly the CRC-checked bytes. An empty object has no extents to expect and is
322 * trivially complete, so it is not read back. */
323static ngx_int_t
324writer_verify(brix_vfs_writer_t *w)
325{
326 brix_vfs_file_t *rfh;
327 int verr = 0;
328 ngx_int_t rc;
329
330 if (w->wv == NULL || w->written == 0) {
331 return NGX_OK;
332 }
333 rfh = brix_vfs_open(w->ctx, BRIX_VFS_O_READ, &verr);
334 if (rfh == NULL) {
335 return NGX_ERROR;
336 }
337 rc = brix_vfs_wverify_check(w->wv, rfh);
338 brix_vfs_close(rfh, w->log);
339 return rc;
340}

352/*
353 * WHAT: The staged half of commit: drain a spill, publish the temp onto the
354 * final path, tear the staged session down.
355 * WHY: A staged commit failure must leave nothing published; `pre` rides to
356 * brix_vfs_staged_commit, whose errno (EEXIST/ECANCELED = precondition
357 * refused) must survive the teardown that follows.
358 * HOW: T4 (FAILED) refuses ENOSPC without publishing. T2: a spilled object is
359 * drained sequentially into the staged session first; a coverage hole
360 * refuses (EINVAL). The scratch is unlinked only AFTER the publish
361 * succeeds, so a crash mid-publish leaves the bytes recoverable (C1).
362 */
363static ngx_int_t
364writer_commit_staged(brix_vfs_writer_t *w, brix_sd_precond_t *pre)
365{
366 ngx_int_t crc;
367 int ce;
368
369 if (w->mode == BRIX_VFS_WRITER_FAILED) {
370 /* T4 already aborted the staged session; nothing may publish. */
371 writer_release(w);
372 errno = ENOSPC;
373 return NGX_ERROR;
374 }
375 if (w->mode == BRIX_VFS_WRITER_SPILL
376 && brix_vfs_writer_spill_drain(w) != NGX_OK)
377 {
378 ce = errno;
379 writer_release(w);
380 errno = ce;
381 return NGX_ERROR;
382 }
383
384 crc = brix_vfs_staged_commit(w->st, pre);
385 ce = errno; /* preserve EEXIST/ECANCELED (precondition refused) */
386
387 brix_vfs_staged_abort(w->st, 0 /* already published/closed */);
388 w->st = NULL;
389 brix_vfs_writer_spill_discard(w);
390 if (crc != NGX_OK) {
391 writer_wv_free(w);
392 errno = ce;
393 return NGX_ERROR;
394 }
395 return NGX_OK;
396}

398ngx_int_t
399brix_vfs_writer_commit_pre(brix_vfs_writer_t *w, brix_sd_precond_t *pre)
400{
401 if (w == NULL || w->finished) {
402 return NGX_ERROR;
403 }
404 /* MUTATE_PUBLISH: commit is where the object becomes visible under its final
405 * name — a distinct event from the body writes above, and reported as one
406 * (see the same reasoning in brix_vfs_staged_commit). */
407 if (brix_vfs_gate_mutation(w->ctx, BRIX_VFS_MUTATE_PUBLISH) != NGX_OK)
408 {
409 return NGX_ERROR;
410 }
411 /* The in-place path cannot evaluate a publish precondition: the bytes
412 * already landed on the final object when they were written. NONE and
413 * ABSENT are legitimately settled earlier (ABSENT was the open's O_EXCL);
414 * a MATCH_* here would be a silent pass, so it refuses (§3.5 — a refusal
415 * over an emulation that lies). After the policy gate, so a read-only
416 * endpoint still answers EROFS, never ENOTSUP. */
417 if (w->mode == BRIX_VFS_WRITER_RANDOM
418 && pre != NULL && pre->kind != BRIX_SD_PRECOND_NONE
419 && pre->kind != BRIX_SD_PRECOND_ABSENT)
420 {
421 errno = ENOTSUP;
422 return NGX_ERROR;
423 }
424 w->finished = 1;
425
426 /* Persist: fsync + close the in-place handle, or atomically publish the temp
427 * onto the final path. A staged commit failure leaves nothing published.
428 * `excl` uses RENAME_NOREPLACE on the staged path (S3 If-None-Match → EEXIST);
429 * it is meaningless for the in-place random path (no separate publish). */
430 if (w->mode == BRIX_VFS_WRITER_RANDOM) {
431 (void) brix_vfs_sync(w->fh);
432 brix_vfs_close(w->fh, w->log);
433 w->fh = NULL;
434 } else if (writer_commit_staged(w, pre) != NGX_OK) {
435 return NGX_ERROR;
436 }
437
438 /* Read-back integrity check; a mismatch must never leave a corrupt object. */
439 if (writer_verify(w) != NGX_OK) {
440 (void) brix_vfs_unlink(w->ctx);
441 writer_wv_free(w);
442 return NGX_ERROR;
443 }
444 writer_wv_free(w);
445 return NGX_OK;
446}
