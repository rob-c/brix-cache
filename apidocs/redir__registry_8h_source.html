1/*
2 * redir_registry.h — client-side redirect-collapse / VirtualRedirector registry
3 * (§7.11).
4 *
5 * WHAT: a small per-process table mapping an endpoint URL to the endpoint it was
6 * redirected to, with transitive COLLAPSE — so a chain A→B→C is answered as
7 * A→C and a repeated open of A dials C directly instead of walking the chain
8 * again. Cycle-safe (bounded hop count) and bounded in size (LRU eviction).
9 * WHY: without it the client re-incurs every hop of a multi-level redirect on
10 * each operation, and a redirect loop (A→B→A) could spin. This is the client
11 * half of XrdCl's VirtualRedirector registry (the metalink virtual redirector
12 * feeds the same table via brix_vredir_record).
13 * HOW: a fixed-capacity array of (url, target) pairs; record() resolves the
14 * target through any existing chain before storing, and lookup() follows the
15 * chain with a hop cap. Thread-affinity: single-threaded client process.
16 */
17#ifndef BRIX_NET_REDIR_REGISTRY_H
18#define BRIX_NET_REDIR_REGISTRY_H
19
20/* Record that `url` redirected to `target`. Both are copied. `target` is first
21 * collapsed through the existing table, so the stored mapping is url→FINAL. A
22 * self- or cycle-mapping is refused (never stored). No-op on NULL/empty args. */
23void brix_vredir_record(const char *url, const char *target);
24
25/* Return the collapsed final target `url` resolves to (following the chain, hop-
26 * capped), or NULL when `url` has no recorded redirect. The returned pointer is
27 * owned by the registry and valid until the next record()/clear(). */
28const char *brix_vredir_lookup(const char *url);
29
30/* Drop every mapping (test isolation / reconfiguration). */
31void brix_vredir_clear(void);
32
33/* Live mapping count (diagnostics/tests). */
34unsigned brix_vredir_count(void);
35
36#endif /* BRIX_NET_REDIR_REGISTRY_H */
