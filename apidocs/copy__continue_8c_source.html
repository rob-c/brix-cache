1/*
2 * copy_continue.c — xrdcp --continue: byte-offset download resume (§7.6).
3 *
4 * WHAT: copy_download_continue() handles a --continue download: the
5 * destination file is written DIRECTLY (no atomic temp+rename), and
6 * when it already exists the transfer resumes at its current size —
7 * stock xrdcp's --continue semantics.
8 * WHY: The normal download path deliberately never leaves a partial
9 * destination (temp+rename, abort unlinks). That fail-closed posture
10 * makes interrupted multi-gigabyte transfers start over from byte 0;
11 * --continue is the operator's explicit opt-OUT: keep what arrived,
12 * finish the rest. The audit (§7.6) lists it as the one missing
13 * resume mode (the journal --resume is whole-file only).
14 * HOW: Gate first in copy_download (before the destination-exists check —
15 * an existing partial is the POINT). stat the local file, bound-check
16 * against the remote size, then a plain offset read loop appending at
17 * the partial's tail. fsync before the verdict. --cksum, when given,
18 * verifies the COMPLETED file exactly like the normal path (and drops
19 * it on a genuine mismatch — a finished-but-corrupt file is never kept;
20 * only IN-PROGRESS partials are preserved).
21 */
22#include "copy_internal.h"
23
24#include <fcntl.h>
25#include <sys/stat.h>

27/* Resume-mode read granularity: a chunk lost mid-flight to a sever is
28 * progress the NEXT attempt must refetch, so --continue trades a little
29 * pipeline depth for durability — 1 MiB bounds the refetch per fault while
30 * the normal (non-resume) paths keep the full 8 MiB chunk. */
31#define XRDC_CONTINUE_CHUNK (1024u * 1024u)
32
33
34/* ---- Pull [start, size) from the open source into fd at the same offset ----
35 *
36 * WHAT: Chunked offset read loop: brix_rfile reads from `start` to the known
37 * remote size, pwrite'ing each chunk at its absolute offset. Feeds
38 * o->progress with absolute completion. 0 / -1 (st set).
39 *
40 * WHY: download_stream_body always pumps from byte 0; resume needs an
41 * arbitrary starting offset, and the plain pread/pwrite pair keeps the
42 * resumed region byte-addressed (no rename dance to preserve the head).
43 *
44 * HOW: 1. Open the remote for read (resilient handle, same as stream_file).
45 * 2. Loop XRDC_COPY_CHUNK reads; short reads advance by what arrived;
46 * EOF before `size` is a protocol error (size was authoritative).
47 * 3. Progress callback per chunk with (start + done, size).
48 */
49static int
50continue_pull_tail(const download_job_t *job, int fd, int64_t start,
51 brix_status *st)
52{
53 const brix_copy_opts *o = job->o;
54 brix_rfile rf;
55 uint8_t *buf;
56 int64_t off = start;
57 uint64_t t0 = brix_mono_ns();
58 int rc = 0;
59
60 if (brix_rfile_open_read(job->c, job->su->path, NULL, 0, -1, &rf,
61 st) != 0) {
62 return -1;
63 }
64 buf = (uint8_t *) malloc(XRDC_COPY_CHUNK);
65 if (buf == NULL) {
66 brix_status_set(st, XRDC_EPROTO, 0, "out of memory");
67 brix_rfile_close(&rf, st);
68 return -1;
69 }
70 while (off < job->si->size) {
71 size_t cap = brix_pump_pace_cap(o, XRDC_CONTINUE_CHUNK);
72 size_t want = (size_t) ((job->si->size - off < (int64_t) cap)
73 ? (size_t) (job->si->size - off)
74 : cap);
75 ssize_t n = brix_rfile_pread(&rf, off, buf, want, st);
76
77 if (n < 0) {
78 rc = -1;
79 break;
80 }
81 if (n == 0) {
82 brix_status_set(st, XRDC_EPROTO, 0,
83 "source ended early at %lld (size said %lld)",
84 (long long) off, (long long) job->si->size);
85 rc = -1;
86 break;
87 }
88 if (pwrite(fd, buf, (size_t) n, (off_t) off) != n) { /* vfs-seam-allow: local copy destination, not export data */
89 brix_status_set(st, XRDC_ESOCK, errno, "local write failed: %s",
90 strerror(errno));
91 rc = -1;
92 break;
93 }
94 off += n;
95 if (o->progress != NULL) {
96 o->progress(o->progress_arg, (long long) off,
97 (long long) job->si->size);
98 }
99 /* --xrate pacing counts only the RESUMED bytes (off - start). */
100 if (brix_pump_pace(o, t0, off - start, st) != 0) {
101 rc = -1;
102 break;
103 }
104 if (brix_copy_quit_requested()) {
105 brix_status_set(st, XRDC_ESOCK, EINTR, "cancelled (signal)");
106 rc = -1;
107 break;
108 }
109 }
110 free(buf);
111 {
112 brix_status tw;
113 brix_status_clear(&tw);
114 brix_rfile_close(&rf, rc == 0 ? st : &tw);
115 }
116 return rc;
117}

120/* ---- Run a --continue download end-to-end ----
121 *
122 * WHAT: Returns 1 when it handled the transfer (verdict in *out_rc), 0 when
123 * --continue was not requested (caller proceeds normally). Same
124 * handled? contract as the xcp/parallel gates.
125 *
126 * WHY: §7.6 — resume an interrupted download at the partial's size instead
127 * of restarting; see the file header for the fail-open-partial /
128 * fail-closed-complete split.
129 *
130 * HOW: 1. stat the destination: absent → start 0; a non-regular file or one
131 * LARGER than the source are usage errors; equal size skips straight
132 * to the (optional) checksum verdict.
133 * 2. Open O_WRONLY|O_CREAT (never O_TRUNC), pull the tail, fsync.
134 * 3. Transfer failure keeps the partial (the mode's contract); a
135 * completed file failing --cksum is unlinked like the normal path.
136 */
137int
138copy_download_continue(const download_job_t *job, int *out_rc, brix_status *st)
139{
140 const brix_copy_opts *o = job->o;
141 struct stat sb;
142 int64_t start = 0;
143 int fd, rc = 0;
144
145 if (!o->cont) {
146 return 0;
147 }
148
149 if (stat(job->du->path, &sb) == 0) {
150 if (!S_ISREG(sb.st_mode)) {
151 brix_status_set(st, XRDC_EUSAGE, 0,
152 "--continue: destination is not a regular file: %s",
153 job->du->path);
154 *out_rc = -1;
155 return 1;
156 }
157 start = (int64_t) sb.st_size;
158 }
159 if (start > job->si->size) {
160 brix_status_set(st, XRDC_EUSAGE, 0,
161 "--continue: local file (%lld bytes) is larger than "
162 "the source (%lld) — refusing to guess",
163 (long long) start, (long long) job->si->size);
164 *out_rc = -1;
165 return 1;
166 }
167
168 if (start < job->si->size) {
169 fd = open(job->du->path, O_WRONLY | O_CREAT, 0644);
170 if (fd < 0) {
171 brix_status_set(st, XRDC_ESOCK, errno, "open %s failed: %s",
172 job->du->path, strerror(errno));
173 *out_rc = -1;
174 return 1;
175 }
176 rc = continue_pull_tail(job, fd, start, st);
177 if (rc == 0 && fsync(fd) != 0) {
178 brix_status_set(st, XRDC_ESOCK, errno, "fsync failed: %s",
179 strerror(errno));
180 rc = -1;
181 }
182 close(fd);
183 if (rc != 0) {
184 *out_rc = -1; /* the partial stays — that is the mode's point */
185 return 1;
186 }
187 }
188
189 if (o->cksum != NULL) {
190 rc = download_reconcile_cksum(job, job->du->path, st);
191 if (rc != 0) {
192 unlink(job->du->path); /* completed-but-corrupt: never kept */
193 }
194 }
195 *out_rc = rc;
196 return 1;
197}
