10/* ---- op_path_backend_host --------------------------------------------------
11 *
12 * WHAT: Extract the bare host from a "root://host:port" storage-backend URL as a
13 * borrowed view into `url` — handling the "roots://" scheme, bracketed IPv6
14 * ([::1]:port), and an optional trailing "/path". Empty view when absent.
15 *
16 * WHY: §5.7 krb5 EXCHANGE. The modern brix_storage_backend grammar parses the
17 * driver host lazily in the sd_xroot factory, so conf->cache_origin_host is
18 * empty on a plain root:// export; the delegated-TGT origin SPN derivation
19 * (brix_krb5_deleg_origin_spn) needs the host, recovered from the URL here.
20 * The derived SPN is only a fallback — the origin's advertised "&P=krb5,<spn>"
21 * wins at auth time — but a non-empty host is required for the bind to run.
22 *
23 * HOW: Strip the scheme prefix, then take the bracketed IPv6 body or the token up
24 * to the first ':' or '/'. A borrowed view (no allocation); the bytes live on
25 * the config and outlive the op. */
26static ngx_str_t
27op_path_backend_host(const ngx_str_t *url)
28{
29 ngx_str_t h = { 0, NULL };
30 u_char *p, *end, *stop, *rb;
31
32 if (url == NULL || url->len == 0) {
33 return h;
34 }
35 p = url->data;
36 end = url->data + url->len;
37
38 if ((size_t) (end - p) > sizeof("roots://") - 1
39 && ngx_strncmp(p, "roots://", sizeof("roots://") - 1) == 0)
40 {
41 p += sizeof("roots://") - 1;
42 } else if ((size_t) (end - p) > sizeof("root://") - 1
43 && ngx_strncmp(p, "root://", sizeof("root://") - 1) == 0)
44 {
45 p += sizeof("root://") - 1;
46 }
47
48 if (p < end && *p == '[') { /* [IPv6]:port */
49 rb = ngx_strlchr(p, end, ']');
50 if (rb == NULL) {
51 return h;
52 }
53 h.data = p + 1;
54 h.len = (size_t) (rb - (p + 1));
55 return h;
56 }
57
58 stop = p; /* host ends at ':' or '/' */
59 while (stop < end && *stop != ':' && *stop != '/') {
60 stop++;
61 }
62 h.data = p;
63 h.len = (size_t) (stop - p);
64 return h;
65}

103/* ---- brix_root_vfs_bind_session ----------------------------------------------
104 *
105 * WHAT: Bind the session's captured raw bearer JWT onto a cred-bound VFS ctx for
106 * backend PASSTHROUGH. See op_path.h for the full contract.
107 *
108 * WHY: Lets a remote-backed root:// export authenticate the backend leg AS the
109 * inbound user. Two forwardable credentials ride the GSI login: the raw
110 * bearer JWT (ctx->bearer_token) and — when the client opts in and the DN
111 * is proven (phase-70 §5.1, gsi_promote_fullproxy) — a full x509 proxy
112 * (ctx->deleg_proxy_pem, chain + private key) for backend PASSTHROUGH.
113 *
114 * HOW: No-op on the default SELECT export. Otherwise wraps whichever
115 * credential(s) the session captured as ngx_str_t (bytes owned by the
116 * session ctx, outliving the op) and hands them to brix_vfs_deleg_bind on
117 * the VFS ctx's pool. The full proxy is forwarded only in PASSTHROUGH
118 * mode — the only strategy that replays the user's own credential
119 * verbatim. If neither credential is present, nothing is bound. */
120void
121brix_root_vfs_bind_session(brix_ctx_t *ctx,
122 ngx_stream_brix_srv_conf_t *conf,
123 brix_vfs_ctx_t *vctx)
124{
125 ngx_str_t bearer;
126 const ngx_str_t *bearer_arg = NULL;
127 const ngx_str_t *proxy_arg = NULL;
128
129 if (ctx == NULL || vctx == NULL) {
130 return;
131 }
132
133 /* phase-110 W3: every VFS ctx the session builds folds into the session's
134 * I/O monitor (context.h) — this is the root plane's ONE per-session
135 * post-init hook, called at all 14 ctx-build sites, which is what makes
136 * the stream $brix_* surface complete by construction rather than by
137 * remembering each site. Unconditional and before the delegation
138 * early-return: monitoring does not depend on the credential mode. The
139 * monitor is embedded in the pcalloc'd per-connection ctx, so this is an
140 * event-loop-allocated target the offload thread may scalar-write. */
141 vctx->io_monitor = &ctx->io_monitor;
142 /* phase-110 W7: the client address for the JSON access log's `remote`.
143 * ctx->peer_ip is an already-NUL-terminated cstr (populated for authdb HOST
144 * rules), so this is a borrow, not an alloc — safe on any thread. */
145 if (ctx->login.peer_ip[0] != '\0') {
146 vctx->peer = ctx->login.peer_ip;
147 }
148
149 op_path_bind_authz(ctx, conf, vctx);
150
151 if (conf == NULL || conf->common.backend_delegation == BRIX_CRED_SELECT) {
152 return;
153 }
154
155 bearer.data = (u_char *) ctx->bearer_token;
156 bearer.len = ngx_strlen(ctx->bearer_token);
157 if (bearer.len > 0) {
158 /* Backend audience gate (phase-70 §5.2 / P90-70.9): a bearer forwarded
159 * verbatim must name the backend in its aud — on refusal nothing is
160 * bound and the service-cred policy applies. */
161 bearer_arg = brix_proto_deleg_gate_bearer(&bearer, &conf->common,
162 vctx->log);
163 }
164
165 /* A full proxy is a PASSTHROUGH-only credential: it is presented to the
166 * upstream unmodified, so it makes sense only when the resolved mode replays
167 * the user's own credential. */
168 if (ctx->deleg_proxy_pem.len > 0
169 && (enum brix_cred_mode) conf->common.backend_delegation
170 == BRIX_CRED_PASSTHROUGH)
171 {
172 proxy_arg = &ctx->deleg_proxy_pem;
173 }
174
175 /* Phase 70 §5.7: an inbound-captured forwarded TGT (ctx->krb5.ccache) binds
176 * as a krb5 EXCHANGE credential so the origin leg re-authenticates AS the
177 * user over GSSAPI. The origin service principal is derived from the
178 * configured origin host + the gateway realm; independent of the bearer/proxy
179 * binding, and it allocates its own live-cred bag (like the STS/SSS stamps),
180 * so it must run before the no-bearer/no-proxy early return. */
181#if (BRIX_HAVE_KRB5)
182 {
183 ngx_str_t cc;
184 ngx_str_t spn;
185 ngx_str_t origin_host = conf->cache_origin_host;
186
187 /* The modern "brix_storage_backend root://host:port" grammar parses the
188 * origin host lazily in the driver factory (sd_xroot), so on a plain
189 * root:// export conf->cache_origin_host is empty here — only the legacy
190 * tier grammar fills it. Recover the host from the backend URL so the
191 * delegated-TGT origin SPN can still be derived; without it the krb5
192 * EXCHANGE bind would silently no-op and the origin leg would fall back
193 * to the (absent) service credential. */
194 if (origin_host.len == 0) {
195 origin_host = op_path_backend_host(&conf->common.storage_backend);
196 }
197
198 cc.data = (u_char *) ctx->krb5.ccache;
199 cc.len = ngx_strlen(ctx->krb5.ccache);
200
201 if (brix_krb5_deleg_origin_spn(&cc, conf->common.backend_krb5_forwardable,
202 &origin_host, &conf->krb5.principal,
203 vctx->pool, &spn) == NGX_OK)
204 {
205 brix_vfs_deleg_set_krb5(vctx,
206 (enum brix_cred_mode) conf->common.backend_delegation, &cc, &spn);
207 }
208 }
209#endif
210
211 if (bearer_arg == NULL && proxy_arg == NULL) {
212 return;
213 }
214
215 (void) brix_vfs_deleg_bind(vctx->pool, vctx,
216 (enum brix_cred_mode) conf->common.backend_delegation,
217 bearer_arg, proxy_arg);
218
219 brix_proto_deleg_stamp_conf(vctx, &conf->common);
220
221 /* P90-70.4: stamp the server's GSI trust store so the VFS deleg gate
222 * re-runs the RFC-3820 chain-trust check on the pushed proxy before
223 * materialising it (gsi_promote_fullproxy only DN-matches the push).
224 * Depth 0 = OpenSSL default, matching the root:// login verify. */
225 brix_vfs_deleg_set_ca_store(vctx, conf->gsi_store, 0);
226}
