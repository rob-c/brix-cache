17/*
18 * WHAT: ngx_stream_brix_send() is the write event handler called by nginx
19 * whenever the kernel socket send buffer has room again after a
20 * previous c->send()/c->send_chain() returned NGX_AGAIN.
21 *
22 * Drives the async flush cycle:
23 * (1) calls brix_flush_pending() to drain wbuf/wchain into the socket;
24 * (2) if NGX_AGAIN, returns and waits for next write event;
25 * (3) on success with state==XRD_ST_SENDING, transitions to
26 * XRD_ST_REQ_HEADER and re-enters recv loop for next request;
27 * (4) if tls_pending after flushing kXR_haveTLS response, triggers
28 * TLS handshake via brix_start_tls().
29 *
30 * Security invariant: timeout → disconnect immediately.
31 *
32 * Thread safety: single-owner per connection on nginx event thread —
33 * no locking required.
34 *
35 * Returns only when flush=NGX_AGAIN or state≠SENDING.
36 *
37 * WHY: This handler is the critical counterpart to recv() in the async I/O
38 * cycle — without it, response buffers would stall until the next
39 * natural epoll write event detection.
40 *
41 * Flush pending ensures all queued response data (wire protocol headers
42 * + payload) are sent efficiently to the client socket rather than
43 * accumulating in memory indefinitely.
44 *
45 * TLS handshake trigger enables in-protocol upgrade for roots://
46 * clients when kXR_haveTLS response has been flushed and the server is
47 * ready to negotiate secure transport.

98{

100 ngx_stream_session_t *s;
101 ngx_stream_brix_srv_conf_t *conf;
102 brix_ctx_t *ctx;
103 ngx_int_t rc;
104
105 c = wev->data;
106 s = c->data;
107 ctx = ngx_stream_get_module_ctx(s, ngx_stream_brix_module);
108 conf = ngx_stream_get_module_srv_conf(s, ngx_stream_brix_module);
109
110 if (wev->timedout) {
111 ngx_log_error(NGX_LOG_INFO, c->log, NGX_ETIMEDOUT,
112 "brix: write timed out");
113 /* Phase 39: the response-drain deadline fired — the consumer made no
114 * progress for brix_send_timeout (slow / half-open reader). Attribute
115 * it and tear down via the single disconnect funnel. */
116 ctx->deadline.send_armed = 0;
117 BRIX_SRV_METRIC_INC(ctx, send_drain_timeouts_total);
118 if (brix_defer_teardown_if_writing(ctx, c, NGX_STREAM_OK)) {
119 return;
120 }
121 brix_on_disconnect(ctx, c);
122 brix_close_all_files(ctx);
123 ngx_stream_finalize_session(s, NGX_STREAM_OK);
124 return;
125 }
126
127 rc = brix_flush_pending(ctx, c);
128 if (rc == NGX_ERROR) {
129 if (brix_defer_teardown_if_writing(ctx, c,
130 NGX_STREAM_INTERNAL_SERVER_ERROR)) {
131 return;
132 }
133 brix_on_disconnect(ctx, c);
134 brix_close_all_files(ctx);
135 ngx_stream_finalize_session(s, NGX_STREAM_INTERNAL_SERVER_ERROR);
136 return;
137 }
138
139 if (rc == NGX_AGAIN) {
140 return;
141 }
142
143 if (ctx->state != XRD_ST_SENDING) {
144 return;
145 }
146
147 /*
148 * Phase 31 W2.1: a windowed read's chunk just finished draining from
149 * read_scratch — read and send the next window before accepting any new
150 * request. The pump posts the next window (state XRD_ST_AIO) or finishes
151 * and resumes the read side itself.
152 */
153 if (ctx->rd.win_active) {
154 brix_read_window_pump(ctx, c, conf);
155 return;
156 }
157
158 ctx->state = XRD_ST_REQ_HEADER;
159 ctx->recv.hdr_pos = 0;
160 ngx_log_debug3(NGX_LOG_DEBUG_STREAM, c->log, 0,
161 "brix: send_done avail=%d ready=%d active=%d",
162 c->read->available, (int) c->read->ready,
163 (int) c->read->active);
164
165 if (ctx->tls_pending) {
166 brix_start_tls(ctx, c, conf);
167 return;
168 }
169
170 ngx_stream_brix_recv(c->read);
171}
