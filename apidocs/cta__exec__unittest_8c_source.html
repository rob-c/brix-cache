21static char g_alerts[8][64];
22static int g_nalerts;
23static void rec_alert(void *ctx, const char *msg)
24{
25 (void) ctx;
26 if (g_nalerts < 8) {
27 size_t n = strlen(msg);
28 if (n >= sizeof(g_alerts[0])) n = sizeof(g_alerts[0]) - 1;
29 memcpy(g_alerts[g_nalerts], msg, n);
30 g_alerts[g_nalerts][n] = '\0';
31 g_nalerts++;
32 }
33}

43/*
44 * Run one op through a vtable on a throwaway queue, hand back the entry's
45 * final state, and leave the executor's return in *rc. The five cases below
46 * differ only in vtable, op, and expected outcome — the setup was identical
47 * in all of them.
48 */
49static cta_state_t
50run_one(const cta_exec_vtbl_t *vt, cta_op_t op, int *rc)
51{
52 brix_cta_queue_t *q = cta_queue_create();
53 cta_req_t *e = submit(q, op);
54 cta_progress_t prog = { q, NULL, rec_alert, NULL };
55 cta_state_t st;
56
57 g_nalerts = 0;
58 *rc = cta_exec_run(vt, e, &prog);
59 st = e->state;
60 cta_queue_destroy(q);
61 return st;
62}

102/* ---- the transition seam ------------------------------------------------
103 *
104 * The executor moves entries through cta_progress_t::transition so that the
105 * nginx build can take the shared-zone lock once per transition instead of
106 * holding it for the length of an archive or retrieve. These pin both halves
107 * of that seam: the hook is used when supplied, and the queue is called
108 * directly when it is not.
109 */
110
111static int g_hook_calls;
112static int
113counting_transition(brix_cta_queue_t *q, cta_req_t *e, cta_state_t to)
114{
115 g_hook_calls++;
116 return cta_queue_transition(q, e, to);
117}

119static void test_the_executor_transitions_through_the_hook(void)
120{
121 brix_cta_queue_t *q = cta_queue_create();
122 cta_req_t *e = submit(q, CTA_OP_ARCHIVE);
123 cta_progress_t prog = { q, counting_transition, rec_alert, NULL };
124 g_hook_calls = 0;
125 CHECK(cta_exec_run(cta_exec_test_vtbl(), e, &prog) == 0);
126 CHECK(e->state == CTA_ST_COMPLETE);
127 CHECK(g_hook_calls == 3); /* QUEUED, ACTIVE, COMPLETE */
128 cta_queue_destroy(q);
129}

143/* Security-negative: an executor handed no queue must refuse, not transition
144 * an entry it cannot journal. This is the shape a caller hits when the SHM
145 * zone is absent — the case that must NOT silently fall back to a private
146 * queue and fabricate a durability guarantee. */
147static void test_no_queue_refuses_every_op(void)
148{
149 brix_cta_queue_t *q = cta_queue_create();
150 cta_req_t *arch = submit(q, CTA_OP_ARCHIVE);
151 cta_req_t *retr = submit(q, CTA_OP_RETRIEVE);
152 cta_req_t *canc = submit(q, CTA_OP_CANCEL);
153 cta_progress_t none = { NULL, NULL, rec_alert, NULL };
154
155 CHECK(cta_exec_run(cta_exec_test_vtbl(), arch, &none) == -1);
156 CHECK(arch->state == CTA_ST_SUBMITTED);
157 CHECK(cta_exec_run(cta_exec_test_vtbl(), retr, &none) == -1);
158 CHECK(retr->state == CTA_ST_SUBMITTED);
159 CHECK(cta_exec_run(cta_exec_test_vtbl(), canc, &none) == -1);
160 CHECK(canc->state == CTA_ST_SUBMITTED);
161
162 /* and with no sink at all */
163 CHECK(cta_exec_run(cta_exec_test_vtbl(), arch, NULL) == -1);
164 CHECK(arch->state == CTA_ST_SUBMITTED);
165 cta_queue_destroy(q);
166}
