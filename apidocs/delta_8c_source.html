1/* delta.c — phase-87 G10 cross-revision delta transfer (server leg).
2 *
3 * WHAT: a CAS data GET carrying `X-Brix-Delta-Base: <40-hex>` (the sha1 of
4 * a CAS object the client already holds — typically the revision-N
5 * catalog while fetching the N+1 catalog) may be answered as a zstd
6 * DELTA of the target against that base (Content-Encoding: zstd-delta)
7 * when the base is cache-RESIDENT and the delta is strictly smaller
8 * than identity. Gated brix_cvmfs_delta (default off).
9 * WHY: frequent-publish repos (nightlies, calibration) republish catalogs
10 * that are 99% identical to the previous revision; shipping the whole
11 * object every time is RTT+bandwidth waste the client can prove it
12 * doesn't need. A delta against the client's own held base captures
13 * exactly the changed bytes.
14 * HOW: the delta IS the G3 dictionary codec with the base object as a raw-
15 * content zstd dictionary (cvmfs_dict_compress — zstd loads non-ZDICT
16 * bytes as a raw prefix, i.e. patch-from semantics), so no new codec
17 * exists. The base is resolved through the cache tier's own cstore
18 * (cinfo COMPLETE + serve_open): a miss NEVER fans out to the origin —
19 * this endpoint cannot amplify one GET into extra origin fetches, the
20 * client just gets the whole object (NGX_DECLINED ⇒ identity serve).
21 * TRUST IS UNCHANGED: the coding is a reversible transform of the
22 * STORED bytes, and a raw-content dictionary has no embedded dictID,
23 * so a client reconstructing against the wrong base gets bytes whose
24 * CAS hash does not match the name — its ordinary CAS verify rejects
25 * them and it refetches whole. The delta path adds no trust surface;
26 * it only ever changes how many bytes cross the wire.
27 */

41/* Base/target size cap: the win case is catalogs (100s of KiB — a few MiB);
42 * 4 MiB each keeps base+target+output request-pool buffering bounded at
43 * 12 MiB and both sides inside zstd level-19's 8 MiB match window without
44 * advanced-parameter plumbing. */
45#define CVMFS_DELTA_MAX_OBJ (4u * 1024u * 1024u)
46#define CVMFS_DELTA_HEXLEN 40u

48static int
49cvmfs_delta_hexlc(char c)
50{
51 return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f');
52}

54/* Read the whole resident base object into `buf` through the cstore's own
55 * serve path (resident-only by construction). 0 on success, -1 on any
56 * short/failed read — callers treat that as "no base" and serve identity. */
57static int
58cvmfs_delta_read_base(brix_cstore_t *cs, const char *key,
59 unsigned char *buf, size_t len)
60{
61 brix_sd_obj_t *obj;
62 size_t off;
63 int verr = 0;
64
65 obj = brix_cstore_serve_open(cs, key, &verr);
66 if (obj == NULL) {
67 return -1;
68 }
69 for (off = 0; off < len; ) {
70 ssize_t rd = obj->driver->pread(obj, buf + off, len - off, (off_t) off);
71 if (rd <= 0) {
72 brix_sd_obj_release(obj);
73 return -1;
74 }
75 off += (size_t) rd;
76 }
77 brix_sd_obj_release(obj);
78 return 0;
79}

81/* Parse + validate the opt-in header into `base_id` (NUL-terminated
82 * lowercase 40-hex). -1 = no usable opt-in (identity serve); base == target
83 * names a zero delta, so identity is already optimal there too. */
84static int
85cvmfs_delta_base_id(ngx_http_request_t *r, ngx_http_brix_cvmfs_ctx_t *ctx,
86 char *base_id)
87{
88 ngx_str_t base;
89 size_t i;
90
91 base = brix_http_get_header(r, "X-Brix-Delta-Base");
92 if (base.len != CVMFS_DELTA_HEXLEN) {
93 return -1;
94 }
95 for (i = 0; i < base.len; i++) {
96 if (!cvmfs_delta_hexlc((char) base.data[i])) {
97 return -1;
98 }
99 }
100 if (ngx_memcmp(base.data, ctx->url.cas_hex, CVMFS_DELTA_HEXLEN) == 0) {
101 return -1;
102 }
103 memcpy(base_id, base.data, CVMFS_DELTA_HEXLEN);
104 base_id[CVMFS_DELTA_HEXLEN] = '\0';
105 return 0;
106}

108/* Resolve the RESIDENT base: cache store, per-repo key and COMPLETE bounded
109 * cinfo. NULL = no usable base (identity serve) — a miss NEVER fans out. */
110static brix_cstore_t *
111cvmfs_delta_resolve_base(ngx_http_request_t *r,
112 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
113 const char *base_id, char *basekey, size_t keycap, size_t *baselen)
114{
115 brix_sd_instance_t *sd;
116 brix_cstore_t *cs;
117 brix_cache_cinfo_t ci;
118
119 sd = cvmfs_resolve_sd(r, lcf);
120 cs = (sd != NULL) ? brix_sd_cache_cstore(sd) : NULL;
121 if (cs == NULL) {
122 return NULL;
123 }
124
125 /* The base inherits the TARGET's suffix class: a catalog's (…C) delta
126 * base is the previous catalog, a plain chunk's base a plain chunk. */
127 if ((size_t) snprintf(basekey, keycap,
128 "/cvmfs/%.*s/data/%.2s/%.38s%.*s",
129 (int) ctx->url.repo_len, ctx->url.repo,
130 base_id, base_id + 2,
131 ctx->url.cas_suffix != 0 ? 1 : 0,
132 &ctx->url.cas_suffix) >= keycap)
133 {
134 return NULL;
135 }
136
137 /* Residency gate: COMPLETE cinfo, bounded size, NO origin fan-out. */
138 if (brix_cstore_cinfo_load(cs, basekey, &ci) != NGX_OK
139 || !(ci.flags & BRIX_CINFO_F_COMPLETE)
140 || ci.size == 0
141 || ci.size > CVMFS_DELTA_MAX_OBJ)
142 {
143 return NULL;
144 }
145 *baselen = (size_t) ci.size;
146 return cs;
147}

149/* Stat the target and read base + target into fresh request-pool buffers
150 * (`*out` is capped at srclen: the delta must WIN or compress fails).
151 * -1 = identity serve. */
152static int
153cvmfs_delta_load(ngx_http_request_t *r, brix_cstore_t *cs,
154 const char *basekey, size_t baselen, brix_vfs_file_t *fh,
155 brix_vfs_stat_t *vst, size_t *srclen,
156 unsigned char **basebuf, unsigned char **src, unsigned char **out)
157{
158 size_t off, n;
159
160 if (brix_vfs_file_stat(fh, vst) != NGX_OK
161 || vst->is_directory
162 || vst->size <= 0
163 || (size_t) vst->size > CVMFS_DELTA_MAX_OBJ)
164 {
165 return -1;
166 }
167 n = (size_t) vst->size;
168
169 *basebuf = ngx_pnalloc(r->pool, baselen);
170 *src = ngx_pnalloc(r->pool, n);
171 *out = ngx_pnalloc(r->pool, n);
172 if (*basebuf == NULL || *src == NULL || *out == NULL) {
173 return -1;
174 }
175
176 if (cvmfs_delta_read_base(cs, basekey, *basebuf, baselen) != 0) {
177 return -1; /* vanished/short under us — identity serve */
178 }
179 for (off = 0; off < n; ) {
180 ssize_t rd = brix_vfs_file_pread(fh, *src + off, n - off, (off_t) off);
181 if (rd <= 0) {
182 return -1;
183 }
184 off += (size_t) rd;
185 }
186 *srclen = n;
187 return 0;
188}

190/* Emit the committed delta-coded response (headers + single in-memory buf). */
191static ngx_int_t
192cvmfs_delta_emit(ngx_http_request_t *r, time_t mtime, const char *base_id,
193 unsigned char *out, size_t srclen, size_t outlen)
194{
195 ngx_table_elt_t *h = NULL;
196 ngx_chain_t chain;
197 ngx_buf_t *b;
198 ngx_int_t rc;
199
200 r->headers_out.status = NGX_HTTP_OK;
201 r->headers_out.content_length_n = (off_t) outlen;
202 r->headers_out.last_modified_time = mtime;
203 ngx_str_set(&r->headers_out.content_type, "application/octet-stream");
204 r->headers_out.content_type_len = r->headers_out.content_type.len;
205
206 if (brix_http_set_header(r, "Content-Encoding", "zstd-delta", &h) != NGX_OK
207 || brix_http_set_header(r, "X-Brix-Delta-Base", base_id, NULL) != NGX_OK
208 || brix_http_set_header(r, "Vary", "X-Brix-Delta-Base", NULL) != NGX_OK)
209 {
210 return NGX_HTTP_INTERNAL_SERVER_ERROR;
211 }
212 r->headers_out.content_encoding = h;
213
214 b = ngx_calloc_buf(r->pool);
215 if (b == NULL) {
216 return NGX_HTTP_INTERNAL_SERVER_ERROR;
217 }
218 b->memory = 1;
219 b->pos = out;
220 b->last = out + outlen;
221 b->start = b->pos;
222 b->end = b->last;
223 b->last_buf = 1;
224 b->last_in_chain = 1;
225 chain.buf = b;
226 chain.next = NULL;
227
228 ngx_log_error(NGX_LOG_DEBUG, r->connection->log, 0,
229 "cvmfs-delta: coded serve %uz -> %uz base=%s", srclen, outlen,
230 base_id);
231
232 rc = ngx_http_send_header(r);
233 if (rc == NGX_ERROR || rc > NGX_OK || r->header_only) {
234 return rc;
235 }
236 return ngx_http_output_filter(r, &chain);
237}

239ngx_int_t
240brix_cvmfs_delta_try_serve(ngx_http_request_t *r,
241 ngx_http_brix_cvmfs_loc_conf_t *lcf, ngx_http_brix_cvmfs_ctx_t *ctx,
242 brix_vfs_file_t *fh)
243{
244 brix_cstore_t *cs;
245 brix_vfs_stat_t vst;
246 unsigned char *basebuf, *src, *out;
247 char basekey[512];
248 char base_id[CVMFS_DELTA_HEXLEN + 1];
249 size_t baselen, srclen, outlen = 0;
250
251 if (!lcf->cvmfs.delta
252 || r->method != NGX_HTTP_GET
253 || ctx == NULL
254 || ctx->url.cls != CVMFS_URL_CAS
255 || ctx->url.cas_hex_len != CVMFS_DELTA_HEXLEN
256 || r->headers_in.range != NULL)
257 {
258 return NGX_DECLINED;
259 }
260
261 if (cvmfs_delta_base_id(r, ctx, base_id) != 0) {
262 return NGX_DECLINED;
263 }
264
265 cs = cvmfs_delta_resolve_base(r, lcf, ctx, base_id, basekey,
266 sizeof(basekey), &baselen);
267 if (cs == NULL) {
268 return NGX_DECLINED;
269 }
270
271 if (cvmfs_delta_load(r, cs, basekey, baselen, fh, &vst, &srclen,
272 &basebuf, &src, &out) != 0)
273 {
274 return NGX_DECLINED;
275 }
276
277 if (cvmfs_dict_compress(basebuf, baselen, src, srclen, out, srclen,
278 &outlen) != 0)
279 {
280 return NGX_DECLINED; /* no gain (or error) → identity serve */
281 }
282
283 /* commit: from here the response is ours and fh is consumed */
284 brix_vfs_close(fh, r->connection->log);
285
286 return cvmfs_delta_emit(r, vst.mtime, base_id, out, srclen, outlen);
