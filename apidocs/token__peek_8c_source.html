1/*
2 * token_peek.c — the UNVERIFIED claim readers.
3 *
4 * WHAT: brix_token_peek_iss() and brix_token_peek_exp() read one claim each out
5 * of a compact JWS payload WITHOUT checking the signature.
6 *
7 * WHY: both are needed before, or entirely outside, verification. peek_iss runs
8 * first by necessity — the registry has to pick an issuer, and thus a key set,
9 * before anything about the token can be trusted; the claim is re-derived from
10 * verified claims afterwards. peek_exp runs where a signature check would be
11 * meaningless: the exchange cache reads the lifetime a token it just minted
12 * declares about ITSELF so an entry never outlives its credential, and the TPC
13 * outbound renewal leg reads the same claim to decide when to mint another.
14 * Neither authorises anything.
15 *
16 * They live in their own translation unit because they are the only token
17 * functions whose entire dependency set is xrdjwt_split + b64url_decode +
18 * json_get_*. Keeping them here means a caller that wants nothing but the
19 * declared lifetime -- exchange_cache.c, and its C unit test -- links three
20 * small objects instead of dragging in the issuer registry, the signature
21 * pipeline, the scope parser and the subject mapfile. Before this split the
22 * exchange_cache unit could not be linked at all.
23 *
24 * HOW: split the compact JWS, base64url-decode segment 1, read the one claim.
25 * Every failure -- not a compact JWS, an undecodable payload, a missing or
26 * non-positive claim -- returns -1 with the out parameter already cleared, so a
27 * caller that ignores the return value still cannot read a stale value.
28 */

38/* brix_token_peek_iss — read the "iss" claim WITHOUT trusting the signature
39 * (xrdjwt_split + b64url_decode + json_get_string): the registry must pick an
40 * issuer, and thus its verification keys, before it can trust anything, so this
41 * read is explicitly untrusted and re-derived from verified claims afterwards. */
42int
43brix_token_peek_iss(const char *token, size_t token_len,
44 char *out, size_t outsz)
45{
46 xrdjwt_seg seg[3];
47 u_char pay[BRIX_BEARER_TOKEN_MAX];
48 ssize_t n;
49
50 out[0] = '\0';
51
52 if (xrdjwt_split(token, token_len, seg) != 0) {
53 return -1; /* not a compact JWS */
54 }
55 n = b64url_decode(seg[1].p, seg[1].n, pay, sizeof(pay) - 1);
56 if (n < 0) {
57 return -1;
58 }
59 pay[n] = '\0';
60 if (json_get_string((char *) pay, (size_t) n, "iss", out, outsz) < 0) {
61 return -1;
62 }
63 return 0;
64}

66/* brix_token_peek_exp — read the "exp" claim WITHOUT trusting the signature.
67 * Promoted out of exchange_cache.c (where it was tx_cache_minted_exp) when TPC
68 * outbound renewal became a second caller: both need the lifetime a token
69 * declares about ITSELF, and neither authorises anything on it — the cache uses
70 * it to avoid outliving the credential, the renewal leg to decide when to mint
71 * another. A signature check would be meaningless here: for the cache the token
72 * was just minted by the issuer we asked, and for renewal it is a credential
73 * this server already presented and the source already validated. */
74int
75brix_token_peek_exp(const char *token, size_t token_len, time_t *out)
76{
77 xrdjwt_seg seg[3];
78 u_char payload[BRIX_B64_DECODE_MAX];
79 ssize_t plen;
80 int64_t exp = 0;
81
82 *out = 0;
83
84 if (xrdjwt_split(token, token_len, seg) != 0) {
85 return -1; /* not a compact JWS */
86 }
87 plen = b64url_decode(seg[1].p, seg[1].n, payload, sizeof(payload));
88 if (plen <= 0) {
89 return -1;
90 }
91 if (json_get_int64((const char *) payload, (size_t) plen, "exp", &exp) != 0
92 || exp <= 0)
93 {
94 return -1;
95 }
96
97 *out = (time_t) exp;
98 return 0;
99}
