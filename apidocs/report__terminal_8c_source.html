26/* ---- Print detailed evidence for one actionable result ----
27 *
28 * WHAT: Emits id, verdict, reason, expectation, observation and authority.
29 * WHY: A non-green row must be independently understandable and reproducible.
30 * HOW: Print only populated optional fields after the stable leading line.
31 */
32static void
33terminal_detail(FILE *out, const brix_conform_result *result)
34{
35 fprintf(out, " %s %s", result->clause->id,
36 brix_conform_verdict_name(result->verdict));
37 if (result->reason[0] != '\0') fprintf(out, " %s", result->reason);
38 fputc('\n', out);
39 if (result->ecosystem_verdict != result->strict_verdict)
40 fprintf(out, " ecosystem: %s strict: %s (%s)\n",
41 brix_conform_verdict_name(result->ecosystem_verdict),
42 brix_conform_verdict_name(result->strict_verdict),
43 result->strict_reason);
44 if (result->expected[0] != '\0')
45 fprintf(out, " expected: %s\n", result->expected);
46 if (result->received[0] != '\0')
47 fprintf(out, " received: %s\n", result->received);
48 fprintf(out, " ref: %s oracle-mask: 0x%x source: %s\n",
49 brix_conform_reference_name(result->clause->reference),
50 result->clause->oracles, result->clause->source);
51 fprintf(out, " negative-fixture: %s\n",
52 result->clause->negative_fixture);
53}

55/* ---- Print actionable result details ----
56 *
57 * WHAT: Emits red/unknown rows, filtered to changes when a baseline is active.
58 * WHY: Baseline mode should show transitions without hiding aggregate truth.
59 * HOW: Walk once, skip unchanged rows, delegate stable detail formatting.
60 */
61static void
62terminal_details(FILE *out, const brix_conform_run_results *results)
63{
64 size_t index;
65
66 for (index = 0; index < results->count; index++) {
67 const brix_conform_result *result = &results->results[index];
68
69 if (results->baseline_applied && !result->changed) continue;
70 if (result->verdict == BRIX_CONFORM_VERDICT_DIVERGENT
71 || result->verdict == BRIX_CONFORM_VERDICT_INCONCLUSIVE)
72 terminal_detail(out, result);
73 }
74}

76/* ---- Render the complete terminal report ----
77 *
78 * WHAT: Prints metadata, every family aggregate and all actionable rows.
79 * WHY: The terminal contract is the operator's immediate certification view.
80 * HOW: Exploit registry family ordering for one pass, then detail red/unknown.
81 */
82void
83brix_conform_report_terminal(FILE *out, const brix_conform_report_meta *meta,
84 const brix_conform_run_results *results)
85{
86 size_t index = 0;
87
88 if (out == NULL || meta == NULL || results == NULL) return;
89 fprintf(out, "xrd conform target=%s profile=%s shape=%s budget=%s reference=%s\n",
90 meta->target, meta->profile, meta->shape, meta->budget, meta->reference);
91 fprintf(out, "xrd conform version=%s usage=requests:%u/%u bytes:%llu/%llu wall:%us\n",
92 meta->tool_version, meta->requests, meta->request_limit, meta->bytes,
93 meta->byte_limit, meta->wall_seconds);
94 for (index = 0; index < meta->also_target_count; index++)
95 fprintf(out, "xrd conform also=%s\n", meta->also_targets[index]);
96 for (index = 0; index < meta->negative_verify_count; index++)
97 fprintf(out, "xrd conform verify-negative=%s result=%s\n",
98 meta->negative_verify[index],
99 meta->negative_verify_passed ? "passed" : "failed");
100 index = 0;
101 while (index < results->count) {
102 const char *family = results->results[index].clause->family;
103 size_t ok = 0, divergent = 0, skipped = 0, inconclusive = 0;
104
105 while (index < results->count
106 && strcmp(results->results[index].clause->family, family) == 0) {
107 switch (results->results[index].verdict) {
108 case BRIX_CONFORM_VERDICT_CONFORMANT: ok++; break;
109 case BRIX_CONFORM_VERDICT_DIVERGENT: divergent++; break;
110 case BRIX_CONFORM_VERDICT_SKIPPED: skipped++; break;
111 case BRIX_CONFORM_VERDICT_INCONCLUSIVE: inconclusive++; break;
112 }
113 index++;
114 }
115 terminal_family(out, family, ok, divergent, skipped, inconclusive);
116 }
117 terminal_details(out, results);
118 fprintf(out, "TOTAL %zu conformant · %zu divergent · %zu skipped · %zu inconclusive\n",
119 results->conformant, results->divergent, results->skipped,
120 results->inconclusive);
121 if (results->baseline_applied)
122 fprintf(out, "BASELINE %zu clause verdict(s) changed\n", results->changed);
123}
