1/*
2 * write_staged.c — root:// block-write → whole-object staged-commit adapter.
3 *
4 * WHAT: The write/sync/close hooks for a root:// write handle whose backend LEAF
5 * advertises NO random write (BRIX_SD_CAP_RANDOM_WRITE) and has no pwrite
6 * slot — a whole-object store (sd_http/s3 and any driver whose write is a
7 * single commit-time PUT). brix_open_dispatch_staged put such a handle in
8 * STAGED mode (ctx->files[idx].writer != NULL — a unified brix_vfs_writer
9 * session opened with the per-user credential already resolved). This file
10 * APPENDS each kXR_write/pgwrite block to that session and COMMITS the whole
11 * object on kXR_sync / kXR_close.
12 *
13 * WHY: The block-oriented root:// write model (open-for-write → write/pgwrite at
14 * offsets → sync/close → driver pwrite) requires a random-write driver; a
15 * whole-object backend cannot pwrite, so a root:// upload used to fail EROFS.
16 * GridFTP STOR and (via http_body) WebDAV/S3 PUT stream a block body through
17 * the same unified brix_vfs_writer; this routes the root:// path through it
18 * too, so every filesystem shares one verified-write call to the VFS layer.
19 *
20 * HOW: Uploads are appended via brix_vfs_writer_write; the writer itself keeps
21 * the sequential contract, spilling an out-of-order extent to local scratch
22 * (phase-107 C1) and refusing only when it cannot (no spill root, capacity,
23 * overlap) — that surfaces here as an I/O error whose errno maps to the
24 * kXR code (ENOSPC → kXR_NoSpace). sync/close
25 * call brix_staged_commit_handle → brix_vfs_writer_commit (one whole-object
26 * PUT, plus the optional read-back CRC check when brix_verify_write is on),
27 * which on success consumes the session; brix_free_fhandle aborts an
28 * uncommitted session so no partial object is published. No goto; each stage
29 * is a small single-purpose function with explicit data flow.
30 */

36/*
37 * brix_staged_append — append `len` bytes at `offset` to the write session,
38 * WITHOUT sending a success reply (the caller chooses the reply frame: kXR_ok for
39 * kXR_write, kXR_status for pgwrite).
40 *
41 * Ordering is the writer's problem now (phase-107 C1: reordered extents spill to
42 * local scratch); a failure — including "cannot spill" ENOSPC — is replied here
43 * with the kXR code mapped from errno, and *rc holds the reply's return value.
44 * Returns NGX_OK when appended (no reply sent yet); NGX_ERROR when the caller
45 * must return *rc immediately.
46 */
47/*
48 * brix_staged_append_raw — the reply-free core of a staged append: forward the
49 * block to the write session (which spills a reordered extent to local scratch,
50 * phase-107 C1), with the success-path byte accounting. Returns
51 * BRIX_STAGED_APPEND_OK on success or BRIX_STAGED_APPEND_IO on a writer error
52 * (errno preserved — ENOSPC when a reordered extent could not be spilled).
53 * Sends nothing and touches no metrics
54 * — the caller chooses the reply/log framing. This is the primitive the chunked
55 * streaming writer (write_stream.c) applies per chunk without acking mid-stream.
56 */
57int
58brix_staged_append_raw(brix_ctx_t *ctx, int idx, int64_t offset,
59 const u_char *buf, size_t len)
60{
61 brix_file_t *file = &ctx->files[idx];
62
63 if (brix_vfs_writer_write(file->writer, buf, len, offset) != NGX_OK) {
64 return BRIX_STAGED_APPEND_IO;
65 }
66
67 file->bytes_written += len;
68 ctx->totals.bytes_written += len;
69 brix_rl_charge_ctx(ctx, len);
70
71 if (file->dashboard_slot >= 0 && ngx_brix_dashboard_shm_zone != NULL) {
72 brix_transfer_slot_update(ngx_brix_dashboard_shm_zone->data,
73 file->dashboard_slot,
74 (ngx_atomic_int_t) len,
75 (int64_t) ngx_current_msec);
76 brix_transfer_slot_count_op(ngx_brix_dashboard_shm_zone->data,
77 file->dashboard_slot, "write");
78 }
79 return BRIX_STAGED_APPEND_OK;
80}

82ngx_int_t
83brix_staged_append(brix_ctx_t *ctx, ngx_connection_t *c, int idx,
84 int64_t offset, const u_char *buf, size_t len, ngx_int_t *rc)
85{
86 char detail[64];
87 int ar;
88
89 ar = brix_staged_append_raw(ctx, idx, offset, buf, len);
90 if (ar == BRIX_STAGED_APPEND_OK) {
91 return NGX_OK;
92 }
93
94 snprintf(detail, sizeof(detail), "%lld+%zu", (long long) offset, len);
95
96 {
97 /* errno → kXR so a refused spill surfaces as kXR_NoSpace (ENOSPC), not a
98 * generic I/O error the client would retry against the same wall. */
99 uint16_t code = brix_kxr_from_errno(errno);
100 const char *ioerr = strerror(errno);
101
102 brix_log_access(ctx, c, "WRITE", ctx->files[idx].path, detail, 0,
103 code, ioerr, 0);
104 BRIX_OP_ERR(ctx, BRIX_OP_WRITE);
105 *rc = brix_send_error(ctx, c, code, ioerr);
106 }
107 return NGX_ERROR;
108}

143/*
144 * brix_staged_commit_handle — commit the whole staged object (one backend PUT,
145 * plus the read-back CRC check when brix_verify_write is on). The kXR_sync /
146 * kXR_close hook. Idempotent: once committed, a later call is a no-op success (a
147 * client that syncs then closes). On failure sets *err_out and returns NGX_ERROR;
148 * brix_vfs_writer_commit already unlinked any published-then-mismatched object, so
149 * no partial/corrupt object is left behind.
150 */
151ngx_int_t
152brix_staged_commit_handle(brix_ctx_t *ctx, int idx, int *err_out)
153{
154 brix_file_t *file = &ctx->files[idx];
155
156 if (file->writer == NULL || file->staged_committed) {
157 return NGX_OK; /* nothing staged, or already published */
158 }
159
160 /* staged_excl carries the open's kXR_new intent to the publish: the storage
161 * enforces create-if-absent atomically here (EEXIST → kXR_ItExists), rather
162 * than trusting the open-time existence check that raced (phase-107 C1). */
163 if (brix_vfs_writer_commit_ex(file->writer, file->staged_excl) != NGX_OK) {
164 if (err_out != NULL) {
165 *err_out = errno ? errno : EIO;
166 }
167 return NGX_ERROR;
168 }
169
170 /* A successful commit published (and optionally verified) the object and
171 * consumed the session's staged state; mark committed so brix_free_fhandle's
172 * abort is a no-op and no second commit runs. */
173 file->staged_committed = 1;
174 return NGX_OK;
175}
