94/* WHAT: Attempt bounded TLS-1.2 and TLS-1.1-only upgrades to the same target.
95 * WHY: A local library floor alone cannot prove the server rejects legacy
96 * ClientHello messages; the endpoint must make that decision on the wire.
97 * HOW: Require a TLS-1.2 control connection, then lower only an isolated

99static void
100probe_protocol_floor(xrd_conform_execution *execution, const brix_url *url)
101{
102 brix_opts options = tls12_probe_options(execution);
103 brix_conn connection;
104 brix_status status;
105 int control;
106 int legacy;
107
108 if (!pending(execution, "TLS-006")) return;

110 control = brix_connect_no_login(&connection, url, &options, &status) == 0;
111 if (control) brix_close(&connection);

130/* WHAT: Ask a TLS-1.2 session to renegotiate and observe the peer decision.
131 * WHY: Negotiating TLS 1.3 does not prove the same endpoint's TLS-1.2 path has
132 * disabled client-initiated renegotiation.
133 * HOW: Constrain an isolated control to 1.2, then drive OpenSSL's renegotiation
134 * state machine under the ordinary connection timeout. */
135static void
136probe_renegotiation(xrd_conform_execution *execution, const brix_url *url)
137{
138 brix_opts options = tls12_probe_options(execution);
139 brix_conn connection;
140 brix_status status;
141 int connected;
142 int result;
143
144 if (!pending(execution, "TLS-008")) return;
145 brix_status_clear(&status);
146 connected = brix_connect_no_login(&connection, url, &options, &status) == 0;
147 if (!connected) {
148 xrd_conform_result_set(execution, "TLS-008",
149 execution->root_tls13_seen ? BRIX_CONFORM_VERDICT_CONFORMANT
150 : BRIX_CONFORM_VERDICT_INCONCLUSIVE,
151 execution->root_tls13_seen ? "endpoint-is-tls13-only"
152 : "tls12-control-handshake-unavailable",
153 NULL, status.msg);
154 return;
155 }
156 result = brix_tls_probe_renegotiate(&connection, &status);
157 brix_close(&connection);
158 xrd_conform_result_set(execution, "TLS-008",
159 result == 0 ? BRIX_CONFORM_VERDICT_CONFORMANT
160 : result > 0 ? BRIX_CONFORM_VERDICT_DIVERGENT
161 : BRIX_CONFORM_VERDICT_INCONCLUSIVE,
162 result == 0 ? "tls12-renegotiation-refused"
163 : result > 0 ? "tls12-renegotiation-accepted"
164 : "tls12-renegotiation-indeterminate",
165 "client-initiated renegotiation refused", status.msg);
166}

190/* WHAT: Resume an authenticated TLS session while suppressing application auth.
191 * WHY: TLS resumption may restore transport parameters, never ROOT identity or
192 * authorization state from the connection that minted the ticket.
193 * HOW: Prove the authenticated baseline, retain an OpenSSL session reference,
194 * reconnect with force_anon, require confirmed resumption, then demand an
195 * exact authorization refusal for the protected object. */
196static void
197probe_resumed_identity(xrd_conform_execution *execution, const brix_url *url,
198 const brix_conn *authenticated)
199{
200 brix_opts options = execution->run->opts;
201 brix_conn resumed;
202 brix_statinfo info;
203 brix_status status;
204 void *session;
205 char reason[64];
206 int baseline;
207 int denied;
208
209 if (!pending(execution, "TLS-009")
210 || authenticated->sec_list[0] == '\0') return;
211 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) {
212 xrd_conform_result_set(execution, "TLS-009",
213 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
214 "request-ceiling-before-resumption-baseline", NULL, NULL);
215 return;
216 }
217 options.capture = NULL;
218 options.no_retry = 1;
219 options.tls_resume_session = NULL;
220 session = authenticated_baseline_session(url, &options, &status, &baseline);
221 if (session == NULL) {
222 xrd_conform_result_set(execution, "TLS-009",
223 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
224 baseline ? "tls-session-ticket-unavailable"
225 : "authenticated-baseline-unavailable",
226 NULL, status.msg);
227 return;
228 }
229 options.force_anon = 1;
230 options.tls_resume_session = session;
231 brix_status_clear(&status);
232 if (brix_connect(&resumed, url, &options, &status) != 0) {
233 brix_tls_session_free(session);
234 xrd_conform_result_set(execution, "TLS-009",
235 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
236 "resumed-anonymous-connection-unavailable", NULL, status.msg);
237 return;
238 }
239 brix_tls_session_free(session);
240 if (!resumed.tls_session_reused) {
241 brix_close(&resumed);
242 xrd_conform_result_set(execution, "TLS-009",
243 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
244 "server-did-not-resume-session", NULL, NULL);
245 return;
246 }
247 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) {
248 brix_close(&resumed);
249 xrd_conform_result_set(execution, "TLS-009",
250 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
251 "request-ceiling-before-resumed-stat", NULL, NULL);
252 return;
253 }
254 brix_status_clear(&status);
255 denied = brix_stat(&resumed, url->path, &info, &status) != 0
256 && (status.kxr == kXR_NotAuthorized || status.kxr == kXR_AuthFailed);
257 brix_close(&resumed);
258 xrd_conform_result_set(execution, "TLS-009",
259 denied ? BRIX_CONFORM_VERDICT_CONFORMANT
260 : BRIX_CONFORM_VERDICT_DIVERGENT,
261 denied ? "resumed-session-did-not-carry-authorization"
262 : "resumed-session-inherited-authorization",
263 "authorization refusal after confirmed TLS resumption", status.msg);
264}

315/* WHAT: Collect live TLS evidence from one ROOT plane.
316 * WHY: A paired --also run may need observations from two distinct listeners.
317 * HOW: Accumulate claims and grade rows supported by this particular session. */
318void
319xrd_conform_probe_tls(xrd_conform_execution *execution, const brix_url *url,
320 const brix_conn *connection, int object_observed)
321{
322 const char *version = NULL;
323 const char *cipher = NULL;
324 int tls_active;
325 uint32_t required;
326
327 if (!xrd_conform_family_active(execution, "TLS")) return;
328 tls_active = brix_tls_info(connection, &version, &cipher);
329 required = connection->server_flags & (kXR_tlsAny | kXR_tlsGPFA);
330 execution->root_tls_active_seen |= tls_active;
331 execution->root_tls_auth_seen |= connection->sec_list[0] != '\0';
332 execution->root_tls_required_seen |= required;
333 execution->root_tls13_seen |= version != NULL
334 && strcmp(version, "TLSv1.3") == 0;
335 probe_upgrade(execution, connection, tls_active, version);
336 probe_certificate(execution, url, tls_active);
337 if (tls_active) probe_protocol_floor(execution, url);
338 if (tls_active) probe_renegotiation(execution, url);
339 if (tls_active) probe_resumed_identity(execution, url, connection);
340 if (object_observed) probe_object_tls(execution, tls_active, required);
341 (void) cipher;
342}

426/* WHAT: Exercise each operator-supplied incapable-client TLS fixture.
427 * WHY: An ordinary client upgrades automatically and therefore cannot prove a
428 * server refuses the corresponding request class on cleartext.
429 * HOW: Force anonymous credential selection, suppress only the test upgrade,
430 * require the advertised class bit, and accept only kXR_TLSRequired. */
431void
432xrd_conform_probe_tls_fixtures(xrd_conform_execution *execution)
433{
434 size_t index;
435
436 if (!xrd_conform_family_active(execution, "TLS")) return;
437 for (index = 0; index < execution->run->tls_fixture_count; index++) {
438 const xrd_conform_tls_fixture *fixture =
439 &execution->run->tls_fixtures[index];
440 brix_opts options = execution->run->opts;
441 brix_url url;
442 brix_conn connection = { 0 };
443 brix_status status;
444 uint32_t bit = tls_fixture_bit(fixture->request_class);
445 char reason[64];
446 int connected;
447 int refused = 0;
448
449 execution->root_tls_fixture_attempted = 1;
450 if (bit == 0 || brix_endpoint_parse(fixture->target, &url, &status) != 0
451 || url.scheme != XRDC_SCHEME_ROOT
452 || !xrd_conform_reserve(execution, 0, reason, sizeof(reason)))
453 continue;
454 options.capture = NULL;
455 options.no_retry = 1;
456 options.force_anon = 1;
457 options.auth_force = NULL;
458 options.diagnostic_no_tls_upgrade = 1;
459 brix_status_clear(&status);
460 connected = brix_connect(&connection, &url, &options, &status) == 0;
461 execution->root_tls_required_seen |= connection.server_flags
462 & (kXR_tlsAny | kXR_tlsGPFA);
463 if ((connection.server_flags & bit) == 0) {
464 if (connected) brix_close(&connection);
465 continue;
466 }
467 execution->root_tls_cleartext_tested |= bit;
468 if (strcmp(fixture->request_class, "login") == 0) {
469 refused = !connected && tls_required(&status);
470 } else if (connected) {
471 refused = probe_cleartext_class(&connection, &url,
472 fixture->request_class, &status);
473 }
474 if (connected) brix_close(&connection);
475 if (refused) execution->root_tls_cleartext_refused |= bit;
476 }
477}
