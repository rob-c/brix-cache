1/*
2 * vfs_secgate.c — generic per-capability TLS gating (parser + gate + setter).
3 *
4 * WHAT: brix_tls_require_parse (directive grammar → 4-bit mask),
5 * brix_tls_gate_refused (mask × op-caps × transport → refused bits),
6 * brix_tls_cap_name (bit → token for refusal messages), and
7 * brix_conf_set_tls_require (offset-based ngx_conf_t setter shared by
8 * the stream and HTTP directive tables).
9 *
10 * WHY: One implementation of the stock `xrootd.tls` capability policy for
11 * every plane — see vfs_secgate.h. Keeping parse/check pure lets the
12 * config-parse negatives and the enforcement paths be tested without
13 * touching each other.
14 *
15 * HOW: Left-to-right token fold: `all` → BRIX_TLSREQ_ALL, a bare cap ORs its
16 * bit, `-cap` clears its bit, `none` → 0 and must be the only token.
17 * Unknown tokens (and `-all`/`-none`) fail the parse. The gate is one
18 * expression: bits that are (required by mask) AND (used by this op)
19 * AND (transport is cleartext); SESSION is a required bit on every
20 * post-handshake op its callers pass, so `session` alone locks the
21 * whole session exactly like stock.
22 */
23
24#include "vfs_secgate.h"

26/* token → bit; "all"/"none" are handled structurally in the parser. */
27static ngx_uint_t
28secgate_cap_bit(ngx_str_t *tok)
29{
30 if (tok->len == 5 && ngx_strncmp(tok->data, "login", 5) == 0) {
31 return BRIX_TLSREQ_LOGIN;
32 }
33 if (tok->len == 7 && ngx_strncmp(tok->data, "session", 7) == 0) {
34 return BRIX_TLSREQ_SESSION;
35 }
36 if (tok->len == 4 && ngx_strncmp(tok->data, "data", 4) == 0) {
37 return BRIX_TLSREQ_DATA;
38 }
39 if (tok->len == 3 && ngx_strncmp(tok->data, "tpc", 3) == 0) {
40 return BRIX_TLSREQ_TPC;
41 }
42 return 0;
43}

45/* brix_tls_require_parse — fold directive tokens into a capability mask.
46 *
47 * Grammar (stock xrootd.tls shape): `none` alone, or any sequence of
48 * `all|login|session|data|tpc` and `-login|-session|-data|-tpc` applied
49 * left-to-right. Returns NGX_ERROR on an unknown token, `-all`/`-none`,
50 * or `none` mixed with other tokens. */
51ngx_int_t
52brix_tls_require_parse(ngx_str_t *args, ngx_uint_t nargs, ngx_uint_t *mask_out)
53{
54 ngx_uint_t i, bit, mask = 0;
55 ngx_str_t tok;
56
57 for (i = 0; i < nargs; i++) {
58 tok = args[i];
59
60 if (tok.len == 4 && ngx_strncmp(tok.data, "none", 4) == 0) {
61 if (nargs != 1) {
62 return NGX_ERROR; /* `none` must stand alone */
63 }
64 *mask_out = 0;
65 return NGX_OK;
66 }
67 if (tok.len == 3 && ngx_strncmp(tok.data, "all", 3) == 0) {
68 mask = BRIX_TLSREQ_ALL;
69 continue;
70 }
71 if (tok.len > 1 && tok.data[0] == '-') {
72 tok.data++;
73 tok.len--;
74 bit = secgate_cap_bit(&tok);
75 if (bit == 0) {
76 return NGX_ERROR; /* -all / -none / unknown */
77 }
78 mask &= ~bit;
79 continue;
80 }
81 bit = secgate_cap_bit(&tok);
82 if (bit == 0) {
83 return NGX_ERROR;
84 }
85 mask |= bit;
86 }
87
88 *mask_out = mask;
89 return NGX_OK;
90}

92/* brix_tls_gate_refused — the one enforcement expression.
93 *
94 * `caps` is the capability set the current operation exercises (e.g. a
95 * kXR_read passes SESSION|DATA); the result is the required-but-cleartext
96 * subset, 0 when allowed. TLS transports are never refused. */
97ngx_uint_t
98brix_tls_gate_refused(ngx_uint_t mask, ngx_uint_t caps, ngx_uint_t is_tls)
99{
100 if (is_tls) {
101 return 0;
102 }
103 return mask & caps;
104}

106/* First refused capability's token, for kXR_TLSRequired / 403 messages. */
107const char *
108brix_tls_cap_name(ngx_uint_t bits)
109{
110 if (bits & BRIX_TLSREQ_LOGIN) { return "login"; }
111 if (bits & BRIX_TLSREQ_SESSION) { return "session"; }
112 if (bits & BRIX_TLSREQ_DATA) { return "data"; }
113 if (bits & BRIX_TLSREQ_TPC) { return "tpc"; }
114 return "none";
115}

117/* brix_conf_set_tls_require — `brix_tls_require <caps...>;` setter.
118 *
119 * Offset-based so the same function serves the HTTP common conf and the
120 * stream srv conf (both hold the mask in their shared preamble). Rejects a
121 * duplicate directive and any token the parser refuses. */
122char *
123brix_conf_set_tls_require(ngx_conf_t *cf, ngx_command_t *cmd, void *conf)
124{
125 char *p = conf;
126 ngx_uint_t *field;
127 ngx_str_t *value;
128
129 field = (ngx_uint_t *) (p + cmd->offset);
130 if (*field != NGX_CONF_UNSET_UINT) {
131 return "is duplicate";
132 }
133
134 value = cf->args->elts;
135 if (brix_tls_require_parse(value + 1, cf->args->nelts - 1, field)
136 != NGX_OK)
137 {
138 ngx_conf_log_error(NGX_LOG_EMERG, cf, 0,
139 "invalid brix_tls_require capability; expected "
140 "\"none\" or [all|login|session|data|tpc|-<cap>]...");
141 return NGX_CONF_ERROR;
142 }
143
144 return NGX_CONF_OK;
145}
