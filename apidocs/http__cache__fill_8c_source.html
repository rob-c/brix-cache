1/*
2 * http_cache_fill.c - off-event-loop cache-miss fill for the HTTP read plane.
3 * See http_cache_fill.h for the WHAT/WHY/HOW.
4 *
5 * This unit is the public entry point + async-pool resolver. The concurrency
6 * machinery it drives was split out mechanically (zero behaviour change) for
7 * auditability — see http_cache_fill_internal.h:
8 * http_cache_fill_registry.c coalescing waiter registry (attach/detach/hold/abort)
9 * http_cache_fill_worker.c worker thread + resolve + done finalize
10 *
11 * Phase-68 additions:
12 * - COALESCING: every concurrent request for one (inst,key) parks on a
13 * single heap-owned fill — a 40-request stampede is exactly ONE origin
14 * fetch. The waiter list is event-loop-only (no locks).
15 * - NEVER-DROP (T20): with a client-hold configured, the fill worker
16 * retries transient origin failures with jittered backoff until the
17 * hold deadline; a waiter whose hold expires detaches and receives
18 * 504 + Retry-After on a KEPT-ALIVE connection (a TCP close is never
19 * an error signal — convention #6). A client abort detaches its waiter
20 * but never cancels the fill: the fill keeps retrying (max-life
21 * deadline) and publishes so the client's retry is a hit.
22 *
23 * Ownership: the fill ctx (+ its ngx_thread_task_t) is one calloc block,
24 * freed in the done handler. Each waiter is freed by ITS REQUEST's pool
25 * cleanup (which always fires exactly once), so a late cleanup can never
26 * use-after-free a waiter the done handler already resolved.
27 */

51static ngx_int_t
52fill_cred_copy_fields(brix_http_fill_cred_t *out,
53 const brix_sd_cred_t *cred)
54{
55 fill_cred_copy_t fields[] = {
56 { out->x509_proxy, sizeof(out->x509_proxy), cred->x509_proxy },
57 { out->bearer, sizeof(out->bearer), cred->bearer },
58 { out->s3_ak, sizeof(out->s3_ak), cred->s3_ak },
59 { out->s3_sk, sizeof(out->s3_sk), cred->s3_sk },
60 { out->s3_region, sizeof(out->s3_region), cred->s3_region },
61 { out->s3_session, sizeof(out->s3_session), cred->s3_session },
62 { out->ceph_keyring, sizeof(out->ceph_keyring), cred->ceph_keyring },
63 { out->ceph_user, sizeof(out->ceph_user), cred->ceph_user },
64 { out->sss_keytab, sizeof(out->sss_keytab), cred->sss_keytab },
65 { out->krb5_ccache, sizeof(out->krb5_ccache), cred->krb5_ccache },
66 { out->krb5_princ, sizeof(out->krb5_princ), cred->krb5_princ },
67 { out->key, sizeof(out->key), cred->key },
68 { out->principal, sizeof(out->principal), cred->principal },
69 { out->vos, sizeof(out->vos), cred->vos },
70 { out->cred_dir, sizeof(out->cred_dir), cred->cred_dir },
71 };
72 size_t i;
73
74 for (i = 0; i < sizeof(fields) / sizeof(fields[0]); i++) {
75 size_t len = fields[i].src == NULL ? 0 : ngx_strlen(fields[i].src);
76
77 if (len >= fields[i].cap) {
78 errno = ENAMETOOLONG;
79 return NGX_ERROR;
80 }
81 if (len > 0) {
82 ngx_memcpy(fields[i].dst, fields[i].src, len);
83 }
84 fields[i].dst[len] = '\0';
85 }
86 return NGX_OK;
87}

89ngx_int_t
90brix_http_fill_cred_resolve(brix_vfs_ctx_t *vctx,
91 brix_http_fill_cred_t *out)
92{
93 brix_sd_ucred_t store;
94 brix_sd_cred_t cred;
95 int use_cred = 0;
96 int gate_err = 0;
97 ngx_int_t rc;
98
99 ngx_memzero(out, sizeof(*out));
100 if (vctx == NULL) {
101 return NGX_OK;
102 }
103 ngx_memzero(&store, sizeof(store));
104 ngx_memzero(&cred, sizeof(cred));
105 rc = brix_vfs_backend_cred(vctx, &store, &cred, &use_cred, &gate_err);
106 if (rc == NGX_OK && use_cred) {
107 rc = fill_cred_copy_fields(out, &cred);
108 out->mode = cred.mode;
109 out->fallback_deny = cred.fallback_deny ? 1 : 0;
110 out->use_cred = (rc == NGX_OK) ? 1 : 0;
111 }
112 brix_sd_ucred_wipe(&store);
113 if (rc != NGX_OK && gate_err != 0) {
114 errno = gate_err;
115 }
116 return rc;
117}

164/* Lazily resolve the export's async thread pool: server postconfig fills only the
165 * server-level loc_conf, so a nested location block resolves on first use (the

167static ngx_thread_pool_t *
168brix_http_cache_fill_pool(ngx_http_brix_shared_conf_t *common)
169{
170 ngx_thread_pool_t *pool = common->thread_pool;
171
172 if (pool == NULL) {
173 static ngx_str_t default_name = ngx_string("default");
174 ngx_str_t *pname = common->thread_pool_name.len > 0
175 ? &common->thread_pool_name : &default_name;
176
177 pool = ngx_thread_pool_get((ngx_cycle_t *) ngx_cycle, pname);
178 if (pool != NULL) {
179 common->thread_pool = pool;
180 }
181 }
182 return pool;

185ngx_int_t
186brix_http_cache_fill_if_needed(ngx_http_request_t *r,
187 brix_sd_instance_t *inst, const char *key,
188 ngx_http_brix_shared_conf_t *common,
189 brix_vfs_ctx_t *vctx,
190 brix_http_cache_reenter_pt reenter, void *reenter_data,
191 brix_http_fill_fail_pt on_fail)
192{
193 ngx_thread_task_t *task;
194 brix_http_cache_fill_ctx_t *t;
195 ngx_thread_pool_t *pool;
196 u_char *block;
197 brix_http_fill_cred_t cred;
198
199 if (inst == NULL || key == NULL || reenter == NULL || common == NULL
200 || !brix_sd_cache_fill_needs_offload(inst, key))
201 {
202 return NGX_DECLINED; /* serve inline (hit / local / none) */
203 }
204
205 if (brix_http_fill_cred_resolve(vctx, &cred) != NGX_OK) {
206 return NGX_ERROR; /* deny before coalescing/origin I/O */
207 }
208
209 /* Coalesce onto an in-flight fill of the same object: the stampede case
210 * (N concurrent cold reads) is exactly ONE origin fetch, but only within
211 * one credential scope. */
212 t = brix_http_fill_find(inst, key, &cred);
213 if (t != NULL) {
214 ngx_int_t arc = brix_http_fill_attach(t, r, reenter, reenter_data,
215 on_fail);
216 brix_http_fill_cred_wipe(&cred);
217 return arc;
218 }
219
220 pool = brix_http_cache_fill_pool(common);
221 if (pool == NULL) {
222 /* No pool: nothing can run off-loop, so fall through to the inline path
223 * (preserves the pre-SP2 behaviour - a remote miss may stall/fail). */
224 ngx_log_error(NGX_LOG_WARN, r->connection->log, 0,
225 "brix: cache miss on \"%s\" needs an async thread pool to fill a "
226 "remote tier; none configured - serving inline (may stall)", key);
227 brix_http_fill_cred_wipe(&cred);
228 return NGX_DECLINED;
229 }
230
231 /* Heap-owned task + ctx: the fill is shared by every parked request and
232 * must not live in any one request's pool (an aborted first requester
233 * would otherwise free the memory under the running task). */
234 block = calloc(1, sizeof(ngx_thread_task_t)
235 + sizeof(brix_http_cache_fill_ctx_t));
236 if (block == NULL) {
237 brix_http_fill_cred_wipe(&cred);
238 return NGX_ERROR;
239 }
240 task = (ngx_thread_task_t *) block;
241 task->ctx = block + sizeof(ngx_thread_task_t);
242 t = task->ctx;
243 t->inst = inst;
244 t->task = task;
245 t->result = NGX_ERROR;
246 t->client_hold = common->cache_client_hold; /* 0 = single-pass fill */
247 t->max_life = common->cache_fill_max_life;
248 t->started_ms = ngx_current_msec;
249 ngx_memcpy(&t->cred, &cred, sizeof(cred));
250 brix_http_fill_cred_wipe(&cred);
251 ngx_cpystrn((u_char *) t->key, (u_char *) key, sizeof(t->key));
252 t->sess = brix_sess_begin(common->session_log,
253 brix_http_shared_access_log_fd(common), BRIX_SESS_PROTO_FILL,
254 BRIX_SESS_DIR_OUT, "cache-origin", sizeof("cache-origin") - 1,
255 BRIX_SESS_AM_ANON, NULL);
256 brix_sess_auth_once(t->sess, BRIX_SESS_AM_ANON, "-", "-");
257 brix_sess_attempt(t->sess, t->key, BRIX_SESS_MODE_READ);
258 brix_sess_xfer_start(t->sess, &t->sess_xfer, t->key,
259 BRIX_SESS_MODE_READ, -1);
260
261 brix_task_bind(task, brix_http_cache_fill_thread,
262 brix_http_cache_fill_done);
263 task->event.log = r->connection->log;
264
265 if (brix_http_fill_attach(t, r, reenter, reenter_data, on_fail)
266 != NGX_DONE)
267 {
268 brix_sess_end(t->sess, BRIX_SESS_END_ERROR);
269 brix_http_fill_cred_wipe(&t->cred);
270 free(block);
271 return NGX_ERROR;
272 }
273
274 if (ngx_thread_task_post(pool, task) != NGX_OK) {
275 brix_http_fill_detach(t->waiters); /* undo the attach */
276 r->main->count--;
277 brix_sess_end(t->sess, BRIX_SESS_END_ERROR);
278 brix_http_fill_cred_wipe(&t->cred);
279 free(block);
280 return NGX_ERROR;
281 }
282
283 t->next = brix_http_fills; /* publish for coalescing */
284 brix_http_fills = t;
285
286 ngx_log_error(NGX_LOG_INFO, r->connection->log, 0,
287 "brix: offloaded cache fill of \"%s\" to the thread pool", key);
288 return NGX_DONE;
289}
