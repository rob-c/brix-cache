1/*
2 * cred_load.h — load an outbound X.509 credential (PEM chain + private key).
3 *
4 * The server acts as a GSI *client* on three paths — the cache-fill origin
5 * (src/fs/cache/origin_auth_gsi.c), the TPC destination (src/tpc/gsi/) and the
6 * transparent upstream redirector (src/net/upstream/auth_gsi.c). Each needs
7 * the same two reads of operator-configured files: the proxy certificate chain
8 * as one PEM blob, and the private key. One loader keeps the O_NOFOLLOW
9 * hardening and the "certs only, re-serialised" contract in a single place.
10 */

19/* Every CERTIFICATE block of the PEM file at `path`, re-serialised as one
20 * contiguous PEM blob (certs only — a key in the same file is dropped). The
21 * path is operator-configured (trusted like the server's own cert/key) and is
22 * opened O_NOFOLLOW so a planted symlink cannot redirect it. malloc'd, *outlen
23 * set; NULL when the file is unreadable or holds no certificate. Mirrors
24 * client/lib/sec/sec_gsi.c load_proxy_pem. */
