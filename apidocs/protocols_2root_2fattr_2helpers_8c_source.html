30uint16_t
31fattr_errno_to_xrd(int err)
32{
33 switch (err) {
34 case ENODATA: return kXR_AttrNotFound;
35 case ERANGE: return kXR_ArgTooLong;
36 default: return brix_kxr_from_errno(err);
37 }
38}
39/*
40 * WHAT: Maps POSIX errno values to XRootD kXR error codes.
41 * - ENODATA → kXR_AttrNotFound (attribute not found)
42 * - ERANGE → kXR_ArgTooLong (value too long for buffer)
43 * - All other errnos delegate to brix_kxr_from_errno()
44 *
45 * WHY: POSIX filesystem xattr syscalls return ENODATA/ERANGE/etc but XRootD
46 * wire protocol expects kXR codes. This translation ensures error responses
47 * match the XRootD spec regardless of underlying filesystem behavior.
48 *
49 * HOW: switch(err) — ENODATA returns kXR_AttrNotFound; ERANGE returns
50 * kXR_ArgTooLong; any other value calls brix_kxr_from_errno(err).
51 * Returns uint16_t kXR code. Used by fattr_set_rc() and all fattr handlers
52 * when getxattr/fgetxattr/setxattr/removexattr return negative with errno set.
53 */

55void
56fattr_set_rc(brix_fattr_entry_t *attr, uint16_t rc)
57{
58 uint16_t rc_be;
59
60 attr->errcode = rc;
61
62 rc_be = htons(rc);
63 ngx_memcpy(attr->rc_ptr, &rc_be, 2);
64}
65/*
66 * WHAT: Encodes a per-attribute result code into the wire-format response buffer.
67 * - Sets the in-memory errcode field
68 * - Writes the network-byte-order (big-endian) 16-bit value at rc_ptr
69 *
70 * WHY: fattr responses embed per-attribute status codes as 2-byte big-endian
71 * values at fixed positions within nvec_copy buffers. The caller allocates
72 * these buffers and rc_ptr points back into them so later operations can
73 * overwrite only the result slot without shifting other entries.
74 *
75 * HOW: Sets attr->errcode = rc in memory, then calls htons(rc) to convert to
76 * network byte order, copies 2 bytes via ngx_memcpy(attr->rc_ptr, &rc_be, 2).
77 * No validation — accepts any uint16_t rc value including 0 (success).

80ssize_t
81fattr_parse_nvec(ngx_log_t *log, u_char *nvec_copy, size_t buflen,
82 int numattr, brix_fattr_entry_t *attrs)
83{
84 u_char *cursor;
85 int attr_index;
86
87 /*
88 * nvec is repeated as:
89 * [2-byte per-attribute result slot][NUL-terminated attribute name]
90 *
91 * The request payload is copied before parsing, so rc_ptr can safely point
92 * back into nvec_copy. Later operations overwrite only the result slot.
93 */
94/*
95 * WHY: nvec is the XRootD request payload for set/del/fattrGet — it carries
96 * attribute names with embedded 2-byte result slots that handlers overwrite
97 * with per-attribute status codes. Parsing into attr[] entries enables
98 * subsequent operations to read names, compute xkey prefixes, and write rc
99 * values at fixed positions.
100 *
101 * HOW: Initializes cursor = nvec_copy, end = nvec_copy + buflen. Loop over
102 * numattr entries: checks cursor+2 <= end (truncation guard), sets
103 * attrs[i].rc_ptr=cursor, errcode=0, value=NULL, vlen=0; advances cursor by 2.
104 * Scans for NUL terminator to find name length — if cursor >= end logs
105 * "name not null-terminated" and returns -1. Validates name_len > 0 && <=
106 * kXR_faMaxNlen — otherwise logs invalid length and returns -1. Sets
107 * attrs[i].name=name_start, nlen=name_len; snprintf xkey with
108 * BRIX_FATTR_XKEY_PFX prefix + %.*s format. Advances cursor past NUL. Returns
109 * (ssize_t)(cursor - nvec_copy) = bytes consumed.
110 */
111 cursor = nvec_copy;
112
113 for (attr_index = 0; attr_index < numattr; attr_index++) {
114 const uint8_t *name_p;
115 size_t name_len, off, next;
116
117 /* Shared read-only scan of one [int16 rc][name\0] entry (libxrdproto).
118 * The rc slot lives at cursor; we hold that pointer ourselves so the
119 * set/del handlers can rewrite it in place (fattr_set_rc) — the shared
120 * parser never touches the buffer. */
121 off = (size_t) (cursor - nvec_copy);
122 if (xrdp_fattr_nvec_parse(nvec_copy, buflen, off, NULL,
123 &name_p, &name_len, &next) != 0) {
124 ngx_log_error(NGX_LOG_WARN, log, 0,
125 "brix: fattr nvec truncated/unterminated at entry %d",
126 attr_index);
127 return -1;
128 }
129
130 /* Name-length policy is server-side (the wire parser is policy-free). */
131 if (name_len == 0 || name_len > kXR_faMaxNlen) {
132 ngx_log_error(NGX_LOG_WARN, log, 0,
133 "brix: fattr name length %uz invalid",
134 name_len);
135 return -1;
136 }
137
138 attrs[attr_index].rc_ptr = cursor; /* the 2-byte rc slot */
139 attrs[attr_index].errcode = 0;
140 attrs[attr_index].value = NULL;
141 attrs[attr_index].vlen = 0;
142 attrs[attr_index].name = (char *) name_p;
143 attrs[attr_index].nlen = name_len;
144 snprintf(attrs[attr_index].xkey, sizeof(attrs[attr_index].xkey),
145 BRIX_FATTR_XKEY_PFX "%.*s", (int) name_len, name_p);
146
147 cursor = nvec_copy + next;
148 }
149
150 return (ssize_t) (cursor - nvec_copy);
151}

153ngx_int_t
154fattr_send_vector_status(brix_ctx_t *ctx, ngx_connection_t *c,
155 u_char *nvec_copy, size_t nvec_len, int numattr,
156 brix_fattr_entry_t *attrs)
157{
158 ngx_pool_t *pool;
159 u_char *response;
160 size_t response_size;
161 int error_count;
162 int attr_index;
163
164 pool = c->pool;
165 error_count = 0;
166
167 for (attr_index = 0; attr_index < numattr; attr_index++) {
168 if (attrs[attr_index].errcode) {
169 error_count++;
170 }
171 }
172
173 response_size = 2 + nvec_len;
174 BRIX_PALLOC_OR_RETURN(response, pool, response_size, brix_send_error(ctx, c, kXR_NoMemory, "out of memory"));
175
176 response[0] = (u_char) error_count;
177 response[1] = (u_char) numattr;
178 ngx_memcpy(response + 2, nvec_copy, nvec_len);
179
180 BRIX_OP_OK(ctx, BRIX_OP_FATTR);
181 return brix_send_ok(ctx, c, response, (uint32_t) response_size);
182}

185 * - Encodes error_count + numattr header bytes
186 * - Copies nvec payload
187 * - Sends via brix_send_ok()
188 *
189 * WHY: After setxattr/removexattr operations complete, each attribute has an
190 * errcode set in attrs[]. This function counts errors for the response header
