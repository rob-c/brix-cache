1/*
2 * sd_stage.c - the generic write-stage decorator CORE (section 12.2). See header.
3 *
4 * The decorator forwards every read / namespace / xattr / dir op to the wrapped
5 * `source` (open returns the source's own object, so read byte-I/O bypasses the
6 * decorator) and interposes only the staged-write path. This file owns the
7 * decorator core — the open dispatch, the read/namespace/xattr/dir forwarders,
8 * the driver descriptor, and the instance lifecycle (create/destroy/predicates/
9 * reflush). The two interposed WRITE paths (the write-back byte-I/O object and
10 * the staged-upload path) live in sd_stage_write.c; the driver table below
11 * dispatches to them and the shared seam is declared in sd_stage_internal.h.
12 * A posix stage store is byte-equivalent to phase-63's local-temp promote.
13 */

23/* sd_stage_store_mkparents — create `key`'s parent chain inside the stage STORE.
24 *
25 * WHAT: EEXIST-tolerant prefix walk calling the store driver's mkdir slot for
26 * every directory component before the key's last '/'.
27 * WHY: The stage store is a PRIVATE spool, not a client-visible namespace: a
28 * client mkdir (or the kXR_mkpath/kXR_async pre-create) builds the chain in
29 * the EXPORT, never in the spool, so the store's create-open of a nested key
30 * failed ENOENT — with a stage tier configured, a write to ANY subdirectory
31 * was impossible. The staged (whole-object) leg never hit this because the
32 * POSIX store's staged_open mkpaths its own parents; this is that same rule
33 * for the write-back leg.
34 * HOW: Copies the parent prefix and mkdirs each component in turn with mode 0700
35 * (service-owned spool), tolerating EEXIST. A flat key, or a store with no
36 * mkdir slot, is NGX_OK — the open then decides, exactly as before. */
37static ngx_int_t
38sd_stage_store_mkparents(sd_stage_inst_state *is, const char *key)
39{
40 char acc[PATH_MAX];
41 const char *last = strrchr(key, '/');
42 size_t plen, j;
43
44 if (last == NULL || last == key || is->store->driver->mkdir == NULL) {
45 return NGX_OK;
46 }
47
48 plen = (size_t) (last - key);
49 if (plen >= sizeof(acc)) {
50 errno = ENAMETOOLONG;
51 return NGX_ERROR;
52 }
53 memcpy(acc, key, plen);
54 acc[plen] = '\0';
55
56 for (j = 1; j <= plen; j++) {
57 char sep = acc[j];
58
59 if (sep != '/' && sep != '\0') {
60 continue;
61 }
62 acc[j] = '\0';
63 if (is->store->driver->mkdir(is->store, acc, 0700) != NGX_OK
64 && errno != EEXIST)
65 {
66 return NGX_ERROR;
67 }
68 acc[j] = sep;
69 }
70 return NGX_OK;
71}

75/* The one write-open entry both open slots share: build the spool-side parent
76 * chain for a creating open, then hand off to the write-back open. Only a CREATE
77 * may materialise a new key, so a plain update open never mkdirs. */
78static brix_sd_obj_t *
79sd_stage_open_write(brix_sd_instance_t *inst, sd_stage_inst_state *is,
80 const char *path, int sd_flags, mode_t mode, const brix_sd_cred_t *cred,
81 int *err_out)
82{
83 if ((sd_flags & BRIX_SD_O_CREATE)
84 && sd_stage_store_mkparents(is, path) != NGX_OK)
85 {
86 if (err_out != NULL) { *err_out = errno ? errno : EIO; }
87 return NULL;
88 }
89 return sd_stage_open_writeback(inst, is, path, sd_flags, mode, cred,
90 err_out);
91}

93static brix_sd_obj_t *
94sd_stage_open(brix_sd_instance_t *inst, const char *path, int sd_flags,
95 mode_t mode, int *err_out)
96{
97 sd_stage_inst_state *is = inst->state;
98
99 /* Write open → a write-back object on the stage store (pwrite buffers, fsync/close
100 * flush to the backend). Read open → the source's own object, so read byte-I/O
101 * bypasses the decorator entirely. */
102 if (sd_flags & BRIX_SD_O_WRITE) {
103 return sd_stage_open_write(inst, is, path, sd_flags, mode, NULL,
104 err_out);
105 }
106 return is->source->driver->open(is->source, path, sd_flags, mode, err_out);
107}

109/* Credential-scoped open: records the caller's per-user identity in the
110 * write-back state so the eventual flush authenticates as the original user.
111 *
112 * WHAT: Write opens route through sd_stage_open_writeback with the cred so the
113 * owner key/dir/deny are embedded in the durable wb state; read opens
114 * forward to the source via brix_sd_open_maybe_cred.
115 *
116 * WHY: Without this slot the stage decorator drops a caller-supplied cred on
117 * the floor: write opens use the service account for the flush, and read
118 * opens use the service account for the source open on credential-aware
119 * backends.
120 *
121 * HOW: Write → sd_stage_open_write(... cred); read →
122 * brix_sd_open_maybe_cred(source, ..., cred). */
123static brix_sd_obj_t *
124sd_stage_open_cred(brix_sd_instance_t *inst, const char *path, int sd_flags,
125 mode_t mode, const brix_sd_cred_t *cred, int *err_out)
126{
127 sd_stage_inst_state *is = inst->state;
128
129 if (sd_flags & BRIX_SD_O_WRITE) {
130 return sd_stage_open_write(inst, is, path, sd_flags, mode, cred,
131 err_out);
132 }
133 return brix_sd_open_maybe_cred(is->source, path, sd_flags, mode, cred,
134 err_out);
135}

137/* ---- namespace / xattr / dir forwarders (delegate to the source) ----------
138 *
139 * Each op appears twice: a plain slot and its credential-scoped twin, both
140 * landing on brix_sd_<op>_maybe_cred against the SOURCE (cred=NULL for the
141 * plain one, which is exactly the pre-existing behaviour). The twins are not
142 * optional politeness: brix_sd_<op>_maybe_cred keys off the instance it is
143 * CALLED on, so a decorator publishing `.mkdir` and no `.mkdir_cred` looks
144 * from above like a driver with no per-user support, whatever the source can
145 * actually do. That is why every VFS namespace site unwraps to the LEAF today
146 * (brix_vfs_ns_leaf) — a bypass that also skips this decorator's own work, so
147 * each bypassing site has to re-add the cache eviction by hand. See the same
148 * block in sd_cache_forward.c. ENOTSUP is preserved where the pre-split code
149 * reported it: for the xattr ops it means "no extended attributes here". */

467brix_sd_instance_t *
468brix_sd_stage_create(brix_sd_instance_t *source, brix_sd_instance_t *store,
469 const brix_stage_policy_t *policy, const char *root_canon, ngx_log_t *log)
470{
471 brix_sd_instance_t *inst;
472 sd_stage_inst_state *is;
473
474 if (source == NULL || store == NULL) {
475 errno = EINVAL;
476 return NULL;
477 }
478 inst = calloc(1, sizeof(*inst));
479 is = calloc(1, sizeof(*is));
480 if (inst == NULL || is == NULL) {
481 free(inst);
482 free(is);
483 errno = ENOMEM;
484 return NULL;
485 }
486 is->source = source;
487 is->store = store;
488 is->log = log;
489 if (root_canon != NULL) {
490 ngx_cpystrn((u_char *) is->root_canon, (u_char *) root_canon,
491 sizeof(is->root_canon));
492 }
493 if (policy != NULL) {
494 is->policy = *policy;
495 } else {
496 ngx_memzero(&is->policy, sizeof(is->policy));
497 is->policy.flush_mode = BRIX_WT_MODE_SYNC;
498 }
499
500 inst->driver = &brix_sd_stage_driver;
501 inst->log = log;
502 inst->pool = NULL;
503 inst->state = is;
504 inst->domain = BRIX_VFS_DOMAIN_EXPORT; /* the decorator FRONTS the export;
505 * only its store is DOMAIN_STAGE (C9) */
506 return inst;
507}

543/* SP4 restart-reconcile: re-flush the durable staged object `key` from the stage
544 * store to the backend (the FLUSH a crash interrupted), dropping the stage copy on
545 * success - exactly the sync staged_commit tail, run again.

548 * persisted brix_sreq_t.cred) is threaded into the flush and presented to the
549 * backend driver. A NULL cred uses the service credential.
550 *

552 * service account — for per-user quota / audit / ACL enforcement.
553 *
554 * HOW: Same as the pre-cred path but calls _cred instead of _inline, passing the
555 * caller-supplied cred unchanged. Returns NGX_OK / NGX_DECLINED (not a stage
556 * instance) / NGX_ERROR (errno set; the record is kept for retry). */

559 const brix_stage_cred_t *cred)
560{

562 ngx_int_t rc;
563
564 if (!brix_sd_stage_instance_is(inst) || key == NULL) {
565 return NGX_DECLINED;
566 }
567 is = inst->state;
568 rc = brix_stage_run_inline_cred(BRIX_STAGE_FLUSH, is->store, key, is->source,
569 key, cred);
570 if (rc == NGX_OK && is->store->driver->unlink != NULL) {
571 (void) is->store->driver->unlink(is->store, key, 0); /* drop stage copy */
572 }
573 return rc;
574}
