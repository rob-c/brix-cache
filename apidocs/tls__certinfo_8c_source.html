21/* Join the cert's DNS subjectAltNames into out[] as a comma-separated list. */
22static void
23peer_collect_sans(X509 *cert, char *out, size_t outsz)
24{
25 GENERAL_NAMES *gens;
26 int i, n;
27 size_t off = 0;
28
29 out[0] = '\0';
30 gens = (GENERAL_NAMES *) X509_get_ext_d2i(cert, NID_subject_alt_name, NULL, NULL);
31 if (gens == NULL) {
32 return;
33 }
34 n = sk_GENERAL_NAME_num(gens);
35 for (i = 0; i < n; i++) {
36 GENERAL_NAME *g = sk_GENERAL_NAME_value(gens, i);
37 const char *dns;
38 int len;
39 if (g->type != GEN_DNS) {
40 continue;
41 }
42 dns = (const char *) ASN1_STRING_get0_data(g->d.dNSName);
43 len = ASN1_STRING_length(g->d.dNSName);
44 if (len <= 0 || off + (size_t) len + 2 >= outsz) {
45 continue;
46 }
47 if (off > 0) { out[off++] = ','; }
48 memcpy(out + off, dns, (size_t) len);
49 off += (size_t) len;
50 out[off] = '\0';
51 }
52 GENERAL_NAMES_free(gens);
53}

55int
56brix_tls_peer_cert_info(const brix_conn *c, brix_cert_info *out)
57{
58 SSL *ssl = (SSL *) c->io.ssl;
59 X509 *cert;
60 const ASN1_TIME *nb, *na;
61 struct tm tmv;
62 time_t now = time(NULL);
63 int dd = 0, ds = 0;
64
65 memset(out, 0, sizeof(*out));
66 if (ssl == NULL) {
67 return -1; /* cleartext: no peer cert */
68 }
69 cert = SSL_get_peer_certificate(ssl); /* bumps refcount; free below */
70 if (cert == NULL) {
71 return -1;
72 }
73 out->have = 1;
74 X509_NAME_oneline(X509_get_subject_name(cert), out->subject, sizeof(out->subject));
75 X509_NAME_oneline(X509_get_issuer_name(cert), out->issuer, sizeof(out->issuer));
76 out->self_signed = (X509_check_issued(cert, cert) == X509_V_OK);
77 out->host_match = (c->host[0] != '\0'
78 && X509_check_host(cert, c->host, 0, 0, NULL) == 1);
79
80 nb = X509_get0_notBefore(cert);
81 na = X509_get0_notAfter(cert);
82 if (ASN1_TIME_to_tm(nb, &tmv)) { out->not_before = (long) timegm(&tmv); }
83 if (ASN1_TIME_to_tm(na, &tmv)) { out->not_after = (long) timegm(&tmv); }
84 out->expired = (out->not_after != 0 && out->not_after < (long) now);
85 out->not_yet_valid = (out->not_before != 0 && out->not_before > (long) now);
86 if (ASN1_TIME_diff(&dd, &ds, NULL, na)) {
87 out->days_left = dd; /* whole days; negative once expired */
88 }
89 peer_collect_sans(cert, out->sans, sizeof(out->sans));
90
91 X509_free(cert);
92 return 0;
93}
