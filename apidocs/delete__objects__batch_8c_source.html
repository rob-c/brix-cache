1/*
2 * delete_objects_batch.c - the S3 DeleteObjects two-phase batch stages
3 * (phase-107 C4): collect every key, dispose them in ONE
4 * brix_vfs_delete_many() call, render per-key results in client order.
5 *
6 * WHAT: s3_delete_collect_one() extracts, validates and CONFINES one <Object>
7 * key (s3_resolve_key_ex before any delete - security-load-bearing);
8 * s3_delete_execute() hands the confinable subset to the VFS batch
9 * entry; s3_delete_render() turns per-key verdicts into
10 * <Deleted>/<Error> elements.
11 * WHY: The old handler called brix_vfs_unlink once per key: 1,000 policy
12 * checks, 1,000 metric lines and - over a remote backend - 1,000 signed
13 * round trips for one request whose purpose was to avoid exactly that.
14 * Collect-then-execute also means a malformed <Object> is now rejected
15 * BEFORE anything is deleted, where the old parse-and-delete loop had
16 * already removed the keys ahead of it.
17 * HOW: Per-key error vocabulary is unchanged (AccessDenied, BucketNotEmpty,
18 * InternalError; ENOENT renders as <Deleted> per S3 idempotency; a
19 * RESERVED key reads exactly as an absent one). EROFS fails the WHOLE
20 * request 403 before any key is examined - a read-only export discloses
21 * nothing about which keys exist. ECANCELED (pre-filled by the VFS)
22 * marks a key the batch never reached; it renders as an <Error>, never
23 * as <Deleted>.
24 */

39/*
40 * s3_delete_collect_one - phase 1 for one <Object> node: extract and
41 * length-validate its <Key>, copy the key text onto the request pool (the
42 * parsed document is freed before rendering), then confine it.
43 *
44 * Verdicts land in *it: err.code set = pre-disposed <Error> (bad key, escape);
45 * fs set = confined, goes to the batch; neither = pre-disposed <Deleted> (a
46 * RESERVED key must read exactly as an absent one, and DeleteObjects is
47 * idempotent - an <Error> of any code would single the name out).
48 *
49 * Returns NGX_OK (verdict recorded - continue), or NGX_ABORT (malformed
50 * object: missing <Key>, unreadable content, or allocation failure; the
51 * *malformed_500 out-flag separates client MalformedXML from a 500).
52 */
53ngx_int_t
54s3_delete_collect_one(s3_del_ctx_t *dc, xmlNodePtr obj, s3_del_item_t *it,
55 int *malformed_500)
56{
57 xmlNodePtr key_node;
58 xmlChar *key_text;
59 size_t key_len;
60 char key_str[S3_MAX_KEY];
61 char fs_path[PATH_MAX];
62 int rrc;
63
64 *malformed_500 = 0;
65 ngx_memzero(it, sizeof(*it));
66
67 key_node = s3_delete_xml_find_child(obj, "Key");
68 if (key_node == NULL) {
69 return NGX_ABORT; /* missing <Key> -> MalformedXML */
70 }
71
72 key_text = xmlNodeGetContent(key_node);
73 if (key_text == NULL) {
74 *malformed_500 = 1;
75 return NGX_ABORT; /* unreadable content -> 500 */
76 }
77 key_len = (size_t) xmlStrlen(key_text);
78
79 it->key = ngx_pnalloc(dc->r->pool, key_len + 1);
80 if (it->key == NULL) {
81 xmlFree(key_text);
82 *malformed_500 = 1;
83 return NGX_ABORT;
84 }
85 ngx_memcpy(it->key, key_text, key_len);
86 it->key[key_len] = '\0';
87 it->key_len = key_len;
88 xmlFree(key_text);
89
90 if (key_len == 0 || key_len >= S3_MAX_KEY) {
91 it->err.code = "InvalidArgument";
92 it->err.message = "Object key is empty or too long.";
93 return NGX_OK;
94 }
95
96 /* Confinement BEFORE any delete: a key that resolves outside the export
97 * root is rejected here without ever touching the filesystem, and the
98 * refusal does not leak whether the escaped path exists. */
99 ngx_memcpy(key_str, it->key, key_len + 1);
100 rrc = s3_resolve_key_ex(dc->cf->common.root_canon, key_str, fs_path,
101 sizeof(fs_path),
102 dc->cf->common.cache_store_endpoint);
103 if (rrc == 404) {
104 return NGX_OK; /* pre-disposed <Deleted>: reserved reads as absent */
105 }
106 if (rrc != 0) {
107 s3_key_error_t kerr;
108
109 s3_resolve_key_error(rrc, &kerr);
110 it->err.code = kerr.code;
111 it->err.message = kerr.message;
112 return NGX_OK;
113 }
114
115 it->fs = ngx_pnalloc(dc->r->pool, ngx_strlen(fs_path) + 1);
116 if (it->fs == NULL) {
117 *malformed_500 = 1;
118 return NGX_ABORT;
119 }
120 ngx_memcpy(it->fs, fs_path, ngx_strlen(fs_path) + 1);
121 return NGX_OK;
122}

124/*
125 * s3_delete_execute - phase 2: ONE brix_vfs_delete_many() call for every key
126 * that survived collection (one write gate, one OP_DELETE observation, one
127 * driver batch where the leaf has unlink_many).
128 *
129 * WHAT: Builds the confinable subset (fs != NULL) with a batch->item index
130 * map, runs the VFS batch, writes each per-key errno back into its
131 * item's batch_err, and emits a CNS retraction per actually-removed key.
132 * WHY: EROFS is the one whole-request refusal: it is raised BEFORE any key is
133 * examined, so answering 403 for the request as a whole discloses
134 * nothing per key (the doc's security-negative). Every other batch
135 * failure keeps the per-key vocabulary - untried keys hold the
136 * ECANCELED the VFS pre-filled.
137 * HOW: Returns NGX_OK (per-key verdicts recorded - render next) or NGX_DONE
138 * (response finalised here: EROFS 403 or allocation 500).
139 */
140ngx_int_t
141s3_delete_execute(s3_del_ctx_t *dc, s3_del_item_t *items, size_t count)
142{
143 ngx_http_request_t *r = dc->r;
144 ngx_http_s3_req_ctx_t *s3ctx =
145 ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
146 brix_vfs_ctx_t vctx;
147 const char **fs;
148 int *errs;
149 size_t *map;
150 size_t i, m = 0, done = 0;
151 ngx_int_t rc;
152
153 fs = ngx_palloc(r->pool, count * sizeof(*fs));
154 errs = ngx_palloc(r->pool, count * sizeof(*errs));
155 map = ngx_palloc(r->pool, count * sizeof(*map));
156 if (fs == NULL || errs == NULL || map == NULL) {
157 s3_metrics_finalize_request_method(r, dc->method_slot,
158 NGX_HTTP_INTERNAL_SERVER_ERROR);
159 return NGX_DONE;
160 }
161
162 for (i = 0; i < count; i++) {
163 if (items[i].fs != NULL) {
164 fs[m] = items[i].fs;
165 map[m] = i;
166 m++;
167 }
168 }
169 if (m == 0) {
170 return NGX_OK; /* every key was pre-disposed at collection */
171 }
172
173 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log, BRIX_PROTO_S3,
174 dc->cf->common.root_canon, dc->cf->common.cache_root_canon,
175 brix_vfs_policy_from_write_enable(dc->cf->common.allow_write),
176 brix_http_request_is_tls(r),
177 (s3ctx != NULL) ? s3ctx->identity : NULL,
178 dc->cf->common.root_canon);
179 s3_vfs_bind_deleg(r, dc->cf, &vctx);
180
181 rc = brix_vfs_delete_many(&vctx, fs, m, errs, &done);
182 if (rc != NGX_OK && errno == EROFS) {
183 s3_metrics_finalize_request_method(r, dc->method_slot,
184 s3_send_xml_error(r, NGX_HTTP_FORBIDDEN, "AccessDenied",
185 "The export is read-only."));
186 return NGX_DONE;
187 }
188 if (rc != NGX_OK && errno == EBUSY) {
189 /* phase-107 C7: the per-key lock gate refused BEFORE any arm ran, so
190 * the refusal is atomic - no key was attempted (errs all ECANCELED).
191 * Answer the whole request with the same 409 OperationAborted the
192 * single-key DELETE gives a locked key (§5.0); per-key InternalError
193 * rows would hide the lock and imply a server fault. */
194 s3_metrics_finalize_request_method(r, dc->method_slot,
195 s3_send_xml_error(r, NGX_HTTP_CONFLICT, "OperationAborted",
196 "A conflicting conditional operation is "
197 "currently in progress against this "
198 "resource."));
199 return NGX_DONE;
200 }
201
202 for (i = 0; i < m; i++) {
203 items[map[i]].batch_err = errs[i];
204 if (errs[i] == 0) {
205 /* phase-97 §5: a real removal - retract it from the manager
206 * inventory. ENOENT removed nothing and emits nothing. */
207 brix_cns_emit_at(dc->cf->common.root_canon, BRIX_CNS_DEL,
208 fs[i], 0, 0);
209 }
210 }
211 return NGX_OK;
212}

214/* Map a batch per-key errno onto the S3 error pair - the same vocabulary the
215 * per-key handler used, plus ECANCELED for a key the batch never reached
216 * (which must never render as <Deleted>). */
217static void
218s3_delete_batch_err_pair(int e, s3_del_err_t *err)
219{
220 if (e == EACCES || e == EPERM) {
221 err->code = "AccessDenied";
222 err->message = "Access Denied.";
223 } else if (e == ENOTEMPTY) {
224 err->code = "BucketNotEmpty";
225 err->message = "The directory is not empty.";
226 } else if (e == ECANCELED) {
227 err->code = "InternalError";
228 err->message = "The key was not attempted because the batch failed.";
229 } else {
230 err->code = "InternalError";
231 err->message = "Internal server error.";
232 }
233}

235/*
236 * s3_delete_render - phase 3: one <Deleted> or <Error> element per item, in
237 * the client's original order. <Deleted> covers: pre-disposed at collection
238 * (reserved-absent), removed (batch_err 0), and idempotent-missing (ENOENT).
239 * Returns NGX_ERROR only on output-buffer overflow (caller maps to 500).
240 */
241ngx_int_t
242s3_delete_render(s3_del_ctx_t *dc, const s3_del_item_t *items, size_t count)
243{
244 size_t i;
245
246 for (i = 0; i < count; i++) {
247 const s3_del_item_t *it = &items[i];
248 s3_del_err_t err;
249 ngx_int_t rc;
250
251 if (it->err.code != NULL) {
252 rc = s3_delete_xml_append_error(dc->xml_buf, dc->xml_len,
253 it->key, it->key_len, &it->err);
254 } else if (it->fs == NULL || it->batch_err == 0
255 || it->batch_err == ENOENT)
256 {
257 rc = s3_delete_xml_append_deleted(dc->xml_buf, dc->xml_len,
258 it->key, it->key_len);
259 } else {
260 s3_delete_batch_err_pair(it->batch_err, &err);
261 rc = s3_delete_xml_append_error(dc->xml_buf, dc->xml_len,
262 it->key, it->key_len, &err);
263 }
264 if (rc != NGX_OK) {
265 return NGX_ERROR;
266 }
267 }
268 return NGX_OK;
269}
