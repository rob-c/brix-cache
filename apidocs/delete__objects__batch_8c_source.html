1/*
2 * delete_objects_batch.c - the S3 DeleteObjects batch stages (phase-107 C4):
3 * collect and confine every key, authorize every mutatable resource, dispose
4 * them in ONE brix_vfs_delete_many() call, then render results in client order.
5 *
6 * WHAT: s3_delete_collect_one() extracts, validates and CONFINES one <Object>
7 * key (s3_resolve_key_ex before any delete - security-load-bearing);
8 * s3_delete_execute() hands the confinable subset to the VFS batch
9 * entry; s3_delete_render() turns per-key verdicts into
10 * <Deleted>/<Error> elements.
11 * WHY: The old handler called brix_vfs_unlink once per key: 1,000 policy
12 * checks, 1,000 metric lines and - over a remote backend - 1,000 signed
13 * round trips for one request whose purpose was to avoid exactly that.
14 * Collect-then-execute also means a malformed <Object> is now rejected
15 * BEFORE anything is deleted, where the old parse-and-delete loop had
16 * already removed the keys ahead of it.
17 * HOW: Per-key error vocabulary is unchanged (AccessDenied, BucketNotEmpty,
18 * InternalError; ENOENT renders as <Deleted> per S3 idempotency; a
19 * RESERVED key reads exactly as an absent one). EROFS fails the WHOLE
20 * request 403 before any key is examined - a read-only export discloses
21 * nothing about which keys exist. ECANCELED (pre-filled by the VFS)
22 * marks a key the batch never reached; it renders as an <Error>, never
23 * as <Deleted>.
24 */

41/*
42 * s3_delete_collect_one - phase 1 for one <Object> node: extract and
43 * length-validate its <Key>, copy the key text onto the request pool (the
44 * parsed document is freed before rendering), then confine it.
45 *
46 * Verdicts land in *it: err.code set = pre-disposed <Error> (bad key, escape);
47 * fs set = confined, goes to the batch; neither = pre-disposed <Deleted> (a
48 * RESERVED key must read exactly as an absent one, and DeleteObjects is
49 * idempotent - an <Error> of any code would single the name out).
50 *
51 * Returns NGX_OK (verdict recorded - continue), or NGX_ABORT (malformed
52 * object: missing <Key>, unreadable content, or allocation failure; the
53 * *malformed_500 out-flag separates client MalformedXML from a 500).
54 */
55ngx_int_t
56s3_delete_collect_one(s3_del_ctx_t *dc, xmlNodePtr obj, s3_del_item_t *it,
57 int *malformed_500)
58{
59 xmlNodePtr key_node;
60 xmlChar *key_text;
61 size_t key_len;
62 char key_str[S3_MAX_KEY];
63 char fs_path[PATH_MAX];
64 int rrc;
65
66 *malformed_500 = 0;
67 ngx_memzero(it, sizeof(*it));
68
69 key_node = s3_delete_xml_find_child(obj, "Key");
70 if (key_node == NULL) {
71 return NGX_ABORT; /* missing <Key> -> MalformedXML */
72 }
73
74 key_text = xmlNodeGetContent(key_node);
75 if (key_text == NULL) {
76 *malformed_500 = 1;
77 return NGX_ABORT; /* unreadable content -> 500 */
78 }
79 key_len = (size_t) xmlStrlen(key_text);
80
81 it->key = ngx_pnalloc(dc->r->pool, key_len + 1);
82 if (it->key == NULL) {
83 xmlFree(key_text);
84 *malformed_500 = 1;
85 return NGX_ABORT;
86 }
87 ngx_memcpy(it->key, key_text, key_len);
88 it->key[key_len] = '\0';
89 it->key_len = key_len;
90 xmlFree(key_text);
91
92 if (key_len == 0 || key_len >= S3_MAX_KEY) {
93 it->err.code = "InvalidArgument";
94 it->err.message = "Object key is empty or too long.";
95 return NGX_OK;
96 }
97
98 /* Confinement BEFORE any delete: a key that resolves outside the export
99 * root is rejected here without ever touching the filesystem, and the
100 * refusal does not leak whether the escaped path exists. */
101 ngx_memcpy(key_str, it->key, key_len + 1);
102 rrc = s3_resolve_key_ex(dc->cf->common.root_canon, key_str, fs_path,
103 sizeof(fs_path),
104 dc->cf->common.cache_store_endpoint);
105 if (rrc == 404) {
106 return NGX_OK; /* pre-disposed <Deleted>: reserved reads as absent */
107 }
108 if (rrc != 0) {
109 s3_key_error_t kerr;
110
111 s3_resolve_key_error(rrc, &kerr);
112 it->err.code = kerr.code;
113 it->err.message = kerr.message;
114 return NGX_OK;
115 }
116
117 it->fs = ngx_pnalloc(dc->r->pool, ngx_strlen(fs_path) + 1);
118 if (it->fs == NULL) {
119 *malformed_500 = 1;
120 return NGX_ABORT;
121 }
122 ngx_memcpy(it->fs, fs_path, ngx_strlen(fs_path) + 1);
123 return NGX_OK;
124}

126/* ---- Derive the canonical XrdAcc path for one collected key ----
127 *
128 * WHAT: Writes the path that an ordinary /bucket/key request would authorize,
129 * using the collected key's confined canonical filesystem target. Returns
130 * NGX_OK on success or NGX_ERROR if the target is not beneath the export or
131 * the output would overflow.
132 *
133 * WHY: Reusing raw XML text would authorize a different spelling from the one
134 * storage resolved (leading slashes, dot segments, and percent encodings).
135 * Authorization must describe the exact resource that the VFS will delete.
136 *
137 * HOW: Verify the root boundary, derive the canonical export-relative suffix,
138 * prepend the configured path-style bucket, and reject truncation.
139 */
140static ngx_int_t
141s3_delete_build_acc_path(const s3_del_ctx_t *dc, const s3_del_item_t *item,
142 char *path, size_t path_size)
143{
144 const char *root = dc->cf->common.root_canon;
145 const char *relative;
146 size_t root_len = ngx_strlen(root);
147 int written;
148
149 if (ngx_strncmp(item->fs, root, root_len) != 0) {
150 return NGX_ERROR;
151 }
152 if (root_len == 1 && root[0] == '/') {
153 relative = item->fs;
154 } else {
155 relative = item->fs + root_len;
156 if (relative[0] != '\0' && relative[0] != '/') {
157 return NGX_ERROR;
158 }
159 }
160 if (relative[0] == '\0') {
161 relative = "/";
162 }
163
164 if (dc->cf->bucket.len == 0) {
165 written = snprintf(path, path_size, "%s", relative);
166 } else if (dc->cf->bucket.len > INT_MAX) {
167 return NGX_ERROR;
168 } else {
169 written = snprintf(path, path_size, "/%.*s%s",
170 (int) dc->cf->bucket.len,
171 (const char *) dc->cf->bucket.data, relative);
172 }
173 if (written < 0 || (size_t) written >= path_size) {
174 return NGX_ERROR;
175 }
176 return NGX_OK;
177}

179/* ---- Authorize every body-selected key before batch mutation ----
180 *
181 * WHAT: Returns NGX_OK only when the configured XrdAcc engine grants DELETE
182 * on every collected, mutatable key; returns NGX_DECLINED on denial and
183 * NGX_ERROR on allocation/path-construction failure. It performs no storage
184 * mutation or response I/O.
185 *
186 * WHY: The outer POST names only the bucket, while the XML body selects up to
187 * 1,000 independent resources. One bucket-level CREATE decision cannot grant
188 * DELETE on those later paths, and partial mutation on denial is unsafe.
189 *
190 * HOW: Skip pre-disposed entries, derive each exact canonical /bucket/key path,
191 * and call the shared S3 XrdAcc decision helper with BRIX_AOP_DELETE. The caller
192 * invokes the VFS batch only after this complete pass succeeds.
193 */
194ngx_int_t
195s3_delete_authorize(s3_del_ctx_t *dc, const s3_del_item_t *items,
196 size_t count)
197{
198 ngx_http_s3_req_ctx_t *request_ctx;
199 char *logical_path;
200 size_t path_size;
201 size_t index;
202
203 if (dc->cf->common.acc.format != BRIX_AUTHDB_FORMAT_XRDACC) {
204 return NGX_OK;
205 }
206
207 if (dc->cf->bucket.len > (size_t) -1 - PATH_MAX - 2) {
208 return NGX_ERROR;
209 }
210 path_size = PATH_MAX + dc->cf->bucket.len + 2;
211 logical_path = ngx_pnalloc(dc->r->pool, path_size);
212 if (logical_path == NULL) {
213 return NGX_ERROR;
214 }
215
216 request_ctx = ngx_http_get_module_ctx(dc->r, ngx_http_brix_s3_module);
217 for (index = 0; index < count; index++) {
218 if (items[index].fs == NULL) {
219 continue;
220 }
221 if (s3_delete_build_acc_path(dc, &items[index], logical_path,
222 path_size) != NGX_OK)
223 {
224 return NGX_ERROR;
225 }
226 if (s3_acc_authorize_path(dc->r, dc->cf,
227 request_ctx != NULL ? request_ctx->identity : NULL,
228 BRIX_AOP_DELETE, logical_path) != NGX_OK)
229 {
230 return NGX_DECLINED;
231 }
232 }
233 return NGX_OK;
234}

236/*
237 * s3_delete_execute - phase 3: ONE brix_vfs_delete_many() call for every key
238 * that survived collection (one write gate, one OP_DELETE observation, one
239 * driver batch where the leaf has unlink_many).
240 *
241 * WHAT: Builds the confinable subset (fs != NULL) with a batch->item index
242 * map, runs the VFS batch, writes each per-key errno back into its
243 * item's batch_err, and emits a CNS retraction per actually-removed key.
244 * WHY: EROFS is the one whole-request refusal: it is raised BEFORE any key is
245 * examined, so answering 403 for the request as a whole discloses
246 * nothing per key (the doc's security-negative). Every other batch
247 * failure keeps the per-key vocabulary - untried keys hold the
248 * ECANCELED the VFS pre-filled.
249 * HOW: Returns NGX_OK (per-key verdicts recorded - render next) or NGX_DONE
250 * (response finalised here: EROFS 403 or allocation 500).
251 */
252ngx_int_t
253s3_delete_execute(s3_del_ctx_t *dc, s3_del_item_t *items, size_t count)
254{
255 ngx_http_request_t *r = dc->r;
256 ngx_http_s3_req_ctx_t *s3ctx =
257 ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
258 brix_vfs_ctx_t vctx;
259 const char **fs;
260 int *errs;
261 size_t *map;
262 size_t i, m = 0, done = 0;
263 ngx_int_t rc;
264
265 fs = ngx_palloc(r->pool, count * sizeof(*fs));
266 errs = ngx_palloc(r->pool, count * sizeof(*errs));
267 map = ngx_palloc(r->pool, count * sizeof(*map));
268 if (fs == NULL || errs == NULL || map == NULL) {
269 s3_metrics_finalize_request_method(r, dc->method_slot,
270 NGX_HTTP_INTERNAL_SERVER_ERROR);
271 return NGX_DONE;
272 }
273
274 for (i = 0; i < count; i++) {
275 if (items[i].fs != NULL) {
276 fs[m] = items[i].fs;
277 map[m] = i;
278 m++;
279 }
280 }
281 if (m == 0) {
282 return NGX_OK; /* every key was pre-disposed at collection */
283 }
284
285 brix_vfs_ctx_init(&vctx, r->pool, r->connection->log, BRIX_PROTO_S3,
286 dc->cf->common.root_canon, dc->cf->common.cache_root_canon,
287 brix_vfs_policy_from_write_enable(dc->cf->common.allow_write),
288 brix_http_request_is_tls(r),
289 (s3ctx != NULL) ? s3ctx->identity : NULL,
290 dc->cf->common.root_canon);
291 s3_vfs_bind_deleg(r, dc->cf, &vctx);
292
293 rc = brix_vfs_delete_many(&vctx, fs, m, errs, &done);
294 if (rc != NGX_OK && errno == EROFS) {
295 s3_metrics_finalize_request_method(r, dc->method_slot,
296 s3_send_xml_error(r, NGX_HTTP_FORBIDDEN, "AccessDenied",
297 "The export is read-only."));
298 return NGX_DONE;
299 }
300 if (rc != NGX_OK && errno == EBUSY) {
301 /* phase-107 C7: the per-key lock gate refused BEFORE any arm ran, so
302 * the refusal is atomic - no key was attempted (errs all ECANCELED).
303 * Answer the whole request with the same 409 OperationAborted the
304 * single-key DELETE gives a locked key (§5.0); per-key InternalError
305 * rows would hide the lock and imply a server fault. */
306 s3_metrics_finalize_request_method(r, dc->method_slot,
307 s3_send_xml_error(r, NGX_HTTP_CONFLICT, "OperationAborted",
308 "A conflicting conditional operation is "
309 "currently in progress against this "
310 "resource."));
311 return NGX_DONE;
312 }
313
314 for (i = 0; i < m; i++) {
315 items[map[i]].batch_err = errs[i];
316 if (errs[i] == 0) {
317 /* phase-97 §5: a real removal - retract it from the manager
318 * inventory. ENOENT removed nothing and emits nothing. */
319 brix_cns_emit_at(dc->cf->common.root_canon, BRIX_CNS_DEL,
320 fs[i], 0, 0);
321 }
322 }
323 return NGX_OK;
324}

326/* Map a batch per-key errno onto the S3 error pair - the same vocabulary the
327 * per-key handler used, plus ECANCELED for a key the batch never reached
328 * (which must never render as <Deleted>). */
329static void
330s3_delete_batch_err_pair(int e, s3_del_err_t *err)
331{
332 if (e == EACCES || e == EPERM) {
333 err->code = "AccessDenied";
334 err->message = "Access Denied.";
335 } else if (e == ENOTEMPTY) {
336 err->code = "BucketNotEmpty";
337 err->message = "The directory is not empty.";
338 } else if (e == ECANCELED) {
339 err->code = "InternalError";
340 err->message = "The key was not attempted because the batch failed.";
341 } else {
342 err->code = "InternalError";
343 err->message = "Internal server error.";
344 }
345}

347/*
348 * s3_delete_render - phase 4: one <Deleted> or <Error> element per item, in
349 * the client's original order. <Deleted> covers: pre-disposed at collection
350 * (reserved-absent), removed (batch_err 0), and idempotent-missing (ENOENT).
351 * Returns NGX_ERROR only on output-buffer overflow (caller maps to 500).
352 */
353ngx_int_t
354s3_delete_render(s3_del_ctx_t *dc, const s3_del_item_t *items, size_t count)
355{
356 size_t i;
357
358 for (i = 0; i < count; i++) {
359 const s3_del_item_t *it = &items[i];
360 s3_del_err_t err;
361 ngx_int_t rc;
362
363 if (it->err.code != NULL) {
364 rc = s3_delete_xml_append_error(dc->xml_buf, dc->xml_len,
365 it->key, it->key_len, &it->err);
366 } else if (it->fs == NULL || it->batch_err == 0
367 || it->batch_err == ENOENT)
368 {
369 rc = s3_delete_xml_append_deleted(dc->xml_buf, dc->xml_len,
370 it->key, it->key_len);
371 } else {
372 s3_delete_batch_err_pair(it->batch_err, &err);
373 rc = s3_delete_xml_append_error(dc->xml_buf, dc->xml_len,
374 it->key, it->key_len, &err);
375 }
376 if (rc != NGX_OK) {
377 return NGX_ERROR;
378 }
379 }
380 return NGX_OK;
