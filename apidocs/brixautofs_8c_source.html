1/*
2 * brixautofs.c — CVMFS-brix-autofs: the /cvmfs automount umbrella daemon.
3 *
4 * WHAT: `brixMount autofs <etc-root|-> <mountdir>` mounts a FUSE umbrella on
5 * <mountdir> (normally /cvmfs). Resolving a valid repository FQRN
6 * fork/execs `brixMount cvmfs <fqrn> <mntbase>/<fqrn> -f -o …` as a real
7 * FUSE mount in an EXTERNAL mount farm and answers the name with a
8 * symlink to it — the first path resolution blocks until the repo is
9 * live (autofs UX), every later access follows the link straight into
10 * the mounted fs.
11 * WHY: the stock CVMFS client's autofs experience with zero dependency on
12 * autofs or systemd — the out-of-the-box answer for WSL2 and containers,
13 * and a plain daemon for everything else. One process per repo also
14 * isolates a hung stratum to that repo.
15 * HOW: children CANNOT mount over the umbrella's own virtual subdir: the
16 * kernel serializes every lookup of one name on the in-lookup dentry
17 * (d_alloc_parallel), so while our LOOKUP handler waits for the child,
18 * the child's own mount(2) path walk of the same name waits on US — a
19 * guaranteed deadlock the autofs kernel module exists to solve. A plain
20 * FUSE umbrella therefore mounts children in a farm dir it never path-
21 * walks itself (default <cachebase>/.mnt) and presents each repo as a
22 * symlink; readlink is the (blocking) mount trigger, lstat/readdir never
23 * mount. Works fully unprivileged (no bind mounts needed). Idle repos
24 * expire via umount2(MNT_EXPIRE) two-phase (root only); SIGTERM unmounts
25 * every child before the umbrella itself goes away. Honors
26 * CVMFS_REPOSITORIES / CVMFS_STRICT_MOUNT from the stock config cascade.
27 * Pure parts (FQRN gate, slot table) live in brixautofs.h and compile
28 * without libfuse for the unit test (-DBRIXAUTOFS_UNIT).
29 */

37int brixautofs_valid_fqrn(const char *name) {
38 if (name == NULL || name[0] == '\0') return 0;
39 size_t len = strlen(name);
40 if (len > BRIXAUTOFS_FQRN_MAX - 1) return 0;
41
42 int labels = 0;
43 size_t lab = 0; /* current label length */
44 for (size_t i = 0; i <= len; i++) {
45 char c = name[i];
46 if (c == '.' || c == '\0') {
47 if (lab == 0 || lab > 63) return 0; /* empty/overlong label */
48 if (name[i - 1] == '-') return 0; /* label ends with '-' */
49 labels++;
50 lab = 0;
51 continue;
52 }
53 int ok = (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-';
54 if (!ok) return 0; /* '/', '_', upper, meta… */
55 if (lab == 0 && c == '-') return 0; /* label starts with '-' */
56 lab++;
57 }
58 return labels >= 2; /* needs an org AND a domain part */
59}

61int brixautofs_repo_listed(const char *list, const char *fqrn) {
62 if (list == NULL || fqrn == NULL) return 0;
63 size_t fl = strlen(fqrn);
64 const char *p = list;
65 while (*p) {
66 while (*p == ',' || *p == ':' || *p == ' ' || *p == '\t') p++;
67 const char *tok = p;
68 while (*p && *p != ',' && *p != ':' && *p != ' ' && *p != '\t') p++;
69 if ((size_t)(p - tok) == fl && strncmp(tok, fqrn, fl) == 0) return 1;
70 }
71 return 0;
72}

148/* 1 iff `path` is a mount point per /proc/self/mountinfo (field 5). FQRNs
149 * never contain the \040-style escapes mountinfo uses, so plain compare. */
150int af_is_mounted(const char *path) {
151 FILE *f = fopen("/proc/self/mountinfo", "r");
152 if (f == NULL) return 0;
153 char line[1024];
154 int found = 0;
155 while (!found && fgets(line, sizeof(line), f) != NULL) {
156 /* fields: id parent maj:min root mountpoint … */
157 char *save = NULL, *tok = strtok_r(line, " ", &save);
158 for (int i = 0; tok != NULL && i < 4; i++) tok = strtok_r(NULL, " ", &save);
159 if (tok != NULL && strcmp(tok, path) == 0) found = 1;
160 }
161 fclose(f);
162 return found;
163}

171/* mkdir -p (two fixed levels are enough: farm base, then the per-repo dir). */
172int af_mkdir_p(const char *path) {
173 char buf[768];
174 snprintf(buf, sizeof(buf), "%s", path);
175 for (char *p = buf + 1; *p; p++) {
176 if (*p != '/') continue;
177 *p = '\0';
178 if (mkdir(buf, 0755) != 0 && errno != EEXIST) return -1;
179 *p = '/';
180 }
181 if (mkdir(buf, 0755) != 0 && errno != EEXIST) return -1;
182 return 0;
183}

185/* Unmount `path`: umount2 when root (MNT_DETACH on EBUSY), else fusermount3
186 * (-u, then lazy -uz). Best-effort; returns 0 if the mount is gone. */
187int af_umount_path(const char *path) {
188 if (geteuid() == 0) {
189 if (umount2(path, 0) == 0) return 0;
190 if (errno == EBUSY && umount2(path, MNT_DETACH) == 0) return 0;
191 if (!af_is_mounted(path)) return 0;
192 }
193 const char *modes[] = { "-u", "-uz" };
194 for (int m = 0; m < 2; m++) {
195 pid_t pid = fork();
196 if (pid == 0) {
197 int devnull = open("/dev/null", O_RDWR);
198 if (devnull >= 0) { dup2(devnull, 1); dup2(devnull, 2); }
199 execlp("fusermount3", "fusermount3", modes[m], path, (char *) NULL);
200 _exit(127);
201 }
202 if (pid > 0) {
203 int st = 0;
204 waitpid(pid, &st, 0);
205 }
206 if (!af_is_mounted(path)) return 0;
207 }
208 return af_is_mounted(path) ? -1 : 0;
209}

211/* Build the child's `-o` string. `cache=` pins a per-repo cache dir (which
212 * also disables brixcvmfs "clever" mode — it must never write onto the
213 * mountpoint under the umbrella); nodev,nosuid harden the nested mount.
214 * $BRIXAUTOFS_CHILD_OPTS appends extra tokens (tests). Must stay < 512
215 * (brixcvmfs parse_opts buffer cap). */
216static void af_child_opts(const char *fqrn, char *out, size_t cap) {
217 const char *extra = getenv("BRIXAUTOFS_CHILD_OPTS");
218 snprintf(out, cap, "cache=%s/%s,nodev,nosuid%s%s%s",
219 g_af.o.cache_base, fqrn,
220 g_af.o.allow_other ? ",allow_other" : "",
221 extra ? "," : "", extra ? extra : "");
222}

224/* Fork/exec the per-repo mount (`brixMount cvmfs <fqrn> <mnt>/<fqrn> -f -o …`)
225 * and poll /proc/self/mountinfo until the nested mount is live. The child runs
226 * -f (no daemonize) so its pid stays ours to reap and its exit before the
227 * mount lands is a definitive failure. Returns the child pid, or -1. */
228static pid_t af_spawn_child(const char *fqrn) {
229 char mntpath[768], opts[512], exe[512];
230 af_child_path(fqrn, mntpath, sizeof(mntpath));
231 af_child_opts(fqrn, opts, sizeof(opts));
232
233 ssize_t el = readlink("/proc/self/exe", exe, sizeof(exe) - 1);
234 if (el > 0) {
235 exe[el] = '\0';
236 } else {
237 const char *env = getenv("BRIXMOUNT_BIN");
238 if (env == NULL) { af_log("cannot resolve own binary path"); return -1; }
239 snprintf(exe, sizeof(exe), "%s", env);
240 }
241
242 if (af_mkdir_p(mntpath) != 0) {
243 af_log("cannot create mountpoint %s: %s", mntpath, strerror(errno));
244 return -1;
245 }
246
247 pid_t pid = fork();
248 if (pid < 0) { af_log("fork: %s", strerror(errno)); return -1; }
249 if (pid == 0) {
250 int devnull = open("/dev/null", O_RDONLY);
251 if (devnull >= 0) dup2(devnull, 0);
252 execl(exe, exe, "cvmfs", fqrn, mntpath, "-f", "-o", opts, (char *) NULL);
253 _exit(127);
254 }
255
256 int deadline_ms = g_af.o.spawn_timeout_s * 1000;
257 for (int waited = 0; waited < deadline_ms; waited += 100) {
258 if (af_is_mounted(mntpath)) return pid;
259 int st = 0;
260 if (waitpid(pid, &st, WNOHANG) == pid) {
261 af_log("mount of %s failed (child exit %d)", fqrn,
262 WIFEXITED(st) ? WEXITSTATUS(st) : -1);
263 return -1;
264 }
265 struct timespec ts = { 0, 100 * 1000 * 1000 };
266 nanosleep(&ts, NULL);
267 }
268 af_log("mount of %s timed out after %ds", fqrn, g_af.o.spawn_timeout_s);
269 kill(pid, SIGTERM);
270 waitpid(pid, NULL, 0);
271 af_umount_path(mntpath); /* in case it landed between poll and kill */
272 return -1;
273}

275/* Admission gate shared by every op: is `name` a repo this umbrella may show
276 * at all? (validity + strict-mount policy; never touches the slot table). */
277static int af_admit(const char *name) {
278 if (!brixautofs_valid_fqrn(name)) return 0;
279 if (g_af.strict && !brixautofs_repo_listed(g_af.repos, name)) return 0;
280 return !g_af.shutting_down;
281}

283/* The mount trigger: drive the slot state machine and block until the child
284 * mount is live in the farm. Called from readlink only — the triggering path
285 * resolution waits (autofs UX); the child's own mount(2) walk happens in the
286 * farm, far away from the umbrella's dentries, so nothing re-enters us.
287 * Returns 0 when <farm>/<name> is a live mount, else -ENOENT. */
288static int af_ensure_repo(const char *name) {
289 if (!af_admit(name)) return -ENOENT;
290
291 pthread_mutex_lock(&g_af.tab.mu);
292 int idx = brixautofs_find_locked(&g_af.tab, name);
293 if (idx >= 0) {
294 /* MOUNTED: done. MOUNTING: another resolution is bringing it up; its
295 * kernel-side lookup already serializes ours, so just answer. */
296 pthread_mutex_unlock(&g_af.tab.mu);
297 return 0;
298 }
299 idx = brixautofs_claim_locked(&g_af.tab, name);
300 pthread_mutex_unlock(&g_af.tab.mu);
301 if (idx < 0) {
302 af_log("repo table full (%d), refusing %s", BRIXAUTOFS_MAX_REPOS, name);
303 return -ENOENT;
304 }
305
306 pid_t pid = af_spawn_child(name); /* blocking: the autofs UX — the
307 * triggering resolution waits */
308 pthread_mutex_lock(&g_af.tab.mu);
309 if (pid > 0) brixautofs_commit_locked(&g_af.tab, idx, pid);
310 else brixautofs_release_locked(&g_af.tab, idx);
311 pthread_mutex_unlock(&g_af.tab.mu);
312 return pid > 0 ? 0 : -ENOENT;
313}

338/* getattr NEVER mounts: an admissible name is a symlink into the farm, so a
339 * bare lstat / colorized ls of ghosts stays cheap. Following the link (stat,
340 * open, cd) goes through readlink — that is where the mount happens. */
341static int af_getattr(const char *path, struct stat *st, struct fuse_file_info *fi) {
342 (void) fi;
343 if (strcmp(path, "/") == 0) { af_fill_dir_stat(st); return 0; }
344 const char *name = path + 1;
345 if (strchr(name, '/') != NULL) return -ENOENT; /* depth 1 only */
346 if (!af_admit(name)) return -ENOENT;
347 af_fill_link_stat(st, name);
348 return 0;
349}

351/* readlink IS the mount trigger: block until the repo is live, then hand the
352 * kernel the farm path to walk into. */
353static int af_readlink(const char *path, char *buf, size_t cap) {
354 const char *name = path + 1;
355 if (strchr(name, '/') != NULL) return -ENOENT;
356 int rc = af_ensure_repo(name);
357 if (rc != 0) return rc;
358 char target[768];
359 af_child_path(name, target, sizeof(target));
360 snprintf(buf, cap, "%s", target);
361 return 0;
362}

370/* readdir("/") = "." ".." ∪ mounted repos ∪ configured repos (ghost listing:
371 * CVMFS_REPOSITORIES + the config.d .conf entries). NEVER spawns a mount. */
372/* Copy the idx-th token of a comma/colon/space-separated repo list into out.
373 * Returns 1 if that token exists, 0 once the list is exhausted. */
374static int af_repos_nth_token(const char *list, int idx, char *out, size_t outsz) {
375 const char *p = list;
376 int tok = 0;
377 while (*p) {
378 while (*p == ',' || *p == ':' || *p == ' ') p++;
379 const char *s = p;
380 while (*p && *p != ',' && *p != ':' && *p != ' ') p++;
381 if (p == s) break;
382 if (tok++ == idx) {
383 snprintf(out, outsz, "%.*s", (int)(p - s), s);
384 return 1;
385 }
386 }
387 return 0;
388}

390/* The idx-th ghost-candidate name for source src (0 = the ghost array, 1 = the
391 * configured repos list). Returns 1 with the name in out, or 0 to stop src. */
392static int af_ghost_name(int src, int idx, char *out, size_t outsz) {
393 if (src == 0) {
394 if (idx >= g_af.nghost) return 0;
395 snprintf(out, outsz, "%s", g_af.ghost[idx]);
396 return 1;
397 }
398 return af_repos_nth_token(g_af.repos, idx, out, outsz);
399}

409/* Emit the live-mounted repos (under the table lock), recording each in seen. */
410static void af_fill_mounted(void *buf, fuse_fill_dir_t fill,
411 char seen[][BRIXAUTOFS_FQRN_MAX], int *nseen) {
412 pthread_mutex_lock(&g_af.tab.mu);
413 for (int i = 0; i < BRIXAUTOFS_MAX_REPOS; i++) {
414 if (g_af.tab.slot[i].st == BRIXAUTOFS_FREE) continue;
415 snprintf(seen[*nseen], BRIXAUTOFS_FQRN_MAX, "%s", g_af.tab.slot[i].fqrn);
416 fill(buf, seen[(*nseen)++], NULL, 0, 0);
417 }
418 pthread_mutex_unlock(&g_af.tab.mu);
419}

421static int af_readdir(const char *path, void *buf, fuse_fill_dir_t fill,
422 off_t off, struct fuse_file_info *fi,
423 enum fuse_readdir_flags flags) {
424 (void) off; (void) fi; (void) flags;
425 fill(buf, ".", NULL, 0, 0);
426 fill(buf, "..", NULL, 0, 0);
427 if (strcmp(path, "/") != 0) return 0; /* unreachable: opendir admits "/" only */
428
429 char seen[BRIXAUTOFS_MAX_REPOS * 2][BRIXAUTOFS_FQRN_MAX];
430 int nseen = 0;
431
432 af_fill_mounted(buf, fill, seen, &nseen);
433
434 /* ghost entries: configured but not (yet) mounted */
435 for (int src = 0; src < 2; src++) {
436 for (int i = 0; ; i++) {
437 char name[BRIXAUTOFS_FQRN_MAX] = "";
438 if (!af_ghost_name(src, i, name, sizeof(name))) break;
439 if (af_seen_has(seen, nseen, name) || !brixautofs_valid_fqrn(name))
440 continue;
441 if (nseen >= (int)(sizeof(seen) / sizeof(seen[0]))) break;
442 snprintf(seen[nseen], BRIXAUTOFS_FQRN_MAX, "%s", name);
443 fill(buf, seen[nseen++], NULL, 0, 0);
444 }
445 }
446 return 0;
447}
