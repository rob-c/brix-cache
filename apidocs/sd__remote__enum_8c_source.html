1/*
2 * sd_remote_enum.c — the backend-catalog verb (driver->enumerate) for s3://.
3 *
4 * WHAT: sd_remote_enumerate() reports every object the bucket holds, once each,
5 * with its size and mtime, to the shared brix_sd_catalog_cb.
6 * WHY: inventory, drift detection and the background scrub ask a driver what it
7 * STORES, which is not the same question as what a directory contains. An
8 * object store has no namespace to walk: before this slot, an S3 export
9 * fell back to brix_vfs_walk recursing through opendir/readdir, i.e. one
10 * signed delimited LIST per pseudo-directory, and every entry it found
11 * then cost a HEAD to stat. The flat listing answers with 1000 keys per
12 * request AND carries size/mtime in the same response, so a stat-bearing
13 * enumeration of a bucket costs ceil(n/1000) requests instead of O(n).
14 * HOW: page sd_s3_list_flat_page from the bucket root, translating each key
15 * into a catalog entry: `key` is the S3 key as stored, `path` is the
16 * export-relative logical path (a leading '/' + the key), and the entry is
17 * skipped when it is namespace scaffolding rather than a stored object.
18 * Blocking, transport-injected, malloc-only — safe on the fill thread,
19 * never on the event loop.
20 */
21

29/* The logical path buffer: one leading '/' plus an S3 key. S3 caps a key at
30 * 1024 bytes, so this holds any key the store can legally have returned. */
31#define SD_REMOTE_ENUM_PATH_MAX 1100
32
33typedef struct {
34 brix_sd_catalog_cb cb;
35 void *ctx;
36 int want_stat;
37 int stopped; /* the callback asked to stop */
38} sd_remote_enum_ctx;

40/* sd_s3_list_flat_page callback: translate one listed key into a catalog entry.
41 *
42 * A key ending in '/' is a DIRECTORY-MARKER object — the zero-byte object this
43 * driver's mkdir writes to give S3 a namespace it does not natively have. It is
44 * scaffolding, not stored content: reporting it would have an inventory count a
45 * directory as a file and a drift check treat every folder as an orphan. This
46 * mirrors sd_ceph_enumerate skipping striper data stripes.
47 *
48 * Returns non-zero to stop the enumeration, which is how the caller's own stop
49 * signal propagates back through the pager. */
50static int
51sd_remote_enum_entry(void *ud, const char *key, uint64_t size, time_t mtime)
52{
53 sd_remote_enum_ctx *e = ud;
54 brix_sd_catalog_ent_t ent;
55 char path[SD_REMOTE_ENUM_PATH_MAX];
56 size_t klen = strlen(key);
57
58 if (klen == 0 || key[klen - 1] == '/') {
59 return 0; /* directory marker — not a stored object */
60 }
61
62 ngx_memzero(&ent, sizeof(ent));
63 ent.key = key;
64 /* The bucket root IS the export root here (sd_remote_s3_key composes
65 * "/bucket" + the export-relative path), so the logical path is the key with
66 * the leading '/' restored. A key too long to represent is reported with
67 * path=NULL — an orphan candidate — rather than dropped: it exists, and an
68 * inventory that silently omitted it would under-report the store. */
69 if (klen + 2 <= sizeof(path)) {
70 path[0] = '/';
71 memcpy(path + 1, key, klen + 1);
72 ent.path = path;
73 }
74 if (e->want_stat && mtime != 0) {
75 ent.have_stat = 1;
76 ent.size = (off_t) size;
77 ent.mtime = mtime;
78 }
79 if (e->cb(e->ctx, &ent) != 0) {
80 e->stopped = 1;
81 return 1;
82 }
83 return 0;
84}

86/* driver->enumerate — see the file header. NGX_OK when the enumeration ran (the
87 * callback may have aborted it early, which is not an error), or NGX_ERROR with
88 * errno set when a page could not be listed. */
89ngx_int_t
90sd_remote_enumerate(brix_sd_instance_t *inst, int want_stat,
91 brix_sd_catalog_cb cb, void *ctx)
92{
93 const brix_sd_remote_cfg_t *cfg;
94 sd_remote_enum_ctx e;
95 sd_s3_open_params p;
96 char root[300];
97 char cont[2048];
98 char cont_out[2048];
99 char errbuf[256];
100 int started = 0;
101 int truncated = 0;
102
103 if (inst == NULL || inst->state == NULL || cb == NULL) {
104 errno = EINVAL;
105 return NGX_ERROR;
106 }
107 cfg = inst->state;
108
109 e.cb = cb;
110 e.ctx = ctx;
111 e.want_stat = want_stat;
112 e.stopped = 0;
113 cont[0] = '\0';
114
115 snprintf(root, sizeof(root), "/%s/", cfg->bucket); /* bucket-root canon URI */
116 sd_remote_s3_params(cfg, root, &p);
117
118 /* Signed as the EXPORT, deliberately: this is not a request path. The
119 * catalog verb answers the inventory/scrub engine about what the export
120 * holds — there is no requesting user to scope it to, and a per-user view
121 * would make an inventory report a subset of the store as the whole of it.
122 * The credential-scoped listing a user's request DOES get is opendir_cred. */
123 do {
124 errno = 0;
125 if (sd_s3_list_flat_page(&p, "", started ? cont : "",
126 sd_remote_enum_entry, &e, &truncated, cont_out,
127 sizeof(cont_out), errbuf, sizeof(errbuf)) != 0)
128 {
129 if (errno == 0) { errno = EIO; }
130 return NGX_ERROR;
131 }
132 started = 1;
133 snprintf(cont, sizeof(cont), "%s", cont_out);
134 /* A truncated page with no continuation token would loop forever on the
135 * same page: stop instead. S3 never does this, but the token is the only
136 * thing advancing the cursor and a proxy that drops it must not wedge a
137 * scrub thread. */
138 if (truncated && cont[0] == '\0') {
139 break;
140 }
141 } while (truncated && !e.stopped);
142
143 return NGX_OK;
144}
