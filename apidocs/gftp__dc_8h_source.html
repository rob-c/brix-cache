4/*
5 * gftp_dc.h — one outbound GridFTP data connection.
6 *
7 * WHAT: open a passive data connection to the origin (EPSV, falling back to
8 * PASV), optionally raise it to PROT P, and read/write/close it. A gftp_dc_t is
9 * the ONLY handle the transfer code (gftp_data.c, gftp_mode_e.c) holds on the
10 * data socket.
11 *
12 * WHY: before phase-115 the data socket was a bare int and every transfer
13 * called recv()/send() on it directly. PROT P puts a TLS session in front of
14 * that socket, and MODE E opens several of them at once; both changes would have
15 * to be made in each transfer function separately if the fd stayed bare. One
16 * handle with one open/read/write/close surface means the protection decision is
17 * made once, at open time, and a caller cannot accidentally write cleartext onto
18 * a channel the store line asked to protect.
19 *
20 * HOW: gftp_dc_open() runs EPSV/PASV and connects to the control channel's
21 * PINNED numeric peer (never a fresh name lookup — the reply's address is
22 * attacker-influenced). On a PROT P session the TLS handshake is a SEPARATE
23 * step, gftp_dc_secure(), which the transfer code runs only AFTER the transfer
24 * command has been accepted — see the note on that function for why the order
25 * is not free. read/write dispatch on dc->tls, so a protected connection
26 * cannot fall back to the clear path: if the handshake fails the connection is
27 * closed and the transfer fails.
28 */
29

54/* Raise an already-dialled connection to PROT P, or do nothing on a clear
55 * session. Returns 0 (secured, or nothing to do) or -1 with the session error
56 * set and `dc` closed.
57 *
58 * CALL IT AFTER THE TRANSFER COMMAND, never at dial time. A passive FTP server
59 * has no reason to look at the data connection until it has a transfer to run:
60 * it accepts and starts its side of the handshake when RETR/STOR arrives, not
61 * when the socket appears. A client that handshakes at dial time is waiting
62 * for a ServerHello the server will not send until it reads a command the
63 * client has not sent yet, and both sides sit there until something times out.
64 * That is not a slow path — it is a deadlock, and it is what made every PROT P
65 * transfer fail: the origin logged an idle control channel and the client an
66 * SSL_connect with an empty error queue, neither of which names the cause. */
67int gftp_dc_secure(gftp_session_t *session, gftp_dc_t *dc);
