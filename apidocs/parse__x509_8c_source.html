10/*
11 * gsi_persist_session_cipher — stash the negotiated GSI session cipher on the
12 * connection so a later kXGS_pxyreq/kXGC_sigpxy delegation round (phase-57 §F6)
13 * can encrypt/decrypt its main with the same key. Purely additive: the key is
14 * already derived for the kXGC_cert decrypt; this copies it (≤32 bytes) + the
15 * cipher name + the IV flag. Inert unless brix_tpc_delegate consumes it. The
16 * key is cleansed once delegation completes (auth.c) or at disconnect.
17 */
18void
19gsi_persist_session_cipher(brix_ctx_t *ctx, const char *name,
20 const u_char *key, int keylen, int use_iv)
21{
22 int n = (keylen > 32) ? 32 : (keylen < 0 ? 0 : keylen);
23 size_t nl = ngx_strlen(name);
24
25 ngx_memcpy(ctx->gsi.sess_key, key, (size_t) n);
26 ctx->gsi.sess_keylen = n;
27 ctx->gsi.sess_use_iv = use_iv;
28 if (nl > sizeof(ctx->gsi.sess_cipher) - 1) {
29 nl = sizeof(ctx->gsi.sess_cipher) - 1;
30 }
31 ngx_memcpy(ctx->gsi.sess_cipher, name, nl);
32 ctx->gsi.sess_cipher[nl] = '\0';
33}

60/*
61 * gsi_capture_fullproxy — stash an OPTIONAL client-pushed full proxy PEM
62 * (kXRS_x509_fullproxy) from the decrypted kXGC_cert inner buffer (phase-70
63 * §5.1). Purely additive: absent in every stock client, present only when the
64 * user opted in. Heap-copies the raw bytes onto ctx->gsi.client_fullproxy_pem;
65 * auth.c later validates (DN == authenticated DN) and promotes them. The bytes
66 * carry a PRIVATE KEY, so they are never logged. No-op when the bucket is
67 * missing, empty, or a prior one was already captured.
68 */
69void
70gsi_capture_fullproxy(brix_ctx_t *ctx, const u_char *plain, size_t plain_len)
71{
72 const u_char *pem = NULL;
73 size_t pemlen = 0;
74 u_char *copy;
75
76 if (ctx->gsi.client_fullproxy_pem != NULL) {
77 return;
78 }
79 if (gsi_find_bucket(plain, plain_len, (uint32_t) kXRS_x509_fullproxy,
80 &pem, &pemlen) != 0 || pemlen == 0) {
81 return;

92STACK_OF(X509) *
93gsi_chain_from_plaintext(const u_char *plain, int plain_len,
94 const u_char expected_rtag[BRIX_GSI_RTAG_LEN], ngx_log_t *log)
95{

97 const u_char *signed_rtag = NULL;
98 size_t x509_len = 0;
99 size_t signed_rtag_len = 0;
100 BIO *bio;
101 X509 *cert;
102 EVP_PKEY *leaf_key;
103 STACK_OF(X509) *chain;
104
105 if (gsi_find_bucket(plain, (size_t) plain_len, (uint32_t) kXRS_x509,
106 &x509_data, &x509_len) != 0) {
107 BRIX_LOG(NGX_LOG_WARN, log, 0,
108 "brix: GSI kXGC_cert: kXRS_x509 not found "
109 "in decrypted inner buffer");
110 return NULL;
111 }
112
113 bio = BIO_new_mem_buf(x509_data, (int) x509_len);
114 chain = sk_X509_new_null();
115 if (!bio || !chain) {
116 BIO_free(bio);
117 sk_X509_free(chain);
118 return NULL;
119 }
120 while ((cert = PEM_read_bio_X509(bio, NULL, NULL, NULL)) != NULL) {
121 sk_X509_push(chain, cert);
122 }
123 ERR_clear_error(); /* the terminating PEM_read failure is expected */
124 BIO_free(bio);
125
126 if (sk_X509_num(chain) == 0) {
127 BRIX_LOG(NGX_LOG_WARN, log, 0,
128 "brix: GSI kXGC_cert: kXRS_x509 contained no certs");
129 sk_X509_pop_free(chain, X509_free);
130 return NULL;
131 }
132
133 if (brix_gsi_find_unique_bucket(plain, (size_t) plain_len,
134 (uint32_t) kXRS_signed_rtag,
135 &signed_rtag, &signed_rtag_len) != 0) {
136 BRIX_LOG(NGX_LOG_WARN, log, 0,
137 "brix: GSI kXGC_cert: unique signed server challenge missing");
138 sk_X509_pop_free(chain, X509_free);
139 return NULL;
140 }
141 leaf_key = X509_get_pubkey(sk_X509_value(chain, 0));
142 if (leaf_key == NULL
143 || !brix_gsi_rsa_verify_raw(leaf_key, signed_rtag, signed_rtag_len,
144 expected_rtag, BRIX_GSI_RTAG_LEN)) {
145 BRIX_LOG(NGX_LOG_WARN, log, 0,
146 "brix: GSI kXGC_cert: client leaf-key proof failed");
147 brix_evp_pkey_free(leaf_key);
148 sk_X509_pop_free(chain, X509_free);
149 return NULL;
150 }
151 brix_evp_pkey_free(leaf_key);
152 BRIX_LOG_DEBUG(NGX_LOG_DEBUG_STREAM, log, 0,
153 "brix: GSI parsed %d cert(s) with leaf-key proof",
154 sk_X509_num(chain));
155 return chain;
156}

158/*
159 * gsi_arm_request_signing — WHAT: arm kXR_sigver verification by copying the
160 * just-persisted session cipher (ctx->gsi.sess_*) into the sigver-owned fields
161 * and setting signing_active. WHY: the signature scheme is stock XrdSecProtect
162 * secver 0 — the covered-bytes SHA-256 encrypted with the SESSION cipher — and
163 * the verifier must outlive a §F6 delegation round, which cleanses
164 * ctx->gsi.sess_key; sigver therefore keeps its own copy. HOW: validates that a
165 * cipher was persisted and resolves in the allowlist with a key long enough;
166 * only then copies name + key + IV flag and activates. Call immediately after
167 * gsi_persist_session_cipher (both round-2 paths). Returns 1 if armed, 0 not.
168 */
169int
170gsi_arm_request_signing(brix_ctx_t *ctx)
171{

173
174 if (ctx->gsi.sess_keylen <= 0
175 || !brix_gsi_cipher_lookup(ctx->gsi.sess_cipher, &cipher)
176 || ctx->gsi.sess_keylen < cipher.key_len)
177 {
178 return 0;
179 }
180 ngx_memcpy(ctx->sigver.sig_cipher, ctx->gsi.sess_cipher,
181 sizeof(ctx->sigver.sig_cipher));
182 ngx_memcpy(ctx->sigver.sig_key, ctx->gsi.sess_key,
183 (size_t) ctx->gsi.sess_keylen);
184 ctx->sigver.sig_keylen = ctx->gsi.sess_keylen;
185 ctx->sigver.sig_use_iv = ctx->gsi.sess_use_iv;
