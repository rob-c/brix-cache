4 * The CMS clustering config group — the cluster role/response enums, one

55/*
56 * §2.13 (cms.fsxeq): one operator program registered for one or more forwarded
57 * namespace ops. A directive line contributes exactly one of these and every
58 * op the line names points at it, so `brix_cms_fsxeq mv rm /usr/local/bin/nsop`
59 * stores the command once. argv holds the FIXED prefix tokens (the program and
60 * any literal arguments the operator baked into the line, NUL-terminated, from
61 * cf->pool); the runner appends the op's own arguments after them — the same
62 * shape stock XrdOucProg uses, where Run()'s arguments are appended to the
63 * configured command line, so an operator can disambiguate a program shared
64 * between ops by naming the op as a literal argument on the line.
65 */
66typedef struct {
67 char **argv; /* NULL-terminated fixed prefix (cf->pool) */
68 ngx_uint_t argc; /* its token count (argv[argc] == NULL) */
69 ngx_str_t display; /* the whole command line, for log and error text */
70} brix_cms_fsxeq_prog_t;
