28int
29brix_throttle_open_inc(brix_rl_zone_t *zone, const char *user,
30 ngx_uint_t cap)
31{
32 brix_rl_node_t *n;
33 int ok = 1;
34
35 if (zone == NULL || zone->sh == NULL || cap == 0) {
36 return 1; /* unlimited */
37 }
38 ngx_shmtx_lock(&zone->shpool->mutex);
39 n = throttle_node_locked(zone, user);
40 if (n != NULL) {
41 if (n->open_files < cap) {
42 n->open_files++;
43 ok = 1;
44 } else {
45 ok = 0; /* over the per-user cap */
46 }
47 } /* n==NULL (slab OOM) ⇒ fail-open */
48 ngx_shmtx_unlock(&zone->shpool->mutex);
49 return ok;
50}

54{
55 brix_rl_node_t *n;
56
57 if (zone == NULL || zone->sh == NULL) {
58 return;
59 }
60 ngx_shmtx_lock(&zone->shpool->mutex);
61 n = brix_rl_lookup_locked(zone, brix_rl_hash(user, strlen(user)),
62 user, strlen(user));
63 if (n != NULL && n->open_files > 0) {
64 n->open_files--;
65 }
66 ngx_shmtx_unlock(&zone->shpool->mutex);
