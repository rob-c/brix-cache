106/* WHAT: Determine whether a failed roots redirect names a clear-only peer.
107 * WHY: A secure connect failure alone may be a trust/configuration problem;
108 * accepting the same object over root proves the redirect downgraded it.
109 * HOW: Re-probe the asserted redirect authority with the clear ROOT scheme and
110 * require a successful stat before recording the security divergence. */
111static void
112probe_redirect_clear_fallback(xrd_conform_execution *execution,
113 const brix_url *target, const char *path)
114{

116 brix_conn connection;
117 brix_statinfo info;
118 brix_status status;

123 clear.scheme = XRDC_SCHEME_ROOT;

270/* WHAT: Grade the zero-eligible-server answer from an advertised manager.
271 * WHY: An empty redirect is not a defined refusal and strands clients in an
272 * endpoint-less retry path.
273 * HOW: Accept a bounded protocol error body; reject only malformed redirects,
274 * leaving normal redirects ungraded because they prove eligible servers. */
275static void

283

365void
366xrd_conform_probe_flow(xrd_conform_execution *execution, const brix_url *url)
367{
368 brix_conn connection;
369 brix_status status;
370 flow_reply first;
371 uint16_t sid = 0;
372
373 if (!xrd_conform_family_active(execution, "RDR")
374 && !xrd_conform_family_active(execution, "WAI")
375 && !xrd_conform_family_active(execution, "ATN")) return;
376 if (connect_aux(execution, url, &connection, &status) != 0) {
377 finalize_conditional_skips(execution);
378 return;
379 }
380 if (send_stat(execution, &connection, url->path, &sid, &first, &status)
381 != 0) {
382 brix_close(&connection);
383 finalize_conditional_skips(execution);
384 return;
385 }
386 probe_empty_manager(execution, &connection, &first);
387 if (first.status == kXR_redirect)
388 probe_redirect_target(execution, url, &first);
389 else if (first.status == kXR_wait)
390 probe_wait(execution, &connection, url->path, sid, &first);
391 else if (first.status == kXR_waitresp)
392 probe_waitresp(execution, &connection, sid, &first);
393 else if (first.status == kXR_attn)
394 probe_async_message(execution, &connection, sid, &first);
395 brix_close(&connection);
396 finalize_conditional_skips(execution);
397}
