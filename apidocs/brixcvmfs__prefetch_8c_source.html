1/*
2 * brixcvmfs_prefetch.c — CVMFS-brix predictive subtree prefetch (Phase-38 split).
3 *
4 * WHAT: the opt-in F4 readahead worker — the first readdir of a directory
5 * queues it, a background thread walks the catalog subtree and pre-pulls
6 * every referenced CAS object into the shared cache (G2 .cvmfs-bundle
7 * batching when armed), so later opens are cache hits.
8 * WHY: split from brixcvmfs.c to keep each TU within the file-size budget; the
9 * worker owns its own failover/scratch/CAS-store state and is a pure
10 * best-effort accelerator — its errors are always swallowed and can never
11 * fail a foreground FUSE op.
12 * HOW: the FUSE thread enqueues (pf_enqueue); the detached worker (pf_main)
13 * borrows a libcurl handle from the shared pool per fetch (exclusive
14 * checkout, so it never races the foreground loop) and is bounded by a
15 * byte budget. Single-writer packed caches share the mount's store.
16 */

29/* ---- predictive prefetch (phase-85 F4) ---------------------------------- *
30 * Opt-in subtree readahead: the first readdir of a directory queues it for a
31 * background worker that walks the catalog subtree (cvmfs_walk_subtree) and
32 * pre-pulls every referenced CAS object into the shared cache, so subsequent
33 * opens are cache hits. The worker owns its OWN failover state, scratch and
34 * CAS-store handle (puts are O_EXCL+rename atomic, so sharing the cache
35 * DIRECTORY with the foreground is safe); its libcurl handle is borrowed from
36 * the shared g_curl_pool per fetch (exclusive checkout, so it never races the
37 * FUSE loop's handle). Bounded by a byte budget;
38 * prefetch errors are always swallowed — they can never fail a foreground op. */
39
40#define BRIX_PF_QCAP 32
41#define BRIX_PF_SEENCAP 512

80static uint64_t pf_fnv1a(const char *s) {
81 uint64_t h = 1469598103934665603ull;
82 for (; *s; s++) { h ^= (unsigned char) *s; h *= 1099511628211ull; }
83 return h;
84}

90/* Flush the batched walk items: one .cvmfs-bundle POST for the not-yet-cached
91 * members, ingest the reply (per-member CAS verify — fetch_bundle.c), then
92 * pull EVERY batched object through cvmfs_fetch_object as usual. Bundle-stored
93 * members resolve as verified cache hits; misses, rejects and any bundle
94 * failure at all fall back to the ordinary single fetches — the bundle is an
95 * RTT optimization, never a second trust or error path. */
96/* Build the want-list for the current batch: every member not already in the
97 * cache, as "data/<object-path>" lines in g_pf.want. Returns the list length. */
98static size_t pf_bundle_want(cvmfs_fetch_ctx_t *fx) {
99 size_t wn = 0;
100
101 for (int i = 0; i < g_pf.nbatch; i++) {
102 char key[160], obj[160];
103 int need;
104 if (cvmfs_hash_to_hex(&g_pf.batch[i].hash, g_pf.batch[i].suffix,
105 key, sizeof(key)) < 0
106 || brix_cas_has(fx->cache, key)
107 || cvmfs_hash_to_object_path(&g_pf.batch[i].hash,
108 g_pf.batch[i].suffix,
109 obj, sizeof(obj)) < 0)
110 continue;
111 need = snprintf(g_pf.want + wn, sizeof(g_pf.want) - wn, "data/%s\n", obj);
112 if (need < 0 || wn + (size_t) need >= sizeof(g_pf.want)) break;
113 wn += (size_t) need;
114 }
115
116 return wn;
117}

119static void pf_bundle_flush(pf_walk_ud_t *p) {
120 cvmfs_fetch_ctx_t *fx = p->fx;
121 size_t wn = pf_bundle_want(fx);
122 int i;
123
124 if (wn > 0) {
125 if (g_pf.resp == NULL) g_pf.resp = malloc(BRIX_PF_RESPCAP);
126 cvmfs_fo_route_t route;
127 if (g_pf.resp != NULL
128 && cvmfs_failover_select(fx->fo, mono_now(), &route) == 0) {
129 const char *proxy = route.proxy >= 0
130 ? fx->fo->proxies[route.proxy].url : NULL;
131 size_t rn = 0;
132 if (bundle_http_post(proxy, fx->fo->hosts[route.host].url,
133 g_pf.want, wn,
134 g_pf.resp, BRIX_PF_RESPCAP, &rn) == 0) {
135 unsigned stored = 0, fb = 0;
136 (void) cvmfs_bundle_ingest(fx, g_pf.resp, rn, &stored, &fb);
137 }
138 }
139 }
140
141 /* This loop drives cvmfs_fetch_object directly, so it owns the landing-buffer
142 * sizing the read path does for itself. The batch does not carry per-object
143 * sizes, so reserve once for the largest object the sweep will accept —
144 * prefetch is advisory, and a reserve failure just leaves the batch unfetched. */
145 {
146 cvmfs_client_t *pf_cl = brixcvmfs_client();
147 if (pf_cl == NULL
148 || cvmfs_client_scratch_reserve(pf_cl, BRIX_PF_OBJCAP) != 0) {
149 g_pf.nbatch = 0;
150 return;
151 }
152 }
153
154 for (i = 0; i < g_pf.nbatch; i++) {
155 size_t n = 0;
156 if (g_pf.budget > 0 && g_pf.spent >= g_pf.budget)
157 break; /* pf_visit fires the one prefetchcap audit line */
158 if (cvmfs_fetch_object(fx, &g_pf.batch[i].hash, g_pf.batch[i].suffix,
159 p->out, BRIX_PF_OBJCAP, &n, mono_now()) == 0)
160 g_pf.spent += (long) n;
161 else
162 g_pf.fo = g_pf.fo0; /* same reset as the unbatched path */
163 }
164 g_pf.nbatch = 0;
165}

167static int pf_visit(const cvmfs_walk_item_t *it, void *ud) {
168 pf_walk_ud_t *p = ud;
169 if (it->kind == CVMFS_WALK_CATALOG) return 0; /* the walk itself cached it */
170 if (g_pf.budget > 0 && g_pf.spent >= g_pf.budget) {
171 if (!g_pf.capped) {
172 g_pf.capped = 1;
173 fprintf(stderr, "brixcvmfs: audit signal=prefetchcap repo=%s "
174 "budget=%ld spent=%ld (prefetch stopped, foreground unaffected)\n",
175 g_cl->config.name, g_pf.budget, g_pf.spent);
176 }
177 return 1;
178 }
179 if (g_pf.bundle) { /* G2: batch up; flush POSTs + pulls in bulk */
180 g_pf.batch[g_pf.nbatch].hash = it->hash;
181 g_pf.batch[g_pf.nbatch].suffix = it->suffix;
182 if (++g_pf.nbatch == (int) CVMFS_BUNDLE_MAX_ITEMS)
183 pf_bundle_flush(p);
184 return 0;
185 }
186 size_t n = 0;
187 if (cvmfs_fetch_object(p->fx, &it->hash, it->suffix,
188 p->out, BRIX_PF_OBJCAP, &n, mono_now()) == 0)
189 g_pf.spent += (long) n;
190 else
191 g_pf.fo = g_pf.fo0; /* one bad object blacklists its route — restore the
192 * pristine snapshot so it can't shadow the sweep */
193 return 0; /* fetch errors never stop the sweep */
194}

196static void *pf_main(void *arg) {
197 (void) arg;
198 brix_cas_store_t own;
199 brix_cas_store_t *cache;
200 int crc = 0;
201 if (g_cl->cache.pack != NULL) {
202 /* Packed backend (G4) is a single-writer log: a second handle on the
203 * same dir would interleave appends and corrupt it. Share the mount's
204 * store — every pack op is serialized by its internal mutex. */
205 cache = &g_cl->cache;
206 } else {
207 crc = g_pf.cache_dirfd >= 0
208 ? brix_cas_init_at(&own, g_pf.cache_dirfd, g_pf.quota)
209 : brix_cas_init(&own, g_pf.cache_dir, g_pf.quota);
210 cache = &own;
211 }
212 unsigned char *scratch = malloc(BRIX_PF_OBJCAP);
213 unsigned char *out = malloc(BRIX_PF_OBJCAP);
214 if (crc != 0 || scratch == NULL || out == NULL) {
215 free(scratch); free(out);
216 return NULL; /* prefetch unavailable, mount unaffected */
217 }
218
219 cvmfs_fetch_ctx_t fx;
220 memset(&fx, 0, sizeof(fx));
221 fx.fo = &g_pf.fo;
222 fx.cache = cache;
223 fx.transport = brixcvmfs_transport;
224 fx.transport_ud = NULL; /* handle borrowed from g_curl_pool per fetch */
225 fx.store_form = CVMFS_STORE_COMPRESSED;
226 fx.scratch = scratch;
227 fx.scratch_cap = BRIX_PF_OBJCAP;
228
229 for (;;) {
230 pthread_mutex_lock(&g_pf.mu);
231 while (g_pf.qn == 0) pthread_cond_wait(&g_pf.cv, &g_pf.mu);
232 char path[1024];
233 cvmfs_hash_t root = g_pf.q[g_pf.qh].root;
234 snprintf(path, sizeof(path), "%s", g_pf.q[g_pf.qh].path);
235 g_pf.qh = (g_pf.qh + 1) % BRIX_PF_QCAP;
236 g_pf.qn--;
237 pthread_mutex_unlock(&g_pf.mu);
238
239 if (g_pf.capped) continue; /* drain silently once the budget is hit */
240 pf_walk_ud_t ud = { &fx, out };
241 cvmfs_walk_subtree(&fx, &root, g_pf.tmp_dir, path, g_pf.depth,
242 pf_visit, &ud, mono_now());
243 if (g_pf.bundle && g_pf.nbatch > 0)
244 pf_bundle_flush(&ud); /* tail batch of this subtree */
245 }
246 return NULL;
247}

249/* FUSE-thread side: queue `path` once. Drops silently when the queue is full
250 * or the path was already prefetched — advisory readahead, never a failure. */
251void pf_enqueue(const char *path) {
252 if (!g_pf.enabled) return;
253 uint64_t h = pf_fnv1a(path);
254 for (int i = 0; i < g_pf.nseen; i++)
255 if (g_pf.seen[i] == h) return;
256 if (g_pf.nseen < BRIX_PF_SEENCAP) g_pf.seen[g_pf.nseen++] = h;
257
258 pthread_mutex_lock(&g_pf.mu);
259 if (g_pf.qn < BRIX_PF_QCAP) {
260 int slot = (g_pf.qh + g_pf.qn) % BRIX_PF_QCAP;
261 snprintf(g_pf.q[slot].path, sizeof(g_pf.q[slot].path), "%s", path);
262 g_pf.q[slot].root = g_cl->pin_set ? g_cl->pin_root
263 : g_cl->manifest.root_catalog;
264 g_pf.qn++;
265 pthread_cond_signal(&g_pf.cv);
266 }
267 pthread_mutex_unlock(&g_pf.mu);
268}

270/* Called once between mount and fuse_main (still single-threaded): snapshot the
271 * failover set and start the worker. Failure to start just disables prefetch. */
272void pf_start(int depth, long budget, const char *tmp_dir,
273 const char *cache_dir, int cache_dirfd, long quota,
274 int bundle) {
275 g_pf.bundle = bundle;
276 g_pf.depth = depth;
277 g_pf.budget = budget;
278 snprintf(g_pf.tmp_dir, sizeof(g_pf.tmp_dir), "%s", tmp_dir);
279 snprintf(g_pf.cache_dir, sizeof(g_pf.cache_dir), "%s", cache_dir);
280 g_pf.cache_dirfd = cache_dirfd >= 0 ? dup(cache_dirfd) : -1;
281 g_pf.quota = quota;
282 g_pf.fo = g_pf.fo0 = g_cl->fo; /* snapshot; worker state evolves alone */
283 pthread_mutex_init(&g_pf.mu, NULL);
284 pthread_cond_init(&g_pf.cv, NULL);
285 if (pthread_create(&g_pf.tid, NULL, pf_main, NULL) == 0) {
286 pthread_detach(g_pf.tid);
287 g_pf.enabled = 1;
288 }
