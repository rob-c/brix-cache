19/* brix_serve_memory_backed — stream [start, start+len) of `fh` to the client by
20 * reading through the storage driver into pool buffers and pushing them through
21 * nginx's output filter. Used when the backend exposes no single sendfile fd —
22 * an object/block backend (e.g. pblock) whose bytes span multiple block files,
23 * which brix_http_send_file_range (a single in_file buffer) cannot serve. The
24 * bytes still flow proto -> VFS -> driver, just memory-backed instead of
25 * zero-copy. The driver preads run on the event loop; a thread-pool/AIO streaming
26 * variant for very large objects is a follow-up. Returns the output-filter rc or
27 * NGX_ERROR. The caller still owns closing `fh`. */
28#define BRIX_SERVE_MEM_CHUNK (256 * 1024)
29
30/* Read-ahead for the memory-backed loop: after each chunk, hint the next
31 * WINDOW bytes off the read cursor via the driver's read_advise slot
32 * (WILLNEED) — on a slice partial-cache handle that queues a background block
33 * fill ahead of the foreground preads. The driver keeps a per-handle rolling
34 * frontier, so the repeated per-chunk hints cost one bitmap peek and
35 * speculation stays a continuous runway bounded by ITS prefetch window (and
36 * the object size) — the hint is deliberately NOT clamped to the request
37 * range, so a client walking a file in sequential range GETs gets successor
38 * blocks warmed beyond each request. Ignored entirely unless
39 * brix_cache_prefetch is on. Requests below ADV_MIN never speculate — a small
40 * metadata GET must not amplify origin traffic. The first hint fires only
41 * after the first chunk completes, so the block the foreground is filling
42 * right now is present and gets skipped, never fetched twice. */
43#define BRIX_SERVE_MEM_ADV_WINDOW (8 * 1024 * 1024)
44#define BRIX_SERVE_MEM_ADV_MIN (1024 * 1024)

62/*
63 * brix_serve_mem_prime — the first driver pread, taken before any header.
64 *
65 * WHAT: allocate one chunk and read it at `start`, reporting the driver's own
66 * count and errno to the caller.
67 * WHY: separated so the pre-header read and the in-loop reads stay one
68 * statement each in the streaming body, and so the allocation failure is
69 * distinguishable from a backend refusal (ENOMEM vs the driver's errno).
70 * HOW: pool-allocate `want`, hand the buffer back through `out`, return the
71 * pread result verbatim (-1 with errno set, 0 for an origin that delivered
72 * nothing, else the byte count).
73 */
74static ssize_t
75brix_serve_mem_prime(ngx_http_request_t *r, brix_vfs_file_t *fh, off_t start,
76 size_t want, u_char **out)
77{
78 u_char *buf = ngx_palloc(r->pool, want);
79
80 if (buf == NULL) {
81 errno = ENOMEM;
82 return -1;
83 }
84 *out = buf;
85
86 return brix_vfs_file_pread(fh, buf, want, start);
87}

90static size_t
91brix_serve_mem_want(off_t len, off_t done)
92{
93 off_t want = len - done;

99/*
100 * brix_serve_mem_status — terminal HTTP status for a driver read that failed
101 * before any byte reached the wire.
102 *
103 * WHAT: map the errno left by brix_vfs_file_pread to the status the client
104 * gets instead of a bare connection close.
105 * WHY: the shared errno table owns the security-load-bearing classes
106 * (ENOENT 404, EACCES/EPERM/EXDEV/ELOOP 403) and must keep owning them here
107 * — a driver read must never turn a confinement rejection into a 5xx. Its
108 * "no idea" answer is 500, but this path is reached only for a backend with
109 * no single sendfile fd, i.e. an object/remote origin reached through the
110 * driver, so an unclassified delivery failure is a gateway failure (502),
111 * matching every other origin-failure site in the tree.
112 * HOW: run the shared table, promote its 500 default to 502, leave the rest.
113 */
114static ngx_int_t
115brix_serve_mem_status(int err)
116{
117 ngx_int_t status = brix_http_map_errno(err);
118
119 return (status == NGX_HTTP_INTERNAL_SERVER_ERROR) ? NGX_HTTP_BAD_GATEWAY
120 : status;
121}

123/* Push one already-read chunk through the output filter. */
124static ngx_int_t
125brix_serve_mem_emit(ngx_http_request_t *r, u_char *buf, ssize_t n, int last)
126{
127 ngx_buf_t *b = ngx_calloc_buf(r->pool);
128 ngx_chain_t out;
129
130 if (b == NULL) {
131 return NGX_ERROR;
132 }
133 b->pos = buf;
134 b->last = buf + n;
135 b->memory = 1;
136 b->flush = 1;
137 b->last_buf = last;
138 out.buf = b;
139 out.next = NULL;
140
141 return ngx_http_output_filter(r, &out);
142}

144/* One chunk already read but not yet pushed through the output filter — how

151/*
152 * brix_serve_mem_stream — push [start, start+len) through the output filter.
153 *
154 * WHAT: emit `first` (the chunk the caller already read), then read and emit
155 * the remainder chunk by chunk, hinting the read-ahead window after each.
156 * WHY: split from brix_serve_memory_backed so the pre-header contract (which
157 * owns the only failure that still has a status line available) and the
158 * on-the-wire streaming (where NGX_ERROR is the only report left) are two
159 * readable bodies instead of one over-branched one.
160 * HOW: `first->buf` is consumed on the opening iteration and nulled; every
161 * later iteration allocates and preads its own. A failure here is always
162 * post-header: NGX_ERROR for a driver error, NGX_OK for a short EOF, the
163 * filter's own rc otherwise.
164 */
165static ngx_int_t
166brix_serve_mem_stream(ngx_http_request_t *r, brix_vfs_file_t *fh, off_t start,
167 off_t len, brix_serve_mem_chunk_t *first)
168{
169 off_t done = 0;
170
171 while (done < len) {
172 size_t want = brix_serve_mem_want(len, done);
173 ngx_int_t rc;
174
175 if (first->buf == NULL) {
176 first->buf = ngx_palloc(r->pool, want);
177 if (first->buf == NULL) {
178 return NGX_ERROR;
179 }
180 first->n = brix_vfs_file_pread(fh, first->buf, want, start + done);
181 if (first->n <= 0) {
182 /* error / short EOF — the header is already gone, so the
183 * only report available is the connection itself. */
184 return first->n < 0 ? NGX_ERROR : NGX_OK;
185 }
186 }
187
188 rc = brix_serve_mem_emit(r, first->buf, first->n,
189 (done + first->n >= len) ? 1 : 0);
190 if (rc == NGX_ERROR || rc > NGX_HTTP_SPECIAL_RESPONSE) {
191 return rc;
192 }
193 done += first->n;
194 first->buf = NULL;
195
196 if (len >= BRIX_SERVE_MEM_ADV_MIN) {
197 (void) brix_vfs_file_read_advise(fh, start + done,
198 BRIX_SERVE_MEM_ADV_WINDOW,
199 BRIX_SD_ADV_WILLNEED);
200 }
201 }
202
203 return NGX_OK;
204}

206ngx_int_t
207brix_serve_memory_backed(ngx_http_request_t *r, brix_vfs_file_t *fh,
208 off_t start, off_t len)
209{
210 brix_serve_mem_chunk_t first = { NULL, 0 };
211 ngx_int_t hrc;
212
213 /* The FIRST driver pread runs BEFORE the header is sent. For every
214 * backend that lands here the bytes come from an object/remote origin,
215 * and that first read is the first moment the origin is asked for them —
216 * so it is where an origin refusal surfaces. Sending the header first
217 * left no way to report one: nginx had a 200 buffered and unflushed, the
218 * NGX_ERROR return closed the connection, and the client saw an empty
219 * reply with no status code at all. Read first and the failure still has
220 * a status line to travel in. Once a byte HAS been flushed the old
221 * behaviour is the only one available and is kept in the stream helper. */
222 if (!r->header_only && len > 0) {
223 errno = 0;
224 first.n = brix_serve_mem_prime(r, fh, start,
225 brix_serve_mem_want(len, 0),
226 &first.buf);
227 if (first.n <= 0) {
228 /* n == 0 is an origin that answered the read with no bytes at
229 * all; it is a delivery failure, not an EOF (the range was
230 * clamped to the statted size before we got here). */
231 return brix_serve_mem_status(first.n < 0 ? errno : EIO);
232 }
233 }
234
235 /* The response headers were set (set_file_headers) but not yet sent — the
236 * sendfile path sends them inside send_file_range; do the same here. */
237 hrc = ngx_http_send_header(r);
238 if (hrc == NGX_ERROR || hrc > NGX_OK) {
239 return hrc;
240 }
241
242 if (r->header_only || len <= 0) {
243 return brix_serve_send_empty_body(r);
244 }
245
246 return brix_serve_mem_stream(r, fh, start, len, &first);
247}
