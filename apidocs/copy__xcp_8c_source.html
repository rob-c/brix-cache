1/*
2 * copy_xcp.c — extreme copy: multi-source block-stealing download (phase-100).
3 *
4 * WHAT: copy_download_xcp() downloads one known-size file from up to
5 * --sources N replicas concurrently: the file is cut into blocks, each
6 * worker owns one connection to one replica and claims blocks from a
7 * shared table; when no unclaimed block remains, idle workers STEAL
8 * blocks still in flight on slower sources (bounded duplicate fetch,
9 * first-writer-wins on identical bytes), so the tail finishes at the
10 * speed of the fastest replicas.
11 * WHY: The upstream XrdCl XCpCtx behavior (xrdcp --sources): aggregate
12 * bandwidth across replicas and survive a dead/slow mirror mid-transfer.
13 * The phase-94 striped --parallel path fans one server across bound
14 * streams; this engine fans DISTINCT servers, with dynamic instead of
15 * static range assignment.
16 * HOW: Replica URLs come from copy_xcp_sources.c (metalink mirrors → locate
17 * → duplication). A bounded join gate holds the claim start until every
18 * source's open resolves, so each opened source gets its first claim.
19 * Per-block atomic state bytes coordinate claim/steal; disjoint pwrites
20 * into the shared VFS temp reassemble by offset (io_uring OFF, the
21 * phase-94 thread-safety pattern). Fail-closed: losing every worker
22 * aborts the temp; losing some is survived by claim-return + stealing.
23 */

30/* ---- Record the first worker failure (for the final verdict) ----
31 *
32 * WHAT: First caller stores its status into sh->first_err; later callers are
33 * ignored.

35 * WHY: The transfer's failure message should name the FIRST fault, not

51/* ---- Claim the next block for a worker ----
52 *
53 * WHAT: Return a block index to fetch and set *stole (0 = fresh claim,
54 * 1 = stealing a BUSY block), or -1 when nothing is claimable right now
55 * (all DONE, or every BUSY block already has its one stealer).
56 *
57 * WHY: Dynamic assignment IS the extreme-copy scheduler: fast sources come
58 * back sooner and naturally take more blocks; the steal arm keeps the
59 * transfer from ending at the slowest source's pace.
60 *
61 * HOW: 1. One rotating-scan pass CASing XCP_TODO→XCP_BUSY (the per-worker
62 * start hint spreads workers across the table so they do not contend
63 * block 0).
64 * 2. A second pass latches the first un-stolen XCP_BUSY block. At most
65 * one stealer per block bounds duplicate fetches to 2x.
66 */
67static ssize_t
68xcp_claim(xcp_shared_t *sh, unsigned hint, int *stole)
69{
70 size_t i;
71
72 for (i = 0; i < sh->nblocks; i++) {
73 size_t idx = (hint + i) % sh->nblocks;
74 unsigned char expected = XCP_TODO;
75 if (atomic_compare_exchange_strong(&sh->state[idx], &expected,
76 XCP_BUSY)) {
77 *stole = 0;
78 return (ssize_t) idx;
79 }
80 }
81 for (i = 0; i < sh->nblocks; i++) {
82 unsigned char expected = 0;
83 if (atomic_load(&sh->state[i]) == XCP_BUSY
84 && atomic_compare_exchange_strong(&sh->stealer[i], &expected, 1)) {
85 /* Re-check: the owner may have finished between the two loads. */
86 if (atomic_load(&sh->state[i]) == XCP_BUSY) {
87 atomic_fetch_add(&sh->steals, 1);
88 *stole = 1;
89 return (ssize_t) i;
90 }
91 atomic_store(&sh->stealer[i], 0);
92 }
93 }
94 return -1;
95}

113/* ---- Fetch one block from this worker's replica into the dest temp ----
114 *
115 * WHAT: Read [idx*block, min(size, (idx+1)*block)) from the open handle and
116 * pwrite it at its absolute offset; mark the block DONE (counting its
117 * bytes exactly once, whichever racer gets there first). 0 / -1.
118 *
119 * WHY: The per-block unit of work both the claim and steal arms share.
120 *
121 * HOW: Short-read loop; between reads, bail out early on operator cancel or
122 * when the racing fetcher already finished the block (a stolen block's
123 * owner, or a stealer racing this owner).
124 */
125static int
126xcp_fetch_block(xcp_worker_t *w, brix_conn *c, brix_file *f, size_t idx,
127 uint8_t *buf)
128{
129 xcp_shared_t *sh = w->sh;
130 int64_t start = (int64_t) idx * (int64_t) sh->block;
131 int64_t end = start + (int64_t) sh->block;
132 int64_t off = start;
133
134 if (end > sh->size) {
135 end = sh->size;
136 }
137 while (off < end) {
138 ssize_t n;
139
140 if (brix_copy_quit_requested()) {
141 brix_status_set(&w->st, XRDC_ESOCK, EINTR, "cancelled (signal)");
142 return -1;
143 }
144 if (atomic_load(&sh->state[idx]) == XCP_DONE) {
145 return 0; /* the racing fetcher finished this block first */
146 }
147 n = brix_file_read(c, f, off, buf + (off - start),
148 (size_t) (end - off), &w->st);
149 if (n <= 0) {
150 if (n == 0) {
151 brix_status_set(&w->st, XRDC_EPROTO, 0,
152 "short read at %lld from %s",
153 (long long) off, w->url);
154 }
155 return -1;
156 }
157 off += n;
158 }
159 if (brix_vfs_pwrite(sh->vf, start, buf, (size_t) (end - start),
160 &w->st) != 0) {
161 return -1;
162 }
163 if (atomic_exchange(&sh->state[idx], XCP_DONE) != XCP_DONE) {
164 atomic_fetch_add(&sh->done_bytes,
165 (unsigned long long) (end - start));
166 w->blocks_done++;
167 }
168 return 0;
169}

172/* ---- Undo a claim this worker cannot finish ----
173 *
174 * WHAT: Return a fresh-claimed block to XCP_TODO, or release the stealer latch
175 * of a stolen one (unless the block completed meanwhile).
176 *
177 * WHY: A dying worker must never strand a block in XCP_BUSY — the survivors'
178 * claim scan only picks up XCP_TODO entries.
179 *
180 * HOW: CAS XCP_BUSY→XCP_TODO for claims (an XCP_DONE result means the stealer
181 * won — leave it); plain latch clear for steals.
182 */
183static void
184xcp_release_block(xcp_shared_t *sh, size_t idx, int stole)
185{
186 if (stole) {
187 atomic_store(&sh->stealer[idx], 0);
188 return;
189 }
190 {
191 unsigned char expected = XCP_BUSY;
192 atomic_compare_exchange_strong(&sh->state[idx], &expected, XCP_TODO);
193 }
194}

200/* ---- Worker thread: one replica connection driving the block loop ----
201 *
202 * WHAT: Connect + open this worker's replica URL, then claim/steal/fetch until
203 * every block is DONE, an unrecoverable local fault hits, or the replica
204 * dies (w->rc = -1, block returned, engine survives on the others).
205 *
206 * WHY: One thread per replica keeps each connection's request pipeline full —
207 * the whole point of multi-source — while the atomics above make worker
208 * death safe at any instant.
209 *
210 * HOW: 1. Parse/connect/open (any failure kills only this worker), then count
211 * this source RESOLVED either way and hold at the join gate so every
212 * opened sibling gets its first claim. 2. Loop: claim else steal else
213 * (blocks remain? 20 ms idle wait : exit). 3. Fetch errors release the
214 * block and kill the worker. 4. Teardown always closes handle +
215 * connection and drops the live count.
216 */
217static void *
218xcp_worker_main(void *arg)
219{
220 xcp_worker_t *w = (xcp_worker_t *) arg;
221 xcp_shared_t *sh = w->sh;
222 brix_url u;
223 brix_conn c;
224 brix_file f;
225 uint8_t *buf = NULL;
226 int opened = 0, connected = 0;
227
228 w->rc = -1;
229 if (brix_url_parse(w->url, &u, &w->st) == 0
230 && brix_connect(&c, &u, sh->co, &w->st) == 0) {
231 connected = 1;
232 if (brix_file_open_read(&c, u.path, &f, &w->st) == 0) {
233 opened = 1;
234 buf = (uint8_t *) malloc(sh->block);
235 if (buf == NULL) {
236 brix_status_set(&w->st, XRDC_EPROTO, 0, "out of memory");
237 }
238 }
239 }
240
241 atomic_fetch_add(&sh->resolved, 1);
242
243 if (buf != NULL) {
244 /* Hold until every sibling's open resolves (join gate), then run the
245 * reservation-aware claim/steal loop so no fast source drains the block
246 * table before a slower mirror gets its first claim. */
247 xcp_join_gate(sh);
248 w->rc = xcp_worker_blocks(w, &c, &f, buf);
249 }
250
251 if (w->initial_reserved) {
252 xcp_release_reserved_block(sh, w->initial_idx);
253 w->initial_reserved = 0;
254 }
255
256 if (w->rc != 0) {
257 xcp_note_error(sh, &w->st);
258 }
259 free(buf);
260 if (opened) {
261 brix_status throwaway;
262 brix_status_clear(&throwaway);
263 brix_file_close(&c, &f, &throwaway);
264 }
265 if (connected) {
266 brix_close(&c);
267 }
268 atomic_fetch_sub(&sh->live, 1);
269 return NULL;
270}

273/* ---- Spawn the workers, report progress, join, and give the verdict ----
274 *
275 * WHAT: Run n workers to completion. 0 when every block is DONE, else -1 with
276 * *st carrying the first worker failure.
277 *
278 * WHY: Keeping the thread lifecycle + the single-threaded progress feed in one
279 * helper leaves the orchestrator a flat setup/commit sequence. The
280 * coordinator (not the workers) invokes o->progress, preserving the
281 * serial pump's "progress callback runs on one thread" contract.
282 *
283 * HOW: 1. Spawn each worker (a spawn failure burns that slot's live count and
284 * counts it resolved so the join gate never waits for it — the engine
285 * needs only >= 1 running). 2. Poll live/donebytes at 100 ms, feeding
286 * progress. 3. Join, final progress tick, verdict.
287 */
288static int
289xcp_run_workers(xcp_shared_t *sh, xcp_worker_t *w, pthread_t *th, size_t n,
290 const brix_copy_opts *o)
291{
292 size_t k;
293
294 sh->nworkers = (unsigned) n;
295 atomic_store(&sh->live, (unsigned) n);
296 for (k = 0; k < n; k++) {
297 if (k < sh->nblocks) {
298 w[k].initial_idx = k;
299 w[k].initial_reserved = 1;
300 atomic_store(&sh->state[k], XCP_RESERVED);
301 }
302 if (pthread_create(&th[k], NULL, xcp_worker_main, &w[k]) != 0) {
303 th[k] = 0;
304 if (w[k].initial_reserved) {
305 xcp_release_reserved_block(sh, w[k].initial_idx);
306 w[k].initial_reserved = 0;
307 }
308 atomic_fetch_sub(&sh->live, 1);
309 atomic_fetch_add(&sh->resolved, 1);
310 }
311 }
312 while (atomic_load(&sh->live) > 0) {
313 struct timespec ts = { 0, 100 * 1000 * 1000 };
314 nanosleep(&ts, NULL);
315 if (o->progress != NULL) {
316 o->progress(o->progress_arg,
317 (long long) atomic_load(&sh->done_bytes),
318 (long long) sh->size);
319 }
320 }
321 for (k = 0; k < n; k++) {
322 if (th[k] != 0) {
323 pthread_join(th[k], NULL);
324 }
325 }
326 if (o->progress != NULL) {
327 o->progress(o->progress_arg,
328 (long long) atomic_load(&sh->done_bytes),
329 (long long) sh->size);
330 }
331 return xcp_all_done(sh) ? 0 : -1;
332}

335/* ---- Eligibility: may this download run the extreme-copy engine? ----
336 *
337 * WHAT: 1 only for an explicit --sources >= 2 download to a real local file of
338 * known size >= 2 blocks, with no per-request payload transforms.
339 *
340 * WHY: pgread framing and inline decompression are per-connection negotiated
341 * states the block engine does not carry; tiny/unknown sizes cannot be
342 * cut into blocks worth two connections.
343 *
344 * HOW: Cheap predicate over the job; the caller falls back to the serial (or
345 * --parallel) path on 0.
346 */
347static int
348xcp_eligible(const download_job_t *job, size_t block)
349{
350 const brix_copy_opts *o = job->o;
351
352 return o->sources >= 2
353 && job->du->scheme != XRDC_SCHEME_STDIO
354 && !o->pgrw
355 && !(o->compress != NULL && o->compress[0] != '\0')
356 && job->si->size >= 2 * (int64_t) block;
357}

395/* ---- Run the multi-source block-stealing download end-to-end ----
396 *
397 * WHAT: Returns 1 when it handled the transfer (verdict in *out_rc), or 0 when
398 * the request is not eligible and the caller should fall through to the
399 * --parallel / serial paths. Same contract as copy_download_parallel.
400 *
401 * WHY: The audit's §7.2 extreme-copy gap: replica-parallel downloads with
402 * block stealing, fed by metalink mirrors or locate discovery.
403 *
404 * HOW: 1. Eligibility gate. 2. Build the replica list (>= 2 workers). 3. Open
405 * the dest via the VFS (atomic temp+rename, io_uring OFF so pwrite(2)
406 * is thread-safe for the disjoint blocks). 4. Allocate block tables +
407 * run the workers. 5. Debug line, then the shared commit/abort +
408 * checksum-reconcile helper gives the final verdict.
409 */
410int
411copy_download_xcp(const download_job_t *job, int *out_rc, brix_status *st)
412{
413 const brix_copy_opts *o = job->o;
414 size_t block = xcp_block_size();
415 size_t want, n, k;
416 xcp_shared_t sh;
417 xcp_worker_t *w;
418 pthread_t *th;
419 brix_vfs_open_opts vopts = {0};
420 int rc = -1;
421
422 if (!xcp_eligible(job, block)) {
423 return 0;
424 }
425 want = (size_t) o->sources;
426 if (want > XRDC_XCP_MAX_SRCS) {
427 want = XRDC_XCP_MAX_SRCS;
428 }
429
430 memset(&sh, 0, sizeof(sh));
431 sh.size = job->si->size;
432 sh.block = block;
433 sh.nblocks = (size_t) ((sh.size + (int64_t) block - 1) / (int64_t) block);
434 sh.co = job->co;
435
436 w = (xcp_worker_t *) calloc(want, sizeof(*w));
437 th = (pthread_t *) calloc(want, sizeof(*th));
438 sh.state = (atomic_uchar *) calloc(sh.nblocks, sizeof(atomic_uchar));
439 sh.stealer = (atomic_uchar *) calloc(sh.nblocks, sizeof(atomic_uchar));
440 if (w == NULL || th == NULL || sh.state == NULL || sh.stealer == NULL) {
441 brix_status_set(st, XRDC_EPROTO, 0, "out of memory");
442 free(w); free(th); free(sh.state); free(sh.stealer);
443 *out_rc = -1;
444 return 1;
445 }
446
447 for (k = 0; k < want; k++) {
448 w[k].sh = &sh;
449 /* Spread the initial claim scans across the table. */
450 w[k].claim_hint = (unsigned) ((sh.nblocks / want) * k);
451 brix_status_clear(&w[k].st);
452 }
453 n = xcp_build_sources(job, w, want);
454 if (n < 2) {
455 free(w); free(th); free(sh.state); free(sh.stealer);
456 return 0; /* could not even duplicate: serial path handles it */
457 }
458
459 vopts.io_uring = XRDC_IO_URING_OFF;
460 vopts.expected_size = sh.size;
461 if (brix_vfs_open(job->du->path,
462 XRDC_VFS_WRITE | (o->force ? XRDC_VFS_FORCE : 0),
463 &vopts, &sh.vf, st) != 0) {
464 free(w); free(th); free(sh.state); free(sh.stealer);
465 *out_rc = -1;
466 return 1;
467 }
468
469 rc = xcp_run_workers(&sh, w, th, n, o);
470 if (rc != 0) {
471 if (atomic_load(&sh.have_err)) {
472 *st = sh.first_err;
473 } else {
474 brix_status_set(st, XRDC_ESOCK, 0, "xcp: no worker completed");
475 }
476 }
477 xcp_debug_line(&sh, w, n);
478
479 rc = download_commit_or_abort(job, sh.vf, rc, st);
480
481 free(w); free(th); free(sh.state); free(sh.stealer);
482 *out_rc = rc;
483 return 1;
484}
