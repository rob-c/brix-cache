21static int
22large_offset_io(xrd_conform_execution *execution, brix_conn *connection,
23 brix_file *file)
24{
25 uint8_t expected[CONFORM_L_PROBE];
26 uint8_t actual[CONFORM_L_PROBE];
27 brix_status status;
28 char reason[64];
29 unsigned partials;
30 ssize_t got;
31 int exact;
32
33 brix_conform_fill(&execution->run->key, (uint64_t) CONFORM_L_OFFSET,
34 expected, sizeof(expected));
35 if (!xrd_conform_reserve(execution, sizeof(expected), reason,
36 sizeof(reason))) return -1;
37 brix_status_clear(&status);
38 if (brix_file_write(connection, file, CONFORM_L_OFFSET, expected,
39 sizeof(expected), &status) != 0) return -1;
40 if (!xrd_conform_reserve(execution, sizeof(actual), reason,
41 sizeof(reason))) return -1;
42 brix_status_clear(&status);
43 got = brix_file_read(connection, file, CONFORM_L_OFFSET, actual,
44 sizeof(actual), &status);
45 exact = got == (ssize_t) sizeof(actual)
46 && memcmp(actual, expected, sizeof(actual)) == 0;
47 xrd_conform_result_set(execution, "RD-011",
48 exact ? BRIX_CONFORM_VERDICT_CONFORMANT
49 : BRIX_CONFORM_VERDICT_DIVERGENT,
50 exact ? "large-offset-bytes-position-correct"
51 : "large-offset-read-mismatch", NULL, status.msg);
52
53 if (!xrd_conform_reserve(execution, sizeof(actual), reason,
54 sizeof(reason))) return -1;
55 partials = 0;
56 brix_status_clear(&status);
57 got = brix_file_pgread_profile(connection, file, CONFORM_L_OFFSET, actual,
58 sizeof(actual), &partials, &status);
59 exact = got == (ssize_t) sizeof(actual)
60 && memcmp(actual, expected, sizeof(actual)) == 0;
61 xrd_conform_result_set(execution, "PGR-028",
62 exact ? BRIX_CONFORM_VERDICT_CONFORMANT
63 : BRIX_CONFORM_VERDICT_DIVERGENT,
64 exact ? "large-offset-page-crcs-position-correct"
65 : "large-offset-pgread-mismatch", NULL, status.msg);
66 return exact ? 1 : 0;
67}

103 brix_status_clear(&status);

108static int
109open_write_only(brix_conn *connection,
110 const xrd_conform_scratch_object *object, brix_file *file,
111 brix_status *status)
112{
113 ClientOpenRequest request;
114 xrdw_open_req_t fields = {
115 .mode = 0644,
116 .options = kXR_open_wrto | kXR_delete | kXR_mkpath,
117 .optiont = 0
118 };
119 brix_payload payload = { object->path, (uint32_t) strlen(object->path) };
120 uint16_t wire_status = 0;
121 uint8_t *body = NULL;
122 uint32_t body_length = 0;
123 brix_resp_out response = { &wire_status, &body, &body_length };
124 int rc;
125
126 memset(&request, 0, sizeof(request));
127 request.requestid = htons(kXR_open);
128 xrdw_open_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
129 rc = brix_roundtrip(connection, &request, &payload, &response, status);
130 if (rc == 0 && body_length >= sizeof(file->fhandle)) {
131 memcpy(file->fhandle, body, sizeof(file->fhandle));
132 file->read_codec = 0;
133 file->write_codec = 0;
134 } else if (rc == 0) {
135 brix_status_set(status, XRDC_EPROTO, 0,
136 "write-only open reply too short");
137 rc = -1;
138 }
139 free(body);
140 return rc;
141}

197static int
198probe_standard_object(xrd_conform_execution *execution,
199 brix_conn *connection, xrd_conform_scratch_object *object, uint64_t size)
200{
201 brix_file file;
202 brix_status status;
203 uint8_t *baseline = NULL;
204 int opened = 0;
205 int rc = -1;
206
207 brix_status_clear(&status);
208 if (xrd_conform_plant_object(execution, connection, object, size,
209 &status) == 0
210 && brix_file_open_read(connection, object->path, &file, &status) == 0) {
211 opened = 1;
212 baseline = xrd_conform_read_baseline(execution, connection, &file,
213 size, &status);
214 if (baseline != NULL) {
215 if (size == 0)
216 xrd_conform_grade_read(execution, connection, &file, baseline,
217 size, "RD-010", 0, 1);
218 xrd_conform_probe_pgread_standard(execution, connection, &file,
219 baseline, size);
220 if (size == XRDC_CONFORM_CHUNKED_SIZE
221 && (xrd_conform_family_active(execution, "RD")
222 || xrd_conform_family_active(execution, "PGR")
223 || xrd_conform_family_active(execution, "CHU")))
224 xrd_conform_probe_chunked_reads(execution, connection, &file,
225 baseline, (size_t) size);
226 if (size == XRDC_CONFORM_CHUNKED_SIZE
227 && xrd_conform_family_active(execution, "RDV")) {
228 xrd_conform_probe_readv(execution, connection, &file,
229 object->path, baseline, size);
230 xrd_conform_probe_readv_chunked(execution, connection, &file,
231 baseline, (size_t) size);
232 }
233 rc = 0;
234 }
235 }
236 free(baseline);
237 if (opened) {
238 brix_status close_status;
239 brix_status_clear(&close_status);
240 if (brix_file_close(connection, &file, &close_status) != 0) rc = -1;
241 }
242 brix_status_clear(&status);
243 if (brix_rm(connection, object->path, &status) != 0) rc = -1;
244 return rc;
245}

251xrd_conform_probe_standard_corpus(xrd_conform_execution *execution,
252 brix_conn *connection)
253{
254 static const struct { const char *leaf; uint64_t size; } objects[] = {
255 { "brix-conform-Z", 0 }, { "brix-conform-P0", 4095 },
256 { "brix-conform-C-plus", XRDC_CONFORM_CHUNKED_SIZE }
257 };
258 size_t index;
259
260 for (index = 0; index < sizeof(objects) / sizeof(objects[0]); index++) {

264 objects[index].leaf, object.path, sizeof(object.path)) != 0
265 || probe_standard_object(execution, connection, &object,
266 objects[index].size) != 0) return -1;
267 }
268 return 0;
