6/*
7 * pgreads.c — thread-pool offload for the stream kXR_pgread opcode.
8 *
9 * WHAT: The pgread half of the AIO read family; reads.c keeps the windowed
10 * memory read and plain kXR_read. Output is the interleaved wire form
11 * [CRC32C(4)][page data(4096)] × N behind a ServerStatusResponse_pgRead
12 * header (invariant #1), which brix_build_chunked_chain cannot produce —
13 * hence a dedicated chain builder rather than the kXR_read path.
14 *
15 * WHY: Split out of reads.c when that file crossed the 600-line cap (see
16 * coding-standards §1). The seam is the natural one: pgread is the only
17 * read path that owns a checksum stage and its own framing, and it shares
18 * no file-local state with the plain-read half.
19 *
20 * HOW: Same two-half discipline as reads.c — a *_thread function runs on a
21 * pool worker and may touch nothing but the brix_pgread_aio_t task
22 * struct; *_done runs back on the event loop, re-validates the stream
23 * through brix_aio_restore_request, owns all protocol/state mutation,
24 * and ends in brix_aio_resume(). The P44-B hybrid io_uring path adds a
25 * third entry point, brix_pgread_aio_crc_thread, for the case where the
26 * ring already delivered the bytes and only the CRC pass is left.
27 */

30/*
31 * brix_pgread_aio_thread — thread-pool worker for kXR_pgread.
32 *
33 * Reads file data DIRECTLY into the final interleaved [CRC32C(4)][data] wire
34 * buffer (t->scratch, starting at offset 0) and computes each page CRC32C in
35 * place — no separate flat-data copy pass. This runs on the worker thread so
36 * both the (batched preadv) I/O and the CRC stay off the nginx event loop.
37 * t->out_size is the encoded byte count; t->nread the file bytes read (<0 = I/O
38 * error, t->io_errno set). See brix_pgread_read_encode_inplace().
39 */
40void
41brix_pgread_aio_thread(void *data, ngx_log_t *log)
42{
43 brix_pgread_aio_t *t = data;
44 brix_vfs_job_t job;
45
46 brix_vfs_job_read_init(&job, t->fd, t->offset, t->rlen,
47 t->scratch, t->rlen, 0);
48 job.op = BRIX_VFS_IO_PGREAD;
49 brix_vfs_job_set_obj(&job, &t->obj); /* Layer 3: route via driver if bound */
50 brix_vfs_io_execute(&job);
51
52 t->out_size = job.out_size;
53 t->nread = job.nio;
54 t->io_errno = job.io_errno;
55}

140 brix_aio_resume(c);

142 }

149 if (rconf->access_log_fd != NGX_INVALID_FILE) {
