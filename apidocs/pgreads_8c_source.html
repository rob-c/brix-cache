8/*
9 * pgreads.c — thread-pool offload for the stream kXR_pgread opcode.
10 *
11 * WHAT: The pgread half of the AIO read family; reads.c keeps the windowed
12 * memory read and plain kXR_read. Output is the interleaved wire form
13 * [CRC32C(4)][page data(4096)] × N behind a ServerStatusResponse_pgRead
14 * header (invariant #1), which brix_build_chunked_chain cannot produce —
15 * hence a dedicated chain builder rather than the kXR_read path.
16 *
17 * WHY: Split out of reads.c when that file crossed the 600-line cap (see
18 * coding-standards §1). The seam is the natural one: pgread is the only
19 * read path that owns a checksum stage and its own framing, and it shares
20 * no file-local state with the plain-read half.
21 *
22 * HOW: Same two-half discipline as reads.c — a *_thread function runs on a
23 * pool worker and may touch nothing but the brix_pgread_aio_t task
24 * struct; *_done runs back on the event loop, re-validates the stream
25 * through brix_aio_restore_stream, owns all protocol/state mutation,
26 * and ends in brix_aio_resume(). The P44-B hybrid io_uring path adds a
27 * third entry point, brix_pgread_aio_crc_thread, for the case where the
28 * ring already delivered the bytes and only the CRC pass is left.
29 */

32/*
33 * brix_pgread_aio_thread — thread-pool worker for kXR_pgread.
34 *
35 * Reads file data DIRECTLY into the final interleaved [CRC32C(4)][data] wire
36 * buffer (t->scratch, starting at offset 0) and computes each page CRC32C in
37 * place — no separate flat-data copy pass. This runs on the worker thread so
38 * both the (batched preadv) I/O and the CRC stay off the nginx event loop.
39 * t->out_size is the encoded byte count; t->nread the file bytes read (<0 = I/O
40 * error, t->io_errno set). See brix_pgread_read_encode_inplace().
41 */
42void
43brix_pgread_aio_thread(void *data, ngx_log_t *log)
44{
45 brix_pgread_aio_t *t = data;
46 brix_vfs_job_t job;
47
48 if (t->sec_c != NULL && t->pool_send && brix_pgread_pool_stream(t)) {
49 return; /* §1.3: streamed (or error-shaped) entirely in-thread */
50 }
51
52 brix_vfs_job_read_init(&job, t->fd, t->offset, t->rlen,
53 t->scratch, t->rlen, 0);
54 job.op = BRIX_VFS_IO_PGREAD;
55 brix_vfs_job_set_obj(&job, &t->obj); /* Layer 3: route via driver if bound */
56 brix_vfs_io_execute(&job);
57
58 t->out_size = job.out_size;
59 t->nread = job.nio;
60 t->io_errno = job.io_errno;
61
62 if (t->sec_c != NULL && t->pool_send) {
63 brix_pgread_pool_send(t); /* §1.2: this thread sends the frame */
64 }
65}

142/* §1.2 pool-send wire disposition — MUST run before every early return in
143 * the offload epilogue: the worker thread may have put bytes (or a held send
144 * token) on the SECONDARY, and both belong to it regardless of the primary's
145 * fate. Returns 1 when the worker owned the wire (bytes sent, tail parked,
146 * or the buffer already retired on a send error). */
147static ngx_flag_t
148brix_pgread_pool_disposition(brix_pgread_aio_t *t, u_char *fbuf)
149{
150 brix_ctx_t *sec_ctx = t->sec_ctx;
151 ngx_connection_t *sec_c = t->sec_c;
152
153 if (!t->pool_send
154 || !(t->pool_sent_all || t->pool_token_held || t->pool_send_errno
155 || t->chunk_error))
156 {
157 return 0;
158 }
159
160 if (t->pool_sent > 0) {
161 sec_c->sent += (off_t) t->pool_sent;
162 BRIX_SRV_METRIC_ADD(sec_ctx, wire_bytes_tx_total, t->pool_sent);
163 }
164
165 if (t->pool_sent_all) {
166 BRIX_SRV_METRIC_ADD(sec_ctx, response_frames_total, t->pool_frames);
167 brix_release_read_buffer(sec_ctx, sec_c, fbuf);
168 if (sec_ctx->out.count > 0 && !sec_ctx->destroyed) {
169 /* frames parked while the worker held the token */
170 (void) brix_ensure_write_event(sec_c);
171 }
172
173 } else if (t->pool_token_held) {
174 if (sec_ctx->destroyed) {
175 brix_release_read_buffer(sec_ctx, sec_c, fbuf);
176 } else {
177 BRIX_SRV_METRIC_ADD(sec_ctx, response_frames_total,
178 t->pool_frames);
179 (void) brix_park_front_buf(sec_ctx, sec_c,
180 fbuf + t->pool_sent,
181 t->pool_image_len - t->pool_sent,
182 fbuf);
183 }
184 sec_ctx->out.send_token = 0; /* park first, release second */
185
186 } else {
187 /* hard send error mid-image, or a chunk_error whose committed
188 * frames were all sent: either way the buffer is done */
189 brix_release_read_buffer(sec_ctx, sec_c, fbuf);
190 }
191
192 return 1;
193}

210 * it consumed the completion (error frame queued, recv loop resumed). */
211static ngx_flag_t

213{

217 brix_release_read_buffer(t->sec_ctx, t->sec_c, fbuf);
218 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
219 brix_send_error(ctx, t->c, kXR_IOError,
220 t->io_errno ? strerror(t->io_errno)

223 return 1;
224 }
225
226 if (t->pool_send_errno != 0 && !t->pool_sent_all) {

235 }
236

239 * the SECONDARY — the terminating kXR_error must ride the same
240 * channel under the request sid (any parked tail drains first: the
241 * ring keeps wire order, and queue_response parks behind it). */

245 t->io_errno ? strerror(t->io_errno) : "async pgread error");
246 brix_aio_resume(t->c);
247 return 1;
248 }
249
250 return 0;
251}

253/*
254 * brix_pgread_aio_done_offload — completion epilogue for a pgread whose reply
255 * rides a bound SECONDARY channel (t->sec_c != NULL, §1.1 offload-AIO).
256 *
257 * WHAT: Drops both connections' aio_inflight counts, then routes by liveness:
258 * both alive — stamp the pgRead status header at the head of the
259 * secondary's slot buffer and queue the contiguous [status|pages] frame
260 * on the SECONDARY's out-ring (release decided by the out.count delta,
261 * exactly as in brix_pgread_aio_done); primary dead — return the slot
262 * and fire the primary's deferred teardown if this was its last task;
263 * secondary dead — answer with kXR_error on the PRIMARY control stream
264 * (correlation is by streamid, so this is protocol-legal), return the
265 * slot, and fire the secondary's deferred teardown if last.
266 *
267 * WHY: The buffer belongs to the secondary and the request/recv state to the
268 * primary; each teardown was deferred on its own rd.aio_inflight, so
269 * both sides' memory is guaranteed live here and each release/teardown
270 * must land on its owner.
271 */
272static void
273brix_pgread_aio_done_offload(brix_pgread_aio_t *t)
274{
275 brix_ctx_t *ctx = t->ctx; /* primary: request owner */
276 ngx_connection_t *c = t->c;
277 brix_ctx_t *sec_ctx = t->sec_ctx; /* secondary: buffer owner */
278 ngx_connection_t *sec_c = t->sec_c;
279 u_char *fbuf;
280 ngx_uint_t out_before;
281 ngx_flag_t primary_alive;
282 ngx_flag_t pool_handled;
283
284 fbuf = t->scratch - sizeof(ServerStatusResponse_pgRead);
285
286 if (t->sec_counted && sec_ctx->rd.aio_inflight > 0) {
287 sec_ctx->rd.aio_inflight--;
288 }
289
290 pool_handled = brix_pgread_pool_disposition(t, fbuf);
291
292 primary_alive = brix_aio_restore_stream(ctx, t->streamid);
293
294 if (!primary_alive) {
295 if (!pool_handled) {
296 brix_release_read_buffer(sec_ctx, sec_c, fbuf);
297 }
298 /* both sides died with this task in flight: the secondary's deferred
299 * teardown must fire too, or its session leaks */
300 brix_pgread_offload_reap(t->sec_counted && sec_ctx->destroyed,
301 sec_ctx, sec_c);
302 brix_pgread_offload_reap(t->counted, ctx, c); /* frees primary ctx */
303 return;
304 }
305
306 if (ctx->state == XRD_ST_AIO) {
307 ctx->state = XRD_ST_REQ_HEADER;
308 ctx->recv.hdr_pos = 0;
309 }
310
311 if (sec_ctx->destroyed) {
312 /* Data channel lost mid-read: unless the worker already put the whole
313 * frame on the wire (§1.2), the reply cannot ride it and its slot
314 * buffer dies with it — error the request on the control stream. */
315 if (!pool_handled) {
316 brix_release_read_buffer(sec_ctx, sec_c, fbuf);
317 }
318 brix_pgread_offload_reap(t->sec_counted, sec_ctx, sec_c);
319 if (!t->pool_sent_all) {
320 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
321 brix_send_error(ctx, c, kXR_IOError, "pgread data channel lost");
322 brix_aio_resume(c);
323 return;
324 }
325 /* the frame reached the socket before the channel died — success */
326 }
327
328 brix_aio_metric_done(t->start_ns, BRIX_METRIC_OP_READ);
329
330 if (brix_pgread_offload_error_done(t, fbuf)) {
331 return;
332 }
333
334 /* EOF/short reads need no special casing: a zero out_size queues the bare
335 * 32-byte status frame (dlen 0), which the client reads as end-of-data. */
336 brix_pgread_done_account(t);
337
338 if (pool_handled) {
339 /* §1.2: the worker thread already sent (or parked) the frame — only
340 * the observability tick and the recv-loop resume remain. */
341 brix_metric_offload(BRIX_PROTO_ROOT);
342 brix_aio_resume(c);
343 return;
344 }
345
346 brix_build_pgread_status(ctx, t->offset, (uint32_t) t->out_size,
347 (ServerStatusResponse_pgRead *) fbuf);
348 sec_ctx->out.resp_pipelinable = 1; /* self-contained frame in its own slot */
349
350 out_before = sec_ctx->out.count;
351 (void) brix_queue_response_base(sec_ctx, sec_c, fbuf,
352 sizeof(ServerStatusResponse_pgRead)
353 + t->out_size, fbuf);
354 if (sec_ctx->out.count == out_before) {
355 brix_release_read_buffer(sec_ctx, sec_c, fbuf);
356 }
357 brix_metric_offload(BRIX_PROTO_ROOT); /* §1.1 observability */
358 brix_aio_resume(c);
359}

361/*
362 * brix_pgread_aio_done — response builder for pgread AIO completion.
363 *
364 * pgread wire format ([CRC32C(4)][page data] × N) cannot use
365 * brix_build_chunked_chain and requires direct chain construction with a
366 * ServerStatusResponse_pgRead header.
367 */
368void
369brix_pgread_aio_done(ngx_event_t *ev)
370{
371 ngx_thread_task_t *task = ev->data;
372 brix_pgread_aio_t *t = task->ctx;
373 brix_ctx_t *ctx = t->ctx;
374 ngx_connection_t *c = t->c;
375 ngx_chain_t *rsp_chain;
376 ngx_uint_t out_before;
377
378 /* §1.1 offload-AIO: a secondary-targeted reply has its own epilogue —
379 * it must drop BOTH connections' aio counts and land each release and
380 * deferred teardown on its owner. The primary decrement below still
381 * applies to it, so branch before touching any counter. */
382 if (t->sec_c != NULL) {
383 if (t->counted && ctx->rd.aio_inflight > 0) {
384 ctx->rd.aio_inflight--;
385 }
386 brix_pgread_aio_done_offload(t);
387 return;
388 }
389
390 /*
391 * Pipelined-pgread parity with brix_read_aio_done: this counted task is no
392 * longer in flight. Decrement BEFORE the liveness guard so a deferred
393 * teardown (the client vanished while pipelined pgreads were running)
394 * fires on the LAST completion — once no worker thread references any
395 * rd_pool buffer.
396 */
397 if (t->counted && ctx->rd.aio_inflight > 0) {
398 ctx->rd.aio_inflight--;
399 }
400
401 if (!brix_aio_restore_stream(ctx, t->streamid)) {
402 /* frees ctx when this was its last pending task — return now */
403 brix_pgread_offload_reap(t->counted, ctx, c);
404 return;
405 }
406
407 /*
408 * Re-arm the request state only when recv is actually suspended on this
409 * completion (XRD_ST_AIO). Under pipelining the recv loop kept receiving
410 * after the post — it may be mid-header or mid-payload of the NEXT
411 * request, or parked in XRD_ST_SENDING behind an earlier response — and
412 * blindly resetting state/hdr_pos would corrupt that receive.
413 */
414 if (ctx->state == XRD_ST_AIO) {
415 ctx->state = XRD_ST_REQ_HEADER;
416 ctx->recv.hdr_pos = 0;
417 }
418
419 /* phase-56 D-2: pgread files as a READ op on the latency histogram. */
420 brix_aio_metric_done(t->start_ns, BRIX_METRIC_OP_READ);
421
422 if (t->nread < 0) {
423 brix_release_read_buffer(ctx, c, t->scratch);
424 BRIX_OP_ERR(ctx, BRIX_OP_PGREAD);
425 brix_send_error(ctx, c, kXR_IOError,
426 t->io_errno ? strerror(t->io_errno) : "async pgread error");
427 brix_aio_resume(c);
428 return;
429 }
430
431 if (t->nread == 0 || t->out_size == 0) {
432 brix_pgread_done_eof(t);
433 return;
434 }
435
436 /* PGREAD: the encoded page data (in t->scratch from offset 0) carries its
437 * own per-page CRC32c and must be sent verbatim behind the pgRead status
438 * header — never through brix_build_chunked_chain (wrong kXR_ok framing).
439 * Shared with the synchronous handler via brix_build_pgread_chain. */
440 rsp_chain = brix_build_pgread_chain(ctx, c, t->offset, t->scratch,
441 (uint32_t) t->out_size);
442 if (rsp_chain == NULL) {
443 brix_release_read_buffer(ctx, c, t->scratch);
444 brix_aio_resume(c);
445 return;
446 }
447
448 brix_pgread_done_account(t);
449
450 /* Self-contained frame (per-response palloc'd header, data in this
451 * request's own rd_pool slot): if it parks, the next pgread may safely
452 * queue behind it while it drains (brix_recv_try_pipeline_read). */
453 ctx->out.resp_pipelinable = 1;
454
455 /*
456 * Release ownership is decided by whether the out-ring took the chain
457 * (out.count delta), NEVER by ctx->state: the recv loop parks itself in
458 * XRD_ST_SENDING at the admission bound even with out.count == 0 (all
459 * depth slots held by in-flight AIO), and under that stale SENDING a
460 * fully-inline send would skip BOTH releases here and at drain — leaking
461 * the rd_pool slot and (via the depth bound admitting one request too
462 * many) killing the whole pipelined connection on acquire failure.
463 */
464 out_before = ctx->out.count;
465 brix_queue_response_chain(ctx, c, rsp_chain, t->scratch);
466 if (ctx->out.count == out_before) {
467 brix_release_read_buffer(ctx, c, t->scratch);
468 }
469 brix_aio_resume(c);
470}
