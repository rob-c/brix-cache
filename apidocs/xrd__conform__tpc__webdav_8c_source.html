55static void
56run_webdav_copy(xrd_conform_execution *execution, const brix_weburl *source,
57 const brix_weburl *destination, const char *token, const char *fields,

60{

63 if (request(execution, source, "PUT", token, NULL, native_bytes,
64 native_length, 4096, &response, status) == 0) {
65 *planted = successful_status(response.status);
66 brix_http_resp_free(&response);
67 }
68 if (*planted && request(execution, destination, "COPY", token, fields,
69 NULL, 0, 4096, &response, status) == 0) {
70 *copied = successful_status(response.status);
71 brix_http_resp_free(&response);
72 }
73 if (*copied && request(execution, destination, "GET", token, NULL,
74 NULL, 0, native_length, &response, status) == 0) {
75 *exact = response.status == 200 && response.framing_valid

77 && memcmp(response.body, native_bytes, native_length) == 0;

100/* WHAT: Compare a bearer-authenticated WebDAV pull with the native TPC result.
101 * WHY: Advertising both transfer planes is useful only when they land the same
102 * bytes and the HTTP path really authenticates the COPY request.
103 * HOW: Derive two scratch objects from explicit WebDAV endpoints, PUT the
104 * native oracle bytes, COPY with Source/Credential headers, then GET and
105 * compare independently before bounded best-effort cleanup. */
106void
107xrd_conform_probe_tpc_webdav(xrd_conform_execution *execution,
108 const uint8_t *native_bytes, size_t native_length, int native_exact)
109{
110 const char *kind = execution->run->tpc_fixture_kind;
111 brix_weburl source;
112 brix_weburl destination;
113 brix_status status;
114 char fields[XRDC_PATH_MAX + 384];
115 char *token;
116 int planted = 0;
117 int copied = 0;
118 int exact = 0;
119
120 if (kind == NULL || strcmp(kind, "native-and-webdav") != 0) return;
121 if (!prepare_webdav_pair(execution, &source, &destination, fields,
122 sizeof(fields))) {
123 xrd_conform_result_set(execution, "TPC-006",
124 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
125 "fixture-required:native-and-webdav-tpc", NULL, NULL);
126 return;
127 }
128 token = brix_token_discover();
129 if (token == NULL || !xrd_conform_credential_safe(token)) {
130 xrd_conform_result_set(execution, "TPC-006",
131 BRIX_CONFORM_VERDICT_INCONCLUSIVE,
132 "fixture-required:webdav-bearer", "safe discovered bearer", NULL);
133 free(token);
134 return;
135 }
136 xrd_conform_webdav_cleanup(execution, &destination, token);
137 xrd_conform_webdav_cleanup(execution, &source, token);
138 run_webdav_copy(execution, &source, &destination, token, fields,
139 native_bytes, native_length, &planted, &copied, &exact, &status);
140 grade_webdav_copy(execution, native_exact, planted, copied, exact,
141 &status);
142 xrd_conform_webdav_cleanup(execution, &destination, token);
143 xrd_conform_webdav_cleanup(execution, &source, token);
