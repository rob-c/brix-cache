53typedef struct {

63 /* Exclusively-VFS eviction: enumerate + remove through the cache STORAGE
64 * driver instead of raw POSIX. `inst` is the read-cache instance and

67 void *inst; /* brix_sd_instance_t* */

73 const char *state_root; /* POSIX sidecar root (==cache_root if co-located) */
74} brix_cache_evict_list_t;

86/*

88 *
89 * Bundles the state that stays fixed while brix_cache_evict_one() walks the
90 * candidate list — conf, optional per-connection metrics ctx / manager socket,
91 * log, the candidate list, the running FS usage, the cold-tier decorator, and the
92 * evicted counters. Shared by the ppm engine (evict_policy.c) and the owned-bytes
93 * purge (reap_watermark.c), so it lives here rather than in one .c.

101 brix_cache_fs_usage_t *usage;
102 brix_sd_instance_t *decorator; /* composed sd_cache (tier grammar)
103 * or NULL — phase-85 F7 victims

147/*
148 * Try to acquire the cross-worker eviction lock (an O_CREAT|O_EXCL sentinel
149 * file under conf->cache_root). On success writes the sentinel path into
150 * lock_path (caller-owned buffer of lock_pathsz bytes); the caller must later
151 * release it with brix_cache_evict_unlock(). A sentinel older than
152 * conf->cache_lock_timeout seconds is treated as stale and reclaimed.
153 * Returns NGX_OK (lock held), NGX_DECLINED (another worker holds a fresh lock
154 * or a race lost the reclaim — skip eviction), or NGX_ERROR (path too long /
155 * unexpected OS error; errno set).
156 */
157ngx_int_t brix_cache_try_evict_lock(ngx_stream_brix_srv_conf_t *conf,
158 char *lock_path, size_t lock_pathsz, ngx_log_t *log);
159
160/*
161 * Release the eviction sentinel lock by unlinking lock_path; no-op if
162 * lock_path is the empty string. Call exactly once per NGX_OK from
163 * brix_cache_try_evict_lock(); errors from unlink() are ignored.
164 */
165void brix_cache_evict_unlock(const char *lock_path);

167/*
168 * Recursively scan directory `dir`, appending every regular file as an
169 * eviction candidate to *list (allocating list->elts via malloc/realloc;
170 * each candidate->path is a freshly malloc'd copy owned by the list).
171 * Skips special names (. / .. / lock sentinel / *.part / *.lock) and entries
172 * whose st_dev differs from list->root_dev (other mounts).
173 * Returns NGX_OK, or NGX_ERROR if any entry failed (opendir/lstat/path-overflow
174 * or candidate allocation); partial results are still left in *list.
175 */
176ngx_int_t brix_cache_collect_dir(brix_cache_evict_list_t *list,
177 const char *dir, ngx_log_t *log);

179/*

188/*

197/*
198 * brix_cache_purge_to_target — the fill-task-free eviction engine shared by the
199 * on-fill safety net (brix_cache_evict_if_needed) and the background watermark
200 * reaper (brix_cache_watermark_purge). Collects candidates under cache_root,
201 * sorts oldest-first, and evicts (large-files pass then LRU pass) until occupancy
202 * ≤ target_ppm. `ctx` (per-conn metrics) and `c` (manager unregister socket) are
203 * OPTIONAL — pass NULL from the timer. The CALLER owns the cross-worker eviction
204 * lock + threshold pre-check. Writes evicted file/byte counts (out-pointers may
205 * be NULL); NGX_OK, or NGX_ERROR on a mid-purge usage re-measurement failure.
