9/*
10 * s3_chunk_finalize — post-decode steps that must run on the event loop:
11 * commit the staged object, set the ETag, verify the (trailer or default)
12 * checksum, store tags, and send 200. Shared by the synchronous fallback and
13 * the offload completion; `writer` has been written but not yet committed.
14 */
15void
16s3_chunk_finalize(ngx_http_request_t *r, const char *root_canon,
17 const char *fs_path, brix_vfs_writer_t *writer,
18 const s3_chunk_trailer_t *trailer, uint64_t expected, ngx_uint_t body_mode)
19{
20 if (s3_commit_put(r, r->connection->log, root_canon, writer,
21 fs_path) != NGX_OK)
22 {
23 if (s3_put_commit_conflict(r)) {
24 return;
25 }
26 brix_log_safe_path(r->connection->log, NGX_LOG_ERR, ngx_errno,
27 "s3: staged commit to \"%s\" failed", fs_path);
28 s3_put_finalize_error(r);
29 return;
30 }
31
32 {
33 brix_vfs_ctx_t fctx;
34 brix_vfs_stat_t fst;
35 char etag_buf[48];
36
37 brix_vfs_ctx_init(&fctx, r->pool, r->connection->log, BRIX_PROTO_S3,
38 root_canon, NULL, BRIX_VFS_MUTATION_READ_ONLY, 0 /* is_tls */, NULL, fs_path);
39 {
40 ngx_http_s3_loc_conf_t *acf =
41 ngx_http_get_module_loc_conf(r, ngx_http_brix_s3_module);
42 if (acf != NULL) {
43 /* Decorates a probe/stat-only ctx; Phase-2-forward — namespace
44 * scoping already routes the credential via the VFS ns gate. */
45 brix_vfs_ctx_bind_backend_cred(&fctx,
46 &acf->common.storage_credential_dir,
47 acf->common.storage_credential_fallback);
48 s3_vfs_bind_deleg(r, acf, &fctx);
49 }
50 }
51 if (brix_vfs_probe(&fctx, 1 /* no-follow */, &fst) == NGX_OK) {
52 struct stat sb;
53
54 ngx_memzero(&sb, sizeof(sb));
55 sb.st_mtime = fst.mtime;
56 sb.st_size = fst.size;
57 s3_etag(&sb, etag_buf, sizeof(etag_buf));
58 (void) s3_set_header(r, "ETag", etag_buf);
59 }
60 }
61
62 if (trailer->algo_token[0] != '\0') {
63 switch (s3_put_trailer_checksum_apply(r, fs_path, root_canon,
64 trailer->algo_token,
65 trailer->value))
66 {
67 case S3_CKSUM_MISMATCH:
68 s3_put_finalize_bad_digest(r);
69 return;
70 case S3_CKSUM_CONFLICT:
71 s3_put_finalize_invalid_request(r);
72 return;
73 default:
74 break;
75 }
76 } else if (s3_put_checksum_failed(r, fs_path, root_canon)) {
77 return;
78 }
79
80 (void) s3_apply_put_tagging_header(r, fs_path, root_canon);
81 (void) s3_apply_put_user_metadata(r, fs_path, root_canon);
82 s3_put_finalize_ok(r, expected, body_mode);
83}

86/* Decode-failure path: abort the write session and send the mapped S3 error. */
87void
88s3_chunk_decode_failed(ngx_http_request_t *r, const char *root_canon,
89 brix_vfs_writer_t *writer, int http_status)
90{
91 brix_vfs_writer_abort(writer);
92 if (http_status >= 500) {
93 s3_put_finalize_error(r);
94 } else if (http_status == NGX_HTTP_FORBIDDEN) {
95 /* W6a: a per-chunk signature mismatch — surface as SignatureDoesNotMatch. */
96 s3_put_finalize_client_error(r, NGX_HTTP_FORBIDDEN,
97 "SignatureDoesNotMatch",
98 "The chunk signature does not match the calculated signature.");
99 } else {
100 s3_put_finalize_client_error(r, NGX_HTTP_BAD_REQUEST, "InvalidRequest",
101 "The aws-chunked request body could not be decoded.");
102 }
103}

107 * phase-46 W1b: aws-chunked decode offload. The decode is a blocking
108 * pread/pwrite state machine over the (often spooled) body, so it runs on the
109 * thread pool; the completion commits/checksums/finalizes on the event loop.
110 * The worker uses the task's own `window` scratch — it never touches r->pool.

127/*
128 * s3_build_chunk_verify — W6a: assemble the per-chunk verification context from
129 * the location flag + the SigV4 material auth retained on the request ctx. Left
130 * disabled (zeroed) unless verification is configured AND a signed request was
131 * authenticated (an anonymous endpoint has no secret to verify against).
132 */
133void
134s3_build_chunk_verify(ngx_http_request_t *r, ngx_http_s3_loc_conf_t *cf,
135 s3_chunk_verify_t *v)
136{
137 ngx_http_s3_req_ctx_t *rx;
138
139 ngx_memzero(v, sizeof(*v));
140 if (!cf->verify_chunk_signatures) {
141 return;
142 }
143 rx = ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
144 if (rx == NULL || !rx->have_sigv4) {
145 return;
146 }
147 v->enabled = 1;
148 ngx_memcpy(v->signing_key, rx->sigv4_signing_key, 32);
149 ngx_cpystrn((u_char *) v->seed_signature,
150 (u_char *) rx->sigv4_seed_signature, sizeof(v->seed_signature));
151 ngx_cpystrn((u_char *) v->amz_date,
152 (u_char *) rx->sigv4_amz_date, sizeof(v->amz_date));
153 ngx_cpystrn((u_char *) v->scope,
154 (u_char *) rx->sigv4_scope, sizeof(v->scope));
155}

158void
159s3_chunk_aio_done(ngx_event_t *ev)
160{
161 ngx_thread_task_t *task = ev->data;
162 s3_chunk_aio_t *t = task->ctx;
163 ngx_http_request_t *r = t->r;
164 ngx_http_s3_req_ctx_t *rx =
165 ngx_http_get_module_ctx(r, ngx_http_brix_s3_module);
166
167 /* Balance the r->main->count++ at the task post (webdav_put_aio_done
168 * precedent) - without it the request stays one reference high and the
169 * connection never re-enters keepalive after the response. */
170 r->main->count--;
171
172 /* The finalize/decode-abort below are PATH-based (commit rename, unlink) and
173 * must run as the MAPPED user; this event-loop completion lost the per-request
174 * impersonation bracket, so re-establish it (the threaded write was fd-based).
175 * Mirrors the WebDAV/s3-put async fix. */
176 brix_imp_request_begin(rx != NULL ? rx->identity : NULL);
177 if (t->rc != NGX_OK) {
178 s3_chunk_decode_failed(r, t->root_canon, t->writer, t->http_status);
179 } else {
180 s3_chunk_finalize(r, t->root_canon, t->fs_path, t->writer, &t->trailer,
181 t->expected, t->body_mode);
182 }
183 brix_imp_request_end();
184}
