9/*
10 * kXR_set (3018) — client sends advisory configuration hints to the server.
11 *
12 * WHAT: Handles the SET opcode where a client provides optional metadata about
13 * its identity or session parameters. The modifier byte determines which
14 * hint type is being sent; unrecognized modifiers are accepted with kXR_ok.
15 *
16 * WHY: CMS grid infrastructure sends "cms.space" appid reports to advertise
17 * disk capacity at storage nodes. Session TTL hints (clttl) let clients
18 * suggest keep-alive duration. The server accepts all SET requests per spec,
19 * enabling monitoring and operational visibility without strict enforcement.
20 *
21 * HOW: Parse modifier byte from the request header → switch on known values
22 * (appid, clttl) → handle payload for recognized types → log details at
23 * debug/info level → always return kXR_ok regardless of modifier value.
24 */

26/*
27 * CMS space report constants.
28 *
29 * The "cms.space" appid payload format is: "cms.space <total_bytes> <free_bytes>"
30 * where both values are ASCII decimal unsigned integers. Used by the CMS grid
31 * to advertise storage node capacity in kXR_set(appid) requests.
32 */
33#define CMS_SPACE_PREFIX "cms.space"
34#define CMS_SPACE_PREFIX_LEN (sizeof(CMS_SPACE_PREFIX) - 1)

36/*
37 * brix_set_handle_cms_space — parse and log CMS space-availability report.
38 *
39 * WHAT: Extracts total and free byte counts from a "cms.space" payload string,
40 * then logs the capacity breakdown at INFO level for operational visibility.
41 *
42 * WHY: The CMS grid sends kXR_set(appid) with "cms.space <total> <free>" payloads
43 * to advertise storage node disk availability. This function parses those
44 * reports so operators can monitor capacity via nginx access logs.
45 *
46 * HOW: Skip the "cms.space" prefix, parse two ASCII decimal unsigned integers
47 * separated by a space using strtoull, validate separator presence between
48 * them, compute used = total - free, log at INFO level. Returns silently on
49 * malformed payload (warn logged).
50 */
51static void
52brix_set_handle_cms_space(ngx_connection_t *c, const char *payload,
53 size_t payload_len)
54{
55 /* Expected format: "cms.space <total_bytes> <free_bytes>" */
56 const char *p;
57 char *end;
58 unsigned long long total_bytes, free_bytes;
59
60 p = payload + CMS_SPACE_PREFIX_LEN;
61 while (p < payload + payload_len && *p == ' ') {
62 p++;
63 }
64
65 total_bytes = strtoull(p, &end, 10);
66 if (end == p || *end != ' ') {
67 ngx_log_error(NGX_LOG_WARN, c->log, 0,
68 "brix: kXR_set cms.space: malformed payload");
69 return;
70 }
71 p = end;
72 while (*p == ' ') {
73 p++;
74 }
75 free_bytes = strtoull(p, &end, 10);
76 if (end == p) {
77 ngx_log_error(NGX_LOG_WARN, c->log, 0,
78 "brix: kXR_set cms.space: missing free_bytes");
79 return;
80 }
81
82 ngx_log_error(NGX_LOG_INFO, c->log, 0,
83 "brix: cms.space report: total=%lluB free=%lluB "
84 "used=%lluB",
85 total_bytes, free_bytes,
86 (total_bytes > free_bytes) ? total_bytes - free_bytes : 0ULL);
87}

96/* ---- Operator cache-evict command (stock xrdfs "cache evict|fevict") ----
97 *
98 * WHAT: Handles a kXR_set payload of "cache evict <path>" / "cache fevict
99 * <path>": authorizes the caller, then removes the path's cached copy
100 * (data + cinfo + L1 entry) from the export's cache tier. Replies
101 * kXR_ok whether or not a copy was cached (the engine is idempotent),
102 * kXR_Unsupported when the export has no cache tier, kXR_NotAuthorized
103 * on a failed gate. Returns the response's queue result.
104 *
105 * WHY: Parity audit §4.11/§7.12 — the programmatic evict existed but no
106 * operator command reached it; stock ships `xrdfs cache evict`.
107 * Eviction destroys cached state, so it is gated like a delete
108 * (allow_write first — invariant 3 — then the BRIX_AUTH_DELETE
109 * authz/token-scope chain on the CONFINED path — invariant 4).
110 *
111 * HOW: 1. Extract and NUL-terminate the path operand; refuse empty or
112 * over-long operands and any ".." component outright.
113 * 2. Confine beneath root_canon + run the shared auth gate
114 * (need_write ⇒ allow_write is checked before token scope).
115 * 3. Build a transient VFS ctx; no cache decorator on the export ⇒
116 * kXR_Unsupported.
117 * 4. brix_sd_cache_evict on the export-relative key; log the freed
118 * bytes. evict and fevict act identically here — the engine has no
119 * in-use refusal — a documented divergence from stock's evict.
120 */
121static ngx_int_t
122brix_set_handle_cache_evict(brix_ctx_t *ctx, ngx_connection_t *c,
123 ngx_stream_brix_srv_conf_t *conf, const char *operand, size_t operand_len,
124 int force)
125{
126 brix_vfs_ctx_t vctx;
127 char reqpath[BRIX_MAX_PATH + 1];
128 char full_path[PATH_MAX];
129 char physical[PATH_MAX];
130 uint64_t freed_bytes;
131 const char *verb = force ? "fevict" : "evict";
132
133 while (operand_len > 0
134 && (operand[operand_len - 1] == '\0'
135 || operand[operand_len - 1] == '\n'
136 || operand[operand_len - 1] == ' '))
137 {
138 operand_len--;
139 }
140 if (operand_len == 0 || operand_len > BRIX_MAX_PATH) {
141 BRIX_OP_ERR(ctx, BRIX_OP_SET);
142 return brix_send_error(ctx, c, kXR_ArgInvalid,
143 "cache evict: bad path operand");
144 }
145 ngx_memcpy(reqpath, operand, operand_len);
146 reqpath[operand_len] = '\0';
147
148 if (brix_reject_dotdot_path(ctx, c, BRIX_OP_SET, "SET", reqpath)) {
149 return ctx->write_rc;
150 }
151
152 /* Invariant 3: the global write gate comes BEFORE any token-scope or
153 * authdb evaluation. kXR_set is normally advisory and routes through the
154 * session dispatcher (no write gate there), so the destructive cache
155 * command enforces it here itself. */
156 if (!conf->common.allow_write) {
157 brix_log_access(ctx, c, "SET", reqpath, verb, 0,
158 kXR_NotAuthorized, "read-only export", 0);
159 BRIX_OP_ERR(ctx, BRIX_OP_SET);
160 return brix_send_error(ctx, c, kXR_NotAuthorized,
161 "cache evict: server is read-only");
162 }
163
164 brix_beneath_full_path(conf->common.root_canon, reqpath,
165 full_path, sizeof(full_path));
166
167 if (brix_auth_gate(ctx, c, BRIX_OP_SET, "SET", reqpath, full_path,
168 conf, BRIX_AUTH_DELETE, 1 /* need_write */)
169 != NGX_OK)
170 {
171 return ctx->write_rc;
172 }
173
174 /* phase-105: the only mutation below is brix_sd_cache_evict — the SERVICE
175 * cache store, which Appendix H.1 places outside export policy (dropping a
176 * cached copy changes no exported object). The ctx is ALLOWED so that
177 * intent is stated at the construction site rather than inferred from the
178 * fact that eviction happens not to route through a VFS mutator. */
179 brix_vfs_ctx_init(&vctx, c->pool, c->log, BRIX_PROTO_ROOT,
180 conf->common.root_canon, NULL, BRIX_VFS_MUTATION_ALLOWED, 0 /* is_tls */,
181 ctx->identity, full_path);
182 brix_root_vfs_bind_session(ctx, conf, &vctx);
183
184 if (vctx.sd == NULL || !brix_sd_cache_instance_is(vctx.sd)) {
185 BRIX_OP_ERR(ctx, BRIX_OP_SET);
186 return brix_send_error(ctx, c, kXR_Unsupported,
187 "no cache tier on this export");
188 }
189
190 if (brix_path_resolved_to_pfn(&vctx, full_path, physical,
191 sizeof(physical)) != NGX_OK)
192 {
193 BRIX_OP_ERR(ctx, BRIX_OP_SET);
194 return brix_send_error(ctx, c, kXR_IOError,
195 "cache key translation failed");
196 }
197 freed_bytes = brix_sd_cache_evict(vctx.sd, physical);
198 ngx_log_error(NGX_LOG_INFO, c->log, 0,
199 "brix: cache %s \"%s\": %uL bytes evicted",
200 verb, reqpath, (uint64_t) freed_bytes);
201
202 BRIX_RETURN_OK(ctx, c, BRIX_OP_SET, "SET", reqpath, verb, 0);
203}

207 *
208 * WHAT: Dispatches the SET opcode based on its modifier byte, handling known
209 * hint types (appid with CMS space reports, clttl session TTL) and
210 * returning kXR_ok for all modifiers including unrecognized ones.
211 *
212 * WHY: The XRootD protocol allows clients to send advisory hints about their
213 * identity or session parameters. The server accepts all SET requests per
214 * spec — it does not enforce TTL or validate appid format strictly, but

220 * → default → accept with kXR_ok. Always return kXR_ok regardless of
221 * modifier value. Payload snippet logged at debug level for all types.

224 * ctx — xrootd connection context containing parsed request header and payload

228/* Copy up to cap-1 bytes of the SET payload into `snippet` for display, stripped
229 * of trailing newline/NUL. Empty snippet when the payload is empty. */
230static void
231set_build_snippet(const char *payload, size_t payload_len, char *snippet,
232 size_t cap)
233{
234 size_t snap;
235
236 snippet[0] = '\0';
237 if (payload_len == 0) {
238 return;
239 }
240 snap = payload_len < cap - 1 ? payload_len : cap - 1;
241 ngx_memcpy(snippet, payload, snap);
242 snippet[snap] = '\0';
243 while (snap > 0 && (snippet[snap - 1] == '\n' || snippet[snap - 1] == '\0')) {
244 snippet[--snap] = '\0';
245 }
246}

300 {

302 payload + CACHE_FEVICT_PREFIX_LEN,
303 payload_len
304 - CACHE_FEVICT_PREFIX_LEN, 1);
305 }
306
307 snprintf(detail, sizeof(detail), "modifier=0x%02x(%s) val=\"%s\"",
308 (unsigned) modifier, mod_name, payload_snippet);
309
310 ngx_log_debug1(NGX_LOG_DEBUG_STREAM, c->log, 0,
