6/* WHAT: Request a read on the primary and receive its body on a bound path.
7 * WHY: Merely issuing requests directly on BriX extension streams does not
8 * prove the protocol's pathid-directed response behavior.
9 * HOW: Put the assigned pathid in read_args, send on primary, recv on bound. */
10static int
11bound_read(xrd_conform_execution *execution, brix_conn *primary,
12 brix_conn *secondary, brix_file *file, size_t length,
13 uint8_t **body, uint32_t *body_length, brix_status *status)
14{
15 ClientReadRequest request;
16 xrdw_read_req_t fields = { .offset = 0, .rlen = (int32_t) length };
17 uint8_t args[BRIX_READ_ARGS_LEN] = { 0 };
18 brix_payload payload = { args, sizeof(args) };
19 uint16_t wire_status = 0;
20 uint16_t stream_id;
21 brix_resp_out response = { &wire_status, body, body_length };
22 char reason[64];
23
24 if (!xrd_conform_reserve(execution, length, reason, sizeof(reason))) {
25 brix_status_set(status, XRDC_EUSAGE, 0, "%s", reason);
26 return -1;
27 }
28 memcpy(fields.fhandle, file->fhandle, sizeof(fields.fhandle));
29 args[0] = secondary->bound_pathid;
30 memset(&request, 0, sizeof(request));
31 request.requestid = htons(kXR_read);
32 xrdw_read_req_pack(&fields, ((ClientRequestHdr *) &request)->body);
33 brix_status_clear(status);
34 if (brix_send(primary, &request, &payload, &stream_id, status) != 0)
35 return -1;
36 if (brix_recv(secondary, stream_id, &response, status) != 0) return -1;
37 return wire_status == kXR_ok ? 0 : -1;
38}

40/* WHAT: Round-trip one primary-stream ping after a secondary closes.
41 * WHY: The close and subsequent request arrive on different sockets; a ping
42 * gives the owning event loop a bounded synchronization turn before BND-005.
43 * HOW: Send the empty typed header and require an empty successful response. */
44static int
45primary_ping(xrd_conform_execution *execution, brix_conn *connection)
46{
47 ClientRequestHdr request;
48 uint16_t wire_status = 0;
49 uint16_t stream_id;
50 uint8_t *body = NULL;
51 uint32_t body_length = 0;
52 brix_resp_out response = { &wire_status, &body, &body_length };
53 brix_status status;
54 char reason[64];
55 int rc;
56
57 if (!xrd_conform_reserve(execution, 0, reason, sizeof(reason))) return -1;
58 memset(&request, 0, sizeof(request));
59 request.requestid = htons(kXR_ping);
60 xrdw_empty_req_pack(request.body);
61 brix_status_clear(&status);
62 rc = brix_send(connection, &request, NULL, &stream_id, &status);
63 if (rc == 0) rc = brix_recv(connection, stream_id, &response, &status);
64 free(body);
65 return rc == 0 && wire_status == kXR_ok && body_length == 0 ? 0 : -1;
66}

148/* WHAT: Bind exactly the advertised number of secondary streams plus one.
149 * WHY: BND-004 must observe the remote enforcement boundary; a config claim
150 * alone cannot establish that excess substreams are refused.
151 * HOW: Parse bind_max, keep every admitted stream live, and require the next
152 * bind to return the server's defined count-limit error. */
153static void
154probe_bind_limit(xrd_conform_execution *execution, brix_conn *connection)
155{
156 brix_streamset streams;
157 brix_conn overflow;
158 brix_status status;
159 char reply[64];
160 char reason[64];
161 unsigned long advertised;
162 int all_bound = 1;
163 int refused = 0;
164
165 memset(&streams, 0, sizeof(streams));
166 memset(&overflow, 0, sizeof(overflow));
167 brix_status_clear(&status);
168 all_bound = query_bind_max(execution, connection, &advertised, &status,
169 reply);
170 if (all_bound < 0) {
171 xrd_conform_result_set(execution, "BND-004",
172 BRIX_CONFORM_VERDICT_INCONCLUSIVE, "bind-limit-query-failed",
173 "advertised live streams then kXR_ArgTooLong", status.msg);
174 return;
175 }
176 if (all_bound == 0) {
177 xrd_conform_result_set(execution, "BND-004",
178 BRIX_CONFORM_VERDICT_INCONCLUSIVE, "bind-limit-not-probeable",
179 "bind_max in range 1..15", reply);
180 return;
181 }
182 all_bound = bind_advertised_streams(execution, connection, &streams,
183 advertised, &status);
184 if (all_bound
185 && xrd_conform_reserve(execution, 0, reason, sizeof(reason)))
186 {
187 brix_status_clear(&status);
188 refused = brix_bind(&overflow, connection, &status) != 0
189 && status.kxr == kXR_ArgTooLong;
190 }
191 brix_streams_close(&streams);
192 xrd_conform_result_set(execution, "BND-004",
193 all_bound && refused ? BRIX_CONFORM_VERDICT_CONFORMANT
194 : BRIX_CONFORM_VERDICT_DIVERGENT,
195 all_bound && refused ? "advertised-bind-limit-enforced"
196 : "bind-limit-mismatch",
197 "bind_max successes then kXR_ArgTooLong",
198 all_bound && refused ? NULL : status.msg);
199}

248{

304/* WHAT: Ping the primary while a pathid-directed read is awaiting collection.
305 * WHY: ENV-002 requires same-session control traffic not to corrupt another
306 * stream's pending data response.
307 * HOW: Bind, send read on primary, ping primary, then receive exact data on
308 * the secondary. The request is bounded to the first 4096 adopted bytes. */
