13/* ---- Emit Markdown table text safely ----
14 *
15 * WHAT: Escapes pipes and folds CR/LF to spaces.
16 * WHY: Clause/evidence text must not change the report's table structure.
17 * HOW: Walk bytes and prefix pipes with a backslash.
18 */
19static void
20markdown_text(FILE *out, const char *value)
21{
22 const char *cursor = value == NULL ? "" : value;
23
24 while (*cursor != '\0') {
25 if (strchr("\\`*_{}[]<>#+-.!|", *cursor) != NULL) fputc('\\', out);
26 if (*cursor == '\r' || *cursor == '\n') fputc(' ', out);
27 else fputc(*cursor, out);
28 cursor++;
29 }
30}

32/* ---- Render the complete Markdown report ----
33 *
34 * WHAT: Writes header metadata, summary counts and one row per clause.
35 * WHY: Every skip and oracle decision must survive outside terminal output.
36 * HOW: Print fixed headings then escape each user/server-derived cell.
37 */
38void
39brix_conform_report_markdown(FILE *out,
40 const brix_conform_report_meta *meta,
41 const brix_conform_run_results *results)
42{
43 size_t index;
44
45 fputs("# BriX protocol-conformance report\n\n- Target: ", out);
46 markdown_text(out, meta->target);
47 for (index = 0; index < meta->also_target_count; index++) {
48 fputs("\n- Also: ", out);
49 markdown_text(out, meta->also_targets[index]);
50 }
51 {
52 const brix_conform_report_field fields[] = {
53 { "\n- Profile: ", meta->profile },
54 { "\n- Shape: ", meta->shape },
55 { "\n- Budget: ", meta->budget },
56 { "\n- Reference: ", meta->reference },
57 { "\n- Run key: ", meta->run_key },
58 { "\n- Timestamp: ", meta->timestamp },
59 { "\n- Tool version: ", meta->tool_version },
60 };
61
62 brix_conform_report_fields(out, fields,
63 sizeof(fields) / sizeof(fields[0]), markdown_text);
64 }
65 for (index = 0; index < meta->negative_verify_count; index++) {
66 fputs("\n- Negative verification: ", out);
67 markdown_text(out, meta->negative_verify[index]);
68 fputs(meta->negative_verify_passed ? " (passed)" : " (failed)", out);
69 }
70 fprintf(out, "\n- Usage: %u/%u requests; %llu/%llu bytes; %us wall ceiling\n\n",
71 meta->requests, meta->request_limit, meta->bytes, meta->byte_limit,
72 meta->wall_seconds);
73 fprintf(out, "Summary: **%zu conformant**, **%zu divergent**, **%zu skipped**, "
74 "**%zu inconclusive**.\n\n"
75 "| ID | Family | Verdict | Ecosystem | Strict | Reference | Assertion | Negative fixture | Reason/evidence |\n"
76 "|---|---|---|---|---|---|---|---|---|\n",
77 results->conformant, results->divergent, results->skipped,
78 results->inconclusive);
79 for (index = 0; index < results->count; index++) {
80 const brix_conform_result *result = &results->results[index];
81
82 fprintf(out, "| %s | %s | %s | %s | %s | %s | ", result->clause->id,
83 result->clause->family, brix_conform_verdict_name(result->verdict),
84 brix_conform_verdict_name(result->ecosystem_verdict),
85 brix_conform_verdict_name(result->strict_verdict),
86 brix_conform_reference_name(result->clause->reference));
87 markdown_text(out, result->clause->text);
88 fputs(" | ", out);
89 markdown_text(out, result->clause->negative_fixture);
90 fputs(" | ", out);
91 markdown_text(out, result->reason);
92 if (result->strict_reason[0] != '\0') {
93 fputs(" strict=", out);
94 markdown_text(out, result->strict_reason);
95 }
96 if (result->expected[0] != '\0') {
97 fputs(" expected=", out);
98 markdown_text(out, result->expected);
99 }
100 if (result->received[0] != '\0') {
101 fputs(" received=", out);
102 markdown_text(out, result->received);
103 }
104 fputs(" |\n", out);
105 }
106}
