2 * cache/origin_ns_dirlist.c — origin-side kXR_dirlist for cache
3 * write-through/mirroring, plus the small name-vector container the listing is
4 * accumulated into (brix_cache_dirlist_*).
5 *
6 * Split out of origin_ns.c when that file crossed the 600-line cap
7 * (coding-standards §1). The seam is a natural one: every other op in

34/* brix_cache_dirlist_free — release every owned name string plus the array,
35 * then reset *dl to the zeroed (empty) state so a second call is a no-op.

37void
38brix_cache_dirlist_free(brix_cache_dirlist_t *dl)
39{
40 size_t i;
41

43 return;
44 }

52/* brix_cache_dirlist_add — append a strdup of name[0, len) to *dl, growing
53 * the array (doubling from 16) as needed. Returns 0 on success, -1 (errno
54 * ENOMEM) on allocation failure — caller aborts the fetch; the caller-owned
55 * dl still holds every name added so far and must still be freed. */
56static int
57brix_cache_dirlist_add(brix_cache_dirlist_t *dl, const char *name, size_t len)
58{
59 char *copy;
60
61 copy = malloc(len + 1);
62 if (copy == NULL) {
63 errno = ENOMEM;
64 return -1;
65 }
66 ngx_memcpy(copy, name, len);
67 copy[len] = '\0';
68
69 if (dl->count == dl->cap) {
70 size_t new_cap = (dl->cap == 0) ? 16 : dl->cap * 2;
71 char **grown = realloc(dl->names, new_cap * sizeof(*grown));
72
73 if (grown == NULL) {
74 free(copy);
75 errno = ENOMEM;
76 return -1;
77 }
78 dl->names = grown;
79 dl->cap = new_cap;
80 }
81
82 dl->names[dl->count] = copy;
83 dl->count++;
84 return 0;
85}

87/* brix_cache_dirlist_name_is_dot — 1 iff name (len bytes, not NUL-terminated)
88 * is "." or "..". The VFS readdir contract (brix_vfs_readdir, vfs_dir.c)
89 * never yields these to a protocol handler, so a remote origin that includes
90 * them in its kXR_dirlist body must be filtered here to match every other
91 * backend's opendir/readdir behaviour. */
92static int
93brix_cache_dirlist_name_is_dot(const char *name, size_t len)
94{
95 return (len == 1 && name[0] == '.')
96 || (len == 2 && name[0] == '.' && name[1] == '.');
97}

99/* brix_cache_dirlist_split — split one response frame's body on '\n' (the
100 * kXR_dirlist wire delimiter — src/protocols/root/dirlist/handler.c writes
101 * "<name>\n" per entry, plain-names mode has no stat/checksum trailer lines
102 * to skip) into individual names, appending each non-dot entry to *dl. A
103 * trailing partial name (no terminating '\n' — the final frame's last byte
104 * is NUL-terminated by the server, not '\n', per handler.c) is still emitted
105 * if non-empty, since it is the last name in the whole listing. Returns 0 on
106 * success, -1 (errno set) on an allocation failure partway through. */
107static int
108brix_cache_dirlist_split(brix_cache_dirlist_t *dl, const u_char *body,
109 uint32_t dlen)
110{
111 size_t start = 0, i;
112
113 for (i = 0; i < dlen; i++) {
114 if (body[i] != '\n') {
115 continue;
116 }
117 if (i > start && !brix_cache_dirlist_name_is_dot(
118 (const char *) body + start, i - start))
119 {
120 if (brix_cache_dirlist_add(dl, (const char *) body + start,
121 i - start) != 0)
122 {
123 return -1;
124 }
125 }
126 start = i + 1;
127 }
128 /* Trailing bytes with no '\n' (final frame's NUL-terminated tail, or a
129 * lone name with no delimiter at all) — treat as one more name. */
130 if (start < dlen && !brix_cache_dirlist_name_is_dot(
131 (const char *) body + start, dlen - start))
132 {
133 if (brix_cache_dirlist_add(dl, (const char *) body + start,
134 dlen - start) != 0)
135 {
136 return -1;
137 }
138 }
139 return 0;
140}

142/* brix_cache_origin_dirlist — kXR_dirlist <path> on the origin: plain name
143 * listing (no kXR_dstat/kXR_dcksm — dirlist.c's server handler only needs
144 * per-entry stat when a caller explicitly asks for it, and sd_xroot's
145 * opendir/readdir only need names, matching every other backend's directory
146 * iteration contract). Wire request: kXR_dirlist opcode, body = reserved(15)
147 * + options(1) at body[15] (xrdw_dirlist_req_pack; options=0 here — no
148 * kXR_dstat/kXR_dcksm/kXR_online), payload = the path (no NUL — dlen is the
149 * length, matching kXR_rm/kXR_rmdir's payload framing above). The server
150 * streams the listing as kXR_oksofar frames of newline-delimited names,
151 * ending with a single kXR_ok frame (src/protocols/root/dirlist/handler.c);
152 * this loops brix_cache_read_response until that final kXR_ok, splitting and
153 * appending each frame's body into *dl via brix_cache_dirlist_split.
154 * Returns 0 (dl populated), or -1 with errno set (dl left however far it
155 * got — still owned by the caller). */
156int
157brix_cache_origin_dirlist(brix_cache_fill_t *t,
158 brix_cache_origin_conn_t *oc, const char *path, brix_cache_dirlist_t *dl)
159{
160 uint8_t body[XRDW_BODY_LEN];
161 size_t pl = (path != NULL) ? strlen(path) : 0;
162 size_t total = sizeof(ClientRequestHdr) + pl;
163 u_char *buf;
164 ClientRequestHdr *req;
165
166 if (pl == 0 || pl > 0x7fff) {
167 errno = EINVAL;
168 return -1;
169 }
170
171 {
172 xrdw_dirlist_req_t b = { .options = 0 }; /* plain names, no stat */
173 xrdw_dirlist_req_pack(&b, body);
174 }
175
176 buf = malloc(total);
177 if (buf == NULL) {
178 errno = ENOMEM;
179 return -1;
180 }
181 ngx_memzero(buf, sizeof(ClientRequestHdr));
182 req = (ClientRequestHdr *) buf;
183 req->streamid[1] = 8; /* unused stream slot */
184 req->requestid = htons(kXR_dirlist);
185 ngx_memcpy(req->body, body, XRDW_BODY_LEN);
186 req->dlen = htonl((kXR_int32) pl);
187 ngx_memcpy(buf + sizeof(ClientRequestHdr), path, pl);
188
189 if (brix_cache_io_send(oc, buf, total) != 0) {
190 free(buf);
191 errno = EIO;
192 return -1;
193 }
194 free(buf);
195
196 for (;;) {
197 uint16_t status;
198 uint32_t dlen;
199 u_char *rbody = NULL;
200
201 if (brix_cache_read_response(t, oc, &status, &rbody, &dlen,
202 BRIX_CACHE_FETCH_CHUNK) != 0)
203 {
204 errno = EIO;
205 return -1;
206 }
207
208 if (status == kXR_error) {
209 errno = brix_cache_origin_status_errno(status, rbody, dlen);
210 free(rbody);
211 return -1;
212 }
213
214 if (status != kXR_ok && status != kXR_oksofar) {
215 free(rbody);
216 brix_cache_set_error(t, kXR_ServerError, 0,
217 "cache origin dirlist returned invalid status");
218 errno = EIO;
219 return -1;
220 }
221
222 if (dlen > 0 && brix_cache_dirlist_split(dl, rbody, dlen) != 0) {
223 int saved_errno = errno;
224
225 free(rbody);
226 errno = saved_errno;
227 return -1;
228 }
229 free(rbody);
230
231 if (status == kXR_ok) {
232 return 0;
233 }
234 }
235}
