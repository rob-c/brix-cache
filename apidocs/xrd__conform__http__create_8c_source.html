38/* WHAT: Prove a zero-length PUT creates a visible zero-length object.
39 * WHY: Empty payloads must remain valid data rather than becoming no-ops.
40 * HOW: PUT under scratch, GET it with a one-byte ceiling, then DELETE it. */
41static void
42probe_zero_put(xrd_conform_execution *execution, const brix_weburl *base,
43 const char *token)
44{
45 brix_weburl scratch;
46 brix_http_resp put_response;
47 brix_http_resp get_response;
48 brix_status status;
49 int put_rc;
50 int get_rc = -1;
51 int pass;
52
53 if (!xrd_conform_row_active(execution, "HTC-003")) return;
54 if (scratch_url(execution, base, "brix-conform-http-zero", &scratch) != 0) {
55 xrd_conform_result_set(execution, "HTC-003",
56 BRIX_CONFORM_VERDICT_SKIPPED,
57 "fixture-absent:mutable-http-scratch", NULL, NULL);
58 return;
59 }
60 memset(&put_response, 0, sizeof(put_response));
61 memset(&get_response, 0, sizeof(get_response));
62 brix_status_clear(&status);
63 put_rc = xrd_conform_http_request_body(execution, &scratch, "PUT", token,
64 NULL, NULL, 0, 0, &put_response, &status);
65 if (put_rc == 0) brix_http_resp_free(&put_response);
66 if (put_rc == 0)
67 get_rc = xrd_conform_http_request_fields(execution, &scratch, "GET",
68 token, NULL, 1, &get_response, &status);
69 pass = put_rc == 0 && get_rc == 0
70 && get_response.status >= 200 && get_response.status < 300
71 && get_response.body_len == 0;
72 xrd_conform_result_set(execution, "HTC-003",
73 pass ? BRIX_CONFORM_VERDICT_CONFORMANT
74 : BRIX_CONFORM_VERDICT_DIVERGENT,
75 pass ? "zero-length-put-created-empty-object"
76 : "zero-length-put-object-missing-or-nonempty",
77 "successful GET with zero response bytes", status.msg);
78 if (get_rc == 0) brix_http_resp_free(&get_response);
79 cleanup(execution, &scratch, token);
80}

103/* WHAT: Send one intentionally malformed PUT and prove no object committed.
104 * WHY: Request-framing failures must be atomic from the client's perspective.
105 * HOW: Use exact raw bytes under scratch, close the request side, then GET. */
106static void
107probe_malformed_put(xrd_conform_execution *execution,

117
